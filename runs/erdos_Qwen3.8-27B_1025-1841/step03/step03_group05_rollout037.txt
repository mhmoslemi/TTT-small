def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_shift(h, shift):
    """Subgradient of full_correlation[shift + n - 1] w.r.t. h"""
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        # corr = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
        # g[j] = (1-h[j-s]) * I(s<=j) + (-h[j+s]) * I(j<=n-1-s)
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:

            g[:n-s] += (-h[s:])  # -h[j+s] for j=0..n-1-s
    else:

        g[:n-t] += (1.0 - h[t:])  # (1-h[j+t]) for j=0..n-1-t
        g[t:] += (-h[:n-t])  # -h[j-t] for j=t..n-1
    return g

def weighted_subgrad(h, T, top_k=50):
    """Softmax-weighted subgradient of max correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

        return subgrad_shift(h, idx - (n-1)) * dx
    w = exps / tot
    # Use only top-k for efficiency

        idx = np.argsort(w)[::-1][:top_k]
        w_top = w[idx]
        w_top = w_top / w_top.sum()
    else:
        idx = np.arange(len(w))
        w_top = w
    g = np.zeros(n)
    for i in range(len(idx)):
        if w_top[i] < 1e-14:
            break
        g += w_top[i] * subgrad_shift(h, idx[i] - (n-1))
    return g * dx

def fista_opt(h, n_iter, T_start, T_end, lr_start, lr_end, top_k=50, check_every=30):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** f
        lr = lr_start + (lr_end - lr_start) * f
        g = weighted_subgrad(z, T, top_k)
        x_new = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = x_new + (tk - 1.0) / tn * (x_new - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = x_new
        tk = tn
        if (it + 1) % check_every == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    
    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

def slsqqpolish(h, n_iter=400):
    """Polish using SLSQP with smooth max objective"""
    n = len(h)
    dx = 2.0 / n
    T = 15.0  # Sharp but not too sharp for numerical stability
    
    def obj(h_flat):
        corr = np.correlate(h_flat, 1.0 - h_flat, mode='full')
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    
    def con(h_flat):
        return np.sum(h_flat) - n / 2.0
    
    res = minimize(obj, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con}],
                   options={'maxiter': n_iter, 'ftol': 1e-16})
    if res.success or res.x is not None:
        h_new = project_c(res.x, n)
        return h_new
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_shift(h, shift, n):
    """Subgradient of full_correlation[shift+n-1] w.r.t. h, shape (n,)"""
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        if s == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] += (-h[s:])
    else:
        t = -shift
        g[:n-t] += (1.0 - h[t:])
        g[t:] += (-h[:n-t])
    return g

            for ii in range(len(top_idx)):
                if w_sub[ii] < 1e-14:
                    break
                g += w_sub[ii] * subgrad_shift(z, top_idx[ii] - (n-1), n)
            g *= dx
        x_new = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = x_new + (tk - 1.0) / tn * (x_new - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = x_new
        tk = tn
        if (it + 1) % check_every == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    
    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

Hmm, the coordinate descent is getting complicated with the bookkeeping. Let me simplify: just recompute the correlation from scratch at the end of each pass. The cost is O(n^2) per pass, and each pass is O(n^2) anyway (for the n coordinates, each O(n) for the coefficient computation + O(1) for the candidate evaluation).

Wait, the coefficient computation per coordinate is O(n) (iterating over 2n-1 lags). So the total per pass is O(n * n) = O(n^2). And the final correlation computation is also O(n^2). So total per pass: ~2*O(n^2) ≈ 4M for n=960. That's about 4ms per pass.

In 200 seconds: 50,000 passes. That should be more than enough.

But actually, I realize the coefficient computation per coordinate can be vectorized! Instead of a Python loop over 2n-1 lags, I can use numpy operations.

For a fixed j:
- For s > 0: coeff[n-1+s] = (1-h[j-s])*I(s≤j) - h[j+s]*I(j≤n-1-s)
  - First term: for s=1..min(j, n-1), coeff[n-1+s] += (1-h[j-s])
  - Second term: for s=max(1, n-j)..n-1, coeff[n-1+s] -= h[j+s]
  
- For s < 0 (t = -s): coeff[n-1-t] = (1-h[j+t])*I(j≤n-1-t) - h[j-t]*I(j≥t)
  - First term: for t=1..min(n-1-j, n-1), coeff[n-1-t] += (1-h[j+t])
  - Second term: for t=1..min(j, n-1), coeff[n-1-t] -= h[j-t]
