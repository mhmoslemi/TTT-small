    s = abs(shift)

    g = np.zeros(n)

    norm_val = (np.sum(corr_pos ** p)) ** (1.0 / p)
    weights = cp / norm_val ** (p - 1)

        else:
            T = T0 * (T1 / T0) ** f
            g = sm_grad(z, T)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def downsample(h, factor=2):
    n = len(h)
    m = n // factor
    idx = np.arange(m) * factor
    return h[idx]

def downsample_avg(h, factor=2):
    n = len(h)
    m = n // factor
    reshaped = h[:m * factor].reshape(m, factor)
    return reshaped.mean(axis=1)

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

_OB = {}

def _sobj(h):
    T = _OB['T']
    n = _OB['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    logsum = np.log(np.sum(np.exp(T * (corr - mc))))
    return (mc + logsum / T) * dx

def _sgrad(h):
    T = _OB['T']
    n = _OB['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    w = exps / np.sum(exps)
    top_k = min(150, 2 * n)
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def _scon(h):
    return np.sum(h) - _OB['n'] / 2.0

def _sjac(h):
    return np.ones(_OB['n'])

def slsqp(h, T=5.0, maxiter=400, tol=1e-10):
    n = len(h)
    _OB['T'] = T
    _OB['n'] = n
    try:
        r = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': maxiter, 'ftol': tol})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].copy()

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bc = float('inf')
    bh = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = project_c(tm_seq(480), 480)

    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"I: n={n0} C5={bc:.6f}", flush=True)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    # Phase 1: FISTA softmax at parent resolution
    if remaining() > 200:
        h1, c1 = fista(bh, 4000, 50.0, 1.0, 0.13, 30)
        if c1 < bc:
            bc, bh = c1, h1.copy()
        print(f"F1: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 2: FISTA p-norm at parent resolution
    if remaining() > 150:
        h2, c2 = fista(bh, 3500, 50.0, 1.0, 0.12, 30, use_pn=True, p0=20, p1=200)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        print(f"F2: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 3: FISTA fine-tune (low T)
    if remaining() > 120:
        h3, c3 = fista(bh, 3000, 3.0, 0.05, 0.03, 20)
        if c3 < bc:
            bc, bh = c3, h3.copy()
        print(f"F3: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 4: Downsample and SLSQP polish at multiple resolutions
    for target_n in [480, 384, 512]:
        if remaining() < 80:
            break
        if target_n >= n0:
            hd = bh.copy()
        else:
            hd = downsample_avg(bh, factor=max(1, n0 // target_n))
            if len(hd) != target_n:
                hd = upsample(hd, target_n)
        hd = project_c(hd, target_n)
        cd = compute_c5(hd)
        if cd < bc:
            bc, bh = cd, hd.copy()
        # FISTA warm start at this resolution
        if remaining() > 100:
            hg, cg = fista(hd, 2000, 20.0, 0.3, 0.08, 25)
            if cg < cd:
                hd, cd = hg, cg
        # SLSQP polish
        if remaining() > 60:
            for TV in [10.0, 4.0, 2.0]:
                if remaining() < 30:
                    break
                hd = slsqp(hd, TV, 300)
                cd = compute_c5(hd)
        if cd < bc:
            bc, bh = cd, hd.copy()
        print(f"R{target_n}: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 5: Try Thue-Morse and other structured starts at n=480
    for ns in [480, 384]:
        if remaining() < 80:
            break
        starts = []
        starts.append(tm_seq(ns))
        pat5 = np.zeros(5)
        pat5[:3] = 1.0
        starts.append(project_c(np.tile(pat5, ns // 5 + 1)[:ns], ns))
        pat7 = np.zeros(7)
        pat7[:4] = 1.0
        starts.append(project_c(np.tile(pat7, ns // 7 + 1)[:ns], ns))
        for hs in starts:
            if remaining() < 40:
                break
            hs = project_c(hs, ns)
            hr, cr = fista(hs, 2000, 30.0, 0.5, 0.1, 30)
            if remaining() > 30:
                hr = slsqp(hr, 5.0, 200)
                cr = compute_c5(hr)
            if cr < bc:
                bc, bh = cr, hr.copy()
                print(f"  SS{ns}: C5={bc:.6f}", flush=True)

    # Phase 6: Upsample best to higher resolution and FISTA
    for nn in [max(960, n0), 1200]:
        if remaining() < 80:
            break
        if nn <= n0:
            continue
        hu = upsample(bh, nn)
        cu = compute_c5(hu)
        if remaining() > 60:
            hg, cg = fista(hu, 2500, 15.0, 0.2, 0.06, 30)
            if cg < cu:
                hu, cu = hg, cg
        if remaining() > 30:
            hg, cg = fista(hu, 1500, 2.0, 0.02, 0.02, 20)
            if cg < cu:
                hu, cu = hg, cg
        if cu < bc:
            bc, bh = cu, hu.copy()
        print(f"U{nn}: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 7: Final SLSQP if resolution allows
    np_ = len(bh)
    if np_ <= 480 and remaining() > 30:
        bh = slsqp(bh, 3.0, 400)
        bc = compute_c5(bh)

    # Phase 8: Ultra-fine FISTA
    if remaining() > 20:
        hf, cf = fista(bh, 1500, 2.0, 0.01, 0.008, 20)
        if cf < bc:
            bc, bh = cf, hf.copy()

    # Phase 9: Perturbation + FISTA restarts
    np_ = len(bh)
    for trial in range(3):
        if remaining() < 20:
            break
        sc = 0.015 + 0.02 * trial
        hp = project_c(bh + rng.randn(np_) * sc, np_)
        hr, cr = fista(hp, 1500, 15.0, 0.1, 0.05, 30)
        if cr < bc:
            bc, bh = cr, hr.copy()

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={elapsed():.0f}", flush=True)
    return bh, bc, len(bh)
