```python
def lse_objective_and_gradient(h, T, top_m=50):
    """Compute log-sum-exp of correlations and its gradient w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -100, 100)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    obj = (m + np.log(np.sum(exp_vals)) / T) * dx
    
    # Gradient: sum over top_m lags
    top_idx = np.argsort(-weights)[:top_m]
    grad = np.zeros(n)
    for idx in top_idx:
        if weights[idx] < 1e-10:
            break
        k = idx - (n - 1)  # actual lag
        # dc_k/dh[j]
        if k >= 0:
            # dc_k/dh[j] = (1-h[j-k])*I[k<=j<n] - h[j+k]*I[0<=j<n-1-k]
            j1 = np.arange(k, n)
            grad[j1] += weights[idx] * (1.0 - h[j1 - k])
            j2 = np.arange(0, n - k)
            grad[j2] -= weights[idx] * h[j2 + k]
        else:
            mlag = -k
            j1 = np.arange(0, n - mlag)
            grad[j1] += weights[idx] * (1.0 - h[j1 + mlag])
            j2 = np.arange(mlag, n)
            grad[j2] -= weights[idx] * h[j2 - mlag]
    
    # Add constraint gradient
    viol = np.sum(h) - n / 2.0
    grad += 2.0 * 100.0 * viol  # penalty
    
    return obj, grad
```

And h[j] in h[n-m] position: n-m = j, so n = j+m. Valid if m ≤ j+m ≤ n-1, i.e., 0 ≤ j ≤ n-1-m.
The term is h[j] * (1-h[j+m]). Change: δ * (1-h[j+m]).
So contribution is δ * (1-h[j+m]) * I[0 ≤ j ≤ n-1-m].

OK so for a binary swap (h[a]: 1→0, so δ_a = -1; h[b]: 0→1, so δ_b = +1):

Δcorr[idx] = δ_a * [ (1-h[a-k])*I[valid1_a] - h[a+k]*I[valid2_a] ] + δ_b * [ (1-h[b-k])*I[valid1_b] - h[b+k]*I[valid2_b] ]

where the "valid" conditions depend on the sign of k.

This is O(1) per idx, so O(N) total. Good.

But implementing this correctly in vectorized form is tricky. Let me just do it in a loop for clarity (N=640 is small enough).

Actually, for the local search, I don't need to compute the full new_corr. I just need to check if the new max is less than the current max. I can do this more efficiently:

1. Find the current max value M and the set of indices where corr ≈ M (within some tolerance).
2. For a swap, compute Δcorr[idx] only for those critical indices.
3. If all critical indices decrease (or stay the same), the swap is good.
4. Also need to check that no other index becomes larger than the new max.

This is more complex but could be faster. For now, let me just do the O(N) computation per swap.

For N=640, each swap evaluation is O(640) = 640 operations. If I try 10000 swaps per second, that's 6.4M operations per second, which is fine.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lse_grad_optimize(h_start, n, T, max_iter=500, penalty=100.0):
    """L-BFGS-B with analytic gradient of log-sum-exp smoothed max."""
    dx = 2.0 / n
    
    def obj_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = np.max(corr)
        shifted = T * (corr - m)
        shifted = np.clip(shifted, -100, 100)
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        weights = exp_vals / sum_exp
        
        obj = (m + np.log(sum_exp) / T) * dx
        
        # Gradient
        grad = np.zeros(n)
        # Only sum over lags with significant weight
        sorted_w = np.sort(weights)[::-1]
        threshold = max(1e-8, sorted_w[min(100, len(sorted_w)-1)])
        active = np.where(weights > threshold)[0]
        
        for idx in active:
            w = weights[idx]
            if w < 1e-10:
                continue
            k = idx - (n - 1)
            if k >= 0:
                # dc_k/dh[j] = (1-h[j-k])*I[k<=j<n] - h[j+k]*I[0<=j<n-k]
                j1 = np.arange(k, n)
                grad[j1] += w * (1.0 - h[j1 - k])
                if n - k > 0:
                    j2 = np.arange(0, n - k)
                    grad[j2] -= w * h[j2 + k]
            else:
                mlag = -k
                if n - mlag > 0:
                    j1 = np.arange(0, n - mlag)
                    grad[j1] += w * (1.0 - h[j1 + mlag])
                j2 = np.arange(mlag, n)
                grad[j2] -= w * h[j2 - mlag]
        
        # Constraint penalty gradient
        viol = np.sum(h) - n / 2.0
        grad += 2.0 * penalty * viol
        
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    
    def f(h):
        obj, _ = obj_and_grad(h)
        return obj
    
    def g(h):
        _, grad = obj_and_grad(h)
        return grad
    
    res = minimize(f, h_start, jac=g, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def swap_eval(corr, h, a, b):
    """Compute new max correlation after swapping h[a] and h[b]."""
    n = len(h)
    da = h[b] - h[a]
    db = h[a] - h[b]
    
    if abs(da) < 1e-10:
        return float(np.max(corr))
    
    new_corr = corr.copy()
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        delta = 0.0
        # Position a contribution
        if k >= 0:
            if k <= a < n:
                delta += da * (1.0 - h[a - k])
            if 0 <= a < n - k:
                delta -= da * h[a + k]
        else:
            mlag = -k
            if 0 <= a < n - mlag:
                delta += da * (1.0 - h[a + mlag])
            if mlag <= a < n:
                delta -= da * h[a - mlag]
        # Position b contribution
        if k >= 0:
            if k <= b < n:
                delta += db * (1.0 - h[b - k])
            if 0 <= b < n - k:
                delta -= db * h[b + k]
        else:
            mlag = -k
            if 0 <= b < n - mlag:
                delta += db * (1.0 - h[b + mlag])
            if mlag <= b < n:
                delta -= db * h[b - mlag]
        new_corr[idx] += delta
    
    return float(np.max(new_corr))

                improved = True

        if not improved:
            break
    
    return h, current_max
