def binary_project(h, n):
    """Project to binary with exactly n/2 ones"""
    h = np.array(h, dtype=float)
    threshold_idx = np.argpartition(h, n//2)[n//2]
    h_bin = (h >= threshold_idx).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n//2:
        # Set some 1s to 0
        ones = np.where(h_bin == 1)[0]
        # Prioritize setting to 0 those with lowest h value
        order = ones[np.argsort(h[ones])]
        h_bin[order[:n_ones - n//2]] = 0.0
    elif n_ones < n//2:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        h_bin[order[:n//2 - n_ones]] = 1.0
    return h_bin

One thing I want to try that's different from the previous program: **simulated annealing / stochastic local search**.

For binary h, I can do:

def make_feasible(h, n):
