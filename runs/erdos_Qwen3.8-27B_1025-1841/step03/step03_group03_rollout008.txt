    old_x = np.linspace(0.0, 2.0, old_n)

def subgradient_step(h, n, alpha, time_limit, start_time):
    """One step of projected subgradient descent."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find all lags at (or near) the maximum
    threshold = max_val - 1e-10
    active = np.where(corr >= threshold)[0]
    
    # Compute gradient as average of active gradients
    grad = np.zeros(n)
    for m in active:
        k = m - (n - 1)  # offset
        # grad[j] = (1-h[j-k]) * 1[0<=j-k<n] - h[j+k] * 1[0<=j+k<n]
        if k >= 0:
            # j from k to n-1: term1 contribution (1-h[j-k])
            grad[k:] += 1.0 - h[:n-k]
            # j from 0 to n-k-1: term2 contribution -h[j+k]
            grad[:n-k] -= h[k:]
        else:
            kp = -k  # positive
            # j from 0 to n-kp-1: term1 contribution (1-h[j+k]) = (1-h[j-kp])
            # Wait, j-k = j+kp, need 0<=j+kp<n, so j < n-kp
            grad[:n-kp] += 1.0 - h[kp:]
            # j from kp to n-1: term2 contribution -h[j+k] = -h[j-kp]
            # j+k = j-kp, need 0<=j-kp<n, so j >= kp
            grad[kp:] -= h[:n-kp]
    
    grad /= len(active)
    grad *= dx  # because C5 = max(corr) * dx
    
    # Step
    h_new = h - alpha * grad
    h_new = project_to_simplex(h_new, n)
    return h_new

def project_to_simplex(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Binary search on lambda: p = clip(h + lambda, 0, 1), find lambda s.t. sum(p) = target
    lo = -max(h) - 1.0
    hi = 1.0 - min(h) + 1.0
    for _ in range(60):

        p = np.clip(h + mid, 0.0, 1.0)

        if s < target:
            lo = mid
        else:

    # Fix rounding

                    if new_l < -1e-9 or new_l > 1+1e-9:
                        continue
                    trial = h.copy()
                    trial[j] = max(0.0, min(1.0, new_j))
                    trial[l] = max(0.0, min(1.0, new_l))
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_obj - 1e-14:
                        best_obj = c
                        best_h = trial
            if best_obj < best_c - 1e-14:
                h = best_h
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    # Get parent solution
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n = 640
        h = (rng.rand(n) < 0.5).astype(float)
        idx_ones = np.where(h == 1)[0]
        idx_zeros = np.where(h == 0)[0]
        n_ones = n // 2
        if len(idx_ones) > n_ones:
            h[idx_ones[n_ones:]] = 0.0
        elif len(idx_ones) < n_ones:
            h[idx_zeros[:n_ones - len(idx_ones)]] = 1.0
    n = len(h)
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    best_h = h.copy()
    
    # Phase 1: SLSQP with increasing temperature at current n
    for t, mi in [(30, 300), (80, 400), (200, 500), (500, 500), (1000, 500)]:
        if time.time() - start_time > budget_s - 500:
            break
        h_new = slsqp_run(best_h, n, t, mi, start_time, budget_s)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new
    
    # Phase 2: CD (coarse) at n
    h_cd, c_cd = cd_pass(best_h, n, rounds=25, start_time=start_time, time_limit=budget_s)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd
    
    # Phase 3: CD (fine) at n
    h_cd, c_cd = cd_fine(best_h, n, rounds=30, start_time=start_time, time_limit=budget_s)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd
    
    # Phase 4: Binary annealing
    h_bin = (best_h > 0.5).astype(float)
    h_bin = make_feasible(h_bin, n)
    c_bin = compute_c5(h_bin)
    if time.time() - start_time < budget_s - 300:
        h_bin, c_bin = binary_anneal(h_bin, n, start_time, budget_s, max_iter=4000)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin
    
    # Phase 5: Pair swap CD
    if time.time() - start_time < budget_s - 250:
        h_ps, c_ps = pair_swap_cd(best_h, n, rounds=8, start_time=start_time, time_limit=budget_s)
        if c_ps < best_c:
            best_c = c_ps
            best_h = h_ps
    
    # Phase 6: Another SLSQP pass with high temp
    if time.time() - start_time < budget_s - 200:
        h_new = slsqp_run(best_h, n, 800, 400, start_time, budget_s)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-10:
            best_c = c_new
            best_h = h_new
    
    # Phase 7: Upsample to 2*n and optimize
    n2 = n * 2
    if time.time() - start_time < budget_s - 180:
        h_up = upsample_h(best_h, n2)
        h_up = slsqp_run(h_up, n2, 100, 300, start_time, budget_s)
        c_up = compute_c5(h_up)
        if c_up < best_c - 1e-10:
            best_c = c_up
            best_h = h_up
        h_cd, c_cd = cd_pass(best_h, len(best_h), rounds=10, start_time=start_time, time_limit=budget_s)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
        h_cd, c_cd = cd_fine(best_h, len(best_h), rounds=15, start_time=start_time, time_limit=budget_s)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    # Phase 8: Try upsampling to 4*original_n
    n4 = n * 4
    if time.time() - start_time < budget_s - 100:
        h_up4 = upsample_h(best_h, n4)
        c_up4 = compute_c5(h_up4)
        h_cd, c_cd = cd_pass(h_up4, n4, rounds=8, start_time=start_time, time_limit=budget_s)
        if c_cd < best_c:
            best_c = c_cd
            best_h = h_cd
    
    # Phase 9: Final fine-tuning CD
    h_cd, c_cd = cd_fine(best_h, len(best_h), rounds=15, start_time=start_time, time_limit=budget_s)
    if c_cd < best_c:
        best_c = c_cd
        best_h = h_cd
    
    # Phase 10: Binary threshold + CD
    if time.time() - start_time < budget_s - 20:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, len(h_bin))
        c_bin = compute_c5(h_bin)
        if c_bin < best_c:
            best_c = c_bin
            best_h = h_bin
            h_cd, c_cd = cd_fine(best_h, len(best_h), rounds=5, start_time=start_time, time_limit=budget_s)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd
    
    best_h = make_feasible(best_h, len(best_h))
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    
    return best_h, best_c, len(best_h)
