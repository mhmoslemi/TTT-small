def make_feasible_u(u, n):
    """Project u onto {-1 <= u <= 1, sum(u) = 0}"""
    u = np.clip(u, -1.0, 1.0)

    target = n / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible_h(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-14:

            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)

    h_new = np.interp(new_x, old_x, h)
    return make_feasible_h(h_new, new_n)
