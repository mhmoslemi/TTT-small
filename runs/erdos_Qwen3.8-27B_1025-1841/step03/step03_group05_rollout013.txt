def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    lo, hi = -1.5, 1.5

        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:

        else:

    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s == 0:
            return 1.0 - 2.0 * h
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        t = -s
        g[t:] += 1.0 - h[:n-t]
        g[:n-t] -= h[t:]
    return g

    x = h.astype(float).copy()
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(1, n_iter+1):
        if it % check_every == 0:
            if time.time() - t0 > budget - 5:
                break
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        # Get top-k lags
        if top_k < len(corr):
            idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            idx = np.arange(len(corr))
        # Uniform weights on top-k (simple and effective)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(50):

        c = np.clip(h - mid, 0.0, 1.0)

        else:
            hi = mid

    n = len(h)
    g = np.zeros(n)
    if s == 0:
        return 1.0 - 2.0 * h
    if s > 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        t = -s
        g[t:] += 1.0 - h[:n-t]
        g[:n-t] -= h[t:]
    return g

def pgd_fista(h, t0, t_end, n_iter, top_k=8, alpha0=0.4, decay=1000.0, use_momentum=True, check=200):

    x = h.astype(float).copy()

    t_k = 1.0

    bh = x.copy()

            break
        corr = np.correlate(z, 1.0 - z, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n * 0.5
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def add_subgrad(g, z, s, w, n):
    if s == 0:
        g += w * (1.0 - 2.0 * z)
    elif s > 0:
        g[:n - s] += w * (1.0 - z[s:])
        g[s:] -= w * z[:n - s]
    else:
        t = -s
        g[t:] += w * (1.0 - z[:n - t])
        g[:n - t] -= w * z[t:]

def pgd_fista(h, t0, t_end, n_iter, top_k=8, alpha0=0.35, decay=5000.0,
              momentum=True, check=200):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    t_k = 1.0
    g = np.zeros(n)
    bc = compute_c5(x)
    bh = x.copy()
    clen = 2 * n - 1
    for it in range(1, n_iter + 1):
        if it % check == 0 and time.time() >= t_end:
            break
        corr = np.correlate(z, 1.0 - z, mode='full')
        if top_k < clen:
            idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            idx = np.arange(clen)
        w = 1.0 / len(idx)
        g[:] = 0.0
        for i in idx:
            add_subgrad(g, z, i - (n - 1), w, n)
        alpha = alpha0 / (1.0 + it / decay)
        if momentum:
            xn = project_c(z - alpha * g, n)
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) * 0.5
            z = xn + ((t_k - 1.0) / t_new) * (xn - x)
            np.clip(z, -0.5, 1.5, out=z)
            x = xn
            t_k = t_new
        else:
            x = project_c(x - alpha * g, n)
        if it % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc = c
                bh = x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def upsample_h(h, nn):
    n = len(h)
    if nn == n:
        return h.copy()
    ox = np.arange(n) * (2.0 / n)
    nx = np.arange(nn) * (2.0 / nn)
    hi = np.interp(nx, ox, h)
    return project_c(hi, nn)

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

    print(f"P2: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 3: Very fine step, no momentum
    t_end = t0 + min(620, budget_s * 0.52)
    bh, bc = pgd_fista(bh, t0, t_end, 3000000, top_k=3, alpha0=0.06,
                       decay=1500, momentum=False, check=150)
    print(f"P3: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 4: Upsample to 2x

        nn = n0 * 2
        hu = upsample_h(bh, nn)

        t_end = t0 + min(time.time() - t0 + 160, t0 + budget_s * 0.72)

    # Phase 5: Perturbation restarts
    if remaining() > 80:
        np_ = len(bh)
        for trial in range(6):
            if remaining() < 25:
                break
            sc = 0.02 + 0.015 * trial
            hp = project_c(bh + rng.randn(np_) * sc, np_)

        t_end = t0 + budget_s - 3
        bh, bc = pgd_fista(bh, t0, t_end, 2000000, top_k=3, alpha0=0.03,
                           decay=800, momentum=False, check=100)

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
