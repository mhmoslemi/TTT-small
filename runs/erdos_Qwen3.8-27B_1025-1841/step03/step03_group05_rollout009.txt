```python
import numpy as np
from scipy.optimize import minimize
import time

def _ac(h):
    n = len(h)
    H = np.fft.rfft(h, 2 * n)
    return np.fft.irfft(H * np.conj(H), 2 * n)[:n]

def _ov_all(h):
    n = len(h)
    ac = _ac(h)
    ps = np.concatenate([[0], np.cumsum(h)])
    k = np.arange(n)
    op = ps[n - k] - ac
    on = ps[n] - ps[k] - ac
    ov = np.concatenate([op, on[1:]])
    lag = np.concatenate([k, -k[1:]])
    return ov, lag

def _c5(h):
    ov, _ = _ov_all(h)
    return float(np.max(ov) * 2.0 / len(h))

def _proj(h, n):
    tgt = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > tgt:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def _sg(h, s):
    n = len(h)
    k = abs(s)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-k] += 1.0 - h[k:]
        g[k:] -= h[:n-k]
    else:
        g[k:] += 1.0 - h[:n-k]
        g[:n-k] -= h[k:]
    return g

def _smgrad(h, T, topk=15):
    n = len(h)
    dx = 2.0 / n
    ov, lag = _ov_all(h)
    mc = np.max(ov)
    ev = np.exp(T * (ov - mc))
    tot = ev.sum()
    if tot < 1e-300:
        k = int(np.argmax(ov))
        return _sg(h, lag[k]) * dx
    w = ev / tot
    idx = np.argsort(w)[::-1][:topk]
    g = np.zeros(n)
    for i in idx:
        if w[i] < 1e-12:
            break
        g += w[i] * _sg(h, int(lag[i]))
    return g * dx

def _fista(h, niter, T0, T1, lr, chk):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = _c5(x)
    bh = x.copy()
    for it in range(niter):
        f = it / max(niter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.5 * f)
        g = _smgrad(z, T)
        xn = _proj(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % chk == 0:
            c = _c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = _c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def _slsqp_polish(h, T=4.0, mi=400):
    n = len(h)
    def obj(x):
        ov, lag = _ov_all(x)
        mc = np.max(ov)
        return (mc + np.log(np.sum(np.exp(T * (ov - mc)))) / T) * 2.0 / n
    def obj_jac(x):
        ov, lag = _ov_all(x)
        mc = np.max(ov)
        ev = np.exp(T * (ov - mc))
        w = ev / ev.sum()
        g = np.zeros(n)
        for i in range(len(ov)):
            if w[i] > 1e-10:
                g += w[i] * _sg(x, int(lag[i]))
        return g * 2.0 / n
    
    r = minimize(obj, h, method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}],
                 options={'maxiter': mi, 'ftol': 1e-12})
    if r.x is not None:
        hx = _proj(r.x, n)
        if _c5(hx) < _c5(h):
            return hx
    return h

def _upsample(h, nn):
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return _proj(np.interp(nx, ox, h), nn)

def _tm_se(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def _spec_eq(h, alpha=0.3):
    """Spectral equalization: flatten power spectrum"""
    n = len(h)
    H = np.fft.rfft(h, 2 * n)
    P = np.abs(H) ** 2
    P[P < 1e-10] = 1e-10
    # Target: flat spectrum (white)
    # Blended: P_new = P * (mean(P) / P)^alpha = mean(P)^(alpha) * P^(1-alpha)
    target = np.full_like(P, np.mean(P))
    P_new = P * (target / P) ** alpha
    H_new = H * np.sqrt(P_new / P)
    h_new = np.fft.irfft(H_new, 2 * n)[:n]
    return _proj(h_new, n)

    rng = np.random.RandomState(seed)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        h0 = _proj(rng.rand(480), 480)
    
    n0 = len(h0)
    bc = _c5(h0)
    bh = h0.copy()
    print(f"I: n={n0} C5={bc:.6f}", flush=True)
    
    # Phase 1: FISTA exploitation (low T, fine-tuning)
    if rem() > 200:
        h1, c1 = _fista(bh, 8000, 5.0, 0.05, 0.06, 30)
        if c1 < bc:
            bc, bh = c1, h1
        print(f"F1: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    
    # Phase 2: FISTA exploration (high T)
    if rem() > 300:
        h2, c2 = _fista(bh, 6000, 80.0, 1.0, 0.12, 40)
        if c2 < bc:
            bc, bh = c2, h2
        # Follow up with exploitation
        if rem() > 100:
            h3, c3 = _fista(bh, 6000, 3.0, 0.03, 0.04, 25)
            if c3 < bc:
                bc, bh = c3, h3
        print(f"F2: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    
    # Phase 3: SLSQP polish
    if n0 <= 512 and rem() > 80:
        for TV in [10.0, 4.0, 2.0]:
            if rem() < 25:
                break
            hx = _slsqp_polish(bh, TV, 350)
            cx = _c5(hx)
            if cx < bc:
                bc, bh = cx, hx
        print(f"SL: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    
    # Phase 4: Spectral equalization + refine
    if rem() > 150:
        for alpha in [0.2, 0.4, 0.6]:
            if rem() < 50:
                break
            hs = _spec_eq(bh, alpha)
            hsr, csr = _fista(hs, 4000, 5.0, 0.1, 0.05, 30)
            if csr < bc:
                bc, bh = csr, hsr
        print(f"SE: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    
    # Phase 5: Diverse starts
    starts = [_tm_se(n0), _proj(rng.rand(n0), n0)]
    # Block patterns
    for p in [4, 6, 8, 12, 16]:
        pat = np.zeros(n0)
        seg = n0 // p
        for i in range(p):
            if i % 2 == 0:
                pat[i*seg:(i+1)*seg] = 1.0
        starts.append(_proj(pat, n0))
    # Sinusoidal modulations
    x = np.linspace(0, 2, n0)
    for freq in [1.5, 2.5, 3.5]:
        starts.append(_proj(0.5 + 0.3 * np.sin(2 * np.pi * freq * x), n0))
    
    for hs in starts:
        if rem() < 60:
            break
        hr, cr = _fista(hs, 4000, 20.0, 0.2, 0.08, 40)
        if cr < bc:
            bc, bh = cr, hr
            print(f"  DS: C5={bc:.6f}", flush=True)
    print(f"DA: C5={bc:.6f} t={time.time()-t0:.0f}",
