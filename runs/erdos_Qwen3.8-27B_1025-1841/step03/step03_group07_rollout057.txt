def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def soft_grad(h, n, tau, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    max_c = top_vals.max()
    shifted = np.clip(tau * (top_vals - max_c), -80.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx]
        k = int(top_idx[i_idx])
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return grad


def adam_opt(h_init, n, time_limit, tau_start, tau_end, top_k, lr0,
             beta1=0.8, beta2=0.999, eps=1e-8, use_nesterov=True):
    dx = 2.0 / n
    h = project_to_feasible(h_init.copy(), n)
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    nest_t = 1.0
    stall = 0
    cur_lr = lr0

    while get_remaining(start, time_limit) > 0.5:
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)
        tau = tau_start * (tau_end / tau_start) ** frac

        if use_nesterov:
            mu = min((nest_t - 1.0) / (nest_t + 2.0), 0.9)
            y = h + mu * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
        else:
            y = h

        grad = soft_grad(y, n, tau, top_k)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = y - cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, n)

        if t % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 15000:
                cur_lr *= 0.5
                stall = 0
                if cur_lr < 1e-13:
                    break

        if use_nesterov:
            nest_t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * nest_t * nest_t))
            h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_polish(h_init, n, time_limit, lr0, top_k, beta=0.7):
    dx = 2.0 / n
    h = project_to_feasible(h_init.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    lr = lr0

    while get_remaining(start, time_limit) > 0.3:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2500:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.ones(len(top_idx))
        w[np.argmax(top_vals)] = 2.0
        w /= w.sum()

        grad = np.zeros(n)
        for i_idx in range(len(top_idx)):
            pk = w[i_idx]
            k = int(top_idx[i_idx])
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def block_perturb(h, n, rng, block_size=20, n_blocks=5):
    h_p = h.copy()
    for _ in range(n_blocks):
        start_i = rng.randint(0, max(1, n - block_size))
        end_i = start_i + block_size
        h_p[start_i:end_i] = 1.0 - h_p[start_i:end_i]
    return project_to_feasible(h_p, n)
