def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def slsqp_smooth(h, T, maxiter):
    n = len(h)
    dx = 2.0 / n
    def obj(hv):
        corr = corr_array(hv)
        mc = np.max(corr)
        s = np.sum(np.exp(T * (corr - mc)))
        return (mc + np.log(s) / T) * dx
    def obj_grad(hv):
        corr = corr_array(hv)
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        w = exps / np.sum(exps)
        idx = np.argsort(w)[::-1][:50]
        g = np.zeros(n)
        for i in idx:
            if w[i] < 1e-10:
                break
            g += w[i] * subgrad_one(hv, i - (n-1))
        return g * dx
    def con_eq(hv):
        return np.sum(hv) - n / 2.0
    def con_eq_jac(hv):
        return np.ones(n)
    
    try:
        r = minimize(obj, h, jac=obj_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,

                     options={'maxiter': maxiter, 'ftol': 1e-12})

            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

        return np.ones(n)

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_array(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    return g

def fista_opt(h, n_iter, T0, T1, lr_init, lr_end, top_k, check_iv):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr_init + (lr_end - lr_init) * f
        corr = corr_array(z)
        mc = np.max(corr)
        if T < 0.05:
            active = int(np.argmax(corr)) - (n - 1)
            g = subgrad_one(z, active) * dx
        else:
            exps = np.exp(T * (corr - mc))
            tot = np.sum(exps)
            if tot < 1e-300:
                active = int(np.argmax(corr)) - (n - 1)
                g = subgrad_one(z, active) * dx
            else:
                w = exps / tot
                idx = np.argsort(w)[::-1][:top_k]
                g = np.zeros(n)
                for i in idx:
                    wi = w[i]
                    if wi < 1e-10:
                        break
                    g += wi * subgrad_one(z, i - (n - 1))

        tk = tn

                bc = c
                bh = x.copy()

    if c < bc:
        bc = c
        bh = x.copy()
    return bh, bc

def _sobj_grad(hv):
    n = _SS['n']
    T = _SS['T']
    dx = 2.0 / n
    corr = corr_array(hv)
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    w = exps / np.sum(exps)
    idx = np.argsort(w)[::-1][:60]
    g = np.zeros(n)
    for i in idx:
        if w[i] < 1e-10:
            break
        g += w[i] * subgrad_one(hv, i - (n - 1))
    return g * dx

def _scon_fn(hv):
    return np.sum(hv) - _SS['n'] / 2.0

def _scon_jac(hv):
    return np.ones(_SS['n'])

_SS = {}

def slsqp_smooth(h, T, maxiter):
    n = len(h)
    _SS['n'], _SS['T'] = n, T
    try:
        r = minimize(_sobj_fn, h, jac=_sobj_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon_fn, 'jac': _scon_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-13})
        hx = project_c(r.x, n)
        if r.x is not None and compute_c5(hx) < compute_c5(h):
            return hx
    except Exception:
        pass
    return h

def qp_polish(h, top_k, maxiter):
    n = len(h)
    dx = 2.0 / n
    corr = corr_array(h)
    top_idx = np.argsort(corr)[::-1][:top_k]
    shifts = top_idx - (n - 1)
    def qp_obj(hv):
        c = corr_array(hv)
        return np.sum(c[top_idx]) * dx
    def qp_grad(hv):
        g = np.zeros(n)
        for s in shifts:
            g += subgrad_one(hv, s)
        return g * dx

                     bounds=[(0.0, 1.0)] * n,

                     options={'maxiter': maxiter, 'ftol': 1e-14})
        _SS['n'] = n
        hx = project_c(r.x, n)
        if r.x is not None and compute_c5(hx) < compute_c5(h):
