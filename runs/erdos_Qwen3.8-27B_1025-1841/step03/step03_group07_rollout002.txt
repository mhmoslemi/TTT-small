def max_subgrad(h, n, top_k=1):
    """Compute subgradient of max_k corr[k]*dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    # Hard max: only the argmax contributes
    max_val = top_vals.max()
    is_max = np.equal(top_vals, max_val)
    w = is_max.astype(float)
    w /= w.sum()
    
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = w[i_idx]
        if pk == 0:
            continue

        s = k - n + 1
        # corr[k] = sum_i h[i] * (1-h[i+s]) for valid i
        # d/dh[i] = (1-h[i+s]) for valid i (when h[i] is the variable)
        # d/dh[i+s] = -h[i] for valid i+s (when h[i+s] is the variable)
        j0 = max(0, s)
        j1 = min(n, s + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - s])
        j0b = max(0, -s)
        j1b = min(n, n - s)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + s]
    grad *= dx
    obj = corr.max() * dx
    return obj, grad, corr

        h_p, c5_p = nesterov_opt(h_p, n_p, tau=20.0, time_limit=tl, lr=0.04, top_k=3)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 5: Final refinement with very low tau
    rem = get_remaining(start_time, budget_s)
    if rem > 50:
        tl = min(45, rem * 0.5)
        h_work, c5 = nesterov_opt(best_h, len(best_h), tau=2.0, time_limit=tl, lr=0.005, top_k=1)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 6: Try at higher resolution
    rem = get_remaining(start_time, budget_s)
    if rem > 100:
        n_hi = 8000
        h_hi = interp_periodic(best_h, n_hi)
        h_hi = project_to_feasible(h_hi, n_hi)
        tl = min(80, rem * 0.35)
        h_hi, c5_hi = nesterov_opt(h_hi, n_hi, tau=15.0, time_limit=tl, lr=0.04, top_k=3)
        if c5_hi < best_c5:
            best_c5, best_h = c5_hi, h_hi.copy()

    # Final
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
