import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_lag(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_v2(h, n_iter, T0, T1, lr0, lr1, top_k=20, check=30):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr0 + (lr1 - lr0) * f
        # Compute smoothed gradient
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        shift_corr = corr - mc
        # Use top_k for efficiency
        exps = T * shift_corr
        exps = np.exp(np.clip(exps, -50, 50))
        tot = np.sum(exps)
        if tot < 1e-300:
            idx = np.argmax(corr)
            g = subgrad_lag(z, idx - (n-1)) * dx
        else:
            w = exps / tot
            # Get top_k indices
            top_idx = np.argsort(w)[::-1][:top_k]

def slsqp_solve(h, T=5.0, maxiter=400):
    n = len(h)
    dx = 2.0 / n
    
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    
    def obj_grad(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        w = exps / tot
        g = np.zeros(n)
        for i in range(len(corr)):
            wi = w[i]
            if wi < 1e-12:
                continue
            g += wi * subgrad_lag(hv, i - (n-1))
        return g * dx
    
    try:
        r = minimize(obj, h, jac=obj_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

    """Quaternary Thue-Morse-like sequence"""
    t = np.array([0.0])
    target = n
    while len(t) < target:
        t = np.concatenate([t, 1.0 - t])
    return t[:target].astype(float)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_lag(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_opt(h, n_iter, T0, T1, lr0, lr1, top_k=16, check=30):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr0 + (lr1 - lr0) * f
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(np.clip(T * (corr - mc), -60, 60))
        tot = np.sum(exps)
        if tot < 1e-300:
            g = subgrad_lag(z, int(np.argmax(corr)) - (n - 1)) * dx
        else:
            w = exps / tot
            top_idx = np.argsort(w)[::-1][:top_k]
            g = np.zeros(n)
            for i in top_idx:
                wi = w[i]
                if wi < 1e-14:
                    break
                g += wi * subgrad_lag(z, i - (n - 1))
            g *= dx
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + ((tk - 1.0) / tn) * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def _sg_obj(hv, T, n):
    dx = 2.0 / n
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(np.clip(T * (corr - mc), -200, 200)))) / T) * dx

def _sg_grad(hv, T, n, top_k=30):
    dx = 2.0 / n
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(corr)
    exps = np.exp(np.clip(T * (corr - mc), -200, 200))
    tot = np.sum(exps)
    if tot < 1e-300:
        return subgrad_lag(hv, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    top_idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in top_idx:
        wi = w[i]
        if wi < 1e-14:
            break
        g += wi * subgrad_lag(hv, i - (n - 1))
    return g * dx

def slsqp_solve(h, T=4.0, maxiter=250, top_k=30):
    n = len(h)
    try:
        r = minimize(lambda x: _sg_obj(x, T, n), h,
                     jac=lambda x: _sg_grad(x, T, n, top_k),
                     method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
                                   'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def qtm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].astype(float)

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def downsample(h, nn):

        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)
