import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def topk_subgrad(h, corr, k_offset, top_k):
    """Compute subgradient using top_k lags (0-indexed in corr)."""
    n = len(h)
    # corr has length 2n-1, index 0 corresponds to lag -(n-1)
    # lag = idx - (n-1)
    mc = np.max(corr)
    # Get top_k indices
    idx = np.argsort(corr)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        lag = i - (n - 1)
        s = abs(lag)
        if s >= n:
            continue
        if lag >= 0:
            g[s:] += 1.0 - h[:n-s]
            g[:n-s] -= h[s:]
        else:
            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
    return g / top_k

    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** f
        lr = lr_start * (lr_end / lr_start) ** f
        K = int(K_start + (K_end - K_start) * f)
        
        corr = corr_full(z)
        mc = np.max(corr)

            g = topk_subgrad(z, corr, n-1, K_use)
        else:
            w = exps / tot

            g = np.zeros(n)

                lag = i - (n-1)

                if s >= n:

                    g[:n-s] -= wi * z[s:]

                    g[:n-s] += wi * (1.0 - z[s:])
                    g[s:] -= wi * z[:n-s]

def coord_descent(h, n_passes=200, lr=0.001, seed=42):

    n = len(h)
    dx = 2.0 / n

    best_c = compute_c5(x)

                    if j >= s:

            proj_lo = max(0.0, x[j] - lr * gj * dx)

        # Project to constraint

        c = compute_c5(x)
        if c < best_c:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_arr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_lag(h, lag):
    n = len(h)
    s = abs(lag)
    g = np.zeros(n)
    if s >= n:
        return g
    if lag >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_adaptive(h, n_iter, T_s, T_e, lr_s, lr_e, K_s, K_e, check, t_deadline):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()

    for it in range(n_iter):
        if time.time() > t_deadline:
            break
        f = it / max(n_iter - 1, 1)
        lr = lr_s * (lr_e / lr_s) ** f
        K = int(K_s + (K_e - K_s) * f)

        corr = corr_arr(z)
        mc = np.max(corr)
        top_idx = np.argsort(corr)[::-1]
        gap = corr[top_idx[0]] - corr[min(top_idx.shape[0]-1, K)]
        T = max(1.0, min(T_s, 25.0 / (gap + 1e-8)))
        T = max(T, T_e)

        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            sel = top_idx[:1]

def coord_descent(h, n_passes, lr_init, t_deadline, seed=0):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    rng = np.random.RandomState(seed)
    best_c = compute_c5(x)
    best_x = x.copy()
    order = np.arange(n)

    for p in range(n_passes):
        if time.time() > t_deadline:
            break
        rng.shuffle(order)
        lr = lr_init * (0.7 ** (p // 30))
        corr = corr_arr(x)
        mc = np.max(corr)
        top_k = min(40, 2 * n - 1)
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_lags = [ti - (n - 1) for ti in top_idx]

        for j in order:
            gj = 0.0
            for lag in top_lags:
                s = abs(lag)
                if s >= n:
                    continue
                if lag >= 0:
                    if j >= s:
                        gj += (1.0 - x[j - s])
                    if j < n - s:
                        gj -= x[j + s]
                else:
                    if j < n - s:
                        gj += (1.0 - x[j + s])
                    if j >= s:
                        gj -= x[j - s]
            val = x[j] - lr * gj * dx
            x[j] = min(1.0, max(0.0, val))

        x = project_c(x, n)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_x = c, x.copy()

    return best_x, best_c

def slsqp_free(h, T=4.0, maxiter=300, t_deadline=0):
    n = len(h)
    free = (h > 0.03) & (h < 0.97)
    nf = int(np.sum(free))
    if nf < 10 or nf > 700:
        return h
    fixed_sum = np.sum(h[~free])
    target_f = n / 2.0 - fixed_sum

    def obj(v, _h=h, _free=free, _n=n, _T=T, _fs=target_f):
        hx = _h.copy()
        hx[_free] = np.clip(v, 0.0, 1.0)
        dx = 2.0 / _n
        c = corr_arr(hx)
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(_T * (c - mc)))) / _T) * dx

    def grad(v, _h=h, _free=free, _n=n, _T=T):
        hx = _h.copy()
        hx[_free] = np.clip(v, 0.0, 1.0)
        dx = 2.0 / _n
        c = corr_arr(hx)
        mc = np.max(c)
        exps = np.exp(_T * (c - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            w = np.zeros_like(exps); w[np.argmax(c)] = 1.0
        else:
            w = exps / tot
        gf = np.zeros(nf)
        for li in np.argsort(w)[::-1][:30]:
            wi = w[li]
            if wi < 1e-12:
                break
            lag = li - (_n - 1)
            gs = subgrad_lag(hx, lag)
            gf += wi * gs[_free]
        return gf * dx

    def con_fun(v, _fs=target_f):
        return np.sum(v) - _fs
    def con_jac(v):
        return np.ones(len(v))

    v0 = h[free].copy()
    try:
        r = minimize(obj, v0, jac=grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * nf,
                     constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        if r.success or r.fun < obj(v0):
            hx = h.copy()
            hx[free] = np.clip(r.x, 0.0, 1.0)
            hx = project_c(hx, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h
