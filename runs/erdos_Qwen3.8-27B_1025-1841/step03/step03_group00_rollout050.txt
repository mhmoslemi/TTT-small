import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    # Ensure 0 <= h <= 1 and sum(h) = n/2 exactly
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff/cnt, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff))+1:]
            for i in idx:
                h[i] = min(1.0, h[i] + diff)
        else:
            idx = np.argsort(h)[:int(abs(diff))+1:]
            for i in idx:
                h[i] = max(0.0, h[i] + diff)
    return h

def generate_m_sequence(k):
    """Generate binary m-sequence of length 2^k - 1"""
    n = (1 << k) - 1
    # Simple LFSR
    # Use a primitive polynomial - for k=4, x^4+x+1
    if k == 4:
        poly = 0b10011  # x^4 + x + 1
    elif k == 5:
        poly = 0b100111  # x^5 + x^2 + 1
    elif k == 6:
        poly = 0b1000001  # x^6 + x + 1
    else:
        return None
    state = 1
    seq = []
    for _ in range(n):
        seq.append(state & 1)
        bit = bin(state & poly).count('1') & 1
        if bit:
            state = (state >> 1) | (1 << (k-1))
        else:
            state >>= 1
    return np.array(seq, dtype=float)

def tile_pattern(pat, n):
    """Tile pattern to length n"""
    p = len(pat)
    reps = n // p + 1
    out = np.tile(pat, reps)[:n]
    return out

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_optimize(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -100, 100)
        logsum = mc + np.log(np.sum(np.exp(shifted))) / T
        viol = np.sum(hv) - n / 2.0
        return logsum * dx + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def simulated_annealing(h_start, n, rng, n_iterations=50000):
    """SA on continuous values [0,1] with binary swappable values"""
    h = h_start.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    T_start = 0.05
    T_end = 0.001
    for it in range(n_iterations):
        T = T_start * (T_end / T_start) ** (it / n_iterations)
        # Random mutation: pick a few indices and perturb
        n_mut = rng.randint(1, 5)
        idx = rng.choice(n, n_mut, replace=False)
        old_vals = h[idx].copy()
        h[idx] = np.clip(h[idx] + rng.randn(n_mut) * 0.3, 0, 1)
        h = make_feasible(h, n)
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[idx] = old_vals
            h = make_feasible(h, n)
    return best_h, best_c5
