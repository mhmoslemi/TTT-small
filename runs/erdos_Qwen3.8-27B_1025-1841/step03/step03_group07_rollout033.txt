def subgrad_direct(h, n, time_limit, top_k=5):

    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5, _ = compute_c5_full(h)
    best_h = h.copy()
    start = time.time()

    beta = 0.7
    stall = 0
    best_obj = best_c5
    it = 0

        max_c = corr.max() * dx
        if max_c < best_c5 - 1e-14:
            best_c5 = max_c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            lr *= 0.7
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.exp(10.0 * (top_vals - top_vals.max()))
        w /= w.sum()
        
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = w[ii]
            k = top_idx[ii]
            sh = k - n + 1
            j0 = max(0, -sh)
            j1 = min(n, n - sh)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0+sh:j1+sh])
            j0b = max(0, sh)
            j1b = min(n, n + sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b-sh:j1b-sh]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        it += 1
    
    return best_h, best_c5

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_grad_softmax(h, n, tau, top_k=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    max_tv = top_vals.max()
    shifted = tau * (top_vals - max_tv)
    shifted = np.clip(shifted, -80.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        pk = p[ii]
        if pk < 1e-12:
            continue
        k = top_idx[ii]
        sh = k - n + 1
        # Term 1: (1 - h[j+sh]) for valid j
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0+sh:j1+sh])
        # Term 2: -h[j-sh] for valid j
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b-sh:j1b-sh]
    grad *= dx
    return grad, corr.max() * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    eps = 1e-8
    
    while time.time() - start < time_limit:
        t += 1
        grad, c5_now = compute_grad_softmax(h, n, tau, top_k)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            lr *= 0.7
            stall = 0
            if lr < 1e-10:
                break
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    
    while time.time() - start < time_limit:
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.95)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        grad, c5_now = compute_grad_softmax(y, n, tau, top_k)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            lr *= 0.7
            stall = 0
            if lr < 1e-12:
                break
        
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def binary_proj(h, n):
    """Project to {0,1} while maintaining sum constraint"""
    target = n // 2
    thresh = np.sort(h)[::-1][target - 1] if target > 0 else 0
    h_bin = (h >= thresh).astype(float)
    # Fix sum
    s = h_bin.sum()
    if s > target:
        # Need to turn off some 1s
        excess = s - target
        ones = np.where(h_bin == 1)[0]
        # Turn off the ones with lowest h values
        order = np.argsort(h[ones])[::-1]
        for i in range(min(excess, len(order))):
            h_bin[ones[order[i]]] = 0
    elif s < target:
        deficit = target - s
        zeros = np.where(h_bin == 0)[0]
        order = np.argsort(h[zeros])[::-1]
        for i in range(min(deficit, len(order))):
            h_bin[zeros[order[i]]] = 1
    return h_bin

def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))
