def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    ox = np.linspace(0.0, 2.0, old_n)
    nx = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(nx, ox, h), new_n)

def symmetric_h(h, n):
    half = n // 2
    hs = h.copy()
    for i in range(half):
        v = (h[i] + 1.0 - h[n - 1 - i]) / 2.0
        hs[i] = v
        hs[n - 1 - i] = 1.0 - v
    hs[half] = 0.5
    return hs

def half_to_h(z, n):
    half = n // 2
    dz = np.clip(z, 0.0, 1.0)
    h = np.empty(n)
    h[:half] = dz
    h[half + 1:] = 1.0 - dz
    h[half] = 0.5
    return h

def h_to_half(h, n):
    return h[:n // 2].copy()

# --- Objective functions for SLSQP (full dimension) ---

def _full_lse_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(_G_T * (corr - mc)))) / _G_T) * dx

def _full_active_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.mean(corr[_G_TOP])) * dx

def _full_con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _full_con_jac(x):
    return np.ones(_G_N)

# --- Objective functions for L-BFGS-B (half dimension, symmetric) ---

def _half_lse_obj(z):
    n = _G_N
    h = half_to_h(z, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(_G_T * (corr - mc)))) / _G_T) * dx

def _half_active_obj(z):
    n = _G_N
    h = half_to_h(z, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.mean(corr[_G_TOP])) * dx

def binary_snap(h, n, tlimit):
    best = h.copy()
    best_c = compute_c5(best)
    t0 = time.time()
    for j in range(n):
        if time.time() - t0 > tlimit:
            break
        for val in (0.0, 1.0):
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def coord_descent(h, n, tlimit):
    best_c = compute_c5(h)
    t0 = time.time()
    for r in range(50):
        improved = False
        for j in range(n):
            if time.time() - t0 > tlimit:
                return h, best_c
            old = h[j]
            bt = old
            bc = best_c
            for cd in (0.0, 1.0, 0.5,
                       max(0.0, old - 0.03), max(0.0, old - 0.01),
                       min(1.0, old + 0.01), min(1.0, old + 0.03)):
                if abs(cd - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cd
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-14:
                    bc = c
                    bt = cd
            if bt != old:
                h[j] = bt
                h = make_feasible(h, n)
                best_c = bc
                improved = True
        if not improved:
            break
    return h, best_c

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-15:

        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:

    ox = np.linspace(0.0, 2.0, old_n)
    nx = np.linspace(0.0, 2.0, new_n)
