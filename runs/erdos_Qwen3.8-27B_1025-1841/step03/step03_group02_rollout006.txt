def make_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)

        if abs(diff) < 1e-14:

    target = n // 2
    diff = target - b.sum()
    if diff > 0:
        idx = np.where(b == 0)[0]
        order = idx[np.argsort(-h[idx])]
        b[order[:int(diff)]] = 1

_slq_n = 0
_slq_T = 1.0

def slq_obj(x):
    n = _slq_n
    N = 2 * n
    corr = np.real(ifft(np.conj(fft(x, N)) * fft(1.0 - x, N)))
    mc = np.max(corr)
    T = _slq_T
    logsum = np.log(np.sum(np.exp(T * (corr - mc))))
    return (mc + logsum / T) * 2.0 / n

def slq_con_v(x):
    return x.sum() - len(x) / 2.0

def slq_con_j(x):
    return np.ones(len(x))

    _slq_n = n
    _slq_T = T
    res = minimize(slq_obj, h.copy(), method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq','fun':slq_con_v,'jac':slq_con_j}],
                   options={'maxiter':maxiter, 'ftol':1e-15})
    return res.x

import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def compute_c5_fft(h):
    n = len(h)
    N = 2 * n
    corr = np.real(ifft(np.conj(fft(h, N)) * fft(1.0 - h, N)))
    return float(np.max(corr) * 2.0 / n)

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(np.float64)
    target = n // 2
    diff = int(target - b.sum())
    if diff > 0:
        idx = np.where(b == 0)[0]
        order = idx[np.argsort(-h[idx])]
        b[order[:diff]] = 1.0
    elif diff < 0:
        idx = np.where(b == 1)[0]
        order = idx[np.argsort(h[idx])]
        b[order[:abs(diff)]] = 0.0
    return b

def sa_point(b, n, iters, T0, T1, rs, deadline):
    curr = compute_c5_fft(b)
    best = curr
    best_b = b.copy()
    for it in range(iters):

            break
        T = T0 * (T1 / T0) ** (it / max(iters, 1))

        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rs.randint(len(ones))]
        j = zeros[rs.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        new = compute_c5_fft(b)
        if new <= curr or (T > 1e-15 and rs.random() < np.exp(-(new - curr) / T)):

            if new < best:

                best_b = b.copy()
        else:

_slq_n = 0
_slq_T = 1.0

def slq_obj(x):
    n = _slq_n
    N = 2 * n
    corr = np.real(ifft(np.conj(fft(x, N)) * fft(1.0 - x, N)))
    mc = np.max(corr)
    T = _slq_T
    logsum = np.log(np.sum(np.exp(T * (corr - mc))))
    return (mc + logsum / T) * 2.0 / n

def slq_v(x):
    return x.sum() - len(x) / 2.0

def slq_j(x):
    return np.ones(len(x))

def slsqp_opt(h, n, T, maxiter, deadline):
    global _slq_n, _slq_T
    _slq_n = n
    _slq_T = T
    res = minimize(slq_obj, h.copy(), method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': slq_v, 'jac': slq_j}],
                   options={'maxiter': maxiter, 'ftol': 1e-16})
    if time.time() > deadline:
        return None
    return res.x

def coord_desc(h, n, rounds, deadline):
    best_c5 = compute_c5_fft(h)
    for r in range(rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline - 3:
                break
            old = h[j]
            best_v = old
            best_c = best_c5
            cands = [0.0, 1.0, max(0.0, old - 0.05), max(0.0, old - 0.01),
                     min(1.0, old + 0.01), min(1.0, old + 0.05)]
            for cv in cands:
                if abs(cv - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5_fft(trial)
                if c < best_c:
                    best_c = c
                    best_v = cv
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def gen_thue_morse(n):
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    b = t.astype(np.float64)
    target = n // 2
    diff = int(target - b.sum())
    if diff > 0:
        idx = np.where(b == 0)[0]
        b[idx[:diff]] = 1.0
    elif diff < 0:
        idx = np.where(b == 1)[0]
        b[idx[:abs(diff)]] = 0.0
    return b
