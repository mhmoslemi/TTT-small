```python
def grad_max_corr(h, n, top_k=1):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    if top_k >= m:
        top_idx = np.arange(m)
        w = np.ones(m, dtype=np.float64) / m
    else:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        tv = corr[top_idx].copy()
        mv = float(tv.max())
        ev = np.exp(tv - mv)
        w = ev / ev.sum()

                grad[:n-s] += wi * (1.0 - h[s:])
                grad[s:] -= wi * h[:n-s]
    return grad * dx
```

```python
    start = time.time()
    
    while time.time() - start < time_limit:
        improved = False
        # Try random swaps
        for _ in range(1000):
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if h_b[i] == h_b[j]:
                continue
            # Swap: h_b[i] ↔ h_b[j]
            h_b[i], h_b[j] = h_b[j], h_b[i]
            c5_new = compute_c5(h_b)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_b.copy()
                improved = True
            else:
                # Undo
                h_b[i], h_b[j] = h_b[j], h_b[i]
        
        if not improved:
            break
    
    return best_h, best_c5
```
