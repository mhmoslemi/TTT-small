    v = 1.0 - h

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        shift = int(np.argmax(corr)) - (n - 1)
        return _subgrad_one(h, shift) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter=5000, T0=40.0, T1=0.1, lr=0.15, check=40, top_k=80):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    best_it = 0
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.3 * f)
        g = sm_grad(z, T, top_k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh, best_it = c, x.copy(), it
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def downsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def sharpen(h, alpha, n):
    hs = 0.5 + alpha * (h - 0.5)
    return project_c(hs, n)

def targeted_fix(h, n, n_iter=500, lr=0.05):
    x = h.copy()
    best_c = compute_c5(x)
    best_h = x.copy()
    for it in range(n_iter):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        active = np.where(corr > mc - 0.01 * mc)[0]
        g = np.zeros(n)
        wts = corr[active] - mc * 0.5 + 1e-10
        wts = np.maximum(wts, 0)
        wts = wts / (wts.sum() + 1e-30)
        for i, wi in zip(active, wts):
            shift = i - (n - 1)
            g += wi * _subgrad_one(x, shift)
        x = project_c(x - lr * g * (2.0 / n), n)
        c = compute_c5(x)
        if c < best_c:
            best_c, best_h = c, x.copy()
    return best_h, best_c

_OB = {}

def _sobj(h):
    T = _OB['T']
    n = _OB['n']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _scon(h):
    return np.sum(h) - _OB['n'] / 2.0

def _sjac(h):
    return np.ones(_OB['n'])

def slsqp(h, T=5.0, mi=400):
    n = len(h)
    _OB['T'], _OB['n'] = T, n
    try:
        r = minimize(_sobj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h
