    valid1 = (idx1 >= 0) & (idx1 < n)
    delta[valid1] -= (1 - h[idx1[valid1]])

    idx2 = j - shifts
    valid2 = (idx2 >= 0) & (idx2 < n)
    delta[valid2] += (1 - h[idx2[valid2]])
    
    # Term 3: +h[i+s] for valid i+s
    idx3 = i + shifts
    valid3 = (idx3 >= 0) & (idx3 < n)
    delta[valid3] += h[idx3[valid3]]
    
    # Term 4: -h[j+s] for valid j+s
    idx4 = j + shifts
    valid4 = (idx4 >= 0) & (idx4 < n)
    delta[valid4] -= h[idx4[valid4]]
    
    return delta

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def swap_delta_vec(h, n, i, j):
    shifts = np.arange(-(n-1), n)
    delta = np.zeros(2*n - 1)
    idx = i - shifts
    v = (idx >= 0) & (idx < n)
    delta[v] -= (1.0 - h[idx[v]])
    idx = j - shifts
    v = (idx >= 0) & (idx < n)
    delta[v] += (1.0 - h[idx[v]])
    idx = i + shifts
    v = (idx >= 0) & (idx < n)
    delta[v] += h[idx[v]]
    idx = j + shifts
    v = (idx >= 0) & (idx < n)
    delta[v] -= h[idx[v]]
    return delta

def binary_local_search(h, n, time_limit, seed=42, max_no_improve=500):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = (h >= 0.5).astype(np.float64)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        pos = np.where(h == 1)[0]
        h[rng.choice(pos, n_ones - target, replace=False)] = 0
    elif n_ones < target:
        pos = np.where(h == 0)[0]
        h[rng.choice(pos, target - n_ones, replace=False)] = 1
    
    corr = np.correlate(h, 1.0 - h, 'full')
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    no_improve = 0
    
    while time.time() - start < time_limit:
        if time.time() - start > time_limit:
            break
        max_idx = np.argmax(corr)
        curr_max = corr[max_idx]
        s_star = max_idx - (n - 1)
        
        if s_star >= 0:
            i_range = np.arange(s_star, n)
        else:
            i_range = np.arange(0, n + s_star)
        i_shifted = i_range - s_star
        bad_mask = (h[i_range] == 1) & (h[i_shifted] == 0)
        bad_positions = i_range[bad_mask]
        
        improved = False
        for _ in range(300):
            if time.time() - start > time_limit:
                return best_h, best_c5
            if len(bad_positions) > 0 and rng.random() < 0.6:
                i = bad_positions[rng.randint(len(bad_positions))]
            else:

            while h[j] != 0:
                j = rng.randint(n)
            if i == j:
                continue

            if (corr + delta).max() < curr_max - 1e-10:

            if no_improve > max_no_improve:

def nesterov_optimize(h, n, tau, time_limit, lr=None, top_k=10):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0
    
    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = corr.max()
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = p[ii]
            k = top_idx[ii]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - y[np.arange(j0, j1) - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * y[np.arange(j0b, j1b) + sh]
        grad *= dx
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        if it % 50 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 50
            if stall > 5000:
                lr *= 0.6
                stall = 0
                if lr < 1e-15:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5
