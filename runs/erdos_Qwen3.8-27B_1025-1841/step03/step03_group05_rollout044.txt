```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        clipped = np.clip(h - mid, 0.0, 1.0)
        if np.sum(clipped) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if T > 1000:
        idx = np.argmax(corr)
        return _subgrad_one(h, idx - (n - 1)) * dx
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = np.argmax(corr)
        return _subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    # Get top-k weights
    sorted_idx = np.argsort(w)[::-1]
    sorted_idx = sorted_idx[:top_k]
    g = np.zeros(n)
    for i in sorted_idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=120):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if T > 500:
        idx = int(np.argmax(corr))
        return _subgrad_one(h, idx - (n - 1)) * dx
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return _subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    sidx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in sidx:
        wi = w[i]
        if wi < 1e-11:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

        T = T0 + (T1 - T0) * (1.0 + np.cos(np.pi * f)) / 2.0
        lrt = lr * (1.0 - 0.5 * f * f)
        g = sm_grad(z, T, top_k)

        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)

        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def block_refine(h, n_blocks=None, block_size=32, n_passes=3, budget_check=None, t0=None, budget_s=1300):
    n = len(h)
    h_work = h.copy()
    best_c = compute_c5(h_work)
    if n_blocks is None:
        n_blocks = n // block_size
    for p in range(n_passes):
        if budget_check is not None and (budget_check() > 0.85 * budget_s):
            break
        perm = np.random.RandomState(42 + p).permutation(n_blocks)
        for bi in perm:
            start = bi * block_size
            end = min(start + block_size, n)
            B = end - start
            S = float(np.sum(h_work[start:end]))
            h_fixed = h_work.copy()
            def bobj(xb):
                h_try = h_fixed.copy()
                h_try[start:end] = np.clip(xb, 0, 1)
                c = max(0, np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * (2.0 / n))
                return c
            res = minimize(bobj, h_work[start:end].copy(), method='Nelder-Mead',
                           options={'maxiter': 200, 'xatol': 1e-5, 'fatol': 1e-8})
            h_new = np.clip(res.x, 0, 1)
            if abs(np.sum(h_new) - S) < 0.5:
                h_work[start:end] = h_new
                h_work = project_c(h_work, n)
                c = compute_c5(h_work)
                if c < best_c:
                    best_c = c
    return h_work, best_c

    starts.append(tm_seq(n))
