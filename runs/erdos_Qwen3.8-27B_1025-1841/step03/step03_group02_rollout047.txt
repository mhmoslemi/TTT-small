def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_feasible(v, n):

    target = n / 2.0
    lo = -1.0
    hi = 2.0

        else:

        bs = rng.randint(2, 8)
        start = rng.randint(0, max(n - bs, 1))
        end = start + bs
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def projected_gradient(h, n, T_start, T_end, n_iters, step, start_time, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        frac = float(it) / max(n_iters - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        g = smooth_grad(best_h, T)
        alpha = step
        for _ in range(20):
            v = best_h - alpha * g
            p = proj_feasible(v, n)
            c5_new = compute_c5(p)
            if c5_new < best_c5 + 0.001 * alpha * np.max(np.abs(g)):
                break
            alpha *= 0.5
            if alpha < 1e-6:
                break
        p = proj_feasible(best_h - alpha * g, n)
        c5_new = compute_c5(p)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = p
    return best_h, best_c5

def slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mx = np.max(corr)
    shifted = corr - mx
    return (mx + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_sum(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_sum_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum, 'jac': slsqp_sum_jac}]
    def obj(x):
        return slsqp_obj(x, n, T)
    try:
        res = minimize(obj, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    improved = True
    for rnd in range(max_rounds):
        if not improved:
            break
        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 2:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binary SA from parent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        b, c5b = binary_sa(h_init, n0, 70000, 0.012, 0.00001, rng)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()

    # Phase 2: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        b, c5b = binary_sa_block(best_h, n0, 40000, 0.008, 0.00005, rng)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()

    # Phase 3: Projected gradient (multi-T)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        best_h, best_c5 = projected_gradient(best_h, n0, 80.0, 5.0, 150, 0.05, start_time, budget_s)

    # Phase 4: Binary SA from continuous
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        b, c5b = binary_sa(best_h, n0, 50000, 0.006, 0.00001, rng)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()

    # Phase 5: Projected gradient again
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = projected_gradient(best_h, n0, 40.0, 2.0, 120, 0.05, start_time, budget_s)

    # Phase 6: SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        x_res = slsqp_polish(best_h, n0, 50.0, 200, 1e-14)
        x_res = make_feasible(x_res, n0)
        c5r = compute_c5(x_res)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = x_res.copy()

    # Phase 7: ILS - perturbation + re-optimization
    remaining = budget_s - (time.time() - start_time)
    ills_rounds = max(1, min(5, int(remaining // 80)))
    for ir in range(ills_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 70:
            break
        k_swap = rng.randint(4, 12)
        pert = best_h.copy()
        ones = np.where(pert > 0.5)[0]
        zeros = np.where(pert < 0.5)[0]
        if len(ones) < k_swap or len(zeros) < k_swap:
            k_swap = min(k_swap, len(ones), len(zeros))
        if k_swap == 0:
            break
        fo = rng.choice(ones, size=k_swap, replace=False)
        fz = rng.choice(zeros, size=k_swap, replace=False)
        pert[fo] = 0.0
        pert[fz] = 1.0
        pert = make_feasible(pert, n0)
        b_ip, c5_ip = binary_sa(pert, n0, 20000, 0.008, 0.00002, rng)
        if c5_ip < best_c5:
            best_c5 = c5_ip
            best_h = b_ip.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            best_h, best_c5 = projected_gradient(best_h, n0, 30.0, 3.0, 80, 0.04, start_time, budget_s)
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            x_res = slsqp_polish(best_h, n0, 30.0, 150, 1e-14)
            x_res = make_feasible(x_res, n0)
            c5r = compute_c5(x_res)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = x_res.copy()

    # Phase 8: Try coarser resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120 and n0 > 480:
        n_co = n0 // 2
        h_co = upsample(best_h, n_co)
        b_co, c5_co = binary_sa(h_co, n_co, 30000, 0.01, 0.00002, rng)
        if c5_co < best_c5:
            best_c5 = c5_co
            best_h = b_co.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 50:
            best_h, best_c5 = projected_gradient(best_h, len(best_h), 50.0, 3.0, 80, 0.05, start_time, budget_s)
            remaining = budget_s - (time.time() - start_time)
            if remaining > 30:
                x_res = slsqp_polish(best_h, len(best_h), 20.0, 150, 1e-14)
                x_res = make_feasible(x_res, len(best_h))
                c5r = compute_c5(x_res)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = x_res.copy()

    # Phase 9: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 20, start_time, budget_s)

    # Phase 10: Final SLSQP + gradient
    remaining = budget_s - (time.time() - start_time)
    n_pts = len(best_h)
    if remaining > 40:
        x_res = slsqp_polish(best_h, n_pts, 100.0, 200, 1e-15)
        x_res = make_feasible(x_res, n_pts)
        c5r = compute_c5(x_res)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = x_res.copy()
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, best_c5 = projected_gradient(best_h, len(best_h), 20.0, 1.0, 80, 0.03, start_time, budget_s)

    # Final polish
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n_pts = len(best_h)
    if remaining > 15:
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 5, start_time, budget_s)
        best_h = make_feasible(best_h, n_pts)
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
