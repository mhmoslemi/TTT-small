import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_shifted = np.exp(T * shifted)
    sum_exp = np.sum(exp_shifted)
    probs = exp_shifted / sum_exp
    obj = (mc + np.log(sum_exp) / T) * dx
    
    grad = np.zeros(n)
    # For each m (0 to 2n-2), k = m - (n-1)
    # If k >= 0: grad[0:n-k] += probs[m]*(1-x[k:n])
    #            grad[k:n] -= probs[m]*x[0:n-k]
    # If k < 0, k' = -k: grad[k':n] += probs[m]*(1-x[0:n-k'])
    #              grad[0:n-k'] -= probs[m]*x[k':n]
    for m in range(2 * n - 1):
        pm = probs[m]
        if pm < 1e-18:
            continue
        k = m - (n - 1)
        if k >= 0:
            # First term: j in [0, n-1-k]
            grad[:n-k] += pm * (1.0 - x[k:n])
            # Second term: j in [k, n-1]
            grad[k:] -= pm * x[:n-k]
        else:
            kp = -k
            # First term: j in [kp, n-1]
            grad[kp:] += pm * (1.0 - x[:n-kp])
            # Second term: j in [0, n-1-kp]
            grad[:n-kp] -= pm * x[kp:n]
    
    grad *= dx
    return obj, grad

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(x_init, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    
    def fun(x):
        return obj_grad(x, n, T)[0]
    
    def jac(x):
        return obj_grad(x, n, T)[1]
    
    res = minimize(fun, x_init.copy(), jac=jac, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:

        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()

        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac

        j = zeros[rng.randint(len(zeros))]

        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:

        improved = False

            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def upsample_h(h, new_n):

    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
