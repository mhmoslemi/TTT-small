from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def sm_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-14:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def binary_snap(h, n):
    """Create a binary-like solution by thresholding and projecting"""
    # Try multiple thresholds
    best_h = None
    best_c = float('inf')
    for t in np.linspace(0.2, 0.8, 13):
        b = (h >= t).astype(float)
        # Adjust to have exactly n/2 ones
        n_ones = int(np.sum(b))
        if n_ones != n // 2:
            # Adjust: if too many ones, flip some to 0; if too few, flip some to 1
            if n_ones > n // 2:
                # Flip the ones with lowest h values
                ones = np.where(b == 1)[0]
                order = np.argsort(h[ones])[::-1]
                flip_count = n_ones - n // 2
                for i in range(flip_count):
                    b[ones[order[i]]] = 0
            else:
                zeros = np.where(b == 0)[0]
                order = np.argsort(h[zeros])[::-1]
                flip_count = n // 2 - n_ones
                for i in range(flip_count):
                    b[zeros[order[i]]] = 1
        c = compute_c5(b)
        if c < best_c:
            best_c = c
            best_h = b.copy()
    return best_h, best_c

def coord_descent(h, n_max_iter=1000, budget=30.0):
    """Coordinate descent minimizing the max correlation"""
    t0 = time.time()
    n = len(h)
    h = h.astype(float).copy()
    bc = compute_c5(h)
    bh = h.copy()

    # shift s means: corr[n-1+s] involves h[i] as h[n+s] when n=i-s, or as 1-h[n] when n=i
    
    for it in range(n_max_iter):
        if time.time() - t0 > budget * 0.9:
            break
        improved = False
        for i in range(n):
            # For coordinate i, the relevant shifts are:
            # - As first factor h[i]: shifts s where 0 <= i-s < n, i.e., s in [i-n+1, i]
            # - As second factor (1-h[i]): shifts s where 0 <= i+s < n, i.e., s in [-i, n-1-i]
            
            # The contribution of h[i] to shift s:
            # If i-s >= 0 and i-s < n: h[i] * (1-h[i-s])
            # If i+s >= 0 and i+s < n: h[i+s] * (1-h[i])
            
            # Total derivative w.r.t. h[i] for shift s:
            # (1-h[i-s]) - h[i+s] (summing over valid terms)
            
            # For the max, we want to find the h[i] that minimizes the max
            # Simple approach: try a few candidate values
            
            old = h[i]
            # Try values: 0, 1, and the value that equalizes the active constraints
            candidates = [0.0, 1.0, old]
            
            # Also try values that would reduce the top correlations
            corr = get_corr(h)
            top_idx = np.argsort(corr)[-5:]
            best_c = compute_c5(h)
            
            for cv in candidates:
                h[i] = cv
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_cv = cv
            if best_cv != old:
                h[i] = best_cv
                improved = True
        
        # Project to maintain constraint
        h = project_c(h, n)
        c = compute_c5(h)
        if c < bc:
            bc, bh = c, h.copy()
        
        if not improved:
            break
    
    return bh, bc

def slsqp_smooth(h, T=5.0, mi=400):
    """SLSQP on smoothed objective"""
    n = len(h)
    dx = 2.0 / n
    
    def obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx
    
    def obj_grad(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        w = exps / np.sum(exps)
        g = np.zeros(n)
        top_k = min(50, n * 3 - 1)
        idx = np.argsort(w)[::-1][:top_k]
        for i in idx:
            if w[i] < 1e-12:
                break
            g += w[i] * _subgrad_one(hv, i - (n - 1))
        return g * dx
    
    def con_fun(hv):
        return np.sum(hv) - n / 2.0
    
    def con_jac(hv):
        return np.ones(n)
    
    try:
        r = minimize(obj, h, method='SLSQP', jac=obj_grad,
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                     options={'maxiter': mi, 'ftol': 1e-12})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-14:
            break
        g += wi * _subgrad_one(h, int(i) - (n - 1))
    return g * dx

def fista(h, n_iter=5000, T0=40.0, T1=0.1, lr=0.15, top_k=80, check=40):

    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()

        T = T0 * (T1 / T0) ** f

        f = min(it / max(5000, 1), 1.0)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.4 * f)
        g = sm_grad(z, T, top_k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        it += 1
        if it % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def _sobj_wrap(hv, T, n, dx):
    c = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _scon_wrap(hv, n):
    return np.sum(hv) - n / 2.0

def _sjac_wrap(hv, n):
    return np.ones(n)

def slsqp_small(h, T=5.0, mi=400):
    n = len(h)
    dx = 2.0 / n
    def obj(hv):
        return _sobj_wrap(hv, T, n, dx)
    def con(hv):
        return _scon_wrap(hv, n)
    def con_j(hv):
        return _sjac_wrap(hv, n)
    try:
        r = minimize(obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con, 'jac': con_j}],
                     options={'maxiter': mi, 'ftol': 1e-13})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def binary_probe(h, n):
    best_h = None
    best_c = float('inf')
    for t in np.linspace(0.3, 0.7, 9):
        b = (h >= t).astype(float)
        n_ones = int(np.sum(b))
        if n_ones != n // 2:
            if n_ones > n // 2:
                ones = np.where(b == 1)[0]
                order = ones[np.argsort(h[ones])[::-1]]
                for i in range(n_ones - n // 2):
                    b[order[i]] = 0
            else:
                zeros = np.where(b == 0)[0]
                order = zeros[np.argsort(h[zeros])[::-1]]
                for i in range(n // 2 - n_ones):
                    b[order[i]] = 1
        c = compute_c5(b)
        if c < best_c:
            best_c = c
            best_h = b.copy()
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    el = lambda: time.time() - t0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = project_c(rng.rand(480), 480)

    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"I: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1: FISTA high-T at n0
    if budget_s - el() > 80:
        b1 = min(80, (budget_s - el()) * 0.25)
        h1, c1 = fista_time(bh, b1, T0=60.0, T1=1.0, lr=0.13, top_k=60, check=30)
        if c1 < bc:
            bc, bh = c1, h1.copy()
        print(f"P1: C5={bc:.6f} t={el():.0f}", flush=True)

    # Phase 2: FISTA low-T at n0
    if budget_s - el() > 60:
        b2 = min(60, (budget_s - el()) * 0.2)
        h2, c2 = fista_time(bh, b2, T0=3.0, T1=0.03, lr=0.035, top_k=120, check=30)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        print(f"P2: C5={bc:.6f} t={el():.0f}", flush=True)

    # Phase 3: SLSQP at n0
    if n0 <= 1024 and budget_s - el() > 50:
        for TV in [15.0, 6.0, 2.5, 1.2]:
            if budget_s - el() < 15:
                break
            hx = slsqp_small(bh, TV, 350)
            cx = compute_c5(hx)
            if cx < bc:
                bc, bh = cx, hx
        print(f"P3: C5={bc:.6f} t={el():.0f}", flush=True)

    # Phase 4: Binary exploration at n0
    if budget_s - el() > 50:
        hb, cb = binary_probe(bh, n0)
        if cb < bc:
            bf = min(30, (budget_s - el()) * 0.4)
            hr, cr = fista_time(hb, bf, T0=20.0, T1=0.5, lr=0.08, top_k=50, check=30)
            if cr < cb:
                bh, bc = hr, cr
            else:
                bh, bc = hb, cb
        print(f"P4: C5={bc:.6f} t={el():.0f}", flush=True)

    # Phase 5: Upsample to n0*2
    n1 = n0 * 2
    if n1 <= 2560 and budget_s - el() > 120:
        hu = upsample(bh, n1)
        cu = compute_c5(hu)
        print(f"  U{n1}: init={cu:.6f}", flush=True)
        if cu < bc:
            bc, bh = cu, hu.copy()
        b5 = min(80, (budget_s - el()) * 0.3)
        h5, c5 = fista_time(bh, b5, T0=40.0, T1=0.5, lr=0.1, top_k=80, check=50)
        if c5 < bc:
            bc, bh = c5, h5.copy()
        b5b = min(50, (budget_s - el()) * 0.2)
        h5b, c5b = fista_time(bh, b5b, T0=2.0, T1=0.03, lr=0.03, top_k=150, check=40)
        if c5b < bc:
            bc, bh = c5b, h5b.copy()
        if budget_s - el() > 30:
            hx5 = slsqp_small(bh, 4.0, 300)
            if compute_c5(hx5) < bc:
                bc, bh = compute_c5(hx5), hx5
        print(f"P5: C5={bc:.6f} t={el():.0f}", flush=True)

    # Phase 6: Upsample to n0*3 (if ≤ 3200)
    n2 = min(n0 * 3, 3200)
    if n2 > n1 and budget_s - el() > 100:
        hu2 = upsample(bh, n2)
        cu2 = compute_c5(hu2)
        if cu2 < bc:
            bc, bh = cu2, hu2.copy()
        b6 = min(70, (budget_s - el()) * 0.35)
        h6, c6 = fista_time(bh, b6, T0=30.0, T1=0.3, lr=0.08, top_k=100, check=60)
        if c6 < bc:
            bc, bh = c6, h6.copy()
        b6b = min(40, (budget_s - el()) * 0.2)
        h6b, c6b = fista_time(bh, b6b, T0=1.5, T1=0.02, lr=0.02, top_k=200, check=50)
        if c6b < bc:
            bc, bh = c6b, h6b.copy()
        print(f"P6: C5={bc:.6f} t={el():.0f}", flush=True)

    # Phase 7: Diverse starts at best resolution
    np_ = len(bh)
    if budget_s - el() > 60:
        # Thue-Morse
        t = np.array([0.0])
        while len(t) < np_:
            t = np.concatenate([t, 1.0 - t])
        tm = project_c(t[:np_], np_)
        ctm = compute_c5(tm)
        if ctm < bc:
            bh, bc = tm, ctm
            if budget_s - el() > 30:
                hr, cr = fista_time(tm, min(30, (budget_s - el()) * 0.4), T0=20.0, T1=0.3, lr=0.07, top_k=60, check=30)
                if cr < bc:
                    bc, bh = cr, hr.copy()
        # Noise perturbations
        for trial in range(3):
            if budget_s - el() < 25:
                break
            sc = 0.03 + 0.04 * trial
            hp = project_c(bh + rng.randn(np_) * sc, np_)
            hr2, cr2 = fista_time(hp, min(25, (budget_s - el()) * 0.4), T0=30.0, T1=0.3, lr=0.08, top_k=60, check=30)
            if cr2 < bc:
                bc, bh = cr2, hr2.copy()
                print(f"  SR{trial}: C5={bc:.6f}", flush=True)

    # Phase 8: Final ultra-fine FISTA
    if budget_s - el() > 15:
        bf = min(12, (budget_s - el()) * 0.6)
        hf, cf = fista_time(bh, bf, T0=2.0, T1=0.01, lr=0.01, top_k=200, check=10)
        if cf < bc:
            bc, bh = cf, hf.copy()

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={el():.0f}", flush=True)
    return bh, bc, len(bh)
