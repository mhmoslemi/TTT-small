    corr = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(flipped, N), N)
    return float(np.max(corr[:2*n-1]) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = h < 1.0 - 1e-15 if diff > 0 else h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    for it in range(n_iters):
        if time_limit_fn() < 5:
            break
        # Cooling schedule
        frac = it / max(n_iters - 1, 1)

        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        
        h[oi] = 0.0
        h[oj] = 1.0
        new_c5 = compute_c5_fft(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            # Accept
            current_c5 = new_c5
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, oj)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    flipped = (1.0 - h)[::-1]
    corr = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(flipped, N), N)
    return float(np.max(corr[:2 * n - 1]) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = h < 1.0 - 1e-15 if diff > 0 else h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def fix_binary_count(h, n):
    n_target = n // 2
    n_ones = int(np.sum(h > 0.5))
    if n_ones != n_target:
        if n_ones > n_target:
            idx = np.where(h > 0.5)[0]
            rng2 = np.random.RandomState(0)
            rng2.shuffle(idx)
            for k in range(n_ones - n_target):
                h[idx[k]] = 0.0
        else:
            idx = np.where(h <= 0.5)[0]
            rng2 = np.random.RandomState(0)
            rng2.shuffle(idx)
            for k in range(n_target - n_ones):
                h[idx[k]] = 1.0
    return h

def binary_sa_global(h_bin, n, rng, T_init, T_final, time_limit_fn):
    h = h_bin.copy()
    current_c5 = compute_c5_fft(h)
    best_c5 = current_c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    n_ones = len(ones)
    n_zeros = len(zeros)
    it = 0
    while time_limit_fn() > 3:
        it += 1
        T = T_init * (T_final / T_init) ** (it / 200000.0)
        if T < T_final:
            T = T_final
        ii = rng.randint(n_ones)
        jj = rng.randint(n_zeros)
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        new_c5 = compute_c5_fft(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            current_c5 = new_c5
            ones[ii] = oj
            zeros[jj] = oi
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best_c5

def binary_sa_local(h_bin, n, rng, T_init, T_final, window, time_limit_fn):
    h = h_bin.copy()
    current_c5 = compute_c5_fft(h)
    best_c5 = current_c5
    best_h = h.copy()
    it = 0
    while time_limit_fn() > 3:
        it += 1
        T = T_init * (T_final / T_init) ** (it / 100000.0)
        if T < T_final:
            T = T_final
        # Pick a random position and try to swap with a neighbor within window
        pos = rng.randint(n)
        if h[pos] > 0.5:
            # Find a nearby zero
            found = False
            for _ in range(30):
                d = rng.randint(1, window + 1)
                direction = 1 if rng.rand() < 0.5 else -1
                target = (pos + direction * d) % n
                if h[target] <= 0.5:
                    found = True
                    break
            if not found:
                target = rng.randint(n)
                while h[target] > 0.5:
                    target = rng.randint(n)
        else:
            found = False
            for _ in range(30):

                target = (pos + direction * d) % n

            if not found:

                while h[target] <= 0.5:
                    target = rng.randint(n)
        swap1, swap2 = int(h[pos]), int(h[target])

        h[target] = 1.0 - h[target]

    return best_h, best_c5

def lbgfb_smooth(h_start, n, T, mu, max_iter=200):
    dx = 2.0 / n

    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    # Make it have exactly n/2 ones
    if int(np.sum(tm)) != n // 2:
        ones = np.where(tm > 0.5)[0]
        zeros = np.where(tm <= 0.5)[0]
        cur = int(np.sum(tm))
        if cur > n // 2:
            for k in range(cur - n // 2):
                tm[ones[k]] = 0
        else:
            for k in range(n // 2 - cur):
                tm[zeros[k]] = 1
    return tm

def golden_pattern(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    return fix_binary_count(h, n)
