    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = np.sum(np.exp(t * (corr - mc)))
    val = (mc + np.log(s) / t) * dx
    val += lam * (np.sum(h) - n / 2.0) ** 2

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    ox = np.linspace(0, 2, old_n)
    nx = np.linspace(0, 2, new_n)
    return make_feasible(np.interp(nx, ox, h), new_n)

def slsqp_pass(h, n, t, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        val = (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx
        return val
    def con_f(x):
        return np.sum(x) - n / 2.0
    def con_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h, n)
