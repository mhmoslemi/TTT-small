```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        step = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(h) if diff > 0 else np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(diff, 1.0 - h[i])
                h[i] += add
                diff -= add
            else:
                sub = min(-diff, h[i])
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h
```

```python
        # T2: -w * h[j-d] for j where 0 <= j-d < n
        
        a = max(0, -d)
        b = min(n, n - d)
        if a < b:
            grad[a:b] += w * (1.0 - h[a+d:b+d])
        
        c = max(0, d)
        e = min(n, n + d)
        if c < e:
            grad[c:e] -= w * h[c-d:e-d]
    return grad
```

Now let me think about the coordinate descent more carefully. The key insight is that for each coordinate j, the problem is:
min_{x ∈ [0,1]} max_m (c[m]_0 + x * g[m])

where g[m] = ∂c[m]/∂h[j] (computed as before) and c[m]_0 = c[m] - h[j] * g[m] (the value if h[j]=0).

The optimal x can be found by:
1. If max_m c[m]_0 is achieved at an index where g < 0, then increasing x might help
2. If max_m (c[m]_0 + g[m]) is achieved at an index where g > 0, then decreasing x might help
3. The optimal x is where the upper envelope is minimized

A practical approach:
- Start with x = h[j]
- Compute the current max: c_max = max(c[m])
- Find which m achieves the max (or is close to it)
- If g[m_max] > 0: try decreasing x (set x to a value that reduces c[m_max] without increasing others too much)
- If g[m_max] < 0: try increasing x

For the 1D minimax, a good heuristic is:
- Find the two "most constraining" lines: one going up (g > 0, highest at current x) and one going down (g < 0, highest at current x)
- Compute their intersection: x* = (c[m_down] - c[m_up]) / (g[m_up] - g[m_down])
- Clip x* to [0, 1]
- Check if this actually reduces the max

This is O(n) per coordinate (to find the relevant m's), so a full sweep is O(n²) ≈ 8M for n=2048. Each sweep takes about 10-20ms. We can do hundreds of sweeps in the time budget.

Let me also think about the sum constraint in coordinate descent. If I change h[j] from old to new, the sum changes by (new - old). To maintain the sum, I need to adjust other coordinates.

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Force exact
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Find a coordinate to adjust
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def logsumexp_objective(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # in units of the final objective
    max_c = np.max(corr)
    lse = max_c + np.log(np.sum(np.exp(T * (corr - max_c)))) / T
    penalty = mu * (np.sum(h) - n / 2.0) ** 2
    return lse + penalty

def logsumexp_gradient(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = np.max(corr)
    
    # Weights for log-sum-exp
    shifted = T * (corr - max_c)
    # For numerical stability
    shifted_max = np.max(shifted)
    exp_shifted = np.exp(shifted - shifted_max)
    weights = exp_shifted / np.sum(exp_shifted)
    # weights sums to 1, and grad of LSE = sum weights[m] * d corr_dx[m] / d h[j]
    # where corr_dx = corr * dx, so d corr_dx / d h = dx * d corr / d h
    
    # Compute d corr[m] / d h[j] for all m, j
    # As derived: for each lag d, the gradient contribution is:
    # grad[j] += weights[m] * (1-h)[j+d] - weights[m] * h[j-d] (with bounds)
    
    grad = np.zeros(n)
    # m index corresponds to lag = m - (n-1)
    for d in range(-(n-1), n):
        w = weights[d + n - 1]
        if abs(w) < 1e-18:
            continue
        a = max(0, -d)
        b = min(n, n - d)
        if a < b:
            grad[a:b] += w * (1.0 - h[a+d:b+d])
        c = max(0, d)
        e = min(n, n + d)
        if c < e:
            grad[c:e] -= w * h[c-d:e-d]
    
    grad *= dx  # because we're using corr*dx as the objective
    
    # Penalty gradient
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    
    return grad

def lbfgsb_optimize(h, n, T, mu=1000.0, maxiter=300):

                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
    return make_feasible(res.x, n)
```

But wait, I can do better. When I change h[j] from old to new, the correlation changes as:

where g[m] is the gradient I computed. So I only need to:

For coordinate j, the current max is c_max at index m*. 
- If g[m*] > 0 (increasing h[j] increases c[m*]): we should try to decrease h[j]
- If g[m*] < 0: try to increase h[j]

```python
def coordinate_descent_simple(h, n, max_sweeps=100, time_limit=60):
    start = time.time()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    best_c5 = np.max(corr)
    mu = 100.0
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        any_improved = False
        
        for j in range(n):
            # Compute g[m] = d corr[m] / d h[j] (in dx units)
            # g[m] = dx * [(1-h[j+lag_m])*I - h[j-lag_m]*I]
            
            # For efficiency, compute only the relevant part
            # The key question: what's the best new value for h[j]?
            
            # Current max: c_max = max(corr)
            c_max = np.max(corr)
            m_max = np.argmax(corr)
            d_max = m_max - (n - 1)  # lag of the max
            
            # g at m_max:
            g_mmax = 0.0
            if 0 <= j + d_max < n:
                g_mmax += (1.0 - h[j + d_max])
            if 0 <= j - d_max < n:
                g_mmax -= h[j - d_max]
            
            if abs(g_mmax) < 1e-15:
                continue
            
            # Try to reduce c[m_max]
            # c[m_max] = corr[m_max] + (x - h[j]) * g_mmax * dx
            # We want corr[m_max] + delta * g_mmax * dx to be as low as possible
            # But also check other correlations
            
            if g_mmax > 0:
                # Decrease h[j] to reduce c[m_max]
                # Max decrease: can go to 0
                # c[m_max] reduces by h[j] * g_mmax * dx
                target_reduce = h[j]  # decrease to 0
            else:
                # Increase h[j] to reduce c[m_max]
                target_reduce = 1.0 - h[j]  # increase to 1
            
            # Try the full move
            old_val = h[j]
            if g_mmax > 0:
                new_val = 0.0
            else:
                new_val = 1.0
            
            if abs(new_val - old_val) < 1e-15:
                continue
            
            # Compute new corr
            delta = new_val - old_val
            corr_new = corr.copy()  # O(n) - expensive!
            
            # Update corr: corr[m] += delta * g[m] * dx for all m
            # g[m] for this j:
            # g[m] = (1-h[j+lag_m])*I - h[j-lag_m]*I where h is the OLD h
            # But wait, after the update, h[j] changes, so g for future coords is different
            # For this update, we use the old h to compute g
            
            for d in range(-(n-1), n):
                m = d + n - 1
                g_m = 0.0
                if 0 <= j + d < n:
                    g_m += (1.0 - h[j + d])
                if 0 <= j - d < n:
                    g_m -= h[j - d]
                if g_m != 0:
                    corr_new[m] += delta * g_m * dx
            
            new_max = np.max(corr_new)
            
            if new_max < c_max - 1e-14:
                h[j] = new_val
                corr = corr_new
                any_improved = True
        
        if not any_improved:
            break
```

```python
def make_feasible(h, n):

def lse_obj(h, T, mu):

    mc = np.max(corr)

    return lse + mu * (np.sum(h) - n / 2.0) ** 2

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)

    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return grad

def lbfgsb_opt(h, n, T, mu, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(lse_obj, h.copy(), method='L-BFGS-B',
                   jac=lse_grad, args=(T, mu),
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-7})
    return make_feasible(res.x, n)

def coord_descent(h, n, max_sweeps=80, time_limit=60.0):
    start_t = time.time()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    n_corr = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() - start_t > time_limit:
            break
        improved = False
        for j in range(n):
            c_max = np.max(corr)
            m_max = np.argmax(corr)
            d_max = m_max - (n - 1)
            g_m = 0.0
            if 0 <= j - d_max < n:
                g_m += (1.0 - h[j - d_max])
            if 0 <= j + d_max < n:
                g_m -= h[j + d_max]
            if abs(g_m) < 1e-15:
                continue
            if g_m > 0:
                new_val = 0.0
            else:
                new_val = 1.0
            if abs(new_val - h[j]) < 1e-15:
                continue
            delta = new_val - h[j]
            corr_new = corr + delta * np.zeros(n_corr)
            for d in range(-(n-1), n):
                m = d + n - 1
                g_m2 = 0.0
                if 0 <= j - d < n:
                    g_m2 += (1.0 - h[j - d])
                if 0 <= j + d < n:
                    g_m2 -= h[j + d]
                if g_m2 != 0.0:
                    corr_new[m] = corr[m] + delta * g_m2 * dx
            if np.max(corr_new) < c_max - 1e-14:
                h[j] = new_val
                corr = corr_new
                improved = True
        if not improved:
            break
    return make_feasible(h, n)

def binary_search_refine(h, n, lo, hi, max_iter=30, time_limit=45.0):
    start_t = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        if hi - lo < 1e-7:
            break
        mid = (lo + hi) / 2.0
        mu = 5000.0
        T = 100.0
        try:
            h_try = lbfgsb_opt(h, n, T, mu, 80)
            c5_try = compute_c5(h_try)
            if c5_try <= mid + 1e-9:
                hi = c5_try
                best_h = h_try.copy()
                best_c5 = c5_try
            else:
                lo = mid
                h = h_try.copy() if compute_c5(h_try) < best_c5 else h
                if compute_c5(h_try) < best_c5:
                    best_c5 = compute_c5(h_try)
                    best_h = h_try.copy()
        except Exception:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    def time_left():
        return budget_s - (time.time() - start_time)

    # Phase 1: L-BFGS-B at increasing resolutions
    resolutions = [1024, 2048, 3072, 4096]
    for n_pts in resolutions:
        if time_left() < 120:
            break
        h_curr = upsample_h(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        T_list = [40, 100, 300]
        mu_list = [500, 2000, 8000]
        maxit_list = [200, 150, 100]

        for i, (T, mu, mit) in enumerate(zip(T_list, mu_list, maxit_list)):
            if time_left() < 40:
                break
            try:
                h_opt = lbfgsb_opt(h_curr, n_pts, T, mu, mit)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_curr = h_opt.copy()
            except Exception:
                pass

        # Coordinate descent polish
        if time_left() > 60:
            h_cd = coord_descent(h_curr.copy(), n_pts, max_sweeps=60,
                                 time_limit=min(60.0, time_left() - 30))
            c5_cd = compute_c5(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
                h_curr = h_cd.copy()

        # Another L-BFGS pass with higher T from polished solution
        if time_left() > 50:
            try:
                h_opt = lbfgsb_opt(h_curr, n_pts, 500, 5000, 100)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 2: Random restarts at best resolution
    n_best = len(best_h)
    if time_left() > 80:
        for restart in range(5):
            if time_left() < 30:
                break
            noise = rng.normal(0, 0.15, n_best)
            h_r = np.clip(best_h + noise, 0.0, 1.0)
            h_r = make_feasible(h_r, n_best)
            try:
                h_opt = lbfgsb_opt(h_r, n_best, 80, 3000, 150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 3: Final coordinate descent polish
    if time_left() > 20:
        h_final = coord_descent(best_h.copy(), n_best, max_sweeps=100,
                                time_limit=min(30.0, time_left() - 5))
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
