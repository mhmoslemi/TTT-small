    def con_fun(h):

    if rem() > 20:
        h_cd = make_feasible(best_h.copy(), len(best_h))
        cd_time = min(rem() - 10, 80)
        h_cd, c5_cd = coord_descend(h_cd, len(h_cd), max_rounds=10, max_time=cd_time)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 7: SLSQP final polish
    if rem() > 15:
        n_s = min(len(best_h), 1024)
        h_s = upsample_h(best_h, n_s)
        h_s = slsqp_run(h_s, n_s, T=500, max_iter=100)
        cs = compute_c5(h_s)
        if cs < best_c5:
            best_c5, best_h = cs, h_s.copy()

    # Final verification
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
