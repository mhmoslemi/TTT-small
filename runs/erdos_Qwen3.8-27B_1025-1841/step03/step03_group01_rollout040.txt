from scipy.optimize import minimize
import time

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_c5(h):
    return float(np.max(compute_lags(h)))

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def compute_gradient_sum_top_k(h, k_indices):
    """Compute gradient of sum of lags at given indices k_indices.
    k_indices are offsets from -(n-1) to (n-1).
    In the full correlation array of length 2n-1, index m corresponds to lag m-(n-1).
    """
    n = len(h)
    grad = np.zeros(n)
    for k in k_indices:
        # g_k = sum_i h[i]*(1-h[i+k]) for valid i
        # dg_k/dh[j] = (1-h[j+k])*[0<=j+k<n] - h[j-k]*[0<=j-k<n]
        # First term: for j where 0 <= j+k < n, i.e., -k <= j < n-k
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            grad[lo1:hi1] += (1.0 - h[lo1+k:hi1+k])
        # Second term: for j where 0 <= j-k < n, i.e., k <= j < n+k
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            grad[lo2:hi2] -= h[lo2-k:hi2-k]
    return grad

def obj_sum_top_k(h, k_indices):
    """Sum of lag values at given k indices (offsets from -(n-1) to (n-1))."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # k_indices are actual lag offsets; convert to correlation array indices
    # corr[m] corresponds to lag m - (n-1)
    idxs = [k + n - 1 for k in k_indices]
    return float(np.sum(corr[idxs]) * dx)
