from scipy.fft import fft, ifft

def project_to_feasible(g, n):
    # Project to {0 ≤ h ≤ 1, sum(h) = n/2} using bisection on threshold
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def fft_correlate(h):
    # Compute full linear cross-correlation of h with (1-h) using FFT
    n = len(h)
    N = 2 * n  # pad to 2n for linear correlation
    H = fft(h, N)
    S = fft(1.0 - h, N)
    corr = ifft(conj(H) * S).real
    # The 'full' correlation in numpy has lags -(n-1) to (n-1)
    # Our FFT result: corr[k] for k=0..N-1 corresponds to circular correlation
    # We need to reorganize: lags 0 to n-1 are in corr[0:n], lags -(n-1) to -1 are in corr[N-(n-1):N]
    full = np.concatenate([corr[N-(n-1):], corr[0:n]])
    return full  # length 2n-1, matching np.correlate(h, 1-h, 'full')

import numpy as np
import time
from scipy.fft import fft, ifft

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    # Simple linear interpolation (not periodic, just linear)
    return np.interp(x_new, x_old, np.concatenate([h, [h[0]])))[:target_n]

def fft_full_correlate(h, g):
    """Compute full cross-correlation of h and g (same length) using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = fft(h, N)
    G = fft(g, N)
    # Cross-correlation: corr[m] = sum_i h[i+m] * g[i]
    # Using FFT: ifft(conj(FFT(h)) * FFT(g)) gives circular cross-correlation
    # But we need to be careful with the indexing
    r = ifft(conj(H) * G).real
    # r[0] = sum h[i]*g[i], r[1] = sum h[i+1]*g[i] (positive lag)
    # r[N-1] = sum h[i-1]*g[i] (negative lag)
    # Full correlation has lags -(n-1) to (n-1), total 2n-1 values
    # r[N-(n-1):] gives lags -(n-1) to -1
    # r[0:n] gives lags 0 to n-1
    full = np.concatenate([r[N-(n-1):], r[0:n]])
    return full

def compute_c5(h):
    n = len(h)
    corr = fft_full_correlate(h, 1.0 - h)
    return corr.max() * (2.0 / n)

def compute_c5_direct(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

import numpy as np
import time
from scipy.fft import fft, ifft

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def fft_correlate_full(h):
    n = len(h)
    N = 1
    while N < 2 * n:
        N *= 2
    H = fft(h, N)
    S = fft(1.0 - h, N)
    r = ifft(conj(H) * S).real
    # Reorganize to match np.correlate(h, 1-h, 'full')
    # full[k] for k=0..2n-2, lag = k-(n-1)
    # r[n-1::-1] gives lags -(n-1)..0
    # r[N-n+1:N][::-1] gives lags 1..(n-1)
    return np.concatenate([r[n-1::-1], r[N-n+1:N][::-1]])

def compute_c5(h):
    n = len(h)
    corr = fft_correlate_full(h)
    return corr.max() * (2.0 / n)

def nesterov_optimize_fft(h, n, tau, time_limit, lr=None, top_k=20):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.5 / (tau * dx)
    h_prev = h.copy()
    t_mom = 1.0
    stall = 0
    two_n_minus_1 = 2 * n - 1
    n_active = min(top_k, two_n_minus_1)
    
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        mu = min((t_mom - 1.0) / (t_mom + 2.0), 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        corr_raw = fft_correlate_full(y)
        
        if n_active < two_n_minus_1:
            top_idx = np.argpartition(corr_raw, -n_active)[-n_active:]
        else:
            top_idx = np.arange(two_n_minus_1)
        top_vals = corr_raw[top_idx]
        tv_shifted = top_vals - top_vals.max()
        exp_v = np.exp(tau * dx * tv_shifted)
        p = exp_v / exp_v.sum()
        
        grad = np.zeros(n)
        for i_idx in range(n_active):
            w = p[i_idx] * dx
            k = top_idx[i_idx]
            sh = k - (n - 1)
            # h[j] as first factor: j+sh in [0,n-1]
            j0 = max(0, -sh)
            j1 = min(n, n - sh)
            if j0 < j1:
                grad[j0:j1] += w * (1.0 - y[j0 + sh:j1 + sh])
            # h[j] as second factor: j-sh in [0,n-1]
            j0b = max(0, sh)
            j1b = min(n, n + sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w * y[j0b - sh:j1b - sh]
        
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        if it % 50 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:

                stall = 0

    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def coordinate_descent(h, n, time_limit, n_passes=10):
    """Exact coordinate descent: optimize each h[j] by 1D minimization."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    two_n_minus_1 = 2 * n - 1
    rng = np.random.default_rng(12345)
    
    for p in range(n_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        corr = fft_correlate_full(h)
        improved = False
        
        for j in order:
            if time.time() - start > time_limit:
                break
            old_val = h[j]
            
            # Compute d_k for this j
            # d_k = [1-h[j+s]]*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n) where s = k-(n-1)
            s_range = np.arange(-(n-1), n)  # s values
            j_plus = j + s_range
            j_minus = j - s_range
            valid_p = (j_plus >= 0) & (j_plus < n)
            valid_m = (j_minus >= 0) & (j_minus < n)
            d = np.zeros(two_n_minus_1)
            d[valid_p] = 1.0 - h[j_plus[valid_p]]
            d[valid_m] -= h[j_minus[valid_m]]
            # Note: where both valid, d = 1 - h[j+s] - h[j-s]
            # where only valid_p: d = 1 - h[j+s]
            # where only valid_m: d = -h[j-s]
            
            # c_k = corr[k] - old_val * d[k]
            c = corr - old_val * d
            
            # Find t in [0,1] minimizing max_k(c[k] + t * d[k])
            # Ternary search (convex function)
            t_lo, t_hi = 0.0, 1.0
            for _ in range(30):
                t1 = t_lo + (t_hi - t_lo) / 3.0
                t2 = t_hi - (t_hi - t_lo) / 3.0
                f1 = (c + t1 * d).max()
                f2 = (c + t2 * d).max()
                if f1 < f2:
                    t_hi = t2
                else:
                    t_lo = t1
            t_new = (t_lo + t_hi) / 2.0
            
            # Also check boundaries
            f0 = c.max()
            f1v = (c + d).max()
            ft = (c + t_new * d).max()
            if f0 <= ft and f0 <= f1v:
                t_new = 0.0
            elif f1v <= ft and f1v <= f0:
                t_new = 1.0
            
            delta = t_new - old_val
            if abs(delta) > 1e-12:
                corr += delta * d
                h[j] = t_new
                improved = True

        h = project_to_feasible(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5
