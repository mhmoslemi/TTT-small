```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj_and_grad(h, T, penalty=0.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_terms = np.exp(T * shifted)
    sum_exp = np.sum(exp_terms)
    obj_val = (mc + np.log(sum_exp) / T) * dx
    
    if penalty > 0:
        violation = np.sum(h) - N / 2.0
        obj_val += penalty * violation ** 2
    
    w = T * exp_terms / sum_exp
    g = 1.0 - 2.0 * h
    grad = (dx / T) * np.correlate(w, g, mode='valid')
    
    if penalty > 0:
        grad += 2.0 * penalty * violation * np.ones(N)
    
    return obj_val, grad
```

```python
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

    obj = (mc + np.log(np.sum(exp_terms)) / T) * dx
    if penalty > 0:
        v = np.sum(h) - N / 2.0
        obj += penalty * v * v
    return obj

def smooth_grad(h, T, penalty):

    dx = 2.0 / N

        grad += 2.0 * penalty * v
```
