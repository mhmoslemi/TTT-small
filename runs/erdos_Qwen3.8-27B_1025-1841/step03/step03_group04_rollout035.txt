    return _T_cur, n

def _make_obj(T, penalty=1e6):
    def obj_fn(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = np.max(corr)
        shifted = T * (corr - M)
        z_max = np.max(shifted)
        log_sum = z_max + np.log(np.sum(np.exp(shifted - z_max)))
        return (M + log_sum / T) * dx + penalty * (np.sum(h) - n / 2.0) ** 2
    return obj_fn

def _make_grad(T, penalty=1e6):
    def grad_fn(h):
        n = len(h)
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = np.max(corr)
        shifted = T * (corr - M)
        z_max = np.max(shifted)
        exp_z = np.exp(shifted - z_max)
        w = exp_z / np.sum(exp_z)
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            wk = w[k]
            if wk < 1e-12:
                continue
            l = k - n + 1
            if l == 0:
                grad += wk * (1.0 - 2.0 * h)
            elif l > 0:
                grad[l:] += wk * (1.0 - h[:n-l])
                grad[:n-l] -= wk * h[l:]
            else:
                m = -l
                grad[:n-m] += wk * (1.0 - h[m:])
                grad[m:] -= wk * h[:n-m]
        grad *= dx
        grad += 2.0 * penalty * (np.sum(h) - n / 2.0)
        return grad
    return grad_fn

def lbfgsb_optimize(h, T, maxiter=400, maxfun=4000):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    obj_fn = _make_obj(T)
    grad_fn = _make_grad(T)
    res = minimize(obj_fn, h, method='L-BFGS-B', jac=grad_fn,
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-12, 'maxfun': maxfun})
    return make_feasible(res.x, n)

def coord_descent(h, n, passes=2, steps=None):
    if steps is None:
        steps = [0.001, 0.005, 0.01, 0.02, 0.05]
    for p in range(passes):
        for j in range(n):
            old_val = h[j]
            best_val = old_val
            best_c = compute_c5(h)
            for s in steps:
                for sign in [-1, 1]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                h[j] = best_val
                h = make_feasible(h, n)
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Phase 1: L-BFGS-B at initial resolution with increasing T
    for T in [300.0, 800.0, 1500.0]:
        if remaining() < 20:
            break
        try:
            h_res = lbfgsb_optimize(best_h, T, maxiter=400, maxfun=4000)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
        except Exception:
            pass
    
    # Phase 2: Upsample to 2x resolution and optimize
    n_up = n_init * 2
    if remaining() > 200 and n_up <= 3840:
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up <= best_c5 + 0.005:
            for T in [300.0, 800.0, 2000.0]:
                if remaining() < 30:
                    break
                try:
                    h_res = lbfgsb_optimize(h_up, T, maxiter=350, maxfun=3500)
                    c5_res = compute_c5(h_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = h_res.copy()
                        h_up = h_res.copy()
                except Exception:
                    pass
    
    # Phase 3: Coordinate descent on best
    if remaining() > 30:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n_pts, passes=2,
                                         steps=[0.002, 0.005, 0.01, 0.02, 0.05])
    
    # Phase 4: Random perturbations + reoptimize
    for restart in range(5):
        if remaining() < 40:
            break
        n_pts = len(best_h)
        scale = 0.02 + restart * 0.01
        h_try = make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
        for T in [500.0, 1500.0]:
            if remaining() < 20:
                break
            try:
                h_res = lbfgsb_optimize(h_try, T, maxiter=300, maxfun=3000)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
                    h_try = h_res.copy()
            except Exception:
                pass
    
    # Phase 5: Higher T polish at best resolution
    if remaining() > 30:
        n_pts = len(best_h)
        for T in [3000.0, 5000.0]:
            if remaining() < 15:
                break
            try:
                h_res = lbfgsb_optimize(best_h, T, maxiter=300, maxfun=3000)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
            except Exception:
                pass
    
    # Phase 6: Fine coordinate descent
    if remaining() > 20:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n_pts, passes=1,
                                         steps=[0.0005, 0.001, 0.002, 0.005])
    
    # Phase 7: Binary snap heuristic
    if remaining() > 20:
        n_pts = len(best_h)
        h_bin = (best_h > 0.5).astype(float)
        # Fix count
        target = n_pts // 2
        ones_count = int(np.sum(h_bin))
        if ones_count > target:
            one_idx = np.where(h_bin == 1)[0]
            order = np.argsort(best_h[one_idx])
            for i in range(ones_count - target):
                h_bin[one_idx[order[i]]] = 0.0
        elif ones_count < target:
            zero_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zero_idx])
            for i in range(target - ones_count):
                h_bin[zero_idx[order[i]]] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Use binary as starting point for continuous opt
        if remaining() > 20:
            try:
                h_res = lbfgsb_optimize(h_bin, 800.0, maxiter=300, maxfun=3000)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
            except Exception:
                pass
    
    # Phase 8: Final ultra-fine coordinate descent
    if remaining() > 10:
        n_pts = len(best_h)
        for j in range(n_pts):
            if remaining() < 3:
                break
            old_val = best_h[j]
            best_val = old_val
            best_c = best_c5
            for val in [0.0, 1.0]:
                if abs(old_val - val) < 1e-8:
                    continue
                trial = best_h.copy()
                trial[j] = val
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = val
            for s in [0.0002, 0.0005, 0.001]:
                for sign in [-1, 1]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
