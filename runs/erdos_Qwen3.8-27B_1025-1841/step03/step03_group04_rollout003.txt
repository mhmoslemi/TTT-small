import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    mask = (h < 1.0) if diff > 0 else (h > 0.0)
    cnt = float(np.sum(mask))
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

def top_shift_gradient(h, top_idx, n):
    """Compute gradient contribution from top shifts only."""
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        for j in range(n):
            val = -h[j - s] if 0 <= j - s < n else 0.0
            if 0 <= j + s < n:
                val += (1.0 - h[j + s])
            else:
                val += 0.0
            grad[j] += val
    return grad / len(top_idx)

def pgd_optimize(h, n, budget_s, start_time, top_m=5, T_schedule=None):
    """Projected gradient descent with adaptive step and Nesterov momentum."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    step_size = 0.1
    momentum = 0.9
    it = 0
    no_improve = 0
    max_no_improve = 30
    
    while time.time() - start_time < budget_s:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = np.max(corr) * dx
        
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step_size *= 0.5
                if step_size < 1e-6:
                    break
                no_improve = 0
        
        # Find top shifts
        top_idx = np.argsort(corr)[-top_m:]
        
        # Compute gradient from top shifts
        grad = top_shift_gradient(h, top_idx, n)
        
        # Nesterov step
        y = h - momentum * velocity
        corr_y = np.correlate(np.clip(y, 0, 1), 1.0 - np.clip(y, 0, 1), mode='full')
        top_idx_y = np.argsort(corr_y)[-top_m:]
        grad_y = top_shift_gradient(y, top_idx_y, n)
        velocity = momentum * velocity - step_size * grad_y

                improved = True

def slsqp_polish(h, n, budget_s, start_time, T=1000.0, top_m=None):
    """SLSQP optimization with smooth max."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None and top_m < len(corr):
            top_idx = np.argsort(corr)[-top_m:]
            mc = np.max(corr[top_idx])
            shifted = corr[top_idx] - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 400, 'ftol': 1e-16})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h):
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h, compute_c5(h)
