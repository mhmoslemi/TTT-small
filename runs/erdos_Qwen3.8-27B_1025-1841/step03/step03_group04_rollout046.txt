    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def grad_softmax(h, T, n_top=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    ev = np.exp(T * shifted)
    w = ev / np.sum(ev)
    top_idx = np.argsort(-w)[:n_top]
    grad = np.zeros(n)
    for k in top_idx:
        pk = w[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        j1 = np.arange(max(0, s), min(n, n + s))
        if len(j1) > 0:
            grad[j1] += pk * (1.0 - h[j1 - s])
        j2 = np.arange(max(0, -s), min(n, n - s))
        if len(j2) > 0:
            grad[j2] -= pk * h[j2 + s]
    grad *= dx
    return grad

def obj_softmax(x, T, n):
    dx = 2.0 / n
    cv = np.correlate(x, 1.0 - x, mode='full')
    mcv = np.max(cv)
    return (mcv + np.log(np.sum(np.exp(T * (cv - mcv)))) / T) * dx

def con_eq(x, n):
    return np.sum(x) - n / 2.0

def con_eq_jac(x, n):
    return np.ones(n)
