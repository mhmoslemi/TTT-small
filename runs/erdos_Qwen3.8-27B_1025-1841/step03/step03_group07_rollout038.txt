import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr

def grad_from_shifts(h, n, shifts, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(shifts)):
        s = shifts[idx]
        w = weights[idx] * dx
        if s >= 0:
            if s == 0:
                grad[:] += w * (1.0 - 2.0 * h)
            else:
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] -= w * h[s:]
        else:
            t = -s
            grad[:n-t] += w * (1.0 - h[t:])
            grad[t:] -= w * h[:n-t]
    return grad

def compute_corr_at_shift(h, s):
    n = len(h)
    if s >= 0:
        return np.dot(h[s:], 1.0 - h[:n-s])
    else:
        return np.dot(h[:n+s], 1.0 - h[-s:])

def nesterov_efficient(h, n, tau, time_limit, lr=None, top_k=15, full_every=100, momentum_clip=0.9):
    """Nesterov with efficient gradient (only compute active shifts)."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    
    # Initial full correlation
    _, corr_full = compute_c5(h)
    max_c = corr_full.max()
    best_c5 = max_c * dx
    best_h = h.copy()
    
    # Get initial active set
    full_idx = np.argsort(corr_full)[-top_k:]
    # Convert to shifts: shift = idx - (n-1)
    active_shifts = (full_idx - (n - 1)).astype(int)
    
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0
    
    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        
        # Periodically update active set
        if it % full_every == 0:
            _, corr_full = compute_c5(h)
            max_c = corr_full.max()
            c5_now = max_c * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += full_every
            if stall > 2000:
                lr *= 0.7
                h_prev = h.copy()
                stall = 0
                if lr < 1e-16:
                    break
            full_idx = np.argsort(corr_full)[-top_k:]
            active_shifts = (full_idx - (n - 1)).astype(int)
        
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, momentum_clip)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute objective and gradient using active shifts only
        corr_vals = np.array([compute_corr_at_shift(y, s) * dx for s in active_shifts])
        max_cv = corr_vals.max()
        shifted = tau * (corr_vals - max_cv)
        shifted = np.clip(shifted, -500.0, 0.0)
        exp_c = np.exp(shifted)
        s_sum = exp_c.sum()
        p = exp_c / s_sum
        obj = max_cv + np.log(s_sum) / tau
        
        grad = grad_from_shifts(y, n, active_shifts, p)
        
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    
    # Final evaluation
    c5_final, _ = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(35):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr


def compute_corr_at_shift(h, s):
    n = len(h)
    if s >= 0:
        if s == 0:
            return np.dot(h, 1.0 - h)
        return np.dot(h[s:], 1.0 - h[:n - s])
    else:
        return np.dot(h[:n + s], 1.0 - h[-s:])


def grad_from_shifts(h, n, shifts, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx in range(len(shifts)):
        s = shifts[idx]
        w = weights[idx] * dx
        if s >= 0:
            if s == 0:
                grad[:] += w * (1.0 - 2.0 * h)
            else:
                grad[s:] += w * (1.0 - h[:n - s])
                grad[:n - s] -= w * h[s:]
        else:
            t = -s
            grad[:n - t] += w * (1.0 - h[t:])
            grad[t:] -= w * h[:n - t]
    return grad


def nesterov_efficient(h, n, tau, time_limit, lr=None, top_k=15, full_every=80, momentum_clip=0.9):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    _, corr_full = compute_c5(h)
    max_c = corr_full.max()
    best_c5 = max_c * dx
    best_h = h.copy()
    full_idx = np.argsort(corr_full)[-top_k:]
    active_shifts = (full_idx - (n - 1)).astype(int)
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0
    for it in range(100000000):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        if it % full_every == 0:
            _, corr_full = compute_c5(h)
            max_c = corr_full.max()
            c5_now = max_c * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 40:
                lr *= 0.7
                h_prev = h.copy()
                stall = 0
                if lr < 1e-16:
                    break
            full_idx = np.argsort(corr_full)[-top_k:]
            active_shifts = (full_idx - (n - 1)).astype(int)
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, momentum_clip)
        y = h + mu * (h - h_prev)
        np.clip(y, 0.0, 1.0, out=y)
        corr_vals = np.empty(len(active_shifts))
        for ii in range(len(active_shifts)):
            corr_vals[ii] = compute_corr_at_shift(y, active_shifts[ii]) * dx
        max_cv = corr_vals.max()
        shifted = tau * (corr_vals - max_cv)
        np.clip(shifted, -500.0, 0.0, out=shifted)
        exp_c = np.exp(shifted)
        s_sum = exp_c.sum()
        p = exp_c / s_sum
        grad = grad_from_shifts(y, n, active_shifts, p)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    c5_final, _ = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5
