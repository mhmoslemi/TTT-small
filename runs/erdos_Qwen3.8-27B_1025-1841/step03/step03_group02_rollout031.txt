def swap_update_vec(corr, h, p, q):
    """Vectorized swap update."""
    N = len(h)
    s = np.arange(-(N-1), N)  # shape (2N-1,)
    # p+s, q+s, p-s, q-s (with bounds)
    ps = p + s  # p-s_arr
    qs = q + s
    pms = p - s
    qms = q - s
    # Gather with bounds (out of bounds = 0)
