def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, target_sum=None):

    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
        # Fine adjustment
        diff2 = target_sum - np.sum(h)
        if abs(diff2) > 1e-14:
            if diff2 > 0:
                mask2 = h < 1.0
            else:
                mask2 = h > 0.0
            idx = np.where(mask2)[0]
            if len(idx) > 0:
                h[idx[0]] += diff2
                h = np.clip(h, 0.0, 1.0)
    return h

def correlate_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def corr_gradient(corr_vec, h, t):
    """Gradient of smooth max: (1/t)*log(sum exp(t*corr*dx))"""
    n = len(h)
    dx = 2.0 / n
    tc = t * corr_vec * dx
    tc -= tc.max()
    p = np.exp(tc)
    p /= p.sum()
    
    n_half = n - 1
    grad = np.zeros(n)
    
    # For each correlation index k with significant weight:
    for k in range(len(corr_vec)):
        pk = p[k]
        if pk < 1e-14:
            continue
        L = k - n_half  # lag
        if L >= 0:
            # c = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # dc/dh[i] for i in [0, n-1-L]: += (1-h[i+L])
            # dc/dh[i+L] for i in [0, n-1-L]: -= h[i]
            idx1 = np.arange(n - L)  # 0 to n-1-L
            idx2 = idx1 + L  # L to n-1
            grad[idx1] += pk * (1.0 - h[idx2])
            grad[idx2] -= pk * h[idx1]
        else:
            M = -L  # M > 0
            # c(-M) = sum_{i=M}^{n-1} h[i]*(1-h[i-M])
            # dc/dh[i] for i in [M, n-1]: += (1-h[i-M])
            # dc/dh[i-M] for i in [M, n-1]: -= h[i]
            idx2 = np.arange(M, n)  # M to n-1
            idx1 = idx2 - M  # 0 to n-1-M
            grad[idx2] += pk * (1.0 - h[idx1])
            grad[idx1] -= pk * h[idx2]
    
    return dx * grad

def project_feasible(h):
    """Project onto {0 <= h <= 1, sum(h) = n/2}"""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:

        cnt = np.sum(mask)
        if cnt > 0:

def smooth_obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_full(h)
    tc = t * corr * dx
    tc -= tc.max()
    e = np.exp(tc)
    val = (tc.max() + np.log(e.sum()) / t) * dx
    grad = corr_gradient(corr, h, t)
    return val, grad

def projected_grad_descent(h, t, n_steps, step_decay=0.99, start_lr=0.01):
    """Projected gradient descent with Polyak-type step size"""
    n = len(h)
    target = n / 2.0
    lr = start_lr
    best_h = h.copy()
    best_c = compute_c5(h)
    
    for step in range(n_steps):
        val, grad = smooth_obj_and_grad(h, t)
        # Gradient step
        h_new = h - lr * grad
        h_new = project_feasible(h_new)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        h = h_new
        lr *= step_decay
        if lr < 1e-10:
            break
    
    return best_h, best_c

def coord_descent_1d(h, max_rounds=50, time_limit=None, start_time=None):
    """Coordinate descent with golden section search on each coordinate"""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    best_c = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 20:
                    return h, best_c
            old_val = h[j]
            # Try a few values around old_val
            candidates = np.array([max(0, old_val - 0.15), max(0, old_val - 0.05),
                                   old_val - 0.02, old_val + 0.02,
                                   min(1, old_val + 0.05), min(1, old_val + 0.15),
                                   0.0, 1.0, 0.5])
            best_val = old_val
            best_obj = best_c
            for cv in candidates:
                cv = np.clip(cv, 0.0, 1.0)
                if abs(cv - old_val) < 1e-6:
                    continue
                trial = h.copy()
                trial[j] = cv
                # Fix constraint
                diff = target - np.sum(trial)
                if abs(diff) > 1e-10:
                    # Distribute to other non-bound coordinates
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    if diff > 0:
                        mask &= trial < 1.0
                    else:
                        mask &= trial > 0.0
                    cnt = np.sum(mask)
                    if cnt > 0:
                        trial[mask] += diff / cnt
                        trial = np.clip(trial, 0.0, 1.0)
                    # Adjust j
                    diff2 = target - np.sum(trial)
                    trial[j] = max(0, min(1, trial[j] + diff2))
                    trial = np.clip(trial, 0.0, 1.0)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old_val:
                h = h.copy()
                h[j] = best_val
                h = project_feasible(h)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def pairwise_local_search(h, n_pairs=200, time_limit=None, start_time=None):
    """Try pairwise changes to escape local minima"""
    n = len(h)
    target = n / 2.0
    best_c = compute_c5(h)
    rng = np.random.RandomState(42)
    
    for it in range(n_pairs):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 30:
                break
        i, j = rng.choice(n, size=2, replace=False)
        # Try swapping or adjusting pair
        for di, dj in [(0.1, -0.1), (-0.1, 0.1), (0.05, -0.05), (-0.05, 0.05),
                       (0.2, -0.2), (1.0 - h[j], h[i])]:  # last: swap values
            trial = h.copy()
            trial[i] = h[i] + di
            trial[j] = h[j] + dj
            trial = np.clip(trial, 0.0, 1.0)
            trial = project_feasible(trial)
            c = compute_c5(trial)
            if c < best_c - 1e-10:
                h = trial
                best_c = c
                break
    
    return h, best_c

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new)

For efficiency, instead of looping over all k values, I can vectorize:

For each j, the gradient is:
grad[j] = dx · Σ_k p(k) · [(1-h[j+L_k])·[0≤j+L_k<n] - h[j-L_k]·[0≤j-L_k<n]]

This is equivalent to:
grad[j] = dx · [Σ_k p(k)·(1-h[j+L_k])·[valid] - Σ_k p(k)·h[j-L_k]·[valid]]

The first sum: for each j, sum over k where j+L_k is valid.
The second sum: for each j, sum over k where j-L_k is valid.

I can compute this as a weighted sum of shifted h vectors. Specifically:
First term for grad: Σ_k p(k) * (shifted (1-h) by L_k, evaluated at j, with zero padding)
Second term for grad: Σ_k p(k) * (shifted h by -L_k, evaluated at j, with zero padding)

This can be done with numpy by:
- Creating a 2D array where row k is (1-h) shifted by L_k (with zeros outside [0,n-1])
- Computing weighted sum with weights p(k)

But this is O(n²) memory and computation, which for n=640 is ~400K, totally fine.

Actually, an even simpler vectorized approach:
For each lag L (which ranges from -(n-1) to n-1):
- The contribution to grad from lag L with weight w = p(k) where L = k-(n-1):
  - For i = 0 to n-1-|L|: grad[i] += w*(1-h[i+|L|]) if L≥0, or grad[i+|L|] += w*(1-h[i]) if L<0
  
This is O(n) per lag and O(n²) total. Let me just loop over the significant lags.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    if target is None:
        target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-14:
            if diff2 > 0:
                mask2 = h < 1.0
            else:
                mask2 = h > 0.0
            idx = np.where(mask2)[0]
            if len(idx) > 0:
                h[idx[0]] += diff2
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new)

def smooth_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    tc = t * corr * dx
    tc_max = tc.max()
    e = np.exp(tc - tc_max)
    val = (tc_max + np.log(e.sum())) / t * dx
    p = e / e.sum()
    n_half = n - 1
    grad = np.zeros(n)
    for k in range(len(corr)):
        pk = p[k]
        if pk < 1e-13:
            continue
        L = k - n_half
        if L >= 0:
            idx1 = np.arange(n - L)
            idx2 = idx1 + L
            grad[idx1] += pk * (1.0 - h[idx2])
            grad[idx2] -= pk * h[idx1]
        else:
            M = -L
            idx2 = np.arange(M, n)
            idx1 = idx2 - M
            grad[idx2] += pk * (1.0 - h[idx1])
            grad[idx1] -= pk * h[idx2]
    return val, dx * grad

def proj_grad_descent(h, t, n_steps, start_lr, decay, time_budget, start_time):
    n = len(h)
    target = n / 2.0
    lr = start_lr
    best_h = h.copy()
    best_c = compute_c5(h)
    for step in range(n_steps):
        if time.time() - start_time > time_budget - 30:
            break
        val, grad = smooth_obj_grad(h, t)
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
        lr *= decay
        if lr < 1e-12:
            lr = 1e-6
    return best_h, best_c

def coord_descent(h, max_rounds=40, time_budget=None, start_time=None):
    n = len(h)
    target = n / 2.0
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time_budget is not None and time.time() - start_time > time_budget - 25:
            break
        improved = False
        for j in range(n):
            if time_budget is not None and time.time() - start_time > time_budget - 25:
                return h, best_c
            old_val = h[j]
            candidates = [0.0, 1.0, 0.5,
                          max(0.0, old_val - 0.2), max(0.0, old_val - 0.08),
                          max(0.0, old_val - 0.03), min(1.0, old_val + 0.03),
                          min(1.0, old_val + 0.08), min(1.0, old_val + 0.2)]
            best_val = old_val
            best_obj = best_c
            for cv in candidates:
                cv = np.clip(cv, 0.0, 1.0)
                if abs(cv - old_val) < 1e-7:
                    continue
                trial = h.copy()
                trial[j] = cv
                diff = target - np.sum(trial)
                if abs(diff) > 1e-10:
                    mask = np.ones(n, dtype=bool)
                    mask[j] = False
                    if diff > 0:
                        mask &= trial < 1.0
                    else:
                        mask &= trial > 0.0
                    cnt = np.sum(mask)
                    if cnt > 0:
                        trial[mask] += diff / cnt
                        trial = np.clip(trial, 0.0, 1.0)
                    diff2 = target - np.sum(trial)
                    trial[j] = np.clip(trial[j] + diff2, 0.0, 1.0)
                c = compute_c5(trial)
                if c < best_obj - 1e-11:
                    best_obj = c
                    best_val = cv
            if best_val != old_val:
                h = h.copy()
                h[j] = best_val
                h = project_feasible(h, target)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def pairwise_sa(h, n_iters, T_init, T_final, time_budget, start_time, seed=123):
    n = len(h)
    target = n / 2.0
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = compute_c5(h)
    curr_c = best_c
    for it in range(n_iters):
        if time.time() - start_time > time_budget - 30:
            break
        frac = it / max(n_iters, 1)
        T = T_init * (T_final / T_init) ** frac
        i, j = rng.choice(n, size=2, replace=False)
        di = rng.uniform(-0.3, 0.3)
        dj = rng.uniform(-0.3, 0.3)
        trial = h.copy()
        trial[i] = np.clip(h[i] + di, 0.0, 1.0)
        trial[j] = np.clip(h[j] + dj, 0.0, 1.0)
        trial = project_feasible(trial, target)
        c_new = compute_c5(trial)
        delta = c_new - curr_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-8)):
            h = trial
            curr_c = c_new
            if curr_c < best_c:
                best_c = curr_c
                best_h = h.copy()
    return best_h, best_c

def binary_refine(h, time_budget, start_time, seed=1):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    for thresh in [0.5, 0.45, 0.55, 0.4, 0.6]:
        h_bin = (h > thresh).astype(float)
        n_ones = int(np.sum(h_bin))
        if n_ones > target:
            ones = np.where(h_bin == 1)[0]
            rng = np.random.RandomState(seed + int(thresh * 100))
            remove = rng.choice(ones, size=n_ones - int(target), replace=False)
            h_bin[remove] = 0.0
        elif n_ones < target:
            zeros = np.where(h_bin == 0)[0]
            rng = np.random.RandomState(seed + int(thresh * 100))
            add = rng.choice(zeros, size=int(target) - n_ones, replace=False)
            h_bin[add] = 1.0
        h_bin = project_feasible(h_bin, target)
        c = compute_c5(h_bin)
        if c < best_c - 1e-10:
            best_c = c
            best_h = h_bin
            h_cd, c_cd = coord_descent(best_h, max_rounds=15,
                                       time_budget=time_budget, start_time=start_time)
            if c_cd < best_c:
                best_c = c_cd
                best_h = h_cd
    return best_h, best_c
