```python
            # = (base_c_k + h[j+k]*1[j+k valid]) + t * ((1-h[j-k])*1[j-k valid] - h[j+k]*1[j+k valid])
            #
            # So D_k = base_c_k + h[j+k]*1[j+k<n] (h[j+k] from current h with h[j]=0)
            # A_k = (1-h[j-k])*1[j-k valid] - h[j+k]*1[j+k valid]
            
            # h[j] is currently 0, so h[j+k] is just h[j+k] (the actual value)
            # and h[j-k] is just h[j-k]
            
            # A_k for lag k (where k ranges from -(n-1) to n-1, index k+n-1 in corr):
            # A_k = (1-h[j-k])*1[0<=j-k<n] - h[j+k]*1[0<=j+k<n]
            # Since h[j]=0 currently:
            # A_k = (1-h[j-k])*1[j-k<n and j-k>=0] - h[j+k]*1[j+k<n]
            
            A = np.zeros(2*n-1)
            if j - (n-1) >= 0 and j + (n-1) < 2*n:  # always true for 0<=j<n
                # h[j-k] is valid when 0 <= j-k < n, i.e., j <= k <= j+n-1... 
                # k ranges from -(n-1) to n-1, index k+n-1 from 0 to 2n-2
                # h[j-k] valid: 0 <= j-k < n → j-n+1 <= k <= j
                # h[j+k] valid: 0 <= j+k < n → -j <= k <= n-1-j
                
                # Vectorized:
                ks = np.arange(-(n-1), n)  # all lag values
                A[j-n+1:j+1] += (1.0 - h[max(0, j-(n-1)):min(n, j+1) if j+1<n else j+1])
                # Hmm this is getting complicated. Let me just compute A directly.
            
            # Actually, let me simplify. The range of k where h[j-k] is valid:
            # 0 <= j-k < n → j-n < k <= j → k ∈ {j-n+1, ..., j} ∩ {-(n-1), ..., n-1}
            # The range of k where h[j+k] is valid:
            # 0 <= j+k < n → -j <= k < n-j → k ∈ {-j, ..., n-1-j} ∩ {-(n-1), ..., n-1}
            
            A[:] = 0.0
            # For h[j-k] term: k from max(-(n-1), j-n+1) to min(n-1, j)
            k_lo1 = max(-(n-1), j - n + 1)
            k_hi1 = min(n-1, j)
            if k_lo1 <= k_hi1:
                idx = np.arange(k_lo1 + n - 1, k_hi1 + n)  # indices in corr array
                h_vals = h[j - k_lo1 : j - k_hi1 + 1][::-1]  # h[j-k] for k=k_lo1..k_hi1
                A[idx] += (1.0 - h_vals)
            
            # For h[j+k] term: k from max(-(n-1), -j) to min(n-1, n-1-j)
            k_lo2 = max(-(n-1), -j)
            k_hi2 = min(n-1, n - 1 - j)
            if k_lo2 <= k_hi2:
                idx = np.arange(k_lo2 + n - 1, k_hi2 + n)
                h_vals = h[j + k_lo2 : j + k_hi2 + 1]  # h[j+k] for k=k_lo2..k_hi2
                A[idx] -= h_vals
            
            # D_k = base_c_k (since h[j]=0, the "base" already accounts for h[j] being 0)
            # Wait, base_c_k was computed with h[j]=0. 
            # The actual c_k with h[j]=t is:
            # = base_c_k + t * A_k (where A_k is as computed above)
            # BUT: base_c_k was computed with h[j]=0, so it INCLUDES the effect of h[j]=0
            # on the (1-h[i]) terms. When we set h[j]=t, the (1-h[j]) term changes.
            #
            # Let me redo: c_k = ∑_i h[i+k]*(1-h[i])
            # With h[j]=t:
            # = ∑_{i≠j} h[i+k]*(1-h[i]) + h[j+k]*(1-t)·1[j+k valid] + t*(1-h[j-k])·1[j-k valid]
            # = ∑_{i≠j} h[i+k]*(1-h[i]) + h[j+k]·1[j+k valid] - t·h[j+k]·1[j+k valid] + t·(1-h[j-k])·1[j-k valid]
            # = [∑_{i≠j} h[i+k]*(1-h[i])] + h[j+k]·1[j+k valid] + t·[(1-h[j-k])·1[j-k valid] - h[j+k]·1[j+k valid]]
            #
            # With h[j]=0 in base:
            # base_c_k = ∑_{i≠j} h[i+k]*(1-h[i]) + 0 + 0 (since h[j]=0)
            # Wait no. With h[j]=0:
            # c_k = ∑_i h[i+k]*(1-h[i]) where h[j]=0
            # = ∑_{i≠j} h[i+k]*(1-h[i]) + h[j+k]*(1-0)·1[j+k valid] + 0·(1-h[j-k])·1[j-k valid]
            # = ∑_{i≠j} h[i+k]*(1-h[i]) + h[j+k]·1[j+k valid]
            #
            # So base_c_k = ∑_{i≠j} h[i+k]*(1-h[i]) + h[j+k]·1[j+k valid]
            # And D_k = base_c_k, A_k = (1-h[j-k])·1[j-k valid] - h[j+k]·1[j+k valid]
            # 
            # So c_k = base_c_k + t * A_k = D_k + t * A_k ✓
            
            # Now find min_{0≤t≤1} max_k (D_k + t*A_k) * dx
            # This is a 1D minimax. Evaluate at t=0, t=1, and critical points.
            # For efficiency, just evaluate at a grid:
            ts = np.array([0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0])
            best_t = 0.0
            best_val = np.max(base_corr + ts[0] * A)
            for ti in ts[1:]:
                val = np.max(base_corr + ti * A)
                if val < best_val:
                    best_val = val
                    best_t = ti
            
            # Refine around best_t
            if 0 < best_t < 1:
                for ti in np.linspace(best_t - 0.1, best_t + 0.1, 11):
                    if 0 <= ti <= 1:
                        val = np.max(base_corr + ti * A)
                        if val < best_val:
                            best_val = val
                            best_t = float(ti)
            
            h[j] = best_t
            if abs(h[j] - old_val) > 1e-8:
                improved = True
        
        # Re-project after each pass
        h = project_c(h, n)
        if not improved:
            break
    return h
```

Hmm, this coordinate descent is O(n * (2n)) per coordinate = O(n * 2n) = O(2n²) per full pass. For n=960, that's about 1.8M operations per pass. With 10 passes, that's 18M. Fine.

But wait, I'm computing base_corr (via np.correlate) for each coordinate j! That's O(n * n²) = O(n³) total. For n=960, that's 900M operations. That's too slow.

I need to update base_corr incrementally. When I change h[j] from 0 to t, the correlation changes in a specific way. But since I'm setting h[j]=0 first and then evaluating, I need the correlation of h with h[j]=0.

Alternative: don't zero out h[j]. Instead, compute the effect analytically.

Let me reconsider. The current h has h[j] = old_val. I want to compute c_k as a function of t = h[j]:
c_k(t) = c_k(current) + (t - old_val) * A_k

where A_k = (1-h[j-k])·1[j-k valid] - h[j+k]·1[j+k valid]

This avoids recomputing the correlation! I just need the current correlation (computed once) and the A_k values.

So the algorithm is:
1. Compute corr = np.correlate(h, 1-h, mode='full') once
2. For each j:
   a. Compute A_k for this j
   b. Find best t = argmin_{0≤t≤1} max_k (corr[k] + (t-h[j])*A_k)
   c. Update h[j] = t
   d. Update corr: corr += (t - old_val) * A_k (only for the affected k's)
3. Re-project

This makes each coordinate update O(n) (computing A and finding best t), and the full pass is O(n²). 

Let me implement this properly:

```python
def coord_descent_fast(h, n_passes=5):
    n = len(h)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for p in range(n_passes):
        for j in range(n):
            old_val = h[j]
            
            # Compute A_k: coefficient of h[j] in c_k
            # A_k = (1-h[j-k])*1[0<=j-k<n] - h[j+k]*1[0<=j+k<n]
            A = np.zeros(2 * n - 1)
            
            # h[j-k] valid when 0 <= j-k < n, i.e., j-n+1 <= k <= j
            k_lo = max(-(n-1), j - n + 1)
            k_hi = min(n-1, j)
            if k_lo <= k_hi:
                # j-k ranges from j-k_lo to j-k_hi, i.e., from min(j, n-1) to max(0, j-n+1)
                # idx in A: k + n - 1
                for k in range(k_lo, k_hi + 1):
                    A[k + n - 1] += (1.0 - h[j - k])
            
            # h[j+k] valid when 0 <= j+k < n, i.e., -j <= k <= n-1-j
            k_lo = max(-(n-1), -j)
            k_hi = min(n-1, n - 1 - j)
            if k_lo <= k_hi:
                for k in range(k_lo, k_hi + 1):
                    A[k + n - 1] -= h[j + k]
            
            # Find best t: min_{0<=t<=1} max_k (corr[k] + (t - old_val) * A[k])
            # = min_{0<=t<=1} max_k (corr[k] + old_val*A[k]) - old_val*A[k] + t*A[k])
            # Let B[k] = corr[k] - old_val * A[k]  (this is c_k with h[j]=0... wait)
            # c_k(t) = corr[k] + (t - old_val) * A[k]
            
            # Evaluate at grid points
            best_t = old_val
            best_max = np.max(corr + (old_val - old_val) * A)  # = max(corr)
            for t in [0.0, 0.25, 0.5, 0.75, 1.0]:
                m = np.max(corr + (t - old_val) * A)
                if m < best_max:
                    best_max = m
                    best_t = t
            
            # Refine
            if best_t > 0 and best_t < 1:
                lo_t, hi_t = max(0, best_t - 0.15), min(1, best_t + 0.15)
                for _ in range(5):
                    mid_t = (lo_t + hi_t) / 2
                    m1 = np.max(corr + (lo_t - old_val) * A)
                    m2 = np.max(corr + (hi_t - old_val) * A)
                    m3 = np.max(corr + (mid_t - old_val) * A)
                    if m3 <= m1 and m3 <= m2:
                        pass  # mid is best, narrow around it
                        hi_t = mid_t
                    elif m1 <= m2:
                        hi_t = mid_t
                    else:
                        lo_t = mid_t
                # Just evaluate at a few points
                for t in np.linspace(lo_t, hi_t, 9):
                    m = np.max(corr + (t - old_val) * A)
                    if m < best_max:
                        best_max = m
                        best_t = float(t)
            
            # Update
            h[j] = best_t
            corr += (best_t - old_val) * A
            corr = np.maximum(corr, 0.0)  # numerical safety
        
        # Re-project
        h = project_c(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    return h
```

Wait, the inner loops over k for computing A are O(n) each, and we do this for n coordinates, so the full pass is O(n²). With the grid evaluation being O(n) per coordinate (since A has 2n-1 entries), total is O(n²) per pass. For n=960, that's about 1M per pass. Very fast!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad(h, shift, n):
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def fista_opt(h, n_iter, T0, T1, lr, top_k, check_every):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.3 * f)
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            g = subgrad(z, int(np.argmax(corr)) - (n-1), n)
        else:
            w = exps / tot
            idx = np.argsort(w)[::-1][:top_k]
            g = np.zeros(n)
            for i in idx:
                wi = w[i]
                if wi < 1e-10:
                    break
                g += wi * subgrad(z, i - (n-1), n)
        g *= dx
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check_every == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def coord_descent(h, n_passes=5):
    n = len(h)
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    
    for p in range(n_passes):
        for j in range(n):
            old_val = h[j]
            A = np.zeros(2 * n - 1)
            
            # Part 1: (1-h[j-k]) for valid j-k
            k_lo = max(-(n-1), j - n + 1)
            k_hi = min(n-1, j)
            if k_lo <= k_hi:
                a_idx = np.arange(k_lo + n - 1, k_hi + n)
                h_idx = np.arange(j - k_hi, j - k_lo + 1)
                A[a_idx] += (1.0 - h[h_idx][::-1])
            
            # Part 2: -h[j+k] for valid j+k
            k_lo2 = max(-(n-1), -j)
            k_hi2 = min(n-1, n - 1 - j)
            if k_lo2 <= k_hi2:
                a_idx2 = np.arange(k_lo2 + n - 1, k_hi2 + n)
                h_idx2 = np.arange(j + k_lo2, j + k_hi2 + 1)
                A[a_idx2] -= h[h_idx2]
            
            # Find best t
            best_t = old_val
            best_max = np.max(corr)  # at t = old_val, corr + 0*A = corr
            
            # Grid evaluation
            for t in [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0]:
                m = np.max(corr + (t - old_val) * A)
                if m < best_max:
                    best_max = m
                    best_t = t
            
            # Refine
            lo_t, hi_t = 0.0, 1.0
            if best_t > 0.05 and best_t < 0.95:
                lo_t, hi_t = max(0, best_t - 0.1), min(1, best_t + 0.1)
            for t in np.linspace(lo_t, hi_t, 13):
                m = np.max(corr + (t - old_val) * A)
                if m < best_max:
                    best_max = m
                    best_t = float(t)
            
            # Update
            delta = best_t - old_val
            if abs(delta) > 1e-10:
                h[j] = best_t
                corr += delta * A
                corr = np.maximum(corr, 0.0)
        
        h = project_c(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    
    return h

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].astype(float)

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bc = float('inf')
    bh = None
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = project_c(rng.rand(960), 960)
    
    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"INIT: n={n0} C5={bc:.6f}", flush=True)
    
    def remaining():
        return budget_s - (time.time() - t0)
    
    # Phase 1: FISTA with moderate T
    if remaining() > 200:
        h1, c1 = fista_opt(bh, 4000, 30.0, 0.5, 0.12, 150, 30)
        if c1 < bc:
            bc, bh = c1, h1
        print(f"F1: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)
    
    # Phase 2: Coordinate descent
    if remaining() > 150:
        h2 = coord_descent(bh, 5)
        c2 = compute_c5(h2)
        if c2 < bc:
            bc, bh = c2, h2
        print(f"CD1: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)
    
    # Phase 3: FISTA with low T (exploitation)
    if remaining() > 150:
        h3, c3 = fista_opt(bh, 4000, 3.0, 0.05, 0.05, 100, 20)
        if c3 < bc:
            bc, bh = c3, h3
        print(f"F2: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)
    
    # Phase 4: More coordinate descent
    if remaining() > 100:
        h4 = coord_descent(bh, 5)
        c4 = compute_c5(h4)
        if c4 < bc:
            bc, bh = c4, h4
        print(f"CD2: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)
    
    # Phase 5: Try Thue-Morse based start
    if remaining() > 120:
        tm = project_c(tm_seq(n0), n0)
        h5, c5v = fista_opt(tm, 3000, 20.0, 0.3, 0.1, 100, 30)
        c5v = compute_c5(h5)
        if c5v < bc:
            bc, bh = c5v, h5
            print(f"TM: C5={bc:.6f}", flush=True)
        # Also try CD on TM result
        if remaining() > 60:
            h5b = coord_descent(h5, 3)
            c5b = compute_c5(h5b)
            if c5b < bc:
                bc, bh = c5b, h5b
                print(f"TM-CD: C5={bc:.6f}", flush=True)
    
    # Phase 6: Try downsample and re-upsample (smoothing)
    if remaining() > 100:
        # Downsample to n/2, then upsample back (acts as smoothing)
        n_half = n0 // 2
        ox = np.linspace(0, 2, n0)
```
