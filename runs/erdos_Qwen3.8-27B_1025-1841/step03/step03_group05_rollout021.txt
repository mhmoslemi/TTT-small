import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    """Project to [0,1]^n with sum = n/2 using bisection on threshold."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    """Subgradient of corr[shift] with respect to h."""
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=50):
    """Smooth gradient using softmax weighting over top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if mc < 1e-15:
        return np.zeros(n)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return _subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    # Focus on top-k
    idx = np.argsort(w)[::-1][:top_k]
    w_subset = w[idx]
    w_subset = w_subset / w_subset.sum()  # renormalize
    g = np.zeros(n)
    for i in idx:
        g += w_subset[i] * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter=5000, T0=30.0, T1=0.05, lr=0.12, check=50, top_k0=10, top_k1=30):
    """FISTA with adaptive temperature and top-k."""
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        # Adaptive top-k: start focused, widen later
        tk_cur = int(top_k0 + (top_k1 - top_k0) * f)
        lrt = lr * (1.0 - 0.3 * f)
        
        g = sm_grad(z, T, tk_cur)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def tm_seq(n):
    """Thue-Morse sequence."""
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def tm_seq_offset(n, offset=1):
    """Thue-Morse sequence with offset."""
    t = tm_seq(2 * n)
    return t[offset:offset + n].astype(float)

def upsample(h, nn):
    """Upsample h to nn points using linear interpolation and projection."""
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def binary_project(h, n):
    """Project to binary sequence with exactly n/2 ones."""
    h = h.copy()
    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def binary_local_search(h_bin, max_iters=1000, time_limit=30):
    """Local search on binary sequences: try swaps to reduce max corr."""
    t0 = time.time()
    n = len(h_bin)
    g = h_bin.astype(float).copy()
    bc = compute_c5(g)
    
    ones = np.where(g == 1)[0]
    zeros = np.where(g == 0)[0]
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        # Try a random sample of swaps
        best_delta = 0
        best_i, best_j = -1, -1
        
        # Sample up to 200 random swaps
        n_try = min(200, len(ones) * len(zeros))
        for _ in range(n_try):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            
            # Evaluate swap
            g[i], g[j] = 0, 1
            c = compute_c5(g)
            delta = bc - c
            g[i], g[j] = 1, 0
            
            if delta > best_delta + 1e-10:
                best_delta = delta
                best_i, best_j = i, j
        
        if best_i >= 0:
            g[best_i], g[best_j] = 0, 1
            bc -= best_delta
            ones = np.where(g == 1)[0]
            zeros = np.where(g == 0)[0]
        else:
            break
    
    return g, bc

def slsqp_refine(h, T=5.0, max_iter=400):
    """SLSQP refinement using smooth objective."""
    n = len(h)
    _slsqp_state = {'T': T, 'n': n}
    
    def sobj(h):
        T = _slsqp_state['T']
        n = _slsqp_state['n']
        dx = 2.0 / n
        c = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx
    
    def scon(h):
        return np.sum(h) - _slsqp_state['n'] / 2.0
    
    def sjac(h):
        return np.ones(_slsqp_state['n'])
    
    try:
        r = minimize(sobj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': scon, 'jac': sjac}],
                     options={'maxiter': max_iter, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if mc < 1e-15:
        return np.zeros(n)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return _subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    ws = w[idx]
    ws = ws / ws.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if ws[i] < 1e-12:
            break
        g += ws[i] * _subgrad_one(h, idx[i] - (n - 1))
    return g * dx

            hu2, cu2 = fista(hu, 2500, 1.0, 0.02, 0.02, 25, 3, 8)
            if cu2 < cu:
                hu, cu = hu2, cu2
        cf = compute_c5(hu)
        if cf < bc:
            bc, bh = cf, hu.copy()
        print(f"  U{nn2}: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 7: Perturbation restarts
    np_ = len(bh)
    for trial in range(5):
        if budget_s - elapsed() < 25:
            break
        sc = 0.015 + 0.02 * trial
        hp = project_c(bh + rng.randn(np_) * sc, np_)
        hr, cr = fista(hp, 2500, 10.0, 0.1, 0.05, 35, 4, 12)
        if cr < bc:
            bc, bh = cr, hr.copy()
            print(f"  SA{trial}: C5={bc:.6f}", flush=True)

    # Phase 8: Final SLSQP + FISTA
    np_ = len(bh)
    if np_ <= 1200 and budget_s - elapsed() > 20:
        bh = slsqp_refine(bh, 1.0, 300)
        bc = compute_c5(bh)

    if budget_s - elapsed() > 15:
        hf, cf = fista(bh, 1500, 1.0, 0.01, 0.008, 20, 3, 6)
        if cf < bc:
            bc, bh = cf, hf.copy()

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={elapsed():.0f}", flush=True)
    return bh, bc, len(bh)
