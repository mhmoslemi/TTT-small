    target = n / 2.0

    old_x = np.linspace(0.0, 2.0, old_n)

    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Load initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    N = len(h_init)
    h_init = make_feasible(h_init, N)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Phase 1: Fast coordinate descent at initial resolution
    # Budget: 40% of total time
    cd_budget = budget_s * 0.4
    cd_start = time.time()
    
    while time.time() - cd_start < cd_budget and remaining() > 100:
        h = best_h.copy()
        corr = np.correlate(h, 1.0 - h, 'full').copy()
        improved = False
        
        for j in range(N):
            if time.time() - cd_start > cd_budget:
                break
            old = h[j]
            old_max = np.max(corr)
            
            # Try candidates
            cands = [0.0, 1.0]
            for step in [0.02, 0.05, 0.1]:
                for s in [-1, 1]:
                    v = old + s * step
                    if 0.0 <= v <= 1.0:
                        cands.append(v)
            
            best_val = old
            best_max = old_max
            save_corr = corr.copy()
            
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                delta = cand - old
                update_corr_inplace(corr, h, j, delta, N)
                new_max = np.max(corr)
                if new_max < best_max - 1e-15:
                    best_max = new_max
                    best_val = cand
            
            # Restore and apply best
            corr[:] = save_corr
            if best_val != old:
                delta = best_val - old
                update_corr_inplace(corr, h, j, delta, N)
                h[j] = best_val
                improved = True
        
        # Repair
        h = make_feasible(h, N)
        corr = np.correlate(h, 1.0 - h, 'full')
        curr_c5 = np.max(corr) * (2.0 / N)
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        
        if not improved:
            break
    
    # Phase 2: SLSQP polish
    if remaining() > 80:
        # Use a smoothed objective
        n_pts = len(best_h)
        T = 500.0
        dx = 2.0 / n_pts
        
        def smooth_obj(x):
            c = np.correlate(x, 1.0 - x, 'full')
            mc = np.max(c)
            return mc * dx + np.log(np.sum(np.exp(T * (c - mc)))) / T * dx
        
        def smooth_grad(x):
            c = np.correlate(x, 1.0 - x, 'full')
            k_star = np.argmax(c)
            # Gradient of max(c) at k_star
            # dc[k_star]/dx[j] for each j
            # This is complex, let me just use numerical gradient
            return None  # Let SLSQP handle it
        
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_pts / 2.0,
                 'jac': lambda x: np.ones(n_pts)}]
        
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14})
            if res.success or res.fun < best_c5 + 0.001:
                h_res = make_feasible(res.x, n_pts)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res
        except Exception:
            pass

    if remaining() > 100:
