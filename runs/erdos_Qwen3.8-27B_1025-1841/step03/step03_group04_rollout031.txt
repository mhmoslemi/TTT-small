def compute_gradient(h, w, dx):
    n = len(h)
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        w_s = w[s + n - 1]
        if w_s < 1e-14:
            continue
        if s >= 0:
            grad[:n-s] += w_s * (1.0 - h[s:n])
            grad[s:] -= w_s * h[:n-s]
        else:
            grad[-s:] += w_s * (1.0 - h[:n+s])
            grad[:n+s] -= w_s * h[-s:]
    grad *= dx
    return grad

def adam_projected(h, n, dx, T_schedule, max_iters, start_time, budget_s):
    """Optimize using Adam with projection and adaptive T."""
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.01
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 30:
            break
        t += 1
        T = T_schedule(it)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        logits = T * shifted
        max_logit = np.max(logits)
        exp_l = np.exp(logits - max_logit)
        w = exp_l / np.sum(exp_l)
        
        grad = compute_gradient(h, w, dx)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h_new, n)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

            continue
        if s >= 0:
            grad[:n - s] += w_s * (1.0 - h[s:])
            grad[s:] -= w_s * h[:n - s]
        else:
            grad[-s:] += w_s * (1.0 - h[:n + s])
            grad[:n + s] -= w_s * h[-s:]
    grad *= dx
    return grad

def adam_optimize(h, n, dx, T_start, T_end, max_iters, start_time, budget_s, lr=0.005):
    h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 30:
            break
        t += 1
        frac = min(1.0, it / max(1, max_iters // 2))
        T = T_start + (T_end - T_start) * frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        w, mc = soft_max_weights(corr, T)
        grad = compute_gradient(h, w, dx)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_feasible(h - step, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.8
                if lr < 1e-6:
                    break
                stall = 0
        h = h_new
    return best_h, best_c5

def coord_descent(h, n, start_time, budget_s, max_passes=3, fine=False):
    best_c5 = compute_c5(h)
    if fine:
        steps = [0.001, 0.002, 0.003, 0.005]
        snaps = [0.0, 1.0]
    else:
        steps = [0.005, 0.01, 0.02, 0.05]
        snaps = [0.0, 1.0, 0.25, 0.5, 0.75]
    for p in range(max_passes):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                return h, best_c5
            old = h[j]
            best_val = old
            best_c = best_c5
            for val in snaps:
                if abs(val - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c:
                    best_c = c
                    best_val = val
            for st in steps:
                for sign in [-1, 1]:
                    val = old + sign * st
                    if 0.0 <= val <= 1.0:
                        trial = h.copy()
                        trial[j] = val
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c:
                            best_c = c
                            best_val = val
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_polish(h, n, start_time, budget_s):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        T = 500.0
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con_f(x):
        return np.sum(x) - n / 2.0
    def con_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-14})
        c5 = compute_c5(res.x)
        if c5 < compute_c5(h):
            return make_feasible(res.x, n), c5
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    dx = 2.0 / n
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Adam with moderate T
    if remaining() > 200:
        h, c5 = adam_optimize(best_h, n, dx, 100, 400, 1500, start_time, budget_s, lr=0.008)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 2: Adam with high T
    if remaining() > 150:
        h, c5 = adam_optimize(best_h, n, dx, 300, 1000, 800, start_time, budget_s, lr=0.004)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 3: Coordinate descent
    if remaining() > 80:
        h, c5 = coord_descent(best_h, n, start_time, budget_s, max_passes=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 4: SLSQP polish
    if remaining() > 80:
        h, c5 = slsqp_polish(best_h, n, start_time, budget_s)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 5: Random restarts
    for r in range(6):
        if remaining() < 80:
            break
        scale = 0.01 + 0.005 * r
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        h, c5 = adam_optimize(h_try, n, dx, 150, 600, 400, start_time, budget_s, lr=0.005)
        if c5 < best_c5:
            best_c5, best_h = c5, h
            if remaining() > 40:
                h, c5 = coord_descent(best_h, n, start_time, budget_s, max_passes=1)
                if c5 < best_c5:
                    best_c5, best_h = c5, h

    # Phase 6: Very high T Adam
    if remaining() > 100:
        h, c5 = adam_optimize(best_h, n, dx, 800, 2000, 500, start_time, budget_s, lr=0.002)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 7: Try upsample
    if remaining() > 300:
        n2 = n * 2
        if n2 <= 2000:
            old_x = np.linspace(0.0, 2.0, n)
            new_x = np.linspace(0.0, 2.0, n2)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = make_feasible(h_up, n2)
            dx2 = 2.0 / n2
            c5_up = compute_c5(h_up)
            if c5_up < best_c5 + 0.003:
                h, c5 = adam_optimize(h_up, n2, dx2, 100, 500, 800, start_time, budget_s, lr=0.005)
                if c5 < best_c5:
                    best_c5, best_h = c5, h
                if remaining() > 60:
                    h, c5 = coord_descent(best_h, n2, start_time, budget_s, max_passes=2)
                    if c5 < best_c5:
                        best_c5, best_h = c5, h

    # Phase 8: Fine coordinate descent
    if remaining() > 20:
        h, c5 = coord_descent(best_h, len(best_h), start_time, budget_s, max_passes=2, fine=True)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
