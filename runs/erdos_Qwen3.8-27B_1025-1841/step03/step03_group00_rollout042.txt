def project_onto_simplex_box(h, target_sum):
    # Project h onto {x: 0≤x≤1, sum(x) = target_sum}
    # Binary search on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any numerical issues
    diff = target_sum - h_proj.sum()
    if abs(diff) > 1e-12:
        # Add/subtract from the most "interior" elements
        if diff > 0:
            mask = h_proj < 1 - 1e-12
        else:
            mask = h_proj > 1e-12
        idx = np.where(mask)[0]
        h_proj[idx[0]] += diff
        h_proj[idx[0]] = np.clip(h_proj[idx[0]], 0, 1)
    return h_proj

I think the projected gradient with log-sum-exp smoothing is the way to go, with FFT acceleration.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    """Compute C5 using FFT-based linear correlation."""
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    R = np.fft.irfft(H * np.conj(H), N)[:n].real
    PS = np.concatenate([[0.0], np.cumsum(h)])
    S = PS[n]
    c_pos = PS[n:0:-1] - R
    c_neg = (S - PS[1:n+1]) - R
    return np.max(np.maximum(c_pos, c_neg)) * dx

def project_box_sum(h, target):
    """Project h onto {0<=x<=1, sum(x)=target}."""
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h_p = np.clip(h - lam, 0, 1)
    diff = target - h_p.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_p < 1 - 1e-13
        else:
            mask = h_p > 1e-13
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h_p[idx[0]] = np.clip(h_p[idx[0]] + diff, 0, 1)
    return h_p

def make_feasible(h, n):
    h = np.clip(h, 0, 1).copy()
    target = n / 2.0
    return project_box_sum(h, target)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    x_old = np.linspace(0, 2, old_n, endpoint=False)
    x_new = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return make_feasible(h_new, new_n)

def fft_grad_step(h, T, dx):
    """Compute gradient of smoothed max correlation."""
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    R = np.fft.irfft(H * np.conj(H), N)[:n].real
    PS = np.concatenate([[0.0], np.cumsum(h)])
    S = PS[n]
    c_pos = PS[n:0:-1] - R
    c_neg = (S - PS[1:n+1]) - R
    
    f = np.concatenate([c_pos, c_neg[1:]]) * dx  # 2n-1 values
    
    # Log-sum-exp
    f_max = f.max()
    exp_f = np.exp(T * (f - f_max))
    w_all = exp_f / exp_f.sum()
    
    w_pos = w_all[:n].copy()
    w_neg = np.zeros(n)
    w_neg[1:] = w_all[n:]
    w = w_pos + w_neg  # w[0] = w_pos[0] + 0
    
    # A[j] = sum_{d=0}^{n-1-j} w_pos[d]
    cumsum_wp = np.cumsum(w_pos)
    A = cumsum_wp[n-1::-1][:n].copy()
    
    # B[j] = sum_{d=1}^{j} w_neg[d]
    cumsum_wn = np.cumsum(w_neg)
    B = cumsum_wn.copy()
    
    # D[j] = sum_{d=0}^{j} w[d]*h[j-d] = (w*h)[j]
    D = np.convolve(w, h, 'full')[:n]
    
    # C[j] = sum_{d=0}^{n-1-j} w[d]*h[j+d] = (w * flip(h))[n-1-j]
    h_flip = h[::-1]
    C_full = np.convolve(w, h_flip, 'full')
    C = C_full[n-1::-1][:n].copy()
    
    grad = dx * (A - B - C - D)
    obj = f_max + np.log(exp_f.sum()) / T
    return grad, obj, w_all

def projected_gradient(h, n, T_schedule, max_iters, alpha0=0.01):
    """Projected gradient with backtracking on smoothed max."""
    dx = 2.0 / n
    target = n / 2.0
    h = h.copy()
    best_obj = compute_c5_fft(h)
    
    for it in range(max_iters):
        grad, obj, _ = fft_grad_step(h, T_schedule[it % len(T_schedule)], dx)
        
        # Backtracking line search
        alpha = alpha0
        h_trial = project_box_sum(h - alpha * grad, target)
        c5_trial = compute_c5_fft(h_trial)
        
        while c5_trial > obj and alpha > 1e-8:
            alpha *= 0.5
            h_trial = project_box_sum(h - alpha * grad, target)
            c5_trial = compute_c5_fft(h_trial)
        
        if c5_trial < best_obj:
            best_obj = c5_trial
            h = h_trial.copy()
        
        h = h_trial
    
    return h, best_obj

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

The previous code uses:

2. Binary local search
3. SLSQP polish

The main limitation is probably:
- The resolution (n=2048 max in the previous code)
- The number of optimization iterations (limited by the O(n²) cost of np.correlate)

With FFT, I can:
- Use n=4096 or 8192
- Do many more iterations
- Compute exact gradients (faster convergence)

Let me focus on a clean, efficient implementation:

1. Start from parent (n=640), upsample to n=2048
2. Run projected gradient with FFT (fast, many iterations)
3. Run at n=4096
4. Binary local search at each resolution
5. Final coordinate descent polish

For the binary local search, I should also use FFT to make it fast.

Let me also consider: what if I use a different parametrization? Instead of optimizing h directly, I could optimize the "excess" above 1/2:
h = 1/2 + g, where sum(g) = 0 and -1/2 ≤ g ≤ 1/2.

The overlap becomes:
O(d) = dx * sum (1/2 + g[i]) * (1/2 - g[i+d])
= dx * sum [1/4 - g[i+d]/2 + g[i]/2 - g[i]g[i+d]]
= dx * [(n-d)/4 + (PS_g[n-d] - PS_g[d])/2 - R_g[d]]
where PS_g is the prefix sum of g and R_g is the linear autocorrelation of g.

This doesn't seem to simplify things much. Let me stick with the direct approach.

One more thought: the log-sum-exp temperature T controls the "sharpness" of the max approximation. For the final answer, I need T → ∞ (to recover the exact max). But for optimization, I want T finite (for smoothness). A good strategy is:
- Start with T = 10 (smooth, easy to optimize)
- Gradually increase T to 100, 500, 1000 (sharper, more accurate)
- At each level, run several gradient steps

This is similar to simulated annealing (cooling schedule).
