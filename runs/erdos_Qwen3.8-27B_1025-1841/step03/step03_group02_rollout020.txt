So the new correlation at affected indices is: corr_new[k] = corr[k] + dcorr[k] * (v - h[j])

Then the new objective is the log-sum-exp of the new correlations.

def coord_descent_logsumexp(h, n, max_rounds, T, start_time, budget_s):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        # Compute full correlation and derivatives
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        
        for j in range(n):
            # Compute dcorr/dh[j]
            # corr[m] = sum_i h[i] * (1-h[i + m - (n-1)])
            # Let k = m - (n-1)
            # corr[n-1+k] = sum_{i: valid} h[i] * (1-h[i+k])
            # 
            # dcorr[n-1+k]/dh[j]:
            # If j appears as h[i] (i=j): contributes (1-h[j+k]) if 0≤j+k<n
            # If j appears as h[i+k] (i+k=j, i=j-k): contributes -h[j-k] if 0≤j-k<n
            
            dcorr = np.zeros(2*n-1)
            # j as first factor: k ranges such that 0≤j+k<n, i.e., -j ≤ k < n-j
            for k in range(-j, n-j):
                dcorr[n-1+k] += (1.0 - h[j+k])
            # j as second factor: k ranges such that 0≤j-k<n, i.e., j-n+1 < k ≤ j
            for k in range(j-n+1, j+1):
                dcorr[n-1+k] -= h[j-k]
            
            # Find best v in [0,1] for h[j]
            old = h[j]
            # Try a few values
            best_v = old
            best_obj = (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
            
            for v in [0.0, 1.0, old - 0.1, old - 0.01, old + 0.01, old + 0.1]:
                v = max(0.0, min(1.0, v))
                if abs(v - old) < 1e-10:
                    continue
                # New correlation
                new_corr = corr + dcorr * (v - old)
                new_mc = np.max(new_corr)
                new_obj = (new_mc + np.log(np.sum(np.exp(T * (new_corr - new_mc)))) / T) * dx
                if new_obj < best_obj - 1e-15:
                    best_obj = new_obj
                    best_v = v
            
            if best_v != old:
                h[j] = best_v
        
        # Project to feasible set
        h = make_feasible(h, n)
        new_c5 = compute_c5(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return h, best_c5

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        b3 = to_binary(best_h, n0)
        b3, c5_3 = targeted_sa(b3, n0, 500000, 0.005, 0.000001, rng, start_time, budget_s * 0.8, target_frac=0.7)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()

    # Phase 6: Final LSE coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        h_final = best_h.copy()
        for T in [500.0, 2000.0]:
            rem = budget_s - (time.time() - start_time)
            if rem < 30:
                break
            h_final, c5_fin = coord_descent_lse(h_final, n0, T, 20, start_time, budget_s)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_final.copy()

    # Phase 7: Final direct coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        h_final2 = best_h.copy()
        h_final2, c5_fd = coord_descent_direct(h_final2, n0, 20, start_time, budget_s)
        if c5_fd < best_c5:
            best_c5 = c5_fd
            best_h = h_final2.copy()

    # Phase 8: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 400:
        n1 = n0 + 480
        h_up = upsample_h(best_h, n1)
        b_up = to_binary(h_up, n1)
        b_up, c5_up = targeted_sa(b_up, n1, 200000, 0.008, 0.000005, rng, start_time, budget_s * 0.9, target_frac=0.6)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = b_up.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 80:
            h_up2 = best_h.copy()
            h_up2, c5_up2 = coord_descent_direct(h_up2, len(h_up2), 10, start_time, budget_s)
            if c5_up2 < best_c5:
                best_c5 = c5_up2
                best_h = h_up2.copy()

    # Final cleanup
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
