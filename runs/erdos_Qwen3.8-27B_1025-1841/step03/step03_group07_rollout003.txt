import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_correlation(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def top_k_shifts(corr, top_k):
    if top_k >= len(corr):
        return np.arange(len(corr))
    return np.argpartition(corr, -top_k)[-top_k:]

def compute_shift_values(h, shifts):
    """Compute correlation values at specific shifts (shifts are 0-indexed in [0, 2n-2])"""
    n = len(h)
    vals = np.zeros(len(shifts))
    for idx, sh in enumerate(shifts):
        lag = sh - (n - 1)
        if lag >= 0:
            m = n - lag
            vals[idx] = np.dot(h[:m], 1.0 - h[lag:])
        else:
            m = n + lag  # lag is negative, so m = n - |lag|
            vals[idx] = np.dot(h[-lag:], 1.0 - h[:m])
    return vals

def grad_from_shifts(h, shifts, weights, dx):
    """Compute gradient given specific shifts and their weights"""
    n = len(h)
    grad = np.zeros(n)
    for idx, sh in enumerate(shifts):
        w = weights[idx]
        if w < 1e-15:
            continue
        lag = sh - (n - 1)
        if lag >= 0:
            m = n - lag
            # corr = sum_{i=0}^{m-1} h(i)(1-h(i+lag))
            # d/dh(j): for j in [0, m-1]: +(1-h(j+lag)), for j in [lag, n-1]: -h(j-lag)
            if m > 0:
                grad[:m] += w * (1.0 - h[lag:lag+m])
                grad[lag:lag+m] -= w * h[:m]
        else:
            m = n + lag  # negative lag, m = n - |lag|
            # corr = sum_{i=0}^{m-1} h(i+|lag|)(1-h(i))
            # d/dh(j): for j in [0, m-1]: -(h(j+|lag|)), for j in [m, n-1]: +(1-h(j-m))
            alag = -lag
            grad[alag:alag+m] -= w * h[:m]
            grad[:m] += w * (1.0 - h[alag:alag+m])
    grad *= dx
    return grad

def binary_local_search(h, n, time_limit, max_iter=None):
    """Local search on binary configurations"""
    dx = 2.0 / n
    # Round to binary
    target = n // 2
    h_bin = (h > 0.5).astype(float)
    # Adjust to get exactly target ones
    n_ones = int(h_bin.sum())
    if n_ones > target:
        # Remove excess ones (remove the "least important" ones)
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by h value (lower h means less confident about being 1)
        order = np.argsort(h[ones_idx])
        for idx in order[:n_ones - target]:
            h_bin[ones_idx[idx]] = 0
    elif n_ones < target:
        # Add needed ones
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])  # highest h first
        for idx in order[:target - n_ones]:
            h_bin[zeros_idx[idx]] = 1
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    start = time.time()
    iter_count = 0
    stall = 0
    
    while time.time() - start < time_limit:
        iter_count += 1
        if max_iter and iter_count >= max_iter:
            break
        
        corr = compute_correlation(h_bin)
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        c5_now = max_val * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h_bin.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break
        
        # Find positions to flip
        lag = max_k - (n - 1)
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        
        # Bad pairs: h(i)=1 and h(i+lag)=0 (contribute to the max correlation)
        if lag >= 0:
            bad_ones = [i for i in ones if i + lag < n and h_bin[i + lag] == 0]
            good_zeros = [j for j in zeros if j - lag >= 0 and h_bin[j - lag] == 1]
        else:
            alag = -lag
            bad_ones = [i for i in ones if i - alag >= 0 and h_bin[i - alag] == 0]
            good_zeros = [j for j in zeros if j + alag < n and h_bin[j + alag] == 1]
        
        if len(bad_ones) == 0 or len(good_zeros) == 0:
            # Random swap
            if len(ones) > 0 and len(zeros) > 0:
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            else:
                break
        else:
            # Try the best swap: flip a bad one and a good zero
            i = bad_ones[np.random.randint(len(bad_ones))]
            j = good_zeros[np.random.randint(len(good_zeros))]
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    
    return best_h, best_c5

Also, I realize the binary_local_search should be more careful. Let me think about what makes a good move:

For the max shift k* with lag L:
- corr(L) = sum_{i: h(i)=1, h(i+L)=0} 1 (for L ≥ 0, with bounds)
- To decrease corr(L), we want to either:
  a) Set h(i)=0 where h(i)=1 and h(i+L)=0
  b) Set h(i+L)=1 where h(i)=1 and h(i+L)=0

But we must maintain sum(h) = n/2, so every change must be a swap.

A good swap: find i where h(i)=1, h(i+L)=0, AND find j where h(j)=0, h(j+L)=1.
Swapping h(i)↔h(j): 
- Removes contribution at i (h(i) was 1, h(i+L) was 0 → now h(i)=0, no contribution)
- Adds contribution at j (h(j) was 0 → now h(j)=1, but h(j+L) was 1, so no new contribution at j for shift L)
- But it might create new contributions at other shifts!

So we need to check the full C₅ after each swap, not just at shift L. This is the expensive part.

For efficiency, we can first do a "cheap check": estimate the change in the top few shifts, and only do the full check if the cheap check is promising.

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def compute_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')


def top_k_shifts(corr, top_k):
    if top_k >= len(corr):
        return np.arange(len(corr))
    return np.argpartition(corr, -top_k)[-top_k:]


def compute_shift_values(h, shifts):
    n = len(h)
    vals = np.zeros(len(shifts))
    for idx, sh in enumerate(shifts):
        lag = sh - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                vals[idx] = np.dot(h[:m], 1.0 - h[lag:lag + m])
        else:
            m = n + lag
            if m > 0:
                vals[idx] = np.dot(h[-lag:], 1.0 - h[:m])
    return vals


def grad_from_shifts(h, shifts, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for idx, sh in enumerate(shifts):
        w = weights[idx]
        if w < 1e-15:
            continue
        lag = sh - (n - 1)
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += w * (1.0 - h[lag:lag + m])
                grad[lag:lag + m] -= w * h[:m]
        else:
            alag = -lag
            m = n - alag
            if m > 0:
                grad[alag:alag + m] -= w * h[:m]
                grad[:m] += w * (1.0 - h[alag:alag + m])
    grad *= dx
    return grad


def adam_optimize(h, n, tau, time_limit, lr=0.01, top_k=15, recompute_int=40):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    is_m = np.zeros(n)
    is_v = np.zeros(n)
    beta1, beta2, eps = 0.75, 0.999, 1e-8
    t = 0
    stall = 0
    cached_shifts = None

    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        t += 1

        if cached_shifts is None or t % recompute_int == 1:
            corr = compute_correlation(h)
            cached_shifts = top_k_shifts(corr, top_k)

        vals = compute_shift_values(h, cached_shifts)
        shifted = np.clip(tau * (vals - vals.max()), -50.0, 0.0)
        exp_v = np.exp(shifted)
        w = exp_v / exp_v.sum()

        grad = grad_from_shifts(h, cached_shifts, w, dx)

        is_m = beta1 * is_m + (1.0 - beta1) * grad
        is_v = beta2 * is_v + (1.0 - beta2) * grad * grad
        m_hat = is_m / (1.0 - beta1 ** t)
        v_hat = is_v / (1.0 - beta2 ** t)

        h_new = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        h = h_new

        if t % 80 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 80
            if stall > 4000:
                lr *= 0.5
                is_m *= 0.5
                is_v *= 0.5
                stall = 0
                if lr < 1e-12:
                    break

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def nesterov_optimize(h, n, tau, time_limit, lr=None, top_k=12):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.3 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0
    cached_shifts = None
    recompute_int = 30

    for it in range(2000000):
        if time.time() - start > time_limit:
            break

        if cached_shifts is None or it % recompute_int == 0:
            corr = compute_correlation(h)
            cached_shifts = top_k_shifts(corr, top_k)

        mu = min((t - 1.0) / (t + 2.0), 0.85)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        vals = compute_shift_values(y, cached_shifts)
        shifted = np.clip(tau * (vals - vals.max()), -50.0, 0.0)
        exp_c = np.exp(shifted)
        w = exp_c / exp_c.sum()
        grad = grad_from_shifts(y, cached_shifts, w, dx)

        h_new = project_to_feasible(y - lr * grad, n)

        if it % 50 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 50
            if stall > 3000:
                lr *= 0.6
                stall = 0
                if lr < 1e-15:
                    break

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def binary_local_search(h, n, time_limit, max_stall=300):
