def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    tc = t * corr
    m = np.max(tc)
    return (m + np.log(np.sum(np.exp(tc - m)))) / t * dx

def slsqp_optimize(h_start, n, t, maxiter):
    def obj(h):
        return soft_obj(h, n, t)
    def con_fun(h):
        return np.sum(h) - n / 2.0
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_feasible(res.x, n)
        return h_new
    except Exception:
        return make_feasible(h_start, n)

def coord_descent(h, n, max_rounds=40, time_limit=None, start_time=None,
                  candidate_delta=[0.01, 0.02, 0.05, 0.1, 0.15, 0.2, 0.3, 0.4]):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        
        improved = False
        order = np.random.permutation(n)
        
        for j in order:
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 20:
                    return best_h, best_c
            
            old = best_h[j]
            best_val = old
            best_obj = best_c
            
            # Generate candidates
            candidates = [0.0, 1.0, 0.5]
            for d in candidate_delta:
                candidates.append(np.clip(old + d, 0, 1))
                candidates.append(np.clip(old - d, 0, 1))
            
            for cv in candidates:
                cv = float(cv)
                if abs(cv - old) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-13:
                    best_obj = c
                    best_val = cv
            
            if best_val != old and best_obj < best_c - 1e-13:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c = best_obj
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c
