def _obj_lse(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)

    # Phase 8: Final pair search squeeze
    if remaining() > 15:
        best_h, best_c5 = local_search_pair(best_h, len(best_h), 5000, seed=999, time_limit=remaining()-3, start_time=start_time)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
