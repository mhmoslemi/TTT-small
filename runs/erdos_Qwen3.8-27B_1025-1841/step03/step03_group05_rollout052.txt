def _obj(h):

    n = _OBJ['n']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    # Use top-k softmax for tighter approximation
    topk = _OBJ['topk']
    idx = np.argsort(c)[::-1][:topk]
    ct = c[idx]
    cmt = mc
    exps = np.exp(T * (ct - cmt))
    logsum = np.log(np.sum(exps))
    return (mc + logsum / T) * dx

def _obj_grad(h):
    T = _OBJ['T']
    n = _OBJ['n']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    topk = _OBJ['topk']

    exps = np.exp(T * (ct - mc))
    w = exps / np.sum(exps)

    for i, wi in zip(idx, w):
        if wi < 1e-14:
            break
        g += wi * _subgrad_one(h, i - (n-1))
    return g * dx

def _con(h):
    return np.sum(h) - _OBJ['n'] / 2.0

def _con_jac(h):
    return np.ones(_OBJ['n'])

    n = len(h)
    _OBJ['T'], _OBJ['n'], _OBJ['topk'] = T, n, topk
    try:
        r = minimize(_obj, h, jac=_obj_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _con, 'jac': _con_jac}],
                     options={'maxiter': max_iter, 'ftol': 1e-12})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

    bc = compute_c5(x)
    bh = x.copy()

        tot = np.sum(exps)
        if tot < 1e-300:

        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)

    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def _interp(h, nn):
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

# SLSQP globals
_SG = {}

def _sobj(h):
    T, n, tk = _SG['T'], _SG['n'], _SG['tk']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    idx = np.argsort(c)[::-1][:tk]
    ct = c[idx]
    e = np.exp(T * (ct - mc))
    return (mc + np.log(np.sum(e)) / T) * dx

def _sgrad(h):
    T, n, tk = _SG['T'], _SG['n'], _SG['tk']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    idx = np.argsort(c)[::-1][:tk]
    e = np.exp(T * (c[idx] - mc))
    w = e / np.sum(e)
    g = np.zeros(n)
    for i, wi in zip(idx, w):
        if wi < 1e-14:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def _scon(h):
    return np.sum(h) - _SG['n'] / 2.0

def _sjac(h):
    return np.ones(_SG['n'])

def slsqp(h, T=5.0, mi=600, tk=40):
    n = len(h)
    _SG['T'], _SG['n'], _SG['tk'] = T, n, tk
    try:
        r = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': mi, 'ftol': 1e-13})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h
