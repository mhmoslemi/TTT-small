    def f(v):
        return np.max(a + b * v)

    for _ in range(60):
        m1 = L + (U - L) / 3
        m2 = U - (U - L) / 3
        if f(m1) < f(m2):
            U = m2
        else:
            L = m1
    v_best = (L + U) / 2
    # Also check boundaries
    candidates = [L, U, v_best]
    best_v = min(candidates, key=lambda v: f(v))
    return best_v

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_descent(h, M, n_passes, time_budget, start_time, seed=0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    rng = np.random.RandomState(seed)
    stale = 0

    for p in range(n_passes):
        if time.time() - start_time > time_budget - 20:
            break
        top_idx = np.argpartition(corr, -M)[-M:]
        improved = False
        order = rng.permutation(N)
        for j in order:
            active_vals = corr[top_idx]
            grads = np.zeros(M)
            for i in range(M):
                idx = top_idx[i] - j
                if 0 <= idx < N:
                    grads[i] = 1.0 - 2.0 * h[idx]
            if np.all(np.abs(grads) < 1e-14):
                continue
            L = -h[j]
            U = 1.0 - h[j]
            if U - L < 1e-14:
                continue
            a2, b2 = L, U
            for _ in range(35):
                m1 = a2 + (b2 - a2) / 3.0
                m2 = b2 - (b2 - a2) / 3.0
                if np.max(active_vals + m1 * grads) < np.max(active_vals + m2 * grads):
                    b2 = m2
                else:
                    a2 = m1
            vs = (a2 + b2) / 2.0
            fL = np.max(active_vals + L * grads)
            fU = np.max(active_vals + U * grads)
            fS = np.max(active_vals + vs * grads)
            if fL <= fU and fL <= fS:
                vs = L
            elif fU <= fL and fU <= fS:
                vs = U
            new_val = h[j] + vs
            if abs(vs) < 1e-14:
                continue
            cur_max = float(np.max(corr))
            delta = new_val - h[j]
            m_start = j
            m_end = min(j + N, 2 * N - 1)
            idx_arr = np.arange(m_start, m_end) - j
            corr[m_start:m_end] += delta * (1.0 - 2.0 * h[idx_arr])
            h[j] = new_val
            new_full_max = float(np.max(corr))
            if new_full_max < cur_max - 1e-12:
                improved = True
                if new_full_max * dx < best_c5:
                    best_c5 = new_full_max * dx
            else:
                corr[m_start:m_end] -= delta * (1.0 - 2.0 * h[idx_arr])
                h[j] = h[j] - delta
        if not improved:
            stale += 1
            if stale > 10:
                break
        else:
            stale = 0
    return h, corr, best_c5
