```python
def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    for _ in range(100):
        # Find tau such that sum(clip(h - tau, 0, 1)) = target
        # Bisection
        lo, hi = h.min() - 1, h.max() + 1
        for _ in range(60):
            mid = (lo + hi) / 2
            s = np.sum(np.clip(h - mid, 0, 1))
            if s > target:
                lo = mid
            else:
                hi = mid
        tau = (lo + hi) / 2
        h_new = np.clip(h - tau, 0, 1)
        if abs(np.sum(h_new) - target) < 1e-10:
            break
        # If still not close, do a correction
        diff = target - np.sum(h_new)
        mask = (h_new > 1e-10) & (h_new < 1 - 1e-10)
        if np.sum(mask) > 0:
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0, 1)
    return h_new
```

```python
def compute_gradient(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_t = np.exp(t * shifted)
    w = exp_t / np.sum(exp_t)  # length 2n-1
    f_val = (max_c + np.log(np.sum(exp_t)) / t) * dx
    
    # Gradient
    grad = np.zeros(n)
    # For each lag, compute contribution
    # lag = k - (n-1) for k = 0, ..., 2n-2
    for k in range(2*n-1):
        wl = w[k]
        if wl < 1e-15:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += wl * (1 - 2*h)
        elif lag > 0:
            # C(lag) = sum_{n=0}^{n-1-lag} h[n+lag]*(1-h[n])
            # d/dh[j] = (1-h[j-lag])*I(lag<=j<=n-1) - h[j+lag]*I(0<=j<=n-1-lag)
            grad[lag:] += wl * (1 - h[:-lag])
            grad[:-lag] += wl * (-h[lag:])
        else:
            m = -lag
            # C(-m) = sum_{n=m}^{n-1} h[n-m]*(1-h[n])

            grad[:n-m] += wl * (1 - h[m:])
            grad[m:] += wl * (-h[:-m])
    
    grad *= dx
    return f_val, grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    """Euclidean projection onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Water-filling: find tau such that sum(clip(h-tau, 0, 1)) = target
    lo = h.min() - 0.1
    hi = h.max() + 0.1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    return np.clip(h - tau, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max_and_grad(h, n, t):
    """Compute smoothed max objective and gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exp_vals = np.exp(t * shifted)
    exp_sum = np.sum(exp_vals)
    w = exp_vals / exp_sum  # softmax weights over lags
    f_val = (max_c + np.log(exp_sum) / t) * dx
    
    # Compute gradient
    grad = np.zeros(n)
    # Loop over lags
    for k in range(2 * n - 1):
        wl = w[k]
        if wl < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wl * (1.0 - 2.0 * h)
        elif lag > 0:
            # C(lag) = sum_{n=0}^{n-1-lag} h[n+lag]*(1-h[n])
            # d/dh[j]: (1-h[j-lag]) for j in [lag, n-1], -h[j+lag] for j in [0, n-1-lag]
            grad[lag:] += wl * (1.0 - h[:-lag])
            grad[:n-lag] += wl * (-h[lag:])
        else:
            m = -lag
            # C(-m) = sum_{n=m}^{n-1} h[n-m]*(1-h[n])
            # d/dh[j]: (1-h[j+m]) for j in [0, n-1-m], -h[j-m] for j in [m, n-1]
            grad[:n-m] += wl * (1.0 - h[m:])
            grad[m:] += wl * (-h[:-m])
    
    grad *= dx
    return f_val, grad

def projected_grad_descent(h, n, t_schedule, max_iter, time_limit, start_time):
    """Projected gradient descent with backtracking line search."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.1
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        f_val, grad = smooth_max_and_grad(h, n, t_schedule[it])
        
        # Backtracking line search
        alpha = step
        for bt in range(20):
            h_trial = h - alpha * grad
            h_trial = project_feasible(h_trial, n)
            f_trial, _ = smooth_max_and_grad(h_trial, n, t_schedule[it])
            if f_trial < f_val - 1e-10 * alpha * np.sum(grad**2):
                break
            alpha *= 0.5
        
        h = project_feasible(h - alpha * grad, n)
        
        # Check actual C5
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
        
        # Adjust step size
        if f_trial < f_val:
            step = min(step * 1.1, 1.0)
        else:
            step = max(step * 0.5, 0.001)
    
    return best_h, best_c5
