            current = best_new_max

            zeros.remove(best_j)
            ones.append(best_j)
            zeros.append(best_i)
            improved = True

    return h, current

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        mask = h < 1.0 - 1e-14 if diff > 0 else h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def softmax_obj(h, n, T):
    corr = compute_corr(h)
    mc = corr.max()
    s = np.clip(T * (corr - mc), -70, 70)
    return (mc + np.log(np.sum(np.exp(s))) / T) * (2.0 / n)

def lbgfb_opt(h0, n, T, mu=500.0, max_iter=250):
    def f(h):
        return softmax_obj(h, n, T) + mu * (h.sum() - n / 2.0) ** 2
    res = minimize(f, h0, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def binarize_best(h, n):
    """Binarize while keeping exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:n // 2]] = 1.0
    return out

def exhaustive_binary_search(h_bin, n, max_passes=100, window=None):
    h = h_bin.astype(float).copy()
    corr = compute_corr(h)
    current = float(corr.max())
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    n_o, n_z = len(ones), len(zeros)
    for rnd in range(max_passes):
        improved = False
        best_delta = 1e-15
        best_i, best_j = -1, -1
        best_new_max = current
        best_new_corr = corr
        if window is None:
            window = n
        w2 = window * 2
        for ii in range(n_o):
            i = ones[ii]
            for jj in range(n_z):
                j = zeros[jj]
                d = min(abs(i - j), n - abs(i - j))
                if window < n and d > window:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                nc = compute_corr(h)
                nm = float(nc.max())
                h[i] = 1.0
                h[j] = 0.0
                delta = current - nm
                if delta > best_delta:
                    best_delta = delta
                    best_i, best_j = i, j
                    best_new_max = nm
                    best_new_corr = nc
        if best_delta > 1e-14:
            h[best_i] = 0.0
            h[best_j] = 1.0
            corr = best_new_corr
            current = best_new_max
            ones.remove(best_i)
            zeros.remove(best_j)
            ones.append(best_j)
            zeros.append(best_i)
            improved = True
        if not improved:
            break
    return h, current

def anneal_binary(h_bin, n, rng, n_iter=50000, T0=0.01, T1=1e-5):
    h = h_bin.astype(float).copy()
    corr = compute_corr(h)
    current = float(corr.max())
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    best_h = h.copy()
    best_val = current
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(n_iter - 1, 1))
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i, j = ones[ii], zeros[jj]
        h[i] = 0.0
        h[j] = 1.0
        nc = compute_corr(h)
        nm = float(nc.max())
        delta = nm - current
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current = nm
            corr = nc
            ones[ii] = j
            zeros[jj] = i
            if nm < best_val:
                best_val = nm
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_val

def structured_init(n, freqs, phases, rng):
    """Generate sinusoidal/sturmian-like binary sequences."""
    cands = []
    x = np.linspace(0, 2, n, endpoint=False)
    for f in freqs:
        for ph in phases:
            s = np.sin(2 * np.pi * f * x / 2.0 + ph)
            b = (s > 0).astype(float)
            b = make_feasible(b, n)
            cands.append(b)
    # Sturmian-like
    for alpha in [0.61803, 0.5, 0.4, 0.7, 0.3, 0.25, 0.61803 / 2]:
        for ph in [0.0, 0.25, 0.5, 0.75]:
            idx = np.arange(n)
            vals = np.mod(idx * alpha + ph, 1.0)
            b = (vals < 0.5).astype(float)
            b = make_feasible(b, n)
            cands.append(b)
    # Random
    for _ in range(4):
        b = (rng.rand(n) < 0.5).astype(float)
        b = make_feasible(b, n)
        cands.append(b)
    return cands

def slsqp_polish(h_start, n, T, max_iter=100):
    def obj(h):
        return softmax_obj(h, n, T)
    def con(h):
        return h.sum() - n / 2.0
    def jac(h):
        return np.ones(n)
    res = minimize(obj, h_start, method='SLSQP', bounds=[(0, 1)] * n,
                   constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def coordinate_polish(h, n, max_rounds=5, rng=None):
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for rnd in range(max_rounds):
        improved = False
        order = np.arange(n)
        if rng is not None:
            rng.shuffle(order)
        for j in order:
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < compute_c5(h) - 1e-13:
                    h = tr
                    improved = True
                    break
            bv = 0.0 if h[j] > 0.5 else 1.0
            if abs(bv - h[j]) > 1e-9:
                tr = h.copy()
                tr[j] = bv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < compute_c5(h) - 1e-13:
                    h = tr
                    improved = True
        if not improved:
            break
    return h
