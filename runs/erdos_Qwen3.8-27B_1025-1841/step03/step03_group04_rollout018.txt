        cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
        try:
            res = minimize(slsqp_obj, h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-14})
            c5 = compute_c5(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = make_feasible(res.x, n)
        except Exception:
            pass
        h = best_h.copy()
    
    # Phase 2: Many CD passes (main workhorse)
    prev_mc = compute_c5(h) / (2.0 / n)
    stagnant = 0
    while remaining() > 150:
        h, mc = cd_pass(h, n, n_grid=50)
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        if mc < prev_mc - 1e-10:
            prev_mc = mc
            stagnant = 0
        else:
            stagnant += 1
        if stagnant > 200:
            break
    
    # Phase 3: SLSQP polish with higher T
    if remaining() > 100:
        _G_N = n
        _G_T = 500.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
        try:
            res = minimize(slsqp_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-15})
            c5 = compute_c5(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = make_feasible(res.x, n)
        except Exception:
            pass
    
    # Phase 4: Finer CD passes
    h = best_h.copy()
    while remaining() > 80:
        h, mc = cd_pass(h, n, n_grid=100)
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 5: Another SLSQP round
    if remaining() > 80:
        _G_N = n
        _G_T = 800.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
        try:
            res = minimize(slsqp_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-15})
            c5 = compute_c5(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = make_feasible(res.x, n)
        except Exception:
            pass
    
    # Phase 6: Random restarts
    for restart in range(4):
        if remaining() < 100:
            break
        scale = 0.01 + 0.01 * restart
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        for _ in range(80):
            if remaining() < 80:
                break
            h_try, mc = cd_pass(h_try, n, n_grid=50)
            h_try = make_feasible(h_try, n)
        c5 = compute_c5(h_try)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_try.copy()
    
    # Phase 7: Final ultra-fine CD
    if remaining() > 30:
        h = best_h.copy()
        for _ in range(10):
            if remaining() < 20:
                break
            h, mc = cd_pass(h, n, n_grid=200)
            h = make_feasible(h, n)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
