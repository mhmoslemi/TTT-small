def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_lag(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def slsqp_polish(h, T, max_iter):
    n = len(h)
    dx = 2.0 / n
    
    def obj(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx
    
    def jac(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))

        pass
    return h

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_lag(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_opt(h, n_iter, T0, T1, lr0, lr1, km, kn, ck, damp):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    t_val = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    stale = 0
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** frac
        lr = lr0 + (lr1 - lr0) * frac
        top_k = max(1, int(km + (kn - km) * frac))
        
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        
        if T < 0.05:
            ix = int(np.argmax(corr))
            g = subgrad_lag(z, ix - (n - 1)) * dx
        else:
            ex = np.exp(T * (corr - mc))
            tt = np.sum(ex)
            if tt < 1e-300:
                ix = int(np.argmax(corr))
                g = subgrad_lag(z, ix - (n - 1)) * dx
            else:
                w = ex / tt
                ii = np.argsort(w)[::-1][:top_k]
                g = np.zeros(n)
                for j in ii:
                    if w[j] < 1e-14:
                        break
                    g += w[j] * subgrad_lag(z, j - (n - 1))
                g *= dx
        
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
        mom = damp * (t_val - 1.0) / tn
        z = xn + mom * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        t_val = tn
        
        if (it + 1) % ck == 0:
            c = compute_c5(x)
            if c < best_c - 1e-13:
                best_c = c
                best_h = x.copy()
                stale = 0
            else:
                stale += 1
            if stale > 6:
                z = project_c(z + np.random.randn(n) * 0.003, n)
                x = z.copy()
                t_val = 1.0
                stale = 0
    
    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

def cycle_opt(h, n_iter, nl, lr0, lr1, ck):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    best_c = compute_c5(x)
    best_h = x.copy()
    
    for it in range(n_iter):
        lr = lr0 + (lr1 - lr0) * (it / max(n_iter - 1, 1))
        corr = np.correlate(x, 1.0 - x, mode='full')
        top = np.argsort(corr)[::-1][:nl]
        
        for li in top:
            sh = li - (n - 1)
            g = subgrad_lag(x, sh) * dx
            x = project_c(x - lr * g, n)
        
        if (it + 1) % ck == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    
    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

def _sobj(v, T, n):
    dx = 2.0 / n
    c = np.correlate(v, 1.0 - v, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _sjac(v, T, n):
    dx = 2.0 / n
    c = np.correlate(v, 1.0 - v, mode='full')
    mc = np.max(c)
    ex = np.exp(T * (c - mc))
    w = ex / np.sum(ex)
    g = np.zeros(n)
    for j in range(len(c)):
        if w[j] > 1e-10:
            g += w[j] * subgrad_lag(v, j - (n - 1))
    return g * dx

def _sfun(v, n):
    return np.sum(v) - n / 2.0

def _sjacv(v, n):
    return np.ones(n)

def slsqp_run(h, T, mi):
    n = len(h)
    try:
        r = minimize(lambda v: _sobj(v, T, n), h, method='SLSQP',
                     jac=lambda v: _sjac(v, T, n),
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': lambda v: _sfun(v, n),
                                    'jac': lambda v: _sjacv(v, n)}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        if r.x is not None and np.all(np.isfinite(r.x)):
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h
```

- Take a small step: increase coords with negative gradient, decrease coords with positive gradient
- Project

This is essentially what FISTA does, but without the momentum and soft-max. Let me see if cycling through lags helps.

Pick 10 random coordinates, try all 2^10 = 1024 binary assignments (or a subset), pick the one that reduces C5 most. This is very local but can find improvements that gradient methods miss.

But 1024 evaluations × O(n²) each = too slow for n=960. Let me use 5 coordinates: 2^5 = 32 evaluations. Still 32 × 1ms = 32ms per block. Over 1000 random blocks: 32 seconds. Feasible.

import numpy as np
from scipy.optimize import minimize

_SP = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_lag(h, shift):
    n = len(h)
    s = abs(int(shift))
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_opt(h, n_iter, T0, T1, lr0, lr1, km, kn, ck, damp):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    t_val = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    stale = 0
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** frac
        lr = lr0 + (lr1 - lr0) * frac
        top_k = max(1, int(km + (kn - km) * frac))
        
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        
        if T < 0.05:
            ix = int(np.argmax(corr))
            g = subgrad_lag(z, ix - (n - 1)) * dx
        else:
            ex = np.exp(T * (corr - mc))
            tt = np.sum(ex)
            if tt < 1e-300:
                ix = int(np.argmax(corr))
                g = subgrad_lag(z, ix - (n - 1)) * dx
            else:
                w = ex / tt
                ii = np.argsort(w)[::-1][:top_k]
                g = np.zeros(n)

                    if w[j] < 1e-14:
                        break
                    g += w[j] * subgrad_lag(z, int(j) - (n - 1))

                best_h = x.copy()
                stale = 0

                x = z.copy()

def cycle_opt(h, n_iter, nl, lr0, lr1, ck):

def _sjac_slsqp(v):

    n = _SP['n']

            g += w[j] * subgrad_lag(v, j - (n - 1))
    return g * dx

    return np.sum(v) - _SP['n'] / 2.0

    _SP['T'] = T
    _SP['n'] = n
    try:
