    dx = 2.0 / n
    N = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(x, N)
    G = np.fft.rfft(1.0 - x, N)
    corr = np.fft.irfft(H * np.conj(G), N).real
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_eq(x):
    return np.sum(x) - len(x) / 2.0

def con_eq_jac(x):
    return np.ones(len(x))

def continuous_slsqp(h, n, T, maxiter, time_limit):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
    
    def obj(x):
        return lsq_smooth_obj(x, n, T)
    
    t0 = time.time()
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-12,
                            'disp': False,
                            'callback': None})
    return res.x

    global initial_h_values
    start_time = time.time()
    rng = np.random.RandomState(seed)
    elapsed = lambda: time.time() - start_time
    remaining = lambda: budget_s - elapsed()
    
    def time_ok(minimum):
        return remaining() > minimum
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    b_init = to_binary(h_init, n0)
    best_c5 = compute_c5_fast(b_init)
    best_h = b_init.copy()
    
    # Phase 1: Extended SA on initial binary (swap + reverse + block)
    if time_ok(200):
        b1, c5_1 = sa_mixed(b_init.copy(), n0, 180, rng, 0.005, 1e-5, 300000)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = b1.copy()
    
    # Phase 2: Greedy improvement
    if time_ok(100):
        b2, c5_2 = greedy_improve(best_h.copy(), n0, min(80, remaining() - 10), rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()
    
    # Phase 3: Continuous SLSQP polish
    if time_ok(200):
        try:
            x_res = continuous_slsqp(best_h, n0, 100.0, 500, remaining() - 210)
            x_res = make_feasible(x_res, n0)
            c5_res = compute_c5_fast(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass

                    c5_res = compute_c5_fast(x_res)
