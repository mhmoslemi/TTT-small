For the projection onto {h: 0 ≤ h ≤ 1, Σh = N/2}:
This is a standard problem. Given a vector v, find the closest point in the constraint set.
Algorithm: binary search on λ to find the value such that Σ clip(v[i] - λ, 0, 1) = N/2.

Let me code this up:

```python
def project_onto_constraint(v, target_sum):
    """Project v onto {h: 0 <= h <= 1, sum(h) = target_sum}"""
    # Binary search on lambda
    lo, hi = v.min() - 1, v.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0, 1))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(v - (lo + hi) / 2, 0, 1)
```

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_box_sum(v, target):
    """Project v onto {x: 0<=x<=1, sum(x)=target} via bisection on offset."""
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5*(lo+hi), 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_onto_box_sum(h_new, new_n / 2.0)

def soft_max_val(h, t):
    """Smooth approximation of max of correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    return m + np.log(np.sum(np.exp(t * (corr - m)))) / t

def soft_max_grad(h, t):
    """Gradient of soft_max_val with respect to h."""
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = corr.max()
    w = np.exp(t * (corr - m))
    w /= w.sum()
    
    grad = np.zeros(N)
    # For each lag contribution
    # corr[idx] = sum_n h[n + lag] * (1 - h[n]) where lag = idx - (N-1)
    # d corr[idx] / d h[j]:
    #   From h[n+lag]: when n+lag = j, i.e., n = j-lag. Contribution: (1-h[j-lag]) if valid
    #   From (1-h[n]): when n = j. Contribution: -h[j+lag] if valid
    
    # Vectorized: for lag >= 0:
    #   grad[lag:] += w[idx] * (1-h[:N-lag])
    #   grad[:N-lag] -= w[idx] * h[lag:]
    # For lag < 0 (lag_abs = -lag):
    #   grad[:N-lag_abs] += w[idx] * (1-h[lag_abs:])
    #   grad[lag_abs:] -= w[idx] * h[:N-lag_abs]
    
    for idx in range(len(corr)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lag = idx - (N - 1)
        if lag >= 0:
            nv = N - lag
            grad[lag:] += wi * (1.0 - h[:nv])
            grad[:nv] -= wi * h[lag:]
        else:
            la = -lag
            nv = N - la
            grad[:nv] += wi * (1.0 - h[la:])
            grad[la:] -= wi * h[:nv]
    
    return grad * (2.0 / N)  # dx factor

def run_gradient_descent(h, start_time, budget, target, max_iters=3000):

            no_improve = 0

def coord_descent(h, start_time, budget):
    """Paired coordinate descent."""
    N = len(h)
    h = h.copy()
    best_c = compute_c5(h)
    target = N / 2.0
    rng = np.random.RandomState(123)
    
    for rnd in range(200):
        if time.time() - start_time > budget - 15:
            break
        improved = False
        for _ in range(N // 2):
            if time.time() - start_time > budget - 15:
                return h, best_c
            i, j = rng.randint(0, N), rng.randint(0, N)
            if i == j:
                continue
            # Try swapping/moving mass between i and j
            old_i, old_j = h[i], h[j]
            for delta in [0.01, 0.02, 0.05, 0.1, -0.01, -0.02, -0.05, -0.1]:

                nj = np.clip(old_j - delta, 0, 1)

                h[i], h[j] = ni, nj
                c = compute_c5(h)
                if c < best_c - 1e-12:
                    best_c = c
                    improved = True
                    break
                h[i], h[j] = old_i, old_j
        if not improved:
            break
    
    return h, best_c

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_box_sum(v, target):
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(v - 0.5 * (lo + hi), 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_onto_box_sum(h_new, new_n / 2.0)

def soft_max_grad(h, t):
    N = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')

        return np.zeros(N)
