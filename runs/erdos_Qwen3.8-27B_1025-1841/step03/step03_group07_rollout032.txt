import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_overlaps_fft(h):
    """Compute circular overlaps using FFT."""
    n = len(h)
    H = np.fft.rfft(h, n=2*n)
    autocorr = np.fft.irfft(H * np.conj(H), n=2*n)[:n]
    return n/2.0 - autocorr

def binary_local_search(h_bin, n, time_limit, max_passes=100):
    """Greedy bit-flip local search for binary h."""
    h = h_bin.copy()
    start = time.time()
    overlap = compute_overlaps_fft(h)
    best_max = overlap.max()
    
    for it in range(max_passes * n):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            delta = (1.0 - np.roll(h, -i)) - np.roll(h, i)
            if h[i] == 0:
                new_max = (overlap + delta).max()
            else:
                new_max = (overlap - delta).max()
            if new_max < best_max - 1e-10:
                # Apply flip
                if h[i] == 0:
                    overlap += delta
                    h[i] = 1
                else:
                    overlap -= delta
                    h[i] = 0
                best_max = new_max
                improved = True
        if not improved:
            break
    
    return h, best_max * (2.0 / n)

    start = time.time()
    h = h.copy()
    N = 2 * n  # FFT length
    
    # Precompute FFT of h
    H = np.fft.rfft(h, n=N)
    # FFT of 1-h
    one_minus_h = 1.0 - h
    OH = np.fft.rfft(one_minus_h, n=N)
    
    # Precompute twiddle factors
    twiddle = np.exp(-2j * np.pi * np.arange(N//2+1) / N)  # rfft only goes to N//2
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    stall = 0
    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        
        # Compute current full correlation
        corr = np.fft.irfft(H * np.conj(OH), n=N)[:2*n-1]
        current_max = corr.max()
        
        if current_max / n * 2 < best_c5:
            best_c5 = current_max * 2.0 / n
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 50 * n:  # no improvement for many flips
            break
        
        # Try all flips
        best_improvement = 0
        best_flip = -1
        for i in range(n):
            # Flip h[i]
            if h[i] == 0:
                delta_h = 1.0  # h[i] goes from 0 to 1
            else:
                delta_h = -1.0  # h[i] goes from 1 to 0
            
            # Update FFTs
            new_H = H + delta_h * twiddle
            new_OH = OH - delta_h * twiddle
            
            new_corr = np.fft.irfft(new_H * np.conj(new_OH), n=N)[:2*n-1]
            new_max = new_corr.max()

                best_improvement = improvement
                best_flip = i
        
        if best_flip < 0:
            break  # No improving flip found
        
        # Apply the best flip
        i = best_flip
        if h[i] == 0:
            delta_h = 1.0
        else:
            delta_h = -1.0
        h[i] += delta_h
        H += delta_h * twiddle
        OH -= delta_h * twiddle

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def nesterov_opt(h, n, tau, time_limit, lr, top_k=10):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        mu = min((t - 1.0) / (t + 2.0), 0.9)
        y = np.clip(h + mu * (h - h_prev), 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = corr.max()
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = p[ii]
            k = top_idx[ii]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - y[j0:j1] if sh <= 0 else (1.0 - y[j0 - sh:j1 - sh]))
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * y[j0b + sh:j1b + sh] if sh >= 0 else pk * y[j0b:j1b]
        grad *= dx
        h_new = project_to_feasible(y - lr * grad, n)
        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 2000:
                lr *= 0.65
                stall = 0
                if lr < 1e-15:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def subgrad_opt(h, n, time_limit, lr, top_k=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-16:
                break
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = w[ii]
            k = top_idx[ii]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh]) if sh <= 0 else pk * (1.0 - h[j0:j1])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh] if sh >= 0 else pk * h[j0b:j1b]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5

def binary_greedy(h_init, n, time_limit):
    start = time.time()
    h = (h_init[:n] > 0.5).astype(float) if len(h_init) >= n else (h_init > 0.5).astype(float)
    if len(h) < n:
        h = np.zeros(n)
        h[:len(h_init)] = (h_init > 0.5).astype(float)
    # Fix sum
    cur_sum = h.sum()
    target = n // 2
    if cur_sum > target:
        idx = np.where(h == 1)[0]
        np.random.shuffle(idx)
        for i in idx:
            h[i] = 0
            if h.sum() == target:
                break
    elif cur_sum < target:
        idx = np.where(h == 0)[0]
        np.random.shuffle(idx)
        for i in idx:
            h[i] = 1
            if h.sum() == target:
                break
    best_c5 = compute_c5(h)
    best_h = h.copy()
    N = 2 * n
    # Precompute twiddle factors for each position
    # W[i, k] = exp(-2*pi*i*k/N) for k = 0...N//2
    ks = np.arange(N // 2 + 1)
    while time.time() - start < time_limit:
        H = np.fft.rfft(h, n=N)
        OH = np.fft.rfft(1.0 - h, n=N)
        corr = np.fft.irfft(H * np.conj(OH), n=N)[:2 * n - 1]
        current_max = corr.max()
        c5_now = current_max * 2.0 / n
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
        best_improve = 1e-12
        best_i = -1
        best_dh = 0.0
        for i in range(n):
            dh = 1.0 if h[i] == 0 else -1.0
            tw = np.exp(-2j * np.pi * ks * i / N)
            new_H = H + dh * tw
            new_OH = OH - dh * tw
            new_corr = np.fft.irfft(new_H * np.conj(new_OH), n=N)[:2 * n - 1]
            new_max = new_corr.max()
            improve = current_max - new_max
            if improve > best_improve:
                best_improve = improve
                best_i = i
                best_dh = dh
        if best_i < 0:
            break
        h[best_i] += best_dh
    return best_h, best_c5

def get_rem(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))
