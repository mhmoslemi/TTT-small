import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    max_s = np.max(shifted)
    if max_s > 30:
        shifted -= max_s
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = (mc + max_s + np.log(sum_exp) / t) * dx
    w = t * exp_s / sum_exp

        else:
            a = -m
            grad[a:] += wkm * (1.0 - h[:n-a])
            grad[:n-a] -= wkm * h[a:]
    grad *= dx
    return obj, grad

def lbfgs_optimize(h_start, n, t, maxiter, time_limit, start_time):
    def obj_grad(h):
        o, g = soft_obj_grad(h, t)
        return o, g
    try:
        if time.time() - start_time > time_limit:
            return h_start
        res = minimize(obj_grad, h_start, method='L-BFGS-B',
                      jac=True, bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-14,
                               'gtol': 1e-10, 'maxls': 50})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def pairwise_swap(h, n, n_swaps=200, time_limit=1300, start_time=None):

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    obj = (mc + np.log(sum_exp) / t) * dx
    w = t * exp_s / sum_exp
    grad = np.zeros(n)
    for m in range(-(n - 1), n):
        wkm = w[m + n - 1]
        if wkm < 1e-15:
            continue
        if m >= 0:
            grad[:n - m] += wkm * (1.0 - h[m:])
            grad[m:] -= wkm * h[:n - m]
        else:
            a = -m
            grad[a:] += wkm * (1.0 - h[:n - a])
            grad[:n - a] -= wkm * h[a:]
    grad *= dx
    return obj, grad

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def soft_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Numerical stability
    ms = np.max(shifted)
    if ms > 50:
        shifted -= ms
        mc2 = mc + ms
    else:
        mc2 = mc
    exp_s = np.exp(shifted)
    se = np.sum(exp_s)
    obj = (mc2 + np.log(se) / t) * dx
    w = t * exp_s / se
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        wk = w[m + n - 1]
        if wk < 1e-15:
            continue
        if m >= 0:
            grad[:n-m] += wk * (1.0 - h[m:])
            grad[m:] -= wk * h[:n-m]
        else:
            a = -m
            grad[a:] += wk * (1.0 - h[:n-a])
            grad[:n-a] -= wk * h[a:]
    grad *= dx
    return obj, grad

def lbfgs(h0, n, t, maxiter, dl):
    def og(h):
        return soft_obj_grad(h, t)
    try:
        if time.time() > dl:
            return h0
        r = minimize(og, h0, method='L-BFGS-B', jac=True,
                     bounds=[(0.0, 1.0)]*n,
                     options={'maxiter': maxiter, 'ftol': 1e-15,
                              'gtol': 1e-12, 'maxls': 40})
        return make_feasible(r.x, n)
    except:
        return make_feasible(h0, n)
