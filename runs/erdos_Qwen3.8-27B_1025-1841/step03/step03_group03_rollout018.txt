            no_improve_count += 1
    
    return h, best_c, c_full

def top_k_balanced_cd(h, c_full, n, time_limit, start_time, top_k=3):
    """CD that balances the top-k lags instead of just the max"""
    best_c = float(np.max(c_full))
    
    while time.time() - start_time < time_limit:
        # Get top-k lags
        top_indices = np.argsort(c_full)[::-1][:top_k]
        # Weighted gradient: average of gradients of top-k lags
        grad = np.zeros(n)
        for idx in top_indices:
            k_star = int(idx)
            k_abs = abs(k_star - (n - 1))
            k_sign = 1 if k_star >= n - 1 else -1
            if k_abs == 0:
                grad += (1.0 - 2.0 * h)
            elif k_sign > 0:
                k = k_abs
                mask1 = np.arange(n) <= (n - 1 - k)
                mask2 = np.arange(n) >= k
                if np.any(mask1):
                    j_idx = np.where(mask1)[0]
                    grad[j_idx] += (1.0 - h[j_idx + k])
                if np.any(mask2):
                    j_idx = np.where(mask2)[0]
                    grad[j_idx] -= h[j_idx - k]
            else:
                k = k_abs
                mask1 = np.arange(n) >= k
                mask2 = np.arange(n) <= (n - 1 - k)
                if np.any(mask1):
                    j_idx = np.where(mask1)[0]
                    grad[j_idx] += (1.0 - h[j_idx - k])
                if np.any(mask2):
                    j_idx = np.where(mask2)[0]
                    grad[j_idx] -= h[j_idx + k]

        j_down = int(np.argmax(grad))

            break

            break

        old_jd = h[j_down]

            new_jd = max(old_jd - step, 0.0)
            if new_ju <= old_ju and new_jd >= old_jd:

            apply_change(c_full, h, j_down, new_jd)

                best_step = step

            apply_change(c_full, h, j_up, old_ju)
