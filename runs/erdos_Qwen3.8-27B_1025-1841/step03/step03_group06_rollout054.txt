def _fft_corr(h, N):
    """Compute np.correlate(h, 1-h, 'full') via FFT."""
    n = len(h)
    b = 1.0 - h
    H = np.fft.fft(h, N)
    B = np.fft.fft(b, N)
    C = np.fft.ifft(H * np.conj(B)).real
    return np.concatenate([C[N-(n-1):N], C[0:n]])

def _fft_grad(h, w, N):

    return float(np.max(corr) * dx)

    old_x = np.linspace(0.0, 2.0, old_n)

import numpy as np
import time

def _fft_full_corr(h, N):
    n = len(h)
    b = 1.0 - h
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    C = np.fft.irfft(H * np.conj(B), N)
    return np.concatenate([C[N-(n-1):N], C[0:n]])

def _fft_corr_grad(h, w, N):
    n = len(h)
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    W = np.fft.rfft(np.concatenate([w[:n], np.zeros(N//2 - n + 1) if N//2 >= n else np.zeros(0)]), N)
    # Pad w to N for FFT
    w_pad = np.zeros(N)
    w_pad[:n] = w
    Wfull = np.fft.rfft(w_pad, N)
    term1 = np.fft.irfft(Wfull * B, N)[:n]
    term2 = np.fft.irfft(np.conj(Wfull) * H, N)[:n]
    return term1 - term2

def _project(h, target_sum):
    h = h.copy()
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)

def _make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return _make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return _make_feasible(h_new, new_n)

def _soft_objective(h, N, T):
    corr = _fft_full_corr(h, N)
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / len(h))

def _grad_step(h, N, T, alpha, beta, h_prev):
    n = len(h)
    dx = 2.0 / n
    corr = _fft_full_corr(h, N)
    mc = np.max(corr)
    shifted = corr - mc
    w = np.exp(T * shifted)
    w /= w.sum()
    grad = _fft_corr_grad(h, w, N)
    momentum = beta * (h - h_prev) if h_prev is not None else 0.0
    h_new = h + momentum - alpha * dx * grad
    h_new = _project(h_new, n / 2.0)
    return h_new, h

def _gradient_descent(h, N, T, alpha, beta, max_iter, time_limit_fn):
    n = len(h)
    h_prev = h.copy()
    cur_obj = _soft_objective(h, N, T)
    for it in range(max_iter):
        if time_limit_fn() < 40:
            break
        h_new, h_prev = _grad_step(h, N, T, alpha, beta, h_prev)
        new_obj = _soft_objective(h_new, N, T)
        if new_obj > cur_obj:
            alpha *= 0.5
            if alpha < 1e-8:
                break
            h_new, h_prev = _grad_step(h, N, T, alpha, beta, h_prev)
            new_obj = _soft_objective(h_new, N, T)
        else:
            if it % 20 == 0 and it > 0:
                alpha = min(alpha * 1.1, 5.0)
        h = h_new
        cur_obj = new_obj
    return h, alpha

def _coord_descent(h, max_rounds, time_limit_fn):
    n = len(h)
    best_c = _compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit_fn() < 15:
            break
        improved = False
        for j in range(n):
            if time_limit_fn() < 5:
                break
            old = h[j]
            best_c_local = best_c
            best_val = old
            cands = [0.0, 1.0, 0.5, max(0.0, old - 0.05), min(1.0, old + 0.05),
                     max(0.0, old - 0.01), min(1.0, old + 0.01)]
            for cand in cands:
                trial = h.copy()
                trial[j] = cand
                trial = _make_feasible(trial, n)
                c = _compute_c5(trial)
                if c < best_c_local - 1e-13:
                    best_c_local = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = _make_feasible(h, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return h, best_c

def _binary_snap(h, n):
    best = h.copy()
    best_c = _compute_c5(h)
    for j in range(n):
        for val in [0.0, 1.0]:
            trial = best.copy()
            trial[j] = val
            trial = _make_feasible(trial, n)
            c = _compute_c5(trial)
            if c < best_c - 1e-13:
                best = trial
                best_c = c
    return best, best_c
