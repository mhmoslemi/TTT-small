from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n"""
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def fista(h, n_iter=5000, T0=40.0, T1=0.1, lr=0.15, check=40):

    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.4 * f)
        
        # Soft-max gradient
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        w = exps / tot
        
        # Subgradient of the weighted sum
        g = np.zeros(n)
        # For each lag l, the subgradient of corr[n-1+l] w.r.t. h is:
        # (1-h shifted by -l) - (h shifted by l)
        # We only need the top few lags
        idx = np.argsort(w)[::-1][:60]
        for i in idx:
            wi = w[i]
            if wi < 1e-10:
                break
            l = i - (n - 1)
            if l >= 0:
                g[l:] += wi * (1.0 - z[:n-l])

        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc
