def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-10:
        return h
    # Distribute the difference
    if diff > 0:
        indices = np.argsort(h)  # smallest first (most room to increase)
    else:
        indices = np.argsort(-h)  # largest first (most room to decrease)
    remaining = abs(diff)
    per = remaining / n
    for i in indices:
        if remaining <= 1e-15:
            break
        if diff > 0:
            can_add = 1.0 - h[i]
            add = min(can_add, per, remaining)
            h[i] += add
            remaining -= add
        else:
            can_sub = h[i]
            sub = min(can_sub, per, remaining)
            h[i] -= sub
            remaining -= sub
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
