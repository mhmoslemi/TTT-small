def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = np.clip(g - theta, 0.0, 1.0)
    # Fix floating point: adjust to get exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        # Find indices where h is in (0,1) and adjust
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

def nesterov_adaptive(h, n, tau, time_limit, lr_init=0.1, top_k=15, tau_anneal=True):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    h_prev = h.copy()
    t_nes = 1.0
    stall = 0
    patience = 8000
    it = 0
    
    while time.time() - start < time_limit:
        if it % 200 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 200
                if stall > patience:
                    lr *= 0.6
                    h_prev = h.copy()
                    t_nes = 1.0
                    stall = 0
                    patience = max(2000, patience * 0.8)
                    if lr < 1e-12:
                        break
        
        mu = (t_nes - 1.0) / (t_nes + 2.0)
        mu = min(mu, 0.85)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        obj, grad, corr = smooth_objective_and_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        t_nes = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nes * t_nes))
        h_prev = h
        h = h_new
        it += 1
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def coordinate_descent_polish(h, n, time_limit, max_iter=50000):
    """Try pairwise swaps to reduce C5"""
    dx = 2.0 / n
    h = np.array(h, dtype=float)
    h = project_to_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    n_local = len(h)
    
    corr_full = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr_full)
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        if it % 200 == 0:
            corr_full = np.correlate(h, 1.0 - h, mode='full')
            c5_now = corr_full.max() * dx
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
            max_idx = np.argmax(corr_full)
        
        # Find the max shift and try to improve
        sh = max_idx - n_local + 1
        j0 = max(0, -sh)
        j1 = min(n_local, n_local - sh)
        
        if j0 >= j1:
            continue
        
        # Find positions where h[j]=1 and h[j+sh]=0 (can reduce by setting h[j]=0)
        # and positions where h[j]=0 and h[j+sh]=1 (can't directly, but swap)
        # Try: find j where h[j] is high and h[j+sh] is low, swap with j' where h[j'] is low
        # Actually, simpler: try random single-point adjustments
        
        # Pick a random j in the active range where reducing h[j] would help
        candidates = np.where(h[j0:j1] > 0.3)[0] + j0
        if len(candidates) == 0:
            candidates = np.arange(n_local)
        j_pick = candidates[np.random.randint(len(candidates))]
        
        # Try decreasing h[j_pick] by delta (and compensating elsewhere)
        delta = 0.01
        h_trial = h.copy()
        h_trial[j_pick] = max(0, h_trial[j_pick] - delta)
        # Find where to add delta
        mask_low = h < 0.5
        if mask_low.sum() > 0:
            j_add = np.where(mask_low)[0][np.random.randint(mask_low.sum())]
            h_trial[j_add] = min(1.0, h_trial[j_add] + delta)
            h_trial = project_to_feasible(h_trial, n_local)
            corr_t = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            c5_t = corr_t.max() * dx
            if c5_t < best_c5 - 1e-10:
                h = h_trial
        else:
            h_trial = h  # no change
    
    return best_h, best_c5

def project_to_feasible(g, n):
    target = n / 2.0
    g = g.copy()
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) / 2.0
    h = np.clip(g - theta, 0.0, 1.0)
    # Fix to exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
        else:
            # Adjust the median value
            sort_idx = np.argsort(h)
            mid_idx = sort_idx[n // 2]
            h[mid_idx] = np.clip(h[mid_idx] + diff, 0.0, 1.0)

def obj_grad(h, n, tau, top_k):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    nc = len(corr)
    if top_k < nc:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(nc)
    max_c = np.max(corr[top_idx])
    shifted = np.clip(tau * (corr[top_idx] - max_c), -400.0, 0.0)
    ec = np.exp(shifted)
    s = ec.sum()
    p = ec / s
    obj = dx * (max_c + np.log(s) / tau)
    
    grad = np.zeros(n)

        if pk < 1e-14:

        cc = top_idx[ii]

        if a0 < a1:

    grad *= dx
    return obj, grad
