def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0/old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0/new_n
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def block_coord_descent(h, n, block_sizes, max_rounds, time_limit, start_time):
    """Coordinate descent that changes blocks of consecutive values."""
    best_c = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for bs in block_sizes:
            for start in range(0, n - bs + 1, max(1, bs // 2)):
                if time.time() - start_time > time_limit:
                    return h, best_c
                old_block = h[start:start+bs].copy()
                # Try setting the block to various patterns
                candidates = [np.ones(bs) * 0.0, np.ones(bs) * 1.0,
                             np.ones(bs) * 0.5, old_block]
                # Try flipping
                candidates.append(1.0 - old_block)
                best_cand = old_block
                best_obj = best_c
                for cand in candidates:
                    trial = h.copy()
                    trial[start:start+bs] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_obj - 1e-10:
                        best_obj = c
                        best_cand = cand.copy()
                if not np.allclose(best_cand, old_block):
                    h[start:start+bs] = best_cand
                    h = make_feasible(h, n)
                    best_c = best_obj
                    improved = True
        if not improved:
            break
    return h, best_c

def local_coord_descent(h, n, max_rounds, time_limit, start_time):
    """Single-coordinate descent with finer candidates."""
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                return h, best_c
            old = h[j]
            candidates = [0.0, 1.0, 0.5, old + 0.1, old - 0.1, old + 0.05, old - 0.05]
            best_val = old
            best_obj = best_c
            for cv in candidates:
                if cv < 0 or cv > 1:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def anneal_block(h, n, time_limit, start_time, T0=0.5, cooling=0.95, steps_per_T=5):
    """Simulated annealing with block flips."""
    best_h = h.copy()
    best_c = compute_c5(h)
    T = T0
    n_attempts = 0
    
    while time.time() - start_time < time_limit:
        for _ in range(steps_per_T):
            if time.time() - start_time > time_limit:
                return best_h, best_c
            # Random block move
            bs = np.random.randint(1, min(20, n//4) + 1)
            start = np.random.randint(0, n - bs + 1)
            old_block = h[start:start+bs].copy()
            # Flip or randomize
            if np.random.random() < 0.5:
                new_block = 1.0 - old_block
            else:
                new_block = np.random.random(bs)
            trial = h.copy()
            trial[start:start+bs] = new_block
            trial = make_feasible(trial, n)
            c_trial = compute_c5(trial)
            delta = c_trial - best_c
            if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-6)):
                h = trial
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = trial.copy()
        T *= cooling
        if T < 0.01:
            break
    return best_h, best_c

            best_val = old

            for cv in cands:
                if cv < -0.01 or cv > 1.01:
                    continue
                cv = np.clip(cv, 0.0, 1.0)
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-11:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def block_cd(h, n, time_limit, start_time, block_sizes=(2, 3, 5, 8)):
    best_c = compute_c5(h)
    improved_any = True
    while improved_any and time.time() - start_time < time_limit:
        improved_any = False
        for bs in block_sizes:
            step = max(1, bs // 2)
            for start in range(0, n - bs + 1, step):
                if time.time() - start_time > time_limit:
                    return h, best_c
                old_block = h[start:start + bs].copy()
                cands = [np.zeros(bs), np.ones(bs), 1.0 - old_block, np.full(bs, 0.5)]
                best_cand = old_block
                best_obj = best_c
                for cand in cands:
                    trial = h.copy()
                    trial[start:start + bs] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_obj - 1e-10:
                        best_obj = c
                        best_cand = cand.copy()
                if not np.allclose(best_cand, old_block):
                    h[start:start + bs] = best_cand
                    h = make_feasible(h, n)
                    best_c = best_obj
                    improved_any = True
    return h, best_c

def kick_perturb(h, n, rng, strength=0.1):
    """Add a smooth random perturbation to escape local minima."""
    freqs = rng.randint(1, 5, size=4)
    phases = rng.uniform(0, 2 * np.pi, size=4)
    amps = rng.uniform(0.3, 1.0, size=4)
    x = np.linspace(0, 2, n, endpoint=False)
    pert = np.zeros(n)
    for f, p, a in zip(freqs, phases, amps):
        pert += a * np.sin(2 * np.pi * f * x + p)
    h_new = h + strength * pert
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25
    rng = np.random.RandomState(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        n_init = 512
        h0 = rng.rand(n_init)
        h0 = make_feasible(h0, n_init)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: L-BFGS-B at current resolution with increasing temperature
    for t, pen, mi in [(15, 50, 80), (40, 100, 120), (150, 300, 150), (500, 800, 120)]:
        if time.time() > deadline - 500:
            break
        h_new = lbfgs_optimize(best_h, n, t, pen, mi)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-10:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Local coordinate descent
    if time.time() < deadline - 250:
        h_new, c_new = local_cd(best_h, n, max_rounds=8,
                                time_limit=deadline - 200, start_time=start_time)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 3: Block coordinate descent
    if time.time() < deadline - 150:
        h_new, c_new = block_cd(best_h, n,
                                time_limit=deadline - 100, start_time=start_time)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 4: Kick + re-optimize (try to escape local minima)
    for kick_i in range(3):
        if time.time() > deadline - 200:
            break
        h_kick = kick_perturb(best_h, n, rng, strength=0.08)
        h_new = lbfgs_optimize(h_kick, n, 100, 300, 100)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-10:
            best_c5 = c_new
            best_h = h_new
            if time.time() < deadline - 80:
                h_cd, c_cd = local_cd(best_h, n, max_rounds=4,
                                      time_limit=deadline - 50, start_time=start_time)
                if c_cd < best_c5:
                    best_c5 = c_cd
                    best_h = h_cd

    # Phase 5: Upsample and optimize at higher resolution
    n2 = min(n * 2, 1280)
    if time.time() < deadline - 250:
        h2 = upsample_h(best_h, n2)
        for t, pen, mi in [(40, 100, 80), (150, 300, 120), (500, 800, 100)]:
            if time.time() > deadline - 150:
                break
            h2_new = lbfgs_optimize(h2, n2, t, pen, mi)
            c2 = compute_c5(h2_new)
            if c2 < best_c5 - 1e-10:
                best_c5 = c2
                best_h = h2_new
            h2 = h2_new
        if time.time() < deadline - 80:
            h_new, c_new = local_cd(best_h, len(best_h), max_rounds=5,
                                    time_limit=deadline - 40, start_time=start_time)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
        if time.time() < deadline - 30:
            h_new, c_new = block_cd(best_h, len(best_h),
                                    time_limit=deadline - 10, start_time=start_time)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Phase 6: Try complement symmetry (1-h is also feasible)
    if time.time() < deadline - 30:
        h_comp = 1.0 - best_h
        h_comp = make_feasible(h_comp, len(h_comp))
        c_comp = compute_c5(h_comp)
        if c_comp < best_c5 - 1e-10:
            best_c5 = c_comp
            best_h = h_comp
            h_new, c_new = local_cd(best_h, len(best_h), max_rounds=3,
                                    time_limit=deadline, start_time=start_time)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
