            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def block_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, 8)
        start = rng.randint(0, n - bsize)
        end = start + bsize
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.01), max(0.0, old - 0.003),
                         min(1.0, old + 0.003), min(1.0, old + 0.01)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def multi_temp_slsqp(h, n, start_time, budget_s):
    temps = [80.0, 300.0, 1000.0, 5000.0, 20000.0]
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 40:
            break
        try:
            x_res = slsqp_optimize(h, n, T, 150, 1e-13)
            c5_res = compute_c5(x_res)
            if c5_res < compute_c5(h):
                h = make_feasible(x_res, n)
        except Exception:
            pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Temperature sweep SLSQP from start
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s)

    # Phase 2: Structured perturbations + SLSQP refinement
    perts = [fourier_perturb, segment_perturb, blend_perturb]
    for pf in perts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 120:
            break
        for _ in range(2):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
            try:
                hp = pf(best_h, n0, rng)
                hr, cr = multi_temp_slsqp(hp, n0, start_time, budget_s)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 3: Binary SA (point swaps)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 30, 120)
        b, c5_b = binary_sa(best_h, n0, 100000, 0.006, 0.00001, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 80:
                best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s)

    # Phase 4: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 30, 120)
        b, c5_b = block_sa(best_h, n0, 60000, 0.005, 0.00003, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 80:
                best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s)

    # Phase 5: Higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 350:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        h_up, c5_up = multi_temp_slsqp(h_up, n1, start_time, budget_s)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        # Additional perturbation at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 120:
            try:
                hp = fourier_perturb(best_h, len(best_h), rng, n_modes=3, amp=0.03)
                hr, cr = multi_temp_slsqp(hp, len(best_h), start_time, budget_s)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 6: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 20, start_time, budget_s)

    # Phase 7: Final high-temperature SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        n_pts = len(best_h)
        try:
            x_res = slsqp_optimize(best_h, n_pts, 30000.0, 200, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 8: One more coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 10, start_time, budget_s)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
