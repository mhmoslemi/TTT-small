```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_c5(h, t):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(-t * shifted))) / t) * dx

def slsqp_optimize(h_start, n, t, maxiter):
    def obj(h):
        return soft_c5(h, t)
    def con_fun(h):
        return np.sum(h) - n / 2.0
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or np.sum(res.x**2) > 0:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return make_feasible(h_start, n)

def binary_sa(h, n, start_time, time_limit, rng, T0=1.0, T_end=0.01, max_iter=None):
    """Simulated annealing on binary solutions using swap moves."""
    h = h.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    best_h = h.copy()
    best_c = compute_c5(h)
    current_c = best_c
    it = 0
    
    if max_iter is None:
        max_iter = 100000
    
    while it < max_iter:
        if time.time() - start_time > time_limit - 30:
            break
        
        # Temperature schedule
        progress = min(1.0, it / max_iter)
        T = T0 * (T_end / T0) ** progress
        
        # Pick a random 1 and a random 0
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Swap
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        
        delta = new_c - current_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        it += 1
    
    return best_h, best_c
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n <= old_n:
        return h[:new_n].copy() if new_n < old_n else h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)  # No, don't use period
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_c5(h, t):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Use log-sum-exp for numerical stability
    shifted = t * (corr - mc)
    max_shifted = np.max(shifted)
    lse = mc + (max_shifted + np.log(np.sum(np.exp(shifted - max_shifted)))) / t
    return lse * dx

def slsqp_optimize(h_start, n, t, maxiter):
    def obj(h):
        return soft_c5(h, t)
    def con_fun(h):
        return np.sum(h) - n / 2.0
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.sum(res.x**2) > 0:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return make_feasible(h_start, n)

def coord_descent(h, n, max_rounds=50, time_limit=None, start_time=None):
    dx = 2.0 / n
    best_c = compute_c5(h)
    h = h.copy()
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 20:
                    return h, best_c
            old = h[j]
            # Try several candidates
            candidates = set()
            candidates.add(0.0)
            candidates.add(1.0)
            candidates.add(0.5)
            for delta in [0.1, 0.05, 0.02, 0.01, 0.005]:
                candidates.add(max(0.0, min(1.0, old + delta)))
                candidates.add(max(0.0, min(1.0, old - delta)))
            
            best_obj = best_c
            best_val = old
            for cv in candidates:
                if cv == old:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def sa_binary(h, n, rng, T0, T1, iters, start_time, deadline):

    h = (h > 0.5).astype(float)

        frac = it / max(1, iters - 1)

        j_idx = zeros[rng.randint(len(zeros))]

        h[j_idx] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c

            # Update ones/zeros tracking

            zeros = np.where(h == 0.0)[0]
            if new_c < best_c - 1e-12:

                best_h = h.copy()
            # Occasionally re-derive exact best

            h[j_idx] = 0.0

            c_check = compute_c5(h)
            if c_check < best_c:

                best_h = h.copy()

    make_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c

        if time.time() > deadline:
            break
        
        frac = it / max(1, iters - 1)
        T = T0 * (T1 / T0) ** frac
        
        # Block swap: pick two random positions and swap blocks of length 1-5
        blen = rng.randint(1, 6)
        i = rng.randint(0, n - blen)
        j = rng.randint(0, n - blen)
        
        if i == j:
            continue
        
        # Check that block at i is mostly 1s and block at j is mostly 0s (or vice versa)
        block_i = h[i:i+blen]
        block_j = h[j:j+blen]
        
        # Only swap if it maintains the sum constraint
        sum_i = block_i.sum()
        sum_j = block_j.sum()
        if sum_i != sum_j:
            # Find positions to swap
            # Swap a 1 from block_i with a 0 from block_j
            ones_in_i = [i+k for k in range(blen) if block_i[k] > 0.5]
            zeros_in_j = [j+k for k in range(blen) if block_j[k] < 0.5]
            ones_in_j = [j+k for k in range(blen) if block_j[k] > 0.5]
            zeros_in_i = [i+k for k in range(blen) if block_i[k] < 0.5]
            
            if len(ones_in_i) > 0 and len(zeros_in_j) > 0:
                a = ones_in_i[rng.randint(len(ones_in_i))]
                b = zeros_in_j[rng.randint(len(zeros_in_j))]
                h[a] = 0.0
                h[b] = 1.0
                new_c = compute_c5(h)
                delta = new_c - cur_c
                if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
                    cur_c = new_c
                    if new_c < best_c - 1e-12:
                        best_c = new_c
                        best_h = h.copy()
                else:
                    h[a] = 1.0
                    h[b] = 0.0
            continue
        
        # Equal sums: swap blocks directly
        h[i:i+blen], h[j:j+blen] = block_j.copy(), block_i.copy()
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-10 and rng.rand() < np.exp(-delta / T)):
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i:i+blen], h[j:j+blen] = block_i.copy(), block_j.copy()
    
    return best_h, best_c
```

```python
    old_n = len(h)
    if new_n <= old_n:
        return h[:new_n].copy() if new_n < old_n else h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_c5(h, t):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = t * (corr - mc)
    ms = np.max(shifted)
    return (mc + (ms + np.log(np.sum(np.exp(shifted - ms))) / t)) * dx

def slsqp_run(h_start, n, t, maxiter):
    def obj(h):
        return soft_c5(h, t)
    def cfun(h):
        return np.sum(h) - n / 2.0
    def cjac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return make_feasible(h_start, n)

def cd_refine(h, n, max_rounds=50, start_time=None, deadline=None):
    best_c = compute_c5(h)
    h = h.copy()
    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if deadline is not None and time.time() > deadline:
                return h, best_c
            old = h[j]
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old-0.1), min(1.0, old+0.1),
                     max(0.0, old-0.05), min(1.0, old+0.05),
                     max(0.0, old-0.01), min(1.0, old+0.01)]
            best_obj = best_c
            best_val = old
            for cv in cands:
                if abs(cv - old) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def sa_single_swap(h, n, rng, T0, T1, deadline):
    """SA with single 1-0 swap moves."""
    h = (h > 0.5).astype(float)
    make_feasible(h, n)
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    it = 0
    while time.time() < deadline:
        it += 1
        # Adaptive T
        T = T0 * (T1 / T0) ** min(1.0, it / 80000.0)
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        d = new_c - cur_c
        if d < 0 or (T > 1e-12 and rng.rand() < np.exp(-d / T)):
            cur_c = new_c
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def hill_climb_swap(h, n, start_time, deadline):
    """Steepest-descent single swap."""
    h = (h > 0.5).astype(float)
    make_feasible(h, n)
    best_c = compute_c5(h)
    improved = True
    while improved and time.time() < deadline:
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_d = 0.0
        best_i, best_j = -1, -1
        # Try a subset of swaps (can't try all n²/4)
        n_trials = min(len(ones) * len(zeros), 2000)
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h)
            h[i] = 1.0
            h[j] = 0.0
            d = c - best_c
            if d < best_d - 1e-12:
                best_d = d
                best_i, best_j = i, j
        if best_i >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            best_c = compute_c5(h)
            improved = True
    return h, best_c
```
