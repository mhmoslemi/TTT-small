def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

import numpy as np
from scipy.optimize import minimize
import time

_CUR_N = 480
_CUR_T = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _smobj(x):
    n = _CUR_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_CUR_T * shifted))) / _CUR_T) * dx

def _con_fun(x):
    return np.sum(x) - _CUR_N / 2.0

def _con_jac(x):
    return np.ones(_CUR_N)

def slsqp_solve(h_start, n, T, maxiter=200):
    global _CUR_N, _CUR_T
    _CUR_N = n
    _CUR_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_smobj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def coord_descent(h, n, max_rounds, time_limit, rng):
    start_t = time.time()
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_t > time_limit * 0.9:
            break
        improved = False
        for j in range(n):
            if time.time() - start_t > time_limit:
                return h, best_c5
            old = h[j]
            best_val = old
            best_c = best_c5
            cands = [0.0, 1.0, 0.5]
            cands.append(max(0.0, old - 0.05))
            cands.append(min(1.0, old + 0.05))
            cands.append(max(0.0, old - 0.02))
            cands.append(min(1.0, old + 0.02))
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def binary_snap(h, n, budget):
    best_c5 = compute_c5(h)
    for j in range(n):
        if time.time() - budget[0] > budget[1]:
            break
        for val in [0.0, 1.0]:
            trial = h.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-14:
                best_c5 = c
                h = trial
                break
    return h, best_c5

    return best_h, best_c5, len(best_h)
