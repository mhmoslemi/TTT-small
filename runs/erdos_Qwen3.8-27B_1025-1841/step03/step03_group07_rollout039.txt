    shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx]
        k = top_idx[i_idx]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def nesterov_step(h, n, tau, time_limit, lr=None, top_k=10):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0
    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj, grad = soft_obj_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 3000:
                lr *= 0.6
                stall = 0
                if lr < 1e-15:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def coord_descent(h, n, time_limit, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    ndx = 2.0 / n
    best_c5 = corr.max() * ndx
    best_h = h.copy()
    start = time.time()
    nc = len(corr)
    factor = np.zeros(nc)
    no_improve = 0
    while time.time() - start < time_limit:
        order = rng.permutation(n)
        pass_impr = 0
        cl = 0.0
        for idx in range(n):
            i = order[idx]
            v = h[i]
            if v < 1e-8 or v > 1.0 - 1e-8:
                continue
            factor.fill(0.0)
            k_f_lo = max(0, n - 1 - i)
            k_f_hi = min(nc - 1, n - 1 + i)
            g_start = i + k_f_lo - (n - 1)
            glen = k_f_hi - k_f_lo + 1
            factor[k_f_lo:k_f_hi + 1] += g[g_start:g_start + glen]
            k_b_lo = max(0, i)
            k_b_hi = min(nc - 1, i + n - 1)
            h_start = i - k_b_hi + (n - 1)
            blen = k_b_hi - k_b_lo + 1
            factor[k_b_lo:k_b_hi + 1] -= h[h_start:h_start + blen]
            cur_max = corr.max()
            d0 = -v
            d1 = 1.0 - v
            mc0 = np.max(corr + d0 * factor)
            mc1 = np.max(corr + d1 * factor)
            if mc0 <= mc1:
                if mc0 < cur_max - 1e-10:
                    corr += d0 * factor
                    h[i] = 0.0
                    g[i] = 1.0
                    pass_impr += 1
            else:
                if mc1 < cur_max - 1e-10:
                    corr += d1 * factor
                    h[i] = 1.0
                    g[i] = 0.0
                    pass_impr += 1
        c5_now = corr.max() * ndx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 12:
            n_pert = max(3, n // 100)
            pert_idx = rng.choice(n, size=n_pert, replace=False)
            h[pert_idx] = rng.uniform(0.1, 0.9, size=n_pert)
            g = 1.0 - h
            corr = np.correlate(h, g, mode='full')
            no_improve = 0
    return best_h, best_c5


def refine_subgrad(h, n, time_limit, lr=0.001, top_k=5):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-15:
                break
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for i_idx in range(len(top_idx)):
            pk = w[i_idx]
            k = top_idx[i_idx]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[j0:j1] += pk * (1.0 - h[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[j0b:j1b] -= pk * h[idx + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    best_n = n_init

    rem = max(0.0, budget_s - (time.time() - start_time))
    if rem > 250:
        tl = min(100, rem * 0.12)
        h1, c5 = nesterov_step(h_init, n_init, tau=80.0, time_limit=tl, lr=0.25, top_k=15)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h1.copy(), n_init

    rem = max(0.0, budget_s - (time.time() - start_time))
    if rem > 300:
        tl = min(400, rem * 0.38)
        h2, c5 = coord_descent(best_h, best_n, time_limit=tl, seed=seed)
        h2p = project_to_feasible(h2, best_n)
        c5p = compute_c5(h2p)
        if c5p < best_c5:
            best_c5, best_h, best_n = c5p, h2p.copy(), best_n

    rem = max(0.0, budget_s - (time.time() - start_time))
    if rem > 250:
        n_hi = best_n * 2
        h_hi = interp_periodic(best_h, n_hi)
        h_hi = project_to_feasible(h_hi, n_hi)
        tl = min(350, rem * 0.4)
        h3, c5 = coord_descent(h_hi, n_hi, time_limit=tl, seed=seed + 100)
        h3p = project_to_feasible(h3, n_hi)
        c5p = compute_c5(h3p)
        if c5p < best_c5:
            best_c5, best_h, best_n = c5p, h3p.copy(), n_hi

    rem = max(0.0, budget_s - (time.time() - start_time))
    if rem > 150:
        tl = min(200, rem * 0.4)
        h4, c5 = coord_descent(best_h, best_n, time_limit=tl, seed=seed + 200)
        h4p = project_to_feasible(h4, best_n)
        c5p = compute_c5(h4p)
        if c5p < best_c5:
            best_c5, best_h, best_n = c5p, h4p.copy(), best_n

    rem = max(0.0, budget_s - (time.time() - start_time))
    if rem > 60:
        tl = min(50, rem * 0.5)
        h5, c5 = refine_subgrad(best_h, best_n, time_limit=tl, lr=0.0008, top_k=4)
        if c5 < best_c5:
            best_c5, best_h, best_n = c5, h5.copy(), best_n

    rem = max(0.0, budget_s - (time.time() - start_time))
    if rem > 40:
        tl = min(35, rem * 0.6)
        h6, c5 = coord_descent(best_h, best_n, time_limit=tl, seed=seed + 300)
        h6p = project_to_feasible(h6, best_n)
        c5p = compute_c5(h6p)
        if c5p < best_c5:
            best_c5, best_h, best_n = c5p, h6p.copy(), best_n

    best_h = project_to_feasible(best_h, best_n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
