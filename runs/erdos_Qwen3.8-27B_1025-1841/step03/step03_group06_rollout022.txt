import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 960
_OPT_T = 100.0

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    sorted_idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[sorted_idx[:n//2]] = 1.0

    size = 2 * n
    H = np.fft.rfft(h, size)
    OH = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(H * np.conj(OH), size)
    return corr[:2*n-1]

def c5_from_h(h):
    n = len(h)
    corr = corr_full_fft(h)
    return float(np.max(corr) * (2.0/n))

    best_h = h.copy()

            corr = corr_full_fft(best_h)

            elif -n < shift <= 0:

            else:

def simulated_annealing_binary(h, n, max_iters, rng, time_limit, T_start=1.0, T_end=0.01):
    """Simulated annealing for binary optimization."""
    start_time = time.time()
    cur_h = h.copy()
    cur_c = c5_from_h(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    
    ones = np.where(cur_h > 0.5)[0]
    zeros = np.where(cur_h < 0.5)[0]
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        trial = cur_h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        c = c5_from_h(trial)
        
        delta = c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / (T * 2.0/n)):
            cur_h = trial
            cur_c = c
            ones = np.where(cur_h > 0.5)[0]
            zeros = np.where(cur_h < 0.5)[0]
            if cur_c < best_c:
                best_h = cur_h.copy()
                best_c = cur_c

def continuous_refine(h, n, T, maxiter, time_limit):
    """SLSQP continuous refinement."""
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    
    start = time.time()
    try:
        res = minimize(_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        c5_res = compute_c5(res.x)

    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

    return np.sum(x) - _OPT_N / 2.0

                        best_c5 = c
                        improved = True
                        break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
