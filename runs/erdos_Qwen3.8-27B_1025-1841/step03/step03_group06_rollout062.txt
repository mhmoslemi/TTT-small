import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_full(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    mv = np.max(vals)
    shifted = vals - mv
    return float(mv + np.log(np.sum(np.exp(T * shifted))) / T)

def smooth_obj_active(x, n, T, active_idx):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr[active_idx] * dx
    mv = np.max(vals)
    shifted = vals - mv
    return float(mv + np.log(np.sum(np.exp(T * shifted))) / T)

def smooth_obj_top(x, n, T, m):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    top = np.argpartition(vals, -m)[-m:]
    vals_top = vals[top]
    mv = np.max(vals_top)
    shifted = vals_top - mv
    return float(mv + np.log(np.sum(np.exp(T * shifted))) / T)

def _con_eq(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def _obj_wrap_full(x, n, T):
    return smooth_obj_full(x, n, T)

def _obj_wrap_active(x, n, T, active_idx):
    return smooth_obj_active(x, n, T, active_idx)

def _obj_wrap_top(x, n, T, m):
    return smooth_obj_top(x, n, T, m)

def get_active(corr, m=30, margin_frac=0.02):
    vals = corr * (2.0 / len(corr // 2))
    mx = np.max(vals)
    threshold = mx - margin_frac * mx
    active = np.where(vals >= threshold)[0]
    if len(active) > m:
        top = np.argsort(vals)[::-1][:m]
        active = top
    if len(active) < 5:
        active = np.argsort(vals)[::-1][:5]
    return active

def binary_snap(h, n, max_rounds=5):
    best = h.copy()
    best_c = compute_c5(h)
    for _r in range(max_rounds):
        improved = False
        for j in range(n):
            for val in (0.0, 1.0):
                if abs(best[j] - val) < 1e-12:
                    continue
                trial = best.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best = trial
                    best_c = c
                    improved = True
        if not improved:
            break
    return best, best_c

def local_search(h, n, time_limit=120):

    return best, best_c

                    best_local = cv
                    best_trial = c
            if best_trial != old:
                best[j] = best_trial
                best = make_feasible(best, n)
                best_c = best_local
                improved = True
        if not improved:
            break
    return best, best_c

def subgradient_polish(h, n, steps=5000, margin=0.001):
    """Polish using subgradient on the active max."""
    best = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / n
    alpha = 0.01
    t_hist = 0.0
    # Pre-compute the gradient indices for each shift
    for step in range(steps):
        corr = compute_corr_full(best)
        vals = corr * dx
        k_star = np.argmax(vals)
        current_max = vals[k_star]
        if current_max < best_c - 0.001:
            best_c = current_max
            best = best.copy()
        # Subgradient of corr[k_star] w.r.t. h
        k = k_star
        sg = np.zeros(n)
        if k < n:
            # corr[k] = sum_{i=max(0,k-n+1)}^{min(n-1,k)} h[i]*(1-h[n-1-k+i])
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                j = i  # h[i] term
                sg[j] += (1.0 - best[n - 1 - k + i])
                j2 = n - 1 - k + i  # (1-h[j2]) term
                if 0 <= j2 < n:
                    sg[j2] += (-best[i])
        else:
            k2 = k - (n - 1)
            i_lo = max(0, k - n + 1)
            i_hi = min(n - 1, k)
            for i in range(i_lo, i_hi + 1):
                j = i
                sg[j] += (1.0 - best[n - 1 - k + i])
                j2 = n - 1 - k + i
                if 0 <= j2 < n:
                    sg[j2] += (-best[i])
        # Normalize subgradient
        sg_norm = np.linalg.norm(sg)
        if sg_norm < 1e-15:
            break
        sg = sg / sg_norm
        # Step
        t_hist += alpha ** 2
        h_new = best - alpha * sg
        # Project to {0<=h<=1, sum(h)=n/2}
        h_new = np.clip(h_new, 0.0, 1.0)
        target = n / 2.0
        for _ in range(10):
            diff = target - np.sum(h_new)
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = h_new < 1.0
            else:
                mask = h_new > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best = h_new
            best_c = c_new
        alpha *= 0.999
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP polish at initial resolution
    if remaining() > 80:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n,)}]
        for T in (50, 120):
            if remaining() < 40:
                break
            for restart in range(2):
                if remaining() < 30:
                    break
                if restart == 0:
                    x0 = best_h.copy()
                else:
                    x0 = best_h + rng.randn(n) * 0.02
                    x0 = make_feasible(x0, n)
                try:
                    res = minimize(_obj_wrap_full, x0, args=(n, T),
                                   method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 200, 'ftol': 1e-13})
                    c5r = compute_c5(res.x)
                    if c5r < best_c5:
                        best_c5 = c5r
                        best_h = make_feasible(res.x, n)
                except Exception:
                    pass

    # Phase 2: Active-set iterative SLSQP
    if remaining() > 120:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n,)}]
        for it in range(10):
            if remaining() < 40:
                break
            corr = compute_corr_full(best_h)
            dx = 2.0 / n
            active = get_active(corr, m=20, margin_frac=0.0)
            # Also include a broader set
            vals = corr * dx
            thresh = np.max(vals) - 0.005
            broad = np.where(vals >= thresh)[0]
            if len(broad) > 40:
                broad = np.argsort(vals)[::-1][:40]

            for T in (80, 200):
                if remaining() < 25:
                    break
                try:
                    res = minimize(_obj_wrap_active, best_h,
                                   args=(n, T, broad),
                                   method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 150, 'ftol': 1e-13})
                    c5r = compute_c5(res.x)
                    if c5r < best_c5:
                        best_c5 = c5r
                        best_h = make_feasible(res.x, n)
                except Exception:
                    pass

            # Also try top-m variant
            if remaining() > 25:
                try:
                    res = minimize(_obj_wrap_top, best_h,
                                   args=(n, 150, 15),
                                   method='SLSQP', bounds=bounds,
                                   constraints=cons,
                                   options={'maxiter': 100, 'ftol': 1e-13})
                    c5r = compute_c5(res.x)
                    if c5r < best_c5:
                        best_c5 = c5r
                        best_h = make_feasible(res.x, n)
                except Exception:
                    pass

    # Phase 3: Binary snap
    if remaining() > 20:
        n = len(best_h)
        best_h, best_c5 = binary_snap(best_h, n, max_rounds=3)

    # Phase 4: Local search
    if remaining() > 60:
        n = len(best_h)
        tl = min(remaining() - 20, 200)
        if tl > 20:
            best_h, best_c5 = local_search(best_h, n, time_limit=tl)

    # Phase 5: SLSQP to break binary, then re-snap
    if remaining() > 80:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n,)}]
        for T in (100, 300):
            if remaining() < 40:
                break
            try:
                res = minimize(_obj_wrap_full, best_h, args=(n, T),
                               method='SLSQP', bounds=bounds,
                               constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-14})
                c5r = compute_c5(res.x)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = make_feasible(res.x, n)
            except Exception:
                pass
        if remaining() > 20:
            best_h, best_c5 = binary_snap(best_h, n, max_rounds=3)

    # Phase 6: Subgradient polish
    if remaining() > 60:
        n = len(best_h)
        steps = min(3000, int(remaining() * 2))
        if steps > 100:
            best_h, best_c5 = subgradient_polish(best_h, n, steps=steps)

    # Phase 7: Fine local search
    if remaining() > 30:
        n = len(best_h)
        tl = min(remaining() - 10, 80)
        if tl > 10:
            best_h, best_c5 = fine_local_search(best_h, n, time_limit=tl)

    # Phase 8: Try higher resolution if time permits
    if remaining() > 150:
        n_high = 1920
        h_high = upsample_h(best_h, n_high)
        bounds_h = [(0.0, 1.0)] * n_high
        cons_h = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n_high,)}]
        for T in (80, 200):
            if remaining() < 80:
                break
            try:
                res = minimize(_obj_wrap_full, h_high, args=(n_high, T),
                               method='SLSQP', bounds=bounds_h,
                               constraints=cons_h,
                               options={'maxiter': 80, 'ftol': 1e-12})
                c5r = compute_c5(res.x)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = make_feasible(res.x, n_high)
            except Exception:
                pass
        # Active set at high res
        if remaining() > 60:
            n_h2 = len(best_h)
            bounds_h2 = [(0.0, 1.0)] * n_h2
            cons_h2 = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac, 'args': (n_h2,)}]
            corr_h = compute_corr_full(best_h)
            active_h = get_active(corr_h, m=25, margin_frac=0.0)
            try:
                res = minimize(_obj_wrap_active, best_h, args=(n_h2, 200, active_h),
                               method='SLSQP', bounds=bounds_h2,
                               constraints=cons_h2,
                               options={'maxiter': 80, 'ftol': 1e-12})
                c5r = compute_c5(res.x)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = make_feasible(res.x, n_h2)
            except Exception:
                pass
        if remaining() > 15:
            best_h, best_c5 = binary_snap(best_h, len(best_h), max_rounds=2)

    # Phase 9: Final polish
    if remaining() > 20:
        n = len(best_h)
        best_h, best_c5 = fine_local_search(best_h, n, time_limit=min(remaining() - 5, 40))

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
