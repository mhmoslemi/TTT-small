import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 5.0
    hi = g.max() + 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)


def interp_to_n(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) / n
    x_new = np.arange(target_n) / target_n
    return np.interp(x_new, x_old, h)


def nesterov_phase(h, n, tau_start, tau_end, lr, time_limit, top_k=15):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    max_iter = 5000000

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        frac = min(1.0, it / (time_limit * 80.0))
        tau_cur = tau_start * (1.0 - frac) + tau_end * frac

        mu = (t - 1.0) / (t + 2.0)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        max_c = top_vals.max()
        shifted = (top_vals - max_c) / tau_cur
        p = np.exp(shifted)
        p_sum = p.sum()
        if p_sum < 1e-15:
            p = np.ones(len(top_idx)) / len(top_idx)
        else:
            p /= p_sum

        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = p[ii] * dx
            if pk < 1e-12:
                continue
            sh = top_idx[ii] - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[j0:j1] += pk * (1.0 - y[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[j0b:j1b] -= pk * y[idx + sh]

        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)

        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 8000:
                lr *= 0.5
                stall = 0
                if lr < 1e-16:
                    break

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_phase(h, n, lr, time_limit, top_k=8, beta=0.85):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    max_iter = 5000000

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-16:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 6000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-16:
                break

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals.max())
        w /= w.sum()

        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = w[ii] * dx
            if pk < 1e-12:
                continue
            sh = top_idx[ii] - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[j0:j1] += pk * (1.0 - h[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[j0b:j1b] -= pk * h[idx + sh]

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def coord_descent(h, n, time_limit, block_size=200):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    n_blocks = n // block_size

    while True:
        improved = False
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                return best_h, best_c5
            i0 = b * block_size
            i1 = (b + 1) * block_size
            target_block = (n / 2.0) - (np.sum(h) - np.sum(h[i0:i1]))
            h_block = h[i0:i1].copy()
            for _ in range(30):
                h_block = np.clip(h_block, 0.0, 1.0)
                h[i0:i1] = h_block
                new_sum = np.sum(h)
                diff = target_block - (new_sum - np.sum(h[i0:i1]) + np.sum(h_block))
                h_block += diff / block_size
            h[i0:i1] = h_block
            h = project_to_feasible(h, n)
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                improved = True
        if not improved:
            break

    return best_h, best_c5


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))
