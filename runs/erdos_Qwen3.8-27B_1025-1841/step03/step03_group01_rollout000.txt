        if abs(diff) < 1e-13:

        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _slsqp_softmax(h_start, n, T, K, max_iter):
    dx = 2.0 / n
    full_len = 2 * n - 1

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if K is not None and K < full_len:
            top_idx = np.argpartition(corr, -K)[-K:]
            vals = corr[top_idx]
        else:
            vals = corr
        mc = np.max(vals)
        shifted = vals - mc
        log_sum = np.log(np.sum(np.exp(T * shifted)))
        return (mc + log_sum / T) * dx

    def cons_fun(h):
        return np.sum(h) - n / 2.0

    def cons_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def _grad_lag(h, n, lag):
    grad = np.zeros(n)
    j_lo1 = max(0, lag)
    j_hi1 = min(n, n + lag)
    if j_hi1 > j_lo1:
        grad[j_lo1:j_hi1] += (1.0 - h[j_lo1 - lag:j_hi1 - lag])
    j_lo2 = max(0, -lag)
    j_hi2 = min(n, n - lag)
    if j_hi2 > j_lo2:
        grad[j_lo2:j_hi2] -= h[j_lo2 + lag:j_hi2 + lag]
    return grad

def _softmax_gradient_descent(h, n, steps, T, K, lr_init, time_limit):
    dx = 2.0 / n
    full_len = 2 * n - 1
    h_curr = h.copy()
    lr = lr_init
    best_local = compute_c5(h_curr)
    start_t = time.time()

    for step in range(steps):
        if time.time() - start_t > time_limit:
            break
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        if K is not None and K < full_len:
            top_idx = np.argpartition(corr, -K)[-K:]
            vals = corr[top_idx]
            weights_full = np.zeros(full_len)
        else:
            top_idx = np.arange(full_len)
            vals = corr
            weights_full = np.ones(full_len)

        mc = np.max(vals)
        exp_shifted = np.exp(T * (vals - mc))
        weights = exp_shifted / exp_shifted.sum()

        # Gradient of soft-max objective w.r.t. h
        grad_obj = np.zeros(n)
        for idx_i, w_i in enumerate(zip(top_idx, weights)):
            ki, wi = idx_i
            lag = ki - (n - 1)
            grad_obj += wi * _grad_lag(h_curr, n, lag)

        # Projected gradient step (minimize, so negative gradient)
        h_trial = h_curr - lr * grad_obj
        h_trial = np.clip(h_trial, 0.0, 1.0)
        h_trial = make_feasible(h_trial, n)
        c_trial = compute_c5(h_trial)
        if c_trial < best_local:
            best_local = c_trial
            h_curr = h_trial
            lr = min(lr * 1.05, 0.02)
        else:
            lr *= 0.5
            if lr < 1e-10:
                break
    return h_curr, best_local

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: SLSQP at n=1280 to get a good starting point
    n1 = 1280
    h1 = upsample_h(best_h, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    if time_left() > 60:
        try:
            h1 = _slsqp_softmax(h1, n1, 150.0, None, 200)
            c5_1 = compute_c5(h1)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()
        except Exception:
            pass

    # Phase 2: Main optimization at n=2048
    n2 = 2048
    h2 = upsample_h(best_h, n2)
    c5_2 = compute_c5(h2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()

    # SLSQP passes at n=2048
    slsqp_configs = [
        (200.0, None, 200),
        (500.0, None, 150),
        (1000.0, 5, 150),
        (500.0, 3, 150),
        (2000.0, None, 100),
    ]
    for T, K, max_it in slsqp_configs:
        if time_left() < 30:
            break
        try:
            h2 = _slsqp_softmax(h2, n2, T, K, max_it)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
        except Exception:
            pass

    # Gradient descent polish at n=2048
    if time_left() > 120:
        h2, c5_gd = _softmax_gradient_descent(h2, n2, steps=500, T=200.0, K=10, lr_init=0.005, time_limit=60)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h2.copy()

    # More SLSQP after GD
    if time_left() > 40:
        try:
            h2 = _slsqp_softmax(h2, n2, 800.0, 5, 100)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()
        except Exception:
            pass

    # Phase 3: Try n=3072
    if time_left() > 180:
        n3 = 3072
        h3 = upsample_h(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

        if time_left() > 50:
            try:
                h3 = _slsqp_softmax(h3, n3, 300.0, None, 150)
                c5_3 = compute_c5(h3)
                if c5_3 < best_c5:
                    best_c5 = c5_3
                    best_h = h3.copy()
            except Exception:
                pass

        # GD at n=3072
        if time_left() > 60:
            h3, c5_gd3 = _softmax_gradient_descent(h3, n3, steps=400, T=200.0, K=10, lr_init=0.004, time_limit=50)
            if c5_gd3 < best_c5:
                best_c5 = c5_gd3
                best_h = h3.copy()

        # Final SLSQP at n=3072
        if time_left() > 30:
            try:
                h3 = _slsqp_softmax(h3, n3, 1000.0, 3, 100)
                c5_3 = compute_c5(h3)
                if c5_3 < best_c5:
                    best_c5 = c5_3
                    best_h = h3.copy()
            except Exception:
                pass

    # Phase 4: Random restarts with local optimization
    n_cur = len(best_h)
    restart_configs = [
        (0.03, 500.0, 80),
        (0.05, 300.0, 80),
        (0.02, 1000.0, 60),
        (0.08, 200.0, 60),
        (0.01, 500.0, 60),
        (0.04, 800.0, 60),
    ]
    for scale, T, max_it in restart_configs:
        if time_left() < 25:
            break
        try:
            h_pert = best_h + scale * rng.randn(n_cur)
            h_pert = make_feasible(h_pert, n_cur)
            h_opt = _slsqp_softmax(h_pert, n_cur, T, None, max_it)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 5: Final GD polish
    if time_left() > 40:
        n_fin = len(best_h)
        h_fin, c5_fin = _softmax_gradient_descent(best_h, n_fin, steps=300, T=100.0, K=5, lr_init=0.003, time_limit=30)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Phase 6: Top-K focused SLSQP on final solution
    if time_left() > 25:
        n_fin = len(best_h)
        try:
            h_fin = _slsqp_softmax(best_h, n_fin, 2000.0, 3, 80)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin
        except Exception:
            pass

    # Final feasibility
    n_ret = len(best_h)
    best_h = make_feasible(best_h, n_ret)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_ret
