def fft_correlate(h):
    """Compute full linear correlation of h and 1-h using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    a = np.zeros(N)
    a[:n] = h
    b = np.zeros(N)
    b[:n] = 1.0 - h
    V = np.fft.rfft(a) * np.conj(np.fft.rfft(b))
    Z = np.fft.irfft(V, N)
    return np.concatenate([Z[N - n + 1:], Z[:n]])

def compute_c5(h):
    return float(np.max(fft_correlate(h)) * 2.0 / len(h))

def binary_local_search(h, n, rng, max_passes=50, pair_tries_per_pass=500):
    """Exhaustive single-flip + random pair-swap local search."""
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for rnd in range(max_passes):
        improved = False
        # Single flips (try all)
        for i in range(n):
            old = h[i]
            h[i] = 1.0 - old
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                if old > 0.5:
                    ones.append(i)
                    zeros.remove(i)
                else:
                    zeros.append(i)
                    ones.remove(i)
                improved = True
            else:
                h[i] = old
        # Pair swaps (random)
        for _ in range(pair_tries_per_pass):
            if not ones or not zeros:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                improved = True

                h[i] = 1.0
                h[j] = 0.0

            break

    zeros = np.where(h <= 0.5)[0].tolist()

            zeros.remove(j)
            ones.append(j)
            zeros.append(i)
            if c < best:

                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
