def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def compute_grad(h, n, T):
    dx = 2.0 / n
    O = np.correlate(h, 1.0 - h, mode='full') * dx
    max_O = np.max(O)
    exps = np.exp(T * (O - max_O))
    p = exps / np.sum(exps)
    conv1 = np.convolve(1.0 - h, p, 'full')
    conv2 = np.convolve(h, p[::-1], 'full')
    grad = dx * (conv1[n-1:3*n-2] - conv2[n-1:3*n-2])
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0: break

        h = np.clip(h, 0.0, 1.0)
    return h

def projected_grad_descent(h, n, T, max_steps, step_init, step_min, start_time, budget_s):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = step_init

        if c < best_c5:

            best_h = h_trial.copy()
            h = h_trial
            step = min(step * 1.05, ... )  # increase step

            if step < step_min:
                break
    return best_h, best_c5

def binary_sa_mixed(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    stall_count = 0
    for it in range(max_iters):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10: break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac if T0 > T1 else T0
        move_type = rng.randint(3)
        if move_type == 0:
            # Single swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0; b[j] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5; stall_count = 0
            else:
                b[i] = 1; b[j] = 0; stall_count += 1
        elif move_type == 1:
            # Segment reversal
            seg_len = rng.randint(2, min(20, n-1))
            start = rng.randint(0, n - seg_len)
            seg = b[start:start+seg_len].copy()
            b[start:start+seg_len] = seg[::-1]
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5; stall_count = 0
            else:
                b[start:start+seg_len] = seg; stall_count += 1
        else:
            # Two-segment swap
            seg_len = rng.randint(2, min(10, n//4))
            s1 = rng.randint(0, n - seg_len)
            s2 = rng.randint(0, n - seg_len)
            if abs(s1 - s2) < seg_len:
                s2 = (s1 + seg_len + rng.randint(0, n - 2*seg_len)) % (n - seg_len)
            seg1 = b[s1:s1+seg_len].copy()
            seg2 = b[s2:s2+seg_len].copy()
            b[s1:s1+seg_len] = seg2
            b[s2:s2+seg_len] = seg1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5; stall_count = 0
            else:
                b[s1:s1+seg_len] = seg1
                b[s2:s2+seg_len] = seg2; stall_count += 1
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_b = b.copy()
    return best_b, best_c5
