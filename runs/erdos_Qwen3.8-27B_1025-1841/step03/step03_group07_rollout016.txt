```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def grad_at_shift(h, sh):
    n = len(h)
    grad = np.ones(n)
    if sh >= 0:
        if sh < n:
            grad[sh:] -= h[:n-sh]
            grad[:n-sh] -= h[sh:]
    else:
        s = -sh
        if s < n:
            grad[:n-s] -= h[s:]
            grad[s:] -= h[:n-s]
    return grad

def multi_lag_grad(h, corr, top_k):
    """Compute gradient as weighted sum of gradients at top_k lags"""
    n = len(h)
    top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[top_idx]
    # softmax weights
    w = np.exp(top_vals - top_vals.max())
    w /= w.sum()
    
    grad = np.zeros(n)
    for i in range(top_k):
        k = top_idx[i]
        sh = k - n + 1
        grad += w[i] * grad_at_shift(h, sh)
    return grad

def nesterov_multi(h, n, time_limit, tau_init, tau_final, top_k, lr_init):
    """Nesterov with multi-lag softmax gradient, annealed tau"""
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    lr = lr_init
    total_iters = 100000  # will break by time
    
    for it in range(total_iters):
        if time.time() - start > time_limit:
            break
        
        # Anneal tau
        frac = min(1.0, it / (total_iters * 0.5))
        tau = tau_init + (tau_final - tau_init) * frac
        
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.85)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_now = corr.max() * dx
        
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = y.copy()
        
        # Multi-lag gradient with temperature tau
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        top_vals = corr[top_idx]
        shifted = (top_vals - top_vals.max()) * tau
        w = np.exp(np.clip(shifted, -50, 0))
        w /= w.sum()
        
        grad = np.zeros(n)
        for i in range(top_k):
            sh = top_idx[i] - n + 1
            grad += w[i] * grad_at_shift(y, sh)
        grad *= dx
        
        h_new = project_to_feasible(y - lr / (1 + t*0.01) * grad, n)
        
        t = 0.5 * (1 + np.sqrt(1 + 4*t*t))
        h_prev = h
        h = h_new
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def nesterov_argmax(h, n, time_limit, lr_init):

def coordinate_descent(h, n, time_limit, step=0.01):
    """Coordinate descent polish"""
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
```

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def grad_at_shift(h, sh):
    n = len(h)
    grad = np.ones(n)
    if sh >= 0:
        if sh < n:
            grad[sh:] -= h[:n - sh]
            grad[:n - sh] -= h[sh:]
    else:
        s = -sh
        if s < n:
            grad[:n - s] -= h[s:]
            grad[s:] -= h[:n - s]
    return grad


def nesterov_argmax(h0, n, time_limit, lr_init=0.03):
    dx = 2.0 / n
    h = project_to_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    lr = lr_init
    stall = 0

    for it in range(1000000):
        rem = time_limit - (time.time() - start)
        if rem < 0.5:
            break

        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.85)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_now = corr.max() * dx
        k_star = int(np.argmax(corr))
        sh = k_star - n + 1

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            lr *= 0.7
            h_prev = h.copy()
            stall = 0
            if lr < 1e-16:
                break

        grad = grad_at_shift(y, sh) * dx
        h_new = project_to_feasible(y - lr * grad, n)

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def nesterov_multilag(h0, n, time_limit, top_k=15, tau_init=80.0, tau_final=8.0, lr_init=0.06):
    dx = 2.0 / n
    h = project_to_feasible(h0.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    lr = lr_init
    stall = 0
    max_iter_snap = 50000

    for it in range(1000000):
        rem = time_limit - (time.time() - start)
        if rem < 0.5:
            break

        frac = min(1.0, it / max_iter_snap)
        tau = tau_init + (tau_final - tau_init) * frac

        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.85)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        c5_now = corr.max() * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = y.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            lr *= 0.75
            h_prev = h.copy()
            stall = 0
            if lr < 1e-16:
                break

        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        top_vals = corr[top_idx]
        shifted = (top_vals - top_vals.max()) * tau
        w = np.exp(np.clip(shifted, -40.0, 0.0))
        w /= w.sum()

        grad = np.zeros(n)
        for i in range(top_k):
            sh = top_idx[i] - n + 1
            grad += w[i] * grad_at_shift(y, sh)
        grad *= dx

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
```
