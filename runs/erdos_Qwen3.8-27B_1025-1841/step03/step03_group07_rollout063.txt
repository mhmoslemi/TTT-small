def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def corr_shift_grad(h, n, sh):
    dx = 2.0 / n
    if sh >= 0:
        i_lo, i_hi = 0, n - 1 - sh
    else:
        i_lo, i_hi = -sh, n - 1
    if i_lo > i_hi:
        return np.zeros(n)
    grad = np.zeros(n)
    idx_a = np.arange(i_lo, i_hi + 1)
    grad[idx_a] += (1.0 - h[idx_a + sh])
    idx_b = np.arange(i_lo + sh, i_hi + sh + 1)
    if idx_b.size > 0:
        grad[idx_b] += -h[idx_b - sh]
    return grad * dx

def topk_subgrad(h, n, top_k=3, weight_power=2.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    # Weight by proximity to max
    deltas = top_vals - max_c  # ≤ 0
    # Soft weight: exp(deltas / scale) with small scale
    scale = max((top_vals[0] - top_vals[-1]) * 0.1, 1e-10)
    w = np.exp(deltas / scale)
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        sh = k - (n - 1)
        grad += wi * corr_shift_grad(h, n, sh)
    return grad

def nesterov_opt(h, n, time_limit, top_k=3, lr=0.03, weight_power=2.0):
    start = time.time()
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    
    while time.time() - start < time_limit:
        mu = min((t - 1.0) / (t + 2.0), 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        grad = topk_subgrad(y, n, top_k, weight_power)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        # Periodic evaluation
        c5_now = compute_c5(h_new)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 1000:
            lr *= 0.6
            stall = 0
            if lr < 1e-12:
                break
        
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    
    c5_f = compute_c5(h)
    if c5_f < best_c5:
        best_c5, best_h = c5_f, h.copy()
    return best_h, best_c5

def local_swap(h, n, time_limit, rng):

            h = h_test.copy()
    
    return best_h, best_c5

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx

def shift_gradient(h, n, sh):
    """Exact gradient of c(sh) w.r.t. h, scaled by dx."""
    dx = 2.0 / n
    if sh >= 0:
        i_lo, i_hi = 0, n - 1 - sh
    else:
        i_lo, i_hi = -sh, n - 1
    if i_lo > i_hi:
        return np.zeros(n)
    grad = np.zeros(n)
    idx_a = np.arange(i_lo, i_hi + 1)
    grad[idx_a] += (1.0 - h[idx_a + sh])
    idx_b = np.arange(i_lo + sh, i_hi + sh + 1)
    if idx_b.size > 0:
        grad[idx_b] += -h[idx_b - sh]
    return grad * dx

def topk_weighted_grad(h, n, top_k=3):
    """Weighted subgradient of max_k c(k)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    # Geometric weighting: more weight on closer-to-max
    spread = (top_vals[0] - top_vals[-1])
    if spread > 1e-12:
        w = np.exp(-2.0 * (top_vals[0] - top_vals) / spread)
    else:
        w = np.ones(top_k) / top_k
    w /= w.sum()
    
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        sh = k - (n - 1)
        grad += wi * shift_gradient(h, n, sh)
    return grad, max_c, top_idx

    best_h = h.copy()

    stall = 0

        y = h + mu * (h - h_prev)

    if c5_f < best_c5:
        best_c5, best_h = c5_f, h.copy()
