def corr_and_grad(h, T):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Log-sum-exp
    mc = np.max(corr)
    shifted = T * (corr - mc)
    log_sum = mc + np.log(np.sum(np.exp(shifted)))
    obj = log_sum / T * dx
    
    # Softmax weights
    p = np.exp(shifted) / np.sum(np.exp(shifted))
    
    # Gradient: grad[j] = sum_k p[k] * dc_k/dh[j]
    # dc_k/dh[j] = (1-h[j+d_k])*I(valid) - h[j-d_k]*I(valid) where d_k = k-(n-1)
    grad_c = np.zeros(n)
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-14:
            continue
        d = k - (n - 1)
        j_lo = max(0, -d)
        j_hi = min(n, n - d)
        if j_lo >= j_hi:
            continue
        idx1 = slice(j_lo + d, j_hi + d)
        idx2 = slice(j_lo - d, j_hi - d)
        grad_c[j_lo:j_hi] += wk * ((1.0 - h[idx1]) - h[idx2])
    
    return obj, grad_c * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = T * (corr - mc)
    exp_s = np.exp(s)
    log_sum = mc + np.log(np.sum(exp_s))
    obj = log_sum / T * dx
    p = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-13:
            continue
        d = k - (n - 1)
        lo = max(0, -d)
        hi = min(n, n - d)
        if lo >= hi:
            continue
        grad[lo:hi] += wk * ((1.0 - h[lo + d:hi + d]) - h[lo - d:hi - d])
    return obj, grad * dx

def project_feasible(h_raw, n):
    target = n / 2.0
    lo_lam = -np.max(h_raw) - 1.0
    hi_lam = np.max(h_raw) - np.min(h_raw) + 1.0
    for _ in range(80):
        mid = (lo_lam + hi_lam) / 2.0
        h_try = np.clip(h_raw - mid, 0.0, 1.0)
        if np.sum(h_try) > target:
            lo_lam = mid
        else:
            hi_lam = mid
    return np.clip(h_raw - (lo_lam + hi_lam) / 2.0, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_from_continuous(h, n):
    target = n // 2
    sorted_idx = np.argsort(-h)
    b = np.zeros(n)
    b[sorted_idx[:target]] = 1.0
    return b

def m_sequence(n, m=10):
    L = (1 << m) - 1
    period = 1 << (m - 1)
    seq = np.zeros(L)
    for k in range(L):
        t = bin(k).count('1')
        seq[k] = 0.0 if t % 2 == 0 else 1.0
    if n <= L:
        return seq[:n]
    reps = int(np.ceil(n / L))
    full = np.tile(seq, reps)[:n]
    ones = np.sum(full)
    if ones > n / 2:
        excess = int(ones - n / 2)
        one_positions = np.where(full == 1.0)[0]
        to_flip = one_positions[:excess]
        full[to_flip] = 0.0
    elif ones < n / 2:
        deficit = int(n / 2 - ones)
        zero_positions = np.where(full == 0.0)[0]
        to_flip = zero_positions[:deficit]
        full[to_flip] = 1.0
    return full

def slsqp_run(h0, n, T, maxiter, seed=42):
    bounds = [(0.0, 1.0)] * n
    def objfun(x):
        o, _ = smooth_obj_grad(x, T)
        return o
    def gradfun(x):
        _, g = smooth_obj_grad(x, T)
        return g
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(objfun, h0, jac=gradfun, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return make_feasible(res.x, n)
    except Exception:
        return h0

def projected_gradient(h0, n, T, n_steps, seed=42):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = 0.1
    for step in range(n_steps):
        obj, grad = smooth_obj_grad(h, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        h_new = project_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new)
        if c_new < compute_c5(h) - 1e-14:
            h = h_new
            lr = min(lr * 1.1, 0.5)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            lr *= 0.7
    return best_h, best_c

def simulated_annealing_binary(b, n, n_steps, T_start, T_end, seed=42):
    rng = np.random.RandomState(seed)
    h = b.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    one_pos = np.where(h == 1.0)[0]
    zero_pos = np.where(h == 0.0)[0]
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        i_idx = rng.randint(len(one_pos))
        j_idx = rng.randint(len(zero_pos))
        i_pos = one_pos[i_idx]
        j_pos = zero_pos[j_idx]
        h[i_pos], h[j_pos] = h[j_pos], h[i_pos]
        c_new = compute_c5(h)
        delta = c_new - c_curr
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            c_curr = c_new
            one_pos = np.where(h == 1.0)[0]
            zero_pos = np.where(h == 0.0)[0]
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[i_pos], h[j_pos] = h[j_pos], h[i_pos]
    return best_h, best_c

def local_swap_search(h, n, n_rounds, seed=42):
    rng = np.random.RandomState(seed)
    h = h.copy()
    c_curr = compute_c5(h)
    best_h = h.copy()
    best_c = c_curr
    for rnd in range(n_rounds):
        improved = False
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        n_trials = min(2000, len(ones) * len(zeros))
        for _ in range(n_trials):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c_new = compute_c5(h)
            if c_new < c_curr - 1e-13:
                c_curr = c_new
                improved = True
                if c_new < best_c:
                    best_c = c_new
                    best_h = h.copy()
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return best_h, best_c

def coordinate_descent(h, n, n_rounds):
    h = h.copy()
    c_curr = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_val = old
            best_c = c_curr
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old - 0.1), max(0.0, old - 0.05), max(0.0, old - 0.01),
                     min(1.0, old + 0.01), min(1.0, old + 0.05), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                h[j] = cand
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    best_val = cand
            h[j] = best_val
            h = make_feasible(h, n)
            if best_val != old:
                c_curr = best_c
                improved = True
        if not improved:
            break
    return h, c_curr
