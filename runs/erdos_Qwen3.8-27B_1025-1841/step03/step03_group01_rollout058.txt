def compute_c5(h):
    n = len(h)

    target = n / 2.0

    diff = target - np.sum(h)

        idx = np.argmin(h) if diff > 0 else np.argmax(h)

    return h

def upsample(h, new_n):

    # For each j, valid k are those with 0 <= j - p_k < n

    for j in range(n):

```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15: break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0: break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def obj_and_grad(h, T, n):
    """Soft-max objective and gradient for the C5 minimization."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    c_max = np.max(corr)
    shifted = T * (corr - c_max)
    shifted -= np.max(shifted)  # numerical stability
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    s = exp_shifted / sum_exp  # softmax weights
    
    # Objective
    obj = (c_max + np.log(sum_exp) / T) * dx
    
    # Gradient
    cs = np.concatenate([[0.0], np.cumsum(s)])
    contrib1a = np.zeros(n)
    for j in range(n):
        lo = max(0, n + 1 - j)
        hi = min(2*n - 1, n + j + 1)
        contrib1a[j] = cs[hi] - cs[lo]
    
    conv_hs = fftconvolve(h, s)  # length 3n-1
    conv_fhs = fftconvolve(h[::-1], s)  # length 3n-1
    
    contrib1b = conv_hs[n-1 : 2*n-1]  # length n
    T2 = conv_fhs[2*n-2 : n-1 : -1]  # length n
    
    grad = dx * (contrib1a - contrib1b - T2)
    
    return obj, grad
```
