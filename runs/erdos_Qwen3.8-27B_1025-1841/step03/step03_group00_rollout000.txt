            h_sa3, c5_sa3 = sa_binary(h_b2, n1, rng, T_start=0.004, T_end=0.0001, n_iter=80000, max_rounds=2)
            if c5_sa3 < best_c5:
                best_c5, best_h = c5_sa3, h_sa3.copy()

    # Phase 4: Progressive resolution - n=1024
    n2 = 1024
    if tl() > 100:
        h_c = upsample_h(best_h, n2)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()
        for T in [30, 100, 400, 1200, 4000]:
            if tl() < 25:
                break
            h_opt = lbgfb_opt(h_c, n2, T, mu=300.0, max_iter=200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
                h_c = h_opt.copy()
        # SA at n=1024
        if tl() > 60:
            h_b = (h_c > 0.5).astype(float)
            h_b = make_feasible(h_b, n2)
            h_sa4, c5_sa4 = sa_binary(h_b, n2, rng, T_start=0.006, T_end=0.0002, n_iter=120000, max_rounds=3)
            if c5_sa4 < best_c5:
                best_c5, best_h = c5_sa4, h_sa4.copy()
            # Continuous polish from SA result
            h_opt2 = lbgfb_opt(h_sa4, n2, 1000, mu=300.0, max_iter=150)
            c5_opt2 = compute_c5(h_opt2)
            if c5_opt2 < best_c5:
                best_c5, best_h = c5_opt2, h_opt2.copy()

    # Phase 5: n=2048
    n3 = 2048
    if tl() > 80:
        h_c = upsample_h(best_h, n3)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()
        for T in [50, 300, 1000, 3000]:
            if tl() < 25:
                break
            h_opt = lbgfb_opt(h_c, n3, T, mu=200.0, max_iter=150)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
                h_c = h_opt.copy()
        if tl() > 40:
            h_b = (h_c > 0.5).astype(float)
            h_b = make_feasible(h_b, n3)
            h_sa5, c5_sa5 = sa_binary(h_b, n3, rng, T_start=0.005, T_end=0.0001, n_iter=100000, max_rounds=3)
            if c5_sa5 < best_c5:
                best_c5, best_h = c5_sa5, h_sa5.copy()

    # Phase 6: n=4096 quick pass
    n4 = 4096
    if tl() > 50:
        h_c = upsample_h(best_h, n4)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()
        for T in [100, 500, 2000]:
            if tl() < 20:
                break
            h_opt = lbgfb_opt(h_c, n4, T, mu=200.0, max_iter=100)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
                h_c = h_opt.copy()

    # Phase 7: Coordinate descent at current best resolution
    n_cd = len(best_h)
    if n_cd > 4096:
        h_cd = upsample_h(best_h, 4096)
        n_cd = 4096
    else:
        h_cd = best_h.copy()
    if tl() > 30:
        h_cd, c5_cd = coord_descent(h_cd, n_cd, rng, n_rounds=40, time_limit_fn=tl)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 8: SLSQP final polish
    n_f = min(len(best_h), 2048)
    h_f = upsample_h(best_h, n_f) if n_f < len(best_h) else best_h.copy()
    if tl() > 20:
        h_opt = slsqp_opt(h_f, n_f, 2000, max_iter=80)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5, best_h = c5_opt, h_opt.copy()

    # Phase 9: Final coordinate descent squeeze
    if tl() > 20:
        n_cd2 = min(len(best_h), 2048)
        h_cd2 = upsample_h(best_h, n_cd2) if n_cd2 < len(best_h) else best_h.copy()
        h_cd2, c5_cd2 = coord_descent(h_cd2, n_cd2, rng, n_rounds=30, time_limit_fn=tl)
        if c5_cd2 < best_c5:
            best_c5, best_h = c5_cd2, h_cd2.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
