def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            h[:] = target / n
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def soft_max_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def soft_max_grad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(T * shifted)

    return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_jacobian}]
    
    def obj(x):
        return soft_max_obj(x, T, n)
    
    def grad(x):
        return soft_max_grad(x, T, n)
    
    res = minimize(obj, h.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

                best_b = b.copy()

            b[j] = 0

        if time.time() - start_time > budget_s - reserve:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        block_size = rng.randint(2, 8)
        start = rng.randint(0, max(1, n - block_size))
        end = start + block_size
        seg = b[start:end]
        ones_in = np.where(seg == 1)[0]
        zeros_in = np.where(seg == 0)[0]
        if len(ones_in) == 0 or len(zeros_in) == 0:
            continue
        swap_cnt = min(len(ones_in), len(zeros_in))
        chosen_ones = rng.choice(ones_in, size=swap_cnt, replace=False)
        chosen_zeros = rng.choice(zeros_in, size=swap_cnt, replace=False)
        b[start + chosen_ones] = 0
        b[start + chosen_zeros] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + chosen_ones] = 1
            b[start + chosen_zeros] = 0
    return best_b, best_c5

def multi_temp_slsqp(h, n, temps, maxiter_each, ftol, start_time, budget_s, reserve):
    cur = h.copy()
    cur_c5 = compute_c5(cur)
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < reserve + 40:
            break
        cur = slsqp_soft(cur, n, T, maxiter_each, ftol, start_time, budget_s, reserve)
        c5 = compute_c5(cur)
        if c5 < cur_c5:
            cur_c5 = c5
    return cur, cur_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binary SA (point swaps) - quick
    b0, c5_b0 = binary_sa(h_init, n0, 40000, 0.012, 0.00001, rng, start_time, budget_s, 800)
    if c5_b0 < best_c5:
        best_c5 = c5_b0
        best_h = b0.copy()

    # Phase 2: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 600:
        b1, c5_b1 = binary_sa_block(best_h, n0, 40000, 0.008, 0.00002, rng, start_time, budget_s, 600)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = b1.copy()

    # Phase 3: Multi-temperature SLSQP with exact gradient
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        temps = [30.0, 80.0, 200.0, 500.0, 1200.0]
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, temps, 250, 1e-14, start_time, budget_s, 400)

    # Phase 4: Binary refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        b2, c5_b2 = binary_sa(best_h, n0, 50000, 0.004, 0.000003, rng, start_time, budget_s, 250)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = b2.copy()

    # Phase 5: High-temperature SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        temps = [300.0, 800.0, 2000.0]
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, temps, 200, 1e-15, start_time, budget_s, 150)

    # Phase 6: Try alternative resolutions
    try_resolutions = [480, 640, 720, 960, 1120, 1280]
    for n_try in try_resolutions:
        if n_try == n0:
            continue
        remaining = budget_s - (time.time() - start_time)
        if remaining < 250:
            break
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, n_try)
        h_try = np.interp(new_x, old_x, best_h)
        h_try = make_feasible(h_try, n_try)

        b_try, c5_try = binary_sa(h_try, n_try, 25000, 0.01, 0.00001, rng, start_time, budget_s, 200)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = b_try.copy()

        remaining = budget_s - (time.time() - start_time)
        if remaining > 150:
            h_start = best_h if len(best_h) == n_try else make_feasible(np.interp(new_x, old_x, best_h), n_try)
            best_h_r, best_c5_r = multi_temp_slsqp(h_start, n_try, [50.0, 200.0, 600.0], 200, 1e-14,
                                                    start_time, budget_s, 100)
            if best_c5_r < best_c5:
                best_c5 = best_c5_r
                best_h = best_h_r.copy()

    # Phase 7: Final block SA at best resolution
    n_f = len(best_h)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        b_f, c5_f = binary_sa_block(best_h, n_f, 30000, 0.004, 0.00001, rng, start_time, budget_s, 100)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = b_f.copy()

    # Phase 8: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        best_h, best_c5 = multi_temp_slsqp(best_h, len(best_h), [500.0, 1500.0, 4000.0], 150, 1e-15,
                                           start_time, budget_s, 50)

    # Phase 9: Final binary check
    b_fin = to_binary(best_h, len(best_h))
    c5_fin = compute_c5(b_fin)
    if c5_fin < best_c5:
        best_c5 = c5_fin
        best_h = b_fin

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
