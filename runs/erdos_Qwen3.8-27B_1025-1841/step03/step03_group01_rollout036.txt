```python
_OPT_N = 0
_OPT_T = 1.0

def _obj_fn(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Use a safer log-sum-exp
    m = np.max(shifted)
    log_sum = m + np.log(np.sum(np.exp(T * (shifted - m))))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _cons_jac(h):
    return np.ones(_OPT_N)

def slsqp_optimize(h_start, n, T=200.0, max_iter=200):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_fn, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except:
        return h_start
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

_OPT_N = 0
_OPT_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def build_MA(n, vec):
    A = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = n - 1 + k
        if k >= 0:
            A[row, :n - k] = vec[k:]
        else:
            A[row, -k:] = vec[:n + k]
    return A

def build_MB(n, vec):
    A = np.zeros((2 * n - 1, n))
    for k in range(-(n - 1), n):
        row = n - 1 + k
        if k >= 0:
            A[row, k:n] = vec[:n - k]
        else:
            A[row, :n + k] = vec[-k:]
    return A

def solve_lp(C, n):
    dx = 2.0 / n
    x_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        C @ x_var <= t_var / dx,
        x_var >= 0,
        x_var <= 1,
        cp.sum(x_var) == n / 2.0
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, verbose=False, max_iters=20000)
    except Exception:
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=50000)
        except Exception:
            return None
    if x_var.value is None:
        return None
    return np.clip(x_var.value, 0.0, 1.0)

def bcd_optimize(h_init, n, max_iter=300, time_limit=200):
    start = time.time()
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stall = 0

    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        frac = it / max(max_iter - 1, 1)
        alpha = 1.0 - 0.4 * frac

        g = 1.0 - h
        MA = build_MA(n, g)
        h_lp = solve_lp(MA, n)
        if h_lp is not None:
            h = (1 - alpha) * h + alpha * h_lp
            h = make_feasible(h, n)

        MB = build_MB(n, h)
        g_lp = solve_lp(MB, n)
        if g_lp is not None:
            h = (1 - alpha) * h + alpha * (1.0 - g_lp)
            h = make_feasible(h, n)

        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 80:
            break

    return best_h, best_c5

def _obj_fn(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    m = np.max(shifted)
    log_sum = m + np.log(np.sum(np.exp(T * (shifted - m))))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _cons_jac(h):
    return np.ones(_OPT_N)

def slsqp_optimize(h_start, n, T=200.0, max_iter=200):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_fn, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        h_out = make_feasible(res.x, n)
        return h_out
    except Exception:
        return h_start
```
