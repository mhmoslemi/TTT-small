This is getting really complicated. Let me just use the O(n²) Python loop and accept the ~1s per gradient evaluation. With 50 SLSQP iterations, that's 50s per T value. With 3 T values, that's 150s. That's a lot but manageable within 1300s budget.

Actually wait, I can vectorize the inner loop! For each k, the operation is:
- m in [m_lo, m_hi]: grad[m] += σ_k * (1-h[m+s]) and grad[m+s] -= σ_k * h[m]

For a given k with shift s:
- m_lo = max(0, -s), m_hi = min(n-1, n-1-s)
- grad[m_lo:m_hi+1] += σ_k * (1 - h[m_lo+s:m_hi+1+s])
- grad[m_lo+s:m_hi+1+s] -= σ_k * h[m_lo:m_hi+1]

This is vectorized! The total work is still O(n²) but in vectorized numpy operations, much faster.

```python
def logsumexp_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = T * (corr - max_c)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    sigma = exp_vals / sum_exp
    
    obj = (max_c + np.log(sum_exp) / T) * dx
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        sk = sigma[k]
        if sk < 1e-14:
            continue
        s = k - (n - 1)
        m_lo = max(0, -s)
        m_hi = min(n-1, n-1-s)
        if m_lo > m_hi:
            continue
        seg = h[m_lo:m_hi+1]
        grad[m_lo:m_hi+1] += sk * (1.0 - seg)
        # h[m+s] for m in [m_lo, m_hi]: indices [m_lo+s, m_hi+s]
        lo2 = m_lo + s
        hi2 = m_hi + s
        # These should be in [0, n-1] by construction
        grad[lo2:hi2+1] -= sk * seg
    
    grad *= dx / T
    return obj, grad
```

This loop runs 2n-1 = 1919 times, each with a vectorized numpy operation on at most n elements. Total numpy operations: still O(n²) but vectorized. Should be a few ms per gradient.

import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        b[zeros[order[:diff]]] = 1
    elif diff < 0:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        b[ones[order[:(-diff)]]] = 0
    return b

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def downsamp_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_sa(h, n, max_iters, T0, T1, rng, block_frac=0.3):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    ones_arr = np.where(b == 1)[0]
    zeros_arr = np.where(b == 0)[0]
    refresh_interval = 5000
    for it in range(max_iters):
        if it % refresh_interval == 0:
            ones_arr = np.where(b == 1)[0]
            zeros_arr = np.where(b == 0)[0]
        T = T0 * (T1 / T0) ** (float(it) / max_iters)
        accepted = False
        if rng.random() < block_frac:
            bs = rng.randint(2, 8)
            s = rng.randint(0, n - bs)
            seg = b[s:s+bs]
            oi = np.where(seg == 1)[0]
            oz = np.where(seg == 0)[0]
            if len(oi) == 0 or len(oz) == 0:
                continue
            sw = min(len(oi), len(oz), 4)
            ci = rng.choice(oi, sw, replace=False)
            cj = rng.choice(oz, sw, replace=False)
            b[s+ci] = 0
            b[s+cj] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5)/T)):
                curr_c5 = new_c5
                accepted = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[s+ci] = 1
                b[s+cj] = 0
        else:
            if len(ones_arr) == 0 or len(zeros_arr) == 0:
                continue
            i = ones_arr[rng.randint(len(ones_arr))]
            j = zeros_arr[rng.randint(len(zeros_arr))]
            if b[i] != 1 or b[j] != 0:
                continue
            b[i] = 0
            b[j] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5)/T)):
                curr_c5 = new_c5
                accepted = True
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i] = 1
                b[j] = 0
    return best_b, best_c5

def lse_obj_grad(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    max_c = np.max(corr)
    shifted = T * (corr - max_c)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    sum_exp = np.sum(exp_vals)
    sigma = exp_vals / sum_exp
    obj = (max_c + np.log(sum_exp) / T) * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        sk = sigma[k]
        if sk < 1e-14:
            continue
        s = k - (n - 1)
        m_lo = max(0, -s)
        m_hi = min(n - 1, n - 1 - s)
        if m_lo > m_hi:
            continue
        seg = x[m_lo:m_hi + 1]
        grad[m_lo:m_hi + 1] += sk * (1.0 - seg)
        lo2 = m_lo + s
        hi2 = m_hi + s
        grad[lo2:hi2 + 1] -= sk * seg
    grad *= dx / T
    return obj, grad

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_lse(h_init, n, T, maxiter=200, ftol=1e-14):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(lse_obj_grad, h_init.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x if res.success or res.x is not None else h_init.copy()

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.02), max(0.0, old - 0.005),
                     min(1.0, old + 0.005), min(1.0, old + 0.02)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def multi_start_binary_sa(rng, n, num_starts, max_iters_each, T0, T1):
    best_b = None
    best_c5 = 1e9
    for s in range(num_starts):

            h_rand = np.full(n, 0.5)
