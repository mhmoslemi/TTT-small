import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(h) if diff > 0 else np.argmin(h)  # wrong direction
        # If diff > 0, we need to add, so pick the one closest to 1
        # If diff < 0, we need to subtract, so pick the one closest to 0
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # For numerical stability, use log-sum-exp
    max_shifted = np.max(shifted)  # should be 0
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    obj = (mc + log_sum / T) * dx
    # Penalty for sum constraint
    penalty = mu * (np.sum(h) - n / 2.0) ** 2
    return obj + penalty

def lBFGS_optimize(h_start, n, T, mu, max_iter=300):
    bounds = [(0.0, 1.0)] * n
    res = minimize(soft_max_obj, h_start, args=(n, T, mu),
                   method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
    return make_feasible(res.x, n)

For larger n, the structure matters more. The key insight is:
- We want to minimize the max over ALL lags of the cross-correlation
- At lag 0: sum h[i](1-h[i]) = n/2 - sum h[i]². For binary, this is 0.
- At intermediate lags: this involves the "structure" of h.

The optimal h should be as "smooth" as possible to avoid sharp transitions (which create large values at specific lags), but also as "peaked" as possible (to minimize the lag-0 contribution).

This is a tension: peaked (binary) minimizes lag 0 but creates large peaks at transition lags. Smooth (all 0.5) minimizes all lags uniformly but has a large lag-0 value.

The optimal solution balances these. For the continuous problem, the optimal h is likely a smooth function that's close to 0 or 1 but with smooth transitions.

Given this understanding, the previous approach of starting from [1,0,1,0,...] and optimizing with SLSQP makes sense - the optimizer smooths out the transitions while maintaining the 0-1 structure.

    n = len(h)

    size = 1
    while size < n + m - 1:
