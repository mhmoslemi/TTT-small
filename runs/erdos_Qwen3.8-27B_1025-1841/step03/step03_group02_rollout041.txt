            break

        ac = np.correlate(y, y, mode='full')
        if top_k == 1:
            d_star = int(np.argmin(ac)) - (n - 1)
            sg = shift_pad(y, d_star) + shift_pad(y, -d_star)
        else:
            sorted_idx = np.argpartition(ac, top_k)[:top_k]
            vals = ac[sorted_idx]
            w = 1.0 / (vals + 1e-10)
            w /= w.sum()
            sg = np.zeros(n)
            for wi, idx in zip(w, sorted_idx):
                d = int(idx) - (n - 1)
                sg += wi * (shift_pad(y, d) + shift_pad(y, -d))
        
        # Nesterov update
        t_k = 2.0 / (1.0 + np.sqrt(1.0 + 4.0 * it * it))
        y_new = make_feasible(y - alpha * sg, n)
        h_new = make_feasible((1 - t_k) * y_new + t_k * y, n)
        # Hmm, this isn't quite right for the projection...
        
        # Actually, let me use the simpler "heavy ball" method:
        # y = h
        # v = beta*v + alpha*sg
        # h_new = project(h + v)
        
    return best_h, best_c5

The terms b[i+d] and b[i-d] are fixed for a given i. The terms b[j+d] and b[j-d] depend on j.

So Δ_i,j[d] = (b[i+d] + b[i-d]) - (b[j+d] + b[j-d])

Let A_i = b[i+d] + b[i-d] (as a function of d, this is a vector of length 2n-1)
Let B_j = b[j+d] + b[j-d] (similarly)

Then Δ_i,j[d] = A_i[d] - B_j[d]

And corr_new[d] = corr[d] + A_i[d] - B_j[d]

We want to minimize max_d (corr[d] + A_i[d] - B_j[d]).

For a fixed i, we want to find j that minimizes this max. This is still O(n) per j if we compute the max naively, giving O(n^2) total for fixed i, and O(n^3) overall.

But we can do better: for a fixed i, the function we're minimizing over j is:
f(j) = max_d (corr[d] + A_i[d] - B_j[d])

Note that B_j[d] = b[j+d] + b[j-d]. For binary b, this is 0, 1, or 2.

The function f(j) is the max over d of (corr[d] + A_i[d] - B_j[d]). Since B_j[d] ∈ {0,1,2}, this is approximately corr[d] + A_i[d] (with small perturbations).

def local_search_binary_fast(b, n, max_rounds, K, start_time, budget_s, rng):
    """Local search with K random candidates per round."""
    b = b.copy()
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 20:
            break
        
        corr = np.correlate(b, 1.0 - b, mode='full')
        corr_max = np.max(corr)
        
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        
        # Sample K random (i, j) pairs
        n_pairs = min(K, len(ones) * len(zeros))
        i_idx = rng.randint(0, len(ones), n_pairs)
        j_idx = rng.randint(0, len(zeros), n_pairs)
        i_arr = ones[i_idx]
        j_arr = zeros[j_idx]

        for p in range(n_pairs):
            i = i_arr[p]
            j = j_arr[p]
            # Compute A_i = b[i+d] + b[i-d] for all d
            # d ranges from -(n-1) to (n-1), indexed as k = n-1+d in [0, 2n-2]
            # b[i+d] is valid when 0 <= i+d < n, i.e., -i <= d < n-i
            # b[i-d] is valid when 0 <= i-d < n, i.e., i-n < d <= i

            Ai = np.zeros(2*n-1)
            # b[i+d] for valid d: d from max(-(n-1), -i) to min(n-1, n-1-i)

            d_max1 = min(n-1, n-1-i)
            if d_min1 <= d_max1:
                Ai[n-1+d_min1 : n-1+d_max1+1] = b[i+d_min1 : i+d_max1+1]

            d_min2 = max(-(n-1), i-(n-1))
            d_max2 = min(n-1, i)
            if d_min2 <= d_max2:

            Bj = np.zeros(2*n-1)
            d_min1 = max(-(n-1), -j)

            if d_min1 <= d_max1:
                Bj[n-1+d_min1 : n-1+d_max1+1] = b[j+d_min1 : j+d_max1+1]
            
            d_min2 = max(-(n-1), j-(n-1))

            if d_min2 <= d_max2:

            if new_c5 < best_c5:

                best_b = b.copy()
        else:
            break  # No improvement
    
    return best_b, best_c5
