def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    if target * 2 != n:
        target = n // 2  # handle odd n
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    tabu = {}
    for it in range(iters):
        T = T0 * (T1/T0) ** (it/iters)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        key = (min(i,j), max(i,j))

        b[i1], b[j1] = 0.0, 1.0
        b[i2], b[j2] = 0.0, 1.0
        new = compute_c5(b)
        if new <= curr or (T > 1e-15 and rng.random() < np.exp(-(new-curr)/T)):

            if new < best:

                best_b = b.copy()
        else:
            b[i1], b[j1] = 1.0, 0.0
            b[i2], b[j2] = 1.0, 0.0
    return best_b, best

Let me also think about the SLSQP objective. The objective is:
f(h) = max_k sum_i h[i] * (1-h[i+k-(n-1)]) * dx

This is max of bilinear functions. To make it smooth, I can use the log-sum-exp trick:
f_smooth(h) = (1/T) * log(sum_k exp(T * corr[k])) * dx

But this changes the problem. For SLSQP, I'll just use the max directly (it's piecewise smooth, and SLSQP can handle it).

Actually, for SLSQP, the objective needs to be differentiable. The max of C¹ functions is C⁰ but not C¹ at kinks. However, in practice SLSQP handles this fine because the subgradient exists (it's the gradient of the active function).

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[np.argmin(h[ones])]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[np.argmax(h[zeros])]] = 1
    return b

def random_binary(n, rng):
    b = np.zeros(n)
    idx = rng.permutation(n)
    b[idx[:n // 2]] = 1.0
    return b

def sa_swap(b, n, iters, T0, T1, rng, t_start=None, t_limit=None):
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    stall = 0
    for it in range(iters):
        if t_start is not None and t_limit is not None and time.time() - t_start > t_limit:
            break
        T = T0 * (T1 / T0) ** (it / max(iters, 1))
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        new = compute_c5(b)
        if new <= curr + 1e-15 or (T > 1e-15 and rng.random() < np.exp(-(new - curr) / T)):
            curr = new
            stall = 0
            if new < best:
                best = new
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
            stall += 1
    return best_b, best

def sa_two_swap(b, n, iters, T0, T1, rng, t_start=None, t_limit=None):
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    for it in range(iters):
        if t_start is not None and t_limit is not None and time.time() - t_start > t_limit:
            break
        T = T0 * (T1 / T0) ** (it / max(iters, 1))
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        if i2 == i1:
            i2 = ones[(np.where(ones == i1)[0][0] + 1) % len(ones)]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if j2 == j1:
            j2 = zeros[(np.where(zeros == j1)[0][0] + 1) % len(zeros)]
        if j1 == i1 or j1 == i2 or j2 == i1 or j2 == i2:
            continue
        b[i1] = 0.0
        b[i2] = 0.0
        b[j1] = 1.0
        b[j2] = 1.0
        new = compute_c5(b)
        if new <= curr + 1e-15 or (T > 1e-15 and rng.random() < np.exp(-(new - curr) / T)):
            curr = new
            if new < best:
                best = new
                best_b = b.copy()
        else:
            b[i1] = 1.0
            b[i2] = 1.0
            b[j1] = 0.0
            b[j2] = 0.0
    return best_b, best

def slsqp_obj(x):
    n = _global_n
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    return float(np.max(corr)) * dx

def slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def run_slsqp(h, n, maxiter=300, ftol=1e-14):
    global _global_n
    _global_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    res = minimize(slsqp_obj, h.astype(float).copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, t_start, t_budget):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        if time.time() - t_start > t_budget - 5:
            break
        for j in range(n):
            if time.time() - t_start > t_budget - 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def guided_local_search(b, n, iters, rng, t_start, t_limit):
    """Local search focusing on critical shifts."""
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    for it in range(iters):
        if time.time() - t_start > t_limit:
            break
        corr = np.correlate(b, 1.0 - b, mode='full')
        top_shifts = np.argsort(corr)[-5:]
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        best_move = None
        best_new_c = curr
        for _ in range(20):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = compute_c5(b)
            if new_c < best_new_c:
                best_new_c = new_c
                best_move = (i, j)
            b[i] = 1.0
            b[j] = 0.0
        if best_move is not None:
            i, j = best_move
            b[i] = 0.0
            b[j] = 1.0
            curr = best_new_c
            if curr < best:
                best = curr
                best_b = b.copy()
        else:
            # Random kick
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = compute_c5(b)
            if new_c < curr * 1.001:
                curr = new_c
            else:
                b[i] = 1.0
                b[j] = 0.0
    return best_b, best
