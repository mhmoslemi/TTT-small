- Find the lag k* that achieves the max
- Try to reduce corr[k*] by swapping:
  - If h[i]=1 and h[i+k*]=0, changing h[i] to 0 and h[j] to 1 (for some j) might help
  - But we need to check the effect on ALL lags, not just k*

A smarter move: given the current corr vector, for a swap (i: 1→0, j: 0→1), the change in corr[k] is:
- corr[k] decreases by 1 if h[i]=1 and h[i+k]=0 (removing a "1→0" pair at lag k)
- corr[k] increases by 1 if h[j]=0 and h[j+k]=1 (adding a "1→0" pair at lag k), but wait, we're changing h[j] from 0 to 1, so...

Let me think more carefully. If we change h[i] from 1 to 0 and h[j] from 0 to 1:

Effect on corr[k] = sum_m h[m] * (1-h[m+k]):
- The term at m=i: was h[i]*(1-h[i+k]) = 1*(1-h[i+k]) = 1-h[i+k], now 0. Change: -(1-h[i+k]) = h[i+k]-1
- The term at m=i-k: was h[i-k]*(1-h[i]) = h[i-k]*0 = 0 (since h[i] was 1). Now h[i-k]*(1-0) = h[i-k]. Change: +h[i-k]
- The term at m=j: was h[j]*(1-h[j+k]) = 0. Now 1*(1-h[j+k]) = 1-h[j+k]. Change: +(1-h[j+k])
- The term at m=j-k: was h[j-k]*(1-h[j]) = h[j-k]*1 = h[j-k]. Now h[j-k]*0 = 0. Change: -h[j-k]

So total change in corr[k]:
Δcorr[k] = (h[i+k]-1) + h[i-k] + (1-h[j+k]) - h[j-k]
= h[i+k] - 1 + h[i-k] + 1 - h[j+k] - h[j-k]
= h[i+k] + h[i-k] - h[j+k] - h[j-k]

(with the convention that h[out_of_range] = 0)

This is a really nice formula! For binary h:
Δcorr[k] = [h[i+k] + h[i-k]] - [h[j+k] + h[j-k]]

So the change depends only on 4 values of h! This makes it very fast to evaluate the effect of a swap.

The max of corr changes based on whether the max lags are affected. If the current max is at lag k*, and we change corr[k*] by Δ, then:
- If Δ < 0 and no other lag exceeds the new corr[k*], the max decreases
- If Δ > 0, the max might increase

This gives us a very fast local search: for each swap, we can quickly compute the effect on the critical lags.

Even better: we can precompute for each position i the "fingerprint" f[i] = h[i+1] + h[i-1] (or more generally, the vector of h[i+k] for all k). Then the effect of swapping i and j on corr[k] is f_i[k] - f_j[k] where f_i[k] = h[i+k] + h[i-k].

For a practical implementation, I'll focus on the lags that are near the maximum (top few lags) and check if the swap helps.

```python
    curr_corr = np.correlate(b, 1.0 - b, mode='full')
    curr_c5 = np.max(curr_corr) * dx
    best_c5 = curr_c5

    zeros = np.where(b == 0)[0]

    T_end = 0.001 / dx

        T = T0 * (T_end / T0) ** frac

        j = zeros[rng.randint(len(zeros))]

        # where shifted_b_i_plus[k] = b[i+k] if valid else 0

        # And δ[k] ≠ 0 only when at least one of b[i+k], b[i-k], b[j+k], b[j-k] is 1
        # For a "dense" binary sequence (n/2 ones), this is most k values.

        # δ[k] = (b[i+k] if 0≤i+k<n else 0) + (b[i-k] if 0≤i-k<n else 0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    """Project h onto {0<=x<=1, sum(x)=target_sum} using bisection on shift"""
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2, 0, 1)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa_fast(b, n, time_limit, rng, T0=0.005, T_end=0.0001):
    """Fast binary SA with incremental correlation tracking"""
    dx = 2.0 / n
    curr_corr = np.correlate(b, 1.0 - b, mode='full').astype(float)
    curr_max = float(np.max(curr_corr))
    curr_c5 = curr_max * dx
    best_c5 = curr_c5
    best_b = b.copy()
    
    ones = np.where(b == 1)[0].tolist()
    zeros = np.where(b == 0)[0].tolist()
    
    start = time.time()
    ks = np.arange(-(n-1), n)
    iterations = 0
    
    while time.time() - start < time_limit:
        iterations += 1
        frac = min(1.0, (time.time() - start) / time_limit)
        T = T0 * (T_end / T0) ** frac
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta for all lags
        # delta[k] = b[i+k] + b[i-k] - b[j+k] - b[j-k] (out-of-range = 0)
        idx1 = i + ks
        m1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - ks
        m2 = (idx2 >= 0) & (idx2 < n)
        idx3 = j + ks
        m3 = (idx3 >= 0) & (idx3 < n)
        idx4 = j - ks
        m4 = (idx4 >= 0) & (idx4 < n)
        
        delta = np.zeros(2*n-1)
        delta[m1] += b[idx1[m1]]
        delta[m2] += b[idx2[m2]]
        delta[m3] -= b[idx3[m3]]
        delta[m4] -= b[idx4[m4]]
        
        new_max = float(np.max(curr_corr + delta))
        new_c5 = new_max * dx
        
        if new_c5 <= curr_c5 or (T > 1e-15 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            b[i] = 0
            b[j] = 1
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            curr_corr += delta
            curr_max = new_max
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
    
    return best_b, best_c5
```

```python
def continuous_descent(h, n, time_limit, rng):
    """Log-sum-exp gradient descent with projection"""
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    T = 50.0  # smoothing parameter (1/T ≈ effective number of top lags)
    lr = 0.1
    
    start = time.time()
    ks = np.arange(-(n-1), n)
    
    while time.time() - start < time_limit:
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Log-sum-exp
        max_c = np.max(corr)
        shifted = T * (corr - max_c)
        exp_shifted = np.exp(shifted)
        sum_exp = np.sum(exp_shifted)
        w = exp_shifted / sum_exp  # weights over lags
        
        # Smooth objective
        c5_smooth = (max_c + np.log(sum_exp) / T) * dx
        
        # Compute gradient: grad[j] = dx * sum_k w[k] * ((1-h[j+k]) - h[j-k])
        # First part: sum_k w[k] * (1-h[j+k]) for valid j+k
        # Second part: sum_k w[k] * h[j-k] for valid j-k
        
        omh = 1.0 - h
        grad = np.zeros(n)
        
        # For each lag k, w[k] contributes to grad[j] for valid j
        # w is indexed as w[i] = weight for lag i-(n-1)
        for lag_idx in range(2*n-1):
            k = lag_idx - (n-1)
            wk = w[lag_idx]
            if wk < 1e-10:
                continue
            if k >= 0:
                # First part: (1-h[j+k]) for j in [0, n-k)
                grad[:n-k] += wk * omh[k:]
                # Second part: h[j-k] for j in [k, n)
                grad[k:] += -wk * h[:n-k]
            else:
                kp = -k
                # First part: (1-h[j+k]) = (1-h[j-kp]) for j in [kp, n)
                grad[kp:] += wk * omh[:n-kp]
                # Second part: h[j-k] = h[j+kp] for j in [0, n-kp)
                grad[:n-kp] += -wk * h[kp:]
        
        grad *= dx
        
        # Take step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Check if it's better
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        
        # Adaptive learning rate
        if new_c5 > c5_smooth + 0.01:
            lr *= 0.5
        else:
            lr = min(lr * 1.05, 0.5)
        
        # Anneal T
        T *= 0.999
    
    return best_h, best_c5
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa_fast(b_in, n, time_limit, rng, T0=0.005, T_end=0.00002):
    b = b_in.astype(float).copy()
    dx = 2.0 / n
    curr_corr = np.correlate(b, 1.0 - b, mode='full').astype(float)
    curr_c5 = float(np.max(curr_corr) * dx)
    best_c5 = curr_c5
    best_b = b.copy()
    ones_list = list(np.where(b == 1)[0])
    zeros_list = list(np.where(b == 0)[0])
    start = time.time()
    ks = np.arange(-(n - 1), n)
    no_improve = 0
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / max(time_limit, 1e-9))
        T = T0 * (T_end / T0) ** frac
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        idx1 = i + ks
        m1 = (idx1 >= 0) & (idx1 < n)
        idx2 = i - ks
        m2 = (idx2 >= 0) & (idx2 < n)
        idx3 = j + ks
        m3 = (idx3 >= 0) & (idx3 < n)
        idx4 = j - ks
        m4 = (idx4 >= 0) & (idx4 < n)
        delta = np.zeros(2 * n - 1)
        delta[m1] += b[idx1[m1]]
        delta[m2] += b[idx2[m2]]
        delta[m3] -= b[idx3[m3]]
        delta[m4] -= b[idx4[m4]]
        new_max = float(np.max(curr_corr + delta))
        new_c5 = new_max * dx
        if new_c5 <= curr_c5 or (T > 1e-15 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            b[i] = 0.0
            b[j] = 1.0
            oi = ones_list.index(i)
            ones_list[oi] = j
            oj = zeros_list.index(j)
            zeros_list[oj] = i
            curr_corr += delta
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
            if no_improve > 50000:
                for _ in range(100):
                    ri = ones_list[rng.randint(len(ones_list))]
                    rj = zeros_list[rng.randint(len(zeros_list))]
                    b[ri], b[rj] = 1.0, 0.0
                    oi = ones_list.index(ri)
                    ones_list[oi] = rj
                    oj = zeros_list.index(rj)
                    zeros_list[oj] = ri
                no_improve = 0
    return best_b, best_c5

def continuous_descent(h_in, n, time_limit, T_init=300.0):
    h = h_in.astype(float).copy()
    dx = 2.0 / n
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = T_init
    lr = 0.02
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = float(np.max(corr))
        shifted = T * (corr - max_c)
        exp_s = np.exp(shifted - np.max(shifted))
        sum_exp = float(np.sum(exp_s))
        w = exp_s / sum_exp
        omh = 1.0 - h
        grad = np.zeros(n)
        active = np.where(w > 1e-5)[0]
        for lag_idx in active:
            k = int(lag_idx) - (n - 1)
            wk = float(w[lag_idx])
            if k >= 0:
                nk = n - k
                grad[:nk] += wk * omh[k:]
                grad[k:] -= wk * h[:nk]
            else:
                kp = -k
                nk = n - kp
                grad[kp:] += wk * omh[:nk]
                grad[:nk] -= wk * h[kp:]
        grad *= dx
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        h = h_new
        c5_smooth = (max_c + (np.log(sum_exp) + np.max(shifted)) / T) * dx
        if new_c5 > c5_smooth + 0.003:
            lr *= 0.85
        elif new_c5 < best_c5 + 1e-8:
            lr = min(lr * 1.03, 0.15)
        T = max(T * 0.998, 10.0)
    return best_h, best_c5

def coord_descent(h_in, n, time_limit, steps=None):
    h = h_in.astype(float).copy()
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if steps is None:
        steps = [0.0, 1.0, 0.02, -0.02, 0.05, -0.05]
    improved = True
    while improved and time.time() - start < time_limit:
        improved = False
        for j in range(n):
            if time.time() - start > time_limit - 1:
                return best_h, best_c5
            old = h[j]
            best_local = best_c5
            best_val = old
            h_opt = h.copy()
            for s in steps:
                cand = old + s
                if cand < 0.0 or cand > 1.0:
                    cand = float(np.clip(cand, 0.0, 1.0))
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = project_feasible(trial, target_sum)
                c = compute_c5(trial)
                if c < best_local - 1e-14:
                    best_local = c
                    best_val = cand
                    h_opt = trial.copy()
            if best_val != old:
                h = h_opt
                best_c5 = best_local
                best_h = h.copy()
                improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
```
