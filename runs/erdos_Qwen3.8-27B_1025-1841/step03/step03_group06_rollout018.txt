        grad[:n-kp] += w * (1.0 - h[kp:])
        grad[kp:] -= w * h[:n-kp]
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

_G_T = 50.0
_G_PEN = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _get_corr_softmax(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = np.clip(T * (corr - c_max), -700.0, 700.0)
    exp_vals = np.exp(shifted)
    S = np.sum(exp_vals)
    if S < 1e-300:
        sm = np.zeros(2 * n - 1)
        sm[np.argmax(corr)] = 1.0
    else:
        sm = exp_vals / S
    return corr, c_max, sm

def _grad_corr(h, sm):
    n = len(h)
    grad = np.zeros(n)
    grad += sm[n - 1] * (1.0 - 3.0 * h)
    for k in range(1, n):
        w = sm[n - 1 + k]
        if w < 1e-14:
            continue
        grad[k:] += w * (1.0 - h[:n - k])
        grad[:n - k] -= w * h[k:]
    for kp in range(1, n):
        w = sm[n - 1 - kp]
        if w < 1e-14:
            continue
        grad[:n - kp] += w * (1.0 - h[kp:])
        grad[kp:] -= w * h[:n - kp]
    return grad

def _obj_val(x):
    T = _G_T
    pen = _G_PEN
    n = len(x)
    dx = 2.0 / n
    corr, c_max, sm = _get_corr_softmax(x, T)
    logS = np.log(np.sum(np.exp(np.clip(T * (corr - c_max), -700.0, 700.0))))
    val = dx * (c_max + logS / T)
    sd = np.sum(x) - n / 2.0
    val += pen * sd * sd
    return val

def _obj_grad(x):
    T = _G_T
    pen = _G_PEN
    n = len(x)
    dx = 2.0 / n
    corr, c_max, sm = _get_corr_softmax(x, T)
    grad = dx * _grad_corr(x, sm)
    sd = np.sum(x) - n / 2.0
    grad += 2.0 * pen * sd
    return grad

def _coord_descent(h, n, max_rounds, time_limit):
    t0 = time.time()
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            bc = best_c
            bt = old
            for cand in [0.0, 1.0, 0.5, max(0.0, old - 0.05), min(1.0, old + 0.05)]:
                if cand == old:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-14:
                    bc = c
                    bt = cand
            if bt != old:
                h[j] = bt
                h = make_feasible(h, n)
                best_c = bc
                improved = True
        if not improved and rnd >= 3:
            break
    return h, best_c
