import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_and_worst(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = int(np.argmax(corr))
    return float(corr[idx] * dx), idx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)
```

def slsqp_with_grad(h0, n, T, max_iter=100):
    def obj(x):
        corr = np.correlate(x, 1-x, mode='full')
        shifted = corr - corr.max()
        val = np.log(np.sum(np.exp(T * shifted))) / T
        return val * (2.0/n)
    
    def grad(x):
        corr = np.correlate(x, 1-x, mode='full')
        shifted = corr - corr.max()
        exps = np.exp(T * shifted)
        w = exps / exps.sum()
        # Compute gradient
        g = np.zeros(n)
        for s in range(n):
            # k = n-1-s (left side)
            wk = w[n-1-s] if n-1-s < len(w) else 0
            if wk > 1e-15:
                g[0:n-s] += wk * (1 - x[s:n])
                g[s:n] -= wk * x[0:n-s]
            # k = n-1+s (right side, s>=1)
            if s >= 1:
                wk2 = w[n-1+s] if n-1+s < len(w) else 0
                if wk2 > 1e-15:
                    g[s:n] += wk2 * (1 - x[0:n-s])
                    g[0:n-s] -= wk2 * x[s:n]
        return g * (2.0/n)
    
    def eq_con_fun(x):
        return np.sum(x) - n/2.0
    
    def eq_con_jac(x):
        return np.ones(n)
    
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=[(0,1)]*n,
                   constraints=[{'type':'eq', 'fun':eq_con_fun, 'jac':eq_con_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x
```
