    target = n / 2.0

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)  # or with endpoint
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)

    """Compute coefficient vector for changing h[j]."""
    corr_len = 2 * n - 1
    coeff = np.zeros(corr_len)
    for k in range(corr_len):

        c = 0.0

        if 0 <= jp < n:
            c += (1.0 - h[jp])
        jm = j - lag
        if 0 <= jm < n:
            c += (-h[jm])

def golden_coord_descent(h, n, max_rounds=50, time_limit=1200, start_time=0, mu=100.0, gss_iters=25):
    """Fast coordinate descent with golden section search."""
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    base_sum = float(np.sum(h))
    
    def current_obj(corr, ms):
        return np.max(corr) * dx + mu * (ms - n/2)**2
    
    obj = current_obj(corr, base_sum)
    best_obj = obj
    inv_phi = 0.38196601125  # (sqrt(5)-1)/2
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 10:
            break
        improved = False
        for j in range(n):
            jh = h[j]
            coeff = compute_coeff(h, j, n)
            
            # Golden section search for optimal h[j]
            lo, hi = 0.0, 1.0
            c1 = lo + inv_phi * (hi - lo)
            c2 = hi - inv_phi * (hi - lo)

                new_sum = base_sum + (v - jh)
                return np.max(vals) * dx + mu * (new_sum - n/2)**2, vals
            
            f1, _ = eval_v(c1)
            f2, _ = eval_v(c2)

                if f1 < f2:
                    hi = c2
                    c2 = c1
                    f2 = f1
                    c1 = lo + inv_phi * (hi - lo)
                    f1, _ = eval_v(c1)
                else:
                    lo = c1
                    c1 = c2
                    f1 = f2
                    c2 = hi - inv_phi * (hi - lo)

        tm[i] = c
    n_ones = int(np.sum(tm))
    target = n // 2
    rng = np.random.RandomState(seed)
    if n_ones > target:
        idx = np.where(tm == 1)[0]
        rng.shuffle(idx)
        for i in idx[:n_ones - target]:
            tm[i] = 0
    elif n_ones < target:
        idx = np.where(tm == 0)[0]
        rng.shuffle(idx)
        for i in idx[:target - n_ones]:
            tm[i] = 1
    return make_feasible(tm, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial solution
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass

    if h0 is None:
        n_init = 1024
        h0 = thue_morse_init(n_init, seed=seed)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent at current resolution
    n_cur = len(best_h)
    h_new, c_new = golden_coord_descent(best_h, n_cur, max_rounds=40,
                                         time_limit=budget_s, start_time=start_time)
    if c_new < best_c5:
        best_c5 = c_new
        best_h = h_new

    # Phase 2: L-BFGS-B with soft max at current resolution
    if time.time() - start_time < budget_s - 400:
        for t in [30, 80, 200]:
            if time.time() - start_time > budget_s - 350:
                break
            h_new = lbfgs_soft(best_h, n_cur, t, maxiter=200,
                               time_limit=budget_s, start_time=start_time)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                best_h = h_new

    # Phase 3: Upsample to 1024
    n2 = 1024
    if time.time() - start_time < budget_s - 300:
        h2 = upsample_h(best_h, n2)
        h2, c2 = golden_coord_descent(h2, n2, max_rounds=60,
                                       time_limit=budget_s, start_time=start_time)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

        for t in [50, 150, 400]:
            if time.time() - start_time > budget_s - 250:
                break
            h2 = lbfgs_soft(best_h, n2, t, maxiter=250,
                            time_limit=budget_s, start_time=start_time)
            c2 = compute_c5(h2)
            if c2 < best_c5 - 1e-10:
                best_c5 = c2
                best_h = h2

        h2, c2 = golden_coord_descent(best_h, n2, max_rounds=50,
                                       time_limit=budget_s, start_time=start_time)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 4: Try Thue-Morse initialization at 1024
    if time.time() - start_time < budget_s - 250:
        for s in [1, 7, 42]:
            if time.time() - start_time > budget_s - 220:
                break
            h_tm = thue_morse_init(1024, seed=s)
            h_tm, c_tm = golden_coord_descent(h_tm, 1024, max_rounds=40,
                                               time_limit=budget_s, start_time=start_time)
            if c_tm < best_c5:
                best_c5 = c_tm
                best_h = h_tm

    # Phase 5: Random restart at 1024
    if time.time() - start_time < budget_s - 200:
        for s in [123, 456]:
            if time.time() - start_time > budget_s - 180:
                break
            h_r = rng.rand(1024)
            h_r = make_feasible(h_r, 1024)
            h_r = lbfgs_soft(h_r, 1024, t=40, maxiter=200,
                             time_limit=budget_s, start_time=start_time)
            h_r, c_r = golden_coord_descent(h_r, 1024, max_rounds=30,
                                             time_limit=budget_s, start_time=start_time)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r

    # Phase 6: Upsample to 2048 and refine
    n3 = 2048
    if time.time() - start_time < budget_s - 150:
        h3 = upsample_h(best_h, n3)
        h3, c3 = golden_coord_descent(h3, n3, max_rounds=50,
                                       time_limit=budget_s, start_time=start_time)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

        if time.time() - start_time < budget_s - 100:
            h3 = lbfgs_soft(best_h, n3, t=100, maxiter=200,
                            time_limit=budget_s, start_time=start_time)
            c3 = compute_c5(h3)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3

            h3, c3 = golden_coord_descent(best_h, n3, max_rounds=40,
                                           time_limit=budget_s, start_time=start_time)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3

    # Phase 7: Final coordinate descent pass
    n_f = len(best_h)
    if time.time() - start_time < budget_s - 30:
        h_f, c_f = golden_coord_descent(best_h, n_f, max_rounds=30,
                                         time_limit=budget_s, start_time=start_time)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    # Phase 8: Try rounding near-binary values then re-optimize
    if time.time() - start_time < budget_s - 20:
        h_rounded = best_h.copy()
        h_rounded[h_rounded < 0.15] = 0.0
        h_rounded[h_rounded > 0.85] = 1.0
        h_rounded = make_feasible(h_rounded, len(h_rounded))
        h_rounded, c_rounded = golden_coord_descent(h_rounded, len(h_rounded),
                                                     max_rounds=20,
                                                     time_limit=budget_s, start_time=start_time)
        if c_rounded < best_c5:
            best_c5 = c_rounded
            best_h = h_rounded

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
