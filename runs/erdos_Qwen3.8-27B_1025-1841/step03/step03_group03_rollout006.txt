def project_box_sum(x, target_sum):
    """Project x onto {y : 0 <= y[i] <= 1, sum(y) = target_sum}"""
    n = len(x)
    lo, hi = -1.0, 1.0  # range for the Lagrange multiplier
    for _ in range(100):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo+hi)/2, 0.0, 1.0)
    # Fine-tune
    diff = target_sum - np.sum(y)
    if abs(diff) > 1e-12:
        mask = (y > 0) & (y < 1)
        cnt = np.sum(mask)
        if cnt > 0:
            y[mask] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y

        h_sg, c_sg = subgradient_optimize(best_h, n0, 3000, start_time, budget_s - 200)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg
        h_pn, c_pn = project_newton_optimize(best_h, n0, 200, 300, start_time, budget_s - 100)
        if c_pn < best_c5:
            best_c5 = c_pn
            best_h = h_pn

    # Phase 3: Upsample and optimize at higher resolutions
    for target_n in [800, 1000]:
        if time.time() - start_time > budget_s - 250:
            break
        h_up = upsample_h(best_h, target_n)
        for t, mi in [(40, 250), (150, 350), (500, 400)]:
            if time.time() - start_time > budget_s - 150:
                break
            h_new = slsqp_optimize(h_up, target_n, t, mi)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new
                h_up = h_new
            else:
                h_up = h_new

    # Phase 4: Coordinate descent at current resolution
    n_cd = len(best_h)
    h_cd, c_cd = coord_descent(best_h, n_cd, 12, start_time, budget_s - 60)
    if c_cd < best_c5:
        best_c5 = c_cd
        best_h = h_cd

    # Phase 5: Binary + swap search
    if time.time() - start_time < budget_s - 50:
        n_sw = len(best_h)
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, n_sw)
        c_bin = compute_c5(h_bin)
        h_cur = best_h.copy()
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin
            h_cur = h_bin
        h_sw, c_sw = swap_search(h_cur, n_sw, 30, start_time, budget_s - 25, rng)
        if c_sw < best_c5:
            best_c5 = c_sw
            best_h = h_sw

    # Phase 6: One more SLSQP polish at final resolution
    if time.time() - start_time < budget_s - 80:
        n_fin = len(best_h)
        h_fin = slsqp_optimize(best_h, n_fin, 2000, 300)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5 - 1e-12:
            best_c5 = c_fin
            best_h = h_fin

    # Final coordinate descent
    h_f, c_f = coord_descent(best_h, len(best_h), 8, start_time, budget_s - 5)
    if c_f < best_c5:
        best_c5 = c_f
        best_h = h_f

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
