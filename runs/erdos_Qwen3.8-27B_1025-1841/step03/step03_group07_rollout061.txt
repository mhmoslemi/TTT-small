        if s > target:
            lo = mid
        else:
            hi = mid

        h_try = project_to_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < compute_c5(h) or (c5_try - compute_c5(h)) < 0 or rng.random() < np.exp((compute_c5(h) - c5_try) / T):
            h = h_try
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                best_h = h.copy()
        T *= cool
        if T < 1e-7:
            T = 1e-7
    return best_h, best_c5


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    n1 = 4000
    h1 = interp_periodic(h_init, n1)
    h1 = project_to_feasible(h1, n1)

    # Phase 1: Nesterov tau=120, broad convergence
    rem = get_remaining(start_time, budget_s)
    if rem > 250:
        tl = min(100, rem * 0.15)
        h1, c5 = nesterov_optimize(h1, n1, tau=120.0, time_limit=tl, lr=0.12, top_k=15)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Nesterov tau=40, medium focus
    rem = get_remaining(start_time, budget_s)
    if rem > 200:
        tl = min(80, rem * 0.12)
        h1, c5 = nesterov_optimize(h1, n1, tau=40.0, time_limit=tl, lr=0.04, top_k=10)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 3: Nesterov tau=15, sharp focus
    rem = get_remaining(start_time, budget_s)
    if rem > 180:
        tl = min(70, rem * 0.12)
        h1, c5 = nesterov_optimize(h1, n1, tau=15.0, time_limit=tl, lr=0.018, top_k=7)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 4: Subgradient at n=4000
    rem = get_remaining(start_time, budget_s)
    if rem > 150:
        tl = min(80, rem * 0.15)
        h1, c5 = subgrad_refine(best_h, n1, time_limit=tl, lr=0.001, top_k=6)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 5: Upsample to n=8000
    rem = get_remaining(start_time, budget_s)
    n2 = 8000
    if rem > 250:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        # Short Nesterov at higher res to settle
        tl = min(50, rem * 0.1)
        h2, c5 = nesterov_optimize(h2, n2, tau=20.0, time_limit=tl, lr=0.015, top_k=8)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()
        else:
            c5_check = compute_c5(h2)
            if c5_check < best_c5:
                best_c5, best_h = c5_check, h2.copy()
        n_best = n2
    else:
        n_best = n1

    # Phase 6: Long subgradient at best resolution
    rem = get_remaining(start_time, budget_s)
    if rem > 150:
        tl = min(150, rem * 0.25)
        h_b, c5 = subgrad_refine(best_h, n_best, time_limit=tl, lr=0.0005, top_k=4)
        if c5 < best_c5:
            best_c5, best_h = c5, h_b.copy()

    # Phase 7: Multi-start perturbations
    rem = get_remaining(start_time, budget_s)
    perturb_configs = [
        (0.003, 25.0, 0.035, 6),
        (0.005, 35.0, 0.05, 8),
        (0.002, 15.0, 0.02, 5),
    ]
    for p_scale, p_tau, p_lr, p_topk in perturb_configs:
        rem = get_remaining(start_time, budget_s)
        if rem < 60:
            break
        h_p = best_h.copy() + np.random.randn(n_best) * p_scale
        h_p = project_to_feasible(h_p, n_best)
        tl = min(50, rem * 0.15)
        h_p, c5_p = nesterov_optimize(h_p, n_best, tau=p_tau, time_limit=tl, lr=p_lr, top_k=p_topk)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 8: Local perturbation refine
    rem = get_remaining(start_time, budget_s)
    if rem > 50:
        tl = min(45, rem * 0.4)
        h_b, c5 = local_perturb_refine(best_h, n_best, time_limit=tl, seed=777)
        if c5 < best_c5:
            best_c5, best_h = c5, h_b.copy()

    # Phase 9: Very fine subgradient polish
    rem = get_remaining(start_time, budget_s)
    if rem > 30:
        tl = min(30, rem * 0.7)
        h_b, c5 = subgrad_refine(best_h, n_best, time_limit=tl, lr=0.0002, top_k=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h_b.copy()

    # Final: one more local perturb
    rem = get_remaining(start_time, budget_s)
    if rem > 20:
        tl = min(20, rem * 0.8)
        h_b, c5 = local_perturb_refine(best_h, n_best, time_limit=tl, seed=314)
        if c5 < best_c5:
            best_c5, best_h = c5, h_b.copy()

    best_h = project_to_feasible(best_h, n_best)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_best
