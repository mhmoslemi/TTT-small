import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_exact(h, n_iter=5000, lr0=0.12, check=30):

    dx = 2.0 / n

    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        lr = lr0 * (1.0 - 0.5 * f)
        
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc_idx = np.argmax(corr)
        k_star = mc_idx - (n - 1)
        g = _subgrad_one(z, k_star) * dx
        
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0

# SLSQP helpers
_SLSQP = {}

def _sobj(h):
    T = _SLSQP['T']
    n = _SLSQP['n']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _scon(h):
    return np.sum(h) - _SLSQP['n'] / 2.0

def _sjac(h):
    return np.ones(_SLSQP['n'])

def slsqp(h, T=5.0, mi=500):
    n = len(h)
    _SLSQP['T'], _SLSQP['n'] = T, n
    try:
        r = minimize(_sobj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

    ox = np.linspace(0, 2, len(h))

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))

    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    bc = float('inf')
    bh = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = project_c(rng.rand(480), 480)

    n0 = len(h0)
    bh = h0.copy()
    bc = compute_c5(bh)
    print(f"INIT: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1: FISTA at full resolution
    if rem() > 200:
        bh, bc = fista(bh, 4000, 30.0, 0.05, 0.09, 40, 30)
        print(f"P1: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 2: Downsample to 480, extensive SLSQP
    n480 = min(n0, 480)
    if rem() > 250:
        hs = resample(bh, n480) if n0 > n480 else bh.copy()
        for T in [50.0, 20.0, 8.0, 4.0, 2.0, 1.0]:
            if rem() < 30:
                break
            hs = slsqp(hs, T, 600)
        cs = compute_c5(hs)
        print(f"P2 SLSQP: C5={cs:.6f} t={time.time()-t0:.0f}s", flush=True)
        if n0 > n480:
            hu = resample(hs, n0)
            cu = compute_c5(hu)
            if cu < bc:
                bc, bh = cu, hu.copy()
            if rem() > 100:
                bh, bc = fista(bh, 2500, 15.0, 0.03, 0.06, 35, 30)
        else:
            if cs < bc:
                bc, bh = cs, hs.copy()
        print(f"P2 final: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 3: Multi-start at n=480
    if rem() > 200:
        nms = 480
        starts = [resample(bh, nms), tm_seq(nms)]
        for p in [7, 11, 13, 17, 19, 23]:
            pat = np.zeros(p)
            pat[:p // 2] = 1.0
            starts.append(project_c(np.tile(pat, nms // p + 1)[:nms], nms))
        xms = np.linspace(0, 2, nms)
        for fr, ph in [(3, 0.0), (5, np.pi / 4), (7, np.pi / 2), (11, 0.7)]:
            starts.append(project_c(0.5 + 0.45 * np.sin(2 * np.pi * fr * xms / 2.0 + ph), nms))
        starts.append(project_c(rng.rand(nms), nms))

        for si, hs in enumerate(starts):
            if rem() < 50:
                break
            hr, cr = fista(hs, 2000, 40.0, 0.1, 0.1, 30, 40)
            if rem() < 30:
                break
            hr = slsqp(hr, 6.0, 400)
            hr = slsqp(hr, 2.0, 400)
            cr = compute_c5(hr)
            if cr < bc:
                if n0 > nms:
                    hr_up = resample(hr, n0)
                    hr_up, cr = fista(hr_up, 1500, 10.0, 0.03, 0.05, 30, 30)
                    bh = hr_up
                else:
                    bh = hr
                bc = cr
                print(f"  MS{si}: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)
        print(f"P3: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 4: Perturbation restarts at n=480
    if rem() > 120:
        np_ = 480
        hp = resample(bh, np_)
        for trial in range(3):
            if rem() < 35:
                break
            sc = 0.025 + 0.02 * trial
            hp = project_c(bh + rng.randn(len(bh)) * sc * (len(bh) / np_) if len(bh) == np_ else resample(bh, np_) + rng.randn(np_) * sc, np_)
            hr, cr = fista(hp, 2000, 25.0, 0.05, 0.07, 30, 30)
            hr = slsqp(hr, 3.0, 400)
            cr = compute_c5(hr)
            if cr < bc:
                if n0 > np_:
                    hr_up = resample(hr, n0)
                    hr_up, cr = fista(hr_up, 1200, 8.0, 0.02, 0.04, 30, 30)
                bc = cr
                bh = hr_up if n0 > np_ else hr
                print(f"  PT{trial}: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)
        print(f"P4: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 5: Second SLSQP pass at n=480 on best
    if rem() > 80:
        hf = resample(bh, min(len(bh), 480))
        hf = slsqp(hf, 4.0, 500)
        hf = slsqp(hf, 1.5, 400)
        cf = compute_c5(hf)
        if len(bh) > len(hf):
            hf = resample(hf, len(bh))
            cf = compute_c5(hf)
            if cf < bc:
                bc, bh = cf, hf.copy()
        elif cf < bc:
            bc, bh = cf, hf.copy()
        print(f"P5: C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 6: Final FISTA polish at full resolution
    if rem() > 25:
        n_it = max(800, int(rem() / 0.04))
        bh, bc = fista(bh, min(n_it, 2500), 5.0, 0.01, 0.03, 30, 20)

    # Final validation
    bh = project_c(bh, len(bh))
    fc = compute_c5(bh)
    if fc < bc:
        bc = fc
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}s", flush=True)
    return bh, bc, len(bh)
