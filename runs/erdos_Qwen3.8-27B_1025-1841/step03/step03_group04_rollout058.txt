def smooth_max_with_grad(h, top_idx, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr[top_idx]
    mc = np.max(vals)
    exps = np.exp(T * (vals - mc))
    weights = exps / np.sum(exps)
    obj_val = (mc + np.log(np.sum(exps)) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    for idx_in_top, k in enumerate(top_idx):
        # ∂corr[k]/∂h[i] for all i
        w = weights[idx_in_top]
        # First term: (1-h[i+k-n+1]) for valid i
        # i ranges from max(0, n-1-k) to min(n-1, 2n-1-k)
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, 2 * n - 1 - k)
        for i in range(i_lo, i_hi + 1):
            j = i + k - n + 1  # index into (1-h)
            if 0 <= j < n:
                grad[i] += w * (1.0 - h[j])
        # Second term: -h[i-k+n-1] for valid i
        # i ranges from max(0, k-n+1) to min(n-1, k)
        i_lo2 = max(0, k - n + 1)
        i_hi2 = min(n - 1, k)
        for i in range(i_lo2, i_hi2 + 1):
            m = i - k + n - 1
            if 0 <= m < n:
                grad[i] -= w * h[m]
    
    grad *= dx
    return obj_val, grad

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0
_G_TOP = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def _obj_sm(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr[_G_TOP]
    mc = np.max(vals)
    shifted = vals - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _obj_grad(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top = _G_TOP
    vals = corr[top]
    mc = np.max(vals)
    exps = np.exp(T * (vals - mc))
    sw = np.sum(exps)
    weights = exps / sw
    obj_val = (mc + np.log(sw) / T) * dx
    grad = np.zeros(n)
    for ii in range(len(top)):
        k = top[ii]
        w = weights[ii]
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, 2 * n - 1 - k)
        idx1 = np.arange(i_lo, i_hi + 1)
        j1 = idx1 + k - n + 1
        m1 = (j1 >= 0) & (j1 < n)
        grad[idx1[m1]] += w * (1.0 - x[j1[m1]])
        i_lo2 = max(0, k - n + 1)
        i_hi2 = min(n - 1, k)
        idx2 = np.arange(i_lo2, i_hi2 + 1)
        m2 = idx2 - k + n - 1
        m2v = (m2 >= 0) & (m2 < n)
        grad[idx2[m2v]] -= w * x[m2[m2v]]
    grad *= dx
    return obj_val, grad

def _con_eq(x):
    return np.sum(x) - _G_N / 2.0

def _con_eq_jac(x):
    return np.ones(_G_N)

def _slsqp_call(x0, top_k, T, maxiter):
    global _G_N, _G_T, _G_TOP
    n = _G_N
    corr = get_corr(x0)
    mc = np.max(corr)
    thr = mc - 0.002 * mc * top_k / 10.0
    idx = np.where(corr >= thr)[0]
    if len(idx) > top_k:
        idx = idx[np.argsort(-corr[idx])[:top_k]]
    elif len(idx) == 0:
        idx = np.array([int(np.argmax(corr))])
    _G_TOP = idx
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    res = minimize(_obj_sm, x0, method='SLSQP', bounds=bounds,
                   constraints=cons, jac=_obj_grad,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x
