import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj_grad(h, t, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(np.clip(t * shifted, -50, 0))
    se = np.sum(exp_s)
    obj = (mc + np.log(se) / t) * dx
    s = np.sum(h) - n / 2.0
    obj += pen * s * s
    w = exp_s / se
    active = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for ki in active:
        lag = ki - (n - 1)
        wk = w[ki]
        jlo = max(0, lag)
        jhi = min(n - 1, lag + n - 1)
        if jlo <= jhi:
            grad[jlo:jhi + 1] += wk * (1.0 - h[jlo - lag:jhi - lag + 1])
        jlo2 = max(0, -lag)
        jhi2 = min(n - 1, n - 1 - lag)
        if jlo2 <= jhi2:
            grad[jlo2:jhi2 + 1] -= wk * h[jlo2 + lag:jhi2 + lag + 1]
    grad *= dx
    grad += 2.0 * pen * s
    return obj, grad

def run_lfbfgs(h0, t, pen, maxiter):
    n = len(h0)
    def fun(x):
        return soft_obj_grad(x, t, pen)[0]
    def jac(x):
        return soft_obj_grad(x, t, pen)[1]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-9})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h0, n)

def proj_grad_opt(h0, t, lr_init, n_iter, tlimit, tstart):
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for it in range(n_iter):
        if time.time() - tstart > tlimit - 20:
            break
        lr = lr_init / (1.0 + 0.01 * it)
        obj, grad = soft_obj_grad(h, t, 0.0)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c

def coord_desc(h, max_rounds, tlimit, tstart):
    n = len(h)
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - tstart > tlimit - 25:
            break
        improved = False
        for j in range(n):
            if time.time() - tstart > tlimit - 25:
                return h, best_c
            old = h[j]
            candidates = [0.0, 1.0, old + 0.05, old - 0.05,
                          old + 0.02, old - 0.02,
                          old + 0.01, old - 0.01,
                          old + 0.005, old - 0.005]
            best_val = old
            best_obj = best_c
            for cv in candidates:
                if cv < -1e-9 or cv > 1.0 + 1e-9:
                    continue
                cv = max(0.0, min(1.0, cv))
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-14:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def binary_local_search(h, tlimit, tstart, max_rounds=10):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    if abs(np.sum(h_bin) - n / 2) > 1e-6:
        h_bin = make_feasible(h_bin, n)
    best_c = compute_c5(h_bin)
    for rnd in range(max_rounds):
        if time.time() - tstart > tlimit - 25:
            break
        improved = False
        for j in range(n):
            if time.time() - tstart > tlimit - 25:
                return h_bin, best_c
            trial = h_bin.copy()
            trial[j] = 1.0 - trial[j]
            if abs(np.sum(trial) - n / 2) > 1e-6:
                continue
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best_c = c
                h_bin = trial
                improved = True
        if not improved:
            break
    return h_bin, best_c
