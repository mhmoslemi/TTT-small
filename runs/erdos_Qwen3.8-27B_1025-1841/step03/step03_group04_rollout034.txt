import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_project(h, n):
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

# Globals for SLSQP
_G_N = 0
_G_T = 500.0
_G_ACTIVE = None

def _obj_smooth(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_ACTIVE is not None:
        vals = corr[_G_ACTIVE]
    else:
        vals = corr
    mc = np.max(vals)
    shifted = vals - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    return np.sum(x) - _G_N / 2.0

Wait, the directed swap approach has an issue: computing the full correlation just to find the worst lag is O(n²), which defeats the purpose. Let me keep the corr vector updated.

Actually, for n=960, np.correlate is fast enough (~0.1-0.5ms). With 80000 iterations, that's 8-40 seconds just for correlation computations. That's fine within a 400s budget for SA.

But the directed approach requires computing corr every iteration to find the worst lag. Instead, I can:
- Compute corr every K iterations (e.g., K=100)
- Cache the worst lag and bad positions
- Use the cached info for K iterations

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_project(h, n):
    idx = np.argsort(-h)
    out = np.zeros(n)
    out[idx[:n // 2]] = 1.0
    return out

def binary_sa(h_start, n, time_limit, rng, T_start=0.004, T_end=0.00003, max_iter=60000):

    best = curr.copy()
    best_c5 = curr_c5
    ones = np.where(curr > 0.5)[0].copy()
    zeros = np.where(curr < 0.5)[0].copy()
    
    alpha = (T_end / T_start) ** (1.0 / max(1, max_iter))
    t0 = time.time()
    corr = get_corr(curr)
    cache_period = 200
    
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if it % cache_period == 0:
            T = T_start * (alpha ** it)
            corr = get_corr(curr)
            k_max = int(np.argmax(corr))
        else:
            T = T_start * (alpha ** it)
        
        do_directed = (rng.random() < 0.6 and it >= cache_period)
        if do_directed:
            s = k_max - (n - 1)
            if s >= 0:
                valid = np.arange(0, n - s)
            else:

            a = int(ones[rng.randint(len(ones))])
            b = int(zeros[rng.randint(len(zeros))])

        curr[b] = 1.0

        delta = new_c5 - curr_c5
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if a in ones:

def _obj_smooth(x):

    T = _G_T

    else:
        vals = corr
