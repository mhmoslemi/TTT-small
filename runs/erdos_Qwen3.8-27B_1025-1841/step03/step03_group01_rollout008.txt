            # corr[k] = sum_i h[i]*(1-h[i+L]) for i in [0, n-L-1]

            #           if j-L in [0,n-1] i.e. j in [L,n-1], term is -h[j-L]
            if L == 0:

        lc = 1.0 - (t + 1) * lr / (n_iters * 0.5 + 1)
        if lc < 0.1:
            lc = 0.1
        mhat = m / (1 - b1 ** (t + 1))
        vhat = v / (1 - b2 ** (t + 1))
        h_new = h - lc * mhat / (np.sqrt(vhat) + eps)
        h = project_feasible(h_new, n)
        if (t + 1) % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _obj_fn(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _cons_jac(h):
    return np.ones(_OPT_N)

def slsqp_opt(h0, n, T, max_it):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_fn, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_it, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h0, n)

def sim_anneal(h, n, n_swaps, T_start, T_end, t_end):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    rng = np.random.RandomState(12345)
    n_s = len(n_swaps)
    for it in range(n_swaps):
        if time.time() > t_end:
            break
        frac = it / max(n_swaps - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        j1, j2 = rng.randint(0, n, 2)
        if j1 == j2:
            continue
        dv = h[j1] - h[j2]
        if abs(dv) < 1e-12:
            continue
        h[j1] -= dv
        h[j2] += dv
        if h[j1] < -1e-10 or h[j1] > 1.0 + 1e-10 or h[j2] < -1e-10 or h[j2] > 1.0 + 1e-10:
            h[j1] += dv
            h[j2] -= dv
            continue
        h = np.clip(h, 0.0, 1.0)
        h = make_feasible(h, n)
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < 0:
            best_c5 = c5_new
            best_h = h.copy()
        else:
            if delta > T * 0.01 or rng.rand() > np.exp(-delta / max(T, 1e-10)):
                h[j1] += dv
                h[j2] -= dv
                h = np.clip(h, 0.0, 1.0)
                h = make_feasible(h, n)
    return best_h, best_c5

def smart_cd(h, n, max_rounds, t_end):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > t_end:
            break
        improved = False
        grad = compute_subgrad(h, 20)
        order = np.argsort(np.abs(grad))[::-1]
        for j in order[:n]:
            if time.time() > t_end:
                break
            if abs(grad[j]) < 1e-8:
                break
            direction = -np.sign(grad[j])
            steps = [0.04 * direction, 0.01 * direction, -0.01 * direction,
                     0.02 * direction, 0.005 * direction, -0.005 * direction]
            for delta in steps:
                trial = h.copy()
                trial[j] = h[j] + delta
                if trial[j] < -1e-10 or trial[j] > 1.0 + 1e-10:
                    continue
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = trial
                    h = trial
                    improved = True
            if not improved and h[j] > 0.05 and h[j] < 0.95:
                for bv in (0.0, 1.0):
                    trial = h.copy()
                    trial[j] = bv
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        best_h = trial
                        h = trial
                        improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_N, _OPT_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 512)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: n=1024 heavy optimization
    n1 = 1024
    h = upsample_h(best_h, n1)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()

    if _tl() > 350:
        h, c5 = adam_subgrad(h, n1, 3000, lr=0.08, top_k=50, t_end=start_time + _tl() - 50)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if _tl() > 80:
        h = slsqp_opt(h, n1, T=80.0, max_it=200)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if _tl() > 50:
        h = slsqp_opt(h, n1, T=300.0, max_it=150)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if _tl() > 60:
        h, c5 = sim_anneal(h, n1, 3000, T_start=0.01, T_end=0.0001, t_end=start_time + _tl() - 20)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    if _tl() > 40:
        h, c5 = smart_cd(h, n1, max_rounds=4, t_end=start_time + _tl() - 10)
        if c5 < best_c5:
            best_c5, best_h = c5, h

    # Phase 2: n=2048
    if _tl() > 250:
        n2 = 2048
        h = upsample_h(best_h, n2)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

        if _tl() > 180:
            h, c5 = adam_subgrad(h, n2, 1500, lr=0.04, top_k=60, t_end=start_time + _tl() - 60)
            if c5 < best_c5:
                best_c5, best_h = c5, h

        if _tl() > 50:
            h = slsqp_opt(h, n2, T=200.0, max_it=100)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h

        if _tl() > 40:
            h, c5 = smart_cd(h, n2, max_rounds=3, t_end=start_time + _tl() - 10)
            if c5 < best_c5:
                best_c5, best_h = c5, h

    # Phase 3: restart with perturbation at n=1024
    if _tl() > 120:
        n3 = 1024
        h3 = upsample_h(best_h, n3)
        mask = rng.rand(n3) < 0.08
        h3[mask] = np.clip(h3[mask] + 0.15 * rng.randn(int(mask.sum())), 0, 1)
        h3 = make_feasible(h3, n3)
        h3, c5 = adam_subgrad(h3, n3, 1500, lr=0.06, top_k=40, t_end=start_time + _tl() - 60)
        if _tl() > 30:
            h3 = slsqp_opt(h3, n3, T=150.0, max_it=120)
            c5 = compute_c5(h3)
        if c5 < best_c5:
            best_c5, best_h = c5, h3

    # Phase 4: binary-like restart
    if _tl() > 100:
        n4 = 1024
        h4 = (rng.rand(n4) < 0.5).astype(float)
        h4 = make_feasible(h4, n4)
        h4 = 0.3 * upsample_h(best_h, n4) + 0.7 * h4
        h4 = make_feasible(h4, n4)
        h4, c5 = adam_subgrad(h4, n4, 1500, lr=0.07, top_k=40, t_end=start_time + _tl() - 50)
        if _tl() > 30:
            h4 = slsqp_opt(h4, n4, T=100.0, max_it=100)
            c5 = compute_c5(h4)
        if c5 < best_c5:
            best_c5, best_h = c5, h4

    # Phase 5: if we have a 2048 solution, try n=4096 briefly
    if _tl() > 120 and len(best_h) >= 1024:
        n5 = 4096
        h5 = upsample_h(best_h, n5)
        c5 = compute_c5(h5)
        if c5 < best_c5:
            best_c5, best_h = c5, h5
        if _tl() > 60:
            h5, c5 = adam_subgrad(h5, n5, 500, lr=0.02, top_k=60, t_end=start_time + _tl() - 20)
            if c5 < best_c5:
                best_c5, best_h = c5, h5

    # Final polish on best
    if _tl() > 25:
        nn = len(best_h)
        best_h = slsqp_opt(best_h, nn, T=500.0, max_it=80)
        c5 = compute_c5(best_h)
        if c5 < best_c5:
            best_c5 = c5

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
