def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

    return np.correlate(h, 1.0 - h, mode='full')

def exact_coord_descent(h, n, max_rounds=50, time_limit=None, start_time=None):
    """Exact coordinate descent: for each coordinate, find the optimal value."""
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c = float(np.max(corr)) * dx
    
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        
        improved = False
        for j in range(n):

            # corr[m] = sum_i h[i+m] * (1-h[i]) (with appropriate indexing)
            # Actually, let's use the formula: 
            # np.correlate(h, 1-h, 'full')[k] = sum_i h[i] * (1-h[i-(k-(n-1))])
            # For k in range 0 to 2n-2, the lag is k-(n-1)
            # So corr_full[k] = sum_i h[i] * (1-h[i-(k-(n-1))]) for valid i
            
            # When h[j] changes by delta:
            # - h[j] appears as h[i] when i=j: contributes delta*(1-h[j-(k-(n-1))])
            # - h[j] appears as h[i-(k-(n-1))] when i-(k-(n-1))=j, i.e., i=j+(k-(n-1)):
            #   contributes -delta*h[j+(k-(n-1))]

            # s[l_idx] = (1-h[j-l]) if j-l in [0,n-1] else 0
            #           - h[j+l] if j+l in [0,n-1] else 0
            
            j_minus_l = j - lag  # indices j - l for each lag l
            j_plus_l = j + lag   # indices j + l for each lag l

            valid2 = (j_plus_l >= 0) & (j_plus_l < n)
            s[valid1] += 1.0 - h[j_minus_l[valid1]]
            s[valid2] -= h[j_plus_l[valid2]]

            # subject to -h[j] <= delta <= 1 - h[j]
            lo = -h[j]
            hi = 1.0 - h[j]

                    best_c = c
                    improved = True
        
        if not improved:
            break
    
    return h, best_c

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def exact_coord_descent(h, n, corr, max_rounds=100, time_limit=None, start_time=None, tol=1e-11):

    dx = 2.0 / n
    best_c = float(np.max(corr)) * dx
    lags = np.arange(2*n-1) - (n-1)
    
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        
        max_improve = 0.0
        for j in range(n):
            jml = j - lags  # j - m for each lag m
            jpl = j + lags   # j + m for each lag m
            
            s = np.zeros(2*n-1)
            v1 = (jml >= 0) & (jml < n)
            v2 = (jpl >= 0) & (jpl < n)
            s[v1] += 1.0 - h[jml[v1]]
            s[v2] -= h[jpl[v2]]
            
            lo = -h[j]
            hi = 1.0 - h[j]
            if hi - lo < 1e-15:

                m1 = lo + (hi - lo) * (1.0/3.0)
                m2 = hi - (hi - lo) * (1.0/3.0)
                f1 = np.max(corr + m1 * s)
                f2 = np.max(corr + m2 * s)
                if f1 <= f2:
                    hi = m2
                else:
                    lo = m1

            new_val = h[j] + d_opt
