        g[j] = val

        diff = target - np.sum(h)

            else:
                mask = h > 0.0

    """Coordinate descent with many candidate values."""
    best_h = h.copy()
    best_c5 = compute_c5(h)

        if time.time() - start_time > time_limit - 5:
            break
        improved = False
        order = rng.permutation(n)
        for j in order:

            candidates = np.concatenate([

            candidates = np.unique(candidates)

            best_obj = best_c5

                    best_val = cv

                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_obj
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

```python
def compute_subgradient(h, n, k):
    """Compute subgradient dc_k/dh for shift k."""
    g = np.zeros(n)
    # For each j, g[j] = (1-h[j-k]) if j-k in [0,n), plus -h[j+k] if j+k in [0,n)
    # j-k in [0,n) means j in [k, k+n-1), intersect [0,n-1] = [k, n-1] (if k>=0) or [0, n-1-k] (if k<0)
    # j+k in [0,n) means j in [-k, n-1-k), intersect [0,n-1]
    
    if k >= 0:
        # (1-h[j-k]) term: valid for j in [k, n-1]
        g[k:n] += (1.0 - h[:n-k])
        # -h[j+k] term: valid for j in [0, n-1-k]
        g[:n-k] -= h[k:n]
    else:
        kk = -k
        # (1-h[j-k]) = (1-h[j+kk]): valid for j in [0, n-1-kk]
        g[:n-kk] += (1.0 - h[kk:n])
        # -h[j+k] = -h[j-kk]: valid for j in [kk, n-1]
        g[kk:] -= h[:n-kk]
    
    return g
```

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = corr - np.max(corr)

    h_new = make_feasible(h_new, n)

            i, j = rng.randint(0, n, 2)
            if i != j:
                tr[i], tr[j] = tr[j], tr[i]
        elif r < 0.60:
            j = rng.randint(0, n)
            tr[j] = np.clip(tr[j] + noise * rng.randn(), 0.0, 1.0)
            tr = make_feasible(tr, n)
        elif r < 0.85:
            bs = rng.randint(2, 6)
            st = rng.randint(0, n - bs)
            tr[st:st+bs] = np.clip(tr[st:st+bs] + noise * 0.7 * rng.randn(bs), 0.0, 1.0)
            tr = make_feasible(tr, n)
        else:
            bs = min(15, n // 6)
            st = rng.randint(0, n - bs)
            tr[st:st+bs] = np.clip(tr[st:st+bs] + noise * 1.5 * rng.randn(bs), 0.0, 1.0)
            tr = make_feasible(tr, n)
        nc = compute_c5(tr)
        d = nc - cc
        if d < -1e-14 or (T > Tmin and rng.random() < np.exp(-min(d / T, 300))):
            h = tr
            cc = nc
            if nc < bc - 1e-14:
                bc = nc
                bh = h.copy()
        if it % 80 == 0:
            T = max(Tmin, T * cool)
        it += 1
    return bh, bc

def coord_pass(h, n, tlim, t0, rng):
    """Multi-pass coordinate descent."""
    bh = h.copy()
    bc = compute_c5(h)
    progress = True
    while progress:
        if time.time() - t0 > tlim - 3:
            break
        progress = False
        for j in rng.permutation(n):
            if time.time() - t0 > tlim - 3:
                return bh, bc
            old = bh[j]
            # Deterministic candidates + a few random
            cands = np.unique(np.concatenate([
                np.array([0.0, 1.0, 0.5]),
                np.linspace(0.0, 1.0, 11),
                np.clip(old + (rng.rand(6) - 0.5) * 0.06, 0, 1)
            ]))
            bv = old
            bo = bc
            for cv in cands:
                if abs(cv - old) < 5e-5:
                    continue
                tr = bh.copy()
                tr[j] = cv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < bo - 1e-14:
                    bo = c
                    bv = cv
            if bv != old:
                bh[j] = bv
                bh = make_feasible(bh, n)
                bc = bo
                progress = True
    return bh, bc

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        n_i = 640
        h0 = (rng.rand(n_i) < 0.5).astype(float)
        h0 = make_feasible(h0, n_i)
    
    bh = h0.copy()
    bc = compute_c5(bh)
    n = len(bh)
    
    # Phase 1: SA exploration (30%)
    tl = t0 + budget_s * 0.30
    h1, c1 = blocked_sa(bh, n, tl - time.time(), time.time(), rng, T0=0.05, Tmin=0.001, cool=0.9997, noise=0.12)
    if c1 < bc:
        bc, bh = c1, h1
    
    # Phase 2: Coordinate refinement (30%)
    tl2 = t0 + budget_s * 0.60
    h2, c2 = coord_pass(bh, n, tl2 - time.time(), time.time(), rng)
