def compute_c5(h):

def project_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 3.0
    hi = g.max() + 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0

        if s > target:

        else:

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_grad_exact(h, n, tau, top_k=20):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    # Soft weights
    shifted = (top_vals - max_c) / tau
    w = np.exp(shifted)
    w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):

        k = top_idx[idx]
        sh = k - n + 1  # shift

        h2, c5 = subgrad_polish(best_h, len(best_h), tl, 0.0008, 5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2
    rem = remaining()
    if rem > 30:
        tl = min(35, rem * 0.4)
        h2, c5 = subgrad_polish(best_h, len(best_h), tl, 0.0003, 3)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2

    # Phase 5: Multi-start perturbations
    rem = remaining()
    for pert in [0.004, 0.002, 0.008, 0.0015, 0.006]:
        rem = remaining()
        if rem < 45:
            break
        h_p = best_h + np.random.randn(len(best_h)) * pert
        h_p = project_feasible(h_p, len(best_h))
        tl = min(35, rem * 0.2)
        h_p, c5_p = nesterov_full(h_p, len(best_h), 5.0, tl, 0.006, 8)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = h_p

    # Phase 6: Final tiny-step subgradient
    rem = remaining()
    if rem > 15:
        tl = min(20, rem * 0.5)
        h2, c5 = subgrad_polish(best_h, len(best_h), tl, 0.0001, 2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h2

    # Final projection
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
