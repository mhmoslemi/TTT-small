import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_feasible(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    mid = 0.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        h = np.clip(v + mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
        if abs(s - target) < 1e-14:
            break
    return np.clip(v + mid, 0.0, 1.0)

def soft_max_obj(h, n, T, top_k, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    k = min(top_k, len(corr))
    top_idx = np.argpartition(corr, -k)[-k:]
    top_vals = corr[top_idx]
    shifted = top_vals - mc
    log_sum = np.log(np.sum(np.exp(T * shifted))) / T
    viol = (np.sum(h) - n / 2.0) ** 2
    return (mc + log_sum + lam * viol) * dx

def soft_max_grad(h, n, T, top_k, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    k = min(top_k, len(corr))
    top_idx = np.argpartition(corr, -k)[-k:]
    top_vals = corr[top_idx]
    shifted = top_vals - mc
    exp_vals = np.exp(T * shifted)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        m = idx - (n - 1)
        lo1 = max(0, -m)
        hi1 = min(n, n - m)
        if hi1 > lo1:
            grad[lo1:hi1] += w * (1.0 - h[lo1 + m:hi1 + m])
        lo2 = max(0, m)
        hi2 = min(n, n + m)
        if hi2 > lo2:
            grad[lo2:hi2] -= w * h[lo2 - m:hi2 - m]
    sum_viol = np.sum(h) - n / 2.0
    grad += 2.0 * lam * sum_viol
    return dx * grad

def lbfgsb_optimize(h_start, n, T, top_k, max_iter, lam=1e6):
    def obj(h_v):
        return soft_max_obj(h_v, n, T, top_k, lam)
    def grad(h_v):
        return soft_max_grad(h_v, n, T, top_k, lam)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, method='L-BFGS-B', jac=grad,
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-15,
                                'gtol': 1e-10, 'maxfun': max_iter * 4,
                                'maxls': 40})
        return project_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def coord_descent(h, n, n_rounds, time_limit=None, start_time=None):
