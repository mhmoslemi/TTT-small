import numpy as np
from scipy.optimize import minimize
import time

def fft_circ_corr(h):
    n = len(h)
    g = 1.0 - h
    H = np.fft.rfft(h)
    G = np.fft.rfft(g)
    return np.fft.irfft(H * np.conj(G), n)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmax(h) if diff < 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

        cnt = np.sum(mask)
        if cnt == 0:

    diff = target - np.sum(h)

    def f(h):

        mc = np.max(c)
        shifted = T * (c - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    def fprime(h):
        c = fft_circ_corr(h)
        mc = np.max(c)
        shifted = T * (c - mc)
        shifted = np.clip(shifted, -80, 80)
        exps = np.exp(shifted)
        sum_exp = np.sum(exps)
        weights = exps / sum_exp  # softmax weights
        # Gradient of soft max: dx * (max + logsumexp/T)
        # Gradient of c[i] wrt h[j]: (1-h[(j+i)%n]) - h[(j-i)%n]
        # Full gradient: dx * sum_i weights[i] * grad_ci
        grad_c = np.zeros(n)
        for i in range(n):
            if weights[i] > 1e-8:
                # d(c_i)/d(j) = (1-h[(j+i)%n]) - h[(j-i)%n]
                # This is: 1 - h_shifted_forward - h_shifted_backward
                # where h_shifted_forward[j] = h[(j+i)%n]
                # and h_shifted_backward[j] = h[(j-i)%n] = h[(-i+j)%n]
                hf = np.roll(h, -i)  # h[(j+i)%n]
                hb = np.roll(h, i)   # h[(j-i)%n]
                grad_c += weights[i] * (1 - hf - hb)
        viol = np.sum(h) - n / 2.0
        return dx * (mc + np.log(sum_exp) / T) * 0 + dx * grad_c + 2 * mu * viol * np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10})
    return make_feasible(res.x, n)

Hmm, the gradient computation with the roll for each i is O(n²) which defeats the purpose. Let me think of a better way.

Actually, for L-BFGS-B, I don't need to provide the gradient - it can use numerical differentiation. But that's slow too (n function evaluations per gradient).

Alternatively, I can compute the gradient using FFT:
grad_c[j] = sum_i weights[i] * (1 - h[(j+i)%n] - h[(j-i)%n])
= n - sum_i weights[i]*h[(j+i)%n] - sum_i weights[i]*h[(j-i)%n]
= n - circ_conv(h, weights)[j] - circ_conv(h, flip(weights))[j]

Where circ_conv is circular convolution computed via FFT in O(n log n)!

So:
grad_c = n - ifft(FFT(h) * FFT(weights)) - ifft(FFT(h) * FFT(flip(weights)))

Wait, circular convolution: (h * w)[j] = sum_i h[(j-i)%n] * w[i]
This is ifft(FFT(h) * FFT(w)).

And (h * flip(w))[j] = sum_i h[(j-i)%n] * w[-i mod n] = sum_i h[(j+i)%n] * w[i]

So:
sum_i weights[i] * h[(j+i)%n] = (h * flip(weights))[j]
sum_i weights[i] * h[(j-i)%n] = (h * weights)[j]

grad_c = n - (h*flip(w))[j] - (h*w)[j]

Computable in O(n log n) via FFT!

import numpy as np
from scipy.optimize import minimize
import time

def fft_circ_corr(h):
    n = len(h)
    H = np.fft.rfft(h)
    G = np.fft.rfft(1.0 - h)
    return np.fft.irfft(H * np.conj(G), n)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmax(h)) if diff < 0 else int(np.argmin(h))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample(h, new_n):

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)

    h = np.asarray(h, dtype=float).copy()
    h = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h))
    n_target = n // 2
    if n_ones > n_target:

        rng2 = np.random.RandomState(0)
        rng2.shuffle(one_idx)

    elif n_ones < n_target:

        rng2 = np.random.RandomState(0)
        rng2.shuffle(zero_idx)

def binary_hull(h, n, ones, zeros, k_worst, max_tries, rng):
    """Targeted binary local search focusing on worst lags."""
    curr_c = fft_circ_corr(h)
    curr_max = float(np.max(curr_c))
    for _ in range(max_tries):
        k_star = k_worst % n
        # Find good swap candidates targeting k_star
        best_delta = 0
        best_i, best_j = -1, -1
        n_cand = min(80, len(ones) * len(zeros))
        for _ in range(n_cand):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            i, j = ones[ii], zeros[jj]
            # Delta at k_star: h[(i+k)%n] - h[(j+k)%n] + h[(i-k)%n] - h[(j-k)%n]
            dk = (h[(i + k_star) % n] - h[(j + k_star) % n] +
                  h[(i - k_star) % n] - h[(j - k_star) % n])
            if dk < best_delta:
                best_delta = dk
                best_i, best_j = ii, jj
        if best_i < 0:
            break
        i, j = ones[best_i], zeros[best_j]
        h[i] = 0.0
        h[j] = 1.0

        zeros[best_j] = i

def subgrad_step(h, n, alpha, nr_steps):
    """Sub-gradient descent targeting worst lag."""
    for _ in range(nr_steps):
        c = fft_circ_corr(h)
        k_star = int(np.argmax(c))
        # grad_j = (1 - h[(j+k)%n]) - h[(j-k)%n]
        # Use FFT: grad = n - circ_conv(h, delta_k) - circ_conv(h, delta_-k)
        e_k = np.zeros(n)
        e_k[k_star] = 1.0
        e_mk = np.zeros(n)
        e_mk[(-k_star) % n] = 1.0
        Hf = np.fft.rfft(h)
        Ek = np.fft.rfft(e_k)
        Emk = np.fft.rfft(e_mk)
        conv_f = np.fft.irfft(Hf * Ek, n)
        conv_b = np.fft.irfft(Hf * Emk, n)
        grad = n - conv_f - conv_b
        h = h - alpha * grad
        h = np.clip(h, 0.0, 1.0)
        h = make_feasible(h, n)
    return h
