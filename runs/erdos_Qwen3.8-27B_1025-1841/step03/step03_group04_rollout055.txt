    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

        h = np.clip(h, 0.0, 1.0)

def snap_binary(h, n):

    b = h.copy()

    b = (b >= thresh).astype(float)

    ones = int(np.sum(b))

        # Re-threshold
        sorted_idx = np.argsort(b)[::-1]
        b = np.zeros(n)
        b[sorted_idx[:n//2]] = 1.0
    return b

    dx = 2.0 / n

    n_iter = 0
