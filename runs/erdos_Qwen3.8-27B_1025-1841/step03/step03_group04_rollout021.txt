    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        if diff > 0:

        else:

        else:

    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def corr_gradient(h, lag):
    """Gradient of correlation at given lag w.r.t. h"""
    N = len(h)
    grad = np.zeros(N)
    if lag >= 0:
        # dC/dh[j] = (1-h[j-lag]) if lag <= j <= N-1
        #           - h[j+lag] if 0 <= j <= N-1-lag
        j_range1 = np.arange(lag, N)
        if len(j_range1) > 0:
            grad[j_range1] += (1.0 - h[j_range1 - lag])
        j_range2 = np.arange(0, N - lag)
        if len(j_range2) > 0:
            grad[j_range2] -= h[j_range2 + lag]
    else:
        p = -lag
        # dC/dh[j] = (1-h[j+p]) if 0 <= j <= N-1-p
        #           - h[j-p] if p <= j <= N-1
        j_range1 = np.arange(0, N - p)
        if len(j_range1) > 0:
            grad[j_range1] += (1.0 - h[j_range1 + p])
        j_range2 = np.arange(p, N)
        if len(j_range2) > 0:
            grad[j_range2] -= h[j_range2 - p]
    return grad

def soft_max_value_and_grad(h, top_idx, T):
    """Compute soft-max value and gradient"""
    N = len(h)
    dx = 2.0 / N
    corr = get_corr(h)
    if top_idx is None:
        vals = corr
        idxs = np.arange(len(corr))
    else:
        vals = corr[top_idx]
        idxs = top_idx
    
    mc = np.max(vals)
    exp_vals = np.exp(T * (vals - mc))
    w = exp_vals / np.sum(exp_vals)
    val = (mc + np.log(np.sum(exp_vals)) / T) * dx
    
    # Gradient: weighted sum of correlation gradients
    grad = np.zeros(N)
    for i, m in enumerate(idxs):
        if w[i] < 1e-12:
            continue
        lag = m - (N - 1)
        grad += w[i] * corr_gradient(h, lag)
    grad *= dx
    
    return val, grad

def _con_fun(h, n):
    return np.sum(h) - n / 2.0

def _con_jac(n):
    return np.ones(n)
