        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def binary_round(h, n):
    """Round h to binary while maintaining sum = n/2"""
    target = n // 2
    # Start with h > 0.5 as 1
    b = (h > 0.5).astype(float)
    s = b.sum()
    if s > target:
        # Need to flip s - target ones to zero
        # Flip the ones with lowest h values
        ones = np.where(b == 1)[0]
        order = h[ones].argsort()[:s - target]
        b[ones[order]] = 0
    elif s < target:
        # Need to flip target - s zeros to one
        zeros = np.where(b == 0)[0]
        order = h[zeros].argsort()[::-1][:target - s]
        b[zeros[order]] = 1
    return b

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)

def downsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)
