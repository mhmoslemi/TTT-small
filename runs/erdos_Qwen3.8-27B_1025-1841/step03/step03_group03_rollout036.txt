```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # Second pass for exact feasibility
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_objective_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    log_sum = max_corr + np.log(np.sum(np.exp(t * (corr - max_corr))))
    obj = (max_corr + np.log(np.sum(np.exp(t * (corr - max_corr)))) / t) * dx
    
    p = np.exp(t * (corr - max_corr))
    p /= p.sum()
    
    # Gradient
    j = np.arange(n)[:, None]
    k = np.arange(2*n-1)[None, :]
    idx1 = j + k - (n-1)
    idx2 = j - k + (n-1)
    
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    h1 = np.where(valid1, h[np.clip(idx1, 0, n-1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    A = (p[None, :] * h1).sum(axis=1)
    B = (p[None, :] * h2).sum(axis=1)
    
    grad = (p.sum() - A - B) * dx
    return obj, grad

def lbfgsb_optimize(h_start, n, t, maxiter=200):
    bounds = [(0.0, 1.0)] * n
    
    def obj_and_grad(x):
        x_clipped = np.clip(x, 0.0, 1.0)
        o, g = softmax_objective_and_grad(x_clipped, t)
        # Add penalty for sum constraint
        sum_val = np.sum(x_clipped) - n/2.0
        o += 100.0 * sum_val**2
        g = g + 200.0 * sum_val / n * np.ones(n)
        return o, g
    
    res = minimize(obj_and_grad, h_start, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'gtol': 1e-8, 'ftol': 1e-14})
    h_opt = make_feasible(res.x, n)
    return h_opt

def coordinate_descent(h, n, max_rounds=20, time_limit=None, start_time=None):
    dx = 2.0 / n
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 20:
                    return h, best_c
            old = h[j]
            # Try a few candidates around old
            candidates = [0.0, 1.0, old - 0.1, old - 0.05, old - 0.01,
                          old + 0.01, old + 0.05, old + 0.1]
            best_obj = best_c
            best_val = old
            for cv in candidates:
                if cv < -0.01 or cv > 1.01:
                    continue
                cv = np.clip(cv, 0.0, 1.0)
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c
```

```python
def compute_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_t = np.exp(t * (corr - max_c))
    p = exp_t / exp_t.sum()
    
    # Direct computation using broadcasting
    j = np.arange(n)
    k = np.arange(2*n-1)
    # idx1 = j[:,None] + (k[None,:] - (n-1))  → j + k - (n-1), shape (n, 2n-1)
    # idx2 = j[:,None] - (k[None,:] - (n-1))  → j - k + (n-1), shape (n, 2n-1)
    
    shift = k - (n-1)  # shape (2n-1,), values from -(n-1) to (n-1)
    idx1 = j[:, None] + shift[None, :]  # j + shift[k], shape (n, 2n-1)
    idx2 = j[:, None] - shift[None, :]  # j - shift[k], shape (n, 2n-1)
    
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    
    h1 = np.where(valid1, h[np.clip(idx1, 0, n-1)], 0.0)
    h2 = np.where(valid2, h[np.clip(idx2, 0, n-1)], 0.0)
    
    A = (p[None, :] * h1).sum(axis=1)
    B = (p[None, :] * h2).sum(axis=1)
    
    grad = (p.sum() - A - B) * dx
    return grad
```

- The log-sum-exp approximation might not be accurate enough at high t
- The penalty for the sum constraint might interfere

To address these:
- Use multiple random restarts
- Use a very high t (like 1000+) in the final stages to make the approximation very close to the true max
- Use a separate correction step for the sum constraint

Let me also consider: after L-BFGS-B converges, the sum constraint might not be exactly satisfied (due to the penalty). I should always re-project to feasibility.
