def compute_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    mc = np.max(corr)
    exp_vals = np.exp(T * (corr - mc))
    w = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        wi = w[idx]
        if wi < 1e-30:
            continue
        if k >= 0:
            # pos: j in [k, n-1], contributes wi*(1-h[j-k])
            grad[k:] += wi * (1.0 - h[:n-k])
            # neg: j in [0, n-1-k], contributes -wi*h[j+k]
            if n - 1 - k >= 0:
                grad[:n-k] -= wi * h[k:n]
        else:
            ak = -k  # |k|
            # pos: j in [0, n-1-ak], contributes wi*(1-h[j+ak])
            grad[:n-ak] += wi * (1.0 - h[ak:])
            # neg: j in [ak, n-1], contributes -wi*h[j-ak]
            grad[ak:] -= wi * h[:n-ak]
    
    return grad * dx
