                best_c = best_c_t
                improved = True
        if not improved:
            break
    return best, best_c

def adam_grad_descent(h, n, max_iters, lr, deadline):
    x = h.copy()
    best = h.copy()
    best_c = compute_c5(x)
    m = np.zeros(n)
    v = np.zeros(n)
    for t in range(max_iters):
        if time.time() > deadline - 20:
            break
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        T = 150.0
        weights = np.exp(T * (corr - mc))
        weights /= weights.sum()
        grad = weighted_corr_gradient(x, n, weights)
        b1, b2, eps = 0.8, 0.999, 1e-8
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** (t + 1))
        v_hat = v / (1.0 - b2 ** (t + 1))
        x_new = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x_new = np.clip(x_new, 0.0, 1.0)
        target = n / 2.0
        diff = target - np.sum(x_new)
        if abs(diff) > 1e-10:
            if diff > 0:
                mask = x_new < 1.0 - 1e-10
            else:
                mask = x_new > 1e-10
            cnt = np.sum(mask)
            if cnt > 0:
                x_new[mask] += diff / cnt
                x_new = np.clip(x_new, 0.0, 1.0)
        x = x_new
        c = compute_c5(x)
        if c < best_c:
            best = x.copy()
            best_c = c
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_N, _OPT_T
    start_time = time.time()
    deadline = start_time + budget_s
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Multi-resolution SLSQP
    configs = [
        (960, 100.0, 400),
        (960, 350.0, 350),
        (1280, 100.0, 300),
        (1280, 350.0, 300),
        (1600, 150.0, 250),
        (1600, 400.0, 250),
        (1920, 200.0, 200),
        (1920, 500.0, 180),
    ]

    for n_pts, T, maxiter in configs:
        if time.time() > deadline - 120:
            break
        h_start = upsample_h(best_h, n_pts)
        n_restarts = 3 if n_pts <= 1280 else 2
        for r in range(n_restarts):
            if time.time() > deadline - 80:
                break
            if r == 0:
                x0 = h_start.copy()
            elif r == 1:
                x0 = h_start + rng.randn(n_pts) * 0.012
                x0 = make_feasible(x0, n_pts)
            else:
                x0 = h_start + rng.randn(n_pts) * 0.035
                x0 = make_feasible(x0, n_pts)

            _OPT_N = n_pts
            _OPT_T = T
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_obj, x0, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': maxiter, 'ftol': 1e-13})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_pts)
            except Exception:
                pass

    # Phase 2: Adam gradient descent from multiple starts
    if time.time() < deadline - 120:
        n_pts = len(best_h)
        h_starts = [best_h.copy()]
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, n_pts)
        h_starts.append(h_bin)
        h_noisy = best_h + rng.randn(n_pts) * 0.01
        h_noisy = make_feasible(h_noisy, n_pts)
        h_starts.append(h_noisy)

        for hs in h_starts:
            if time.time() > deadline - 60:
                break
            n_iters = min(80, max(20, int((deadline - time.time() - 40) // 1.5)))
            if n_iters < 15:
                break
            h_res, c5_res = adam_grad_descent(hs, n_pts, max_iters=n_iters, lr=0.004, deadline=deadline)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res

    # Phase 3: Binary swap search
    if time.time() < deadline - 80:
        n_pts = len(best_h)
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, n_pts)
        h_sw, c5_sw = binary_swap_search(h_bin, n_pts, max_rounds=20, rng=rng)
        if c5_sw < best_c5:
            best_c5 = c5_sw
            best_h = h_sw

        for thresh in [0.3, 0.4, 0.6, 0.7]:
            if time.time() > deadline - 30:
                break
            h_t = (best_h > thresh).astype(float)
            h_t = make_feasible(h_t, n_pts)
            h_sw2, c5_sw2 = binary_swap_search(h_t, n_pts, max_rounds=8, rng=rng)
            if c5_sw2 < best_c5:
                best_c5 = c5_sw2
                best_h = h_sw2

    # Phase 4: Coordinate descent (multi-pass)
    if time.time() < deadline - 40:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n_pts, max_rounds=40, deadline=deadline)

    # Phase 5: Final SLSQP polish at very high temperature
    if time.time() < deadline - 40:
        n_pts = len(best_h)
        for T_f in [600.0, 1200.0, 2000.0]:
            if time.time() > deadline - 30:
                break
            _OPT_N = n_pts
            _OPT_T = T_f
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_obj, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-14})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_pts)
            except Exception:
                pass

    # Phase 6: Micro coordinate descent
    if time.time() < deadline - 8:
        n_pts = len(best_h)
        for j in range(n_pts):
            if time.time() > deadline - 4:
                break
            old = best_h[j]
            for delta in [-0.004, 0.004, -0.001, 0.001, -0.008, 0.008]:
                cand = max(0.0, min(1.0, old + delta))
                if abs(cand - old) < 1e-8:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c
                    break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
