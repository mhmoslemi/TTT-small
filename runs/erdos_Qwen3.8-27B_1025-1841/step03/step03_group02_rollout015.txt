def slsqp_polish(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        mc = float(np.max(corr))
        shifted = T * (corr - mc)
        lse = mc + np.log(np.sum(np.exp(shifted))) / T
        return lse * dx

    def cfun(x):
        return np.sum(x) - n / 2.0

    def cjac(x):
        return np.ones(n)

    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=[(0.0, 1.0)]*n,
                   constraints=[{'type':'eq','fun':cfun,'jac':cjac}],
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    except Exception:

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: Binary SA at parent resolution (long)
    b1, c5_1 = binary_sa(h_init, n0, 300000, 0.008, 0.000005, rng, 180)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, b1.copy()

    # Phase 2: SLSQP continuous polish
    rem = budget_s - (time.time() - start_time)
    if rem > 120:
        n_cur = len(best_h)
        try:
            x_r = slsqp_polish(best_h, n_cur, 80.0, 250, 1e-14)
            c5_r = compute_c5(x_r)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, make_feasible(x_r, n_cur)
        except Exception:
            pass

    # Phase 3: Binary SA from continuous result
    rem = budget_s - (time.time() - start_time)
    if rem > 200:
        n_cur = len(best_h)
        b2, c5_2 = binary_sa(best_h, n_cur, 250000, 0.004, 0.000001, rng, min(rem-80, 200))
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, b2.copy()

    # Phase 4: Higher resolution
    rem = budget_s - (time.time() - start_time)
    if rem > 350:
        for n_try in [n0 + 480, 1440, 1920]:
            if n_try <= len(best_h) or n_try > 2000:
                continue
            rem = budget_s - (time.time() - start_time)
            if rem < 200:
                break
            h_up = upsample(best_h, n_try)
            b_up, c5_up = binary_sa(h_up, n_try, 150000, 0.005, 0.00001, rng, min(rem-100, 150))
            if c5_up < best_c5:
                best_c5, best_h = c5_up, b_up.copy()
            else:
                break

    # Phase 5: Lower resolution exploration
    rem = budget_s - (time.time() - start_time)
    if rem > 250 and len(best_h) > 640:
        n_lo = max(len(best_h) // 2, 480)
        h_lo = upsample(best_h, n_lo)
        b_lo, c5_lo = binary_sa(h_lo, n_lo, 200000, 0.008, 0.00001, rng, min(rem-50, 150))
        if c5_lo < best_c5:
            best_c5, best_h = c5_lo, b_lo.copy()

    # Phase 6: SLSQP polish (high T)
    rem = budget_s - (time.time() - start_time)
    if rem > 100:
        n_cur = len(best_h)
        try:
            x_r = slsqp_polish(best_h, n_cur, 300.0, 300, 1e-15)
            c5_r = compute_c5(x_r)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, make_feasible(x_r, n_cur)
        except Exception:
            pass

    # Phase 7: Final binary SA (low T, long)
    rem = budget_s - (time.time() - start_time)
    if rem > 80:
        n_cur = len(best_h)
        b_f, c5_f = binary_sa(best_h, n_cur, 200000, 0.002, 0.0000005, rng, min(rem-30, 100))
        if c5_f < best_c5:
            best_c5, best_h = c5_f, b_f.copy()

    # Phase 8: Coordinate descent
    rem = budget_s - (time.time() - start_time)
    if rem > 40:
        n_cur = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_cur, start_time, budget_s, 10)

    # Phase 9: Final SLSQP
    rem = budget_s - (time.time() - start_time)
    if rem > 60:
        n_cur = len(best_h)
        try:
            x_r = slsqp_polish(best_h, n_cur, 800.0, 200, 1e-16)
            c5_r = compute_c5(x_r)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, make_feasible(x_r, n_cur)
        except Exception:
            pass

    # Phase 10: One more binary SA if time
    rem = budget_s - (time.time() - start_time)
    if rem > 50:
        n_cur = len(best_h)
        b_l, c5_l = binary_sa(best_h, n_cur, 100000, 0.001, 0.0000001, rng, min(rem-10, 50))
        if c5_l < best_c5:
            best_c5, best_h = c5_l, b_l.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
