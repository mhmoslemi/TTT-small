- Evaluating at the breakpoints (where the argmax of the correlation changes)
- Or using ternary search (since it's convex)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_onto_constraints(z, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(z - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(proj)
    if abs(diff) > 1e-10:

        nm = np.sum(mask)
        if nm > 0:
            proj[mask] = np.clip(proj[mask] + diff / nm, 0.0, 1.0)

    sm = np.max(s)

def coordinate_descent(h, n, n_pass=3, time_limit=60.0):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    t0 = time.time()
    for pass_idx in range(n_pass):
        if time.time() - t0 > time_limit:
            break
        improved = False
        candidate_vals = [0.0, 1.0]
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h_curr[j]
            if old < 1e-10 or old > 1.0 - 1e-10:
                test_vals = [0.0, 1.0]
            else:
                test_vals = [0.0, 1.0, old - 0.1, old - 0.05, old - 0.01,
                            old + 0.01, old + 0.05, old + 0.1,
                            0.5 * (old + 0.0), 0.5 * (old + 1.0)]
            test_vals = [v for v in test_vals if 0.0 <= v <= 1.0 and abs(v - old) > 1e-12]
            for v in test_vals:
                trial = h_curr.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h_curr = trial
                    improved = True
        if not improved:
            break
    return h_curr, best_c5

    def obj(x):
        return smoothed_obj(x, n, T)

        return np.ones(n)

                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': cons, 'jac': cons_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return make_feasible(res.x, n)
