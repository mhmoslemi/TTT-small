    # Phase 1: Nesterov gradient at parent resolution with varying t
    n1 = len(best_h)
    h1 = best_h.copy()
    for t, lr in [(30, 0.05), (80, 0.03), (200, 0.015), (500, 0.008)]:
        if remaining() < 250:
            break
        h1, c1 = nesterov_optimize(h1, n1, t, lr, 3000, start_time, start_time + 200)
        if c1 < best_c5 - 1e-9:
            best_c5 = c1
            best_h = h1

    # Phase 2: Pair coordinate descent
    if remaining() > 200:
        best_h, best_c5, _ = pair_cd(best_h, len(best_h), best_c5, 30, start_time, start_time + 180)

    # Phase 3: DE exploration at lower resolution to find new basins
    if remaining() > 250:
        n_de = 256
        h_de, c_de = de_explore(n_de, 30, 300, start_time, start_time + 180, rng)
        if c_de < best_c5 + 0.01:
            h_up = upsample_h(h_de, n1)
            h_up, c_up = nesterov_optimize(h_up, n1, 100, 0.02, 2000, start_time, start_time + 150)
            if c_up < best_c5 - 1e-9:
                best_c5 = c_up
                best_h = h_up

    # Phase 4: Simulated annealing for global exploration
    if remaining() > 200:
        h_sa, c_sa = sa_optimize(best_h, len(best_h), start_time, start_time + 150, rng, t0=0.5, t1=0.005)
        if c_sa < best_c5 - 1e-9:
            best_c5 = c_sa
            best_h = h_sa

    # Phase 5: Upsample to higher resolution
    n2 = max(len(best_h) * 2, 1280)
    if remaining() > 250:
        h2 = upsample_h(best_h, n2)
        for t, lr in [(50, 0.03), (150, 0.015), (400, 0.007)]:
            if remaining() < 200:
                break
            h2, c2 = nesterov_optimize(h2, n2, t, lr, 2000, start_time, start_time + 120)
            if c2 < best_c5 - 1e-9:
                best_c5 = c2
                best_h = h2
        if remaining() > 150:
            best_h, best_c5, _ = pair_cd(best_h, len(best_h), best_c5, 15, start_time, start_time + 130)

    # Phase 6: Single-coordinate CD (dense candidates)
    if remaining() > 100:
        n_cd = len(best_h)
        best_h, best_c5 = single_cd(best_h, n_cd, best_c5, 15, start_time, start_time + 100)

    # Phase 7: Final pair CD polish
    if remaining() > 50:
        best_h, best_c5, _ = pair_cd(best_h, len(best_h), best_c5, 10, start_time, start_time + 45)

    # Phase 8: Try binary projection
    if remaining() > 30:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, len(best_h))
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5 - 1e-9:
            best_c5 = c_bin
            best_h = h_bin
            best_h, best_c5 = single_cd(best_h, len(best_h), best_c5, 5, start_time, start_time + 25)

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
