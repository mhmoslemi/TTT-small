def gradient_step(h, T, lr, momentum_buf, beta=0.9):
    N = len(h)
    dx = 2.0 / N
    H = np.fft.rfft(h)
    H1h = np.fft.rfft(1.0 - h)
    R = np.fft.irfft(H * np.conj(H1h), N)  # circular cross-correlation
    
    # Softmax weights
    R_scaled = R * dx / T
    R_scaled = R_scaled - np.max(R_scaled)
    w = np.exp(R_scaled)
    w /= np.sum(w)
    
    # Gradient
    W = np.fft.rfft(w)
    conv = np.fft.irfft(W * H, N)
    grad = (dx / N) * (1.0 - 2.0 * conv)
    
    # Momentum update
    momentum_buf[:] = beta * momentum_buf - lr * grad
    h_new = h + momentum_buf
    
    # Project
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, N)
    return h_new

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):

            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)

def upsample_h(h, new_n):

    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def circ_corr(h):
    """Circular cross-correlation of h and (1-h)"""
    N = len(h)
    H = np.fft.rfft(h)
    H1 = np.fft.rfft(1.0 - h)
    return np.fft.irfft(H * np.conj(H1), N)

    W = np.fft.rfft(w)
    conv = np.fft.irfft(W * H, N)
    grad = (2.0 / N) * (1.0 - 2.0 * conv)
    
    return obj, grad

            h_new = np.clip(h_new, 0.0, 1.0)

            h = h_new
        
        # Check improvement

        if best_c5 is None or c5 < best_c5:

            best_h = h.copy()
    
    return best_h, best_c5

def binary_local_search(h_bin, n, rng, max_rounds=50):
    """Local search on binary patterns"""
    h = h_bin.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()

                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)

                zeros = np.append(zeros, oi)
                improved = True

                h[oj] = 0.0
        if not improved:

This is a spectral design problem. The optimal solution would have |c_k|² decreasing with k (to keep the sum bounded) while making the autocorrelation as flat as possible.

But this is hard to optimize with the [0,1] constraint. The gradient descent in primal space is probably more practical.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0 - 1e-10) if diff > 0 else (h > 1e-10)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff/cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return make_feasible(np.interp(new_x, old_x, h, period=2.0), new_n)

def circ_corr_and_grad(h, T):

    dx = 2.0 / N
    H = np.fft.rfft(h)
    H1 = np.fft.rfft(1.0 - h)
    R = np.fft.irfft(H * np.conj(H1), N)
    Rv = R * dx
    # Normalized log-sum-exp
    Rm = Rv.max()
    logsum = Rm + (1.0/T) * np.log(np.sum(np.exp(Rv - Rm)))
    obj = T * logsum
    # Weights

    w /= w.sum()
    # Gradient: (2/N) * (1 - 2 * circ_conv(w, h))
    W = np.fft.rfft(w)
    conv = np.fft.irfft(W * H, N)
    grad = (2.0 / N) * (1.0 - 2.0 * conv)
    return obj, grad
