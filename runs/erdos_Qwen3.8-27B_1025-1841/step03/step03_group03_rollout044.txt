import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} via bisection on lambda"""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    # Ensure the range brackets the solution
    while np.sum(np.clip(h - lo, 0, 1)) < target:
        lo -= 1.0
    while np.sum(np.clip(h - hi, 0, 1)) > target:
        hi += 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_proj = np.clip(h - lam, 0.0, 1.0)
    # Fix tiny numerical errors
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0, 1)
    return h_proj

def compute_subgradient(h, n, corr=None, t=50.0):
    """Compute weighted subgradient of max_c corr[c]*dx using softmax weighting"""
    dx = 2.0 / n
    if corr is None:
        corr = np.correlate(h, 1.0 - h, mode='full')
    
    max_c = np.max(corr)
    # Softmax weights
    exp_vals = np.exp(t * (corr - max_c))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute gradient as weighted sum
    grad = np.zeros(n)
    n_corr = len(corr)
    # Only process lags with significant weight
    thresh = np.max(weights) * 0.01
    active = np.where(weights > thresh)[0]
    
    for k in active:
        L = k - (n - 1)  # actual lag
        w = weights[k]
        if L >= 0:
            if L < n:
                grad[L:] += w * (1.0 - h[:n-L])
                if n - L > 0:
                    grad[:n-L] -= w * h[L:]
        else:
            l = -L
            if l < n:
                if n - l > 0:
                    grad[:n-l] += w * (1.0 - h[l:])
                    grad[l:] -= w * h[:n-l]
    
    grad *= dx
    return grad, max_c * dx

def subgradient_descent(h, n, max_iter=5000, t=50.0, start_time=None, time_limit=None):
    """Projected subgradient descent with adaptive step size"""
    dx = 2.0 / n
    best_c = compute_c5(h)
    best_h = h.copy()
    step = 0.1  # initial step size
    
    for it in range(max_iter):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit - 30:
                break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        c_val = max_c * dx
        
        if c_val < best_c:
            best_c = c_val
            best_h = h.copy()
        
        grad, _ = compute_subgradient(h, n, corr, t)
        
        # Backtracking line search
        step = 0.1
        for ls in range(10):
            h_new = h - step * grad
            h_new = project_feasible(h_new, n)
            c_new = compute_c5(h_new)
            if c_new < c_val - 1e-10 or ls == 9:
                break
            step *= 0.5
        
        h = h_new
        
        # Adaptive: if we didn't improve, reduce t (focus more on max)
        if it % 500 == 0:
            t = max(10.0, t * 0.9)
    
    return best_h, best_c

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_descent(h, n, max_rounds=20, time_limit=None, start_time=None):
    dx = 2.0 / n
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 20:
                    return h, best_c
            old = h[j]
            best_obj = best_c
            best_val = old
            for cv in [0.0, 1.0, 0.5, old+0.1, old-0.1, old+0.05, old-0.05]:
                cv = np.clip(cv, 0.0, 1.0)
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:

            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h_p = np.clip(h - lam, 0, 1)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-10:
        mask = (h_p > 1e-10) & (h_p < 1 - 1e-10)
        cnt = mask.sum()
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0, 1)

def smooth_obj_grad(h, n, t, corr=None):
    dx = 2.0 / n
    if corr is None:
        corr = compute_corr(h)
    mc = np.max(corr)
    ev = np.exp(t * (corr - mc))
    w = ev / np.sum(ev)
    obj_val = (mc + np.log(np.sum(ev)) / t) * dx
    grad = np.zeros(n)
    w_thresh = w.max() * 0.003
    active = np.where(w > w_thresh)[0]

        L = k - (n - 1)
        wk = w[k]
        if L >= 0:

                if L == 0:

                grad[:n-l] += wk * (1.0 - h[l:])
                grad[l:] -= wk * h[:n-l]
    grad *= dx * t
    return obj_val, grad

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_descent(h, n, max_rounds, start_time, time_limit):
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                return h, best_c
            old = h[j]
            best_obj = best_c
            best_val = old
            cands = [0.0, 1.0, 0.5, old+0.1, old-0.1, old+0.05, old-0.05]
            for cv in cands:
                cv = np.clip(cv, 0.0, 1.0)
                if abs(cv - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-13:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

        best_h = h2.copy()
    cutoff2 = start_time + budget_s * 0.40
    t_sched2 = [20, 40, 80, 150]
    h_p2, c_p2 = projected_grad_descent(h2, n2, t_sched2, start_time, cutoff2)
    if c_p2 < best_c5:
        best_c5 = c_p2
        best_h = h_p2
    
    # Phase 3: Coordinate descent at 1280
    cutoff3 = start_time + budget_s * 0.55
    h_cd, c_cd = coord_descent(best_h, n2, max_rounds=30, start_time=start_time, time_limit=cutoff3)
    if c_cd < best_c5:
        best_c5 = c_cd
        best_h = h_cd
    
    # Phase 4: Upsample to 2048
    n3 = 2048
    h3 = upsample_h(best_h, n3)
    c3 = compute_c5(h3)
    if c3 < best_c5:
        best_c5 = c3
        best_h = h3.copy()
    cutoff4 = start_time + budget_s * 0.75
    t_sched3 = [30, 60, 120]
    h_p3, c_p3 = projected_grad_descent(h3, n3, t_sched3, start_time, cutoff4)
    if c_p3 < best_c5:
        best_c5 = c_p3
        best_h = h_p3
    
    # Phase 5: Coordinate descent at 2048
    cutoff5 = start_time + budget_s * 0.85
    h_cd2, c_cd2 = coord_descent(best_h, n3, max_rounds=15, start_time=start_time, time_limit=cutoff5)
    if c_cd2 < best_c5:
        best_c5 = c_cd2
        best_h = h_cd2
    
    # Phase 6: Binary local search
    cutoff6 = start_time + budget_s * 0.93
    h_bin, c_bin = binary_local_search(best_h, len(best_h), start_time, cutoff6)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin
        # Quick coord descent on binary
        h_cd3, c_cd3 = coord_descent(h_bin, len(h_bin), max_rounds=5, start_time=start_time, time_limit=start_time + budget_s - 30)
        if c_cd3 < best_c5:
            best_c5 = c_cd3
            best_h = h_cd3
    
    # Final: also try at the original parent resolution with binary
    if len(best_h) != n1:
        h_res = np.interp(np.linspace(0, 2, n1), np.linspace(0, 2, len(best_h)), best_h)
        h_res = make_feasible(h_res, n1)
        c_res = compute_c5(h_res)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res
            h_cd4, c_cd4 = coord_descent(h_res, n1, max_rounds=10, start_time=start_time, time_limit=start_time + budget_s - 20)
            if c_cd4 < best_c5:
                best_c5 = c_cd4
                best_h = h_cd4
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
