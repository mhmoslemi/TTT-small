def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0

        (0.012, 80.0, 0.08),
    ]
    for scale, tau, lr_p in perturb_params:
        rem = get_remaining(start_time, budget_s)
        if rem < 55:
            break
        h_p = best_h + np.random.randn(n_init) * scale
        h_p = project_to_feasible(h_p, n_init)
        tl = min(45, rem * 0.3)
        h_p, c5_p = soft_nesterov(h_p, n_init, tau=tau, time_limit=tl, lr=lr_p, top_k=10, check_interval=5)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()
            rem = get_remaining(start_time, budget_s)
            if rem > 30:
                h_f, c5_f = ultra_fine(best_h, n_init, time_limit=min(25, rem * 0.5))
                if c5_f < best_c5:
                    best_c5, best_h = c5_f, h_f.copy()

    # Phase 6: Micro-step final polish
    rem = get_remaining(start_time, budget_s)
    if rem > 40:
        tl = min(35, rem * 0.6)
        h1, c5 = micro_step(best_h, n_init, time_limit=tl)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 7: One more ultra-fine
    rem = get_remaining(start_time, budget_s)
    if rem > 20:
        tl = min(18, rem * 0.7)
        h1, c5 = ultra_fine(best_h, n_init, time_limit=tl)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Final projection and evaluation
    best_h = project_to_feasible(best_h, n_init)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_init
