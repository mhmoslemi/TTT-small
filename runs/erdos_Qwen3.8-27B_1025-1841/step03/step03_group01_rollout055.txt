```python
def compute_grad_soft(h, sw, dx, mu):
    n = len(h)
    d = n - 1 - np.arange(2*n - 1)  # d[c] = n-1-c
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for c in range(2*n - 1):
        w = sw[c]
        if w < 1e-15:
            continue
        dc = d[c]
        # First term: (1-h[j+dc]) for valid j
        j_range = np.arange(max(0, -dc), min(n, n-dc))
        if len(j_range) > 0:
            idx = j_range + dc
            grad[j_range] += w * one_minus_h[idx]
        # Second term: -h[j-dc] for valid j
        j_range2 = np.arange(max(0, dc), min(n, n+dc))
        if len(j_range2) > 0:
            idx2 = j_range2 - dc
            grad[j_range2] -= w * h[idx2]
    grad *= dx
    grad += 2 * mu * (np.sum(h) - n/2.0)
    return grad
```
