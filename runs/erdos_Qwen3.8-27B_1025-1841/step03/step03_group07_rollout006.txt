def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def grad_overlap(h, top_idx, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for t_i in range(len(top_idx)):
        pk = weights[t_i]
        k = top_idx[t_i]

        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    return grad

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def grad_topk(h, top_idx, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for t_i in range(len(top_idx)):
        pk = weights[t_i]
        k = int(top_idx[t_i])
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    return grad


def adam_nest_opt(h_in, n, tl, tau0, tau_end, tk0, tk_end, lr0):
    dx = 2.0 / n
    h = project_to_feasible(h_in.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t_ad = 0
    h_prev = h.copy()
    t_nest = 1.0
    lr = lr0
    stall = 0

    for it in range(20000000):
        if time.time() - start > tl:
            break
        frac = min(1.0, it / max(200, tl * 40.0))
        tau = tau0 * (tau_end / tau0) ** frac
        tk = max(3, int(tk0 + (tk_end - tk0) * frac))
        mu = min(0.9, (t_nest - 1.0) / (t_nest + 2.0))
        y = np.clip(h + mu * (h - h_prev), 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = corr.max()
        if tk < len(corr):
            top_idx = np.argpartition(corr, -tk)[-tk:]
        else:
            top_idx = np.arange(len(corr))
        tv = corr[top_idx]
        sh_val = np.clip(tau * (tv - max_c), -100.0, 0.0)
        ec = np.exp(sh_val)
        p = ec / ec.sum()
        grad = grad_topk(y, top_idx, p, n)
        t_ad += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_ad)
        v_hat = v / (1.0 - beta2 ** t_ad)
        h_new = np.clip(y - lr * m_hat / (np.sqrt(v_hat) + eps), 0.0, 1.0)
        h_new = project_to_feasible(h_new, n)
        if it % 80 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 80
            if stall > 6000:
                lr *= 0.65
                m *= 0.5
                v *= 0.5
                stall = 0
                if lr < 1e-15:
                    break
        t_nest = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_nest * t_nest))
        h_prev = h
        h = h_new

    c5f = compute_c5(h)
    if c5f < best_c5:
        best_c5 = c5f
        best_h = h.copy()
    return best_h, best_c5


def subgrad_opt(h_in, n, tl, lr0, tk):
    h = project_to_feasible(h_in.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.6
    lr = lr0
    stall = 0

    for it in range(20000000):
        if time.time() - start > tl:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            lr *= 0.65
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break
        if tk < len(corr):
            top_idx = np.argpartition(corr, -tk)[-tk:]
        else:
            top_idx = np.arange(len(corr))
        w = np.ones(len(top_idx)) / len(top_idx)
        grad = grad_topk(h, top_idx, w, n)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    c5f = compute_c5(h)
    if c5f < best_c5:
        best_c5 = c5f
        best_h = h.copy()
    return best_h, best_c5


def coord_swap(h, n, rng):
    """Random coordinate improvements: try setting h[j] to reduce local max."""
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    tl = 15.0
    for _ in range(50000):
        if time.time() - start > tl:
            break
        j = rng.randint(n)
        old_val = h[j]
        for new_val in [0.0, 1.0, old_val * 0.5, old_val * 1.5]:
            new_val = float(np.clip(new_val, 0.0, 1.0))
            if abs(new_val - old_val) < 1e-9:
                continue
            h[j] = new_val
            h = project_to_feasible(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5_test = corr.max() * (2.0 / n)
            if c5_test < best_c5 - 1e-14:
                best_c5 = c5_test
                best_h = h.copy()
            else:
                h[j] = old_val
                h = project_to_feasible(h, n)
                break
    return best_h, best_c5


def get_rem(t0, budget):
    return max(0.0, budget - (time.time() - t0))
