One more idea: **use the structure of the correlation to guide local search more effectively**.

When I change h[j] from v to v', the change in corr[k] is:
Δcorr[k] = (v' - v) * [(1-h[j+s])*[0≤j+s<n] - h[j-s]*[0≤j-s<n]]

For the critical lags (near the max), I want Δcorr[k] < 0. This gives me a direction for h[j].

So for each j, the "ideal" direction is:
- If at a critical lag s, (1-h[j+s]) > h[j-s], then increasing h[j] increases corr at that lag → I want to decrease h[j]
- If at a critical lag s, h[j-s] > (1-h[j+s]), then increasing h[j] decreases corr at that lag → I want to increase h[j]

This gives me a "gradient" specific to the critical lags, which I can use to determine which direction to push each h[j].

Let me implement this as a "directed local search":

            res = minimize(_obj_only, best_h, jac=_obj_jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 400, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 5: Fine local search
    if remaining() > 40:
        best_h, best_c5 = fine_local_search(best_h, n, best_c5, max_time=min(100, remaining() - 30))

    # Phase 6: Another round of top-shift + local
    if remaining() > 60:
        corr = get_corr(best_h)
        mc = np.max(corr)
        thr = mc - 0.01 * mc
        top_idx = np.where(corr >= thr)[0]
        if len(top_idx) > 20:
            top_idx = top_idx[np.argsort(-corr[top_idx])[:20]]
        elif len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(corr))])
        _G_N = n
        _G_T = 800.0
        _G_TOP_IDX = top_idx
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_only, best_h, jac=_obj_jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass
        if remaining() > 40:
            best_h, best_c5 = directed_local_search(best_h, n, best_c5, max_time=min(80, remaining() - 30))

    # Phase 7: Random restarts
    for r in range(4):
        if remaining() < 50:
            break
        sc = 0.01 + 0.01 * r
        h_try = make_feasible(best_h + rng.randn(n) * sc, n)
        _G_N = n
        _G_T = 500.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_only, h_try, jac=_obj_jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 250, 'ftol': 1e-15})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 8: Final ultra-fine polish
    if remaining() > 20:
        _G_N = n
        _G_T = 3000.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_only, best_h, jac=_obj_jac, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-17})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Final fine search
    if remaining() > 15:
        best_h, best_c5 = fine_local_search(best_h, n, best_c5, max_time=min(25, remaining() - 5))

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
