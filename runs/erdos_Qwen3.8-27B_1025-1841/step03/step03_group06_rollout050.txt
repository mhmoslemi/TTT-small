        bounds = [(0.0, 1.0)] * n3
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
        for T_val, miter in [(100.0, 120), (250.0, 120), (500.0, 100)]:
            el = time.time() - start_time
            if budget_s - el < 70:
                break
            _T = T_val
            res = minimize(obj_topk, h_up, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': miter, 'ftol': 1e-14})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, n3)
                h_up = best_h.copy()
            # Try with full LSE at high T
            el = time.time() - start_time
            if budget_s - el > 50 and T_val >= 250:
                res2 = minimize(obj_full_lse, h_up, method='SLSQP',
                                bounds=bounds, constraints=cons,
                                options={'maxiter': 80, 'ftol': 1e-14})
                c2 = compute_c5(res2.x)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = make_feasible(res2.x, n3)
                    h_up = best_h.copy()

    # Phase 4: Binary local search with pair swaps
    n = len(best_h)
    el = time.time() - start_time
    if budget_s - el > 200:
        # Snap to binary
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin > 0.5))
        target_ones = n // 2
        if n_ones > target_ones:
            idx_1 = np.where(h_bin == 1.0)[0]
            order = np.argsort(best_h[idx_1])
            for i in range(n_ones - target_ones):
                h_bin[idx_1[order[i]]] = 0.0
        elif n_ones < target_ones:
            idx_0 = np.where(h_bin == 0.0)[0]
            order = np.argsort(-best_h[idx_0])
            for i in range(target_ones - n_ones):
                h_bin[idx_0[order[i]]] = 1.0
        h_bin = make_feasible(h_bin, n)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()

        # Single-flip local search
        for rd in range(30):
            el = time.time() - start_time
            if budget_s - el < 25:
                break
            improved = False
            for i in range(n):
                if abs(best_h[i] - 0.5) < 0.1:
                    candidates = [0.0, 1.0]
                else:
                    candidates = [0.0 if best_h[i] > 0.5 else 1.0]
                for tv in candidates:
                    if abs(tv - best_h[i]) < 0.05:
                        continue
                    trial = best_h.copy()
                    trial[i] = tv
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_h = trial
                        improved = True
            if not improved:
                break

        # Pair-swap search
        el = time.time() - start_time
        if budget_s - el > 60:
            for rd in range(10):
                el = time.time() - start_time
                if budget_s - el < 20:
                    break
                improved = False
                ones = np.where(best_h > 0.5)[0]
                zeros = np.where(best_h <= 0.5)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    # Sample a subset of pairs for efficiency
                    n_pairs_to_try = min(200, len(ones) * len(zeros))
                    for _ in range(n_pairs_to_try):
                        el = time.time() - start_time
                        if budget_s - el < 5:
                            break
                        i = ones[rng.randint(len(ones))]
                        j = zeros[rng.randint(len(zeros))]
                        if i == j:
                            continue
                        trial = best_h.copy()
                        trial[i] = 0.0
                        trial[j] = 1.0
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-14:
                            best_c5 = c
                            best_h = trial
                            improved = True
                if not improved:
                    break

    # Phase 5: Coordinate descent (continuous)
    n = len(best_h)
    el = time.time() - start_time
    if budget_s - el > 80:
        for rd in range(40):
            el = time.time() - start_time
            if budget_s - el < 15:
                break
            improved = False
            for j in range(n):
                if time.time() - start_time > budget_s - 10:
                    break
                old = best_h[j]
                best_c = best_c5
                best_trial = old
                cands = [0.0, 1.0, 0.5,
                         max(0.0, old - 0.1), max(0.0, old - 0.04), max(0.0, old - 0.01),
                         min(1.0, old + 0.01), min(1.0, old + 0.04), min(1.0, old + 0.1)]
                for cand in cands:
                    if abs(cand - old) < 1e-10:
                        continue
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_trial = cand
                if best_trial != old:
                    best_h[j] = best_trial
                    best_h = make_feasible(best_h, n)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break

    # Phase 6: Final high-T SLSQP polish
    el = time.time() - start_time
    if budget_s - el > 50:
        n = len(best_h)
        _N = n
        _K_SMOOTH = 5
        _T = 600.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
        res = minimize(obj_topk, best_h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 150, 'ftol': 1e-15})
        c = compute_c5(res.x)
        if c < best_c5:
            best_c5 = c
            best_h = make_feasible(res.x, n)
        # Also try full LSE
        el = time.time() - start_time
        if budget_s - el > 25:
            _K_SMOOTH = len(best_h) * 2  # effectively full
            _T = 800.0
            res2 = minimize(obj_full_lse, best_h, method='SLSQP',
                            bounds=bounds, constraints=cons,
                            options={'maxiter': 100, 'ftol': 1e-15})
            c2 = compute_c5(res2.x)
            if c2 < best_c5:
                best_c5 = c2
                best_h = make_feasible(res2.x, n)

    # Phase 7: Ultra-fine coordinate descent
    n = len(best_h)
    el = time.time() - start_time
    if budget_s - el > 20:
        for rd in range(15):
            el = time.time() - start_time
            if budget_s - el < 5:
                break
            improved = False
            for j in range(n):
                if time.time() - start_time > budget_s - 3:
                    break
                old = best_h[j]
                best_c = best_c5
                best_trial = old
                for d in [-0.003, -0.001, 0.001, 0.003]:
                    cand = max(0.0, min(1.0, old + d))
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_trial = cand
                if best_trial != old:
                    best_h[j] = best_trial
                    best_h = make_feasible(best_h, n)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
