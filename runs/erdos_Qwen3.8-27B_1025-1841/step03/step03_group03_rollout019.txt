```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    return make_feasible(h_new, new_n)

def subgradient(h, n, lag):
```

```python
    vh = 1.0 - h
    corr = np.correlate(h, vh, mode='full')  # length 2n-1
    c_max = np.max(corr)

    ncorr = len(corr)  # 2n-1
    grad = np.zeros(n)

                grad[idx1] += w[k] * (1.0 - h[idx1 + p])

                grad[idx2] -= w[k] * h[idx2 - p]
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_objective_and_grad(h, n, t, top_k=30):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    obj = (c_max + np.log(np.sum(np.exp(t * shifted))) / t) * dx
    
    # Gradient using top_k lags
    ncorr = len(corr)
    if top_k >= ncorr:
        top_idx = np.arange(ncorr)
    else:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    corr_top = corr[top_idx]
    exp_top = np.exp(t * (corr_top - c_max))
    s = np.sum(exp_top)
    if s < 1e-300:
        w_top = np.ones(len(top_idx)) / len(top_idx)
    else:
        w_top = exp_top / s
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        w_k = w_top[i]
        if w_k < 1e-12:
            continue
        k = int(top_idx[i])
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += w_k * (1.0 - 2.0 * h)
            else:
                idx1 = np.arange(lag, n)
                grad[idx1] += w_k * (1.0 - h[idx1 - lag])
                idx2 = np.arange(0, n - lag)
                grad[idx2] -= w_k * h[idx2 + lag]
        else:
            p = -lag
            idx1 = np.arange(0, n - p)
            grad[idx1] += w_k * (1.0 - h[idx1 + p])
            idx2 = np.arange(p, n)
            grad[idx2] -= w_k * h[idx2 - p]
    return obj, grad * dx

def slsqp_smooth(h_start, n, t, maxiter=300, top_k=30):
    def obj_fun(h):
        return softmax_objective_and_grad(h, n, t, top_k)[0]
    def obj_grad(h):
        return softmax_objective_and_grad(h, n, t, top_k)[1]
    def con_fun(h):
        return np.sum(h) - n / 2.0
    def con_jac(h):
        return np.ones(n)

        h_out = make_feasible(res.x, n)
        return h_out
    except Exception:
        return make_feasible(h_start, n)
```

```python
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.full(320, 0.5)
        best_h = make_feasible(best_h, 320)

    best_c5 = compute_c5(best_h)
    n1 = len(best_h)

    # Phase 1: SLSQP at original resolution
    for t, mi, tk in [(100, 300, 50), (400, 400, 30), (1500, 500, 20), (5000, 600, 15)]:
        if time.time() - start_time > budget_s - 600:
            break
        h_new = slsqp_smooth(best_h, n1, t, mi, tk)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-10:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Coordinate descent
    h_cd, c_cd = coord_descent(best_h, n1, max_rounds=12, time_lim=budget_s, t0=start_time)
    if c_cd < best_c5:
        best_c5 = c_cd
        best_h = h_cd

    # Phase 3: Simulated annealing
    if time.time() - start_time < budget_s - 300:
        h_sa, c_sa = simulated_annealing(best_h, n1, 8000, 0.01, 0.0001,
                                          budget_s, start_time, rng)
        if c_sa < best_c5:
            best_c5 = c_sa
            best_h = h_sa
        h_cd, c_cd = coord_descent(best_h, n1, max_rounds=6, time_lim=budget_s, t0=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 4: Upsample to 1024
    n2 = 1024
    if time.time() - start_time < budget_s - 350:
        h2 = upsample_h(best_h, n2)
        for t, mi, tk in [(200, 300, 40), (800, 400, 25), (3000, 500, 15)]:
            if time.time() - start_time > budget_s - 350:
                break
            h_new = slsqp_smooth(h2, n2, t, mi, tk)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                best_h = h_new
                h2 = h_new
            else:
                h2 = h_new
        h_cd, c_cd = coord_descent(h2, n2, max_rounds=6, time_lim=budget_s, t0=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd
            h2 = h_cd

    # Phase 5: Upsample to 1536
    n3 = 1536
    if time.time() - start_time < budget_s - 250:
        h3 = upsample_h(best_h, n3)
        for t, mi, tk in [(400, 300, 30), (2000, 400, 18), (6000, 400, 12)]:
            if time.time() - start_time > budget_s - 250:
                break
            h_new = slsqp_smooth(h3, n3, t, mi, tk)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                best_h = h_new
                h3 = h_new
            else:
                h3 = h_new
        h_cd, c_cd = coord_descent(h3, n3, max_rounds=5, time_lim=budget_s, t0=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd
            h3 = h_cd

    # Phase 6: Random restarts
    n_r = len(best_h)
    for restart in range(4):
        if time.time() - start_time > budget_s - 80:
            break
        h_r = best_h + rng.randn(n_r) * 0.08
        h_r = make_feasible(h_r, n_r)
        h_new = slsqp_smooth(h_r, n_r, 800, 250, 20)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-10:
            best_c5 = c_new
            best_h = h_new
        h_cd, c_cd = coord_descent(h_new, n_r, max_rounds=4, time_lim=budget_s, t0=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 7: Final intensive coordinate descent
    n_f = len(best_h)
    h_cd, c_cd = coord_descent(best_h, n_f, max_rounds=12, time_lim=budget_s, t0=start_time)
    if c_cd < best_c5:
        best_c5 = c_cd
        best_h = h_cd

    # Phase 8: One more SLSQP polish at final resolution
    if time.time() - start_time < budget_s - 40:
        h_new = slsqp_smooth(best_h, n_f, 10000, 300, 10)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-10:
            best_c5 = c_new
            best_h = h_new
        h_cd, c_cd = coord_descent(best_h, n_f, max_rounds=5, time_lim=budget_s, t0=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
```
