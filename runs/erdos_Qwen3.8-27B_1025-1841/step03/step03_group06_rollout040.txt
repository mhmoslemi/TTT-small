def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    Objective: (1/T)*log(sum(exp(T*f_k))) + lam*(sum(h)-n/2)^2

    # Penalty term
    s = np.sum(h) - n/2.0
    value_pen = lam * s * s
    value = value_f + value_pen

    grad_pen = 2.0 * lam * s * np.ones(n)
