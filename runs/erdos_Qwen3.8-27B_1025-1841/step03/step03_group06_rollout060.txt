import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(y, n):
    """Project y onto [0,1]^n ∩ {sum = n/2}."""
    target = n / 2.0
    # Binary search for lambda
    lo, hi = y.min() - 1, y.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(y - mid, 0, 1)
        s = proj.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(y - (lo + hi) / 2, 0, 1)
    # Fix any remaining error
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0, 1)
    return proj

def compute_grad_softmax(h, T):
    """Compute gradient of soft-max objective w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    # Softmax weights
    exp_vals = np.exp(T * (corr - c_max))
    w = exp_vals / np.sum(exp_vals)
    
    # Build W_pos and W_neg
    # corr is indexed from -(n-1) to (n-1)
    # w[n-1 + k] corresponds to lag k
    W_pos = np.zeros(n)  # W_pos[k] = w_{k} for k=0..n-1
    W_neg = np.zeros(n)  # W_neg[k] = w_{-k} for k=0..n-1 (W_neg[0]=0)
    for k in range(n):
        W_pos[k] = w[n-1 + k]  # lag k
    for k in range(1, n):
        W_neg[k] = w[n-1 - k]  # lag -k
    
    # Gradient using FFT convolutions
    one_minus_h = 1.0 - h
    # A[j] = correlate(W_pos, 1-h, 'full')[n-1-j]
    corr_wp_omh = np.correlate(W_pos, one_minus_h, mode='full')
    A = corr_wp_omh[n-1::-1]  # A[j] = corr_wp_omh[n-1-j]
    
    # B[j] = convolve(W_neg, 1-h, 'full')[j]
    conv_wn_omh = fftconvolve(W_neg, one_minus_h, mode='full')
    B = conv_wn_omh[:n]
    
    # C[j] = convolve(W_pos, h, 'full')[j]
    conv_wp_h = fftconvolve(W_pos, h, mode='full')
    C = conv_wp_h[:n]
    
    # D[j] = correlate(W_neg, h, 'full')[n-1-j]
    corr_wn_h = np.correlate(W_neg, h, mode='full')
    D = corr_wn_h[n-1::-1]
    
    grad = (A + B - C - D) * dx
    return grad

def projected_gradient_step(h, alpha, T):
    """One step of projected gradient descent."""
    n = len(h)
    grad = compute_grad_softmax(h, T)
    y = h - alpha * grad
    return project_feasible(y, n)
    
def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(y, n):
    target = n / 2.0
    lo = y.min() - 1.0
    hi = y.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(y - mid, 0.0, 1.0)
        if proj.sum() > target:
            hi = mid
        else:
            lo = mid
    proj = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            proj[mask] += diff / cnt
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    c_best = compute_c5(h)
    alpha = 1.0 / max(np.max(np.abs(compute_grad_softmax(h, T))), 1e-10)
    alpha = min(alpha, 1.0)
