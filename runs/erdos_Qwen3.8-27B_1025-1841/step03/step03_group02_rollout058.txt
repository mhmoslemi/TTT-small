    dx = 2.0 / n
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    # Find the max element
    max_idx = np.argmax(corr)
    # Compute subgradient for this lag
    lag = max_idx - (n - 1)  # Convert to lag
    # Subgradient: for the active constraint
    grad = np.zeros(n)

        n_sa = min(int((time_left - 30) * 5000), 200000)
        b_hi, c_hi = sa_mixed(b_hi, n_hi, max(n_sa, 30000), 0.005, 0.00002, rng, time.time(), start_time + budget_s - 100)
        if c_hi < best_c5:
            best_c5 = c_hi
            best_h = b_hi.copy()

    # Phase 5: Continuous SLSQP polish
    time_left = budget_s - (time.time() - start_time)
    if time_left > 80:
        np_pts = len(best_h)
        for lam in (50.0, 200.0):
            if time.time() - start_time > budget_s - 30:
                break
            try:
                x_res = slsqp_opt(best_h, np_pts, lam, 150, 1e-15)
                c_res = compute_c5(x_res)
                if c_res < best_c5:
                    best_c5 = c_res
                    best_h = make_feasible(x_res, np_pts)
            except Exception:
                pass

    # Phase 6: Coordinate descent refinement
    time_left = budget_s - (time.time() - start_time)
    if time_left > 60:
        np_pts = len(best_h)
        best_h, best_c5 = coord_desc(best_h.copy(), np_pts, 15, time.time(), start_time + budget_s - 20)

    # Phase 7: Final binary SA polish
    time_left = budget_s - (time.time() - start_time)
    if time_left > 40:
        np_pts = len(best_h)
        b_f = to_binary(best_h, np_pts)
        n_sa = min(int((time_left - 10) * 8000), 100000)
        b_f, c_f = sa_mixed(b_f, np_pts, max(n_sa, 10000), 0.003, 0.00001, rng, time.time(), start_time + budget_s - 10)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = b_f.copy()
        else:
            # Try continuous if binary didn't help
            if time.time() - start_time < budget_s - 20:
                try:
                    x_res = slsqp_opt(best_h, np_pts, 500.0, 100, 1e-16)
                    c_res = compute_c5(x_res)
                    if c_res < best_c5:
                        best_c5 = c_res
                        best_h = make_feasible(x_res, np_pts)
                except Exception:
                    pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
