```python
def lbgfb_with_grad(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    nm1 = n - 1
    
    def f_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        
        # Softmax weights
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -100, 100)
        exp_s = np.exp(shifted)
        sum_exp = np.sum(exp_s)
        sw = exp_s / sum_exp  # length 2n-1
        
        obj = (mc + np.log(sum_exp) / T) * dx
        viol = np.sum(h) - n / 2.0
        obj += mu * viol * viol
        
        # Gradient
        W = sw * dx  # length 2n-1
        first_part = np.convolve(1.0 - h, W, 'full')[nm1 : 2*nm1+1]
        second_part = np.correlate(W, h, 'full')[nm1 : 2*nm1+1]
        second_part = second_part[::-1]
        
        grad = first_part - second_part
        grad += 2.0 * mu * viol
        
        return obj, grad
    
    def f(h):
        return f_and_grad(h)[0]
    
    def fprime(h):
        return f_and_grad(h)[1]
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, jac=fprime, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
```

```python
def subgrad_optimize(h_start, n, num_iters=1000, step_decay=True):
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    step = 0.01
    for it in range(num_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        c5 = mc * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Subgradient: pick a random argmax lag
        max_idx = np.random.choice(np.where(corr == mc)[0])
        lag = max_idx - (n - 1)
        
        # grad_corr at this single lag
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:n] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
        else:
            grad[:n+lag] += (1.0 - h[-lag:])
            grad[-lag:] -= h[:n+lag]
        grad *= dx
        
        # Step
        h = h - step * grad
        h = make_feasible(h, n)
        
        if step_decay:
            step *= 0.999
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_optimize(h_start, n, T, mu, max_iter=400):
    dx = 2.0 / n
    nm1 = n - 1

    def f_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -100, 100)
        exp_s = np.exp(shifted)
        sum_exp = np.sum(exp_s)
        sw = exp_s / sum_exp
        obj = (mc + np.log(sum_exp) / T) * dx
        viol = np.sum(h) - n / 2.0
        obj += mu * viol * viol
        W = sw * dx
        first_part = np.convolve(1.0 - h, W, 'full')[nm1:nm1 + n]
        second_part = np.correlate(W, h, 'full')[nm1:nm1 + n]
        second_part = second_part[::-1]
        grad = first_part - second_part
        grad += 2.0 * mu * viol
        return obj, grad

    def f(h):
        return f_and_grad(h)[0]

    def fprime(h):
        return f_and_grad(h)[1]

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, jac=fprime, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 60})
    return make_feasible(res.x, n)

def subgrad_polish(h_start, n, max_iters=5000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    dx = 2.0 / n
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.02
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, 'full')
        mc = np.max(corr)
        c5 = mc * dx
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
        max_mask = corr >= mc - 1e-12
        candidates = np.where(max_mask)[0]
        max_idx = candidates[rng.randint(len(candidates))]
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[lag:n] += (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
        else:
            grad[:n + lag] += (1.0 - h[-lag:])
            grad[-lag:] -= h[:n + lag]
        grad *= dx
        h = h - step * grad
        h = make_feasible(h, n)
        step *= 0.9995
    return best_h, best_c5

def topk_lag_optimize(h_start, n, T_base, k_lags, mu, max_iter=200, rng=None):
    if rng is None:
        rng = np.random.RandomState(1)
    dx = 2.0 / n
    nm1 = n - 1
    h = h_start.copy()

    def f_and_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T_base * (corr - mc)
        shifted = np.clip(shifted, -100, 100)
        exp_s = np.exp(shifted)
        sum_exp = np.sum(exp_s)
        sw = exp_s / sum_exp
        topk_mask = np.zeros(len(corr), dtype=int)
        topk_idx = np.argsort(corr)[-k_lags:]
        topk_mask[topk_idx] = 1
        sw_topk = sw * topk_mask
        sw_topk_sum = np.sum(sw_topk)
        if sw_topk_sum < 1e-12:
            sw_final = sw
        else:
            sw_final = sw * 0.5 + sw_topk / sw_topk_sum * 0.5
        obj = (mc + np.log(sum_exp) / T_base) * dx
        viol = np.sum(h) - n / 2.0
        obj += mu * viol * viol
        W = sw_final * dx
        first_part = np.convolve(1.0 - h, W, 'full')[nm1:nm1 + n]
        second_part = np.correlate(W, h, 'full')[nm1:nm1 + n]
        second_part = second_part[::-1]
        grad = first_part - second_part
        grad += 2.0 * mu * viol
        return obj, grad

    def f(h):
        return f_and_grad(h)[0]

    def fprime(h):
        return f_and_grad(h)[1]

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h, jac=fprime, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=150, n_tries=3000):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current:
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def single_flip_search(h, n, rng, max_rounds=50):
    current = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        idxs = rng.permutation(n)
        for j in idxs:
            old = h[j]
            nv = 1.0 - old
            h[j] = nv
            h = make_feasible(h, n)
            c = compute_c5(h)
            if c < current - 1e-14:
                current = c
                improved = True
            else:
                h[j] = old
                h = make_feasible(h, n)
        if not improved:
            break
    return h, current

                    h_cd = make_feasible(h_cd, n_cd)
                    c = compute_c5(h_cd)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        improved = True
                        break
                    else:
                        h_cd[j] = ov
                        h_cd = make_feasible(h_cd, n_cd)
                for bv in (0.0, 1.0):
                    if abs(bv - h_cd[j]) < 1e-9:
                        continue
                    h_cd[j] = bv
                    h_cd = make_feasible(h_cd, n_cd)
                    c = compute_c5(h_cd)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        improved = True
                        break
                if tl() < 5:
                    break
            if not improved:
                break

    # Final subgradient squeeze
    if tl() > 20:
        n_f = len(best_h)
        h_sg, c5_sg = subgrad_polish(best_h, n_f, max_iters=2000, rng=rng)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()

    # Final feasibility and evaluation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
