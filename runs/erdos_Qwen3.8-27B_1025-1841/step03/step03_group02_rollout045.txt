import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_binary(b, n, n_iters, T0, T1, rng, start_time, budget_s):
    """General binary SA with multiple move types."""
    curr_c5, _ = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(n_iters):
        if time.time() - start_time > budget_s - 10:
            break
        frac = it / max(n_iters, 1)
        T = T0 * (T1 / T0) ** frac
        
        r = rng.random()
        if r < 0.6:
            # Single swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0, 1
            new_c5, _ = compute_c5(b)
            if new_c5 <= curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-12)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5

                b[i], b[j] = 1, 0
        elif r < 0.85:

            j = rng.randint(0, n - blen)

            seg_j = b[j:j+blen].copy()

            b[j:j+blen] = seg_i
            new_c5, _ = compute_c5(b)
            if new_c5 <= curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-12)):

                b[j:j+blen] = seg_j

    curr_c5, curr_corr = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(n_iters):
        if time.time() - start_time > budget_s - 5:
            break
        T = 0.002 * (0.2 ** (it / max(n_iters, 1)))
        max_val = np.max(curr_corr)
        max_lags = np.where(curr_corr >= max_val - 1e-10)[0]
        k_idx = max_lags[rng.randint(len(max_lags))]
        d = k_idx - (n - 1)
        if d == 0:
            continue
        if d > 0:
            valid = (b[d:n] == 1) & (b[0:n-d] == 0)
            ms = np.where(valid)[0]
            if len(ms) == 0:
                continue
            m = ms[rng.randint(len(ms))]
            i, j = m + d, m
        else:
            ad = -d
            valid = (b[0:n-ad] == 1) & (b[ad:n] == 0)
            ms = np.where(valid)[0]
            if len(ms) == 0:
                continue
            m = ms[rng.randint(len(ms))]
            i, j = m, m + ad
        b[i], b[j] = b[j], b[i]
        new_c5, new_corr = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            curr_corr = new_corr
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
            curr_corr = np.correlate(b, 1.0 - b, mode='full')
    return best_b, best_c5

def coord_descent(h, n, start_time, budget_s, max_rounds=30):
    curr_c5, _ = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for i in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            old = h[i]
            for val in [0.0, 1.0, max(0.0, old - 0.05), min(1.0, old + 0.05)]:
                if abs(val - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[i] = val
                trial = make_feasible(trial, n)
                c, _ = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    h = trial.copy()
                    best_h = trial.copy()
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, n, T, maxiter, start_time, budget_s):
    global _SL_N, _SL_T
    if time.time() - start_time > budget_s - 30:
        return h
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.fun < compute_c5(h)[0]:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h
