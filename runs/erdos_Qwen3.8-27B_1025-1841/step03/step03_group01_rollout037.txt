        if it % 500 == 0 and val >= best_val * 1.001:
            lr_init *= 0.5
    return best_h, best_val

def run(seed=42, budget_s=1300, **kwargs):
    global _SN, _ST
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - t0)

    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = np.tile([1.0, 0.0], 160)
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)

    # Phase 1: SLSQP at original resolution with increasing T
    h_curr = best_h.copy()
    for T in [30, 60, 100, 200, 500, 1000, 2000, 5000]:
        if _tl() < 40:
            break
        iters = min(3000, max(300, int(50000 / T)))
        try:
            h_new = do_slsqp(h_curr, n0, T, iters)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_curr = h_new.copy()
        except Exception:
            pass

    # Phase 2: Perturbation restarts
    if _tl() > 80:
        for _ in range(4):
            if _tl() < 40:
                break
            h_pert = best_h + 0.04 * (rng.rand(n0) - 0.5)
            h_pert = make_feasible(h_pert, n0)
            try:
                h_new = do_slsqp(h_pert, n0, 1000, 1000)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 3: Binary initialization + local search
    if _tl() > 100:
        for n_try in [n0, 1280]:
            if _tl() < 60:
                break
            h_blk = np.zeros(n_try)
            block = n_try // 8
            for b in range(0, n_try - block + 1, block * 2):
                h_blk[b:b + block] = 1.0
            h_blk = make_feasible(h_blk, n_try)
            try:
                h_new = do_slsqp(h_blk, n_try, 200, 500)
                c5_new = compute_c5(h_new)
                if n_try == n0 and c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 4: Upsample and optimize at higher resolutions
    for n_new in [1280, 2560]:
        if _tl() < 150:
            break
        h_up = upsample_h(best_h, n_new)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()

        for T in [100, 300, 700, 2000]:
            if _tl() < 50:
                break
            iters = min(2000, max(200, int(30000 / T)))
            try:
                h_new = do_slsqp(h_up, n_new, T, iters)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_up = h_new.copy()
            except Exception:
                pass

    # Phase 5: Projected gradient at highest resolution
    if _tl() > 100:
        n_pg = 5120
        h_pg = upsample_h(best_h, n_pg)
        c5_pg = compute_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
        for T in [200, 500, 1000, 3000]:
            if _tl() < 50:
                break
            try:
                h_new, val_new = proj_grad(h_pg, n_pg, T, 3000, lr_init=0.02)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_pg = h_new.copy()
            except Exception:
                pass

    # Phase 6: Final SLSQP polish at current best resolution
    if _tl() > 40:
        nf = len(best_h)
        if nf <= 2560:
            try:
                h_new = do_slsqp(best_h, nf, 8000, 1000)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass
        else:
            try:
                h_new, _ = proj_grad(best_h, nf, 5000, 2000, lr_init=0.01)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    # Phase 7: One more SLSQP pass with very high T on the best
    if _tl() > 30:
        nf = len(best_h)
        if nf <= 1280:
            try:
                h_new = do_slsqp(best_h, nf, 20000, 500)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
            except Exception:
                pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
