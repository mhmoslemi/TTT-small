    if new_n == old_n:
        return make_feasible(h.copy(), new_n)

def objective_and_grad(h, n, T, penalty_coef):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted_max = np.max(shifted)
    exp_shifted = np.exp(shifted - shifted_max)
    log_sum = np.log(np.sum(exp_shifted)) + shifted_max
    obj = (mc + log_sum / T) * dx
    obj += penalty_coef * (np.sum(h) - n / 2.0) ** 2
    
    # Gradient
    weights = exp_shifted / np.sum(exp_shifted)
    # Compute gradient of max-smoothed correlation
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-14:
            continue
        lag = m - (n - 1)
        if lag >= 0:
            if lag < n:
                j_idx = np.arange(0, n - lag)
                grad[j_idx] += w * (1.0 - h[j_idx + lag])
                if lag > 0:
                    j_idx2 = np.arange(lag, n)
                    grad[j_idx2] -= w * h[j_idx2 - lag]
        else:
            lag2 = -lag
            if lag2 < n:
                j_idx = np.arange(lag2, n)
                grad[j_idx] += w * (1.0 - h[j_idx - lag2])
                j_idx2 = np.arange(0, n - lag2)
                grad[j_idx2] -= w * h[j_idx2 + lag2]
    grad *= dx
    grad += 2 * penalty_coef * (np.sum(h) - n / 2.0)
    return obj, grad

def l_bfgs_b_optimize(h_start, n, T, penalty_coef, max_iter, time_limit):
    """Run L-BFGS-B with our objective."""
    state = {'time_start': time.time(), 'time_limit': time_limit}
    
    def fun_and_grad(x):
        if time.time() - state['time_start'] > state['time_limit']:
            raise TimeoutError()

        return res.x

def top_k_gradient(h, n, K=10):

    dx = 2.0 / n

    T = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

                j1 = np.arange(lag, n)
                grad[j1] += w * (1.0 - h[j1 - lag])
                j2 = np.arange(0, n - lag)
                grad[j2] -= w * h[j2 + lag]
        else:
            lag2 = -lag
            if lag2 < n:
                j1 = np.arange(0, n - lag2)
                grad[j1] += w * (1.0 - h[j1 + lag2])
                j2 = np.arange(lag2, n)
                grad[j2] -= w * h[j2 - lag2]
    return grad * dx

def _obj_grad_h(h, n, T, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    obj = (mc + np.log(s) / T) * dx + pen * (np.sum(h) - n / 2.0) ** 2
    weights = exp_s / s
    grad = _corr_grad(h, n, weights)
    grad += 2.0 * pen * (np.sum(h) - n / 2.0)
    return obj, grad

def _topk_obj_grad(h, n, K, T_k, pen):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    w_top = np.exp(T_k * (top_vals - np.max(top_vals)))
    w_top /= np.sum(w_top)
    obj = mc * dx + pen * (np.sum(h) - n / 2.0) ** 2
    full_w = np.zeros(2 * n - 1)
    full_w[top_idx] = w_top
    grad = _corr_grad(h, n, full_w)
    grad += 2.0 * pen * (np.sum(h) - n / 2.0)
    return obj, grad

def lbb_optimize(h0, n, T, pen, max_iter, time_limit, topK=0, T_k=50.0):
    t0 = time.time()
    def fg(x):
        if time.time() - t0 > time_limit:
            raise TimeoutError
        if topK > 0:
            return _topk_obj_grad(x, n, topK, T_k, pen)
        return _obj_grad_h(x, n, T, pen)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fg, h0, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'gtol': 1e-6, 'maxfun': max_iter * 2})
        return res.x
    except TimeoutError:
        return h0
    except Exception:
        return h0

def structured_perturb(h, n, rng, amp=0.15):
    """Add a random Fourier mode perturbation."""
    freq = rng.randint(2, max(3, n // 4))
    phase = rng.uniform(0, 2 * np.pi)
    x = np.linspace(0, 2, n, endpoint=False)
    pert = amp * amp * np.sin(2 * np.pi * freq * x + phase)
    h_new = h + pert
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    t_start = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t_start)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: L-BFGS-B at multiple resolutions and temperatures
    n_res = [1280, 2048, 3200]

    for n_pts in n_res:
        if tl() < 150:
            break
        h_cur = upsample_h(best_h, n_pts)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5, best_h = c, h_cur.copy()

        # Temperature annealing with L-BFGS-B
        temps = [
            (10.0, 300, 50.0),
            (40.0, 200, 50.0),
            (150.0, 150, 50.0),
            (500.0, 100, 50.0),
        ]
        for T, mi, pen in temps:
            if tl() < 25:
                break
            alloc = min(80.0, tl() * 0.35)
            h_opt = lbb_optimize(h_cur, n_pts, T, pen, mi, alloc)
            h_opt = make_feasible(h_opt, n_pts)
            c_opt = compute_c5(h_opt)
            h_cur = h_opt
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()

        # Top-K refinement
        if tl() > 40:
            for K in [5, 10, 20]:
                if tl() < 15:
                    break
                alloc = min(40.0, tl() * 0.3)
                h_opt = lbb_optimize(h_cur, n_pts, 0, 50.0, 100, alloc, topK=K, T_k=30.0)
                h_opt = make_feasible(h_opt, n_pts)
                c_opt = compute_c5(h_opt)
                h_cur = h_opt
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()

        # Structured perturbation restarts
        for _ in range(4):
            if tl() < 25:
                break
            h_pert = structured_perturb(upsample_h(best_h, n_pts), n_pts, rng, amp=0.2)
            alloc = min(30.0, tl() * 0.4)
            h_opt = lbb_optimize(h_pert, n_pts, 100.0, 50.0, 80, alloc)
            h_opt = make_feasible(h_opt, n_pts)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h_cur = h_opt.copy()

    # Phase 2: Final high-temperature polish at current resolution
    n_fin = len(best_h)
    for T in [200.0, 800.0, 3000.0]:
        if tl() < 15:
            break
        alloc = min(25.0, tl() * 0.4)
        h_opt = lbb_optimize(best_h, n_fin, T, 50.0, 50, alloc)
        h_opt = make_feasible(h_opt, n_fin)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()

    # Phase 3: Top-K gradient descent (direct, no L-BFGS-B overhead)
    if tl() > 30:
        h_cd = best_h.copy()
        n_cd = len(h_cd)
        step = 0.03
        for it in range(500):
            if tl() < 10:
                break
            obj, g = _topk_obj_grad(h_cd, n_cd, 8, 20.0, 50.0)
            h_new = make_feasible(h_cd - step * g, n_cd)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5, best_h = c_new, h_new.copy()
                h_cd = h_new
                step = min(step * 1.05, 0.1)
            else:
                step *= 0.85
                if step < 0.001:
                    break

    # Phase 4: Neighborhood flip search
    if tl() > 20:
        n_fl = len(best_h)
        for radius in [1, 2, 3, 5, 8]:
            for _ in range(80):
                if tl() < 5:
                    break
                j = rng.randint(0, n_fl)
                lo, hi = max(0, j - radius), min(n_fl, j + radius + 1)
                trial = best_h.copy()
                trial[lo:hi] = 1.0 - trial[lo:hi]
                trial = make_feasible(trial, n_fl)
                c = compute_c5(trial)
                if c < best_c5 - 1e-10:
                    best_c5, best_h = c, trial

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
