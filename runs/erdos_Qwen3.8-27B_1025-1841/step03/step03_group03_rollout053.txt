import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    # Project onto sum = target with h in [0,1]
    # Use bisection on lambda
    def project(lam):
        return np.clip(h - lam, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.sum(project(mid)) > target:
            lo = mid
        else:
            hi = mid
    h_proj = project((lo + hi) / 2)
    # Fine adjust
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def project_onto_constraint(v, n):
    """Project v onto {h: sum(h)=n/2, 0<=h<=1}"""
    target = n / 2.0
    def proj(lam):
        return np.clip(v - lam, 0.0, 1.0)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(proj(mid)) > target:
            lo = mid
        else:
            hi = mid
    h = proj((lo + hi) / 2)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_c5_and_grad(h, t):
    """Compute soft C5 and its gradient using shift-based approach"""
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')  # length 2n-1
    c_scaled = corr * dx
    
    max_c = np.max(c_scaled)
    exp_c = np.exp(t * (c_scaled - max_c))
    s = np.sum(exp_c)
    w = exp_c / s
    c_soft = max_c + np.log(s) / t
    
    # Gradient: grad[i] = sum_k w[k] * v[i + k - (n-1)] for valid indices
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-15:
            continue
        i_start = max(0, (n - 1) - k)
        i_end = min(n, 2 * (n - 1) - k + 1)
        if i_start >= i_end:
            continue
        v_start = i_start + k - (n - 1)
        v_len = i_end - i_start
        grad[i_start:i_end] += w[k] * v[v_start:v_start + v_len]
    
    grad *= dx  # because c_scaled = corr * dx
    
    return c_soft, grad

def grad_descent(h, n, t, max_iters, time_limit, start_time):
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    step = 0.01
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 30:
            break
        c_soft, grad = softmax_c5_and_grad(h, t)
        # Check convergence
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        # Backtracking line search
        alpha = step / gnorm if gnorm > 1e-12 else 0.01
        for _ in range(30):
            h_new = project_onto_constraint(h - alpha * grad, n)
            c_new = compute_c5(h_new)
            if c_new < best_c - 1e-12 or c_new < c_soft - 0.5 * alpha * gnorm**2:
                h = h_new
                best_c = min(best_c, c_new)
                break
            alpha *= 0.5
        else:
            # No improvement, try smaller step
            step *= 0.7
            if step < 1e-6:
                break
    return h, best_c

def coord_descent(h, n, max_rounds, time_limit, start_time):
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 20:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 20:
                return h, best_c
            old = h[j]
            # Try golden section on [0,1] for h[j]
            best_val = old
            best_obj = best_c
            # Try a few candidate values
            for cv in [0.0, 1.0, 0.5, old+0.1, old-0.1, old+0.05, old-0.05,
                       old+0.02, old-0.02, old+0.01, old-0.01]:
                if cv < -0.01 or cv > 1.01:
                    continue
                cv = np.clip(cv, 0.0, 1.0)
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-13:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def binary_round(h, n):
    """Round to binary, keeping exactly n/2 ones"""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Set excess ones to 0 (prefer those with lowest h value)
        one_idx = np.where(h_bin == 1)[0]
        h_sorted = np.sort(h[one_idx])[::-1]
        # Set the ones with smallest original h to 0
        excess = n_ones - target
        idx_to_zero = one_idx[np.argsort(h[one_idx])[:excess]]
        h_bin[idx_to_zero] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        need = target - n_ones
        idx_to_one = zero_idx[np.argsort(h[zero_idx])[::-1][:need]]
        h_bin[idx_to_one] = 1.0
    return h_bin

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_onto_constraint(v, n):
    target = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(v - lam, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h = h.copy()
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def make_feasible(h, n):
    return project_onto_constraint(np.asarray(h, dtype=float), n)

def soft_c5_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    c_scaled = corr * dx
    mc = np.max(c_scaled)
    exp_c = np.exp(t * (c_scaled - mc))
    s = np.sum(exp_c)
    w = exp_c / s
    c_soft = mc + np.log(s) / t
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-14:
            continue
        i_start = max(0, (n - 1) - k)
        i_end = min(n, 2 * (n - 1) - k + 1)
        if i_start >= i_end:
            continue
        v_start = i_start + k - (n - 1)
        v_len = i_end - i_start
        grad[i_start:i_end] += wk * v[v_start:v_start + v_len]
    grad *= dx
    return c_soft, grad, c_scaled

def proj_grad_descent(h, n, t, max_iters, time_limit, start_time):
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    alpha = 0.05
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 20:
            break
        c_soft, grad, _ = soft_c5_and_grad(h, t)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        step = alpha
        improved = False
        for _ls in range(20):
            h_try = project_onto_constraint(h - step * grad, n)
            c_try = compute_c5(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                improved = True
                break
            step *= 0.5
        if not improved:
            alpha *= 0.7
            if alpha < 1e-8:

    return h, best_c

def coord_descent_fast(h, n, max_rounds, time_limit, start_time):
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 15:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 15:
                return h, best_c
            old = h[j]
            best_val = old
            best_obj = best_c
            cands = [0.0, 1.0, 0.5, old + 0.1, old - 0.1, old + 0.04, old - 0.04,
                     old + 0.01, old - 0.01]
            for cv in cands:

                best_c = best_obj

        if not improved:

    return h, best_c

def binary_round(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        excess = n_ones - target
        h_bin = h_bin.copy()
        h_bin[one_idx[order[:excess]]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(h[zero_idx])[::-1]
        need = target - n_ones
        h_bin = h_bin.copy()
        h_bin[zero_idx[order[:need]]] = 1.0
    return h_bin

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    best_h = h.copy()

        T = T0 * (T1 / T0) ** (it / max(n_iters, 1))

        i = rng.choice(ones_idx)

        c_new = compute_c5(h)
        delta = c_new - best_c

            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

    if time.time() - start_time < budget_s - 250:
        h_g, c_g = proj_grad_descent(best_h, n, 200, 100, budget_s, start_time)
        h_g, c_g = coord_descent_fast(h_g, n, 8, budget_s, start_time)
        if c_g < best_c5:
            best_c5 = c_g
            best_h = h_g

    # Phase 5: Complement trick
    if time.time() - start_time < budget_s - 200:
        h_comp = 1.0 - best_h
        h_comp = make_feasible(h_comp, n)
        c_comp = compute_c5(h_comp)
        h_comp, c_comp = proj_grad_descent(h_comp, n, 100, 150, budget_s, start_time)
        h_comp, c_comp = coord_descent_fast(h_comp, n, 5, budget_s, start_time)
        if c_comp < best_c5:
            best_c5 = c_comp
            best_h = h_comp

    # Phase 6: Upsample to 2n
    n2 = n * 2
    if time.time() - start_time < budget_s - 200:
        h2 = upsample_h(best_h, n2)
        for t in [50, 200, 600]:
            if time.time() - start_time > budget_s - 100:
                break
            h2_new, c2_new = proj_grad_descent(h2, n2, t, 200, budget_s, start_time)
            if c2_new < best_c5 - 1e-10:
                best_c5 = c2_new
                best_h = h2_new
                h2 = h2_new
            else:
                h2 = h2_new
        h2, c2 = coord_descent_fast(h2, n2, 5, budget_s, start_time)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Phase 7: Binary at high res + anneal
    if time.time() - start_time < budget_s - 80:
        h_b2 = binary_round(best_h, len(best_h))
        c_b2 = compute_c5(h_b2)
        h_b2, c_b2 = binary_anneal(h_b2, len(best_h), 2000,
                                    budget_s, start_time, T0=0.1, T1=0.005, rng=rng)
        h_b2, c_b2 = coord_descent_fast(h_b2, len(best_h), 5, budget_s, start_time)
        if c_b2 < best_c5:
            best_c5 = c_b2
            best_h = h_b2

    # Phase 8: Final gradient polish
    if time.time() - start_time < budget_s - 20:
        n_f = len(best_h)
        h_f, c_f = proj_grad_descent(best_h, n_f, 500, 100, budget_s, start_time)
        h_f, c_f = coord_descent_fast(h_f, n_f, 3, budget_s, start_time)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    # Final: try binary
    if time.time() - start_time < budget_s - 5:
        h_fb = binary_round(best_h, len(best_h))
        c_fb = compute_c5(h_fb)
        if c_fb < best_c5:
            best_c5 = c_fb
            best_h = h_fb

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
