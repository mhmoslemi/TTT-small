def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    x = T * corr
    x_max = np.max(x)
    return dx * (x_max / T + np.log(np.sum(np.exp(x - x_max))) / (len(corr) * T))

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _smmax_fn(h, T, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = T * dx * corr
    zmax = np.max(z)
    return (zmax + np.log(np.sum(np.exp(z - zmax)))) / T

def _smmax_grad(h, T, N):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = T * dx * corr
    zmax = np.max(z)
    p = np.exp(z - zmax)
    p /= p.sum()
    
    # A[m] = sum of p[k] for valid k
    p_prefix = np.concatenate([[0.0], np.cumsum(p)])
    A = np.zeros(N)
    for m in range(N):
        k_lo = max(0, N - 1 - m)
        k_hi = min(2 * N - 2, N - 1 + m)
        A[m] = p_prefix[k_hi + 1] - p_prefix[k_lo]
    
    # B[m] = np.correlate(h, p, 'full')[2*N-3-m]
    B_full = np.correlate(h, p, mode='full')
    B = B_full[2 * N - 3 : N - 2 : -1]
    
    # C[m] = np.convolve(p, h, 'full')[m+N-1]
    C_full = np.convolve(p, h, mode='full')
    C = C_full[N - 1 : 2 * N - 1]
    
    return dx * (A - B - C)

def _slsqp_optimize(h_start, N, T, max_iter=200):
    dx = 2.0 / N
    
    def obj(x):
        return _smmax_fn(x, T, N)
    
    def grad(x):
        return _smmax_grad(x, T, N)
    
    def con_fun(x):
        return np.sum(x) - N / 2.0
    
    def con_jac(x):
        return np.ones(N)
    
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    res = minimize(obj, h_start, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, N)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _smmax_fn(x, T, N):
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode='full')
    z = T * dx * corr
    zmax = np.max(z)
    return (zmax + np.log(np.sum(np.exp(z - zmax)))) / T

def _smmax_grad(x, T, N):
    dx = 2.0 / N
    corr = np.correlate(x, 1.0 - x, mode='full')
    z = T * dx * corr
    zmax = np.max(z)
    p = np.exp(z - zmax)
    p /= p.sum()
    p_prefix = np.concatenate([[0.0], np.cumsum(p)])
    A = np.zeros(N)
    for m in range(N):
        k_lo = max(0, N - 1 - m)
        k_hi = min(2 * N - 2, N - 1 + m)
        A[m] = p_prefix[k_hi + 1] - p_prefix[k_lo]
    B_full = np.correlate(x, p, mode='full')
    B = B_full[2 * N - 3 : N - 2 : -1]
    C_full = np.convolve(p, x, mode='full')
    C = C_full[N - 1 : 2 * N - 1]
    return dx * (A - B - C)

def _slsqp_opt(h0, N, T, max_iter=200):
    def obj(x):
        return _smmax_fn(x, T, N)
    def grad(x):
        return _smmax_grad(x, T, N)
    def con_f(x):
        return np.sum(x) - N / 2.0
    def con_j(x):
        return np.ones(N)
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    res = minimize(obj, h0, jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'maxls': 50})
    if not np.isfinite(res.fun):
        return make_feasible(h0, N)
    return make_feasible(res.x, N)

def _cd_light(h, N, cur_c5, time_left_fn, max_sweeps=3, n_indices=50):
    h = h.copy()
    rng = np.random.RandomState(123)
    for sweep in range(max_sweeps):
        if time_left_fn() < 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-10:]
        grad = _smmax_grad(h, 1.0, N)
        idx_candidates = np.argsort(np.abs(grad))[-n_indices:]
        rng.shuffle(idx_candidates)
        improved = False
        for m in idx_candidates:
            if time_left_fn() < 15:
                break
            old_v = h[m]
            candidates = [0.0, 1.0]
            if 0.05 < old_v < 0.95:
                candidates += [old_v * 0.5, old_v + (1.0 - old_v) * 0.5]
            for cv in candidates:
                if abs(cv - old_v) < 1e-10:
                    continue
                trial = h.copy()
                trial[m] = cv
                trial = make_feasible(trial, N)
                c = compute_c5(trial)
                if c < cur_c5 - 1e-12:
                    h = trial
                    cur_c5 = c
                    old_v = cv
                    improved = True
        if not improved:
            break
    return h, cur_c5
