```python
def compute_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_shifted = np.exp(T * shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp * dx
    
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    
    for lag in range(min(2*n-1, int(np.sum(weights > 1e-10)) * 2)):
        w = weights[lag]
        if w < 1e-12:
            continue
        if lag < n:
            # d(corr[lag])/dh[j]
            # For j in [lag, n-1]: (1-h[j-lag])
            # For j in [0, n-1-lag]: -h[j+lag]
            grad[lag:] += w * one_minus_h[:n-lag]
            grad[:n-lag] += w * (-h[lag:])
        else:
            k = lag - n
            if k == 0:
                continue  # same as lag 0
            # d(corr[n+k])/dh[j]
            # For j in [0, n-1-k]: (1-h[j+k])
            # For j in [k, n-1]: -h[j-k]
            grad[:n-k] += w * one_minus_h[k:]
            grad[k:] += w * (-h[:n-k])
    
    return grad
```

For the direct approach, here's the idea:
- At the optimum, the gradient of the active constraint is in the null space of the feasible set
- We can use a "max gradient descent" approach:
  1. Compute all overlaps, find the argmax
  2. Compute gradient of that overlap w.r.t. h
  3. Take a step in the negative gradient direction
  4. Project to feasible set
  5. Repeat

This is subgradient descent for the max function. It converges slowly but can get stuck in good local optima.

To speed it up, use a "barrier" approach:
- Maintain a set of "active" shifts (top K)
- Minimize the sum (or max) of the active overlaps
- Periodically update the active set

Let me also think about using cvxpy. The problem is non-convex (bilinear), but if we linearize:
- Fix h and optimize (1-h_shifted): linear
- Fix (1-h_shifted) and optimize h: linear

This is an alternating linear program, which at each step is a QP (or LP if we ignore the product term).

Actually, for a single shift k:
minimize sum_i h[i] * (1-h[i+k])
subject to 0 ≤ h[i] ≤ 1, sum h[i] = n/2

This is a linear objective in h if h[i+k] is fixed! So it's an LP.

Alternating:
1. Fix h "base" values
2. Optimize h to minimize sum_k alpha_k * (overlap_k) (linear in h if base is fixed)
3. Update the "base" to the new h
4. Repeat

This is a Fixed-Point / Alternating Direction approach. Each inner step is an LP (solvable with cvxpy or even just sign(threshold) for linear objectives with box + sum constraints).

Actually, for a linear objective c^T h with constraints 0≤h≤1, sum h = n/2:
The solution is: set h[i] = 1 for the n/2 indices with smallest c[i], h[i] = 0 for the rest (with one fractional value if there's a tie). This is O(n log n)!

So the alternating algorithm becomes:
1. h_old = current solution
2. Compute cost vector: c[j] = sum_k alpha_k * d(overlap_k)/dh[j] |_{h_old}
3. h_new = threshold(c) to satisfy {0≤h≤1, sum h = n/2}
4. Blend: h = (1-t)*h + t*h_new
5. Repeat

This is a "proximal" alternating method. Very fast per iteration!

```python
def alternating_minmax(h, n, n_iter=1000, blend=0.5):
    dx = 2.0 / n
    h_cur = h.copy()
    
    for it in range(n_iter):
        # Compute all overlaps
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        mc = np.max(corr)
        
        # Find active shifts (top K)
        threshold = mc - 0.01 * mc  # or some fraction
        active = corr > threshold
        K = np.sum(active)
        if K == 0:
            active = np.zeros(len(corr), bool)
            active[np.argmax(corr)] = True
        
        # Compute gradient of the "soft max" of active overlaps
        # For the alternating approach, just use uniform weights on active set
        weights = np.zeros(len(corr))
        weights[active] = 1.0 / K * dx
        
        # Cost vector for linearization
        c = np.zeros(n)
        one_minus_h = 1.0 - h_cur
        for lag in range(len(corr)):
            w = weights[lag]
            if w < 1e-15:
                continue
            if lag < n:
                c[lag:] += w * one_minus_h[:n-lag]
                c[:n-lag] += w * (-h_cur[lag:])
            else:
                k = lag - n
                if k == 0:
                    continue
                c[:n-k] += w * one_minus_h[k:]
                c[k:] += w * (-h_cur[:n-k])
        
        # Threshold to get binary-like solution
        # Solve: minimize c^T h s.t. 0<=h<=1, sum h = n/2
        # Solution: sort c, set h=1 for smallest c, h=0 for largest
        idx = np.argsort(c)
        h_new = np.zeros(n)
        h_new[idx[:n//2]] = 1.0
        
        # Blend
        h_new = (1 - blend) * h_cur + blend * h_new
        h_new = make_feasible(h_new, n)
        
        # Check convergence
        if np.max(np.abs(h_new - h_cur)) < 1e-10:
            h_cur = h_new
            break
        h_cur = h_new
    
    return h_cur
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < (1.0 - 1e-15)
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def obj_and_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Clip for numerical stability
    shifted = np.clip(shifted, -50.0/T, 50.0/T)
    exp_s = np.exp(T * shifted)
    sum_exp = np.sum(exp_s)
    val = (mc + np.log(sum_exp) / T) * dx
    
    weights = exp_s / sum_exp * dx
    # Compute gradient
    grad = np.zeros(n)
    omh = 1.0 - x  # one minus h
    # Only process lags with significant weight
    active = np.where(weights > 1e-10)[0]
    for lag in active:
        w = weights[lag]
        if lag < n:
            grad[lag:] += w * omh[:n-lag]
            grad[:n-lag] -= w * x[lag:]
        else:
            k = lag - n
            if k == 0:
                continue
            grad[:n-k] += w * omh[k:]
            grad[k:] -= w * x[:n-k]
    return val, grad

    start = time.time()
    def fun(x):
        return obj_and_grad(x, n, T)[0]

        return obj_and_grad(x, n, T)[1]

        v, g = obj_and_grad(x, n, T)
        return v, g
    
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                   'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(fun_combined, h0, method='L-BFGS-B',
                   bounds=bounds, constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-8,
                            'maxfun': 5000, 'disp': False})
    if time.time() - start > time_limit:
        return h0
    return res.x
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < (1.0 - 1e-12)
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

def compute_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0 / T, 50.0 / T)
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def compute_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0 / T, 50.0 / T)
    exp_s = np.exp(T * shifted)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp * dx
    grad = np.zeros(n)
    omh = 1.0 - x
    active = np.where(weights > 1e-10)[0]
    for i in active:
        w = weights[i]
        k = i - (n - 1)
        if k >= 0:
            grad[k:] += w * omh[:n - k]
            grad[:n - k] += w * (-x[k:])
        else:
            kp = -k
            grad[:n - kp] += w * omh[kp:]
            grad[kp:] += w * (-x[:n - kp])
    return grad

def project_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
```
