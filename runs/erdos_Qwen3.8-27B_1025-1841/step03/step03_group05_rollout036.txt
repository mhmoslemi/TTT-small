import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shift = corr - mc
    shift -= shift.max()  # for numerical stability
    exps = np.exp(T * shift)
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = np.argmax(corr)
        return _subgrad_one(h, int(idx) - (n - 1)) * dx
    w = exps / tot
    # Use top_k for efficiency
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-14:
            break
        g += wi * _subgrad_one(h, int(i) - (n - 1))
    return g * dx

    t = np.array([0.0])

        t = np.concatenate([t, 1.0 - t])

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def _sobj(h, T, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _sobj_grad(h, T, n):

def binary_local_search_fast(h, n_iter=200, t0=0, budget=1300):
    n = len(h)
    dx = 2.0 / n
    rng_local = np.random.RandomState(42)
    
    # Round to binary and fix sum
    h_bin = (h > 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target = n // 2
    if ones > target:
        # Flip (ones - target) 1s to 0s, starting from those closest to 0.5
        idx1 = np.where(h_bin == 1)[0]
        order = idx1[np.argsort(h[idx1])[::-1]]  # highest h first (closest to 0.5)
        for k in range(ones - target):
            h_bin[order[k]] = 0
    elif ones < target:
        idx0 = np.where(h_bin == 0)[0]
        order = idx0[np.argsort(-h[idx0])]  # highest h first (closest to 0.5)
        for k in range(target - ones):
            h_bin[order[k]] = 1
    
    # Precompute correlation array
    corr = np.correlate(h_bin, 1 - h_bin, mode='full')  # length 2n-1
    bc = float(np.max(corr) * dx)
    bh = h_bin.copy()
    
    for it in range(n_iter):
        if time.time() - t0 > budget:
            break

        ones_idx = np.where(bh == 1)[0]
        zeros_idx = np.where(bh == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        n_swaps = min(1000, len(ones_idx) * len(zeros_idx))
        for _ in range(n_swaps):

            # Actually, np.correlate(a, b, mode='full')[k] = sum_m a[m] * b[m+k-(len(b)-1)]

def binary_ls(h_init, max_iter=50, swaps_per_iter=500, t0=0, budget=1300, seed=42):
    n = len(h_init)
    dx = 2.0 / n
    rng_ls = np.random.RandomState(seed)
    
    # Initialize binary solution
    bh = (h_init > 0.5).astype(float)
    target = n // 2
    ones = int(np.sum(bh))
    if ones > target:
        idx1 = np.where(bh == 1)[0]
        # Flip 1s with highest h_init value (closest to 0.5)
        order = idx1[np.argsort(h_init[idx1])[::-1]]
        for k in range(min(ones - target, len(order))):
            bh[order[k]] = 0
    elif ones < target:
        idx0 = np.where(bh == 0)[0]
        order = idx0[np.argsort(-h_init[idx0])]
        for k in range(min(target - ones, len(order))):
            bh[order[k]] = 1
    
    corr = np.correlate(bh, 1.0 - bh, mode='full')
    bc = float(np.max(corr) * dx)
    print(f"  BLS init: C5={bc:.6f}", flush=True)
    
    for it in range(max_iter):
        if time.time() - t0 > budget:
            break
        improved = False
        ones_idx = np.where(bh == 1)[0]
        zeros_idx = np.where(bh == 0)[0]
        
        for _ in range(swaps_per_iter):
            i = ones_idx[rng_ls.randint(len(ones_idx))]
            j = zeros_idx[rng_ls.randint(len(zeros_idx))]
            
            bh[i] = 0.0
            bh[j] = 1.0
            new_corr = np.correlate(bh, 1.0 - bh, mode='full')
            new_mc = float(np.max(new_corr))
            
            if new_mc < np.max(corr) - 1e-10:
                corr = new_corr
                bc = new_mc * dx
                improved = True
                # Update indices
                ones_idx = np.where(bh == 1)[0]
                zeros_idx = np.where(bh == 0)[0]
            else:
                bh[i] = 1.0
                bh[j] = 0.0
        
        if not improved:
            break
        if (it+1) % 5 == 0:
            print(f"  BLS it{it}: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    
    return bh, bc

**Faster correlation update**: When we change h[i] from 1 to 0, the change in corr[k] is:
- For the term involving h[i] as the first factor: h[i]*(1-h[i+k-n+1]) for each k where i+k-n+1 is valid
  - corr[k] decreases by (1-h[i+k-n+1]) for each valid k
- For the term involving h[i] as the second factor (i.e., h[m]*(1-h[i]) where m+k-n+1 = i, so m = i-k+n-1):
  - Previously: h[m]*(1-1) = 0
  - Now: h[m]*(1-0) = h[m]
  - corr[k] increases by h[m] = h[i-k+n-1] for each valid m

So the change in corr when flipping h[i] from 1 to 0 is:
Δ[k] = -(1-h[i+k-n+1]) + h[i-k+n-1] (for valid indices)

This is O(n) to compute (update n entries of corr). Similarly for h[j] from 0 to 1:
Δ[k] = +(1-h[j+k-n+1]) - h[j-k+n-1] (for valid indices)

So the total update for a swap is O(n) = O(960) operations, which is very fast!

    Compute the change in corr array when bh[i] changes by val_change (e.g., -1 for 1→0, +1 for 0→1).

s=-2 (k=1): First: 0+(-2)=-2, not valid. Second: 0-(-2)=2, valid. -1*(1-bh[2]) = -1*(1-1) = 0. So Δcorr[1]=0. Original=0. ✓

s=0 (k=3): Δcorr=0. Original=0. ✓

s=1 (k=4): First: 0+1=1, valid. -(-1)*bh[1] = 1*0 = 0. Second: 0-1=-1, not valid. Δcorr[4]=0. Original=1. ✓

s=2 (k=5): First: 0+2=2, valid. 1*bh[2] = 1*1 = 1. Second: 0-2=-2, not valid. Δcorr[5]=1. Original=0. New: 0+1=1. ✓

s=3 (k=6): First: 0+3=3, valid. 1*bh[3] = 1*0 = 0. Second: 0-3=-3, not valid. Δcorr[6]=0. Original=0. ✓

So new corr = [0, 0, 1, 0, 1, 1, 0]. ✓ 

The formula works. Now I can implement the fast binary local search.

For a swap (i: 1→0, j: 0→1):
Δcorr = delta_flip(i, -1) + delta_flip(j, +1)

where delta_flip(i, Δ) is computed in O(n) time.

Total cost per swap: O(n) = O(960). Much faster than O(n²) for full recomputation!
