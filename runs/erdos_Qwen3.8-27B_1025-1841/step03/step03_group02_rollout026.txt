def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

_SLS_N = 0
_SLS_T = 0.0

def sls_obj(x):
    n = _SLS_N
    T = _SLS_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def sls_con(x):
    return np.sum(x) - len(x) / 2.0

def sls_con_jac(x):
    return np.ones(len(x))

def sls_run(h, n, T, maxiter, ftol):
    global _SLS_N, _SLS_T
    _SLS_N = n
    _SLS_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sls_con, 'jac': sls_con_jac}]
    res = minimize(sls_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    remaining = budget_s - (time.time() - start_time)
    if remaining > 120 and n >= 960:
        n_lo = 480
        h_lo = upsample(best_h, n_lo)
        b_lo = to_binary(h_lo, n_lo)
        b_lo, c5_lo = sa_binary(b_lo, n_lo, 80000, 0.015, 0.00001, rng, start_time, start_time + remaining - 10)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = b_lo.copy()

    # Phase 8: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        n_hi = min(n + 480, 1440)
        h_hi = upsample(best_h, n_hi)
        b_hi = to_binary(h_hi, n_hi)
        b_hi, c5_hi = sa_binary(b_hi, n_hi, 60000, 0.01, 0.000005, rng, start_time, start_time + remaining - 10)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = b_hi.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 30:
                try:
                    x_res = sls_run(best_h, n_hi, 300.0, 150, 1e-14)
                    c5_res = compute_c5(make_feasible(x_res, n_hi))
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(x_res, n_hi)
                except Exception:
                    pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
