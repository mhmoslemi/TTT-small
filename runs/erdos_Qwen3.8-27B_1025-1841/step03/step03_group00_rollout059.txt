import numpy as np
from scipy.optimize import minimize
import time

def fft_full_corr(h):
    n = len(h)
    Z = 1
    while Z < 2 * n:
        Z *= 2
    A = np.fft.rfft(h, Z)
    B = np.fft.rfft(1.0 - h, Z)
    return np.fft.irfft(A * np.conj(B), Z)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_full_corr(h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = np.clip(diff / cnt, -0.99, 0.99)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(h)[-abs(int(diff)):]
            for i in idx:
                h[i] = np.clip(h[i] + diff / len(idx), 0.0, 1.0)
        else:
            idx = np.argsort(h)[:abs(int(diff)):]
            for i in idx:
                h[i] = np.clip(h[i] + diff / len(idx), 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_optimize(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    
    def obj(h):
        corr = fft_full_corr(h)
        mc = np.max(corr)
        # Log-sum-exp smoothing
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -50, 50)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / len(corr)) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def binary_swap_search(h_bin, n, rng, max_rounds=200, n_tries=300):
    h = h_bin.astype(float).copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_tries):
            if len(ones) == 0 or len(zeros) == 0:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            a, b = ones[ii], zeros[jj]
            h[a] = 0.0
            h[b] = 1.0
            c = compute_c5(h)
            if c < current - 1e-14:
                current = c
                ones = np.append(ones[np.delete(ones, ii)], b)
                zeros = np.append(zeros[np.delete(zeros, jj)], a)
                improved = True
            else:
                h[a] = 1.0
                h[b] = 0.0
        if not improved:
            break
    return h, current

    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_opt(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def obj(h):
        corr = fft_full_corr(h)
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -50, 50)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / len(corr)) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=300, n_tries=500):
    h = h_bin.astype(float).copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_tries):
            if len(ones) == 0 or len(zeros) == 0:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            a, b = ones[ii], zeros[jj]
            h[a] = 0.0
            h[b] = 1.0
            c = compute_c5(h)
            if c < current - 1e-14:
                current = c
                ones = np.append(ones[np.delete(ones, ii)], b)
                zeros = np.append(zeros[np.delete(zeros, jj)], a)
                improved = True
            else:
                h[a] = 1.0
                h[b] = 0.0
        if not improved:
            break
    return h, current

def sim_anneal_binary(h_bin, n, rng, max_iters=50000, T0=0.05, T1=0.005):
    h = h_bin.astype(float).copy()
    current = compute_c5(h)
    best_h = h.copy()
    best_c = current
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for it in range(max_iters):
        T = T0 * (T1 / T0) ** (it / max_iters)
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        a, b = ones[ii], zeros[jj]
        h[a] = 0.0
        h[b] = 1.0
        c = compute_c5(h)
        delta = c - current
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current = c
            ones = np.append(ones[np.delete(ones, ii)], b)
            zeros = np.append(zeros[np.delete(zeros, jj)], a)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
    return best_h, best_c

def subgrad_step(h, n, alpha, top_k=5):
    corr = fft_full_corr(h)
    n_lags = 2 * n - 1
    valid = corr[:n_lags] if n_lags <= len(corr) else corr
    idx_sorted = np.argsort(valid)[::-1][:top_k]
    weights = np.exp(T * (valid[idx_sorted] - valid[idx_sorted[0]]))
    weights /= weights.sum()
    grad = np.zeros(n)
    for w, k in zip(weights, idx_sorted):
        if k == 0:
            for i in range(n):
                grad[i] += w * 1.0
                if i < n:
                    grad[i] -= w * (1.0 - h[i]) if False else 0
        elif k < n:
            for i in range(n - k):
                grad[i] += w * (1.0 - h[i + k])
                if i + k < n:
                    grad[i + k] -= w * h[i]
        else:
            kk = k - (2 * n - 1)
            for i in range(-kk):
                grad[i] -= w * h[i - kk]
                grad[i - kk] += w * (1.0 - h[i])
    return h, grad

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    n0 = len(best_h)

    # Phase 1: L-BFGS-B at n=1024
    n1 = 1024
    h_w = upsample_h(best_h, n1)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()

    for T in [100, 500, 2000]:
        if tl() < 30:
            break
        try:
            h_opt = lbgfb_opt(h_w, n1, T, mu=1000.0, max_iter=200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h_w = h_opt.copy()
        except Exception:
            pass

    # Phase 2: Binary local search at n=1024
    if tl() > 80:
        h_bin = (h_w > 0.5).astype(float)
        n_ones = int(np.sum(h_bin > 0.5))
        n_target = n1 // 2
        if n_ones > n_target:
            oi = np.where(h_bin == 1)[0]
            for k in range(n_ones - n_target):
                h_bin[oi[k]] = 0.0
        elif n_ones < n_target:
            zi = np.where(h_bin == 0)[0]
            for k in range(n_target - n_ones):
                h_bin[zi[k]] = 1.0
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()
        h_bs, c5_bs = binary_local_search(h_bin, n1, rng, max_rounds=200, n_tries=800)
        if c5_bs < best_c5:
            best_c5, best_h = c5_bs, h_bs.copy()

    # Phase 3: Upsample to n=2048 and optimize
    if tl() > 80:
        n2 = 2048
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
        for T in [200, 800, 3000]:
            if tl() < 25:
                break
            try:
                h_opt = lbgfb_opt(h2, n2, T, mu=2000.0, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h2 = h_opt.copy()
            except Exception:
                pass

        # Binary search at n=2048
        if tl() > 60:
            h_b2 = (h2 > 0.5).astype(float)
            n_on = int(np.sum(h_b2 > 0.5))
            n_tg = n2 // 2
            if n_on > n_tg:
                oi = np.where(h_b2 == 1)[0]
                for k in range(n_on - n_tg):
                    h_b2[oi[k]] = 0.0
            elif n_on < n_tg:
                zi = np.where(h_b2 == 0)[0]
                for k in range(n_tg - n_on):
                    h_b2[zi[k]] = 1.0
            c5b2 = compute_c5(h_b2)
            if c5b2 < best_c5:
                best_c5, best_h = c5b2, h_b2.copy()
            h_bs2, c5_bs2 = binary_local_search(h_b2, n2, rng, max_rounds=150, n_tries=1000)
            if c5_bs2 < best_c5:
                best_c5, best_h = c5_bs2, h_bs2.copy()

    # Phase 4: Simulated annealing at n=2048
    if tl() > 60:
        n_sa = 2048
        h_sa = (best_h if len(best_h) == n_sa else upsample_h(best_h, n_sa) > 0.5).astype(float)
        n_on = int(np.sum(h_sa > 0.5))
        n_tg = n_sa // 2
        if n_on > n_tg:
            oi = np.where(h_sa == 1)[0]
            for k in range(n_on - n_tg):
                h_sa[oi[k]] = 0.0
        elif n_on < n_tg:
            zi = np.where(h_sa == 0)[0]
            for k in range(n_tg - n_on):
                h_sa[zi[k]] = 1.0
        h_sa_out, c5_sa = sim_anneal_binary(h_sa, n_sa, rng, max_iters=min(30000, int(tl() * 3)), T0=0.03, T1=0.003)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa_out.copy()

    # Phase 5: Higher resolution n=4096
    if tl() > 60:
        n3 = 4096
        h3 = upsample_h(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()
        for T in [500, 2000]:
            if tl() < 30:
                break
            try:
                h_opt = lbgfb_opt(h3, n3, T, mu=5000.0, max_iter=100)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h3 = h_opt.copy()
            except Exception:
                pass
        if tl() > 40:
            h_b3 = (h3 > 0.5).astype(float)
            n_on = int(np.sum(h_b3 > 0.5))
            n_tg = n3 // 2
            if n_on > n_tg:
                oi = np.where(h_b3 == 1)[0]
                for k in range(n_on - n_tg):
                    h_b3[oi[k]] = 0.0
            elif n_on < n_tg:
                zi = np.where(h_b3 == 0)[0]
                for k in range(n_tg - n_on):
                    h_b3[zi[k]] = 1.0
            h_bs3, c5_bs3 = binary_local_search(h_b3, n3, rng, max_rounds=80, n_tries=500)
            if c5_bs3 < best_c5:
                best_c5, best_h = c5_bs3, h_bs3.copy()

    # Phase 6: Final coordinate descent polish
    if tl() > 20:
        n_cd = min(len(best_h), 2048)
        h_cd = best_h.copy() if len(best_h) == n_cd else upsample_h(best_h, n_cd)
        deltas = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]
        for rnd in range(5):
            if tl() < 10:
                break
            improved = False
            for j in range(n_cd):
                if tl() < 10:
                    break
                ov = h_cd[j]
                for d in deltas:
                    nv = ov + d
                    if nv < -1e-12 or nv > 1 + 1e-12:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        improved = True
                        ov = nv
                        break
                for bv in (0.0, 1.0):
                    if abs(bv - h_cd[j]) < 1e-9:
                        continue
                    tr = h_cd.copy()
                    tr[j] = bv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        improved = True
            if not improved:
                break
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Final SLSQP squeeze
    if tl() > 20:
        n_f = min(len(best_h), 1024)
        h_f = best_h.copy() if len(best_h) == n_f else upsample_h(best_h, n_f)
        dx = 2.0 / n_f
        def obj_f(h):
            corr = fft_full_corr(h)
            mc = np.max(corr)
            shifted = 100.0 * (corr - mc)
            shifted = np.clip(shifted, -50, 50)
            return (mc + np.log(np.sum(np.exp(shifted))) / len(corr)) * dx
        def con_f(h):
            return np.sum(h) - n_f / 2.0
        bounds_f = [(0.0, 1.0)] * n_f
        cons_f = [{'type': 'eq', 'fun': con_f, 'jac': lambda h: np.ones(n_f)}]
        try:
            res = minimize(obj_f, h_f, method='SLSQP', bounds=bounds_f,
                           constraints=cons_f, options={'maxiter': 80, 'ftol': 1e-14})
            h_opt = make_feasible(res.x, n_f)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
