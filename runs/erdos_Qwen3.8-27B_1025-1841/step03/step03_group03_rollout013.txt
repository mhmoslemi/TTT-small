import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project_binary(h, n):
    """Project to binary with exactly n//2 ones."""
    h = np.asarray(h, dtype=float)
    # Sort indices by value, keep top n//2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def binary_local_search(h, n, max_iters=10000, time_limit=None, start_time=None, rng=None):
    """Greedy local search: try all swaps that reduce the max correlation."""
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = np.max(corr) * dx
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    for it in range(max_iters):
        if time_limit and start_time and time.time() - start_time > time_limit - 30:
            break
        
        # Find all improving swaps (expensive but thorough for small n)
        # Or: random swaps
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        
        # Try a subset of random swaps
        n_trials = min(200, n_ones * n_zeros)
        for _ in range(n_trials):
            i = ones[rng.randint(n_ones)]
            j = zeros[rng.randint(n_zeros)]
            if i == j:
                continue
            
            # Try the swap
            h[i], h[j] = h[j], h[i]  # h[i] was 1→0, h[j] was 0→1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_c = np.max(new_corr) * dx
            h[i], h[j] = h[j], h[i]  # undo
            
            if new_c < best_c - 1e-12:
                h[i], h[j] = h[j], h[i]  # do the swap
                corr = new_corr
                best_c = new_c
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                improved = True
                break  # restart with new state
        
        if not improved:
            break
    
    return h, best_c

**Structured local search**: 
- Find the current bottleneck lag k* (argmax of corr)
- Generate candidate swaps specifically designed to reduce corr[k*]
- For lag k*, corr[k*] = sum_i h[i]*(1-h[i-k*+n-1]) over valid i
- To reduce this, we want to either:
  - Change h[i] from 1 to 0 where h[i-k*+n-1]=0 (reduces by 1)
  - Change h[i-k*+n-1] from 0 to 1 where h[i]=1 (reduces by 1)
- But we must maintain the constraint (equal number of 1s and 0s), so we do a swap.

This is essentially: find a 1 at position i and a 0 at position j such that the swap reduces the bottleneck most.

For a given bottleneck lag k*, the positions that matter are the "mismatches": where h[i]=1 and h[i-k*+n-1]=0 (for valid indices in the lag-k* window).

If we swap a 1 at position i (where i is in the mismatch set for lag k*) with a 0 at position i-k*+n-1, that directly reduces corr[k*] by 1. But it might increase other lags.

This is a directed search: target the bottleneck, accept if it doesn't hurt others too much.

Let me implement this:

def structured_local_search(h, n, max_iters=100000, time_limit=None, start_time=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = np.max(corr) * dx
    ones = set(np.where(h == 1)[0].tolist())
    zeros = set(np.where(h == 0)[0].tolist())
    
    for it in range(max_iters):
        if time_limit and start_time and time.time() - start_time > time_limit - 30:
            break
        
        # Find bottleneck lags
        max_val = np.max(corr)
        # Get indices where corr is within 1 of max
        bottleneck_lags = np.where(corr >= max_val - 1)[0]
        
        # For each bottleneck lag, find potential improving swaps
        best_improvement = 0
        best_swap = None
        
        for k in bottleneck_lags[:5]:  # limit to top 5
            # corr[k] = sum_i h[i]*(1-h[i-k+n-1])
            # Mismatches: h[i]=1 and h[i-k+n-1]=0
            for i in list(ones):
                q = i - k + n - 1
                if 0 <= q < n and h[q] == 0:
                    # Swapping i (1→0) with q (0→1) would reduce corr[k] by 1
                    # But might affect other lags
                    # Quick check: evaluate the full effect
                    h[i], h[q] = h[q], h[i]
                    new_corr = np.correlate(h, 1-h, mode='full')
                    new_max = np.max(new_corr)
                    h[i], h[q] = h[q], h[i]
                    
                    if new_max < max_val - best_improvement:
                        best_improvement = max_val - new_max
                        best_swap = (i, q)
        
        if best_swap is None:
            # No structured improvement found, try random swaps
            improved = False
            for _ in range(50):
                i = next(iter(ones))  # hmm, need random
                # ...
            break
        
        # Execute the best swap
        i, q = best_swap
        h[i], h[q] = h[q], h[i]
        corr = np.correlate(h, 1-h, mode='full')
        best_c = np.max(corr) * dx
        
        ones.discard(i)
        zeros.discard(q)
        ones.add(q)
        zeros.add(i)
    
    return h, best_c
