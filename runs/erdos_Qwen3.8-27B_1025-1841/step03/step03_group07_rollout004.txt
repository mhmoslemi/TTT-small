```python
def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def gradient_for_shift(h, k_offset, n):

    # shift = k_offset - (n-1)

    # If i = j+shift: -h[j] for valid j

    hi2 = min(n, n + shift)
```

```python
import cvxpy as cp

def lp_subgrad_step(h, n, t_current, top_k=10):
    """One step of linearized subgradient using LP."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    
    # Linearize: C(k) ≈ C(k)_current + grad_k · δ
    # where grad_k is the gradient of C(k) at h
    # C(k) = sum_i h[i](1-h[i+k])
    # dC(k)/dh[i] = (1-h[i+k]) - h[i-k] (with bounds)
    
    t = cp.Variable()
    delta = cp.Variable(n)
    
    constraints = [
        cp.sum(delta) == 0,  # Maintain sum constraint
        delta >= -h,          # h + delta >= 0
        delta <= 1 - h        # h + delta <= 1
    ]
    
    # Linear constraints for top correlations
    for k in top_idx:
        shift = k - (n - 1)
        # grad = (1-h[i+shift]) - h[i-shift]
        g = np.zeros(n)
        lo1 = max(0, -shift)
        hi1 = min(n, n - shift)
        if lo1 < hi1:
            g[lo1:hi1] += (1.0 - h[lo1+shift:hi1+shift])
        lo2 = max(0, shift)
        hi2 = min(n, n + shift)
        if lo2 < hi2:
            g[lo2:hi2] -= h[lo2-shift:hi2-shift]
        
        # Linearized: C(k) + dx * (g · δ) ≤ t
        constraints.append(corr[k] * dx + dx * cp.sum(g * delta) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)
    
    if delta.value is not None:
        h_new = h + delta.value
        h_new = np.clip(h_new, 0, 1)
        h_new = project_to_feasible(h_new, n)
        return h_new
    return h.copy()
```

```python
            grad[lo2:hi2] -= pk * h[lo2 - shift:hi2 - shift]
    return grad


def nesterov_opt(h, n, tau, time_limit, lr_init, top_k):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    h_prev = h.copy()
    t_val = 1.0
    stall = 0
    best_lr = lr

    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        mu = min((t_val - 1.0) / (t_val + 2.0), 0.85)
        y = np.clip(h + mu * (h - h_prev), 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = corr.max()

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))

        top_vals = corr[top_idx]
        shifted = tau * (top_vals - max_c)
        shifted = np.clip(shifted, -80.0, 0.0)
        w = np.exp(shifted)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(top_idx)) / len(top_idx)
        else:
            w /= ws

        grad = compute_grad_weighted(y, top_idx, w, n)
        grad *= dx

        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)

        if it % 30 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
                if best_lr > lr:
                    best_lr = lr * 1.1
            else:
                stall += 30
            if stall > 1200:
                lr *= 0.6
                stall = 0
                if lr < 1e-14:
                    break

        t_val = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val))
        h_prev = h
        h = h_new

    c5_f = compute_c5(h)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h.copy()
    return best_h, best_c5


def subgrad_polish(h, n, time_limit, top_k, lr_init):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.7
    stall = 0

    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2500:
            lr *= 0.55
            v *= 0.2
            stall = 0
            if lr < 1e-14:
                break

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        mx = top_vals.max()
        w = np.exp(top_vals - mx)
        w /= w.sum()

        grad = compute_grad_weighted(h, top_idx, w, n)
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    c5_f = compute_c5(h)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h.copy()
    return best_h, best_c5


def get_remaining(st, bs):
    return max(0.0, bs - (time.time() - st))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    h_work = h_init.copy()

    # Phase 1: Very smooth, fast convergence
    rem = get_remaining(start_time, budget_s)
    if rem > 170:
        tl = min(150, rem * 0.22)
        h_work, c5 = nesterov_opt(h_work, n, tau=500.0, time_limit=tl, lr_init=0.10, top_k=20)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 2: Medium smooth
    rem = get_remaining(start_time, budget_s)
    if rem > 140:
        tl = min(120, rem * 0.2)
        h_work, c5 = nesterov_opt(h_work, n, tau=120.0, time_limit=tl, lr_init=0.05, top_k=10)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 3: Focused
    rem = get_remaining(start_time, budget_s)
    if rem > 110:
        tl = min(95, rem * 0.18)
        h_work, c5 = nesterov_opt(h_work, n, tau=40.0, time_limit=tl, lr_init=0.02, top_k=6)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 4: Sharp
    rem = get_remaining(start_time, budget_s)
    if rem > 90:
        tl = min(75, rem * 0.16)
        h_work, c5 = nesterov_opt(h_work, n, tau=12.0, time_limit=tl, lr_init=0.008, top_k=4)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 5: Subgradient refinement
    rem = get_remaining(start_time, budget_s)
    if rem > 70:
        tl = min(60, rem * 0.15)
        h_work, c5 = subgrad_polish(h_work, n, time_limit=tl, top_k=4, lr_init=0.0006)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 6: Multi-start perturbations
    rem = get_remaining(start_time, budget_s)
    perturbations = [0.004, 0.007, 0.002, 0.010, 0.003, 0.005]
    for pert in perturbations:
        rem = get_remaining(start_time, budget_s)
        if rem < 50:
            break
        h_p = best_h.copy() + np.random.randn(n) * pert
        h_p = project_to_feasible(h_p, n)
        tl = min(35, rem * 0.1)
        h_p, c5_p = nesterov_opt(h_p, n, tau=30.0, time_limit=tl, lr_init=0.025, top_k=6)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 7: Try circular shift of best solution then optimize
    rem = get_remaining(start_time, budget_s)
    if rem > 50:
        for frac in [0.3, 0.7, 0.5]:
            rem = get_remaining(start_time, budget_s)
            if rem < 30:
                break
            shift_amt = int(n * frac)
            h_s = np.roll(best_h, shift_amt)
            h_s = project_to_feasible(h_s, n)
            tl = min(20, rem * 0.25)
            h_s, c5_s = nesterov_opt(h_s, n, tau=25.0, time_limit=tl, lr_init=0.02, top_k=5)
            if c5_s < best_c5:
                best_c5, best_h = c5_s, h_s.copy()

    # Phase 8: Final subgradient polish
    rem = get_remaining(start_time, budget_s)
    if rem > 30:
        tl = min(25, rem * 0.35)
        h_work, c5 = subgrad_polish(best_h, n, time_limit=tl, top_k=3, lr_init=0.0003)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 9: Very final Nesterov at low lr
    rem = get_remaining(start_time, budget_s)
    if rem > 20:
        tl = min(18, rem * 0.4)
        h_work, c5 = nesterov_opt(best_h, n, tau=8.0, time_limit=tl, lr_init=0.003, top_k=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Final projection and evaluation
    best_h = project_to_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
```
