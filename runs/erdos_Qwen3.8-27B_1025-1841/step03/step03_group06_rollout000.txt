        if time.time() - start_t > time_limit:
            break
        for rnd in range(50):
            if time.time() - start_t > time_limit:
                break
            improved = False
            for j in range(n):
                if time.time() - start_t > time_limit:
                    break
                old = best_h[j]
                cands = [0.0, 1.0,
                         max(0.0, old - step), min(1.0, old + step),
                         max(0.0, old - step * 0.5), min(1.0, old + step * 0.5)]
                best_trial = old
                best_tc = best_c
                for cand in cands:
                    if abs(cand - old) < 1e-8:
                        continue
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_tc - 1e-13:
                        best_tc = c
                        best_trial = cand
                if best_trial != old:
                    best_h = best_h.copy()
                    best_h[j] = best_trial
                    best_h = make_feasible(best_h, n)
                    best_c = best_tc
                    improved = True
            if not improved:
                break
    return best_h, best_c

def simulated_annealing(h, n, time_limit, seed_val):
    rng = np.random.RandomState(seed_val)
    best_h = h.copy()
    best_c = compute_c5(best_h)
    cur_h = h.copy()
    cur_c = best_c
    start_t = time.time()
    n_iter = 0
    T_start, T_end = 0.08, 0.0005
    while time.time() - start_t < time_limit:
        progress = (time.time() - start_t) / time_limit
        T = T_start * (T_end / T_start) ** progress
        trial = cur_h.copy()
        mt = rng.randint(4)
        if mt == 0:
            length = int(rng.choice([6, 12, 24, 48]))
            start = int(rng.randint(0, max(1, n - length)))
            trial[start:start + length] = 1.0 - trial[start:start + length]
        elif mt == 1:
            length = int(rng.choice([12, 24, 48, 96]))
            start = int(rng.randint(0, max(1, n - length)))
            shift = rng.uniform(-0.08, 0.08)
            trial[start:start + length] = np.clip(
                trial[start:start + length] + shift, 0, 1)
        elif mt == 2:
            j = int(rng.randint(n))
            trial[j] = float(rng.choice([0.0, 1.0, 0.5]))
        else:
            j1 = int(rng.randint(n))
            j2 = int(rng.randint(n))
            trial[j1], trial[j2] = trial[j2], trial[j1]
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < cur_c or (c - cur_c < 5 * T and
                         rng.random() < np.exp(-(c - cur_c) / max(T, 1e-12))):
            cur_h = trial
            cur_c = c
            if c < best_c:
                best_c = c
                best_h = trial.copy()
        n_iter += 1
        if n_iter % 800 == 0 and cur_c > best_c * 1.005:
            cur_h = best_h.copy()
            cur_c = best_c
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = thue_morse_h(960)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP at initial resolution, multiple T
    for T_val in [60.0, 150.0, 350.0, 700.0]:
        if remaining() < 130:
            break
        try:
            h_new = slsqp_optimize(best_h, n_init, T_val, 400, 1e-14)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 2: Upsample to higher resolution and optimize
    for n_target in [1280, 1920, 2560]:
        if remaining() < 200:
            break
        h_up = upsample_h(best_h, n_target)
        c_up = compute_c5(h_up)
        for T_val in [150.0, 400.0, 800.0]:
            if remaining() < 120:
                break
            try:
                h_new = slsqp_optimize(h_up, n_target, T_val, 300, 1e-14)
                c_new = compute_c5(h_new)
                if c_new < c_up:
                    c_up = c_new
                    h_up = h_new
            except Exception:
                pass
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up

    # Phase 3: Block coordinate descent
    n_cur = len(best_h)
    for bs in [6, 12, 24]:
        if remaining() < 80:
            break
        best_h, best_c5 = block_cd(best_h, n_cur, bs, 4,
                                   remaining() - 80, seed + bs)

    # Phase 4: Simulated annealing
    if remaining() > 120:
        sa_time = min(120.0, remaining() - 40)
        best_h, best_c5 = simulated_annealing(
            best_h, n_cur, sa_time, seed + 1000)

    # Phase 5: Binary snap
    if remaining() > 30:
        best_h, best_c5 = binary_snap(best_h, n_cur)

    # Phase 6: Coordinate descent (medium steps)
    if remaining() > 80:
        best_h, best_c5 = coord_descent(
            best_h, n_cur, remaining() - 40, [0.03, 0.015, 0.008])

    # Phase 7: Final SLSQP polish
    if remaining() > 50:
        n_cur = len(best_h)
        try:
            h_new = slsqp_optimize(best_h, n_cur, 1000.0, 200, 1e-15)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new
        except Exception:
            pass

    # Phase 8: Fine coordinate descent
    if remaining() > 30:
        n_cur = len(best_h)
        best_h, best_c5 = coord_descent(
            best_h, n_cur, remaining() - 10, [0.01, 0.005, 0.002, 0.001])

    # Phase 9: Ultimate binary snap + tiny adjustments
    if remaining() > 15:
        best_h, best_c5 = binary_snap(best_h, len(best_h))
        n_cur = len(best_h)
        best_h, best_c5 = coord_descent(
            best_h, n_cur, remaining() - 3, [0.003, 0.001])

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
