    target = n / 2.0

        ones = np.where(b == 1)[0]

    """Update corr after h[p]: 1->0, h[q]: 0->1. Returns new corr (modified in place for speed)."""
    # For each affected lag L, compute delta
    # p was 1, now 0:
    #   as h[i+L] where i=p-L: delta = -(1-h[p-L]) if 0<=p-L<=n-1
    #   as (1-h[i]) where i=p: delta = +h[p+L] if 0<=p+L<=n-1
    # q was 0, now 1:
    #   as h[i+L] where i=q-L: delta = +(1-h[q-L]) if 0<=q-L<=n-1
    #   as (1-h[i]) where i=q: delta = -h[q+L] if 0<=q+L<=n-1

    L_min = max(p - n + 1, q - n + 1, -(n-1))
    L_max = min(p, q, n-1)

    # For p: L in [p-n+1, p] (as h[i+L]) and L in [-p, n-1-p] (as (1-h[i]))
    # For q: L in [q-n+1, q] (as h[i+L]) and L in [-q, n-1-q] (as (1-h[i]))
    
    L_min = max(min(p-n+1, -p), min(q-n+1, -q), -(n-1))
    L_max = min(max(p, n-1-p), max(q, n-1-q), n-1)

        # p as h[i+L], i = p-L, valid if 0 <= p-L <= n-1
        if 0 <= p - L <= n - 1:

        # p as (1-h[i]), i = p, valid if 0 <= p+L <= n-1
        if 0 <= p + L <= n - 1:
            delta += h[p + L]
        # q as h[i+L], i = q-L, valid if 0 <= q-L <= n-1
        if 0 <= q - L <= n - 1:

        # q as (1-h[i]), i = q, valid if 0 <= q+L <= n-1
        if 0 <= q + L <= n - 1:
            delta -= h[q + L]
        
        idx = L + n - 1
        if 0 <= idx < len(corr):
            corr[idx] += delta
    
    # Update h
    h[p] = 0.0
    h[q] = 1.0
    return corr

def swap_update_reverse(corr, h, p, q, n):
    """Reverse: h[p]: 0->1, h[q]: 1->0"""
    # Same as swap_update with p and q swapped, and signs flipped
    # Actually, it's the same formula with p and q roles reversed
    # p was 0, now 1: delta = +(1-h[p-L]) - h[p+L] (same as q's contribution in forward)
    # q was 1, now 0: delta = -(1-h[q-L]) + h[q+L] (same as p's contribution in forward)
    # So it's the same as the forward formula with p and q swapped!
    
    L_min = max(min(q-n+1, -q), min(p-n+1, -p), -(n-1))
    L_max = min(max(q, n-1-q), max(p, n-1-p), n-1)
    
    for L in range(L_min, L_max + 1):
        delta = 0.0
        # p (was 0, now 1) as h[i+L]: +(1-h[p-L])
        if 0 <= p - L <= n - 1:
            delta += (1.0 - h[p - L])
        # p (was 0, now 1) as (1-h[i]): -h[p+L]
        if 0 <= p + L <= n - 1:
            delta -= h[p + L]
        # q (was 1, now 0) as h[i+L]: -(1-h[q-L])
        if 0 <= q - L <= n - 1:

        # q (was 1, now 0) as (1-h[i]): +h[q+L]
        if 0 <= q + L <= n - 1:
            delta += h[q + L]
        
        idx = L + n - 1
        if 0 <= idx < len(corr):
            corr[idx] += delta
    
    h[p] = 1.0
    h[q] = 0.0
    return corr

                best_h = h.copy()
            # Update ones/zeros tracking
            # p was 1, now 0; q was 0, now 1
            # For simplicity, just rebuild

        else:
