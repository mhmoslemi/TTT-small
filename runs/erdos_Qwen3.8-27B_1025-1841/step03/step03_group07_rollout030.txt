- When j = i (i.e., sh <= i <= n-1): contributes h[i-sh]
- When j-sh = i (i.e., j = i+sh, so sh <= i+sh <= n-1, i.e., 0 <= i <= n-1-sh): contributes h[i+sh]

So d/dh[i] of the second sum = h[i-sh]*[sh<=i<=n-1] + h[i+sh]*[0<=i<=n-1-sh]

Therefore:
d/dh[i] of corr[k] = [sh<=i<=n-1] - h[i-sh]*[sh<=i<=n-1] - h[i+sh]*[0<=i<=n-1-sh]

For i in [sh, n-1-sh] (when sh <= n-1-sh, i.e., sh <= (n-1)/2):
= 1 - h[i-sh] - h[i+sh]

For i in [n-sh, n-1] (only first bracket active):
= 1 - h[i-sh]

For i in [0, sh-1] (only second bracket active, and first bracket is 0):
= -h[i+sh]

```python
def fft_correlate(h, n):
    """Compute np.correlate(h, 1-h, mode='full') efficiently using FFT."""
    b = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(b, N)
    corr = np.fft.irfft(np.conj(fa) * fb, N)
    out = np.empty(2 * n - 1)
    out[n-1:] = corr[:n]
    out[:n-1] = corr[N-n+1:][::-1]
    return out

def compute_c5_fft(h):
    n = len(h)

    corr = fft_correlate(h, n)

    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]

    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)

    obj = dx * (max_c + np.log(s) / tau)
```

```python
def fft_correlate(a, v):
    na, nv = len(a), len(v)
    N = 1
    while N < na + nv - 1:
        N <<= 1
    fa = np.fft.rfft(a, N)
    fv = np.fft.rfft(v, N)
    # This gives circular cross-correlation: r[m] = sum_n a[n+m] * v[n] (with wrap)
    r = np.fft.irfft(np.conj(fa) * fv, N)
    # r[m] for m=0,...,na-1: positive lags (a advanced)
    # r[N-m] for m=1,...,nv-1: negative lags (a delayed)
    # np.correlate(a, v, mode='full') has:
    #   index 0 = lag -(nv-1), index 1 = lag -(nv-2), ..., index nv-1 = lag 0, ..., index na+nv-2 = lag na-1
    out = np.empty(na + nv - 1)
    out[nv-1:] = r[:na]  # lags 0 to na-1
    if nv > 1:
        out[:nv-1] = r[N-nv+1:][::-1]  # lags -(nv-1) to -1
    return out
```
