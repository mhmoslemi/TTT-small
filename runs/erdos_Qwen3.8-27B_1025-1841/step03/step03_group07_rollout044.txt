        corr = np.correlate(y, 1.0 - y, mode='full')
        max_c = corr.max()
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        shifted = np.clip(cur_tau * (top_vals - max_c), -100.0, 0.0)
        exp_c = np.exp(shifted)
        s_exp = exp_c.sum()
        p = exp_c / s_exp

        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = p[ii]
            k = top_idx[ii]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - y[j0 + sh:j1 + sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * y[j0b + sh:j1b + sh]
        grad *= dx

        h_new = y - cur_lr * grad
        h_new = project_to_feasible(h_new, n)

        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 2000:
                lr *= 0.65
                cur_lr *= 0.65
                stall = 0
                if lr < 1e-16:
                    break

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def coord_descent(h, n, time_limit, top_k=4):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    h = h.copy()
    no_improve = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argsort(corr)[-top_k:]
        improved = False

        for k in max_idx:
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 >= j1:
                continue
            contrib = h[j0:j1] * (1.0 - h[j0 + sh:j1 + sh])
            top_local = np.argsort(contrib)[-min(5, j1 - j0):][::-1]
            for ti in top_local:
                mi = j0 + ti
                partner = mi + sh
                if h[mi] > 0.005:
                    h_try = h.copy()
                    h_try[mi] -= min(0.015, h[mi])
                    h_try = project_to_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h.copy()
                        improved = True
                        break
                if 0 <= partner < n and h[partner] < 0.995:
                    h_try = h.copy()
                    h_try[partner] += min(0.015, 1.0 - h[partner])
                    h_try = project_to_feasible(h_try, n)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-13:
                        h = h_try
                        best_c5 = c5_try
                        best_h = h.copy()
                        improved = True
                        break
            if improved:
                break

        if not improved:
            for _ in range(30):
                idx = np.random.randint(0, n)
                d = np.random.uniform(-0.008, 0.008)
                h_try = h.copy()
                h_try[idx] = np.clip(h[idx] + d, 0, 1)
                h_try = project_to_feasible(h_try, n)
                c5_try = compute_c5(h_try)
                if c5_try < best_c5 - 1e-13:
                    h = h_try
                    best_c5 = c5_try
                    best_h = h.copy()
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve > 80:
                break
        else:
            no_improve = 0

    return best_h, best_c5


def bang_bang_round(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    new_h = np.zeros(n)
    new_h[idx[:target]] = 1.0
    return new_h


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase A: Optimize at n=4000 (refine parent)
    n = n_init
    h = h_init.copy()
    h = project_to_feasible(h, n)

    rem = get_remaining(start_time, budget_s)
    if rem > 200:
        tl = min(160, rem * 0.14)
        h, c5 = nesterov_optimize(h, n, 500.0, 8.0, tl, lr=0.25, top_k=12)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    rem = get_remaining(start_time, budget_s)
    if rem > 120:
        tl = min(100, rem * 0.12)
        h, c5 = nesterov_optimize(h, n, 30.0, 3.0, tl, lr=0.04, top_k=6)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase B: Upsample to n=8000 and optimize
    n2 = 8000
    rem = get_remaining(start_time, budget_s)
    if rem > 350:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)

        # Stage 1: broad
        tl = min(130, rem * 0.14)
        h2, c5 = nesterov_optimize(h2, n2, 400.0, 20.0, tl, lr=0.22, top_k=14)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

        # Stage 2: medium
        rem = get_remaining(start_time, budget_s)
        tl = min(110, rem * 0.13)
        h2, c5 = nesterov_optimize(h2, n2, 50.0, 5.0, tl, lr=0.06, top_k=10)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

        # Stage 3: fine
        rem = get_remaining(start_time, budget_s)
        tl = min(90, rem * 0.12)
        h2, c5 = nesterov_optimize(h2, n2, 10.0, 2.0, tl, lr=0.02, top_k=6)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase C: Upsample to n=12000 for final refinement
    n3 = 12000
    rem = get_remaining(start_time, budget_s)
    if rem > 250:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)

        tl = min(100, rem * 0.13)
        h3, c5 = nesterov_optimize(h3, n3, 200.0, 15.0, tl, lr=0.18, top_k=12)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

        rem = get_remaining(start_time, budget_s)
        tl = min(90, rem * 0.12)
        h3, c5 = nesterov_optimize(h3, n3, 20.0, 3.0, tl, lr=0.03, top_k=8)
        if c5 < best_c5:
            best_c5, best_h = c5, h3.copy()

    # Phase D: Coordinate descent
    rem = get_remaining(start_time, budget_s)
    if rem > 60:
        tl = min(55, rem * 0.1)
        h, c5 = coord_descent(best_h, len(best_h), tl, top_k=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase E: Bang-bang rounding
    rem = get_remaining(start_time, budget_s)
    if rem > 10:
        h_bb = bang_bang_round(best_h, len(best_h))
        c5_bb = compute_c5(h_bb)
        if c5_bb < best_c5:
            best_c5, best_h = c5_bb, h_bb.copy()

    # Phase F: Multi-start perturbations
    rem = get_remaining(start_time, budget_s)
    for pert in [0.004, 0.002, 0.008, 0.0015, 0.006]:
        rem = get_remaining(start_time, budget_s)
        if rem < 50:
            break
        h_p = best_h.copy() + np.random.randn(len(best_h)) * pert
        h_p = project_to_feasible(h_p, len(best_h))
        tl = min(40, rem * 0.15)
        h_p, c5_p = nesterov_optimize(h_p, len(h_p), 30.0, 4.0, tl, lr=0.05, top_k=8)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase G: Final coordinate descent
    rem = get_remaining(start_time, budget_s)
    if rem > 30:
        tl = min(28, rem * 0.4)
        h, c5 = coord_descent(best_h, len(best_h), tl, top_k=4)
        if c5 < best_c5:
            best_c5, best_h = c5, h.copy()

    # Phase H: Final bang-bang check
    h_bb2 = bang_bang_round(best_h, len(best_h))
    c5_bb2 = compute_c5(h_bb2)
    if c5_bb2 < best_c5:
        best_c5, best_h = c5_bb2, h_bb2.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
