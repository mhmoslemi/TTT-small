import numpy as np
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    H = fft(h)
    autocorr = np.real(ifft(H * np.conj(H)))
    c = np.sum(h) - autocorr
    return float(np.max(c) * dx)

def water_fill(h, target_sum):
    lo, hi = -1.5, 1.5
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return np.interp(new_x, old_x, h, period=2.0)

def compute_grad_fft(h, p, p_flipped):
    n = len(h)
    H_h = fft(h)
    H_p = fft(p)
    H_pf = fft(p_flipped)
    term1 = np.real(ifft(H_pf * H_h))
    term2 = np.real(ifft(H_p * H_h))
    return 1.0 - term1 - term2

def fft_grad_descent(h, n, max_iters, T_init, lr_init, momentum, start_time, time_budget):
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    T = T_init
    no_improve = 0
    patience = 150
    global_start = start_time
    global_budget = time_budget

    for it in range(max_iters):
        if global_budget - (time.time() - global_start) < 15:
            break

        H = fft(h)
        autocorr = np.real(ifft(H * np.conj(H)))
        c = np.sum(h) - autocorr

        c_max = np.max(c)
        exp_c = np.exp(T * (c - c_max))
        p = exp_c / np.sum(exp_c)

        p_flipped = np.zeros(n)
        p_flipped[0] = p[0]
        if n > 1:
            p_flipped[1:] = p[:0:-1]

        grad = compute_grad_fft(h, p, p_flipped)

        v = momentum * v + (1.0 - momentum) * grad

        # Backtracking line search
        step = lr
        best_trial = None
        best_obj = compute_c5(h)
        for bt in range(12):
            h_trial = h - step * v
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = water_fill(h_trial, n / 2.0)
            obj = compute_c5(h_trial)
            if obj < best_obj - 1e-14:
                best_trial = h_trial
                break
            step *= 0.5
            if step < 1e-8:
                break
        if best_trial is not None:
            h = best_trial
            lr = min(step * 2.0, lr_init * 2.0)
            obj = best_obj if best_trial is None else compute_c5(h)
            # Recompute obj properly
            obj = compute_c5(h)
            if obj < best_c5 - 1e-12:
                best_c5 = obj
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
            lr = max(lr * 0.8, 1e-6)

        if no_improve > patience:
            # Perturb and restart momentum
            h = best_h + 0.005 * np.random.randn(n)
            h = water_fill(h, n / 2.0)
            v = np.zeros(n)
            lr = lr_init
            T = max(T * 0.7, 20.0)
            no_improve = 0

        if it % 400 == 399:
            T = min(T * 1.3, 8000.0)

    return best_h, best_c5
