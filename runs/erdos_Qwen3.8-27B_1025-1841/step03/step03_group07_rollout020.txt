import numpy as np
import time

def project_to_feasible(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2} using bisection on the threshold."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def soft_max_and_grad(corr, tau, top_k):
    """Compute soft-max value and gradient weights for top-k shifts."""
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_corr = corr[top_idx]
    max_c = top_corr.max()
    shifted = tau * (top_corr - max_c)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = max_c + np.log(s) / tau
    return obj, top_idx, p

def grad_for_shifts(h, shifts, weights, n):
    """Compute gradient contribution from weighted shifts."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(shifts)):
        k = shifts[i]
        w = weights[i]
        if w < 1e-10:
            continue
        sh = k - n + 1
        # h[i+k] * (1-h[i]) contribution
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += w * (1.0 - h[idx - sh])
        # h[i] * (1-h[i+k]) contribution (negative)
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= w * h[idx + sh]
    return grad * dx

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def grad_correlate(h, shift, n):
    """Gradient of correlation at a specific shift."""
    dx = 2.0 / n
    sh = shift - n + 1  # convert to np.correlate indexing
    grad = np.zeros(n)
    j0 = max(0, sh)
    j1 = min(n, sh + n)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[j0:j1] += (1.0 - h[idx - sh])
    j0b = max(0, -sh)
    j1b = min(n, n - sh)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        grad[j0b:j1b] -= h[idx + sh]
    return grad * dx
