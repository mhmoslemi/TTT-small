import numpy as np
import time

def fft_correlate(a, v):
    n = len(a)
    m = len(v)
    N = 1
    while N < n + m - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    V = np.fft.rfft(v, N)
    result = np.fft.irfft(A * np.conj(V), N)
    shifts = np.arange(n + m - 1) - (m - 1)
    return result[shifts % N]

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(g - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5*(lo+hi), 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, target_n, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h, 1.0 - h)
    return corr.max() * (2.0 / n)

def compute_grad(h, n, weights, top_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        pk = weights[ii]
        sh = int(top_idx[ii]) - (n - 1)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    return grad * dx

def nesterov_phase(h, n, tau, time_limit, lr, top_k):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    
    while time.time() - start < time_limit:
        mu = min((t - 1.0) / (t + 2.0), 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        corr = fft_correlate(y, 1.0 - y)
        max_c = corr.max()
        
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        
        shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        
        grad = compute_grad(y, n, p, top_idx)
        h_new = project_to_feasible(y - lr * grad, n)
        
        t += 1
        if t % 80 == 1:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 80
            if stall > 4000:
                lr *= 0.65
                stall = 0
                if lr < 1e-18:
                    break

        h_prev = h

    if c5_final < best_c5:

        best_h = h.copy()
    return best_h, best_c5

    return best_h, best_c5

import numpy as np
import time


def fft_correlate(a, v):
    n = len(a)
    m = len(v)
    N = 1
    while N < n + m - 1:
        N <<= 1
    A = np.fft.rfft(a, N)
    V = np.fft.rfft(v, N)
    result = np.fft.irfft(A * np.conj(V), N)
    shifts = np.arange(n + m - 1) - (m - 1)
    return result[shifts % N]


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        s = np.clip(g - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5 * (lo + hi), 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0, 2, n, endpoint=False)
    x_new = np.linspace(0, 2, target_n, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h, 1.0 - h)
    return corr.max() * (2.0 / n)


def compute_grad(h, n, weights, top_idx):
    dx = 2.0 / n
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        pk = weights[ii]
        sh = int(top_idx[ii]) - (n - 1)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    return grad * dx


def nesterov_phase(h, n, tau, time_limit, lr, top_k):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    it = 0

    while time.time() - start < time_limit:
        it += 1
        mu = min((t - 1.0) / (t + 2.0), 0.9)
        y = h + mu * (h - h_prev)
        np.clip(y, 0.0, 1.0, out=y)

        corr = fft_correlate(y, 1.0 - y)
        max_c = corr.max()

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]

        shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s

        grad = compute_grad(y, n, p, top_idx)
        h_new = project_to_feasible(y - lr * grad, n)

        if it % 60 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 60
            if stall > 3500:
                lr *= 0.65
                stall = 0
                if lr < 1e-18:
                    break

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_phase(h, n, time_limit, lr, top_k):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    while time.time() - start < time_limit:
        corr = fft_correlate(h, 1.0 - h)
        c5_now = corr.max() * (2.0 / n)

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            lr *= 0.8
            v *= 0.5
            stall = 0
            if lr < 1e-20:
                break

        max_c = corr.max()
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        w = np.exp(top_vals - max_c)
        w /= w.sum()

        grad = compute_grad(h, n, w, top_idx)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))
