import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def project_box_sum(y, target):
    n = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        if np.sum(mask) > 0:
            x[mask] = np.clip(x[mask] + diff / np.sum(mask), 0.0, 1.0)
    return x

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def fista_smooth(h_init, n, T, max_iter, time_limit_s):
    """FISTA minimization of smooth approximation to max correlation."""
    start = time.time()
    h = h_init.astype(float).copy()
    h_prev = h.copy()
    t = 1.0
    step = 1.0 / n  # initial guess
    target = n / 2.0
    dx = 2.0 / n
    
    for it in range(max_iter):
        if time.time() - start > time_limit_s:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        weights = np.exp(T * shifted)
        wsum = np.sum(weights)
        weights /= wsum
        
        # Gradient computation via convolution
        conv = np.convolve(weights[::-1], 1.0 - h, mode='full')
        grad = dx * conv[n-1:n-1+n]
        
        # Nesterov momentum
        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        if it == 0:
            y = h.copy()
        else:
            y = h + (t_prev - 1.0) / t * (h - h_prev)
        
        # Backtracking line search (simplified: accept if step is small enough)
        h_new = project_box_sum(y - step * grad, target)
        h_prev = h
        h = h_new
        
        # Adaptive step size (increase slowly)
        if it % 50 == 0 and it > 0:
            step = min(step * 1.1, 10.0 / n)
    
    return h

def coordinate_descent(h, n, max_rounds, time_limit_s):
    """Polish with coordinate descent on the top-contributing coordinates."""
    start = time.time()
    target = n / 2.0
    best_c5 = compute_c5(h)
    hm = h.copy()
    
    deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit_s:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit_s:
                break
            old = hm[j]
            for d in deltas:
                nv = old + d
                if nv < 0 or nv > 1:
                    continue
                trial = hm.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    hm = trial
                    old = nv
                    improved = True
        if not improved:
            break
    return hm, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Load initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except:
        h0 = np.tile([1.0, 0.0], 320)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: FISTA at moderate resolution
    for n_pts in [1024, 2048]:
        if time_left() < 60:
            break
        h_curr = upsample(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
        
        # Multiple temperature levels
        T_values = [30.0, 80.0, 200.0, 500.0]
        for T in T_values:
            if time_left() < 20:
                break
            h_opt = fista_smooth(h_curr, n_pts, T, max_iter=2000, time_limit_s=min(120, time_left()-10))
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_curr = h_opt.copy()
            else:
                h_curr = h_opt.copy()  # Still use as starting point
        
        # Random restart
        if time_left() > 60:
            h_rand = 0.8 * h_curr + 0.2 * rng.rand(n_pts)
            h_rand = make_feasible(h_rand, n_pts)
            h_opt = fista_smooth(h_rand, n_pts, 100.0, max_iter=1000, time_limit_s=min(60, time_left()-10))
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    
    # Phase 2: Coordinate descent polish
    n_cd = min(len(best_h), 1024)
    h_cd = upsample(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    h_cd, c5_cd = coordinate_descent(h_cd, n_cd, max_rounds=5, time_limit_s=min(120, time_left()-20))
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = upsample(h_cd, len(best_h)) if len(best_h) > n_cd else h_cd.copy()
    
    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

import numpy as np
from scipy.optimize import minimize
import time

_opt_n = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def project_box_sum(y, target):
    n = len(y)
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = x < 1.0 - 1e-15
        else:
            mask = x > 1e-15
        if np.sum(mask) > 0:
            x[mask] = np.clip(x[mask] + diff / np.sum(mask), 0.0, 1.0)
    return x

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _smooth_obj_and_grad(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(T * shifted)
    wsum = np.sum(weights)
    obj = dx * (mc + np.log(wsum) / T)
    weights /= wsum
    conv = np.convolve(weights[::-1], 1.0 - h, mode='full')
    grad = dx * conv[n - 1:n - 1 + n]
    return obj, grad

def fista_optimize(h_init, n, T, max_iter, time_limit):

    h = h_init.astype(float).copy()
    h_prev = h.copy()
    t = 1.0
    target = n / 2.0
    step = 2.0 / n
    best_h = h.copy()
    best_obj = _smooth_obj_and_grad(h, T, n)[0]
    stall = 0

    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        obj, grad = _smooth_obj_and_grad(h, T, n)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                step *= 0.8
                stall = 0
                if step < 1e-10:
                    break

        t_prev = t
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_prev * t_prev))
        if it == 0:
            y = h.copy()

        h_prev = h
        h = h_new
        if it > 0 and it % 100 == 0:

    return best_h

def subgrad_optimize(h_init, n, max_iter, time_limit):

def coord_descent(h, n, max_rounds, time_limit):
    start_t = time.time()
    best_c5 = compute_c5(h)
    hm = h.copy()
    deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005, 0.05, -0.05]
    for rnd in range(max_rounds):
        if time.time() - start_t > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_t > time_limit:
                break
            old = hm[j]
            for d in deltas:
                nv = old + d
                if nv < -0.01 or nv > 1.01:
                    continue
                trial = hm.copy()
                trial[j] = np.clip(nv, 0.0, 1.0)

                    best_c5 = c
                    hm = trial
                    old = nv
                    improved = True
            if abs(old - hm[j]) < 1e-12 and improved:

        if not improved:

def slsqp_pass(h_start, n, T, max_iter):
    global _opt_n
    _opt_n = n
    dx = 2.0 / n
    h0 = h_start.copy()

    def obj_fn(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return dx * (mc + np.log(np.sum(np.exp(T * shifted))) / T)

    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        weights = np.exp(T * shifted)
        weights /= np.sum(weights)
        conv = np.convolve(weights[::-1], 1.0 - h, mode='full')
        return dx * conv[n - 1:n - 1 + n]

    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:

                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-15})
