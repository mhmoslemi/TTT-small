                    hx = slsqp(hr, 1.0, 350)
                    cx = compute_c5(hx)
                    if cx < cr:
                        hr, cr = hx, cx
                if cr < bc:
                    bc, bh = cr, hr.copy()
                    print(f"  D: C5={bc:.6f}", flush=True)

    # Phase 7: Perturbation restarts (annealing-style)
    np_cur = len(bh)
    if rem() > 80:
        for trial in range(8):
            if rem() < 15:
                break
            sc = 0.015 + 0.01 * trial
            hp = project_c(bh + rng.randn(np_cur) * sc, np_cur)
            hr, cr = fista(hp, 2000, 8.0, 0.03, 0.05, 50, 100)
            if cr < bc and rem() > 12:
                if np_cur <= 1024:
                    hx = slsqp(hr, 0.5, 350)
                    cx = compute_c5(hx)
                    if cx < cr:
                        hr, cr = hx, cx
                if cr < bc:
                    bc, bh = cr, hr.copy()
                    print(f"  P{trial}: C5={bc:.6f}", flush=True)

    # Phase 8: Final ultra-fine polish
    np_cur = len(bh)
    if rem() > 20:
        hf, cf = fista(bh, 3000, 2.0, 0.01, 0.008, 30, 130)
        if cf < bc:
            bc, bh = cf, hf.copy()

    if np_cur <= 1024 and rem() > 15:
        bh2 = slsqp(bh, 0.15, 800)
        bc = compute_c5(bh2)
        bh = bh2

    if rem() > 10:
        hf, cf = fista(bh, 1500, 1.0, 0.005, 0.004, 30, 130)
        if cf < bc:
            bc, bh = cf, hf.copy()

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
