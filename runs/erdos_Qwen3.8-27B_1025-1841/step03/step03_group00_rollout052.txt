def compute_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def compute_corr(h):

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_hill_climb(h_bin, n, rng, n_iter):
    """Single-swap hill climbing"""
    h = h_bin.copy()
    curr = compute_c5(h)
    best = curr
    best_h = h.copy()
    for _ in range(n_iter):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        oi = ones[rng.randint(len(ones))]
        zi = zeros[rng.randint(len(zeros))]
        h[oi] = 0.0
        h[zi] = 1.0
        new_c = compute_c5(h)
        if new_c < curr:
            curr = new_c
            if curr < best:
                best = curr
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[zi] = 0.0
    return best_h, best

def binary_2swap_search(h_bin, n, rng, n_iter):
    """Two-swap hill climbing (4-opt)"""
    h = h_bin.copy()
    curr = compute_c5(h)
    best = curr
    best_h = h.copy()
    for _ in range(n_iter):
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = ones[rng.randint(len(ones))]
        j1 = zeros[rng.randint(len(zeros))]
        j2 = zeros[rng.randint(len(zeros))]
        if len(set([i1, i2, j1, j2])) < 4:
            continue
        h[i1] = 0; h[i2] = 0; h[j1] = 1; h[j2] = 1
        new_c = compute_c5(h)
        if new_c < curr:
            curr = new_c
            if curr < best:
                best = curr
                best_h = h.copy()
        else:
            h[i1] = 1; h[i2] = 1; h[j1] = 0; h[j2] = 0
    return best_h, best

def block_rotate_search(h_bin, n, rng, n_iter, block_sizes):
    """Block rotation hill climbing"""
    h = h_bin.copy()
    curr = compute_c5(h)
    best = curr
    best_h = h.copy()
    for _ in range(n_iter):
        blen = block_sizes[rng.randint(len(block_sizes))]
        if blen >= n:
            continue
        start = rng.randint(0, n - blen + 1)
        saved = h[start:start+blen].copy()
        # Rotate left by 1
        h[start:start+blen] = np.concatenate([saved[1:], saved[:1]])
        new_c = compute_c5(h)
        if new_c < curr:
            curr = new_c
            if curr < best:
                best = curr
                best_h = h.copy()
        else:
            h[start:start+blen] = saved
            # Try rotate right
            h[start:start+blen] = np.concatenate([saved[-1:], saved[:-1]])
            new_c = compute_c5(h)
            if new_c < curr:
                curr = new_c
                if curr < best:
                    best = curr
                    best_h = h.copy()
            else:
                h[start:start+blen] = saved
    return best_h, best
