    """Project h onto {x: sum(x) = target, 0 ≤ x ≤ 1}."""
    # Water filling
    h = h.copy()

        if abs(s - target) < 1e-14:

        diff = target - s
        if diff > 0:
            mask = h < 1.0

            mask = h > 0.0
        if not np.any(mask):

        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0, 1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):

        if abs(s - target) < 1e-14:

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:

        shifted = corr - mc
        m2 = np.max(shifted)
        if m2 > 50.0 / T:
            val = mc + m2 + np.log(np.sum(np.exp(T * (shifted - m2)))) / T
        else:
            val = mc + np.log(np.sum(np.exp(T * shifted))) / T
        return val * dx
    def con_eq(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)

    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
