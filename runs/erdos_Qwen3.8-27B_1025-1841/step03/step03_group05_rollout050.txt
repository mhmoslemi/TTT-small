def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def smooth_grad(h, T, top_k=300):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx_max = int(np.argmax(corr))
        return subgrad_one(h, idx_max - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * subgrad_one(h, i - (n - 1))
    return g * dx

        tk = tn

            if c < bc:

    c = compute_c5(x)
    if c < bc:

            bc, bh = ca, ha.copy()
        print(f"AS1: C5={bc:.8f} t={time.time()-t0:.0f}", flush=True)

    # Phase 5: SLSQP after active set
    for TV in [8.0, 3.0, 1.0]:
        if remaining() < 50:
            break
        bh_new = slsqp(bh, TV, 350)
        cx = compute_c5(bh_new)
        if cx < bc:
            bc, bh = cx, bh_new
        print(f"SL2({TV:.0f}): C5={bc:.8f} t={time.time()-t0:.0f}", flush=True)

    # Phase 6: More active-set (tighter)
    if remaining() > 80:
        ha2, ca2 = active_set_refine(bh, 3, 300, 0.02)
        if ca2 < bc:
            bc, bh = ca2, ha2.copy()
        print(f"AS2: C5={bc:.8f} t={time.time()-t0:.0f}", flush=True)

    # Phase 7: Final FISTA ultra-fine
    if remaining() > 80:
        hf, cf = fista(bh, 3000, 2.0, 0.01, 0.008, 0.001, 25, 100)
        if cf < bc:
            bc, bh = cf, hf.copy()
        print(f"FF: C5={bc:.8f} t={time.time()-t0:.0f}", flush=True)

    # Phase 8: Final SLSQP
    if remaining() > 40:
        bh_new = slsqp(bh, 1.0, 400)
        cx = compute_c5(bh_new)
        if cx < bc:
            bc, bh = cx, bh_new

    # Phase 9: Final active set
    if remaining() > 25:
        ha3, ca3 = active_set_refine(bh, 3, 150, 0.01)
        if ca3 < bc:
            bc, bh = ca3, ha3.copy()

    bp = project_c(bh, len(bh))
    final_c = compute_c5(bp)
    if final_c < bc:
        bc = final_c
        bh = bp
    print(f"DONE: n={len(bh)} C5={bc:.8f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
