    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    stall = 0
    
    while time.time() - start < time_limit:
        obj, grad = soft_max_obj_grad(h, n, tau, top_k)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

- A "move" is: pick one position with h=1 and one position with h=0, swap them.
- The effect on the correlation: we can compute the change in each corr[k].
- We want to minimize the maximum corr[k].

For a swap of positions a (h[a]=1→0) and b (h[b]=0→1):
- corr[k] changes by:
  - Position a: was contributing h[a]*(1-h[a+s]) = 1*(1-h[a+s]) = 1-h[a+s] to corr[k] (for shift s=k-n+1). Now contributes 0.
  - Position b: was contributing 0. Now contributes 1*(1-h[b+s]) = 1-h[b+s].

So the net change in corr[k] for shift s = k-n+1:
Δcorr[k] = -(1-h[a+s]) + (1-h[b+s]) - h[a-s] + h[b-s]
= h[b+s] - h[a+s] - h[a-s] + h[b-s]
= (h[b+s] - h[a+s]) + (h[b-s] - h[a-s])

(with boundary handling: h[i] = 0 for i < 0 or i ≥ n)

        if pk < 1e-10:

        k = top_idx[idx]
        s = k - n + 1
        # d(corr[k])/d(h[i]) = (1-h[i+s]) - h[i-s]
        # First part: (1-h[i+s]) for i in [max(0,-s), min(n, n-s))
        j0 = max(0, -s)
        j1 = min(n, n - s)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0+s:j1+s])
        # Second part: -h[i-s] for i in [max(0,s), min(n, n+s))
        j0b = max(0, s)
        j1b = min(n, n + s)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b-s:j1b-s]
    
    return obj, grad

def adam_opt(h, n, tau, time_limit, lr=0.01, top_k=50):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    t = 0
    stall = 0
    
    while time.time() - start < time_limit:
        obj, grad = smooth_grad(h, n, tau, top_k)
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, n)
        
        if t % 15 == 0:
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 15
            if stall > 400:
                lr *= 0.7
                m *= 0.3
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
        
        h = h_new
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def binary_snap(h, n):
    """Round h to binary and fix the constraint."""
    hb = (h > 0.5).astype(float)
    # Count ones
    n_ones = int(hb.sum())
    target = n // 2
    if n_ones == target:
        return hb
    if n_ones > target:
        # Need to turn off (n_ones - target) positions
        # Turn off the smallest values of h (least confident 1s)
        excess = n_ones - target
        idx = np.where(hb == 1)[0]
        # Sort by h value ascending (least confident first)
        order = idx[np.argsort(h[idx])]
        turn_off = order[:excess]
        hb[turn_off] = 0
    else:
        # Need to turn on (target - n_ones) positions
        deficit = target - n_ones
        idx = np.where(hb == 0)[0]
        order = idx[np.argsort(h[idx], descending=True)]
        turn_on = order[:deficit]
        hb[turn_on] = 1
    return hb
