def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

    if tot < 1e-300:

import cvxpy as cp

def lp_refine(h, n_active=50, max_iter=50, tol=1e-8):
    n = len(h)
    dx = 2.0 / n
    h_current = h.copy()
    for it in range(max_iter):
        corr = np.correlate(h_current, 1.0 - h_current, mode='full')
        mc = np.max(corr)
        # Find active lags (within some tolerance of the max)
        active = np.where(corr > mc - 1e-3 * n)[0]
        active = active[:n_active]  # Limit to top n_active
        
        # Linearize: cross[k] ≈ cross[k](h0) + grad_k . (h - h0)
        # grad_k[i] = subgrad of cross[k] w.r.t. h[i]
        # For cross[k] = sum_j h[j+k']*(1-h[j]):
        #   grad[i] = (1-h[i-k']) if i is in the "first" part
        #            + (-h[i+k']) if i is in the "second" part
        # where k' = k - (n-1) is the actual lag
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [cp.sum(h_var) == n / 2.0, h_var >= 0, h_var <= 1]
        
        for idx in active:
            lag = idx - (n - 1)
            g = _subgrad_one(h_current, lag)
            linearized = corr[idx] + g @ (h_var - h_current)
            constraints.append(linearized <= t_var)
        
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.SCS, verbose=False)
            except:
                break
        
        if h_var.value is None:
            break
        
        h_new = np.clip(h_var.value, 0, 1)
        h_new = project_c(h_new, n)
        
        c_new = compute_c5(h_new)
        c_old = compute_c5(h_current)
        
        if c_new < c_old - tol:
            h_current = h_new
        else:
            # Try a step back
            for alpha in [0.5, 0.25, 0.1]:
                h_trial = h_current + alpha * (h_new - h_current)
                h_trial = project_c(h_trial, n)
                if compute_c5(h_trial) < c_old:
                    h_current = h_trial
                    break
            else:
                break
    
    return h_current

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_adv(h, n_iter, T0, T1, lr0, lr1, top_k0, top_k1, check=20, restart_every=500):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    last_restart = 0
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr0 * (lr1 / lr0) ** f
        tk_actual = int(top_k0 * (top_k1 / top_k0) ** f)
        tk_actual = max(5, tk_actual)
        
        # Compute smoothed gradient
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            g = _subgrad_one(z, int(np.argmax(corr)) - (n - 1)) * (2.0/n)
        else:
            w = exps / tot
            idx = np.argsort(w)[::-1][:tk_actual]
            g = np.zeros(n)
            for i in idx:
                wi = w[i]
                if wi < 1e-12:
                    break
                g += wi * _subgrad_one(z, i - (n - 1))
            g *= (2.0 / n)
        
        # FISTA update
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        
        # Periodic restart (momentum reset)
        if (it + 1) % restart_every == 0:
            c_now = compute_c5(x)
            if c_now > bc:
                z = bh.copy()
                tk = 1.0
                x = bh.copy()
            last_restart = it
        
        # Check improvement
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def lag_refine(h, n_lags=10, n_iter=200, step_size=0.01):
    n = len(h)
    dx = 2.0 / n
    h_cur = h.copy()
    bc = compute_c5(h_cur)
    
    for it in range(n_iter):
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        mc = np.max(corr)
        # Get top lags
        top_idx = np.argsort(corr)[::-1][:n_lags]
        
        for idx in top_idx:
            lag = idx - (n - 1)
            g = _subgrad_one(h_cur, lag)
            # Project gradient to sum constraint
            g_mean = np.mean(g)
            g_proj = g - g_mean
            # Normalize
            norm = np.linalg.norm(g_proj)
            if norm < 1e-10:
                continue
            g_proj /= norm
            # Take step
            for alpha in [step_size, step_size*0.5, step_size*0.25]:
                h_trial = project_c(h_cur - alpha * g_proj, n)
                c_trial = compute_c5(h_trial)
                if c_trial < bc:
                    h_cur = h_trial
                    bc = c_trial
                    break
        
        # Early stopping
        if (it + 1) % 20 == 0:
            c_now = compute_c5(h_cur)
            if abs(c_now - bc) < 1e-10:
                break
    
    return h_cur, bc

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_run(h, n_iter, T0, T1, lr0, lr1, k0, k1, check=25, restart=400):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr0 * (lr1 / lr0) ** f
        kt = max(5, int(k0 * (k1 / k0) ** f))
        
        dx = 2.0 / n
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            g = _subgrad_one(z, int(np.argmax(corr)) - (n - 1)) * dx
        else:

            idx = np.argsort(w)[::-1][:kt]
            g = np.zeros(n)

        x = xn

def slsqp_polish(h, T=5.0, max_iter=400):
    n = len(h)
    dx = 2.0 / n
    
    def obj(v):
        T_local = T
        c = np.correlate(v, 1.0 - v, mode='full')
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(T_local * (c - mc)))) / T_local) * dx
    
    def grad(v):
        c = np.correlate(v, 1.0 - v, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        w = exps / np.sum(exps)
        idx = np.argsort(w)[::-1][:50]
        g = np.zeros(n)
        for i in idx:
            if w[i] < 1e-12:
                break
            g += w[i] * _subgrad_one(v, i - (n - 1))
        return g * dx
    
    def con_fun(v):
        return np.sum(v) - n / 2.0
    
    def con_jac(v):
        return np.ones(n)
    
    try:
        r = minimize(obj, h, jac=grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                     options={'maxiter': max_iter, 'ftol': 1e-12})
        if r.success or r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except:
        pass
    return h

def upsample(h, nn):
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].copy()

    # Alternating

    starts.append(project_c(al, n))
    return starts
