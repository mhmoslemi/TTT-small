import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def compute_c5_fft(h, N_fft=None):
    n = len(h)
    if N_fft is None:
        N_fft = 1
        while N_fft < 2*n - 1:
            N_fft *= 2
    dh = 1.0 - h
    # Cross-correlation via FFT
    # np.correlate(h, dh, 'full')[k] = sum_i h[i+k-n+1] * dh[i]
    # = sum_i h[i+j] * dh[i] where j = k-n+1 ranges from -(n-1) to n-1
    # FFT approach: c = ifft(fft(h) * conj(fft(dh)))
    # c[j] = sum_i h[i] * dh[(i-j) mod N]
    # For zero-padded (N >= 2n-1): c[j] = sum_i h[i] * dh[i-j] for valid i, i-j
    # This gives lag j correlation: sum_i h[i] * dh[i-j]
    # = sum_i h[i+j] * dh[i] (reindexing)
    # So c[j] = corr at lag j (for j=0,...,n-1)
    # And c[N-j] = corr at lag -j (for j=1,...,n-1)
    
    fh = rfft(h, N_fft)
    fdh = rfft(dh, N_fft)
    c = irfft(fh * np.conj(fdh), N_fft)
    # c[0] to c[n-1]: lags 0, 1, ..., n-1
    # c[N_fft-(n-1)] to c[N_fft-1]: lags -(n-1), ..., -1
    full = np.concatenate([c[N_fft-(n-1):], c[:n]])
    return np.max(full) * (2.0 / n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    # Adjust to get exactly target ones
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

    curr_c5 = compute_c5_fft(b, N_fft)

    start = time.time()
    
    ones_idx = np.where(b == 1)[0]
    zeros_idx = np.where(b == 0)[0]

        T = T0 * (T1 / T0) ** frac
        
        # Random block move

        chosen_zeros = rng.choice(zeros_in, size=k, replace=False)

import numpy as np
from scipy.optimize import minimize
from scipy.fft import rfft, irfft
import time

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_fft(h, N_fft):
    n = len(h)
    dh = 1.0 - h
    fh = rfft(h, N_fft)
    fdh = rfft(dh, N_fft)
    c = irfft(fh * np.conj(fdh), N_fft)
    full = np.concatenate([c[N_fft - (n - 1):], c[:n]])
    return float(np.max(full) * (2.0 / n))

def get_N_fft(n):
    N = 1
    while N < 2 * n - 1:
        N *= 2
    return N

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_block(h_init, n, N_fft, dur, T0, T1, rng, bmax=8):
    b = to_binary(h_init, n)
    curr = compute_c5_fft(b, N_fft)
    best = curr
    best_b = b.copy()
    t0 = time.time()
    while time.time() - t0 < dur:
        frac = min(1.0, (time.time() - t0) / dur)
        T = T0 * (T1 / T0) ** frac
        bs = rng.randint(1, bmax + 1)
        si = rng.randint(0, n - bs)
        ei = si + bs
        seg = b[si:ei]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        k = rng.randint(1, min(len(oi), len(zi)) + 1)
        co = rng.choice(oi, size=k, replace=False)
        cz = rng.choice(zi, size=k, replace=False)
        b[si + co] = 0
        b[si + cz] = 1
        new_c = compute_c5_fft(b, N_fft)
        if new_c <= curr or (T > 1e-14 and rng.random() < np.exp(-(new_c - curr) / T)):
            curr = new_c
            if new_c < best:
                best = new_c
                best_b = b.copy()
        else:
            b[si + co] = 1
            b[si + cz] = 0
    return best_b, best

def sa_single_swap(h_init, n, N_fft, dur, T0, T1, rng):
    b = to_binary(h_init, n)
    curr = compute_c5_fft(b, N_fft)
    best = curr
    best_b = b.copy()
    t0 = time.time()
    half = n // 2
    while time.time() - t0 < dur:
        frac = min(1.0, (time.time() - t0) / dur)
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if b[i] == b[j]:
            continue
        b[i], b[j] = b[j], b[i]
        new_c = compute_c5_fft(b, N_fft)
        if new_c <= curr or (T > 1e-14 and rng.random() < np.exp(-(new_c - curr) / T)):
            curr = new_c
            if new_c < best:
                best = new_c
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
    return best_b, best

def slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, T, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(slsqp_obj, h.copy(), args=(n, T), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x

def coord_descend(h, n, N_fft, max_rounds, t_start, budget):
    best_c = compute_c5_fft(h, N_fft)
    for rnd in range(max_rounds):
        if time.time() - t_start > budget - 5:
            break
        improved = False
        for j in range(n):
            if time.time() - t_start > budget - 3:
                break
            old = h[j]
            best_val = old
            best_c_local = best_c
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5_fft(trial, N_fft)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return h, best_c

def make_periodic(n, period, ones_per_period):
    h = np.zeros(n)
    for i in range(n):
        if (i % period) < ones_per_period:
            h[i] = 1.0
    return make_feasible(h, n)

def make_random_binary(n, rng):
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n // 2]] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None

    if h_init is not None and len(h_init) > 100:
        n0 = len(h_init)
    else:
        n0 = 960
        h_init = make_random_binary(n0, rng)

    h_init = make_feasible(h_init, n0)
    N0 = get_N_fft(n0)
    best_c5 = compute_c5_direct(h_init)
    best_h = h_init.copy()
    n_best = n0

    # Verify FFT matches direct
    c_fft = compute_c5_fft(h_init, N0)
    c_dir = compute_c5_direct(h_init)
    # If mismatch, fall back to direct in SA
    use_fft = abs(c_fft - c_dir) < 1e-6
    if not use_fft:
        # Try the other convention
        pass  # We'll just use direct

    def c5_eval(h, Nf):
        if use_fft:
            return compute_c5_fft(h, Nf)
        return compute_c5_direct(h)

    def sa_eval(h, n, Nf, dur, T0, T1, rm, bmx, single):
        if use_fft:
            if single:
                return sa_single_swap(h, n, Nf, dur, T0, T1, rm)
            return sa_block(h, n, Nf, dur, T0, T1, rm, bmx)
        else:
            # Slow fallback: just use direct compute in a simple SA
            b = to_binary(h, n)
            curr = compute_c5_direct(b)
            best = curr
            best_b = b.copy()
            t0 = time.time()
            while time.time() - t0 < dur:
                frac = min(1.0, (time.time() - t0) / dur)
                T = T0 * (T1 / T0) ** frac
                bs = rm.randint(1, bmx + 1)
                si = rm.randint(0, n - bs)
                ei = si + bs
                seg = b[si:ei]
                oi = np.where(seg == 1)[0]
                zi = np.where(seg == 0)[0]
                if len(oi) == 0 or len(zi) == 0:
                    continue
                k = rm.randint(1, min(len(oi), len(zi)) + 1)
                co = rm.choice(oi, size=k, replace=False)
                cz = rm.choice(zi, size=k, replace=False)
                b[si + co] = 0
                b[si + cz] = 1
                new_c = compute_c5_direct(b)
                if new_c <= curr or (T > 1e-14 and rm.random() < np.exp(-(new_c - curr) / T)):
                    curr = new_c
                    if new_c < best:
                        best = new_c
                        best_b = b.copy()
                else:
                    b[si + co] = 1
                    b[si + cz] = 0
            return best_b, best

    # Phase 1: SA from parent (block moves)
    rem = budget_s - (time.time() - start_time)
    if rem > 80:
        dur = min(250, rem * 0.25)
        b1, c1 = sa_eval(h_init, n0, N0, dur, 0.008, 0.00001, rng, 7, False)
        if c1 < best_c5:
            best_c5 = c1
            best_h = b1.copy()
            n_best = n0

    # Phase 2: SA single-swap refinement
    rem = budget_s - (time.time() - start_time)
    if rem > 80:
        dur = min(150, rem * 0.2)
        b2, c2 = sa_eval(best_h, n_best, get_N_fft(n_best), dur, 0.004, 0.000005, rng, 2, True)
        if c2 < best_c5:
            best_c5 = c2
            best_h = b2.copy()

    # Phase 3: Diverse structured starts
    rem = budget_s - (time.time() - start_time)
    if rem > 200:
        patterns = []
        for p in [5, 7, 9, 11, 13, 17]:
            o1 = p // 2
            o2 = p - o1
            patterns.append((make_periodic(n0, p, o1), make_periodic(n0, p, o2)))
        patterns.append((make_random_binary(n0, rng), make_random_binary(n0, rng)))

        t_each = max(30, min(100, rem / (len(patterns) + 1)))
        for pat in patterns:
            rem = budget_s - (time.time() - start_time)
            if rem < 50:
                break
            for p in [pat[0], pat[1]]:
                rem = budget_s - (time.time() - start_time)
                if rem < 30:
                    break
                dur = min(t_each, rem * 0.4)
                bb, cc = sa_eval(p, n0, N0, dur, 0.01, 0.00005, rng, 5, False)
                if cc < best_c5:
                    best_c5 = cc
                    best_h = bb.copy()
                    n_best = n0

    # Phase 4: Long SA block refinement on best
    rem = budget_s - (time.time() - start_time)
    if rem > 150:
        dur = min(300, rem * 0.4)
        bb, cc = sa_eval(best_h, n_best, get_N_fft(n_best), dur, 0.003, 0.000005, rng, 5, False)
        if cc < best_c5:
            best_c5 = cc
            best_h = bb.copy()

    # Phase 5: SLSQP polish
    rem = budget_s - (time.time() - start_time)
    if rem > 100:
        try:
            x_res = slsqp_polish(best_h, n_best, 100.0, 500)
            c_res = compute_c5_direct(x_res)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = make_feasible(x_res, n_best)
        except Exception:
            pass

    # Phase 6: SA again after polish
    rem = budget_s - (time.time() - start_time)
    if rem > 120:
        dur = min(200, rem * 0.5)
        bb, cc = sa_eval(best_h, n_best, get_N_fft(n_best), dur, 0.002, 0.000002, rng, 3, False)
        if cc < best_c5:
            best_c5 = cc
            best_h = bb.copy()

    # Phase 7: Try higher resolution
    rem = budget_s - (time.time() - start_time)
    if rem > 350:
        n_hi = n_best * 2
        N_hi = get_N_fft(n_hi)
        old_x = np.linspace(0.0, 2.0, n_best)
        new_x = np.linspace(0.0, 2.0, n_hi)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = to_binary(h_up, n_hi)
        dur = min(200, rem * 0.4)
        bb, cc = sa_eval(h_up.astype(float), n_hi, N_hi, dur, 0.006, 0.00001, rng, 5, False)
        if cc < best_c5:
            best_c5 = cc
            best_h = bb.copy()
            n_best = n_hi
        # Polish
        rem = budget_s - (time.time() - start_time)
        if rem > 80 and n_best == n_hi:
            try:
                x_res = slsqp_polish(best_h, n_hi, 100.0, 300)
                c_res = compute_c5_direct(x_res)
                if c_res < best_c5:
                    best_c5 = c_res
                    best_h = make_feasible(x_res, n_hi)
            except Exception:
                pass

    # Phase 8: Try lower resolution
    rem = budget_s - (time.time() - start_time)
    if rem > 250 and n_best > 480:
        n_lo = max(480, n_best // 2)
        N_lo = get_N_fft(n_lo)
        old_x = np.linspace(0.0, 2.0, n_best)
        new_x = np.linspace(0.0, 2.0, n_lo)
        h_lo = np.interp(new_x, old_x, best_h)
        h_lo = to_binary(h_lo, n_lo)
        dur = min(150, rem * 0.4)
        bb, cc = sa_eval(h_lo.astype(float), n_lo, N_lo, dur, 0.006, 0.00001, rng, 5, False)
        if cc < best_c5:
            best_c5 = cc
            best_h = bb.copy()
            n_best = n_lo
        rem = budget_s - (time.time() - start_time)
        if rem > 60 and n_best == n_lo:
            try:
                x_res = slsqp_polish(best_h, n_lo, 100.0, 200)
                c_res = compute_c5_direct(x_res)
                if c_res < best_c5:
                    best_c5 = c_res
                    best_h = make_feasible(x_res, n_lo)
            except Exception:
                pass

    # Phase 9: Coordinate descent
    rem = budget_s - (time.time() - start_time)
    if rem > 30:
        n_pts = len(best_h)
        N_pts = get_N_fft(n_pts)
        dur_cd = min(60, rem * 0.5)
        best_h, best_c5 = coord_descend(best_h.copy(), n_pts, N_pts, 10, time.time() - dur_cd, budget_s)

    # Phase 10: Final SLSQP with high T
    rem = budget_s - (time.time() - start_time)
    if rem > 60:
        n_pts = len(best_h)
        try:
            x_res = slsqp_polish(best_h, n_pts, 300.0, 400)
            c_res = compute_c5_direct(x_res)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 11: Final tiny SA to squeeze out last bit
    rem = budget_s - (time.time() - start_time)
    if rem > 20:
        n_pts = len(best_h)
        N_pts = get_N_fft(n_pts)
        dur = min(15, rem * 0.6)
        bb, cc = sa_eval(best_h, n_pts, N_pts, dur, 0.001, 0.000001, rng, 2, True)
        if cc < best_c5:
            best_c5 = cc
            best_h = bb.copy()

    # Final
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5_direct(best_h)
    return best_h, best_c5, len(best_h)
