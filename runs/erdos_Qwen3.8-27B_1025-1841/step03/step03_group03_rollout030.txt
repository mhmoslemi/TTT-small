        # Let m = n-k: corr(-k) = sum_{m=0}^{n-1-k} h[m]*(1-h[m+k])
        for j in range(0, n - k):
            grad[j] += (1 - h[j + k])
        for j in range(k, n):
            grad[j] -= h[j - k]
    return grad

def lbfgsb_optimize(h0, n, T, top_k, maxiter, penalty):
    def obj(x):
        return soft_max_obj(x, T, top_k) + penalty * (np.sum(x) - n/2.0)**2
    
    def grad(x):
        g = soft_max_grad(x, T, top_k) + 2 * penalty * (np.sum(x) - n/2.0) * np.ones(n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        return res.x
    except Exception:
        return h0

def coordinate_descent(h, n, max_rounds=10, time_limit=None, start_time=None, n_candidates=8):
    dx = 2.0 / n
    best_c = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 20:
                    return h, best_c
            # Try several values for h[j]
            old = h[j]
            candidates = np.linspace(0, 1, n_candidates)
            # Add old value neighborhood
            candidates = np.unique(np.concatenate([candidates, [old, max(0,old-0.1), min(1,old+0.1)]]))
            best_val = old
            best_obj = best_c
            for cv in candidates:
                h[j] = cv
                trial = project_box_sum(h, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            h[j] = old  # restore
            if best_val != old:
                h[j] = best_val
                h = project_box_sum(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def grad_descent(h, n, n_steps=1000, time_limit=None, start_time=None):
    """Subgradient descent on the max correlation"""
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = 0.01
    
    for step in range(n_steps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 30:
                break
        
        h = best_h.copy()
        corr_vals = compute_c5_vec(h)
        # Find all shifts at or near max
        max_val = np.max(corr_vals)
        # Weight: soft attention on top shifts
        T = 50.0
        weights = np.exp(T * corr_vals - T * max_val)
        weights /= np.sum(weights)
        
        # Compute weighted gradient
        grad = np.zeros(n)
        # Only use top shifts for efficiency
        top_k = 10
        top_idx = np.argsort(weights)[::-1][:top_k]
        for idx in top_idx:
            if weights[idx] < 1e-8:
                break
            lag = idx - n + 1
            grad += weights[idx] * _corr_grad(h, lag)
        grad *= 2.0 / n  # dx factor
        
        # Step
        h_new = h - alpha * grad
        h_new = project_box_sum(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
        
        # Decrease alpha
        alpha *= 0.995
        if alpha < 1e-5:
            alpha = 1e-5
    
    return best_h, best_c

        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def project_box_sum(h, n):
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _corr_grad_vec(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        if k > 0:
            grad[k:] = 1.0
            grad[k:] -= h[:n-k]
        grad[:n-k] -= h[k:]
        if k > 0:
            grad[k:] += 1.0
            grad[k:] -= h[:n-k]
    else:
        k2 = -k
        if k2 > 0:
            grad[:n-k2] = 1.0
            grad[:n-k2] -= h[k2:]
            grad[k2:] -= h[:n-k2]
        else:
            grad[:] = 1.0
            grad[:] -= h
            grad[:] -= h
    return grad

def _corr_grad_vec2(h, k):
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        lo, hi = k, n
        grad[lo:hi] += 1.0
        grad[lo:hi] -= h[:hi-lo]
        lo2, hi2 = 0, n - k
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2+k:hi2+k]
    else:
        k2 = -k
        lo, hi = 0, n - k2
        if lo < hi:
            grad[lo:hi] += 1.0
            grad[lo:hi] -= h[lo+k2:hi+k2]
        lo2, hi2 = k2, n
        if lo2 < hi2:
            grad[lo2:hi2] -= h[:hi2-lo2]
    return grad

def gradient_descent_max(h, n, n_steps=2000, time_limit=None, start_time=None, cool=0.997):
    best_h = h.copy()
    best_c = compute_c5(h)
    alpha = 0.05
    for step in range(n_steps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 30:
                break
        corr_vals = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        max_val = np.max(corr_vals)
        T_soft = 30.0
        w = np.exp(T_soft * (corr_vals - max_val))
        w /= np.sum(w)
        top_idx = np.argsort(w)[::-1][:10]
        grad = np.zeros(n)
        for idx in top_idx:
            if w[idx] < 1e-10:
                break
            lag = idx - n + 1
            grad += w[idx] * _corr_grad_vec2(h, lag)
        grad *= 2.0 / n
        h_new = h - alpha * grad
        h_new = project_box_sum(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new
            h = h_new
        alpha *= cool
        if alpha < 1e-6:
            alpha = 1e-6
    return best_h, best_c

def lbfgsb_run(h0, n, T, top_k, maxiter, penalty=50.0):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        s = np.sort(corr)[::-1][:top_k]
        val = float(np.log(np.sum(np.exp(T * (s - s[0])))) / T + s[0])
        return val + penalty * (np.sum(x) - n / 2.0) ** 2

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * (2.0 / n)
        s = np.sort(corr)[::-1][:top_k]
        exp_s = np.exp(T * (s - s[0]))
        w_s = exp_s / np.sum(exp_s)
        top_corr_vals = s
        all_sorted = np.sort(corr)[::-1]
        grad = np.zeros(n)
        for ti in range(top_k):
            idx_in_sorted = np.where(np.isclose(corr, top_corr_vals[ti]))[0]
            if len(idx_in_sorted) == 0:
                continue
            idx = idx_in_sorted[0]
            lag = idx - n + 1
            grad += w_s[ti] * _corr_grad_vec2(x, lag) * (2.0 / n)
        grad += 2.0 * penalty * (np.sum(x) - n / 2.0)
        return grad

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-7,
                                'maxls': 50})
        return res.x
    except Exception:
        return h0

def coord_descent_fast(h, n, max_rounds=5, time_limit=None, start_time=None):
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 20:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 20:
                    return h, best_c
            old = h[j]
            good = False
            for cv in [0.0, 0.25, 0.5, 0.75, 1.0]:
                if abs(cv - old) < 1e-9:
                    continue
                h[j] = cv
                h = project_box_sum(h, n)
                c = compute_c5(h)
                if c < best_c - 1e-13:
                    best_c = c
                    good = True
                    break
            if not good:
                h[j] = old
                h = project_box_sum(h, n)
            if good:
                improved = True
        if not improved:
            break
    return h, best_c

def binary_search(h, n, max_iters=5000, time_limit=None, start_time=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    h = (h > 0.5).astype(float)
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    for it in range(max_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 30:
                break
        improved = False
        for _ in range(30):
            i1 = ones[rng.randint(len(ones))]
            i0 = zeros[rng.randint(len(zeros))]
            if i1 == i0:
                continue
            h[i1] = 0.0
            h[i0] = 1.0
            c = compute_c5(h)
            if c < best_c - 1e-13:
                best_c = c
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                improved = True
            else:
                h[i1] = 1.0
                h[i0] = 0.0
        if not improved:
            break
    return h, best_c

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    h_new = np.interp(np.linspace(0, 2, new_n, endpoint=False),
                      np.linspace(0, 2, old_n, endpoint=False), h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        n_init = 640
        h0 = np.zeros(n_init)
        h0[:n_init // 2] = 1.0

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: L-BFGS-B at current resolution with increasing T
    for T, mi, pk in [(10, 300, 5), (20, 300, 3), (50, 400, 2), (100, 400, 1), (200, 500, 1), (500, 500, 1)]:
        if time.time() - start_time > budget_s - 400:
            break
        h_new = lbfgsb_run(best_h, n, T, pk, mi)
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Gradient descent
    if time.time() - start_time < budget_s - 300:
        h_new, c_new = gradient_descent_max(best_h, n, n_steps=3000,
                                             time_limit=budget_s, start_time=start_time)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 3: More L-BFGS-B after gradient descent
    for T, mi, pk in [(50, 300, 2), (200, 400, 1), (500, 400, 1)]:
        if time.time() - start_time > budget_s - 250:
            break
        h_new = lbfgsb_run(best_h, n, T, pk, mi)
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-12:
            best_c5 = c_new
            best_h = h_new

    # Phase 4: Coordinate descent
    if time.time() - start_time < budget_s - 150:
        h_cd, c_cd = coord_descent_fast(best_h, n, max_rounds=8,
                                         time_limit=budget_s, start_time=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 5: Gradient descent again
    if time.time() - start_time < budget_s - 200:
        h_new, c_new = gradient_descent_max(best_h, n, n_steps=2000,
                                             time_limit=budget_s, start_time=start_time)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 6: Upsample to 1280
    n2 = 1280
    if time.time() - start_time < budget_s - 300:
        h2 = upsample(best_h, n2)
        for T, mi, pk in [(20, 200, 3), (50, 300, 2), (150, 300, 1), (400, 400, 1)]:
            if time.time() - start_time > budget_s - 200:
                break
            h2 = lbfgsb_run(h2, n2, T, pk, mi)
            h2 = make_feasible(h2, n2)
            c2 = compute_c5(h2)
            if c2 < best_c5 - 1e-12:
                best_c5 = c2
                best_h = h2
        if time.time() - start_time < budget_s - 150:
            h2, c2 = gradient_descent_max(h2, n2, n_steps=1500,
                                           time_limit=budget_s, start_time=start_time)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2
        if time.time() - start_time < budget_s - 80:
            h_cd2, c_cd2 = coord_descent_fast(best_h, len(best_h), max_rounds=4,
                                               time_limit=budget_s, start_time=start_time)
            if c_cd2 < best_c5:
                best_c5 = c_cd2
                best_h = h_cd2

    # Phase 7: Binary local search
    if time.time() - start_time < budget_s - 40:
        h_bin, c_bin = binary_search(best_h, len(best_h), max_iters=8000,
                                      time_limit=budget_s, start_time=start_time, rng=rng)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin

    # Phase 8: Final L-BFGS-B polish if time remains
    if time.time() - start_time < budget_s - 30:
        n_final = len(best_h)
        for T, mi in [(100, 200), (500, 300)]:
            if time.time() - start_time > budget_s - 20:
                break
            h_new = lbfgsb_run(best_h, n_final, T, 1, mi)
            h_new = make_feasible(h_new, n_final)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-12:
                best_c5 = c_new
                best_h = h_new
        h_cd, c_cd = coord_descent_fast(best_h, len(best_h), max_rounds=3,
                                         time_limit=budget_s, start_time=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
