    return h_bin

def sa_with_kicks(h_start, n, time_limit, start_time, rng, T0=0.15, cooling=0.9997, kick_interval=2000, kick_size=5):
    dx = 2.0 / n
    h = to_binary(h_start, n, rng)
    c_cur = compute_c5(h)
    h_best = h.copy()
    c_best = c_cur
    
    ones_pos = np.where(h == 1)[0]
    zeros_pos = np.where(h == 0)[0]
    
    iter_count = 0
    no_improve_count = 0
    
    while time.time() - start_time < time_limit:
        # Check for time
        if iter_count % 100 == 0 and time.time() - start_time > time_limit:
            break
        
        # Kick: occasionally make multiple random swaps
        if iter_count > 0 and iter_count % kick_interval == 0 and no_improve_count > 1000:
            for _ in range(kick_size):
                o = rng.randint(len(ones_pos))
                z = rng.randint(len(zeros_pos))
                op, zp = ones_pos[o], zeros_pos[z]
                h[op], h[zp] = h[zp], h[op]
                ones_pos[o], zeros_pos[z] = zeros_pos[z], ones_pos[o]
            c_cur = compute_c5(h)
            no_improve_count = 0
            # Reset temperature
            T = T0 * 0.5
        else:
            # Normal move
            o = rng.randint(len(ones_pos))
            z = rng.randint(len(zeros_pos))
            op, zp = ones_pos[o], zeros_pos[z]
            h[op] = 0
            h[zp] = 1
            
            c_new = compute_c5(h)
            delta = c_new - c_cur
            
            if delta < 0 or rng.random() < np.exp(-max(delta, -10) / T):
                c_cur = c_new
                ones_pos[o] = zp
                zeros_pos[z] = op
                if c_new < c_best - 1e-15:
                    c_best = c_new
                    h_best = h.copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                h[op] = 1
                h[zp] = 0
            
            T *= cooling
        
        iter_count += 1
    
    return h_best, c_best, c_cur

def targeted_local_search(h, n, max_rounds=100, time_limit=None, start_time=None):
    """Local search that targets the bottleneck lags."""
    dx = 2.0 / n
    h = h.copy()
    c_cur = compute_c5(h)
    
    for rnd in range(max_rounds):
        if time_limit and start_time and time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        max_lags = np.where(corr >= mc - 1e-10)[0]
        # Convert to actual lags (shift by n-1)
        actual_lags = max_lags - (n - 1)
        
        # For each bottleneck lag, try to reduce it
        improved = False
        ones_pos = np.where(h == 1)[0]
        zeros_pos = np.where(h == 0)[0]
        n_o, n_z = len(ones_pos), len(zeros_pos)
        
        for lag in actual_lags[:5]:  # Focus on top 5 bottleneck lags
            if time_limit and start_time and time.time() - start_time > time_limit:
                break
            
            abs_lag = abs(lag)
            if abs_lag == 0:
                continue
            
            # Try swaps that specifically reduce correlation at this lag
            # A swap of position i (h=1) with position j (h=0) affects lag k if
            # |i-j| is related to k... actually it affects all lags.
            # Simpler: try random swaps and check if they reduce the max
            for attempt in range(50):
                oi = np.random.randint(n_o)
                zi = np.random.randint(n_z)
                op, zp = ones_pos[oi], zeros_pos[zi]
                h[op] = 0
                h[zp] = 1
                c_try = compute_c5(h)
                if c_try < c_cur - 1e-15:
                    c_cur = c_try
                    ones_pos[oi] = zp
                    zeros_pos[zi] = op
                    improved = True
                    break
                else:
                    h[op] = 1
                    h[zp] = 0
        
        if not improved:
            break
    
    return h, c_cur

def golden_ratio_init(n, seed):
    """Initialize using golden ratio distribution."""
    phi = (1 + 5**0.5) / 2
    h = np.zeros(n)
    positions = np.round(np.arange(n // 2) * phi * n / (n // 2)).astype(int) % n
    # Deduplicate
    positions = np.unique(positions)
    if len(positions) < n // 2:
        rng = np.random.RandomState(seed)
        remaining = np.setdiff1d(np.arange(n), positions)
        extra = rng.choice(remaining, n // 2 - len(positions), replace=False)
        positions = np.concatenate([positions, extra])
    h[positions] = 1.0
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_c5(h):
    n = len(h)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary_strict(h, n, rng):
    """Convert to strictly binary with exactly n/2 ones."""
    h_b = np.round(h).astype(float)
    h_b = np.clip(h_b, 0, 1)
    n_ones = int(np.sum(h_b))
    target = n // 2

            h_b[idx] = 1
            n_ones += 1

def compute_c5(h):

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary_strict(h, n, rng):
    h_b = np.round(h).astype(float)
    h_b = np.clip(h_b, 0, 1)
    n_ones = int(np.sum(h_b))

    while n_ones > target:

        if h_b[idx] == 1:
            h_b[idx] = 0
            n_ones -= 1
    while n_ones < target:
        idx = rng.randint(n)
        if h_b[idx] == 0:
            h_b[idx] = 1
            n_ones += 1
    return h_b

    h = to_binary_strict(h_start, n, rng)

    c_best = c_cur

    stagnant = 0

        if iters > 0 and iters % kick_every == 0 and stagnant > kick_every // 3:
            for _ in range(kick_mag):

        else:

                    stagnant = 0

    h = to_binary_strict(h, n, np.random.RandomState(42))
    c_cur = compute_c5(h)
