So c_k = c_k^base + h[i+m_k]*(1-h[i]) + h[i]*(1-h[i-m_k])
where c_k^base is c_k with those two terms removed.

= c_k^base + h[i+m_k] - h[i]*h[i+m_k] + h[i] - h[i]*h[i-m_k]
= c_k^base + h[i+m_k] + h[i]*(1 - h[i+m_k] - h[i-m_k])
= c_k^base + h[i+m_k] + h[i]*(-h[i+m_k] - h[i-m_k] + 1)

So c_k is linear in h[i] with coefficient (1 - h[i+m_k] - h[i-m_k]) (when both neighbors exist).

This means in CD, for a fixed set of active lags, C₅ as a function of h[i] is the max of linear functions, which is convex! So we can find the optimal h[i] by considering the breakpoints.

But the issue is that changing h[i] changes which lags are active (the max). However, for a small change in h[i], the active set is likely the same, so we can do:
1. Identify active lags (those where c_k is near the max)
2. Find h[i] that minimizes max over active lags of (linear function of h[i])
3. This is a 1D linear programming problem (minimize max of linear functions)

For the 1D LP: minimize max_j (a_j * t + b_j) over t ∈ [0,1]
The optimal t is where the two "most constraining" lines cross, or at a boundary.

This could be much more efficient than evaluating C₅ at each candidate value!

Let me implement this approach:

```python
def cd_step(h, i, n):
    """Optimize h[i] holding others fixed. Returns optimal value."""
    dx = 2.0 / n
    # Compute base correlations (without h[i]'s contribution)
    # Actually, let's just compute the coefficients and constants
    
    # For each lag k (0 to 2n-2), m_k = k - (n-1)
    # c_k = sum of h[i'+m_k]*(1-h[i']) over valid i'
    # Terms with h[i]:
    #   If i+m_k valid: h[i+m_k]*(1-h[i]) → contributes -h[i]*h[i+m_k] to c_k, and +h[i+m_k] is constant
    #   If i-m_k valid: h[i]*(1-h[i-m_k]) → contributes h[i]*(1-h[i-m_k]) to c_k
    
    # So c_k = base_k + h[i] * coeff_k
    # where coeff_k = (1-h[i-m_k]) if i-m_k valid, minus h[i+m_k] if i+m_k valid
    # and base_k = c_k - (the h[i] dependent terms evaluated at current h[i])
    
    # Simpler: just compute c_k as a function of t = h[i]
    # c_k(t) = A_k + t * B_k
    # where B_k = (1-h[i-m_k])*I(i-m_k valid) - h[i+m_k]*I(i+m_k valid)
    # and A_k = c_k - t_current * B_k
    
    # Then C5(t) = max_k (A_k + t*B_k) * dx
    # Minimize over t in [0,1]
    
    # Compute B_k for all k
    # B_k = (1-h[i-m_k]) if 0<=i-m_k<n else 0, minus h[i+m_k] if 0<=i+m_k<n else 0
    
    B = np.zeros(2*n-1)
    for k in range(2*n-1):
        m_k = k - (n-1)
        b = 0.0
        if 0 <= i - m_k < n:
            b += 1.0 - h[i - m_k]
        if 0 <= i + m_k < n:
            b -= h[i + m_k]
        B[k] = b
    
    # Compute A_k = c_k - h[i] * B_k
    t_cur = h[i]
    corr = np.correlate(h, 1.0 - h, mode='full')
    A = corr - t_cur * B
    
    # Now minimize max_k (A_k + t*B_k) over t in [0,1]
    # This is a 1D LP: min s.t. A_k + t*B_k <= s for all k, 0<=t<=1
    # Optimal: find t where the max is minimized
    
    # The function f(t) = max_k (A_k + t*B_k) is convex in t
    # Minimum is where the subgradient contains 0, or at a boundary
    
    # Simple approach: ternary search or check all breakpoints
    # Breakpoints are where A_j + t*B_j = A_k + t*B_k → t = (A_k - A_j)/(B_j - B_k)
    
    # For efficiency, just use a golden section search or check O(n) candidates
    
    # Actually, for a convex piecewise-linear function on [0,1], the minimum is at a breakpoint or boundary.
    # There are O(n) breakpoints, but finding them all is O(n^2).
    # Instead, use bisection on the derivative (subgradient).
    
    # f(t) is convex, so we can use the fact that the left derivative is non-decreasing.
    # At the minimum, 0 is in the subgradient.
    # Subgradient at t: {sum of B_k for k in argmax(A_k + t*B_k)}... 
    # Actually, the subdifferential of max of linear functions at t is the convex hull of the B_k for active k.
    
    # So at the optimum, 0 should be in the convex hull of {B_k : k in argmax}.
    # This means min active B_k <= 0 <= max active B_k, OR the optimum is at t=0 or t=1.
    
    # Practical approach: evaluate f at t=0, t=1, and a few interior points
    
    # Or: since f is convex and piecewise linear, use a simple bisection:
    # Check if f is increasing or decreasing at t=0.5 by looking at the subgradient
    
    # Let me just do a fine grid search + local refinement
    t_candidates = np.linspace(0, 1, 101)
    f_values = np.max(A[:, None] + B[:, None] * t_candidates[None, :], axis=0)
    best_t = t_candidates[np.argmin(f_values)]
    
    # Refine around best_t
    t_fine = np.linspace(max(0, best_t - 0.01), min(1, best_t + 0.01), 201)
    f_fine = np.max(A[:, None] + B[:, None] * t_fine[None, :], axis=0)
    best_t = t_fine[np.argmin(f_fine)]
    
    return best_t
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-10:
        idx = np.argmin(h) if d > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + d, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def cd_optimize(h, n, max_rounds=20, time_limit=None, start_time=None):
    """Coordinate descent optimization."""
    dx = 2.0 / n
    best_c5 = np.max(np.correlate(h, 1.0 - h, mode='full')) * dx
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    
    t_candidates_coarse = np.linspace(0, 1, 21)
    t_candidates_fine = 21  # number of fine points
    
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        for i in range(n):
            if time_limit is not None and start_time is not None:
                if (rnd * n + i) % 200 == 0 and time.time() - start_time > time_limit:
                    break
            
            # Compute B for coordinate i
            B = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                m_k = k - (n - 1)
                b = 0.0
                p1 = i - m_k
                p2 = i + m_k
                if 0 <= p1 < n:
                    b += 1.0 - h[p1]
                if 0 <= p2 < n:
                    b -= h[p2]
                B[k] = b
            
            old_t = h[i]
            A = corr - old_t * B
            
            # Coarse search
            f_coarse = np.max(A[:, None] + B[:, None] * t_candidates_coarse[None, :], axis=0)
            idx_best = np.argmin(f_coarse)
            best_t = t_candidates_coarse[idx_best]
            
            # Fine search around best
            lo = max(0.0, best_t - 0.05)
            hi = min(1.0, best_t + 0.05)
            t_fine = np.linspace(lo, hi, 51)
            f_fine = np.max(A[:, None] + B[:, None] * t_fine[None, :], axis=0)
            best_t = t_fine[np.argmin(f_fine)]
            
            # Try boundaries too
            for bv in [0.0, 1.0]:
                f_bv = np.max(A + bv * B)
                f_best = np.max(A + best_t * B)
                if f_bv < f_best:
                    best_t = bv
            
            if abs(best_t - old_t) > 1e-10:
                # Update
                corr += (best_t - old_t) * B
                h[i] = best_t
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                improved = True
        
        if not improved:
            break
    
    return h, best_c5
```
