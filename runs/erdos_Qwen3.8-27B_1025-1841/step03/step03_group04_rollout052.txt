import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = h.sum()
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
            mask = room > 1e-12
            if not mask.any():
                break
            total_room = room[mask].sum()
            if total_room < 1e-12:
                break
            frac = min(1.0, diff / total_room)
            h[mask] += room[mask] * frac
        else:
            mask = h > 1e-12
            if not mask.any():
                break
            total = h[mask].sum()
            if total < 1e-12:
                break
            frac = min(1.0, (-diff) / total)
            h[mask] -= h[mask] * frac
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_gradient_and_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    exp_c = np.exp(T * (c - cmax) * dx)
    wsum = exp_c.sum()
    if wsum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = exp_c / wsum
    obj = (cmax + np.log(wsum) / T) * dx
    G = np.zeros(n)
    for m in range(2 * n - 1):
        wm = w[m]
        if wm < 1e-14:
            continue
        k = m - (n - 1)
        lo1 = max(0, -k)
        hi1 = min(n, n - k)
        if hi1 > lo1:
            G[lo1:hi1] += wm * (1.0 - h[lo1 + k:hi1 + k])
        lo2 = max(0, k)
        hi2 = min(n, n + k)
        if hi2 > lo2:
            G[lo2:hi2] -= wm * h[lo2 - k:hi2 - k]
    G *= dx
    return obj, G, c, w

def gradient_descent(h, T_start, T_end, n_iter, lr_start, lr_min):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_start
    for T_frac in np.linspace(0, 1, n_iter):
        T = T_start + (T_end - T_start) * T_frac
        for it in range(5):
            obj, G, c, w = smooth_gradient_and_obj(h, T)
            G_proj = G - G.mean()
            for _ in range(30):
                h_new = project_feasible(h - lr * G_proj, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    h = h_new
                    best_c5 = c5_new
                    best_h = h.copy()
                    break
                lr *= 0.5
            lr = max(lr * 1.1, lr_min)
    return best_h, best_c5

def focused_coord_descent(h, n, num_passes=3):

    best_h = h.copy()
    for p in range(num_passes):

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return project_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def slsqp_polish(h, n, maxiter=300):
    dx = 2.0 / n
    T_slsqp = 400.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = c.max()
        e = np.exp(T_slsqp * (c - mc) * dx)
        return (mc + np.log(e.sum()) / T_slsqp) * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = c.max()
        e = np.exp(T_slsqp * (c - mc) * dx)
        ws = e / e.sum()
        G = np.zeros(n)
        for m in range(2 * n - 1):
            wm = ws[m]
            if wm < 1e-14:
                continue
            k = m - (n - 1)
            lo1 = max(0, -k)
            hi1 = min(n, n - k)
            if hi1 > lo1:
                G[lo1:hi1] += wm * (1.0 - x[lo1 + k:hi1 + k])
            lo2 = max(0, k)
            hi2 = min(n, n + k)
            if hi2 > lo2:
                G[lo2:hi2] -= wm * x[lo2 - k:hi2 - k]
        return G * dx
    def con_fun(x):
        return x.sum() - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16})
        h_res = project_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return h, compute_c5(h)
