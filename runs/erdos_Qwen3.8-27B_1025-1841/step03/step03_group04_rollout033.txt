from scipy.optimize import minimize

_G_N = 0
_G_T = 100.0
_G_TOP_IDX = None

def _obj_full(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def corr_update(corr, h, j, delta, n):
    """Incrementally update correlation when h[j] changes by delta."""
    new_corr = corr.copy()
    kn1 = n - 1
    # Term 1: k in [j, j+n-1] ∩ [0, 2n-2]
    k_lo = j
    k_hi = min(j + n - 1, 2 * n - 2)
    if k_lo <= k_hi:
        h_idx = j - np.arange(k_lo, k_hi + 1) + n - 1
        new_corr[k_lo:k_hi+1] += delta * (1.0 - h[h_idx])
    # Term 2: k in [n-1-j, 2n-2-j] ∩ [0, 2n-2]
    k_lo2 = n - 1 - j
    k_hi2 = min(2 * n - 2 - j, 2 * n - 2)
    if k_lo2 <= k_hi2:
        h_idx2 = j + np.arange(k_lo2, k_hi2 + 1) - n + 1
        new_corr[k_lo2:k_hi2+1] -= delta * h[h_idx2]
    # Fix k=n-1: subtract delta^2
    new_corr[kn1] -= delta * delta
    return new_corr
