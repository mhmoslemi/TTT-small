def fft_cross_corr(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    return corr[:2*n-1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_cross_corr(h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.random.argmax(h < 1.0) if diff > 0 else np.random.argmin(h > 0.0)
        h[idx] += diff
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    h = h.copy()
    # Sort by value, set top n/2 to 1, rest to 0
    thresh = np.sort(h)[n//2]
    h_bin = (h >= thresh).astype(float)
    # Fix count
    n_ones = h_bin.sum()
    if n_ones > n/2:
        # Randomly set some to 0
        one_idx = np.where(h_bin == 1)[0]
        extra = n_ones - n/2
        rng_choice = np.random.choice(one_idx, int(extra), replace=False)
        h_bin[rng_choice] = 0
    elif n_ones < n/2:
        zero_idx = np.where(h_bin == 0)[0]
        missing = n/2 - n_ones
        rng_choice = np.random.choice(zero_idx, int(missing), replace=False)
        h_bin[rng_choice] = 1
    return h_bin

def soft_max(corr, dx, T):
    """Smooth approximation of max(corr) * dx."""
    x = corr * dx
    x_max = x.max()
    shifted = T * (x - x_max)
    shifted = np.clip(shifted, -50, 50)
    return (x_max + np.log(np.sum(np.exp(shifted))) / T)

    h = h.copy()
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    
    dx = 2.0 / n
    corr = fft_cross_corr(h)
    current_energy = soft_max(corr, dx, T_init)
    best_energy = current_energy
    best_h = h.copy()
    
    start = time.time()
    
    for it in range(n_iters):
        if it % 1000 == 0 and time.time() - start > time_limit:
            break
        # Adaptive temperature
        frac = it / max(n_iters - 1, 1)
        T = T_init * (T_final / T_init) ** frac

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i] = 0; h[j] = 1
        new_corr = fft_cross_corr(h)
        new_energy = soft_max(new_corr, dx, T)
        
        delta = new_energy - current_energy
        if delta < 0 or rng.random() < np.exp(-delta / max(T * 0.01, 1e-10)):

            ones[ones.index(i)] = j
            zeros[zeros.index(j)] = i
            corr = new_corr
            if current_energy < best_energy:

                best_h = h.copy()
        else:
            # Reject
            h[i] = 1; h[j] = 0
            corr = fft_cross_corr(h)
    
    return best_h, best_energy
