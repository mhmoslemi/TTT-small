```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    tc = t * corr
    tc_max = np.max(tc)
    exp_tc = np.exp(tc - tc_max)
    p = exp_tc / np.sum(exp_tc)
    obj = (tc_max + np.log(np.sum(exp_tc))) / t * dx
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        if p[m] < 1e-14:
            continue
        s = m - (n - 1)
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += p[m] * (1.0 - h[s:])
                grad[s:] -= p[m] * h[:ns]
        else:
            ns = -s
            cnt = n - ns
            if cnt > 0:
                grad[ns:] += p[m] * (1.0 - h[:cnt])
                grad[:cnt] -= p[m] * h[ns:]
    grad *= dx
    return obj, grad

def lbfgs_optimize(h, n, t_list, maxiter_per_t):
    best_h = h.copy()
    best_c = compute_c5(best_h)
    for t in t_list:
        def obj(h):
            return soft_obj_grad(h, t)[0]
        def obj_grad(h):
            return soft_obj_grad(h, t)[1]
        bounds = [(0.0, 1.0)] * n
        # Use L-BFGS-B with constraint handling
        # L-BFGS-B doesn't directly support equality constraints
        # We'll use a penalty method or just ignore the constraint
        # and project afterwards
        try:
            res = minimize(obj, best_h, jac=obj_grad, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxiter_per_t, 'ftol': 1e-14})
            h_new = make_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c:
                best_c = c_new
                best_h = h_new
        except Exception:
            pass
    return best_h, best_c
```

4. Just use L-BFGS-B and project afterwards (the constraint might be slightly violated but make_feasible will fix it)
