                val = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
                val += lam * (np.sum(x) - n / 2.0) ** 2
                return val
            
            bounds = [(0.0, 1.0)] * n_pts
            try:
                res = minimize(obj_lbfgs, h_work, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 500, 'ftol': 1e-16, 'gtol': 1e-10})
                h_work = make_feasible(res.x, n_pts)
                c5_res = compute_c5(h_work)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_work.copy()
            except Exception:
                pass
    
    # Phase 4: Aggressive coordinate descent with binary snap
    n_pts = len(best_h)
    if remaining() > 30:
        best_h, best_c5 = coordinate_descent(best_h, n_pts, max_rounds=5,
                                              candidates=[0.0, 1.0, 0.5])
    
    # Phase 5: Fine coordinate descent
    if remaining() > 20:
        fine_cands = [0.0, 1.0]
        best_h, best_c5 = coordinate_descent(best_h, n_pts, max_rounds=3,
                                              candidates=fine_cands)
    
    # Phase 6: Final SLSQP polish with very high temperature
    if remaining() > 40:
        n_pts = len(best_h)
        h_work = best_h.copy()
        T_final = 800.0
        dx = 2.0 / n_pts
        
        def obj_final(x, n=n_pts, T=T_final):
            corr = np.correlate(x, 1.0 - x, mode='full')
            mc = np.max(corr)
            shifted = corr - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        
        def con_fun(x, n=n_pts):
            return np.sum(x) - n / 2.0
        
        def con_jac(x, n=n_pts):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj_final, h_work, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass
    
    # Phase 7: Final fine-tuning
    if remaining() > 10:
        n_pts = len(best_h)
        fine_cands = []
        for d in [0.001, 0.005, 0.01, -0.001, -0.005, -0.01]:
            fine_cands.append(d)
        # Add 0 and 1
        fine_cands = [0.0, 1.0] + fine_cands
        best_h, best_c5 = coordinate_descent(best_h, n_pts, max_rounds=2,
                                              candidates=fine_cands)
    
    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
