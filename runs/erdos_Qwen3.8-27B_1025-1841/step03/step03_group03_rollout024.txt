def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)

            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-15:

        else:

    target = n / 2.0

        idx = np.where(mask)[0]
        if len(idx) > 0:

            h_proj = np.clip(h_proj, 0.0, 1.0)

    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    # Term 1: j appears as i+s (j = i+s, i = j-s)
    # Valid: 0 <= j < n and 0 <= j-s < n => max(0,s) <= j < min(n, n+s)
    lo1 = max(0, s)

    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if lo2 < hi2:

    return grad

    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def subgradient_optimize(h, n, max_iter, step_init, time_limit, start_time):

    best_c = compute_c5(h)
    step = step_init
    for it in range(max_iter):
        if time.time() - start_time > time_limit - 30:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        c_current = corr[k_star] * (2.0/n)
        
        if c_current < best_c:
            best_c = c_current
        
        grad = grad_corr(h, k_star)
        
        # Line search
        best_step = step
        best_obj = c_current
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05, 2.0, 4.0]:
            s = step * factor
            h_trial = h - s * grad
            h_trial = project_feasible(h_trial, n)
            corr_t = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            c_t = np.max(corr_t) * (2.0/n)
            if c_t < best_obj:
                best_obj = c_t
                best_step = s
                h = h_trial
        
        # Adaptive step
        if best_step < step * 0.1:
            step *= 0.5
        elif best_step > step * 5:
            step *= 2.0
        
        if best_obj < best_c:
            best_c = best_obj
            h = project_feasible(h - best_step * grad, n)
    
    return h, best_c

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = (h < 1.0).astype(int)
        else:
            mask = (h > 0.0).astype(int)
        idx = np.where(mask > 0)[0]
        if len(idx) > 0:
            h[idx[0]] = np.clip(h[idx[0]] + diff, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) * 0.5
        h_p = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_p < 1.0
        else:
            mask = h_p > 0.0
        idx = np.where(mask)[0]
        if len(idx) > 0:
            h_p[idx[0]] = np.clip(h_p[idx[0]] + diff, 0.0, 1.0)
    return h_p

def grad_corr_at(h, k):
    n = len(h)
    s = k - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, s)
    hi1 = min(n, n + s)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1 - s: hi1 - s])
    lo2 = max(0, -s)
    hi2 = min(n, n - s)
    if lo2 < hi2:
        grad[lo2:hi2] += -h[lo2 + s: hi2 + s]
    return grad

def gradient_optimize(h, n, max_iter, time_limit, start_time, top_k=5, momentum=0.5):
    h = h.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    v = np.zeros(n)
    h_bar = h.copy()
    step = 0.3
    it_count = 0
    for it in range(max_iter):
        if time.time() - start_time > time_limit - 30:
            break
        it_count += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        c_max = np.max(c_vals)
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
        top_idx = np.argsort(c_vals)[-top_k:][::-1]
        active_vals = c_vals[top_idx]

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
