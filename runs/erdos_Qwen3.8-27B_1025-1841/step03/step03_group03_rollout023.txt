        if not improved:
            break
        if kick_interval > 0 and (sweep + 1) % kick_interval == 0:
            n_kick = max(1, int(n * 0.08))
            kick_idx = rng.choice(n, n_kick, replace=False)
            h[kick_idx] += (rng.rand(n_kick) - 0.5) * 2.0 * kick_size
            h = make_feasible(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full')
            c5 = float(np.max(corr) * dx)
    return best_h, best_c5

def pair_transfer(h_init, max_iters, time_limit, start_time, seed, transfer_frac=0.02):
    """Try transferring mass between pairs of coordinates."""
    n = len(h_init)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h_init.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 15:
            break
        j1, j2 = rng.choice(n, 2, replace=False)
        lo = max(0.0, h[j1] - 1.0)
        hi = min(1.0, h[j1])
        if hi - lo < 1e-8:
            continue
        delta = (rng.rand() - 0.5) * 2.0 * transfer_frac
        new_h1 = np.clip(h[j1] + delta, 0.0, 1.0)
        new_h2 = np.clip(h[j2] - delta, 0.0, 1.0)
        actual_delta = new_h1 - h[j1]
        if abs(actual_delta) < 1e-10:
            continue
        new_h2 = np.clip(h[j2] - actual_delta, 0.0, 1.0)
        trial = h.copy()
        trial[j1] = new_h1
        trial[j2] = new_h2
        trial = make_feasible(trial, n)
        c_trial = compute_c5(trial)
        if c_trial < best_c5 - 1e-12:
            best_c5 = c_trial
            best_h = trial
            h = trial
    return best_h, best_c5

def structured_init(n, rng, kind=0):
    """Generate structured initial guesses."""
    h = np.zeros(n)
    if kind == 0:
        # Alternating blocks of size ~1
        block = max(1, n // 12)
        for i in range(0, n, 2 * block):
            h[i:i+block] = 1.0
    elif kind == 1:
        # Random with bias
        h = (rng.rand(n) < 0.5).astype(float)
    elif kind == 2:
        # Sine-based
        x = np.linspace(0, 2, n)
        h = (0.5 + 0.4 * np.sin(2 * np.pi * x)).clip(0, 1)
    elif kind == 3:
        # Three-phase
        h = (np.sin(3 * np.pi * np.linspace(0, 2, n)) > 0).astype(float)
    return make_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n = 640
        h0 = (rng.rand(n) < 0.5).astype(float)
        h0 = make_feasible(h0, n)

    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Coordinate descent from parent
    h1, c1 = coord_descent(best_h, max_sweeps=60, time_limit=budget_s,
                           start_time=start_time, seed=seed,
                           kick_interval=10, kick_size=0.03)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Phase 2: More CD with different kick parameters
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        for ki, ks in [(7, 0.02), (5, 0.05), (4, 0.08)]:
            if time.time() - start_time > budget_s - 200:
                break
            h2, c2 = coord_descent(best_h, max_sweeps=40, time_limit=budget_s,
                                   start_time=start_time, seed=seed + 10 + ki,
                                   kick_interval=ki, kick_size=ks)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2

    # Phase 3: Try structured starts
    if time.time() - start_time < budget_s - 200:
        for kind in range(4):
            if time.time() - start_time > budget_s - 150:
                break
            h_s = structured_init(n0, rng, kind=kind)
            h_s, c_s = coord_descent(h_s, max_sweeps=40, time_limit=budget_s,
                                     start_time=start_time, seed=seed + 100 + kind,
                                     kick_interval=8, kick_size=0.04)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s

    # Phase 4: Upsample and refine
    if time.time() - start_time < budget_s - 120:
        n2 = n0 * 2
        h_u = upsample_h(best_h, n2)
        h_u, c_u = coord_descent(h_u, max_sweeps=30, time_limit=budget_s,
                                 start_time=start_time, seed=seed + 200,
                                 kick_interval=10, kick_size=0.02)
        if c_u < best_c5:
            best_c5 = c_u
            best_h = h_u

    # Phase 5: Pair transfer refinement
    if time.time() - start_time < budget_s - 60:
        n_pt = min(len(best_h), 640)
        h_pt = upsample_h(best_h, n_pt) if len(best_h) != n_pt else best_h.copy()
        h_pt, c_pt = pair_transfer(h_pt, max_iters=5000, time_limit=budget_s,
                                   start_time=start_time, seed=seed + 300,
                                   transfer_frac=0.015)
        if c_pt < best_c5:
            best_c5 = c_pt
            best_h = h_pt

    # Phase 6: Final CD without kicks (pure exploitation)
    if time.time() - start_time < budget_s - 30:
        h_f, c_f = coord_descent(best_h, max_sweeps=50, time_limit=budget_s,
                                 start_time=start_time, seed=seed + 400,
                                 kick_interval=0, kick_size=0.0)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
