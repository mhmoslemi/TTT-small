def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h
```

def optimal_coord(h, j, corr, n_active=50):
    """Find optimal value of h[j] given other coordinates fixed."""
    n = len(h)
    dx = 2.0 / n
    h_j = h[j]
    
    # Compute a_k and b_k for active correlations
    mc = np.max(corr)
    top_idx = np.argsort(-corr)[:n_active]
    
    # For each active correlation, compute coefficient of h[j]
    candidates = [0.0, 1.0]
    breakpoints = []
    
    for k in top_idx:
        lag = k - (n - 1)
        a = 0.0
        if lag >= 0:
            if j < n - lag:
                a += (1 - h[j + lag]) * dx
            if j >= lag:
                a -= h[j - lag] * dx
        else:
            L = -lag
            if j >= L:
                a += (1 - h[j - L]) * dx
            if j < n - L:
                a -= h[j + L] * dx
        
        b = corr[k] - a * h_j
        candidates.append(b)  # value at h[j]=0 is b, we track the function
        # Store (a, b) for breakpoint computation
        breakpoints.append((a, b))
    
    # The objective at h[j]=v is: max_k (a_k * v + b_k)
    # Find minimum over v in [0,1]
    # Try v=0, v=1, and breakpoints
    
    best_v = h_j
    best_val = np.max([a * h_j + b for a, b in breakpoints])
    
    for v in [0.0, 1.0]:
        val = np.max([a * v + b for a, b in breakpoints])
        if val < best_val:
            best_val = val
            best_v = v
    
    # Find breakpoints
    for i in range(len(breakpoints)):
        for jj in range(i+1, len(breakpoints)):
            a1, b1 = breakpoints[i]
            a2, b2 = breakpoints[jj]
            if abs(a1 - a2) > 1e-15:
                v_bp = (b2 - b1) / (a1 - a2)
                if 0.0 <= v_bp <= 1.0:
                    val = np.max([a * v_bp + b for a, b in breakpoints])
                    if val < best_val:
                        best_val = val
                        best_v = v_bp
    
    return best_v
```

But wait, we can be smarter. The function max_k(a_k*v + b_k) is convex PL. Its minimum on [0,1] can be found by:
1. Evaluating at v=0 and v=1
2. Checking if the slope changes sign (i.e., the function goes from decreasing to increasing)

The slope at any v is a_{k*(v)} where k* is the argmax at v. The function decreases while the argmax has negative slope and increases while it has positive slope. The minimum is where the slope changes from negative to positive, which is at a breakpoint.

A simpler approach: binary search for the minimum. The function is convex, so we can ternary search. Each evaluation is O(K). Ternary search takes O(log(1/eps)) iterations ≈ 50 for double precision. Total: O(K * 50) per coordinate. For K=50: 2500 per coordinate, 960*2500 = 2.4M per pass. Very fast!

```python
def optimal_coord_l1(h, j, corr, n, n_active=50):
    dx = 2.0 / n
    h_j = h[j]
    top_idx = np.argsort(-corr)[:n_active]
    
    # Precompute (a, b) pairs
    ab = np.zeros((n_active, 2))
    for idx, k in enumerate(top_idx):
        lag = k - (n - 1)
        a = 0.0
        if lag >= 0:
            if j < n - lag:
                a += (1 - h[j + lag]) * dx
            if j >= lag:
                a -= h[j - lag] * dx
        else:
            L = -lag
            if j >= L:
                a += (1 - h[j - L]) * dx
            if j < n - L:
                a -= h[j + L] * dx
        b = corr[k] - a * h_j
        ab[idx] = [a, b]
    
    # Ternary search for min of max_k(a_k*v + b_k) on [0,1]
    lo, hi = 0.0, 1.0
    for _ in range(60):
        m1 = lo + (hi - lo) / 3
        m2 = hi - (hi - lo) / 3
        f1 = np.max(ab[:,0]*m1 + ab[:,1])
        f2 = np.max(ab[:,0]*m2 + ab[:,1])
        if f1 < f2:
            hi = m2
        else:
            lo = m1
    
    best_v = (lo + hi) / 2
    # Also check endpoints
    f0 = np.max(ab[:,1])  # at v=0
    f1 = np.max(ab[:,0] + ab[:,1])  # at v=1
    fmid = np.max(ab[:,0]*best_v + ab[:,1])
    
    if f0 < fmid and f0 < f1:
        best_v = 0.0
    elif f1 < fmid and f1 < f0:
        best_v = 1.0
    # else best_v stays as ternary result
    
    return best_v
```

    dx = 2.0 / n
    top_idx = np.argsort(-corr)[:n_active]
    grad = np.zeros(n)
    for k in top_idx:

        if lag > 0:
            grad[lag:] += (1.0 - h[:n - lag]) * dx
            grad[:n - lag] -= h[lag:] * dx

            grad[:n - L] += (1.0 - h[L:]) * dx

                lag = k - (n - 1)
                if lag > 0:
                    grad2[lag:] += (1.0 - x2[:n - lag]) * dx
                    grad2[:n - lag] -= x2[lag:] * dx
                elif lag < 0:
                    L = -lag
                    grad2[:n - L] += (1.0 - x2[L:]) * dx
                    grad2[L:] -= x2[:n - L] * dx
                else:
                    grad2 += (1.0 - 2.0 * x2) * dx
            grad2 /= len(top_idx2)
            v2 = 0.85 * v2 - step2 * grad2
            x2_new = project_feasible(x2 + v2, n)
            f2 = compute_c5(x2_new)
            if f2 < compute_c5(x2) - 1e-12:
                x2 = x2_new
                if f2 < best_c5:
                    best_c5 = f2
                    best_h = x2.copy()
                step2 = min(step2 * 1.005, 0.02)
            else:
                x2 = x2_new
                step2 *= 0.85
                if step2 < 0.0005:
                    step2 = 0.0005
        print(f"After targeted SG: C5={best_c5:.6f}", flush=True)

    # Phase 6: Final coordinate descent
    if remaining() > 40:
        h_cd, c5_cd = exact_coord_descent(best_h.copy(), n, n_active=60, max_passes=4)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd

    # Phase 7: Binary snap for near-binary
    if remaining() > 20:
        for j in range(n):
            if remaining() < 5:
                break
            for val in [0.0, 1.0]:
                if abs(best_h[j] - val) < 1e-6:
                    continue
                trial = best_h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial

    # Phase 8: Final SLSQP
    if remaining() > 30:
        _RUN_N = n
        _RUN_T = 800.0
        _RUN_TOP_FRAC = 0.005
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun_wrapper, 'jac': con_jac_wrapper}]
        try:
            res = minimize(obj_smooth_wrapper, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-16})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 9: Tiny perturbations
    if remaining() > 15:
        for _ in range(200):
            if remaining() < 5:
                break
            j = rng.randint(n)
            for delta in [0.0005, -0.0005, 0.001, -0.001, 0.002, -0.002]:
                trial = best_h.copy()
                trial[j] = best_h[j] + delta
                if trial[j] < 0 or trial[j] > 1:
                    continue
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_h = trial
                    break

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    print(f"Final: C5={best_c5:.6f}, n={n}, time={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, n
