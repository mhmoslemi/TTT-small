```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_with_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx, np.argmax(corr)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def fista_nonsmooth(h, n_iter=3000, lr=0.05, check=50):
    """FISTA with exact max (nonsmooth) objective."""
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        # Compute gradient of the active constraint
        corr = np.correlate(z, 1.0 - z, mode='full')
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        
        # Gradient of g_{k_star}(z)
        g = np.zeros(n)
        if shift >= 0:
            g[shift:] += 1.0 - z[:n-shift]
            g[:n-shift] -= z[shift:]
        else:
            g[:n-shift] += 1.0 - z[shift:]
            g[shift:] -= z[:n-shift]
        g *= dx
        
        # FISTA update
        lrt = lr / (tk)  # adaptive step
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = x.copy(), c
    return bh, bc
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, s):
    """Subgradient of correlation at shift s."""
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        s = -s
        g[:n-s] -= h[s:]
        g[s:] += 1.0 - h[:n-s]
    return g

def smooth_grad(h, T, top_k=50):
    """Smoothed gradient using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        k_star = int(np.argmax(corr)) - (n - 1)
        return subgrad_one(h, k_star) * dx
    w = exps / tot
    # Use top-k for efficiency
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter, T0, T1, lr, check=40, top_k=50):
    """FISTA with smoothed gradient."""
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.3 * f)
        g = smooth_grad(z, T, top_k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def solve_lp_relax(n, max_shift, h_init=None):
    """Solve LP relaxation with McCormick envelopes for shifts |k| <= max_shift."""
    h = cp.Variable(n)
    C = cp.Variable()
    u = {}  # u[(s, i)] for each shift s and valid index i
    
    # Build constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    # For each shift, add McCormick constraints and the C constraint
    for s in range(-max_shift, max_shift + 1):
        if s >= 0:
            valid = list(range(0, n - s))
        else:
            valid = list(range(-s, n))
        
        u_s = cp.Variable(len(valid))
        constraints.append(u_s >= 0)
        
        for idx, i in enumerate(valid):
            j = i + s
            # McCormick: h[i] + h[j] - 1 <= u <= min(h[i], h[j])
            constraints.append(u_s[idx] >= h[i] + h[j] - 1)
            constraints.append(u_s[idx] <= h[i])
            constraints.append(u_s[idx] <= h[j])
        
        # C constraint: C >= dx * (sum_{valid i} h[i] - sum_{valid i} u[i])
        dx = 2.0 / n
        total_h = cp.sum([h[i] for i in valid])
        total_u = cp.sum(u_s)
        constraints.append(C >= dx * (total_h - total_u))
    
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLOP)
        if h.value is not None:
            return np.clip(h.value, 0, 1), C.value
    except Exception:
        pass
    
    try:
        prob.solve(solver=cp.CBC)
        if h.value is not None:
            return np.clip(h.value, 0, 1), C.value
    except Exception:
        pass
    
    return None, None

def binary_local_search(h, n_max_flips=5000):
    """Local search: flip individual bits to reduce C5."""
    n = len(h)
    # First make it binary
    target = n / 2.0
    h_bin = np.zeros(n)
    # Find the n/2 largest values and set them to 1
    idx = np.argsort(h)[::-1][:int(target)]
    h_bin[idx] = 1.0
    
    bc = compute_c5(h_bin)
    
    for flip_count in range(n_max_flips):
        best_improve = 0
        best_idx = -1
        # Try flipping some random positions
        candidates = np.random.choice(n, min(n, 200), replace=False)
        for i in candidates:
            old_val = h_bin[i]
            h_bin[i] = 1.0 - old_val
            # Check sum constraint
            if h_bin[i] == 1.0 and np.sum(h_bin) > target:
                h_bin[i] = old_val
                continue
            if h_bin[i] == 0.0 and np.sum(h_bin) < target:
                h_bin[i] = old_val
                continue
            c = compute_c5(h_bin)
            h_bin[i] = old_val
            if bc - c > best_improve:
                best_improve = bc - c
                best_idx = i
        
        if best_idx < 0:
            break
        h_bin[best_idx] = 1.0 - h_bin[best_idx]
        bc = compute_c5(h_bin)
    
    return h_bin, bc
```

```python
        best_new_mc = mc
        best_i = -1
        best_dir = 0
        best_corr = None
        
        for i in candidates:
            old = h_b[i]
            # Compute delta for all shifts
            # delta_s = (1-2*old)*(1-h_b[i+s]) * valid1 + h_b[i-s]*(2*old-1) * valid2
            deltas = np.zeros(2*n - 1)
            
            # Contribution 1: i as first factor
            # Valid s: 0 ≤ i+s < n, i.e., -i ≤ s < n-i
            s_lo = max(-(n-1), -i)
            s_hi = min(n-1, n-1-i)
            if s_lo <= s_hi:
                s_range = np.arange(s_lo, s_hi + 1)
                # In the corr array, shift s is at index s + n - 1
                idx_range = s_range + n - 1
                deltas[idx_range] += (1 - 2*old) * (1 - h_b[i + s_range])
            
            # Contribution 2: i-s as first factor (i is second factor)
            # j = i-s, valid: 0 ≤ j < n, 0 ≤ j+s < n → 0 ≤ i-s < n, 0 ≤ i < n
            # s range: i-n+1 ≤ s ≤ i, intersected with [-(n-1), n-1]
            s_lo2 = max(-(n-1), i-n+1)
            s_hi2 = min(n-1, i)
            if s_lo2 <= s_hi2:
                s_range2 = np.arange(s_lo2, s_hi2 + 1)
                idx_range2 = s_range2 + n - 1
                j_range2 = i - s_range2  # = i - s
                deltas[idx_range2] += h_b[j_range2] * (2*old - 1)
            
            new_corr = corr + deltas
            new_mc = np.max(new_corr)
            if new_mc < best_new_mc:
                best_new_mc = new_mc
                best_i = i
                best_dir = (1 - 2*old)  # +1 if old=0 (0→1), -1 if old=1 (1→0)
                best_corr = new_corr
        
        if best_i < 0:
            break
        
        # Apply flip
        h_b[best_i] = 1.0 - h_b[best_i]
        corr = best_corr
        mc = best_new_mc
        bc = mc * dx
    
    return h_b, bc
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, s):
    n = len(h)
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    else:
        s = -s
        g[:n-s] -= h[s:]
        g[s:] += 1.0 - h[:n-s]
    return g
