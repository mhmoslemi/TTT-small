                if new_val < 0 or new_val > 1:

                m_vals = np.arange(k_start - j, k_end - j + 1)
                corr[k_start:k_end+1] += delta * (1.0 - 2.0 * h[m_vals])
                h[j] = new_val
                improved = True
        
        # Fix constraint
        h = make_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')  # Recompute to avoid drift
        
        if not improved:

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def smooth_max(corr, T):
    mc = np.max(corr)
    c = corr - mc
    return mc + np.log(np.sum(np.exp(T * c))) / T

def lbfgs_run(h, T, penalty, maxiter=300, time_limit=30):
    n = len(h)
    dx = 2.0 / n
    
    def _obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max(corr, T) * dx + penalty * (np.sum(x) - n/2.0)**2
    
    def _grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        p = np.exp(T * (corr - mc))
        s = np.sum(p)
        p /= s
        q = 1.0 - 2.0 * x
        conv = np.convolve(p, q[::-1], 'full')
        g = dx * conv[n-1:2*n-1]
        g += 2.0 * penalty * (np.sum(x) - n/2.0)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj, h, method='L-BFGS-B', jac=_grad, bounds=bounds,
                  options={'maxiter': maxiter, 'maxfun': 500, 'ftol': 1e-15})
    return res.x

    dx = 2.0 / n
    h = h.copy()
    
    for _ in range(n_passes):

            k_start = max(j, 0)
            k_end = min(j + n - 1, 2*n - 2)
            if k_start > k_end:

            m_vals = np.arange(k_start - j, k_end - j + 1)
            a = 1.0 - 2.0 * h[m_vals]
            corr_base = corr.copy()
            
            def eval_x(x):
                if x < 0 or x > 1:
                    return np.inf
                delta = x - old_val
                corr_new = corr_base + delta * a  # only the relevant slice changes
                # Wait, this isn't right. corr_base already has the old contribution.

                # So: corr_new = corr_base + delta * a (where a is zero-padded)
                # Actually a already has the right length and the full corr is:

                # corr_new[k_start:k_end+1] += delta * a

                sv = np.sum(h) - h[j] + x - n/2.0

_G = {'T': 300.0, 'penalty': 1e5, 'n': 960}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def smooth_max(corr, T):
    mc = float(np.max(corr))
    c = corr - mc
    return mc + float(np.log(np.sum(np.exp(T * c)))) / T

def _lbfgs_obj(x):
    T = _G['T']
    penalty = _G['penalty']
    n = _G['n']
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = float(np.max(corr))
    sm = mc + float(np.log(np.sum(np.exp(T * (corr - mc))))) / T
    return sm * (2.0 / n) + penalty * (np.sum(x) - n / 2.0) ** 2

def _lbfgs_grad(x):
    T = _G['T']
    penalty = _G['penalty']
    n = _G['n']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = float(np.max(corr))
    p = np.exp(T * (corr - mc))
    s = float(np.sum(p))
    p /= s
    q = 1.0 - 2.0 * x
    conv = np.convolve(p, q[::-1], 'full')
    g = dx * conv[n - 1:2 * n - 1]
    g += 2.0 * penalty * (np.sum(x) - n / 2.0)
    return g

    _G['T'] = T
    _G['penalty'] = penalty
    _G['n'] = n

    res = minimize(_lbfgs_obj, h, method='L-BFGS-B', jac=_lbfgs_grad,
                   bounds=bounds, options={'maxiter': maxiter, 'maxfun': 400,
                                           'ftol': 1e-15, 'gtol': 1e-10})
    return res.x

def coord_descent(h, T, n_passes=3, penalty=1e4):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for _ in range(n_passes):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        h_sum = float(np.sum(h))
        for j in range(n):
            old_val = h[j]
            k_start = j
            k_end = min(j + n - 1, 2 * n - 2)
            if k_start > k_end:
                continue
            m_vals = np.arange(k_start - j, k_end - j + 1)
            a = 1.0 - 2.0 * h[m_vals]
            rest_sum = h_sum - old_val
            lo = 0.0
            hi = 1.0
            for _ts in range(28):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                d1 = m1 - old_val
                d2 = m2 - old_val
                cn1 = corr.copy()
                cn1[k_start:k_end + 1] += d1 * a
                f1 = smooth_max(cn1, T) * dx + penalty * (rest_sum + m1 - n / 2.0) ** 2
                cn2 = corr.copy()
                cn2[k_start:k_end + 1] += d2 * a
                f2 = smooth_max(cn2, T) * dx + penalty * (rest_sum + m2 - n / 2.0) ** 2
                if f1 < f2:

                else:
                    lo = m1
            best_x = (lo + hi) / 2.0
            if abs(best_x - old_val) > 1e-10:
                delta = best_x - old_val
                corr[k_start:k_end + 1] += delta * a
                h[j] = best_x
                h_sum += delta
                improved = True
        h = make_feasible(h, n)
        if not improved:
            break
    return h
