    exp_s = np.exp(shifted)
    log_s = np.log(exp_s.sum())
    p = exp_s / exp_s.sum()

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def lse_obj_grad(h, n, tau, top_k=20):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c_top = corr[idx]
    else:
        idx = np.arange(len(corr))
        c_top = corr.copy()
    max_c = c_top.max()
    shifted = (c_top - max_c) / tau
    exp_s = np.exp(shifted)
    s_exp = exp_s.sum()
    p = exp_s / s_exp
    obj = dx * (max_c + tau * np.log(s_exp)) / tau
    grad = np.zeros(n)
    for ii in range(len(idx)):
        pk = p[ii]
        if pk < 1e-12:
            continue
        k = idx[ii]
        sh = k - n + 1
        if sh >= 0:
            if sh < n:
                grad[sh:] += pk * (1.0 - h[:n - sh])
                grad[:n - sh] -= pk * h[sh:]
        else:
            sp = -sh
            grad[:n - sp] += pk * (1.0 - h[sp:])
            grad[sp:] -= pk * h[:n - sp]
    grad *= dx
    return obj, grad, corr


def nesterov_opt(h, n, tau_init, tau_end, time_limit, lr_init=None, lr_end=None, top_k=20):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr_init is None:
        lr_init = 0.5 / tau_init
    if lr_end is None:
        lr_end = max(lr_init * 0.1, 1e-4)
    lr = lr_init
    h_prev = h.copy()
    t = 1.0
    stall = 0

    for it in range(5000000):
        if time.time() - start > time_limit:
            break
        elapsed = time.time() - start
        frac = min(elapsed / time_limit, 1.0)
        tau = tau_init * (tau_end / tau_init) ** frac
        lr_cur = lr_init * (lr_end / lr_init) ** frac
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj, grad, corr = lse_obj_grad(y, n, tau, top_k)
        h_new = y - lr_cur * grad
        h_new = project_to_feasible(h_new, n)
        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 2000:
                lr_init *= 0.5
                lr_end *= 0.5
                stall = 0
                if lr_init < 1e-16:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_polish(h, n, time_limit, top_k=5, lr_init=0.0008):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(5000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            break
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        max_v = top_vals.max()
        w = np.exp(top_vals - max_v)
        w /= w.sum()
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = w[ii]
            if pk < 1e-12:
                continue
            k = top_idx[ii]
            sh = k - n + 1
            if sh >= 0:
                if sh < n:
                    grad[sh:] += pk * (1.0 - h[:n - sh])
                    grad[:n - sh] -= pk * h[sh:]
            else:
                sp = -sh
                grad[:n - sp] += pk * (1.0 - h[sp:])
                grad[sp:] -= pk * h[:n - sp]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def try_binary_round(h, n):
    """Try rounding to binary and check if it improves C5."""
    target = n // 2
    thresh = np.sort(h)[::-1][target]
    h_bin = np.where(h >= thresh, 1.0, 0.0)
    if h_bin.sum() != target:
        # Fix count
        diff = target - h_bin.sum()
        if diff > 0:
            zero_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zero_idx])[:diff]
            h_bin[zero_idx[order]] = 1.0
        elif diff < 0:
            one_idx = np.where(h_bin == 1)[0]
            order = np.argsort(h[one_idx])[: -diff]
            h_bin[one_idx[order]] = 0.0
    c5_bin = compute_c5(h_bin)
    return h_bin, c5_bin


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))
