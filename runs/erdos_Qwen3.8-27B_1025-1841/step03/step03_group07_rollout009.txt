```python
import numpy as np
import time

# Helper functions (top-level)

def project_to_feasible(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def fft_corr(a, b):
    """FFT-based full cross-correlation"""
    na, nb = len(a), len(b)
    nfull = na + nb - 1
    N = 1
    while N < nfull:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = np.fft.irfft(A * np.conj(B), N)
    idx = (np.arange(nfull) - (nb - 1)) % N
    return C[idx]

def compute_c5(h):
    n = len(h)
    corr = fft_corr(h, 1.0 - h)
    return corr.max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def obj_grad(h, tau, top_k):
    """Soft-max objective and gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h, 1.0 - h)
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    shifted = tau * (top_vals - max_c)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    # Gradient of (1/tau) * log(sum(exp(tau * (c_k - c_max))))
    # d/dh = (1/tau) * sum_k p_k * dc_k/dh
    # dc_k/dh_j where k = shift:
    #   corr[k] = sum_i h[i] * (1-h[i+k-(n-1)]) * dx
    #   Actually corr[k] in our indexing: corr[k] = sum_i h[i] * (1-h[i + k - n + 1]) * dx
    #   Let sh = k - n + 1
    #   corr[k] = sum_i h[i] * (1 - h[i+sh]) * dx (for valid i, i+sh)
    #   d/dh_j: from h[j] term: (1-h[j+sh])*dx (if j+sh valid)
    #             from (1-h[j]) term in h[i]*... : -h[i]*dx where i+sh=j, i=j-sh
    #             so -h[j-sh]*dx (if j-sh valid)
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx] * dx
        k = top_idx[i_idx]
        sh = k - n + 1
        # h[j] * (1 - h[j+sh]): contributes (1-h[j+sh]) to d/dh[j] for j+sh in [0,n)
        # (1-h[j]) * h[j-sh]: contributes -h[j-sh] to d/dh[j] for j-sh in [0,n)
        # Term 1: for j in [0, n), j+sh in [0, n) → j in [max(0,-sh), min(n, n-sh))
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j1 > j0:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        # Term 2: for j in [0, n), j-sh in [0, n) → j in [max(0,sh), min(n, n+sh))
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j1b > j0b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def nesterov(h, n, tau, time_limit, lr=None, top_k=10):
    """Nesterov accelerated gradient with soft-max smoothing"""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0

    for it in range(5000000):
        if time.time() - start > time_limit:
            break
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj, grad = obj_grad(y, tau, top_k)
```

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 3.0
    hi = g.max() + 3.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def fft_corr(a, b):
    na, nb = len(a), len(b)
    nfull = na + nb - 1
    N = 1
    while N < nfull:
        N <<= 1
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b, N)
    C = np.fft.irfft(A * np.conj(B), N)
    idx = (np.arange(nfull) - (nb - 1)) % N
    return C[idx]


def compute_c5(h):
    n = len(h)
    corr = fft_corr(h, 1.0 - h)
    return corr.max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def obj_grad(h, tau, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h, 1.0 - h)
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    shifted = np.clip(tau * (top_vals - max_c), -60.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx] * dx
        k = int(top_idx[i_idx])
        sh = k - n + 1

        j1b = min(n, n + sh)
        if j1b > j0b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

        h_p = project_to_feasible(h_p, nb)
        tl = min(50, r * 0.3)
        h_p, c5_p = nesterov_opt(h_p, nb, 25.0, tl, 0.035, 8)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p

    # Phase 7: Low-tau Nesterov refinement
    r = rem_time(start_time, budget_s)
    nb = len(best_h)
    if r > 60:
        tl = min(55, r * 0.3)
        h1, c5 = nesterov_opt(best_h, nb, 6.0, tl, 0.008, 5)
        if c5 < best_c5:
            best_c5, best_h = c5, h1

    # Phase 8: Subgradient polish
    r = rem_time(start_time, budget_s)
    nb = len(best_h)
    if r > 40:
        tl = min(35, r * 0.45)
        h1, c5 = subgrad_polish(best_h, nb, tl, 0.0008, 4)
        if c5 < best_c5:
            best_c5, best_h = c5, h1

    # Phase 9: Final tiny subgradient
    r = rem_time(start_time, budget_s)
    nb = len(best_h)
    if r > 15:
        tl = min(12, r * 0.6)
        h1, c5 = subgrad_polish(best_h, nb, tl, 0.0002, 2)
        if c5 < best_c5:
            best_c5, best_h = c5, h1

    # Final validation
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
