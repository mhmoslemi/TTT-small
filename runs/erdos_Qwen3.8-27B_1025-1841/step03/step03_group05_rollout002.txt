import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def compute_all_corrs(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def sm_grad(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter=5000, T0=40.0, T1=0.1, lr=0.15, check=40, top_k=80):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.4 * f)

        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0

        tk = tn
        if (it + 1) % check == 0:

            if c < bc:

    c = compute_c5(x)
    if c < bc:

    return bh, bc

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

    T = _SLSQP['T']

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
