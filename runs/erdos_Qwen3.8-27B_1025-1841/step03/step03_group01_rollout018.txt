```python
def compute_c5_fft(h):

    n = len(h)

    size = 2 * n
```

```python
def project_sum(h, target_sum):
    """Project h to {h : sum h = target_sum, 0 ≤ h ≤ 1}"""
    n = len(h)
    h = np.clip(h, 0, 1).copy()
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to increase sum: add to coordinates < 1
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0, 1)
    else:
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0, 1)
    # Second pass if needed
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0, 1)
    return h
```

Better approach: For each coordinate i, I can analytically determine the effect on c_k:

c_k(h) = sum_j h_j (1-h_{j+k})

When I change h_i from old to new:
- Term j=i: changes from old*(1-h_{i+k}) to new*(1-h_{i+k}), delta = (new-old)*(1-h_{i+k})
- Term where j+k = i, i.e., j = i-k: changes from h_{i-k}*(1-old) to h_{i-k}*(1-new), delta = (new-old)*h_{i-k}

So c_k changes by: (new-old) * [(1-h_{i+k}) + h_{i-k}] (with boundary handling)

Let me define: conv_k(i) = (1-h_{i+k}) + h_{i-k} (the "convolution coefficient" at position i, lag k)

Then: c_k(new_h) = c_k(old_h) + (new-old) * conv_k(i)

And the new c5 = max_k (c_k(old_h) + (new-old) * conv_k(i))

This is the maximum of linear functions in (new-old)! So for a fixed i, I can find the optimal new value by minimizing:
max_k (c_k + t * conv_k(i)) over t = new - old, where new ∈ [0,1]

Plus the sum constraint: we need to adjust other coordinates. But if we ignore the sum constraint for now (or handle it separately), the 1D optimization is:

min_{t} max_k (c_k + t * conv_k(i))  subject to old + t ∈ [0, 1]

This is 1D convex piecewise linear! We can solve it by:
1. Evaluating at t = -old (new=0) and t = 1-old (new=1)
2. Finding the minimum of the piecewise linear function

The minimum of max of linear functions occurs either at a breakpoint or at a boundary. The breakpoints are where c_k + t*conv_k(i) = c_l + t*conv_l(i), i.e., t = (c_l - c_k) / (conv_k(i) - conv_l(i)).

There are O(K²) potential breakpoints where K = 2n-1. But we don't need to check all of them. Since the function is convex, we can use bisection on the subgradient.

Subgradient at t: {conv_k(i) : k ∈ argmax_k(c_k + t*conv_k(i))}

The minimum is where 0 ∈ subgradient, i.e., there exists k* in argmax such that conv_k*(i) ≤ 0 ≤ conv_l(i) for all l in argmax.

Simpler numerical approach:
- Evaluate f(t) = max_k(c_k + t * conv_k(i)) at t = 0, and determine the argmax
- If conv_argmax < 0: f is decreasing at t=0, so try larger t
- If conv_argmax > 0: f is increasing at t=0, so try smaller t
- Bisection: about 30-40 iterations to get precision 1e-10

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def compute_conv_matrix(h):
    n = len(h)
    conv = np.zeros((2*n-1, n))
    for k in range(n):
        conv[k, k:] += (1.0 - h[:n-k])
        conv[k, :n-k] -= h[k:]
    for d in range(1, n):
        k_idx = n + d - 1
        conv[k_idx, d:] += (1.0 - h[:n-d])
        conv[k_idx, :n-d] -= h[d:]
    return conv

def cd_optimize_coord(corr, conv_col, h_i):
    """Find optimal h_i minimizing max_k(corr + t*conv) for t in [-h_i, 1-h_i]"""
    t_lo = -h_i
    t_hi = 1.0 - h_i
    if t_lo == t_hi:
        return h_i
    
    # Evaluate at endpoints and 0
    f_lo = np.max(corr + t_lo * conv_col)
    f_hi = np.max(corr + t_hi * conv_col)
    f_0 = np.max(corr)
    
    best_t = 0.0
    best_f = f_0
    if f_lo < best_f:
        best_t, best_f = t_lo, f_lo
    if f_hi < best_f:
        best_t, best_f = t_hi, f_hi
    
    # Bisection for interior minimum
    lo, hi = t_lo, t_hi
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = corr + mid * conv_col
        idx = np.argmax(vals)
        if conv_col[idx] < -1e-14:
            lo = mid
        elif conv_col[idx] > 1e-14:
            hi = mid
        else:
            lo = hi = mid
            break
    t_cand = (lo + hi) * 0.5
    f_cand = np.max(corr + t_cand * conv_col)
    if f_cand < best_f - 1e-15:
        return h_i + t_cand
    return h_i + best_t

def coordinate_descent(h, n_sweeps, time_limit_fn):
    n = len(h)
    target = n / 2.0
    h = h.copy()
    c5 = compute_c5(h)

def _slsqp_obj(x):

    T = _G_T
    dx = 2.0 / n

    mx = np.max(corr)

    max_shifted = np.max(vals)
    lse = np.log(np.sum(np.exp(vals - max_shifted))) + max_shifted
    return (mx + lse / T) * dx

    return np.sum(x) - _G_N / 2.0

    return np.ones(_G_N)
```

```python
def compute_conv_matrix(h):
    n = len(h)
    conv = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        lag = m - (n-1)
        if lag >= 0:
            # First term: i >= lag, contributes (1-h[i-lag])
            if lag < n:
                conv[m, lag:] += (1.0 - h[:n-lag])
            # Second term: i <= n-1-lag, contributes -h[i+lag]
            if n-1-lag >= 0:
                conv[m, :n-lag] -= h[lag:]
        else:
            # lag < 0, let d = -lag > 0

            # First term: i <= n-1-d, contributes (1-h[i+d])
            if n-1-d >= 0:
                conv[m, :n-d] += (1.0 - h[d:])
            # Second term: i >= d, contributes -h[i-d]
            if d < n:
                conv[m, d:] -= h[:n-d]
```

    return (mx + lse / T) * dx
