    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff/cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h[np.argmin(h)] += diff if diff > 0 else 0
        h[np.argmax(h)] += diff if diff < 0 else 0
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h, new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def lse_objective(h, n, T, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -700, 700)
    lse = mc + np.log(np.sum(np.exp(shifted))) / T
    viol = np.sum(h) - n/2.0
    return lse * (2.0/n) + mu * viol * viol

def lse_grad(h, n, T, mu):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -700, 700)
    e = np.exp(shifted)
    w = e / e.sum()
    
    grad = np.zeros(n)
    # Only process lags where w is significant

        bk = w[k]
        if bk < 1e-12:
            continue
        d = k - (n-1)
        # First term: j where 0<=j<n and 0<=j+d<n
        js = max(0, -d)
        je = min(n-1, n-1-d)
        if js <= je:
            grad[js:je+1] += bk * (1.0 - h[js+d:je+1+d])
        # Second term: j where 0<=j-d<n
        js2 = max(0, d)
        je2 = min(n-1, d+n-1)
        if js2 <= je2:
            grad[js2:je2+1] -= bk * h[js2-d:je2+1-d]
    
    grad *= (2.0/n)
    # Add constraint gradient
    grad += 2*mu*(np.sum(h) - n/2.0)
    return grad

def lbgfb_run(h, n, T, mu, max_iter):
    def f(x):
        return lse_objective(x, n, T, mu)
    def g(x):
        return lse_grad(x, n, T, mu)
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-10, 'maxls': 50})
    return make_feasible(res.x, n)

        return (mc + np.log(np.sum(np.exp(shifted)))/T) * dx

        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con, 'jac': jac_con}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-16})
    return make_feasible(res.x, n)
