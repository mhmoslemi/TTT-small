import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    n_ones = n // 2
    # Get top n_ones values
    indices = np.argsort(h)[::-1][:n_ones]
    result = np.zeros(n)
    result[indices] = 1.0
    return result

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def simulated_annealing_binary(h_bin, n, rng, start_time, budget_s, max_time_frac=0.7):
    """Simulated annealing on binary h with swap moves."""
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    n_iter = 0
    max_time = max_time_frac * budget_s
    
    # Temperature schedule
    T_start = 0.1
    T_end = 0.001
    
    while (time.time() - start_time) < (budget_s - max_time):
        n_iter += 1
        # Temperature
        t_frac = min(1.0, n_iter / max(1, int(max_time / 0.01)))  # rough progress
        T = T_start * (T_end / T_start) ** t_frac
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            # Update ones/zeros
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

    """Convert to binary with exactly n/2 ones using threshold + adjustment."""
    h = np.asarray(h, dtype=float)
    n_ones = n // 2
    # Sort by value, take top n_ones
    idx = np.argsort(h)[::-1][:n_ones]
    result = np.zeros(n)
    result[idx] = 1.0
    return result

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

Wait, let me recalculate. If I swap one bit, how much can c[k] change?
c[k] = sum_{t} h[t+k]*(1-h[t])
Changing h[i] from 1 to 0: affects c[k] for all k where i is in the valid range. Specifically:
- As the "first" element: (1-h[i]) changes from 0 to 1, so c[k] changes by +h[i+k] for all valid k
- As the "second" element: h[i] changes from 1 to 0, so c[k] changes by -(1-h[i-k]) for all valid k

The maximum change in any single c[k] is at most 2 (one from being the first element, one from being the second). So the maximum change in C₅ = max(c[k]) * dx is at most 2 * dx = 4/n.

For n=2048: max change in C₅ per swap = 4/2048 ≈ 0.002.
For n=4096: max change = 4/4096 ≈ 0.001.

So the "step size" in C₅ space is about 0.001-0.002. The total range of C₅ we care about is about 0.38-0.5, so there are about 100-200 "steps" in the interesting range.

For SA to work well, the temperature should be comparable to the typical step size. So T_start ≈ 0.005-0.01 and T_end ≈ 0.0001-0.001 would be appropriate.

    """Generate first n bits of Thue-Morse sequence."""
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()

        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    # Force exact
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            for k in range(n):
                room = 1.0 - h[k]
                add = min(room, diff)
                h[k] += add
                diff -= add
                if abs(diff) < 1e-15:
                    break

            for k in range(n-1, -1, -1):

                rem = min(room, -diff)

        c5_2 = min(c5_2c, c5_2b)
        
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = candidate.copy()
        
        # Quick SA at higher resolution
        if tl() > 80 and (best_h == 1).any():
            h2b = to_binary(best_h, n2)
            sa_budget = min(tl() * 0.4, 100)
