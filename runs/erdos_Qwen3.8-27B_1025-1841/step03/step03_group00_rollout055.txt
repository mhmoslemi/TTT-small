    return h, current

def run(seed=42, budget_s=1300, **kwargs):
    global _lbgfb_T, _lbgfb_mu, _lbgfb_n, _slsqp_T, _slsqp_n
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)
    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Multi-temperature L-BFGS-B at native resolution
    T_list = [15, 40, 100, 250, 600, 1500, 4000]
    for T in T_list:
        if time_left() < 40:
            break
        h_work = best_h.copy()
        for r in range(3):
            if time_left() < 25:
                break
            if r == 1:
                h_work = fourier_perturb(best_h, n0, rng, 8, 0.25)
            elif r == 2:
                h_work = best_h + 0.08 * rng.randn(n0)
                h_work = make_feasible(h_work, n0)
            try:
                h_opt = lbgfb_opt(h_work, n0, T, 500.0, max_iter=250)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5 - 1e-12:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_work = h_opt.copy()
            except Exception:
                pass

    # Phase 2: Binary strategies at native resolution
    if time_left() > 80:
        h_bin = (best_h > 0.5).astype(float)
        # Strict count fix
        n_target = n0 // 2
        n_ones = int(np.sum(h_bin))
        if n_ones > n_target:
            idx = np.where(h_bin == 1.0)[0]
            rng.shuffle(idx)
            h_bin[idx[:n_ones - n_target]] = 0.0
        elif n_ones < n_target:
            idx = np.where(h_bin == 0.0)[0]
            rng.shuffle(idx)
            h_bin[idx[:n_target - n_ones]] = 1.0

        # Annealing rounds
        for nsw in [50000, 80000]:
            if time_left() < 40:
                break
            h_a, c5_a = annealed_binary(h_bin, n0, rng, n_swaps=nsw)
            if c5_a < best_c5 - 1e-12:
                best_c5 = c5_a
                best_h = h_a.copy()
                h_bin = h_a.copy()

        # Local search
        if time_left() > 30:
            h_ls, c5_ls = local_binary_search(h_bin, n0, rng, max_rounds=50, tries=4000)
            if c5_ls < best_c5 - 1e-12:
                best_c5 = c5_ls
                best_h = h_ls.copy()

        # Also try annealing from a fresh binary start
        if time_left() > 40:
            h_rand = (rng.rand(n0) < 0.5).astype(float)
            h_rand = make_feasible(h_rand, n0)
            h_rand = (h_rand > 0.5).astype(float)
            n_ones = int(np.sum(h_rand))
            if n_ones != n_target:
                if n_ones > n_target:
                    idx = np.where(h_rand == 1.0)[0]
                    rng.shuffle(idx)
                    h_rand[idx[:n_ones - n_target]] = 0.0
                else:
                    idx = np.where(h_rand == 0.0)[0]
                    rng.shuffle(idx)
                    h_rand[idx[:n_target - n_ones]] = 1.0
            h_a, c5_a = annealed_binary(h_rand, n0, rng, n_swaps=30000, T_start=0.5, T_end=0.01)
            if c5_a < best_c5 - 1e-12:
                best_c5 = c5_a
                best_h = h_a.copy()

    # Phase 3: Upsample to higher resolutions and optimize
    for n_up in [800, 1024, 1280, 1600]:
        if time_left() < 80:
            break
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 - 1e-12:
            best_c5 = c5_up
            best_h = h_up.copy()
            h_up = h_up.copy()

        for T in [40, 150, 500, 1500]:
            if time_left() < 30:
                break
            try:
                h_opt = lbgfb_opt(h_up, n_up, T, 300.0, max_iter=200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5 - 1e-12:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_up = h_opt.copy()
            except Exception:
                pass

        if time_left() > 30:
            h_f = fourier_perturb(h_up, n_up, rng, 10, 0.2)
            try:
                h_opt = lbgfb_opt(h_f, n_up, 300, 300.0, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5 - 1e-12:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 4: SLSQP polish at moderate resolution
    n_s = min(len(best_h), 800)
    h_s = upsample_h(best_h, n_s) if n_s < len(best_h) else best_h.copy()
    if time_left() > 40:
        for T in [400, 1500, 5000]:
            if time_left() < 15:
                break
            try:
                h_opt = slsqp_opt(h_s, n_s, T, max_iter=80)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5 - 1e-12:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_s = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Coordinate descent polish
    n_cd = min(len(best_h), 800)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    if time_left() > 25:
        h_cd, c5_cd = coord_descent_polish(h_cd, n_cd, max_rounds=4)
        if c5_cd < best_c5 - 1e-12:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 6: Final L-BFGS-B squeeze at best resolution
    if time_left() > 20:
        n_f = len(best_h)
        try:
            h_opt = lbgfb_opt(best_h, n_f, 5000, 2000.0, max_iter=200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5 - 1e-12:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Final feasibility and evaluation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
