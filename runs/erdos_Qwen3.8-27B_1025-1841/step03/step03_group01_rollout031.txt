        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n, period=2.0):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=period)
    return make_feasible(h_new, new_n)

def compute_correlations(h):
    """Compute full cross-correlation of h with 1-h"""
    return np.correlate(h, 1.0 - h, mode='full')

def logsumexp_objective(h, T):
    """Smoothed max of correlations * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx
