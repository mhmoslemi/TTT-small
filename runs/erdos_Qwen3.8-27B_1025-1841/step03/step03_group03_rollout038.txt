from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones, using the fractional parts."""
    h = np.asarray(h, dtype=float)
    # Sort by fractional part descending, pick top n/2
    frac = h - np.floor(h)
    # For values exactly 0 or 1, frac is 0
    # We want to pick n/2 positions to be 1
    order = np.argsort(-frac)  # descending by fractional part
    h_bin = np.zeros(n)
    h_bin[order[:n//2]] = 1.0
    return h_bin

def pairwise_cd(h, n, n_iterations, time_limit, start_time, rng):
    best_c = compute_c5(h)
    
    for it in range(n_iterations):
        if time.time() - start_time > time_limit:
            break
        
        j1 = rng.randint(n)
        j2 = rng.randint(n)
        if j1 == j2:
            continue
        
        v1, v2 = h[j1], h[j2]
        target = v1 + v2
        
        lo = max(0.0, target - 1.0)
        hi = min(1.0, target)
        
        if lo >= hi:
            continue
        
        # Grid search + refinement
        # First: coarse grid
        candidates = np.linspace(lo, hi, 21)
        best_t = v1
        best_val = np.inf
        for t in candidates:
            h[j1] = t
            h[j2] = target - t
            c = compute_c5(h)
            if c < best_val:
                best_val = c
                best_t = t
        
        # Refine around best
        if best_t > lo and best_t < hi:
            width = (hi - lo) / 20
            candidates = np.linspace(best_t - width, best_t + width, 11)
            candidates = np.clip(candidates, lo, hi)
            for t in candidates:
                h[j1] = t
                h[j2] = target - t
                c = compute_c5(h)
                if c < best_val:
                    best_val = c
                    best_t = t
        
        h[j1] = best_t
        h[j2] = target - best_t
        
        if best_val < best_c:
            best_c = best_val
    
    return h, best_c
