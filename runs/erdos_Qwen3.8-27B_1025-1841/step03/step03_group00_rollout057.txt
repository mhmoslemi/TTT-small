```python
def project_feasible(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(100):
        lam = (lo + hi) / 2
        h = np.clip(g + lam, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            hi = lam
        elif s < target:
            lo = lam
        else:
            break
    return np.clip(g + (lo+hi)/2, 0.0, 1.0)
```

```python
def binary_swap_search(h0, n, rng, max_rounds=50, n_tries_per_round=5000):

    h = h0.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]

        n_ones = len(ones)
        n_zeros = len(zeros)
        n_tries = min(n_tries_per_round, n_ones * n_zeros)
        for _ in range(n_tries):
            ii = rng.randint(n_ones)
            jj = rng.randint(n_zeros)
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:

                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:

                h[oj] = 0.0

        return logsum * dx + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def coordinate_descent(h0, n, n_rounds=5, dt=0.02):
    """Coordinate descent with binary and continuous moves."""
    h = h0.copy()
    c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            # Try continuous moves
            for d in [dt, -dt, dt/2, -dt/2]:
                nv = h[j] + d
                if nv < 0 or nv > 1:
                    continue
                h[j] = nv
                h2 = make_feasible(h, n)
                c = compute_c5(h2)
                if c < c5 - 1e-14:
                    c5 = c
                    h = h2
                    improved = True
                    break
                else:
                    h[j] = nv  # restore (make_feasible might have changed others)
            # Try binary
            for bv in (0.0, 1.0):
                if abs(h[j] - bv) < 1e-9:
                    continue
                h[j] = bv
                h2 = make_feasible(h, n)
                c = compute_c5(h2)
                if c < c5 - 1e-14:
                    c5 = c
                    h = h2
                    improved = True
        if not improved:
            break
    return h, c5

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
```
