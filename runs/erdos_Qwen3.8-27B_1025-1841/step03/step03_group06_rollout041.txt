import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def lse_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    s = corr - mc
    return mc + np.log(np.sum(np.exp(T * s))) / T

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def coord_descent(h, n, max_rounds, remaining_fn, n_cands=20):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if remaining_fn() < 10:
            break
        improved = False
        for i in range(n):
            if remaining_fn() < 5:
                break
            old = h[i]
            best_v = old
            best_c = best_c5
            # Generate candidates
            cands = np.unique(np.clip(np.round(np.linspace(0, 1, n_cands), 4), 0, 1))
            # Add current value and some fine values around it
            extra = np.clip(old + np.array([-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]), 0, 1)
            cands = np.unique(np.concatenate([cands, extra, [0.0, 1.0, 0.5]]))
            for v in cands:
                if abs(v - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[i] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_v != old:
                h[i] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def sa_binary(h, n, remaining_fn, seed=42):
    rng = np.random.RandomState(seed)
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    target = n // 2
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    while len(ones) > target:
        idx = ones.pop()
        h_bin[idx] = 0.0
        zeros = np.append(zeros, idx)
    while len(ones) < target:
        idx = zeros.pop()
        h_bin[idx] = 1.0
        ones = np.append(ones, idx)
    
    cur_c5 = compute_c5(h_bin)
    best_c5 = cur_c5
    best_h = h_bin.copy()
    
    T = 0.02
    T_min = 0.0005
    cool = 0.9995
    
    while remaining_fn() > 10 and T > T_min:
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        # Update ones/zeros
        ones = ones[ones != i]
        ones = np.append(ones, j)
        zeros = zeros[zeros != j]
        zeros = np.append(zeros, i)
        
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - cur_c5

                best_h = h_bin.copy()
        else:

            ones = ones[ones != j]
            ones = np.append(ones, i)
            zeros = zeros[zeros != i]
            zeros = np.append(zeros, j)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def lse_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    s = corr - mc
    return mc + np.log(np.sum(np.exp(T * s))) / T

def sum_con(x):
    return np.sum(x) - len(x) / 2.0

def sum_jac(x):
    return np.ones(len(x))

def to_binary(h, n):
    """Convert to binary, maintaining exactly n/2 ones."""
    h = np.where(h >= 0.5, 1.0, 0.0)
    target = n // 2
    num_ones = int(np.sum(h))
    if num_ones > target:
        ones = np.where(h == 1.0)[0]
        # Sort by value (most "committed" to 1 first)
        # Actually just take the last ones
        for i in range(num_ones - target):
            h[ones[i]] = 0.0
    elif num_ones < target:
        zeros = np.where(h == 0.0)[0]
        for i in range(target - num_ones):
            h[zeros[i]] = 1.0
    return h

def sa_binary(h, n, remaining_fn, seed=42, t_start=0.005, t_min=0.00005, n_restarts=3):
    """Simulated annealing on binary h."""
    rng = np.random.RandomState(seed)
    h_bin = to_binary(h, n)
    cur_c5 = compute_c5(h_bin)
    best_c5 = cur_c5
    best_h = h_bin.copy()
    
    for restart in range(n_restarts):
        if remaining_fn() < 20:
            break
        T = t_start
        no_improve = 0
        max_no_improve = 20000
        
        while T > t_min and no_improve < max_no_improve:
            if remaining_fn() < 15:
                break
            
            # Get current ones and zeros
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
            
            # Random pair swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            
            # Swap
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - cur_c5
            
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c5 = new_c5
                if new_c5 < best_c5 - 1e-15:
                    best_c5 = new_c5
                    best_h = h_bin.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                # Revert
                h_bin[i] = 1.0
                h_bin[j] = 0.0
                no_improve += 1
            
            # Cool
            T *= 0.9998
    
    return best_h, best_c5

def coord_descent_fast(h, n, remaining_fn, max_rounds=10):
