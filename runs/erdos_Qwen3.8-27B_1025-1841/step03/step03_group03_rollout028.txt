    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    C = np.correlate(h, 1-h, 'full')  # length 2n-1
    current_c5 = C.max()  # will multiply by dx at the end
    
    for round in range(max_rounds):
        if time.time() - start_time > time_limit - 30:
            break
        
        best_swap = None
        best_c5 = current_c5
        
        # Try a subset of swaps
        # For each 1-position, try a few random 0-positions
        n_tries = min(200, len(ones))
        idx = np.random.choice(len(ones), n_tries, replace=False)
        
        for idx_i in idx:
            i = ones[idx_i]
            # Try a few random zeros
            n_zero_tries = min(50, len(zeros))
            idx_j = np.random.choice(len(zeros), n_zero_tries, replace=False)
            for idx_j_val in idx_j:
                j = zeros[idx_j_val]
                # Compute delta
                m = np.arange(-(n-1), n)  # all lags
                # h[i+m] where valid
                im = i + m
                im[jm = i-m]
                jm = j + m
                jm2 = j - m
                delta = np.zeros(2*n-1)
                # h[i+m] where 0 <= i+m < n
                mask1 = (im >= 0) & (im < n)
                delta[mask1] += h[im[mask1]]
                # h[i-m] where 0 <= i-m < n
                mask2 = (im2 >= 0) & (im2 < n)
                delta[mask2] += h[im2[mask2]]
                # -h[j+m]
                mask3 = (jm >= 0) & (jm < n)
                delta[mask3] -= h[jm[mask3]]
                # -h[j-m]
                mask4 = (jm2 >= 0) & (jm2 < n)
                delta[mask4] -= h[jm2[mask4]]
                
                new_c5 = (C + delta).max()
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_swap = (i, j, delta)
        
        if best_swap is not None:
            i, j, delta = best_swap
            h[i], h[j] = 0, 1
            C = C + delta
            current_c5 = best_c5
            # Update ones and zeros
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
            break
    
    return h, current_c5 * (2.0/n)

Let me use the bottleneck test approach:
- m_star = argmax(C), lag_star = m_star - (n-1)
- For a swap (i,j) to be improving, we need C(m_star) + Δ(m_star) < current_max
- Δ(m_star) = h[i+lag_star] + h[i-lag_star] - h[j+lag_star] - h[j-lag_star]
- Since C(m_star) = current_max, we need Δ(m_star) < 0
- This is an O(1) check per pair!

So the algorithm becomes:
1. For each i (320 or 1024 positions):
   a. Compute left_i = h[i+lag_star] + h[i-lag_star] (O(1))
   b. If left_i >= 2, skip this i (can't get Δ < 0)
   c. For each j (320 or 1024 positions):
      i. Compute right_j = h[j+lag_star] + h[j-lag_star] (O(1))
      ii. If left_i >= right_j, skip
      iii. Full evaluation: O(corr_len)

Step 1b and 1c can be vectorized:
- For all i: left_all = h[i+lag_star] + h[i-lag_star] (vectorized, O(n))
- For all j: right_all = h[j+lag_star] + h[j-lag_star] (vectorized, O(n))
- Valid (i,j) pairs: left_all[i] < right_all[j]

The number of valid pairs is roughly (n/2) * (n/2) * P(left < right). Since left and right are each 0, 1, or 2:
- P(left=0) ≈ 0.25, P(left=1) ≈ 0.5, P(left=2) ≈ 0.25 (for random binary)
- P(right=0) ≈ 0.25, P(right=1) ≈ 0.5, P(right=2) ≈ 0.25
- P(left < right) = P(left=0,right≥1) + P(left=1,right=2) = 0.25*0.75 + 0.5*0.25 = 0.1875 + 0.125 = 0.3125

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def binarize_h(h, n):
    """Binarize h to have exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    target = n // 2
    # Threshold: top target values become 1
    threshold = np.sort(h)[::-1][target - 1] if target > 0 else 1.0
    h_bin = (h >= threshold).astype(float)
    # Fix sum
    s = int(np.sum(h_bin))
    if s > target:
        # Turn some 1s to 0s (lowest valued first)
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(-h[one_idx])  # ascending by value
        for k in range(s - target):
            h_bin[one_idx[order[k]]] = 0
    elif s < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])  # descending by value
        for k in range(target - s):
            h_bin[zero_idx[order[k]]] = 1
    return h_bin

def upsample_binary(h, factor):
    """Upsample binary h by given integer factor."""
    return np.repeat(h, factor)

def compute_correlation(h):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def compute_delta(h, i, j, n, m_vals):
    """Compute delta for swap (i: 1->0, j: 0->1)."""
    corr_len = 2 * n - 1
    delta = np.zeros(corr_len)
    pm = i + m_vals
    mask = (pm >= 0) & (pm < n)
    delta[mask] += h[pm[mask]]
    pm2 = i - m_vals
    mask2 = (pm2 >= 0) & (pm2 < n)
    delta[mask2] += h[pm2[mask2]]
    pm = j + m_vals
    mask = (pm >= 0) & (pm < n)
    delta[mask] -= h[pm[mask]]
    pm2 = j - m_vals
    mask2 = (pm2 >= 0) & (pm2 < n)
    delta[mask2] -
