```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not np.any(mask):
            break
        step = diff / np.count_nonzero(mask)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            h[np.argmin(h)] = np.clip(h[np.argmin(h)] + diff, 0, 1)
        else:
            h[np.argmax(h)] = np.clip(h[np.argmax(h)] + diff, 0, 1)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            h[np.argmin(h)] = np.clip(h[np.argmin(h)] + diff, 0, 1)
        else:
            h[np.argmax(h)] = np.clip(h[np.argmax(h)] + diff, 0, 1)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def get_top_k_indices(corr, K):
    return np.argsort(corr)[-K:]

def subgradient_step(h, n, K, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-K:]
    corr_top = corr[top_k]
    max_corr = np.max(corr)
    active_mask = corr_top >= max_corr * dx - 1e-8
    if np.count_nonzero(active_mask) == 0:
        active_mask = corr_top >= np.max(corr_top) - 1e-10
    active_k = top_k[active_mask]
    grad = np.zeros(n)
    for m in active_k:
        m_int = int(m)
        k_lo = max(0, n - 1 - m_int)
        k_hi = min(n - 1, 2 * n - 2 - m_int)
        idx = np.arange(k_lo, k_hi + 1)
        partner = idx + m_int - n + 1
        grad[idx] += (1.0 - h[partner])
        grad[partner] += -h[idx]
    grad /= np.count_nonzero(active_mask)
    h_new = h + alpha * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    return h_new

def coordinate_descent(h, n, max_passes, time_budget, start_time, K=30):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for passe in range(max_passes):
        if time.time() - start_time > time_budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        top_idx = np.argsort(corr)[-K:]
        improved = False
        for j in range(n):
            if time.time() - start_time > time_budget:
                break
            old = h[j]
            deltas = [-old, 1.0 - old, -0.05, 0.05, -0.1, 0.1, -0.02, 0.02]
            deltas = [d for d in deltas if d != 0]
            best_delta = 0
            best_val = max_c
            for d in deltas:
                nv = old + d
                if nv < 0 or nv > 1:
                    continue
                h[j] = nv
                c = np.correlate(h, 1.0 - h, mode='full')
                mv = np.max(c)
                h[j] = old
                if mv < best_val - 1e-14:
                    best_val = mv
                    best_delta = d
            if best_delta != 0:
                h[j] = np.clip(h[j] + best_delta, 0, 1)
                h = make_feasible(h, n)
                mc = compute_c5(h)
                if mc < best_c5 - 1e-14:
                    best_c5 = mc
                    improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_topk(h_start, n, K, max_iter):
    target_sum = n / 2.0
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top = np.sort(corr)[-K:]
        return float(np.mean(top) * dx + 0.01 * np.max(corr) * dx)
    def obj_grad(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-K:]
        max_k = int(np.argmax(corr))
        grad = np.zeros(n)
        for m in top_k:
            m_int = int(m)
            k_lo = max(0, n - 1 - m_int)
            k_hi = min(n - 1, 2 * n - 2 - m_int)
            idx = np.arange(k_lo, k_hi + 1)
            partner = idx + m_int - n + 1
            grad[idx] += (1.0 - h[partner])
            grad[partner] += -h[idx]
        if max_k not in top_k:
            m_int = max_k
            k_lo = max(0, n - 1 - m_int)
            k_hi = min(n - 1, 2 * n - 2 - m_int)
            idx = np.arange(k_lo, k_hi + 1)
            partner = idx + m_int - n + 1
            grad[idx] += (1.0 - h[partner])
            grad[partner] += -h[idx]
        grad = grad / K * dx + 0.01 * dx * grad / K
        return grad
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum,
             'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_start, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-16})
    return make_feasible(res.x, n)

    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.zeros(640)
        h0[::2] = 1.0
    n0 = len(h0)

    def tl():

    phases = [

        (min(n0 * 2, 1280), 0.15),

    for phase_idx, (n_pts, time_frac) in enumerate(phases):

        if phase_idx > 0:

        else:
            h_curr = best_h.copy()
        c_curr = compute_c5(h_curr)
        if c_curr < best_c5:

        # Subgradient descent phase
        n_sub_steps = 200 if n_pts <= 640 else 100

            alpha = alpha_start * (0.95 ** s)

            h_curr = subgradient_step(h_curr, n_pts, K=10, alpha=alpha)

            if c < best_c5 - 1e-14:

                best_h = h_curr.copy()

        # SLSQP with top-K

                                   max_iter=300 if n_pts <= 640 else 150)
```
