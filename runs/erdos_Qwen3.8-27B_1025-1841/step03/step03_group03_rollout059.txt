def upsample_h(h, new_n):

    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)

def lse_and_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    obj = (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx
    w = np.exp(t * shifted)
    wsum = np.sum(w)
    w /= wsum
    grad = np.zeros(n)

            grad[s:] -= wm * h[:v]
        else:

            v = n - sp
            grad[sp:] += wm * A[:v]
            grad[:v] -= wm * h[sp:]
    grad *= dx
