    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def project_sum(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} using bisection on lambda."""
    # Find lambda such that sum(clamp(h - lambda, 0, 1)) = target
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def soft_max_val(corr, T):
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T

def soft_max_grad_fft(h, T, N_fft):
    """Compute gradient of soft_max_val(compute_corr(h), T) * dx using FFT."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    exp_arg = T * (corr - mc)
    p = np.exp(exp_arg)
    p /= np.sum(p)
    
    # Build c_circ (centered p in circular array)
    c_circ = np.zeros(N_fft, dtype=float)
    # p[m] for m=0..2n-2 corresponds to shift s = m-(n-1)
    # c[k] = p[k+(n-1)] for k=-(n-1)..n-1
    # In circular array: c[k] goes to position k mod N_fft
    # c[0..n-1] = p[n-1..2n-2] → position 0..n-1
    c_circ[0:n] = p[n-1:2*n-1]
    # c[-(n-1)..-1] = p[0..n-2] → position N_fft-(n-1)..N_fft-1
    c_circ[N_fft-(n-1):N_fft] = p[0:n-1]
    
    H = np.fft.rfft(h, N_fft)
    C = np.fft.rfft(c_circ, N_fft)
    
    # term1[j] = sum_k c[k]*h[j+k] = ifft(conj(H)*C)[j]
    term1 = np.fft.irfft(np.conj(H) * C, N_fft)[:n]
    # term2[j] = sum_k c[k]*h[j-k] = ifft(H*C)[j]
    term2 = np.fft.irfft(H * C, N_fft)[:n]
    
    grad = dx * (1.0 - term1 + term2)
    return grad, mc + np.log(np.sum(np.exp(exp_arg))) / T

def fista_step(h, T, N_fft, max_iter=1000, step_init=1.0, time_limit=None, start_time=None):
    """FISTA with backtracking for the soft-max objective."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    
    h_curr = h.copy()
    y = h.copy()
    t_mom = 1.0
    step = step_init
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    
    for it in range(max_iter):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        # Compute gradient at y
        grad_y, f_y = soft_max_grad_fft(y, T, N_fft)
        
        # Backtracking line search
        while True:
            y_new = y - step * grad_y
            # Project
            h_p = project_sum(y_new, target)
            corr_h = compute_corr(h_p)
            f_h = soft_max_val(corr_h, T)
            
            # Check sufficient decrease
            if f_h <= f_y - 0.5 * step * np.dot(grad_y, y_new - y):
                break
            step *= 0.5
            if step < 1e-12:
                break
        
        h_new = h_p
        corr_new = compute_corr(h_new)
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Momentum update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        y = h_new + ((t_mom - 1.0) / t_new) * (h_new - h_curr)
        y = project_sum(y, target)  # Keep y feasible
        t_mom = t_new
        h_curr = h_new
        
        # Restore step if we reduced it
        # (optional: keep the reduced step or restore)
    
    return best_h, best_c5

Hmm wait, there's an issue. The FISTA objective is the soft-max, but I want to minimize the true max (which is what C5 measures). The soft-max is an upper bound on the max, so minimizing it is valid (if the soft-max decreases, the max also decreases or stays the same... actually no, the soft-max always >= max, and as T→∞, soft-max → max).

Actually, soft_max = max + (1/T)*log(sum(exp(T*(corr-max)))) >= max, and as T→∞, it approaches max. So minimizing the soft-max with a large T is a good proxy for minimizing the max.

But there's a subtlety: the soft-max value and the true max can differ. When I compare solutions, I should use the true max (C5), not the soft-max value.

In my FISTA, I'll use the soft-max for the optimization (it's smooth) but track the true C5 for comparison.

Let me also think about the step size. The initial step should be related to 1/L where L is the Lipschitz constant of the gradient.

For the soft-max objective f(h) = max(C) + (1/T)*log(sum exp(T*(C-max))) :
The gradient involves the Hessian of C (which is bounded since C is bilinear in h ∈ [0,1]) times T (from the softmax).

A rough bound: ||∇f|| ≤ T * ||H_C|| * ||p||₁ where H_C is the Hessian of C.
Since C is bilinear, ||H_C|| ≤ O(n) (each entry is at most 1, and there are O(n) non-zero entries per row).
So L ≈ T * n * dx (with the dx factor from the objective).

For T=500, n=960, dx=2/960: L ≈ 500. So step ≈ 1/500 = 0.002.

But this is very conservative. In practice, the effective Lipschitz constant is much smaller. Backtracking will handle this.

Let me set step_init = 1.0 and let backtracking reduce it. Or better, estimate L from the first few iterations.

Actually, for FISTA to work well, I need the step to not be too small (or convergence is slow) and not too large (or it diverges). Backtracking from a large initial step is standard.

Let me set step_init = 0.01 (somewhat conservative) and increase it by 1.5x after each successful step (barzilai-borwein-like adaptive step).

Now let me think about the binary local search.

For binary h (exactly n/2 ones), a pairwise swap (move a 1 from position i to position j) changes:
delta_corr[s] = (h[i+s] + h[i-s]) - (h[j+s] + h[j-s])

where h[<0] = h[>=n] = 0.

For the max to decrease, we need delta_corr[s] < 0 for all s where corr[s] = max.

The number of positions with value 1 is n/2 = 480. The number of pairwise swaps to consider is C(480, 480) which is huge. But we can be smarter:
- Only consider swaps where the new C5 would decrease
- Focus on the "active" shifts (those near the max)

A good heuristic: for the current max shift s*, find pairs (i, j) where h[i]=1, h[j]=0, and (h[i+s*]+h[i-s*]) < (h[j+s*]+h[j-s*]). These are candidates that would decrease the max correlation.

Then evaluate the full C5 for the top candidates.

Actually, for n=960, there are 480 * 480 = 230K possible swaps. Evaluating each swap's effect on the full correlation takes O(n) = O(960) operations (to update all correlations and find the max). So total is 230K * 960 ≈ 222M operations. That's too slow in pure Python.
