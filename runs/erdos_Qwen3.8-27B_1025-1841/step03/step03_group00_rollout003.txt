        else:

    diff = target - h.sum()

        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    if new_n == old_n:

    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)

The overlap at shift k* is: O(k*) = Σ_i h[i] * (1-h[i+k*]) (for valid i)

To reduce this, I want to decrease h[i] where h[i+k*] is also high (both 1 → overlap), or increase h[i] where h[i+k*] is low.

A simple gradient for O(k*) w.r.t. h[j]:
- If j is the "i" in h[i]: dO/dh[j] = (1-h[j+k*]) for j+k* < n
- If j is the "i+k*" in (1-h[i+k*]): dO/dh[j] = -h[j-k*] for j-k* >= 0

g[j] = (1-h[j+k*]) [if j+k* valid] + (-h[j-k*]) [if j-k* valid]

And we want to decrease O(k*), so we move h in the direction -g (i.e., h -= α*g).

But we also want to not increase other O(k'), so it's a trade-off. The subgradient of the max is the average of gradients of active constraints.

def compute_c5(h):
    n = len(h)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)


def lbgfs_opt(h_start, n, T, mu, max_iter):
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -60, 60)

        viol = np.sum(h) - n / 2.0
        return val + mu * viol * viol

                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)


def subgradient_refine(h, n, n_iter, step_size, top_k):

    best_c = compute_c5(h)

        top_idx = np.argsort(corr)[-top_k:]
        sg = np.zeros(n)
        for ki in top_idx:
            lag = int(ki) - (n - 1)
            if lag >= 0:
                sg[:n - lag] += (1.0 - h[lag:n])
                sg[lag:] += -h[:n - lag]
            else:
                la = -lag
                sg[la:] += -h[:n - la]
                sg[:n - la] += (1.0 - h[la:n])
        sg /= top_k
        alpha = step_size / (1.0 + 0.05 * it)
        for frac in [1.0, 0.5, 0.25]:
            h_try = h - alpha * frac * sg
            h_try = make_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c:
                h = h_try
                best_c = c_try
                break

                if remaining() < 10:
                    break
                old_v = h_cd[j]
                for new_v in [0.0, 1.0, 0.5, old_v * 0.5, max(0.0, 1.0 - old_v * 0.3)]:
                    if abs(new_v - old_v) < 1e-9:
                        continue
                    h_try = h_cd.copy()
                    h_try[j] = new_v
                    h_try = make_feasible(h_try, n_cd)
                    c = compute_c5(h_try)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = h_try
                        improved = True
                        break
            if not improved:
                break
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
