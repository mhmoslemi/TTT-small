def corr_gradient(h, m, N):
    """Compute gradient of correlation c_m = sum h[n+m]*(1-h[n]) w.r.t. h"""
    grad = np.zeros(N)
    if m >= 0:
        # c_m = sum_{n=0}^{N-1-m} h[n+m]*(1-h[n])
        # dc_m/dh[j] = (1-h[j-m]) for m <= j <= N-1
        #             - h[j+m] for 0 <= j <= N-1-m
        idx_pos = np.arange(m, N)
        grad[idx_pos] = (1.0 - h[idx_pos - m])
        idx_neg = np.arange(0, N - m)
        grad[idx_neg] -= h[idx_neg + m]
    else:
        p = -m
        # c_{-p} = sum_{j=0}^{N-1-p} h[j]*(1-h[j+p])
        # dc/dh[j] = (1-h[j+p]) for 0 <= j <= N-1-p
        #           - h[j-p] for p <= j <= N-1
        idx_pos = np.arange(0, N - p)
        grad[idx_pos] = (1.0 - h[idx_pos + p])
        idx_neg = np.arange(p, N)
        grad[idx_neg] -= h[idx_neg - p]
    return grad

    for k in range(len(corr)):
        if weights[k] < 1e-12:
            continue
        m = k - (N - 1)
        grad += weights[k] * corr_grad(h, m, N)
    grad *= dx
    
    # Penalty for sum constraint
    sv = np.sum(h) - N / 2.0
    obj += penalty * sv * sv
    grad += 2.0 * penalty * sv * np.ones(N)
    
    return obj, grad

def lbfgsb_optimize(h_start, N, T, penalty, max_iter=300):
    def fun(h):
        return lse_objective_and_grad(h, T, penalty, N)[0]
    
    def fun_and_grad(h):
        return lse_objective_and_grad(h, T, penalty, N)
    
    bounds = [(0.0, 1.0)] * N
    res = minimize(fun_and_grad, h_start, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8})
    return make_feasible(res.x, N)
