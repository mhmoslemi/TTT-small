import numpy as np
import time


def fft_correlate(h, g):
    n = len(h)
    m = len(g)
    result_len = n + m - 1
    fft_size = 1
    while fft_size < result_len:
        fft_size *= 2
    H = np.fft.rfft(h, fft_size)
    G = np.fft.rfft(g[::-1], fft_size)
    result = np.fft.irfft(H * G, fft_size)[:result_len]
    return result


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return corr.max() * dx


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def softmax_grad(h, n, tau, top_k):
    dx = 2.0 / n
    one_mh = 1.0 - h
    corr = fft_correlate(h, one_mh)
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    max_c = top_vals.max()
    shifted = np.clip(tau * (top_vals - max_c), -500.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        pk = p[ii]
        m = int(top_idx[ii])
        shift = m - (n - 1)
        i_lo = max(0, shift)
        i_hi = min(n - 1, shift + n - 1)
        if i_lo <= i_hi:
            grad[i_lo:i_hi + 1] += pk * (1.0 - h[i_lo - shift:i_hi - shift + 1])
        i_lo2 = max(0, -shift)
        i_hi2 = min(n - 1, n - 1 - shift)
        if i_lo2 <= i_hi2:
            grad[i_lo2:i_hi2 + 1] -= pk * h[i_lo2 + shift:i_hi2 + shift + 1]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr


def nesterov_loop(h, n, tau_start, tau_end, time_limit, lr, top_k):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    it_cnt = 0

    while True:
        if time.time() - start > time_limit:
            break
        elapsed = time.time() - start
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** frac
        mu = min((t - 1.0) / (t + 2.0), 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj, grad, _ = softmax_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        it_cnt += 1
        if it_cnt % 15 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 15
            if stall > 2500:
                lr *= 0.5
                stall = 0
                if lr < 1e-15:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

            best_h = h.copy()
            stall = 0
        else:

        if top_k < len(corr):

            if i_lo2 <= i_hi2:
                grad[i_lo2:i_hi2 + 1] -= pk * h[i_lo2 + shift:i_hi2 + shift + 1]
        grad *= dx

        h = project_to_feasible(h + v, n)
