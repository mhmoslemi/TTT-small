def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_obj_softmin(v, n, T, mu):
    dx = 2.0 / n
    h = v
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -700, 700)
    soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    viol = np.sum(h) - n / 2.0
    return soft + mu * viol * viol

    def f(v):
        return lbgfb_obj_softmin(v, n, T, mu)

        res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 80})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def oneshot_jac(h):
    return np.ones(len(h))

def slsqp_obj(v, n, T):
    dx = 2.0 / n
    h = v
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -700, 700)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def slsqp_con(v, n):
    return np.sum(v) - n / 2.0

def slsqp_polish(h_start, n, T, max_iter=100):
    def obj(v):
        return slsqp_obj(v, n, T)
    def con(v):
        return slsqp_con(v, n)
    nval = len(h_start)
    bounds = [(0.0, 1.0)] * nval
    cons = [{'type': 'eq', 'fun': lambda v: slsqp_con(v, nval), 'jac': lambda v: oneshot_jac(v)}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-16})
        return make_feasible(res.x, nval)
    except Exception:
        return make_feasible(h_start, nval)

def single_swap_search(h_bin, n, rng, max_rounds=100, n_tries=5000):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        nt = min(n_tries, max(100, len(ones) * len(zeros) // 10))
        for _ in range(nt):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def block_swap_search(h_bin, n, rng, max_rounds=50, max_block=6, T_start=0.1, max_tries=4000):

    zeros_set = set(np.where(h <= 0.5)[0])

        improved = False
        for _ in range(max_tries):
            blen = rng.randint(1, max_block + 1)
            if len(ones_set) < blen or len(zeros_set) < blen:
                continue
            oi = rng.choice(list(ones_set))
            oj = rng.choice(list(zeros_set))
            one_block = [oi + d for d in range(blen) if 0 <= oi + d < n and (oi + d) in ones_set]
            zero_block = [oj + d for d in range(blen) if 0 <= oj + d < n and (oj + d) in zeros_set]
            if len(one_block) != blen or len(zero_block) != blen:
                continue

                h[idx] = 0.0
            for idx in zero_block:
                h[idx] = 1.0

            delta = new_c - current
            accept = delta < -1e-15

                    ones_set.discard(idx)
                    zeros_set.add(idx)

                    zeros_set.discard(idx)
                    ones_set.add(idx)

                    h[idx] = 1.0
                for idx in zero_block:
                    h[idx] = 0.0

        if not improved:
            break
