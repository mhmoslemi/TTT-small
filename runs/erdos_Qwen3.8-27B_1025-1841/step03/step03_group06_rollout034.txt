import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_worst_lag_grad(h):
    """Compute C5 value, worst lag, and gradient of the worst lag's overlap."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = np.max(corr) * dx
    j_star = np.argmax(corr)
    ell = j_star - (n - 1)  # lag
    # Compute gradient of g_ell with respect to h
    grad = np.zeros(n)
    if ell >= 0:
        # Term 1: (1-h[j-ell]) for j in [ell, n-1]
        grad[ell:] += (1.0 - h[:n-ell])
        # Term 2: -h[j+ell] for j in [0, n-1-ell]
        grad[:n-ell] -= h[ell:]
    else:
        m = -ell
        # Term 1: (1-h[j+m]) for j in [0, n-1-m]
        grad[:n-m] += (1.0 - h[m:])
        # Term 2: -h[j-m] for j in [m, n-1]
        grad[m:] -= h[:n-m]
    grad *= dx
    return c5_val, grad, ell

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = h.copy()
    # Binary search on the shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_trial)
        target = n / 2.0
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = n / 2.0 - np.sum(h_proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h_proj < 1.0
        else:
            mask = h_proj > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_local_search(h, n, max_rounds=100, time_limit=None, start_time=None):
    """Flip coordinates to 0 or 1, accepting improvements."""
    best_c = compute_c5(h)
    for r in range(max_rounds):
        if time_limit and start_time and (time.time() - start_time) > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit and start_time and (time.time() - start_time) > time_limit:
                break
            for val in [0.0, 1.0]:
                if abs(h[j] - val) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    h = trial
                    best_c = c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c

def _obj_smooth(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    return np.sum(x) - n / 2.0

    return np.ones(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def get_worst_lag_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5_val = np.max(corr) * dx
    j_star = np.argmax(corr)
    ell = j_star - (n - 1)
    grad = np.zeros(n)
    if ell >= 0:
        grad[ell:] += (1.0 - h[:n-ell])
        grad[:n-ell] -= h[ell:]
    else:
        m = -ell
        grad[:n-m] += (1.0 - h[m:])
        grad[m:] -= h[:n-m]
    grad *= dx
    return c5_val, grad

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    return np.sum(x) - n / 2.0

    return np.ones(n)
