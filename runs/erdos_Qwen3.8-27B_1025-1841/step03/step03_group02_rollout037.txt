import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

_SLS_N = 0
_SLS_T = 100.0

def slsqp_obj(x):
    n = _SLS_N
    T = _SLS_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    ms = np.max(shifted)
    lse = mc + ms + np.log(np.sum(np.exp(T * (shifted - ms)))) / T
    return lse * dx

def slsqp_con_fun(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol, deadline):
    global _SLS_N, _SLS_T
    _SLS_N = n
    _SLS_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
    try:
        res = minimize(slsqp_obj, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if time.time() > deadline:
            return None
        return res.x
    except Exception:
        return None

def perturbed_h(h, rng, epsilon):
    p = h.copy()
    mask = (p > 0.01) & (p < 0.99)
    cnt = np.sum(mask)
    if cnt > 0:
        p[mask] += rng.normal(0, epsilon, cnt)
        p = np.clip(p, 0.0, 1.0)
        p = make_feasible(p, len(p))
    return p
