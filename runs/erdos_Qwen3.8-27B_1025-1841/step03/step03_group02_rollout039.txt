import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def exact_project(y, total):
    """Project y onto {x: 0<=x<=1, sum(x)=total}"""
    n = len(y)
    # Bisection on tau: sum(clamp(y_i - tau, 0, 1)) = total
    lo = y.min() - 1.0
    hi = y.max()
    for _ in range(100):
        tau = (lo + hi) / 2.0
        s = np.sum(np.clip(y - tau, 0.0, 1.0))
        if s > total:
            lo = tau
        else:
            hi = tau
    tau = (lo + hi) / 2.0
    return np.clip(y - tau, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    w = np.exp(T * (corr - max_c))
    w_sum = np.sum(w)
    w = w / w_sum
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            # f_0 = sum h[i]*(1-h[i]) = sum h[i] - sum h[i]^2
            # ∂f_0/∂h[j] = 1 - 2*h[j]
            grad += w[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            # ∂f/∂h[j] = (1-h[j+lag]) for j < n-lag, minus h[j-lag] for j >= lag
            g = np.zeros(n)
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
            grad += w[k] * g
        else:
            l = -lag
            g = np.zeros(n)
            g[l:] += (1.0 - h[:n-l])
            g[:n-l] -= h[l:]
            grad += w[k] * g
    
    return (dx / T) * grad

    n = len(h)
    total = n / 2.0

    best_c5 = cur_c5
    best_h = h.copy()

        grad = compute_grad_lse(h, T)

        lr = min(1.0 / (np.max(np.abs(grad)) + 1e-10), 0.5)
        accepted = False
        for _ in range(20):

            h_new = exact_project(y, total)
            c5_new = compute_c5(h_new)
            if c5_new < cur_c5 + 1e-12:

                cur_c5 = c5_new
                accepted = True
                break

            c5_new = compute_c5(h_new)
            if c5_new < cur_c5:
                h = h_new
                cur_c5 = c5_new
        
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def exact_project(y, total):
    """Project onto {x: 0<=x<=1, sum(x)=total} using bisection"""
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > total:
            lo = mid
        else:
            hi = mid
    return np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)

def compute_grad(h, T):
    """Analytical gradient of LSE-smoothed C5"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    exponents = T * (corr - max_c)
    # Clip for numerical stability
    exponents = np.clip(exponents, -500, 0)
    w = np.exp(exponents)
    sw = w.sum()
    if sw < 1e-100:
        return np.zeros(n)
    w /= sw
    
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:n - lag]
        else:
            l = -lag
            grad[l:] += wk * (1.0 - h[:n - l])
            grad[:n - l] -= wk * h[l:]
    return (dx / T) * grad

def gradient_descent(h, T_start, T_end, n_steps, lr_base, start_time, budget_s, reserve):
    n = len(h)
    total = n / 2.0
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    lr = lr_base
    
    for step in range(n_steps):
        if time.time() - start_time > budget_s - reserve:
            break
        T = T_start * (T_end / T_start) ** (step / max(n_steps - 1, 1))
        grad = compute_grad(h, T)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-14:
            break
        
        # Adaptive step with backtracking
        step_size = lr
        accepted = False
        for _ in range(8):
            y = h - step_size * grad
            h_new = exact_project(y, total)
            c5_new = compute_c5(h_new)
            if c5_new <= cur_c5 + 1e-14:
                h = h_new
                cur_c5 = c5_new
                accepted = True
                break
            step_size *= 0.5
        
        if accepted:
            lr = min(lr * 1.1, 200.0)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            lr = max(lr * 0.5, 0.1)
    
    return best_h, best_c5

def binary_sa(h, n, max_iters, T0, T1, start_time, budget_s, rng, reserve):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = int(np.sum(b) - target)
    if diff > 0:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])[:diff]
        b[ones[order]] = 0
    elif diff < 0:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])[:(-diff)]
        b[zeros[order]] = 1
    
    cur_c5 = compute_c5(b)
    best_c5 = cur_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - reserve:
            break
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or b[i] == b[j]:
            continue
        b[i], b[j] = b[j], b[i]
        new_c5 = compute_c5(b)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i], b[j] = b[j], b[i]
    
    return best_b, best_c5
