def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def coord_descent_step(h, j, corr):
    """Optimize coordinate j given current corr values."""
    n = len(h)
    hj = h[j]
    
    # Compute alpha[m] for all m
    # alpha[m] = (1-h[j-lag]) - h[j+lag] where lag = m-(n-1)
    # Valid ranges:
    # term1 valid when 0 <= j-lag < n, i.e., j-n+1 <= m <= j+n-1 (clipped to [0, 2n-2])
    # term2 valid when 0 <= j+lag < n, i.e., n-1-j <= m <= 2n-2-j (clipped to [0, 2n-2])
    
    alpha = np.zeros(2*n-1)
    
    # Term 1: 1-h[j-lag] = 1-h[n-1+j-m] for valid m
    m1_start = max(0, j - n + 1)
    m1_end = min(2*n-2, j + n - 1)
    if m1_start <= m1_end:
        indices = np.arange(m1_start, m1_end + 1)
        h_indices = n - 1 + j - indices  # = j - (m - (n-1)) = j - lag
        alpha[m1_start:m1_end+1] += 1.0 - h[h_indices]
    
    # Term 2: -h[j+lag] = -h[m-(n-1-j)] for valid m
    m2_start = max(0, n - 1 - j)
    m2_end = min(2*n-2, 2*n - 2 - j)
    if m2_start <= m2_end:
        indices = np.arange(m2_start, m2_end + 1)
        h_indices = indices - (n - 1 - j)  # = j + (m-(n-1)) = j + lag
        alpha[m2_start:m2_end+1] += -h[h_indices]
    
    # beta[m] = corr[m] - h[j]*alpha[m] (valid for m != n-1)
    beta = corr - hj * alpha
    
    # For m = n-1, the relationship is quadratic: t - t^2 + beta_true
    # beta_true[n-1] = corr[n-1] - h[j]*(1-h[j])
    beta_true_n1 = corr[n-1] - hj * (1.0 - hj)
    
    # Define g(t) = max over all m of expr_m(t)
    # For m != n-1: alpha[m]*t + beta[m]
    # For m = n-1: t - t^2 + beta_true_n1
    
    def g(t):
        vals = alpha * t + beta  # shape (2n-1,)
        # Override m=n-1 with quadratic
        vals[n-1] = t - t*t + beta_true_n1
        return np.max(vals)
    
    # Minimize g(t) over t in [0,1] using ternary search
    # Also check endpoints and t=0.5
    a, b = 0.0, 1.0
    for _ in range(50):
        t1 = a + (b - a) / 3.0
        t2 = b - (b - a) / 3.0
        if g(t1) < g(t2):
            b = t2
        else:
            a = t1
    
    t_ternary = (a + b) / 2.0
    candidates = [0.0, 1.0, 0.5, t_ternary, hj]
    best_t = hj
    best_g = g(hj)
    for t in candidates:
        val = g(t)
        if val < best_g:
            best_g = val
            best_t = t
    
    return best_t
