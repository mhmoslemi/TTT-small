import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -100, 100)
    logsum = mc + np.log(np.sum(np.exp(shifted))) / T
    viol = np.sum(h) - n / 2.0
    return (logsum + mu * viol * viol) * dx

def soft_max_obj_no_penalty(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -100, 100)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def lbgfb_optimize(h_start, n, T, mu, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    
    def f(h):
        return soft_max_obj(h, T, mu)
    
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def slsqp_optimize(h_start, n, T, max_iter=100):
    dx = 2.0 / n
    
    def obj(h):
        return soft_max_obj_no_penalty(h, T)
    
    def con(h):
        return np.sum(h) - n / 2.0
    
    def con_jac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_run(h_start, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -100, 100)
        val = (mc + np.log(np.sum(np.exp(sh))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return val + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def slsqp_run(h_start, n, T, max_iter=80):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -100, 100)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    def c(h):
        return np.sum(h) - n / 2.0
    def cj(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': c, 'jac': cj}]
    res = minimize(f, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def sa_binary(h_bin, n, rng, n_evals=8000, T0=0.6, T1=0.003):
    h = h_bin.copy()
    current = compute_c5(h)
    best_c = current
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for i in range(n_evals):
        frac = i / max(n_evals - 1, 1)
        T = T0 * (T1 / T0) ** frac
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        c = compute_c5(h)
        d = c - current
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-12)):
            current = c
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, oj)
            zeros = np.append(zeros, oi)
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    h = best_h.copy()
    current = best_c
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for _ in range(3000):
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        c = compute_c5(h)
        if c < current:
            current = c
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, oj)
            zeros = np.append(zeros, oi)
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return h, current

def coord_descent(h_start, n, max_rounds=5):
    h = h_start.copy()
    current = compute_c5(h)
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-14:
                    h = tr
                    current = c
                    ov = nv
                    improved = True
                    break
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                tr = h.copy()
                tr[j] = bv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-14:
                    h = tr
                    current = c
                    improved = True
        if not improved:
            break
    return h, current

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: n=256 multi-temp L-BFGS-B
    n1 = 256
    h_w = upsample_h(best_h, n1)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()
    for T in [8, 20, 50, 120, 300, 800, 2000]:
        if tl() < 15:
            break
        for r in range(2):
            if tl() < 8:
                break
            h_init = h_w.copy() if r == 0 else make_feasible(h_w + 0.25 * rng.randn(n1), n1)
            try:
                h_opt = lbgfb_run(h_init, n1, T, 800.0, 180)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
                h_w = h_opt.copy()
            except Exception:
                pass

    # Phase 2: n=512 L-BFGS-B
    n2 = 512
    h_w = upsample_h(best_h, n2)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()
    for T in [25, 60, 150, 400, 1000, 2500]:
        if tl() < 15:
            break
        h_opt = lbgfb_run(h_w, n2, T, 400.0, 150)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5, best_h = c, h_opt.copy()
        h_w = h_opt.copy()
        # random restart
        if tl() > 30:
            h_r = make_feasible(h_w + 0.15 * rng.randn(n2), n2)
            try:
                h_opt2 = lbgfb_run(h_r, n2, T, 400.0, 100)
                c2 = compute_c5(h_opt2)
                if c2 < best_c5:
                    best_c5, best_h = c2, h_opt2.copy()
                    h_w = h_opt2.copy()
            except Exception:
                pass

    # Phase 3: Binary SA at n=512 (multiple restarts)
    if tl() > 120:
        h_bin = (h_w > 0.5).astype(float)
        n_ones = int(round(np.sum(h_bin)))
        n_tgt = n2 // 2
        if n_ones > n_tgt:
            idx = np.where(h_bin == 1)[0]
            for k in range(n_ones - n_tgt):
                h_bin[idx[k]] = 0.0
        elif n_ones < n_tgt:
            idx = np.where(h_bin == 0)[0]
            for k in range(n_tgt - n_ones):
                h_bin[idx[k]] = 1.0
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

        h_sa, c5_sa = sa_binary(h_bin, n2, rng, n_evals=10000)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()

        # Second SA from perturbed start
        if tl() > 60:
            h_pert = best_h.copy()
            flip_count = 40
            ones_p = np.where(h_pert > 0.5)[0]
            zeros_p = np.where(h_pert <= 0.5)[0]
            for _ in range(flip_count):
                if len(ones_p) > 0 and len(zeros_p) > 0:
                    ii = rng.randint(len(ones_p))
                    jj = rng.randint(len(zeros_p))
                    h_pert[ones_p[ii]] = 0.0
                    h_pert[zeros_p[jj]] = 1.0
                    ones_p = np.append(np.delete(ones_p, ii), zeros_p[jj])
                    zeros_p = np.append(np.delete(zeros_p, jj), ones_p[ii] if False else h_pert[ones_p[jj]] if False else 0)
                    ones_p = np.where(h_pert > 0.5)[0]
                    zeros_p = np.where(h_pert <= 0.5)[0]
            h_pert = make_feasible(h_pert, n2)
            h_sa2, c5_sa2 = sa_binary(h_pert, n2, rng, n_evals=6000)
            if c5_sa2 < best_c5:
                best_c5, best_h = c5_sa2, h_sa2.copy()

    # Phase 4: SLSQP polish at n=512
    if tl() > 60:
        h_s = best_h.copy() if len(best_h) == n2 else upsample_h(best_h, n2)
        for T in [100, 500, 2000, 5000]:
            if tl() < 15:
                break
            try:
                h_opt = slsqp_run(h_s, n2, T, 80)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
                    h_s = h_opt.copy()
            except Exception:
                pass

    # Phase 5: n=1024
    if tl() > 80:
        n3 = 1024
        h_w = upsample_h(best_h, n3)
        c5w = compute_c5(h_w)
        if c5w < best_c5:
            best_c5, best_h = c5w, h_w.copy()
        for T in [60, 200, 600, 1500]:
            if tl() < 20:
                break
            try:
                h_opt = lbgfb_run(h_w, n3, T, 200.0, 100)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
                h_w = h_opt.copy()
            except Exception:
                pass

        # Binary SA at n=1024
        if tl() > 80:
            h_bin3 = (h_w > 0.5).astype(float)
            n_ones3 = int(round(np.sum(h_bin3)))
            n_tgt3 = n3 // 2
            if n_ones3 > n_tgt3:
                idx = np.where(h_bin3 == 1)[0]
                for k in range(n_ones3 - n_tgt3):
                    h_bin3[idx[k]] = 0.0
            elif n_ones3 < n_tgt3:
                idx = np.where(h_bin3 == 0)[0]
                for k in range(n_tgt3 - n_ones3):
                    h_bin3[idx[k]] = 1.0
            c5b3 = compute_c5(h_bin3)
            if c5b3 < best_c5:
                best_c5, best_h = c5b3, h_bin3.copy()
            h_sa3, c5_sa3 = sa_binary(h_bin3, n3, rng, n_evals=8000)
            if c5_sa3 < best_c5:
                best_c5, best_h = c5_sa3, h_sa3.copy()

        # Coordinate descent at n=1024
        if tl() > 30:
            h_cd, c5_cd = coord_descent(h_w, n3, max_rounds=3)
            if c5_cd < best_c5:
                best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 6: n=2048
    if tl() > 30:
        n4 = 2048
        h_w = upsample_h(best_h, n4)
        c5w = compute_c5(h_w)
        if c5w < best_c5:
            best_c5, best_h = c5w, h_w.copy()
        for T in [100, 400, 1000]:
            if tl() < 15:
                break
            try:
                h_opt = lbgfb_run(h_w, n4, T, 100.0, 60)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
                h_w = h_opt.copy()
            except Exception:
                pass

    # Final squeeze
    if tl() > 10:
        n_f = len(best_h)
        try:
            h_opt = slsqp_run(best_h, n_f, 5000, 50)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5, best_h = c, h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
