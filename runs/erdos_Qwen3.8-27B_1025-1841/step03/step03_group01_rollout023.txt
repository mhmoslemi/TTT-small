def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def compute_gradient(h, T, penalty_M):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc

    p = np.exp(T * shifted)
    p /= np.sum(p)
    
    # q = p (indexed 0 to 2n-2)
    q = p

    S1 = conv[n-1 : 2*n-1]  # indices n-1 to 2n-2, length n

OK the full recomputation is too slow. Let me use the incremental update:

When h[j] changes from old to new (delta = new - old):
For each lag k (indexed 0 to 2n-2, corresponding to actual lag = k - (n-1)):
  The correlation c[k] changes by:
  delta_c = delta * [(1-h[j - (k-(n-1))]) * I(0 ≤ j-(k-(n-1)) < n) - h[j + (k-(n-1))]) * I(0 ≤ j+(k-(n-1)) < n)]

Let me define lag = k - (n-1), so lag ranges from -(n-1) to n-1.
delta_c(lag) = delta * [(1-h[j-lag]) * I(0 ≤ j-lag < n) - h[j+lag] * I(0 ≤ j+lag < n)]

The affected lags are those where either j-lag or j+lag is in [0, n-1]:
- j-lag ∈ [0, n-1] → lag ∈ [j-n+1, j]
- j+lag ∈ [0, n-1] → lag ∈ [-j, n-1-j]

Union: lag ∈ [j-n+1, n-1-j] (approximately, the union of the two ranges)

The number of affected lags is at most 2n-1, but typically about n.

For each affected lag, delta_c can be computed in O(1).
The new max is: max(old_max, max of new c values over affected lags) if the max was not at an affected lag, or max of all new c values if it was.

To track the max efficiently, I can maintain a sorted structure, but for simplicity, let me just:
1. Compute delta_c for all affected lags (O(n))
2. Update corr for those lags (O(n))
3. Find the new max: if the old max was not affected, new_max = max(old_max, max of updated values at affected lags). If the old max was affected, new_max = max of all corr values (O(n)).

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    T = _g_T
    M = _g_M
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = corr - mc
    p = np.exp(T * shifted)
    p_sum = np.sum(p)
    p = p / p_sum
    
    q = p  # length 2n-1
    one_minus_h = 1.0 - h
    
    # S1[j] = (q * (1-h))[j+n-1] for j=0..n-1
    conv = np.convolve(q, one_minus_h, mode='full')
    S1 = conv[n-1 : 2*n-1]  # length n
    
    # S2[j] = correlate(q, h)[n-1-j] for j=0..n-1
    corr_qh = np.correlate(q, h, mode='full')
    S2 = corr_qh[n-1 : -1 : -1]  # elements n-1, n-2, ..., 0, reversed = elements 0..n-1 reversed
    # Wait: S2[j] = corr_qh[n-1-j], so S2 = corr_qh[n-1], corr_qh[n-2], ..., corr_qh[0]
    S2 = corr_qh[0 : n][::-1]
    
    grad_c5 = dx * (S1 - S2)
    diff = np.sum(h) - n / 2.0
    grad_pen = 2.0 * M * diff
    return grad_c5 + grad_pen

def _obj_wrap(h):
    return _obj_fn(h)

def _grad_wrap(h):
