    if n == target_n:
        return h.copy()
    # Simple linear interpolation with periodic extension
    h_ext = np.concatenate([h[-n//10:], h, h[:n//10]])

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    pad = max(n // 20, 10)
    h_ext = np.concatenate([h[-pad:], h, h[:pad]])
    dx = 2.0 / n
    x_old = np.arange(len(h_ext)) * dx - pad * dx
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h_ext)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def soft_obj_grad(h, n, tau, top_k=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx] - max_c
    shifted = np.clip(tau * top_vals, -100.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    for i_idx in range(top_idx.size):
        pk = p[i_idx]
        sh = int(top_idx[i_idx]) - n + 1
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    pad = max(n // 20, 10)
    h_ext = np.concatenate([h[-pad:], h, h[:pad]])
    dx_old = 2.0 / n
    x_old = np.arange(len(h_ext)) * dx_old - pad * dx_old
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h_ext)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def soft_obj_grad(h, n, tau, top_k=10):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx] - max_c
    shifted = np.clip(tau * top_vals, -100.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    for ii in range(top_idx.size):
        pk = p[ii]
        sh = int(top_idx[ii]) - n + 1
        j0, j1 = max(0, -sh), min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        j0b, j1b = max(0, sh), min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    return obj, grad


def binary_nudge(h, strength=0.06):
    h = h.copy()
    mask = (h > 0.08) & (h < 0.92)
    if mask.any():
        push = np.where(h[mask] < 0.5, -1.0, 1.0)
        h[mask] += push * strength
    return h


def nesterov(h, n, tau, tl, lr=None, top_k=10, nudge_every=0):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.35 / np.sqrt(tau)
    h_prev = h.copy()
    t = 1.0
    stall = 0
    for it in range(2000000):
        if time.time() - start > tl:
            break
        mu = min((t - 1.0) / (t + 2.0), 0.95)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        _, grad = soft_obj_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        if nudge_every > 0 and it > 0 and it % nudge_every == 0:
            h_new = binary_nudge(h_new, 0.04)
        h_new = project_to_feasible(h_new, n)
        if it % 35 == 0:
            c5 = compute_c5(h_new)
            if c5 < best_c5 - 1e-14:
                best_c5, best_h, stall = c5, h_new.copy(), 0
            else:
                stall += 35
            if stall > 2000:
                lr *= 0.5
                stall = 0
                if lr < 1e-16:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    c5f = compute_c5(h)
    if c5f < best_c5:
        best_c5, best_h = c5f, h.copy()
    return best_h, best_c5


def subgrad(h, n, tl, lr=None, top_k=3):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.0007
    v = np.zeros(n)
    stall = 0
    for it in range(2000000):
        if time.time() - start > tl:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-14:
            best_c5, best_h, stall = c5, h.copy(), 0
        else:
            stall += 1
        if stall > 1200:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-16:
                break
        if top_k < len(corr):
            ti = np.argpartition(corr, -top_k)[-top_k:]
        else:
            ti = np.arange(len(corr))
        tv = corr[ti]
        w = np.exp(tv - tv.max())
        w /= w.sum()
        grad = np.zeros(n)
        for ii in range(ti.size):
            sh = int(ti[ii]) - n + 1
            j0, j1 = max(0, -sh), min(n, n - sh)
            if j0 < j1:
                grad[j0:j1] += w[ii] * (1.0 - h[j0 + sh:j1 + sh])
            j0b, j1b = max(0, sh), min(n, n + sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[ii] * h[j0b - sh:j1b - sh]
        v = 0.85 * v - lr * grad * dx
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def rem_time(t0, budget):
    return max(0.0, budget - (time.time() - t0))
