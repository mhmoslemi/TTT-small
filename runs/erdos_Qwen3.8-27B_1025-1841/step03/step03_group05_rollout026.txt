```python
import numpy as np
from scipy.optimize import minimize
import time

def fft_corr(h):
    """Compute np.correlate(h, 1-h, 'full') using FFT."""
    n = len(h)
    b = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fh = np.fft.rfft(h, N)
    fb = np.fft.rfft(b, N)
    full = np.fft.irfft(fh * np.conj(fb), N)
    return np.concatenate([full[n-1:0:-1], [full[0]], full[N-1:N-n+1:-1]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, s):
    """Subgradient of correlation at lag s with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if s >= 0:
        if s < n:
            g[:n-s] += 1.0 - h[s:]
            g[s:] -= h[:n-s]
    else:
        s = -s
        if s < n:
            g[s:] += 1.0 - h[:n-s]
            g[:n-s] -= h[s:]
    return g

def smoothed_grad(h, T, top_k=200):
    """Gradient of smoothed max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return subgrad_one(h, idx - (n-1)) * dx
    w = exps / tot
    # Get top-k indices by weight
    k = min(top_k, len(w))
    top_idx = np.argpartition(w, -k)[-k:]
    top_idx = top_idx[top_idx[w[top_idx] > 1e-12]]
    g = np.zeros(n)
    for i in top_idx:
        wi = w[i]
        lag = i - (n - 1)
        g += wi * subgrad_one(h, lag)
    return g * dx
```

            g[:n-k] -= h[k:n]
        else:
            g[:] += 1.0 - 2.0 * h
    else:
        sp = -k
        g[:n-sp] += 1.0 - h[sp:n]
        g[sp:n] -= h[:n-sp]
    return g

def fista_fft(h_init, n_iter, T0, T1, lr, top_k, check, n_budget_check=None, t0=None):
    n = len(h_init)
    dx = 2.0 / n
    x = h_init.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    
    for it in range(n_iter):
        if n_budget_check is not None and t0 is not None:
            if time.time() - t0 > n_budget_check:
                break
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.5 * f)
        
        omz = 1.0 - z
        corr = fft_corr(z, omz)
        mc = np.max(corr)
        
        exps = np.exp(T * (corr - mc))
        tot = exps.sum()
        
        k_used = min(top_k, len(corr))
        if tot > 1e-300:
            top_idx = np.argpartition(exps, -k_used)[-k_used:]
            w = exps[top_idx] / tot
            mask = w > 1e-10
            top_idx = top_idx[mask]
            w = w[mask]
        else:
            top_idx = np.array([int(np.argmax(corr))])
            w = np.array([1.0])
        
        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            lag = int(top_idx[ii]) - (n - 1)
            grad += w[ii] * _sg_one(z, lag, n)
        grad *= dx
        
        xn = project_c(z - lrt * grad, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    
    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    
    return best_h, best_c

def tm_seq(n):
    """Thue-Morse sequence of length n."""
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def upsample(h, nn):
    ox = np.linspace(0, 2.0, len(h), endpoint=False)
    nx = np.linspace(0, 2.0, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)
