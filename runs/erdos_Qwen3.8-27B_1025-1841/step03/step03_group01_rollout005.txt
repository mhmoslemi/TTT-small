from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 0
_G_TOP = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def subgrad_dir(h, n, top_idx):
    corr = compute_corr(h)
    mid = n - 1
    grad = np.zeros(n)
    idx = np.arange(n)
    for ki in top_idx:
        l = ki - mid
        if abs(l) >= n:
            continue
        grad += (1.0 - h[(idx - l) % n]) - h[(idx + l) % n]
    grad /= max(1.0, float(len(top_idx)))
    return grad, corr

def _obj_topk(v):
    corr = np.correlate(v, 1.0 - v, mode='full')
    return float(np.sum(corr[_G_TOP])) * (2.0 / _G_N)

def _obj_soft(v):
    corr = np.correlate(v, 1.0 - v, mode='full')
    mc = np.max(corr)
    sh = corr - mc
    ls = np.log(np.sum(np.exp(np.clip(_G_T * sh, -400, 400))))
    return (mc + ls / _G_T) * (2.0 / _G_N)

def _cons_fn(v):
    return np.sum(v) - _G_N / 2.0

def _cons_jac(v):
    return np.ones(_G_N)

    _G_T = T
    _G_TOP = use_topk

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fn, 'jac': _cons_jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

        best_c5, best_h = c5A, hA.copy()

    hA = subgrad_phase(hA, nA, 2000, tl)
    c5A = compute_c5(hA)
    if c5A < best_c5:
        best_c5, best_h = c5A, hA.copy()

    hA = iter_active_slq(hA, nA, 8, 120, tl)
    c5A = compute_c5(hA)
    if c5A < best_c5:
        best_c5, best_h = c5A, hA.copy()

    # High-T SLSQP at n=1024
    if tl() > 60:
        for T in [2000, 8000]:
            if tl() < 25:
                break
            try:
                htmp = slsqp_run(hA, nA, T, 150, use_topk=None)
                ct = compute_c5(htmp)
                if ct < best_c5:
                    best_c5, best_h = ct, htmp.copy()
                    hA = htmp.copy()
            except Exception:
                pass

    # Phase B: n=2048
    nB = 2048
    if tl() > 180:
        hB = upsample_h(best_h, nB)
        c5B = compute_c5(hB)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB.copy()

        hB = subgrad_phase(hB, nB, 2000, tl)
        c5B = compute_c5(hB)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB.copy()

        hB = iter_active_slq(hB, nB, 6, 100, tl)
        c5B = compute_c5(hB)
        if c5B < best_c5:
            best_c5, best_h = c5B, hB.copy()

        if tl() > 50:
            for T in [5000, 15000]:
                if tl() < 25:
                    break
                try:
                    htmp = slsqp_run(hB, nB, T, 120, use_topk=None)
                    ct = compute_c5(htmp)
                    if ct < best_c5:
                        best_c5, best_h = ct, htmp.copy()
                        hB = htmp.copy()
                except Exception:
                    pass

    # Phase C: Coordinate descent polish at n=1024
    nCd = 1024
    hCd = upsample_h(best_h, nCd)
    hCd, c5Cd = cd_polish(hCd, nCd, 6, tl)
    if c5Cd < best_c5:
        best_c5 = c5Cd
        best_h = upsample_h(hCd, nB) if len(best_h) > nCd else hCd.copy()

    # Phase D: Restart from binary-quantized version
    if tl() > 100:
        hq = (best_h > 0.5).astype(float)
        hq = make_feasible(hq, len(hq))
        c5q = compute_c5(hq)
        if c5q < best_c5:
            best_c5, best_h = c5q, hq.copy()
        if tl() > 60:
            nq = len(hq)
            hq = subgrad_phase(hq, nq, 1000, tl)
            c5q = compute_c5(hq)
            if c5q < best_c5:
                best_c5, best_h = c5q, hq.copy()
            if tl() > 30:
                try:
                    hq = slsqp_run(hq, nq, 5000, 100, use_topk=None)
                    c5q = compute_c5(hq)
                    if c5q < best_c5:
                        best_c5, best_h = c5q, hq.copy()
                except Exception:
                    pass

    # Phase E: Random restart blend
    if tl() > 80:
        nf = len(best_h)
        hrnd = 0.6 * best_h + 0.4 * rng.rand(nf)
        hrnd = make_feasible(hrnd, nf)
        hrnd = subgrad_phase(hrnd, nf, 1000, tl)
        c5r = compute_c5(hrnd)
        if c5r < best_c5:
            best_c5, best_h = c5r, hrnd.copy()
        if tl() > 30:
            try:
                hrnd = slsqp_run(hrnd, nf, 5000, 80, use_topk=None)
                c5r = compute_c5(hrnd)
                if c5r < best_c5:
                    best_c5, best_h = c5r, hrnd.copy()
            except Exception:
                pass

    # Final SLSQP squeeze
    if tl() > 20:
        try:
            hF = slsqp_run(best_h, len(best_h), 20000, 80, use_topk=None)
            c5F = compute_c5(hF)
            if c5F < best_c5:
                best_c5, best_h = c5F, hF.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
