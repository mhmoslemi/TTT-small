The minimum of a convex piecewise-linear function can be found by:
1. Find the "lower envelope" of the -functions, i.e., minimize max = find the lowest point of the upper envelope.
2. The upper envelope is determined by the "upper hull" of the lines.
3. The minimum is at a vertex of this upper hull that lies in [0,1].

A simple O(n log n) algorithm:

- Filter to those in [0,1]
- Evaluate f at all kink points + endpoints {0, 1}
- Return the t with minimum f

- f'(t-) = min of B[m] among active m at t
- f'(t+) = max of B[m] among active m at t
- If f'(t-) ≤ 0 ≤ f'(t+), t is optimal
- Otherwise, move in direction -sign(0 - subgradient)

Actually, for a convex piecewise-linear function with n pieces, the minimum is found in at most n steps by following the subgradient (which is constant on each piece). But in practice, it converges much faster.

    """Solve min_{t in [0,1]} max_m(corr[m] + B[m]*t) using the linear model."""
    n_lags = len(corr)
    t = 0.5  # start in middle
    
    for it in range(n_lags):  # at most n_lags iterations
        vals = corr + B * t
        max_val = np.max(vals)
        active = vals >= max_val - 1e-12
        active_B = B[active]
        
        min_B = np.min(active_B)
        max_B = np.max(active_B)
        
        if min_B <= 1e-10 and max_B >= -1e-10:
            break  # optimal (0 in subgradient)
        
        if min_B > 0:
            # Need to decrease t. Find closest kink below t.
            # A kink occurs where a new line becomes active.
            # For lines with B < min_B, find where they intersect the current max.
            # The nearest such intersection below t gives the next kink.
            inactive = ~active
            if not np.any(inactive):
                t = 0.0
                break
            # For inactive lines, find where corr[m] + B[m]*t' = max_val + (B[m]-min_B)*(t'-t)
            # This is getting complex. Just do a binary search.
            lo, hi = 0.0, t

            # Actually just bisect
            for _ in range(30):
                mid = (lo + hi) / 2
                vals_mid = corr + B * mid
                active_mid = vals_mid >= np.max(vals_mid) - 1e-12
                min_B_mid = np.min(B[active_mid])
                if min_B_mid <= 1e-10:
                    hi = mid
                else:
                    lo = mid
            t = (lo + hi) / 2
        else:  # max_B < 0
            # Need to increase t
            lo, hi = t, 1.0
            for _ in range(30):
                mid = (lo + hi) / 2
                vals_mid = corr + B * mid
                active_mid = vals_mid >= np.max(vals_mid) - 1e-12
                max_B_mid = np.max(B[active_mid])
                if max_B_mid >= -1e-10:
                    lo = mid
                else:
                    hi = mid
            t = (lo + hi) / 2
    
    # Final comparison with endpoints
    best_t = t
    best_val = np.max(corr + B * t)
    for tc in [0.0, 1.0]:
        vc = np.max(corr + B * tc)
        if vc < best_val:
            best_val = vc
            best_t = tc

    corr = np.correlate(h, 1.0 - h, mode='full')
    best_mc = float(corr.max())
    rng = np.random.RandomState(seed)

    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit - 80:
            break
        any_improve = False
        if sweep % 3 == 0:
            idx = rng.permutation(n)
        else:
            idx = np.arange(n)
        for j in idx:
            B = compute_B(h, j, n)
            t_opt = solve_1d(corr, B)
            old_t = h[j]
            delta = old_t - t_opt
            if abs(delta) < 1e-10:
                continue
            h_trial = h.copy()
            h_trial[j] = t_opt
            mask = np.ones(n, dtype=bool)
            mask[j] = False
            h_trial[mask] -= delta / (n - 1)
            h_trial = np.clip(h_trial, 0.0, 1.0)
            h_trial = make_feasible(h_trial, n)
            corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            mc_trial = float(corr_trial.max())
            if mc_trial < best_mc - 1e-11:
                h = h_trial
                corr = corr_trial
                best_mc = mc_trial
                any_improve = True
        if not any_improve:
            break
    return h, best_mc * dx

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def pair_swap_cd(h, n, max_sweeps, time_limit, start_time, seed=100):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_mc = float(corr.max())
    rng = np.random.RandomState(seed)
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit - 80:
            break
        any_improve = False
        for _ in range(n):
            j, jp = rng.randint(0, n), rng.randint(0, n)
            if j == jp:
                continue
            Bj = compute_B(h, j, n)
            Bjp = compute_B(h, jp, n)
            D = Bj - Bjp
            t_opt = solve_1d(corr, D)
            old_j = h[j]
            old_jp = h[jp]
            new_j = old_j + t_opt
            new_jp = old_jp - t_opt
            if new_j < -1e-9 or new_j > 1 + 1e-9 or new_jp < -1e-9 or new_jp > 1 + 1e-9:
                continue
            new_j = np.clip(new_j, 0.0, 1.0)
            new_jp = np.clip(new_jp, 0.0, 1.0)
            if abs(new_j - old_j) < 1e-10:
                continue
            h_trial = h.copy()
            h_trial[j] = new_j
            h_trial[jp] = new_jp
            h_trial = make_feasible(h_trial, n)
            corr_trial = np.correlate(h_trial, 1.0 - h_trial, mode='full')
            mc_trial = float(corr_trial.max())
            if mc_trial < best_mc - 1e-11:
                h = h_trial
                corr = corr_trial
                best_mc = mc_trial
                any_improve = True
        if not any_improve:
            break
    return h, best_mc * dx
