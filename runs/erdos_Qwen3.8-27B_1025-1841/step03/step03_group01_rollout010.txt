    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def smoothed_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc
    w = np.exp(T * shifted)
    sw = float(np.sum(w))
    if sw < 1e-300:
        w = np.zeros(2 * n - 1)
        w[int(np.argmax(corr))] = 1.0
    else:
        w = w / sw
    k_star_idx = int(np.argmax(w))
    w_max = w[k_star_idx]
    k_star = k_star_idx - (n - 1)
    if w_max > 0.99:
        grad = np.ones(n)
        k = k_star
        if k > 0:
            grad[:n - k] -= h[k:]
            grad[k:] -= h[:n - k]
        elif k < 0:
            ka = -k
            grad[ka:] -= h[:n - ka]
            grad[:n - ka] -= h[ka:]
        return grad * dx
    ks = np.arange(2 * n - 1) - (n - 1)
    thresh = 1e-10
    active_idx = np.where(w > thresh)[0]
    grad = np.zeros(n)
    grad[:] = float(np.sum(w[active_idx]))
    for idx in active_idx:
        ki = int(ks[idx])
        wi = float(w[idx])
        if ki > 0:
            grad[:n - ki] -= wi * h[ki:]
            grad[ki:] -= wi * h[:n - ki]
        elif ki < 0:
            ka = -ki
            grad[ka:] -= wi * h[:n - ka]
            grad[:n - ka] -= wi * h[ka:]
    return grad * dx

def fista_optimize(h_init, n, T, n_iter, time_cap):
    t0 = time.time()
    h = h_init.copy()
    z = h.copy()
    t_mom = 1.0
    L = float(n)
    step = 1.0 / (L * 0.5)
    for it in range(n_iter):
        if time.time() - t0 > time_cap:
            break
        if time.time() - _G_START > _G_BUDGET - 10:
            break
        grad = smoothed_grad(z, n, T)
        h_new = z - step * grad
        h_new = project_feasible(h_new, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_mom * t_mom)) / 2.0
        z = h_new + ((t_mom - 1.0) / t_new) * (h_new - h)
        z = project_feasible(z, n)
        t_mom = t_new
        h = h_new
    return h

def binary_flip_search(h, n, max_passes, time_cap):
    t0 = time.time()
    best_c5 = compute_c5(h)
    h_work = h.copy()
    ones_idx = np.where(h_work > 0.5)[0]
    zeros_idx = np.where(h_work <= 0.5)[0]
    for p in range(max_passes):
        if time.time() - t0 > time_cap:
            break
        if time.time() - _G_START > _G_BUDGET - 15:
            break
        improved = False
        n1 = len(ones_idx)
        nz = len(zeros_idx)
        if n1 == 0 or nz == 0:
            break
        n_try = min(n1 * nz, 2000)
        for t in range(n_try):
            if time.time() - t0 > time_cap:
                break
            if time.time() - _G_START > _G_BUDGET - 15:
                break
            i = ones_idx[t % n1]
            j = zeros_idx[t % nz]
            h_work[i] = 0.0
            h_work[j] = 1.0
            trial = make_feasible(h_work, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-12:
                best_c5 = c
                h_work = trial
                improved = True
                ones_idx = np.where(h_work > 0.5)[0]
                zeros_idx = np.where(h_work <= 0.5)[0]
            else:
                h_work[i] = 1.0
                h_work[j] = 0.0
        if not improved:
            break
    return h_work, best_c5

def pair_transfer_search(h, n, max_passes, time_cap):
    t0 = time.time()
    best_c5 = compute_c5(h)
    h_work = h.copy()
    deltas = [0.05, -0.05, 0.1, -0.1, 0.2, -0.2]
    for p in range(max_passes):
        if time.time() - t0 > time_cap:
            break
        if time.time() - _G_START > _G_BUDGET - 15:
            break
        improved = False
        idx = np.random.permutation(n)
        for t in range(min(len(idx), 1500)):
            if time.time() - t0 > time_cap:
                break
            i = idx[t]
            if h_work[i] <= 0.01 or h_work[i] >= 0.99:
                continue
            for d in deltas:
                new_v = h_work[i] + d
                if new_v < 0.0 or new_v > 1.0:
                    continue
                trial = h_work.copy()
                trial[i] = new_v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-11:
                    best_c5 = c
                    h_work = trial
                    improved = True
                    break
        if not improved:
            break
    return h_work, best_c5

def _obj_fn(h):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _G_N / 2.0

def _cons_jac(h):
    return np.ones(_G_N)

def slsqp_step(h, n, T, max_iter):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj_fn, h, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = budget_s
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = np.tile([1.0, 0.0], 320)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    def tl():
        return _G_BUDGET - (time.time() - _G_START)

    # Phase 1: Upsample to n=2048, FISTA optimization
    n_opt = 2048
    h_cur = upsample_h(best_h, n_opt)
    c5_cur = compute_c5(h_cur)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h_cur.copy()

    temps = [50.0, 100.0, 200.0, 500.0, 1000.0, 3000.0, 8000.0]
    for T in temps:
        if tl() < 60:
            break
        t_cap = min(30.0, tl() * 0.25)
        try:
            h_f = fista_optimize(h_cur, n_opt, T, int(2000.0 / max(T, 1.0)), t_cap)
            c5_f = compute_c5(h_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()
                h_cur = h_f.copy()
        except Exception:
            pass

    # Phase 2: SLSQP at a couple of temperatures
    for T in [500.0, 2000.0]:
        if tl() < 40:
            break
        try:
            h_s = slsqp_step(h_cur, n_opt, T, 200)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s.copy()
                h_cur = h_s.copy()
        except Exception:
            pass

    # Phase 3: Binary flip search
    if tl() > 40:
        h_b, c5_b = binary_flip_search(best_h, n_opt, max_passes=8,
                                       time_cap=min(120.0, tl() * 0.5))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

    # Phase 4: Pair transfer (continuous) local search
    if tl() > 30:
        h_t, c5_t = pair_transfer_search(best_h, n_opt, max_passes=5,
                                         time_cap=min(80.0, tl() * 0.4))
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h_t.copy()

    # Phase 5: Try higher resolution n=4096
    if tl() > 80:
        n_hi = 4096
        h_hi = upsample_h(best_h, n_hi)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
        # Quick FISTA at high res
        for T in [2000.0, 8000.0]:
            if tl() < 30:
                break
            try:
                h_hi2 = fista_optimize(h_hi, n_hi, T, 300, min(25.0, tl() * 0.3))
                c5_hi2 = compute_c5(h_hi2)
                if c5_hi2 < best_c5:
                    best_c5 = c5_hi2
                    best_h = h_hi2.copy()
                    h_hi = h_hi2.copy()
            except Exception:
                pass
        # Binary flip at high res
        if tl() > 30:
            h_b2, c5_b2 = binary_flip_search(best_h, len(best_h),
                                             max_passes=4, time_cap=min(60.0, tl() * 0.5))
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()

    # Phase 6: Final SLSQP polish
    if tl() > 25:
        n_fin = len(best_h)
        try:
            h_fin = slsqp_step(best_h, n_fin, 5000.0, 100)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        except Exception:
            pass

    # Phase 7: Final binary flip
    if tl() > 15:
        n_fin = len(best_h)
        h_bf, c5_bf = binary_flip_search(best_h, n_fin, max_passes=3,
                                         time_cap=min(20.0, tl() * 0.7))
        if c5_bf < best_c5:
            best_c5 = c5_bf
            best_h = h_bf.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
