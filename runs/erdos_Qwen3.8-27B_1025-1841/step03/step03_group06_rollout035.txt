    # Phase 4: Simulated annealing for escape
    if remaining() > 200:
        h_sa = best_h.copy()
        h_sa, c5_sa = simulated_annealing(h_sa, n_curr, time_limit_s=min(300, remaining() - 100),
                                           T_start=0.015, T_end=0.002)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # Greedy refinement after SA
        if remaining() > 100:
            h_g, c5_g = greedy_local_search(h_sa, n_curr, max_rounds=100,
                                            time_limit_s=min(100, remaining() - 60))
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g

    # Phase 5: SLSQP polish
    if remaining() > 80:
        n_curr = len(best_h)
        h_pol = slsqp_optimize(best_h, n_curr, T=400, maxiter=200, ftol=1e-14)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 6: Final binary local search
    if remaining() > 60:
        n_curr = len(best_h)
        h_bin = binary_project(best_h, n_curr)
        h_bin, c5_bin = greedy_local_search(h_bin, n_curr, max_rounds=200,
                                            time_limit_s=min(150, remaining() - 30))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Phase 7: Final SLSQP with very high T
    if remaining() > 40:
        n_curr = len(best_h)
        h_pol = slsqp_optimize(best_h, n_curr, T=800, maxiter=150, ftol=1e-14)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 8: One more local search pass
    if remaining() > 30:
        n_curr = len(best_h)
        h_bin = binary_project(best_h, n_curr)
        h_bin, c5_bin = greedy_local_search(h_bin, n_curr, max_rounds=100,
                                            time_limit_s=min(60, remaining() - 10))
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
