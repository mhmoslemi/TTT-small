        if time.time() - start > time_limit:
            break
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)

            p = np.zeros(len(top_idx))
            p[0] = 1.0  # np.argmax

            j_hi2 = min(n, n + s)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= pk * y[j_lo2 - s:j_hi2 - s]
        grad *= dx

        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)

        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-16:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 4000:
                lr *= 0.55
                stall = 0
                if lr < 1e-17:
                    break

        t_param = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def exact_subgrad_opt(h0, n, time_limit, lr=0.003):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.7

    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(corr.argmax())
        s = (n - 1) - k_star
        grad = grad_shift(h, s, n, dx)

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if it % 30 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-16:
                best_c5 = c5_now
                best_h = h_new.copy()
            else:
                lr *= 0.85
                v *= 0.5
                if lr < 1e-18:
                    break
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def binary_push(h, n):
    h_bin = (h > 0.5).astype(float)
    target = n / 2.0
    cur_sum = h_bin.sum()
    if cur_sum != target:
        thresholds = h[np.abs(h - 0.5) > 1e-6]
        if cur_sum > target:
            too_many = int(cur_sum - target)
            h_bin[h > 0.5] = 0.0
            top_idx = np.argsort(h)[::-1][:too_many]
            h_bin[top_idx] = 1.0
        else:
            too_few = int(target - cur_sum)
            h_bin[h < 0.5] = 1.0
            top_idx = np.argsort(-h)[:too_few]
            h_bin[top_idx] = 1.0
    h_bin = project_to_feasible(h_bin, n)
    return h_bin


def local_coord_search(h0, n, time_limit, step_size=0.05):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    rng = np.random.RandomState(123)

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(corr.argmax())
        s = (n - 1) - k_star
        grad = grad_shift(h, s, n, dx)

        candidates = np.where((h > 0.05) & (h < 0.95))[0]
        if len(candidates) == 0:
            break
        idxs = rng.choice(candidates, size=min(200, len(candidates)), replace=False)

        improved = False
        for idx in idxs:
            if time.time() - start > time_limit:
                break
            g_i = grad[int(idx)]
            if g_i > 0:
                h[int(idx)] = max(0.0, h[int(idx)] - step_size)
            elif g_i < 0:
                h[int(idx)] = min(1.0, h[int(idx)] + step_size)
            h = project_to_feasible(h, n)
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-16:
                best_c5 = c5_now
                best_h = h.copy()
                improved = True
            else:
                h = best_h.copy()

        if not improved:
            step_size *= 0.7
            if step_size < 0.001:
                break

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    n1 = 4000
    h1 = interp_periodic(h_init, n1)
    h1 = project_to_feasible(h1, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5, best_h = c5_1, h1.copy()

    # Phase 1: Nesterov tau=80
    rem = get_remaining(start_time, budget_s)
    if rem > 100:
        tl = min(80, rem * 0.12)
        h1, c5 = nesterov_opt(h1, n1, tau=80.0, time_limit=tl, lr=0.12, top_k=12)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: Nesterov tau=30
    rem = get_remaining(start_time, budget_s)
    if rem > 80:
        tl = min(70, rem * 0.1)
        h1, c5 = nesterov_opt(h1, n1, tau=30.0, time_limit=tl, lr=0.06, top_k=8)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 3: Nesterov tau=10
    rem = get_remaining(start_time, budget_s)
    if rem > 70:
        tl = min(60, rem * 0.09)
        h1, c5 = nesterov_opt(h1, n1, tau=10.0, time_limit=tl, lr=0.025, top_k=5)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 4: Nesterov tau=4
    rem = get_remaining(start_time, budget_s)
    if rem > 55:
        tl = min(50, rem * 0.08)
        h1, c5 = nesterov_opt(h1, n1, tau=4.0, time_limit=tl, lr=0.01, top_k=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 5: Exact max subgradient
    rem = get_remaining(start_time, budget_s)
    if rem > 50:
        tl = min(40, rem * 0.07)
        h1, c5 = exact_subgrad_opt(best_h, len(best_h), time_limit=tl, lr=0.002)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 6: Local coordinate search
    rem = get_remaining(start_time, budget_s)
    if rem > 40:
        tl = min(35, rem * 0.06)
        h1, c5 = local_coord_search(best_h, len(best_h), time_limit=tl, step_size=0.08)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 7: Binary push + refine
    rem = get_remaining(start_time, budget_s)
    if rem > 40:
        h_bin = binary_push(best_h, len(best_h))
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin.copy()
        tl = min(25, rem * 0.05)
        h1, c5 = nesterov_opt(h_bin, len(best_h), tau=5.0, time_limit=tl, lr=0.012, top_k=3)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 8: Multi-start with structured perturbations
    rem = get_remaining(start_time, budget_s)
    for pert in [0.01, 0.005, 0.02, 0.003]:
        rem = get_remaining(start_time, budget_s)
        if rem < 50:
            break
        h_p = best_h.copy() + np.random.randn(len(best_h)) * pert
        h_p = project_to_feasible(h_p, len(best_h))
        tl = min(35, rem * 0.08)
        h_p, c5_p = nesterov_opt(h_p, len(best_h), tau=5.0, time_limit=tl, lr=0.015, top_k=4)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()
        # Local search after each perturbation
        rem = get_remaining(start_time, budget_s)
        if rem > 25:
            tl2 = min(15, rem * 0.05)
            h2, c5_2 = exact_subgrad_opt(best_h, len(best_h), time_limit=tl2, lr=0.0015)
            if c5_2 < best_c5:
                best_c5, best_h = c5_2, h2.copy()

    # Phase 9: Try n=2000 (coarser, might find different basin)
    rem = get_remaining(start_time, budget_s)
    if rem > 60:
        n2 = 2000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(50, rem * 0.1)
        h2, c5_2 = nesterov_opt(h2, n2, tau=5.0, time_limit=tl, lr=0.012, top_k=4)
        # Upsample result back to n=4000
        h2_up = interp_periodic(h2, n1)
        h2_up = project_to_feasible(h2_up, n1)
        c5_up = compute_c5(h2_up)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h2_up.copy()
        c5_2_up = compute_c5(h2)
        if c5_2_up < best_c5:
            best_c5, best_h = c5_2_up, h2.copy()

    # Phase 10: Final polish
    rem = get_remaining(start_time, budget_s)
    if rem > 20:
        tl = min(15, rem * 0.4)
        h1, c5 = exact_subgrad_opt(best_h, len(best_h), time_limit=tl, lr=0.001)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 11: Final local search
    rem = get_remaining(start_time, budget_s)
    if rem > 15:
        tl = min(12, rem * 0.5)
        h1, c5 = local_coord_search(best_h, len(best_h), time_limit=tl, step_size=0.03)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Final projection
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
