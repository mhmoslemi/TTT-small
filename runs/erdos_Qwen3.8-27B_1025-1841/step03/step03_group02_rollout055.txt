**Custom projected gradient for the continuous case:**

The objective f(h) = max_k (c_k * dx) where c = correlate(h, 1-h, 'full')
This is a DC (difference of convex) function since c_k = sum_i h[i] - sum_i h[i]h[i+k] (roughly).

For a subgradient step:
- Find k* = argmax_k c_k
- subgradient: ∇f ≈ dx * ∇c_{k*}
- ∇c_k[i] = (1 - h[i-k]) - h[i+k] (with boundary handling)
- Update: h ← h - α * subgradient
- Project to [0,1] and sum(h) = n/2

The projection to the set {h : 0≤h≤1, sum(h)=n/2} can be done iteratively (like the make_feasible function but more careful).

```python
def proj_onto_constraint(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    h = h.copy()
    for _ in range(100):
        h = np.clip(h, 0.0, 1.0)
        diff = n/2.0 - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
    return h

def subgrad_step(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0-h, mode='full')
    k_star = np.argmax(corr)
    # Compute gradient of corr[k_star] w.r.t. h
    grad = np.zeros(n)
    # corr[k_star] = sum_i h[i] * (1-h[i-k_star+(n-1)])  (numpy convention)
    # Actually for numpy correlate(a, v, 'full'):
    # c[k] = sum_n a[n] * v[n - (k - (len(v)-1))]
    # With a=h (len n), v=1-h (len n):
    # c[k] = sum_i h[i] * (1-h)[i - (k-(n-1))]
    # = sum_i h[i] * (1 - h[i - k + n - 1])
    # Valid when 0<=i<n and 0<=i-k+n-1<n, i.e., k-n+1 <= i <= k
    # Let s = k - (n-1). Then c[k] = sum_i h[i] * (1-h[i-s]) for valid i.
    # d(c[k])/dh[i] = (1-h[i-s]) + sum_j h[j] * d(1-h[j-s])/dh[i]
    # = (1-h[i-s]) - h[i+s] (if j=i and i-s=i means... 
    # Wait: d(1-h[j-s])/dh[i] = -delta(j-s, i) = -1 if j-s=i, i.e., j=i+s
    # So: dc[k]/dh[i] = (1-h[i-s]) - h[i+s] (with validity conditions)
    s = k_star - (n - 1)
    for i in range(n):
        g = 0.0
        if 0 <= i - s < n:
            g += (1.0 - h[i - s])
        if 0 <= i + s < n:
            g -= h[i + s]
        grad[i] = g
    h_new = h - alpha * dx * grad
    return proj_onto_constraint(h_new, n)
```

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0

        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1

    curr_vals = np.correlate(b, 1.0 - b, mode='full')
    curr_c5 = float(np.max(curr_vals) * dx)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        
        # Every 10000 iters, identify worst shifts
        if it % 10000 == 0:
            top_k = np.argsort(curr_vals)[-5:]
            worst_shifts = top_k  # store for bias (not really using in this simple version)
        
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        b[i] = 0.0
        b[j] = 1.0
        new_vals = np.correlate(b, 1.0 - b, mode='full')
        new_c5 = float(np.max(new_vals) * dx)
        
        if new_c5 <= curr_c5 or (T > 1e-10 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            curr_vals = new_vals
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_c5

def nudge_optimize(h, n, max_iters, start_time, budget_s):
    """Coordinate nudge optimization for continuous h."""
    dx = 2.0 / n
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()
    rng = np.random.RandomState(777)
    
    deltas = [-0.02, -0.01, -0.005, -0.002, 0.002, 0.005, 0.01, 0.02, -0.05, 0.05]
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 5:
            break
        i = rng.randint(n)
        old = h[i]
        best_val = old
        best_c = curr_c5
        for d in deltas:
            nv = old + d
            if nv < 0.0 or nv > 1.0:
                continue
            trial = h.copy()
            trial[i] = nv
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-15:
                best_c = c
                best_val = nv
        if best_val != old:
            h = h.copy()
            h[i] = best_val
            h = make_feasible(h, n)
            curr_c5 = best_c
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
    return best_h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def slsqp_polish(h, n, T, maxiter, ftol):

        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def obj_grad(x):

        mc = np.max(corr)

        # grad[i] = (dx/T) * sum_s w_norm[s+(n-1)] * [(1-x[i-s])_valid - x[i+s]_valid]

                if 0 <= i + s < n:

            # Actually let me redo:
            g = 0.0

                if wv < 1e-12:
                    continue
                term = 0.0
                if 0 <= i - s < n:
                    term += (1.0 - x[i - s])
                if 0 <= i + s < n:

                g += wv * term

    bounds = [(0.0, 1.0)] * n
