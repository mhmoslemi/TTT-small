def top_k_penalty(h, n, k=5, T=20):
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k correlations
    idx = np.argpartition(-corr, k)[:k]
    top_vals = corr[idx]
    # Soft-sum of top values
    mc = np.max(top_vals)
    shifted = np.clip(T * (top_vals - mc), -80, 80)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)
