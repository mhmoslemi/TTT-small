_G_TOP_IDX = None
_G_T = 300.0

def time_remaining():
    return _G_BUDGET - (time.time() - _G_START)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def _corr_of(x):
    return np.correlate(x, 1.0 - x, mode='full')

def _obj_direct(x):
    corr = _corr_of(x)
    if _G_TOP_IDX is not None:
        return float(np.max(corr[_G_TOP_IDX]))
    return float(np.max(corr))

def _obj_direct_grad(x):
    n = _G_N
    corr = _corr_of(x)
    if _G_TOP_IDX is not None:
        vals = corr[_G_TOP_IDX]
        kk = int(_G_TOP_IDX[np.argmax(vals)])
    else:
        kk = int(np.argmax(corr))
    s = kk - (n - 1)
    val = float(corr[kk])
    grad = np.zeros(n)
    if s >= 0:
        grad[0:n-s] += (1.0 - x[s:n])
        grad[s:n] -= x[0:n-s]
    else:
        neg_s = -s
        grad[neg_s:n] += (1.0 - x[0:n-s])
        grad[0:n-s] -= x[neg_s:n]
    return val, grad

def _obj_sm(x):
    T = _G_T
    n = _G_N
    dx = 2.0 / n
    corr = _corr_of(x)
    if _G_TOP_IDX is not None:
        vals = corr[_G_TOP_IDX]
    else:
        vals = corr
    m = np.max(vals)
    shifted = vals - m
    return (m + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _get_active_shifts(h, frac=0.01, max_k=12):
    n = len(h)
    corr = _corr_of(h)
    mc = np.max(corr)
    threshold = mc - frac * mc
    active = np.where(corr >= threshold)[0]
    if len(active) > max_k:
        s = np.argsort(-corr[active])
        active = active[s[:max_k]]
    if len(active) == 0:
        active = np.array([int(np.argmax(corr))])
    return active

def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET, _G_N, _G_TOP_IDX, _G_T
    _G_START = time.time()
    _G_BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_pts = len(h_init)
    h_init = make_feasible(h_init, n_pts)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP with analytic gradient on top shifts
    for slsqp_round in range(6):
        if time_remaining() < 80:
            break
        n = len(best_h)
        _G_N = n
        active = _get_active_shifts(best_h, 0.01, 12)
        _G_TOP_IDX = active
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_direct, best_h, jac=_obj_direct_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 500, 'ftol': 1e-14})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 2: Soft-max SLSQP for smoothing
    for T_val in [150.0, 400.0]:
        if time_remaining() < 60:
            break
        n = len(best_h)
        _G_N = n
        _G_T = T_val
        active = _get_active_shifts(best_h, 0.005, 10)
        _G_TOP_IDX = active
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_sm, best_h, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 400, 'ftol': 1e-14})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 3: Coordinate descent (multiple passes, adaptive)
    for cd_pass in range(20):
        if time_remaining() < 25:
            break
        n = len(best_h)
        improved = False
        for j in range(n):
            if time_remaining() < 8:
                break
            old_val = best_h[j]
            best_local = old_val
            best_c_local = best_c5
            cands = [0.0, 1.0]
            for s in [0.002, 0.005, 0.01, 0.02]:
                cands.append(old_val + s)
                cands.append(old_val - s)
            for cand in cands:
                if cand < -1e-9 or cand > 1.0 + 1e-9:
                    continue
                if abs(cand - old_val) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = float(np.clip(cand, 0.0, 1.0))
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_local - 1e-15:
                    best_c_local = c
                    best_local = cand
            if best_local != old_val:
                best_h[j] = float(np.clip(best_local, 0.0, 1.0))
                best_h = make_feasible(best_h, n)
                best_c5 = best_c_local
                improved = True
        if not improved:
            break

    # Phase 4: Random perturbations + quick local opt
    for restart in range(10):
        if time_remaining() < 40:
            break
        n = len(best_h)
        scale = 0.005 + 0.004 * (restart % 5)
        h_try = best_h + rng.randn(n) * scale
        h_try = make_feasible(h_try, n)
        # 2-pass coord descent on subset
        for _ in range(2):
            for j in range(0, n, 2):
                if time_remaining() < 8:
                    break
                old_val = h_try[j]
                best_local = old_val
                best_c_local = compute_c5(h_try)
                for cand in [0.0, 1.0, old_val - 0.01, old_val + 0.01]:
                    if cand < 0 or cand > 1:
                        continue
                    trial = h_try.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c_local - 1e-14:
                        best_c_local = c
                        best_local = cand
                h_try[j] = best_local
                h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            # Continue from improved point
            for _ in range(2):
                if time_remaining() < 20:
                    break
                for j in range(0, len(best_h), 3):
                    old_val = best_h[j]
                    best_local = old_val
                    best_c_local = best_c5
                    for cand in [0.0, 1.0, old_val - 0.005, old_val + 0.005]:
                        if cand < 0 or cand > 1:
                            continue
                        trial = best_h.copy()
                        trial[j] = cand
                        trial = make_feasible(trial, len(best_h))
                        c = compute_c5(trial)
                        if c < best_c_local - 1e-14:
                            best_c_local = c
                            best_local = cand
                    if best_local != old_val:
                        best_h[j] = best_local
                        best_h = make_feasible(best_h, len(best_h))
                        best_c5 = best_c_local

    # Phase 5: Final SLSQP polish with gradient
    if time_remaining() > 60:
        n = len(best_h)
        _G_N = n
        active = _get_active_shifts(best_h, 0.005, 10)
        _G_TOP_IDX = active
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_direct, best_h, jac=_obj_direct_grad, method='SLSQP',
                         bounds=bounds, constraints=cons,
                         options={'maxiter': 400, 'ftol': 1e-15})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 6: Ultra-fine coordinate descent
    if time_remaining() > 15:
        n = len(best_h)
        for j in range(n):
            if time_remaining() < 4:
                break
            old_val = best_h[j]
            best_local = old_val
            best_c_local = best_c5
            for s in [0.0005, 0.001, 0.0015, 0.002]:
                for sign in [-1, 1]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c_local - 1e-16:
                            best_c_local = c
                            best_local = v
            if best_local != old_val:
                best_h[j] = best_local
                best_h = make_feasible(best_h, n)
                best_c5 = best_c_local

    # Phase 7: Binary snap
    if time_remaining() > 10:
        n = len(best_h)
        for j in range(n):
            if time_remaining() < 4:
                break
            for val in [0.0, 1.0]:
                if 0.001 < abs(best_h[j] - val) < 0.05:
                    trial = best_h.copy()
                    trial[j] = val
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_h = trial

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
