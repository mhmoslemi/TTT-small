def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_shift(h, shift, n):
    """Subgradient of f_shift(h) = sum_i h[i]*(1-h[i+shift]) w.r.t. h"""
    g = np.zeros(n)
    s = abs(shift)
    if s >= n:
        return g
    if shift >= 0:
        # f = sum_{i=0}^{n-1-s} h[i] * (1 - h[i+s])
        # df/dh[i] for i in [0, n-1-s]: (1 - h[i+s])
        # df/dh[i] for i in [s, n-1]: -h[i-s]
        g[s:] += -h[:n-s]  # h[i] appears as second factor for i >= s: coeff = -h[i-s]
        g[:n-s] += 1.0 - h[s:]  # h[i] appears as first factor: coeff = 1-h[i+s]
    else:
        # f = sum_{i=s}^{n-1} h[i] * (1 - h[i+s]) where s = -shift > 0
        # Let j = i+s, so f = sum_{i=s}^{n-1} h[i]*(1-h[i-s+s]) ... 
        # Actually for shift < 0, correlate gives:
        # result[n-1+shift] = sum_i h[i] * (1-h[i+shift])
        # where i ranges so that both i and i+shift are in [0,n-1]
        # If shift = -s < 0: i ranges from s to n-1, and i+shift = i-s ranges from 0 to n-1-s
        # f = sum_{i=s}^{n-1} h[i] * (1 - h[i-s])
        # df/dh[i] for i in [s, n-1]: (1 - h[i-s])  [first factor]
        # df/dh[i] for i in [0, n-1-s]: -h[i+s]  [second factor, since h[i] = h[(i+s)-s]]
        g[:n-s] += -h[s:]
        g[s:] += 1.0 - h[:n-s]
    return g

def active_grad(h, n, top_k=30, T=10.0):
    """Compute gradient using softmax-weighted active shifts"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if T > 1e-6:
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        w = exps / tot
    else:
        idx = np.argmax(corr)
        w = np.zeros(len(corr))
        w[idx] = 1.0
    
    # Only use top_k weights
    top_idx = np.argsort(w)[::-1][:top_k]
    w_sum = np.sum(w[top_idx])
    if w_sum < 1e-15:
        return subgrad_shift(h, int(np.argmax(corr)) - (n-1), n) * dx
    
    g = np.zeros(n)
    for i in top_idx:
        wi = w[i] / w_sum
        if wi < 1e-10:
            break
        shift = i - (n - 1)
        g += wi * subgrad_shift(h, shift, n)
    return g * dx

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    val = (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    return val

def project_c(h, n):
    target = n / 2.0

    g = np.zeros(n)

        return g
    if shift >= 0:

        g[s:] -= h[:n-s]

                print(f"  DIV: C5={bc:.6f}", flush=True)
    print(f"P6: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 7: Final SLSQP + peel
    if tl() > 60 and n <= 1024:
        if tl() > 40:
            bh = slsqp_smooth(bh, 1.5, 100)
            bc = compute_c5(bh)
        if tl() > 20:
            hp, cp = peel_step(bh, n, 150)
            if cp < bc:
                bc, bh = cp, hp.copy()
    print(f"P7: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 8: Final micro-FISTA
    if tl() > 15:
        hf, cf = fista(bh, 4000, 1.0, 0.01, 0.008, 0.002, 5, 20)
        if cf < bc:
            bc, bh = cf, hf.copy()

    bh = project_c(bh, n)
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c

    print(f"DONE: n={n} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, n
