        n_up = n0 * 2
        h_up = _upsample(best_h, n_up)
        c5_up = _compute_c5(h_up)
        if c5_up < best_c5 + 0.003:
            for T_val in [300.0, 1000.0, 2500.0]:
                if remaining() < 120:
                    break
                result = _run_slsqp(h_up, n_up, T_val, None, 300, 1e-16)
                c5_new = _compute_c5(result)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = result
                    h_up = result
            # Shift-targeted at higher res
            if remaining() > 100:
                n_pts = len(best_h)
                corr = np.correlate(best_h, 1.0 - best_h, mode='full')
                mc = np.max(corr)
                thresh = mc * 0.995
                top_idx = np.where(corr >= thresh)[0]
                if len(top_idx) > 10:
                    top_idx = top_idx[np.argsort(-corr[top_idx])[:10]]
                elif len(top_idx) == 0:
                    top_idx = np.array([int(np.argmax(corr))])
                result = _run_slsqp(best_h, n_pts, 2000.0, top_idx, 300, 1e-16)
                c5_new = _compute_c5(result)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = result

    # Phase 5: Binary snap (efficient - only check top shifts region)
    n_pts = len(best_h)
    for j in range(n_pts):
        if remaining() < 5:
            break
        for val in [0.0, 1.0]:
            if abs(best_h[j] - val) < 1e-7:
                continue
            trial = best_h.copy()
            trial[j] = val
            trial = _make_feasible(trial, n_pts)
            c = _compute_c5(trial)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_h = trial

    # Phase 6: Fine coordinate descent (fewer candidates for speed)
    for cd_pass in range(3):
        if remaining() < 20:
            break
        improved = False
        for j in range(n_pts):
            if remaining() < 8:
                break
            old_val = best_h[j]
            best_val = old_val
            best_c = best_c5
            for s in [0.02, 0.05, 0.1]:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 < v < 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = _make_feasible(trial, n_pts)
                        c = _compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = _make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    # Phase 7: Random restarts with SLSQP
    for r in range(5):
        if remaining() < 50:
            break
        n_pts = len(best_h)
        scale = 0.008 if r < 3 else 0.02
        h_try = _make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
        result = _run_slsqp(h_try, n_pts, 500.0, None, 200, 1e-15)
        c5_new = _compute_c5(result)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = result
        # Also try shift-targeted from perturbed point
        if remaining() > 40:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            mc = np.max(corr)
            thresh = mc * 0.996
            top_idx = np.where(corr >= thresh)[0]
            if len(top_idx) > 8:
                top_idx = top_idx[np.argsort(-corr[top_idx])[:8]]
            elif len(top_idx) == 0:
                top_idx = np.array([int(np.argmax(corr))])
            result2 = _run_slsqp(best_h, n_pts, 1500.0, top_idx, 200, 1e-15)
            c5_new2 = _compute_c5(result2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = result2

    # Phase 8: Final ultra-high T polish
    if remaining() > 30:
        n_pts = len(best_h)
        result = _run_slsqp(best_h, n_pts, 5000.0, None, 200, 1e-17)
        c5_new = _compute_c5(result)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = result

    # Phase 9: Final binary snap
    n_pts = len(best_h)
    for j in range(n_pts):
        if remaining() < 2:
            break
        for val in [0.0, 1.0]:
            if abs(best_h[j] - val) < 1e-7:
                continue
            trial = best_h.copy()
            trial[j] = val
            trial = _make_feasible(trial, n_pts)
            c = _compute_c5(trial)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_h = trial

    best_h = _make_feasible(best_h, len(best_h))
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, len(best_h)
