    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def slsqp_smooth(h, T, maxiter=400):
    """SLSQP minimizing smoothed max of correlations."""
    n = len(h)
    dx = 2.0 / n
    
    def obj(hv):
        corr = get_corr(hv)
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    
    def obj_grad(hv):
        corr = get_corr(hv)
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        w = exps / tot
        g = np.zeros(n)
        for i, wi in enumerate(w):
            if wi < 1e-10:
                continue
            g += wi * subgrad_shift(hv, i - (n-1))
        return g * dx
    
    def eq_fun(hv):
        return np.sum(hv) - n / 2.0
    
    def eq_jac(hv):
        return np.ones(n)
    
    res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-13})
    if res.x is not None:
        hh = project_c(res.x, n)
        if compute_c5(hh) < compute_c5(h):
            return hh
    return h

def peak_shave(h, n_iter=200, step=0.01):
    """Iteratively reduce the max correlation by stepping in its negative subgradient."""
    n = len(h)
    dx = 2.0 / n
    x = h.copy()
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        decay = 1.0 - it / n_iter
        corr = get_corr(x)
        mc = np.max(corr)
        # Top 5 shifts
        top5 = np.argsort(corr)[::-1][:5]
        g = np.zeros(n)
        for k in top5:
            g += subgrad_shift(x, k - (n-1))
        g /= 5.0
        # Normalize
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        g = g / gn
        lr = step * decay
        xn = project_c(x - lr * g * dx, n)
        c = compute_c5(xn)
        if c < bc:
            bc, bh = c, xn.copy()
            x = xn
        else:
            # Halve step
            xn = project_c(x - lr * 0.5 * g * dx, n)
            c = compute_c5(xn)
            if c < bc:
                bc, bh = c, xn.copy()
                x = xn
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

_SLS = {}

def _sls_obj(hv):

    dx = _SLS['dx']
    T = _SLS['T']
    center = n - 1
    corr = get_corr(hv)
    mc = np.max(corr)
    s = T * (corr - mc)
    s -= np.max(s)
    return (mc + np.max(s) + np.log(np.sum(np.exp(s))) / T) * dx

def _sls_obj_grad(hv):
    n = _SLS['n']
    dx = _SLS['dx']
    T = _SLS['T']
    center = n - 1
    corr = get_corr(hv)

    s -= np.max(s)

def slsqp_smooth(h, T, maxiter=400):
    n = len(h)
    _SLS['n'] = n
    _SLS['dx'] = 2.0 / n
    _SLS['T'] = T
    cur = compute_c5(h)
    try:
        res = minimize(_sls_obj, h, method='SLSQP', jac=_sls_obj_grad,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _sls_eq_fun, 'jac': _sls_eq_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hh = project_c(res.x, n)
            if compute_c5(hh) < cur:
                return hh
    except Exception:
        pass
    return h

def peak_shave(h, n_iter=300, base_step=0.008):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    x = h.copy()

                x = xn

    return bh, bc

    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])

def upsample_h(h, nn):
    if nn == len(h):
        return h.copy()

    nx = np.linspace(0, 2, nn, endpoint=False) + 1.0 / nn
    return project_c(np.interp(nx, ox, h), nn)
