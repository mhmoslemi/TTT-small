        c5_c = compute_c5(h_curr)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_curr.copy()

        for T, miter in temps:
            if tl() < 25:
                break
            h_opt = do_slsqp(h_curr, n_pts, T, miter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_curr = h_opt.copy()

        # Random restart at this resolution
        if tl() > 40 and n_pts <= 2560:
            h_rand = block_init(n_pts, rng)
            h_rand = 0.6 * h_rand + 0.4 * upsample_h(best_h, n_pts)
            h_rand = make_feasible(h_rand, n_pts)
            h_opt = do_slsqp(h_rand, n_pts, 50, 100)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 2: Targeted perturbation loop at highest res
    n_hi = len(best_h)
    for it in range(20):
        if tl() < 30:
            break
        strength = 0.05 * (1.0 - 0.7 * it / 20.0)
        h_p = targeted_perturb(best_h, n_hi, rng, strength)
        h_opt = do_slsqp(h_p, n_hi, 500, 60)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Greedy binary push + small moves
    n_cd = min(n_hi, 2048)
    h_cd = upsample_h(best_h, n_cd) if n_cd < n_hi else best_h.copy()
    for rnd in range(4):
        if tl() < 15:
            break
        improved = False
        for j in range(n_cd):
            if tl() < 15:
                break
            for bv in (0.0, 1.0):
                if abs(h_cd[j] - bv) < 1e-9:
                    continue
                trial = h_cd.copy()
                trial[j] = bv
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    h_cd = trial
                    improved = True
            for delta in (0.05, -0.05):
                cv = h_cd[j] + delta
                if cv < 0 or cv > 1:
                    continue
                trial = h_cd.copy()
                trial[j] = cv
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    h_cd = trial
                    improved = True
        if not improved:
            break

    c5_cd = compute_c5(h_cd)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy() if n_cd < n_hi else h_cd

    # Phase 4: Final refinement
    if tl() > 20:
        n_f = len(best_h)
        h_opt = do_slsqp(best_h, n_f, 2000, 80)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = make_feasible(best_h, len(best_h))
    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_c5 = c5_final

    return best_h, best_c5, len(best_h)
