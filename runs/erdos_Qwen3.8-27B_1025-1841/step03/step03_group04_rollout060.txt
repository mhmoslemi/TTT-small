So: current_corr[m] = C_m + h[j]*(1-h[j+l_m])*I(0≤j+l_m<n) + h[j-l_m]*(1-h[j])*I(0≤j-l_m<n)

where C_m is the sum over all other i.

Solving for C_m:
C_m = current_corr[m] - h[j]*(1-h[j+l_m])*I(0≤j+l_m<n) - h[j-l_m]*(1-h[j])*I(0≤j-l_m<n)

Now if I change h[j] to t:
new_corr[m] = C_m + t*(1-h[j+l_m])*I(0≤j+l_m<n) + h[j-l_m]*(1-t)*I(0≤j-l_m<n)
            = C_m + t*(1-h[j+l_m])*I(0≤j+l_m<n) + h[j-l_m]*I(0≤j-l_m<n) - t*h[j-l_m]*I(0≤j-l_m<n)

So: new_corr[m] = b_m + a_m * t
where:
a_m = (1-h[j+l_m])*I(0≤j+l_m<n) - h[j-l_m]*I(0≤j-l_m<n)
b_m = C_m + h[j-l_m]*I(0≤j-l_m<n)
    = current_corr[m] - h[j]*(1-h[j+l_m])*I(0≤j+l_m<n) - h[j-l_m]*(1-h[j])*I(0≤j-l_m<n) + h[j-l_m]*I(0≤j-l_m<n)
    = current_corr[m] - h[j]*(1-h[j+l_m])*I(0≤j+l_m<n) + h[j-l_m]*h[j]*I(0≤j-l_m<n)

Let me simplify:
b_m = current_corr[m] - h[j]*((1-h[j+l_m])*I(0≤j+l_m<n) - h[j-l_m]*I(0≤j-l_m<n))
    = current_corr[m] - h[j]*a_m

So: new_corr[m] = current_corr[m] - h[j]*a_m + a_m*t = current_corr[m] + a_m*(t - h[j])

That's much simpler! new_corr[m] = current_corr[m] + a_m * (t - h[j])

where a_m = (1-h[j+l_m])*I(0≤j+l_m<n) - h[j-l_m]*I(0≤j-l_m<n)

Great, so for a given j and a change t = h[j], the new correlation at shift m is:
new_corr[m] = corr[m] + a_m * (t - h[j])

And I want to choose t ∈ [0,1] to minimize max_m (corr[m] + a_m*(t-h[j])) over the top K shifts (approximately).

This is: minimize max_{m∈K} (corr[m] - a_m*h[j] + a_m*t) = minimize max_{m∈K} (b'_m + a_m*t)

This is a 1D convex minimax. The optimal t* is where the maximum transitions from one line to another.

Algorithm to find t*:
1. For each m in K, the line is L_m(t) = a_m*t + b'_m where b'_m = corr[m] - a_m*h[j]
2. At t=0: max_val_0 = max_m L_m(0)
3. At t=1: max_val_1 = max_m L_m(1)
4. The minimum of max_m L_m(t) over t∈[0,1] is at:
   - t=0 if the function is increasing at 0
   - t=1 if the function is decreasing at 1
   - Or at an intersection of two lines where the max changes

I can find the optimal t by:
- Evaluate the max at t=0, t=1, and at all pairwise intersections clipped to [0,1]
- Return the t with minimum max

For K lines, there are O(K²) intersections. For K=100, that's 5000, which is fine.

def optimize_single_coord(h, j, corr, top_idx, n):
    lags = top_idx - (n - 1)
    current_hj = h[j]
    
    # Compute a_m for each top correlation
    a = np.zeros(len(top_idx))
    for idx, m in enumerate(top_idx):
        l = lags[idx]
        am = 0.0
        if 0 <= j + l < n:
            am += (1.0 - h[j + l])
        if 0 <= j - l < n:
            am -= h[j - l]
        a[idx] = am
    
    # L_m(t) = corr[m] + a_m * (t - current_hj) = a_m * t + (corr[m] - a_m * current_hj)
    b = corr[top_idx] - a * current_hj
    
    # Find t in [0,1] minimizing max_m (a_m * t + b_m)
    candidates = [0.0, 1.0]
    
    # Pairwise intersections
    for i in range(len(top_idx)):
        for k in range(i+1, len(top_idx)):
            if abs(a[i] - a[k]) < 1e-15:
                continue
            t_int = (b[k] - b[i]) / (a[i] - a[k])
            if 0.0 < t_int < 1.0:
                candidates.append(t_int)
    
    # Also check where individual lines are flat (a_m = 0)
    # Those don't create intersections, just constants
    
    best_t = current_hj
    best_val = np.inf
    for t in candidates:
        t = np.clip(t, 0.0, 1.0)
        val = np.max(a * t + b)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t

        else:

    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    return np.sum(x) - n / 2.0

    return np.ones(n)

        if c5_up < best_c5 + 0.002:
            # SLSQP on upsampled
            bounds_u = [(0.0, 1.0)] * n_up
            cons_u = [{'type': 'eq', 'fun': lambda x: _con_fun(x, n_up),
                       'jac': lambda x: _con_jac(x, n_up)}]
            for T_u in [100.0, 300.0]:
                if remaining() < 30:
                    break
                obj_u = lambda x: _slsqp_obj(x, n_up, T_u)
                try:
                    res_u = minimize(obj_u, h_up, method='SLSQP', bounds=bounds_u,
                                    constraints=cons_u,
                                    options={'maxiter': 150, 'ftol': 1e-14})
                    c5_ru = compute_c5(res_u.x)
                    if c5_ru < c5_up:
                        c5_up = c5_ru
                        h_up = make_feasible(res_u.x, n_up)
                except Exception:
                    pass
            # Coordinate descent on upsampled
            if remaining() > 60:
                dl = start_time + budget_s - 30
                h_up, c5_up = exact_coord_descent(h_up, n_up, top_k=80, n_passes=10, deadline=dl)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
                h = best_h.copy()
                n = n_up

    # Phase 6: Random perturbations + reoptimize
    for restart in range(3):
        if remaining() < 40:
            break
        scale = 0.01 if restart < 2 else 0.02
        n_cur = len(h)
        h_try = make_feasible(h + rng.randn(n_cur) * scale, n_cur)
        if remaining() > 20:
            bounds = [(0.0, 1.0)] * n_cur
            cons = [{'type': 'eq', 'fun': lambda x: _con_fun(x, n_cur),
                     'jac': lambda x: _con_jac(x, n_cur)}]
            obj = lambda x: _slsqp_obj(x, n_cur, 300.0)
            try:
                res = minimize(obj, h_try, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 150, 'ftol': 1e-14})
                c5_r = compute_c5(res.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res.x, n_cur)
                    h = best_h.copy()
            except Exception:
                pass

    # Phase 7: Final coordinate descent polish
    if remaining() > 30:
        n_fin = len(h)
        deadline = start_time + budget_s - 5
        h_cd3, c5_cd3 = exact_coord_descent(h, n_fin, top_k=40, n_passes=10, deadline=deadline)
        if c5_cd3 < best_c5:
            best_c5 = c5_cd3
            best_h = h_cd3.copy()

    # Phase 8: Binary snap attempts
    if remaining() > 15:
        n_fin = len(h)
        h_try = best_h.copy()
        for j in range(n_fin):
            if remaining() < 3:
                break
            for val in [0.0, 1.0]:
                if abs(h_try[j] - val) < 1e-5:
                    continue
                h_try[j] = val
                h_try = make_feasible(h_try, n_fin)
                c = compute_c5(h_try)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = h_try.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
