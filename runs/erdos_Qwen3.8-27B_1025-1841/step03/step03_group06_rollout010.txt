    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    """Convert to binary with exactly n/2 ones using largest-remainder method."""
    # Sort by value descending, take top n/2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0

    return make_feasible(h_new, new_n)

def make_binary(h, n):
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2 if n % 2 == 0 else (n+1)//2]] = 1.0
    # Ensure exactly n/2 ones
    num_ones = n // 2
    if n % 2 == 0:
        h_bin = np.zeros(n)
        h_bin[idx[:num_ones]] = 1.0
    return h_bin

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def corr_after_swap(h, j1, j2, corr_base):
    """Compute new correlation after swapping h[j1]=1→0 and h[j2]=0→1.
    j1 is 1, j2 is 0. corr_base is the current full correlation array."""
    n = len(h)
    new_corr = corr_base.copy()
    
    # Effect of j1 going 1→0:
    # For each lag l, the change involves:
    # - h[j1] as the "first" factor: was contributing (1-h[j1-l]) or h[j1+l] depending on sign of l
    # - h[j1] as the "second" factor (inside 1-h[...]): was making that term 0, now it's h[...]
    
    # Let me just directly compute the change
    # corr_full[m] = sum_i h[i] * (1-h[i-m+n-1]) for valid i
    # Let lag = m - (n-1)
    # corr at lag l = sum_{valid i} h[i+l] * (1-h[i]) for l>=0
    #                = sum_{valid i} h[i] * (1-h[i-l]) for l<0 (substituting)
    
    # Actually, let me just use the definition directly:
    # np.correlate(a, b, 'full')[m] = sum_i a[i] * b[i-(m-n+1)]
    # with a=h, b=1-h:
    # result[m] = sum_i h[i] * (1 - h[i - m + n - 1])
    
    # For j1: h[j1] changes from 1 to 0
    # As "a": result[m] loses h[j1]*(1-h[j1-m+n-1]) = 1*(1-h[j1-m+n-1]) for valid index
    #   i.e., for m such that 0 <= j1-m+n-1 < n, i.e., j1-n+1 <= m <= j1
    #   Change: -(1-h[j1-m+n-1])
    # As "b" (inside 1-h[...]): result[m] had h[i]*(1-h[j1]) where i-m+n-1=j1, i=j1+m-n+1
    #   1-h[j1] was 0, now 1. So result[m] gains h[i] = h[j1+m-n+1]
    #   For valid: 0 <= j1+m-n+1 < n, i.e., n-1-j1 <= m <= n-1
    
    # For j2: h[j2] changes from 0 to 1
    # As "a": result[m] gains 1*(1-h[j2-m+n-1]) for valid index
    #   For 0 <= j2-m+n-1 < n: j2-n+1 <= m <= j2
    #   Change: +(1-h[j2-m+n-1])
    # As "b": result[m] had h[i]*(1-h[j2]) where i=j2+m-n+1
    #   1-h[j2] was 1, now 0. So result[m] loses h[i] = h[j2+m-n+1]
    #   For 0 <= j2+m-n+1 < n: n-1-j2 <= m <= n-1
    
    # This is getting complex but O(n) per swap. Let me just do it in a vectorized way.
    
    # Actually, for simplicity and correctness, let me just recompute:
    h[j1] = 0.0
    h[j2] = 1.0
    new_corr = np.correlate(h, 1.0 - h, mode='full')
    h[j1] = 1.0  # restore
    h[j2] = 0.0
    return new_corr

    best_sa_h = h_sa.copy()
    
    ones = np.where(h_sa == 1.0)[0]
    zeros = np.where(h_sa == 0.0)[0]
    
    T_start = 0.01
    T_end = 0.0001
    n_iter = 200000
    sa_time_limit = min(remaining() * 0.4, 400)
    sa_deadline = time.time() + sa_time_limit
    
    for it in range(n_iter):
        if time.time() > sa_deadline:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Temperature
        frac = it / n_iter
        T = T_start * (T_end / T_start) ** frac
        
        # Directed move: focus on the max lag
        # Get a random 1-position and a random 0-position
        # But bias towards positions that contribute to the max
        j1 = int(rng.choice(ones))
        j2 = int(rng.choice(zeros))
        
        if j1 == j2:
            continue

        h_sa[j1] = 0.0
        h_sa[j2] = 1.0
        corr_trial = np.correlate(h_sa, 1.0 - h_sa, mode='full')
        c5_trial = float(np.max(corr_trial) * dx)
        
        delta = c5_trial - c5_sa
        if delta < 0 or rng.random() < np.exp(-delta / T):
            # Accept
            c5_sa = c5_trial
            corr_sa = corr_trial
            ones = np.setdiff1d(ones, [j1])
            zeros = np.append(zeros, j1)
            zeros = np.setdiff1d(zeros, [j2])
            ones = np.append(ones, j2)
            if c5_sa < best_sa_c5:
                best_sa_c5 = c5_sa
                best_sa_h = h_sa.copy()
        else:
            # Reject
            h_sa[j1] = 1.0
            h_sa[j2] = 0.0
    
    if best_sa_c5 < best_c5:
        best_c5 = best_sa_c5
        best_h = best_sa_h.copy()
    
    # Phase 2: Try different resolutions
    for n_new in [480, 640, 960, 1280, 1920, 2560]:
        if remaining() < 200:
            break
        if n_new == n0:
            continue
        h_res = upsample_h(best_h, n_new)
        h_res_bin = make_binary(h_res, n_new)
        c5_res = compute_c5(h_res_bin)

            j1 = int(rng.choice(ones_s))
            j2 = int(rng.choice(zeros_s))
            if j1 == j2:
                continue
            h_s[j1] = 0.0
            h_s[j2] = 1.0
            corr_t = np.correlate(h_s, 1.0 - h_s, mode='full')
            c5_t = float(np.max(corr_t) * dx_new)
            delta = c5_t - c5_s
            if delta < 0 or rng.random() < np.exp(-delta / T):
                c5_s = c5_t
                corr_s = corr_t
                ones_s = np.setdiff1d(ones_s, [j1])
                zeros_s = np.append(zeros_s, j1)
                zeros_s = np.setdiff1d(zeros_s, [j2])
                ones_s = np.append(ones_s, j2)
                if c5_s < best_s_c5:
                    best_s_c5 = c5_s
                    best_s_h = h_s.copy()
            else:
                h_s[j1] = 1.0
                h_s[j2] = 0.0
        
        if best_s_c5 < best_c5:
            best_c5 = best_s_c5
            best_h = best_s_h.copy()
    
    # Phase 3: Non-binary SLSQP polish at best resolution
    n_best = len(best_h)
    if remaining() > 80:
        # Smooth max approximation
        _T = 100.0
        def obj_fn(x, n=n_best, T=_T):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            mc = np.max(corr)
            shifted = corr - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        
        def con_fun(x, n=n_best):
            return np.sum(x) - n / 2.0
        
        def con_jac(x, n=n_best):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n_best
        constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        try:
            res = minimize(obj_fn, best_h, method='SLSQP',
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': 200, 'ftol': 1e-13})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_best)
        except Exception:
            pass
    
    # Phase 4: Coordinate descent on non-binary
    if remaining() > 30:
        n_pts = len(best_h)
        dx_pts = 2.0 / n_pts
        for rnd in range(30):
            if remaining() < 10:
                break
            improved = False
            for j in range(n_pts):
                if remaining() < 5:
                    break
                old = best_h[j]
                best_c = best_c5
                best_val = old
                for d in [-0.02, -0.01, 0.0, 0.01, 0.02, -0.05, 0.05, -0.1, 0.1]:
                    cand = max(0.0, min(1.0, old + d))
                    if cand == old:
                        continue
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n_pts)
                    c = compute_c5(trial)
                    if c < best_c - 1e-13:
                        best_c = c
                        best_val = cand
                if best_val != old:
                    best_h = best_h.copy()
                    best_h[j] = best_val
                    best_h = make_feasible(best_h, n_pts)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break
    
    # Final
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)

Hmm wait, the SA approach with full np.correlate each time is going to be slow for large n. For n=2560, np.correlate takes maybe 5ms. 100000 iterations * 5ms = 500s. Too slow.

Let me use the efficient O(n) delta computation instead. Let me precompute the delta for a swap in O(n):

For swap (j1: 1→0, j2: 0→1), the change in corr at index m (0-indexed in the 'full' output) is:

The 'full' output has indices 0 to 2n-2.
result[m] = sum_i h[i] * (1 - h[i - m + n - 1])

Let shift = m - (n-1). Then:
result[m] = sum_i h[i] * (1 - h[i - shift])

For h[j1] going 1→0:
- As the outer h[i]: i=j1. Term was (1-h[j1-shift]) = (1-h[j1-shift]). Now 0.
  Change: -(1-h[j1-shift]) if 0 <= j1-shift < n, i.e., j1-n+1 <= shift <= j1
  In terms of m: j1-1 <= m <= j1+n-2
  
- As the inner h[i-shift]: i-shift = j1, i = j1+shift. Term was h[j1+shift]*(1-1) = 0. Now h[j1+shift]*(1-0) = h[j1+shift].
  Change: +h[j1+shift] if 0 <= j1+shift < n, i.e., -j1 <= shift <= n-1-j1
  In terms of m: n-1-j1 <= m <= 2n-2-j1

For h[j2] going 0→1:
- As the outer h[i]: i=j2. Term was 0. Now (1-h[j2-shift]).
  Change: +(1-h[j2-shift]) if 0 <= j2-shift < n, i.e., j2-n+1 <= shift <= j2
  In terms of m: j2-1 <= m <= j2+n-2
  
- As the inner h[i-shift]: i-shift = j2, i = j2+shift. Term was h[j2+shift]*(1-0) = h[j2+shift]. Now h[j2+shift]*(1-1) = 0.
  Change: -h[j2+shift] if 0 <= j2+shift < n, i.e., -j2 <= shift <= n-1-j2
  In terms of m: n-1-j2 <= m <= 2n-2-j2

So total delta[m] for the correlation array:
- From j1 outer: -(1-h[j1-shift]) for m in [j1-1, j1+n-2]
- From j1 inner: +h[j1+shift] for m in [n-1-j1, 2n-2-j1]
- From j2 outer: +(1-h[j2-shift]) for m in [j2-1, j2+n-2]
- From j2 inner: -h[j2+shift] for m in [n-1-j2, 2n-2-j2]

Where shift = m - (n-1).

Let me implement this:

```python
def compute_delta(h, j1, j2, n):
    """Compute the change in the 'full' correlation array when h[j1] 1->0, h[j2] 0->1."""
    delta = np.zeros(2*n-1)
    shift_arr = np.arange(2*n-1) - (n-1)  # -n+1 to n-1
    
    # j1 outer: -(1-h[j1-shift]) for valid shift
    mask1 = (j1 - shift_arr >= 0) & (j1 - shift_arr < n)
    idx1 = j1 - shift_arr[mask1]
    delta[mask1] -= (1.0 - h[idx1])
    
    # j1 inner: +h[j1+shift] for valid shift
    mask2 = (j1 + shift_arr >= 0) & (j1 + shift_arr < n)
    idx2 = j1 + shift_arr[mask2]
    delta[mask2] += h[idx2]
    
    # j2 outer: +(1-h[j2-shift]) for valid shift
    mask3 = (j2 - shift_arr >= 0) & (j2 - shift_arr < n)
    idx3 = j2 - shift_arr[mask3]
    delta[mask3] += (1.0 - h[idx3])
    
    # j2 inner: -h[j2+shift] for valid shift
    mask4 = (j2 + shift_arr >= 0) & (j2 + shift_arr < n)
    idx4 = j2 + shift_arr[mask4]
    delta[mask4] -= h[idx4]
    
    return delta
```

This is O(n) with vectorized numpy operations. Each call creates a (2n-1) array and does 4 masked operations. For n=960, this is very fast (~50μs perhaps).

Then the new max is: np.max(corr + delta) * dx

This makes the SA much faster: ~100μs per evaluation instead of ~1ms. So 10x faster.

    """Delta in 'full' correlation when h[j1] 1->0, h[j2] 0->1."""
    delta = np.zeros(2*n-1)
    shift_arr = np.arange(2*n-1) - (n-1)

    mask1 = (j1 - shift_arr >= 0) & (j1 - shift_arr < n)
    idx1 = j1 - shift_arr[mask1]
    delta[mask1] -= (1.0 - h[idx1])

    idx2 = j1 + shift_arr[mask2]
    delta[mask2] += h[idx2]

    idx3 = j2 - shift_arr[mask3]

    idx4 = j2 + shift_arr[mask4]

    # At shift = j2-j1: j1+shift = j2, j2-shift = j1

            # new: sum_i h_new[i]*(1-h_new[i-s])
            # Only terms involving j1 or j2 change
            # i=j1: old h[j1]*(1-h[j1-s]), new 0
            #   if j1-s == j2: old 1*(1-0)=1, new 0, delta=-1
            #   else: old 1*(1-h[j1-s]), new 0, delta=-(1-h[j1-s])
            # i=j2: old 0, new 1*(1-h[j2-s])
            #   if j2-s == j1: new 1*(1-0)=0, delta=0
            #   else: new 1*(1-h[j2-s]), delta=(1-h[j2-s])
            # i=j1+s (inner j1): old h[j1+s]*(1-h[j1])=0, new h_new[j1+s]*1
            #   if j1+s == j2: new 1, delta=1
            #   if j1+s == j1: (s=0, impossible if j1!=j2) new 0
            #   else: new h[j1+s], delta=h[j1+s]
            # i=j2+s (inner j2): old h[j2+s]*(1-h[j2])=h[j2+s], new h_new[j2+s]*0=0
            #   if j2+s == j1: new 0, old h[j1], delta=-h[j1]=-1
            #   if j2+s == j2: (s=0) old h[j2], new 0, delta=-h[j2]=0
            #   else: old h[j2+s], new 0, delta=-h[j2+s]
            
            old_terms = 0.0
            new_terms = 0.0
            # i = j1
            if 0 <= j1 - s < n:
                old_v = h[j1] * (1 - h[j1-s])
                if j1 - s == j2:
                    new_v = 0.0 * (1 - 1.0)  # h_new[j1]=0, h_new[j2]=1
                else:
                    new_v = 0.0 * (1 - h[j1-s])
                old_terms += old_v
                new_terms += new_v
            # i = j2
            if 0 <= j2 - s < n:
                old_v = h[j2] * (1 - h[j2-s])
                if j2 - s == j1:
                    new_v = 1.0 * (1 - 0.0)  # h_new[j2]=1, h_new[j1]=0
                else:
                    new_v = 1.0 * (1 - h[j2-s])
                old_terms += old_v
                new_terms += new_v
            # i = j1+s (inner j1)
            if 0 <= j1+s < n:
                old_v = h[j1+s] * (1 - h[j1])
                if j1+s == j2:
                    new_v = 1.0 * (1 - 0.0)  # h_new[j2]=1, h_new[j1]=0
                elif j1+s == j1:
                    new_v = 0.0 * (1 - 0.0)  # impossible if s!=0
                else:
                    new_v = h[j1+s] * (1 - 0.0)  # h_new[j1]=0
                old_terms += old_v
                new_terms += new_v
            # i = j2+s (inner j2)
            if 0 <= j2+s < n:
                old_v = h[j2+s] * (1 - h[j2])
                if j2+s == j1:
                    new_v = 0.0 * (1 - 0.0)  # h_new[j1]=0, h_new[j2]=1
                elif j2+s == j2:
                    new_v = 1.0 * (1 - 1.0)  # s=0
                else:
                    new_v = h[j2+s] * (1 - 1.0)  # h_new[j2]=1
                old_terms += old_v
                new_terms += new_v
            
            true_delta = new_terms - old_terms
            # Replace the computed delta at position m
            delta[m] = true_delta
    
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def slsqp_polish(h, n, time_limit=80):
    dx = 2.0 / n
    T_val = 200.0
    
    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T_val * shifted))) / T_val) * dx
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        deadline = time.time() + time_limit
        res = minimize(obj_fn, h.copy(), method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': 300, 'ftol': 1e-14})
        c5_res = compute_c5(res.x)
        return make_feasible(res.x, n), c5_res
    except Exception:
        return make_feasible(h, n), compute_c5(h)
