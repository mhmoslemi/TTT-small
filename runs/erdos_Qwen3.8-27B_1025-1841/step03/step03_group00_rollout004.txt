            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2

    # Phase 5: SLSQP polish at best resolution
    if remain() > 30:
        try:
            h_pol = slsqp_polish(best_h, len(best_h), 2000, max_iter=80)
            c5_pol = compute_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
        except:
            pass

    # Phase 6: Final intensive binary SA
    if remain() > 40:
        h_fin = (best_h > 0.5).astype(float).copy()
        nf = len(h_fin)
        nt = nf // 2
        nr = int(round(np.sum(h_fin)))
        if nr > nt:
            idx = np.where(h_fin == 1)[0]
            for k in range(nr - nt):
                h_fin[idx[k]] = 0.0
        elif nr < nt:
            idx = np.where(h_fin == 0)[0]
            for k in range(nt - nr):
                h_fin[idx[k]] = 1.0
        t6 = time.time() + min(remain() * 0.7, 250)
        h_fin2, c5_fin = sa_binary(h_fin, nf, rng, t6)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin2

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
