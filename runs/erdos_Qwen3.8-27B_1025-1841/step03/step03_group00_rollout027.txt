    """One subgradient step on the top-K worst lags"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Find top K lags
    top_k = np.argsort(corr)[-K:]
    grad = np.zeros(n)
    for k_idx in top_k:
        s = k_idx - (n - 1)  # actual shift
        A = np.zeros(n)
        B = np.zeros(n)
        if s >= 0:
            A[:n-s] = h[s:]  # A[j] = h[j+s] for j < n-s

            else:

                term2 = np.zeros(n)
                term1[sn:] = 1.0 - y[:n-sn]
                term2[:n-sn] = -y[sn:]
                grad += term1 + term2
        grad *= dx

        h = make_feasible(h, n)
        y = h + beta * (h - h_old)
        y = np.clip(y, 0, 1)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def exact_max_subgradient(h, n, r=None):
    """Returns (value, subgradient) of max(corr)*dx with random tie-breaking"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

5. But we need to make sure other lags don't increase too much
