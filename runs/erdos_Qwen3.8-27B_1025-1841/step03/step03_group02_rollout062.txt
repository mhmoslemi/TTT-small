import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_binary(h, n, n_iters, T0, T1, rng, start_time, budget_s, block_max=8):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(n_iters):
        if time.time() - start_time > budget_s - 60:
            break
        frac = float(it) / max(n_iters, 1)
        T = T0 * (T1 / T0) ** frac
        # Choose move type
        r = rng.random()
        if r < 0.3:
            # Single bit swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0; b[j] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i] = 1; b[j] = 0
        elif r < 0.7:
            # Block swap
            blen = rng.randint(2, block_max + 1)
            s1 = rng.randint(0, n - blen)
            s2 = rng.randint(0, n - blen)
            if abs(s1 - s2) < blen:
                continue
            seg1 = b[s1:s1+blen].copy()
            seg2 = b[s2:s2+blen].copy()
            # Swap ones and zeros between blocks
            ones1 = np.where(seg1 == 1)[0]
            zeros1 = np.where(seg1 == 0)[0]
            ones2 = np.where(seg2 == 1)[0]
            zeros2 = np.where(seg2 == 0)[0]
            k = min(len(ones1), len(zeros2), len(ones2), len(zeros1))
            if k == 0:
                continue
            io1 = rng.choice(ones1, size=k, replace=False)
            iz2 = rng.choice(zeros2, size=k, replace=False)
            io2 = rng.choice(ones2, size=k, replace=False)
            iz1 = rng.choice(zeros1, size=k, replace=False)
            b[s1+io1] = 0; b[s2+iz2] = 1
            b[s2+io2] = 0; b[s1+iz1] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[s1+io1] = 1; b[s2+iz2] = 0
                b[s2+io2] = 1; b[s1+iz1] = 0
        else:
            # Multi-bit flip (keep count constant)
            n_flip = rng.randint(2, 6)
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            flip_from = rng.choice(ones, size=n_flip, replace=False)
            flip_to = rng.choice(zeros, size=n_flip, replace=False)
            b[flip_from] = 0; b[flip_to] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[flip_from] = 1; b[flip_to] = 0
    return best_b, best_c5

    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_binary(h, n, n_iters, T0, T1, rng, start_time, budget_s, min_remain=60):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(n_iters):
        if time.time() - start_time > budget_s - min_remain:
            break
        frac = float(it) / max(n_iters, 1)
        T = T0 * (T1 / T0) ** frac
        r = rng.random()
        if r < 0.35:
            # Single swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0; b[j] = 1
            new_c5 = compute_c5(b)
            accept = new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T))
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5; best_b = b.copy()
            else:
                b[i] = 1; b[j] = 0
        elif r < 0.65:
            # Block swap
            blen = rng.randint(2, 10)
            s1 = rng.randint(0, n - blen)
            s2 = rng.randint(0, n - blen)
            if abs(s1 - s2) < blen:
                continue
            seg1_ones = np.where(b[s1:s1+blen] == 1)[0]
            seg1_zeros = np.where(b[s1:s1+blen] == 0)[0]
            seg2_ones = np.where(b[s2:s2+blen] == 1)[0]
            seg2_zeros = np.where(b[s2:s2+blen] == 0)[0]
            k = min(len(seg1_ones), len(seg2_zeros), len(seg2_ones), len(seg1_zeros))
            if k == 0:
                continue
            if rng.random() < 0.5:
                i1 = rng.choice(seg1_ones, size=k, replace=False)
                i2 = rng.choice(seg2_zeros, size=k, replace=False)
                j1 = rng.choice(seg2_ones, size=k, replace=False)
                j2 = rng.choice(seg1_zeros, size=k, replace=False)
            else:
                # Just swap the segments entirely (if same number of ones)
                if len(seg1_ones) != len(seg2_ones):
                    continue
                b[s1:s1+blen], b[s2:s2+blen] = b[s2:s2+blen].copy(), b[s1:s1+blen].copy()
                new_c5 = compute_c5(b)
                accept = new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T))
                if accept:
                    curr_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5; best_b = b.copy()
                else:
                    b[s1:s1+blen], b[s2:s2+blen] = b[s2:s2+blen].copy(), b[s1:s1+blen].copy()
                continue
            b[s1+i1] = 0; b[s2+i2] = 1
            b[s2+j1] = 0; b[s1+j2] = 1
            new_c5 = compute_c5(b)
            accept = new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T))
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5; best_b = b.copy()
            else:
                b[s1+i1] = 1; b[s2+i2] = 0
                b[s2+j1] = 1; b[s1+j2] = 0
        else:
            # Multi-swap (2-5 random pairs)
            nsw = rng.randint(2, 6)
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            fl = rng.choice(ones, size=nsw, replace=False)
            ft = rng.choice(zeros, size=nsw, replace=False)
            b[fl] = 0; b[ft] = 1
            new_c5 = compute_c5(b)
            accept = new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T))
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5; best_b = b.copy()
            else:
                b[fl] = 1; b[ft] = 0
    return best_b, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h):

        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)
    return h

    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0

        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

# Analytical gradient for soft-max objective
def softmax_obj_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    obj = (mc + log_sum / T) * dx
    p = exp_s / np.sum(exp_s)
    # A[i] = convolve(x, p, 'full')[i+n-1]
    # B[i] = correlate(x, p, 'full')[n-1+i]
    A = np.convolve(x, p, mode='full')[n-1 : n-1+n]
    B = np.correlate(x, p, mode='full')[n-1 : n-1+n]
    grad = dx * (1.0 - A - B)
    return obj, grad

def slsqp_optimize(h0, n, T, maxiter, ftol, start_time, budget_s):
    x0 = h0.copy()
    def fun(x):
        o, _ = softmax_obj_grad(x, T)
        return o
    def jac(x):
        _, g = softmax_obj_grad(x, T)
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, x0, method='SLSQP', jac=jac,
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

            best_h = b2.copy()

    # Phase 5: SLSQP T=500 (very sharp)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        try:
            x_res = slsqp_optimize(best_h, 500.0, 200, 1e-14, start_time, budget_s)
            x_res = make_feasible(x_res, n0)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass

    # Phase 6: Multi-start - perturb and re-optimize
    remaining = budget_s - (time.time() - start_time)
    if remaining > 400:
        for restart in range(2):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 180:
                break
            # Perturb: flip random 8% of bits
            h_pert = to_binary(best_h, n0).copy()
            n_flip = max(1, int(0.08 * n0 / 2))
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            flip_from = rng.choice(ones, size=n_flip, replace=False)
            flip_to = rng.choice(zeros, size=n_flip, replace=False)
            h_pert[flip_from] = 0
            h_pert[flip_to] = 1
            # Quick SA
            b_r, c5_r = sa_binary(h_pert, n0, 60000, 0.008, 0.00001, rng, start_time, budget_s, min_remain=120)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = b_r.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60 and c5_r < best_c5:
                try:
                    x_res = slsqp_optimize(best_h, 200.0, 200, 1e-14, start_time, budget_s)
                    x_res = make_feasible(x_res, n0)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = x_res.copy()
                except Exception:
                    pass

    # Phase 7: Try lower resolution n=480
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250 and n0 > 480:
        n_lo = 480
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, n_lo)
        h_lo = np.interp(new_x, old_x, best_h)
        h_lo = make_feasible(h_lo, n_lo)
        b_lo, c5_lo = sa_binary(h_lo, n_lo, 100000, 0.01, 0.00001, rng, start_time, budget_s, min_remain=100)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = b_lo.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            try:
                x_res = slsqp_optimize(best_h, 200.0, 200, 1e-14, start_time, budget_s)
                x_res = make_feasible(x_res, len(best_h))
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = x_res.copy()
            except Exception:
                pass

    # Phase 8: Coordinate descent on binding lags
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h = best_h.copy()
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent_binding(best_h, n_pts, 15, start_time, budget_s)

    # Phase 9: Final SLSQP with very sharp max
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        n_pts = len(best_h)
        try:
            x_res = slsqp_optimize(best_h, 1000.0, 150, 1e-15, start_time, budget_s)
            x_res = make_feasible(x_res, n_pts)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass

    # Phase 10: Final fine coordinate descent
    n_pts = len(best_h)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, best_c5 = coord_descent_binding(best_h.copy(), n_pts, 10, start_time, budget_s)

    # Final: ensure feasibility and recompute
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
