import numpy as np
from scipy.optimize import minimize
import time

def make_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15 * target:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    if T < 2.0:
        return mc * dx
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def make_con(n):
    return {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}

def slsqp_optimize(h_start, n, T, maxiter=200):
    def obj(x):
        return soft_max_obj(x, n, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x, nn=n: np.sum(x) - nn / 2.0, 'jac': lambda x, nn=n: np.ones(nn)}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_res = make_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        return h_res, c5_res
    except Exception:
        return h_start.copy(), compute_c5(h_start)

def lbfgs_optimize(h_start, n, T, maxiter=300):

    try:
        res = minimize(obj, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        h_res = make_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        return h_res, c5_res
    except Exception:
        return h_start.copy(), compute_c5(h_start)

def binary_snap(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        for val in [0.0, 1.0]:
            if abs(best[j] - val) < 1e-14:
                continue
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def coord_descent(h, n, n_rounds=5):
    best_c = compute_c5(h)
    cands_fixed = [0.0, 1.0, 0.5]
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):

                                        max(0.0, min(1.0, old - 0.005)),
                                        max(0.0, min(1.0, old + 0.005))]
            for cand in all_cands:

                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)

                    best_c_local = c
                    best_val = cand
            if best_val != old:
                h = best.copy() if best_val == old else h
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return h, best_c
