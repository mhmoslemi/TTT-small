            k = i2 - s

            cur_max = new_max

        else:
            # Metropolis acceptance
            diff = new_max - cur_max
            if rng.random() < np.exp(-diff / max(T, 1e-10)):
                ov = new_ov
                cur_max = new_max
                h[i1], h[i2] = h[i2], h[i1]
    
    return best_h, best_max * 2.0 / n

def compute_c5(h):

    target = n / 2.0
