import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def topk_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:K]
    g = np.zeros(n)
    for i in top_idx:
        g += _subgrad_one(h, i - (n - 1))
    return g * dx / K

def fista_topk(h, n_iter, K, lr_init, lr_end, check):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        lr = lr_init + (lr_end - lr_init) * f
        g = topk_grad(z, K)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:
            break
        g = g / gnorm * max(gnorm, 1e-10)  # keep as is actually
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def fw_step(h, g, n):
    """Frank-Wolfe step: find s in V minimizing g·s"""
    idx = np.argsort(g)[:n // 2]
    s = np.zeros(n)
    s[idx] = 1.0
    return s

def frank_wolfe(h, n_iter, K, check=50):
    n = len(h)
    x = h.astype(float).copy()
    bc = compute_c5(x)
    bh = x.copy()
    for t in range(1, n_iter + 1):
        g = topk_grad(x, K)
        s = fw_step(x, g, n)
        gamma = 2.0 / (t + 2.0)
        x_new = (1 - gamma) * x + gamma * s
        # Line search
        best_c = compute_c5(x_new)
        if best_c < bc:
            bc, bh = best_c, x_new.copy()
        x = x_new
        if t % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    return bh, bc

_OB = {}

def _sobj_topk(h):
    K = _OB['K']
    n = _OB['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Top K mean
    topk_vals = np.sort(corr)[::-1][:K]
    return np.mean(topk_vals) * dx

def _sobj_soft(h):
    T = _OB['T']
    n = _OB['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def _scon(h):
    return np.sum(h) - _OB['n'] / 2.0

def _sjac(h):
    return np.ones(_OB['n'])

def slsqp_topk(h, K, maxiter=500):

    _OB['K'], _OB['n'] = K, n
    try:
        r = minimize(_sobj_topk, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):

    except Exception:

    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(int(shift))
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def topk_gradient(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[::-1][:K]
    g = np.zeros(n)
    for i in top_idx:
        g += _subgrad_one(h, i - (n - 1))
    return g * dx / K

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    top_k_idx = np.argsort(w)[::-1][:min(K, 100)]
    g = np.zeros(n)
    for i in top_k_idx:
        if w[i] < 1e-12:
            break
        g += w[i] * _subgrad_one(h, i - (n - 1))
    return g * dx

    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()

        f = it / max(n_iter - 1, 1)
        lr = lr_init + (lr_end - lr_init) * f
        if T > 0:
            g = softmax_gradient(z, T)
        else:
            g = topk_gradient(z, K)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-15:

        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)

    c = compute_c5(x)

    return bh, bc

def frank_wolfe(h, n_iter, K, check=50):
    n = len(h)
    x = h.astype(float).copy()
    bc = compute_c5(x)
    bh = x.copy()

_OB = {}

def _obj_topk(h):
    K = _OB['K']
    n = _OB['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    tv = np.sort(corr)[::-1][:K]
    return float(np.mean(tv) * dx)

def _obj_soft(h):
    T = _OB['T']
    n = _OB['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx)

def _scon(h):
    return np.sum(h) - _OB['n'] / 2.0

def _sjac(h):
    return np.ones(_OB['n'])

def slsqp_topk(h, K, maxiter=400):
    n = len(h)
    _OB['K'], _OB['n'] = K, n
    try:
        r = minimize(_obj_topk, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-13})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h
