def smooth_max_optimize(h, n, T, max_iter=200):
    """Optimize using smooth max with L-BFGS-B."""
    dx = 2.0 / n
    
    def obj(x):
        # Project to feasible set
        x = np.clip(x, 0, 1)
        diff = n/2.0 - np.sum(x)
        x = x + diff / n
        x = np.clip(x, 0, 1)
        # Renormalize (approximate)
        for _ in range(5):
            diff = n/2.0 - np.sum(x)
            if abs(diff) < 1e-12:
                break
            x = x + diff / n
            x = np.clip(x, 0, 1)
        
        h_curr = x
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        logsum = np.log(np.sum(np.exp(T * shifted)))
        return (mc + logsum / T) * dx
    
    def grad(x):
        x = np.clip(x, 0, 1)
        diff = n/2.0 - np.sum(x)
        x = x + diff / n
        x = np.clip(x, 0, 1)
        for _ in range(5):
            diff = n/2.0 - np.sum(x)
            if abs(diff) < 1e-12:
                break
            x = x + diff / n
            x = np.clip(x, 0, 1)
        
        h_curr = x
        corr = np.array(np.correlate(h_curr, 1.0 - h_curr, mode='full'), dtype=float)
        mc = np.max(corr)
        weights = np.exp(T * (corr - mc))
        weights = weights / np.sum(weights)
        
        # Gradient of smooth max wrt h
        # d/dh_j of (mc + logsum/T)*dx
        # = dx * (1 + (1/T)*sum_k weights[k] * dlog(corr[k])/dh_j)
        # Since mc is one of the corr values, dmc/dh_j is the gradient of the max corr
        # This gets complicated. Let me just use the gradient of the smooth max directly.
        # 
        # smooth_max = mc + log(sum exp(T*(c_k - mc)))/T
        # d(smooth_max)/dh_j = dmc/dh_j + (1/T) * sum_k [T*weights_k * dc_k/dh_j - T*weights_k * dmc/dh_j]
        # = dmc/dh_j * (1 - sum_k weights_k) + (1/T) * sum_k T*weights_k * dc_k/dh_j
        # = sum_k weights_k * dc_k/dh_j  (since sum weights = 1)
        
        # So gradient is weighted sum of gradients of each corr[k]
        g = np.zeros(n)
        for k in range(len(corr)):
            if weights[k] < 1e-10:
                continue
            lag = k - (n - 1)
            js = np.arange(n)
            idx1 = js + lag
            mask1 = (idx1 >= 0) & (idx1 < n)
            g[mask1] += weights[k] * (1.0 - h_curr[idx1[mask1]])
            idx2 = js - lag
            mask2 = (idx2 >= 0) & (idx2 < n)
            g[mask2] -= weights[k] * h_curr[idx2[mask2]]
        
        return g * dx
    
    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-8})
    return make_feasible(res.x, n)

def coordinate_descent(h, n, time_limit=60):
    """Simple coordinate descent polish."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    dx = 2.0 / n
    target_sum = n / 2.0
    
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    
    for sweep in range(50):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old = h[j]
            for d in deltas:
                new_val = old + d
                if new_val < 0 or new_val > 1:
                    continue
                trial = h.copy()
                trial[j] = new_val
                # Fix sum
                diff = target_sum - np.sum(trial)
                if abs(diff) > 1e-10:
                    # Spread correction
                    mask = trial < 1.0 if diff > 0 else trial > 0.0
                    cnt = np.sum(mask)
                    if cnt == 0:
                        continue
                    trial[mask] = np.clip(trial[mask] + diff/cnt, 0, 1)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = trial
                    old = new_val
                    improved = True
            # Try binary values
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = bv
                diff = target_sum - np.sum(trial)
                if abs(diff) > 1e-10:
                    mask = trial < 1.0 if diff > 0 else trial > 0.0
                    cnt = np.sum(mask)
                    if cnt == 0:
                        continue
                    trial[mask] = np.clip(trial[mask] + diff/cnt, 0, 1)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = trial
                    improved = True
        if not improved:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def compute_gradients(h, n, top_idx):
    m = len(top_idx)
    grads = np.zeros((m, n))
    js = np.arange(n)
    for ii, i in enumerate(top_idx):
        lag = i - (n - 1)
        idx1 = js + lag
        mask1 = (idx1 >= 0) & (idx1 < n)
        grads[ii, mask1] += (1.0 - h[idx1[mask1]])
        idx2 = js - lag
        mask2 = (idx2 >= 0) & (idx2 < n)
        grads[ii, mask2] -= h[idx2[mask2]]
    return grads

def sca_optimize(h, n, n_active=50, max_iters=40, time_limit=120):
    h = h.copy()
    start = time.time()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-n_active:]
        c0 = corr[top_idx].copy()
        grads = compute_gradients(h, n, top_idx)
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        rhs = grads @ h - c0
        for ii in range(len(top_idx)):
            constraints.append(cp.sum(grads[ii] * h_var) - t_var <= rhs[ii])
        prob = cp.Problem(cp.Minimize(t_var), constraints)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.CLP]:
            try:
                prob.solve(solver=solver, verbose=False)
                if h_var.value is not None:
                    solved = True
                    break
            except Exception:
                continue
        if not solved:
            break
        h_new = np.clip(h_var.value, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5_new
        else:
            h_damp = 0.5 * h + 0.5 * h_new
            h_damp = make_feasible(h_damp, n)
            c5_damp = compute_c5(h_damp)
            if c5_damp < best_c5 - 1e-13:
                h = h_damp
                best_c5 = c5_damp
            else:
                break
    return h, compute_c5(h)

def smooth_max_optimize(h, n, T, max_iter=150):
    dx = 2.0 / n
    target_sum = n / 2.0

    def _project(x):
        x = np.clip(x, 0.0, 1.0)
        for _ in range(20):
            diff = target_sum - np.sum(x)
            if abs(diff) < 1e-13:
                break
            if diff > 0:
                mask = x < 1.0 - 1e-12
            else:
                mask = x > 1e-12
            cnt = np.sum(mask)
            if cnt == 0:
                break
            x[mask] = np.clip(x[mask] + diff / cnt, 0.0, 1.0)
        return x

    def obj(x):
        h_c = _project(x)
        corr = np.correlate(h_c, 1.0 - h_c, mode='full').astype(float)
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    def grad(x):
        h_c = _project(x)
        corr = np.array(np.correlate(h_c, 1.0 - h_c, mode='full'), dtype=float)
        mc = np.max(corr)
        weights = np.exp(T * (corr - mc))
        weights /= np.sum(weights)
        g = np.zeros(n)
        js = np.arange(n)
        for k in range(len(corr)):
            if weights[k] < 1e-12:
                continue
            lag = k - (n - 1)
            idx1 = js + lag
            m1 = (idx1 >= 0) & (idx1 < n)
            g[m1] += weights[k] * (1.0 - h_c[idx1[m1]])
            idx2 = js - lag
            m2 = (idx2 >= 0) & (idx2 < n)
            g[m2] -= weights[k] * h_c[idx2[m2]]
        return g * dx

    try:
        res = minimize(obj, _project(h), jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-9})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h.copy(), n)

def coordinate_descent(h, n, time_limit=45):
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    target_sum = n / 2.0
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]

    for sweep in range(100):
        if time.time() - start > time_limit:
            break
        improved = False
        rng_local = np.random.RandomState(sweep * 7 + 3)
        order = np.arange(n)
        rng_local.shuffle(order)
        for j in order:
            if time.time() - start > time_limit:
                break
            old = h[j]
            for d in deltas:
                nv = old + d
                if nv < 0 or nv > 1:
                    continue
                trial = h.copy()
                trial[j] = nv
                diff = target_sum - np.sum(trial)
                if abs(diff) > 1e-12:
                    mask = trial < 1.0 if diff > 0 else trial > 0.0
                    cnt = np.sum(mask)
                    if cnt == 0:
                        continue
                    trial[mask] = np.clip(trial[mask] + diff / cnt, 0, 1)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = trial
                    old = nv
                    improved = True
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = bv
                diff = target_sum - np.sum(trial)
                if abs(diff) > 1e-12:
                    mask = trial < 1.0 if diff > 0 else trial > 0.0
                    cnt = np.sum(mask)
                    if cnt == 0:
                        continue
                    trial[mask] = np.clip(trial[mask] + diff / cnt, 0, 1)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = trial
                    improved = True
        if not improved:
            break
    return h, best_c5
