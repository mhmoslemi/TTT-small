import numpy as np
import time

def project_to_feasible(g, n):
    """Project g to [0,1]^n with sum = n/2."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def softmax_grad(h, n, tau, top_k):
    """Compute soft-max approximation and its gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    max_c = top_vals.max()
    shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        pk = p[idx]
        k = int(top_idx[idx])
        s = k - (n - 1)
        if s >= 0:
            grad[s:] += pk * (1.0 - h[:n-s])
            grad[:n-s] -= pk * h[s:]
        else:
            m = -s
            grad[:n-m] += pk * (1.0 - h[m:])
            grad[m:] -= pk * h[:n-m]
    grad *= dx
    obj = dx * (max_c + np.log(exp_c.sum()) / tau)
    return obj, grad

    best_h = h.copy()

    v = np.zeros(n)
