def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def smooth_obj(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -100.0, 100.0)
    soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    viol = np.sum(h) - n / 2.0
    return soft + mu * viol * viol

        try:
            if attempt == 0:
                hx = h0.copy()
            else:
                hx = best_x + 0.1 * np.random.randn(n)
                hx = make_feasible(hx, n)

                          method='L-BFGS-B', bounds=bounds,
                          options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
            x = make_feasible(res.x, n)
            f = compute_c5(x)
            if f < best_f:
                best_f, best_x = f, x.copy()
        except Exception:
            pass
    return best_x, best_f

def binary_local_search(h_bin, n, rng, max_rounds=200, tries_per_round=5000):
    h = h_bin.astype(float).copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        for _ in range(tries_per_round):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-16:
                current = c
                ones = np.concatenate([ones[:ii], ones[ii+1:]])
                zeros = np.concatenate([zeros[:jj], zeros[jj+1:]])
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def to_binary(h, n):
    hb = (h > 0.5).astype(float)
    n_ones = int(np.sum(hb))
    n_target = n // 2
    if n_ones > n_target:
        idx = np.where(hb == 1)[0]
        for k in range(n_ones - n_target):
            hb[idx[k]] = 0.0
    elif n_ones < n_target:
        idx = np.where(hb == 0)[0]
        for k in range(n_target - n_ones):
            hb[idx[k]] = 1.0
    return make_feasible(hb, n)

def coord_descent(h, n, max_rounds=3, time_limit=1200, t0=None):
    if t0 is None:
        t0 = time.time()
    current = compute_c5(h)
    deltas = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for rnd in range(max_rounds):
        if (time.time() - t0) > time_limit:
            break
        improved = False
        for j in range(n):
            if (time.time() - t0) > time_limit:
                break
            ov = h[j]
            new_val = ov
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                h[j] = nv
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < current - 1e-14:
                    current = c
                    new_val = nv
                    improved = True
                    break
                else:
                    h[j] = ov
            if h[j] != new_val:
                continue
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                h[j] = bv
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < current - 1e-14:
                    current = c
                    improved = True
                    break
                else:
                    h[j] = new_val
        if not improved:
            break
    return h, current
