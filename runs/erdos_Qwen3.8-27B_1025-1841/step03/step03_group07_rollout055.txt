import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.5
    hi = g.max() + 1.5
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def soft_max_obj_grad(h, tau, top_k=15):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    tc = tau * (corr - max_c)
    exp_tc = np.exp(tc)
    p_sum = exp_tc.sum()
    p = exp_tc / p_sum
    obj = dx * (max_c + np.log(p_sum) / tau)
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    grad = np.zeros(n)
    for idx in top_idx:
        pk = p[idx]
        if pk < 1e-9:
            continue
        s = idx - n + 1
        j1_lo = max(0, -s)
        j1_hi = min(n, n - s)
        j2_lo = max(0, s)
        j2_hi = min(n, n + s)
        if j1_lo < j1_hi:
            grad[j1_lo:j1_hi] += pk * (1.0 - h[j1_lo + s:j1_hi + s])
        if j2_lo < j2_hi:
            grad[j2_lo:j2_hi] -= pk * h[j2_lo - s:j2_hi - s]
    grad *= dx
    return obj, grad

    h = project_to_feasible(h.copy(), n)

    t = 1.0

        mu = (t - 1.0) / (t + 2.0)

        h_new = project_to_feasible(h_new, n)
        obj_new, _ = soft_max_obj_grad(h_new, tau, top_k)
