_G_T = 1.0

def _obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n * 0.5
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def fast_coord_eval(overlaps, h, j, new_val, n):
    """O(n) evaluation of max overlap after changing h[j] to new_val."""
    old_val = h[j]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return overlaps.max()
    center = n - 1
    new_ov = overlaps.copy()
    # First factor: h[j]*(1-h[j+d]), d = n-1-m
    # m ranges over valid indices where j+d is in [0,n)
    lo_m = max(0, n - 1 - (n - 1 - j))  # simplify: m where j + (n-1-m) in [0,n)
    # j + n - 1 - m >= 0 → m <= j + n - 1
    # j + n - 1 - m < n → m > j - 1 → m >= j
    m_lo = max(0, j)
    m_hi = min(2 * n - 2, j + n - 1)
    for m in range(m_lo, m_hi + 1):
        d = n - 1 - m
        partner = j + d
        if 0 <= partner < n:
            new_ov[m] += delta * (1.0 - h[partner])
    # Second factor: h[j-d]*(1-h[j]), where j-d in [0,n)
    # m where j - (n-1-m) in [0,n) → m-1-(n-1) <= j-m+n-1... 
    # j - d = j - (n-1-m) = j - n + 1 + m in [0, n)
    # m in [n-1-j, n-1-j+n-1] = [n-1-j, 2n-2-j]
    m_lo2 = max(0, n - 1 - j)
    m_hi2 = min(2 * n - 2, 2 * n - 2 - j)
    for m in range(m_lo2, m_hi2 + 1):
        d = n - 1 - m
        partner = j - d
        if 0 <= partner < n and m != center:
            new_ov[m] -= delta * h[partner]
    # Center: nonlinear
    new_ov[center] += new_val * (1.0 - new_val) - old_val * (1.0 - old_val)
    return new_ov.max()

def coord_descent_fast(h, n, max_rounds=8, time_limit=90):
    start = time.time()
    best_c = compute_c5(h)
    overlaps = compute_overlaps(h)
    candidates = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old = h[j]
            best_cv = best_c
            best_cand = old
            # Quick check: 0 and 1
            for cv in [0.0, 1.0]:
                if abs(cv - old) < 1e-12:
                    continue
                mc = fast_coord_eval(overlaps, h, j, cv, n)
                if mc < best_cv - 1e-14:
                    best_cv = mc
                    best_cand = cv
            # Try intermediate values
            if best_cand == old:
                for cv in [0.25, 0.5, 0.75]:
                    if abs(cv - old) < 1e-12:
                        continue
                    mc = fast_coord_eval(overlaps, h, j, cv, n)
                    if mc < best_cv - 1e-14:
                        best_cv = mc
                        best_cand = cv
            if best_cand != old:
                h = h.copy()
                h[j] = best_cand
                h = make_feasible(h, n)
                overlays = compute_overlaps(h)
                actual_c = float(overlaps.max())
                if actual_c < best_c - 1e-15:
                    best_c = actual_c
                    overlaps = overlays
                    improved = True
                else:
                    # Revert
                    h[j] = old
                    h = make_feasible(h, n)
                    overlaps = compute_overlaps(h)
        if not improved:
            break
    return h, best_c

def slsqp_optimize(h, n, T, maxiter, ftol=1e-13):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(res.x, n)
    except Exception:
        return h

def binary_local_search(h, n, time_limit=60):
    """Local search on binary representation: swap pairs of bits."""
    start = time.time()
    target = n // 2
    b = (h >= 0.5).astype(float)
    # Ensure exactly target ones
    num_ones = int(np.sum(b))
    if num_ones > target:
        ones_idx = np.where(b == 1)[0]
        if len(ones_idx) > 0:
            order = np.argsort(-h[ones_idx])
            for i in range(num_ones - target):
                b[ones_idx[order[i]]] = 0.0
    elif num_ones < target:
        zeros_idx = np.where(b == 0)[0]
        if len(zeros_idx) > 0:
            order = np.argsort(-h[zeros_idx])
            for i in range(target - num_ones):
                b[zeros_idx[order[i]]] = 1.0
    best_c = compute_c5(b)
    best_b = b.copy()
    rng = np.random.RandomState(123)
    for _ in range(500):
        if time.time() - start > time_limit:
            break
        ones = np.where(best_b == 1)[0]
        zeros = np.where(best_b == 0)[0]
        if len(ones) < 2 or len(zeros) < 2:
            break
        j1 = ones[rng.randint(len(ones))]
        j2 = zeros[rng.randint(len(zeros))]
        best_b[j1] = 0.0
        best_b[j2] = 1.0
        c = compute_c5(best_b)
        if c < best_c:
            best_c = c
        else:
            best_b[j1] = 1.0
            best_b[j2] = 0.0
    return best_b, best_c

        h_trial = slsqp_optimize(best_h.copy(), n, T, maxiter=400)
        c = compute_c5(h_trial)
        if c < best_c5:
            best_c5 = c
            best_h = h_trial

    # Phase 2: Fast coordinate descent
    if time_left() > 120:
        best_h, best_c5 = coord_descent_fast(best_h, n, max_rounds=10, time_limit=80)

    # Phase 3: Try higher resolutions
    resolutions = [1280, 1920, 2560]
    for new_n in resolutions:
        if time_left() < 150:
            break
        h_up = upsample_h(best_h, new_n)
        # SLSQP at new resolution
        h_up = slsqp_optimize(h_up, new_n, T=400.0, maxiter=300)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up
            n = new_n
        # Coordinate descent at new resolution
        if time_left() > 60:
            h_up2, c_up2 = coord_descent_fast(h_up, new_n, max_rounds=6, time_limit=50)
            if c_up2 < best_c5:
                best_c5 = c_up2
                best_h = h_up2

    # Phase 4: Binary local search
    if time_left() > 40:
        n_cur = len(best_h)
        b_sol, b_c = binary_local_search(best_h, n_cur, time_limit=30)
        if b_c < best_c5:
            best_c5 = b_c
            best_h = b_sol
            n_cur = len(best_h)
        # Continuous refinement from binary
        if time_left() > 30:
            h_from_bin = b_sol.astype(float)
            h_refined = slsqp_optimize(h_from_bin, n_cur, T=500.0, maxiter=200)
            c_ref = compute_c5(h_refined)
            if c_ref < best_c5:
                best_c5 = c_ref
                best_h = h_refined

    # Phase 5: Final SLSQP polish
    n_final = len(best_h)
    if time_left() > 40:
        h_pol = slsqp_optimize(best_h.copy(), n_final, T=800.0, maxiter=300, ftol=1e-14)
        c_pol = compute_c5(h_pol)
        if c_pol < best_c5:
            best_c5 = c_pol
            best_h = h_pol

    # Phase 6: Final coordinate descent
    if time_left() > 20:
        best_h, best_c5 = coord_descent_fast(best_h, len(best_h), max_rounds=6, time_limit=int(min(30, time_left() - 5)))

    # Phase 7: Random perturbation restarts
    n_f = len(best_h)
    for pert in [0.02, 0.05, 0.01]:
        if time_left() < 30:
            break
        h_pert = best_h + rng.randn(n_f) * pert
        h_pert = make_feasible(h_pert, n_f)
        h_pert = slsqp_optimize(h_pert, n_f, T=500.0, maxiter=200)
        c_pert = compute_c5(h_pert)
        if c_pert < best_c5:
            best_c5 = c_pert
            best_h = h_pert
        if time_left() > 10:
            h_pert2, c_pert2 = coord_descent_fast(h_pert, n_f, max_rounds=4, time_limit=15)
            if c_pert2 < best_c5:
                best_c5 = c_pert2
                best_h = h_pert2

    # Final: ensure feasibility and compute exact C5
    best_h = make_feasible(best_h, len(best_h))
    np.clip(best_h, 0.0, 1.0, out=best_h)
    # Fix sum exactly
    diff = len(best_h) * 0.5 - np.sum(best_h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(best_h - 0.5))
        best_h[idx] += diff
        best_h = np.clip(best_h, 0.0, 1.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
