import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def max_corr_val(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def project_binary(h, n):
    """Project to binary preserving sum = n/2"""
    h = h.copy()
    # Sort by value descending, set top n/2 to 1
    thresh = np.sort(h)[::-1][n//2 - 1] if n > 0 else 0
    h_bin = (h >= thresh).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    if n_ones != n // 2:
        # Adjust
        if n_ones > n // 2:
            # Need to flip some to 0
            ones_idx = np.where(h_bin == 1)[0]
            # Flip the ones with smallest original values
            order = np.argsort(h[ones_idx])
            for k in range(n_ones - n // 2):
                h_bin[ones_idx[order[k]]] = 0
        else:
            zeros_idx = np.where(h_bin == 0)[0]
            order = np.argsort(-h[zeros_idx])
            for k in range(n // 2 - n_ones):
                h_bin[zeros_idx[order[k]]] = 1
    return h_bin

def simulated_annealing(h, n, start_time, budget, n_iter=50000):
    """SA on binary representation with swap moves"""
    h_bin = project_binary(h, n)
    best_h = h_bin.copy()
    best_c = max_corr_val(h_bin)
    cur_h = h_bin.copy()
    cur_c = best_c
    
    T_start = 5.0
    T_end = 0.01
    total_iters = n_iter
    
    zero_positions = np.where(cur_h == 0)[0]
    one_positions = np.where(cur_h == 1)[0]
    
    for it in range(total_iters):
        if time.time() - start_time > budget - 20:
            break
        
        # Geometric cooling
        T = T_start * (T_end / T_start) ** (it / total_iters)
        
        # Pick random swap
        j = zero_positions[np.random.randint(len(zero_positions))]
        k = one_positions[np.random.randint(len(one_positions))]
        
        # Perform swap
        cur_h[j], cur_h[k] = cur_h[k], cur_h[j]
        new_c = max_corr_val(cur_h)
        
        # Determine acceptance
        delta = new_c - cur_c
        if delta < 0 or np.random.random() < np.exp(-delta / (T * 0.01)):
            # Accept
            cur_c = new_c
            # Update position lists
            zero_positions = np.where(cur_h == 0)[0]
            one_positions = np.where(cur_h == 1)[0]
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            # Reject
            cur_h[j], cur_h[k] = cur_h[k], cur_h[j]
    
    return best_h, best_c

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def project_binary(h, n):
    h = h.copy()
    # Sort indices by value descending
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def sa_binary(h, n, start_time, budget, n_iter=50000, seed=42):
    rng = np.random.RandomState(seed)
    h_bin = project_binary(h, n)
    best_h = h_bin.copy()
    best_mc = float(np.max(np.correlate(h_bin, 1.0 - h_bin, mode='full')))
    cur_h = h_bin.copy()
    cur_mc = best_mc
    
    T = 5.0
    T_min = 0.01
    cool_rate = (T_min / T) ** (1.0 / max(n_iter, 1))
    
    for it in range(n_iter):
        if time.time() - start_time > budget:
            break
        T = max(T * cool_rate, T_min)

        j = rng.randint(n)
        k = rng.randint(n)
        if cur_h[j] == cur_h[k]:
            continue

        new_mc = float(np.max(np.correlate(cur_h, 1.0 - cur_h, mode='full')))
        delta = new_mc - cur_mc

                best_h = cur_h.copy()
