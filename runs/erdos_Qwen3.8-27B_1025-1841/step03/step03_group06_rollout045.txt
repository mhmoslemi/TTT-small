    if remaining() > 60:
        h_start = best_h.copy()
        for T in [500.0, 800.0]:
            if remaining() < 30:
                break
            x_res = slsqp_optimize(h_start, len(h_start), T, 300)
            c5_r = compute_c5(x_res)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(x_res, len(x_res))
            h_start = make_feasible(x_res, len(x_res))

    # Phase 6: Binary snap
    if remaining() > 30:
        best_h, best_c5 = binary_snap(best_h, len(best_h))

    # Phase 7: Final fine coordinate descent
    if remaining() > 20:
        best_h, best_c5 = fine_cd(best_h, len(best_h), start_time + remaining() - 5, max_rounds=20)

    # Phase 8: One more pair descent pass
    if remaining() > 15:
        best_h, best_c5 = pair_descent(best_h, len(best_h), start_time + remaining() - 5, max_rounds=5)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
