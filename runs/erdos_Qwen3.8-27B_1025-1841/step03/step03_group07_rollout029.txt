import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def subgrad_shift(h, d, n):
    """Subgradient of c[k]*dx where d = k-(n-1)."""
    dx = 2.0 / n
    grad = np.zeros(n)
    i_lo = max(0, d)
    i_hi = min(n, n + d)
    if i_lo < i_hi:
        grad[i_lo:i_hi] = 1.0 - h[i_lo - d:i_hi - d]
    j_lo = max(0, i_lo - d)
    j_hi = min(n, i_hi - d)
    if j_lo < j_hi:
        grad[j_lo:j_hi] -= h[j_lo + d:j_hi + d]
    return grad * dx

def momentum_subgrad(h, n, time_limit, top_k=2, lr=0.05, momentum=0.9, 
                     refresh_every=50, explore_every=500, explore_mag=0.01):
    """Momentum subgradient with periodic exploration."""
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    it = 0
    stall = 0
    
    while time.time() - start < time_limit:
        # Periodically refresh the top-k shifts
        if it % refresh_every == 0:
            corr = compute_corr_vals(h)
            if top_k < len(corr):
                top_idx = np.argpartition(corr, -top_k)[-top_k:]
            else:
                top_idx = np.arange(len(corr))
            # Weight by how much each exceeds the median
            c_max = corr.max()
            c_med = np.median(corr)
            weights = np.maximum(corr[top_idx] - c_med, 0.01)
            weights /= weights.sum()
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k_idx in zip(weights, top_idx):
            d = k_idx - (n - 1)
            grad += w * subgrad_shift(h, d, n)
        
        # Momentum step
        v = momentum * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        # Periodic exploration
        if explore_every > 0 and it > 0 and it % explore_every == 0:
            h_new = project_to_feasible(h_new + np.random.randn(n) * explore_mag, n)
        
        # Check improvement every 20 steps
        if it % 20 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h_new.copy()
                # Increase lr slightly on improvement
                lr *= 1.05
                stall = 0
            else:
                stall += 20
                if stall > 1000:
                    lr *= 0.7
                    v *= 0.5
                    stall = 0
                    if lr < 1e-8:
                        break
        
        h = h_new
        it += 1
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def simulated_annealing(h, n, time_limit, T_init=0.05, T_final=0.001, 
                        n_perturb=10, perturb_mag=0.1):
    """Simple simulated annealing: randomly perturb n_perturb positions."""
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = T_init
    total_iter_est = time_limit / 0.01  # rough estimate
    it = 0
    
    while time.time() - start < time_limit:
        # Decide number of iterations this "epoch" (to update T)
        h_trial = h.copy()
        idx = np.random.choice(n, size=n_perturb, replace=False)
        h_trial[idx] = np.clip(h_trial[idx] + np.random.randn(n_perturb) * perturb_mag, 0, 1)
        h_trial = project_to_feasible(h_trial, n)
        
        c5_trial = compute_c5(h_trial)
        c5_curr = compute_c5(h)
        
        if c5_trial < c5_curr or np.random.rand() < np.exp(-(c5_trial - c5_curr) / T):
            h = h_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
        
        it += 1
        # Update temperature
        elapsed = time.time() - start
        T = T_init * (T_final / T_init) ** (elapsed / time_limit)
        
        # Safety: break if T too small
        if T < 1e-6:
            break
    
    return best_h, best_c5

def pairwise_fix(h, d, n, n_iter=100):
    """For shift d, find and fix anti-correlated pairs."""
    dx = 2.0 / n
    i_lo = max(0, d)
    i_hi = min(n, n + d)
    # For i in [i_lo, i_hi): c includes h[i](1-h[i-d])
    # We want to reduce this, so we want h[i] and h[i-d] to be similar
    for _ in range(n_iter):
        i = np.random.randint(i_lo, i_hi)
        j = i - d
        # If h[i] > h[j], reduce h[i] and increase h[j] (or vice versa)
        # But we need to maintain sum(h) = n/2
        # So: h[i] -= eps, h[j] += eps (if h[i] > h[j])
        # or h[i] += eps, h[j] -= eps (if h[i] < h[j])
        eps = 0.01
        if h[i] > h[j] + 0.01:
            delta = min(eps, (h[i] - h[j]) / 2)
            h[i] -= delta
            h[j] += delta
        elif h[j] > h[i] + 0.01:
            delta = min(eps, (h[j] - h[i]) / 2)
            h[i] += delta
            h[j] -= delta
    return h

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_corr_vals(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)

def subgrad_shift(h, d, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    i_lo = max(0, d)
    i_hi = min(n, n + d)
    if i_lo < i_hi:
        grad[i_lo:i_hi] = 1.0 - h[i_lo - d:i_hi - d]
    j_lo = max(0, i_lo - d)
    j_hi = min(n, i_hi - d)
    if j_lo < j_hi:
        grad[j_lo:j_hi] -= h[j_lo + d:j_hi + d]
    return grad * dx

def momentum_subgrad(h, n, time_limit, top_k=1, lr=0.05, momentum=0.9,
                     refresh_every=30, explore_every=500, explore_mag=0.005):
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    it = 0
    stall = 0
    
    # Initial top-k
    corr = compute_corr_vals(h)
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    c_max = corr.max()
    weights = np.ones(top_k) / top_k
    
    while time.time() - start < time_limit:
        # Periodic refresh
        if it % refresh_every == 0:
            corr = compute_corr_vals(h)
            if top_k < len(corr):
                top_idx = np.argpartition(corr, -top_k)[-top_k:]
            else:
                top_idx = np.arange(len(corr))
            c_max = corr.max()
            c_min_top = corr[top_idx].min()
            spread = c_max - c_min_top
            if spread > 1e-10:
                weights = (corr[top_idx] - c_min_top) / spread
                weights /= weights.sum()
            else:
                weights = np.ones(top_k) / top_k
        
        # Compute weighted subgradient
        grad = np.zeros(n)
        for w, k_idx in zip(weights, top_idx):
            d = k_idx - (n - 1)
            grad += w * subgrad_shift(h, d, n)
        
        # Gradient clipping
        gn = np.linalg.norm(grad)
        if gn > 1.0:
            grad *= 1.0 / gn
        
        # Momentum step
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
        
        # Periodic exploration
        if explore_every > 0 and it > 0 and it % explore_every == 0:
            h = project_to_feasible(h + np.random.randn(n) * explore_mag, n)
        
        # Check improvement
        if it % 20 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
                lr = min(lr * 1.02, 0.5)
                stall = 0
            else:
                stall += 20
                if stall > 800:
                    lr *= 0.7
                    v *= 0.3
                    stall = 0
                    if lr < 1e-8:
                        break
        
        it += 1
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def pairwise_refine(h, n, time_limit, n_pairs=50, eps=0.02):
    """Directly fix anti-correlated pairs for the worst shift."""
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    it = 0
    
    while time.time() - start < time_limit:
        corr = compute_corr_vals(h)
        k_star = corr.argmax()
        d = k_star - (n - 1)
        i_lo = max(0, d)
        i_hi = min(n, n + d)
        
        # Find the worst pairs and fix them
        for _ in range(n_pairs):
            i = np.random.randint(i_lo, i_hi)
            j = i - d

            elif h[j] > h[i] + 0.02:

def simulated_annealing(h, n, time_limit, T_init=0.02, n_perturb=10, perturb_mag=0.08):
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    c5_curr = best_c5
    T = T_init
    
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        T = T_init * 0.01 ** (elapsed / time_limit)
        
        h_trial = h.copy()
        idx = np.random.choice(n, size=n_perturb, replace=False)
        h_trial[idx] += np.random.randn(n_perturb) * perturb_mag
        h_trial = project_to_feasible(h_trial, n)
        
        c5_trial = compute_c5(h_trial)
        
        if c5_trial < c5_curr or np.random.rand() < np.exp(-(c5_trial - c5_curr) / max(T, 1e-8)):
            h = h_trial
            c5_curr = c5_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
    
    return best_h, best_c5
