    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, K=10, T=1.0):
    """Gradient using top-K correlations with softmax weighting."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top K indices
    top_idx = np.argpartition(-corr, K)[:K]
    top_corr = corr[top_idx]
    # Softmax over top K
    exps = np.exp(T * (top_corr - top_corr.max()))
    w = exps / exps.sum()
    g = np.zeros(n)
    for i, idx in enumerate(top_idx):
        if w[i] < 1e-12:
            continue
        shift = idx - (n - 1)
        s = abs(shift)
        if s >= n:
            continue
        if shift >= 0:
            g[s:] += w[i] * (1.0 - h[:n-s])
            g[:n-s] -= w[i] * h[s:]
        else:
            g[:n-s] += w[i] * (1.0 - h[s:])
            g[s:] -= w[i] * h[:n-s]
    return g * dx

        tk = tn

            if c < bc:

    c = compute_c5(x)

        mc = np.max(c)
        # Soft-smoothed max

        mc = np.max(c)
        exps = np.exp(T * (c - mc))

        if wi < 1e-12:
            continue
        shift = top_idx[i] - (n - 1)
        s = abs(shift)
        if s >= n:
            continue
        if shift >= 0:
            g[s:] += wi * (1.0 - h[:n - s])
            g[:n - s] -= wi * h[s:]
        else:
            g[:n - s] += wi * (1.0 - h[s:])
            g[s:] -= wi * h[:n - s]
    return g * dx

def fista_opt(h, n_iter, T0, T1, K0, K1, lr0, lr1, check=40):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        K = max(1, int(round(K0 + (K1 - K0) * f)))
        lr = lr0 + (lr1 - lr0) * f
        g = topk_grad(z, K, T)
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def _slsqp_obj(x, T, n, dx):
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _slsqp_jac(x, T, n):
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    exps = np.exp(T * (c - mc))
    w = exps / exps.sum()
    g = np.zeros(n)
    for i in range(len(c)):
        wi = w[i]
        if wi < 1e-11:
            continue
        shift = i - (n - 1)
        s = abs(shift)
        if s >= n:
            continue
        if shift >= 0:
            g[s:] += wi * (1.0 - x[:n - s])
            g[:n - s] -= wi * x[s:]
        else:
            g[:n - s] += wi * (1.0 - x[s:])
            g[s:] -= wi * x[:n - s]
    return g * dx

def slsqp_polish(h, T, maxiter):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        return _slsqp_obj(x, T, n, dx)
    def jac(x):
        return _slsqp_jac(x, T, n)
    def con_eq(x):
        return np.sum(x) - n / 2.0
    try:
        r = minimize(obj, h, jac=jac, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_eq}],
                     options={'maxiter': maxiter, 'ftol': 1e-13})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        bh = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        bh = project_c(rng.rand(960), 960)
    n0 = len(bh)
    bc = compute_c5(bh)
    print(f"I: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1: Broad exploration (high K, moderate T)
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h1, c1 = fista_opt(bh, 12000, 40.0, 2.0, 80, 8, 2.0, 0.8, 60)
        if c1 < bc:
            bc, bh = c1, h1.copy()
        print(f"P1: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 2: Focused exploitation (low K, low T)
    rem = budget_s - (time.time() - t0)
    if rem > 250:
        h2, c2 = fista_opt(bh, 12000, 3.0, 0.05, 5, 1, 15.0, 3.0, 40)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        print(f"P2: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 3: Very fine (K=1, T=0)
    rem = budget_s - (time.time() - t0)
    if rem > 200:
        h3, c3 = fista_opt(bh, 10000, 1.0, 0.01, 1, 1, 40.0, 8.0, 30)
        if c3 < bc:
            bc, bh = c3, h3.copy()
        print(f"P3: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 4: SLSQP polish
    rem = budget_s - (time.time() - t0)
    np_ = len(bh)
    if np_ <= 1200 and rem > 150:
        for TV in [15.0, 6.0, 2.5, 1.0]:
            rem = budget_s - (time.time() - t0)
            if rem < 40:
                break
            hx = slsqp_polish(bh, TV, 500)
            cx = compute_c5(hx)
            if cx < bc:
                bc, bh = cx, hx
        print(f"P4: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 5: Upsample to 1.5x and optimize
    rem = budget_s - (time.time() - t0)
    nn1 = min(int(n0 * 1.5), 1440)
    if nn1 > n0 and rem > 200:
        hu = upsample(bh, nn1)
        cu = compute_c5(hu)
        hg, cg = fista_opt(hu, 10000, 20.0, 0.5, 40, 4, 2.0, 1.0, 50)
        if cg < cu:
            hu, cu = hg, cg
        hn, cn = fista_opt(hu, 10000, 1.5, 0.03, 3, 1, 20.0, 4.0, 40)
        if cn < cu:
            hu, cu = hn, cu
        rem = budget_s - (time.time() - t0)
        if nn1 <= 1200 and rem > 80:
            hu = slsqp_polish(hu, 3.0, 400)
        cf = compute_c5(hu)
        if cf < bc:
            bc, bh, n0 = cf, hu.copy(), nn1
        print(f"P5(n={nn1}): C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 6: Perturbation restarts
    rem = budget_s - (time.time() - t0)
    np_ = len(bh)
    for trial in range(4):
        rem = budget_s - (time.time() - t0)
        if rem < 40:
            break
        sc = 0.02 + 0.025 * trial
        hp = project_c(bh + rng.randn(np_) * sc, np_)
        hr, cr = fista_opt(hp, 8000, 12.0, 0.3, 8, 1, 8.0, 2.0, 50)
        if cr < bc:
            bc, bh = cr, hr.copy()
            print(f"  R{trial}: C5={bc:.6f}", flush=True)

    # Phase 7: Final SLSQP + fine-tune
    rem = budget_s - (time.time() - t0)
    np_ = len(bh)
    if np_ <= 1200 and rem > 30:
        bh = slsqp_polish(bh, 1.5, 400)
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        hf, cf = fista_opt(bh, 3000, 0.8, 0.01, 1, 1, 30.0, 5.0, 20)
        if cf < compute_c5(bh):
            bh, bc = hf, cf

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
