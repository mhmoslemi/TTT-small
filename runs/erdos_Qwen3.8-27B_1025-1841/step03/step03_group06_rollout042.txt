from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    """Project h onto {x: 0<=x_i<=1, sum(x)=n/2} using bisection."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection on threshold
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    return proj

def subgradient_descent(h, n, rng, max_time=30):
    alpha0 = 0.1
    c0 = compute_c5(h)
    best_h = h.copy()
    best_c = c0
    momentum = 0.8
    v = np.zeros(n)
    
    for t in range(10000):
        if time.time() - start_time > max_time + start_time:  # need to pass start_time
            break
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)  # lag
        
        # Compute subgradient
        g = np.zeros(n)
        if k >= 0:
            for j in range(n - k):
                g[j + k] += (1.0 - h[j])
                g[j] -= h[j + k]
        else:
            p = -k
            for i in range(n - p):
                g[i] += (1.0 - h[i + p])
                g[i + p] -= h[i]
        
        # Nesterov momentum
        alpha = alpha0 / (1.0 + 0.01 * t)
        v = momentum * v - alpha * g
        h_new = h + v
        h_new = project_feasible(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo+hi)/2.0, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binarize(h, n):
    """Round h to binary while maintaining sum constraint."""
    h_bin = (h >= 0.5).astype(float)
    return make_feasible(h_bin, n)

def swap_delta(h, i, j, n):
    """Compute Δcorr for swapping h[i] and h[j]."""
    a, b = h[i], h[j]
    if abs(a - b) < 1e-15:
        return None
    d = b - a  # change at position i
    # Δc_k = d * [(1-h[j-k]) - h[j+k]] + (-d) * [(1-h[i-k]) - h[i+k]]
    #       = d * [h[i-k] - h[j-k] + h[i+k] - h[j+k]]
    result = np.zeros(2*n-1)
    h_ext = np.zeros(3*n)  # h embedded in larger array for easy indexing
    h_ext[n:2*n] = h  # h[m] at position n+m in h_ext
    
    # h[i-k]: valid when 0 <= i-k < n → -i < -k < n-i → i-n < k < i+1
    # In result array, position = n-1+k
    # h[i-k] for k from max(1-n, i-n+1) to min(n-1, i)
    for offset in [i, j]:
        pass  # This loop approach is too slow, let me use vectorized ops
    
    # Vectorized approach:
    # Vector v1[k] = h[i-k] for k in [-(n-1), n-1], else 0
    # This is h reversed, shifted
    # h[i-k] = h_rev[k-i+n-1] where h_rev = h[::−1]
    # Actually: h[i-k] for k=0 is h[i], k=1 is h[i-1], etc.
    # In result array index n-1+k:
    #   k=0 → index n-1, value h[i]
    #   k=-1 → index n-2, value h[i+1]
    #   k=1 → index n, value h[i-1]
    # So the values h[i], h[i-1], ..., h[0] go at indices n-1, n, ..., n-1+i
    # And h[i+1], h[i+2], ... go at indices n-2, n-3, ...
    
    # Simpler: just use the formula directly with numpy arrays
    # result[n-1+k] = d * (h[i-k] - h[j-k] + h[i+k] - h[j+k])
    
    # h[i-k] for k in range: place at result[n-1+k]
    # k ranges such that i-k is in [0, n-1], i.e., k in [i-(n-1), i]
    k_lo = max(-(n-1), i-(n-1))
    k_hi = min(n-1, i)
    if k_lo <= k_hi:
        ks = np.arange(k_lo, k_hi+1)
        result[n-1+ks] += d * h[i-ks]
        # h[i+k]
    k_lo2 = max(-(n-1), -i)
    k_hi2 = min(n-1, n-1-i)
    if k_lo2 <= k_hi2:
        ks2 = np.arange(k_lo2, k_hi2+1)
        result[n-1+ks2] += d * h[i+ks2]
    
    k_lo3 = max(-(n-1), j-(n-1))
    k_hi3 = min(n-1, j)
    if k_lo3 <= k_hi3:
        ks3 = np.arange(k_lo3, k_hi3+1)
        result[n-1+ks3] -= d * h[j-ks3]
    
    k_lo4 = max(-(n-1), -j)
    k_hi4 = min(n-1, n-1-j)
    if k_lo4 <= k_hi4:
        ks4 = np.arange(k_lo4, k_hi4+1)
        result[n-1+ks4] -= d * h[j+ks4]
    
    return result

Now let me think about the swap search more carefully. The key question is: how to efficiently find good swaps?

Strategy:
1. Maintain the set of "1" positions and "0" positions
2. At each step, randomly pick a "1" position i and a "0" position j
3. Try the swap i→j (move the 1 from i to j)
4. If it improves, accept

This is a standard "cow path" or "first-improvement" local search.

For better performance, I can use a "simulated annealing" style: accept bad moves with some probability (exp(-Δ/T)).

Or I can use a "worst-lag focused" approach: instead of random swaps, focus on swaps that specifically reduce the maximum correlation lag.

Let me implement the simplest version first (first-improvement with random swaps) and see if it's effective.

One optimization: instead of recomputing the full correlate for each swap, I can use the delta formula. Let me implement it correctly this time.

The delta formula for swapping positions i and j:
Let a = h[i], b = h[j]. After swap: h[i] = b, h[j] = a.
Δc_k = (b-a)*[h[i-k] - h[j-k] + h[i+k] - h[j+k]]

where h[m] = 0 if m < 0 or m >= n.

In the output array of length 2n-1 (indexed 0 to 2n-2), index = n-1+k:
result[n-1+k] = (b-a) * [h(i-k) - h(j-k) + h(i+k) - h(j+k)]

Let me define a helper: for a position p, the vector "h(p-k) for all k" placed in the result array:
- h(p-k) is non-zero when 0 <= p-k < n, i.e., p-n+1 <= k <= p
- In result array: index = n-1+k for k in [p-n+1, p]
  → indices from n-1+p-n+1 = p to n-1+p

So the vector "h(p-k)" placed in the result array has entries at indices p, p+1, ..., p+n-1, with values h(p-0), h(p-1), ..., h(p-(n-1)) = h[p], h[p-1], ..., h[p-(n-1)].

This is the reverse of h[p-(n-1):p+1] placed starting at index p.

Similarly, "h(p+k)" is non-zero when 0 <= p+k < n, i.e., -p <= k <= n-1-p.
In result array: indices from n-1-p to n-1+n-1-p = 2n-2-p.
Values: h(p+0), h(p+1), ..., h(p+(n-1-p)) = h[p], h[p+1], ..., h[n-1].

So "h(p+k)" placed in result array: entries at indices n-1-p to 2n-2-p, with values h[p:n].

Let me define:
vec_minus(p) = array of length 2n-1, where:
  vec_minus(p)[p : p+n] = h[p], h[p-1], ..., h[p-(n-1)]  (i.e., h[p::-1][:n])
  rest = 0

vec_plus(p) = array of length 2n-1, where:
  vec_plus(p)[n-1-p : 2n-1-p] = h[p], h[p+1], ..., h[n-1]  (i.e., h[p:])
  rest = 0

Then:
Δc = (b-a) * (vec_minus(i) - vec_minus(j) + vec_plus(i) - vec_plus(j))

This is 4 array additions/subtractions of length 2n-1, plus construction of 4 sparse vectors. Each construction is O(n) (filling n entries). Total: O(n) per swap.

For n=960, this is very fast. Let me implement it:

```python
def compute_swap_delta_fast(h, i, j, n):
    a, b = h[i], h[j]
    d = b - a
    if abs(d) < 1e-15:
        return None
    result = np.zeros(2*n-1)
    # vec_minus(i): h[i-k] for k=0..n-1 at positions i, i+1, ..., i+n-1
    # h[i], h[i-1], ..., h[i-(n-1)]
    # Only valid entries: h[m] for m in [0, n-1]
    # h[i-k] valid when 0 <= i-k < n, i.e., i-n+1 <= k <= i
    # At position i+k in result:
    k_start = max(0, i-n+1)
    k_end = min(n-1, i)
    positions = np.arange(i+k_start, i+k_end+1)
    values = h[i-np.arange(k_start, k_end+1)]  # h[i-k_start], ... h[i-k_end]
    # Wait: I want h[i-k] for k = k_start, ..., k_end
    # That's h[i-k_start], h[i-k_start-1], ..., h[i-k_end]
    # = h[i-(k_start)], h[i-(k_start+1)], ..., h[i-(k_end)]
    # which is h[i-k_start : i-k_end-1 : -1] if k_end > k_start
    # Simpler: h[i - np.arange(k_start, k_end+1)]
    result[positions] += d * h[i - np.arange(k_start, k_end+1)]
    
    # vec_minus(j): similarly
    k_start = max(0, j-n+1)
    k_end = min(n-1, j)
    positions = np.arange(j+k_start, j+k_end+1)
    result[positions] -= d * h[j - np.arange(k_start, k_end+1)]
    
    # vec_plus(i): h[i+k] for k=0..n-1 at positions n-1, n, ..., 2n-2
    # h[i+k] valid when 0 <= i+k < n, i.e., -i <= k <= n-1-i
    k_start = max(0, -i)  # k >= 0 anyway since k starts at 0
    k_end = min(n-1, n-1-i)
    positions = np.arange(n-1+k_start, n-1+k_end+1)
    result[positions] += d * h[i+k_start : i+k_end+1]
    
    # vec_plus(j)
    k_start = max(0, -j)
    k_end = min(n-1, n-1-j)
    positions = np.arange(n-1+k_start, n-1+k_end+1)
    result[positions] -= d * h[j+k_start : j+k_end+1]
    
    return result
```
