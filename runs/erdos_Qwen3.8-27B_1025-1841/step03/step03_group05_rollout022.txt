from scipy.optimize import minimize
import time

def project_c(h, n):
    """Project h to [0,1]^n with sum = n/2 using bisection."""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgrad_one(h, shift):
    """Subgradient of the shift 'shift' component of the max."""
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_schedule(frac)
        lr = lr_schedule(frac)
        
        # Compute correlation
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)

        tot = np.sum(exps)
        if tot < 1e-300:

        else:

            # Use top-k for efficiency

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def fista_run(h, n_iter, T_start, T_end, lr_start, lr_end, top_k=100, check=30):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** f
        lr = lr_start + (lr_end - lr_start) * f
        
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        
        if tot < 1e-300:
            g = subgrad_one(z, int(np.argmax(corr)) - (n-1)) * (2.0/n)
        else:
            w = exps / tot
            idx = np.argsort(w)[::-1][:top_k]
            ws = w[idx].sum()
            if ws < 1e-300:
                g = np.zeros(n)
            else:
                g = np.zeros(n)
                for ii in idx:
                    wi = w[ii] / ws
                    if wi < 1e-13:
                        break
                    g += wi * subgrad_one(z, ii - (n-1))
                g *= (2.0 / n)
        
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
        
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def local_search(h, n_trials=5000, rng=None):

def downsample(h, nn):
    """Downsample h to nn points, preserving the integral."""
    n = len(h)
    if nn >= n:
        return h[:nn].copy() if nn > n else h.copy()
    # Use block averaging
    block = n // nn
    h_blocks = h[:block*nn].reshape(nn, block).mean(axis=1)
    return project_c(h_blocks, nn)

def upsample(h, nn):
    """Upsample h to nn points."""
    n = len(h)
    if nn == n:
        return h.copy()
    ox = np.linspace(0, 2, n)
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def m_sequence(m):
    """Generate an m-sequence of length 2^m - 1."""
    L = (1 << m) - 1
    # Simple LFSR with primitive polynomial
    # For m=3: x^3 + x + 1
    # For m=4: x^4 + x + 1
    # For m=5: x^5 + x^2 + 1
    polys = {3: [0,1], 4: [0,1], 5: [1,2], 6: [0,3], 7: [0,3], 8: [0,4]}
    if m not in polys:
        m = 5
    taps = polys[m]
    
    state = 1  # Start with single 1
    seq = [1]
    for _ in range(L - 1):
        # Compute next bit
        feedback = 0
        for t in taps:
            feedback ^= (state >> t) & 1
        state = ((state << 1) | feedback) & ((1 << m) - 1)
        seq.append(feedback)
    
    return np.array(seq, dtype=float)

def tm_seq(n):
    """Thue-Morse sequence of length n."""
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def compute_grad(h, top_k=100, T=10.0):
    """Compute smoothed gradient using top-k shifts."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return subgrad_one(h, int(np.argmax(corr)) - (n-1)) * (2.0/n)
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    ws = w[idx].sum()
    if ws < 1e-300:
        return np.zeros(n)
    g = np.zeros(n)
    for ii in idx:
        wi = w[ii] / ws
        if wi < 1e-13:
            break
        g += wi * subgrad_one(h, int(ii) - (n-1))
    return g * (2.0 / n)

def fista_run(h, n_iter, T0, T1, lr0, lr1, top_k=100, check=30):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr0 + (lr1 - lr0) * f
        g = compute_grad(z, top_k, T)
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def local_search_binary(h, n_trials=50000, rng=None):
    """Greedy local search on binary sequence."""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h)
    h = (h > 0.5).astype(float)
    h = project_c(h, n)
    bc = compute_c5(h)
    
    for t in range(n_trials):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_k = np.argmax(corr)
        s = max_k - (n - 1)
        sa = abs(s)
        if sa >= n:
            break
        
        # Find bad positions: h[i]=1 and h[i+s]=0 (for s>0)
        if s > 0:
            bad = (h[:n-sa] > 0.5) & (h[sa:] < 0.5)
            bad_idx = np.where(bad)[0]
        elif s < 0:
            bad = (h[:n-sa] > 0.5) & (h[sa:] < 0.5)
            bad_idx = np.where(bad)[0]
        else:
            break
        
        if len(bad_idx) == 0:
            break
        
        # Pick a bad position and try to fix it
        i = bad_idx[rng.randint(len(bad_idx))]
        
        # Find zeros to swap with
        zeros = np.where(h < 0.5)[0]
        if len(zeros) == 0:
            break
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Try swap
        h[i], h[j] = h[j], h[i]
        c = compute_c5(h)
        if c < bc - 1e-12:
            bc = c
        else:
            h[i], h[j] = h[j], h[i]
            # Try another random zero
            if rng.random() < 0.5 and len(zeros) > 1:
                j2 = zeros[rng.randint(len(zeros))]
                if j2 != i:
                    h[i], h[j2] = h[j2], h[i]
                    c = compute_c5(h)
                    if c < bc - 1e-12:
                        bc = c
                    else:
                        h[i], h[j2] = h[j2], h[i]
    
    return h, bc

def downsample(h, nn):
    n = len(h)
    if nn >= n:
        return h.copy()
    block = n // nn
    h_r = h[:block * nn].reshape(nn, block).mean(axis=1)
    return project_c(h_r, nn)

def upsample(h, nn):
    n = len(h)
    if nn == n:
        return h.copy()
    ox = np.linspace(0, 2, n)
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def slsqp_polish(h, T=3.0, max_iter=400):
    n = len(h)
    _SLS['T'] = T
    _SLS['n'] = n
    try:
        r = minimize(_sobj, h.copy(), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': max_iter, 'ftol': 1e-12})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].copy()
