    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff))+1:] if int(abs(diff)) > 0 else [np.argmin(h)]

The correlation r_k = sum_n h[n+k] * (1-h[n]) for lag k.

For a "good" pattern, we want r_k to be small for ALL k. The sum of all r_k (over all lags from -(n-1) to n-1) is:
sum_k r_k = sum_k sum_n h[n+k]*(1-h[n]) (with appropriate bounds)

For the "full" correlation (linear), the sum over positive lags is:
sum_{k=1}^{n-1} sum_{n=0}^{n-1-k} h[n+k]*(1-h[n])
= number of pairs (i,j) with i<j, h[i]=0, h[j]=1

This is independent of the arrangement! It only depends on the number of 1s and 0s. For n/2 ones and n/2 zeros:
Total pairs = C(n/2, 1) * C(n/2, 1) = (n/2)² = n²/4.

Wait no, it's the number of (i,j) pairs with i<j where h[i]=0 and h[j]=1. This depends on the arrangement.

If all 1s are at the beginning: h = [1,1,...,1,0,0,...,0]. Then pairs (i,j) with h[i]=0, h[j]=1 and i<j: none! (All 1s are before all 0s.)
So sum of positive-lag correlations = 0.

If all 1s are at the end: h = [0,0,...,0,1,1,...,1]. Then all pairs (i,j) with i<j, h[i]=0, h[j]=1 count. Total = (n/2)² = n²/4.

For a "mixed" pattern: somewhere in between.

The average positive-lag correlation (averaged over all n-1 positive lags) is:
(n²/4) / (n-1) ≈ n/4 for a "random" arrangement.

But we care about the MAX, not the average. A "block" pattern (all 1s together) minimizes the sum of positive lags but maximizes individual lags near the boundary.

This is a classic tradeoff! The optimal pattern balances between:
- Spreading 1s and 0s to reduce the max individual lag correlation
- Keeping some clustering to avoid intermediate lags being too large

This is exactly the type of problem where a "blue noise" or "low-discrepancy" pattern works well.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -700, 700)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
    return make_feasible(res.x, n)
