        h_res = run_slsqp(h_try, T_r, None, maxiter=250)
        c5_r = compute_c5(h_res)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_res
            # Quick CD follow-up
            if rem() > 30:
                h_cd, c5_cd = coord_descent(best_h, len(best_h),
                                            [0.005, 0.01, 0.02], max_passes=1,
                                            time_limit=rem() - 20, start_t=start_time)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd

    # Phase 8: Final ultra-fine CD + SLSQP polish
    if rem() > 30:
        h_cd, c5_cd = coord_descent(best_h, len(best_h),
                                    [0.0005, 0.001, 0.002, 0.003, 0.005],
                                    max_passes=2,
                                    time_limit=rem() - 20, start_t=start_time)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
        if rem() > 25:
            h_res = run_slsqp(best_h, 1000.0, None, maxiter=200)
            c5_r = compute_c5(h_res)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_res

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
