def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Convert to binary, ensuring exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s, prefer those with lower h value
        one_idx = np.where(h_bin == 1)[0]
        # Sort by h value ascending (most "uncertain" first)
        order = np.argsort(h[one_idx])
        for k in range(n_ones - target):
            h_bin[one_idx[order[k]]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for k in range(target - n_ones):
            h_bin[zero_idx[order[k]]] = 1.0
    return h_bin

def swap_delta(h, a, b, n):

    bk2 = b - ks

def lbgfb_smooth(h_start, n, T, mu, max_iter=200, deadline=None):
    dx = 2.0 / n

    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    opts = {'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12}
    if deadline and time.time() > deadline - 5:
        return h_start
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds, options=opts)
    return make_feasible(res.x, n)

def try_periodic(n, period, shift, rng, n_ones_target):
    """Create a periodic pattern with given period and shift."""
    h = np.zeros(n)
    for i in range(n):
        pos = (i + shift) % period
        # First half of period is 1
        if pos < period // 2:
            h[i] = 1.0
    # Fix count
    n_ones = int(np.sum(h))
    if n_ones > n_ones_target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        for k in range(n_ones - n_ones_target):
            h[idx[k]] = 0.0
    elif n_ones < n_ones_target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        for k in range(n_ones_target - n_ones):
            h[idx[k]] = 1.0
    return h

Actually, a better approach: instead of computing the full delta array, I can compute just the max of (corr + delta) more efficiently.

Note that delta[k] = h[a+k] + h[a-k] - h[b+k] - h[b-k].

For most k, this is 0 (if h is binary, each term is 0 or 1, so delta is in {-2, -1, 0, 1, 2}).

Actually, for binary h, the non-zero entries of delta are limited:
- h[a+k] is non-zero only when a+k is in [0, n-1], i.e., k in [-a, n-1-a]
- h[a-k] is non-zero only when a-k is in [0, n-1], i.e., k in [a-(n-1), a]
- Similarly for b

So the support of delta is the union of these ranges, which is [-(n-1), n-1] (the full range). But the values are small integers.

Let me think of a faster way. For binary h:
- delta[k] = h[a+k] + h[a-k] - h[b+k] - h[b-k]
- Each term is 0 or 1 (or 0 if out of bounds)
- So delta[k] ∈ {-2, -1, 0, 1, 2}

The key insight: I only need to check positions where delta[k] != 0, because at those positions corr[k] is unchanged.

The positions where delta[k] != 0 are:
- k where h[a+k] = 1 (i.e., a+k is a 1-position): k = pos - a for pos in ones
- k where h[a-k] = 1: k = a - pos for pos in ones
- k where h[b+k] = 1: k = pos - b for pos in ones
- k where h[b-k] = 1: k = b - pos for pos in ones

So the number of non-zero entries of delta is at most 4 * (n/2) = 2n. For n=2048, that's 4096 positions.

But checking max over only these positions + the positions where delta=0 but corr is at its max... this gets complicated.

```python
def swap_delta_fast(h, a, b, n, buf):
    """Compute delta in-place into pre-allocated buffer."""
    buf[:] = 0.0
    # h[a+k] for k in [-a, n-1-a]
    k_lo = max(-(n-1), -a)
    k_hi = min(n-1, n-1-a)
    if k_lo <= k_hi:
        buf[n-1+k_lo:n-1+k_hi+1] += h[a+k_lo:a+k_hi+1]
    # h[a-k] for k in [a-(n-1), a]
    k_lo2 = max(-(n-1), a-n+1)
    k_hi2 = min(n-1, a)
    if k_lo2 <= k_hi2:
        buf[n-1+k_lo2:n-1+k_hi2+1] += h[a-k_hi2:a-k_lo2+1][::-1]
    # -h[b+k] for k in [-b, n-1-b]
    k_lo3 = max(-(n-1), -b)
    k_hi3 = min(n-1, n-1-b)
    if k_lo3 <= k_hi3:
        buf[n-1+k_lo3:n-1+k_hi3+1] -= h[b+k_lo3:b+k_hi3+1]
    # -h[b-k] for k in [b-(n-1), b]
    k_lo4 = max(-(n-1), b-n+1)
    k_hi4 = min(n-1, b)
    if k_lo4 <= k_hi4:
        buf[n-1+k_lo4:n-1+k_hi4+1] -= h[b-k_hi4:b-k_lo4+1][::-1]
    return buf
```
