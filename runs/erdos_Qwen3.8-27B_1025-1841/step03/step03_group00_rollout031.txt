```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:

        if diff > 0:

        else:

        cnt = np.sum(mask)
        if cnt == 0:

        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

    """Compute approximate subgradient of max_k c_k * dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Find top m lags
    idx = np.argsort(corr)[::-1][:top_m]
    # For each active lag, compute gradient
    grad = np.zeros(n)
    for k_idx in idx:
        # k_idx is the index in the full correlation, corresponding to lag k = k_idx - (n-1)
        k = k_idx - (n - 1)
        # c_k = sum_i h[i+k]*(1-h[i]) for valid i
        # grad[j] += (1-h[j-k]) if j-k valid, minus h[j+k] if j+k valid
        # But we need to be careful about the "full" correlation indexing
        # In np.correlate(h, 1-h, 'full'), index k_idx corresponds to:
        # c[k_idx] = sum_i h[i+k]*(1-h[i-k_idx+n-1])... 
        # Actually, let me just compute it directly.
        # The full correlation of a and v has length len(a)+len(v)-1
        # c[m] = sum_i a[i+m] * v[i] for m = -(len(v)-1), ..., len(a)-1
        # So c[k_idx] corresponds to m = k_idx - (n-1)
        m = k_idx - (n - 1)
        # c[m] = sum_i h[i+m] * (1-h[i]) for i where 0<=i<n and 0<=i+m<n
        # grad wrt h[j]: 
        #   if j = i+m for some valid i: contributes (1-h[i]) = (1-h[j-m])
        #   if j = i for some valid i: contributes -h[i+m] = -h[j+m]
        # So grad[j] = (1-h[j-m]) * [0<=j-m<n] - h[j+m] * [0<=j+m<n]
        # Wait, that's the gradient of c[m] wrt h[j]
        # Let me verify: c[m] = sum_i h[i+m]*(1-h[i])
        # dc[m]/dh[j] = sum_i [delta(j,i+m)*(1-h[i]) + h[i+m]*(-delta(j,i))]
        # = (1-h[j-m]) if 0<=j-m<n else 0, minus h[j+m] if 0<=j+m<n else 0
        lower = max(0, -m)
        upper = min(n, n - m)
        # For j in range(lower, upper):
        # grad[j] += (1-h[j-m]) - h[j+m] (with appropriate bounds)
        # But h[j-m] is only defined if 0 <= j-m < n, and h[j+m] if 0 <= j+m < n
        # Since j is in [lower, upper), j-m is in [lower-m, upper-m) = [max(0,-m)-m, min(n,n-m)-m)
        # and j+m is in [lower+m, upper+m)
        
        # Let me just compute it element-wise
        j = np.arange(n)
        jm = j - m  # j - m
        jp = j + m  # j + m
        valid_m = (jm >= 0) & (jm < n)
        valid_p = (jp >= 0) & (jp < n)
        g = np.zeros(n)
        g[valid_m] += (1.0 - h[jm[valid_m]])
        g[valid_p] -= h[jp[valid_p]]
        grad += g
    return grad * dx, mc * dx, corr

def subgradient_descent(h_start, n, lr=0.01, n_iter=5000, top_m=5):
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        grad, mc, corr = max_lag_subgradient(h, n, top_m)
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        # Decrease learning rate
        if it % 1000 == 0 and it > 0:
            lr *= 0.5
    return best_h, best_c5

def simulated_annealing_binary(h_bin, n, rng, n_iterations=500000, T_start=0.1, T_end=0.001):
    h = h_bin.copy()
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    n_target = n // 2
    if n_ones > n_target:
        one_idx = np.where(h == 1)[0]
        rng.shuffle(one_idx)
        for k in range(n_ones - n_target):
            h[one_idx[k]] = 0
    elif n_ones < n_target:
        zero_idx = np.where(h == 0)[0]
        rng.shuffle(zero_idx)
        for k in range(n_target - n_ones):
            h[zero_idx[k]] = 1
    
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_iterations):
        T = T_start * (T_end / T_start) ** (it / n_iterations)
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0
        h[j] = 1
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / T):
            current_c5 = new_c5
            # Update ones and zeros
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def generate_m_sequence(m, target_n):
    """Generate an m-sequence of length 2^m - 1 and extend to target_n"""
    L = (1 << m) - 1
    # Generate m-sequence using LFSR
    # Use a known primitive polynomial for small m
    # For m=5: polynomial x^5 + x^2 + 1 (taps 5,2)
    # For m=6: x^6 + x^5 + x^2 + x + 1 (taps 6,5,2,1)
    # For m=7: x^7 + x^4 + x^3 + x^2 + 1 (taps 7,4,3,2)
    # For m=8: x^8 + x^6 + x^5 + x^4 + 1 (taps 8,6,5,4)
    taps_dict = {
        5: [4, 1],  # x^5 + x^2 + 1
        6: [5, 4, 1, 0],  # x^6 + x^5 + x^2 + x + 1
        7: [6, 3, 2, 1],  # x^7 + x^4 + x^3 + x^2 + 1
        8: [7, 5, 4, 3],  # x^8 + x^6 + x^5 + x^4 + 1
        9: [8, 4, 3, 2],  # x^9 + x^5 + x^4 + x^3 + 1
        10: [9, 7, 6, 3, 0],  # x^10 + x^8 + x^7 + x^4 + 1
    }
    taps = taps_dict.get(m, [m-1, 0])
    # Initialize LFSR
    state = [1] * m
    seq = []
    for _ in range(L):
        seq.append(state[0])
        # Compute new bit
        feedback = 0
        for t in taps:
            feedback ^= state[t]
        # Shift
        state = [feedback] + state[:-1]
    
    # Convert to {0,1} with appropriate balance
    # m-sequence has (L+1)/2 ones and (L-1)/2 zeros (or vice versa)
    # Extend to target_n by tiling
    if L >= target_n:
        h = np.array(seq[:target_n], dtype=float)
    else:
        reps = target_n // L + 1
        h = np.tile(np.array(seq, dtype=float), reps)[:target_n]
    return h

def generate_thue_morse(target_n):
    """Generate Thue-Morse sequence truncated/extended to target_n"""
    # Thue-Morse: t(n) = popcount(n) mod 2
    h = np.array([bin(i).count('1') % 2 for i in range(target_n)], dtype=float)
    return h
```

Let me also think about whether I should use a different approach for the binary optimization. Instead of full correlation recomputation, I can compute the change in correlation for a swap in O(n) time. This would speed up the SA significantly.

For a swap of positions i (1→0) and j (0→1):
- The change in c[k] for each lag k can be computed in O(1) per lag, giving O(n) total.
- But we need the MAX of all c[k], which requires knowing all c[k] values.

```python
    # Term 3: (1-h[j-m]) for 0 <= j-m < n, i.e., j-n < m <= j
    t3 = np.zeros(2*n-1)
    valid_t3 = (m_vals >= j - (n-1)) & (m_vals <= j)
    t3[valid_t3] = 1.0 - h[j - m_vals[valid_t3]]
    
    # Term 4: -h[j+m] for 0 <= j+m < n, i.e., 0 <= m < n-j
    t4 = np.zeros(2*n-1)
    valid_t4 = (m_vals >= 0) & (m_vals < n - j)
    t4[valid_t4] = -h[j + m_vals[valid_t4]]
    
    return t1 + t2 + t3 + t4
```

This is O(n) per swap, which is much faster than O(n^2) for the full correlation. For n=640, this is about 640 operations per swap vs 800K for the full correlation. So we get about 1000x speedup.

```python
def sa_binary_fast(h_start, n, rng, n_iterations=2000000, T_start=0.05, T_end=0.0001):
    h = h_start.copy()
    n_target = n // 2
    # Fix count
    n_ones = int(np.sum(h))
    if n_ones > n_target:
        one_idx = np.where(h > 0.5)[0]
        rng.shuffle(one_idx)
        for k in range(n_ones - n_target):
            h[one_idx[k]] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(h <= 0.5)[0]
        rng.shuffle(zero_idx)
        for k in range(n_target - n_ones):
            h[zero_idx[k]] = 1.0
    
    c = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(c)
    current_c5 = current_max * (2.0 / n)
    best_c5 = current_c5
    best_h = h.copy()
    best_c = c.copy()
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for it in range(n_iterations):
        T = T_start * (T_end / T_start) ** (it / n_iterations)
        
        # Random swap
        idx_i = rng.randint(len(ones))
        idx_j = rng.randint(len(zeros))
        i = ones[idx_i]
        j = zeros[idx_j]
        
        if i == j:
            continue
            
        # Compute delta
        delta = compute_swap_delta(h, i, j)
        c_new = c + delta
        new_max = np.max(c_new)
        new_c5 = new_max * (2.0 / n)
        
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or rng.random() < np.exp(-delta_c5 / T):
            # Accept
            h[i] = 0.0
            h[j] = 1.0
            c = c_new
            current_max = new_max
            current_c5 = new_c5
            
            # Update ones and zeros
            ones = np.delete(ones, idx_i)
            zeros = np.delete(zeros, idx_j)
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                best_c = c.copy()
        else:
            pass  # reject, h and c unchanged
    
    return best_h, best_c5
```
