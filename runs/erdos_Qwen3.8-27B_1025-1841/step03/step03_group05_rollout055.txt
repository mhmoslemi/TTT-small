    bc = compute_c5(bh)
    for t in range(n_trials):
        blen = rng.randint(3, 13)
        if blen >= n - 1:
            blen = max(2, n // 3)
        start = rng.randint(0, max(1, n - blen))
        orig = bh[start:start + blen].copy()
        bsum = orig.sum()
        scale = 0.03 + 0.12 * rng.rand()
        pert = orig + rng.randn(blen) * scale
        pert = np.clip(pert, 0.0, 1.0)
        diff = bsum - pert.sum()
        if abs(diff) > 1e-10:
            mask = (pert > 0.02) & (pert < 0.98)
            if mask.any():
                pert[mask] += diff / mask.sum()
                pert = np.clip(pert, 0.0, 1.0)
            else:
                continue
        bh[start:start + blen] = pert
        c = compute_c5(bh)
        if c < bc - 1e-14:
            bc = c
        else:
            bh[start:start + blen] = orig
    return bh, bc

_sm_state = {'T': 5.0, 'n': 0}

def _sobj(h):
    T = _sm_state['T']
    n = _sm_state['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def _scon(h):
    return np.sum(h) - _sm_state['n'] / 2.0

def _sjac(h):
    return np.ones(_sm_state['n'])

def _sgrad(h):
    T = _sm_state['T']
    n = _sm_state['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = exps.sum()
    if tot < 1e-300:
        idx = np.array([int(np.argmax(corr))])
        w = np.ones(1)
    else:
        w = exps / tot
        idx = np.argsort(w)[::-1][:60]
        w = w[idx]
    g = np.zeros(n)
    for i in range(len(idx)):
        wi = w[i]
        if wi < 1e-13:
            break
        k = int(idx[i]) - (n - 1)
        g += wi * _lag_grad_vec(h, k, n)
    return g * dx

def slsqp_polish(h, T=3.0, maxiter=200):
    n = len(h)
    _sm_state['T'] = T
    _sm_state['n'] = n
    try:
        r = minimize(_sobj, h, jac=_sgrad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].astype(float)

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def elapsed(t0):
    return time.time() - t0

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    bc = float('inf')
    bh = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = project_c(rng.rand(960), 960)

    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"Init: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1: FISTA exploration
    if budget_s - elapsed(t0) > 250:
        h1, c1 = fista_core(bh, 5000, 50.0, 1.0, 0.14, 50, 30)
        if c1 < bc:
            bc, bh = c1, h1.copy()
        print(f"F1: C5={bc:.6f} t={elapsed(t0):.0f}", flush=True)

    # Phase 2: FISTA exploitation
    if budget_s - elapsed(t0) > 180:
        h2, c2 = fista_core(bh, 5000, 3.0, 0.02, 0.04, 30, 25)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        print(f"F2: C5={bc:.6f} t={elapsed(t0):.0f}", flush=True)

    # Phase 3: Block local search
    if budget_s - elapsed(t0) > 60:
        h3, c3 = block_search(bh, 5000, seed)
        if c3 < bc:
            bc, bh = c3, h3.copy()
        print(f"BL1: C5={bc:.6f} t={elapsed(t0):.0f}", flush=True)

    # Phase 4: SLSQP polish with analytical gradient
    np_ = len(bh)
    if np_ <= 640 and budget_s - elapsed(t0) > 100:
        for TV in [20.0, 8.0, 3.0, 1.5]:
            if budget_s - elapsed(t0) < 25:
                break
            bx = slsqp_polish(bh, TV, 250)
            cx = compute_c5(bx)
            if cx < bc:
                bc, bh = cx, bx
            h3b, c3b = block_search(bh, 1500, seed + int(TV))
            if c3b < bc:
                bc, bh = c3b, h3b.copy()
        print(f"SL: C5={bc:.6f} t={elapsed(t0):.0f}", flush=True)

    # Phase 5: Resolution exploration
    tried_ns = set([len(bh)])
    for nn in [n0 - 128, n0 + 64, n0 + 128, n0 * 2]:
        if budget_s - elapsed(t0) < 120:
            break
        if nn < 64 or nn > 2560 or nn in tried_ns:
            continue
        tried_ns.add(nn)
        hu = upsample(bh, nn)
        cu = compute_c5(hu)
        if budget_s - elapsed(t0) > 80:
            hg, cg = fista_core(hu, 3000, 20.0, 0.1, 0.1, 40, 30)
            if cg < cu:
                hu, cu = hg, cg
        if budget_s - elapsed(t0) > 40:
            hn, cn = fista_core(hu, 2000, 2.0, 0.02, 0.03, 25, 25)
            if cn < cu:
                hu, cu = hn, cn
        if budget_s - elapsed(t0) > 20:
            hb, cb = block_search(hu, 1500, seed + nn)
            if cb < cu:
                hu, cu = hb, cb
        if cu < bc:
            bc, bh = cu, hu.copy()
            print(f"  R{nn}: C5={bc:.6f}", flush=True)

    # Phase 6: Diverse starts and re-optimize
    if budget_s - elapsed(t0) > 80:
        np_ = len(bh)
        starts = [tm_seq(np_)]
        for p in [7, 11, 13]:
            pat = np.zeros(p)
            pat[:p // 2] = 1.0
            starts.append(project_c(np.tile(pat, np_ // p + 1)[:np_], np_))
        al = np.zeros(np_)
        al[::2] = 1.0
        starts.append(project_c(al, np_))
        for hs in starts:
            if budget_s - elapsed(t0) < 40:
                break
            hr, cr = fista_core(hs, 2000, 30.0, 0.3, 0.1, 40, 40)
            if budget_s - elapsed(t0) > 15:
                hr2, cr2 = block_search(hr, 1000, seed + 500)
                if cr2 < cr:
                    hr, cr = hr2, cr2
            if cr < bc:
                bc, bh = cr, hr.copy()
                print(f"  DS: C5={bc:.6f}", flush=True)

    # Phase 7: Final fine FISTA
    if budget_s - elapsed(t0) > 40:
        hf, cf = fista_core(bh, 3000, 5.0, 0.01, 0.015, 20, 20)
        if cf < bc:
            bc, bh = cf, hf.copy()

    # Phase 8: Final block search
    if budget_s - elapsed(t0) > 20:
        h8, c8 = block_search(bh, 3000, seed + 999)
        if c8 < bc:
            bc, bh = c8, h8.copy()

    bh = project_c(bh, len(bh))
    bc = compute_c5(bh)
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={elapsed(t0):.0f}", flush=True)
    return bh, bc, len(bh)
