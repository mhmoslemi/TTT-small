def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary_feasible(h, n):
    """Convert h to a binary vector with exactly n/2 ones."""
    # Take the top n/2 values as 1
    threshold = np.sort(h)[-n//2]
    h_bin = (h >= threshold).astype(float)
    # If we have more than n/2 ones, fix it
    if np.sum(h_bin) > n//2:
        # Set the smallest ones to 0
        ones_idx = np.where(h_bin == 1)[0]
        # Sort by h value, set the smallest to 0
        order = np.argsort(h[ones_idx])
        excess = int(np.sum(h_bin)) - n//2
        h_bin[ones_idx[order[:excess]]] = 0
    elif np.sum(h_bin) < n//2:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        deficit = n//2 - int(np.sum(h_bin))
        h_bin[zeros_idx[order[:deficit]]] = 1
    return h_bin

    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=float)

    """Convert to binary with exactly n/2 ones, choosing top n/2 values."""
    h = np.asarray(h, dtype=float)
    if n != len(h):
        # Interpolate to new length
        old_x = np.linspace(0, 2, len(h), endpoint=False)
        new_x = np.linspace(0, 2, n, endpoint=False)
        h = np.interp(new_x, old_x, h)
    # Sort and threshold
    sorted_h = np.sort(h)[::-1]
    threshold = sorted_h[n//2]
    h_bin = (h >= threshold).astype(float)
    # Handle ties
    n_ones = int(np.sum(h_bin))
    if n_ones > n//2:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones_idx])
        for idx in order[:n_ones - n//2]:
            h_bin[ones_idx[idx]] = 0
    elif n_ones < n//2:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zeros_idx])
        for idx in order[:n//2 - n_ones]:
            h_bin[zeros_idx[idx]] = 1
    return h_bin
