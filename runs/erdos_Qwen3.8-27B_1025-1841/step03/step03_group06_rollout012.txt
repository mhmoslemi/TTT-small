def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        h = np.clip(h, 0.0, 1.0)

    """Snap to binary while maintaining exactly n/2 ones."""
    thresholded = (h >= 0.5).astype(float)

        for i in range(target - n_ones):

import numpy as np
from scipy.optimize import minimize
import time

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _thue_morse(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

def _snap_binary(h, n):
    thresholded = (h >= 0.5).astype(float)
    n_ones = int(thresholded.sum())
    target = n // 2
    if n_ones == target:
        return thresholded
    if n_ones > target:
        one_pos = np.where(thresholded == 1)[0]
        order = np.argsort(h[one_pos])
        for i in range(n_ones - target):
            thresholded[one_pos[order[i]]] = 0
    else:
        zero_pos = np.where(thresholded == 0)[0]
        order = np.argsort(-h[zero_pos])
        for i in range(target - n_ones):
            thresholded[zero_pos[order[i]]] = 1
    return thresholded

def _upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return np.interp(new_x, old_x, h)

def _binary_sa(h, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c = float(np.max(corr) * dx)
    best_c = cur_c
    best_h = h.copy()
    start_t = time.time()
    T_start = 0.04
    T_end = 0.0005
    total_iters = 2000000
    no_improve = 0
    MAX_NO_IMP = 40000
    for it in range(total_iters):
        if time.time() - start_t > time_limit:
            break
        frac = it / total_iters
        T = T_start * (T_end / T_start) ** frac
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        delta = c_new - cur_c
        if delta < -1e-15 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            no_improve += 1
        if no_improve > MAX_NO_IMP:
            for _ in range(30):
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                aa = ones[rng.randint(len(ones))]
                bb = zeros[rng.randint(len(zeros))]
                h[aa], h[bb] = h[bb], h[aa]
            corr = np.correlate(h, 1.0 - h, mode='full')
            cur_c = float(np.max(corr) * dx)
            no_improve = 0
    return best_h, best_c

def _hill_climb_binary(h, n, time_limit, seed):
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_c = float(np.max(corr) * dx)
    best_c = cur_c
    best_h = h.copy()
    start_t = time.time()
    while time.time() - start_t < time_limit:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for a in ones:
            if time.time() - start_t > time_limit:
                break
            for b in zeros:
                if time.time() - start_t > time_limit:
                    break
                h[a], h[b] = h[b], h[a]
                c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                if c_new < cur_c - 1e-15:
                    cur_c = c_new
                    if c_new < best_c:
                        best_c = c_new
                        best_h = h.copy()
                    improved = True
                    zeros = np.where(h == 0)[0]
                    break
                else:
                    h[a], h[b] = h[b], h[a]
        if not improved:
            break
    return best_h, best_c

def _softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _sum_eq(x):
    return np.sum(x) - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def _continuous_refine(h, n, time_limit, seed):
    dx = 2.0 / n
    h0 = make_feasible(h.copy(), n)
    best_c = _compute_c5(h0)
    best_x = h0.copy()
    start_t = time.time()
    for T in [200.0, 400.0, 800.0]:
        if time.time() - start_t > time_limit:
            break
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_jac}]
        try:
            maxiter = max(50, int((time_limit - (time.time() - start_t)) / 0.5))
            res = minimize(_softmax_obj, best_x, args=(n, T), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': min(maxiter, 300), 'ftol': 1e-14})
            h_res = _make_feasible(res.x, n)
            c5_res = _compute_c5(h_res)
            if c5_res < best_c:
                best_c = c5_res
                best_x = h_res
        except Exception:
            pass
    return best_x, best_c
