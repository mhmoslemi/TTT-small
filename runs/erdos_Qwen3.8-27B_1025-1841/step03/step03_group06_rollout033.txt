import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_project(h, n):
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin

def _slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_optimize(h, n, T=100.0, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _slsqp_obj(x, n, T), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, n)
    except Exception:
        return h

def apply_swap_binary(corr, h, a, b, n):
    off = n - 1
    new_corr = corr.copy()
    ks = np.arange(max(0, off - a), min(2 * n - 1, off + a + 1))
    idxs = a + ks - off
    v = (idxs >= 0) & (idxs < n)
    new_corr[ks[v]] -= (1.0 - h[idxs[v]])
    i_arr = np.arange(n)
    i_arr = i_arr[i_arr != a]
    ks2 = a + off - i_arr
    v2 = (ks2 >= 0) & (ks2 <= 2 * n - 2)
    new_corr[ks2[v2]] += h[i_arr[v2]]
    ks = np.arange(max(0, off - b), min(2 * n - 1, off + b + 1))
    idxs = b + ks - off
    v = (idxs >= 0) & (idxs < n)
    new_corr[ks[v]] += (1.0 - h[idxs[v]])
    i_arr = np.arange(n)
    i_arr = i_arr[i_arr != b]
    ks3 = b + off - i_arr
    v3 = (ks3 >= 0) & (ks3 <= 2 * n - 2)
    new_corr[ks3[v3]] -= h[i_arr[v3]]
    return new_corr

def binary_local_search(h, n, time_budget=600, seed=42):
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    dx = 2.0 / n
    offset = n - 1
    h = h.copy()
    start_time = time.time()
    stagnation = 0

    while time.time() - start_time < time_budget:
        improved = False
        top_indices = np.argsort(corr)[-8:]
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        n_ones, n_zeros = len(ones), len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        n_tries = min(3000, n_ones * n_zeros)
        a_idx = rng.randint(0, n_ones, size=n_tries)
        b_idx = rng.randint(0, n_zeros, size=n_tries)
        best_new_max = current_max
        best_a, best_b = -1, -1
        best_new_corr = None

        for t in range(n_tries):
            a = int(ones[a_idx[t]])
            b = int(zeros[b_idx[t]])
            promising = True
            for k in top_indices:
                k = int(k)
                delta = 0.0
                idx = a + k - offset
                if 0 <= idx < n:
                    delta -= (1.0 - h[idx])
                ii = a - k + offset
                if 0 <= ii < n and ii != a:
                    delta += h[ii]
                idx = b + k - offset
                if 0 <= idx < n:
                    delta += (1.0 - h[idx])
                ii = b - k + offset
                if 0 <= ii < n and ii != b:
                    delta -= h[ii]
                if delta > 0:
                    promising = False
                    break
            if not promising:
                continue
            new_corr = apply_swap_binary(corr, h, a, b, n)
            nm = float(np.max(new_corr))
            if nm < best_new_max - 1e-12:
                best_new_max = nm
                best_a, best_b = a, b
                best_new_corr = new_corr

        if best_a >= 0:
            h[best_a] = 0.0
            h[best_b] = 1.0
            corr = best_new_corr
            current_max = best_new_max
            improved = True
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 8:
                break

    return h, current_max * dx

def coordinate_descent(h, n, time_budget=60, seed=42):
    rng = np.random.RandomState(seed)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    dx = 2.0 / n
    h = h.copy()
    start_time = time.time()
    n_rounds = 0

    while time.time() - start_time < time_budget and n_rounds < 50:
        n_rounds += 1
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        top_k = np.argsort(corr)[-5:]

        for j in order:
            if time.time() - start_time > time_budget - 5:
                break
            old_val = h[j]
            if old_val < 0.01 or old_val > 0.99:
                continue
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old_val - 0.1), max(0.0, old_val - 0.02),
                     min(1.0, old_val + 0.1), min(1.0, old_val + 0.02)]
            best_c = current_max
            best_v = old_val
            for cv in cands:
                if abs(cv - old_val) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                tc = np.correlate(trial, 1.0 - trial, mode='full')
                tm = float(np.max(tc))
                if tm < best_c - 1e-12:
                    best_c = tm
                    best_v = cv
            if best_v != old_val:
                h[j] = best_v
                h = make_feasible(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                current_max = float(np.max(corr))
                improved = True

        if not improved:
            break

    return h, current_max * dx
