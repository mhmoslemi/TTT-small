    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argsort(-h)
        else:
            idx = np.argsort(h)
        cnt = min(n, int(np.ceil(abs(diff))))
        step = diff / cnt
        h[idx[:cnt]] += step
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        h[mask] += diff / np.sum(mask)
    return h

            j = rng.choice(j_candidates)
            if i == j:
                continue

            n_tried += 1
            if c < c_best - 1e-14:

                improved = True
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return h, c_best

def continuous_refine(h, n, time_limit=None, start_time=None):
    """Refine a binary solution allowing small continuous perturbations"""
    h = h.copy()
    c_best = compute_c5(h)
    # Try perturbing each coordinate slightly
    for rnd in range(5):
        if time_limit and start_time and time.time() - start_time > time_limit - 30:
            break
        improved = False
        for i in range(n):
            old = h[i]
            for delta in [0.1, -0.1, 0.05, -0.05, 0.2, -0.2]:
                cv = old + delta
                if cv < 0 or cv > 1:
                    continue
                h[i] = cv
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < c_best - 1e-10:
                    c_best = c
                    improved = True
                    break
                else:
                    h[i] = old
                    h = make_feasible(h, n)
            if not improved:
                # didn't improve this coordinate, move on
                pass
    return h, c_best

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(-h)
        else:
            idx = np.argsort(h)
        cnt = min(n, int(np.ceil(abs(diff))))
        h[idx[:cnt]] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return h

def tody_init(n, alpha):
    h = np.zeros(n)
    idx = np.arange(n)
    h[np.mod(idx * alpha, 1.0) < 0.5] = 1.0
    return make_feasible(h, n)

def binary_local_search(h, n, rng, max_evals, time_limit, start_time):
    h = (h > 0.5).astype(float)
    h = make_feasible(h, n)
    s = np.sum(h)
    if abs(s - n / 2) > 1e-10:
        if s < n / 2:
            z = np.where(h == 0)[0]
            rng.shuffle(z)
            h[z[:int(n / 2 - s)]] = 1.0
        else:
            o = np.where(h == 1)[0]
            rng.shuffle(o)
            h[o[:int(s - n / 2)]] = 0.0
    c_best = compute_c5(h)
    no_improve = 0
    evals = 0
    while evals < max_evals:
        if time.time() - start_time > time_limit - 20:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:

        for _ in range(20):
            if evals >= max_evals:

            i = rng.choice(ones)
            j = rng.choice(zeros)
            h[i] = 0.0
            h[j] = 1.0
            c = compute_c5(h)
            evals += 1
            if c < c_best - 1e-14:
                c_best = c
                no_improve = 0
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        if no_improve > 40000:
            break
    return h, c_best

def block_shift_search(h, n, rng, time_limit, start_time):
    h = (h > 0.5).astype(float)
    h = make_feasible(h, n)
    c_best = compute_c5(h)
    for _ in range(3000):
        if time.time() - start_time > time_limit - 20:
            break
        start = rng.randint(0, n)
        length = rng.randint(2, 8)
        end = min(start + length, n)
        h_new = h.copy()
        for off in [1, -1, 2, -2, 3]:
            h_try = h.copy()
            s = start + off
            e = end + off
            if s < 0 or e > n:
                continue
            h_try[s:e] = h_try[s:e] * 0 + 1.0
            h_try[start:end] = 0.0
            h_try = make_feasible(h_try, n)
            if np.sum(h_try) != n / 2:
                continue
            c = compute_c5(h_try)
            if c < c_best - 1e-14:
                c_best = c
                h = h_try
                break
    return h, c_best

def continuous_coord_descent(h, n, time_limit, start_time):

    c_best = compute_c5(h)
    for rnd in range(30):
        if time.time() - start_time > time_limit - 15:

            h = make_feasible(h, n)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
