import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        order = np.argsort(np.abs(h - 0.5))
        for idx in order:
            if diff > 0:
                amt = min(1.0 - h[idx], diff)
                h[idx] += amt
                diff -= amt
            else:
                amt = min(h[idx], -diff)
                h[idx] -= amt
                diff += amt
            if abs(diff) < 1e-12:
                break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj_and_grad(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    logsum = np.log(np.sum(np.exp(t * shifted)))
    val = (mc + logsum / t) * dx
    w = np.exp(t * shifted)
    w /= np.sum(w)
    w_pad = np.zeros(4 * n)
    w_pad[n - 1:n - 1 + 2 * n - 1] = w
    h_pad = np.zeros(4 * n)
    h_pad[:n] = h
    Wf = np.fft.fft(w_pad)
    Hf = np.fft.fft(h_pad)
    S = np.real(np.fft.ifft(Hf * np.conj(Wf)))
    one_minus_S = 1.0 - S
    T = np.real(np.fft.ifft(np.fft.fft(one_minus_S) * np.conj(Wf)))
    grad = dx * ((1.0 - S[:n]) - T[:n])
    return val, grad

def slsqp_opt(h_init, n, t, maxiter, ftol=1e-13):
    def obj(h):
        v, _ = soft_obj_and_grad(h, n, t)
        return v
    def obj_grad(h):
        _, g = soft_obj_and_grad(h, n, t)
        return g
    x0 = h_init.copy()
    bounds = list(zip(np.zeros(n), np.ones(n)))
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n / 2.0,
             'jac': lambda h: np.ones(n)}]
    try:
        r = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
                     constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        if r.success or np.sum(r.x ** 2) > 0:
            return make_feasible(r.x, n)
    except Exception:
        pass
    return make_feasible(x0, n)

def hill_climb_pair(h, n, n_iter=500, temp=0.05, time_limit=None, t0=None):

        j = lo[rng3.randint(len(lo))] if len(lo) > 0 else 0

        h2 = h.copy()
        dg = rng3.uniform(0.05, 0.3)

        h2 = make_feasible(h2, n)
        c2 = compute_c5(h2)
        accept = c2 < best_c or (rng3.rand() < np.exp(-(c2 - best_c) / max(temp, 1e-6)))

    c_bin = compute_c5(h_bin)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin

    h_cd, c_cd = pair_swap_cd(best_h, len(best_h), max_rounds=10,
                              time_limit=budget_s, t0=start_time)
    if c_cd < best_c5:
        best_c5 = c_cd
        best_h = h_cd

    # Phase 4: Hill climbing / simulated annealing on binary structure
    h_hc, c_hc = hill_climb_pair(best_h, len(best_h), n_iter=400, temp=0.04,
                                 time_limit=budget_s, t0=start_time)
    if c_hc < best_c5:
        best_c5 = c_hc
        best_h = h_hc

    # Phase 5: Single-coordinate refinement
    h_s, c_s = single_cd(best_h, len(best_h), max_rounds=8,
                         time_limit=budget_s, t0=start_time)
    if c_s < best_c5:
        best_c5 = c_s
        best_h = h_s

    # Phase 6: If time remains, try higher resolution
    if time.time() - start_time < budget_s - 200:
        n3 = 2048
        h3 = upsample_h(best_h, n3)
        for t_val, mi in [(50, 200), (200, 250), (600, 300)]:
            if time.time() - start_time > budget_s - 100:
                break
            h_new = slsqp_opt(h3, n3, t_val, mi)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-9:
                best_c5 = c_new
                best_h = h_new
                h3 = h_new
            else:
                h3 = h_new
        best_h = h3
        h_cd2, c_cd2 = pair_swap_cd(best_h, len(best_h), max_rounds=5,
                                    time_limit=budget_s, t0=start_time)
        if c_cd2 < best_c5:
            best_c5 = c_cd2
            best_h = h_cd2

    # Phase 7: Final binary + refinement
    if time.time() - start_time < budget_s - 60:
        h_b2 = (best_h > 0.5).astype(float)
        h_b2 = make_feasible(h_b2, len(best_h))
        c_b2 = compute_c5(h_b2)
        if c_b2 < best_c5:
            best_c5 = c_b2
            best_h = h_b2
        h_f, c_f = pair_swap_cd(best_h, len(best_h), max_rounds=5,
                                time_limit=budget_s, t0=start_time)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f
        h_f2, c_f2 = single_cd(best_h, len(best_h), max_rounds=5,
                               time_limit=budget_s, t0=start_time)
        if c_f2 < best_c5:
            best_c5 = c_f2
            best_h = h_f2

    # Phase 8: Random restart attempt at current resolution if time allows
    if time.time() - start_time < budget_s - 150:
        n_r = len(best_h)
        h_rand = (rng.rand(n_r) < 0.5).astype(float)
        h_rand = make_feasible(h_rand, n_r)
        h_r_new = slsqp_opt(h_rand, n_r, 50, 200)
        c_r = compute_c5(h_r_new)
        if c_r < best_c5 - 1e-9:
            for t_val, mi in [(200, 200), (600, 250)]:
                if time.time() - start_time > budget_s - 80:
                    break
                h_r_new = slsqp_opt(h_r_new, n_r, t_val, mi)
                c_r = compute_c5(h_r_new)
            h_rc, c_rc = pair_swap_cd(h_r_new, n_r, max_rounds=5,
                                      time_limit=budget_s, t0=start_time)
            if c_rc < best_c5:
                best_c5 = c_rc
                best_h = h_rc

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
