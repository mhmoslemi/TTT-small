import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)

    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def ga_mutate(h, n, sigma, rng, flip_prob=0.05):
    """Gaussian mutation + occasional binary flips"""
    h_new = h.copy()
    noise = rng.normal(0, sigma, n)
    h_new += noise
    h_new = np.clip(h_new, 0, 1)
    # Binary flips
    if rng.random() < flip_prob:
        n_flip = max(1, int(n * 0.02))
        flip_idx = rng.choice(n, n_flip, replace=False)
        h_new[flip_idx] = rng.choice([0.0, 1.0], n_flip)
    return make_feasible(h_new, n)

    """Blend crossover"""
    alpha = rng.uniform(0.2, 0.8)
    child = alpha * p1 + (1 - alpha) * p2
    return make_feasible(child, n)

    """Tournament selection, returns index of winner"""
    contest = rng.choice(len(pop), k, replace=False)
    return contest[np.argmin(fitness[contest])]

    start = time.time()
    pop = [make_feasible(h.copy(), n) for h in initial_h_list]

    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    elite_keep = max(3, pop_size // 10)
    
    for gen in range(n_gen):
        if time.time() - start > time_limit:
            break
        
        # Adaptive sigma
        progress = gen / max(n_gen - 1, 1)
        sigma = sigma_start * (sigma_end / sigma_start) ** progress
        
        new_pop = []
        # Elitism
        elite_idx = np.argsort(fitness)[:elite_keep]
        for i in elite_idx:
            new_pop.append(pop[i].copy())
        
        # Fill rest
        while len(new_pop) < pop_size:
            if rng.random() < 0.7:
                # Crossover + mutation
                i1 = ga_tournament_select(pop, fitness, 4, rng)
                i2 = ga_tournament_select(pop, fitness, 4, rng)
                child = ga_crossover(pop[i1], pop[i2], n, rng)
                child = ga_mutate(child, n, sigma, rng)

                child = ga_mutate(pop[i1], n, sigma, rng)

        fitness = np.array([compute_c5(h) for h in pop])

            best_h = pop[gen_best].copy()

_OPT_T = 1.0

def _obj_fn(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _cons_jac(h):
    return np.ones(_OPT_N)

def _slsqp_optimize(h_start, n, T, max_iter):

    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj_fn, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)
