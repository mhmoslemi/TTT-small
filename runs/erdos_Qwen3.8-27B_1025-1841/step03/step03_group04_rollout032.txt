The smooth max is: f(h) = (max_k corr_k + (1/T) log sum_k exp(T(corr_k - max_k))) * dx

where w_k = exp(T(corr_k - max_k)) / sum_j exp(T(corr_j - max_j))

And grad(corr_k) can be computed for any specific k in O(n) time.

So the total gradient computation is: O(n * |active_set|) if we only include the active set, or O(n * 2n) = O(n²) for all shifts.

For n=960 and active_set of size 30, that's about 28,800 operations per gradient. Very fast!

def smooth_max_grad(x):

    shifted = T * (c_use - mc)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for i, k in enumerate(idx):
        w = weights[i]
        if w < 1e-10:
            continue
        d = abs(k - (n - 1))
        if d == 0:
            # corr = sum h[i]*(1-h[i])

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0
_G_TOP_IDX = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def _obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_TOP_IDX is not None:
        corr = corr[_G_TOP_IDX]
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -500, 500)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def _grad(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_TOP_IDX is not None:
        idx = _G_TOP_IDX
        c_use = corr[idx]
    else:
        idx = np.arange(len(corr))
        c_use = corr
    mc = np.max(c_use)
    shifted = np.clip(T * (c_use - mc), -500, 500)
    weights = np.exp(shifted)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    grad = np.zeros(n)
    for i in range(len(idx)):
        w = weights[i]
        if w < 1e-12:
            continue
        k = idx[i]
        d = abs(k - (n - 1))
        if d == 0:
            grad += w * (1.0 - 2.0 * x)
        else:
            grad[d:] += w * (1.0 - x[:n-d])
            grad[:n-d] -= w * x[d:]
    return dx * grad

    return np.sum(x) - _G_N / 2.0

    return np.ones(_G_N)

    mx = np.max(corr)
    thresh = mx - frac * mx
    idx = np.where(corr >= thresh)[0]
    if len(idx) > max_count:
        order = np.argsort(-corr[idx])
        idx = idx[order[:max_count]]
    elif len(idx) == 0:
        idx = np.array([int(np.argmax(corr))])
    return idx
