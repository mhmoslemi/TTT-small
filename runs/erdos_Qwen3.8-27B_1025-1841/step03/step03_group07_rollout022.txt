import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_autocorr(h):
    """Compute periodic autocorrelation R_h(k) = sum_i h[i]*h[(i+k)%n]*dx"""
    n = len(h)
    dx = 2.0 / n
    # Using FFT for efficient computation
    H = np.fft.rfft(h)
    R = np.fft.irfft(np.abs(H)**2) * dx
    return R

def get_worst_shifts(R, n, top_m=5):
    """Get the shifts with minimum autocorrelation (worst for C5)"""
    # R[k] is the autocorrelation at shift k
    # C5 = 1 - min_k R[k]
    # We want the k with smallest R[k]
    # Note: R is computed for k=0..n-1 (periodic)
    # But we should exclude k=0 since R[0] is always the largest
    R_no0 = R.copy()
    R_no0[0] = -np.inf
    worst = np.argpartition(R_no0, -top_m)[-top_m:]
    worst = worst[np.argsort(R_no0[worst])]  # sort by worst first
    return worst

def grad_autocorr_k(h, k, n):
    """Gradient of R_h(k) with respect to h[i] (times dx)"""
    dx = 2.0 / n
    h_shifted_plus = np.roll(h, -k)  # h[(i+k) mod n]
    h_shifted_minus = np.roll(h, k)   # h[(i-k) mod n]
    return (h_shifted_plus + h_shifted_minus) * dx

def nesterov_optimize(h, n, tau, time_limit, lr=None, top_k=10):
    """Nesterov with soft-max over top-k worst shifts"""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0

    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        # Compute correlation and find top_k worst
        corr = np.correlate(h, 1.0 - h, mode='full')
        # corr has 2n-1 entries, shifted by n-1
        # The max of corr is C5/dx
        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        max_c = corr.max()
        
        shifted = np.clip(tau * (top_vals - max_c), -100.0, 0.0)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        
        grad = np.zeros(n)
        for i_idx in range(len(top_idx)):
            pk = p[i_idx]
            k = top_idx[i_idx]
            sh = k - n + 1
            # h[i] * (1-h[i+sh]) for valid i
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
        grad *= dx
        
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        if it % 50 == 0:

                stall = 0

                stall = 0

        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def balanced_minimax(h, n, time_limit, top_m=5, lr=None):
    """Cycle through worst k values, taking steps to increase R_h(k)"""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.01
    
    v = np.zeros(n)
    beta = 0.7
    stall = 0
    k_counter = 0
    
    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        
        # Compute autocorrelation
        R = compute_autocorr(h)
        c5_now = 1.0 - R.min()
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            lr *= 0.8
            v *= 0.5
            stall = 0
            if lr < 1e-15:
                break
        
        # Get worst shift (cycle through top_m)
        worst = get_worst_shifts(R, n, top_m)
        k = worst[k_counter % top_m]
        k_counter += 1
        
        # Gradient of R_h(k) w.r.t. h
        grad = grad_autocorr_k(h, k, n)
        
        # We want to INCREASE R_h(k), so take step in +grad direction
        v = beta * v + lr * grad
        h_new = h + v
        h = project_to_feasible(h_new, n)
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5
