def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    x_new = np.arange(n_new) / n_new

    return np.fft.irfft(np.conj(H) * G, size)

import numpy as np
from scipy.optimize import minimize
import time

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return make_feasible(h_old.copy(), n_new)
    x_old = np.arange(n_old) * 1.0 / n_old
    x_new = np.arange(n_new) * 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, period=1.0)
    return make_feasible(h_new, n_new)

def fft_corr(h):
    n = len(h)
    size = 2 * n
    H = np.fft.rfft(h, size)
    G = np.fft.rfft(1.0 - h, size)
    return np.fft.irfft(np.conj(H) * G, size)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    return float(np.max(corr) * dx)

def lbgfb_fft(h_start, n, T, mu, max_iter):
    dx = 2.0 / n
    size = 2 * n

    def f(h):
        H = np.fft.rfft(h, size)
        G = np.fft.rfft(1.0 - h, size)
        corr = np.fft.irfft(np.conj(H) * G, size)
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -60, 60)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = h.sum() - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 30})
    return make_feasible(res.x, n)

def binary_search_fast(h_bin, n, rng, time_budget):
    h = h_bin.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    current = compute_c5(h)
    t_start = time.time()
    for rnd in range(300):
        if time.time() - t_start > time_budget:
            break
        improved = False
        n_tries = min(2000, max(300, len(ones) // 4))
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def targeted_perturb(h, n, rng, n_trials):
    dx = 2.0 / n
    best = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        idx = rng.randint(n)
        deltas = rng.choice([-0.1, -0.05, -0.02, 0.02, 0.05, 0.1])
        h_try = best.copy()
        h_try[idx] = np.clip(h_try[idx] + deltas, 0.0, 1.0)
        h_try = make_feasible(h_try, n)
        c = compute_c5(h_try)
        if c < best_c5 - 1e-15:
            best_c5 = c
            best = h_try
    return best, best_c5

def periodic_init(n, period, fill_ratio, rng):
    base = np.zeros(period)
    n_fill = max(1, int(period * fill_ratio))
    idx = rng.choice(period, n_fill, replace=False)
    base[idx] = 1.0
    h = np.tile(base, n // period + 1)[:n]
    return make_feasible(h, n)
