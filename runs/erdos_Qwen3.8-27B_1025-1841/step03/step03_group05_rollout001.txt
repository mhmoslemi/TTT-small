def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def softmax_grad(h, T, top_k=200):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if T < 0.01:
        idx = np.argmax(corr)
        return _subgrad_one(h, idx - (n - 1)) * dx
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = np.argmax(corr)
        return _subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot

    for i in idx:
        wi = w[i]
        if wi < 1e-14:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

    c = compute_c5(x)

    return bh, bc

    n = _SM['n']

        return mc * dx
    logsum = mc + np.log(np.sum(np.exp(T * (c - mc)))) / T

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if T < 0.003:
        idx = np.argmax(corr)
        return _subgrad_one(h, idx - (n - 1)) * dx
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = np.argmax(corr)
        return _subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    idx_arr = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx_arr:
        wi = w[i]
        if wi < 1e-15:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista_r(h, n_iter, T0, T1, lr0, lr1, check, top_k, rng=None):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    stagn = 0
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr0 + (lr1 - lr0) * f
        g = sm_grad(z, T, top_k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc - 1e-10:
                bc, bh = c, x.copy()
                stagn = 0
            else:
                stagn += 1
                if stagn > 3 and rng is not None and f < 0.8:
                    z = project_c(bh + rng.randn(n) * 0.01, n)
                    x = z.copy()
                    tk = 1.0
                    stagn = 0
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

_SM = {}

def _sobj(h):
    T = _SM['T']
    n = _SM['n']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    if T < 0.01:
        return mc * dx
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _scon_fun(h):
    return np.sum(h) - _SM['n'] / 2.0

def _scon_jac(h):
    return np.ones(_SM['n'])

def slsqp_polish(h, T=1.0, mi=500):
    n = len(h)

        r = minimize(_sobj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon_fun, 'jac': _scon_jac}],
                     options={'maxiter': mi, 'ftol': 1e-16, 'iprint': 0})

            hx = project_c(r.x, n)
            cx = compute_c5(hx)
            if cx < c0:
                return hx
    except Exception:
        pass
    return h
