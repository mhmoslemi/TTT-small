def correlate_fft(a, b):

    return make_feasible(h_new, new_n)

def soft_max_obj(x, T, n):
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode='full')
    cmax = float(np.max(corr))
    shifted = T * (corr - cmax)
    shifted -= np.max(shifted)  # for numerical stability
    logsum = float(np.log(np.sum(np.exp(shifted))))
    return (cmax + logsum / T) * dx

def soft_max_obj_grad(x, T, n):
    """Objective and gradient for soft-max approximation of max overlap."""
    dx = 2.0 / n
    g = 1.0 - x
    corr = np.correlate(x, g, mode='full')  # length 2n-1
    cmax = float(np.max(corr))
    shifted = T * (corr - cmax)
    shifted -= np.max(shifted)
    exp_vals = np.exp(shifted)
    zsum = float(np.sum(exp_vals))
    obj = (cmax + float(np.log(zsum)) / T) * dx
    p = exp_vals / zsum  # softmax weights, length 2n-1
    
    # Gradient computation
    # dc_k/dx_j = (1-x[j-shift_k])*1{valid} - x[j+shift_k]*1{valid}
    # where shift_k = k - (n-1)
    grad = np.zeros(n)
    shifts = np.arange(2*n-1) - (n-1)
    
    for k in range(2*n-1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = shifts[k]
        # Term 1: (1-x[j-s]) for valid j: 0<=j-s<n
        if s >= 0:
            lo, hi = s, n
        else:
            lo, hi = 0, n + s
        if lo < hi:
            idx = np.arange(lo, hi)
            grad[lo:hi] += pk * (1.0 - x[idx - s])

    grad *= dx

import numpy as np
from scipy.optimize import minimize
import time

def correlate_fft(a, b):
    n, m = len(a), len(b)
    N = n + m - 1
    Nfft = 1 << (N - 1).bit_length()
    fa = np.fft.rfft(a, Nfft)
    fb = np.fft.rfft(b[::-1], Nfft)
    result = np.fft.irfft(fa * fb, Nfft)[:N]
    return result[::-1]

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_fft(h, 1.0 - h)
    return float(np.max(corr) * dx)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = float(np.max(corr))
    sh = T * (corr - cmax)
    sh -= float(np.max(sh))
    logsum = float(np.log(np.sum(np.exp(sh))))
    return (cmax + logsum / T) * dx

def soft_max_obj_grad(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    cmax = float(np.max(corr))
    sh = T * (corr - cmax)
    sh -= float(np.max(sh))
    ev = np.exp(sh)
    zsum = float(np.sum(ev))
    obj = (cmax + float(np.log(zsum)) / T) * dx
    p = ev / zsum
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            lo1, hi1 = s, n
        else:
            lo1, hi1 = 0, n + s
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - x[lo1 - s:hi1 - s])
        if s >= 0:
            lo2, hi2 = 0, n - s
        else:
            lo2, hi2 = -s, n
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * x[lo2 + s:hi2 + s]
    grad *= dx
    return obj, grad

def make_constraints(n):
    def cfun(x):
        return np.sum(x) - n / 2.0
    return [{'type': 'eq', 'fun': cfun}]

def lfbgsopt(h_start, n, T, maxiter, max_fun):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = make_constraints(n)
    try:
        res = minimize(
            lambda x: soft_max_obj(x, T, n),
            lambda x: soft_max_obj_grad(x, T, n)[1],
            h_start,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'maxfun': max_fun, 'ftol': 1e-15, 'gtol': 1e-10}
        )
        return res.x
    except Exception:
        return h_start

def slsqpopt(h_start, n, T, maxiter):
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = make_constraints(n)
    try:
        res = minimize(
            lambda x: soft_max_obj(x, T, n),
            h_start,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return res.x
    except Exception:
        return h_start

def binary_local_search(h, n, rng, n_rounds=30, n_tries_per_round=3000, time_limit=200):
    h = h.copy()
    best_c = compute_c5_fft(h)
    t_start = time.time()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    c_and_cneg = np.maximum(h, 0.5), np.minimum(h, 0.5)
    h_b = (h > 0.5).astype(float)
    ones = np.where(h_b > 0.5)[0]
    zeros = np.where(h_b < 0.5)[0]
    best_c = compute_c5_fft(h_b)
    h = h_b
    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for _ in range(n_tries_per_round):
            if time.time() - t_start > time_limit:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h_trial = h.copy()
            h_trial[i] = 0.0
            h_trial[j] = 1.0
            c = compute_c5_fft(h_trial)
            if c < best_c - 1e-12:
                h = h_trial
                best_c = c
                improved = True
    return h, best_c

def coord_descent(h, n, rng, n_rounds=50, time_limit=180):
    h = h.copy()
    best_c = compute_c5_fft(h)
    t_start = time.time()
    for rnd in range(n_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        order = np.arange(n)
        rng.shuffle(order)
        for j in order:
            if time.time() - t_start > time_limit:
                break
            old = h[j]
            best_cand = old
            best_cc = best_c
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old - 0.05), max(0.0, old - 0.02), max(0.0, old - 0.005),
                     min(1.0, old + 0.005), min(1.0, old + 0.02), min(1.0, old + 0.05)]
            for cand in cands:
                if abs(cand - old) < 1e-14:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5_fft(trial)
                if c < best_cc - 1e-13:
                    best_cc = c
                    best_cand = cand
            if best_cand != old:
                h[j] = best_cand
                h = make_feasible(h, n)
                best_c = best_cc
                improved = True
        if not improved:
            break
    return h, best_c
