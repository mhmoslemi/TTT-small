def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_shift(h, shift):
    """Compute subgradient of c(shift) w.r.t. h"""
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_smooth(h, n_iter, T0, T1, lr, check, top_k=50):

    n = len(h)
    dx = 2.0 / n

    bh = x.copy()

        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:

            g = subgrad_shift(z, k_star) * dx
        else:
            w = exps / tot

    return bh, bc

                x = xn

        else:

            break
    return bh, bc

def weighted_max_obj(h, top_k=10, T=5.0):
    """Weighted max objective for SLSQP"""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    # Soft max of top correlations
    top = np.sort(corr)[::-1][:top_k]
    return (mc + np.log(np.sum(np.exp(T * (top - mc)))) / T) * dx

def slsqp_topk(h, T=5.0, top_k=10, maxiter=500):
    """SLSQP using top-k weighted objective"""
    n = len(h)
    def obj(hh):
        return weighted_max_obj(hh, top_k, T)
    def obj_grad(hh):
        dx = 2.0 / n
        corr = get_corr(hh)
        mc = np.max(corr)
        top_idx = np.argsort(corr)[::-1][:top_k]
        exps = np.exp(T * (corr[top_idx] - mc))
        tot = np.sum(exps)
        w = exps / tot
        g = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            g += wi * subgrad_shift(hh, int(idx) - (n - 1))
        return g * dx
    def con(hh):
        return np.sum(hh) - n / 2.0
    try:
        r = minimize(obj, h, jac=obj_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def downsample(h, nn):
    if nn >= len(h):
        return project_c(h, len(h))
    # Average blocks
    n = len(h)

    for i in range(nn):
        start = i * block_size
        end = min(start + block_size, n)
        hd[i] = np.mean(h[start:end])
    return project_c(hd, nn)

    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)

    print(f"P5: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    
    # Phase 6: Final SLSQP
    if len(bh) <= 960 and remaining() > 20:
        bh = slsqq_topk(bh, T=3.0, top_k=8, maxiter=300)
        bc = compute_c5(bh)
        print(f"P6: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    
    # Phase 7: Final shift suppression
    if remaining() > 15:
        bh, bc = shift_suppress(bh, 500, lr_base=0.005)
    
    # Phase 8: Ultra-fine FISTA
    if remaining() > 10:
        hf, cf = fista_fixed_T(bh, 1000, 0.1, 0.003, top_k=5)
        if cf < bc:
            bc, bh = cf, hf.copy()
    
    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
