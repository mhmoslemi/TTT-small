def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:

        shifted = np.clip(T * (corr - mc), -50, 50)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(hv) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n

                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})

            h_work = h_ls

    # Phase 4: Get out of local optima - perturb and re-search
    if tl() > 50:
        h_pert = h_work.copy()
        n_perturb = n_main // 20
        for _ in range(n_perturb):
            ones_tmp = np.where(h_pert == 1)[0]
            zeros_tmp = np.where(h_pert == 0)[0]
            if len(ones_tmp) == 0 or len(zeros_tmp) == 0:
                break
            a = int(rng.randint(len(ones_tmp)))
            b = int(rng.randint(len(zeros_tmp)))
            h_pert[ones_tmp[a]], h_pert[zeros_tmp[b]] = 0.0, 1.0
        h_pa, c5_pa = sa_binary(h_pert, n_main, rng, 0.02, 0.001, 200000, tl() * 0.15)
        if c5_pa < best_c5:
            best_c5 = c5_pa
            best_h = h_pa.copy()
            h_work = h_pa

    # Phase 5: Local search on best
    if tl() > 20:
        h_ls2, c5_ls2 = local_search_binary(h_work, n_main, rng, max_rounds=20,
                                            tries_per_round=20000, time_limit=tl() * 0.1)
        if c5_ls2 < best_c5:
            best_c5 = c5_ls2
            best_h = h_ls2.copy()

    # Phase 6: Try n=4096
    if tl() > 80:
        n_hi = 4096
        h_hi = upsample_h(best_h, n_hi)
        h_hi_bin = to_binary(h_hi, n_hi)
        c5_hi = compute_c5(h_hi_bin)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi_bin.copy()
        if tl() > 60:
            h_hi_sa, c5_hi_sa = sa_binary(h_hi_bin, n_hi, rng, 0.03, 0.001,
                                          200000, tl() * 0.35)
            if c5_hi_sa < best_c5:
                best_c5 = c5_hi_sa
                best_h = h_hi_sa.copy()
            if tl() > 20:
                h_hi_ls, c5_hi_ls = local_search_binary(h_hi_sa, n_hi, rng, 10, 20000,
                                                        time_limit=tl() * 0.2)
                if c5_hi_ls < best_c5:
                    best_c5 = c5_hi_ls
                    best_h = h_hi_ls.copy()

    # Phase 7: Continuous relaxation at working resolution
    if tl() > 25:
        n_c = len(best_h)
        h_c = make_feasible(best_h.astype(float), n_c)
        for T in [50, 300, 1000]:
            if tl() < 10:
                break
            try:
                h_opt = lbgfb_smooth(h_c, n_c, T, mu=500, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass
        # Re-binarize
        h_rb = to_binary(h_c, n_c)
        c5_rb = compute_c5(h_rb)
        if c5_rb < best_c5:
            best_c5 = c5_rb
            best_h = h_rb.copy()

    # Phase 8: Structured pattern exploration
    if tl() > 20:
        n_s = 2048
        for period in [7, 11, 13, 17, 19, 23, 31, 37, 43, 53, 61, 71, 83, 97, 113, 131, 151, 181, 211, 251, 311, 379]:
            if tl() < 5:
                break
            half = (period + 1) // 2
            pat = np.zeros(period)
            pat[:half] = 1.0
            rep = np.tile(pat, n_s // period + 1)[:n_s]
            for noise in [0.0, 0.2, 0.5]:
                noisy = rep + noise * rng.randn(n_s)
                h_p = to_binary(noisy, n_s)
                c5_p = compute_c5(h_p)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = h_p.copy()

    # Phase 9: Final polish
    if tl() > 10:
        n_f = len(best_h)
        h_f = make_feasible(best_h.astype(float), n_f)
        try:
            h_opt = lbgfb_smooth(h_f, n_f, 200, mu=1000, max_iter=100)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
