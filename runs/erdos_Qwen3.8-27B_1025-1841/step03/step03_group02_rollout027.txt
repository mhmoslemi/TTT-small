import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_bin(n):
    tm = np.zeros(n)
    for i in range(1, n):
        tm[i] = 1 - tm[i >> 1] if (i & -i) != (i & (i+1)) else tm[i >> 1]
    # Standard construction
    tm = np.zeros(n, dtype=float)
    tm[0] = 0
    for i in range(1, n):
        tm[i] = (i ^ (i >> 1)).bit_count() % 2
    return tm

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    current = int(np.sum(b))
    if current > target:
        ones = np.where(b == 1)[0]
        order = ones[np.argsort(-h[ones])]
        for idx in order[:current - target]:
            b[idx] = 0
    elif current < target:
        zeros = np.where(b == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        for idx in order[:target - current]:
            b[idx] = 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for it in range(max_iters):
        if (it % 1000 == 0) and (time.time() - start_time) > budget_s * 0.85:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def binary_sa_block(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if (it % 500 == 0) and (time.time() - start_time) > budget_s * 0.85:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        bs = rng.randint(1, 5)
        start = rng.randint(0, n - bs)
        end = start + bs
        seg = b[start:end]
        ones_seg = np.where(seg == 1)[0]
        zeros_seg = np.where(seg == 0)[0]
        if len(ones_seg) == 0 or len(zeros_seg) == 0:
            continue
        swp = min(len(ones_seg), len(zeros_seg))
        c1 = rng.choice(ones_seg, size=swp, replace=False)
        c0 = rng.choice(zeros_seg, size=swp, replace=False)
        b[start + c1] = 0
        b[start + c0] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + c1] = 1
            b[start + c0] = 0
    return best_b, best_c5

def _obj_slsqp(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    logsum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + logsum / T) * dx

def _con_sum(x):
    return np.sum(x) - len(x) / 2.0

def _con_sum_jac(x):
    return np.ones(len(x))

    improved = True
    rnd = 0
    while improved and rnd < max_rounds:

        improved = False
        remaining = budget_s - (time.time() - start_time)

        order = np.random.permutation(n)  # Use global rng via closure-free approach
        for j in order:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            if old < 0.01 or old > 0.99:
                cands = [0.0, 1.0]
            else:
                cands = [0.0, 1.0]
                for d in [0.01, 0.03, 0.05, 0.1, 0.2]:
                    cands.append(max(0.0, old - d))
                    cands.append(min(1.0, old + d))
            cands = list(set(cands))
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue

                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def pair_swap_cd(h, n, max_rounds, start_time, budget_s, n_local_rng):
    """Try swapping pairs of values to reduce max correlation."""
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        for _ in range(n // 4):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            i = int(n_local_rng.randint(n))
            j = int(n_local_rng.randint(n))
            if i == j:
                continue
            # Try: h[i] -> 1-h[i], h[j] -> 1-h[j] (swap)
            trial = h.copy()
            vi, vj = trial[i], trial[j]
            trial[i] = 1.0 - vj
            trial[j] = 1.0 - vi
            if np.any(trial < -1e-10) or np.any(trial > 1 + 1e-10):
                continue
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                h = trial
                best_c5 = c
                improved = True
    return h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
