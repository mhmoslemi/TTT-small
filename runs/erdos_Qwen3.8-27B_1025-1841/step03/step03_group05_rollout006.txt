```python
import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(h, one_minus_h, n):
    """Compute full correlation using FFT, O(n log n)."""
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(one_minus_h, N)
    corr = np.fft.irfft(H * np.conj(V), N)
    # Reorder to match np.correlate(h, one_minus_h, 'full')
    # np.correlate index j corresponds to lag j-(n-1)
    # FFT: corr_fft[0] = lag 0, corr_fft[1] = lag 1, ..., corr_fft[n-1] = lag n-1
    #      corr_fft[N-1] = lag -(n-1), ..., corr_fft[N-n+1] = lag -1
    result = np.zeros(2 * n - 1)
    result[n-1:] = corr[:n]       # lags 0 to n-1
    result[:n-1] = corr[N-n+1:]   # lags -(n-1) to -1
    return result

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h, n)
    return float(np.max(corr) * dx)
```

```python
from scipy.optimize import minimize
import time

def fft_corr_full(h, n):
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    vm = 1.0 - h
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(vm, N)
    c = np.fft.irfft(H * np.conj(V), N)
    out = np.empty(2 * n - 1)
    out[n-1:] = c[:n]
    out[:n-1] = c[N-n+1:]
    return out

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr_full(h, n)
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def subgrad_lag(g, h, shift, w, n):
    s = abs(shift)
    if s >= n:
        return
    if shift >= 0:
        g[s:] += w * (1.0 - h[:n-s])
        g[:n-s] -= w * h[s:]
    else:
        g[:n-s] += w * (1.0 - h[s:])
        g[s:] -= w * h[:n-s]

def fista(h0, n_iter, T0, T1, lr0, lr1, tk0, tk1, check=100):

    dx = 2.0 / n
    x = h0.copy()
    z = x.copy()
    tk_acc = 1.0
    bc = compute_c5(x)
    bh = x.copy()

            w[np.argmax(corr)] = 1.0

        g = np.zeros(n)
        for i in idx:

            if wi < 1e-10:

            subgrad_lag(g, z, i - (n - 1), wi, n)
        g *= dx
        xn = project_c(z - lr * g, n)

                bh = x.copy()

    if c < bc:
        bc = c
        bh = x.copy()
```

```python
import time

def fft_corr_full(h, n):
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(1.0 - h, N)
    c = np.fft.irfft(H * np.conj(V), N)
    out = np.empty(2 * n - 1)
    out[n - 1:] = c[:n]
    out[:n - 1] = c[N - n + 1:]
    return out

def compute_c5(h):
    n = len(h)
    return float(np.max(fft_corr_full(h, n)) * (2.0 / n))

def project_c(h, n):
    target = 0.5 * n
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _sg(g, z, omz, lag, w, n):

    if s >= n:
        return

        g[s:] += w * omz[:n - s]
        g[:n - s] -= w * z[s:]

    tau = 1.0

    bh = x.copy()

        f = it / max(niter - 1, 1)
        T = T0 * (T1 / T0) ** f

            s = 1.0

                bh = x.copy()

    if c < bc:
```
