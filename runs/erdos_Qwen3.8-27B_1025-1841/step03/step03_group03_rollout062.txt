import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Bisection projection
        lo, hi = -max(abs(diff)/n, 1e-10), max(abs(diff)/n, 1e-10) + (1.0 if diff > 0 else -1.0)
        # Actually simpler: just add uniformly then clip
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff2 / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_constraint(h, target):
    """Project h onto {0<=x<=1, sum(x)=target} using bisection"""
    n = len(h)

    # If current_sum > target, we need tau > 0 (shift down)
    # If current_sum < target, we need tau < 0 (shift up)
    lo = -1.0

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:

        grad = compute_grad(z, t)
        
        f_curr = soft_obj(h, t)
        alpha = lr
        accepted = False
        for _ in range(20):
            h_new = project_onto(h - alpha * grad, n / 2.0)
            f_new = soft_obj(h_new, t)
            if f_new < f_curr - 1e-16:
                accepted = True
                break
            alpha *= 0.5
        
        if accepted:
            h_prev = h.copy()
            h = h_new
            lr = min(alpha * 1.4, 2.0)
        else:
            lr *= 0.7
        
        mu = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        if mu > 10.0:
            mu = 10.0
    
    c_final = compute_c5(h)
    if c_final < best_c - 1e-12:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c

def binary_local_search(h, n, max_rounds=50, time_limit=1200, start_time=None):
    h = (h > 0.5).astype(float)
    h = project_onto(h, n / 2.0)
    best_c = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if start_time is not None and time.time() - start_time > time_limit:
            break
        improved = False
        indices = np.arange(n)
        np.random.seed(rnd * 7 + 13)
        np.random.shuffle(indices)
        
        for j in indices:
            if start_time is not None and time.time() - start_time > time_limit:
                return h, best_c
            old = h[j]
            new_val = 1.0 - old
            
            # Need to flip another position to maintain sum
            flip_candidates = [i for i in range(n) if i != j]
            np.random.shuffle(flip_candidates)
            
            best_pair = None
            best_cand_c = best_c
            for k in flip_candidates[:30]:
                if h[k] == new_val:
                    h[j] = new_val
                    h[k] = new_val
                    c_test = compute_c5(h)
                    if c_test < best_cand_c - 1e-12:
                        best_cand_c = c_test
                        best_pair = (j, k, new_val)
                    h[j] = old
                    h[k] = old
                    if best_cand_c < best_c - 0.01:
                        break
            
            if best_pair is not None:
                jj, kk, nv = best_pair
                h[jj] = nv
                h[kk] = nv
                best_c = best_cand_c
                improved = True
    
    return h, best_c

def coord_descent_frac(h, n, max_rounds=15, time_limit=1200, start_time=None):
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if start_time is not None and time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if start_time is not None and time.time() - start_time > time_limit - 10:
                return h, best_c
            old = h[j]
            best_val = old
            best_obj = best_c
            for cv in [0.0, 1.0, 0.1, 0.3, 0.5, 0.7, 0.9, old + 0.05, old - 0.05]:
                if cv < -1e-9 or cv > 1.0 + 1e-9:
                    continue
                cv = float(np.clip(cv, 0.0, 1.0))
                trial = h.copy()
                trial[j] = cv
                trial = project_onto(trial, n / 2.0)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = project_onto(h, n / 2.0)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return project_onto(h_new, new_n / 2.0)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            h = project_onto(h, target)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        pass
    except Exception:
        pass
    
    n = len(h0) if h0 is not None else 640
    if h0 is None:
        h0 = (rng.rand(n) < 0.5).astype(float)
        h0 = make_feasible(h0, n)
    
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: PGD at original resolution with increasing t
    n1 = len(best_h)
    for t, mi, lr in [(30, 3000, 0.05), (80, 3000, 0.03), (200, 4000, 0.015), (500, 4000, 0.008)]:
        if time.time() - start_time > budget_s - 500:
            break
        h_new, c_new = pgd_momentum(best_h, n1, t, mi, budget_s - 300, start_time, lr)
        if c_new < best_c5 - 1e-10:
            best_c5 = c_new
            best_h = h_new
    
    # Phase 2: Binary local search at original resolution
    if time.time() - start_time < budget_s - 300:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = project_onto(h_bin, n1 / 2.0)
        h_bl, c_bl = binary_local_search(h_bin, n1, max_rounds=30,
                                          time_limit=budget_s - 200, start_time=start_time)
        if c_bl < best_c5 - 1e-10:
            best_c5 = c_bl
            best_h = h_bl
    
    # Phase 3: Upsample to 1024 and optimize
    n3 = 1024
    if time.time() - start_time < budget_s - 350:
        h3 = upsample_h(best_h, n3)
        for t, mi, lr in [(50, 3000, 0.03), (150, 3000, 0.015), (400, 3000, 0.008)]:
            if time.time() - start_time > budget_s - 250:
                break
            h_new, c_new = pgd_momentum(h3, n3, t, mi, budget_s - 200, start_time, lr)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                best_h = h_new
                h3 = h_new
            else:
                h3 = h_new
    
    # Phase 4: Upsample to 1536
    n4 = 1536
    if time.time() - start_time < budget_s - 250:
        h4 = upsample_h(best_h, n4)
        for t, mi, lr in [(80, 3000, 0.02), (250, 2000, 0.01)]:
            if time.time() - start_time > budget_s - 150:
                break
            h_new, c_new = pgd_momentum(h4, n4, t, mi, budget_s - 120, start_time, lr)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                best_h = h_new
                h4 = h_new
            else:
                h4 = h_new
    
    # Phase 5: Coordinate descent at best resolution
    nc = len(best_h)
    if time.time() - start_time < budget_s - 80:
        h_cd, c_cd = coord_descent_frac(best_h, nc, max_rounds=10,
                                         time_limit=budget_s - 60, start_time=start_time)
        if c_cd < best_c5 - 1e-12:
            best_c5 = c_cd
            best_h = h_cd
    
    # Phase 6: Binary local search at best resolution
    if time.time() - start_time < budget_s - 40:
        nb = len(best_h)
        h_b2 = (best_h > 0.5).astype(float)
        h_b2 = project_onto(h_b2, nb / 2.0)
        h_bl2, c_bl2 = binary_local_search(h_b2, nb, max_rounds=15,
                                            time_limit=budget_s - 20, start_time=start_time)
        if c_bl2 < best_c5 - 1e-12:
            best_c5 = c_bl2
            best_h = h_bl2
    
    # Phase 7: Final fractional refinement
    if time.time() - start_time < budget_s - 20:
        nf = len(best_h)
        h_f, c_f = coord_descent_frac(best_h, nf, max_rounds=5,
                                       time_limit=budget_s - 10, start_time=start_time)
        if c_f < best_c5 - 1e-12:
            best_c5 = c_f
            best_h = h_f
    
    # Final: also try binary at the very end
    nb2 = len(best_h)
    h_b3 = (best_h > 0.5).astype(float)
    h_b3 = project_onto(h_b3, nb2 / 2.0)
    c_b3 = compute_c5(h_b3)
    if c_b3 < best_c5 - 1e-12:
        best_c5 = c_b3
        best_h = h_b3
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
