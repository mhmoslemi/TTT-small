_G_N = 0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        idx = int(np.argmax(h)) if diff < 0 else int(np.argmin(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    n = _G_N
    T = _G_T

    m2 = np.max(shifted)
    exp_vals = np.exp(T * (shifted - m2))
    return mc + m2 + np.log(exp_vals.sum()) / T

    return np.sum(h) - _G_N / 2.0

    return np.ones(_G_N)

                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

        j = rng.randint(n)

    if T * np.min(s) > -30:

            if len(j2) > 0:
                grad[j2] += -p * h[j2 - offset]
