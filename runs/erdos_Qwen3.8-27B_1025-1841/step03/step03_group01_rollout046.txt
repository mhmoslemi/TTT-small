import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def smooth_obj_and_grad(h, T, mu, n):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, 'full')
    m = corr.max()
    shifted = T * (corr - m)
    shifted -= shifted.max()
    w = np.exp(shifted)
    w_sum = w.sum()
    w /= w_sum
    obj_val = (m + np.log(w_sum) / T) * dx
    
    # Gradient of smooth part
    n_corr = 2 * n - 1
    corr_gw = np.correlate(g, w, 'full')
    term1 = corr_gw[3*n-3 : 2*n-2 : -1].copy()
    w_rev = w[::-1]
    corr_hw = np.correlate(h, w_rev, 'full')
    term2 = corr_hw[3*n-3 : 2*n-2 : -1].copy()
    grad = dx * (term1 - term2)
    
    # Penalty gradient
    s = np.sum(h) - n / 2.0
    if mu > 0:
        obj_val += mu * s * s
        grad += 2.0 * mu * s
    
    return obj_val, grad

def projected_h(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile(np.array([1.0, 0.0]), 320)  # n=640
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Upsample to working resolution
    n_work = 2048
    h_curr = upsample_h(best_h, n_work)
    c5_curr = compute_c5(h_curr)
    if c5_curr < best_c5:
        best_c5 = c5_curr
        best_h = h_curr.copy()
    
    n = n_work
    dx = 2.0 / n
    
    # L-BFGS-B optimization with smooth max + penalty
    # Progressively increase T (sharper max) and mu (tighter constraint)
    schedules = [
        # (T, mu, maxiter)
        (10.0, 1.0, 200),
        (30.0, 10.0, 200),
        (100.0, 100.0, 200),
        (300.0, 1000.0, 150),
        (1000.0, 10000.0, 100),
        (5000.0, 100000.0, 50),
    ]
    
    for T, mu, maxiter in schedules:
        if time_left() < 30:

            val, _ = smooth_obj_and_grad(h_x, T, mu, n)

            return gr
