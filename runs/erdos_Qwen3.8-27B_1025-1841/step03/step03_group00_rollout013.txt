        else:
            for i in ones:
                j = i + max_lag
                if 0 <= j < n and h[j] == 0.0:
                    bad_ones.append(i)
        pool = bad_ones if len(bad_ones) > 0 else ones
        n_tries = min(3000, len(pool) * len(zeros))
        for _ in range(n_tries):
            ii = rng.randint(len(pool))
            jj = rng.randint(len(zeros))
            oi = pool[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            new_c = compute_c5(h)
            if new_c < current - 1e-15:
                current = new_c
                if new_c < best - 1e-15:
                    best = new_c
                    best_h = h.copy()
                ones = list(np.where(h == 1.0)[0])
                zeros = list(np.where(h == 0.0)[0])
                if oi in ones:
                    pool = [p for p in pool if p != oi]
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return best_h, best

def lbfgs_smooth(h_start, n, T, mu, max_iter):
    dx = 2.0 / n
    target = n / 2.0
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -50.0, 50.0)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - target
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def slsqp_polish(h_start, n, T, max_iter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -50.0, 50.0)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    def con(h):
        return np.sum(h) - target
    def jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con, 'jac': jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def coordinate_descent(h_start, n, max_rounds=6):
    h = h_start.copy()
    best_c5 = compute_c5(h)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < 0.0 or nv > 1.0:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = tr
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Binary optimization at parent resolution
    if tl() > 150:
        h_bin = to_binary(h0, n0)
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

        # Targeted local search first (faster convergence)
        h_t, c5_t = targeted_local_search(h_bin, n0, rng, max_rounds=200)
        if c5_t < best_c5:
            best_c5, best_h = c5_t, h_t.copy()
            h_bin = h_t.copy()

        # SA round 1
        n_sa = int(min(tl() - 60, 300) * 80000)
        n_sa = max(n_sa, 200000)
        h_sa, c5_sa = sa_binary(h_bin, n0, rng, n_sa, 0.008, 0.0004)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()
            h_bin = h_sa.copy()

        # Targeted local search again from SA result
        if tl() > 80:
            h_t2, c5_t2 = targeted_local_search(h_bin, n0, rng, max_rounds=150)
            if c5_t2 < best_c5:
                best_c5, best_h = c5_t2, h_t2.copy()

            # SA round 2 (cooler)
            n_sa2 = int(min(tl() - 40, 150) * 80000)
            n_sa2 = max(n_sa2, 100000)
            h_sa2, c5_sa2 = sa_binary(h_t2, n0, rng, n_sa2, 0.004, 0.0002)
            if c5_sa2 < best_c5:
                best_c5, best_h = c5_sa2, h_sa2.copy()

    # Phase 2: Continuous refinement at parent resolution
    if tl() > 60:
        h_work = best_h.copy()
        for T in [30, 100, 300, 1000]:
            if tl() < 25:
                break
            try:
                h_opt = lbfgs_smooth(h_work, n0, T, 500.0, 200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_work = h_opt.copy()
            except Exception:
                pass

    # Phase 3: Higher resolution refinement
    for n_pts in [1280, 2560]:
        if tl() < 70:
            break
        h_c = upsample(best_h, n_pts)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()

        # Binary SA at new resolution
        if tl() > 50:
            h_bin_c = to_binary(h_c, n_pts)
            c5b_c = compute_c5(h_bin_c)
            if c5b_c < best_c5:
                best_c5, best_h = c5b_c, h_bin_c.copy()
                h_c = h_bin_c.copy()

            n_sa_c = int(min(tl() - 25, 150) * 40000)
            n_sa_c = max(n_sa_c, 100000)
            h_sa_c, c5_sa_c = sa_binary(h_bin_c, n_pts, rng, n_sa_c, 0.006, 0.0003)
            if c5_sa_c < best_c5:
                best_c5, best_h = c5_sa_c, h_sa_c.copy()
                h_c = h_sa_c.copy()

        # Continuous at new resolution
        if tl() > 30:
            for T in [80, 400]:
                if tl() < 20:
                    break
                try:
                    h_opt = lbfgs_smooth(h_c, n_pts, T, 200.0, 150)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5, best_h = c5_opt, h_opt.copy()
                        h_c = h_opt.copy()
                except Exception:
                    pass

    # Phase 4: SLSQP polish at best resolution
    if tl() > 25:
        n_s = len(best_h)
        if n_s > 2048:
            n_s = 2048
            h_s = upsample(best_h, n_s)
        else:
            h_s = best_h.copy()
        for T in [500, 3000]:
            if tl() < 12:
                break
            try:
                h_opt = slsqp_polish(h_s, n_s, T, 100)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_s = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Final coordinate descent
    if tl() > 15:
        n_cd = len(best_h)
        h_cd, c5_cd = coordinate_descent(best_h.copy(), n_cd, max_rounds=4)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
