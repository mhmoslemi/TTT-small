import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b).astype(int) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b).astype(int) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def fast_binary_sa(h, n, max_time, T0, T1, rng, start_time):
    """Fast binary SA with incremental correlation updates."""
    b = to_binary(h, n)
    g = 1.0 - b
    corr = np.correlate(b, g, mode='full')
    curr_max = float(np.max(corr))
    best_max = curr_max
    best_b = b.copy()
    
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    it = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= max_time:
            break
        
        it += 1
        T = T0 * (T1 / T0) ** (elapsed / max_time)
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta
        # h[i]: 1->0, h[j]: 0->1, g[i]: 0->1, g[j]: 1->0
        n2 = 2 * n - 1  # total corr length
        
        # Affected k range
        k_lo = max(0, min(i, j, n-1-i, n-1-j))
        k_hi = min(n2-1, max(i+n-1, j+n-1, 2*n-2-i, 2*n-2-j))
        
        ks = np.arange(k_lo, k_hi + 1)
        delta = np.zeros(len(ks))
        
        # Term 1: -g[i+n-1-k]
        idx1 = i + n - 1 - ks
        v1 = (idx1 >= 0) & (idx1 < n)
        delta[v1] -= g[idx1[v1]]
        
        # Term 2: +g[j+n-1-k]
        idx2 = j + n - 1 - ks
        v2 = (idx2 >= 0) & (idx2 < n)
        delta[v2] += g[idx2[v2]]
        
        # Term 3: +h[k-n+1+i]
        idx3 = ks - n + 1 + i
        v3 = (idx3 >= 0) & (idx3 < n)
        delta[v3] += b[idx3[v3]]
        
        # Term 4: -h[k-n+1+j]
        idx4 = ks - n + 1 + j
        v4 = (idx4 >= 0) & (idx4 < n)
        delta[v4] -= b[idx4[v4]]
        
        new_corr_seg = corr[k_lo:k_hi+1] + delta
        new_max = float(np.max(new_corr_seg))
        # Also need max of unchanged parts
        if k_lo > 0:
            new_max = max(new_max, float(np.max(corr[:k_lo])))
        if k_hi < n2-1:
            new_max = max(new_max, float(np.max(corr[k_hi+1:])))
        
        if new_max <= curr_max or (T > 1e-12 and rng.random() < np.exp(-(new_max - curr_max) / T)):
            # Accept
            corr[k_lo:k_hi+1] = new_corr_seg
            b[i] = 0
            b[j] = 1
            g[i] = 1
            g[j] = 0
            ones = np.delete(ones, np.where(ones == i)[0][0])
            zeros = np.append(zeros, i)
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            ones = np.append(ones, j)
            curr_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
        else:
            pass  # no change
    
    return best_b, best_max

    h = h.copy().astype(float)
    h = make_feasible(h, n)
    dx = 2.0 / n
    n2 = 2 * n - 1
    alpha = alpha_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    while time.time() - start_time < max_time:
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = float(np.max(corr)) * dx
        
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        
        # Find top M lags
        M = min(10, n2)
        top_idx = np.argsort(corr)[-M:]
        
        # Compute averaged gradient for top M lags
        grad = np.zeros(n)
        for k in top_idx:
            for i in range(n):

                # (with 0 for out-of-bounds)

        # Check improvement

        if c5_new < curr_c5:
            h = h_new
            alpha = min(alpha * 1.05, 0.1)
        else:
            alpha *= 0.9
            if alpha < 1e-6:
                break
    
    return best_h, best_c5

Actually, for a given k, the gradient d(corr[k])/d(h) is:
- For position i: (1-h[i+n-1-k]) - h[k-n+1+i] (with 0 for OOB)

This is essentially: (1-h) shifted by (n-1-k) minus h shifted by (k-n+1).

In numpy terms:
- term1 = zero_padded(1-h, shift_by=n-1-k)
- term2 = zero_padded(h, shift_by=k-n+1)
- grad_k = term1 - term2

Where zero_padded(x, shift) puts x at the appropriate position with zeros elsewhere.

For shift s >= 0: result[s:] = x[:n-s], result[:s] = 0
For shift s < 0: result[:n+s] = x[-s:], result[n+s:] = 0

Let me implement this vectorized:

```python
def grad_of_corr_lag(k, h, n):
    """Gradient of corr[k] w.r.t. h, where corr = correlate(h, 1-h, 'full')"""
    g = 1.0 - h
    grad = np.zeros(n)
    
    # term1: (1-h)[i + n - 1 - k] for valid i
    # i + n - 1 - k in [0, n-1] => k - n + 1 <= i <= k
    i_lo = max(0, k - n + 1)
    i_hi = min(n-1, k)
    if i_lo <= i_hi:
        m_lo = i_lo + n - 1 - k
        m_hi = i_hi + n - 1 - k
        grad[i_lo:i_hi+1] += g[m_lo:m_hi+1]
    
    # term2: -h[k - n + 1 + i] for valid i
    # k - n + 1 + i in [0, n-1] => n - 1 - k <= i <= 2n - 2 - k
    i_lo2 = max(0, n - 1 - k)
    i_hi2 = min(n-1, 2*n - 2 - k)
    if i_lo2 <= i_hi2:
        m_lo2 = k - n + 1 + i_lo2
        m_hi2 = k - n + 1 + i_hi2
        grad[i_lo2:i_hi2+1] -= h[m_lo2:m_hi2+1]
    
    return grad
```

This is O(n) per lag, O(n*M) for M lags. For n=960, M=10: ~10000 ops. Very fast.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    while s > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
        s -= 1
    while s < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
        s += 1
    return b

        v4 = (idx4 >= 0) & (idx4 < n)
        delta[v4] -= b[idx4[v4]]

        new_max = float(np.max(new_seg))

            lm = float(np.max(corr[:k_lo]))
            if lm > new_max:

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    if new_n % old_n == 0:
        factor = new_n // old_n
        h_new = np.repeat(h, factor)
    else:
        old_x = np.linspace(0.0, 2.0, old_n)
        new_x = np.linspace(0.0, 2.0, new_n)
        h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

        M = min(20, n2)

        grad = np.zeros(n)

            # d(corr[k])/d(h[i]) = g[i+n-1-k] - h[k-n+1+i] (with 0 for OOB)
