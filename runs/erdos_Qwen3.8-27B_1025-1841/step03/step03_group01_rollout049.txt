import numpy as np
from scipy.optimize import minimize
import time

_DEF_N = 0
_DEF_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr * dx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _obj_fn(h):
    n = _DEF_N
    T = _DEF_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _DEF_N / 2.0

def _cons_jac(h):
    return np.ones(_DEF_N)

def _slsqp_optimize(h_start, n, T, max_iter):
    global _DEF_N, _DEF_T
    _DEF_N = n
    _DEF_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj_fn, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def _lag_gradient(h, s, n):
    """Gradient of corr at lag s (i.e., corr[n-1+s]) wrt h."""
    g = np.zeros(n)
    if s >= 0:
        L = n - s
        g[s:] += (1.0 - h[:L])
        g[:L] += -h[s:]
    else:
        t = -s
        L = n - t
        g[:L] += (1.0 - h[t:])
        g[t:] += -h[:L]
    return g

def _active_set_gradient(h, n, tol_frac=0.02):
    """Compute gradient of max over active lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * (2.0 / n)
    mc = np.max(vals)
    active_mask = vals >= (mc - tol_frac)
    if not np.any(active_mask):
        active_mask = np.zeros(len(corr), dtype=bool)
        active_mask[int(np.argmax(vals))] = True
    grad = np.zeros(n)
    idxs = np.where(active_mask)[0]
    for idx in idxs:
        s = idx - (n - 1)
        grad += _lag_gradient(h, s, n)
    grad /= len(idxs)
    return grad, mc, active_mask

def projected_gradient_descent(h, n, n_iters=2000, tol=1e-10, time_budget=200):
    """Projected gradient on the true max with active-set gradient."""
    start_t = time.time()
    dx = 2.0 / n
    lr = 0.5 / (n * dx)
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    prev_c5 = best_c5

    for it in range(n_iters):
        if time.time() - start_t > time_budget:
            break
        grad, mc, active_mask = _active_set_gradient(h, n)
        
        # Backtracking line search
        for shrink in range(30):
            step = lr * (0.5 ** shrink)
            h_new = h - step * grad
            h_new = make_feasible(h_new, n)
            c_new, _ = compute_c5(h_new)
            if c_new < prev_c5 - 1e-15:
                h = h_new
                prev_c5 = c_new
                if c_new < best_c5:
                    best_c5 = c_new
                    best_h = h.copy()
                break
        else:
            # Stagnation: try smaller active set
            break
        if abs(prev_c5 - mc * dx) < tol:
            pass
    return best_h, best_c5

def binary_coord_descent(h, n, max_rounds=3, time_budget=120):
    """Try setting individual coordinates to 0 or 1, compensating at another point."""
    start_t = time.time()
    best_c5, _ = compute_c5(h)
    best_h = h.copy()

    for rnd in range(max_rounds):
        if time.time() - start_t > time_budget:
            break
        improved = False
        for i in range(n):
            if time.time() - start_t > time_budget:
                break
            for target in (0.0, 1.0):
                if abs(h[i] - target) < 1e-6:
                    continue
                delta = target - h[i]
                # Find j to compensate
                if delta > 0:
                    mask = h > 1e-6
                else:
                    mask = h < 1.0 - 1e-6
                mask[i] = False
                jnd = np.where(mask)[0]
                if len(jnd) == 0:
                    continue
                # Spread compensation over a few neighbors for smoothness
                neighbors = np.array([max(0, i-1), i+1 if i+1 < n else i-1])
                for j in neighbors:
                    if j == i or not mask[j]:
                        continue
                    trial = h.copy()
                    trial[i] = target
                    trial[j] = h[j] - delta
                    if trial[j] < 0 or trial[j] > 1:
                        continue
                    c, _ = compute_c5(trial)
                    if c < best_c5 - 1e-13:
                        best_h = trial
                        best_c5 = c
                        h = trial.copy()
                        improved = True
                        break
        if not improved:
            break
    return best_h, best_c5

def swap_coord_descent(h, n, max_rounds=4, time_budget=150):
    """Try small mass transfers between pairs while preserving sum exactly."""
    start_t = time.time()
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    eps_values = [0.003, 0.007, 0.015, 0.03]

    for rnd in range(max_rounds):
        if time.time() - start_t > time_budget:
            break
        improved = False
        for i in range(0, n, 1):
            if time.time() - start_t > time_budget:
                break
            for j_off in (3, 5, 7, 11, 17):
                j = (i + j_off) % n
                if j == i:
                    continue
                for eps in eps_values:
                    for sign in (1, -1):
                        d = sign * eps
                        h_i_new = h[i] + d
                        h_j_new = h[j] - d
                        if h_i_new < 0 or h_i_new > 1 or h_j_new < 0 or h_j_new > 1:
                            continue
                        trial = h.copy()
                        trial[i] = h_i_new
                        trial[j] = h_j_new
                        c, _ = compute_c5(trial)
                        if c < best_c5 - 1e-13:
                            best_h = trial
                            best_c5 = c
                            h = trial.copy()
                            improved = True
                            break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _DEF_N, _DEF_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _time_left():
        return budget_s - (time.time() - start_time)

    # Initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 160)

    best_c5, _ = compute_c5(h0)
    best_h = h0.copy()

    # Up
    # Upsample to working resolution
    n = 2560
    h_curr = upsample_h(best_h, n)
    c_curr, _ = compute_c5(h_curr)
    if c_curr < best_c5:
        best_c5 = c_curr
        best_h = h_curr.copy()
    h_curr = best_h.copy()

    # Phase 1: SLSQP with soft-max at increasing temperatures
    temp_schedule = [(75.0, 350), (150.0, 300), (350.0, 250), (800.0, 180), (2000.0, 120)]
    for T, max_it in temp_schedule:
        if _time_left() < 50:
            break
        try:
            h_opt = _slsqp_optimize(h_curr, n, T, max_it)
            c_opt, _ = compute_c5(h_opt)
            h_curr = h_opt.copy()
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 2: Projected gradient on true max
    if _time_left() > 180:
        pgbudget = min(180, _time_left() - 100)
        h_pgd, c_pgd = projected_gradient_descent(h_curr, n, n_iters=3000, time_budget=pgbudget)
        if c_pgd < best_c5:
            best_c5 = c_pgd
            best_h = h_pgd.copy()
            h_curr = h_pgd.copy()

    # Phase 3: Multiple restarts with SLSQP
    restart_strategies = [
        ('noise5', 0.05),
        ('noise10', 0.10),
        ('blend_bin', None),
        ('shift', None),
    ]
    for name, param in restart_strategies:
        if _time_left() < 80:
            break
        if name == 'noise5':
            h_r = h_curr + param * rng.randn(n)
        elif name == 'noise10':
            h_r = h_curr + param * rng.randn(n)
        elif name == 'blend_bin':
            h_bin = (rng.rand(n) < 0.5).astype(float)
            h_r = 0.6 * best_h + 0.4 * h_bin
        else:
            shift = rng.randint(2, n // 8)
            h_r = np.roll(best_h, shift)
        h_r = make_feasible(h_r, n)
        c_r, _ = compute_c5(h_r)
        # SLSQP from restart
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()
        try:
            h_opt = _slsqp_optimize(h_r, n, 200.0, 200)
            c_opt, _ = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
            h_curr = h_opt.copy()
        except Exception:
            pass

    # Phase 4: Swap coordinate descent
    if _time_left() > 180:
        swbudget = min(150, _time_left() - 60)
        h_sw, c_sw = swap_coord_descent(best_h, n, max_rounds=5, time_budget=swbudget)
        if c_sw < best_c5:
            best_c5 = c_sw
            best_h = h_sw.copy()

    # Phase 5: Binary coordinate descent
    if _time_left() > 100:
        bbudget = min(80, _time_left() - 40)
        h_bin_cd, c_bc = binary_coord_descent(best_h, n, max_rounds=3, time_budget=bbudget)
        if c_bc < best_c5:
            best_c5 = c_bc
            best_h = h_bin_cd.copy()

    # Phase 6: Final SLSQP polish
    if _time_left() > 40:
        try:
            h_opt = _slsqp_optimize(best_h, n, 5000.0, 150)
            c_opt, _ = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 7: Try higher resolution
    if _time_left() > 150:
        n2 = 5120
        h_hi = upsample_h(best_h, n2)
        c_hi, _ = compute_c5(h_hi)
        if c_hi < best_c5:
            best_c5 = c_hi
            best_h = h_hi.copy()
        if _time_left() > 80:
            try:
                h_opt = _slsqp_optimize(h_hi, n2, 1000.0, 120)
                c_opt, _ = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5 = c_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
        # PGD at higher res
        if _time_left() > 60:
            hh, ch = projected_gradient_descent(best_h if len(best_h)==n2 else h_hi, 
                                                  len(best_h) if len(best_h)==n2 else n2,
                                                  n_iters=1500, time_budget=min(50, _time_left()-20))
            if ch < best_c5:
                best_c5 = ch
                best_h = hh.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5, _ = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
