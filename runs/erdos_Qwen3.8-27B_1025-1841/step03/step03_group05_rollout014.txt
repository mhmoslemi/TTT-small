def binary_sa(h, n_steps, T0, T1, seed=42):
    """Simulated annealing on binary sequence with incremental correlation update"""
    n = len(h)
    h = h.astype(float).copy()

        j = zeros[rng.randint(len(zeros))]

        # c[k] corresponds to lag = k - (n-1)
        # For lag k: delta = (1-h[i+k])_v - h[i-k]_v - (1-h[j+k])_v + h[j-k]_v
        # where _v means valid (index in [0,n-1])

        mask1 = (i + lags >= 0) & (i + lags < n)
        term1[mask1] = 1.0 - h[i + lags[mask1]]
        
        # h[i-k] valid when 0 <= i-k < n, i.e., i-n < k <= i
        term2 = np.zeros(n_lags)
        mask2 = (i - lags >= 0) & (i - lags < n)
        term2[mask2] = h[i - lags[mask2]]

        term3 = np.zeros(n_lags)
        mask3 = (j + lags >= 0) & (j + lags < n)
        term3[mask3] = -(1.0 - h[j + lags[mask3]])
        
        # h[j-k] valid when 0 <= j-k < n
        term4 = np.zeros(n_lags)
        mask4 = (j - lags >= 0) & (j - lags < n)
        term4[mask4] = h[j - lags[mask4]]
        
        delta = term1 - term2 + term3 + term4
        
        # But we need to handle the case where i and j interact
        # If i+k = j: h[i+k] = h[j] = 1 (old), so (1-h[i+k]) = 0. Correct in formula.
        # If i-k = j: h[i-k] = h[j] = 1. Correct.
        # If j+k = i: h[j+k] = h[i] = 0 (old), so (1-h[j+k]) = 1. And -(1-h[j+k]) = -1. 
        # But wait, if j+k=i, then the term "m=j" in c_k involves h[j](1-h[i]) = 1*(1-0) = 1 (old)
        # After swap: h[j](1-h[i]) = 0*(1-1) = 0. Change: -1. ✓
        # And the term "m+k=i" (m=i-k) involves h[i-k](1-h[i]). If i-k=j: h[j](1-0)=1 (old), h[j](1-1)=0 (after). Change: -1.
        # But wait, this double-counts! If i-k=j AND j+k=i (same condition k=i-j), then we have:
        #   term from "m=i, h[i+k] where i+k=j": contributes to term1
        #   term from "m=i-k=j, h[i-k]=h[j]": contributes to term2
        #   term from "m=j, h[j+k] where j+k=i": contributes to term3
        #   term from "m=j-k, h[j-k]=h[i]": contributes to term4
        # All four are valid and correct. No double counting.
        
        c_new = c + delta
        max_new = float(np.max(c_new))
        max_old = float(np.max(c))
        delta_max = max_new - max_old
        
        if delta_max < 0 or rng.random() < np.exp(-delta_max / max(T, 1e-10)):
            # Accept
            h[i], h[j] = h[j], h[i]
            c = c_new

            if max_new < best_c:
                best_c = max_new
                best_h = h.copy()
    
    return best_h, best_c * (2.0 / n)

def project_c(h, n):
    target = n / 2.0

def binarize(h, n):

def binary_sa(h_input, n_steps, T0, T1, seed=42, verbose=False):

    h = binarize(h_input, n)
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

    rng = np.random.RandomState(seed)
    delta_buf = np.zeros(n_lags)
    
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        T = T0 * (T1 / T0) ** frac
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta_buf[:] = 0.0
        v1 = (i + lags >= 0) & (i + lags < n)
        delta_buf[v1] += 1.0 - h[i + lags[v1]]
        v2 = (i - lags >= 0) & (i - lags < n)
        delta_buf[v2] -= h[i - lags[v2]]
        v3 = (j + lags >= 0) & (j + lags < n)
        delta_buf[v3] -= 1.0 - h[j + lags[v3]]
        v4 = (j - lags >= 0) & (j - lags < n)
        delta_buf[v4] += h[j - lags[v4]]

        old_max = float(np.max(c))
        d = new_max - old_max

            h[i] = 1.0
            h[j] = 0.0
            c += delta_buf

                best_h = h.copy()
        
        if verbose and step % 100000 == 0:
            print(f"  SA iter {step}: best={best_val * 2.0/n:.6f}", flush=True)

def subgrad_step(h, n, k):
    """Compute gradient of c_k with respect to h, where c_k = sum h[m]*(1-h[m+k])"""
    g = np.zeros(n)
    if k >= 0:
        for m in range(n - k):
            g[m] += 1.0 - h[m + k]
            g[m + k] -= h[m]
    else:
        kk = -k
        for m in range(n - kk):
            g[m + kk] += 1.0 - h[m]
            g[m] -= h[m + kk]
    return g

def fista_exact(h, n_iter, lr_init, decay, verbose=False):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def binarize(h, n):
    idx = np.argsort(h)[::-1]
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def binary_sa(h_input, n_steps, T0, T1, seed=42, verbose=False):
    n = len(h_input)
    h = binarize(h_input, n).astype(np.float64)
    c = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    n_lags = len(c)
    lags = np.arange(n_lags) - (n - 1)
    best_val = float(c.max())
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    dbuf = np.zeros(n_lags)
    for step in range(n_steps):
        frac = step / max(n_steps - 1, 1)
        T = T0 * (T1 / T0) ** frac
        ones = np.flatnonzero(h)
        zeros = np.flatnonzero(1.0 - h)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        dbuf[:] = 0.0
        v1 = (i + lags >= 0) & (i + lags < n)
        dbuf[v1] += 1.0 - h[i + lags[v1]]
        v2 = (i - lags >= 0) & (i - lags < n)
        dbuf[v2] -= h[i - lags[v2]]
        v3 = (j + lags >= 0) & (j + lags < n)
        dbuf[v3] -= 1.0 - h[j + lags[v3]]
        v4 = (j - lags >= 0) & (j - lags < n)
        dbuf[v4] += h[j - lags[v4]]
        new_max = float(np.max(c + dbuf))
        old_max = float(np.max(c))
        d = new_max - old_max
        if d < 0 or (T > 1e-12 and rng.random() < np.exp(-d / T)):
            h[i] = 1.0
            h[j] = 0.0
            c += dbuf
            if new_max < best_val:
                best_val = new_max
                best_h = h.copy()
        if verbose and (step + 1) % 200000 == 0:
            print(f"    SA {step+1}/{n_steps}: best={best_val * 2.0 / n:.6f}", flush=True)
    return best_h, best_val * (2.0 / n)

def subgrad_one(h, n, k):
    g = np.zeros(n)
    if k >= 0:
        m = np.arange(n - k)
        g[m] += 1.0 - h[m + k]
        g[m + k] -= h[m]
    else:
        kk = -k
        m = np.arange(n - kk)
        g[m + kk] += 1.0 - h[m]
        g[m] -= h[m + kk]
    return g

def fista_sm(h_init, n_iter, T0, T1, lr0, top_k=60, check=50):
    n = len(h_init)
    dx = 2.0 / n
    x = h_init.astype(np.float64).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = exps.sum()
        if tot < 1e-300:
            k = int(np.argmax(corr)) - (n - 1)
            g = subgrad_one(z, n, k) * dx
        else:
            w = exps / tot
            top = np.argsort(w)[::-1][:top_k]
            g = np.zeros(n)
            for idx in top:
                if w[idx] < 1e-12:
                    break
                g += w[idx] * subgrad_one(z, n, idx - (n - 1))
            g *= dx
        lrt = lr0 * (1.0 - 0.3 * f)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def fista_exact(h_init, n_iter, lr_init, decay=1.0, check=100):
    n = len(h_init)
    dx = 2.0 / n
    x = h_init.astype(np.float64).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        corr = np.correlate(z, 1.0 - z, mode='full')
        k_max = int(np.argmax(corr)) - (n - 1)
        g = subgrad_one(z, n, k_max) * dx
        lr = lr_init / (1.0 + it * decay / 100.0)
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]

def upsample(h, nn):
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)
