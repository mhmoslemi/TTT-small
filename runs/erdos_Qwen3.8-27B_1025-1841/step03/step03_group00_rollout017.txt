from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 5000})
    return make_feasible(res.x, n)

def binary_sa(h_bin, n, rng, n_iter=20000, T_start=0.5, T_end=0.01):
    """Simulated annealing on binary sequence."""
    h = h_bin.copy()
    current_c = compute_c5(h)
    best_c = current_c
    best_h = h.copy()
    ones = set(np.where(h > 0.5)[0].tolist())
    zeros = set(np.where(h <= 0.5)[0].tolist())
    
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        oi = rng.choice(list(ones))
        zi = rng.choice(list(zeros))
        h[oi] = 0.0
        h[zi] = 1.0
        new_c = compute_c5(h)
        if new_c < current_c:
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            ones.discard(oi); zeros.discard(zi)
            ones.add(zi); zeros.add(oi)
        else:
            if rng.random() < np.exp(-(new_c - current_c) / max(T, 1e-10)):
                ones.discard(oi); zeros.discard(zi)
                ones.add(zi); zeros.add(oi)
            else:
                h[oi] = 1.0
                h[zi] = 0.0
    return best_h, best_c

def block_local_search(h, n, rng, max_rounds=30):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n

    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 8000})
    return make_feasible(res.x, n)

def binary_sa(h_bin, n, rng, n_iter, T_start=0.3, T_end=0.005):
    h = h_bin.copy()
    current_c = compute_c5(h)
    best_c = current_c
    best_h = h.copy()
    ones_list = np.where(h > 0.5)[0].copy()
    zeros_list = np.where(h <= 0.5)[0].copy()
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        ii = rng.randint(len(ones_list))
        jj = rng.randint(len(zeros_list))
        oi = ones_list[ii]
        zi = zeros_list[jj]
        h[oi] = 0.0
        h[zi] = 1.0
        new_c = compute_c5(h)
        if new_c < current_c:
            current_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_h = h.copy()
            ones_list[ii] = zi
            zeros_list[jj] = oi
        else:
            if rng.random() < np.exp(-(new_c - current_c) / max(T, 1e-12)):
                ones_list[ii] = zi
                zeros_list[jj] = oi
            else:
                h[oi] = 1.0
                h[zi] = 0.0
    return best_h, best_c

def cyclic_shift_optimize(h, n, n_shifts=200):
    best_c = compute_c5(h)
    best_h = h.copy()
    step = max(1, n // n_shifts)
    for shift in range(step, n, step):
        h_s = np.roll(h, shift)
        c = compute_c5(h_s)
        if c < best_c:
            best_c = c
            best_h = h_s.copy()
    return best_h, best_c

def directed_local_search(h, n, rng, max_rounds=50):
    current = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    for rnd in range(max_rounds):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        thresh = max_val * 0.9
        active = np.where(corr >= thresh)[0]
        for _ in range(800):
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            delta = rng.choice([0.01, 0.02, 0.005, -0.01, -0.02, -0.005])
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0.0, 1.0)
            h_new[j] = np.clip(h_new[j] - delta, 0.0, 1.0)
            if abs(np.sum(h_new) - n / 2.0) > 1e-10:
                continue
            c = compute_c5(h_new)
            if c < current - 1e-15:
                current = c
                h = h_new
                improved = True
        if not improved:
            break
    return h, current

    pat = np.zeros(period)

    pat[:n_ones] = 1.0
    perm = rng.permutation(period)
    pat = pat[perm]
    rep = np.tile(pat, n // period + 1)[:n]
    return make_feasible(rep, n)
