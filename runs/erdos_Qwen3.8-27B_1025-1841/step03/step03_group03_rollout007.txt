```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, np.concatenate([h, [h[0]]]))
    return make_feasible(h_new, new_n)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_at_shift(h, m):
    """Gradient of corr at shift m (where m = k - (n-1)) w.r.t. h"""
    n = len(h)
    grad = np.zeros(n)
    # First term: (1-h[j-m]) for j where 0<=j-m<n, i.e., j in [max(0,m), min(n, n+m))
    j_lo1 = max(0, m)
    j_hi1 = min(n, n + m)
    if j_hi1 > j_lo1:
        grad[j_lo1:j_hi1] += (1.0 - h[j_lo1 - m : j_hi1 - m])
    # Second term: -h[j+m] for j where 0<=j+m<n, i.e., j in [max(0,-m), min(n, n-m))
    j_lo2 = max(0, -m)
    j_hi2 = min(n, n - m)
    if j_hi2 > j_lo2:
        grad[j_lo2:j_hi2] -= h[j_lo2 + m : j_hi2 + m]
    return grad

def projected_subgrad(h, n, max_iter, time_limit, start_time, lr_init=0.01, top_k=1):
    """Projected subgradient method on the max correlation"""
    best_c = compute_c5(h)
    h_best = h.copy()
    lr = lr_init
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        corr = corr_full(h)
        # Find top_k correlations to use (for smoothing)
        top_indices = np.argsort(corr)[-top_k:]
        m_star = top_indices[-1] - (n - 1)  # actual shift
        
        # Compute gradient at the maximizing shift
        grad = grad_at_shift(h, m_star)
        
        # For top_k > 1, average gradients
        if top_k > 1:
            for mk in top_indices[:-1]:
                grad += grad_at_shift(h, mk - (n-1))
            grad /= top_k
        
        # Gradient step (we're minimizing, so go opposite to gradient)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            h_best = h_new.copy()
            # Increase lr slightly
            lr = min(lr * 1.01, 0.05)
        else:
            h = h_new  # still use the new point
            lr = max(lr * 0.95, 0.001)
        h = h_new
    
    return h_best, best_c
```

```python
def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)

        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

    return lse * dx
```

```python
    try:
        res = minimize(obj_fn, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        hx = res.x if np.all(np.isfinite(res.x)) else h_start
        return make_feasible(hx, n)
    except Exception:
        return make_feasible(h_start, n)

def subgrad_refine(h, n, max_iter, tlimit, tstart, topk=5, lr0=0.006):
    best_c = compute_c5(h)
    h_b = h.copy()
    lr = lr0
    nc = 2 * n - 1
    for it in range(max_iter):
        if time.time() - tstart > tlimit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        if topk >= nc:
            tidx = np.arange(nc)
        else:
            tidx = np.argpartition(corr, -topk)[-topk:]
        grad = np.zeros(n)
        for ki in tidx:
            m = int(ki) - (n - 1)
            jlo1 = max(0, m)
            jhi1 = min(n, n + m)
            if jhi1 > jlo1:
                grad[jlo1:jhi1] += (1.0 - h[jlo1 - m:jhi1 - m])
            jlo2 = max(0, -m)
            jhi2 = min(n, n - m)
            if jhi2 > jlo2:
                grad[jlo2:jhi2] -= h[jlo2 + m:jhi2 + m]
        grad /= len(tidx)
        h2 = np.clip(h - lr * grad, 0.0, 1.0)
        h2 = make_feasible(h2, n)
        c2 = compute_c5(h2)
        if c2 < best_c - 1e-15:
            best_c = c2
            h_b = h2.copy()
            lr = min(lr * 1.003, 0.035)
        else:
            lr = max(lr * 0.965, 0.0008)
        h = h2
    return h_b, best_c

def coord_cd(h, n, rounds, tlimit, tstart):
    best_c = compute_c5(h)
    for r in range(rounds):
        if time.time() - tstart > tlimit - 15:
            break
        imp = False
        for j in range(n):
            if time.time() - tstart > tlimit - 15:
                return h, best_c
            old = h[j]
            cands = [0.0, 0.25, 0.5, 0.75, 1.0,
                     old - 0.04, old + 0.04,
                     old - 0.015, old + 0.015,
                     old - 0.005, old + 0.005]
            bv = old
            bo = best_c
            for cv in cands:
                cv = float(np.clip(cv, 0.0, 1.0))
                tr = h.copy()
                tr[j] = cv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < bo - 1e-14:
                    bo = c
                    bv = cv
            if bv != old:
                h[j] = bv
                h = make_feasible(h, n)
                best_c = bo
                imp = True
        if not imp:
            break
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    t_end = t0 + budget_s
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        nn = 640
        h0 = (rng.rand(nn) < 0.5).astype(float)
        perms = rng.permutation(nn)
        ones = int(np.sum(h0))
        if ones > nn // 2:
            for idx in perms[nn//2:]:
                if h0[idx] == 1.0:
                    h0[idx] = 0.0
                    ones -= 1
                    if ones == nn // 2:
                        break
        elif ones < nn // 2:
            for idx in perms:
                if h0[idx] == 0.0:
                    h0[idx] = 1.0
                    ones += 1
                    if ones == nn // 2:
                        break
        h0 = make_feasible(h0, nn)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    n0 = len(best_h)

    # Phase 1: Subgradient at n0
    rem = t_end - time.time() - 850
    if rem > 30:
        h_s, c_s = subgrad_refine(best_h, n0, 5000, rem, t0, topk=5, lr0=0.006)
        if c_s < best_c5:
            best_c5 = c_s
            best_h = h_s

    # Phase 2: SLSQP sweep at n0
    for t, mi in [(150, 350), (400, 350), (1000, 400), (3000, 400)]:
        rem = t_end - time.time() - 600
        if rem < 30:
            break
        h_new = slsqp_opt(best_h, n0, t, min(mi, max(50, int(rem * 0.8))))
        c_new = compute_c5(h_new)
        if c_new < best_c5 - 1e-13:
            best_c5 = c_new
            best_h = h_new

    # Phase 3: More subgradient
    rem = t_end - time.time() - 500
    if rem > 30:
        h_s2, c_s2 = subgrad_refine(best_h, n0, 4000, rem, t0, topk=3, lr0=0.004)
        if c_s2 < best_c5:
            best_c5 = c_s2
            best_h = h_s2

    # Phase 4: Upsample to 1024
    n_up = 1024
    rem = t_end - time.time() - 350
    if rem > 40:
        h_up = upsample_h(best_h, n_up)
        c_up = compute_c5(h_up)
        h4 = h_up.copy()
        c4 = c_up
        # Subgradient at 1024
        rem2 = t_end - time.time() - 200
        if rem2 > 30:
            h4, c4 = subgrad_refine(h4, n_up, 3500, rem2, t0, topk=5, lr0=0.005)
        # SLSQP at 1024
        for t, mi in [(300, 300), (1000, 300)]:
            rem3 = t_end - time.time() - 150
            if rem3 < 30:
                break
            h_new = slsqp_opt(h4, n_up, t, mi)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-13:
                best_c5 = c_new
                best_h = h_new
            else:
                h4 = h_new
                c4 = c_new
        if c4 < best_c5 - 1e-13:
            best_c5 = c4
            best_h = h4
        elif compute_c5(h4) < best_c5:
            best_c5 = compute_c5(h4)
            best_h = h4

    # Phase 5: Try 2048
    rem = t_end - time.time() - 180
    if rem > 50:
        n_up2 = 2048
        h_up2 = upsample_h(best_h, n_up2)
        h5, c5 = subgrad_refine(h_up2, n_up2, 2500, rem, t0, topk=5, lr0=0.004)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h5

    # Phase 6: Coordinate descent
    rem = t_end - time.time() - 50
    if rem > 30:
        h_cd, c_cd = coord_cd(best_h, len(best_h), 6, t_end - 30, t0)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 7: Binary projection attempt
    rem = t_end - time.time() - 40
    if rem > 20:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, len(h_bin))
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5 - 1e-13:
            best_h = h_bin
            best_c5 = c_bin
            h_cd2, c_cd2 = coord_cd(best_h, len(best_h), 4, t_end - 20, t0)
            if c_cd2 < best_c5:
                best_c5 = c_cd2
                best_h = h_cd2

    # Phase 8: Random restart
    rem = t_end - time.time() - 20
    if rem > 40:
        h_r = rng.rand(len(best_h))
        h_r = make_feasible(h_r, len(h_r))
        h_r, c_r = subgrad_refine(h_r, len(h_r), 2000, rem, t0, topk=5, lr0=0.006)
        if c_r < best_c5 - 1e-13:
            best_c5 = c_r
            best_h = h_r

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
