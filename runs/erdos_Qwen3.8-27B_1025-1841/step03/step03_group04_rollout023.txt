    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    # Sigmoid/softmax weights
    weights = np.exp(logits - np.max(logits))
    weights /= np.sum(weights)
    
    # Gradient: sum_k weights[k] * dcross_k/dh
    # dcross_k/dh[j] = (1-h[j+L])*[valid] - h[j-L]*[valid]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def softmax_obj(h, T, top_idx=None):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_idx is not None:
        corr = corr[top_idx]
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def softmax_gradient(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        top_idx = np.argsort(corr)[-top_k:]
        corr_sub = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        corr_sub = corr
    mc = np.max(corr_sub)
    logits = T * (corr_sub - mc)
    logits -= np.max(logits)
    weights = np.exp(logits)
    weights /= np.sum(weights)
    grad_h = np.zeros(n)
    for i in range(len(top_idx)):
        if weights[i] < 1e-12:
            continue
        k = top_idx[i]
        L = k - (n - 1)
        w = weights[i]
        if L == 0:
            grad_h += w * (1.0 - 2.0 * h)
        elif L > 0:
            grad_h[:n-L] += w * (1.0 - h[L:])
            grad_h[L:] += w * (-h[:n-L])
        else:
            La = -L
            grad_h[La:] += w * (1.0 - h[:n-La])
            grad_h[:n-La] += w * (-h[La:])
    return dx * grad_h

    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def gradient_descent(h, T, n_iter, lr_init, top_k, budget_check=None):
    n = len(h)
    lr = lr_init
    momentum = 0.9
    velocity = np.zeros(n)
    best_local = compute_c5(h)
    h_work = h.copy()
    for it in range(n_iter):
        if budget_check is not None and budget_check() < 5:
            break
        grad = softmax_gradient(h_work, T, top_k)
        velocity = momentum * velocity - lr * grad
        h_new = h_work + velocity
        h_new = make_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_local:
            h_work = h_new
            best_local = c5
        else:
            lr *= 0.7
            velocity *= 0.5
            if lr < 1e-8:
                break
    return h_work, best_local

def coord_descent_fine(h, steps, budget_check=None):

        def cfun(x):
            return np.sum(x) - target
        def cjac(x):
            return np.ones(n_pts)
        cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
        try:
            res = minimize(obj_top, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-15})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 6: Paired moves (random)
    if remaining() > 30:
        n_pts = len(best_h)
        h_work = best_h.copy()
        c_work = best_c5
        n_pairs = min(5000, int(remaining() * 50))
        for _ in range(n_pairs):
            if remaining() < 5:
                break
            i = rng.randint(n_pts)
            j = rng.randint(n_pts)
            if i == j:
                continue
            for delta in [0.001, 0.003, 0.005, 0.01]:
                v1 = h_work[i] + delta
                v2 = h_work[j] - delta
                if 0.0 <= v1 <= 1.0 and 0.0 <= v2 <= 1.0:
                    trial = h_work.copy()
                    trial[i] = v1
                    trial[j] = v2
                    c = compute_c5(trial)
                    if c < c_work - 1e-15:
                        c_work = c
                        h_work = trial
                        break
                v1 = h_work[i] - delta
                v2 = h_work[j] + delta
                if 0.0 <= v1 <= 1.0 and 0.0 <= v2 <= 1.0:
                    trial = h_work.copy()
                    trial[i] = v1
                    trial[j] = v2
                    c = compute_c5(trial)
                    if c < c_work - 1e-15:
                        c_work = c
                        h_work = trial
                        break
        if c_work < best_c5:
            best_c5 = c_work
            best_h = h_work

    # Phase 7: Final ultrafine coordinate descent
    if remaining() > 20:
        best_h, best_c5 = coord_descent_fine(best_h, [0.002, 0.001, 0.0005, 0.0003, 0.0001], remaining)

    # Phase 8: Final SLSQP polish
    if remaining() > 15:
        best_h, c = slsqp_pass(best_h, 500, 100, remaining)
        if c < best_c5:
            best_c5 = c

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
