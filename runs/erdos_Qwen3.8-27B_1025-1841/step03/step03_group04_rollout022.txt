        best_h, best_c5 = paired_coord_descent(best_h, len(best_h), best_c5,
                                               time_limit=min(remaining() - 80, 300))

    # Phase 3: SLSQP with high T
    for T in [3000, 6000]:
        if remaining() < 80:
            break
        h_new = slsqp_run(best_h, T, len(best_h), maxiter=1500, ftol=1e-16)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(h_new, len(best_h))

    # Phase 4: Fine paired coordinate descent
    if remaining() > 80:
        best_h, best_c5 = paired_coord_descent(best_h, len(best_h), best_c5,
                                               time_limit=min(remaining() - 60, 200))

    # Phase 5: Another SLSQP round
    if remaining() > 80:
        h_new = slsqp_run(best_h, 8000, len(best_h), maxiter=1500, ftol=1e-16)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(h_new, len(best_h))

    # Phase 6: Upsample to 2n and optimize
    if remaining() > 350:
        n_up = len(best_h) * 2
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_up)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = make_feasible(h_up, n_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.001:
            for T in [500, 2000, 5000]:
                if remaining() < 100:
                    break
                h_new = slsqp_run(h_up, T, n_up, maxiter=1500, ftol=1e-15)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = make_feasible(h_new, n_up)
            if remaining() > 100:
                best_h, best_c5 = paired_coord_descent(best_h, len(best_h), best_c5,
                                                       time_limit=min(remaining() - 80, 200))

    # Phase 7: Random restarts at current best resolution
    for restart in range(10):
        if remaining() < 80:
            break
        n_cur = len(best_h)
        scale = 0.003 + 0.003 * restart
        h_try = make_feasible(best_h + rng.randn(n_cur) * scale, n_cur)
        h_new = slsqp_run(h_try, 2000, n_cur, maxiter=1500, ftol=1e-15)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(h_new, n_cur)
            # Quick coord descent after improvement
            if remaining() > 60:
                best_h, best_c5 = paired_coord_descent(best_h, len(best_h), best_c5,
                                                       time_limit=min(remaining() - 50, 60))

    # Phase 8: Ultra-fine paired coordinate descent
    if remaining() > 40:
        best_h, best_c5 = paired_coord_descent(best_h, len(best_h), best_c5,
                                               time_limit=min(remaining() - 20, 120))

    # Phase 9: Final SLSQP with very high T
    if remaining() > 60:
        h_new = slsqp_run(best_h, 10000, len(best_h), maxiter=1000, ftol=1e-16)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(h_new, len(best_h))

    # Final cleanup
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
