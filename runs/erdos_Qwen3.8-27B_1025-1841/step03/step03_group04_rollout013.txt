def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def _smooth_obj(x, T, top_idx):
    dx = 2.0 / len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = corr[top_idx] if top_idx is not None else corr
    mc = np.max(c)
    s = c - mc
    s -= np.max(s)
    return (mc + np.log(np.sum(np.exp(T * s))) / T) * dx

def _smooth_grad(x, T, top_idx):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_idx is not None:
        c = corr[top_idx]
        lags = top_idx - (n - 1)
    else:
        c = corr
        lags = np.arange(2 * n - 1) - (n - 1)
    mc = np.max(c)
    ex = T * (c - mc)
    ex -= np.max(ex)
    w = np.exp(ex)
    sw = np.sum(w)
    w /= sw
    grad = np.zeros(n)
    for idx in range(len(lags)):
        wi = w[idx]
        if wi < 1e-14:
            continue
        lag = lags[idx]
        if lag == 0:
            grad += wi * (1.0 - 2.0 * x)
        elif lag > 0:
            grad[:n - lag] += wi * (1.0 - x[lag:n])
            grad[lag:] -= wi * x[:n - lag]
        else:
            L = -lag
            grad[L:] += wi * (1.0 - x[:n - L])
            grad[:n - L] -= wi * x[L:n]
    grad *= dx
    return grad

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def _top_shifts(corr, num=12, tol=0.01):
    mc = np.max(corr)
    thresh = mc - tol * mc
    idx = np.where(corr >= thresh)[0]
    if len(idx) > num:
        order = np.argsort(-corr[idx])
        idx = idx[order[:num]]
    elif len(idx) == 0:
        idx = np.array([int(np.argmax(corr))])
    return idx

def _slsqp_smooth(h, n, T, top_idx, bounds, cons, maxiter=300):
    try:
        res = minimize(lambda x: _smooth_obj(x, T, top_idx), h,
                       jac=lambda x: _smooth_grad(x, T, top_idx),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        c5 = compute_c5(res.x)
        return res.x, c5
    except Exception:
        return None, None

def _coord_descent(h, n, best_c5, steps, n_passes=3):
    for p in range(n_passes):
        improved = False
        for j in range(n):
            old = h[j]
            best_v = old
            best_c = best_c5
            for s in steps:
                for v in [old - s, old + s, 0.0, 1.0]:
                    if v < 0 or v > 1 or abs(v - old) < 1e-13:
                        continue
                    trial = h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c:
                        best_c = c
                        best_v = v
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def _binary_swap_search(h, n, best_c5, n_trials=2000):
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, best_c5
    for _ in range(n_trials):
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        trial = h.copy()
        trial[i], trial[j] = trial[j], trial[i]
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c5:
            best_c5 = c
            h = trial
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)
    n = len(h)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

    # Phase 1: SLSQP with analytical gradient, progressive T on top shifts
    for T in [500, 1000, 2000, 3000, 5000]:
        if tl() < 80:
            break
        ti = _top_shifts(get_corr(best_h), 15, 0.02)
        x_new, c5_new = _slsqp_smooth(best_h, n, T, ti, bounds, cons, 350)
        if x_new is not None and c5_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(x_new, n)

    # Phase 2: Full-correlation SLSQP
    for T in [1000, 2000]:
        if tl() < 80:
            break
        x_new, c5_new = _slsqp_smooth(best_h, n, T, None, bounds, cons, 300)
        if x_new is not None and c5_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(x_new, n)

    # Phase 3: Re-adapt top shifts and refine
    for round_i in range(4):
        if tl() < 60:
            break
        ti = _top_shifts(get_corr(best_h), 10, 0.005)
        T = 3000 + round_i * 1000
        x_new, c5_new = _slsqp_smooth(best_h, n, T, ti, bounds, cons, 250)
        if x_new is not None and c5_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(x_new, n)

    # Phase 4: Coordinate descent (medium steps)
    if tl() > 60:
        best_h, best_c5 = _coord_descent(best_h, n, best_c5,
                                         [0.002, 0.005, 0.01, 0.02, 0.05], 4)

    # Phase 5: Subgradient min-max with adaptive step
    if tl() > 120:
        for it in range(800):
            if tl() < 30:
                break
            corr = get_corr(best_h)
            mc = np.max(corr)
            thresh = mc - 0.003 * mc
            active = np.where(corr >= thresh)[0]
            lags = active - (n - 1)
            avg_grad = np.zeros(n)
            for lag in lags:
                if lag == 0:
                    avg_grad += (1.0 - 2.0 * best_h)
                elif lag > 0:
                    avg_grad[:n - lag] += (1.0 - best_h[lag:n])
                    avg_grad[lag:] -= best_h[:n - lag]
                else:
                    L = -lag
                    avg_grad[L:] += (1.0 - best_h[:n - L])
                    avg_grad[:n - L] -= best_h[L:n]
            avg_grad /= len(lags)
            step = 0.003 / (1.0 + 0.02 * it)
            trial = make_feasible(best_h - step * avg_grad, n)
            c = compute_c5(trial)
            if c < best_c5:
                best_c5 = c
                best_h = trial

    # Phase 6: Fine coordinate descent
    if tl() > 40:
        best_h, best_c5 = _coord_descent(best_h, n, best_c5,
                                         [0.0005, 0.001, 0.002, 0.005], 5)

    # Phase 7: Binary swap search
    if tl() > 60:
        best_h, best_c5 = _binary_swap_search(best_h, n, best_c5, 3000)

    # Phase 8: Final SLSQP polish
    if tl() > 50:
        ti = _top_shifts(get_corr(best_h), 8, 0.003)
        x_new, c5_new = _slsqp_smooth(best_h, n, 5000, ti, bounds, cons, 200)
        if x_new is not None and c5_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(x_new, n)

    # Phase 9: Ultra-fine coordinate descent
    if tl() > 20:
        best_h, best_c5 = _coord_descent(best_h, n, best_c5,
                                         [0.0001, 0.0003, 0.001], 3)

    # Phase 10: Binary snap
    if tl() > 15:
        for j in range(n):
            if tl() < 5:
                break
            for v in [0.0, 1.0]:
                if abs(best_h[j] - v) > 1e-5:
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5:
                        best_c5 = c
                        best_h = trial

    # Phase 11: One more SLSQP with very high T
    if tl() > 40:
        ti = _top_shifts(get_corr(best_h), 5, 0.002)
        x_new, c5_new = _slsqp_smooth(best_h, n, 8000, ti, bounds, cons, 150)
        if x_new is not None and c5_new is not None and c5_new < best_c5:
            best_c5 = c5_new
            best_h = make_feasible(x_new, n)

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
