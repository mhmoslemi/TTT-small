            for k in range(k_lo2, k_hi2 + 1):
                i2 = j + k - n + 1
                if 0 <= i2 < n:
                    a[k] += -h[i2]
            
            b = corr - v_old * a
            
            # Ternary search for min of max_k(a[k]*t + b[k]) over t in [0,1]
            if np.max(np.abs(a)) < 1e-15:
                continue
            
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a * m1 + b)
                f2 = np.max(a * m2 + b)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
                if hi - lo < 1e-10:
                    break
            
            t_opt = (lo + hi) / 2.0
            
            if abs(t_opt - v_old) > 1e-8:
                h[j] = t_opt
                h = make_feasible(h, n)
                c_new = compute_c5(h)
                if c_new < best_c:
                    best_c = c_new
                    improved = True
        
        if not improved:
            break
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def get_correlation(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_descent_exact(h, n, passes=1):
    """Coordinate descent with ternary search on piecewise-linear C5(t)."""
    best_c = compute_c5(h)
    for _pass in range(passes):
        improved_any = False
        for j in range(n):
            v_old = h[j]
            corr = get_correlation(h)
            a = np.zeros(2 * n - 1)
            k1_lo = max(0, j - n + 1)
            k1_hi = min(2 * n - 2, j)
            if k1_lo <= k1_hi:
                idx1 = np.arange(k1_lo, k1_hi + 1)
                i1 = j - idx1 + n - 1
                valid1 = (i1 >= 0) & (i1 < n)
                a[idx1[valid1]] += (1.0 - h[i1[valid1]])
            k2_lo = max(0, n - 1 - j)
            k2_hi = min(2 * n - 2, n - 1 + j)
            if k2_lo <= k2_hi:
                idx2 = np.arange(k2_lo, k2_hi + 1)
                i2 = j + idx2 - n + 1
                valid2 = (i2 >= 0) & (i2 < n)
                a[idx2[valid2]] += -h[i2[valid2]]
            if np.max(np.abs(a)) < 1e-14:
                continue
            b = corr - v_old * a
            lo, hi = 0.0, 1.0

                    improved_any = True
        if not improved_any:
            break
    return h, best_c

def slsqp_optimize(h, n, T=300.0, top_idx=None, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_idx is not None:
            corr = corr[top_idx]
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_idx is not None:
            full_corr = corr.copy()
            corr_active = corr[top_idx]
        else:
            full_corr = corr
            corr_active = corr
        mc = np.max(corr_active)
        weights = np.exp(T * (corr_active - mc))
        weights /= np.sum(weights)
        g_full = np.zeros(2 * n - 1)
        if top_idx is not None:
            g_full[top_idx] = weights
        else:
            g_full = weights
        corr_full_1mh = np.correlate(x, 1.0 - x, mode='full')
        # gradient of corr[k] w.r.t. x[i]:
        # x[i] appears in corr[k] as x[i]*(1-x[i+k-(n-1)]) or x[i+k-(n-1)]*(1-x[i])
        grad = np.zeros(n)
        for k in range(len(g_full)):
            if abs(g_full[k]) < 1e-15:
                continue
            w = g_full[k]
            # x[i] as first factor: i goes from 0 to n-1, paired with 1-x[i+k-(n-1)]
            k_off = k - (n - 1)
            for i in range(max(0, -k_off), min(n, n - k_off)):
                other = i + k_off
                if 0 <= other < n:
                    grad[i] += w * (1.0 - x[other])
            # x[i] as second factor: 1-x[i], paired with x[i+k_off]
            for i in range(max(0, -k_off), min(n, n - k_off)):
                other = i + k_off
                if 0 <= other < n:
                    grad[i] += w * (-x[other])
        return grad * dx
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or hasattr(res, 'x'):
            h_res = make_feasible(res.x, n)
            return h_res, compute_c5(h_res)
    except Exception:
        pass
    return h, compute_c5(h)

def binary_local_search(h, n, max_swaps=50000):
    """Local search on binary patterns via swap moves."""
    best_c = compute_c5(h)
    h_bin = np.where(h > 0.5, 1.0, 0.0)

        # Force exact count

        if n_ones > n / 2:
            one_idx = np.where(h_bin == 1.0)[0]
            flip_idx = one_idx[np.random.choice(len(one_idx), n_ones - n // 2, replace=False)]
            h_bin[flip_idx] = 0.0
        elif n_ones < n / 2:
            zero_idx = np.where(h_bin == 0.0)[0]
            flip_idx = zero_idx[np.random.choice(len(zero_idx), n // 2 - n_ones, replace=False)]
            h_bin[flip_idx] = 1.0
    best_bin = h_bin.copy()
    for _ in range(max_swaps):
        i = np.random.randint(n)
        j = np.random.randint(n)

        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    best_bin = make_feasible(best_bin, n)
