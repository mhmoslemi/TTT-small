                max_c = float(np.max(corr))
                ones = np.where(h_sw == 1.0)[0]
                zeros = np.where(h_sw == 0.0)[0]
                no_improve = 0
                c5_new = max_c * (2.0 / n0)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_sw.copy()
            else:
                no_improve += 1
                if no_improve >= 3:
                    break

    # Simulated annealing on binary
    if remaining() > 100 and np.allclose(best_h, np.round(best_h), atol=1e-6):
        h_sa = best_h.copy()
        corr_sa = np.correlate(h_sa, 1.0 - h_sa, mode='full').copy()
        max_sa = float(np.max(corr_sa))
        ones = np.where(h_sa == 1.0)[0]
        zeros = np.where(h_sa == 0.0)[0]
        T = max(2.0, max_sa * 0.02)
        sa_start = elapsed()
        sa_budget = min(180.0, remaining() - 60)

        while elapsed() - sa_start < sa_budget and remaining() > 30:
            a = int(rng.choice(ones))
            b = int(rng.choice(zeros))
            dc = swap_delta(h_sa, a, b, n0)
            new_corr = corr_sa + dc
            new_max = float(np.max(new_corr))
            delta = new_max - max_sa
            if delta < 0 or rng.rand() < np.exp(-delta / T):
                h_sa[a] = 0.0
                h_sa[b] = 1.0
                corr_sa = new_corr
                max_sa = new_max
                ones = np.where(h_sa == 1.0)[0]
                zeros = np.where(h_sa == 0.0)[0]
                c5_new = max_sa * (2.0 / n0)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_sa.copy()
            T = max(0.3, T * 0.999)

    # Phase 2: SLSQP refinement (non-binary)
    if remaining() > 120:
        _G_N = n0
        _G_T = 500.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        for T_v in [300.0, 800.0, 1500.0]:
            if remaining() < 60:
                break
            _G_T = T_v
            try:
                res = minimize(_obj_lse, best_h, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 400, 'ftol': 1e-16})
                c5_r = compute_c5(res.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res.x, n0)
            except Exception:
                pass

    # Phase 3: Shift-targeted SLSQP
    if remaining() > 80:
        _G_N = n0
        _G_T = 600.0
        corr_t = np.correlate(best_h, 1.0 - best_h, mode='full')
        mc_t = np.max(corr_t)
        thr = mc_t - 0.015 * mc_t
        top_idx = np.where(corr_t >= thr)[0]
        if len(top_idx) > 25:
            top_idx = top_idx[np.argsort(-corr_t[top_idx])[:25]]
        elif len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(corr_t))])
        _G_TOP_IDX = top_idx
        bounds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        for rnd in range(3):
            if remaining() < 40:
                break
            try:
                res = minimize(_obj_top, best_h, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 250, 'ftol': 1e-16})
                c5_r = compute_c5(res.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res.x, n0)
            except Exception:
                pass

    # Phase 4: Coordinate descent
    if remaining() > 80:
        n_pts = len(best_h)
        for cd_pass in range(4):
            if remaining() < 20:
                break
            for j in range(n_pts):
                if remaining() < 8:
                    break
                ov = best_h[j]
                bv = ov
                bc = best_c5
                for st in [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]:
                    for sg in [-1.0, 1.0]:
                        v = ov + sg * st
                        if 0.0 <= v <= 1.0:
                            tr = best_h.copy()
                            tr[j] = v
                            tr = make_feasible(tr, n_pts)
                            c = compute_c5(tr)
                            if c < bc - 1e-14:
                                bc = c
                                bv = v
                if bv != ov:
                    best_h[j] = bv
                    best_h = make_feasible(best_h, n_pts)
                    best_c5 = bc

    # Phase 5: Try binary snap on current best
    n_pts = len(best_h)
    h_b2 = binary_snap(best_h, n_pts)
    c5_b2 = compute_c5(h_b2)
    if c5_b2 < best_c5:
        best_c5 = c5_b2
        best_h = h_b2.copy()

    # Phase 6: Second round binary local search on best
    if np.allclose(best_h, np.round(best_h), atol=1e-6) and remaining() > 60:
        h_f = best_h.copy()
        corr_f = np.correlate(h_f, 1.0 - h_f, mode='full').copy()
        max_f = float(np.max(corr_f))
        ones = np.where(h_f == 1.0)[0]
        zeros = np.where(h_f == 0.0)[0]
        for _ in range(200):
            if remaining() < 15:
                break
            found = False
            s_ones = rng.choice(ones, min(30, len(ones)), replace=False)
            s_zeros = rng.choice(zeros, min(30, len(zeros)), replace=False)
            for a in s_ones:
                if found:
                    break
                for b in s_zeros:
                    dc = swap_delta(h_f, int(a), int(b), n_pts)
                    nm = float(np.max(corr_f + dc))
                    if nm < max_f - 1e-10:
                        h_f[int(a)] = 0.0
                        h_f[int(b)] = 1.0
                        corr_f = np.correlate(h_f, 1.0 - h_f, mode='full')
                        max_f = nm
                        ones = np.where(h_f == 1.0)[0]
                        zeros = np.where(h_f == 0.0)[0]
                        c5_f = max_f * (2.0 / n_pts)
                        if c5_f < best_c5:
                            best_c5 = c5_f
                            best_h = h_f.copy()
                        found = True
                        break
            if not found:
                break

    # Phase 7: Try upsample to 2*n0
    if remaining() > 200:
        n_up = n0 * 2
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        # Also try binary at higher res
        h_up_bin = binary_snap(h_up, n_up)
        c5_upb = compute_c5(h_up_bin)
        if c5_upb < c5_up:
            h_up = h_up_bin
            c5_up = c5_upb
        if c5_up < best_c5 + 0.001:
            _G_N = n_up
            _G_T = 400.0
            _G_TOP_IDX = None
            bounds_u = [(0.0, 1.0)] * n_up
            cons_u = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_obj_lse, h_up, method='SLSQP',
                              bounds=bounds_u, constraints=cons_u,
                              options={'maxiter': 200, 'ftol': 1e-15})
                c5_r = compute_c5(res.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res.x, n_up)
            except Exception:
                pass
            # Binary search at higher res
            if remaining() > 80 and np.allclose(best_h, np.round(best_h), atol=0.4):
                h_ub = binary_snap(best_h, len(best_h))
                c5_ub = compute_c5(h_ub)
                if c5_ub <= best_c5:
                    best_c5 = c5_ub
                    best_h = h_ub
                h_fs = best_h.copy()
                corr_fs = np.correlate(h_fs, 1.0 - h_fs, mode='full').copy()
                max_fs = float(np.max(corr_fs))
                ones_fs = np.where(h_fs == 1.0)[0]
                zeros_fs = np.where(h_fs == 0.0)[0]
                n_fs = len(h_fs)
                for _ in range(100):
                    if remaining() < 20:
                        break
                    found = False
                    for a in rng.choice(ones_fs, min(25, len(ones_fs)), replace=False):
                        if found:
                            break
                        for b in rng.choice(zeros_fs, min(25, len(zeros_fs), replace=False) if len(zeros_fs) > 25 else zeros_fs):
                            dc = swap_delta(h_fs, int(a), int(b), n_fs)
                            nm = float(np.max(corr_fs + dc))
                            if nm < max_fs - 1e-10:
                                h_fs[int(a)] = 0.0
                                h_fs[int(b)] = 1.0
                                corr_fs = np.correlate(h_fs, 1.0 - h_fs, mode='full')
                                max_fs = nm
                                ones_fs = np.where(h_fs == 1.0)[0]
                                zeros_fs = np.where(h_fs == 0.0)[0]
                                c5_fs = max_fs * (2.0 / n_fs)
                                if c5_fs < best_c5:
                                    best_c5 = c5_fs
                                    best_h = h_fs.copy()
                                found = True
                                break
                    if not found:
                        break

    # Phase 8: Final micro coordinate descent
    if remaining() > 15:
        n_pts = len(best_h)
        for j in range(n_pts):
            if remaining() < 5:
                break
            ov = best_h[j]
            bv = ov
            bc = best_c5
            for st in [0.0005, 0.001, 0.002, 0.004]:
                for sg in [-1.0, 1.0]:
                    v = ov + sg * st
                    if 0.0 <= v <= 1.0:
                        tr = best_h.copy()
                        tr[j] = v
                        tr = make_feasible(tr, n_pts)
                        c = compute_c5(tr)
                        if c < bc - 1e-15:
                            bc = c
                            bv = v
            if bv != ov:
                best_h[j] = bv
                best_h = make_feasible(best_h, n_pts)
                best_c5 = bc

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
