        h_base = upsample_h(best_h, n_pts)
        for r in range(2):
            if remaining() < 55:
                break
            if r == 0:
                h_start = h_base.copy()
            else:
                h_start = h_base + rng.randn(n_pts) * 0.015
                h_start = make_feasible(h_start, n_pts)
            penalty = 5000.0
            bounds = [(0.0, 1.0)] * n_pts
            try:
                res = minimize(lambda x: obj_pen(x, T, n_pts, penalty), h_start,
                               method='SLSQP', bounds=bounds,
                               options={'maxiter': maxiter, 'ftol': 1e-15})
                res.x = make_feasible(res.x, n_pts)
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_pts)
            except Exception:
                pass

    # Phase 2: Single-coordinate descent with many candidates
    n_pts = len(best_h)
    for round_num in range(80):
        if remaining() < 10:
            break
        improved = False
        for j in range(n_pts):
            if remaining() < 5:
                break
            old = best_h[j]
            best_c = best_c5
            best_trial = old
            cands = [0.0, 1.0, 0.1, 0.3, 0.5, 0.7, 0.9]
            for d in [-0.2, -0.1, -0.05, -0.02, -0.01, -0.005, -0.002,
                      0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2]:
                cands.append(max(0.0, min(1.0, old + d)))
            for cand in cands:
                if abs(cand - old) < 1e-14:
                    continue
                trial = fast_trial(best_h, j, cand, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                best_h = fast_trial(best_h, j, best_trial, n_pts)
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    # Phase 3: Pair-coordinate descent
    n_pts = len(best_h)
    for round_num in range(40):
        if remaining() < 15:
            break
        improved = False
        for j in range(n_pts - 1):
            if remaining() < 8:
                break
            old1, old2 = best_h[j], best_h[j + 1]
            best_c = best_c5
            best_t1, best_t2 = old1, old2
            for d1 in [-0.15, -0.08, -0.04, -0.02, 0.02, 0.04, 0.08, 0.15]:
                for d2 in [-0.15, -0.08, -0.04, -0.02, 0.02, 0.04, 0.08, 0.15]:
                    t1 = max(0.0, min(1.0, old1 + d1))
                    t2 = max(0.0, min(1.0, old2 + d2))
                    if abs(t1 - old1) < 1e-14 and abs(t2 - old2) < 1e-14:
                        continue
                    trial = fast_trial2(best_h, j, t1, j + 1, t2, n_pts)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_t1, best_t2 = t1, t2
            if best_t1 != old1 or best_t2 != old2:
                best_h[j] = best_t1
                best_h[j + 1] = best_t2
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    # Phase 4: Higher resolution SLSQP polish
    for n_up in [1600, 1920]:
        if remaining() < 60:
            break
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 - 1e-6:
            best_c5 = c5_up
            best_h = h_up.copy()
        T = 300
        penalty = 10000.0
        n_pts = len(best_h)
        bounds = [(0.0, 1.0)] * n_pts
        try:
            res = minimize(lambda x: obj_pen(x, T, n_pts, penalty), best_h,
                           method='SLSQP', bounds=bounds,
                           options={'maxiter': 350, 'ftol': 1e-15})
            res.x = make_feasible(res.x, n_pts)
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 5: Fine coordinate descent
    n_pts = len(best_h)
    for round_num in range(50):
        if remaining() < 8:
            break
        improved = False
        for j in range(n_pts):
            if remaining() < 4:
                break
            old = best_h[j]
            best_c = best_c5
            best_trial = old
            for d in [-0.03, -0.01, -0.005, -0.002, -0.001, -0.0005,
                      0.0005, 0.001, 0.002, 0.005, 0.01, 0.03]:
                cand = max(0.0, min(1.0, old + d))
                if abs(cand - old) < 1e-14:
                    continue
                trial = fast_trial(best_h, j, cand, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                best_h = fast_trial(best_h, j, best_trial, n_pts)
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    # Phase 6: Final SLSQP with very high T
    if remaining() > 40:
        n_pts = len(best_h)
        for T in [600, 1000]:
            if remaining() < 20:
                break
            penalty = 20000.0
            bounds = [(0.0, 1.0)] * n_pts
            try:
                res = minimize(lambda x: obj_pen(x, T, n_pts, penalty), best_h,
                               method='SLSQP', bounds=bounds,
                               options={'maxiter': 200, 'ftol': 1e-16})
                res.x = make_feasible(res.x, n_pts)
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_pts)
            except Exception:
                pass

    # Phase 7: Ultra-fine final coordinate pass
    n_pts = len(best_h)
    for round_num in range(30):
        if remaining() < 5:
            break
        improved = False
        for j in range(n_pts):
            if remaining() < 3:
                break
            old = best_h[j]
            best_c = best_c5
            best_trial = old
            for d in [-0.002, -0.001, -0.0005, -0.0002, 0.0002, 0.0005, 0.001, 0.002]:
                cand = max(0.0, min(1.0, old + d))
                if abs(cand - old) < 1e-14:
                    continue
                trial = fast_trial(best_h, j, cand, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                best_h = fast_trial(best_h, j, best_trial, n_pts)
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
