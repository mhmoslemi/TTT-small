import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_all(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = float(n) / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

_OPT_T = 1.0
_OPT_P = 1.0

    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -500, 0)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (mc + log_sum / T) * 1.0

def _lse_grad(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    weights = exp_shifted / np.sum(exp_shifted) / T
    dcorr_max = np.zeros(2 * n - 1, dtype=float)
    dcorr_max += weights
    dcorr_max[np.argmax(corr)] += 1.0
    grad = np.zeros(n, dtype=float)
    for i in range(n):
        gi = 0.0
        for k in range(2 * n - 1):
            j1 = i + n - 1 - k
            j2 = i - (n - 1) + k
            if 0 <= j1 < n:
                gi += dcorr_max[k] * (1.0 - h[j1])
            if 0 <= j2 < n:
                gi -= dcorr_max[k] * h[j2]
        grad[i] = dx * gi
    return grad

def _power_obj(h):
    n = _OPT_N
    p = _OPT_P
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    top = np.sort(corr)[::-1][:max(1, len(corr) // 2)]
    return np.sum(np.maximum(corr, 0.0) ** p) ** (1.0 / p)

    return np.sum(h) - _OPT_N / 2.0

def _cons_jac(h):
    return np.ones(_OPT_N)

def _slsqp(h_start, n, T, max_iter, p=None):
    global _OPT_N, _OPT_T, _OPT_P
    _OPT_N = n
    _OPT_T = T
    if p is not None:
        _OPT_P = p
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_lse_obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return h_start.copy()

def _subgrad_step(h, step_size, n):
    r = compute_c5_all(h)
    k_star = int(np.argmax(r))
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(n):
        j1 = i + n - 1 - k_star
        j2 = i - (n - 1) + k_star
        g = 0.0
        if 0 <= j1 < n:
            g += (1.0 - h[j1])
        if 0 <= j2 < n:
            g -= h[j2]
        grad[i] = dx * g
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new
