from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_grad(h, k=10):
    """Gradient of the top-k mean of correlations"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top k indices
    top_idx = np.argsort(corr)[::-1][:k]
    # Average subgradient over top k
    g = np.zeros(n)
    for i in top_idx:
        shift = i - (n - 1)
        s = abs(shift)
        if s >= n:
            continue
        if shift >= 0:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
        else:
            g[:n - s] += 1.0 - h[s:]
            g[s:] -= h[:n - s]
    return g / k * dx

def fista_topk(h, n_iter=5000, k_start=20, k_end=3, lr=0.1, check=30):
    """FISTA with top-k mean surrogate"""
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        k = max(k_end, int(k_start * (k_end / k_start) ** f))
        lrt = lr * (1.0 - 0.5 * f)
        g = topk_grad(z, k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def binary_local_search(h, n_moves=5000, t_max=0.01, t_min=0.001):
    """Local search on binary version: flip pairs to maintain constraint"""
    n = len(h)
    # Round to binary
    b = (h > 0.5).astype(float)
    # Project to ensure sum = n/2
    target = n // 2
    if int(np.sum(b)) > target:
        # Flip some 1s to 0s
        ones = np.where(b == 1)[0]
        np.random.shuffle(ones)
        for i in ones[:int(np.sum(b)) - target]:
            b[i] = 0
    elif int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        np.random.shuffle(zeros)
        for i in zeros[:target - int(np.sum(b))]:
            b[i] = 1
    
    bc = compute_c5(b)
    bh = b.copy()
    t = t_max
    
    for mv in range(n_moves):
        t = t_max * (t_min / t_max) ** (mv / n_moves)
        # Pick a random 1 and a random 0, flip both
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        b[i], b[j] = 0, 1

            bc, bh = c_new, b.copy()
        else:
            b[i], b[j] = 1, 0  # revert
    
    return bh, bc

        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_at(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def fista_topk(h, n_iter=5000, k_start=40, k_end=1, lr=0.12, check=20):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        k = max(k_end, int(k_start * (k_end / k_start) ** f))
        lrt = lr * (1.0 - 0.3 * f)
        corr = np.correlate(z, 1.0 - z, mode='full')
        top_idx = np.argsort(corr)[::-1][:k]
        g = np.zeros(n)
        for i in top_idx:
            g += subgrad_at(z, i - (n - 1))
        g *= (1.0 / k) * dx
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def fista_softmax(h, n_iter=5000, T0=30.0, T1=0.01, top_n=30, lr=0.1, check=20):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.4 * f)
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            top_idx = np.array([np.argmax(corr)])
            w = np.array([1.0])
        else:
            w = exps / tot
            top_idx = np.argsort(w)[::-1][:top_n]
        g = np.zeros(n)
        for i in top_idx:
            wi = w[i]
            if wi < 1e-12:
                break
            g += wi * subgrad_at(z, i - (n - 1))
        g *= dx
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def binary_local_search(h, n_moves=4000, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    b = (h > 0.5).astype(float)
    target = n // 2
    cur = int(np.sum(b))
    if cur > target:
        ones = np.where(b == 1)[0]
        rng.shuffle(ones)
        for i in ones[:cur - target]:
            b[i] = 0.0
    elif cur < target:
        zeros = np.where(b == 0)[0]
        rng.shuffle(zeros)
        for i in zeros[:target - cur]:
            b[i] = 1.0
    bc = compute_c5(b)
    bh = b.copy()
    for mv in range(n_moves):
        t = 0.008 * (0.0003 / 0.008) ** (mv / n_moves)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        c_new = compute_c5(b)
        if c_new < bc or rng.rand() < np.exp(-(c_new - bc) / max(t, 1e-12)):
            bc, bh = c_new, b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return bh, bc

def slsqp_opt(h, T=5.0, maxiter=300):
    n = len(h)
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(c)
        vals = T * (c - mc)
        vals = np.clip(vals, -500, 500)
        return (mc + np.log(np.sum(np.exp(vals))) / T) * dx
    def objj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(c)
        vals = T * (c - mc)
        vals = np.clip(vals, -500, 500)
        w = np.exp(vals)
        ws = np.sum(w)
        if ws < 1e-300:
            i = np.argmax(c)
            return subgrad_at(x, i - (n - 1)) * dx
        w = w / ws
        g = np.zeros(n)
        for idx in np.argsort(w)[::-1][:20]:
            wi = w[idx]
            if wi < 1e-10:
                break
            g += wi * subgrad_at(x, idx - (n - 1))
        return g * dx
    def cfun(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    try:
        r = minimize(obj, h, jac=objj, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': cfun, 'jac': cjac}],
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        if r.x is not None:
            return project_c(r.x, n)
    except Exception:
        pass
    return h

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)
