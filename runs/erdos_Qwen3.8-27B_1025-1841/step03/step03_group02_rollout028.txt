def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t

            affected_m.add(j0)

            affected_m.add(i0 - kk)
        if 0 <= j0 - kk < n and 0 <= j0 < n:

                old_val = b[m] * (1 - b[m + kk])

                new_pk = b[m + kk]
                if m + kk == i0: new_pk = 0
                elif m + kk == j0: new_pk = 1

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def thue_morse_binary(n, rng):
    """Generate a Thue-Morse-inspired binary sequence with exactly n/2 ones."""
    # Thue-Morse: t(0)=0, t(2n)=t(n), t(2n+1)=1-t(n)
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i // 2] ^ (i % 2)
    # This gives exactly n/2 ones if n is a power of 2
    # For other n, we need to adjust
    b = t.copy()
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        b[zeros[rng.randint(len(zeros))]] = 1
    return b
