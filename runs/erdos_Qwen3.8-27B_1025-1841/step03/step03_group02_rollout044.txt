def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.success or res.x is not None:
        x = np.clip(res.x, 0.0, 1.0)
        x = make_feasible(x, n)
        return x
    return h

def subgradient_step(h, n, alpha, shift_k):
    h_new = h.copy().astype(float)
    k = abs(shift_k)
    if k >= n or k == 0:
        return h_new
    g = np.zeros(n)
    if shift_k > 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] += (-h[:n-k])
    else:
        g[k:] += (1.0 - h[:n-k])
        g[:n-k] += (-h[k:])
    h_new = h - alpha * g
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    return h_new

        ov = compute_ov(best_h)

                    best_h = x_res
            except Exception:
                pass

    # Phase 8: Try n=962
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n_alt = 962
        if n_alt != best_n:
            old_x = np.linspace(0.0, 2.0, best_n)
            new_x = np.linspace(0.0, 2.0, n_alt)
            h_alt = np.interp(new_x, old_x, best_h)
            h_alt = make_feasible(h_alt, n_alt)
            b_alt = to_binary(h_alt, n_alt)
            ba, c5_alt = multi_greedy(b_alt, n_alt, start_time, budget_s, 20000, rng)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = ba.copy()
                best_n = n_alt
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                try:
                    x_res = slsqp_run(best_h, best_n, 100.0, 200, 1e-14)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = x_res
                except Exception:
                    pass

    # Phase 9: Multi-restart on best solution with perturbation
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        for restart in range(3):
            if time.time() - start_time > budget_s - 30:
                break
            perturbed = best_h.copy().astype(float)
            perturb_idx = rng.choice(best_n, size=best_n // 20, replace=False)
            perturbed[perturb_idx] = rng.uniform(0.3, 0.7, size=len(perturb_idx))
            perturbed = make_feasible(perturbed, best_n)
            bp = to_binary(perturbed, best_n)
            br, c5_r = multi_greedy(bp, best_n, start_time, budget_s, 10000, rng)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = br.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 40 and c5_r < best_c5:
                try:
                    x_res = slsqp_run(best_h, best_n, 100.0, 200, 1e-14)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = x_res
                except Exception:
                    pass

    # Phase 10: Final SLSQP with high T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        try:
            x_res = slsqp_run(best_h, best_n, 1000.0, 200, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res
        except Exception:
            pass

    # Final feasibility check
    best_h = make_feasible(best_h, best_n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, best_n
