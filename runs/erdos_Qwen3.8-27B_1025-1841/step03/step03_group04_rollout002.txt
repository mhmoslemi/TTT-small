def compute_sens(h, j, n):

def coord_descent_pass(h, n, order):

    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)

        delta = optimize_1d(corr, sens, h[j], n, dx)

            corr += delta * sens

            new_c = float(np.max(corr) * dx)
            if new_c < best_c5:

    return h, corr, best_c5
