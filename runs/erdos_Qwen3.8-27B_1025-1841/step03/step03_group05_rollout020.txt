    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def corr_and_grad(h, T, top_k=60):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    val = (mc + np.log(tot) / T) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    w_sub = w[idx]
    w_sub = w_sub / np.sum(w_sub)
    g = np.zeros(n)
    for i in range(len(idx)):
        if w_sub[i] < 1e-12:
            break
        shift = idx[i] - (n - 1)
        g += w_sub[i] * _subgrad_one(h, shift)
    return val, g * dx

def fista_v2(h0, n_iter=8000, T0=100.0, T1=0.05, lr0=0.2, check=50):
    n = len(h0)
    x = h0.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f_frac = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f_frac
        lrt = lr0 * (1.0 - 0.5 * f_frac)
        val, g = corr_and_grad(z, T)
        xn = project_c(z - lrt * g, n)
        # Momentum with restart
        c_new = compute_c5(xn)
        if it % check == 0:
            if c_new < bc:
                bc = c_new
                bh = xn.copy()
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z_prev = z.copy()
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = xn
        tk = tn
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def lbfgs_smooth(h0, T=80.0, maxiter=300):
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0

    def obj_and_grad(x):
        val, g = corr_and_grad(x, T, top_k=80)
        # Add penalty for constraint violation
        penalty = 100.0 * (np.sum(x) - target) ** 2
        gpen = 200.0 * (np.sum(x) - target) * np.ones(n)
        return val + penalty, g + gpen

    best_h = h0.copy()
    best_c = compute_c5(h0)
    try:
        res = minimize(obj_and_grad, h0.copy(), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12, 'maxfun': maxiter * 5})
        if res.x is not None:
            hx = project_c(res.x, n)
            cx = compute_c5(hx)
            if cx < best_c:
                best_h, best_c = hx, cx
    except Exception:
        pass
    return best_h, best_c

def active_set_descent(h0, n_iter=5000, tol=0.002, lr0=0.05):
    n = len(h0)
    x = h0.astype(float).copy()
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f_frac = it / max(n_iter - 1, 1)
        lr = lr0 * (1.0 - 0.7 * f_frac)
        tol_cur = tol * (1.0 - 0.9 * f_frac) + 1e-5
        dx_ = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        active = np.where(corr > mc - tol_cur)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        g = np.zeros(n)
        for idx in active:
            shift = idx - (n - 1)
            g += _subgrad_one(x, shift)
        g /= len(active)
        g *= dx_
        # Normalize
        gn = np.linalg.norm(g)
        if gn < 1e-12:
            break
        g = g / gn
        xn = project_c(x - lr * g * gn / max(gn, 1e-10), n)
        # Line search
        best_x = xn
        best_c = compute_c5(xn)
        for ls in range(5):
            if best_c < bc:
                break
            lr_ls = lr * (0.5 ** (ls + 1))
            xn = project_c(x - lr_ls * g * gn, n)
            c_ls = compute_c5(xn)
            if c_ls < best_c:
                best_c = c_ls
                best_x = xn.copy()
        x = best_x
        if best_c < bc:
            bc = best_c
            bh = best_x.copy()
    return bh, bc

def coordinate_descent(h0, n_sweeps=50):
    n = len(h0)
    dx_ = 2.0 / n
    x = h0.astype(float).copy()
    bc = compute_c5(x)
    for sweep in range(n_sweeps):
        improved = False
        for i in range(n):
            # Try small perturbations of x[i]
            best_val = bc
            best_xi = x[i]
            for delta in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]:
                x[i] = min(1.0, max(0.0, x[i] + delta))
                # Adjust to maintain sum constraint (add/subtract from a random other element)
                sum_diff = (target_sum(x, n) - x[i]) - (target_sum(h0, n) - best_xi)
                # Actually just project
                xp = np.copy(x)
                xp[i] = best_xi  # revert for now
                # Try: change x[i] by delta, then project
                x_try = h0.astype(float).copy()  # This is wrong, we should modify current x
                # Let me redo: modify x[i], then project the whole thing
                x[i] = min(1.0, max(0.0, x[i] + delta))
                xp = project_c(x.copy(), n)
                # Check if x[i] actually changed
                if abs(xp[i] - x[i]) < 1e-6:
                    x[i] = best_xi
                    continue
                c_try = compute_c5(xp)
                if c_try < best_val:
                    best_val = c_try
                    best_xi = xp.copy()
            if best_val < bc - 1e-10:
                x = best_xi
                bc = best_val
                improved = True
            else:
                x[i] = best_xi
        if not improved:
            break
    return x, compute_c5(x)

def target_sum(h, n):
    return n / 2.0

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def elapsed():
        return time.time() - t0
    
    def remaining():
        return budget_s - elapsed()

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = project_c(rng.rand(480), 480)
    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"Init: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1: FISTA warmup on parent
    if remaining() > 300:
        h1, c1 = fista_v2(bh, 6000, 80.0, 0.1, 0.18, 50)
        if c1 < bc:
            bc, bh = c1, h1.copy()
        print(f"FISTA1: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 2: L-BFGS polish
    if remaining() > 200:
        h2, c2 = lbfgs_smooth(bh, T=60.0, maxiter=200)
        if c2 < bc:
            bc, bh = c2, h2.copy()
        print(f"LBFGS: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 3: Active set descent
    if remaining() > 150:
        h3, c3 = active_set_descent(bh, 8000, tol=0.003, lr0=0.04)
        if c3 < bc:
            bc, bh = c3, h3.copy()
        print(f"ASD: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 4: Another FISTA with very low T
    if remaining() > 150:
        h4, c4 = fista_v2(bh, 5000, 10.0, 0.02, 0.03, 30)
        if c4 < bc:
            bc, bh = c4, h4.copy()
        print(f"FISTA2: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 5: Try different resolutions
    np_ = len(bh)
    for nn in [np_ // 2, int(np_ * 1.5), np_ * 2]:
        if nn < 128 or nn > 2400 or remaining() < 120:
            continue
        hu = upsample(bh, nn)
        cu = compute_c5(hu)
        if remaining() > 80:
            hg, cg = fista_v2(hu, 4000, 40.0, 0.05, 0.12, 40)
            if cg < cu:
                hu, cu = hg, cg
        if remaining() > 40:
            hn, cn = lbfgs_smooth(hu, T=40.0, maxiter=150)
            if cn < cu:
                hu, cu = hn, cn
        cf = compute_c5(hu)
        if cf < bc:
            bc, bh = cf, hu.copy()
        print(f"  Res n={nn}: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 6: Multiple restarts at current resolution
    np_ = len(bh)
    restarts = [tm_seq(np_), project_c(rng.rand(np_), np_)]
    # Golden ratio pattern
    gr = np.zeros(np_)
    gr[:] = np.floor(np.arange(np_) * 0.6180339887).astype(float)
    gr = (gr - np.floor(gr))
    gr = (gr > 0.5).astype(float)
    restarts.append(project_c(gr, np_))
    for hs in restarts:
        if remaining() < 60:
            break
        hr, cr = fista_v2(hs, 3000, 50.0, 0.1, 0.15, 50)
        if remaining() > 30:
            hr2, cr2 = fista_v2(hr, 2000, 5.0, 0.02, 0.02, 30)
            if cr2 < cr:
                hr, cr = hr2, cr2
        if cr < bc:
            bc, bh = cr, hr.copy()
            print(f"  Restart: C5={bc:.6f}", flush=True)

    # Phase 7: Final fine-tuning
    if remaining() > 60:
        hf, cf = active_set_descent(bh, 5000, tol=0.001, lr0=0.02)
        if cf < bc:
            bc, bh = cf, hf.copy()
    if remaining() > 30:
        hf, cf = fista_v2(bh, 3000, 5.0, 0.01, 0.015, 20)
        if cf < bc:
            bc, bh = cf, hf.copy()
    if remaining() > 20:
        hf, cf = lbfgs_smooth(bh, T=100.0, maxiter=200)
        if cf < bc:
            bc, bh = cf, hf.copy()

    # Final projection and verification
    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={elapsed():.0f}", flush=True)
    return bh, bc, len(bh)
