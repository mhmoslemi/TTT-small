            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    while s > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0; s -= 1
    while s < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1; s += 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng, use_blocks=False):
    b = to_binary(h, n)
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac if T1 > 0 else T0

            co = rng.choice(oc, size=cnt, replace=False)
            cz = rng.choice(zc, size=cnt, replace=False)

            b[s+cz] = 1

                curr = new_c
                if new_c < best:
                    best = new_c; best_b = b.copy()
            else:
                b[s+co] = 1
                b[s+cz] = 0
        else:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0; b[j] = 1
            new_c = compute_c5(b)
            if new_c <= curr or (T > 1e-12 and rng.random() < np.exp(min(0.0, -(new_c - curr)/T))):

                if new_c < best:

def _con_sum(x):
    return np.sum(x) - _SL_N / 2.0

def _con_sum_jac(x):
    return np.ones(_SL_N)

def _obj(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    ls = np.log(np.sum(np.exp(shifted)))
    return (mc + ls / T) * dx

def slsqp_polish(h, n, T_values, maxiter_each, start_time, budget_s):
    global _SL_N, _SL_T
    best = h.copy()
    best_c = compute_c5(h)
    for T in T_values:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        _SL_N = n
        _SL_T = T
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_sum_jac}]
        try:
            res = minimize(_obj, best.copy(), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter_each, 'ftol': 1e-14})
            if res.success or res.fun < best_c:
                x = make_feasible(res.x, n)
                c = compute_c5(x)
                if c < best_c:
                    best_c = c
                    best = x
        except Exception:
            pass
    return best, best_c

def coord_descent(h, n, start_time, budget_s, max_rounds=20):
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            for delta in [0.01, 0.005, 0.001, 0.05]:
                for sign in [1, -1]:
                    cand = old + sign * delta
                    if cand < 0 or cand > 1:
                        continue
                    trial = h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-15:
                        best_c = c
                        h = trial
                        old = cand
                        improved = True
                        break
                if improved:
                    break
    return h, best_c

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    while s > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
        s -= 1
    while s < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
        s += 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng, use_blocks=False):

    curr = compute_c5(b)

    best_b = b.copy()

def _con_fun(x):
    return np.sum(x) - _SL_N / 2.0

def _con_jac(x):
    return np.ones(_SL_N)

def _obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    mx = np.max(shifted)
    if mx < -500:
        ls = 0.0
    else:
        ls = mx + np.log(np.sum(np.exp(shifted - mx)))
    return (mc + ls / T) * dx
