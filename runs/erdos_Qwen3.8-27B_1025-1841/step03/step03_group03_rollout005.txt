    for rnd in range(max_rounds):
        if deadline and time.time() > deadline - 10:
            break
        improved = False
        for j in range(n):
            # Precompute d_k for this j
            d = np.zeros(2*n-1)
            for k_idx in range(2*n-1):
                k = lags[k_idx]
                g = 0.0
                jk = j + k
                jmk = j - k
                if 0 <= jk < n:
                    g += (1.0 - h[jk])
                if 0 <= jmk < n:
                    g -= h[jmk]
                d[k_idx] = g
            # Try different values

            best_delta = 0.0
            best_max_corr = np.max(corr)
            for delta in [-old, 1-old, -old*0.5, (1-old)*0.5, -0.1, 0.1, -0.05, 0.05]:
                new_val = old + delta
                if new_val < -1e-9 or new_val > 1+1e-9:

                actual_delta = new_val - old

                mc = np.max(corr_new)

                if np.max(corr) * dx < best_c - 1e-12:
                    best_c = np.max(corr) * dx

        if not improved:
            break

    term2 = np.zeros(2*n-1)

    return term1 - term2

    old_x = np.linspace(0.0, 2.0, old_n)

            break

            # d_k = (1-h[j+k]) - h[j-k] for each lag k
            idx_pos = j + lags  # j+k for each k

1. For each coordinate j in sequence:

4. Check if improved
