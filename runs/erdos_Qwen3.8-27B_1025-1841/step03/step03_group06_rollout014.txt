import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    ox = np.arange(old_n, dtype=float) * (2.0 / old_n)
    nx = np.arange(new_n, dtype=float) * (2.0 / new_n)
    return make_feasible(np.interp(nx, ox, h), new_n)

def _lbfgs_opt(h, n, T, lam, maxiter):
    def f(v):
        dx = 2.0 / n
        c = np.correlate(v, 1.0 - v, mode='full')
        m = c.max()
        return (m + np.log(np.sum(np.exp(T * (c - m)))) / T) * dx + lam * (v.sum() - n/2.0)**2
    res = minimize(f, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
