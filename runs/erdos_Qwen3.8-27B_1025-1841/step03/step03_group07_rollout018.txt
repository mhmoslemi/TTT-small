def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')

    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def corr_and_grad(h, n, top_k, tau):

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    
    if top_k < len(corr):

    else:
        top_idx = np.arange(len(corr))

    top_vals = corr[top_idx] - max_c
    top_vals = np.clip(top_vals * tau, -100, 0)
    exps = np.exp(top_vals)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for (i, k) in enumerate(top_idx):
        w = weights[i]
        if w < 1e-12:
            continue
        sh = k - n + 1
        # h[j] as h[i+sh]: j in [max(0,sh), min(n, n+sh))
        j1_lo = max(0, sh)
        j1_hi = min(n, n + sh)
        if j1_lo < j1_hi:
            idx = np.arange(j1_lo, j1_hi)
            grad[j1_lo:j1_hi] += w * (1.0 - h[idx - sh])

    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall_count = 0

    h = h_flat.reshape(n)
    dx = 2.0 / n

This is a (weighted) correlation of h with the indicator of the valid range... it's getting complex.

```python
def compute_grad_topk(h, n, top_k, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    
    top_vals = corr[top_idx]
    shifted = tau * (top_vals - max_c)
    shifted = np.clip(shifted, -100, 0)
    exps = np.exp(shifted)
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        w = weights[i]
        if w < 1e-12:
            continue
        sh = top_idx[i] - n + 1
        # Contribution from h[j] as the "first" factor
        j_lo = max(0, sh)
        j_hi = min(n, n + sh)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += w * (1.0 - h[j_lo - sh:j_hi - sh])
        # Contribution from h[j] as the "second" factor (with minus)
        j_lo2 = max(0, -sh)
        j_hi2 = min(n, n - sh)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= w * h[j_lo2 + sh:j_hi2 + sh]
    
    grad *= dx
    soft_obj = (max_c + np.log(exps.sum()) / tau) * dx
    return soft_obj, grad, weights, top_idx
```

import numpy as np
import time
from scipy.optimize import minimize

def project_to_feasible(g, n):
    target = n / 2.0
    lo = float(np.min(g)) - 2.0
    hi = float(np.max(g)) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(g - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def resample(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def soft_max_obj(h_flat, n, tau, lam):
    h = h_flat.reshape(n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = tau * (corr - mc)
    sh = np.clip(sh, -50.0, 0.0)
    val = (mc + np.log(np.exp(sh).sum()) / tau) * dx
    s = h.sum()
    val += lam * (s - n * 0.5) ** 2
    return val

def soft_max_grad(h_flat, n, tau, lam, top_k):
    h = h_flat.reshape(n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if top_k < len(corr):
        ti = np.argpartition(corr, -top_k)[-top_k:]
    else:
        ti = np.arange(len(corr))
    tv = corr[ti]
    sh = np.clip(tau * (tv - mc), -50.0, 0.0)
    ex = np.exp(sh)
    ws = ex / ex.sum()
    g = np.zeros(n)
    for i in range(len(ti)):
        w = ws[i]
        if w < 1e-10:
            continue
        k = int(ti[i])
        s = k - n + 1
        a = max(0, s)
        b = min(n, n + s)
        if a < b:
            g[a:b] += w * (1.0 - h[a-s:b-s])
        a2 = max(0, -s)
        b2 = min(n, n - s)
        if a2 < b2:
            g[a2:b2] -= w * h[a2+s:b2+s]
    g *= dx
    s = h.sum()
    g += 2.0 * lam * (s - n * 0.5)

Since I'm using top_k for the gradient but the full soft max for the objective, they won't be exactly consistent. This might cause L-BFGS-B to have convergence issues.

Solution: Either use top_k for both (compute the soft max only over the top_k values), or use the full correlation for the gradient (which is O(n²) per evaluation, too slow).

def project_to_feasible(g, n):
    target = n / 2.0
    lo = float(np.min(g)) - 2.0
    hi = float(np.max(g)) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = float(np.sum(np.clip(g - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def resample(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def nesterov_run(h, n, time_limit, tau, lr, top_k, momentum_cap=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    h_prev = h.copy()
    t_param = 1.0
    dx = 2.0 / n

    while time.time() - start < time_limit:
        mu = (t_param - 1.0) / (t_param + 2.0)
        mu = min(mu, momentum_cap)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = np.correlate(y, 1.0 - y, mode='full')
        mc = corr.max()
        if top_k < len(corr):
            ti = np.argpartition(corr, -top_k)[-top_k:]
        else:
            ti = np.arange(len(corr))
        tv = corr[ti]
        sh = np.clip(tau * (tv - mc), -100.0, 0.0)
        ex = np.exp(sh)
        ws = ex / ex.sum()

        grad = np.zeros(n)
        for i in range(len(ti)):
            w = ws[i]
            if w < 1e-12:
                continue
            s = int(ti[i]) - n + 1
            a = max(0, s)
