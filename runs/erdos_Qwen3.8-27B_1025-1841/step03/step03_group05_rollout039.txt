    n = len(h)
    g = np.zeros(n)

        # c = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])

def adaptive_fista(h, n_iter, T_start, T_end, lr_start, lr_end):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = float('inf')
    best_h = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        # Adaptive temperature
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        # Count "active" shifts (within 5% of max)
        n_active = np.sum(corr > 0.95 * mc)
        n_total = len(corr)
        # T: high when many active (flat), low when few active (peaked)
        T = T_start * (T_end / T_start) ** f
        if n_active < 10:
            T *= 0.5  # More peaked, use sharper
        elif n_active > n_total * 0.5:
            T *= 1.5  # Very flat, spread more
        
        lrt = lr_start * (lr_end / lr_start) ** f
        
        # Compute smoothed gradient
        exps = np.exp(T * (corr - mc))
        tot = exps.sum()
        w = exps / tot
        # Use top-k for efficiency
        top_k = min(200, n_total)
        idx = np.argsort(w)[::-1][:top_k]
        g = np.zeros(n)
        for i in idx:
            if w[i] < 1e-10:
                break
            g += w[i] * subgrad_at_shift(z, i - (n-1))
        g *= (2.0 / n)  # dx factor
        
        # FISTA update
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        x = xn
        tk = tn
        
        # Track best
        if (it + 1) % 20 == 0:
            c = np.max(np.correlate(x, 1.0 - x, mode='full')) * (2.0/n)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    
    c = np.max(np.correlate(best_h, 1.0 - best_h, mode='full')) * (2.0/n)
    return best_h, c

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_at_shift(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        valid = n - lag
        if valid > 0:
            g[:valid] += (1.0 - h[lag:])
            g[lag:] -= h[:valid]
    else:
        s = -lag
        valid = n - s
        if valid > 0:
            g[s:] += (1.0 - h[:valid])
            g[:valid] -= h[s:]
    return g

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def subgrad_at_shift(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        valid = n - lag
        if valid > 0:
            g[:valid] += (1.0 - h[lag:])
            g[lag:] -= h[:valid]
    else:
        s = -lag
        valid = n - s
        if valid > 0:
            g[s:] += (1.0 - h[:valid])
            g[:valid] -= h[s:]
    return g

def fista_opt(h, n_iter, T_s, T_e, lr_s, lr_e, top_k, check, rng_seed):
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(rng_seed)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    last_imp = 0

    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        lr = 0.5 * (lr_e + (lr_s - lr_e) * np.cos(np.pi * f))

        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)

        thr = mc * 0.90
        n_act = int(np.sum(corr >= thr))
        T_base = T_s * (T_e / T_s) ** f
        if n_act < 5:
            T = T_base * 0.25
        elif n_act < 15:
            T = T_base * 0.5
        elif n_act < 50:
            T = T_base * 0.8
        elif n_act < 100:
            T = T_base
        else:
            T = T_base * 1.4
        T = max(T, 0.05)

        diff = np.clip(corr - mc, -80.0 / T, 0.0)
        exps = np.exp(T * diff)
        tot = exps.sum()

        if tot < 1e-300:
            ki = int(np.argmax(corr))
            g = subgrad_at_shift(z, ki - (n - 1))
        else:
            w = exps / tot
            ks = min(top_k, len(w))
            ti = np.argpartition(w, -ks)[-ks:]
            g = np.zeros(n)
            for i in ti:
                wi = w[i]
                if wi < 1e-14:
                    continue
                g += wi * subgrad_at_shift(z, i - (n - 1))

        g *= dx

        if it > 300 and (it - last_imp) > 600 and (it % 120) < check:
            g += 0.015 * (rng.rand(n) * 2.0 - 1.0)

        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) * 0.5
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 3.0, out=z)
        x = xn
        tk = tn

        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
                last_imp = it

    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

def scale_push(h, factor):
    n = len(h)
    return project_c(0.5 + factor * (h - 0.5), n)

def upsample_h(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0.0, 2.0, len(h), endpoint=False)
    nx = np.linspace(0.0, 2.0, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h, period=len(h)), nn)

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]
