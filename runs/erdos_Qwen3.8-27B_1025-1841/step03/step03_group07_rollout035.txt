def compute_slope(h, j, n):
    """Compute slope vector for coordinate j."""
    # s[k] = coeff of h[j] in C[k]
    # C[k] = sum_i h[i]*(1-h[i+k-n+1])
    # h[j] appears in C[k] as:
    #   h[j] * (1-h[j+k-n+1]) if 0 <= j+k-n+1 < n  [direct term]
    #   h[j-k+n-1] * (1-h[j]) if 0 <= j-k+n-1 < n  [from the i=j-k+n-1 term]
    # So s[k] = (1-h[j+k-n+1])*[valid] - h[j-k+n-1]*[valid]
    #          = [1 if j+k-n+1 valid] - [h[j+k-n+1] if valid] - [h[j-k+n-1] if j-k+n+1 valid]
    
    # Let me use shift = k - (n-1), range [-(n-1), (n-1)]
    # s = 1 - h[j+shift] (if valid) - h[j-shift] (if valid)
    # where h[OOB] = 0
    
    shift = np.arange(-(n-1), n)  # length 2n-1
    j_plus = j + shift  # where h[j+shift] is needed
    j_minus = j - shift  # where h[j-shift] is needed
    
    # Build h[j+shift] with OOB as 0
    h_plus = np.zeros(2*n-1)
    valid_plus = (j_plus >= 0) & (j_plus < n)
    h_plus[valid_plus] = h[j_plus[valid_plus]]
    
    h_minus = np.zeros(2*n-1)
    valid_minus = (j_minus >= 0) & (j_minus < n)
    h_minus[valid_minus] = h[j_minus[valid_minus]]
    
    s = 1.0 - h_plus - h_minus

But actually, I can do better. Since f(d) is piecewise linear, I can find the exact minimum by finding the "kink" point. The minimum of a piecewise linear convex function is at a kink (or any point in a flat minimum region).

The kink points are where the argmax changes. Between two consecutive kinks, the argmax is constant, so f is linear. The minimum is where the slope changes from negative to positive.

The slope of f at a point d (where the argmax is unique, say at index k*) is s[k*]. So:
- If s[k*] < 0: f is decreasing at d, minimum is to the right
- If s[k*] > 0: f is increasing at d, minimum is to the left
- If s[k*] = 0: f is flat at d, we're at the minimum (or in a flat region)

So the algorithm is:
1. Start at d = 0 (current value)
2. Find k* = argmax(C)
3. If s[k*] < 0: move right. The next kink is where C[k*] + d*s[k*] = C[k2] + d*s[k2] for the second-best k2. d_kink = (C[k2] - C[k*]) / (s[k*] - s[k2]). Move to d_kink (or d_hi if beyond).
4. If s[k*] > 0: move left. Similar.
5. If s[k*] = 0: done.

This is O(n) per coordinate (to find argmax and second argmax). Much faster than ternary search!

But there's a subtlety: when multiple indices are tied for the argmax, the "slope" is the max of their s values (for the right derivative) or the min (for the left derivative). The minimum is where the left derivative ≤ 0 and right derivative ≥ 0.

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def compute_slope(h, j, n, shift):
    jp = j + shift
    jm = j - shift
    v_p = (jp >= 0) & (jp < n)
    v_m = (jm >= 0) & (jm < n)
    return 1.0 - h[np.clip(jp, 0, n - 1)] * v_p - h[np.clip(jm, 0, n - 1)] * v_m


def opt_1d(C, s, d_lo, d_hi, n_iter=25):
    if d_lo >= d_hi:
        return 0.0
    a = d_lo
    b = d_hi
    for _ in range(n_iter):
        if b - a < 1e-8:
            break
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        if (C + m1 * s).max() <= (C + m2 * s).max():
            b = m2
        else:
            a = m1
    d = (a + b) / 2.0
    f_mid = (C + d * s).max()
    f_a = (C + a * s).max()
    f_b = (C + b * s).max()
    if f_a < f_mid:
        d = a
    elif f_b < f_mid:
        d = b
    return d


def coord_descent(h, n, time_limit, max_passes=5000, rng=None):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    C = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = C.max() * dx
    best_h = h.copy()
    start = time.time()
    shift = np.arange(-(n - 1), n, dtype=np.int64)
    if rng is None:
        rng = np.random.RandomState()
    stall = 0
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        order = rng.permutation(n)
        for j in order:
            v = h[j]
            if v <= 1e-12 or v >= 1.0 - 1e-12:
                continue
            s = compute_slope(h, j, n, shift)
            d = opt_1d(C, s, -v, 1.0 - v)
            if abs(d) > 1e-14:
                C += d * s
                h[j] = v + d
        h = project_to_feasible(h, n)
        C = np.correlate(h, 1.0 - h, mode='full')
        c5_now = C.max() * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 400:
            break
    return best_h, best_c5


def two_opt(h, n, time_limit, n_moves=5000000, rng=None):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    C = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = C.max() * dx
    best_h = h.copy()
    start = time.time()
    shift = np.arange(-(n - 1), n, dtype=np.int64)
    if rng is None:
        rng = np.random.RandomState()
    stall = 0
    for it in range(n_moves):
        if time.time() - start > time_limit:
            break
        i = rng.randint(0, n)
        j = rng.randint(0, n)
        if i == j:
            continue
        if h[i] <= 1e-12 and h[j] >= 1.0 - 1e-12:
            continue
        if h[i] >= 1.0 - 1e-12 and h[j] <= 1e-12:
            continue
        s_i = compute_slope(h, i, n, shift)
        s_j = compute_slope(h, j, n, shift)
        s_eff = s_i - s_j
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_lo >= d_hi - 1e-12:
            continue
        d = opt_1d(C, s_eff, d_lo, d_hi)
        if abs(d) > 1e-14:
            new_max = (C + d * s_eff).max()
            if new_max < C.max() - 1e-14:
                C += d * s_eff
                h[i] += d
                h[j] -= d
                stall = 0
                c5_now = C.max() * dx
                if c5_now < best_c5 - 1e-13:
                    best_c5 = c5_now
                    best_h = h.copy()
            else:
                stall += 1
        else:
            stall += 1
        if stall > 300000:
            break
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)
