            best_h = h_g.copy()
        if rem() > 30:
            h_s, c_s = sa(h_g, 60000, rng, T0=0.02, T1=0.0003)
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Phase 3: Random restarts at best resolution
    if rem() > 50:
        for rep in range(4):
            if rem() < 20:
                break
            hr = best_h + 0.4 * rng.randn(nb)
            hb_r = make_binary(hr, nb)
            h_g, c_g = greedy(hb_r, rng, max_it=3000)
            if rem() > 20:
                h_s, c_s = sa(h_g, 40000, rng, T0=0.02 + 0.005 * rep, T1=0.0002)
                if c_s < best_c5:
                    best_c5 = c_s
                    best_h = h_s.copy()

    # Phase 4: Continuous L-BFGS refinement
    if rem() > 30:
        nc = min(len(best_h), 1024)
        hc = upsample(best_h, nc) if nc < len(best_h) else best_h.copy()
        hc = make_feasible(hc, nc)
        dxc = 2.0 / nc
        bounds = [(0.0, 1.0)] * nc

        def obj_cont(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            mc = np.max(corr)
            T = 400.0
            sh = np.clip(T * (corr - mc), -50, 50)
            return (mc + np.log(np.sum(np.exp(sh))) / T) * dxc + 500.0 * (np.sum(x) - nc * 0.5) ** 2

        try:
            res = minimize(obj_cont, hc, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 400, 'ftol': 1e-17, 'gtol': 1e-12})
            ho = make_feasible(res.x, nc)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5 = co
                best_h = ho.copy()
        except Exception:
            pass

    # Phase 5: Final binary greedy squeeze
    n_f = len(best_h)
    h_f = upsample(best_h, n_f) if False else best_h.copy()
    h_fb = make_binary(h_f, n_f)
    if rem() > 5:
        h_g, c_g = greedy(h_fb, rng, max_it=10000)
        if c_g < best_c5:
            best_c5 = c_g
            best_h = h_g.copy()

    best_h = make_feasible(best_h, len(best_h))
    fc = compute_c5(best_h)
    if fc < best_c5:
        best_c5 = fc

    return best_h, best_c5, len(best_h)
