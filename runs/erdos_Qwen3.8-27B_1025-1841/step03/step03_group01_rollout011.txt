import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _slsqp_smoothed(h_start, n, T, max_iter, ftol=1e-14):
    dx = 2.0 / n
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = shifted - np.max(shifted)
        log_sum = np.max(shifted) + np.log(np.sum(np.exp(shifted - np.max(shifted))))
        return (mc + log_sum / T) * dx
    def cons_eq(hv):
        return np.sum(hv) - n / 2.0
    def cons_jac(hv):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_eq, 'jac': cons_jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': ftol})
    return make_feasible(res.x, n)

def projected_gradient_step(h, grad, lr, n):
    h_new = h - lr * grad
    return make_feasible(h_new, n)

def compute_max_gradient(C, n, T):
    """Compute gradient of smoothed max correlation w.r.t. h."""
    dx = 2.0 / n
    M = np.max(C)
    shifted = T * (C - M)
    shifted -= np.max(shifted)
    exp_s = np.exp(shifted)
    w = exp_s / np.sum(exp_s)
    
    h = _pg_h
    n_c = 2 * n - 1
    # w_c[d] = w[d + n - 1] for d in [-(n-1), n-1], stored as array of length 2n-1
    # Compute grad_h[m] = dx * sum_d w[d+n-1] * ((1-h[m-d])*I[0<=m-d<n] - h[m+d]*I[0<=m+d<n])
    
    # Term A: sum_d w[d+n-1]*(1-h[m-d]) with 0<=m-d<n
    # = convolve(w_c, 1-h) evaluated at appropriate indices
    term_A = np.convolve(w, 1.0 - h, mode='full')  # length 3n-2
    # term_A[m] = sum_j w[j]*(1-h)[m-j+n-1]... 
    # Actually np.convolve(a,v)[k] = sum_j a[j]*v[k-j]
    # np.convolve(w, 1-h, 'full')[m] = sum_j w[j]*(1-h)[m-j]
    # We want sum_d w[d+n-1]*(1-h)[m-d] where d goes from -(n-1) to n-1
    # Let j = d+n-1: sum_j w[j]*(1-h)[m-j+n-1] where j from 0 to 2n-2
    # = sum_j w[j]*(1-h)[(m+n-1)-j] = np.convolve(w, 1-h, 'full')[m+n-1]
    # Wait: np.convolve(w, 1-h, 'full')[k] = sum_j w[j] * (1-h)[k-j]
    # So np.convolve(w, 1-h, 'full')[m+n-1] = sum_j w[j]*(1-h)[m+n-1-j]
    # = sum_j w[j]*(1-h)[m-(j-(n-1))]
    # Let d = j-(n-1): = sum_d w[d+n-1]*(1-h)[m-d]  YES!
    
    A = term_A[n-1:n-1+n]  # indices n-1 to 2n-2 give m=0 to n-1
    
    # Term B: sum_d w[d+n-1]*h[m+d] with 0<=m+d<n
    # = sum_j w[j-（n-1)]*h[m+j-(n-1)]... 
    # Let me think: sum_d w[d+n-1]*h[m+d], d from max(-(n-1),-m) to min(n-1, n-1-m)
    # = sum_i h[i]*w[i-m+n-1] where i=m+d, so d=i-m
    # = sum_i h[i]*w_c[i-m] where w_c[d]=w[d+n-1]
    # This is a correlation: np.correlate(h, w_c, 'full')?
    # np.correlate(a, v, 'full')[k] in numpy = sum_n a[n+k]*v[n] (for valid indices)
    # Actually let me use: sum_i h[i]*w[i-m+n-1]
    # Let's define rev_w where rev_w[j] = w[2n-2-j] (reversed w)
    # sum_i h[i]*w[i-m+n-1] = sum_i h[i]*rev_w[2n-2-(i-m+n-1)] = sum_i h[i]*rev_w[m+n-1-i]
    # = np.convolve(h, rev_w, 'full')[m+n-1]
    
    rev_w = w[::-1]
    term_B = np.convolve(h, rev_w, mode='full')  # length 3n-2
    B = term_B[n-1:n-1+n]
    
    grad_h = dx * (A - B)
    return grad_h

def binary_local_search(h, n, n_evals, rng):
    """Local search on a near-binary solution: try small continuous adjustments."""
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(n_evals):
        # Pick random position and try to push it toward 0 or 1
        i = rng.randint(n)
        if best[i] > 0.5:
            target = 0.0
        else:
            target = 1.0
        trial = best.copy()
        trial[i] = target
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c - 1e-14:
            best = trial
            best_c = c
    return best, best_c

def swap_local_search(h, n, n_swaps, rng):
    """Try swapping values at two positions to reduce C5."""
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(n_swaps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        trial = best.copy()
        trial[i], trial[j] = best[j], best[i]
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c - 1e-14:
            best = trial
            best_c = c
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def _time_left():
        return budget_s - (time.time() - start_time)
    
    best_c5 = float('inf')
    best_h = None
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: SLSQP at multiple resolutions
    for n_pts in [1024, 2048, 4096]:
        if _time_left() < 120:
            break
        
        h_curr = upsample_h(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
        
        # Temperature annealing with SLSQP
        for T, max_it in [(100.0, 300), (500.0, 250), (2000.0, 150), (5000.0, 100)]:
            if _time_left() < 30:
                break
            try:
                h_opt = _slsqp_smoothed(h_curr, n_pts, T, max_it)
                c5_opt = compute_c5(h_opt)
                h_curr = h_opt.copy()
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
        
        # Random restart at this resolution
        if _time_left() > 60:
            h_rand = 0.6 * upsample_h(best_h, n_pts) + 0.4 * rng.rand(n_pts)
            h_rand = make_feasible(h_rand, n_pts)
            try:
                h_opt = _slsqp_smoothed(h_rand, n_pts, 200.0, 200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
    
    # Phase 2: Projected gradient with momentum
    n_pg = min(len(best_h), 2048)
    h_pg = upsample_h(best_h, n_pg) if n_pg < len(best_h) else best_h.copy()
    global _pg_h
    
    momentum = np.zeros(n_pg)
    mu = 0.9
    for pg_iter in range(3000):
        if _time_left() < 30:
            break
        _pg_h = h_pg
        C = np.correlate(h_pg, 1.0 - h_pg, mode='full')
        c5_now = float(np.max(C) * (2.0 / n_pg))
        
        # Subgradient: push down the max lags
        M = np.max(C)
        active_mask = C > M - 0.5 * (2.0 / n_pg)  # within half a dx of max
        if np.sum(active_mask) == 0:
            active_mask = C == M
        
        # Simple subgradient: sum of gradients of active components
        grad = np.zeros(n_pg)
        n_c = 2 * n_pg - 1
        for k in range(int(np.sum(~active_mask)), int(np.sum(~active_mask)) + int(np.sum(active_mask))):
            if not active_mask[k]:
                continue
            d = k - (n_pg - 1)
            # dC[k]/dh[m] = I[0<=m-d<n_pg]*(1-h[m-d]) - I[0<=m+d<n_pg]*h[m+d]
            # Vectorized:
            if d >= 0:
                valid_m = np.arange(d, n_pg)
                grad[valid_m] += (1.0 - h_pg[valid_m - d])
                valid_m2 = np.arange(0, n_pg - d)
                grad[valid_m2] -= h_pg[valid_m2 + d]
            else:
                dd = -d
                valid_m = np.arange(0, n_pg - dd)
                grad[valid_m] += (1.0 - h_pg[valid_m + dd])
                valid_m2 = np.arange(dd, n_pg)
                grad[valid_m2] -= h_pg[valid_m2 - dd]
        
        grad *= (2.0 / n_pg)  # dx factor
        
        # Skip the expensive loop-based gradient and use it
        lr = 0.01 / (np.sum(active_mask) + 1)
        momentum = mu * momentum + lr * grad
        h_new = make_feasible(h_pg - momentum, n_pg)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5_now - 1e-15:
            h_pg = h_new
        else:
            lr *= 0.5
            momentum = mu * momentum + lr * grad
            h_new = make_feasible(h_pg - momentum, n_pg)
            if compute_c5(h_new) < c5_now - 1e-15:
                h_pg = h_new
    
    c5_pg = compute_c5(h_pg)
    if c5_pg < best_c5:
        best_c5 = c5_pg
        if n_pg < len(best_h):
            best_h = upsample_h(h_pg, len(best_h))
        else:
            best_h = h_pg.copy()
    
    # Phase 3: Coordinate descent polish
    n_cd = min(len(best_h), 2048)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    cd_best = compute_c5(h_cd)
    
    deltas = [0.005, -0.005, 0.01, -0.01, 0.02, -0.02, 0.05, -0.05, 0.1, -0.1]
    for cd_round in range(20):
        if _time_left() < 20:
            break
        improved = False
        for j in range(n_cd):
            if _time_left() < 20:
                break
            old_val = h_cd[j]
            for delta in deltas:
                candidate = old_val + delta
                if candidate < 0.0 or candidate > 1.0:
                    continue
                trial = h_cd.copy()
                trial[j] = candidate
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < cd_best - 1e-13:
                    cd_best = c
                    h_cd = trial
                    old_val = candidate
                    improved = True
            for bv in (0.0, 1.0):
                if abs(bv - h_cd[j]) < 1e-9:
                    continue
                trial = h_cd.copy()
                trial[j] = bv
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < cd_best - 1e-13:
                    cd_best = c
                    h_cd = trial
                    improved = True
        if not improved:
            break
    
    if cd_best < best_c5:
        best_c5 = cd_best
        if n_cd < len(best_h):
            best_h = upsample_h(h_cd, len(best_h))
        else:
            best_h = h_cd.copy()
    
    # Phase 4: Binary-style local search at moderate resolution
    n_bin = 1024
    h_bin = upsample_h(best_h, n_bin)
    # Threshold to near-binary
    h_bin_thresh = (h_bin > 0.5).astype(float)
    h_bin_thresh = make_feasible(h_bin_thresh, n_bin)
    c5_bin = compute_c5(h_bin_thresh)
    
    # Also try the continuous version
    h_bin_cont = h_bin.copy()
    c5_bin_cont = compute_c5(h_bin_cont)
    
    h_for_flip = h_bin_thresh if c5_bin < c5_bin_cont else h_bin_cont
    c5_for_flip = min(c5_bin, c5_bin_cont)
    
    n_flips = min(50000, int(_time_left() * 500))
    for _ in range(n_flips):
        if _time_left() < 15:
            break
        i = rng.randint(n_bin)
        h_for_flip[i] = 1.0 - h_for_flip[i]
        h_for_flip = make_feasible(h_for_flip, n_bin)
        c = compute_c5(h_for_flip)
        if c > c5_for_flip:
            h_for_flip[i] = 1.0 - h_for_flip[i]
            h_for_flip = make_feasible(h_for_flip, n_bin)
    c5_flip = compute_c5(h_for_flip)
    if c5_flip < best_c5:
        best_c5 = c5_flip
        best_h = upsample_h(h_for_flip, len(best_h)) if len(best_h) > n_bin else h_for_flip.copy()
    
    # Phase 5: Final SLSQP polish at high T
    if _time_left() > 20:
        n_final = len(best_h)
        try:
            h_opt = _slsqp_smoothed(best_h, n_final, 3000.0, 80)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
