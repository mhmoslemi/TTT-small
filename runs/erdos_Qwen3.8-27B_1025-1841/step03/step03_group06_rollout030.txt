import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_binary(h):
    """Compute C5 for a binary (or general) h vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _obj_sm(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_f(x, n):
    return np.sum(x) - n / 2.0

def _con_j(x, n):
    return np.ones(n)

def binary_local_search(h, n, rng, max_iters, time_limit, t0):
    """Greedy + annealing local search on binary configurations."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    best_c5 = current_max * 2.0 / n

    s_arr = np.arange(2 * n - 1) - (n - 1)
    temp = 0.5
    cool_rate = 0.9997
    no_improve = 0
    target_no_improve = 2000

    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        if it % 500 == 0 and it > 0:
            temp = max(0.01, temp * 0.95)

        worst_idx = np.argmax(corr)
        worst_s = int(s_arr[worst_idx])

        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        # Source: positions i with h[i]=1 where the flip would help
        if worst_s >= 0:
            src_mask = np.zeros(n, dtype=bool)
            valid_src = np.arange(max(0, n - worst_s))
            src_mask[valid_src] = (h[valid_src] == 1.0) & (h[valid_src + worst_s] == 0.0)
            sources = np.where(src_mask)[0]
            # Target: positions j with h[j]=0 where adding reduces overlap
            tgt_mask = np.zeros(n, dtype=bool)
            valid_tgt = np.arange(0, n - worst_s)
            tgt_mask[valid_tgt] = (h[valid_tgt] == 0.0) & (h[valid_tgt + worst_s] == 1.0)
            targets = np.where(tgt_mask)[0]
        else:
            ss = abs(worst_s)
            src_mask = np.zeros(n, dtype=bool)
            valid_src = np.arange(max(0, ss), n)
            src_mask[valid_src] = (h[valid_src] == 1.0) & (h[valid_src - ss] == 0.0)
            sources = np.where(src_mask)[0]
            tgt_mask = np.zeros(n, dtype=bool)
            valid_tgt = np.arange(0, n - ss)
            tgt_mask[valid_tgt] = (h[valid_tgt] == 0.0) & (h[valid_tgt + ss] == 1.0)
            targets = np.where(tgt_mask)[0]

        if len(sources) == 0 or len(targets) == 0:
            # Try random moves
            i = rng.choice(ones)
            j = rng.choice(zeros)
            s = worst_s
            delta = _corr_delta(h, i, j, s, n, s_arr)
            new_max = float(np.max(corr + delta))
            if new_max < current_max or (new_max > current_max and rng.random() < np.exp(-(new_max - current_max) / temp)):
                h[i] = 0.0
                h[j] = 1.0
                corr = corr + delta
                current_max = new_max
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
                    best_c5 = best_max * 2.0 / n
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
            continue

        # Try top candidates
        n_try = min(30, len(sources) * len(targets))
        if n_try > 0:
            if len(sources) <= 60 and len(targets) <= 60:
                idxs = [(sources[a], targets[b]) for a in range(min(len(sources), 15))
                        for b in range(min(len(targets), 15))][:n_try]
            else:
                idxs = [(rng.choice(sources), rng.choice(targets)) for _ in range(n_try)]

            best_delta_max = current_max
            best_pair = None
            best_d = None

            for (i, j) in idxs:
                s = worst_s
                d = _corr_delta(h, i, j, s, n, s_arr)
                nm = float(np.max(corr + d))
                if nm < best_delta_max - 1e-12:
                    best_delta_max = nm
                    best_pair = (i, j)
                    best_d = d

            if best_pair is not None:
                i, j = best_pair
                h[i] = 0.0
                h[j] = 1.0
                corr = corr + best_d
                current_max = best_delta_max
                if current_max < best_max:
                    best_max = current_max
                    best_h = h.copy()
                    best_c5 = best_max * 2.0 / n
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve >= target_no_improve:
            # Perturb: flip 3-5 random pairs
            n_flip = rng.randint(3, 6)
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) >= n_flip and len(zeros) >= n_flip:
                fi = rng.choice(ones, n_flip, replace=False)
                fj = rng.choice(zeros, n_flip, replace=False)
                for ii, jj in zip(fi, fj):
                    d = _corr_delta(h, ii, jj, int(s_arr[np.argmax(corr)]), n, s_arr)
                    h[ii] = 0.0
                    h[jj] = 1.0
                    corr = corr + d
                current_max = float(np.max(corr))
            no_improve = 0

    return best_h, best_c5

def _corr_delta(h, i, j, s, n, s_arr):
    """Compute delta in correlation vector after flipping h[i]:1->0, h[j]:0->1."""
    delta = np.zeros(2 * n - 1)
    # Term 1: -(1-h[i-s]) for valid i-s
    v1 = (i - s_arr >= 0) & (i - s_arr < n)
    if np.any(v1):
        idx = i - s_arr[v1]
        delta[v1] -= (1.0 - h[idx])
    # Term 2: +h[i+s] for valid i+s
    v2 = (i + s_arr >= 0) & (i + s_arr < n)
    if np.any(v2):
        idx = i + s_arr[v2]
        delta[v2] += h[idx]
    # Term 3: +(1-h[j-s]) for valid j-s
    v3 = (j - s_arr >= 0) & (j - s_arr < n)
    if np.any(v3):
        idx = j - s_arr[v3]
        delta[v3] += (1.0 - h[idx])
    # Term 4: -h[j+s] for valid j+s
    v4 = (j + s_arr >= 0) & (j + s_arr < n)
    if np.any(v4):
        idx = j + s_arr[v4]
        delta[v4] -= h[idx]
    return delta

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - t0)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary local search at original resolution
    h_bin = round_to_binary(h_init, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Multiple restarts of binary search
    n_restarts = 5
    ts = 0.9 * budget_s / n_restarts
    for r in range(n_restarts):
        if remaining() < 120:
            break
        if r == 0:
            h_start = round_to_binary(h_init, n)
        elif r == 1:
            # Perturb the parent
            h_pert = h_init + rng.randn(n) * 0.1
            h_start = round_to_binary(h_pert, n)
        elif r == 2:
            # Random balanced binary
            h_rand = np.zeros(n)
            ones_pos = rng.choice(n, n // 2, replace=False)
            h_rand[ones_pos] = 1.0
            h_start = h_rand
        elif r == 3:
            # Periodic pattern
            h_per = np.zeros(n)
            period = max(4, n // 10)
            for i in range(n):
                if (i // period) % 2 == 0:
                    h_per[i] = 1.0
            h_start = make_feasible(h_per, n)
            h_start = round_to_binary(h_start, n)
        else:
            # Use best found so far with perturbation
            h_pert = best_h + rng.randn(n) * 0.08
            h_start = round_to_binary(h_pert, n)

        h_res, c5_res = binary_local_search(h_start, n, rng,
                                            max_iters=2000000,
                                            time_limit=ts, t0=t0)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res.copy()

    # Phase 2: Try higher resolution
    if remaining() > 200:
        for n2 in [1920, 2880]:
            if remaining() < 150:
                break
            h_up = upsample_h(best_h, n2)
            h_bin2 = round_to_binary(h_up, n2)
            h_res2, c5_res2 = binary_local_search(h_bin2, n2, rng,
                                                  max_iters=1000000,
                                                  time_limit=0.5 * (remaining() - 100), t0=t0)
            if c5_res2 < best_c5:
                best_c5 = c5_res2
                best_h = h_res2.copy()

    # Phase 3: SLSQP polish on the best solution
    if remaining() > 80:
        n_opt = len(best_h)
        T_val = 200.0
        bounds = [(0.0, 1.0)] * n_opt
        cons = [{'type': 'eq', 'fun': lambda x: _con_f(x, n_opt),
                 'jac': lambda x: _con_j(x, n_opt)}]

        def obj_packed(x):
            return _obj_sm(x, n_opt, T_val)

        def con_packed(x):
            return _con_f(x, n_opt)

        def con_jac_packed(x):
            return _con_j(x, n_opt)

        cons2 = [{'type': 'eq', 'fun': con_packed, 'jac': con_jac_packed}]
        try:
            res = minimize(obj_packed, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons2,
                           options={'maxiter': 300, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_opt)
        except Exception:
            pass

    # Phase 4: Quick coordinate descent on final solution
    n_opt = len(best_h)
    for rd in range(30):
        if remaining() < 10:
            break
        improved = False
        for j in range(n_opt):
            if remaining() < 5:
                break
            old = best_h[j]
            best_c = best_c5
            best_v = old
            cands = [0.0, 1.0]
            if 0.0 < old < 1.0:
                cands += [max(0.0, old - 0.02), max(0.0, old - 0.01),
                          min(1.0, old + 0.01), min(1.0, old + 0.02)]
            for cv in cands:
                trial = best_h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n_opt)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = cv
            if best_v != old:
                best_h[j] = best_v
                best_h = make_feasible(best_h, n_opt)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
