def _subgrad_step(h, n, alpha):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Subgradient of max: indicator at argmax
    # grad of corr[k] w.r.t. h[i] for full correlation
    # For the max lag, compute gradient
    k_max = np.argmax(corr)
    # corr[k] = sum_i h[i] * (1-h)[i + k - (n-1)]
    # d corr[k]/d h[i] = (1-h)[i+k-(n-1)] - h[i+k-(n-1)] (for the second term product rule)
    # Actually: h[i]*(1-h[j]) where j = i+k-(n-1)
    # d/dh[i] = (1-h[j]) and d/dh[j] = -h[i]
    grad = np.zeros(n)
    shift = k_max - (n - 1)

        j = i + shift
        if 0 <= j < n:
            grad[i] += (1.0 - h[j])
            grad[j] -= h[i]

    # Project: clip to [0,1], then fix sum

        for T in [40.0, 120.0, 350.0]:
            if tl() < 50:
                break
            h_opt = _slsqp(h_hi, n2, 'sm', T=T, max_iter=200)
            c5o = compute_c5(h_opt)
            if c5o < best_c5:
                best_c5 = c5o
                best_h = h_opt.copy()
                h_hi = h_opt.copy()

        # Top-K at high res
        if tl() > 40:
            h_opt = _slsqp(h_hi, n2, 'topk', K=10, max_iter=150)
            c5o = compute_c5(h_opt)
            if c5o < best_c5:
                best_c5 = c5o
                best_h = h_opt.copy()

    # Phase 3: Random restart at n=2048
    if tl() > 100:
        n3 = 2048
        for attempt in range(3):
            if tl() < 40:
                break
            # Different initialization
            if attempt == 0:
                h_rand = 0.6 * upsample_h(best_h, n3) + 0.4 * rng.rand(n3)
            elif attempt == 1:
                h_rand = (rng.rand(n3) < 0.5).astype(float)
                # Blend with best
                h_rand = 0.6 * upsample_h(best_h, n3) + 0.4 * h_rand
            else:
                # Block-based init
                block = n3 // 16
                h_rand = np.zeros(n3)
                for b in range(16):
                    if b % 3 != 1:
                        h_rand[b*block:(b+1)*block] = 1.0
                h_rand = 0.7 * upsample_h(best_h, n3) + 0.3 * h_rand
            h_rand = make_feasible(h_rand, n3)
            h_opt = _slsqp(h_rand, n3, 'sm', T=80.0, max_iter=250)
            c5o = compute_c5(h_opt)
            if c5o < best_c5:
                best_c5 = c5o
                best_h = h_opt.copy()

    # Phase 4: Final polish
    n_final = len(best_h)
    if tl() > 25:
        h_opt = _slsqp(best_h, n_final, 'sm', T=500.0, max_iter=120)
        c5o = compute_c5(h_opt)
        if c5o < best_c5:
            best_c5 = c5o
            best_h = h_opt.copy()

    if tl() > 15:
        h_opt = _slsqp(best_h, n_final, 'topk', K=5, max_iter=80)
        c5o = compute_c5(h_opt)
        if c5o < best_c5:
            best_c5 = c5o
            best_h = h_opt.copy()

    best_h = make_feasible(best_h, len(best_h))
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
