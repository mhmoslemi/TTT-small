import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Convert to binary: threshold at 0.5, then adjust to get exactly n/2 ones."""
    b = (h >= 0.5).astype(float)
    target = n // 2
    current = int(b.sum())
    if current > target:
        # Need to flip some 1s to 0s
        one_indices = np.where(b == 1.0)[0]
        # Flip the ones with smallest h value
        order = np.argsort(h[one_indices])[::-1]  # actually flip the smallest
        order = np.argsort(h[one_indices])
        for idx in order[:current - target]:
            b[one_indices[idx]] = 0.0
    elif current < target:
        # Need to flip some 0s to 1s
        zero_indices = np.where(b == 0.0)[0]
        order = np.argsort(h[zero_indices])[::-1]
        for idx in order[:target - current]:
            b[zero_indices[idx]] = 1.0
    return b
