    target = n / 2.0
    lo, hi = -2.0, 3.0  # range for lambda
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
        if abs(s - target) < 1e-14:
            break
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any remaining tiny error
    diff = target - np.sum(h)
    if abs(diff) > 0:
        # Adjust the entry closest to 0.5
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
    return h

    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgradient_at_shift(h, shift):
    """Compute subgradient of corr[k] w.r.t. h, where k - (n-1) = shift."""
    n = len(h)
    grad = np.zeros(n)
    if shift >= 0:
        m = shift
        if m == 0:
            # corr = sum h[i]*(1-h[i]), grad[i] = 1 - 2*h[i]
            grad = 1.0 - 2.0 * h
        else:
            # For i in [0, m-1]: grad[i] = -h[i+m]
            grad[:m] = -h[m:]
            # For i in [m, n-1-m]: grad[i] = -h[i+m] + (1-h[i-m])
            if m < n - 1 - m:
                grad[m:n-m] = -h[2*m:n] + (1.0 - h[:n-2*m])
            elif m == n - 1 - m:
                grad[m] = -h[2*m] + (1.0 - h[0])
            # For i in [n-m, n-1]: grad[i] = (1-h[i-m])
            grad[n-m:] = 1.0 - h[:m]
    else:
        m = -shift
        if m == 0:
            grad = 1.0 - 2.0 * h
        else:
            # For i in [0, m-1]: grad[i] = 1 - h[i+m]
            grad[:m] = 1.0 - h[m:2*m]
            # For i in [m, n-1-m]: grad[i] = 1 - h[i+m] - h[i-m]
            if m < n - 1 - m:
                grad[m:n-m] = 1.0 - h[2*m:n] - h[:n-2*m]
            elif m == n - 1 - m:
                grad[m] = 1.0 - h[2*m] - h[0]
            # For i in [n-m, n-1]: grad[i] = -h[i-m]
            grad[n-m:] = -h[:m]
    return grad

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0

        h[idx] += diff
    return h

def softmax_obj(h, T):

    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def softmax_grad(h, T):
    """Compute gradient of softmax objective."""
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    shifted = corr - mc
    exp_val = np.exp(T * shifted)
    total = np.sum(exp_val)
    weights = exp_val / total  # probability distribution over shifts
    
    grad = np.zeros(n)
    for k in range(len(corr)):

            continue
        shift = k - (n - 1)
        grad += weights[k] * subgradient_at_shift(h, shift)
    return grad * dx  # Wait, the objective is (mc + logsumexp/T) * dx

Wait, I need to be more careful with the gradient. The objective is:
f(h) = (mc + (1/T) * log(sum_k exp(T * corr_k))) * dx

where mc = max_k corr_k is itself a function of h. For the softmax approximation, we can ignore the mc term (it's just a constant shift for gradient purposes) and focus on:
g(h) = (1/T) * log(sum_k exp(T * corr_k)) * dx

∂g/∂h_j = (dx/T) * sum_k exp(T*corr_k) * T * ∂corr_k/∂h_j / sum_j exp(T*corr_j)
= dx * sum_k p_k * ∂corr_k/∂h_j

where p_k = exp(T*corr_k) / sum_j exp(T*corr_j)

This is correct. But computing this naively requires O(n) subgradient computations, each O(n), for a total of O(n^2) per gradient evaluation. For n=960, that's about 1M operations per gradient, which is fine.

But wait, we only need the top shifts. If T is large, most p_k will be negligible. We can truncate to the top M shifts.

Actually, for the subgradient approach, we don't even need the full softmax. We can just use:
- Find the top M shifts (those within some threshold of the max)
- Average their subgradients (or weight by exponential)
- Take a step

   - This should converge to a point where all top shifts are balanced
