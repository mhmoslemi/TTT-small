    n = len(x)

    # Soft max
    max_corr = np.max(corr)
    exps = np.exp(T * (corr - max_corr))
    S = np.sum(exps)
    p = exps / S  # softmax weights
    f = dx * (max_corr + np.log(S) / T)
    
    # Gradient
    # p[k] * dcorr[k]/dx[j]
    # dcorr[k]/dx[j] = (1-x[j+s_k])*1[j in range_k] - x[j-s_k]*1[j-s_k in range_k]
    # where s_k = k - (n-1)
    grad = np.zeros(n)
    for k in range(len(corr)):
        if p[k] < 1e-15:
            continue
        s = k - (n - 1)
        # Term 1: (1 - x[j+s]) for j in [max(0,-s), min(n-1, n-1-s)]
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)

            js_hi = j_hi + s

    n = len(h)
    for step in range(n_steps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        # Find shifts achieving near-max
        thresh = mc - 0.005 * mc
        top_k = np.where(corr >= thresh)[0]
        # Pick the worst
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        # Subgradient of corr[k_max] at h
        grad = np.zeros(n)
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        grad[j_lo:j_hi+1] += (1.0 - h[j_lo+s:j_hi+s+1])
        j_lo2 = max(0, s)
        j_hi2 = min(n - 1, s + min(n - 1, n - 1 - s))

import numpy as np
from scipy.optimize import minimize
import time

_rng = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_fg(x, T, top_k=100):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    S = np.sum(exps)
    p = exps / S
    f = dx * (mc + np.log(S) / T)
    if top_k < len(p):
        idx = np.argsort(-p)[:top_k]
    else:
        idx = np.where(p > 1e-12)[0]
    grad = np.zeros(n)
    for k in idx:
        pk = p[k]
        s = k - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n - 1, n - 1 - s)
        if j_lo <= j_hi:
            grad[j_lo:j_hi + 1] += pk * (1.0 - x[j_lo + s:j_hi + s + 1])
        j_lo2 = max(0, s)
        j_hi2 = min(n - 1, s + min(n - 1, n - 1 - s))
        if j_lo2 <= j_hi2:
            grad[j_lo2:j_hi2 + 1] -= pk * x[j_lo2 - s:j_hi2 - s + 1]
    grad *= dx
    return f, grad

def penalty_fg(x, T, lam, n, top_k=100):
    f, g = soft_max_fg(x, T, top_k)
    s = np.sum(x) - n / 2.0
    f += lam * s * s
    g = g + 2.0 * lam * s
    return f, g

def penalty_jac(x, T, lam, n, top_k=100):
    _, g = soft_max_fg(x, T, top_k)
    s = np.sum(x) - n / 2.0
    return g + 2.0 * lam * s

def gradient_chase(h, n_total_steps, batch_size, rng):
    n = len(h)
    h = h.copy()
    n_batches = n_total_steps // batch_size + 1
    for batch in range(n_batches):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        thresh = mc - 0.02 * mc
        top_idx = np.where(corr >= thresh)[0]
        if len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(corr))])
        for step in range(batch_size):
            k = int(top_idx[rng.randint(len(top_idx))])
            s = k - (n - 1)
            dstep = 0.003 * (1.0 - 0.0001 * (batch * batch_size + step))
            grad = np.zeros(n)
            j_lo = max(0, -s)
            j_hi = min(n - 1, n - 1 - s)
            if j_lo <= j_hi:
                grad[j_lo:j_hi + 1] = (1.0 - h[j_lo + s:j_hi + s + 1])
            j_lo2 = max(0, s)
            j_hi2 = min(n - 1, s + min(n - 1, n - 1 - s))
            if j_lo2 <= j_hi2:
                grad[j_lo2:j_hi2 + 1] -= h[j_lo2 - s:j_hi2 - s + 1]
            h -= dstep * grad
            h = np.clip(h, 0, 1)
            diff = n / 2.0 - np.sum(h)
            if abs(diff) > 1e-10:
                if diff > 0:
                    mask = h < 1.0
                else:
                    mask = h > 0.0
                cnt = float(np.sum(mask))
                if cnt > 0:
                    h[mask] += diff / cnt
                    h = np.clip(h, 0, 1)
    return make_feasible(h, n)

def coord_descent(h, n, n_passes=3, time_limit_s=60, start_time=None):
    if start_time is None:
        start_time = time.time()
    h = h.copy()
    best_c = compute_c5(h)
    for p in range(n_passes):
        for j in range(n):
            if time.time() - start_time > start_time + time_limit_s:
                break
            old_val = h[j]
            best_val = old_val
            for delta in [0.01, 0.005, 0.002, 0.05]:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * delta
                    if 0.0 < v < 1.0:
                        trial = h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = compute_c5(h)
        for j in range(n):
            if time.time() - start_time > start_time + time_limit_s:
                break
            old_val = h[j]
            for val in [0.0, 1.0]:
                if abs(old_val - val) < 1e-8:
                    continue
                trial = h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    h = trial
    return h, best_c
