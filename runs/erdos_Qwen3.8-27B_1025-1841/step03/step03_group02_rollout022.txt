def flip_delta(h, p, n):
    """Compute the change in correlation array if we flip h[p]."""
    # h is 0/1 valued
    # m ranges from -(n-1) to (n-1), k = m + n - 1
    delta = np.zeros(2*n-1)
    if h[p] == 1:  # flipping 1 to 0
        # delta[k] = -1 + h[p+m] + h[p-m] where valid
        for m in range(-(n-1), n):
            k = m + n - 1
            # Check validity
            if 0 <= p + m < n:  # p+m is valid (second factor position)
                delta[k] += (h[p+m] - 1)  # -(1-h[p+m]) = h[p+m]-1
            if 0 <= p - m < n:  # p-m is valid (first factor position)
                delta[k] += h[p-m]  # +h[p-m]
    else:  # flipping 0 to 1
        for m in range(-(n-1), n):

            if 0 <= p - m < n:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def swap_delta(h, p, q, n, m_arr):
    delta = np.zeros(2 * n - 1)
    pm = p + m_arr
    v = (pm >= 0) & (pm < n)
    delta[v] -= (1.0 - h[pm[v]])
    pmm = p - m_arr
    v2 = (pmm >= 0) & (pmm < n)
    delta[v2] += h[pmm[v2]]
    qm = q + m_arr
    v3 = (qm >= 0) & (qm < n)
    delta[v3] += (1.0 - h[qm[v3]])
    qmm = q - m_arr
    v4 = (qmm >= 0) & (qmm < n)
    delta[v4] -= h[qmm[v4]]
    return delta

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

    while True:

        if remaining < 3:
            break

        improved_this_pass = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        for ii in range(len(ones)):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            i = ones[ii]

            hi = min(n, i + window + 1)

            for j in candidates:
                d = swap_delta(h, i, j, n, m_arr)
                new_max = float(np.max(corr + d))
                if new_max < curr_max - 1e-15:
                    h[i] = 0
                    h[j] = 1
                    corr += d
                    curr_max = new_max
                    improved_this_pass = True
                    if curr_max < best_max:
                        best_max = curr_max
                        best_h = h.copy()
                    ones = np.where(h == 1)[0]
