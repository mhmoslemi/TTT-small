            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, oj)
            zeros = np.append(zeros, oi)
            if c_new < best_c:
                best_c = c_new
                best = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best, best_c

def slsqp_polish(h_start, n, T, max_iter):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    def con(h):
        return np.sum(h) - n / 2.0
    def jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def coord_descent(h, n, rounds, rng):
    h = h.copy()
    best_c = compute_c5(h)
    for _ in range(rounds):
        improved = False
        order = rng.permutation(n)
        for j in order:
            ov = h[j]
            for d in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.0, 1.0]:
                nv = ov + d
                if abs(d) < 1e-10:
                    nv = 0.0 if ov > 0.5 else 1.0
                if nv < 0 or nv > 1:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < best_c - 1e-14:
                    h = tr
                    best_c = c
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = np.tile([1.0, 0.0], 320)
    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: L-BFGS-B at parent resolution with multiple temperatures
    n1 = n0
    for T in [30, 80, 200, 500, 1200]:
        if tl() < 30:
            break
        h_init = best_h.copy() + 0.1 * rng.randn(n1)
        h_init = make_feasible(h_init, n1)
        try:
            h_opt = lbgfb_smooth(h_init, n1, T, mu=800.0, max_iter=200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass
        if tl() < 15:
            break

    # Phase 2: Subgradient descent on top-K lags
    if tl() > 120:
        for K in [3, 5, 8]:
            if tl() < 60:
                break
            h_sg, c5_sg = subgrad_topk(best_h, n1, K, steps=800, alpha0=0.02, alpha1=0.001)
            if c5_sg < best_c5:
                best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 3: Binary optimization with simulated annealing
    if tl() > 150:
        h_bin = (best_h > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_target = n1 // 2
        if n_ones > n_target:
            one_idx = np.where(h_bin == 1.0)[0]
            for k in range(n_ones - n_target):
                h_bin[one_idx[k]] = 0.0
        elif n_ones < n_target:
            zero_idx = np.where(h_bin == 0.0)[0]
            for k in range(n_target - n_ones):
                h_bin[zero_idx[k]] = 1.0
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

        # Multiple SA restarts
        for sa_round in range(3):
            if tl() < 50:
                break
            h_sa, c5_sa = binary_sa(h_bin, n1, rng, max_iters=8000, T0=0.02, T1=0.0005)
            if c5_sa < best_c5:
                best_c5, best_h = c5_sa, h_sa.copy()
            h_bin = h_bin.copy()

    # Phase 4: L-BFGS-B around binary
    if tl() > 80:
        h_soft = h_bin + 0.05 * rng.randn(n1)
        h_soft = make_feasible(h_soft, n1)
        for T in [50, 200, 600]:
            if tl() < 25:
                break
            try:
                h_opt = lbgfb_smooth(h_soft, n1, T, mu=500.0, max_iter=200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                h_soft = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Multi-resolution refinement
    for n_pts in [1024, 2048, 4096]:
        if tl() < 60:
            break
        h_c = upsample_h(best_h, n_pts)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()

        # Subgradient at higher resolution
        if tl() > 50:
            h_sg, c5_sg = subgrad_topk(h_c, n_pts, 5, steps=500, alpha0=0.015, alpha1=0.001)
            if c5_sg < best_c5:
                best_c5, best_h = c5_sg, h_sg.copy()
                h_c = h_sg.copy()

        # L-BFGS at higher resolution
        for T in [60, 200, 600, 1500]:
            if tl() < 20:
                break
            try:
                h_opt = lbgfb_smooth(h_c, n_pts, T, mu=300.0, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass

        # Binary at higher resolution
        if tl() > 60:
            h_b2 = (h_c > 0.5).astype(float)
            n_ones2 = int(np.sum(h_b2))
            n_tgt2 = n_pts // 2
            if n_ones2 > n_tgt2:
                oi = np.where(h_b2 == 1.0)[0]
                for k in range(n_ones2 - n_tgt2):
                    h_b2[oi[k]] = 0.0
            elif n_ones2 < n_tgt2:
                zi = np.where(h_b2 == 0.0)[0]
                for k in range(n_tgt2 - n_ones2):
                    h_b2[zi[k]] = 1.0
            h_sa2, c5_sa2 = binary_sa(h_b2, n_pts, rng, max_iters=5000, T0=0.015, T1=0.0003)
            if c5_sa2 < best_c5:
                best_c5, best_h = c5_sa2, h_sa2.copy()

    # Phase 6: Final coordinate descent + SLSQP polish
    n_cd = min(len(best_h), 2048)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    if tl() > 30:
        h_cd, c5_cd = coord_descent(h_cd, n_cd, rounds=4, rng=rng)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    if tl() > 20:
        n_f = min(len(best_h), 1024)
        h_f = upsample_h(best_h, n_f) if n_f < len(best_h) else best_h.copy()
        try:
            h_opt = slsqp_polish(h_f, n_f, 800, max_iter=100)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass

    # Final feasibility and evaluation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
