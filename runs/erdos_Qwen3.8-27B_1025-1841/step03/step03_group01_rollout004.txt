from scipy.optimize import minimize
from scipy.fft import fft, ifft

def compute_c5_fft(h):
    """Fast C5 computation using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = fft(h, N)
    GH = fft(1.0 - h, N)
    corr = ifft(np.conj(H) * GH).real
    return float(np.max(corr) * (2.0 / n))

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)

    return N

def compute_c5_fft(h):

    N = _fft_next_pow2(2 * n - 1)

            for p in range(8):
                if not time_ok(15):
                    break
                h_cd, imp, c5_cd = cd_pass(best_h if len(best_h) == n_h else upsample_h(best_h, n_h),
                                           n_h if len(best_h) == n_h else n_target,
                                           rng, lambda: time_ok(10))
                nt = n_h if len(h_cd) == n_h else n_target
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                if not imp:
                    break

        if time_ok(40):
            h_opt3 = slsqp_opt(best_h, len(best_h), 300.0, 80)
            c5_opt3 = compute_c5_fft(h_opt3)
            if c5_opt3 < best_c5:
                best_c5 = c5_opt3
                best_h = h_opt3.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5_fft(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
