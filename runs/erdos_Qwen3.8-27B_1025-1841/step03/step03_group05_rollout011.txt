import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def correlations(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fista(h, n_iter, T0, T1, lr, check):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.3 * f)
        corr = correlations(z)
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            g = subgrad_one(z, int(np.argmax(corr)) - (n-1))
        else:
            w = exps / tot
            # Use top-k for efficiency
            idx = np.argsort(w)[::-1][:100]
            g = np.zeros(n)
            for i in idx:
                wi = w[i]
                if wi < 1e-14:
                    break
                g += wi * subgrad_one(z, i - (n-1))
        xn = project_c(z - lrt * g * (2.0/n), n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

# Helper functions (top-level)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fista(h, n_iter, T0, T1, lr_init, lr_end, check):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr_init + (lr_end - lr_init) * f
        corr = compute_corrs(z)
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            g = subgrad_one(z, int(np.argmax(corr)) - (n-1)) * dx
        else:
            w = exps / tot
            idx = np.where(w > 1e-10)[0]
            g = np.zeros(n)
            for i in idx:
                g += w[i] * subgrad_one(z, i - (n-1))
            g *= dx
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def lbfgsb_smooth(h, T, mu, maxiter=300, time_limit=60):
    n = len(h)
    dx = 2.0 / n
    t_start = time.time()
    
    def obj(v):
        if time.time() - t_start > time_limit:
            return 1e10
        c = compute_corrs(v)
        mc = np.max(c)
        sm = mc + np.log(np.sum(np.exp(T*(c-mc))))/T
        penalty = mu * (np.sum(v) - n/2.0)**2
        return sm * dx + penalty
    
    def grad(v):
        if time.time() - t_start > time_limit:
            return np.zeros(n)
        c = compute_corrs(v)
        mc = np.max(c)
        exps = np.exp(T*(c-mc))
        w = exps / np.sum(exps)
        g = np.zeros(n)
        for i in range(len(c)):
            if w[i] > 1e-12:
                g += w[i] * subgrad_one(v, i-(n-1))
        g *= dx
        g += 2*mu*(np.sum(v)-n/2.0)*np.ones(n)
        return g
    
    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B',
                     bounds=[(0,1)]*n,
                     options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        hx = project_c(r.x, n)
        return hx
    except Exception:
        return h

def upsample(h, nn):
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h, kind='cubic'), nn)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def compute_corrs(h):
    return np.correlate(h, 1.0 - h, mode='full')

def fista(h, n_iter, T0, T1, lr_init, lr_end, check):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr_init + (lr_end - lr_init) * f
        corr = compute_corrs(z)
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        if tot < 1e-300:
            g = subgrad_one(z, int(np.argmax(corr)) - (n - 1)) * dx
        else:
            w = exps / tot
            g = np.zeros(n)
            thresh = 1e-12
            idx = np.where(w > thresh)[0]
            for i in idx:
                g += w[i] * subgrad_one(z, i - (n - 1))
            g *= dx
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def topM_balance(h, n_iter, M=20, lr_start=0.04, lr_end=0.003):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        corr = compute_corrs(x)
        top_idx = np.argsort(corr)[::-1][:M]
        top_vals = corr[top_idx]
        tmax = top_vals[0]
        w = np.exp(40.0 * (top_vals - tmax))
        w /= np.sum(w)
        g = np.zeros(n)
        for i in range(M):
            g += w[i] * subgrad_one(x, top_idx[i] - (n - 1))
        g *= dx
        gn = np.linalg.norm(g)
        if gn < 1e-15:
            break
        f = it / max(n_iter - 1, 1)
        lr = lr_start + (lr_end - lr_start) * f
        xn = project_c(x - lr * g / gn, n)
        c = compute_c5(xn)
        if c < bc:
            bc, bh = c, xn.copy()
        x = xn
    return bh, bc

def lbfgsb_smooth(h, T, mu, maxiter=200, time_limit=80):
    n = len(h)
    dx = 2.0 / n
    t_start = time.time()
    timeout = [False]

    def obj(v):
        if timeout[0]:
            return 1e10
        c = compute_corrs(v)
        mc = np.max(c)
        sm = mc + np.log(np.sum(np.exp(T * (c - mc)))) / T
        pen = mu * (np.sum(v) - n / 2.0) ** 2
        return sm * dx + pen

    def grad(v):
        if timeout[0]:
            return np.zeros(n)
        c = compute_corrs(v)
        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        tot = np.sum(exps)
        w = exps / tot
        g = np.zeros(n)
        for i in range(len(c)):
            if w[i] > 1e-12:
                g += w[i] * subgrad_one(v, i - (n - 1))
        g *= dx
        g += 2.0 * mu * (np.sum(v) - n / 2.0)
        if time.time() - t_start > time_limit:
            timeout[0] = True
        return g

    try:
        r = minimize(obj, h, jac=grad, method='L-BFGS-B',
                     bounds=[(0.0, 1.0)] * n,
                     options={'maxiter': maxiter, 'ftol': 1e-16,
                              'gtol': 1e-10, 'maxfun': 500})
        if not timeout[0]:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def upsample(h, nn):
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)
