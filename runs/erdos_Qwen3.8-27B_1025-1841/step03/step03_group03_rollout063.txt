def make_feasible(h, n):
    target = n / 2.0
    for _ in range(10):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

            h[mask] = np.clip(h[mask] + adj, 0.0, 1.0)
    return h

    dx = 2.0 / n

def soft_obj(h, n, t, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    val = (cm + np.log(np.sum(np.exp(t * (c - cm)))) / t) * dx
    if mu > 0:
        s = h.sum() - n / 2.0
        val += 0.5 * mu * s * s
    return val

def soft_grad(h, n, t, mu):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = c.max()
    e = np.exp(t * (c - cm))
    e /= e.sum()
    g = np.zeros(n)
    nz = np.where(e > 1e-14)[0]
    for i in nz:
        k = i - (n - 1)
        if k >= 0:
            g[k:] += e[i] * (1.0 - h[:n-k])
            g[:n-k] -= e[i] * h[k:]
        else:
            g[:n+k] += e[i] * (1.0 - h[-k:])
            g[-k:] -= e[i] * h[:n+k]
    g *= dx
    if mu > 0:
        g += mu * (h.sum() - n / 2.0) / n
    return g

    def f(x):
        return soft_obj(x, n, t, mu)
    def g(x):
        return soft_grad(x, n, t, mu)
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-9, 'maxls': 30})
        if res.x is not None and res.x.shape == (n,):

    except Exception:
        pass
    return make_feasible(h0, n)

    def g(x):
        return soft_grad(x, n, t, 0.0)

                       bounds=[(0.0, 1.0)] * n, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and res.x.shape == (n,):

    except Exception:
        pass

def coord_desc(h, n, rounds, deadline):

            best_c, best_h = c_cd, h_cd

    # Phase 5: Binary local search
    if time.time() < deadline - 40:
        hb = (best_h > 0.5).astype(float)
        hb = make_feasible(hb, len(hb))
        c_b = compute_c5(hb)
        if c_b < best_c + 0.05:
            hb, c_b = binary_local_search(hb, len(hb), 2000, deadline, rng)
            if c_b < best_c:
                best_c, best_h = c_b, hb
                h_cd2, c_cd2 = coord_desc(best_h, len(best_h), 1, deadline)
                if c_cd2 < best_c:
                    best_c, best_h = c_cd2, h_cd2

    # Phase 6: Try n=4096 if time allows
    if time.time() < deadline - 100 and len(best_h) < 4096:
        n3 = 4096
        h3 = upsample(best_h, n3)
        for t, mi in [(300, 400), (1000, 300)]:
            if time.time() > deadline - 30:
                break
            h_new = lbfgsb_opt(h3, n3, t, mi, deadline, mu=3000.0)
            c_new = compute_c5(h_new)
            h3 = h_new
            if c_new < best_c:
                best_c, best_h = c_new, h_new

    # Final coord descent
    if time.time() < deadline - 15:
        h_f, c_f = coord_desc(best_h, len(best_h), 1, deadline)
        if c_f < best_c:
            best_c, best_h = c_f, h_f

    best_h = make_feasible(best_h, len(best_h))
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    return best_h, best_c, len(best_h)
