        N = 1 << (2*n - 1).bit_length()

    G = np.fft.rfft(g, N)
    # np.correlate(h, g, 'full')[j] = sum_i h[i + j - (len(g)-1)] * g[i]
    # Using: ifft(fft(h) * conj(fft(g))) gives circular cross-correlation
    # r[k] = sum_i h[(i+k) mod N] * g[i]
    # For k=0,...,n-1: r[k] = sum_i h[i+k]*g[i] (linear, no wraparound) = correlate at shift k
    # For k=N-(n-1),...,N-1: r[k] = correlate at shift k-N (negative shifts)
    r = np.fft.irfft(H * np.conj(G), N)
    # Build full correlation vector of length 2n-1
    # Index in full output: j = shift + (n-1)
    # shift ranges from -(n-1) to n-1
    # For shift = -(n-1): r[N-(n-1)]
    # For shift = 0: r[0]
    # For shift = n-1: r[n-1]
    return np.concatenate([r[N-(n-1):], r[:n]])

import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(h, g):
    n = len(h)
    N = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    r = np.fft.irfft(H * np.conj(G), N)
    return np.concatenate([r[N-n+1:], r[:n]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return float(np.max(corr) * dx)

def project_c(h, n):
    # Project to {h: 0<=h<=1, sum(h)=n/2}
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_shift(h, shift):
    """Subgradient of c_shift at h, where c_shift = sum_i h[i+shift]*(1-h[i])."""
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        # c = sum_{i=0}^{n-s-1} h[i+s]*(1-h[i])
        # d(c)/d(h[j]) = (1-h[j-s]) if s<=j<n, minus h[j+s] if 0<=j<n-s
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        # shift = -s, c = sum_{i=s}^{n-1} h[i-s]*(1-h[i])
        # d(c)/d(h[j]) = (1-h[j+s]) if j+s<n, i.e., j<n-s... 
        # Actually: c = sum_{i=s}^{n-1} h[i-s]*(1-h[i])
        # Let j = i-s, so i = j+s: c = sum_{j=0}^{n-s-1} h[j]*(1-h[j+s])
        # d(c)/d(h[j]) = (1-h[j+s]) for j=0,...,n-s-1, minus h[j-s] for j=s,...,n-1
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def fista_fft(h, n_iter=5000, T0=30.0, T1=0.05, lr=0.1, top_k=50, check=20):
    """FISTA with FFT-accelerated correlation and softmax-smoothed gradient."""
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.3 * f)
        
        corr = fft_correlate(z, 1.0 - z)
        mc = np.max(corr)
        
        # Softmax weights
        if T > 1.0:
            exps = np.exp(T * (corr - mc))
            tot = np.sum(exps)
            w = exps / tot
            # Use top-k for efficiency
            if top_k < len(w):
                idx = np.argsort(w)[::-1][:top_k]
                w_other = w.copy()
                w[:] = 0.0
                w[idx] = w_other[idx]
                w /= np.sum(w)
            
            # Compute weighted subgradient
            g = np.zeros(n)
            threshold = np.sum(w) / len(w) * 0.1  # Only process significant weights
            for i in np.where(w > threshold)[0]:
                g += w[i] * subgrad_shift(z, i - (n - 1))
        else:
            # Hard: use argmax
            istar = np.argmax(corr)
            g = subgrad_shift(z, istar - (n - 1))
        
        g *= dx
        
        # FISTA update
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    
    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

def subgrad_method(h, n_iter=20000, lr0=0.5, decay=0.999, check=50):
    """Pure subgradient method with the single worst shift."""
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    best_c = compute_c5(x)
    best_h = x.copy()
    
    for it in range(n_iter):
        corr = fft_correlate(x, 1.0 - x)
        mc = np.max(corr)
        istar = np.argmax(corr)
        s = istar - (n - 1)
        
        g = subgrad_shift(x, s) * dx
        lr = lr0 / (1.0 + decay * it * 0.01)
        
        x = project_c(x - lr * g, n)
        
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    
    c = compute_c5(x)
    if c < best_c:
        best_c = c
        best_h = x.copy()
    return best_h, best_c

def slsqp_smooth(h, T=5.0, maxiter=400):
    """SLSQP on the smoothed objective."""
    n = len(h)
    dx = 2.0 / n
    
    def obj(v):
        corr = fft_correlate(v, 1.0 - v)
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc))))) * dx / T
    
    def obj_grad(v):
        corr = fft_correlate(v, 1.0 - v)
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        tot = np.sum(exps)
        w = exps / tot
        g = np.zeros(n)
        # Only process top-k
        k = min(30, n)
        idx = np.argsort(w)[::-1][:k]
        for i in idx:
            if w[i] < 1e-10:
                break
            g += w[i] * subgrad_shift(v, i - (n - 1))
        return g * dx
    
    def con(v):
        return np.sum(v) - n / 2.0
    
    r = minimize(obj, h, jac=obj_grad, method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': con, 'jac': lambda v: np.ones(n)}],
                 options={'maxiter': maxiter, 'ftol': 1e-14})
    if r.success or r.x is not None:
        hx = project_c(r.x, n)
        if compute_c5(hx) < compute_c5(h):
            return hx
    return h

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)

def fft_corr(h, g):
    """Fast FFT-based cross-correlation matching np.correlate(h, g, 'full')."""
    n = len(h)
    N = 1 << (2 * n - 1).bit_length()
    r = np.fft.irfft(np.fft.rfft(h, N) * np.conj(np.fft.rfft(g, N)), N)
    return np.concatenate([r[N - n + 1:], r[:n]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sg_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

    dx = 2.0 / n
    x = h.astype(float).copy()

    tk = 1.0

    best_h = x.copy()
    for it in range(n_iter):

        T = T0 * (T1 / T0) ** f

            e = np.exp(T * (corr - mc))

        g *= dx / k
        lr = lr0 / (1.0 + 0.0005 * it)
        x = project_c(x - lr * g, n)
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c, best_h = c, x.copy()
    c = compute_c5(x)
    if c < best_c:
        best_c, best_h = c, x.copy()
    return best_h, best_c

def single_subgrad(h, n_iter=15000, lr0=0.4, check=50):
    n = len(h)
    dx = 2.0 / n
    x = h.astype(float).copy()
    best_c = compute_c5(x)
    best_h = x.copy()
    for it in range(n_iter):
        corr = fft_corr(x, 1.0 - x)
        istar = corr.argmax()
        g = sg_shift(x, istar - (n - 1)) * dx
        lr = lr0 / (1.0 + 0.001 * it)
        x = project_c(x - lr * g, n)
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c, best_h = c, x.copy()
    c = compute_c5(x)
    if c < best_c:
        best_c, best_h = c, x.copy()
    return best_h, best_c

def slsqp_smooth(h, T=5.0, maxiter=300):
    n = len(h)
    dx = 2.0 / n
    def obj(v):
        corr = fft_corr(v, 1.0 - v)
        mc = corr.max()
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    def objg(v):
        corr = fft_corr(v, 1.0 - v)
        mc = corr.max()
        e = np.exp(T * (corr - mc))
        w = e / e.sum()
        g = np.zeros(n)
        idx = np.argsort(w)[::-1][:min(40, n)]
        for i in idx:
            if w[i] < 1e-10:
                break
            g += w[i] * sg_shift(v, i - (n - 1))
        return g * dx
    def con(v):
        return np.sum(v) - n / 2.0
    try:
        r = minimize(obj, h, jac=objg, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con, 'jac': lambda v: np.ones(n)}],
                     options={'maxiter': maxiter, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = project_c(rng.rand(960), 960)

    n0 = len(h0)
    best_c = compute_c5(h0)
    best_h = h0.copy()
    print(f"Init: n={n0} C5={best_c:.6f}", flush=True)

    # Phase 1: FISTA broad at n0
    if rem() > 300:
        h1, c1 = fista(best_h, 4000, 50.0, 1.0, 0.12, 80, 30)
        if c1 < best_c:
            best_c, best_h = c1, h1
        print(f"P1: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 2: FISTA exploit at n0
    if rem() > 250:
        h2, c2 = fista(best_h, 5000, 2.0, 0.02, 0.05, 30, 20)
        if c2 < best_c:
            best_c, best_h = c2, h2
        print(f"P2: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 3: Top-k subgradient exploit
    if rem() > 180:
        h3, c3 = topk_subgrad(best_h, 8000, 5, 0.3, 50)
        if c3 < best_c:
            best_c, best_h = c3, h3
        print(f"P3: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 4: Single worst shift subgradient
    if rem() > 120:
        h4, c4 = single_subgrad(best_h, 10000, 0.4, 50)
        if c4 < best_c:
            best_c, best_h = c4, h4
        print(f"P4: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 5: SLSQP at n0
    if n0 <= 1024 and rem() > 60:
        for T in [8.0, 3.0, 1.5]:
            if rem() < 25:
                break
            h5 = slsqp_smooth(best_h, T, 250)
            c5v = compute_c5(h5)
            if c5v < best_c:
                best_c, best_h = c5v, h5
        print(f"P5: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 6: Upsample to 2x
    n1 = n0 * 2
    if rem() > 280:
        hu = upsample(best_h, n1)
        cu = compute_c5(hu)
        print(f"  up{n1}: C5={cu:.6f} t={time.time()-t0:.0f}", flush=True)
        if rem() > 200:
            hu, cu = fista(hu, 4000, 30.0, 0.3, 0.1, 60, 30)
            if cu < best_c:
                best_c, best_h = cu, hu
            print(f"  upF1: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)
        if rem() > 120:
            hu, cu = fista(hu, 4000, 1.0, 0.02, 0.03, 20, 20)
            if cu < best_c:
                best_c, best_h = cu, hu
            print(f"  upF2: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)
        if rem() > 80:
            hu, cu = topk_subgrad(hu, 6000, 3, 0.25, 50)
            if cu < best_c:
                best_c, best_h = cu, hu
            print(f"  upSG: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)
        if rem() > 40:
            hu, cu = single_subgrad(hu, 5000, 0.3, 50)
            if cu < best_c:
                best_c, best_h = cu, hu
            print(f"  upS1: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 7: Try 3x if time allows
    n2 = n0 * 3
    if n2 != n1 and rem() > 200:
        h2u = upsample(best_h, n2)
        c2u = compute_c5(h2u)
        if rem() > 150:
            h2u, c2u = fista(h2u, 3000, 25.0, 0.2, 0.08, 40, 30)
            if c2u < best_c:
                best_c, best_h = c2u, h2u
            print(f"  3xF: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)
        if rem() > 60:
            h2u, c2u = topk_subgrad(h2u, 4000, 3, 0.2, 50)
            if c2u < best_c:
                best_c, best_h = c2u, h2u
            print(f"  3xSG: C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 8: Random perturbation restarts
    if rem() > 100:
        for trial in range(3):
            if rem() < 40:
                break
            sc = 0.01 + 0.02 * trial
            hp = project_c(best_h + rng.randn(len(best_h)) * sc, len(best_h))
            hr, cr = fista(hp, 2000, 10.0, 0.1, 0.05, 30, 30)
            if cr < best_c:
                best_c, best_h = cr, hr
                print(f"  R{trial}: C5={best_c:.6f}", flush=True)

    # Phase 9: Final polish
    if rem() > 15:
        nf = len(best_h)
        if nf <= 1024:
            best_h = slsqp_smooth(best_h, 2.0, 200)
        else:
            best_h, _ = fista(best_h, 1000, 0.5, 0.01, 0.02, 10, 20)
        best_c = compute_c5(best_h)

    best_h = project_c(best_h, len(best_h))
    final_c = compute_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    print(f"DONE: n={len(best_h)} C5={best_c:.6f} t={time.time()-t0:.0f}", flush=True)
    return best_h, best_c, len(best_h)
