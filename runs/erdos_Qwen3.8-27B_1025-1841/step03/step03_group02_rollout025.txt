import numpy as np
from scipy.optimize import minimize

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    N = 1
    nn = 2 * n - 1
    while N < nn:
        N <<= 1
    H = np.fft.rfft(h, N)
    B = np.fft.rfft(b, N)
    corr = np.fft.irfft(H * np.conj(B), N)
    return float(np.max(corr) * dx)

def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)
