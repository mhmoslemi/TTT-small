    target = n / 2.0

        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)

    """Evaluate the change in overlap at worst_shifts when flipping h[i]:1->0, h[j]:0->1"""
    n = len(h)
    dx = 2.0 / n
    effects = []
    for k_idx in worst_shifts:
        k = k_idx - (n - 1)  # Convert from correlate index to actual shift
        # i+k and i-k must be valid
        effect = 0
        ik = i + k
        i_k = i - k
        jk = j + k
        j_k = j - k
        # The effect on O(k) from changing h[i] from 1 to 0:
        # -(1-h[i+k]) if i+k valid, +h[i-k] if i-k valid
        # The effect on O(k) from changing h[j] from 0 to 1:
        # +(1-h[j+k]) if j+k valid, -h[j-k] if j-k valid
        if 0 <= ik < n:

        if 0 <= i_k < n:
            effect += h[i_k]
        if 0 <= jk < n:
            effect += (1 - h[jk])
        if 0 <= j_k < n:
            effect += -h[j_k]
        effects.append(effect * dx)
    return np.array(effects)

        # Term 1: i=j, need j+k valid

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(h)
            for idx in order:
                add = min(diff, 1.0 - h[idx])
                h[idx] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            order = np.argsort(-h)
            for idx in order:
                sub = min(-diff, h[idx])
                h[idx] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return h

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

        mc = float(np.max(corr))
        scaled = (corr - mc) * T
        scaled = np.clip(scaled, -500.0, 500.0)
