**Better approach: Batch SA with vectorized moves**

Instead of evaluating C5 after every single swap, I can:
1. Precompute the current correlation
2. For a proposed swap of positions i and j (changing h_i from 1 to 0 and h_j from 0 to 1), compute the NEW correlation incrementally
3. This gives O(n) update instead of O(n²) per evaluation

For a binary swap (h[i]: 1→0, h[j]: 0→1):
- change in h(x) at position i: -1
- change in h(x) at position j: +1
- change in (1-h(x)) at position i: +1
- change in (1-h(x)) at position j: -1

The correlation at lag k: corr[k] = sum_x h[x] * (1-h[x+k])

After swap:
- At position i: h becomes 0, so h[i]*(1-h[i+k]) goes from 1*(1-h[i+k]) to 0. Change: -(1-h[i+k])
- At position j: h becomes 1, so h[j]*(1-h[j+k]) goes from 0 to 1*(1-h[j+k]). Change: +(1-h[j+k])
- At position i-k (i.e., where x+k = i, so x = i-k): h[x+k] changes from 1 to 0, so (1-h[x+k]) changes from 0 to 1. h[x]*(1-h[x+k]) changes from h[x-k]*0 to h[x-k]*1. Change: +h[i-k]
- At position j-k (i.e., where x+k = j, so x = j-k): h[x+k] changes from 0 to 1, so (1-h[x+k]) changes from 1 to 0. h[x]*(1-h[x+k]) changes from h[j-k]*1 to h[j-k]*0. Change: -h[j-k]

So the total change in corr[k] for a swap (i: 1→0, j: 0→1):
Δcorr[k] = -(1-h[i+k]) + (1-h[j+k]) + h[i-k] - h[j-k]

where we use h with boundary conditions (positions outside [0,n-1] are 0).

This is O(1) per lag k, so O(n) total for the full correlation update. But we only need the MAX, which requires checking all n-1 lags. However, if we maintain a running max and only update the affected lags... wait, the change affects ALL lags (since i+k and j+k and i-k and j-k span all possible values as k varies).

Actually, for each lag k, the change involves positions i+k, j+k, i-k, j-k. As k ranges from -(n-1) to (n-1), these positions range over the full [0, 2n-2] range. So all lags are affected.

But wait - the objective is max_k corr[k]. If we know the current max and which k achieves it, we can check if those specific k values are affected. But the change to corr[k] for each specific k is O(1). So we can:
1. Compute the new value at the current argmax k*

def binary_sa_fast(b, n, budget_s, T0=0.01, T1=0.00001, reheat_every=200000, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    b = b.astype(float).copy()
    target = n // 2
    assert int(np.sum(b)) == target
    
    corr = compute_corr(b)
    curr_max = float(np.max(corr))
    best_max = curr_max
    best_b = b.copy()
    
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    start_time = time.time()
    it = 0
    no_improve = 0
    
    while True:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
        
        # Temperature with periodic reheat
        frac = min(1.0, float(it) / reheat_every)
        T = T0 * (T1 / T0) ** frac
        
        it += 1
        no_improve += 1
        
        # Pick random 1 and 0 positions to swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        
        # Apply swap
        b[i] = 0
        b[j] = 1
        
        # Compute new max corr
        new_corr = compute_corr(b)
        new_max = float(np.max(new_corr))
        
        # Acceptance
        delta = new_max - curr_max
        if delta <= 0 or (T > 1e-15 and rng.random() < np.exp(-delta / T)):
            curr_max = new_max
            corr = new_corr
            # Update ones/zeros lists
            ones = np.delete(ones, ii)  # expensive!
            ones = np.insert(ones, ii, j)
            zeros = np.delete(zeros, jj)
            zeros = np.insert(zeros, jj, i)
            
            # Hmm, delete+insert is O(n) each. For n=960, that's fine.
            
            if new_max < best_max - 1e-15:
                best_max = new_max
                best_b = b.copy()
                no_improve = 0
        else:
            # Reject
            b[i] = 1
            b[j] = 0
        
        # Periodic reheat
        if no_improve > reheat_every // 2:
            T *= 2  # bump temperature
            no_improve = 0
    
    return best_b, best_max * (2.0/n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    n = len(h)
    required = 2 * n - 1
    size = 1 << (required - 1).bit_length()
    fa = np.fft.rfft(h, size)
    fb = np.fft.rfft((1.0 - h)[::-1], size)
    return np.fft.irfft(fa * fb)[:required]

def compute_c5(h):
    n = len(h)
    return float(np.max(compute_corr(h)) * (2.0 / n))

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        # Need more ones; flip zeros with highest h
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(diff):
            b[zeros[order[i]]] = 1
    elif diff < 0:
        # Need fewer ones; flip ones with lowest h
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for i in range(-diff):
            b[ones[order[i]]] = 0
    return b

def binary_sa(b, n, time_budget, T0=0.05, T1=0.0001, reheat_period=100000, rng=None):

    if rng is None:
        rng = np.random.RandomState(0)
    
    b = b.astype(float).copy()
    target = n // 2
    
    # Initial correlation
    corr = compute_corr(b)
    curr_max = float(np.max(corr))
    best_max = curr_max
    best_b = b.copy()
    
    ones = np.where(b == 1)[0].copy()
    zeros = np.where(b == 0)[0].copy()
    
    start = time.time()
    it = 0
    best_it = 0
    stall = 0
    max_stall = reheat_period // 2
    
    while True:
        elapsed = time.time() - start
        if elapsed > time_budget:
            break
        
        # Temperature
        cycle_it = it % reheat_period
        frac = float(cycle_it) / reheat_period
        T = T0 * (T1 / T0) ** frac
        # Reheat multiplier
        T *= (1.0 + 0.5 * (stall / max_stall) if stall > max_stall // 2 else 1.0)
        
        # Propose swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        
        # Apply
        b[i] = 0.0
        b[j] = 1.0
        
        # Evaluate
        new_corr = compute_corr(b)
        new_max = float(np.max(new_corr))
        
        # Accept?
        delta = new_max - curr_max
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):

                best_it = it
                stall = 0
        else:
            b[i] = 1.0
            b[j] = 0.0
        
        it += 1

                h = make_feasible(h, n)
                if best_c < best_c5 - 1e-14:
                    best_c5 = best_c
                    improved = True
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binarize parent
    b0 = to_binary(h_init, n0)
    c5_b0 = compute_c5(b0)
    if c5_b0 < best_c5:
        best_c5 = c5_b0
        best_h = b0.copy()

    # Phase 2: Massive binary SA from parent-binarized
    time_left = budget_s - (time.time() - start_time)
    if time_left > 200:
        sa_time = min(time_left * 0.35, 400)
        b_sa, mc = binary_sa_clean(b0, n0, sa_time, T0=0.05, T1=0.0001, reheat_period=50000, rng=rng)
        c5_sa = mc * 2.0 / n0
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = b_sa.copy()

    # Phase 3: SA from random start
    time_left = budget_s - (time.time() - start_time)
    if time_left > 200:
        b_rand = np.zeros(n0)
        rand_idx = rng.choice(n0, size=n0 // 2, replace=False)
        b_rand[rand_idx] = 1.0
        sa_time = min(time_left * 0.25, 250)
        b_r, mc_r = binary_sa_clean(b_rand, n0, sa_time, T0=0.1, T1=0.0005, reheat_period=40000, rng=rng)
        c5_r = mc_r * 2.0 / n0
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = b_r.copy()

    # Phase 4: Block SA for global exploration
    time_left = budget_s - (time.time() - start_time)
    if time_left > 150:
        sa_time = min(time_left * 0.25, 200)
        b_bl, mc_bl = block_sa_clean(best_h, n0, sa_time, T0=0.1, T1=0.005, reheat_period=30000, rng=rng)
        c5_bl = mc_bl * 2.0 / n0
        if c5_bl < best_c5:
            best_c5 = c5_bl
            best_h = b_bl.copy()

    # Phase 5: Another round of point SA (refinement)
    time_left = budget_s - (time.time() - start_time)
    if time_left > 150:
        sa_time = min(time_left * 0.35, 350)
        b2, mc2 = binary_sa_clean(best_h, n0, sa_time, T0=0.02, T1=0.00001, reheat_period=60000, rng=rng)
        c5_2 = mc2 * 2.0 / n0
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()

    # Phase 6: Try n0+2, n0+4 for potentially easier integer targets
    for n_try in [n0 + 2, n0 + 4, n0 + 6]:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 80:
            break
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, n_try)
        h_try = np.interp(new_x, old_x, best_h)
        b_try = to_binary(h_try, n_try)
        sa_time = min(time_left * 0.25, 120)
        b_t, mc_t = binary_sa_clean(b_try, n_try, sa_time, T0=0.05, T1=0.0001, reheat_period=40000, rng=rng)
        c5_t = mc_t * 2.0 / n_try
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = b_t.copy()

    # Phase 7: Try lower resolution
    for n_try in [n0 - 2, n0 - 4]:
        time_left = budget_s - (time.time() - start_time)
        if time_left < 60 or n_try < 64:
            break
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, n_try)
        h_try = np.interp(new_x, old_x, best_h)
        b_try = to_binary(h_try, n_try)
        sa_time = min(time_left * 0.25, 100)
        b_t, mc_t = binary_sa_clean(b_try, n_try, sa_time, T0=0.05, T1=0.0001, reheat_period=40000, rng=rng)
        c5_t = mc_t * 2.0 / n_try
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = b_t.copy()

    # Phase 8: Final binary SA (polish)
    time_left = budget_s - (time.time() - start_time)
    if time_left > 60 and (best_h == best_h.astype(int)).all():
        sa_time = min(time_left * 0.5, 200)
        b_final, mc_f = binary_sa_clean(best_h, len(best_h), sa_time, T0=0.01, T1=0.000005, reheat_period=40000, rng=rng)
        c5_f = mc_f * 2.0 / len(best_h)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = b_final.copy()

    # Phase 9: Continuous coordinate descent (if not purely binary optimal)
    time_left = budget_s - (time.time() - start_time)
    if time_left > 30:
        best_h, best_c5 = coord_desc_fast(best_h, min(time_left * 0.6, 60), rng)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
