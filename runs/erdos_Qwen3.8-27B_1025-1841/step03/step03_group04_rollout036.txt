```python
def minimal_repair(h, n, changed_idx):
    """After changing h[changed_idx], minimally adjust one other position to maintain sum."""
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h, None
    # Need to adjust by 'diff' at some position other than changed_idx
    if diff > 0:
        # Need to increase sum; find position with most room (lowest h, not changed_idx)
        mask = np.ones(n, dtype=bool)
        mask[changed_idx] = False
        candidates = np.where(mask & (h < 1.0))[0]
        if len(candidates) == 0:
            return h, None
        # Pick the one with lowest h (most room)
        k = candidates[np.argmin(h[candidates])]
        adj = min(diff, 1.0 - h[k])
        h[k] += adj
        # If still not enough, might need another adjustment, but for small changes it should be fine
    else:
        # Need to decrease sum; find position with most excess (highest h, not changed_idx)
        mask = np.ones(n, dtype=bool)
        mask[changed_idx] = False
        candidates = np.where(mask & (h > 0.0))[0]
        if len(candidates) == 0:
            return h, None
        k = candidates[np.argmax(h[candidates])]
        adj = max(diff, -(h[k]))
        h[k] += adj
    return h, k
```

```python
        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def minimal_repair(h, n, changed_idx):
    """Minimally fix sum constraint after changing h[changed_idx]."""
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h, True
    if diff > 0:
        # Need to increase: find position (not changed_idx) with most room
        mask = np.ones(n, dtype=bool)
        mask[changed_idx] = False
        cands = np.where(mask)[0]
        if len(cands) == 0:
            return h, False
        # Prefer positions with low values (more room)
        k = cands[np.argmin(h[cands])]
        adj = min(diff, 1.0 - h[k])
        h[k] += adj
        # If still short, do a full repair
        if abs(target - np.sum(h)) > 1e-13:
            h = make_feasible(h, n)
    else:
        # Need to decrease: find position (not changed_idx) with most excess
        mask = np.ones(n, dtype=bool)
        mask[changed_idx] = False
        cands = np.where(mask)[0]
        if len(cands) == 0:
            return h, False
        k = cands[np.argmax(h[cands])]
        adj = max(diff, -h[k])
        h[k] += adj
        if abs(target - np.sum(h)) > 1e-13:
            h = make_feasible(h, n)
    return h, True

def _slsqp_Objective(x, T, n, top_idx):
```
