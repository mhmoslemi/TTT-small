def smooth_obj_grad(h, n, tau, top_k):
    """Compute smoothed objective and gradient using FFT."""
    dx = 2.0 / n
    corr = correlate_fft(h, 1.0 - h)
    max_c = corr.max()
    
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    
    top_vals = corr[top_idx]
    shifted = tau * (top_vals - max_c)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx]
        k = top_idx[i_idx]
        sh = k - n + 1
        # Gradient contribution from shift sh
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    
    grad *= dx
    obj = dx * (max_c + np.log(max(s, 1e-300)) / tau)

def binary_local_search(h, n, time_limit):

        remove_idx = np.random.choice(ones, excess, replace=False)
        b[remove_idx] = 0.0
    elif n_ones < target_ones:

        add_idx = np.random.choice(zeros, need, replace=False)
        b[add_idx] = 1.0

    best_b = b.copy()

    cool_rate = 0.9995

        T = max(T * cool_rate, T_min)
        # Pick random pair to flip

        i = ones_list[np.random.randint(len(ones_list))]
        j = zeros_list[np.random.randint(len(zeros_list))]

        b[i] = 0.0
        b[j] = 1.0

# FFT-based correlation
def correlate_fft(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    N = 1 << (size - 1).bit_length()
    A = np.fft.rfft(a, N)
    B = np.fft.rfft(b[::-1], N)
    return np.fft.irfft(A * B, N)[:size]

# Compute C5 value
def compute_c5_val(h):
    n = len(h)
    corr = correlate_fft(h, 1.0 - h)
    return corr.max() * (2.0 / n)

# Project to feasible set
def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_obj_grad(h, n, tau, top_k):

    corr = correlate_fft(h, 1.0 - h)
    max_c = corr.max()
    if top_k < len(corr):

    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    shifted = np.clip(tau * (top_vals - max_c), -50.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)

        pk = p[i_idx]
        k = int(top_idx[i_idx])

        if j0b < j1b:

    obj = dx * (max_c + np.log(max(s, 1e-300)) / tau)

        i = ones_arr[np.random.randint(len(ones_arr))]
        j = zeros_arr[np.random.randint(len(zeros_arr))]
        b[i] = 0.0
        b[j] = 1.0
        c5_new = compute_c5_val(b)
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / max(T, 1e-10)):
            if c5_new < best_c5:
                best_c5 = c5_new
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    
    return best_b, best_c5

def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))
