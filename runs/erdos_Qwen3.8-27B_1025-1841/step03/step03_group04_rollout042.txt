def subgradient_step(h, n, step_size, active_threshold=0.01):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Find active shifts
    # corr has 2n-1 elements, centered at index n-1
    # corr[n-1+k] corresponds to shift k
    active = corr >= mc - active_threshold * mc
    # Compute subgradient (average gradient of active shifts)
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        idx = n - 1 + k
        if not active[idx]:
            continue
        weight = (corr[idx] - (mc - active_threshold * mc)) / max(1, np.sum(corr[active] - (mc - active_threshold*mc)))
        # Compute gradient of Q_k at h
        # Q_k = sum_{i=max(0,-k)}^{min(n-1, n-1-k)} h[i] * (1-h[i+k])
        # (with h out of range = 0)
        for j in range(n):
            # Contribution from h[j] in the "first" factor
            jk = j + k
            if 0 <= jk < n and j <= n-1-abs(k):  # need to be careful
                # h[j]*(1-h[j+k])
                grad[j] += weight * (1 - h[jk])
            # Contribution from h[j] in the "second" factor  
            jm = j - k
            if 0 <= jm < n:
                # h[jm]*(1-h[j])
                grad[j] -= weight * h[jm]
    # Normalize
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 1e-15:
        grad /= grad_norm
    # Step
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new, mc, active

        diff = m / 2.0 - np.sum(x)
        if abs(diff) > 1e-6:
            mask = x < 1.0 if diff > 0 else x > 0.0
            if np.sum(mask) > 0:
                x = x.copy()
                x[mask] += diff / np.sum(mask)
                x = np.clip(x, 0.0, 1.0)
        dx_seg = 2.0 / m
        corr = np.correlate(x, 1.0 - x, mode='full')
        return np.max(corr) * dx_seg
    
    from scipy.optimize import differential_evolution
    try:
        best = differential_evolution(de_obj, [(0,1)]*m, seed=seed,
                                       maxiter=maxiter, tol=1e-12,
                                       popsize=15, mutation=(0.3, 1.0),
                                       recombination=0.8, init='sobol')
        h_best_seg = np.clip(best.x, 0.0, 1.0)
        h_full = np.interp(old_x, new_x, h_best_seg)
        h_full = make_feasible(h_full, n_parent)
        return h_full
    except Exception:
        return h_parent

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c = compute_c5(best_h)

    # Phase 1: Fast paired coordinate descent
    if remaining() > 60:
        best_h, best_c = fast_paired_cd(best_h, n, max_passes=60, rng=rng)

    # Phase 2: SLSQP refinement with moderate T
    if remaining() > 60:
        refined = slsqp_refine(best_h, n, T=400.0, maxiter=250)
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_h, best_c = refined, c_ref

    # Phase 3: More paired CD
    if remaining() > 40:
        best_h, best_c = fast_paired_cd(best_h, n, max_passes=40, rng=rng)

    # Phase 4: DE at lower resolution for diversity
    if remaining() > 200:
        for m_try in [120, 160]:
            if remaining() < 120:
                break
            h_de = de_low_res(best_h, n, m_try, seed=seed + m_try, maxiter=150)
            c_de = compute_c5(h_de)
            if c_de < best_c + 0.005:
                # Refine the DE result
                h_de2, c_de2 = fast_paired_cd(h_de, n, max_passes=40, rng=rng)
                if c_de2 < best_c:
                    best_h, best_c = h_de2, c_de2
                if remaining() > 40:
                    h_de3 = slsqp_refine(best_h, n, T=500.0, maxiter=200)
                    c_de3 = compute_c5(h_de3)
                    if c_de3 < best_c:
                        best_h, best_c = h_de3, c_de3

    # Phase 5: SLSQP with high T (sharp max approximation)
    if remaining() > 50:
        refined = slsqp_refine(best_h, n, T=800.0, maxiter=300)
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_h, best_c = refined, c_ref

    # Phase 6: Final paired CD
    if remaining() > 30:
        best_h, best_c = fast_paired_cd(best_h, n, max_passes=50, rng=rng)

    # Phase 7: Final SLSQP polish
    if remaining() > 30:
        refined = slsqp_refine(best_h, n, T=1000.0, maxiter=200)
        c_ref = compute_c5(refined)
        if c_ref < best_c:
            best_h, best_c = refined, c_ref

    # Phase 8: Try higher resolution
    if remaining() > 150:
        n_up = n * 2
        if n_up <= 3840:
            old_x = np.linspace(0.0, 2.0, n)
            new_x = np.linspace(0.0, 2.0, n_up)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = make_feasible(h_up, n_up)
            c_up = compute_c5(h_up)
            if c_up < best_c + 0.002:
                h_up, c_up = fast_paired_cd(h_up, n_up, max_passes=30, rng=rng)
                if remaining() > 40:
                    h_up = slsqp_refine(h_up, n_up, T=500.0, maxiter=150)
                    c_up = compute_c5(h_up)
                if c_up < best_c:
                    best_h, best_c, n = h_up, c_up, n_up

    # Final cleanup
    best_h = make_feasible(best_h, n)
    best_c = compute_c5(best_h)
    return best_h, best_c, n
