def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:

        cnt = np.sum(mask)
        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

    shifted = np.clip(shifted, -700, 700)
    w = np.exp(shifted)
    w_sum = np.sum(w)
    w = w / w_sum  # normalized weights, sum to 1

    # C = correlate(1-h, w, 'full') = fftconvolve(1-h, w[::-1], 'full')

    B = D[n-1 : 2*n-1]  # n elements

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
