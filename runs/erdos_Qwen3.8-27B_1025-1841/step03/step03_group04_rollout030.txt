import numpy as np
from scipy.optimize import minimize, differential_evolution
import time

_G_N = 0
_G_T = 100.0
_G_TOP_IDX = None
_G_HALF = 0.0

def _con_fun(x):
    return np.sum(x) - _G_HALF

def _con_jac(x):
    return np.ones(_G_N)

def _smooth_obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_TOP_IDX is not None:
        corr_sub = corr[_G_TOP_IDX]
    else:
        corr_sub = corr
    mc = np.max(corr_sub)
    shifted = corr_sub - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run_sl sqp(h, n, T, top_idx, maxiter):
    global _G_N, _G_T, _G_TOP_IDX, _G_HALF
    _G_N = n
    _G_T = T
    _G_TOP_IDX = top_idx
    _G_HALF = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_smooth_obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def binary_hill_climb(h, n, max_passes, time_limit, elapsed_fn):
    best_c5 = compute_c5(h)
    for _ in range(max_passes):
        if elapsed_fn() > time_limit:
            break
        improved = False
        for j in range(n):
            if elapsed_fn() > time_limit:
                break
            old = h[j]
            if old > 0.05:
                trial = h.copy()
                trial[j] = max(0.0, old - 0.6)
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
                    continue
            if old < 0.95:
                trial = h.copy()
                trial[j] = min(1.0, old + 0.6)
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
        if not improved:
            break
    return h, best_c5

def swap_hill_climb(h, n, n_swaps, rng, time_limit, elapsed_fn):
    best_c5 = compute_c5(h)
    for _ in range(n_swaps):
        if elapsed_fn() > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        trial = h.copy()
        trial[i], trial[j] = h[j], h[i]
        c = compute_c5(trial)
        if c < best_c5 - 1e-15:
            h = trial
            best_c5 = c
    return h, best_c5

def fine_coord_descent(h, n, steps, max_passes, time_limit, elapsed_fn):
    best_c5 = compute_c5(h)
    for _ in range(max_passes):
        if elapsed_fn() > time_limit:
            break
        improved = False
        for j in range(n):
            if elapsed_fn() > time_limit:
                break
            old = h[j]
            best_val = old
            best_c = best_c5
            for d in steps:
                for s in [-1.0, 1.0]:
                    v = old + s * d
                    if 0.0 <= v <= 1.0:
                        trial = h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5
