    return make_feasible(h_new, new_n)

    """Compute gradient of f_{d*}(h) = sum_i h[i]*(1-h[i+d*])"""
    n = len(h)
    g = np.zeros(n)
    for j in range(n):
        val = 0.0
        # Term 1: (1 - h[j+d*]) if j+d* in [0, n-1]
        idx1 = j + d_star
        if 0 <= idx1 < n:

        # Term 2: -h[j-d*] if j-d* in [0, n-1]
        idx2 = j - d_star
        if 0 <= idx2 < n:
            val -= h[idx2]
        g[j] = val
    return g

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_and_grad(h, T, lam=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    corr_scaled = T * corr
    corr_max = np.max(corr_scaled)
    exp_corr = np.exp(corr_scaled - corr_max)
    Z = np.sum(exp_corr)
    p = exp_corr / Z
    obj = dx * (corr_max + np.log(Z)) / T
    
    # Add penalty for sum constraint
    sum_dev = np.sum(h) - n / 2.0
    obj += 0.5 * lam * sum_dev ** 2
    
    # Gradient
    g = np.zeros(n)
    thresh = 1e-4
    top_idx = np.where(p > thresh)[0]
    
    for idx in top_idx:
        d = idx - (n - 1)
        pf = p[idx]
        
        j_min1 = max(0, d)
        j_max1 = min(n - 1, d + n - 1)
        if j_min1 <= j_max1:
            indices = np.arange(j_min1, j_max1 + 1)
            g[indices] += pf * (1.0 - h[indices - d])
        
        j_min2 = max(0, -d)
        j_max2 = min(n - 1, n - 1 - d)
        if j_min2 <= j_max2:
            indices = np.arange(j_min2, j_max2 + 1)
            g[indices] -= pf * h[indices + d]
    
    g *= dx
    g += lam * sum_dev  # gradient of penalty
    
    return obj, g

def project_box_sum(h_raw, n):
    """Project onto {h : 0 ≤ h ≤ 1, sum(h) = n/2}."""
    target = n / 2.0
    # Binary search for shift
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_try = np.clip(h_raw - mid, 0.0, 1.0)
        s = np.sum(h_try)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h_raw - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:
        mask = (h_proj > 1e-10) & (h_proj < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj = h_proj.copy()
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def lbsqp_smoothed(h, T, lam, maxiter=2000):
    """L-BFGS-B on smoothed objective with sum penalty."""
    n = len(h)
    
    def fun(x):
        obj, _ = smooth_obj_and_grad(x, T, lam)
        return obj
    
    def fprime(x):
        _, g = smooth_obj_and_grad(x, T, lam)
        return g
    
    res = fmin_l_bfgs_b(fun, h, fprime=fprime, 
                         pgtol=1e-12, maxiter=maxiter,
                         iprint=0)
    return res[0], fun(res[0])

        for j in range(n):
