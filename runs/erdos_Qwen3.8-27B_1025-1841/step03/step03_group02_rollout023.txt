                    trial[mask] = other[mask]
                    # Fix to maintain n/2 ones
                    cnt_ones = int(np.sum(trial))
                    target = n // 2
                    if cnt_ones > target:
                        excess = cnt_ones - target
                        ones_pos = np.where(trial == 1)[0]
                        zeros_pos = np.where(trial == 0)[0]
                        if len(zeros_pos) >= excess:
                            flip = rng.choice(ones_pos, excess, replace=False)
                            trial[flip] = 0
                    elif cnt_ones < target:
                        deficit = target - cnt_ones
                        ones_pos = np.where(trial == 1)[0]
                        zeros_pos = np.where(trial == 0)[0]
                        if len(zeros_pos) >= deficit:
                            flip = rng.choice(zeros_pos, deficit, replace=False)
                            trial[flip] = 1
                    if abs(int(np.sum(trial)) - target) < 2:
                        new_corr = np.correlate(trial, 1.0 - trial, mode='full')
                        new_c5 = float(np.max(new_corr) * dx)
                        if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T) * 5.0):
                            b = trial
                            curr_c5 = new_c5
                            corr = new_corr
                            if new_c5 < best_c5:
                                best_c5 = new_c5
                                best_b = b.copy()
                                stale = 0
                            else:
                                stale += 1

    return best_b, best_c5

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, deadline):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline - 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    b0 = to_binary(h_init, n0)
    best_c5 = compute_c5(b0)
    best_h = b0.copy()

    # Build archive with diverse starts
    archive = [b0.copy()]
    for _ in range(3):
        archive.append(random_binary(n0, rng))

    # Phase 1: Multi-start SA (200s each for first 3, keep best)
    for idx in range(3):
        if time.time() > deadline - 500:
            break
        phase_deadline = min(deadline - 500, time.time() + 200)
        b_res, c5_res = sa_main(archive[idx].copy(), n0, 500000, 0.008, 0.00003, rng, phase_deadline, archive)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()
        archive.append(b_res)

    # Phase 2: Long targeted SA from best (500s)
    if time.time() < deadline - 300:
        phase_deadline = min(deadline - 300, time.time() + 500)
        b_res, c5_res = sa_main(best_h.copy(), n0, 800000, 0.006, 0.00001, rng, phase_deadline, archive)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()

    # Phase 3: SLSQP continuous polish
    if time.time() < deadline - 150:
        try:
            x_res = slsqp_run(best_h, n0, 80.0, 300, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 4: Coordinate descent
    if time.time() < deadline - 60:
        best_h, best_c5 = coord_descent(best_h.copy(), n0, 15, deadline - 30)

    # Phase 5: Another SA round from polished solution
    if time.time() < deadline - 100:
        b4 = to_binary(best_h, n0)
        phase_deadline = min(deadline - 30, time.time() + 150)
        b_res, c5_res = sa_main(b4.copy(), n0, 400000, 0.004, 0.00001, rng, phase_deadline, archive)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()

    # Phase 6: Higher resolution attempt (n=1920)
    if time.time() < deadline - 250 and n0 < 1900:
        n1 = n0 * 2 if n0 <= 1000 else 1920
        h_up = upsample_h(best_h, n1)
        b_up = to_binary(h_up, n1)
        phase_deadline = min(deadline - 50, time.time() + 250)
        b_res, c5_res = sa_main(b_up.copy(), n1, 400000, 0.005, 0.00002, rng, phase_deadline, None)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()
        # Quick SLSQP on higher-res
        if c5_res < best_c5 + 0.001 and time.time() < deadline - 60:
            try:
                x_res = slsqp_run(best_h, len(best_h), 60.0, 200, 1e-14)
                c5_res2 = compute_c5(x_res)
                if c5_res2 < best_c5:
                    best_c5 = c5_res2
                    best_h = make_feasible(x_res, len(best_h))
            except Exception:
                pass

    # Phase 7: Final micro-SA
    if time.time() < deadline - 20:
        n_final = len(best_h)
        b_f = to_binary(best_h, n_final)
        phase_deadline = min(deadline - 5, time.time() + 30)
        b_res, c5_res = sa_main(b_f.copy(), n_final, 100000, 0.002, 0.000005, rng, phase_deadline, None)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()

    # Final SLSQP
    if time.time() < deadline - 10:
        n_final = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_final, 200.0, 150, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_final)
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
