    grad = np.zeros(n)
    for i in range(n):

        ik = i - k_shift

def grad_corr_at_lag_vec(h, k):

    k_shift = k - (n - 1)
    grad = np.zeros(n)
    # Term 1: (1-h[i-k_shift]) for valid i-k_shift
    if k_shift >= 0:
        # i-k_shift >= 0 means i >= k_shift
        # i-k_shift < n means i < n + k_shift (always true since i < n and k_shift < n)
        grad[k_shift:] += (1.0 - h[:n-k_shift])
        # Term 2: -h[i+k_shift]
        # i+k_shift < n means i < n-k_shift
        grad[:n-k_shift] -= h[k_shift:]
    else:
        # k_shift < 0, let d = -k_shift
        d = -k_shift
        # i+d < n means i < n-d
        grad[:n-d] += (1.0 - h[d:])
        # i-d >= 0 means i >= d
        grad[d:] -= h[:n-d]
    return grad
```

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        
        corr = compute_corr(h)
        dx = 2.0 / n
        c5_curr = np.max(corr) * dx
        
        # Find top active lags
        top_k = np.argsort(corr)[::-1][:5]
        
        # Average gradient over top lags
        grad = np.zeros(n)
        for tk in top_k:
            grad += grad_corr_at_lag_vec(h, tk)
        grad /= len(top_k)
        
        # Projected gradient step
        # We want to decrease c5, so move in -grad direction
        # But we need to maintain sum(h) = n/2
        # The component of grad along (1,1,...,1) doesn't change sum
        # So project: grad_perp = grad - mean(grad) * 1
        grad_perp = grad - np.mean(grad)
        
        # Adaptive step size
        alpha = 0.01
        for trial_alpha in [0.1, 0.05, 0.01, 0.005, 0.001]:
            h_try = h - trial_alpha * grad_perp
            h_try = make_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < c5_curr:
                alpha = trial_alpha
                break
        else:
            break  # No improvement
        
        h = h - alpha * grad_perp
        h = make_feasible(h, n)
        c5_new = compute_c5(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + T * np.log(np.sum(np.exp((corr - m) / T)))) * dx

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = (corr - m) / T
    exps = np.exp(shifted)
    Z = np.sum(exps)
    weights = exps / Z
    grad = np.zeros(n)
    for j in range(2*n - 1):
        w = weights[j]
        if w < 1e-10:
            continue
        k = j - (n - 1)
        if k >= 0:
            if n - k > 0:
                grad[k:] += w * (1.0 - h[:n-k])
                grad[:n-k] -= w * h[k:]
        else:
            d = -k
            if n - d > 0:
                grad[:n-d] += w * (1.0 - h[d:])
                grad[d:] -= w * h[:n-d]
    grad *= (dx / T)
    return grad

def proj_grad_descent(h, n, max_iters, start_time, budget_s, T0=200.0, T1=5.0):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    no_improve = 0
    for it in range(max_iters):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        T = T0 * (T1 / T0) ** (it / max_iters)
        val, grad = smooth_max_grad(h, T)
        # Project gradient onto sum=0 hyperplane
        grad = grad - np.mean(grad)
        # Adaptive step size
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.001]:
            h_try = h - alpha * grad
            h_try = make_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                h = h_try
                best_c5 = c5_try
                best_h = h.copy()
                no_improve = 0
                break
        else:
            no_improve += 1
            if no_improve > 50:
                break
    return best_h, best_c5

def coord_descent_fine(h, n, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(500):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                return h, best_c5
            old = h[j]
            # Try: 0, 1, old±0.01, old±0.001
            for cand in [0.0, 1.0, old-0.01, old+0.01, old-0.001, old+0.001]:
                if cand < 0 or cand > 1 or abs(cand - old) < 1e-10:
                    continue
                h[j] = cand
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    improved = True
                    old = h[j]  # Update reference
                else:
                    h[j] = old
                    h = make_feasible(h, n)
        if not improved:
            break
    return h, best_c5

def slsqp_smooth(h, n, T, maxiter, start_time, budget_s):
    remaining = budget_s - (time.time() - start_time)
    if remaining < 60:
        return h, compute_c5(h)
    try:
        def obj(x):
            return smooth_max_val(x, T)
        def con(x):
            return np.sum(x) - n / 2.0
        res = minimize(obj, h.copy(), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': con, 'jac': np.ones(n)}],
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        h_try = make_feasible(res.x, n)
        c5_try = compute_c5(h_try)
        return h_try, c5_try
    except:

    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)

    n = len(h)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Projected gradient (adaptive T, 2000 iters)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        best_h, best_c5 = proj_grad_descent(best_h, n, 2000, start_time, budget_s, 200.0, 2.0)

    # Phase 2: SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        try:
            h2 = slsqp_run(best_h, n, 100.0, 300)
            h2 = make_feasible(h2, n)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2
        except Exception:
            pass

    # Phase 3: Another gradient round with lower T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        best_h, best_c5 = proj_grad_descent(best_h, n, 2000, start_time, budget_s, 50.0, 0.5)

    # Phase 4: SLSQP with high T (smoother landscape)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        try:
            h2 = slsqp_run(best_h, n, 500.0, 400)
            h2 = make_feasible(h2, n)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2
        except Exception:
            pass

    # Phase 5: Fine coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = coord_fine(best_h, n, start_time, budget_s)

    # Phase 6: Try n+1 resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        n_alt = n + 1
        old_x = np.linspace(0.0, 2.0, n)
        new_x = np.linspace(0.0, 2.0, n_alt)
        h_alt = np.interp(new_x, old_x, best_h)
        h_alt = make_feasible(h_alt, n_alt)
        h_alt, c_alt = proj_grad_descent(h_alt, n_alt, 1500, start_time, budget_s, 100.0, 2.0)
        if c_alt < best_c5:
            best_c5 = c_alt
            best_h = h_alt
            n = n_alt
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100 and c_alt < 0.381:
            try:
                h2 = slsqp_run(h_alt, n_alt, 200.0, 300)
                h2 = make_feasible(h2, n_alt)
                c2 = compute_c5(h2)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = h2
            except Exception:
                pass

    # Phase 7: Try n-1 resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n_alt = max(n - 1, 480)
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, n_alt)
        h_alt = np.interp(new_x, old_x, best_h)
        h_alt = make_feasible(h_alt, n_alt)
        h_alt, c_alt = proj_grad_descent(h_alt, n_alt, 1500, start_time, budget_s, 100.0, 2.0)
        if c_alt < best_c5:
            best_c5 = c_alt
            best_h = h_alt
            n = n_alt

    # Phase 8: Binary SA from current best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n_pts = len(best_h)
        b2, c5_b2 = binary_sa_quick(best_h, n_pts, 60000, 0.008, 0.0001, rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = b2
        # Polish binary with gradient
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100 and c5_b2 < 0.382:
            best_h, best_c5 = proj_grad_descent(best_h, n_pts, 1000, start_time, budget_s, 100.0, 1.0)

    # Phase 9: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        try:
            h2 = slsqp_run(best_h, n_pts, 800.0, 300)
            h2 = make_feasible(h2, n_pts)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2
        except Exception:
            pass

    # Phase 10: Final coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, best_c5 = coord_fine(best_h, len(best_h), start_time, budget_s)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
