def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:

        else:

        if len(ones) == 0: break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0

        if len(zeros) == 0: break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    b = (h >= 0.5).astype(float)

            break

        b[idx] = 1

    return np.correlate(h, 1.0 - h, mode='full').astype(float)

    n = len(h)

    mc = np.max(corr)

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def init_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').astype(float)

def upd_corr(corr, h, j, new_val):
    n = len(h)
    old = h[j]
    delta = new_val - old
    if abs(delta) < 1e-15:
        return
    h[j] = new_val
    s_lo = min(-j, j - (n - 1))
    s_hi = max(n - 1 - j, j)
    if s_lo > s_hi:
        return
    s = np.arange(s_lo, s_hi + 1)
    m = (n - 1) + s
    change = np.zeros(len(s))
    nz = (s != 0)
    v1 = nz & (j + s >= 0) & (j + s < n)
    if np.any(v1):
        sv = s[v1]
        change[v1] += delta * (1.0 - h[j + sv])
    v2 = nz & (j - s >= 0) & (j - s < n)
    if np.any(v2):
        sv = s[v2]
        change[v2] -= h[j - sv] * delta
    z = (s == 0)
    if np.any(z):
        change[z] = delta * (1.0 - old - new_val)
    corr[m] += change

def sa_single_swap(b_in, n, T0, T1, max_iters, rng, start_time, deadline):
    b = b_in.copy()
    corr = init_corr(b)
    dx = 2.0 / n
    curr_c5 = float(np.max(corr) * dx)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if (it & 4095) == 0:
            if time.time() > deadline:
                break
        frac = min(1.0, float(it) / max(1, max_iters))
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        upd_corr(corr, b, i, 0.0)
        upd_corr(corr, b, j, 1.0)
        new_c5 = float(np.max(corr) * dx)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            upd_corr(corr, b, i, 1.0)
            upd_corr(corr, b, j, 0.0)
    return best_b, best_c5

def sa_block(b_in, n, T0, T1, max_iters, rng, start_time, deadline):
    b = b_in.copy()
    corr = init_corr(b)
    dx = 2.0 / n
    curr_c5 = float(np.max(corr) * dx)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if (it & 1023) == 0:
            if time.time() > deadline:
                break
        frac = min(1.0, float(it) / max(1, max_iters))
        T = T0 * (T1 / T0) ** frac
        blen = rng.randint(2, 10)
        pos = rng.randint(0, n - blen)
        seg = b[pos:pos + blen]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        if sc == 0:
            continue
        fo = rng.choice(oi, size=sc, replace=False)
        fz = rng.choice(zi, size=sc, replace=False)
        idx_o = pos + fo
        idx_z = pos + fz
        for k in range(sc):
            upd_corr(corr, b, int(idx_o[k]), 0.0)
            upd_corr(corr, b, int(idx_z[k]), 1.0)
        new_c5 = float(np.max(corr) * dx)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            for k in range(sc):
                upd_corr(corr, b, int(idx_o[k]), 1.0)
                upd_corr(corr, b, int(idx_z[k]), 0.0)
    return best_b, best_c5

def coord_desc_01(h_in, n, start_time, deadline):
    h = h_in.copy()
    corr = init_corr(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    for rnd in range(100):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for jj in range(n):
            if time.time() > deadline:
                break
            j = int(order[jj])
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-10:
                    continue
                upd_corr(corr, h, j, cand)
                c = float(np.max(corr) * dx)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
                else:
                    upd_corr(corr, h, j, old)
            if best_val != old:
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def coord_desc_fine(h_in, n, start_time, deadline):
    h = h_in.copy()
    corr = init_corr(h)
    dx = 2.0 / n
    best_c5 = float(np.max(corr) * dx)
    for rnd in range(50):
        if time.time() > deadline:
            break
        improved = False
        order = np.random.permutation(n)
        for jj in range(n):
            if time.time() > deadline:
                break
            j = int(order[jj])
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [max(0.0, old - 0.1), max(0.0, old - 0.03),
                     min(1.0, old + 0.03), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                upd_corr(corr, h, j, cand)
                c = float(np.max(corr) * dx)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
                else:
                    upd_corr(corr, h, j, old)
            if best_val != old:
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_obj(x):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    T = _SL_T
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def slsqp_con_fun(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_polish(h_in, n, T, maxiter, ftol, start_time, deadline):
    global _SL_T
    if time.time() > deadline:
        return h_in, compute_c5_full(h_in)
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con_fun, 'jac': slsqp_con_jac}]
    try:
        res = minimize(slsqp_obj, h_in.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if hasattr(res, 'x') and res.x is not None:
            x_new = make_feasible(res.x, n)
            c5 = compute_c5_full(x_new)
            return x_new, c5
    except Exception:
        pass
    return h_in, compute_c5_full(h_in)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5_full(h_init)

    # Phase 1: Long binary SA single-swap (main exploration)
    deadline1 = start_time + min(budget_s * 0.40, 500)
    b = to_binary(h_init, n)
    b, c5 = sa_single_swap(b, n, 0.015, 1e-6, 20000000, rng, start_time, deadline1)
    if c5 < best_c5:
        best_c5 = c5
        best_h = b.copy()

    # Phase 2: SA restart with different schedule
    deadline2 = deadline1 + min(budget_s * 0.20, 250)
    b = to_binary(best_h, n)
    b, c5 = sa_single_swap(b, n, 0.008, 1e-7, 20000000, rng, start_time, deadline2)
    if c5 < best_c5:
        best_c5 = c5
        best_h = b.copy()

    # Phase 3: Block-swap SA
    deadline3 = deadline2 + min(budget_s * 0.15, 180)
    b, c5 = sa_block(best_h, n, 0.006, 1e-6, 5000000, rng, start_time, deadline3)
    if c5 < best_c5:
        best_c5 = c5
        best_h = b.copy()

    # Phase 4: Coordinate descent 0/1
    deadline4 = deadline3 + min(budget_s * 0.10, 100)
    best_h, c5 = coord_desc_01(best_h, n, start_time, deadline4)
    if c5 < best_c5:
        best_c5 = c5

    # Phase 5: SLSQP polish
    deadline5 = deadline4 + min(budget_s * 0.12, 120)
    best_h, c5 = slsqp_polish(best_h, n, 100.0, 300, 1e-14, start_time, deadline5)
    if c5 < best_c5:
        best_c5 = c5

    # Phase 6: SA from continuous solution
    deadline6 = deadline5 + min(budget_s * 0.12, 120)
    b = to_binary(best_h, n)
    b, c5 = sa_single_swap(b, n, 0.005, 1e-7, 10000000, rng, start_time, deadline6)
    if c5 < best_c5:
        best_c5 = c5
        best_h = b.copy()

    # Phase 7: Fine coordinate descent
    deadline7 = deadline6 + min(budget_s * 0.08, 80)
    best_h, c5 = coord_desc_fine(best_h, n, start_time, deadline7)
    if c5 < best_c5:
        best_c5 = c5

    # Phase 8: Final SLSQP with higher T
    deadline8 = deadline7 + min(budget_s * 0.08, 80)
    best_h, c5 = slsqp_polish(best_h, n, 300.0, 200, 1e-15, start_time, deadline8)
    if c5 < best_c5:
        best_c5 = c5

    # Phase 9: Final 0/1 coordinate descent
    deadline9 = deadline8 + min(budget_s * 0.05, 50)
    best_h, c5 = coord_desc_01(best_h, n, start_time, deadline9)
    if c5 < best_c5:
        best_c5 = c5

    # Phase 10: Very final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        deadline10 = time.time() + remaining - 5
        best_h, c5 = slsqp_polish(best_h, n, 500.0, 150, 1e-16, start_time, deadline10)
        if c5 < best_c5:
            best_c5 = c5

    # Final validation
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5_full(best_h)
    return best_h, best_c5, n
