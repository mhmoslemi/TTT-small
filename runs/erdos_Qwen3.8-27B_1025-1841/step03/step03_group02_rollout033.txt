        else:
            k = rng.choice([4, 8, 12, 16])
            curr = perturb_kswap(curr, n, k, rng)
    return best, best_max

def random_binary(n, rng):
    b = np.zeros(n)
    ones_pos = rng.choice(n, size=n // 2, replace=False)
    b[ones_pos] = 1.0
    return b

def structured_init(n, rng, seed_offset):
    """Create a structured initial guess using a periodic pattern."""
    b = np.zeros(n)
    # Try different period structures
    period = rng.choice([3, 4, 5, 7, 8, 11])
    pattern = np.zeros(period)
    n_ones = round(period / 2)
    pattern[rng.choice(period, size=n_ones, replace=False)] = 1.0
    for i in range(n):
        b[i] = pattern[i % period]
    # Fix count
    target = n // 2
    diff = target - np.sum(b)
    if diff > 0:
        zeros = np.where(b == 0)[0]
        b[rng.choice(zeros, size=int(diff), replace=False)] = 1.0
    elif diff < 0:
        ones = np.where(b == 1)[0]
        b[rng.choice(ones, size=int(-diff), replace=False)] = 0.0
    return b

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h_init)
    except Exception:
        n0 = 960
        h_init = np.full(n0, 0.5)
    
    h_init = make_feasible(h_init, n0)
    b_init = to_binary(h_init, n0)
    
    best_h = b_init.copy()
    best_c5 = compute_c5(b_init)
    
    # Phase 1: Strong local search on parent
    b_ls, c5_ls = local_search_steepest(b_init.copy(), n0, 5000, start_time, budget_s)
    c5_ls_full = compute_c5(b_ls)
    if c5_ls_full < best_c5:
        best_c5 = c5_ls_full
        best_h = b_ls.copy()
    
    # Phase 2: Iterated local search from best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        b_i, c5_i = iterated_ls(best_h.copy(), n0, start_time, budget_s, rng, max_perts=500)
        c5_i_full = compute_c5(b_i)
        if c5_i_full < best_c5:
            best_c5 = c5_i_full
            best_h = b_i.copy()
    
    # Phase 3: Try random starts
    remaining = budget_s - (time.time() - start_time)
    n_starts = 8
    start_budget_each = remaining / (n_starts + 3) * 0.7
    for s in range(n_starts):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        b_rand = random_binary(n0, rng)
        b_ls2, _ = local_search_steepest(b_rand, n0, 1000, start_time, budget_s)
        c5_r = compute_c5(b_ls2)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = b_ls2.copy()
    
    # Phase 4: Structured starts
    remaining = budget_s - (time.time() - start_time)
    for s in range(10):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        b_struct = structured_init(n0, rng, seed + s * 100)
        b_ls3, _ = local_search_steepest(b_struct, n0, 1000, start_time, budget_s)
        c5_s = compute_c5(b_ls3)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = b_ls3.copy()
    
    # Phase 5: Continue iterated LS on best with more iterations
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        b_i2, c5_i2 = iterated_ls(best_h.copy(), n0, start_time, budget_s, rng, max_perts=2000)
        c5_i2_full = compute_c5(b_i2)
        if c5_i2_full < best_c5:
            best_c5 = c5_i2_full
            best_h = b_i2.copy()
    
    # Phase 6: Try different resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        for n_try in [n0 + 480, n0 - 480]:
            if n_try < 128 or n_try > 2000:
                continue
            remaining = budget_s - (time.time() - start_time)
            if remaining < 80:
                break
            # Upsample/downsample best to new resolution
            old_x = np.linspace(0.0, 2.0, len(best_h))
            new_x = np.linspace(0.0, 2.0, n_try)
            h_interp = np.interp(new_x, old_x, best_h)
            b_new = to_binary(h_interp, n_try)
            b_new_ls, _ = local_search_steepest(b_new, n_try, 2000, start_time, budget_s)
            c5_new = compute_c5(b_new_ls)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = b_new_ls.copy()
    
    # Phase 7: Final intensive iterated LS on best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_final = len(best_h)
        b_final, c5_final = iterated_ls(best_h.copy(), n_final, start_time, budget_s, rng, max_perts=5000)
        c5_final_full = compute_c5(b_final)
        if c5_final_full < best_c5:
            best_c5 = c5_final_full
            best_h = b_final.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
