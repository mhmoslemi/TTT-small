def compute_subgradient(h, n, shifts):
    """Compute sum of subgradients over given shifts."""
    g = np.zeros(n)
    for s in shifts:
        # s is the shift index (0 to 2n-2), actual shift = s - (n-1)
        actual_s = s - (n-1)
        if actual_s >= 0:
            # c(s) = sum_{n=0}^{n-1-s} h[n+s] * (1-h[n])
            # grad[j] = (1-h[j-s]) for s <= j <= n-1, minus h[j+s] for 0 <= j <= n-1-s
            # First term: for j in [s, n-1], add (1-h[j-s])
            for j in range(actual_s, n):
                g[j] += (1.0 - h[j - actual_s])
            # Second term: for j in [0, n-1-s], subtract h[j+s]
            for j in range(0, n - actual_s):
                g[j] -= h[j + actual_s]
        else:
            # c(-|s|) = sum_{m=0}^{n-1-|s|} h[m] * (1-h[m+|s|])
            # grad[j] = (1-h[j+|s|]) for 0 <= j <= n-1-|s|, minus h[j-|s|] for |s| <= j <= n-1

                g[j] += (1.0 - h[j + abs_s])
            # Second term: for j in [abs_s, n-1], subtract h[j-abs_s]
            for j in range(abs_s, n):

def fw_step(h, g, n):
    """Frank-Wolfe linear minimization oracle: argmin_{s in P} <g, s>"""
    # Sort coordinates by g value
    order = np.argsort(g)
    # s_j = 1 for the (n/2) smallest g_j, 0 for the rest
    # But we need sum(s) = n/2 exactly
    # Since n/2 is an integer (we choose n even), we can set exactly n/2 coordinates to 1
    s = np.zeros(n)
    s[order[:n//2]] = 1.0
    return s

            best_c = c_new
