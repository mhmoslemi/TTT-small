def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 1
    while N < 2 * n:
        N *= 2
    h_pad = np.zeros(N)
    h_pad[:n] = h
    g = 1.0 - h
    g_pad = np.zeros(N)
    g_pad[:n] = g

    G = np.fft.rfft(g_pad)
    corr_circ = np.fft.irfft(H * np.conj(G), N)

        h = np.clip(h, 0.0, 1.0)

    target = n // 2

    s = np.sum(b)

        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
        s -= 1
    while s < target:

        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
        s += 1

        curr_max = np.max(curr_corr)
        
        best_delta = 0
        best_idx = -1
        best_action = 0  # 1 = set to 1, 0 = set to 0
        
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]

            new_corr = get_corr_fft(b)
            new_max = np.max(new_corr)
            delta = new_max - curr_max
            if delta < best_delta:
                best_delta = delta
                best_idx = i
                best_action = 0
            b[i] = 1
        
        # Try flipping each 0 to 1
        for i in zeros:
            b[i] = 1

            new_max = np.max(new_corr)
            delta = new_max - curr_max

                best_idx = i
                best_action = 1
            b[i] = 0

            curr_c5 = compute_c5_fft(b)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_b = b.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            if no_improve_count > 3:
                break
    
    return best_b, best_c5

        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    target = n // 2
    s = int(np.sum(b))
    while s > target:
        ones = np.where(b == 1)[0]
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
        s -= 1
    while s < target:
        zeros = np.where(b == 0)[0]
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
        s += 1
    return b

import numpy as np
from scipy.optimize import minimize
import time

def _np2(n):
    p = 1
    while p < n:
        p <<= 1
    return p

def _corr_full(h):
    """Compute full cross-correlation of h with (1-h) using FFT."""
    n = len(h)
    N = _np2(2 * n - 1)
    a = np.zeros(N); a[:n] = h
    b = np.zeros(N); b[:n] = 1.0 - h
    corr = np.fft.irfft(np.fft.fft(a) * np.conj(np.fft.fft(b)), N)
    return np.concatenate([corr[N - n + 1:], corr[:n]])

def c5_of(h):
    n = len(h)
    return float(np.max(_corr_full(h)) * (2.0 / n))

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-13: break
        m = h < 1.0 if d > 0 else h > 0.0
        c = m.sum()
        if c == 0: break
        h[m] += d / c
        h = np.clip(h, 0, 1)
    return h

def to_bin(h, n):
    b = (h >= 0.5).astype(float)
    tgt = n // 2
    s = int(b.sum())
    while s > tgt:
        idxs = np.where(b == 1)[0]
        b[idxs[np.argmin(h[idxs])]] = 0; s -= 1
    while s < tgt:
        idxs = np.where(b == 0)[0]
        b[idxs[np.argmax(h[idxs])]] = 1; s += 1
    return b

def pair_sa(b, n, iters, T0, T1, rng, start, budget):
    """Simulated annealing with pair moves (flip one 1→0 and one 0→1)."""
    curr = c5_of(b)
    best = curr
    best_b = b.copy()
    dx = 2.0 / n
    for it in range(iters):
        if time.time() - start > budget - 30: break
        T = T0 * (T1 / T0) ** (it / max(iters, 1))
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0.0, 1.0
        new = np.max(_corr_full(b)) * dx
        if new <= curr or rng.random() < np.exp(-(new - curr) / max(T, 1e-15)):
            curr = new
            if new < best:
                best = new; best_b = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
    return best_b, best

def pair_descent(b, n, rounds, rng, start, budget):
    """Steepest-descent-like: each round, try K random pairs, take best improving."""
    curr = c5_of(b)
    best = curr
    best_b = b.copy()
    dx = 2.0 / n
    K = 200  # pairs to try per round
    for rnd in range(rounds):
        if time.time() - start > budget - 30: break
        best_delta = 0
        best_pair = None
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        for _ in range(K):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0.0, 1.0
            new = np.max(_corr_full(b)) * dx
            delta = new - curr
            b[i], b[j] = 1.0, 0.0  # revert
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            b[i], b[j] = 0.0, 1.0
            curr += best_delta
            if curr < best:
                best = curr; best_b = b.copy()
    return best_b, best

import numpy as np
from scipy.optimize import minimize
import time

def _np2(n):
    p = 1
    while p < n:
        p <<= 1
    return p

def _corr(h):
    n = len(h)
    N = _np2(2 * n - 1)
    a = np.zeros(N); a[:n] = h
    b = np.zeros(N); b[:n] = 1.0 - h
    c = np.fft.irfft(np.fft.rfft(a) * np.conj(np.fft.rfft(b)), N)
    return np.concatenate([c[N - n + 1:], c[:n]])

def c5val(h):
    n = len(h)
    return float(np.max(_corr(h)) / (n / 2.0))

def feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    tgt = 0.5 * n
    for _ in range(300):
        d = tgt - h.sum()
        if abs(d) < 1e-13:
            break
        mask = (h < 0.9999) if d > 0 else (h > 0.0001)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        np.clip(h, 0, 1, out=h)
    return h

def smooth_lbfgs(h, n, T, t0, budget, max_iter=300):
    """L-BFGS-B optimization with smooth max + penalty."""
    dx = 2.0 / n
    tgt = 0.5 * n
    lam = 100.0
    
    def obj(x):
        c = _corr(x)
        Tc = T * c
        Tc -= Tc.max()
        smax = (Tc.max() + np.log(np.sum(np.exp(Tc)))) / T
        pen = lam * (x.sum() - tgt) ** 2
        return smax * dx + pen
    
    x0 = feasible(h.copy(), n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, x0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-10,
                            'maxfun': 50000})
    x = feasible(res.x, n)
    return x, c5val(x)
