def project_c(h, n):
    target = n / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_at(h, k):
    n = len(h)
    g = np.zeros(n)
    s = abs(k)
    if s >= n:
        return g
    if k >= 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

def wsubgrad(h, T, top_k=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return subgrad_at(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * subgrad_at(h, int(i) - (n - 1))
    return g * dx

def nesterov_opt(h, n_iter, T0, T_end, top_k, eta0, decay=2.0):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    for it in range(n_iter):
        f = (it + 1) / n_iter
        T = T0 * (T_end / T0) ** f
        eta = eta0 / (1.0 + decay * f)
        g = wsubgrad(z, T, top_k)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        x_new = project_c(z - eta * g, n)
        z = x_new + (tk - 1.0) / tn * (x_new - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = x_new
        tk = tn
        if (it + 1) % 200 == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    c = compute_c5(x)
    if c < best_c:
        best_c, best_h = c, x.copy()
    return best_h, best_c

    n = len(h)

    v = np.zeros(n)

    best_c = compute_c5(best)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_at(h, k):
    n = len(h)
    g = np.zeros(n)
    s = abs(k)
    if s >= n:
        return g
    if k >= 0:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    else:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    return g

def wsubgrad(h, T, top_k=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return subgrad_at(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * subgrad_at(h, int(i) - (n - 1))
    return g * dx

def nesterov_opt(h, n_iter, T0, T_end, top_k, eta0, decay):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    best_c = compute_c5(x)
    best_h = x.copy()
    for it in range(n_iter):
        f = (it + 1) / n_iter
        T = T0 * (T_end / T0) ** f
        eta = eta0 / (1.0 + decay * f)
        g = wsubgrad(z, T, top_k)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        x_new = project_c(z - eta * g, n)
        z = x_new + (tk - 1.0) / tn * (x_new - x)
        np.clip(z, -0.5, 1.5, out=z)
        x = x_new
        tk = tn
        if (it + 1) % 200 == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = x.copy()
    c = compute_c5(x)
    if c < best_c:
        best_c, best_h = c, x.copy()
    return best_h, best_c

def adam_opt(h, n_iter, T, top_k, lr):
    n = len(h)
    x = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.9, 0.999
    best_c = compute_c5(x)
    best_h = x.copy()
    for it in range(n_iter):
        t = it + 1
        g = wsubgrad(x, T, top_k)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        mh = m / (1.0 - b1 ** t)
        vh = v / (1.0 - b2 ** t)
        x = project_c(x - lr * mh / (np.sqrt(vh) + 1e-8), n)
        if t % 200 == 0:
            c = compute_c5(x)
            if c < best_c:
                best_c, best_h = c, x.copy()
    c = compute_c5(x)
    if c < best_c:
        best_c, best_h = c, x.copy()
    return best_h, best_c

_GS = {}

def _gs_obj(h):
    n, T = _GS['n'], _GS['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def _gs_jac(h):
    n, T = _GS['n'], _GS['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    w = exps / np.sum(exps)
    idx = np.argsort(w)[::-1][:20]
    g = np.zeros(n)
    for i in idx:
        if w[i] < 1e-12:
            break
        g += w[i] * subgrad_at(h, int(i) - (n - 1))
    return g * dx

def _gs_con(h):
    return np.sum(h) - _GS['n'] / 2.0

def _gs_con_jac(h):
    return np.ones(_GS['n'])

def slsqp_opt(h, T, maxiter):
    n = len(h)
    _GS['n'], _GS['T'] = n, T
    try:
        r = minimize(_gs_obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n, jac=_gs_jac,
                     constraints=[{'type': 'eq', 'fun': _gs_con, 'jac': _gs_con_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

            else:

def thue_morse(n):
    t = np.zeros(1)
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]

def downsample(h, nn):
    n = len(h)
    if nn == n:
        return h.copy()
    ox = np.linspace(0, 2, n)
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

def t_rem(t0, budget_s):
    return budget_s - (time.time() - t0)

def make_binary(h, seed):
    nb = len(h)
    h_bin = (h > 0.5).astype(float)
    num_ones = int(np.sum(h_bin))
    target = nb // 2
    rng_b = np.random.RandomState(seed)
    if num_ones > target:
        ones = np.where(h_bin == 1.0)[0]
        flip = rng_b.choice(ones, num_ones - target, replace=False)
        h_bin[flip] = 0.0
    elif num_ones < target:
        zeros = np.where(h_bin == 0.0)[0]
        flip = rng_b.choice(zeros, target - num_ones, replace=False)
        h_bin[flip] = 1.0
    return h_bin
