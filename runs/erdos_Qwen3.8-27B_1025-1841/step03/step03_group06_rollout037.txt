from scipy.optimize import minimize
import time

_N = 960
_T = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    dx = 2.0 / n
    T = _T
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    return np.sum(x) - _N / 2.0

    return np.ones(_N)

            h_cand = make_feasible(res.x, n_pts)
            c5_cand = compute_c5(h_cand)
            if c5_cand < best_c5:
                best_c5 = c5_cand
                best_h = h_cand.copy()

            h_rebin = to_binary(h_cand, n_pts)
            c5_rebin = compute_c5(h_rebin)
            if c5_rebin < best_c5:
                best_c5 = c5_rebin
                best_h = h_rebin.copy()
        except Exception:
            pass

    if remaining() > 60:
        n_pts = len(best_h)
        _N = n_pts
        _T = 500.0
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        for attempt in range(3):
            if remaining() < 30:
                break
            noise = 0.01 * (attempt + 1)
            h_start = make_feasible(best_h + noise * rng.randn(n_pts), n_pts)
            try:
                res = minimize(_obj_global, h_start, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 200, 'ftol': 1e-15})
                h_cand = make_feasible(res.x, n_pts)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h_cand.copy()
            except Exception:
                pass

    n_pts = len(best_h)
    dx = 2.0 / n_pts
    for round_num in range(50):
        if remaining() < 5:
            break
        improved = False
        for j in range(n_pts):
            if remaining() < 3:
                break
            old = best_h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.05), min(1.0, old + 0.05)]:
                if cand == old:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    h_final = make_feasible(best_h, len(best_h))
    c5_final = compute_c5(h_final)
    return h_final, c5_final, len(h_final)
