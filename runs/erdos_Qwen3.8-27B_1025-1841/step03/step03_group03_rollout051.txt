def compute_gradient(h, t):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, 'full')
    
    # Soft-max weights
    shifted = corr - np.max(corr)
    weights = np.exp(t * shifted)
    weights /= np.sum(weights)
    
    # Gradient: grad[j] = dx * sum_k weights[k] * dcorr[k]/dh[j]
    grad = np.zeros(n)
    
    for k in range(2*n-1):
        if weights[k] < 1e-10:
            continue
        lag = k - (n-1)
        if lag == 0:
            grad += dx * weights[k] * (1.0 - 2.0 * h)
        elif lag > 0:
            # dcorr[k]/dh[j] = (1-h[j+lag])*1[j+lag<n] - h[j-lag]*1[j>=lag]
            term1 = (1.0 - h[lag:n])  # for j=0 to n-1-lag: (1-h[j+lag])
            term2 = h[:n-lag]         # for j=lag to n-1: h[j-lag]
            # Build full gradient vector
            grad_k = np.zeros(n)
            grad_k[:n-lag] += (1.0 - h[lag:n])
            grad_k[lag:] -= h[:n-lag]
            grad += dx * weights[k] * grad_k
        else:
            l = -lag
            # dcorr[k]/dh[j] = (1-h[j-l])*1[j>=l] - h[j+l]*1[j+l<n]
            term1 = (1.0 - h[:n-l])   # for j=l to n-1: (1-h[j-l])
            term2 = h[l:n]            # for j=0 to n-1-l: h[j+l]
            grad_k = np.zeros(n)

            grad_k[:n-l] -= h[l:n]

def softmax_gradient(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, 'full')
    mc = np.max(corr)
    shifted = corr - mc
    w = np.exp(t * shifted)
    w /= np.sum(w)
    
    grad = np.zeros(n)
    # lag = 0 contribution
    grad += w[n-1] * (1.0 - 2.0 * h)
    
    # Positive lags
    for l in range(1, n):
        wl = w[n-1+l]
        if wl < 1e-12:
            continue
        # (1-h[j+l]) for j=0..n-1-l
        grad[:n-l] += wl * (1.0 - h[l:])
        # -h[j-l] for j=l..n-1
        grad[l:] -= wl * h[:n-l]
    
    # Negative lags
    for l in range(1, n):
        wl = w[n-1-l]
        if wl < 1e-12:
            continue
        # (1-h[j-l]) for j=l..n-1
        grad[l:] += wl * (1.0 - h[:n-l])
        # -h[j+l] for j=0..n-1-l
        grad[:n-l] -= wl * h[l:]
    
    return dx * grad
