import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def obj_topk(h, n, K, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -K)[-K:]
    topk = corr[idx]
    mc = np.max(topk)
    shifted = topk - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def obj_cons(h, n):
    return np.sum(h) - n / 2.0

def slsqp_topk(h_start, n, K, T, max_iter):
    dx = 2.0 / n
    def f(x):
        return obj_topk(x, n, K, T)
    def fgrad(x):
        dx_ = 2.0 / n
        perturb = 1e-7
        grad = np.zeros(n)
        base = f(x)
        for j in range(n):
            xp = x.copy()
            xp[j] = np.clip(xp[j] + perturb, 0.0, 1.0)
            xm = x.copy()
            xm[j] = np.clip(xm[j] - perturb, 0.0, 1.0)
            # Use central difference where possible
            if 0 < x[j] < 1:
                grad[j] = (f(xp) - f(xm)) / (2 * perturb)
            elif x[j] >= 1:
                grad[j] = (f(xp) - base) / (xp[j] - x[j]) if xp[j] != x[j] else 0.0
            else:
                grad[j] = (base - f(xm)) / (x[j] - xm[j]) if x[j] != xm[j] else 0.0
        return grad
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    # Use analytical gradient for speed at moderate n
    if n <= 1280:
        res = minimize(f, h_start, jac=fgrad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    else:
        res = minimize(f, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'iprint': 0})
    return make_feasible(res.x, n)

def binary_round(h, n):
    """Round h to binary while maintaining sum constraint."""
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    target = n // 2
    cur_sum = int(np.sum(h))
    if cur_sum > target:
        excess = cur_sum - target
        zeros = np.where(h == 1.0)[0]
        idx = rng_binary_permutation(zeros, excess)
        h[idx] = 0.0
    elif cur_sum < target:
        deficit = target - cur_sum
        zeros = np.where(h == 0.0)[0]
        idx = rng_binary_permutation(zeros, deficit)
        h[idx] = 1.0
    return h

_rng_state = None
def rng_binary_permutation(arr, k):
    global _rng_state
    if _rng_state is None:
        _rng_state = np.random.RandomState(123)
    k = min(k, len(arr))
    if k <= 0:
        return arr[:0]
    return arr[_rng_state.choice(len(arr), k, replace=False)]

def greedy_binary_search(h_bin, n, max_rounds=50, top_k=50):

    global _rng_state
    if _rng_state is None:
        _rng_state = np.random.RandomState(123)
    h = h_bin.copy()

    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    for rnd in range(max_rounds):
        best_improve = 0
        best_pair = None

        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)

        k_max = np.argmax(corr)
        shift = k_max - (n - 1)
        
        # Generate candidate swaps
        candidates = []
        step = max(1, len(ones) // (top_k * 2))
        for i_idx in range(0, len(ones), step):
            i = ones[i_idx]
            # Find zeros near i+shift (where h[i]=1 overlaps with h[i+shift]=0)
            for j_idx in range(0, len(zeros), step):
                j = zeros[j_idx]
                # Heuristic: prefer j near i+shift or i-shift
                dist = min(abs(j - (i + shift)), abs(j - (i - shift)))
                candidates.append((dist, i, j))
        
        candidates.sort()
        tested = 0
        for dist, i, j in candidates:
            if tested >= top_k:
                break
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h)
            if c < best_c5 - 1e-15:
                best_improve = best_c5 - c
                best_pair = (i, j)
                best_c5 = c
                # Re-extract ones and zeros
                ones = np.where(h == 1.0)[0]
                zeros = np.where(h == 0.0)[0]
            else:
                h[i], h[j] = h[j], h[i]
            tested += 1

        for _ in range(top_k):
            i = ones[_rng_state.randint(len(ones))]
            j = zeros[_rng_state.randint(len(zeros))]

            c = compute_c5(h)
            if c < best_c5 - 1e-15:
                best_improve = best_c5 - c
                best_c5 = c

            else:
                h[i], h[j] = h[j], h[i]

            break
    
    return h, best_c5

def coord_descent_polish(h, n, max_rounds=5, step_list=None):
    """Coordinate descent: try adjusting each element."""
    if step_list is None:
        step_list = [0.01, -0.01, 0.005, -0.005, 0.02, -0.02]
    h = h.copy()

    target = n / 2.0

        for j in range(n):
            old = h[j]
            for delta in step_list:
                nv = old + delta
                if nv < 0 or nv > 1:
                    continue
                h[j] = nv
                # Fix sum constraint approximately
                diff = target - np.sum(h)
                if abs(diff) > 1e-6:
                    # Quick fix: distribute to other elements
                    mask = np.arange(n) != j
                    mask &= (h < 1.0 - 1e-6) if diff > 0 else (h > 1e-6)
                    cnt = np.sum(mask)
                    if cnt == 0:

                        continue

                c = compute_c5(h)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    improved = True
                else:
                    h[j] = old
            # Try binary

                if abs(bv - h[j]) < 1e-9:
                    continue
                h[j] = bv
                diff = target - np.sum(h)
                if abs(diff) > 1e-6:
                    mask = np.arange(n) != j
