import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sg_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    s = abs(k)
    if s >= n:
        return g
    if k >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def fista_opt(h0, n_iter=4000, T0=30.0, T1=0.1, lr0=0.12, lr1=0.01, tk_max=150, check=30):
    n = len(h0)
    dx = 2.0 / n
    x = h0.astype(float).copy()
    z = x.copy()
    t = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr0 * (1.0 - 0.7 * f) + lr1 * 0.7 * f
        corr = np.correlate(z, 1.0 - z, mode='full')
        mc = np.max(corr)
        exps = np.exp(np.clip(T * (corr - mc), -50, 0))
        tot = exps.sum()
        if tot < 1e-300:
            w = np.zeros(len(corr))
            w[np.argmax(corr)] = 1.0
        else:
            w = exps / tot
        idx = np.argsort(w)[::-1][:tk_max]
        w_sub = w[idx]
        w_sub = w_sub / w_sub.sum()
        g = np.zeros(n)
        for ii in range(len(idx)):
            ww = w_sub[ii]
            if ww < 1e-12:
                break
            k = idx[ii] - (n - 1)
            g += ww * sg_shift(z, k)
        g *= dx
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        z = xn + (t - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        t = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc = c
                bh = x.copy()
    c = compute_c5(x)
    if c < bc:
        bc = c
        bh = x.copy()
    return bh, bc

def periodic_optimize(p, n_target, seed=42):
    """Optimize a p-periodic pattern and tile to n_target."""
    rng = np.random.RandomState(seed)
    
    def obj(v):
        # v has period p, sum(v) = p/2
        max_ov = 0.0
        for k in range(p):
            ov = 0.0
            for j in range(p):
                ov += v[j] * (1.0 - v[(j + k) % p])
            max_ov = max(max_ov, ov)
        return (2.0 / p) * max_ov
    
    def obj_grad(v):
        # Need gradient of max_k sum_j v[j]*(1-v[(j+k)%p])
        # Find worst k
        overs = []
        for k in range(p):
            ov = 0.0
            for j in range(p):
                ov += v[j] * (1.0 - v[(j + k) % p])
            overs.append(ov)
        k_best = int(np.argmax(overs))
        g = np.zeros(p)
        for j in range(p):
            # d/dv[j] of sum_i v[i]*(1-v[(i+k)%p])
            # = (1 - v[(j+k)%p]) - v[(j-k)%p]
            g[j] = (1.0 - v[(j + k_best) % p]) - v[(j - k_best) % p]
        return (2.0 / p) * g
    
    best_v = None
    best_c = float('inf')
    for trial in range(20):
        if trial == 0:
            v = np.full(p, 0.5)
        elif trial == 1:
            v = np.where(np.arange(p) < p // 2, 1.0, 0.0)
        elif trial == 2:
            v = np.where(np.arange(p) % 2 == 0, 1.0, 0.0)
        else:
            v = rng.rand(p)
        v = project_periodic(v, p)
        try:
            res = minimize(obj, v, jac=obj_grad, method='SLSQP',
                          bounds=[(0, 1)] * p,
                          constraints=[{'type': 'eq', 'fun': lambda v: np.sum(v) - p / 2.0,
                                       'jac': lambda v: np.ones(p)}],
                          options={'maxiter': 200, 'ftol': 1e-12})
            if res.success and res.fun < best_c:
                best_c = res.fun
                best_v = project_periodic(res.x, p)
        except Exception:
            pass
    
    if best_v is None:
        return None, float('inf')
    
    # Tile to n_target
    h = np.tile(best_v, n_target // p + 1)[:n_target]
    h = project_c(h, n_target)
    c = compute_c5(h)
    return h, c

def slsqp_smooth(h, T=5.0, maxiter=400):
    n = len(h)
    dx = 2.0 / n
    
    def sobj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx
    
    def sgrad(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        w = exps / exps.sum()
        g = np.zeros(n)
        for i in range(len(w)):
            if w[i] < 1e-10:
                continue
            k = i - (n - 1)
            g += w[i] * sg_shift(hv, k)
        # Plus derivative of the log-sum-exp term
        # d/dh of (mc + log(sum(exp(T*(c-mc))))/T)
        # The log-sum-exp derivative: sum_j w_j * dc_j/dh where w_j = exp(T*(c_j-mc))/sum(...)
        # But we already computed this above. The mc term: d(mc)/dh is subgradient of max.
        # Actually the smooth approximation handles this automatically through the softmax.
        return g * dx
    
    def scone(hv):
        return np.sum(hv) - n / 2.0
    
    try:
        r = minimize(sobj, h, jac=sgrad, method='SLSQP',
                    bounds=[(0.0, 1.0)] * n,
                    constraints=[{'type': 'eq', 'fun': scone, 'jac': lambda h: np.ones(n)}],
                    options={'maxiter': maxiter, 'ftol': 1e-14})
        if r.x is not None:
            return project_c(r.x, n)
    except Exception:
        pass
    return h

def upsample(h, nn):

        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
