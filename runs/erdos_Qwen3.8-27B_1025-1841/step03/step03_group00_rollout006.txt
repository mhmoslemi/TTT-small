    for i in range(n_main):
        fr = (i * golden) % 1.0
        gr[i] = 1.0 if fr < 0.5 else 0.0
    gr = make_feasible(gr, n_main)
    starts.append(gr.copy())

    # Phase 1: L-BFGS-B smooth max on each start
    T_list = [50, 150, 500]
    for si, h_s in enumerate(starts):
        if tl() < 60:
            break
        for T in T_list:
            if tl() < 30:
                break
            try:
                h_opt = lbgfb_smooth(h_s, n_main, T, mu=1000.0, max_iter=200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_s = h_opt.copy()
            except Exception:
                pass

    # Phase 2: Active-set iterative optimization
    if tl() > 100:
        for iteration in range(6):
            if tl() < 40:
                break
            active = find_active_lags(best_h, top_m=10)
            for T in [200, 800]:
                if tl() < 20:
                    break
                try:
                    h_opt = lbgfb_active(best_h, n_main, active, T, mu=1000.0, max_iter=150)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                except Exception:
                    pass
            # Re-identify active lags after improvement
            if tl() < 30:
                break

    # Phase 3: Binary local search
    if tl() > 80:
        h_bin = make_binary_from_fractional(best_h)
        h_bin, c5_bin = binary_local_search_fast(h_bin, n_main, rng, max_sweeps=12, tries_per_sweep=4000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # Also try from Thue-Morse binary
        tm_bin = (thue_morse(n_main) > 0).astype(float)
        tm_bin = make_feasible(tm_bin, n_main)
        h_b2, c5_b2 = binary_local_search_fast(tm_bin, n_main, rng, max_sweeps=8, tries_per_sweep=3000)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = h_b2.copy()

    # Phase 4: Upsample to higher resolution and refine
    for n_up in [1536, 2048]:
        if tl() < 80:
            break
        h_c = upsample_h(best_h, n_up)
        c5_c = compute_c5(h_c)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c.copy()
        for T in [100, 400, 1200]:
            if tl() < 30:
                break
            try:
                h_opt = lbgfb_smooth(h_c, n_up, T, mu=500.0, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass
        # Active set at higher res
        if tl() > 40:
            active = find_active_lags(best_h, top_m=10)
            try:
                h_opt = lbgfb_active(best_h, n_up, active, 500, mu=500.0, max_iter=120)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Binary refinement at current best resolution
    n_cur = len(best_h)
    if tl() > 60:
        h_bin = make_binary_from_fractional(best_h)
        h_bin, c5_bin = binary_local_search_fast(h_bin, n_cur, rng, max_sweeps=10, tries_per_sweep=3000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 6: Coordinate descent polish
    n_cd = min(len(best_h), 1024)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    if tl() > 30:
        h_cd, c5_cd = coordinate_descent(h_cd, n_cd, max_rounds=4, time_limit=min(60, max(10, tl() - 20)))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final SLSQP squeeze
    if tl() > 30:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        def slsqp_obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            mc = np.max(corr)
            shifted = 3000.0 * (corr - mc)
            shifted = np.clip(shifted, -80, 80)
            return (mc + np.log(np.sum(np.exp(shifted))) / 3000.0) * dx_f
        def slsqp_con(h):
            return np.sum(h) - n_f / 2.0
        def slsqp_jac(h):
            return np.ones(n_f)
        bounds = [(0.0, 1.0)] * n_f
        cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac}]
        try:
            res = minimize(slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 80, 'ftol': 1e-16})
            h_opt = make_feasible(res.x, n_f)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
