    """Project h onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    # Bisection on lambda
    h_arr = np.asarray(h, dtype=float)
    lo, hi = -1.0, 1.0  # range for lambda (h - lambda)
    # Actually lambda can be outside [-1,1] if h is far from feasible
    lo, hi = h_arr.min() - target, h_arr.max() + (n - target)
    # More carefully: we need sum(clip(h - lam, 0, 1)) = target
    # As lam -> -inf, clip -> 1, sum -> n
    # As lam -> +inf, clip -> 0, sum -> 0
    lo = h_arr.max() + 1  # sum will be 0
    hi = h_arr.min() - 1  # sum will be n
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h_arr - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    h_proj = np.clip(h_arr - lam, 0, 1)
    # Numerical fix
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:

        else:

            h_proj[mask] += diff / np.sum(mask)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def corr_gradient_top(h, n, t, top_m=8):
    """Compute soft-max weighted gradient of correlation max."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    idx = np.argsort(corr)[::-1][:top_m]
    vals = corr[idx]
    w = np.exp(t * (vals - mc))
    sw = np.sum(w)
    w /= sw
    val = (mc + np.log(sw) / t) * (2.0 / n)

    grad = np.zeros(n)
    for wi, k in zip(w, idx):
        if wi < 1e-14:
            continue
        L = k - (n - 1)
        if L >= 0:
            grad[:n - L] += wi * (1.0 - h[L:n])
            grad[L:] -= wi * h[:n - L]
        else:
            M = -L
            grad[M:] += wi * (1.0 - h[:n - M])
            grad[:n - M] -= wi * h[M:n]
    return val, grad * (2.0 / n)

def pgd_optimize(h_init, n, t_values, max_iter_per_t, start_time, time_limit, lr_init=0.05):
    h = h_init.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    lr = lr_init

    for t in t_values:
        if time.time() - start_time > time_limit:
            break
        lr = lr_init
        for it in range(max_iter_per_t):
            if time.time() - start_time > time_limit:
                break
            val, grad = corr_gradient_top(h, n, t, top_m=8)
            c_curr = compute_c5(h)

            # Backtracking line search
            step = lr
            found = False
            for _ in range(8):
                h_trial = np.clip(h - step * grad, 0.0, 1.0)
                h_trial = make_feasible(h_trial, n)
                c_trial = compute_c5(h_trial)
                if c_trial < c_curr - 1e-14:
                    h = h_trial
                    lr = min(step * 1.3, 0.2)
                    found = True
                    if c_trial < best_c:
                        best_c = c_trial
                        best_h = h_trial.copy()
                    break
                step *= 0.5
            if not found:
                lr *= 0.85
                if lr < 1e-7:
                    break
    return best_h, best_c

def coord_descent_opt(h_init, n, max_rounds, start_time, time_limit):

    best_c = compute_c5(h)
    improved_any = True
    for rnd in range(max_rounds):

                h[j] = best_val

        h4, c4 = coord_descent_opt(best_h, n_cd, max_rounds=12, start_time=start_time, time_limit=remaining())
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4

    # Phase 5: Try upsample to even higher res
    if remaining() > 150:
        n5 = len(best_h) * 2
        if n5 <= 4096:
            h5 = upsample_h(best_h, n5)
            h5, c5 = pgd_optimize(h5, n5, [60, 150, 500], 200, start_time, min(remaining() - 60, 200))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h5
            if remaining() > 40:
                h5b, c5b = coord_descent_opt(h5, n5, max_rounds=5, start_time=start_time, time_limit=remaining())
                if c5b < best_c5:
                    best_c5 = c5b
                    best_h = h5b

    # Phase 6: Binary local search
    if remaining() > 60:
        h6, c6 = binary_search_local(best_h, len(best_h), max_rounds=5, start_time=start_time, time_limit=remaining())
        if c6 < best_c5:
            best_c5 = c6
            best_h = h6

    # Final refinement
    if remaining() > 20:
        h_final, c_final = coord_descent_opt(best_h, len(best_h), max_rounds=5, start_time=start_time, time_limit=remaining())
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
