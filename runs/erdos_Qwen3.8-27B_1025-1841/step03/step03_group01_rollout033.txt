    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def logsumexp_grad(h, T):
    """Compute gradient of log-sum-exp smoothed C5 objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = T * (corr - c_max)
    # Avoid overflow
    weights = np.exp(shifted - np.max(shifted))
    weights /= np.sum(weights)
    
    # Gradient: for each i, sum over m of weights[m] * dc_m/dh_i
    # dc_m/dh_i = (1-h[i+s_m]) * 1_{0<=i+s_m<n} + (-h[i-s_m]) * 1_{0<=i-s_m<n}
    # where s_m = m - (n-1)
    
    grad = np.zeros(n)
    for i in range(n):
        for m in range(2*n-1):
            s = m - (n - 1)
            val = 0.0
            if 0 <= i + s < n:
                val += (1.0 - h[i + s])
            if 0 <= i - s < n:
                val += (-h[i - s])
            grad[i] += weights[m] * val
    grad *= dx
    return grad

    """Fast C5 computation for binary h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def sa_binary(h, n, max_iters, start_T, end_T, time_limit):
    """Simulated annealing on binary vectors with exactly n/2 ones."""
    start_time = time.time()
    h = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if len(ones_idx) > n // 2:
        # Remove excess ones
        excess = len(ones_idx) - n // 2
        to_remove = np.random.choice(ones_idx, excess, replace=False)
        h[to_remove] = 0
    elif len(ones_idx) < n // 2:
        # Add ones
        deficit = n // 2 - len(ones_idx)
        to_add = np.random.choice(zeros_idx, deficit, replace=False)
        h[to_add] = 1
    
    best_c5 = compute_c5_binary(h)
    best_h = h.copy()
    current_c5 = best_c5
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        T = start_T * (end_T / start_T) ** (it / max_iters)
        
        # Pick a random 1 and a random 0, swap them
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h[i] == h[j]:
            continue
        
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5_binary(h)
        
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]  # revert
    
    return best_h, best_c5

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

        h[remove] = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = float(n) / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    if abs(h.sum() - target) > 1e-14:
        idx = np.argmin(h) if h.sum() < target else np.argmax(h)
        h[idx] = np.clip(h[idx] + (target - h.sum()), 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n, rng):
    h = (h > 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        h[rng.choice(ones, len(ones) - target, replace=False)] = 0
    elif len(ones) < target:
        h[rng.choice(zeros, target - len(ones), replace=False)] = 1
    return h

def binary_sa(h_init, n, time_limit, rng):
    st = time.time()
    h = h_init.copy()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    stag = 0
    max_stag = 40000
    T0, T1 = 0.004, 0.00003

    for it in range(20_000_000):
        if time.time() - st >= time_limit:
            break
        p = min(1.0, it / 300000.0)
        T = T0 * (T1 / T0) ** p
        r = rng.random()

        if r < 0.55:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i], h[j] = h[j], h[i]
        elif r < 0.80:
            L = rng.randint(3, 21)
            i = rng.randint(0, n - L)
            j = rng.randint(0, n - L)
            if j < i + L:
                j = i + L
                if j >= n - L:
                    j = rng.randint(0, n - L)
                    if j < i + L:
                        continue
            b1, b2 = h[i:i+L].copy(), h[j:j+L].copy()
            h[i:i+L], h[j:j+L] = b2, b1
        else:
            ns = rng.randint(2, 6)
            swaps = []
            for _ in range(ns):
                ii = ones[rng.randint(len(ones))]
                jj = zeros[rng.randint(len(zeros))]
                h[ii], h[jj] = h[jj], h[ii]
                swaps.append((ii, jj))

        new = compute_c5(h)
        d = new - cur
        if d < 0 or (T > 1e-12 and rng.random() < np.exp(-d / T)):
            cur = new

def binary_ls(h, n, max_passes=30, deadline=None):
    cur = compute_c5(h)
    best, best_h = cur, h.copy()
    for _ in range(max_passes):
        if deadline and time.time() >= deadline:
            break
        improved = False
        for i in range(n):
            if deadline and time.time() >= deadline:
                break
            h[i] = 1.0 - h[i]
            v = compute_c5(h)
            if v < cur - 1e-15:
                cur = v
                if v < best:
                    best, best_h = v, h.copy()
                improved = True
            else:
                h[i] = 1.0 - h[i]
        if not improved:
            break
    return best_h, best

def _obj_grad_smoothed(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * corr
    Tc_max = Tc.max()
    exp_Tc = np.exp(Tc - Tc_max)
    norms = exp_Tc.sum()
    softmax = exp_Tc / norms
    obj = (Tc_max + np.log(norms)) / T * dx
    g = np.zeros(n)
    for m in range(2 * n - 1):
        wm = softmax[m]
        if wm < 1e-16:
            continue
        s = m - (n - 1)
        if s >= 0:
            g[:n - s] += wm * (1.0 - h[s:])
            g[s:] -= wm * h[:n - s]
        else:
            ss = -s
            g[ss:] += wm * (1.0 - h[:n - ss])
            g[:n - ss] -= wm * h[ss:]
    g *= dx
    return obj, g

def lbfgsb_aug(h_start, n, T, max_iter=150, n_aug=4):
    target = float(n) / 2.0
    mu = 1.0
    lam = 0.0
    h = h_start.copy()
    for _ in range(n_aug):
        def f_obj(x):
            o, _ = _obj_grad_smoothed(x, T, n)
            s = x.sum() - target
            return o + lam * s + 0.5 * mu * s * s
        def f_grad(x):
            _, g = _obj_grad_smoothed(x, T, n)
            s = x.sum() - target
            return g + (lam + mu * s)
        try:
            res = minimize(f_obj, h, method='L-BFGS-B', jac=f_grad,
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
            h = res.x
        except Exception:
            break
        cv = abs(h.sum() - target)
        lam += mu * (h.sum() - target)
        mu = mu * 0.5 if cv < 1e-6 else mu * 3.0
        if cv < 1e-5:
            break
    return make_feasible(h, n)

def slsqp_opt(h_start, n, T, max_iter=200):
    dx = 2.0 / n
    target = float(n) / 2.0
    def f_obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = corr.max()
        sh = T * (corr - mc) - T * (corr - mc).max()
        return (mc + np.log(np.exp(sh).sum()) / T) * dx
    def f_cons(x):
        return 0.0 if abs(x.sum() - target) < 1e-10 else x.sum() - target
    def f_cons_jac(x):
        return np.ones(n)
    try:
        res = minimize(f_obj, h_start, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': f_cons, 'jac': f_cons_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)
