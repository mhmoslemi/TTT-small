def compute_delta(h, j1, j2, n):
    """Compute ΔS(k) for all shifts k when swapping h[j1]:1→0 and h[j2]:0→1."""
    k_range = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # I1: 1 if 0 <= j1+k < n and h[j1+k] == 0
    jp1 = j1 + k_range  # j1+k for all k
    m1 = (jp1 >= 0) & (jp1 < n)
    I1 = np.zeros(2*n-1, dtype=float)
    I1[m1] = (h[jp1[m1]] == 0).astype(float)
    
    # I2: 1 if 0 <= j2+k < n and h'[j2+k] == 0
    # h'[j2+k] = 0 iff j2+k == j1 OR (h[j2+k] == 0 and j2+k != j1)
    jp2 = j2 + k_range
    m2 = (jp2 >= 0) & (jp2 < n)
    I2 = np.zeros(2*n-1, dtype=float)
    # Where jp2 == j1, h' = 0
    I2[jp2 == j1] = 1.0
    # Where jp2 != j1 and in range, h'[jp2] = h[jp2]
    m2_not_j1 = m2 & (jp2 != j1)
    I2[m2_not_j1] = (h[jp2[m2_not_j1]] == 0).astype(float)
    
    # I3: 1 if 0 <= j1-k < n, j1-k not in {j1, j2}, and h[j1-k] == 1
    jm1 = j1 - k_range
    m3 = (jm1 >= 0) & (jm1 < n) & (jm1 != j1) & (jm1 != j2)
    I3 = np.zeros(2*n-1, dtype=float)
    I3[m3] = (h[jm1[m3]] == 1).astype(float)
    
    # I4: 1 if 0 <= j2-k < n, j2-k not in {j1, j2}, and h[j2-k] == 1
    jm2 = j2 - k_range
    m4 = (jm2 >= 0) & (jm2 < n) & (jm2 != j1) & (jm2 != j2)
    I4 = np.zeros(2*n-1, dtype=float)
    I4[m4] = (h[jm2[m4]] == 1).astype(float)
    
    return -I1 + I2 + I3 - I4

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def to_01(h, n):
    n_ones = n // 2
    sort_idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[sort_idx[:n_ones]] = 1
    return result


def compute_delta(h, j1, j2, n, k_range):
    L = 2 * n - 1
    jp1 = j1 + k_range
    jp2 = j2 + k_range
    jm1 = j1 - k_range
    jm2 = j2 - k_range

    m1 = (jp1 >= 0) & (jp1 < n)
    m2 = (jp2 >= 0) & (jp2 < n)
    m3 = (jm1 >= 0) & (jm1 < n) & (jm1 != j1) & (jm1 != j2)
    m4 = (jm2 >= 0) & (jm2 < n) & (jm2 != j1) & (jm2 != j2)

    I1 = np.zeros(L)
    I1[m1] = (h[jp1[m1]] == 0).astype(np.float64)

    I2 = np.zeros(L)
    mask_jp2_eq_j1 = (jp2 == j1)
    I2[mask_jp2_eq_j1] = 1.0
    m2b = m2 & ~mask_jp2_eq_j1
    I2[m2b] = (h[jp2[m2b]] == 0).astype(np.float64)

    I3 = np.zeros(L)
    I3[m3] = (h[jm1[m3]] == 1).astype(np.float64)

    I4 = np.zeros(L)
    I4[m4] = (h[jm2[m4]] == 1).astype(np.float64)

    return -I1 + I2 + I3 - I4


def local_search_01(h, n, time_limit, start_time, n_candidates=400, n_worst=5):
    h = to_01(h, n).copy()
    dx = 2.0 / n
    k_range = np.arange(-(n - 1), n)
    L = 2 * n - 1

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = corr.max() * dx
    best_h = h.copy()

    no_improve = 0
    total_iters = 0

    while time.time() - start_time < time_limit:
        current_max = corr.max()

        top_count = min(n_worst, L)
        worst_indices = np.argpartition(corr, -top_count)[-top_count:]

        best_move = None
        best_new_max = current_max - 1e-12

        for widx in worst_indices:
            shift = widx - (n - 1)

            if shift >= 0:
                j_min = 0
                j_max = n - shift
            else:
                j_min = -shift
                j_max = n

            if j_max <= j_min:
                continue

            j_range = np.arange(j_min, j_max)
            bad_mask = (h[j_range] == 1) & (h[j_range + shift] == 0)
            n_bad = bad_mask.sum()

            if n_bad == 0:
                continue

            if n_bad > n_candidates:
                idx = np.random.choice(n_bad, n_candidates, replace=False)
                candidates = j_range[bad_mask][idx]
            else:
                candidates = j_range[bad_mask]

            for j1 in candidates:
                j2 = j1 + shift
                if j2 < 0 or j2 >= n or h[j2] != 0:
                    continue

                delta = compute_delta(h, j1, j2, n, k_range)
                new_max = (corr + delta).max()

                if new_max < best_new_max:
                    best_new_max = new_max
                    best_move = (j1, j2)

        total_iters += 1

        if best_move is not None:
            j1, j2 = best_move
            delta = compute_delta(h, j1, j2, n, k_range)
            h[j1] = 0
            h[j2] = 1
            corr = corr + delta
            c5_now = corr.max() * dx
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve == 500:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                n_pert = min(20, len(ones), len(zeros))
                fa = np.random.choice(ones, n_pert, replace=False)
                fb = np.random.choice(zeros, n_pert, replace=False)
                for a, b in zip(fa, fb):
                    delta = compute_delta(h, a, b, n, k_range)
                    h[a] = 0
                    h[b] = 1
                    corr = corr + delta
                no_improve = 0
            elif no_improve == 2000:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                n_pert = min(50, len(ones), len(zeros))
                fa = np.random.choice(ones, n_pert, replace=False)
                fb = np.random.choice(zeros, n_pert, replace=False)
                for a, b in zip(fa, fb):
                    delta = compute_delta(h, a, b, n, k_range)
                    h[a] = 0
                    h[b] = 1
                    corr = corr + delta
                no_improve = 0
            elif no_improve > 5000:
                break

    return best_h, best_c5


def local_search_2swap(h, n, time_limit, start_time):
    """Try 2-swaps to escape local optima."""
    h = to_01(h, n).copy()
    dx = 2.0 / n
    k_range = np.arange(-(n - 1), n)
    L = 2 * n - 1

    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = corr.max() * dx
    best_h = h.copy()

    no_improve = 0

    while time.time() - start_time < time_limit:
        current_max = corr.max()
        top_count = min(3, L)
        worst_indices = np.argpartition(corr, -top_count)[-top_count:]

        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]

        found = False
        for widx in worst_indices:
            shift = widx - (n - 1)
            if shift >= 0:
                j_min, j_max = 0, n - shift
            else:
                j_min, j_max = -shift, n
            if j_max <= j_min:
                continue

            j_range = np.arange(j_min, j_max)
            bad_mask = (h[j_range] == 1) & (h[j_range + shift] == 0)
            bad_pos = j_range[bad_mask]

            if len(bad_pos) < 2:
                continue

            # Try pairs of bad positions
            n_try = min(200, len(bad_pos))
            idx = np.random.choice(len(bad_pos), n_try, replace=False)
            cands = bad_pos[idx]

            for i in range(len(cands)):
                j1 = cands[i]
                j2 = j1 + shift
                if j2 < 0 or j2 >= n or h[j2] != 0:
                    continue
                for m in range(i + 1, min(i + 5, len(cands))):
                    j3 = cands[m]
                    if j3 == j1 or j3 == j2:
                        continue
                    j4 = j3 + shift
                    if j4 < 0 or j4 >= n or h[j4] != 0 or j4 == j1 or j4 == j2:
                        continue

                    d1 = compute_delta(h, j1, j2, n, k_range)
                    h[j1] = 0
                    h[j2] = 1
                    d2 = compute_delta(h, j3, j4, n, k_range)
                    h[j3] = 0
                    h[j4] = 1
                    h[j1] = 1
                    h[j2] = 0

                    new_max = (corr + d1 + d2).max()
                    if new_max < current_max - 1e-12:
                        h[j1] = 0
                        h[j2] = 1
                        h[j3] = 0
                        h[j4] = 1
                        corr = corr + d1 + d2
                        c5_now = corr.max() * dx
                        if c5_now < best_c5:
                            best_c5 = c5_now
                            best_h = h.copy()
                        found = True
                        no_improve = 0
                        break
                if found:
                    break
            if found:
                break

        if not found:
            no_improve += 1
            if no_improve > 2000:
                break

    return best_h, best_c5
