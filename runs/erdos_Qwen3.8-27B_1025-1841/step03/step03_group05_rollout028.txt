import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def full_softmax_grad(h, T, top_k=300):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return _subgrad_one(h, idx - (n - 1)), corr
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g, corr

def fista(h, n_iter=8000, T0=50.0, T1=0.05, lr=0.10, decay_exp=0.5, check=50, top_k=300):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (decay_exp ** f)
        g, _ = full_softmax_grad(z, T, top_k)
        gnorm = np.linalg.norm(g)
        if gnorm > 0:
            lrt = min(lrt, 0.05 / gnorm)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def slsqp_polish(h, T=8.0, maxiter=150, top_k=200):
    n = len(h)
    dx = 2.0 / n

    def obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        return (mc + np.log(np.sum(exps)) / T) * dx

    def obj_grad(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))
        w = exps / np.sum(exps)
        idx = np.argsort(w)[::-1][:top_k]
        g = np.zeros(n)
        for i in idx:
            wi = w[i]
            if wi < 1e-12:
                break
            g += wi * _subgrad_one(hv, i - (n - 1))
        return g * dx

    def con_fun(hv):
        return np.sum(hv) - n / 2.0

    def con_jac(hv):
        return np.ones(n)

    try:
        r = minimize(obj, h, jac=obj_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h), endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    return project_c(np.interp(nx, ox, h), nn)

def multishift_qp_polish(h, n_rounds=20, top_m=5, alpha=0.1):
    """Polish by solving a QP that penalizes the top-m shifts."""
    n = len(h)
    dx = 2.0 / n
    curr = h.copy()
    bc = compute_c5(curr)
    for _ in range(n_rounds):
        corr = np.correlate(curr, 1.0 - curr, mode='full')
        top_idx = np.argsort(corr)[::-1][:top_m]
        T = 2.0
        mc = np.max(corr)
        exps = np.exp(T * (corr - mc))
        w = exps / np.sum(exps)
        g = np.zeros(n)
        for i in top_idx:
            wi = w[i]
            if wi < 1e-10:
                continue
            g += wi * _subgrad_one(curr, i - (n - 1))
        step = 2.0 / (alpha + np.linalg.norm(g)**2 + 1e-10)
        step = min(step, 0.5)
        new_h = project_c(curr - step * g, n)
        c_new = compute_c5(new_h)
        if c_new < bc:
            curr = new_h
            bc = c_new
    return curr, bc
