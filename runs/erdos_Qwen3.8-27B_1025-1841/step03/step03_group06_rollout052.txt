    mc = np.max(corr)
    shifted = corr - mc
    exps = np.exp(T * shifted)
    sw = exps.sum()
    weights = exps / sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[0:n-lag] += w * (1.0 - x[lag:n])
            grad[lag:n] -= w * x[0:n-lag]
        else:
            m = -lag
            grad[m:n] += w * (1.0 - x[0:n-m])
            grad[0:n-m] -= w * x[m:n]
    grad *= dx
    sv = np.sum(x) - n / 2.0
    grad += 2.0 * _G_PEN * sv
    return grad

def _slsqp_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    T = _G_T
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_con(x):
    return np.sum(x) - _G_N / 2.0

def _slsqp_con_jac(x):
    return np.ones(_G_N)

def slsqp_opt(h_start, n, T, maxiter):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.x is not None and not np.any(np.isnan(res.x)):
            h_res = make_feasible(res.x, n)
            return h_res, compute_c5(h_res)
    except Exception:
        pass
    return h_start, compute_c5(h_start)

def lbfgsb_opt(h_start, n, T, maxiter):
    global _G_N, _G_T, _G_PEN
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    best_h = h_start.copy()
    best_c = compute_c5(h_start)
    for pen in [1.0, 10.0, 100.0, 500.0, 2000.0]:
        _G_PEN = pen
        mi = max(50, maxiter // 5)
        try:
            res_x, res_f, info = fmin_l_bfgs_b(_penalty_obj, h_start,
                                                fprime=_penalty_grad,
                                                pgtol=1e-12, gtol=1e-12,
                                                maxiter=mi, maxfun=mi*3,
                                                approx_grad=False)
            if res_x is not None and not np.any(np.isnan(res_x)):
                h_res = make_feasible(res_x, n)
                c = compute_c5(h_res)
                if c < best_c:
                    best_c = c
                    best_h = h_res
                h_start = h_res
        except Exception:
            pass
    return best_h, best_c

def coord_descent(h, n, c5, cands_fn, max_rounds, start_time, time_limit):
    for r in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            old = h[j]
            cands = cands_fn(old, j, h)
            best_c = c5
            best_val = old
            for cand in cands:
                if cand is None or abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def cands_coarse(old, j, h):
    return [0.0, 1.0, 0.5,
            max(0.0, old - 0.1), max(0.0, old - 0.05),
            min(1.0, old + 0.05), min(1.0, old + 0.1)]

def cands_fine(old, j, h):
    return [0.0, 1.0,
            max(0.0, old - 0.02), max(0.0, old - 0.01), max(0.0, old - 0.005),
            min(1.0, old + 0.005), min(1.0, old + 0.01), min(1.0, old + 0.02)]

def cands_ultra(old, j, h):
    return [0.0, 1.0,
            max(0.0, old - 0.005), max(0.0, old - 0.002), max(0.0, old - 0.001),
            min(1.0, old + 0.001), min(1.0, old + 0.002), min(1.0, old + 0.005)]

def binary_snap(h, n, c5, start_time, time_limit):
    for j in range(n):
        if time.time() - start_time > time_limit:
            break
        for val in [0.0, 1.0]:
            if abs(val - h[j]) < 1e-8:
                continue
            trial = h.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < c5 - 1e-14:
                h = trial
                c5 = c
    return h, c5

def structured_start(n, freq, amp, phase):
    x = np.linspace(0, 2.0, n)
    h = 0.5 + amp * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, n)

def block_start(n, block_size, phase_shift=0):
    h = np.zeros(n)
    for i in range(n):
        block_idx = (i + phase_shift) // block_size
        if block_idx % 2 == 0:
            h[i] = 1.0
    return make_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: L-BFGS-B with penalty at n=960
    if remaining() > 100:
        h_try = best_h + rng.randn(n) * 0.015
        h_try = make_feasible(h_try, n)
        h_res, c5_res = lbfgsb_opt(h_try, n, 150, 200)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    if remaining() > 100:
        h_res, c5_res = lbfgsb_opt(best_h, n, 300, 200)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Phase 2: SLSQP at n=960
    if remaining() > 90:
        h_res, c5_res = slsqp_opt(best_h, n, 200, 300)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    if remaining() > 80:
        h_try = best_h + rng.randn(n) * 0.03
        h_try = make_feasible(h_try, n)
        h_res, c5_res = slsqp_opt(h_try, n, 100, 250)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Phase 3: Try structured starts
    if remaining() > 120:
        for freq in [3, 5, 7, 11]:
            if remaining() < 60:
                break
            h_try = structured_start(n, freq, 0.45, rng.uniform(0, 2*np.pi))
            h_try = h_try * 0.3 + best_h * 0.7
            h_try = make_feasible(h_try, n)
            h_res, c5_res = lbfgsb_opt(h_try, n, 150, 150)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res

    # Phase 4: Higher resolution
    for n_new in [1280, 1920]:
        if remaining() < 130:
            break
        h_up = upsample_h(best_h, n_new)
        n = len(h_up)
        if remaining() > 80:
            h_res, c5_res = lbfgsb_opt(h_up, n, 100, 150)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
        if remaining() > 70:
            h_res, c5_res = slsqp_opt(best_h, n, 200, 200)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res

    # Phase 5: Coarse coordinate descent
    n = len(best_h)
    if remaining() > 80:
        best_h, best_c5 = coord_descent(best_h, n, best_c5, cands_coarse,
                                        4, start_time, budget_s - 80)

    # Phase 6: Binary snap
    if remaining() > 60:
        best_h, best_c5 = binary_snap(best_h, n, best_c5, start_time, budget_s - 50)

    # Phase 7: Fine coordinate descent
    n = len(best_h)
    if remaining() > 50:
        best_h, best_c5 = coord_descent(best_h, n, best_c5, cands_fine,
                                        15, start_time, budget_s - 40)

    # Phase 8: Another SLSQP polish
    if remaining() > 70:
        h_res, c5_res = slsqp_opt(best_h, n, 400, 250)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Phase 9: Ultra-fine coordinate descent
    n = len(best_h)
    if remaining() > 50:
        best_h, best_c5 = coord_descent(best_h, n, best_c5, cands_ultra,
                                        10, start_time, budget_s - 30)

    # Phase 10: Final binary snap + micro search
    n = len(best_h)
    if remaining() > 40:
        best_h, best_c5 = binary_snap(best_h, n, best_c5, start_time, budget_s - 30)
        if remaining() > 30:
            best_h, best_c5 = coord_descent(best_h, n, best_c5, cands_fine,
                                            8, start_time, budget_s - 20)

    # Phase 11: Very last L-BFGS-B polish
    if remaining() > 50:
        h_res, c5_res = lbfgsb_opt(best_h, n, 500, 100)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res
        if remaining() > 20:
            n = len(best_h)
            best_h, best_c5 = coord_descent(best_h, n, best_c5, cands_ultra,
                                            5, start_time, budget_s - 15)

    # Final cleanup
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    if time.time() - start_time > budget_s - 2:
        best_h = make_feasible(best_h, len(best_h))
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
