import numpy as np
import time

def compute_c5_fft(h):
    n = len(h)
    N = 1 << (2*n - 1).bit_length()  # next power of 2 >= 2n-1
    A = np.fft.rfft(h, N)
    B = np.fft.rfft((1.0 - h)[::-1], N)
    corr = np.fft.irfft(A * B, N)[:2*n-1]
    dx = 2.0 / n
    return corr * dx, corr  # scaled and raw

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):

        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:

        else:

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def corr_fft(h, N_fft):
    n = len(h)
    A = np.fft.rfft(h, N_fft)
    B = np.fft.rfft((1.0 - h)[::-1], N_fft)
    return np.fft.irfft(A * B, N_fft)[:2*n-1]

    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best = compute_c5(h)
    best_h = h.copy()
    start = time.time() if t_start is None else t_start
    if lr is None:
        lr = 0.5 / tau
    N_fft = 1 << (2*n - 1).bit_length()
    h_prev = h.copy()
    t = 1.0
    stall = 0
    it = 0
    n1 = n - 1

    while time.time() - start < time_limit:
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.9)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)

        corr = corr_fft(y, N_fft) * dx

        if top_k < len(corr):
            top_idx = np.argpartition(corr, -top_k)[-top_k:]
        else:
            top_idx = np.arange(len(corr))
        top_vals = corr[top_idx]
        shifted = np.clip(tau * (top_vals - top_vals.max()), -50.0, 0.0)
        exp_c = np.exp(shifted)
        s_exp = exp_c.sum()
        p = exp_c / s_exp

        grad = np.zeros(n)
        for ii in range(len(top_idx)):
            pk = p[ii]
            k = top_idx[ii]
            s = k - n1
            if s >= 0:
                ns = n - s
                grad[:ns] += pk * (1.0 - y[s:])
                grad[s:] -= pk * y[:ns]
            else:
                s2 = -s
                ns = n - s2
                grad[s2:] += pk * (1.0 - y[:ns])
                grad[:ns] -= pk * y[s2:]
        grad *= dx

        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)

        it += 1
        if it % 40 == 0:
            c5_now = compute_c5(h_new)
            if c5_now < best - 1e-15:
                best = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 40
            if stall > 4000:
                lr *= 0.55
                stall = 0
                if lr < 1e-16:
                    break

        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new

    c5_final = compute_c5(h)
    if c5_final < best:
        best = c5_final
        best_h = h.copy()
    return best_h, best


def binary_local_search(h, n, time_limit):
    dx = 2.0 / n
    h_s = h.copy()
    h_s[h_s < 0.25] = 0.0
    h_s[h_s > 0.75] = 1.0
    h_s = project_to_feasible(h_s, n)
    best = compute_c5(h_s)
    best_h = h_s.copy()
    start = time.time()
    N_fft = 1 << (2*n - 1).bit_length()

    ones = np.where(h_s >= 0.5)[0]
    zeros = np.where(h_s < 0.5)[0]

    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.randint(len(ones))
        j = np.random.randint(len(zeros))
        oi = ones[i]
        oz = zeros[j]
        if oi == oz:
            continue
        h_s[oi] = 0.0
        h_s[oz] = 1.0
        c5_new = compute_c5(h_s)
        if c5_new < best - 1e-14:
            best = c5_new
            best_h = h_s.copy()
            ones = np.where(h_s >= 0.5)[0]
            zeros = np.where(h_s < 0.5)[0]
        else:
            h_s[oi] = 1.0
            h_s[oz] = 0.0

    return best_h, best


def ternary_local_search(h, n, time_limit):
    dx = 2.0 / n
    h_s = h.copy()
    h_s[h_s < 0.2] = 0.0
    h_s[(h_s >= 0.2) & (h_s < 0.45)] = 0.5
    h_s[(h_s >= 0.55) & (h_s <= 0.8)] = 0.5
    h_s[h_s > 0.8] = 1.0
    h_s = project_to_feasible(h_s, n)
    best = compute_c5(h_s)
    best_h = h_s.copy()
    start = time.time()

    lo_mask = h_s < 0.4
    hi_mask = h_s > 0.6
    lo_idx = np.where(lo_mask)[0]
    hi_idx = np.where(hi_mask)[0]

    for it in range(2000000):
        if time.time() - start > time_limit:
            break
        if len(lo_idx) == 0 or len(hi_idx) == 0:
            break
        i = lo_idx[np.random.randint(len(lo_idx))]
        j = hi_idx[np.random.randint(len(hi_idx))]
        if i == j:
            continue
        old_i = h_s[i]
        old_j = h_s[j]
        h_s[i] = old_i + 0.15
        h_s[j] = old_j - 0.15
        h_s = np.clip(h_s, 0.0, 1.0)
        c5_new = compute_c5(h_s)
        if c5_new < best - 1e-14:
            best = c5_new
            best_h = h_s.copy()
            lo_idx = np.where(h_s < 0.4)[0]
            hi_idx = np.where(h_s > 0.6)[0]
        else:
            h_s[i] = old_i
            h_s[j] = old_j

    return best_h, best


def get_remaining(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: n=4000, tau=120 (broad)
    n1 = 4000
    h1 = interp_periodic(h_init, n1)
    h1 = project_to_feasible(h1, n1)
    rem = get_remaining(start_time, budget_s)
    if rem > 250:
        tl = min(280, rem * 0.25)
        h1, c5 = nesterov_fft(h1, n1, tau=120.0, time_limit=tl, lr=0.22, top_k=15, t_start=start_time)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 2: n=4000, tau=40
    rem = get_remaining(start_time, budget_s)
    if rem > 220:
        tl = min(250, rem * 0.25)
        h1, c5 = nesterov_fft(h1, n1, tau=40.0, time_limit=tl, lr=0.08, top_k=12, t_start=start_time)
        if c5 < best_c5:
            best_c5, best_h = c5, h1.copy()

    # Phase 3: upsample to 8000, tau=25
    n2 = 8000
    h2 = interp_periodic(best_h, n2)
    h2 = project_to_feasible(h2, n2)
    rem = get_remaining(start_time, budget_s)
    if rem > 230:
        tl = min(280, rem * 0.28)
        h2, c5 = nesterov_fft(h2, n2, tau=25.0, time_limit=tl, lr=0.05, top_k=12, t_start=start_time)
        if c5 < best_c5:
            best_c5, best_h = c5, h2.copy()

    # Phase 4: n=8000, tau=10
    rem = get_remaining(start_time, budget_s)
    if rem > 180:
        nl = len(best_h)
        tl = min(220, rem * 0.28)
        hb, c5 = nesterov_fft(best_h, nl, tau=10.0, time_limit=tl, lr=0.018, top_k=10, t_start=start_time)
        if c5 < best_c5:
            best_c5, best_h = c5, hb.copy()

    # Phase 5: n=8000, tau=5
    rem = get_remaining(start_time, budget_s)
    if rem > 130:
        nl = len(best_h)
        tl = min(160, rem * 0.28)
        hb, c5 = nesterov_fft(best_h, nl, tau=5.0, time_limit=tl, lr=0.009, top_k=8, t_start=start_time)
        if c5 < best_c5:
            best_c5, best_h = c5, hb.copy()

    # Phase 6: binary local search
    rem = get_remaining(start_time, budget_s)
    if rem > 80:
        tl = min(100, rem * 0.3)
        hb, c5 = binary_local_search(best_h, len(best_h), time_limit=tl)
        if c5 < best_c5:
            best_c5, best_h = c5, hb.copy()

    # Phase 7: post-binary Nesterov tau=4
    rem = get_remaining(start_time, budget_s)
    if rem > 80:
        nl = len(best_h)
        tl = min(100, rem * 0.3)
        hb, c5 = nesterov_fft(best_h, nl, tau=4.0, time_limit=tl, lr=0.007, top_k=6, t_start=start_time)
        if c5 < best_c5:
            best_c5, best_h = c5, hb.copy()

    # Phase 8: perturbation restart
    rem = get_remaining(start_time, budget_s)
    for pert in [0.004, 0.008, 0.002]:
        rem = get_remaining(start_time, budget_s)
        if rem < 80:
            break
        nl = len(best_h)
        h_p = best_h.copy() + np.random.randn(nl) * pert
        h_p = project_to_feasible(h_p, nl)
        tl = min(70, rem * 0.25)
        h_p, c5_p = nesterov_fft(h_p, nl, tau=8.0, time_limit=tl, lr=0.012, top_k=8, t_start=start_time)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 9: final fine Nesterov
    rem = get_remaining(start_time, budget_s)
    if rem > 50:
        nl = len(best_h)
        tl = min(60, rem * 0.5)
        hb, c5 = nesterov_fft(best_h, nl, tau=2.5, time_limit=tl, lr=0.004, top_k=5, t_start=start_time)
        if c5 < best_c5:
            best_c5, best_h = c5, hb.copy()

    # Phase 10: ternary local search
    rem = get_remaining(start_time, budget_s)
    if rem > 40:
        tl = min(50, rem * 0.6)
        hb, c5 = ternary_local_search(best_h, len(best_h), time_limit=tl)
        if c5 < best_c5:
            best_c5, best_h = c5, hb.copy()

    # Final projection and evaluation
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
