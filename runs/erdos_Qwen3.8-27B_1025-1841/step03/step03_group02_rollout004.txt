    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    # Adjust to exactly n/2 ones
    diff = target - int(np.sum(b))
    if diff > 0:
        # Need more ones
        order = np.argsort(-h)  # largest h first (already close to 1)
        for idx in order[:diff]:
            b[idx] = 1
    elif diff < 0:
        # Need fewer ones
        order = np.argsort(h)  # smallest h first (already close to 0)
        for idx in order[-diff:]:
            b[idx] = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        order = np.argsort(-h)
        for idx in order[:diff]:
            b[idx] = 1.0
    elif diff < 0:
        order = np.argsort(h)
        for idx in order[-diff:]:
            b[idx] = 0.0
    return b

def upsample_h(h, new_n):
    old_n = len(h)

    new_x = np.linspace(0.0, 2.0, new_n)

    return make_feasible(h_new, new_n)

def _slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_sum(x):
    return np.sum(x) - len(x) / 2.0

def _con_sum_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, T, maxiter, start_time, time_limit):
    if time.time() - start_time > time_limit - 20:
        return h
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_sum_jac}]
    xx = h.copy()

    def obj(x):
        return _slsqp_obj(x, n, T)

    try:
        res = minimize(obj, xx, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.fun < compute_c5(h):
            out = make_feasible(res.x, n)
            return out
    except Exception:
        pass
    return h

                else:
