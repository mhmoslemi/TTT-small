def coord_sweep(h, n_sweeps=100, n_t=100):

    dx = 2.0 / n
    t_grid = np.linspace(0.0, 1.0, n_t)
    h = h.astype(float).copy()
    
    for sweep in range(n_sweeps):
        improved = 0
        for i in range(n):
            corr = np.correlate(h, 1.0 - h, mode='full')
            s = np.arange(-(n-1), n)
            
            # Compute d[m] for each lag
            d = np.zeros(2*n-1)
            ip = i + s  # i + s, where s is the shift
            im = i - s  # i - s
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            # For lag s (correlate index n-1+s):
            # Terms: h[i]*(1-h[i+s]) if 0<=i+s<n, and h[i-s]*(1-h[i]) if 0<=i-s<n
            # d[s] = (1-h[i+s])_{valid} - h[i-s]_{valid}
            d[vp] += (1.0 - h[np.clip(ip[vp], 0, n-1)])
            d[vm] -= h[np.clip(im[vm], 0, n-1)]
            
            # For s=0 (index n-1): the formula is different
            # h[i]*(1-h[i]) is quadratic in h[i], not linear
            # So d[n-1] should be set differently
            
            # Linear part for all lags (approximate, will fix s=0)
            # new_corr[t, m] = corr[m] + (t - h[i]) * d[m]
            linear = corr[None, :] + (t_grid[:, None] - h[i]) * d[None, :]  # (n_t, 2n-1)
            
            # Fix s=0 column (index n-1):
            # Correct: new_value = corr[n-1] - h[i]*(1-h[i]) + t*(1-t)
            # = corr[n-1] - h[i] + h[i]^2 + t - t^2
            linear[:, n-1] = corr[n-1] - h[i] + h[i]*h[i] + t_grid - t_grid*t_grid
            
            obj = np.max(linear, axis=1) * dx
            best_idx = np.argmin(obj)
            new_val = t_grid[best_idx]
            
            if abs(new_val - h[i]) > 1e-10:
                h[i] = new_val
                improved += 1
        
        if improved == 0:
            break
    
    # Final projection (coordinate descent might slightly violate the sum constraint)
    h = project_c(h, n)
    return h

Wait, but the coordinate descent as I've set it up doesn't maintain the constraint ∑h = n/2! When I change h[i], the sum changes. I need to either:
1. Project after each coordinate update (but this changes other coordinates)
2. Use a Lagrange multiplier for the constraint
3. Only update h[i] to a value that maintains the sum (not possible in general)

Option 2 is best: minimize f(t) + λ*(t + ∑_{j≠i} h[j] - n/2) = f(t) + λ*t + const
Since the constraint is equality, at the optimum the Lagrangian derivative should be zero. But this complicates the piecewise-linear maximization.

Alternative: After each sweep, project to the constraint. The projection is:
- Find θ such that ∑ clip(h[j] - θ, 0, 1) = n/2
- Set h[j] = clip(h[j] - θ, 0, 1)

This is what project_c does. So I'll do coordinate descent without the constraint, then project at the end of each sweep. The projection is O(n log n) or O(n * 80) with bisection.

Actually, a better approach: in the coordinate descent, when I update h[i] from old to new, the sum changes by (new - old). To maintain the sum, I could distribute the change to a few other coordinates. But that's more complex.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def sm_grad(h, T, top_k=100):
    """Gradient of smoothmax: (1/T)*log(sum exp(T*corr)) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = np.argmax(corr)
        return _subgrad_one(h, idx - (n-1)) * dx
    w = exps / tot
    # Focus on top_k
    top_idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in top_idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * _subgrad_one(h, i - (n-1))
    return g * dx

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

Instead of smoothmax (log-sum-exp), use a **power mean** smoothmax:
f_p(h) = dx * (∑ |corr[k]|^p)^(1/p)

For large p, this approaches dx * max_k |corr[k]| (well, max_k corr[k] since all should be positive).

The gradient:
∇f_p = dx * (1/p) * (∑|c|^p)^(1/p - 1) * ∑_k p * |c_k|^(p-2) * c_k * ∇c_k

= dx * (∑|c|^p)^(1/p-1) * ∑_k |c_k|^(p-2) * c_k * ∇c_k

For the Nesterov method:
x_{k+1} = proj(x_k - α_k * ∇f_p(z_k))
z_{k+1} = x_{k+1} + (t_k - 1)/(t_{k+1}) * (x_{k+1} - x_k)

Where α_k can be 1/(L*k) or something adaptive.

The issue is computing ∇f_p efficiently. For each coordinate i, I need:
∂f_p/∂h[i] = dx * (∑|c|^p)^(1/p-1) * ∑_k |c_k|^(p-2) * c_k * ∂c_k/∂h[i]

The inner sum is over all 2n-1 lags. For each lag k, ∂c_k/∂h[i] is non-zero for at most... well, it depends on the structure. Actually, for a given i, ∂c_k/∂h[i] is non-zero for all k where i appears in the correlation sum, which is all k (since h[i] participates in the correlation at every lag).

Wait no. corr[k] = ∑_j h[j]*(1-h[j+k-(n-1)]) (with the shift interpretation). For a fixed i, h[i] appears in corr[k] when j=i or when j+k-(n-1)=i. So for each k, h[i] appears in at most 2 terms. Thus ∂corr[k]/∂h[i] is non-zero for all k but is a simple expression.

So ∂f_p/∂h[i] = A * ∑_k w_k * ∂corr[k]/∂h[i]
where A = dx * (∑|c|^p)^(1/p-1) and w_k = |c_k|^(p-2) * c_k

This is O(n * 2n) = O(n²) per gradient computation. For n=960, that's ~1.8M per gradient. With FISTA needing maybe 5000 iterations, that's 9B operations. Too slow in Python.

The smoothmax (log-sum-exp) approach with top_k restriction is much faster because it only sums over the top_k lags. That's why the previous code used it.

Let me be careful about the coordinate descent implementation. The key operations per coordinate i:
1. Compute d[m] for all m: O(n) 
2. Build new_corr matrix (n_t × (2n-1)): O(n_t * n)
3. Take max along axis 1: O(n_t * n)
4. Find argmin: O(n_t)
5. Update h[i] and corr: O(n)

Total per coordinate: O(n_t * n + n) ≈ O(n_t * n)
Total per sweep: O(n * n_t * n) = O(n² * n_t)
For n=960, n_t=100: 960² * 100 = 92M per sweep
For 200 sweeps: 18.4B... that's too much.

Wait, that can't be right. Let me recount.

Per coordinate i:
- d computation: vectorized numpy, size 2n-1 ≈ 1919 elements. Fast.
- new_corr: t_grid[:,None] (100×1) * d[None,:] (1×1919) → 100×1919 matrix. This is ~192K multiply-adds.
- np.max along axis 1: 100×1919 → 100 values. ~192K comparisons.
- Total per coordinate: ~400K ops

Per sweep (960 coordinates): ~384M ops
Per sweep in Python loop: each iteration does numpy ops on ~2000-element arrays. The Python loop overhead is 960 iterations per sweep. Each iteration: maybe 5-10 numpy calls on small arrays. At ~1μs per numpy call + array creation, maybe 50μs per iteration → 48ms per sweep.

200 sweeps: ~10 seconds. That's fine!

Actually, let me benchmark mentally: for each coordinate, I do:
- ip = i + s_arr (1919 elements)
- im = i - s_arr (1919 elements)  
- vp, vm (boolean masks)
- d = zeros + indexing (1919 elements)
- (t_grid[:,None] - hi) * d[None,:] (100×1919 = 192K)
- + corr[None,:] (192K)
- override column n-1 (100 elements)
- np.max axis=1 (192K comparisons → 100 values)
- argmin (100 comparisons)
- corr update: 1919 adds

The big operation is the 100×1919 matrix multiply-add. In numpy, this is a single BLAS-level operation, ~192K FLOPs. At ~1 GFLOPS for numpy (conservative for such small arrays), that's ~0.2ms per coordinate. 960 coordinates → ~190ms per sweep. 200 sweeps → ~38 seconds.

Actually, I realize I can make the matrix operation faster by precomputing. The (t_grid[:,None] - hi) is just t_grid[:,None] - hi (a scalar). So:
new_corr = corr[None,:] + t_grid[:,None]*d[None,:] - hi*d[None,:]
= (corr - hi*d)[None,:] + t_grid[:,None]*d[None,:]

The first term is precomputed per coordinate (O(n)), the second is an outer product (O(n_t * n)).

Or even: new_corr[m] for all t is a line in t with slope d[m] and intercept (corr[m] - hi*d[m]). The max over m of these lines is a convex piecewise-linear function in t. I could find its minimum more efficiently than a grid search, but the grid search is simpler and fast enough.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n-s]
        g[:n-s] -= h[s:]
    else:
        g[:n-s] += 1.0 - h[s:]
        g[s:] -= h[:n-s]
    return g

def sm_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n-1)) * dx
    w = exps / tot
    top_idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in top_idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * _subgrad_one(h, i - (n-1))
    return g * dx

def fista(h, n_iter, T0, T1, lr0, lr1, top_k=100, check=50):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr0 + (lr1 - lr0) * f
        g = sm_grad(z, T, top_k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def coord_sweep(h, n_sweeps=200, n_t=100):
    n = len(h)
    h = h.astype(float).copy()
    t_grid = np.linspace(0.0, 1.0, n_t)
    s_arr = np.arange(-(n-1), n, dtype=np.int64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    
    for sweep in range(n_sweeps):
        changed = 0
        for i in range(n):
            hi = h[i]
            ip = i + s_arr
            im = i - s_arr
            vp = (ip >= 0) & (ip < n)
            vm = (im >= 0) & (im < n)
            d = np.zeros(2*n - 1)
            d[vp] += (1.0 - h[ip[vp]])
            d[vm] -= h[im[vm]]
            
            # new_corr[t, m] = corr[m] + (t-hi)*d[m] for m≠n-1
            # new_corr[t, n-1] = corr[n-1] + t*(1-t) - hi*(1-hi)
            
            # Use: new_corr = (corr - hi*d)[None,:] + t_grid[:,None]*d[None,:]
            base = corr - hi * d  # shape (2n-1,)
            new_corr = base[None, :] + t_grid[:, None] * d[None, :]
            # Fix s=0 column
            new_corr[:, n-1] = corr[n-1] - hi*(1.0-hi) + t_grid*(1.0-t_grid)
            
            obj = np.max(new_corr, axis=1)
            best = t_grid[np.argmin(obj)]
            
            if abs(best - hi) > 1e-7:
                delta = best - hi
                h[i] = best
                corr += delta * d
                corr[n-1] -= delta * delta  # quadratic correction
                changed += 1
        
        h = project_c(h, n)
        if sweep % 10 == 0 or changed < max(1, n // 100):
            corr = np.correlate(h, 1.0 - h, mode='full')
        if changed < max(1, n // 100):
            break
    
    return h

def _slsqp_f(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def _slsqp_g(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    w = exps / np.sum(exps)
    g = np.zeros(n)
    n_full = 2 * n - 1
    top_k = min(200, n_full)
    top_idx = np.argsort(w)[::-1][:top_k]
    for i in top_idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * _subgrad_one(h, i - (n-1))
    return g * dx

def _scon(h, n):
    return np.sum(h) - n / 2.0

def slsqp(h, T=5.0, maxiter=400):
    n = len(h)
    try:
        res = minimize(
            _slsqp_f, h, args=(n, T),
            jac=lambda x: _slsqp_g(x, n, T),
            method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': lambda x: _scon(x, n), 'jac': lambda x: np.ones(n)}],
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        if res.x is not None:
            hx = project_c(res.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n].copy()

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)
