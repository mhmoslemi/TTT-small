import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def smoothed_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    # Use top k for efficiency
    sorted_idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in sorted_idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * subgrad_one(h, i - (n - 1))
    return g * dx

def fista_opt(h_init, n_iter=5000, T0=50.0, T1=0.1, lr=0.12, check=50, top_k=100):

    x = h_init.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()

        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f

_sl_state = {}

def _sl_obj(h):
    T = _sl_state['T']
    n = _sl_state['n']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _sl_con(h):
    return np.sum(h) - _sl_state['n'] / 2.0

def _sl_jac(h):
    return np.ones(_sl_state['n'])

def slsqp_polish(h, T=5.0, maxiter=400):
    n = len(h)
    _sl_state['T'] = T
    _sl_state['n'] = n
    try:
        r = minimize(_sl_obj, h.astype(float), method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sl_con, 'jac': _sl_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-15})
        if r.success and r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) <= compute_c5(h):

    return h

def project_c(h, n):
    target = n / 2.0

    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def smoothed_grad(h, T, top_k=100):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if T < 0.01:
        idx = int(np.argmax(corr))
        return subgrad_one(h, idx - (n - 1)) * dx
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    sorted_idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in sorted_idx:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * subgrad_one(h, i - (n - 1))

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def smoothed_grad(h, T, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    if T < 0.01:
        idx = int(np.argmax(corr))
        return subgrad_one(h, idx - (n - 1)) * dx
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return subgrad_one(h, idx - (n - 1)) * dx
    w = exps / tot
    top = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in top:
        wi = w[i]
        if wi < 1e-10:
            break
        g += wi * subgrad_one(h, i - (n - 1))
    return g * dx

def fista_opt(h_init, n_iter, T0, T1, lr, check, top_k):

    x = h_init.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        fr = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** fr
        lrt = lr * (1.0 - 0.5 * fr)
        g = smoothed_grad(z, T, top_k)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -2.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:

            if c < bc:
                bc, bh = c, x.copy()

    if c < bc:
