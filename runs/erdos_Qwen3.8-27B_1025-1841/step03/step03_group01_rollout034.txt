        return budget_s - (time.time() - start_time)
    
    # Get initial h
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.zeros(640)
        h0[::2] = 1.0
    
    n = len(h0)
    dx = 2.0 / n
    target_sum = n / 2.0
    
    # Make feasible
    h0 = np.clip(h0, 0.0, 1.0)
    h0 = make_feasible(h0, n)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # --- Phase 1: Binary local search with greedy + annealing ---
    # Threshold to binary
    h_bin = (h0 >= 0.5).astype(float)
    n_ones = int(h_bin.sum())
    if n_ones != n // 2:
        if n_ones > n // 2:
            ones_idx = np.where(h_bin == 1)[0]
            perm = rng.permutation(ones_idx)
            h_bin[perm[:n_ones - n//2]] = 0
        else:
            zeros_idx = np.where(h_bin == 0)[0]
            perm = rng.permutation(zeros_idx)
            h_bin[perm[:n//2 - n_ones]] = 1
    
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Compute correlation and influence vectors
    C = compute_corr(h_bin).astype(float)  # shape (2n-1,)
    full_len = 2 * n - 1
    
    # V[i, k] = change in C[k] when h[i] goes 0->1
    # C[k] = sum_m h[m + k - (n-1)] * (1-h[m])
    # V[i, k] = (1-h[i-l]) * [0<=i-l<n] - h[i+l] * [0<=i+l<n] where l=k-(n-1)
    V = np.zeros((n, full_len))
    lags = np.arange(full_len) - (n - 1)  # shape (full_len,)
    
    for i in range(n):
        # Term 1: (1-h[i-l]) for 0<=i-l<n
        idx1 = i - lags  # shape (full_len,)
        mask1 = (idx1 >= 0) & (idx1 < n)
        V[i, mask1] += (1.0 - h_bin[idx1[mask1]])
        
        # Term 2: -h[i+l] for 0<=i+l<n
        idx2 = i + lags
        mask2 = (idx2 >= 0) & (idx2 < n)
        V[i, mask2] -= h_bin[idx2[mask2]]
    
    # Greedy pair flip search
    greedy_rounds = 200
    for round_num in range(greedy_rounds):
        if time_left() < 100:
            break
        
        zeros_idx = np.where(h_bin == 0)[0]
        ones_idx = np.where(h_bin == 1)[0]
        
        if len(zeros_idx) == 0 or len(ones_idx) == 0:
            break
        
        best_delta_c5 = 0.0
        best_pair = None
        current_max_C = C.max()
        
        for i in zeros_idx:
            # For all j in ones_idx: new_C = C + V[i] - V[j]
            # new_max = max(new_C)
            new_C = C[None, :] + V[i][None, :] - V[ones_idx, :]
            new_max = new_C.max(axis=1)
            best_j_local = ones_idx[np.argmin(new_max)]
            best_val_local = new_max.min()
            
            if best_val_local < current_max_C - 1e-10:
                if best_val_local < best_delta_c5:
                    best_delta_c5 = best_val_local
                    best_pair = (i, best_j_local)
        
        if best_pair is None:
            break  # No improvement possible
        
        # Apply the flip
        i_flip, j_flip = best_pair
        h_bin[i_flip] = 1.0
        h_bin[j_flip] = 0.0
        
        # Update C
        C += V[i_flip] - V[j_flip]
        
        # Update V (only 4 positions per row change)
        # V[m, k] changes when m-l=k related to i_flip or j_flip
        # For h[i_flip] 0->1: 
        #   V[m, k] where m-l = i_flip: k = m - i_flip + (n-1), term (1-h) goes 1->0, V decreases by 1
        #   V[m, k] where m+l = i_flip: k = m + i_flip - (n-1), term -h goes 0->-1, V decreases by 1
        # For h[j_flip] 1->0:
        #   V[m, k] where m-l = j_flip: k = m - j_flip + (n-1), term (1-h) goes 0->1, V increases by 1
        #   V[m, k] where m+l = j_flip: k = m + j_flip - (n-1), term -h goes -1->0, V increases by 1
        
        m_all = np.arange(n)
        
        # i_flip 0->1:
        k1 = m_all - i_flip + (n - 1)  # where m-l = i_flip
        valid1 = (k1 >= 0) & (k1 < full_len)
        V[valid1, k1[valid1]] -= 1.0
        
        k2 = m_all + i_flip - (n - 1)  # where m+l = i_flip
        valid2 = (k2 >= 0) & (k2 < full_len)
        V[valid2, k2[valid2]] -= 1.0
        
        # j_flip 1->0:
        k3 = m_all - j_flip + (n - 1)
        valid3 = (k3 >= 0) & (k3 < full_len)
        V[valid3, k3[valid3]] += 1.0
        
        k4 = m_all + j_flip - (n - 1)
        valid4 = (k4 >= 0) & (k4 < full_len)
        V[valid4, k4[valid4]] += 1.0
        
        c5_new = C.max() * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_bin.copy()
    
    # --- Phase 2: Simulated annealing ---
    if time_left() > 100:
        T_start = 0.05
        T_end = 0.001
        n_sa_steps = 200000
        
        h_sa = h_bin.copy()
        C_sa = compute_corr(h_sa).astype(float)
        V_sa = V.copy()  # Use current V (should match)
        
        # Actually let me recompute V for h_sa to be safe
        # (it should be the same as V since h_sa = h_bin)
        
        zeros_set = set(np.where(h_sa == 0)[0].tolist())
        ones_set = set(np.where(h_sa == 1)[0].tolist())
        
        # For SA, we need to do random pair flips
        # To be efficient, precompute arrays
        zeros_arr = np.array(list(zeros_set))
        ones_arr = np.array(list(ones_set))
        
        for step in range(n_sa_steps):
            if time_left() < 30:
                break
            if step % 10000 == 0 and step > 0:
                # Check progress
                pass
            
            T = T_start * (T_end / T_start) ** (step / n_sa_steps)
            
            if len(zeros_arr) == 0 or len(ones_arr) == 0:
                break
            
            i = zeros_arr[rng.randint(len(zeros_arr))]
            j = ones_arr[rng.randint(len(ones_arr))]
            
            # Compute new C
            new_C = C_sa + V_sa[i] - V_sa[j]
            new_max = new_C.max()
            old_max = C_sa.max()
            
            delta = new_max - old_max
            if delta < 0 or rng.random() < np.exp(-delta / T):
                # Accept
                h_sa[i] = 1.0
                h_sa[j] = 0.0
                C_sa += V_sa[i] - V_sa[j]
                
                # Update V_sa
                m_all = np.arange(n)
                k1 = m_all - i + (n - 1)
                valid1 = (k1 >= 0) & (k1 < full_len)
                V_sa[valid1, k1[valid1]] -= 1.0
                k2 = m_all + i - (n - 1)
                valid2 = (k2 >= 0) & (k2 < full_len)
                V_sa[valid2, k2[valid2]] -= 1.0
                k3 = m_all - j + (n - 1)
                valid3 = (k3 >= 0) & (k3 < full_len)
