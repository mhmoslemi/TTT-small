            cur[j] = 0

            ones.discard(j)
            zeros.discard(i)
            zeros.add(j)

def make_feasible(h, n):

        diff = target - np.sum(h)

            break

            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break

    diff = target - np.sum(h)

            h[int(np.argmin(h))] += diff
        else:

def to_binary(h, n):

        zeros = np.where(b == 0)[0]

    return np.ones(_state_n)
