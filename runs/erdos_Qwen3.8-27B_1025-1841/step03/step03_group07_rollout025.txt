import numpy as np
import time
from scipy.signal import fftconvolve

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def correlate_h(h):
    return fftconvolve(h, (1.0 - h)[::-1], mode='full')

def compute_c5(h):
    n = len(h)
    return correlate_h(h).max() * (2.0 / n)

def compute_c5_exact(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def grad_single_shift(h, lag):
    """Gradient of f_lag(h) = sum_i h[i]*(1-h[i+lag]) wrt h."""
    n = len(h)
    g = np.zeros(n)
    j0 = max(0, lag)
    j1 = min(n, n + lag)
    if j0 < j1:
        idx = np.arange(j0, j1)
        g[j0:j1] += (1.0 - h[idx - lag])
    j0b = max(0, -lag)
    j1b = min(n, n - lag)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        g[j0b:j1b] -= h[idx + lag]
    return g

    shifted = np.clip(tau * (top_vals - max_c), -300.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def correlate_h(h):
    return fftconvolve(h, (1.0 - h)[::-1], mode='full')

    n = len(h)

    grad = np.zeros(n)
    for pk, k in zip(weights, top_idx):

                grad[:n - lag] += pk * (1.0 - h[lag:])

        else:

        hs, c5s = subgrad_run(best_h, len(best_h), time_limit=tl, lr=0.0005, top_k=3)
        if c5s < best_c5:
            best_c5, best_h = c5s, hs.copy()

    rem = get_remaining(start_time, budget_s)
    if rem > 15:
        tl = min(12, rem * 0.4)
        hs, c5s = subgrad_run(best_h, len(best_h), time_limit=tl, lr=0.00015, top_k=2)
        if c5s < best_c5:
            best_c5, best_h = c5s, hs.copy()

    # Final evaluation with exact correlate
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5_exact(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
