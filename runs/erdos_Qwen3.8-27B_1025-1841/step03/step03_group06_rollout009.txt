```python
def binary_flip_delta(m, h, n, j):
    """Compute delta of corr[m] when h[j] flips (binary)"""
    # h[j] is currently 1, flipping to 0 (or vice versa)
    # Let's handle both cases
    val = h[j]
    # Effect on h part: -val * (1-h)[j+m-n+1] if valid
    # Effect on (1-h) part: +val * h[j-m+n-1] if valid  (since (1-h)[j] changes by +val)
    # Wait, if h[j] goes from val to 1-val:
    #   h part: (1-val)*h'[j] * (1-h)[j+m-n+1] - val*h[j] * (1-h)[j+m-n+1]
    #   Actually simpler: Δh[j] = (1-2*val) [goes from val to 1-val, change is 1-2val]
    #   Δ(1-h)[j] = -(1-2*val) [goes from 1-val to val, change is -(1-2val)]
    
    delta = 0.0
    sig = 1.0 - 2.0 * val  # +1 if flipping 0→1, -1 if flipping 1→0
    
    # Contribution from h[j] changing by sig:
    idx1 = j + m - (n - 1)
    if 0 <= idx1 < n:
        delta += sig * (1.0 - h[idx1])
    
    # Contribution from (1-h)[j] changing by -sig:
    idx2 = j - m + (n - 1)
    if 0 <= idx2 < n:
        delta += -sig * h[idx2]
    
    return delta
```

```python
    exp_s = np.exp(T * shifted)
    w = exp_s / np.sum(exp_s)
```
