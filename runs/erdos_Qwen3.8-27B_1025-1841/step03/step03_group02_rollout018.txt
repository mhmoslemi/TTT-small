                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _rng_holder
    start_time = time.time()
    deadline = start_time + budget_s - 5
    rng = np.random.RandomState(seed)
    _rng_holder[0] = np.random.RandomState(seed + 1)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SA with targeted max-lag moves (35% of budget)
    d1 = start_time + budget_s * 0.30
    b1, c5_1 = sa_loop(h_init, n0, 99999999, 0.008, 0.00001, 0.55, rng, d1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()

    # Phase 2: SA targeting multiple top lags (20% of budget)
    d2 = start_time + budget_s * 0.50
    b2, c5_2 = sa_multi_target(best_h, n0, 99999999, 0.006, 0.00001, rng, d2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = b2.copy()

    # Phase 3: Greedy repair
    d3 = start_time + budget_s * 0.55
    if time.time() < d3:
        b3, c5_3 = greedy_repair(best_h, n0, 200, d3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()

    # Phase 4: SLSQP polish (10% of budget)
    d4 = start_time + budget_s * 0.65
    if time.time() < d4:
        x_res = slsqp_polish(best_h, n0, 100.0, 300, d4)
        x_res = make_feasible(x_res, n0)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = x_res.copy()

    # Phase 5: SA again from polished result (15% of budget)
    d5 = start_time + budget_s * 0.80
    if time.time() < d5:
        b5, c5_5 = sa_loop(best_h, n0, 99999999, 0.004, 0.000005, 0.6, rng, d5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = b5.copy()

    # Phase 6: Coordinate descent
    d6 = start_time + budget_s * 0.90
    if time.time() < d6:
        best_h, best_c5 = coord_descent(best_h.copy(), n0, 25, d6)

    # Phase 7: Final SLSQP + SA
    d7 = deadline - 15
    if time.time() < d7:
        x_res = slsqp_polish(best_h, n0, 400.0, 200, d7)
        x_res = make_feasible(x_res, n0)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = x_res.copy()

    # Phase 8: Final SA burst
    if time.time() < deadline - 3:
        b8, c5_8 = sa_loop(best_h, n0, 99999999, 0.002, 0.000001, 0.7, rng, deadline - 3)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = b8.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
