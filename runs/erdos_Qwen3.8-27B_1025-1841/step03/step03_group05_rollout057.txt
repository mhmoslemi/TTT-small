import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[n - s:] -= h[:s]
    return g

def sg_active(h, m=10, eps_frac=0.005):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    thresh = mc * (1.0 - eps_frac)
    active = np.where(corr >= thresh)[0]
    if len(active) > m:
        active = np.argsort(corr)[::-1][:m]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    g = np.zeros(n)
    for idx in active:
        g += _subgrad_one(h, idx - (n - 1))
    return g / len(active) * dx

def sg_lp(h, p=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Clip to avoid 0^(p-1) issues
    corr = np.maximum(corr, 1e-15)
    cp_sum = np.sum(corr ** p)
    w = corr ** (p - 1) / (cp_sum ** ((p - 1) / p))
    g = np.zeros(n)
    for i in range(len(w)):
        if w[i] > 1e-15:
            g += w[i] * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista_run(h, n_iter, grad_fn, lr0=0.1, decay=0.9995, check=15):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        g = grad_fn(z)
        lr = lr0 * (decay ** it)
        xn = project_c(z - lr * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc = c
                bh = x.copy()
    c = compute_c5(x)
    if c < bc:
        bc = c
        bh = x.copy()
    return bh, bc

    nx = np.linspace(0, 2, nn)
    interp = np.interp(nx, ox, h)
    return project_c(interp, nn)

    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]

def _sobj_fn(x, n, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * (2.0 / n)

def _sobj_grad_fn(x, n, T):
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    exps = np.exp(T * (c - mc))
    tot = np.sum(exps)
    w = exps / tot
    g = np.zeros(n)
    for i in range(len(w)):

    return g * (2.0 / n)

def _scon_fn(x, n):
    return np.sum(x) - n / 2.0

def _scon_jac_fn(x, n):
    return np.ones(n)

def slsqp_polish(h, T=5.0, maxiter=300):
    n = len(h)
    def obj(x):

    def con_j(x):

    try:

                     bounds=[(0.0, 1.0)] * n,

                     options={'maxiter': maxiter, 'ftol': 1e-12})
        hx = project_c(r.x, n)
        if compute_c5(hx) < compute_c5(h):
            return hx
    except Exception:
        pass
    return h

def tm_seq(n):
    t = np.array([0.0])
    while len(t) < n:
        t = np.concatenate([t, 1.0 - t])
    return t[:n]

def _so_func(x):
    n = _PARAMS['n']
    T = _PARAMS['T']
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * (2.0 / n)

def _so_grad(x):
    n = _PARAMS['n']
    T = _PARAMS['T']
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    exps = np.exp(T * (c - mc))
    tot = np.sum(exps)
    w = exps / tot
    g = np.zeros(n)
    for i in range(len(w)):
        if w[i] > 1e-12:
            g += w[i] * _subgrad_one(x, i - (n - 1))
    return g * (2.0 / n)

def _sc_func(x):
    return np.sum(x) - _PARAMS['n'] / 2.0

def _sc_jac(x):
    return np.ones(_PARAMS['n'])

def slsqp_polish(h, T=5.0, maxiter=300):
    n = len(h)
    _PARAMS['n'] = n
    _PARAMS['T'] = T
    try:
        r = minimize(_so_func, h, jac=_so_grad, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _sc_func, 'jac': _sc_jac}],
                     options={'maxiter': maxiter, 'ftol': 1e-12})
        hx = project_c(r.x, n)
        if compute_c5(hx) < compute_c5(h):
            return hx
    except Exception:
        pass
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    elapsed = lambda: time.time() - t0
    remaining = lambda: budget_s - elapsed()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = project_c(tm_seq(480), 480)

    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"Init: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1: L_p norm optimization at current resolution
    for p_val in [25, 50]:
        if remaining() < 80:
            break
        _PARAMS['p'] = p_val
        _PARAMS['m'] = 999
        _PARAMS['eps'] = 0.999
        hr, cr = fista_run(bh, 2500, sg_lpnorm, lr0=0.035, decay=0.999, check=20)
        if cr < bc:
            bc, bh = cr, hr.copy()
            print(f"  Lp{p_val}: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 2: Active-set with various m and eps
    for (m_v, ef_v, lr_v, dec_v) in [(8, 0.008, 0.09, 0.999), (15, 0.003, 0.05, 0.9995), (25, 0.001, 0.03, 0.9997)]:
        if remaining() < 70:
            break
        _PARAMS['m'] = m_v
        _PARAMS['eps'] = ef_v
        hr, cr = fista_run(bh, 3000, sg_active, lr0=lr_v, decay=dec_v, check=15)
        if cr < bc:
            bc, bh = cr, hr.copy()
            print(f"  Act(m={m_v},e={ef_v}): C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 3: Coarse-to-fine path (start fresh at low res)
    if remaining() > 400:
        _PARAMS['p'] = 30
        for nn in [192, 384, 768]:
            if remaining() < 120:
                break
            if nn == 192:
                hc = project_c(tm_seq(nn), nn)
                hc, cc = fista_run(hc, 5000, sg_lpnorm, lr0=0.05, decay=0.999, check=20)
                _PARAMS['m'] = 10
                _PARAMS['eps'] = 0.005
                hc, cc = fista_run(hc, 4000, sg_active, lr0=0.07, decay=0.999, check=15)
            else:
                hc = upsample_h(hc, nn)
                cc = compute_c5(hc)
                _PARAMS['p'] = 30
                hc, cc2 = fista_run(hc, 3500, sg_lpnorm, lr0=0.04, decay=0.999, check=15)
                if cc2 < cc:
                    hc, cc = hc, cc2
                _PARAMS['m'] = 15
                _PARAMS['eps'] = 0.003
                hc, cc2 = fista_run(hc, 3000, sg_active, lr0=0.05, decay=0.9995, check=15)
                if cc2 < cc:
                    hc, cc = hc, cc2
            if cc < bc:
                bc, bh = cc, hc.copy()
                print(f"  CF n={nn}: C5={bc:.6f} t={elapsed():.0f}", flush=True)
        # Continue to n0 or higher
        if len(bh) < n0:
            if remaining() > 100:
                bh = upsample_h(bh, n0)
                _PARAMS['m'] = 10
                _PARAMS['eps'] = 0.005
                hr, cr = fista_run(bh, 3000, sg_active, lr0=0.05, decay=0.9995, check=15)
                if cr < bc:
                    bc, bh = cr, hr.copy()
                    print(f"  CF->n0: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 4: Diverse starts at current n
    n_cur = len(bh)
    xg = np.linspace(0, 2, n_cur)
    starts = [tm_seq(n_cur), project_c(rng.rand(n_cur), n_cur)]
    for freq in [2, 3, 5, 7]:
        for amp in [0.15, 0.4]:
            s = 0.5 + amp * np.sin(np.pi * freq * xg)
            starts.append(project_c(s, n_cur))
    for p_b in [6, 10, 14]:
        pat = np.zeros(n_cur)
        pat[::p_b] = 1.0
        pat[1:p_b // 3] = 1.0
        starts.append(project_c(pat, n_cur))

    for hs in starts:
        if remaining() < 40:
            break
        _PARAMS['m'] = 10
        _PARAMS['eps'] = 0.005
        hr, cr = fista_run(hs, 2000, sg_active, lr0=0.07, decay=0.999, check=20)
        if cr < bc:
            bc, bh = cr, hr.copy()
            print(f"  DivStart: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 5: Upsample to 2x and refine
    n_up = len(bh) * 2
    if n_up <= 1920 and remaining() > 150:
        hu = upsample_h(bh, n_up)
        cu = compute_c5(hu)
        _PARAMS['p'] = 40
        hg, cg = fista_run(hu, 3000, sg_lpnorm, lr0=0.03, decay=0.999, check=15)
        if cg < cu:
            hu, cu = hg, cg
        _PARAMS['m'] = 12
        _PARAMS['eps'] = 0.004
        hn, cn = fista_run(hu, 2500, sg_active, lr0=0.04, decay=0.9995, check=10)
        if cn < cu:
            hu, cu = hn, cn
        if cu < bc:
            bc, bh = cu, hu.copy()
            print(f"  Up{n_up}: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 6: SLSQP polish (only at moderate n)
    if len(bh) <= 600 and remaining() > 50:
        for T_v in [8.0, 4.0, 2.0]:
            if remaining() < 25:
                break
            bh = slsqp_polish(bh, T_v, 250)
            c_new = compute_c5(bh)
            if c_new < bc:
                bc = c_new
                print(f"  SLP(T={T_v}): C5={bc:.6f} t={elapsed():.0f}", flush=True)

    # Phase 7: Final ultra-fine FISTA
    if remaining() > 20:
        _PARAMS['m'] = 30
        _PARAMS['eps'] = 0.001
        n_it = min(2000, int(remaining() * 60))
        hr, cr = fista_run(bh, max(n_it, 200), sg_active, lr0=0.02, decay=0.9998, check=10)
        if cr < bc:
            bc, bh = cr, hr.copy()

    # Phase 8: Tiny perturbation kicks
    for trial in range(3):
        if remaining() < 15:
            break
        sc = 0.005 + 0.005 * trial
        hp = project_c(bh + rng.randn(len(bh)) * sc, len(bh))
        _PARAMS['m'] = 15
        _PARAMS['eps'] = 0.003
        hr, cr = fista_run(hp, 1500, sg_active, lr0=0.04, decay=0.999, check=15)
        if cr < bc:
            bc, bh = cr, hr.copy()
            print(f"  Pert{trial}: C5={bc:.6f} t={elapsed():.0f}", flush=True)

    bh = project_c(bh, len(bh))
    bc = compute_c5(bh)
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={elapsed():.0f}", flush=True)
    return bh, bc, len(bh)
