    corr = np.correlate(h, 1.0 - h, mode='full')
    n_pts = 2 * n - 1
    
    for j in range(n):
        old_h = h[j]
        # Compute a[p] for this j
        # a[p] = [0<=j-k<n] - [0<=j+k<n] where k = p-(n-1)
        # Equivalent: a[p] = [0<=j-(p-(n-1))<n] - [0<=j+(p-(n-1))<n]
        # = [n-1+j-p >= 0 and n-1+j-p < n] - [n-1+j+p >= n-1 and ...]
        
        # Let me just compute it directly
        k_vals = np.arange(-(n-1), n)  # length 2n-1
        a = np.zeros(n_pts)
        a[(j - k_vals >= 0) & (j - k_vals < n)] += 1
        a[(j + k_vals >= 0) & (j + k_vals < n)] -= 1
        
        # b[p] = corr[p] - a[p] * old_h
        b = corr - a * old_h
        
        # Now find t in [0,1] minimizing max_p (a[p]*t + b[p])
        # Since a[p] in {-1, 0, 1}:
        B1 = np.max(b[a == 1]) if np.any(a == 1) else -np.inf
        B0 = np.max(b[a == 0]) if np.any(a == 0) else -np.inf
        Bm1 = np.max(b[a == -1]) if np.any(a == -1) else -np.inf
        
        # f(t) = max(t + B1, B0, -t + Bm1)
        # Candidates: 0, 1, (Bm1-B1)/2, B0-B1, Bm1-B0
        candidates = [0.0, 1.0]
        if np.isfinite(B1) and np.isfinite(Bm1):
            candidates.append((Bm1 - B1) / 2.0)
        if np.isfinite(B1):
            candidates.append(B0 - B1)
        if np.isfinite(Bm1):
            candidates.append(Bm1 - B0)
        
        best_t = old_h
        best_f = np.inf
        for t in candidates:
            t = max(0.0, min(1.0, t))
            f = max(t + B1, B0, -t + Bm1)
            if f < best_f:
                best_f = f
                best_t = t
        
        if abs(best_t - old_h) > 1e-15:
            delta = best_t - old_h
            h[j] = best_t
            corr += a * delta
    
    return h, corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def gauss_seidel_pass(h, n, max_bisect=45):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_pts = 2 * n - 1
    k_vals = np.arange(-(n - 1), n)
    for j in range(n):
        old_hj = h[j]
        valid1 = (j - k_vals >= 0) & (j - k_vals < n)
        valid2 = (j + k_vals >= 0) & (j + k_vals < n)
        a = np.zeros(n_pts)
        if np.any(valid1):
            a[valid1] += (1.0 - h[j - k_vals[valid1]])
        if np.any(valid2):

            mid_t = (lo_t + hi_t) / 2.0

            slopes = a[mask]
            min_s = np.min(slopes)
            max_s = np.max(slopes)
            if min_s <= 1e-12 and max_s >= -1e-12:
