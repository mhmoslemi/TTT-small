def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_lbfags(h, n, t, maxiter=200):
    """Optimize using L-BFGS-B with soft-max surrogate."""
    dx = 2.0 / n
    self_idx = n - 1
    
    def obj_and_grad(x):
        # x is the raw variable; h = x with clipping and projection
        h = np.clip(x, 0.0, 1.0)
        # Don't enforce sum constraint here (L-BFGS-B handles bounds)
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr)
        exp_terms = np.exp(t * (corr - max_c) * dx)
        z = np.sum(exp_terms)
        obj = (max_c + np.log(z) / t) * dx
        w = exp_terms / z
        
        # Gradient: for each j, grad[j] = dx * sum_k w[k] * d(corr_k)/d(h_j)
        # Using the linearity insight:
        # d(corr_k)/d(h_j) = (corr_hi_k - corr_lo_k) effectively
        # But that requires two evaluates per step which is expensive.
        # Use the analytical formula:
        # d(corr_k)/d(h_j) = (1 - h[j+s_k]) - h[j-s_k] (with bounds, 0 for out of bounds)
        # where s_k is the shift for lag k.
        # 
        # Alternative: just compute numerically via finite difference for the gradient.
        # For n=640, this is 640 extra correlate calls = expensive.
        # 
        # Better: use the structure. For each lag k in 0..2n-2:
        #   The shift is s = k - (n-1) (positive means h is shifted right)
        #   d(corr_k)/d(h_j) = (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)
        #   (using the convention from np.correlate)
        
        # Actually let me verify the shift convention.
        # np.correlate(h, 1-h, 'full')[k]: 
        # From numpy: for same-length arrays of length n,
        # result[k] = sum_i h[i] * (1-h[i+k-(n-1)])
        # So the "shift" for index k is s = k - (n-1)
        # d(result[k])/d(h_j) = (1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)
        # where s = k-(n-1)
        
        # Compute gradient efficiently:
        # grad[j] = dx * sum_k w[k] * [(1-h[j+s_k])*I(0<=j+s_k<n) - h[j-s_k]*I(0<=j-s_k<n)]
        # = dx * [sum_k w[k]*(1-h[j+s_k])*I(...) - sum_k w[k]*h[j-s_k]*I(...)]
        
        # This is essentially a convolution of w with (1-h) and w with h.
        # Let's compute it directly:
        grad = np.zeros(n)
        shifts = np.arange(2 * n - 1) - (n - 1)  # shifts[k] = k - (n-1)
        
        for k in range(2 * n - 1):
            if w[k] < 1e-15:
                continue
            s = shifts[k]
            wk = w[k] * dx
            if s >= 0:
                # (1-h[j+s]) for j in [0, n-1-s]
                j_range = np.arange(0, n - s)
                grad[j_range] += wk * (1.0 - h[j_range + s])
                # -h[j-s] for j in [s, n-1]
                j_range2 = np.arange(s, n)
                grad[j_range2] -= wk * h[j_range2 - s]
            else:
                p = -s
                # (1-h[j-p]) for j in [p, n-1]
                j_range = np.arange(p, n)
                grad[j_range] += wk * (1.0 - h[j_range - p])
                # -h[j+p] for j in [0, n-1-p]
                j_range2 = np.arange(0, n - p)
                grad[j_range2] -= wk * h[j_range2 + p]
        
        return obj, grad
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_and_grad, h.copy(), method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
    
    if res.x is not None:
        h_new = make_feasible(res.x, n)
        return h_new
    return h

                best_h = h_new
    
    # Phase 3: Coordinate descent after L-BFGS
    if time.time() - start_time < budget_s - 200:
        h_cd, c_cd = exact_coord_descent(best_h, n, max_rounds=10,
                                          time_limit=budget_s, start_time=start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd
    
    # Phase 4: Upsample to higher resolution
    n2 = n * 2 if n * 2 <= 2048 else max(n + 256, 1280)
    if time.time() - start_time < budget_s - 150:
        h2 = upsample_h(best_h, n2)
        best_h2, best_c5_2 = exact_coord_descent(h2, n2, max_rounds=10,
                                                  time_limit=budget_s, start_time=start_time)
        if best_c5_2 < best_c5:
            best_c5 = best_c5_2
            best_h = best_h2
            n = n2
    
    # Phase 5: L-BFGS at higher resolution
    if time.time() - start_time < budget_s - 80:
        for t in [50, 150]:
            if time.time() - start_time > budget_s - 50:
                break
            h_new = soft_max_lbfags(best_h, n, t, maxiter=100)
            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-10:
                best_c5 = c_new
                best_h = h_new
    
    # Phase 6: Final coordinate descent
    h_final, c_final = exact_coord_descent(best_h, n, max_rounds=15,
                                            time_limit=budget_s, start_time=start_time)
    if c_final < best_c5:
        best_c5 = c_final
        best_h = h_final
    
    # Ensure feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
