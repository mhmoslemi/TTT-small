    # np.correlate(a, v) = np.convolve(a, v[::-1])
    # So: correlate(h, 1-h) = convolve(h, (1-h)[::-1])
    # In FFT: ifft(fft(h) * fft((1-h)[::-1]))
    # = ifft(fft(h) * fft(1-h)[::-1]) ... no, that's not right.
    # 
    # Let me use: correlate(a, v) = convolve(a, flip(v))
    # convolve in FFT: ifft(fft(a, N) * fft(v_flip, N))
    # where N >= len(a) + len(v) - 1 = 2n - 1, so N = 2n
    v_flip = one_minus_h[::-1]
    f1 = np.fft.rfft(h, N)
    f2 = np.fft.rfft(v_flip, N)
    result = np.fft.irfft(f1 * f2, N)
    # The result has length N, but the valid part is [0, 2n-2]
    return result[:2*n-1]

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def get_grad(h, n, top_k, tau=None):
    """Compute gradient of soft-max or hard-max of correlation.
    Returns (gradient, c5_value)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = corr.max() * dx
    
    if top_k < len(corr):
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(corr))
    top_vals = corr[top_idx]
    
    if tau is not None and tau < 1e10:
        max_v = top_vals.max()
        shifted = tau * (top_vals - max_v)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
    else:
        # Hard max: equal weight on all top entries that achieve the max
        max_v = top_vals.max()
        mask = top_vals >= max_v - 1e-12
        p = np.zeros_like(top_vals)
        p[mask] = 1.0 / mask.sum()
    
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        pk = p[i_idx]
        k = top_idx[i_idx]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    return grad, c5
