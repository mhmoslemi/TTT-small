    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argsort(h)
        if diff > 0:
            for i in range(n - 1, -1, -1):
                add = min(diff, 1.0 - h[idx[i]])
                h[idx[i]] += add
                diff -= add
                if abs(diff) < 1e-13:
                    break
        else:
            for i in range(n):
                sub = min(-diff, h[idx[i]])
                h[idx[i]] -= sub
                diff += sub
                if abs(diff) < 1e-13:
                    break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

_G_T = 0.0
_G_N = 0
_G_MU = 0.0

def _obj_and_grad(h):
    n = _G_N
    T = _G_T
    mu = _G_MU
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exps = np.exp(T * shifted)
    exp_sum = np.sum(exps)
    weights = exps / exp_sum
    obj = (cmax + np.log(exp_sum) / T) * dx
    obj += mu * (np.sum(h) - n / 2.0) ** 2
    grad = np.zeros(n)
    # Lag 0
    w0 = weights[n - 1]
    grad += w0 * (1.0 - 2.0 * h)
    # Lags s = 1, ..., n-1
    for s in range(1, n):
        w_pos = weights[n - 1 + s]
        w_neg = weights[n - 1 - s]
        if w_pos > 1e-15:
            grad[s:] += w_pos * (1.0 - h[:n - s])
            if 2 * s < n:
                grad[s:n - s] -= w_pos * h[2 * s:n]
        if w_neg > 1e-15:
            grad[:n - s] += w_neg * (1.0 - h[s:])
            if 2 * s < n:
                grad[s:n - s] -= w_neg * h[:n - 2 * s]
    grad *= dx
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return obj, grad

def _lbfgs_step(h, n, T, mu, max_iter, ftol):
    global _G_T, _G_N, _G_MU
    _G_T = T
    _G_N = n
    _G_MU = mu
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_and_grad, h, method='L-BFGS-B',
                   jac=True, bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10, 'maxfun': 3 * max_iter})
    return res.x

def _subgrad_dir(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    L = k_star - (n - 1)
    sg = np.zeros(n)
    if L >= 0:
        s = L
        if s == 0:
            sg[:] = 1.0 - 2.0 * h
        else:
            sg[s:] = (1.0 - h[:n - s])
            if 2 * s < n:
                sg[s:n - s] -= h[2 * s:n]
    else:
        s = -L
        sg[:n - s] = (1.0 - h[s:])
        if 2 * s < n:
            sg[s:n - s] -= h[:n - 2 * s]
    return sg * dx

def run(seed=42, budget_s=1300, **kwargs):
    global _G_T, _G_N, _G_MU
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - start_time)

    h = np.array(initial_h_values, dtype=float).copy()
    n0 = len(h)
    h = make_feasible(h, n0)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: L-BFGS-B optimization at increasing resolutions
    resolutions = [n0, 1280, 2560, 4096]
    for n_t in resolutions:
        if _tl() < 100:
            break
        n_t = min(n_t, 5120)
        h_w = upsample_h(best_h, n_t) if n_t != len(best_h) else best_h.copy()
        c5_w = compute_c5(h_w)
        if c5_w < best_c5:
            best_c5 = c5_w
            best_h = h_w.copy()

        temps = [12.0, 30.0, 80.0, 200.0, 500.0]
        for T in temps:
            if _tl() < 35:
                break
            try:
                mi = 250 if n_t <= 1280 else 150
                h_opt = _lbfgs_step(h_w, n_t, T, 800.0, mi, 1e-13)
                h_opt = make_feasible(h_opt, n_t)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_w = h_opt.copy()
            except Exception:
                pass

        # Random restart
        if _tl() > 70:
            h_r = 0.65 * upsample_h(best_h, n_t) + 0.35 * rng.rand(n_t)
            h_r = make_feasible(h_r, n_t)
            try:
                h_opt = _lbfgs_step(h_r, n_t, 50.0, 800.0, 120, 1e-13)
                h_opt = make_feasible(h_opt, n_t)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 2: Subgradient on exact max
    if _tl() > 80:
        n_s = len(best_h)
        h_s = best_h.copy()
        alpha = 0.03
        for it in range(8000):
            if _tl() < 25:
                break
            sg = _subgrad_dir(h_s, n_s)
            h_new = h_s - alpha * sg
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = make_feasible(h_new, n_s)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_s = h_new
            if it % 1000 == 999:
                alpha *= 0.75
        # Also try opposite direction from current best
        if _tl() > 40:
            h_s2 = best_h.copy()
            for it in range(3000):
                if _tl() < 25:
                    break
                sg = _subgrad_dir(h_s2, n_s)
                h_new = h_s2 + alpha * sg  # note: + to explore
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = make_feasible(h_new, n_s)
                c5_new = compute_c5(h_new)
                if c5_new > 0.385:
                    h_new = h_s2 - 0.5 * alpha * sg
                    h_new = np.clip(h_new, 0.0, 1.0)
                    h_new = make_feasible(h_new, n_s)
                    c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_s2 = h_new
                if it % 1000 == 999:
                    alpha *= 0.75

    # Phase 3: One more L-BFGS-B pass at highest T for smoothing
    if _tl() > 50:
        n_f = len(best_h)
        try:
            h_opt = _lbfgs_step(best_h, n_f, 1500.0, 1000.0, 100, 1e-14)
            h_opt = make_feasible(h_opt, n_f)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 4: Coordinate descent polish
    if _tl() > 40:
        n_cd = len(best_h)
        h_cd = best_h.copy()
        deltas = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
        for rd in range(4):
            if _tl() < 15:
                break
            improved = False
            step = max(1, n_cd // 4096)
            for j in range(0, n_cd, step):
                if _tl() < 10:
                    break
                ov = h_cd[j]
                for d in deltas:
                    nv = ov + d
                    if nv < 0.0 or nv > 1.0:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-13:
                        best_c5 = c
                        h_cd = tr
                        improved = True
                        break
                if 0.1 < h_cd[j] < 0.9:
                    for bv in (0.0, 1.0):
                        tr = h_cd.copy()
                        tr[j] = bv
                        tr = make_feasible(tr, n_cd)
                        c = compute_c5(tr)
                        if c < best_c5 - 1e-13:
                            best_c5 = c
                            h_cd = tr
                            improved = True
                            break
            if not improved:
                break
        best_h = h_cd.copy()

    # Final
    best_h = make_feasible(best_h, len(best_h))
    fc = compute_c5(best_h)
    if fc < best_c5:
        best_c5 = fc
    return best_h, best_c5, len(best_h)
