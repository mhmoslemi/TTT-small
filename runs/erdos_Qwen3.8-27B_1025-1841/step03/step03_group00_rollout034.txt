                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            # Try perturbation: flip a few random pairs to escape local optimum
            if rnd < max_rounds // 2:
                n_perturb = min(5, len(ones) // 2)
                for _ in range(n_perturb):
                    ii = rng.randint(len(ones))
                    jj = rng.randint(len(zeros))
                    h[ones[ii]] = 0.0
                    h[zeros[jj]] = 1.0
                    ones = np.delete(ones, ii)
                    zeros = np.delete(zeros, jj)
                    ones = np.append(ones, zeros[jj] if False else zeros[jj] - 1)  # placeholder
                    # Actually let's just rebuild
                ones = np.where(h > 0.5)[0].copy()
                zeros = np.where(h <= 0.5)[0].copy()
                current = compute_c5(h)
    return h, current

def binary_sa(h, n, rng, max_rounds, start_time, time_limit):
    """Simulated annealing binary search."""
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    n_target = n // 2
    if n_ones > n_target:
        one_idx = np.where(h == 1)[0]
        for k in range(n_ones - n_target):
            h[one_idx[k]] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(h == 0)[0]
        for k in range(n_target - n_ones):
            h[zero_idx[k]] = 1.0

    current = compute_c5(h)
    best_c5 = current
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    T_sa = 0.001

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 5:
            break
        n_tries = 1500
        improved = False
        for _ in range(n_tries):
            if time.time() - start_time > time_limit - 5:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-14 or (c > current and rng.random() < np.exp(-(c - current) / T_sa)):
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
                    improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        T_sa *= 0.97
        if not improved and T_sa < 1e-6:
            break
    return best_h, best_c5

def targeted_cd(h, n, max_rounds, start_time, time_limit):
    h = h.copy()
    current = compute_c5(h)
    n_target = n // 2

    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 5:
            break
        improved = False
        n_ones = int(np.sum(h > 0.5))
        for i in range(0, n, max(1, n // 500)):
            if time.time() - start_time > time_limit - 5:
                break
            old = h[i]
            if old > 0.5:
                nv = 0.0
                if n_ones <= n_target:
                    continue
            elif old < 0.5:
                nv = 1.0
                if n_ones >= n_target:
                    continue
            else:
                nv = 1.0 if n_ones < n_target else 0.0

            trial = h.copy()
            trial[i] = nv
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < current - 1e-14:
                h = trial
                current = c
                n_ones = int(np.sum(h > 0.5))
                improved = True

    return h, current

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    time_limit = start_time + budget_s
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Work at n=1024 with soft-max
    n1 = 1024
    h_w = upsample_h(best_h, n1)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()
    dx1 = 2.0 / n1

    temps = [(50, 2), (100, 2), (300, 2), (600, 2), (1200, 2), (2500, 2), (5000, 2), (10000, 1)]
    for T, n_restart in temps:
        if time.time() - start_time > time_limit - 40:
            break
        for r in range(n_restart):
            if time.time() - start_time > time_limit - 20:
                break
            if r == 0:
                h_init = h_w.copy()
            else:
                h_init = h_w + 0.08 * rng.randn(n1)
                h_init = make_feasible(h_init, n1)
            try:
                h_opt = lbgfb_optimize(h_init, T, dx1, 500.0, n1, max_iter=250)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_w = h_opt.copy()
            except Exception:
                pass

    # Phase 2: Binary SA at n=1024
    if time.time() - start_time < time_limit - 80:
        h_sa, c5_sa = binary_sa(best_h, n1, rng, max_rounds=80,
                                start_time=start_time, time_limit=time_limit - 40)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()

    # Phase 3: Targeted CD at n=1024
    if time.time() - start_time < time_limit - 60:
        h_cd, c5_cd = targeted_cd(best_h, n1, max_rounds=30,
                                   start_time=start_time, time_limit=time_limit - 30)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 4: Work at n=2048
    n2 = 2048
    if time.time() - start_time < time_limit - 180:
        h_2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h_2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h_2.copy()
        dx2 = 2.0 / n2

        for T, n_r in [(100, 2), (500, 2), (1500, 2), (4000, 1)]:
            if time.time() - start_time > time_limit - 60:
                break
            for r in range(n_r):
                if time.time() - start_time > time_limit - 30:
                    break
                if r == 0:
                    h_init = h_2.copy()
                else:
                    h_init = h_2 + 0.05 * rng.randn(n2)
                    h_init = make_feasible(h_init, n2)
                try:
                    h_opt = lbgfb_optimize(h_init, T, dx2, 500.0, n2, max_iter=200)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        h_2 = h_opt.copy()
                except Exception:
                    pass

        # Binary SA at n=2048
        if time.time() - start_time < time_limit - 60:
            h_sa2, c5_sa2 = binary_sa(h_2, n2, rng, max_rounds=50,
                                      start_time=start_time, time_limit=time_limit - 30)
            if c5_sa2 < best_c5:
                best_c5 = c5_sa2
                best_h = h_sa2.copy()

    # Phase 5: Final high-T polish at best resolution
    if time.time() - start_time < time_limit - 20:
        n_fin = len(best_h)
        dx_fin = 2.0 / n_fin
        try:
            h_fin = lbgfb_optimize(best_h, 8000, dx_fin, 500.0, n_fin, max_iter=150)
            c5_fin = compute_c5(h_fin)
            if c5_fin < best_c5:
                best_c5 = c5_fin
                best_h = h_fin.copy()
        except Exception:
            pass

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
