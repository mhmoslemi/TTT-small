def compute_grad(h):
    n = len(h)
    dx = 2.0 / n
    T = _OPT_T
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    max_sh = np.max(shifted)
    w = np.exp(T * (shifted - max_sh))
    w /= np.sum(w)
    
    # w has length 2n-1, indexed by j = 0, ..., 2n-2
    # p_j = j - (n-1), so p ranges from -(n-1) to (n-1)
    
    grad = np.zeros(n)
    p = np.arange(-(n-1), n)  # p_j values
    
    for k in range(n):
        # S1: sum over j where 0 ≤ k-p_j < n of w_j * (1-h[k-p_j])
        valid1 = (k - p >= 0) & (k - p < n)
        d1 = k - p[valid1]
        S1 = np.dot(w[valid1], 1.0 - h[d1])
        
        # S2: sum over j where 0 ≤ k+p_j < n of w_j * h[k+p_j]  
        valid2 = (k + p >= 0) & (k + p < n)
        d2 = k + p[valid2]
        S2 = np.dot(w[valid2], h[d2])
        
        grad[k] = dx * (S1 - S2)
    
    return grad
