def _fft_corr(h):
    """Compute np.correlate(h, 1-h, 'full') using FFT."""

import numpy as np
from scipy.optimize import minimize
import time

_gN = 0
_gT = 50.0

def _fft_corr(h):
    n = len(h)
    N = 1 << (2 * n - 1).bit_length()
    ha = np.zeros(N); ha[:n] = h
    oa = np.zeros(N); oa[:n] = 1.0 - h
    c = np.fft.irfft(np.fft.rfft(ha) * np.conj(np.fft.rfft(oa)), N)
    r = np.empty(2 * n - 1)
    r[n - 1:] = c[:n]
    r[:n - 1] = c[N - 1:N - n - 1:-1]
    return r

def _c5f(h):
    return float(np.max(_fft_corr(h)) * 2.0 / len(h))

def _c5x(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    t = n / 2.0
    for _ in range(80):
        d = t - h.sum()
        if abs(d) < 1e-14: break
        m = h < 1.0 if d > 0 else h > 0.0
        c = m.sum()
        if c == 0: break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    corr = _fft_corr(x)
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_gT * shifted))) / _gT) * (2.0 / _gN)

def _con(x):
    return x.sum() - _gN / 2.0

def _conj(x):
    return np.ones(_gN)

def _tm(n):
    return np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)

def _to_bin(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:n // 2]] = 1.0
    return b

def _phase_transition(h, n):
    """Generate h based on phase: h[i] = 1 if cos(2*pi*i/n) > 0, adjusted for sum."""
    n_half = n // 2
    h = np.zeros(n)
    angles = np.arange(n) * 2.0 * np.pi / n
    vals = np.sort(np.cos(angles))[::-1]
    threshold = vals[n_half - 1]
    h = np.where(np.cos(angles) >= threshold, 1.0, 0.0)
    # Fix sum if needed
    s = h.sum()
    if int(s) != n_half:
        idx1 = np.where(h == 1)[0]
        idx0 = np.where(h == 0)[0]
        if int(s) > n_half:
            for i in range(int(int(s) - n_half)):
                h[idx1[i]] = 0
        else:
            for i in range(int(n_half - int(s))):
                h[idx0[i]] = 1
    return h

def _peak_local_search(h, n, t_deadline, rng):
    """Local search focusing on peak correlation lags."""
    h = h.copy()
    best_c = _c5f(h)
    ones = set(int(i) for i in np.where(h == 1)[0])
    zeros = set(int(i) for i in np.where(h == 0)[0])
    stall = 0
    while time.time() < t_deadline and stall < 5000:
        corr = _fft_corr(h)
        peak_idx = int(np.argmax(corr))
        lag = peak_idx - (n - 1)
        # Find contributing pairs
        improved = False
        if lag > 0:
            pairs = [(i + lag, i) for i in range(n - lag) if h[i + lag] == 1 and h[i] == 0]
        elif lag < 0:
            ll = -lag
            pairs = [(i, i + ll) for i in range(n - ll) if h[i] == 1 and h[i + ll] == 0]
        else:
            pairs = [(i, i) for i in range(n) if h[i] == 1]
        rng.shuffle(pairs)
        tried = 0
        for (a, b) in pairs[:200]:
            if time.time() > t_deadline:
                break
            tried += 1
            # Try swapping h[a] and some zero
            if h[a] == 1 and a in ones:
                for _ in range(3):
                    z = rng.choice(list(zeros))
                    if z == a:
                        continue
                    h[a] = 0.0; h[z] = 1.0
                    ones.discard(a); ones.add(z)
                    zeros.discard(z); zeros.add(a)
                    c = _c5f(h)
                    if c < best_c - 1e-15:
                        best_c = c
                        improved = True
                        stall = 0
                    else:
                        h[a] = 1.0; h[z] = 0.0
                        ones.discard(z); ones.add(a)
                        zeros.discard(a); zeros.add(z)
        if not improved:
            stall += 1
    return h, best_c

        if hv is not None:

        if n == 512:

            cur = _feasible(sp, n)

            for T in [30.0, 60.0, 120.0, 250.0]:

                            cur[j] = bv
                            cur = _feasible(cur, n)
