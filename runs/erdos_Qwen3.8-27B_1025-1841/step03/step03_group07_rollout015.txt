import numpy as np
import time

def project_to_feasible(g, n):
    """Project g onto {0<=x<=1, sum(x)=n/2} using bisection on threshold."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5_fft(h):
    """Compute C5 using FFT for speed."""
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    acorr = np.fft.irfft(H * np.conj(H), N)[:n]
    
    cumsum = np.zeros(n + 1)
    cumsum[1:] = np.cumsum(h)
    total = cumsum[n]
    
    s = np.arange(n)
    val_pos = total - cumsum[s] - acorr
    val_neg = cumsum[n:0:-1] - acorr
    
    max_val = max(val_pos.max(), val_neg.max())
    return max_val * dx

def compute_c5_and_grad(h, n, tau, top_k=10):

    # val_neg[s] corresponds to shift -s (for s=0, same as val_pos[0])
    # All 2n-1 shifts: val_neg[0], val_neg[1], ..., val_neg[n-1], val_pos[0], val_pos[1], ..., val_pos[n-1]
    # Actually val_neg[0] = val_pos[0] so we have 2n-1 distinct values

    shifted = tau * (all_vals - max_val)

    s_exp = exp_c.sum()

def nesterov_fft(h, n, tau, time_limit, lr=None, top_k=10):
    """Nesterov optimization with FFT-based gradient."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5_fft(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.5 / tau
    
    h_prev = h.copy()
    t = 1.0
    stall = 0
    
    for it in range(5000000):
        if time.time() - start > time_limit:
            break
        if it % 2000 == 0 and it > 0:
            # Check for time more frequently at first
            pass
        
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.95)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        obj, grad = compute_c5_and_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        
        if it % 500 == 0:
            c5_now = compute_c5_fft(h_new)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h_new.copy()
                stall = 0
            else:
                stall += 500
            if stall > 20000:
                lr *= 0.6
                stall = 0
                if lr < 1e-16:
                    break
        
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    
    c5_final = compute_c5_fft(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def directed_gradient(h, n, time_limit, lr=0.001, top_k=5):

    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5_fft(h)
    best_h = h.copy()
    start = time.time()

    beta = 0.7
    stall = 0

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

            v *= 0.5

        else:

        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()

        h_new = h + v

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    acorr = np.fft.irfft(H * np.conj(H), N)[:n]
    cumsum = np.zeros(n + 1)
    cumsum[1:] = np.cumsum(h)
    total = cumsum[n]
    s = np.arange(n)
    val_pos = total - cumsum[s] - acorr
    val_neg = cumsum[n:0:-1] - acorr
    max_val = max(val_pos.max(), val_neg.max())
    return max_val * dx


def compute_c5_and_grad(h, n, tau, top_k=10):
    dx = 2.0 / n
    N = 2 * n
    H = np.fft.rfft(h, N)
    acorr = np.fft.irfft(H * np.conj(H), N)[:n]
    cumsum = np.zeros(n + 1)
    cumsum[1:] = np.cumsum(h)
    total = cumsum[n]
    s = np.arange(n)
    val_pos = (total - cumsum[s] - acorr) * dx
    val_neg = (cumsum[n:0:-1] - acorr) * dx
    all_vals = np.concatenate([val_neg[n-1:0:-1], val_pos])
    max_val = all_vals.max()
    shifted = tau * (all_vals - max_val)
    shifted = np.clip(shifted, -50.0, 0.0)
    exp_c = np.exp(shifted)
    s_exp = exp_c.sum()
    obj = max_val + np.log(s_exp) / tau
    if top_k < len(all_vals):
        top_idx = np.argpartition(all_vals, -top_k)[-top_k:]
    else:
        top_idx = np.arange(len(all_vals))
    top_props = np.exp(tau * (all_vals[top_idx] - max_val))
    top_props = np.clip(top_props, 0.0, 1.0)
    top_props /= top_props.sum()
    grad = np.zeros(n)
    for i_idx in range(len(top_idx)):
        idx = top_idx[i_idx]
        pk = top_props[i_idx]
        shift = idx - (n - 1)
        if shift >= 0:
            sv = shift
            if sv == 0:
                grad += (1.0 - 2.0 * h)
            else:
                grad[sv:] += (1.0 - h[:n - sv])
                grad[:n - sv] -= h[sv:]
        else:
            r = -shift
            grad[:n - r] += (1.0 - h[r:])
            grad[r:] -= h[:n - r]
        grad *= pk
    grad *= dx
    return obj, grad


def nesterov_fft(h, n, tau, time_limit, lr=None, top_k=10):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5_fft(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.5 / tau
    h_prev = h.copy()
    t = 1.0
    stall = 0
    eval_count = 0
    for it in range(10000000):
        if time.time() - start > time_limit:
            break
        mu = (t - 1.0) / (t + 2.0)
        mu = min(mu, 0.95)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        obj, grad = compute_c5_and_grad(y, n, tau, top_k)
        h_new = y - lr * grad
        h_new = project_to_feasible(h_new, n)
        eval_count += 1

                stall = 0
            else:
                stall += 400
            if stall > 16000:
                lr *= 0.65
                stall = 0
                if lr < 1e-16:
                    break
        t = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
        h_prev = h
        h = h_new
    c5_final = compute_c5_fft(h)
    if c5_final < best_c5:

        best_h = h.copy()
    return best_h, best_c5

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)
