        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-15})
            res_h = make_feasible(res.x, n)
            c5_r = compute_c5(res_h)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = res_h
        except Exception:
            pass
        _G_TOP = None

    # Phase 5: Final pair swaps
    if remaining() > 60:
        h5 = pair_swap_phase(best_h, n, rng, remaining() - 20)
        c5_5 = compute_c5(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Random restarts with pair swap
    for restart in range(4):
        if remaining() < 60:
            break
        scale = 0.015 + 0.01 * restart
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        h_r = pair_swap_phase(h_try, n, rng, min(80, remaining() - 20))
        c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 7: Final SLSQP with very high T
    if remaining() > 40:
        _G_N = n
        _G_T = 1500.0
        _G_TOP = None
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-16})
            res_h = make_feasible(res.x, n)
            c5_r = compute_c5(res_h)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = res_h
        except Exception:
            pass

    # Final feasibility check
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
