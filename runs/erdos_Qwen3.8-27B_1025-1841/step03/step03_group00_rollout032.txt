    delta = np.zeros(2*n-1)
    # For m in range(2n-1), k = m-(n-1)
    # delta[m] = h[i-k] (if valid) - h[j-k] (if valid) + h[i+k] (if valid) - h[j+k] (if valid)

    k = np.arange(2*n-1) - (n-1)  # range from -(n-1) to n-1

    i_plus_k = i + k

    mask_im = (i_minus_k >= 0) & (i_minus_k < n)
    mask_ip = (i_plus_k >= 0) & (i_plus_k < n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):

    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:

        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones"""
    h = np.asarray(h, dtype=float)
    n_ones = n // 2
    # Sort by value, set top n_ones to 1
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def greedy_local_search(h, n, max_sweeps=50):
    """Greedy: find best improving swap each time"""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = np.max(corr).astype(float)
    k_vals = np.arange(2*n-1) - (n-1)  # lag values
    
    for sweep in range(max_sweeps):
        best_delta = 0
        best_i = -1
        best_j = -1
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        # For each swap, compute the new max and check improvement
        # We want to minimize the max, so we look for swaps that reduce it
        improved = False
        
        for i in ones:
            for j in zeros:
                # Compute delta
                imk = i - k_vals
            ipk = i + k_vals
                jmk = j - k_vals
                jpk = j + k_vals
                
                delta = np.zeros(2*n-1)
                m_im = (imk >= 0) & (imk < n)
                m_ip = (ipk >= 0) & (ipk < n)
                m_jm = (jmk >= 0) & (jmk < n)
                m_jp = (jpk >= 0) & (jpk < n)
                
                if np.any(m_im):
                    delta[m_im] += h[imk[m_im]]
                if np.any(m_ip):
                    delta[m_ip] += h[ipk[m_ip]]
                if np.any(m_jm):
                    delta[m_jm] -= h[jmk[m_jm]]
                if np.any(m_jp):
                    delta[m_jp] -= h[jpk[m_jp]]
                
                new_corr = corr + delta
                new_max = np.max(new_corr)
                if new_max < current_max - 1e-10:
                    if best_i == -1 or new_max < best_delta:
                        best_delta = new_max
                        best_i = i
                        best_j = j
                        best_new_corr = new_corr.copy()
                    improved = True
        
        if not improved:
            break
        
        # Apply best swap
        h[best_i] = 0.0
        h[best_j] = 1.0
        corr = best_new_corr
        current_max = best_delta
    
    return h, float(current_max) * (2.0/n)

Then delta_j[m] = A[m] + B[m] - h[j-k[m]] - h[j+k[m]]

For all j simultaneously, h[j-k[m]] for all j is a circulant-like matrix... but it's not exactly circulant due to boundaries.

Actually, h[j-k[m]] for fixed m and varying j is just h shifted by k[m]. So for all j:
h[j-k[m]] = h[j+k'[m]] where k'[m] = -k[m]

Alternative: For each candidate swap (i,j), the change in the max is determined by which lags are affected. The key insight is:

new_max = max(corr + delta)

We want new_max < current_max. This means:
- For all m where corr[m] == current_max: corr[m] + delta[m] < current_max, i.e., delta[m] < 0
- For all other m: corr[m] + delta[m] ≤ current_max (or at least < current_max + ε)

Actually, the first condition is necessary but not sufficient. We need the max of the entire new_corr to be < current_max.

Let me think of a more efficient approach. Instead of checking all pairs (i,j), what about:

1. For each i (position to turn off), precompute the "profile" P_i[m] = h[i-k[m]] + h[i+k[m]] (the contribution from removing position i)
2. For each j (position to turn on), precompute the "profile" Q_j[m] = h[j-k[m]] + h[j+k[m]] (the contribution from adding position j)
3. Then delta_{i,j}[m] = P_i[m] - Q_j[m]
4. new_corr = corr + P_i - Q_j
5. new_max = max(corr + P_i - Q_j)

For fixed i, we want to find j that minimizes max(corr + P_i - Q_j).

This is still O(n²) in the worst case, but with vectorization over j, we can make each i-iteration O(n) or O(n²) with constant factors.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def to_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(-np.asarray(h, dtype=float))
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def swap_delta(h, i, j, k_vals, n):
    """Vectorized delta for swap i(1->0), j(0->1)"""
    imk = i - k_vals
    ipk = i + k_vals
    jmk = j - k_vals
    jpk = j + k_vals
    
    delta = np.zeros(2*n-1)
    m1 = (imk >= 0) & (imk < n)
    m2 = (ipk >= 0) & (ipk < n)
    m3 = (jmk >= 0) & (jmk < n)
    m4 = (jpk >= 0) & (jpk < n)
    
    if np.any(m1):
        delta[m1] += h[imk[m1]]
    if np.any(m2):
        delta[m2] += h[ipk[m2]]
    if np.any(m3):
        delta[m3] -= h[jmk[m3]]
    if np.any(m4):
        delta[m4] -= h[jpk[m4]]
    return delta

def binary_sa(h_init, n, rng, time_limit, T_init=100.0, T_final=0.1, n_iter=None):
    """Simulated annealing on binary sequence"""
    h = h_init.copy()
    k_vals = np.arange(2*n-1) - (n-1)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    
    start = time.time()
    it = 0
    if n_iter is None:
        n_iter = 10**8  # will be limited by time
    
    while it < n_iter:
        if time.time() - start > time_limit:
            break
        
        if it % 10000 == 0:
            progress = it / n_iter
            T = T_init * (T_final / T_init) ** progress
        else:
            progress = it / n_iter
            T = T_init * (T_final / T_init) ** progress
        
        # Random swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        
        delta = swap_delta(h, i, j, k_vals, n)
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        
        # Accept or reject
        if new_max < current_max:
            accept = True
        else:
            dE = new_max - current_max
            if T > 0.01:
                accept = rng.random() < np.exp(-dE / T)
            else:
                accept = False
        
        if accept:
            h[i] = 0.0
            h[j] = 1.0
            corr = new_corr
            current_max = new_max
            
            # Update ones/zeros
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        
        it += 1
    
    return best_h, best_max

def greedy_binary_search(h_init, n, max_iters=50000, rng=None):
    """Greedy: always take improving swap"""
    h = h_init.copy()
    k_vals = np.arange(2*n-1) - (n-1)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    
    no_improve = 0
    it = 0
    while it < max_iters and no_improve < 500:
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        
        delta = swap_delta(h, i, j, k_vals, n)
        new_max = float(np.max(corr + delta))
        
        if new_max < current_max - 1e-10:
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + delta
            current_max = new_max
            no_improve = 0
            
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            
            if current_max < best_max:
                best_max = current_max
                best_h = h.copy()
        else:
            no_improve += 1
        
        it += 1
    
    return best_h, best_max

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n

    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def to_binary(h, n):
    n_ones = n // 2
    idx = np.argsort(-np.asarray(h, dtype=float))
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0
    return h_bin

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)

def binary_local_search(h_init, n, rng, max_iters=200000, time_limit=60.0):
    h = h_init.copy()
    k_vals = np.arange(2 * n - 1) - (n - 1)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_max = current_max
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].copy()
    zeros = np.where(h == 0.0)[0].copy()
    start = time.time()
    it = 0
    while it < max_iters:
        if time.time() - start > time_limit:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        imk = i - k_vals
        ipk = i + k_vals
        jmk = j - k_vals
        jpk = j + k_vals
        m1 = (imk >= 0) & (imk < n)
        m2 = (ipk >= 0) & (ipk < n)
        m3 = (jmk >= 0) & (jmk < n)
        m4 = (jpk >= 0) & (jpk < n)
        delta = np.zeros(2 * n - 1)
        if m1.any():
            delta[m1] += h[imk[m1]]
        if m2.any():
            delta[m2] += h[ipk[m2]]
        if m3.any():
            delta[m3] -= h[jmk[m3]]
        if m4.any():
            delta[m4] -= h[jpk[m4]]

            h[j] = 1.0

            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)

            zeros = np.append(zeros, i)

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)
