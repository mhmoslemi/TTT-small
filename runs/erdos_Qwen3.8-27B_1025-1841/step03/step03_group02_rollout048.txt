import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_reversal(h, n, max_iters, T0, T1, rng):
    """SA with reversal moves"""
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        move_type = rng.random()
        if move_type < 0.5:
            # Single swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i] = 1
                b[j] = 0
        elif move_type < 0.8:
            # Block swap: pick a block from ones-region and swap with same-size block from zeros-region
            block_size = rng.randint(1, 8)
            if block_size >= n:
                continue
            start1 = rng.randint(0, n - block_size)
            seg1 = b[start1:start1+block_size]
            n_ones1 = np.sum(seg1)
            n_zeros1 = block_size - n_ones1
            if n_ones1 == 0 or n_zeros1 == 0:
                continue
            # Find a region with opposite pattern
            start2 = rng.randint(0, n - block_size)
            if start2 == start1:
                continue
            seg2 = b[start2:start2+block_size]
            if np.sum(seg2) != n_zeros1:
                continue
            b[start1:start1+block_size] = seg2
            b[start2:start2+block_size] = seg1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[start1:start1+block_size] = seg1
                b[start2:start2+block_size] = seg2
        else:
            # Reversal
            i = rng.randint(0, n - 2)
            j = rng.randint(i + 1, n)
            b[i:j+1] = b[i:j+1][::-1]
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i:j+1] = b[i:j+1][::-1]
    return best_b, best_c5

Let me also think about **subgradient methods**. The objective f(h) = max_k corr_k(h) * dx is piecewise linear (for fixed h, each corr_k is linear in h). The subgradient at h is:

We can use a subgradient method or accelerated subgradient method. But this is slow for high-dimensional problems.

Actually, for SLSQP with the log-sum-exp trick, the gradient is:
∇f = dx * [∇max + (1/T) * sum(exp(T*(corr_k - max)) * ∇corr_k)]

For large T, this is approximately dx * ∇corr_{k*} (the gradient of the max correlation).

The question is: can SLSQP handle the non-differentiability at points where two correlations are tied for the max? The log-sum-exp smooths this out, so it should be fine.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def thue_morse(n):
    t = np.zeros(n)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return make_feasible(t, n)

def slsqp_optimize(h, n, T, maxiter=200, ftol=1e-15):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    h_new = np.interp(new_x, old_x, h)

    best_c5 = compute_c5(h)
