def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        order = np.argsort(h)
        b[order[-diff:]] = 1.0
    elif diff < 0:
        order = np.argsort(h)
        b[order[-diff:]] = 0.0

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    max_lag = np.argmax(corr)
    return float(max_val * dx), corr, max_lag

    """Approximate max using top-k mean (CVaR-like)"""
    topk = np.sort(corr)[-k:]
    return np.max(topk)  # or mean for CVaR

def slsqp_objective(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_constraint(x, n):
    return np.sum(x) - n / 2.0

def slsqp_constraint_jac(x, n):
    return np.ones(len(x))

def run_slsqp(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: slsqp_constraint(x, n), 
             'jac': lambda x: slsqp_constraint_jac(x, n)}]
    res = minimize(slsqp_objective, h.copy(), args=(n, T),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def best_periodic_pattern(p, n, target_sum):
    """Find the best periodic pattern of length p to tile to length n."""
    # target_sum = n/2 (number of 1s)
    # When tiling pattern of length p to length n, we get:
    # n = q*p + r, where q = n//p, r = n%p
    # Total 1s = q * (sum of pattern) + (sum of first r elements of pattern)
    # We need this to equal n/2
    
    q, r = divmod(n, p)
    best_c5 = float('inf')
    best_pat = None
    
    for mask in range(2**p):
        pat = [(mask >> i) & 1 for i in range(p)]
        total_ones = q * sum(pat) + sum(pat[:r])
        if total_ones != n // 2:
            continue
        
        # Tile to length n
        h = np.zeros(n)
        for i in range(n):
            h[i] = pat[i % p]
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_pat = pat
    
    return best_pat, best_c5

        if d <= j:
