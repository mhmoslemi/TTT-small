    t = T * (corr - M)
    t -= t.max()
    exp_t = np.exp(t)
    s = exp_t.sum()

    corr = np.correlate(h, 1.0 - h, mode='full')

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _g_obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = float(np.max(corr))
    t = T * (corr - M) - T * (corr - M).max()
    return (M + float(np.log(np.exp(t).sum())) / T) * dx

def _g_jac(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = float(np.max(corr))
    t = T * (corr - M)
    t = t - t.max()
    exp_t = np.exp(t)
    w = exp_t / float(exp_t.sum())
    lags = np.arange(2 * n - 1) - (n - 1)
    grad = np.ones(n)
    thresh = max(1e-14, float(w.max()) * 1e-10)
    for k in range(len(w)):
        wk = w[k]
        if wk < thresh:
            continue
        lag = int(lags[k])
        ms = max(0, -lag)
        me = min(n - 1, n - 1 - lag)
        if ms <= me:
            grad[ms:me + 1] -= wk * x[ms + lag:me + 1 + lag]
        ms2 = max(0, lag)
        me2 = min(n - 1, n - 1 + lag)
        if ms2 <= me2:
            grad[ms2:me2 + 1] -= wk * x[ms2 - lag:me2 + 1 - lag]
    return grad * dx / T

def _g_con(x):
    return x.sum() - _G_N / 2.0

def _g_conj(x):
    return np.ones(_G_N)

def slsqp_run(h0, n, T, maxiter):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    h0 = make_feasible(h0.copy(), n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _g_con, 'jac': _g_conj}]
    try:
        res = minimize(_g_obj, h0, jac=_g_jac, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return h0

def focused_grad_desc(h, n, n_iter, t_end, base_alpha):
    h = h.copy()
    alpha = base_alpha
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        sg = np.ones(n)
        ms = max(0, -lag)
        me = min(n - 1, n - 1 - lag)
        if ms <= me:
            sg[ms:me + 1] -= h[ms + lag:me + 1 + lag]
        ms2 = max(0, lag)
        me2 = min(n - 1, n - 1 + lag)
        if ms2 <= me2:
            sg[ms2:me2 + 1] -= h[ms2 - lag:me2 + 1 - lag]
        h = h - alpha * sg
        h = make_feasible(h, n)
        alpha = base_alpha / (1.0 + 0.05 * it)
    return h

def soft_avg_opt(h, n, n_iter, t_start, t_end):
    h = h.copy()
    for it in range(n_iter):
        frac = it / max(1, n_iter - 1)
        T = t_start * (t_end / t_start) ** frac
        alpha = 0.005 / (1.0 + 0.02 * it)
        corr = np.correlate(h, 1.0 - h, mode='full')
        M = float(np.max(corr))
        t = T * (corr - M)
        t = t - t.max()
        exp_t = np.exp(t)
        w = exp_t / float(exp_t.sum())
        lags = np.arange(2 * n - 1) - (n - 1)
        grad = np.ones(n)
        thresh = float(w.max()) * 1e-10
        for k in range(len(w)):
            wk = w[k]
            if wk < thresh:
                continue
            lag = int(lags[k])
            ms = max(0, -lag)
            me = min(n - 1, n - 1 - lag)
            if ms <= me:
                grad[ms:me + 1] -= wk * h[ms + lag:me + 1 + lag]
            ms2 = max(0, lag)
            me2 = min(n - 1, n - 1 + lag)
            if ms2 <= me2:
                grad[ms2:me2 + 1] -= wk * h[ms2 - lag:me2 + 1 - lag]
        h = h - alpha * grad
        h = make_feasible(h, n)
    return h
