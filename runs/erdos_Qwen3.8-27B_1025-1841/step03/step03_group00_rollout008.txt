        # Alternating min-max
        if rem() > 40:
            h_alt = alternating_minmax(h_cur, n_try, n_iters=40, topk_frac=0.12, max_inner=35)
            c_alt = compute_c5(h_alt)
            if c_alt < best_c5:
                best_c5, best_h = c_alt, h_alt.copy()
                h_cur = h_alt.copy()

        # L-BFGS-B with multiple temperatures
        for T in [80, 300, 1500]:
            if rem() < 20:
                break
            h_opt = lbgfb_smooth(h_cur, n_try, T, mu=800.0, max_iter=250)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h_cur = h_opt.copy()

    # Phase 2: Higher resolution
    for n2 in [1280, 1920]:
        if rem() < 100:
            break
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

        for T in [100, 500, 2000]:
            if rem() < 30:
                break
            h_opt = lbgfb_smooth(h2, n2, T, mu=400.0, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h2 = h_opt.copy()

        if rem() > 40:
            h_alt = alternating_minmax(h2, n2, n_iters=30, topk_frac=0.10, max_inner=30)
            c_alt = compute_c5(h_alt)
            if c_alt < best_c5:
                best_c5, best_h = c_alt, h_alt.copy()

    # Phase 3: Binary local search at moderate n
    if rem() > 80:
        n_bin = min(len(best_h), 1024)
        h_b = upsample_h(best_h, n_bin) if n_bin < len(best_h) else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        target_ones = n_bin // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target_ones - n_ones]] = 1.0
        h_bs, c_bs = binary_local_search(h_bin, n_bin, rng, max_rounds=80, tries=5000)
        if c_bs < best_c5:
            best_c5, best_h = c_bs, h_bs.copy()

    # Phase 4: Multi-start from different patterns at best_n resolution
    n_ms = len(best_h)
    if rem() > 60:
        for restart in range(3):
            if rem() < 30:
                break
            h_r = upsample_h(best_h, n_ms) + 0.12 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            h_opt = lbgfb_smooth(h_r, n_ms, 500, mu=600, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()

    # Phase 5: SLSQP final polish
    if rem() > 20:
        n_sp = min(len(best_h), 1280)
        h_sp = upsample_h(best_h, n_sp) if n_sp < len(best_h) else best_h.copy()
        for T in [2000, 8000]:
            if rem() < 12:
                break
            h_opt = slsqp_polish(h_sp, n_sp, T, max_iter=80)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h_sp = h_opt.copy()

    # Phase 6: Coordinate descent on the best solution
    if rem() > 20:
        n_cd = len(best_h)
        h_cd, c_cd = coord_descent_polish(best_h, n_cd, max_rounds=6,
                                          step_list=[0.03, -0.03, 0.015, -0.015, 0.005, -0.005])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
