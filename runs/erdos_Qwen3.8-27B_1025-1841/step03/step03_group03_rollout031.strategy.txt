def random_fourier_projection(x):

def repair_integral(x, target=0.5):  # target = sum(x*dx) should be 1
  global_dx = 2.0 / len(x)
  current_int = sum(x) * global_dx
  adjustments = (target - current_int) * 2
  return x + adjustments / (len(x)**0.5)
