    # B[m] = (1 - h[j-m]) - h[j+m] (with out-of-range = 0)
    B = np.zeros(2*n-1)
    # For shift m in np.correlate convention:
    # c[m] = sum_i h[i] * (1-h[i-(m-(n-1))])
    # h[j] appears as h[i] when i=j: coefficient is (1-h[j-(m-(n-1))])
    # h[j] appears as h[i-(m-(n-1))] when i-(m-(n-1))=j: i=j+(m-(n-1)): coefficient is -h[j+(m-(n-1))]

    # c[m] = sum_i h[i] * (1-h[i-shift])
    # h[j] as h[i]: i=j, term is (1-h[j-shift]) if 0<=j-shift<n
    # h[j] as h[i-shift]: i-shift=j, i=j+shift, term is -h[j+shift] if 0<=j+shift<n
    
    for m in range(2*n-1):
        shift = m - (n-1)
        bm = 0.0
        # h[j] appears as h[i] with i=j
        j_minus_shift = j - shift
        if 0 <= j_minus_shift < n:
            bm += (1.0 - h[j_minus_shift])
        # h[j] appears as h[i-shift] with i-shift=j, i=j+shift
        j_plus_shift = j + shift
        if 0 <= j_plus_shift < n:
            bm -= h[j_plus_shift]
        B[m] = bm
    
    # Now c[m] + delta * B[m] is the new correlation when h[j] changes by delta
    # We want to minimize max_m (corr[m] + delta * B[m]) * dx subject to 0 <= h[j]+delta <= 1
    # Let t = h[j] + delta, so delta = t - h[j]
    # f(t) = max_m (corr[m] + (t - h[j]) * B[m])
    
    # f(t) is convex piecewise linear, minimize over [0, 1]
    # Use ternary search (works for convex functions)
    
    def f(t):
        return np.max(corr + (t - h[j]) * B)
    
    # Ternary search
    lo, hi = 0.0, 1.0
    for _ in range(50):  # 50 iterations gives precision ~ 1/3^50 which is more than enough
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        if f(m1) < f(m2):
            hi = m2
        else:
            lo = m1
    t_opt = (lo + hi) / 2.0
    return t_opt, f(t_opt)

    shifts = np.arange(2*n-1) - (n-1)  # shape (2n-1,)
    # j - shift, valid where 0 <= j-shift < n
    jm = j - shifts
    valid_m = (jm >= 0) & (jm < n)

    jp = j + shifts
