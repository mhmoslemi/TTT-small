def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)

    target = n // 2

    ones = ones - {k} | {j}
    
    # Update contrib for all rows affected by change in h[j] and h[k]
    # contrib[idx, s] = (1-h[idx-s])*[valid] - h[idx+s]*[valid]
    # h[j] changed: affects contrib[idx, s] where idx-s=j (s=idx-j) or idx+s=j (s=j-idx)
    # h[k] changed: affects contrib[idx, s] where idx-s=k (s=idx-k) or idx+s=k (s=k-idx)
    
    for idx in range(n):
        # Effect of h[j] change (0→1):
        # (1-h[idx-s]) term: if idx-s==j, h[j] was 0, now 1, so (1-0)→(1-1), change = -1
        s_j1 = idx - j  # s where idx - s = j
        if -(n-1) <= s_j1 <= n-1:
            s_idx = s_j1 + (n-1)
            # Was: (1-0)=1, now: (1-1)=0. Change: -1
            contrib[idx, s_idx] += -1  # wait, h[j] went 0→1
            # Actually: old (1-h[j]) = 1, new (1-h[j]) = 0. So contrib decreases by 1.
            # But we already updated h[j]=1, so let's just recompute
        # (h[idx+s]) term: if idx+s==j, h[j] was 0, now 1, so 0→1, change = +1
        s_j2 = j - idx  # s where idx + s = j
        if -(n-1) <= s_j2 <= n-1:
            s_idx = s_j2 + (n-1)
            # Old h[j]=0, new h[j]=1. contrib had -h[j] = 0, now -1. Change: -1.
            contrib[idx, s_idx] += -1
        
        # Effect of h[k] change (1→0):
        # (1-h[idx-s]) term: if idx-s==k, h[k] was 1, now 0, so (1-1)→(1-0), change = +1
        s_k1 = idx - k
        if -(n-1) <= s_k1 <= n-1:
            s_idx = s_k1 + (n-1)
            contrib[idx, s_idx] += 1  # (1-h[k]) went from 0 to 1
        
        # (h[idx+s]) term: if idx+s==k, h[k] was 1, now 0, so 1→0, change = -1
        s_k2 = k - idx
        if -(n-1) <= s_k2 <= n-1:
            s_idx = s_k2 + (n-1)
            contrib[idx, s_idx] += -1  # h[k] went from 1 to 0, so -h[k] went from -1 to 0
    
    return contrib, h, zeros, ones

import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def round_to_binary(h, n, rng):
    target = n // 2
    order = np.argsort(-h, kind='stable')
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    # Check for ties at boundary
    if target > 0 and target < n:
        boundary_val = h[order[target]]
        if boundary_val == h[order[target-1]]:
            tie_mask = (np.abs(h - boundary_val) < 1e-9) & (h_bin == 0)
            tie_idx = np.where(tie_mask)[0]
            if len(tie_idx) > target - np.sum(h_bin):
                # Need to randomly select some
                n_take = target - np.sum(h_bin)
                chosen = rng.choice(tie_idx, size=n_take, replace=False)
                h_bin[chosen] = 1.0
    return h_bin

def binary_local_search(h, n, max_sweeps=200, time_limit=None, start_time=None, rng=None):
    """Efficient binary local search using pair flips."""
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(0)
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_max = float(np.max(corr))
    
    S = np.arange(2*n-1) - (n-1)  # shifts
    K = np.arange(n)[:, None]
    idx1 = K - S[None, :]
    valid1 = (idx1 >= 0) & (idx1 < n)
    idx2 = K + S[None, :]
    valid2 = (idx2 >= 0) & (idx2 < n)
    idx1_safe = np.where(valid1, idx1, 0)
    idx2_safe = np.where(valid2, idx2, 0)
    
    def get_contrib(h):
        return (1.0 - h[idx1_safe]).astype(float) * valid1 - h[idx2_safe].astype(float) * valid2
    
    contrib = get_contrib(h)
    
    best_c5 = current_max * dx
    
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 30:
                break
        
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        # For each j in zeros, find best k
        best_score = current_max  # we want to minimize this
        best_j = -1
        best_k = -1
        
        # Vectorized: for all j's at once
        # base[j, s] = corr[s] + contrib[j, s]  → (n_zeros, 2n-1)
        # scores[j, k] = max_s(base[j, s] - contrib[k, s])  → (n_zeros, n_ones)
        # This is: (base[:, None, :] - contrib[ones][:, :, None]).max(axis=2)
        # But this is O(n_zeros * n_ones * (2n-1)) memory = O(n³) which might be too much
        
        # For n=640: 320 * 320 * 1279 ≈ 131M elements × 8 bytes ≈ 1 GB. Too much!
        # Need to process in batches.
        
        batch_size = 20  # process 20 j's at a time
        for batch_start in range(0, len(zeros), batch_size):
            batch = zeros[batch_start:batch_start+batch_size]
            base_batch = corr[None, :] + contrib[batch]  # (batch_size, 2n-1)
            # scores: (batch_size, n_ones)
            scores = (base_batch[:, None, :] - contrib[ones][None, :, :]).max(axis=2)
            
            batch_best = np.min(scores)
            if batch_best < best_score - 1e-12:
                best_score = float(batch_best)
                local_idx = np.argmin(scores)
                j_idx = local_idx // len(ones)
                k_idx = local_idx % len(ones)
                best_j = batch[j_idx]
                best_k = ones[k_idx]
        
        if best_j == -1:
            break  # No improvement found
        
        # Apply flip
        h[best_j] = 1.0
        h[best_k] = 0.0
        corr = corr + contrib[best_j] - contrib[best_k]
        contrib = get_contrib(h)
        current_max = best_score
        best_c5 = current_max * dx
    
    return h, best_c5
