    target = n / 2.0

        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def topk_softgrad(h, T, K=30):
    """Compute subgradient of top-K soft-max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    
    # Get top K indices
    top_idx = np.argpartition(corr, -K)[-K:]
    top_corr = corr[top_idx]
    
    # Soft-max over top K
    top_mc = np.max(top_corr)
    exps = np.exp(T * (top_corr - top_mc))
    tot = np.sum(exps)
    w = exps / tot
    
    # Compute gradient
    g = np.zeros(n)
    for i, wi in zip(top_idx, w):
        if wi < 1e-14:
            continue
        lag = int(i) - (n - 1)
        if lag >= 0:
            g[lag:] += wi * (1.0 - h[:n - lag])
            g[:n - lag] -= wi * h[lag:]
        else:
            s = -lag
            g[:n - s] += wi * (1.0 - h[s:])
            g[s:] -= wi * h[:n - s]
    return g * dx, mc * dx

    n = len(h)

    def obj(hv):
        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

        c = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(c)
        exps = np.exp(T * (c - mc))

        w = exps / tot

            else:
                s = -lag
                g[:n - s] += wi * (1.0 - hv[s:])
                g[s:] -= wi * hv[:n - s]
        return g * dx

    if rng is None:
        rng = np.random.RandomState(0)
    for trial in range(n_cand):
        bs = block_sizes[trial % len(block_sizes)]
        start = rng.randint(0, max(1, n - bs))
        h_try = bh.copy()
        h_try[start:start + bs] = 1.0 - h_try[start:start + bs]
        h_try = project_c(h_try, n)
        c_try = compute_c5(h_try)
        if c_try < bc - 1e-12:
            bc, bh = c_try, h_try
    return bh, bc

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = project_c(rng.rand(480), 480)
    n0 = len(h0)
    bc = compute_c5(h0)
    bh = h0.copy()
    print(f"I: n={n0} C5={bc:.6f}", flush=True)

    # Phase 1a: FISTA broad (high T, small K)
    if rem() > 220:
        h1, c1 = fista_k(bh, 4000, 60.0, 1.0, 3, 40, 0.12, 40)
        if c1 < bc:
            bc, bh = c1, h1
        print(f"F1a: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 1b: FISTA focused (low T, larger K)
    if rem() > 160:
        h2, c2 = fista_k(bh, 5000, 4.0, 0.02, 10, 55, 0.03, 20)
        if c2 < bc:
            bc, bh = c2, h2
        print(f"F1b: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 2: SLSQP with analytical gradient, decreasing T
    if n0 <= 1024 and rem() > 100:
        for TV in [15.0, 6.0, 3.0, 1.5, 1.0]:
            if rem() < 25:
                break
            hx = slsqp_g(bh, TV, 500)
            cx = compute_c5(hx)
            if cx < bc:
                bc, bh = cx, hx
        print(f"SL: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 3: Local flip search
    if rem() > 50:
        bh, bc = local_flip_search(bh, 300, (4, 8, 16, 24), rng)
        print(f"FL: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 4: Low-resolution optimization
    if rem() > 180:
        n_low = max(96, n0 // 4)
        idx = np.linspace(0, n0 - 1, n_low).astype(int)
        hd = project_c(bh[idx].astype(float), n_low)
        cd = compute_c5(hd)
        hdl, cd2 = fista_k(hd, 8000, 30.0, 0.3, 3, 25, 0.14, 30)
        if cd2 < cd:
            hd, cd = hdl, cd2
        if n_low <= 300 and rem() > 40:
            hd = slsqp_g(hd, 2.0, 700)
            cd = compute_c5(hd)
        hd, cd = local_flip_search(hd, 200, (2, 4, 8), rng)
        hu = upsample(hd, n0)
        cu = compute_c5(hu)
        if cu < bc - 1e-6:
            bc, bh = cu, hu
        elif cu - bc < 0.002 and rem() > 50:
            hu2, cu2 = fista_k(hu, 2000, 2.0, 0.02, 10, 40, 0.02, 30)
            if cu2 < bc:
                bc, bh = cu2, hu2
        print(f"LR: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 5: Diverse starts
    starts = [tm_seq(n0), project_c(rng.rand(n0), n0)]
    for p in [3, 5, 7, 9, 11, 14]:
        pat = np.zeros(p)
        pat[:(p + 1) // 2] = 1.0
        starts.append(project_c(np.tile(pat, n0 // p + 1)[:n0], n0))
    al = np.zeros(n0)
    al[::2] = 1.0
    starts.append(project_c(al, n0))
    for hs in starts:
        if rem() < 35:
            break
        hr, cr = fista_k(hs, 2500, 30.0, 0.3, 5, 30, 0.1, 50)
        if cr < bc:
            bc, bh = cr, hr
            print(f"  DS: C5={bc:.6f}", flush=True)
    print(f"DA: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 6: Upsample for final push
    for nn in [n0 * 2, min(n0 * 3, 1920)]:
        if rem() < 80:
            break
        hu = upsample(bh, nn)
        cu = compute_c5(hu)
        if rem() > 50:
            hg, cg = fista_k(hu, 3000, 12.0, 0.08, 5, 30, 0.08, 40)
            if cg < cu:
                hu, cu = hg, cg
        if rem() > 35:
            hn, cn = fista_k(hu, 2000, 1.0, 0.01, 10, 40, 0.012, 25)
            if cn < cu:
                hu, cu = hn, cn
        if nn <= 512 and rem() > 30:
            hu = slsqp_g(hu, 1.5, 350)
            cu = compute_c5(hu)
        cf = compute_c5(hu)
        if cf < bc:
            bc, bh = cf, hu.copy()
        print(f"  U{nn}: C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)

    # Phase 7: Final SLSQP + flip + ultra-fine FISTA
    np_ = len(bh)
    if np_ <= 512 and rem() > 25:
        bh = slsqp_g(bh, 1.0, 500)
        bc = compute_c5(bh)
    if rem() > 20:
        bh, bc = local_flip_search(bh, 200, (4, 8, 16), rng)
    if rem() > 12:
        hf, cf = fista_k(bh, 1500, 1.5, 0.005, 15, 50, 0.006, 20)
        if cf < bc:
            bc, bh = cf, hf

    bh = project_c(bh, len(bh))
    final_c = compute_c5(bh)
    if final_c < bc:
        bc = final_c
    print(f"DONE: n={len(bh)} C5={bc:.6f} t={time.time()-t0:.0f}", flush=True)
    return bh, bc, len(bh)
