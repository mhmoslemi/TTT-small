import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_smooth(h0, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def binary_sa_local(h_bin, n, rng, max_rounds=150, T_init=0.015, T_final=1e-7, trials_per_round=3000):
    h = h_bin.copy()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        frac = float(rnd + 1) / max_rounds
        T = T_init * (T_final / T_init) ** frac
        improved = 0
        for _ in range(trials_per_round):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            if oi == oj:
                continue
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            delta = c - cur
            if delta < -1e-14:
                cur = c
                ones[ii], zeros[jj] = oj, oi
                improved += 1
                if c < best:
                    best = c
                    best_h = h.copy()
            elif rng.rand() < np.exp(-delta / max(T, 1e-12)):
                cur = c
                ones[ii], zeros[jj] = oj, oi
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if improved == 0 and rnd > max_rounds // 3:
            break
    return best_h, best

def fractional_polish(h0, n, topk=40, max_iter=100, T=2000):
    """Allow only topk fractional variables to move, rest pinned to 0/1"""
    h = h0.copy()
    h_bin = (h > 0.5).astype(float)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    active_lags = np.argsort(corr)[-topk:]
    # Find h values that contribute to active lags
    n_offset = n - 1
    candidates = set()
    for lag in active_lags:
        offset = lag - n_offset
        lo = max(0, -offset)
        hi = min(n, n - offset)
        for i in range(lo, hi):
            candidates.add(i)
            candidates.add(i + offset)
    candidates = list(candidates)
    if len(candidates) > 2*n//3:
        candidates = list(range(n))
    fixed_mask = np.array([i not in set(candidates) for i in range(n)])
    h[fixed_mask] = h_bin[fixed_mask]
    free_idx = np.where(~fixed_mask)[0]
    
    def obj_fr(x):
        xv = h.copy()
        xv[free_idx] = np.clip(x, 0.0, 1.0)
        corr_x = np.correlate(xv, 1.0 - xv, mode='full')
        return soft_max_obj(corr_x, T) * (2.0/n)
    
    x0 = h[free_idx].copy()
    res = minimize(obj_fr, x0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * len(free_idx),
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    h_final = h.copy()
    h_final[free_idx] = np.clip(res.x, 0.0, 1.0)
    h_final = make_feasible(h_final, n)
    return h_final

def thue_morse_init(n):
    h = np.zeros(n)
    for i in range(n):
        parity = bin(i).count('1') % 2
        h[i] = 1.0 if parity == 0 else 0.0
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1.0
    return h

def quadratic_init(n):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * i % (2*n)) < n else 0.0

        h[idx[:n_ones - target]] = 0.0

                h_ai_bin[idx[:n_par // 2 - n_o]] = 1.0
            h_ai_sa, c_ai_sa = binary_sa_local(h_ai_bin, n_par, rng, max_rounds=100, T_init=0.015, T_final=1e-7, trials_per_round=2000)
            if c_ai_sa < best_c5:
                best_c5, best_h = c_ai_sa, h_ai_sa.copy()
    
    # Phase 5: Final intensive polish on best solution
    if rem() > 40:
        h_fin = (best_h > 0.5).astype(float)
        n_f = len(h_fin)
        t_ones = n_f // 2
        n_o = int(h_fin.sum())
        if n_o > t_ones:
            idx = np.where(h_fin == 1)[0]
            h_fin[idx[:n_o - t_ones]] = 0.0
        elif n_o < t_ones:
            idx = np.where(h_fin == 0)[0]
            h_fin[idx[:t_ones - n_o]] = 1.0
        h_final, c_final = binary_sa_local(h_fin, n_f, rng, max_rounds=120, T_init=0.01, T_final=1e-8, trials_per_round=3000)
        if c_final < best_c5:
            best_c5, best_h = c_final, h_final.copy()
    
    # Final fractional polish
    if rem() > 15:
        h_fps = fractional_polish(best_h, len(best_h), topk=30, T=8000, max_iter=100)
        c_fps = compute_c5(h_fps)
        if c_fps < best_c5:
            best_c5, best_h = c_fps, h_fps.copy()
    
    # Ensure feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
