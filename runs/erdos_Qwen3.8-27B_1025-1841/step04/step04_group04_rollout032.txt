        h = np.clip(h, 0.0, 1.0)
    return h

def g_obj(x):
    T = _g_T
    dx = _g_dx
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def g_con_f(x):
    return np.sum(x) - _g_n / 2.0

def g_con_j(x):
    return np.ones(_g_n)

def periodic_obj(params, p, n_total):
    h_period = np.clip(params, 0, 1)
    h = np.tile(h_period, n_total // p)
    rem = n_total - (n_total // p) * p
    if rem > 0:
        h = np.concatenate([h, h_period[:rem]])
    h = make_feasible(h, n_total)
    return compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_T, _g_dx
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)
    
    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    best_n = n
    
    # Phase 1: Quick periodic search at initial resolution
    for p in [12, 16, 20, 24, 32, 40, 48, 64, 80, 96]:
        if time.time() - start_time > budget_s * 0.2:
            break
        if n % p != 0:
            continue
        try:
            res = differential_evolution(
                periodic_obj, [(0.0, 1.0)] * p, args=(p, n),
                seed=seed + p, maxiter=100, tol=1e-10,
                popsize=10, mutation=(0.4, 1.0), recombination=0.7,
                polish=False
            )
            h_p = np.clip(res.x, 0, 1)
            h_full = np.tile(h_p, n // p)
            h_full = make_feasible(h_full, n)
            c5_p = compute_c5(h_full)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_full
                best_n = n
        except Exception:
            pass
    
    # Phase 2: SLSQP at current resolution
    _g_n = best_n
    _g_dx = 2.0 / best_n
    bounds = [(0.0, 1.0)] * best_n
    cons = [{'type': 'eq', 'fun': g_con_f, 'jac': g_con_j}]
    
    for T_try in [80, 150, 300, 600, 1000]:
        if time.time() - start_time > budget_s * 0.55:
            break
        _g_T = float(T_try)
        x0 = best_h.copy()
        try:
            res = minimize(g_obj, x0, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15, 'maxls': 50})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, best_n)
        except Exception:
            pass
    
    # Phase 3: Coordinate descent
    for ss in [0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - start_time > budget_s * 0.8:
            break
        improved = True
        passes = 0
        while improved and passes < 5:
            if time.time() - start_time > budget_s * 0.85:
                break
            improved = False
            passes += 1
            for j in range(best_n):
                if time.time() - start_time > budget_s * 0.87:
                    break
