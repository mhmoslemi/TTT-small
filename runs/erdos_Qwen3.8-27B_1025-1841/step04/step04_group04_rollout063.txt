def corr_diff(h, j, delta):
    """Compute the change in all correlation values when h[j] changes by delta."""
    n = len(h)
    # c[k] for k = -(n-1), ..., n-1 (full correlation)
    # c[k] = sum_i h[i] * (1-h[i+k])
    # ∂c[k]/∂h[j] = (1-h[j+k]) - h[j-k] (with 0 for out-of-range)
    lags = np.arange(-(n-1), n)
    idx1 = j + lags  # j+k for each k
    idx2 = j - lags  # j-k for each k
    term1 = np.where((idx1 >= 0) & (idx1 < n), 1.0 - h[np.clip(idx1, 0, n-1)], 0.0)
    term2 = np.where((idx2 >= 0) & (idx2 < n), -h[np.clip(idx2, 0, n-1)], 0.0)
    return delta * (term1 + term2)
```

```python
def cd_fast(h, n, max_rounds, tleft_fn):
    dx = 2.0 / n
    best_c5 = c5(h)
    best_h = h.copy()
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    bc = np.max(corr) * dx
    
    lags = np.arange(-(n-1), n)
    
    for rnd in range(max_rounds):
        if tleft_fn() < 5:
            break
        imp = False
        for j in range(n):
            if tleft_fn() < 2:
                break
            ov = best_h[j]
            bval, bcv = ov, bc
            # Precompute the gradient terms for this j
            idx1 = j + lags
            idx2 = j - lags
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            g1 = np.where(v1, 1.0 - best_h[np.clip(idx1, 0, n-1)], 0.0)
            g2 = np.where(v2, -best_h[np.clip(idx2, 0, n-1)], 0.0)
            g = g1 + g2  # gradient of corr w.r.t. h[j] for each lag
            
            cands = [0.0, 1.0, 0.5]
            for d in [-0.1,-0.05,-0.02,-0.01,0.01,0.02,0.05,0.1]:
                v = ov + d
                if 0 <= v <= 1:
                    cands.append(v)
            
            for cd_v in cands:
                if abs(cd_v - ov) < 1e-10:
                    continue
                new_corr = corr + (cd_v - ov) * g
                cc = np.max(new_corr) * dx
                if cc < bcv - 1e-15:
                    bcv = cc
                    bval = cd_v
            
            if abs(bval - ov) > 1e-10:
                corr += (bval - ov) * g
                corr = np.clip(corr, -10, 10*n)  # safety
                best_h[j] = bval
                bc = bcv
                best_c5 = bcv
                imp = True
        
        # Recompute from scratch every 10 rounds to avoid drift
        if (rnd + 1) % 10 == 0:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            bc = np.max(corr) * dx
            best_c5 = bc
        
        if not imp:
            break
    
    # Final feasibility fix
    best_h = mkf(best_h, n)
    best_c5 = c5(best_h)
    return best_h, best_c5

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        # Count 1s in binary representation of i
        cnt = 0
        x = i
        while x:
            cnt += x & 1
            x >>= 1
        h[i] = cnt % 2
    return h
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 960
_G_T = 100.0

def _obj(x):
    dx = 2.0 / _G_N
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    sh = np.clip(_G_T * (corr - mc), -100, 500)
    return mc + np.log(np.sum(np.exp(sh))) / _G_T

def _cf(x):
    return np.sum(x) - _G_N / 2.0

def _cj(x):
    return np.ones(_G_N)

def c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def mkf(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    tgt = n / 2.0
    for _ in range(100):
        d = tgt - h.sum()
        if abs(d) < 1e-14:
            break
        msk = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = int(msk.sum())
        if cnt == 0:
            break
        h[msk] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def up(h, nn):
    on = len(h)
    if nn == on:
        return mkf(h.copy(), nn)
    ox = np.linspace(0, 2, on, endpoint=False)
    nx = np.linspace(0, 2, nn, endpoint=False)
    if nn > on:
        hl = np.interp(nx, ox, h)
        hl = mkf(hl, nn)
        r = nn // on
        if r >= 1:
            hs = np.repeat(h, r)[:nn]
            hs = mkf(hs, nn)
        else:
            hs = hl
        return hl if c5(hl) <= c5(hs) else hs
    hn = np.interp(nx, ox, h)
    return mkf(hn, nn)

def slsq(h, n, T, mi):
    global _G_N, _G_T
    _G_N, _G_T = n, T
    try:
        r = minimize(_obj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _cf, 'jac': _cj}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        return mkf(r.x, n)
    except Exception:
        return mkf(h.copy(), n)

def tm(n):
    h = np.zeros(1)
    while len(h) < n:
        h = np.concatenate([h, 1.0 - h])
    return h[:n].copy()

def cd_fast(h, n, max_rounds, tleft_fn):

    best_h = h.copy()

        imp = False
        for j in range(n):
            if tleft_fn() < 2:
                break
            ov = best_h[j]
            idx1 = j + lags
            idx2 = j - lags
            v1 = (idx1 >= 0) & (idx1 < n)
            v2 = (idx2 >= 0) & (idx2 < n)
            g = np.zeros(2 * n - 1)
            g[v1] += 1.0 - best_h[idx1[v1]]
            g[v2] -= best_h[idx2[v2]]
            
            bval = ov
            bcv = bc
            cands = [0.0, 1.0]
            for d in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
                v = ov + d

            if 0.0 < ov < 1.0:
                cands.append(0.5)

                nc = corr + (cv - ov) * g
                mc = np.max(nc)
                if mc < bcv - 1e-12:

                    bval = cv
