import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def flip_change(h, j, n):
    """Compute change in corr vector when flipping h[j] (binary)."""
    delta = 1.0 - 2.0 * h[j]
    change = np.zeros(2 * n - 1)
    change[n-1-j : 2*n-1-j] += delta * (1.0 - h)
    change[j : j+n] += -delta * h[::-1]
    change[n-1] -= 1.0
    return change

def fast_sa(h, n, max_iters, T0, T1, start_time, budget_s, rng):
    """Simulated annealing with fast flip evaluation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    curr_max = float(np.max(corr))
    best_c5 = curr_max * dx
    best_h = h.copy()
    
    n_positions = n  # all positions are candidates
    positions = np.arange(n)
    
    for it in range(max_iters):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        
        if it % 1000 == 0:
            frac = float(it) / max_iters
            T = T0 * (T1 / T0) ** frac
        else:
            T = T0 * (T1 / T0) ** (float(it) / max_iters)
        
        # Pick a random position to flip
        j = positions[rng.randint(n)]
        ch = flip_change(h, j, n)
        new_corr = corr + ch
        new_max = float(np.max(new_corr))
        
        diff = new_max - curr_max
        if diff <= 0 or (T > 1e-15 and rng.random() < np.exp(-diff / (T * dx))):
            # Accept
            h[j] = 1.0 - h[j]
            corr = new_corr
            curr_max = new_max
            if curr_max * dx < best_c5:
                best_c5 = curr_max * dx
                best_h = h.copy()
        # else: reject (h unchanged, corr unchanged)
    
    return best_h, best_c5

def fast_greedy(h, n, max_rounds, start_time, budget_s):
    """Greedy local search: accept any improving flip."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    curr_max = float(np.max(corr))
    best_c5 = curr_max * dx
    best_h = h.copy()

                    best_c5 = curr_max * dx
                    best_h = h.copy()
        if not made_change:

    return best_h, best_c5

def slsqp_objective(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    # Smoothed max
    shifted = corr - mc
    logsumexp = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return logsumexp * dx

def _slsqp_run(h, n, T, maxiter, ftol):
    from scipy.optimize import minimize
    
    def obj(x):
        return slsqp_objective(x, T)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s):
    """Coordinate descent for continuous h."""
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.1), min(1.0, old + 0.1)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5
