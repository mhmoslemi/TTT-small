- For each position i with h[i]=1, compute which shifts s are "critical" (c(s) is near the max)
- For each such critical shift, identify if position i is contributing (i.e., h[i]=1 and h[i+s]=0)
- If so, try swapping i with a nearby 0-position j that doesn't create new critical contributions
- This is a guided local search

def subgradient_optimize(h, n, start_time, time_budget, top_m=5):
    """Subgradient method for the real-valued problem."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05
    
    while time.time() - start_time < time_budget:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr) * dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find top shifts
        top_idx = np.argsort(corr)[-top_m:]
        
        # Compute subgradient (average gradient of top correlations)
        grad = np.zeros(n)
        for k in top_idx:
            s = int(k) - (n - 1)
            # c_k = sum_i h[i] * (1 - h[i+s]) for valid i
            # dc_k/dh[j] = (1-h[j+s]) if j valid and j+s valid (as first factor)
            #             + (-h[j-s]) if j-s valid and j valid (as second factor, from 1-h[j])
            for j in range(n):
                if 0 <= j + s < n:
                    grad[j] += (1.0 - h[j + s])
                if 0 <= j - s < n:
                    grad[j] += (-h[j - s])
        
        grad /= len(top_idx)
        
        # Update

        h_new = make_feasible(h_new, n)
        new_c5 = compute_c5(h_new)

            h = h_new
            step = min(step * 1.1, 0.5)
        else:
            step *= 0.7

            break
    
    return best_h, best_c5
