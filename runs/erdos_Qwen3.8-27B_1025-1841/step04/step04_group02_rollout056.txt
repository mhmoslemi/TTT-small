```python
def projected_subgrad(h, n, max_iters, T_init, T_final, rng):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        corr = corr_raw * dx
        curr_max = corr.max()
        
        # Find active constraints (top few)
        topk = 10
        top_indices = np.argsort(corr)[-topk:]
        weights = np.exp(corr[top_indices] / (0.01 + 0.001))  # soft weights
        weights /= weights.sum()
        
        # Compute subgradient
        grad = np.zeros(n)
        for wi, k in zip(weights, top_indices):
            m = k - (n - 1)
            # ∂c_k/∂h[j] = (1-h[j+m]) - h[j-m] for valid indices
            for j in range(n):
                term = 0.0
                if 0 <= j + m < n:
                    term += 1 - h[j + m]
                if 0 <= j - m < n:
                    term -= h[j - m]
                grad[j] += wi * term
        
        # Line search
        step = 2.0 / (max_iters ** 0.5 + it * 0.001)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0, 1)
        # Project onto sum constraint
        target = n / 2.0
        diff = target - h_new.sum()
        if abs(diff) > 1e-10:
            # Adjust: add diff/n to all, then clip and re-iterate
            h_new += diff / n
            h_new = np.clip(h_new, 0, 1)
            # Simple fix
            for _ in range(10):
                diff = target - h_new.sum()
                if abs(diff) < 1e-10:
                    break
                if diff > 0:
                    mask = h_new < 1
                else:
                    mask = h_new > 0
                if mask.sum() == 0:
                    break
                h_new[mask] += diff / mask.sum()
                h_new = np.clip(h_new, 0, 1)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    if s > target:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        b[ones[order[:s - target]]] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        b[zeros[order[:target - s]]] = 1
    return b

def fast_sa_single(b_init, n, max_iters, T0, T1, rng):
    b = to_binary(b_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(b, 1.0 - b, mode='full') * dx
    curr_max = float(corr.max())
    best_max = curr_max
    best_b = b.copy()
    m_arr = np.arange(2 * n - 1) - (n - 1)
    for it in range(max_iters):
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(n)
        while b[i] != 1.0:
            i = rng.randint(n)
        j = rng.randint(n)
        while b[j] != 0.0:
            j = rng.randint(n)
        if i == j:
            continue
        delta = np.zeros(2 * n - 1)
        idx = i + m_arr
        v = (idx >= 0) & (idx < n)
        delta[v] += b[idx[v]]
        idx = i - m_arr
        v = (idx >= 0) & (idx < n)
        delta[v] += b[idx[v]]
        idx = j + m_arr
        v = (idx >= 0) & (idx < n)
        delta[v] -= b[idx[v]]
        idx = j - m_arr
        v = (idx >= 0) & (idx < n)
        delta[v] -= b[idx[v]]
        new_corr = corr + delta * dx
        new_max = float(new_corr.max())
        if new_max <= curr_max or (T > 1e-15 and rng.random() < np.exp((curr_max - new_max) / T)):
            corr = new_corr
            curr_max = new_max
            b[i] = 0.0
            b[j] = 1.0
            if curr_max < best_max:
                best_max = curr_max
                best_b = b.copy()
    return best_b, best_max

def fast_sa_block(b_init, n, max_iters, T0, T1, rng, max_block=4):
    b = to_binary(b_init, n).copy()
    dx = 2.0 / n
    corr = np.correlate(b, 1.0 - b, mode='full') * dx
    curr_max = float(corr.max())
    best_max = curr_max
    best_b = b.copy()
    for it in range(max_iters):
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        bs = rng.randint(1, max_block + 1)
        if bs >= n:
            bs = 1
        a = rng.randint(0, n - bs)
        bpos = rng.randint(0, n - bs)
        if bpos < a + bs:
            bpos = (bpos + a + bs) % (n - bs)
            if bpos < a + bs:
                bpos = 0
        seg_a = b[a:a + bs].copy()
        seg_b = b[bpos:bpos + bs].copy()
        b[a:a + bs] = seg_b
        b[bpos:bpos + bs] = seg_a
        new_corr = np.correlate(b, 1.0 - b, mode='full') * dx
        new_max = float(new_corr.max())
        if new_max <= curr_max or (T > 1e-15 and rng.random() < np.exp((curr_max - new_max) / T)):
            corr = new_corr
            curr_max = new_max
            if curr_max < best_max:
                best_max = curr_max
                best_b = b.copy()
        else:
            b[a:a + bs] = seg_a
            b[bpos:bpos + bs] = seg_b
    return best_b, best_max

def subgradient_optimize(h, n, max_iters, start_time, budget_s, rng):
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    T = 80.0
    for it in range(max_iters):
        rem = budget_s - (time.time() - start_time)
        if rem < 5:
            break
        if it % 2000 == 0 and it > 0:
            T = max(10.0, T * 0.85)
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        curr_max = float(corr.max())
        topk = min(20, 2 * n - 1)
        top_idx = np.argsort(corr)[-topk:]
        w = np.exp(T * (corr[top_idx] - curr_max))
        w /= w.sum()
        grad = np.zeros(n)
        for wi, k in zip(w, top_idx):
            m = int(k) - (n - 1)
            j = np.arange(n)
            jp = j + m
            jm = j - m
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            if np.any(vp):
                grad[vp] += wi * (1.0 - h[jp[vp]])
            if np.any(vm):
                grad[vm] -= wi * h[jm[vm]]
        step = 0.5 / (np.sqrt(float(it + 1)) + 1.0)
        h_new = h - step * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        target = n / 2.0
        for _ in range(20):
            diff = target - h_new.sum()
            if abs(diff) < 1e-12:
                break
            if diff > 0:
                mask = h_new < 1.0
            else:
                mask = h_new > 0.0
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def slsqp_smooth(x):
    n = len(x)
    dx = 2.0 / n
    T = 150.0
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = corr.max()
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T)

def slsqp_run(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = corr.max()
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T)
    def obj_con(x):
        return np.sum(x) - n / 2.0
    def obj_con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': obj_con, 'jac': obj_con_jac}]
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
