import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def sa_optimize(h_init, n, rng, T_start=15.0, T_end=0.05, n_steps=500000, time_limit=None):
    h = h_init.astype(float).copy()
    # Ensure binary
    h = (h > 0.5).astype(float)
    # Fix count
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        h[ones_idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        h[zeros_idx[:target - n_ones]] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    mLargest = np.zeros(2 * n - 1)
    
    step = 0
    while step < n_steps:
        if time_limit is not None and time_limit() < 5.0:
            break
        step += 1
        frac = step / n_steps
        T = T_start * (T_end / T_start) ** frac
        if T < T_end * 1.2:
            T = T_end
        i = rng.randint(n)
        j = rng.randint(n)
        if h[i] == h[j]:
            continue
        di = 1.0 - 2.0 * h[i]  # +1 for 0->1, -1 for 1->0
        dj = 1.0 - 2.0 * h[j]
        psi = (n - 1) - i
        mxi = 2 * i if 2 * i < n else n - 1
        psj = (n - 1) - j
        mxj = 2 * j if 2 * j < n else n - 1
        mLargest[:] = 0.0
        mLargest[psi:psi + mxi + 1] += di * (1.0 - h[:mxi + 1])
        mLargest[psj:psj + mxj + 1] += dj * (1.0 - h[:mxj + 1])
        new_max = float(np.max(corr + mLargest))
        if new_max <= current_max + 1e-12:
            corr += mLargest
            current_max = new_max
            h[i] += di
            h[j] += dj
            if current_max < best_max - 1e-12:
                best_max = current_max
                best_h = h.copy()
        elif T > 1e-6 and rng.random() < np.exp(-(new_max - current_max) / T):
            corr += mLargest
            current_max = new_max
            h[i] += di
            h[j] += dj
    
    return best_h, best_max * (2.0 / n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def make_binary(h, n):
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[:target - n_ones]] = 1.0
    return h

def sa_optimize(h_init, n, rng, T_start, T_end, n_steps, time_limit):
    h = make_binary(h_init, n).astype(np.float64)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    current_max = float(np.max(corr))
    best_h = h.copy()
    best_max = current_max
    work = np.zeros(2 * n - 1, dtype=np.float64)
    inv_n = 2.0 / n

    step = 0
    while step < n_steps:
        if time_limit is not None and time_limit() < 3.0:
            break
        step += 1
        frac = step / n_steps if n_steps > 1 else 1.0
        T = T_start * (T_end / T_start) ** frac
        if T < T_end:
            T = T_end

        if rng.random() < 0.7:
            i = rng.randint(n)
            j = rng.randint(n)
            if h[i] == h[j]:
                continue
            di = 1.0 - 2.0 * h[i]
            dj = 1.0 - 2.0 * h[j]
            psi = (n - 1) - i
            mxi = 2 * i if 2 * i < n else n - 1
            psj = (n - 1) - j
            mxj = 2 * j if 2 * j < n else n - 1
            work[:] = 0.0
            work[psi:psi + mxi + 1] += di * (1.0 - h[:mxi + 1])
            work[psj:psj + mxj + 1] += dj * (1.0 - h[:mxj + 1])
            new_max = float(np.max(corr + work))
            if new_max < current_max - 1e-12:
                corr += work
                current_max = new_max
                h[i] += di
                h[j] += dj
                if current_max < best_max - 1e-12:
                    best_max = current_max
                    best_h = h.copy()
            elif T > 1e-8 and rng.random() < np.exp(-min((new_max - current_max) / T, 50.0)):
                corr += work
                current_max = new_max
                h[i] += di
                h[j] += dj
        else:
            i = rng.randint(n)
            j = rng.randint(n)
            k = rng.randint(n)
            if h[i] == h[j] or h[i] == h[k] or j == k:
                continue
            if h[i] == 0.0 or h[j] == 0.0:
                if h[i] == 1.0 and h[j] == 1.0 and h[k] == 0.0:
                    i2, j2 = i, j
                    di = dj = dk = -1.0 if h[k] == 0.0 else 1.0
                continue
            if h[i] == 1.0 and h[j] == 1.0 and h[k] == 0.0:
                di = dj = -1.0
                dk = 1.0
                ps1 = (n - 1) - i
                mx1 = 2 * i if 2 * i < n else n - 1
                ps2 = (n - 1) - j
                mx2 = 2 * j if 2 * j < n else n - 1
                ps3 = (n - 1) - k
                mx3 = 2 * k if 2 * k < n else n - 1
                work[:] = 0.0
                work[ps1:ps1 + mx1 + 1] += di * (1.0 - h[:mx1 + 1])
                work[ps2:ps2 + mx2 + 1] += dj * (1.0 - h[:mx2 + 1])
                work[ps3:ps3 + mx3 + 1] += dk * (1.0 - h[:mx3 + 1])
                new_max = float(np.max(corr + work))
                if new_max < current_max - 1e-12:
                    corr += work
                    current_max = new_max
                    h[i] += di
                    h[j] += dj
                    h[k] += dk
                    if current_max < best_max - 1e-12:
                        best_max = current_max
                        best_h = h.copy()
                elif T > 1e-8 and rng.random() < np.exp(-min((new_max - current_max) / (T * 2), 50.0)):
                    corr += work
                    current_max = new_max
                    h[i] += di
                    h[j] += dj
                    h[k] += dk
                    # Rebalance
                    s = np.sum(h)
                    if s != n / 2.0:
                        h = make_binary(h, n)
                        corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
                        current_max = float(np.max(corr))

    return best_h, best_max * inv_n

def lbgfb_smooth(h_start, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)
