import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max(corr, T):
    mc = np.max(corr)
    s = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(s))) / T

        return soft_max(corr, T) * dx + mu * (x.sum() - n * 0.5) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max(corr, T):
    mc = np.max(corr)
    s = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(s))) / T

def lbgfb_opt(h0, n, T, mu, max_iter=400):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max(corr, T) * dx + mu * (x.sum() - n * 0.5) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def to_binary(h, n):
    hb = (h > 0.5).astype(float)
    t = n // 2
    cnt = int(hb.sum())
    if cnt > t:
        idx = np.where(hb == 1)[0]
        order = np.argsort(h[idx])
        hb[idx[order[:cnt - t]]] = 0.0
    elif cnt < t:
        idx = np.where(hb == 0)[0]
        order = np.argsort(-h[idx])
        hb[idx[order[:t - cnt]]] = 1.0
    return hb

def binary_sa(h_bin, n, rng, rounds, tries, T_init=0.0, T_end=0.0):
    h = h_bin.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    do_sa = T_init > 0.01
    for r in range(rounds):
        T = T_init * (T_end / T_init) ** (r / max(1, rounds)) if do_sa else 0.0
        improved = False
        nt = min(tries, max(1, len(ones)) * max(1, len(zeros)))
        for _ in range(nt):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            d = c - cur
            accept = False
            if d < -1e-14:
                accept = True
            elif do_sa:
                if rng.rand() < np.exp(-d / max(T, 1e-10)):
                    accept = True
            if accept:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved and not do_sa:
            break
    return best_h, best_c

def alternating_topk(h0, n, n_iters=40, topk_frac=0.12, max_inner=40):
    dx = 2.0 / n
    h = h0.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        k = max(5, int(topk_frac * len(corr)))
        idx = np.argpartition(corr, -k)[-k:]
        active = np.zeros(len(corr))
        active[idx] = 1.0
        active /= active.sum()
        mu = 800.0
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(active, c)) * dx + mu * (x.sum() - n * 0.5) ** 2
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = make_feasible(res.x, n)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            h = h_new
    return h
