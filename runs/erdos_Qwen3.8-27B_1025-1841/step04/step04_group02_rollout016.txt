_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    if T < 1e-6:
        return mc * dx
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    return np.sum(x) - _SL_N / 2.0

    return np.ones(_SL_N)

    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]

        res = minimize(obj_slsqp, h0.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(res.x, n)
    except Exception:
        return h0

                b = b_old
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
        elif move < 0.75:

            L = rng.randint(2, min(30, n // 10))

def coord_descent(h, n, deadline, steps=4):

    best_c5 = compute_c5(h)
    for s in range(steps):
        if time.time() > deadline:
            break

        improved = True
        passes = 0

            passes += 1

                break

                cands.add(0.0)

                if best_v != old:
                    h[j] = best_v
                    h = make_feasible(h, n)
                    best_c5 = best_c
                    improved = True
    return h, best_c5

    h_new = np.interp(x_new, x_old, h_src)
    return make_feasible(h_new, n_new)

            s1 = rng.randint(0, n - L)
            s2 = rng.randint(0, n - L)
            if abs(s1 - s2) < L:
                b = b_old
                continue
            tmp = b[s1:s1 + L].copy()
            b[s1:s1 + L] = b[s2:s2 + L]
            b[s2:s2 + L] = tmp
        else:
            ns = rng.randint(2, 5)
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if len(ones) < ns or len(zeros) < ns:
                b = b_old
                continue
            io = rng.choice(ones, ns, replace=False)
            iz = rng.choice(zeros, ns, replace=False)
            b[io] = 0
            b[iz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b = b_old
    return best_b, best_c5

def coord_descent(h, n, deadline, n_steps=4):
    best_c5 = compute_c5(h)
    for s in range(n_steps):
        if time.time() > deadline:
            break
        delta = 0.06 / (s + 1)
        improved = True
        passes = 0
        while improved and passes < 40:
            passes += 1
            if time.time() > deadline:
                break
            improved = False
            for j in range(n):
                if time.time() > deadline:
                    break
                old = h[j]
                best_c = best_c5
                best_v = old
                cands = [0.0, 1.0, max(0.0, old - delta), min(1.0, old + delta)]
                for cand in cands:
                    if abs(cand - old) < 1e-12:
                        continue
                    trial = h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_v = cand
                if best_v != old:
                    h[j] = best_v
                    h = make_feasible(h, n)
                    best_c5 = best_c
                    improved = True
    return h, best_c5

def upsample_h(h_src, n_new):
    n_old = len(h_src)
    if n_new == n_old:
        return make_feasible(h_src.copy(), n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_src)
    return make_feasible(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SA diverse at n0
    if time.time() < deadline - 200:
        dl = min(t0 + 220, deadline - 250)
        b1, c1 = sa_diverse(h_init, n0, 0.012, 0.00004, dl, rng)
        if c1 < best_c5:
            best_c5 = c1
            best_h = b1

    # Phase 2: SLSQP multi-T
    for T in [40.0, 150.0, 400.0]:
        if time.time() > deadline - 200:
            break
        x = slsqp_opt(best_h, n0, T, 180, 1e-13)
        c = compute_c5(x)
        if c < best_c5:
            best_c5 = c
            best_h = x

    # Phase 3: SA from continuous + SLSQP
    if time.time() < deadline - 150:
        dl = min(time.time() + 160, deadline - 100)
        b2, c2 = sa_diverse(best_h, n0, 0.006, 0.00002, dl, rng)
        if c2 < best_c5:
            best_c5 = c2
            best_h = b2
        if time.time() < deadline - 60:
            x = slsqp_opt(best_h, n0, 300.0, 150, 1e-14)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x

    # Phase 4: Higher resolution 2*n0
    if time.time() < deadline - 350:
        n1 = 2 * n0
        h_up = upsample_h(best_h, n1)
        dl = min(time.time() + 130, deadline - 220)
        b3, c3 = sa_diverse(h_up, n1, 0.008, 0.00003, dl, rng)
        if c3 < best_c5:
            best_c5 = c3
            best_h = b3
        if time.time() < deadline - 80:
            for T in [60.0, 250.0]:
                if time.time() > deadline - 40:
                    break
                x = slsqp_opt(best_h, n1, T, 120, 1e-13)
                c = compute_c5(x)
                if c < best_c5:
                    best_c5 = c
                    best_h = x

    # Phase 5: Second resolution attempt (1.5*n0) if n0 allowed
    if time.time() < deadline - 200 and n0 <= 960:
        n2 = n0 + n0 // 2
        h_up2 = upsample_h(best_h, n2)
        dl = min(time.time() + 80, deadline - 100)
        b4, c4 = sa_diverse(h_up2, n2, 0.007, 0.00003, dl, rng)
        if c4 < best_c5:
            best_c5 = c4
            best_h = b4
        if time.time() < deadline - 50:
            x = slsqp_opt(best_h, n2, 200.0, 100, 1e-13)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x

    # Phase 6: Coordinate descent on best
    if time.time() < deadline - 40:
        nb = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), nb, min(time.time() + (deadline - time.time()) * 0.25, deadline - 20), n_steps=4)

    # Phase 7: Final SLSQP with high T
    if time.time() < deadline - 20:
        nb = len(best_h)
        x = slsqp_opt(best_h, nb, 600.0, 150, 1e-14)
        c = compute_c5(x)
        if c < best_c5:
            best_c5 = c
            best_h = x

    # Phase 8: Final fine CD
    if time.time() < deadline - 5:
        nb = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), nb, deadline - 2, n_steps=2)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
