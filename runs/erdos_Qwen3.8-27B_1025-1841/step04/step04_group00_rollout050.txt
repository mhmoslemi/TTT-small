        top_idx = np.arange(len(w))
    
    grad = np.zeros(n)
    for k in top_idx:
        wk = w[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        if s >= 0:
            # (1 - x[m+s]) for m in [0, n-1-s]
            grad[:n-s] += wk * (1 - x[s:])
            # -x[m-s] for m in [s, n-1]
            grad[s:] -= wk * x[:n-s]
        else:
            # (1 - x[m+s]) for m in [-s, n-1]
            grad[-s:] += wk * (1 - x[:n+s])
            # -x[m-s] for m in [0, n-1+s]
            grad[:n+s] -= wk * x[-s:]
    
    grad *= dx
    return obj_val, grad

def slsqp_smooth(h0, n, T, maxiter=200, top_M=30):
    def obj(x):
        return smooth_obj_grad(x, n, T, top_M)[0]
    
    def grad(x):
        return smooth_obj_grad(x, n, T, top_M)[1]
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        h_res = np.clip(res.x, 0, 1)
        # Project to exact constraint
        diff = n/2.0 - np.sum(h_res)
        if abs(diff) > 1e-14:
            h_res += diff / n
            h_res = np.clip(h_res, 0, 1)
        return h_res, compute_c5(h_res)
    except Exception:
        return h0, compute_c5(h0)

def projected_.gradient(h, n, T, top_M=30, step_size=0.1):

    # Project to [0,1]
    h_new = np.clip(h_new, 0, 1)

    if diff > 0:
        mask = h_new < 1.0
        cnt = mask.sum()
        if cnt > 0:

    else:
        mask = h_new > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0, 1)
    # Fix any remaining constraint violation
    diff = n/2.0 - np.sum(h_new)

            h_new[idx] += diff
        else:

def pgd_smooth(h0, n, T, top_M=30, max_iter=500, budget_check=None, start_time=0, budget=1300):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.1
    momentum = 0.0
    velocity = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iter):
        if budget_check and time.time() - start_time > budget - 10:
            break
        
        obj_val, grad = smooth_obj_grad(h, n, T, top_M)
        
        if momentum > 0:
            velocity = momentum * velocity - step * grad
        else:
            velocity = -step * grad
        
        h_new = h + velocity
        h_new = np.clip(h_new, 0, 1)
        # Project to constraint
        diff = n/2.0 - np.sum(h_new)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h_new < 1.0
                cnt = mask.sum()
                if cnt > 0: h_new[mask] += diff/cnt
            else:
                mask = h_new > 0.0
                cnt = mask.sum()
                if cnt > 0: h_new[mask] += diff/cnt
            h_new = np.clip(h_new, 0, 1)
            diff = n/2.0 - np.sum(h_new)
            if abs(diff) > 1e-12:
                idx = np.argmin(h_new) if diff>0 else np.argmax(h_new)
                h_new[idx] += diff
                h_new = np.clip(h_new, 0, 1)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        elif c5_new <= compute_c5(h) + 1e-10:
            h = h_new
            no_improve += 1
        else:
            step *= 0.7
            no_improve += 1
            velocity *= 0.5
        
        if no_improve > 50:
            step = max(step * 0.5, 1e-5)
            if step < 1e-6:
                break
            no_improve = 0
        
        # Adaptive step
        if it % 100 == 0 and step < 0.05:
            step = min(step * 1.2, 0.1)
        if it % 200 == 0:
            momentum = min(0.9, momentum + 0.1)
    
    return best_h, best_c5

def binary_sa(h0, n, max_iters=50000, t0=0.1, t_end=0.001, budget_check=None, start_time=0, budget=1300):
    """Simulated annealing on binary representation."""
    h = h0.copy()
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    if np.sum(h_bin) > n/2:
        idx = np.where(h_bin == 1)[0]
        remove = np.random.choice(idx, int(np.sum(h_bin) - n/2), replace=False)
        h_bin[remove] = 0
    elif np.sum(h_bin) < n/2:
        idx = np.where(h_bin == 0)[0]
        add = np.random.choice(idx, int(n/2 - np.sum(h_bin)), replace=False)
        h_bin[add] = 1
    
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    curr_c5 = best_c5
    n_ones = n // 2
    
    for it in range(max_iters):
        if budget_check and time.time() - start_time > budget - 10:
            break
        T_curr = t0 * (t_end/t0) ** (it / max_iters)
        
        # Pick a random pair to swap
        i = np.random.randint(n)
        if h_bin[i] == 1:
            j = np.random.randint(n)
            while h_bin[j] == 1:
                j = np.random.randint(n)
        else:
            j = np.random.randint(n)
            while h_bin[j] == 0:
                j = np.random.randint(n)
        
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_c5 = compute_c5(h_bin)
        delta = new_c5 - curr_c5
        
        if delta < 0 or np.random.random() < np.exp(-delta / T_curr):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_bin.copy()
                best_c5 = new_c5
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_grad(x, n, T, top_M=30):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = np.clip((corr - mc) * T, -50.0, 0.0)
    exp_vals = np.exp(shifted)
    s = exp_vals.sum()
    if s < 1e-300:
        return mc * dx, np.zeros(n)
    log_sum = np.log(s)
    obj_val = (mc + log_sum / T) * dx
    w = exp_vals / s
    if top_M < len(w):
        top_idx = np.argsort(w)[-top_M:]
    else:
        top_idx = np.arange(len(w))
    grad = np.zeros(n)
    for k in top_idx:
        wk = w[k]
        if wk < 1e-14:
            continue
        s_shift = k - (n - 1)
        if s_shift >= 0:
            grad[:n - s_shift] += wk * (1.0 - x[s_shift:])
            grad[s_shift:] -= wk * x[:n - s_shift]
        else:
            grad[-s_shift:] += wk * (1.0 - x[:n + s_shift])
            grad[:n + s_shift] -= wk * x[-s_shift:]
    grad *= dx
    return obj_val, grad

        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        cnt = int(mask.sum())
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
        else:
            idx = int(np.argmin(h_proj)) if diff > 0 else int(np.argmax(h_proj))
            h_proj[idx] += diff
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def slsqp_smooth(h0, n, T, maxiter=200, top_M=30):
    def obj(x):
        return smooth_obj_grad(x, n, T, top_M)[0]
    def grad(x):
        return smooth_obj_grad(x, n, T, top_M)[1]
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h0, method='SLSQP', jac=grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-18, 'iprint': 0})
        h_res = np.clip(res.x, 0.0, 1.0)
        h_res = project_constraint(h_res, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return h0, compute_c5(h0)

def pgd_smooth(h0, n, T, top_M=30, max_iter=500, start_time=0, budget=1300):
    h = project_constraint(h0.copy(), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05
    momentum = 0.0
    velocity = np.zeros(n)
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start_time > budget - 15:
            break
        _, grad = smooth_obj_grad(h, n, T, top_M)
        if momentum > 0:
            velocity = momentum * velocity - step * grad
        else:
            velocity = -step * grad
        h_new = project_constraint(h + velocity, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improve = 0
        elif c5_new <= compute_c5(h) + 1e-10:
            h = h_new
            no_improve += 1
        else:
            step *= 0.7
            no_improve += 1
            velocity *= 0.5
        if no_improve > 40:
            step = max(step * 0.5, 1e-5)
            if step < 1e-6:
                break
            no_improve = 0
        if it % 80 == 0 and step < 0.05:
            step = min(step * 1.3, 0.08)
        if it % 150 == 0 and momentum < 0.8:
            momentum = min(0.8, momentum + 0.1)
    
    return best_h, best_c5

def binary_sa(h0, n, max_iters=50000, t0=0.08, t_end=0.001, start_time=0, budget=1300):
    h = (h0 > 0.5).astype(float)
    ones = int(np.sum(h))
    target = n // 2
    if ones > target:
        one_idx = np.where(h == 1)[0]
        remove = one_idx[np.argsort(h0[one_idx])[:ones - target]]
        h[remove] = 0
    elif ones < target:
        zero_idx = np.where(h == 0)[0]
        add = zero_idx[np.argsort(-h0[zero_idx])[:target - ones]]
        h[add] = 1
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    curr_c5 = best_c5
    
    for it in range(max_iters):
        if time.time() - start_time > budget - 15:
            break
        T_curr = t0 * (t_end / t0) ** (float(it) / max_iters)
        i = np.random.randint(n)
        if h[i] == 1:
            j = np.random.randint(n)
            while h[j] == 1:
                j = np.random.randint(n)
        else:
            j = np.random.randint(n)
            while h[j] == 0:
                j = np.random.randint(n)
        h[i], h[j] = h[j], h[i]
        new_c5 = compute_c5(h)
        delta = new_c5 - curr_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T_curr):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5
