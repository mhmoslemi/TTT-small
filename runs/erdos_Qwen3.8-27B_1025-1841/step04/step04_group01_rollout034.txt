import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_to_binary(h, n, rng):
    """Project h to a binary vector with sum = n/2."""
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    # Fix the sum
    target = n / 2.0
    diff = target - h_bin.sum()
    if diff > 0:
        # Need to flip 'diff' zeros to ones
        zeros = np.where(h_bin == 0)[0]
        # Prefer zeros that are "close to 0.5" (i.e., h values closest to 0.5 from below)
        order = np.argsort(-h[zeros])
        for idx in order[:int(diff)]:
            h_bin[zeros[idx]] = 1.0
    elif diff < 0:
        # Need to flip 'abs(diff)' ones to zeros
        ones = np.where(h_bin == 1)[0]
        order = np.argsort(h[ones])
        for idx in order[:int(-diff)]:
            h_bin[ones[idx]] = 0.0
    return h_bin

def make_feasible(h, n):
    """Make h feasible: 0<=h<=1, sum=h_target."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def _obj_smooth(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_TOP_IDX is not None:
        corr = corr[_G_TOP_IDX]
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _project_binary(h, n, rng):
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    target = n // 2
    current = int(h_bin.sum())
    if current > target:
        excess = current - target
        ones = np.where(h_bin == 1)[0]
        perm = rng.permutation(len(ones))
        for idx in perm[:excess]:
            h_bin[ones[idx]] = 0.0
    elif current < target:
        deficit = target - current
        zeros = np.where(h_bin == 0)[0]
        perm = rng.permutation(len(zeros))
        for idx in perm[:deficit]:
            h_bin[zeros[idx]] = 1.0
    return h_bin
