2. Take a step: h_new = h - step_size * grad

For the step size, I'll use backtracking line search.

- Start with low T (smooth approximation, finds good basin)

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2} using bisection"""
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine-tune
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-15
        else:
            mask = h_proj > 1e-15
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_smooth_obj_and_grad(h, T):
    """Compute the smooth approximation of max(corr) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    
    # Softmax weights
    exp_shifted = np.exp(T * (corr - mc))
    S = exp_shifted / np.sum(exp_shifted)
    
    obj = (mc + np.log(np.sum(exp_shifted)) / T) * dx
    
    # Gradient
    # term2[j] = conv(h, S)[j + n - 1]
    conv2 = np.convolve(h, S, mode='full')
    term2 = conv2[n-1 : n-1+n]
    
    # term3[j] = conv(h, S[::-1])[j + n - 1]
    conv3 = np.convolve(h, S[::-1], mode='full')
    term3 = conv3[n-1 : n-1+n]
    
    # term1[j] = sum of S[p] for p in [max(0, n-1-j), min(2n-2, 2n-2-j)]
    S_cum = np.concatenate([[0.0], np.cumsum(S)])
    lo_idx = np.maximum(0, n - 1 - np.arange(n))
    hi_idx = np.minimum(2*n - 2, 2*n - 2 - np.arange(n))
    term1 = S_cum[hi_idx + 1] - S_cum[lo_idx]
    
    grad = dx * (term1 - term2 - term3)
    
    return obj, grad

def projected_gradient_step(h, T, step_size, n):
    """One projected gradient descent step."""
    obj, grad = compute_smooth_obj_and_grad(h, T)
    h_new = h - step_size * grad
    h_new = project_feasible(h_new, n)
    return obj, h_new

def optimize_pg(h_start, n, T_schedule, max_iters, time_limit):
    """Projected gradient optimization with adaptive step size."""
    h = h_start.copy()
    start_time = time.time()
    best_obj = compute_c5(h)
    best_h = h.copy()
    
    for T in T_schedule:
        step_size = 0.5 / T  # initial step size
        stale = 0
        
        for it in range(max_iters):
            if time.time() - start_time > time_limit:
                break
            
            obj, grad = compute_smooth_obj_and_grad(h, T)
            
            # Backtracking line search
            step = step_size
            h_new = h
            obj_new = obj
            for _ in range(30):
                h_candidate = project_feasible(h - step * grad, n)
                obj_candidate, _ = compute_smooth_obj_and_grad(h_candidate, T)
                if obj_candidate < obj - 1e-14:
                    h_new = h_candidate
                    obj_new = obj_candidate
                    break
                step *= 0.5
            
            h = h_new
            
            c5 = compute_c5(h)
            if c5 < best_obj:
                best_obj = c5
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
            
            if stale > 200:
                break
        
        if time.time() - start_time > time_limit:
            break
    
    return best_h, best_obj
```
