```python
    #   New = h[a+m]*I(a+m valid)
    # Δ_a = h[a+m]*I(a+m) - (1-h[a-m])*I(a-m)
    
    a_p = a + m  # a+m
    a_m = a - m  # a-m
    
    delta_a = np.zeros(2*n-1)
    mask_ap = (a_p >= 0) & (a_p < n)
    delta_a[mask_ap] = h[a_p[mask_ap]]
    mask_am = (a_m >= 0) & (a_m < n)
    delta_a[mask_am] -= (1 - h[a_m[mask_am]])
    
    # For position b going 0→1:
    # Old contribution: 0 [term A with h=0]
    #                    + h[b+m] * (1-0) * I(b+m valid) [term B]
    #   Old = h[b+m]*I(b+m valid)
    # New contribution: (1-h[b-m])*I(b-m valid) [term A with h=1]
    #                    + 0 [term B with h=1]
    #   New = (1-h[b-m])*I(b-m valid)

    b_m = b - m

    mask_bm = (b_m >= 0) & (b_m < n)
    delta_b[mask_bm] = (1 - h[b_m[mask_bm]])
    mask_bp = (b_p >= 0) & (b_p < n)
    delta_b[mask_bp] -= h[b_p[mask_bp]]
    
    return delta_a + delta_b
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def swap_delta(h, a, b, n, lags):
    """O(n) computation of correlation change for swap h[a]:1→0, h[b]:0→1."""
    # lags is precomputed np.arange(-(n-1), n), length 2n-1
    delta = np.zeros(2*n-1)
    
    # Position a (1→0):
    # Δ from term A: -(1-h[a-m]) for valid a-m  [removing h[a]=1 from A]
    # Δ from term B: +h[a+m] for valid a+m     [adding (1-h[a])=1 to B]
    a_p = a + lags
    mask_ap = (a_p >= 0) & (a_p < n)
    delta[mask_ap] += h[a_p[mask_ap]]
    a_m = a - lags
    mask_am = (a_m >= 0) & (a_m < n)
    delta[mask_am] -= (1.0 - h[a_m[mask_am]])
    
    # Position b (0→1):
    # Δ from term A: +(1-h[b-m]) for valid b-m  [adding h[b]=1 to A]
    # Δ from term B: -h[b+m] for valid b+m     [removing (1-h[b])=1 from B]
    b_m = b - lags
    mask_bm = (b_m >= 0) & (b_m < n)
    delta[mask_bm] += (1.0 - h[b_m[mask_bm]])
    b_p = b + lags
    mask_bp = (b_p >= 0) & (b_p < n)
    delta[mask_bp] -= h[b_p[mask_bp]]
    
    return delta

    curr_corr = get_corr(curr).astype(float)

    lags = np.arange(-(n-1), n)
    
    best_max = curr_max
    best_h = curr.copy()
    no_improve = 0
    verify_count = 0
    
    while time.time() - start < max_time:
        if no_improve > 10000:
            break
        
        # Find top shifts
        top_k = np.argsort(curr_corr)[-5:]
        
        found = False
        for max_idx in top_k:
            m_star = lags[max_idx]
            
            # Contributing positions: h[i+m_star]=1, h[i]=0
            if m_star >= 0:
                i_range = np.arange(0, n - m_star)
            else:
                i_range = np.arange(-m_star, n)
            
            cmask = (curr[i_range + m_star] == 1) & (curr[i_range] == 0)
            cpos = i_range[cmask]
            
            if len(cpos) == 0:
                continue
            
            # Try moving some of the 1-positions (src = i + m_star)
            n_try = min(30, len(cpos))
            idxs = rng.choice(len(cpos), size=n_try, replace=False)
            
            for idx in idxs:
                src = int(cpos[idx] + m_star)
                # Find target: a 0-position
                zeros = np.where(curr == 0)[0]
                # Try a few random zeros, preferring distant ones
                n_tgt = min(15, len(zeros))
                tgts = rng.choice(zeros, size=n_tgt, replace=False)
                
                for tgt in tgts:
                    tgt = int(tgt)
                    if tgt == src:
                        continue
                    d = swap_delta(curr, src, tgt, n, lags)
                    nc = curr_corr + d
                    nm = np.max(nc)
                    if nm < curr_max - 1e-10:
                        src, tgt2 = src, tgt
                        found_swap = True
                        break
                if found_swap:
                    break
            if found_swap:
                d = swap_delta(curr, src, tgt2, n, lags)
                curr[src] = 0
                curr[tgt2] = 1
                curr_corr += d
                curr_max = np.max(curr_corr)

                    best_h = curr.copy()

                curr_corr = actual
                curr_max = np.max(curr_corr)
    
    return best_h, best_max * 2.0 / n
```

2. Allow a small window of fractional values around each transition
3. Optimize those fractional values to minimize the max correlation

```python
def soft_boundary_optimize(h_binary, n, rng, max_time=200):
    """Optimize fractional values around transition points."""
    start = time.time()
    h = h_binary.astype(float).copy()
    
    # Find transition points (where h changes from 0 to 1 or 1 to 0)
    transitions = np.where(h[1:] != h[:-1])[0]
    
    # Create a set of "soft" positions (transition ± window)
    window = 3
    soft_mask = np.zeros(n, dtype=bool)
    for t in transitions:
        for d in range(-window, window+1):
            pos = t + d
            if 0 <= pos < n:
                soft_mask[pos] = True
    
    soft_idx = np.where(soft_mask)[0]
    n_soft = len(soft_idx)
    
    # Optimize these positions
    for iteration in range(100000):
        if time.time() - start > max_time:
            break
        
        # Pick a soft position and try to adjust it
        j = soft_idx[rng.randint(n_soft)]
        old_val = h[j]
        best_val = old_val
        best_c = compute_c5(h)
        
        for dv in [-0.2, -0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1, 0.2]:
            new_val = old_val + dv
            if 0 <= new_val <= 1:
                trial = h.copy()
                trial[j] = new_val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = new_val
        
        if best_val != old_val:
            h[j] = best_val
            h = make_feasible(h, n)
    
    return h, compute_c5(h)
```

```python
                        if curr_max < best_max:
                            best_max = curr_max
                            best_h = curr.copy()
                        found = True
                        break
                if found:
                    break
            if found:
                break

        if not found:
            no_improve += 1
            if no_improve % 30 == 0:
                ones = ones_cache
                zeros = zeros_cache
                if len(ones) > 0 and len(zeros) > 0:
                    src = int(rng.choice(ones))
                    tgt = int(rng.choice(zeros))
                    d = swap_delta(curr, src, tgt, n)
                    nc = curr_corr + d
                    nm = np.max(nc)
                    temp = max(0.5, 5.0 / (1.0 + no_improve * 0.001))
                    if nm <= curr_max or rng.random() < np.exp(-(nm - curr_max) / temp):
                        curr[src] = 0
                        curr[tgt] = 1
                        curr_corr = nc
                        curr_max = nm
                        zeros_cache = np.where(curr == 0)[0]
                        ones_cache = np.where(curr == 1)[0]

        verify_cnt += 1
        if verify_cnt % 3000 == 0:
            actual = get_corr(curr)
            if np.max(np.abs(actual - curr_corr)) > 0.5:
                curr_corr = actual
                curr_max = np.max(curr_corr)

    return best_h, best_max * 2.0 / n

def soft_boundary_opt(h, n, rng, max_time=150):
    start = time.time()
    h = h.astype(float).copy()
    trans = np.where(h[1:] != h[:-1])[0]
    window = 4
    soft_mask = np.zeros(n, dtype=bool)
    for t in trans:
        for d in range(-window, window + 1):
            p = t + d
            if 0 <= p < n:
                soft_mask[p] = True
    soft_idx = np.where(soft_mask)[0]
    n_soft = len(soft_idx)
    if n_soft == 0:
        return h, compute_c5(h)

    best_c = compute_c5(h)
    while time.time() - start < max_time:
        j = soft_idx[rng.randint(n_soft)]
        old_val = h[j]
        found_improve = False
        for dv in sorted([-0.3, -0.2, -0.15, -0.1, -0.05, -0.03, -0.01,
                          0.01, 0.03, 0.05, 0.1, 0.15, 0.2, 0.3],
                         key=lambda x: abs(x - old_val)):
            nv = old_val + dv
            if 0.0 <= nv <= 1.0:
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    h = trial
                    best_c = c
                    found_improve = True
                    break
        if not found_improve:
            for dv in [-0.005, 0.005, -0.01, 0.01, -0.02, 0.02]:
                nv = old_val + dv
                if 0.0 <= nv <= 1.0:
                    trial = h.copy()
                    trial[j] = nv
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-15:
                        h = trial
                        best_c = c
                        found_improve = True
                        break
    return h, best_c

def slsqp_polish(h, n, rng, max_time=200, T_list=None):
    start = time.time()
    if T_list is None:
        T_list = [200.0, 500.0, 1500.0, 5000.0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n

    def obj(x):
        T = obj.T
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    for T in T_list:
        if time.time() - start > max_time - 20:
            break
        obj.T = T
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-16})
            c5_r = compute_c5(res.x)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Shift-targeted: focus only on top shifts
    if time.time() - start < max_time - 30:
        corr = get_corr(best_h)
        mc = np.max(corr)
        thr = mc - 0.02 * mc
        top_idx = np.where(corr >= thr)[0]
        if len(top_idx) > 25:
            order = np.argsort(-corr[top_idx])
            top_idx = top_idx[order[:25]]

        def obj_top(x):
            T = obj_top.T
            corr = np.correlate(x, 1.0 - x, mode='full')
            top = corr[top_idx]
            mct = np.max(top)
            shifted = top - mct
            return (mct + np.log(np.sum(np.exp(T * shifted))) / T) * dx + 0.01 * (np.sum(x) - n/2.0)**2

        for T in [400.0, 2000.0]:
            if time.time() - start > max_time - 15:
                break
            obj_top.T = T
            bounds = [(0.0, 1.0)] * n
            try:
                res = minimize(obj_top, best_h, method='SLSQP',
                               bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-16})
                c5_r = compute_c5(res.x)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(res.x, n)
            except Exception:
                pass

    return best_h, best_c5

def coord_descent_polish(h, n, max_time=100):
    start = time.time()
    steps = [0.001, 0.003, 0.005, 0.01, 0.02, 0.05]
    for _ in range(50):
        if time.time() - start > max_time:
            break
        improved = False
        for j in range(n):
            if time.time() - start > max_time:
                break
            old = h[j]
            for dv in steps:
                for s in [-1, 1]:
                    v = old + s * dv
                    if 0.0 <= v <= 1.0:
                        trial = h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < compute_c5(h) - 1e-16:
                            h = trial
                            improved = True
                            break
                if improved:
                    break
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    global _G_LAGS
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(960, 0.5)

    n = len(h0)
    _G_LAGS = np.arange(-(n - 1), n)
    h0 = make_feasible(h0, n)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    def tleft():
        return budget_s - (time.time() - t0)

    # Phase 1: Binary local search from initial
    if tleft() > 400:
        b = (h0 >= 0.5).astype(float)
        tgt_ones = n // 2
        while np.sum(b) > tgt_ones:
            ones = np.where(b == 1)[0]
            b[ones[rng.randint(len(ones))]] = 0
        while np.sum(b) < tgt_ones:
            zeros = np.where(b == 0)[0]
            b[zeros[rng.randint(len(zeros))]] = 1
        b_c5 = compute_c5(b)
        if b_c5 < best_c5:
            best_c5 = b_c5
            best_h = b.copy()
        b_best, b_c5 = binary_local_search(b, n, rng, max_time=min(350, tleft() - 200))
        if b_c5 < best_c5:
            best_c5 = b_c5
            best_h = b_best

    # Phase 2: Try alternative binary start (random)
    if tleft() > 400:
        b_rand = np.zeros(n)
        perm = rng.permutation(n)
        b_rand[perm[:n // 2]] = 1.0
        b_rand_best, b_rand_c5 = binary_local_search(b_rand, n, rng, max_time=min(150, tleft() - 200))
        if b_rand_c5 < best_c5:
            best_c5 = b_rand_c5
            best_h = b_rand_best

    # Phase 3: Soft boundary optimization
    if tleft() > 250:
        h_soft, c_soft = soft_boundary_opt(best_h, n, rng, max_time=min(120, tleft() - 150))
        if c_soft < best_c5:
            best_c5 = c_soft
            best_h = h_soft

    # Phase 4: SLSQP polish
    if tleft() > 150:
        h_sl, c_sl = slsqp_polish(best_h, n, rng, max_time=min(180, tleft() - 80))
        if c_sl < best_c5:
            best_c5 = c_sl
            best_h = h_sl

    # Phase 5: Coordinate descent
    if tleft() > 60:
        h_cd, c_cd = coord_descent_polish(best_h, n, max_time=min(80, tleft() - 20))
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Phase 6: Final SLSQP with very high T
    if tleft() > 50:
        h_final, c_final = slsqp_polish(best_h, n, rng, max_time=min(45, tleft() - 10),
                                         T_list=[3000.0, 10000.0])
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
```
