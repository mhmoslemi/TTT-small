So it's non-convex. But we can still use L-BFGS-B on it.

    h = np.clip(h.astype(float), 0.0, 1.0)

        else:

    m = np.max(corr)
    shifted = corr - m
    return m + np.log(np.sum(np.exp(T * shifted))) / T

    shifted = corr - m

        else:
            s = -shift
            grad[:n - s] += w * (1.0 - x[s:])

        return smooth_max_grad(x, n, T)

        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)

    """Perform subgradient steps targeting the worst lag."""
    dx = 2.0 / n
    current = h.copy()

        worst_idx = np.argmax(corr)
        shift = worst_idx - (n - 1)

            grad[:n - shift] -= current[shift:]

            grad[s:] -= current[:n - s]

            for d in [0.05, 0.02, 0.01, -0.01, -0.02, -0.05]:
                v = old + d
                if 0.0 <= v <= 1.0:
                    cands.append(v)
            best_val = old
            best_cc = best_c
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = current.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_cc - 1e-14:
                    best_cc = c
                    best_val = cand
            if best_val != old:
                current = current.copy()
                current[j] = best_val
                current = make_feasible(current, n)
                best_c = best_cc
    return current, best_c

def run(seed=42, budget_s=1300, **kwargs):
    _G['start'] = time.time()
    rng = np.random.RandomState(seed)
    deadline = _G['start'] + budget_s

    def elapsed():
        return time.time() - _G['start']

    def rem():
        return budget_s - elapsed()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP at initial resolution with increasing T
    n_work = n_init
    phase1_configs = [(30.0, 400), (80.0, 300), (200.0, 250), (500.0, 200)]
    for T, maxiter in phase1_configs:
        if rem() < 150:
            break
        h_opt = _slsqp(best_h, n_work, T, maxiter)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Subgradient chasing at initial resolution
    if rem() > 200:
        for start_s, end_s in [(0.02, 0.001), (0.01, 0.0005), (0.005, 0.0002)]:
            if rem() < 100:
                break
            h_chase = subgradient_chase(best_h, n_work, 300, start_s, end_s)
            c5_chase = compute_c5(h_chase)
            if c5_chase < best_c5:
                best_c5 = c5_chase
                best_h = h_chase
            # SLSQP recovery after chasing
            if rem() > 80:
                h_rec = _slsqp(best_h, n_work, 200.0, 150)
                c5_rec = compute_c5(h_rec)
                if c5_rec < best_c5:
                    best_c5 = c5_rec
                    best_h = h_rec

    # Phase 3: Multi-worst chasing
    if rem() > 200:
        h_multi = multi_worst_chase(best_h, n_work, 600, 0.015, 0.001, n_worst=4)
        c5_multi = compute_c5(h_multi)
        if c5_multi < best_c5:
            best_c5 = c5_multi
            best_h = h_multi
        if rem() > 80:
            h_rec = _slsqp(best_h, n_work, 300.0, 200)
            c5_rec = compute_c5(h_rec)
            if c5_rec < best_c5:
                best_c5 = c5_rec
                best_h = h_rec

    # Phase 4: Upsample to higher resolution and optimize
    if rem() > 300:
        n_high = n_work * 2
        h_high = upsample(best_h, n_high)
        c5_high = compute_c5(h_high)
        for T, maxiter in [(50.0, 200), (150.0, 150), (400.0, 100)]:
            if rem() < 150:
                break
            h_opt = _slsqp(h_high, n_high, T, maxiter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < c5_high:
                c5_high = c5_opt
                h_high = h_opt
        # Chase at high res
        if rem() > 150:
            h_chase = subgradient_chase(h_high, n_high, 200, 0.01, 0.001)
            c5_chase = compute_c5(h_chase)
            if c5_chase < c5_high:
                c5_high = c5_chase
                h_high = h_chase
            if rem() > 80:
                h_rec = _slsqp(h_high, n_high, 200.0, 150)
                c5_rec = compute_c5(h_rec)
                if c5_rec < c5_high:
                    c5_high = c5_rec
                    h_high = h_rec
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high
            n_work = n_high

    # Phase 5: Binary refinement
    if rem() > 60:
        best_h, best_c5 = binary_refine(best_h, n_work, 20, deadline - 30)

    # Phase 6: Coordinate descent
    if rem() > 60:
        best_h, best_c5 = coord_descent(best_h, n_work, deadline - 30)

    # Phase 7: Final SLSQP polish
    if rem() > 50:
        h_final = _slsqp(best_h, n_work, 1000.0, 100)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    # Phase 8: Final binary snap
    if rem() > 20:
        best_h, best_c5 = binary_refine(best_h, n_work, 5, deadline - 5)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
