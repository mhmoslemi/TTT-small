        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = corr.max()
    # stable log-sum-exp
    logsum = np.log(np.sum(np.exp(T * (corr - mc)))) 
    return (mc + logsum / T) * dx

def smooth_max_full_grad(x, T, n):
    """Full gradient of smooth max over all shifts."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = corr.max()
    exp_vals = np.exp(T * (corr - mc))
    s = exp_vals.sum()
    if s == 0:
        return np.zeros(n)
    w = exp_vals / s
    
    # Only process shifts with non-negligible weight
    thresh = 1e-10
    active = np.where(w > thresh)[0]
    
    grad = np.zeros(n)
    for k_idx in active:
        sh = k_idx - (n - 1)
        wk = w[k_idx]
        if sh == 0:
            grad -= 2.0 * wk * x
        elif sh > 0:
            grad[sh:] -= wk * x[:n - sh]
            grad[:n - sh] -= wk * x[sh:]
        else:
            t = -sh
            grad[:n - t] -= wk * x[t:]
            grad[t:] -= wk * x[:n - t]
    
    return grad * dx

        frac = min(1.0, (time.time() - start_time - (budget_s - total_time)) / total_time)
        T = T_init + (T_max - T_init) * frac ** 0.7
        
        obj, grad = smooth_max_obj(h, T, n), smooth_max_full_grad(h, T, n)

        for ls in range(15):
            new_h = make_feasible(h - step * grad, n)
            new_obj = smooth_max_obj(new_h, T, n)
            if new_obj < obj - 1e-4 * step * np.linalg.norm(grad):
                h = new_h
                break
            step *= 0.5

        if stall % 50 == 0:

    c5 = compute_c5(h)
    if c5 < best_c5:

    return best_h, best_c5

def slsqp_polish(h, n, budget_s, start_time, T=1000.0):

        return smooth_max_obj(x, T, n)

        return smooth_max_full_grad(x, T, n)

        return np.ones(n)
