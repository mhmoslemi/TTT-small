```python
    """Compute gradient of sum_k p[k] * output[k] where output = np.correlate(h, 1-h, 'full')"""
    grad = np.zeros(n_pts)
    jarr = np.arange(n_pts)
    for k_idx in range(len(p)):
        pk = p[k_idx]
        if pk < 1e-14:
            continue
        d = k_idx - n_pts + 1
        # valid n range
        n_lo = max(0, -d)
        n_hi = min(n_pts - 1, n_pts - 1 - d)
        # First term: h[n+d]*(1-h[n]), derivative w.r.t. h[j] when j = n+d
        # j = n+d, n in [n_lo, n_hi], so j in [n_lo+d, n_hi+d]
        # (1-h[n]) = (1-h[j-d])
        j_lo1 = max(0, n_lo + d)
        j_hi1 = min(n_pts - 1, n_hi + d)
        if j_lo1 <= j_hi1:
            idx = jarr[(jarr >= j_lo1) & (jarr <= j_hi1)]
            grad[idx] += pk * (1.0 - h[idx - d])
        # Second term: -h[n+d]*h[n], derivative w.r.t. h[j] when j = n
        # -h[j+d], n in [n_lo, n_hi]
        if n_lo <= n_hi:
            idx = jarr[(jarr >= n_lo) & (jarr <= n_hi)]
            grad[idx] += -pk * h[idx + d]
    return grad
```

```python
def alternating_qp(h_init, n, beta_seq, max_rounds, mu=500.0):
    """Alternating: update p (weights), then solve inner QP."""
    h = h_init.copy()
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        # Compute correlate
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Update weights (exponential)
        beta = beta_seq[rnd % len(beta_seq)]
        cmax = np.max(corr)
        p = np.exp(beta * (corr - cmax))
        p /= p.sum()
        
        # Solve inner QP with L-BFGS-B
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(p, c) * dx) + mu * (x.sum() - n/2.0)**2
        
        def grad(x):
            g = corr_grad_weighted(x, p, n) * dx
            g += 2.0 * mu * (x.sum() - n/2.0)
            return g
        
        try:
            res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)]*n,
                          options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = make_feasible(res.x, n)
            c_new = compute_c5(h_new)
            c_old = compute_c5(h)
            if c_new <= c_old + 1e-10:
                h = h_new
        except Exception:
            pass
    
    return h
```

```python
        val[valid] = h[idx[valid]]
        return val

    # Corrections for overlapping cases

    near_i = abs(di) < n  # d = di is within range
    if near_i:

        delta[idx_correct] += 1.0
    near_j = abs(dj) < n
    if near_j:

        delta[idx_correct] += 1.0
```
