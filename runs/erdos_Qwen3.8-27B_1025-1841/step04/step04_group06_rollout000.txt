    fb = np.fft.rfft(b, nfft)
    # correlate(a,b) = convolve(a, b[::-1])
    # convolve via FFT: ifft(fft(a) * fft(b[::-1]))
    # But b[::-1] in frequency domain is... let me use a different approach
    # Actually: correlate(a, b)[k] = sum_i a[i] * b[i-k+M-1] where M = len(b)
    # Simpler: use the identity correlate(a,b,'full') = convolve(a, b[::-1], 'full')
    b_rev = b[::-1]
    fb_rev = np.fft.rfft(b_rev, nfft)
    result = np.fft.irfft(fa * fb_rev, nfft)[:size]
    return result

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = fft_correlate_full(h, one_minus_h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def binary_sa(h_bin, n, rng, time_limit, cool_rate=0.9995, T_start=2.0, T_min=0.01):
    """Simulated annealing on binary sequence with swap moves."""
    h = h_bin.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    current_c5 = compute_c5_fast(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    T = T_start
    end_time = time.time() + time_limit
    iterations = 0
    no_improve = 0
    patience = 50000
    
    while time.time() < end_time:
        if no_improve > patience:
            break
        T = max(T_min, T * cool_rate)
        
        # Batch of swaps
        batch = 200
        for _ in range(batch):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            
            h[oi] = 0.0
            h[oj] = 1.0
            new_c5 = compute_c5_fast(h)
            
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                # Accept
                current_c5 = new_c5
                ones[ii] = oj
                zeros[jj] = oi
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[oi] = 1.0
                h[oj] = 0.0
                no_improve += 1
        iterations += batch
    
    return best_h, best_c5

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    
    def f(h):
        corr = fft_correlate_full(h, 1.0 - h)
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def generate_m_sequence(n_bits):
    """Generate a maximum-length sequence (m-sequence) of given length."""
    # For m-sequences, n_bits must be 2^m - 1
    # We'll use a simpler approach: generate a low-autocorrelation binary sequence
    rng_local = np.random.RandomState(12345)
    # Use a Gold-like construction

    seq = np.zeros(n_bits)
    n_ones = n_bits // 2
    indices = rng_local.permutation(n_bits)[:n_ones]
    seq[indices] = 1.0
    return seq
