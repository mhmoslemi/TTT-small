def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, n):
    """Make h binary with exactly n/2 ones, closest to input h."""
    h = np.asarray(h, dtype=float).copy()
    # Sort indices by value descending
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:n//2]] = 1.0
    return result

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

# Global variables for objective function
_OPT_N = 0
_OPT_T = 1.0

def _obj_fn(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

    return np.ones(_OPT_N)

    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n

    res = minimize(_obj_fn, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

    best_c5 = curr_c5
    best_h = h.copy()

    zeros = np.where(h == 0.0)[0]
    
    curr_temp = 0.005  # annealing temperature

    for it in range(n_iters):
        if time_check() < 30:
            break
        if it % 1000 == 0:
            curr_temp *= cool_rate

            zeros = np.append(zeros, i)

            curr_corr = new_corr
        else:

            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 0
_OPT_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def make_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    idx = np.argsort(-h)
    result = np.zeros(n)
    result[idx[:n // 2]] = 1.0
    return result

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _obj_fn(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

    return np.ones(_OPT_N)

                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})

        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        new_c5 = compute_c5(h)

            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
