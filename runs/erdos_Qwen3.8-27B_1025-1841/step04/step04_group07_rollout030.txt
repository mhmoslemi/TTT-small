**Key optimization for binary local search:**

Instead of recomputing the full correlation from scratch each time (O(n²)), I can update it incrementally. When I swap positions i (1→0) and j (0→1):

The correlation at lag k (for the "full" output, indexed from n-1-k to n-1+k):

Let me denote the full correlation output as `c` of length 2n-1, where c[n-1+k] corresponds to lag k.

For lag k > 0: c[n-1+k] = Σ_{m=0}^{n-1-k} h[m](1-h[m+k])

When h[i] changes from 1 to 0:
- For lag k: if 0 ≤ i ≤ n-1-k, the term h[i](1-h[i+k]) changes from (1-h[i+k]) to 0, so c decreases by (1-h[i+k])
- For lag k: if i-k ≥ 0 (i.e., k ≤ i), the term at m=i-k is h[i-k](1-h[i]), which changes from h[i-k]*1 to h[i-k]*0, so c decreases by h[i-k]

When h[j] changes from 0 to 1:
- For lag k: if 0 ≤ j ≤ n-1-k, the term h[j](1-h[j+k]) changes from 0 to (1-h[j+k]), so c increases by (1-h[j+k])
- For lag k: if j-k ≥ 0 (i.e., k ≤ j), the term at m=j-k is h[j-k](1-h[j-k+... wait.

Let me be more careful. For lag k > 0:
c_k = Σ_{m=0}^{n-1-k} h[m]·(1-h[m+k])

Position i appears in this sum as:
- First factor: h[i] when m = i, requires i ≤ n-1-k, i.e., k ≤ n-1-i
- Second factor: (1-h[i]) when m+k = i, i.e., m = i-k, requires 0 ≤ i-k, i.e., k ≤ i

So for h[i]:-) changes from 1 to 0:
- As first factor (m=i): if k ≤ n-1-i, change = -(1-h[i+k]) [term goes from 1*(1-h[i+k]) to 0]
- As second factor (m=i-k): if k ≤ i, change = -h[i-k] [term goes from h[i-k]*1 to h[i-k]*0]

For h[j]:0→1 changes from 0 to 1:
- As first factor (m=j): if k ≤ n-1-j, change = +(1-h[j+k]) [term goes from 0 to 1*(1-h[j+k])]
- As second factor (m=j-k): if k ≤ j, change = -h[j-k] [term goes from h[j-k]*1 to h[j-k]*0]

Wait, for the second factor of h[j]: the term is h[j-k]·(1-h[j]). Before: h[j-k]·(1-0) = h[j-k]. After: h[j-k]·(1-1) = 0. Change = -h[j-k].

So the total change in c_k for a swap (i: 1→0, j: 0→1):
Δc_k = -(1-h[i+k])·[k≤n-1-i] - h[i-k]·[k≤i] + (1-h[j+k])·[k≤n-1-j] - h[j-k]·[k≤j]

where [condition] is 1 if condition is true, 0 otherwise.

Similarly for negative lags. By the symmetry of the full correlation (c[n-1+k] and c[n-1-k]), we can handle both.

This is O(1) per lag, and there are O(n) lags, so O(n) per swap evaluation. Much better than O(n²)!

    n_ones = int(h_bin.sum())
    if n_ones > target:
        oi = np.where(h_bin == 1)[0]
        order = np.argsort(h[oi])
        for idx in order[:n_ones - target]:
            h_bin[oi[idx]] = 0.0
    elif n_ones < target:
        zi = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zi])
        for idx in order[:target - n_ones]:
            h_bin[zi[idx]] = 1.0
    return h_bin

def greedy_binary(h_bin, n, rng, max_rounds=200, tries_per_round=3000):
    h = h_bin.copy()
    cur_max = max_corr_val(h)
    best_h = h.copy()
    best_max = cur_max
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        np_tries = min(tries_per_round, max(1, len(ones) * len(zeros)))
        for _ in range(np_tries):
            if len(ones) == 0 or len(zeros) == 0:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_max = max_corr_val(h)
            if new_max < cur_max - 1e-15:
                cur_max = new_max
                ones.remove(i)
                ones.append(j)
                zeros.remove(j)
                zeros.append(i)
                improved = True
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not improved:
            break
    return best_h, best_max

def sa_binary(h_bin, n, rng, max_iters=80000, T0=3.0, T_end=0.05):
    h = h_bin.copy()
    cur_max = max_corr_val(h)
    best_h = h.copy()
    best_max = cur_max
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for it in range(max_iters):
        T = T0 * (T_end / T0) ** (it / max_iters)
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_max = max_corr_val(h)
        delta = new_max - cur_max
        if delta < -1e-15 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            ones.remove(i)
            ones.append(j)
            zeros.remove(j)
            zeros.append(i)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_max

def ils_binary(h_bin, n, rng, max_rounds=100, tries=2000, perturb=6, n_restarts=30):
    best_h = h_bin.copy()
    best_max = max_corr_val(h_bin)
    h_curr = h_bin.copy()
    cur_max = max_corr_val(h_curr)
    for restart in range(n_restarts):
        h_g, c_g = greedy_binary(h_curr, n, rng, max_rounds=max_rounds, tries_per_round=tries)
        if c_g < best_max:
            best_max = c_g
            best_h = h_g.copy()
        # Perturb
        h_p = best_h.copy()
        ones = np.where(h_p > 0.5)[0]
        zeros = np.where(h_p <= 0.5)[0]
        for _ in range(perturb):
            if len(ones) == 0 or len(zeros) == 0:
                break
            oi = rng.randint(len(ones))
            zi = rng.randint(len(zeros))
            h_p[ones[oi]] = 0.0
            h_p[zeros[zi]] = 1.0
            ones = np.where(h_p > 0.5)[0]
            zeros = np.where(h_p <= 0.5)[0]
        h_curr = h_p
        cur_max = max_corr_val(h_curr)
    return best_h, best_max

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def continuous_refine(h0, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_obj(corr, T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def slsqp_refine(h0, n, T, max_iter=60):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_obj(corr, T) * dx
    def con_eq(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def thue_morse_init(n):
    t = np.zeros(n)
    for i in range(n):
        t[i] = -1.0 if bin(i).count('1') % 2 else 1.0
    h = (1.0 + t) / 2.0
    return to_binary(h, n)

def periodic_init(n, p):
    q = p // 2
    best_h = None
    best_c = np.inf
    from itertools import combinations
    for combo in combinations(range(p), q):
        pattern = np.zeros(p)
        for c in combo:
            pattern[c] = 1.0
        h_full = np.tile(pattern, n // p)
        if n % p != 0:
            h_full = np.concatenate([h_full, pattern[:n % p]])
        h_full = h_full[:n]
        c = max_corr_val(h_full)
        if c < best_c:
            best_c = c
            best_h = h_full.copy()
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 640
        h0 = np.tile([1.0, 0.0], n0 // 2).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n0

    # Phase 1: Binary search at parent resolution
    h_bin = to_binary(h0, n0)
    binary_c = max_corr_val(h_bin)

    # Greedy from parent binary
    if rem() > 100:
        h_g, c_g = greedy_binary(h_bin, n0, rng, max_rounds=150, tries_per_round=2000)
        if c_g * 2.0 / n0 < best_c5:
            best_c5 = c_g * 2.0 / n0
            best_h = h_g.copy()
            h_bin = h_g.copy()

    # ILS from parent binary
    if rem() > 200:
        h_il, c_il = ils_binary(h_bin, n0, rng, max_rounds=80, tries=1500, perturb=5, n_restarts=15)
        if c_il * 2.0 / n0 < best_c5:
            best_c5 = c_il * 2.0 / n0
            best_h = h_il.copy()
            h_bin = h_il.copy()

    # SA from parent binary
    if rem() > 200:
        h_sa, c_sa = sa_binary(h_bin, n0, rng, max_iters=50000, T0=3.0, T_end=0.05)
        if c_sa * 2.0 / n0 < best_c5:
            best_c5 = c_sa * 2.0 / n0
            best_h = h_sa.copy()
            h_bin = h_sa.copy()

    # Phase 2: Try different initializations at same resolution
    inits = []
    # Thue-Morse
    if rem() > 150:
        h_tm = thue_morse_init(n0)
        inits.append(h_tm)
    # Periodic patterns
    for p in [6, 8, 10, 12]:
        if n0 % p == 0 and rem() > 100:
            h_p, c_p = periodic_init(n0, p)
            inits.append(h_p)
    # Random
    for _ in range(2):
        if rem() > 100:
            h_r = (rng.rand(n0) < 0.5).astype(float)
            h_r = to_binary(h_r, n0)
            inits.append(h_r)

    for h_init in inits:
        if rem() < 80:
            break
        h_g, c_g = greedy_binary(h_init, n0, rng, max_rounds=100, tries_per_round=1500)
        if c_g * 2.0 / n0 < best_c5:
            best_c5 = c_g * 2.0 / n0
            best_h = h_g.copy()
            h_bin = h_g.copy()
        if rem() < 60:
            break
        h_il, c_il = ils_binary(h_g, n0, rng, max_rounds=60, tries=1200, perturb=4, n_restarts=8)
        if c_il * 2.0 / n0 < best_c5:
            best_c5 = c_il * 2.0 / n0
            best_h = h_il.copy()
            h_bin = h_il.copy()

    # Phase 3: Try at different resolutions
    for n_try in [512, 768, 896]:
        if rem() < 150:
            break
        h_up = upsample_h(best_h, n_try)
        h_b = to_binary(h_up, n_try)
        c_b = max_corr_val(h_b)
        if c_b * 2.0 / n_try < best_c5:
            best_c5 = c_b * 2.0 / n_try
            best_n = n_try

        h_g, c_g = greedy_binary(h_b, n_try, rng, max_rounds=120, tries_per_round=2000)
        if c_g * 2.0 / n_try < best_c5:
            best_c5 = c_g * 2.0 / n_try
            best_h = h_g.copy()
            best_n = n_try
            h_b = h_g.copy()

        if rem() > 100:
            h_il, c_il = ils_binary(h_b, n_try, rng, max_rounds=80, tries=1500, perturb=5, n_restarts=10)
            if c_il * 2.0 / n_try < best_c5:
                best_c5 = c_il * 2.0 / n_try
                best_h = h_il.copy()
                best_n = n_try

    # Phase 4: More aggressive search at best resolution
    n_work = best_n
    h_work = to_binary(best_h, n_work)
    if rem() > 100:
        # Multiple SA runs from different perturbations
        for sa_run in range(3):
            if rem() < 50:
                break
            h_pert = h_work.copy()
            ones = np.where(h_pert > 0.5)[0]
            zeros = np.where(h_pert <= 0.5)[0]
            for _ in range(3 + sa_run * 2):
                oi = rng.randint(len(ones))
                zi = rng.randint(len(zeros))
                h_pert[ones[oi]] = 0.0
                h_pert[zeros[zi]] = 1.0
                ones = np.where(h_pert > 0.5)[0]
                zeros = np.where(h_pert <= 0.5)[0]
            h_sa, c_sa = sa_binary(h_pert, n_work, rng, max_iters=40000, T0=2.5, T_end=0.03)
            if c_sa * 2.0 / n_work < best_c5:
                best_c5 = c_sa * 2.0 / n_work
                best_h = h_sa.copy()
                h_work = h_sa.copy()

    # Phase 5: Continuous refinement on best binary solution
    if rem() > 60:
        h_cont = best_h.copy()
        for T in [200, 800, 3000]:
            if rem() < 20:
                break
            h_opt = continuous_refine(h_cont, n_work, T, mu=500.0, max_iter=150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                h_cont = h_opt.copy()

    # SLSQP polish
    if rem() > 30:
        h_sp = best_h.copy()
        for T in [1000, 5000]:
            if rem() < 15:
                break
            h_opt = slsqp_refine(h_sp, n_work, T, max_iter=50)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                h_sp = h_opt.copy()

    # Phase 6: Final coordinate descent
    if rem() > 20:
        h = best_h.copy()
        cur_c = best_c5
        step_list = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
        for rnd in range(5):
            if rem() < 10:
                break
            improved = False
            for j in range(n_work):
                ov = h[j]
                for d in step_list:
                    nv = ov + d
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    trial = h.copy()
                    trial[j] = nv
                    trial = make_feasible(trial, n_work)
                    c = compute_c5(trial)
                    if c < cur_c - 1e-14:
                        cur_c = c
                        h = trial
                        ov = nv
                        improved = True
                        break
            if not improved:
                break
        if cur_c < best_c5:
            best_c5 = cur_c
            best_h = h.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
