            h_act2 = active_set_opt(h2, n2, k_act=7, alpha=4.0, max_iter=60, maxfun=800)
            c_act2 = compute_c5(h_act2)
            if c_act2 < best_c5:
                best_c5 = c_act2
                best_h = h_act2.copy()

    # Phase 4: Binary search at higher resolution
    if rem() > 80:
        n_bin = min(len(best_h), 1280)
        h_b2 = (best_h > 0.5).astype(float)
        if len(h_b2) != n_bin:
            h_b2 = upsample_h(best_h, n_bin)
            h_b2 = (h_b2 > 0.5).astype(float)
        target_ones = n_bin // 2
        n_ones = int(h_b2.sum())
        if n_ones > target_ones:
            idx = np.where(h_b2 == 1)[0]
            h_b2[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_b2 == 0)[0]
            h_b2[idx[:target_ones - n_ones]] = 1.0
        h_il2 = h_b2.copy()
        for restart in range(5):
            if rem() < 20:
                break
            h_il2, c_il2 = greedy_bin(h_il2, n_bin, rng, max_rounds=15, tries=1000)
            if c_il2 < best_c5:
                best_c5 = c_il2
                best_h_up = h_il2
                best_h = best_h_up
                best_n = n_bin
            else:
                h_il2 = perturb_bin(h_il2, rng, 4 + rng.randint(3))

    # Phase 5: SLSQP polish
    if rem() > 30:
        n_sp = min(len(best_h), 1280)
        h_sp = upsample_h(best_h, n_sp) if n_sp < len(best_h) else best_h.copy()
        for T in [5000, 15000]:
            if rem() < 15:
                break
            h_opt = slsqp_polish(h_sp, n_sp, T, 80)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                h_sp = h_opt.copy()

    # Phase 6: Final coordinate descent
    if rem() > 15:
        n_cd = len(best_h)
        h_cd, c_cd = coord_descent(best_h, n_cd, max_rounds=6,
                                   step_list=[0.02, -0.02, 0.008, -0.008, 0.003, -0.003])
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 7: Multi-start continuous from perturbed best
    if rem() > 40:
        n_ms = min(len(best_h), 960)
        h_ms = upsample_h(best_h, n_ms) if n_ms < len(best_h) else best_h.copy()
        for rs in range(3):
            if rem() < 20:
                break
            h_r = h_ms + 0.08 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            h_opt = lbgfb_smooth(h_r, n_ms, 500, 600.0, 80, 1000)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
