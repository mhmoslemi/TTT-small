from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def sa_point(b, n, max_iters, T0, T1, rng):
    """Point-swap SA."""
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    for it in range(max_iters):
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0; b[j] = 1
        new_c = compute_c5(b)
        if new_c <= curr or (T > 1e-14 and rng.random() < np.exp(-(new_c - curr) / T)):
            curr = new_c
            if new_c < best:
                best = new_c
                best_b = b.copy()
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
        else:
            b[i] = 1; b[j] = 0
    return best_b, best

def sa_block(b, n, max_iters, T0, T1, rng, bmax):
    """Block-swap SA."""
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    for it in range(max_iters):
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        bs = 1 + rng.randint(0, bmax - 1)
        if bs >= n:
            bs = 1
        start = rng.randint(0, n - bs)
        end = start + bs
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, sc, replace=False)
        cz = rng.choice(zi, sc, replace=False)
        b[start+co] = 0; b[start+cz] = 1
        new_c = compute_c5(b)
        if new_c <= curr or (T > 1e-14 and rng.random() < np.exp(-(new_c - curr) / T)):
            curr = new_c
            if new_c < best:
                best = new_c
                best_b = b.copy()
        else:
            b[start+co] = 1; b[start+cz] = 0
    return best_b, best
