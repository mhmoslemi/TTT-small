  - (h[m+s]=1, h[m-s]=1): 0-1 = -1
- j+s in range, j-s out: Δ = (1-h[m+s]) - 0
  - h[m+s]=0: +1
  - h[m+s]=1: 0
- m+s out, m-s in range: Δ = 0 - h[m-s]
  - h[m-s]=0: 0
  - h[m-s]=1: -1
- Both out: 0

So the total Δ for the swap:
Δcorr[n-1+s] = [h[j-s] - (1-h[j+s])] + [(1-h[m+s]) - h[m-s]]

This is O(1) per shift s. The total update is O(n) but with very small constant.

def swap_correlation(h, j, m, n):
    """Compute new correlation after swap h[j]:1→0, h[m]:0→1, using incremental update."""
    # Full correlation: corr_full[k] = sum_i h[i]*(1-h[i+k-(n-1)])
    # Let's work with shift s = k-(n-1), so corr[s] = sum_i h[i]*(1-h[i+s])
    
    # Change from h[j]: 1→0
    # Δ_j[s] = h[j-s] - (1-h[j+s]) for valid indices, 0 otherwise
    # Valid: j+s in [0,n-1] AND/OR j-s in [0,n-1]
    
    # Vectorized:
    s = np.arange(-(n-1), n)
    js = j + s  # j+s for each shift
    jm = j - s  # j-s for each shift
    
    # h[j+s] with bounds: 0 if out of range
    h_js = np.zeros(n*2-1)
    mask_js = (js >= 0) & (js < n)
    h_js[mask_js] = h[js[mask_js]]
    
    h_jm = np.zeros(n*2-1)
    mask_jm = (jm >= 0) & (jm < n)
    h_jm[mask_jm] = h[jm[mask_jm]]
    
    delta_j = h_jm - (1 - h_js)
    # But this is only valid where at least one of js, jm is in range
    # If js out of range: the term -(1-h[j+s]) should be 0 (not -1)
    # If jm out of range: the term h[j-s] should be 0
    # So:
    delta_j = np.where(mask_js, -(1-h_js), 0.0) + np.where(mask_jm, h_jm, 0.0)
    
    # Similarly for m: 0→1
    ms = m + s
    mm = m - s
    h_ms = np.zeros(n*2-1)
    mask_ms = (ms >= 0) & (ms < n)
    h_ms[mask_ms] = h[ms[mask_ms]]
    
    h_mm = np.zeros(n*2-1)
    mask_mm = (mm >= 0) & (mm < n)
    h_mm[mask_mm] = h[mm[mask_mm]]
    
    delta_m = np.where(mask_ms, (1-h_ms), 0.0) - np.where(mask_mm, h_mm, 0.0)
    
    return delta_j + delta_m
