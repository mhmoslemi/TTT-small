import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def power_mean_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    c = np.maximum(corr, 0.0)
    return (np.sum(c ** T)) ** (1.0 / T) * dx

def logsumexp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def binary_swap_sa(h, n, rng, iterations, t0, t1):

    cur_h = h.copy()
    cur_c = best_c
    for it in range(iterations):
        frac = it / max(1, iterations - 1)
        T = t0 * (t1 / t0) ** frac
        npair = rng.randint(1, 4)
        new_h = cur_h.copy()
        for _ in range(npair):
            i = rng.randint(n)
            j = rng.randint(n)
            new_h[i], new_h[j] = new_h[j], new_h[i]
        new_h = make_feasible(new_h, n)
        new_c = compute_c5(new_h)
        delta = new_c - cur_c
        if delta < 0 or T > 1e-12 and rng.random() < np.exp(-delta / T):
            cur_h = new_h
            cur_c = new_c
            if cur_c < best_c - 1e-15:
                best_c = cur_c
                best_h = cur_h.copy()
    return best_h, best_c

def coord_descent(h, n, c5, rng, max_rounds, time_limit):
    t0 = time.time()
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            best_trial = old
            best_c = c5
            cands = [0.0, 1.0]
            for d in [-0.2, -0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1, 0.2]:
                v = old + d
                if 0.0 < v < 1.0:
                    cands.append(v)
            if 0.2 < old < 0.8:
                cands.append(0.5)
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                h = h.copy()
                h[j] = best_trial
                h = make_feasible(h, n)
                c5 = best_c
                improved = True
        if not improved:
            break
    return h, c5

def slsqp_run(h, n, T, maxiter, use_logsumexp=True):
    if use_logsumexp:
        obj = lambda x: logsumexp_obj(x, n, T)
    else:
        obj = lambda x: power_mean_obj(x, n, T)
    con_fun = lambda x: np.sum(x) - n / 2.0
    con_jac = lambda x: np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.success or res.x is not None:
            return make_feasible(np.array(res.x, dtype=float), n)
    except Exception:
        pass
    return h
