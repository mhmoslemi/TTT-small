```python
    target = n / 2.0

        diff = target - np.sum(h)

        else:

    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
```

**Approach: Penalty + Projected Gradient**

Instead of using scipy's optimizer (which is slow for constrained problems in high dimensions), I'll implement my own projected gradient method:

1. Compute gradient g of LSE
2. Subtract the Lagrange multiplier for the constraint: g - λ * 1
3. Take a step: h_new = h - step * (g - mean(g)) (the mean subtraction handles the constraint)
4. Project to [0,1]
5. Adjust to satisfy sum constraint
6. Repeat

Actually, for the constraint sum(h) = n/2, the Lagrangian is:
L = LSE(h) - λ*(sum(h) - n/2)
∇L = ∇LSE - λ * 1

At the optimum, the component of ∇LSE orthogonal to the constraint (i.e., the part that's not just a constant) determines the direction. So we can project the gradient onto the subspace sum(x) = 0:
∇_constrained = ∇LSE - mean(∇LSE)

Then do projected gradient:
h_new = h - α * ∇_constrained
Project to [0,1]
Fix sum to n/2

This is clean and simple! Let me use Nesterov acceleration for faster convergence.

**Nesterov + Projected Gradient:**
```
y = h + momentum * (h - h_prev)
g = gradient_LSE(y) - mean(gradient_LSE(y))
h_new = project(y - step * g)
momentum = (1 + sqrt(1 + 4*momentum²)) / 2  (or simpler: t/2 update)
h_prev = h
h = h_new
```

The step size can be chosen via backtracking line search or fixed based on Lipschitz constant.

```python
import numpy as np
from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff/cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lse_value_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c) * dx)
    s = np.sum(exp_c)
    w = exp_c / s
    lse_val = (max_c + np.log(s) / T) * dx
    
    f1 = 1.0 - h
    w_flip = w[::-1]
    S1 = fftconvolve(w_flip, f1, mode='full')
    S2 = fftconvolve(w, h, mode='full')
    
    S1_ext = S1[2*n-2 : 3*n-2]
    S2_ext = S2[n-1 : 3*n-2]
    
    g = T * dx * (S1_ext - S2_ext)
    return lse_val, g

def projected_nesterov(h0, n, T, n_iter, step=None, time_limit=30, start_time=0):
    """Nesterov accelerated projected gradient on LSE with sum constraint."""
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    momentum = 0.0
    
    if step is None:
        # Estimate step size
        # Lipschitz constant of LSE gradient is roughly T^2 * dx * n (very rough)
        step = 1.0 / (T * T * 2.0 / n)  # rough
        step = min(step, 0.1)
    
    best_val = float('inf')
    best_h = h0.copy()
    
    for i in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        # Nesterov momentum
        momentum = (1.0 + np.sqrt(1.0 + 4.0 * momentum * momentum)) / 2.0 - 1.0
        y = h + momentum * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        val, g = lse_value_and_grad(y, T)
        
        # Project gradient onto affine constraint sum(x) = n/2
        g_proj = g - np.mean(g)
        
        # Line search (simple: try a few step sizes)
        for factor in [1.0, 0.5, 0.25, 0.1]:
            h_new = h - factor * step * g_proj
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = make_feasible(h_new, n)
            val_new, _ = lse_value_and_grad(h_new, T)
            if val_new < val:
                break
        
        h_prev = h
        h = h_new
        
        # Track best based on actual C5 (not LSE)
        c5 = compute_c5(h)
        if c5 < best_val:
            best_val = c5
            best_h = h.copy()
    
    return best_h, best_val
```
