    """Compute change in correlation when swapping b[i]:1->0 and b[j]:0->1."""

        T = T0 * (T1 / T0) ** frac

        i_pool = np.where(b == 1)[0]
        j_pool = np.where(b == 0)[0]
        if len(i_pool) == 0 or len(j_pool) == 0:
            break

        best_new_max = np.inf
        best_i = -1
        best_j = -1
        best_delta = None

        for _ in range(n_cand):
            i = i_pool[rng.randint(len(i_pool))]
            j = j_pool[rng.randint(len(j_pool))]
            if i == j:
                continue
            d = compute_swap_delta(i, j, b, n, lags)
            new_max = float(np.max(corr + d))
            if new_max < best_new_max:
                best_new_max = new_max
                best_i, best_j = i, j
                best_delta = d

        if best_i < 0:
            stuck += 1
            continue

        accept = best_new_max <= curr_max
        if not accept and T > 1e-12:
            delta_val = best_new_max - curr_max
            if stuck > 50:
                delta_val *= 0.3
            accept = rng.random() < np.exp(-delta_val / T)

        if accept:
            b[best_i] = 0
            b[best_j] = 1
            corr = corr + best_delta
            curr_max = best_new_max
            stuck = 0
            if curr_max < best_max:
                best_max = curr_max
                best_b = b.copy()
        else:
            stuck += 1

        if stuck > 2000:
            for _ in range(20):
                i = i_pool[rng.randint(len(i_pool))]
                j = j_pool[rng.randint(len(j_pool))]
                if i == j:
                    continue
                b[i] = 0
                b[j] = 1
                corr = compute_corr(b)
                curr_max = float(np.max(corr))
                i_pool = np.where(b == 1)[0]
                j_pool = np.where(b == 0)[0]
            stuck = 0

    return best_b, best_max * (2.0 / n)

def block_sa(h, n, n_iters, rng, deadline, block_max=4):
    b = (h >= 0.5).astype(float)
    corr = compute_corr(b)
    curr_max = float(np.max(corr))
    best_max = curr_max
    best_b = b.copy()

    for it in range(n_iters):
        if time.time() > deadline:
            break

        L = rng.randint(1, block_max + 1)
        if L >= n:
            continue
        start = rng.randint(0, n - L)
        seg = b[start:start + L]
        ones_in = np.where(seg == 1)[0]
        zeros_in = np.where(seg == 0)[0]
        if len(ones_in) == 0 or len(zeros_in) == 0:
            continue
        k = min(len(ones_in), len(zeros_in), rng.randint(1, min(len(ones_in), len(zeros_in)) + 1))
        chosen_ones = rng.choice(ones_in, size=k, replace=False)
        chosen_zeros = rng.choice(zeros_in, size=k, replace=False)

        old_vals = b[start + chosen_ones].copy()
        b[start + chosen_ones] = 0
        b[start + chosen_zeros] = 1
        new_corr = compute_corr(b)
        new_max = float(np.max(new_corr))

        if new_max <= curr_max:
            curr_max = new_max
            corr = new_corr
            if curr_max < best_max:
                best_max = curr_max
                best_b = b.copy()
        else:
            b[start + chosen_ones] = old_vals
            b[start + chosen_zeros] = 0

    return best_b, best_max * (2.0 / n)

def _obj_lse(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def continuous_optimize(h, n, deadline, max_time=120):
    global _SL_N, _SL_T
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

    for T in [80.0, 200.0, 600.0]:
        if time.time() > deadline - 20:
            break
        t0 = time.time()
        _SL_N = n
        _SL_T = T
        try:
            bounds = [(0.0, 1.0)] * n
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            res = minimize(_obj_lse, best_h.copy(), method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 100, 'ftol': 1e-14})
            if res.x is not None:
                x = make_feasible(res.x, n)
                c5 = compute_c5(x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = x
        except Exception:
            pass

    return best_h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    b0 = to_binary(h_init, n0)
    best_c5 = compute_c5(b0)
    best_h = b0.copy()

    # Phase 1: Targeted SA on initial binary
    t1 = deadline - 400
    b1, c5_1 = targeted_sa(b0, n0, 500000, rng, t1, T0=5.0, T1=0.03, n_cand=6)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()

    # Phase 2: Block SA
    if time.time() < deadline - 200:
        b2, c5_2 = block_sa(best_h, n0, 30000, rng, min(deadline - 200, time.time() + 120), block_max=3)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()

    # Phase 3: More targeted SA with lower temperature
    if time.time() < deadline - 200:
        b3, c5_3 = targeted_sa(best_h, n0, 300000, rng, min(deadline - 200, time.time() + 180),
                               T0=2.0, T1=0.01, n_cand=8)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()

    # Phase 4: Continuous optimization
    if time.time() < deadline - 80:
        h_cont, c5_cont = continuous_optimize(best_h, n0, deadline - 80)
        if c5_cont < best_c5:
            best_c5 = c5_cont
            best_h = h_cont.copy()

    # Phase 5: Re-binaryize and polish
    if time.time() < deadline - 60:
        b4 = to_binary(best_h, n0)
        b5, c5_5 = targeted_sa(b4, n0, 100000, rng, min(deadline - 60, time.time() + 100),
                               T0=1.5, T1=0.01, n_cand=8)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = b5.copy()

    # Phase 6: Try different resolutions
    for n_try in [480, 720, 1200, 1440, 1920]:
        if time.time() > deadline - 80:
            break
        if n_try == n0:
            continue

        if n_try < len(best_h):
            old_x = np.linspace(0.0, 2.0, len(best_h))
            new_x = np.linspace(0.0, 2.0, n_try)
            h_try = np.interp(new_x, old_x, best_h)
        else:
            h_try = upsample_h(best_h, n_try)
        h_try = make_feasible(h_try, n_try)
        b_try = to_binary(h_try, n_try)

        alloc = min(150, deadline - 80 - time.time())
        if alloc < 30:
            continue
        b_res, c5_res = targeted_sa(b_try, n_try, 200000, rng, time.time() + alloc,
                                    T0=3.0, T1=0.03, n_cand=6)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()

    # Phase 7: Final continuous polish
    if time.time() < deadline - 40:
        n_final = len(best_h)
        h_final, c5_final = continuous_optimize(best_h, n_final, deadline - 20)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 8: Final binary polish
    if time.time() < deadline - 10:
        n_final = len(best_h)
        b_final = to_binary(best_h, n_final)
        b_f, c5_f = targeted_sa(b_final, n_final, 50000, rng, deadline,
                                T0=1.0, T1=0.005, n_cand=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = b_f.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
