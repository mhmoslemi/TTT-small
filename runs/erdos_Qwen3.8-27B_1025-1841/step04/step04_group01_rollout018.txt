Let me vectorize the gradient computation:

For lag >= 0:
- grad[:n-lag] += (1-h[lag:])  → this is O(n)
- grad[lag:] -= h[:-lag]  → this is O(n)

For lag < 0 (k = -lag):
- grad[k:] += (1-h[:n-k])  → O(n)
- grad[:n-k] -= h[k:]  → O(n)

So for each active lag, it's O(n). With |active| active lags, total is O(|active| * n).

With |active| = 20 and n = 960, that's 19200 operations per iteration. Very fast!

For the coordinate descent, the simplest effective approach:
- For each position i, try a few candidate values: {0, 1, current, current±0.01, current±0.05}
- Evaluate C₅ for each

def update_corr_delta(corr, h, i, delta, n):
    """Update corr array after h[i] changes by delta. Must be called BEFORE updating h[i]."""
    # Case 1: h[i] as first factor a[t=i]
    # k range: n-1-i to n-1+i (clamped to [0, 2n-2])
    k_start = max(0, n - 1 - i)
    k_end = min(2*n - 2, n - 1 + i)
    if k_start <= k_end:
        # result[k] += delta * (1 - h[i + k - n + 1])
        indices = i + np.arange(k_start, k_end + 1) - n + 1
        corr[k_start:k_end+1] += delta * (1.0 - h[indices])
    
    # Case 2: h[i] as second factor v[...=i], so t = i - k + n - 1
    # k range: i to i+n-1 (clamped to [0, 2n-2])
    k_start2 = max(0, i)
    k_end2 = min(2*n - 2, i + n - 1)
    if k_start2 <= k_end2:
        # result[k] += -delta * h[i - k + n - 1]
        indices2 = i - np.arange(k_start2, k_end2 + 1) + n - 1
        corr[k_start2:k_end2+1] += -delta * h[indices2]

def coordinate_descent_step(h, corr, i, candidates, n):
    """Try candidate values for h[i], return best value and updated corr if improved."""
    old = h[i]
    best_v = old
    best_max = np.max(corr)
    
    for v in candidates:
        if v < -1e-12 or v > 1 + 1e-12: continue
        v = np.clip(v, 0, 1)
        if abs(v - old) < 1e-12: continue
        
        delta = v - old
        # Save affected corr values
        k1s, k1e = max(0, n-1-i), min(2*n-2, n-1+i)
        k2s, k2e = max(0, i), min(2*n-2, i+n-1)
        all_ks = np.unique(np.concatenate([np.arange(k1s, k1e+1), np.arange(k2s, k2e+1)]))
        saved = corr[all_ks].copy()
        
        # Apply
        update_corr_delta(corr, h, i, delta, n)
        
        # Check max: only affected indices could change the max
        # New max is max(old_max excluding affected, max of affected)
        others_max = np.max(np.delete(corr, all_ks)) if len(all_ks) < len(corr) else 0
        affected_max = np.max(corr[all_ks])
        new_max = max(others_max, affected_max)

        # For simplicity, skip the sum constraint in the local search and fix at the end
        # Actually, changing h[i] by delta changes sum by delta. We need to compensate.
        # Option 1: don't enforce sum exactly, just check at end
        # Option 2: distribute -delta among other free elements
        
        if new_max < best_max - 1e-12:
            best_max = new_max
            best_v = v
            # Keep the modified corr
        else:
            # Restore
            corr[all_ks] = saved
    
    if best_v != old:
        h[i] = best_v
    return h, corr, best_v

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):

        else:

                    np.arange(k3s, k3e + 1), np.arange(k4s, k4e + 1)
                ]))
                saved = corr_b[all_ks].copy()
                update_corr_delta(corr_b, h_b, i, d_i, n)
                update_corr_delta(corr_b, h_b, j, d_j, n)
                new_max = float(np.max(corr_b))
                if new_max < float(np.max(saved[all_ks[0] == all_ks[0]]) * 0 + np.max(corr_b) * 0 + (np.max(saved) if len(saved) > 0 else 0)):
                    pass
                # Simpler: check if new max of full corr is less than old max
                old_max = float(np.max(corr_b[~np.isin(np.arange(len(corr_b)), all_ks)]) if len(all_ks) < len(corr_b) else float(np.max(corr_b)))
                # Actually just check the full new corr max
                full_new_max = float(np.max(corr_b))
                orig_max = float(np.max(np.concatenate([saved, corr_b[[k for k in range(len(corr_b)) if k not in set(all_ks)]]])) if len(all_ks) < len(corr_b) else float(np.max(corr_b)))
                # Too complex. Just recompute:
                h_b[all_ks == all_ks] = h_b[all_ks == all_ks]  # no-op
                # Let me just check: if the new max over affected indices is still below the global old max
                affected_new_max = float(np.max(corr_b[all_ks]))
                # If affected new max < global old max (minus small epsilon), it's an improvement
                global_old_max = float(np.max(corr_b))  # this is now the NEW max
                # Actually I overwrote corr_b. Let me redo this more carefully.
                corr_b[all_ks] = saved
                # Try a simpler check: recompute
                h_try = h_b.copy()
                h_try[i] = 0.0
                h_try[j] = 1.0
                h_try = make_feasible(h_try, n)
                c_try = float(np.max(np.correlate(h_try, 1.0 - h_try, mode='full')) * (2.0 / n))
                if c_try < best_c5 - 1e-12:
                    h_b = h_try
                    corr_b = np.correlate(h_b, 1.0 - h_b, mode='full')
                    best_c5 = c_try
                    best_h = h_b.copy()
                    improved_any = True
                    break
        if improved_any:
            h = h_b.copy()
        if not improved_any:
            break

    # Phase 3: Fine coordinate descent
    for cd_round in range(30):
        if tl() < 150:
            break
        h_cd = h.copy()
        improved_any = False
        for i in range(n):
            if tl() < 100:
                break
            old = h_cd[i]
            best_v = old
            best_local = compute_c5(h_cd)
            trials = [0.0, 1.0, 0.25, 0.5, 0.75]
            for s in [0.005, 0.01, 0.02, 0.05, 0.1]:
                for sg in [-1.0, 1.0]:
                    v = old + sg * s
                    if 0.0 <= v <= 1.0:
                        trials.append(v)
            for v in trials:
                if abs(v - old) < 1e-12:
                    continue
                trial = h_cd.copy()
                trial[i] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_local - 1e-14:
                    best_local = c
                    best_v = v
            if abs(best_v - old) > 1e-12:
                h_cd[i] = best_v
                h_cd = make_feasible(h_cd, n)
                improved_any = True
                if compute_c5(h_cd) < best_c5:
                    best_c5 = compute_c5(h_cd)
                    best_h = h_cd.copy()
        h = h_cd
        if not improved_any:
            break

    # Phase 4: SLSQP with soft-max at current resolution
    if tl() > 80:
        _n = len(h)
        bounds = [(0.0, 1.0)] * _n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _n / 2.0,
                 'jac': lambda x: np.ones(_n)}]
        for T_val in [500.0, 1000.0]:
            if tl() < 50:
                break
            try:
                res = minimize(lambda x: _obj_full(x, T_val, _n), h,
                               method='SLSQP', bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-16})
                c_res = compute_c5(res.x)
                if c_res < best_c5:
                    best_c5 = c_res
                    best_h = make_feasible(res.x, _n)
                    h = best_h.copy()
            except Exception:
                pass

    # Phase 5: Shift-targeted SLSQP
    if tl() > 60:
        _n = len(h)
        corr_t = np.correlate(h, 1.0 - h, mode='full')
        mc_t = np.max(corr_t)
        thr_t = mc_t - 0.005 * mc_t
        top_idx = np.where(corr_t >= thr_t)[0]
        if len(top_idx) > 20:
            top_idx = top_idx[np.argsort(-corr_t[top_idx])[:20]]
        bounds = [(0.0, 1.0)] * _n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _n / 2.0,
                 'jac': lambda x: np.ones(_n)}]
        try:
            res = minimize(lambda x: _obj_top(x, 400.0, _n, top_idx), h,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-16})
            c_res = compute_c5(res.x)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = make_feasible(res.x, _n)
                h = best_h.copy()
        except Exception:
            pass

    # Phase 6: Upsample to 2n and optimize
    if tl() > 300:
        n_up = len(h) * 2
        if n_up <= 2000:
            old_x = np.linspace(0.0, 2.0, len(h))
            new_x = np.linspace(0.0, 2.0, n_up)
            h_up = np.interp(new_x, old_x, h)
            h_up = make_feasible(h_up, n_up)
            c_up = compute_c5(h_up)
            h_up_opt = h_up.copy()
            bounds_up = [(0.0, 1.0)] * n_up
            cons_up = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_up / 2.0,
                        'jac': lambda x: np.ones(n_up)}]
            for T_val in [200.0, 500.0]:
                if tl() < 100:
                    break
                try:
                    res = minimize(lambda x: _obj_full(x, T_val, n_up), h_up_opt,
                                   method='SLSQP', bounds=bounds_up, constraints=cons_up,
                                   options={'maxiter': 300, 'ftol': 1e-15})
                    c_res = compute_c5(res.x)
                    if c_res < c_up:
                        c_up = c_res
                        h_up_opt = make_feasible(res.x, n_up)
                except Exception:
                    pass
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up_opt
                h = h_up_opt.copy()

    # Phase 7: Random restarts + SLSQP
    for restart in range(6):
        if tl() < 60:
            break
        _n = len(h)
        scales = [0.01, 0.02, 0.03, 0.02, 0.05, 0.01]
        h_try = make_feasible(h + rng.randn(_n) * scales[restart], _n)
        bounds = [(0.0, 1.0)] * _n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - _n / 2.0,
                 'jac': lambda x: np.ones(_n)}]
        try:
            res = minimize(lambda x: _obj_full(x, 300.0, _n), h_try,
                           method='SLSQP', bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15})
            c_res = compute_c5(res.x)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = make_feasible(res.x, _n)
                h = best_h.copy()
        except Exception:
            pass

    # Phase 8: Final ultra-fine coordinate descent
    if tl() > 20:
        _n = len(h)
        for rd in range(3):
            for i in range(_n):
                if tl() < 5:
                    break
                old = h[i]
                best_v = old
                best_c = best_c5
                for s in [0.0005, 0.001, 0.002, 0.003, 0.005, 0.01]:
                    for sg in [-1.0, 1.0]:
                        v = old + sg * s
                        if 0.0 <= v <= 1.0:
                            trial = h.copy()
                            trial[i] = v
                            trial = make_feasible(trial, _n)
                            c = compute_c5(trial)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_v = v
                if abs(best_v - old) > 1e-14:
                    h[i] = best_v
                    h = make_feasible(h, _n)
                    best_c5 = best_c
                    best_h = h.copy()

    # Final check
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
