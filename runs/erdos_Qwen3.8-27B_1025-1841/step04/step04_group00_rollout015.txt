def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    for _ in range(50):
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def eval_c5_at_t(t, corr0, alpha, zero_k):
    vals = corr0 + alpha * t
    vals[zero_k] = corr0[zero_k] + t - t * t
    return np.max(vals)

            old_val = h[j]
            h[j] = 0.0

            alpha = corr1 - corr0

        if pass_improved == 0:
