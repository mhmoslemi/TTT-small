    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    for rnd in range(n_rounds):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        improved = False
        n_tries = 500
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones = np.append(ones[np.delete(ones, ii)], oj)
                zeros = np.append(zeros[np.delete(zeros, jj)], oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def binary_from_smooth(h_smooth, n):
    h = h_smooth.astype(float).copy()
    sorted_h = np.sort(h)[::-1]
    threshold = sorted_h[n // 2 - 1] if n // 2 > 0 else 0.5
    h_bin = (h >= threshold).astype(float)
    n_ones = int(np.sum(h_bin))
    if n_ones > n // 2:
        one_idx = np.where(h_bin == 1)[0]
        sort_order = np.argsort(h[one_idx])
        for k in range(n_ones - n // 2):
            h_bin[one_idx[sort_order[k]]] = 0.0
    elif n_ones < n // 2:
        zero_idx = np.where(h_bin == 0)[0]
        sort_order = np.argsort(-h[zero_idx])
        for k in range(n // 2 - n_ones):
            h_bin[zero_idx[sort_order[k]]] = 1.0
    return h_bin

def make_quasiperiodic(n, f1, f2, a1, a2, theta1, theta2, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + a1 * np.cos(2 * np.pi * f1 * x / 2.0 + theta1) + a2 * np.cos(2 * np.pi * f2 * x / 2.0 + theta2)
    h += 0.05 * rng.randn(n)
    return make_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Multi-resolution p-norm optimization
    for n_res in [256, 512, 1024, 2048]:
        if tl() < 80:
            break
        h_curr = upsample_h(best_h, n_res)
        for p in [10, 30, 100]:
            if tl() < 30:
                break
            h_opt = lbgfb_pnorm(h_curr, n_res, p, 1000.0, 200, min(tl() - 20, 120))
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_curr = h_opt
        # Soft-max refinement
        for T in [50, 200, 800]:
            if tl() < 20:
                break
            h_opt = lbgfb_softmax(h_curr, n_res, T, 500.0, 150, min(tl() - 15, 80))
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_curr = h_opt

    # Phase 2: Binary local search at n=2048
    if tl() > 200:
        n_bin = 2048
        h_b = upsample_h(best_h, n_bin)
        h_b = binary_from_smooth(h_b, n_bin)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()

        # Guided search (target the max shift)
        h_g, c5_g = guided_binary_swap(h_b, n_bin, rng, 30, min(tl() - 60, 120))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

        # Annealing-based search
        h_a, c5_a = binary_anneal(h_g, n_bin, rng, 300000, min(tl() - 60, 250), T_init=0.003)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Phase 3: Smooth + binary alternation at high res
    if tl() > 100:
        n_alt = 2048
        h_alt = upsample_h(best_h, n_alt)
        for rnd in range(4):
            if tl() < 40:
                break
            h_opt = lbgfb_softmax(h_alt, n_alt, 300, 500, 150, min(tl() - 30, 60))
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            h_alt = h_opt
            if tl() < 40:
                break
            h_b = binary_from_smooth(h_alt, n_alt)
            c5_b = compute_c5(h_b)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_b.copy()
            h_g, c5_g = guided_binary_swap(h_b, n_alt, rng, 15, min(tl() - 30, 60))
            if c5_g < best_c5:
                best_c5 = c5_g
                best_h = h_g.copy()
            h_alt = upsample_h(h_g, n_alt)

    # Phase 4: Diverse quasi-periodic restarts
    if tl() > 80:
        n_r = 1024
        golden = (1 + np.sqrt(5)) / 2.0
        for f1 in [3.0, 5.0, 7.0, 11.0, 13.0]:
            for f2 in [f1 * golden % 50, f1 * 1.618, f1 * 2.414, f1 * 0.707]:
                if tl() < 20:
                    break
                a1 = 0.25 + 0.1 * rng.rand()
                a2 = 0.1 + 0.1 * rng.rand()
                th1 = rng.rand() * 2 * np.pi
                th2 = rng.rand() * 2 * np.pi
                h_qp = make_quasiperiodic(n_r, f1, f2, a1, a2, th1, th2, rng)
                h_opt = lbgfb_pnorm(h_qp, n_r, 30, 1000, 150, min(tl() - 10, 45))
                c5 = compute_c5(h_opt)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                # Also try binary
                h_b = binary_from_smooth(h_opt, n_r)
                c5_b = compute_c5(h_b)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
                    h_g, c5_g = guided_binary_swap(h_b, n_r, rng, 10, min(tl() - 10, 30))
                    if c5_g < best_c5:
                        best_c5 = c5_g
                        best_h = h_g.copy()

    # Phase 5: High-resolution refinement
    if tl() > 60:
        n_hr = 4096
        h_hr = upsample_h(best_h, n_hr)
        c5_hr = compute_c5(h_hr)
        if c5_hr < best_c5:
            best_c5 = c5_hr
            best_h = h_hr.copy()
        h_opt = lbgfb_pnorm(h_hr, n_hr, 50, 500, 150, min(tl() - 15, 120))
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
        h_opt = lbgfb_softmax(h_opt, n_hr, 500, 500, 150, min(tl() - 10, 80))
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Phase 6: Final binary polish at best resolution
    if tl() > 30:
        n_f = len(best_h)
        if n_f > 2048:
            n_f = 2048
            h_f = upsample_h(best_h, n_f)
        else:
            h_f = best_h.copy()
        h_b = binary_from_smooth(h_f, n_f)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        h_g, c5_g = guided_binary_swap(h_b, n_f, rng, 20, min(tl() - 10, 60))
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()

    # Phase 7: Final smooth squeeze
    if tl() > 10:
        n_s = len(best_h)
        h_final = lbgfb_softmax(best_h, n_s, 1000, 500, 80, max(tl() - 5, 10))
        c5 = compute_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
