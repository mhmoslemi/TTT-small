def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-14:

        if diff > 0:

        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):

        idx = zeros[np.argmax(h[zeros])]

def slsqp_obj(x, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    # Soft max with large temperature for near-exact max
    T = 50.0
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_sum(x, n):
    return np.sum(x) - n / 2.0

def slsqp_sum_jac(x, n):
    return np.ones(len(x))

    _G_N = n
    _G_T = T_soft
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: slsqp_sum(x, n), 'jac': lambda x: slsqp_sum_jac(x, n)}]
    res = minimize(_slsqp_obj_impl, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s):

    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:

                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

Minimize t
Subject to:

- 0 ≤ h[i] ≤ 1 for all i
- sum_i h[i] * dx = 1

The constraint sum_i h[i] * (1 - h[i+k]) * dx ≤ t is equivalent to:

The term sum_i h[i]*h[i+k] is a quadratic form. Is it convex or concave in h? It's h^T A h where A has 1s at positions (i, i+k) and (i+k, i). This is not convex in general (it's indefinite).

_G_N = 0
_G_T = 0.0

def _g_obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
