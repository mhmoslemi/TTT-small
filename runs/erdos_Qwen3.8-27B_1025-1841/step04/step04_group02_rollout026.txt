import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    stall = 0
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
                stall = 0
            else:
                stall += 1
        else:
            b[i] = 1
            b[j] = 0
            stall += 1
    return best_b, best_c5

def coord_descent_exact(h, n, max_rounds, start_time, budget_s):
    """Coordinate descent with smart candidate selection."""
    best_c5 = compute_c5(h)
    improved = True

        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:

                h = make_feasible(h, n)
                best_c5 = best_c

        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    h_init = make_feasible(h_init, len(h_init))
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    n0 = len(h_init)

    # Phase 1: Binary SA
    b0, c5_b0 = binary_sa(h_init, n0, 200000, 0.012, 0.000005, rng)
    if c5_b0 < best_c5:
        best_c5 = c5_b0
        best_h = b0.copy()

    # Phase 2: Fast coordinate descent
    h_cd, c5_cd = coord_descent_fast(best_h.copy(), n0, 30)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Phase 3: SLSQP with decreasing temperature
    remaining = BUDGET - (time.time() - START)
    if remaining > 100:
        for T in [200.0, 50.0, 15.0, 5.0]:
            remaining = BUDGET - (time.time() - START)
            if remaining < 60:
                break
            try:
                x_res = slsqp_run(best_h, n0, T, 200, 1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n0)
            except Exception:
                pass

    # Phase 4: Second round of coordinate descent
    remaining = BUDGET - (time.time() - START)
    if remaining > 60:
        h_cd, c5_cd = coord_descent_fast(best_h.copy(), n0, 20)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Binary SA again from continuous solution
    remaining = BUDGET - (time.time() - START)
    if remaining > 150:
        b1, c5_b1 = binary_sa(best_h, n0, 150000, 0.006, 0.00001, rng)
        if c5_b1 < best_c5:
            best_c5 = c5_b1
            best_h = b1.copy()
            # Polish
            remaining = BUDGET - (time.time() - START)
            if remaining > 80:
                h_cd, c5_cd = coord_descent_fast(best_h.copy(), n0, 15)
                if c5_cd < best_c5:
                    best_c5 = c5_cd
                    best_h = h_cd.copy()
                remaining = BUDGET - (time.time() - START)
                if remaining > 60:
                    try:
                        x_res = slsqp_run(best_h, n0, 8.0, 200, 1e-15)
                        c5_res = compute_c5(x_res)
                        if c5_res < best_c5:
                            best_c5 = c5_res
                            best_h = make_feasible(x_res, n0)
                    except Exception:
                        pass

    # Phase 6: Try different resolution (higher)
    remaining = BUDGET - (time.time() - START)
    if remaining > 400:
        n1 = n0 * 2
        h_up = upsample_h(best_h, n1)
        b_up, c5_up = binary_sa(h_up, n1, 100000, 0.006, 0.00001, rng)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = b_up.copy()
        remaining = BUDGET - (time.time() - START)
        if c5_up < best_c5 or remaining > 200:
            h_cd, c5_cd = coord_descent_fast(best_h.copy(), n1, 10)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            remaining = BUDGET - (time.time() - START)
            if remaining > 80:
                try:
                    x_res = slsqp_run(best_h, n1, 20.0, 150, 1e-14)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(x_res, n1)
                except Exception:
                    pass

    # Phase 7: Try lower resolution
    remaining = BUDGET - (time.time() - START)
    if remaining > 250 and n0 > 600:
        n2 = n0 // 2
        h_lo = upsample_h(best_h, n2)
        b_lo, c5_lo = binary_sa(h_lo, n2, 120000, 0.01, 0.00002, rng)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = b_lo.copy()
        remaining = BUDGET - (time.time() - START)
        if remaining > 80:
            h_cd, c5_cd = coord_descent_fast(best_h.copy(), n2, 10)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()
            remaining = BUDGET - (time.time() - START)
            if remaining > 60:
                try:
                    x_res = slsqp_run(best_h, n2, 10.0, 200, 1e-14)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(x_res, n2)
                except Exception:
                    pass

    # Phase 8: Final polish on best solution
    remaining = BUDGET - (time.time() - START)
    n_pts = len(best_h)
    if remaining > 40:
        h_cd, c5_cd = coord_descent_fast(best_h.copy(), n_pts, 10)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    remaining = BUDGET - (time.time() - START)
    if remaining > 50:
        try:
            x_res = slsqp_run(best_h, len(best_h), 3.0, 150, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, len(best_h))
        except Exception:
            pass
    remaining = BUDGET - (time.time() - START)
    if remaining > 30:
        h_cd, c5_cd = coord_descent_fast(best_h.copy(), len(best_h), 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
