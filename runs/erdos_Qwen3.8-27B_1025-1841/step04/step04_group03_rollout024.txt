```python
def compute_delta(h, i, j, n):
    """Compute change in correlation when h[i]:0->1 and h[j]:1->0."""
    delta = np.zeros(2*n-1)
    # For position i (0->1, b-a=1):
    # Term 1: (1-h[i-s]) valid if 0<=i-s<n, i.e., -n<i-s<0... wait
    # s ranges from -(n-1) to n-1
    # i-s in [0,n-1] means s in [i-(n-1), i]
    # Term 2: h[i+s] valid if 0<=i+s<n, i.e., s in [-i, n-1-i]
    
    for s in range(-(n-1), n):
        dc = 0.0
        # From position i (0->1):
        if 0 <= i - s < n:
            dc += (1.0 - h[i-s])
        if 0 <= i + s < n:
            dc -= h[i+s]
        # From position j (1->0, b-a=-1):
        if 0 <= j - s < n:
            dc -= (1.0 - h[j-s])
        if 0 <= j + s < n:
            dc += h[j+s]
        delta[s + n - 1] = dc
    return delta
```

```python
def compute_delta_vec(h, i, j, n):
    """Vectorized computation of correlation delta."""
    s = np.arange(-(n-1), n)  # shape (2n-1,)
    delta = np.zeros(2*n-1)
    
    # From position i (0->1):
    mask1 = (0 <= i - s) & (i - s < n)
    delta[mask1] += (1.0 - h[i - s[mask1]])
    mask2 = (0 <= i + s) & (i + s < n)
    delta[mask2] -= h[i + s[mask2]]
    
    # From position j (1->0):
    mask3 = (0 <= j - s) & (j - s < n)
    delta[mask3] -= (1.0 - h[j - s[mask3]])
    mask4 = (0 <= j + s) & (j + s < n)
    delta[mask4] += h[j + s[mask4]]
    
    return delta
```

```python
        h = np.clip(h, 0.0, 1.0)

def binarize(h, n):
    """Convert to binary, maintaining exactly n/2 ones."""
    b = np.zeros(n)
    # Sort by value, set top n/2 to 1
    idx = np.argsort(h)[::-1]
    b[idx[:n//2]] = 1.0
    return b

def fast_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    """Simulated annealing with O(n) cost per iteration."""
    b = binarize(h, n)
    target = n // 2
    # Ensure exact count
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        b[zeros[rng.randint(len(ones))]] = 1
    
    # Precompute correlation
    corr = np.correlate(b, 1.0 - b, mode='full').astype(float)
    curr_max = np.max(corr)
    best_max = curr_max
    best_b = b.copy()
    dx = 2.0 / n
    
    # Precompute shift array
    s = np.arange(-(n-1), n)
    
    n_acc = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if it % 10000 == 0 and it > 0:
            # Periodic cooling
            pass
        frac = min(1.0, float(it) / max_iters)
        T = T0 * (T1 / T0) ** frac
        
        # Pick random swap
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        
        # Compute delta in O(n)
        delta = np.zeros(2*n-1)
        
        # Position i goes 0->1 (b-a = 1):
        # (1-h[i-s]) for valid i-s, minus h[i+s] for valid i+s
        mask_is = (i - s >= 0) & (i - s < n)
        mask_is2 = (i + s >= 0) & (i + s < n)
        delta[mask_is] += (1.0 - b[i - s[mask_is]])
        delta[mask_is2] -= b[i + s[mask_is2]]
        
        # Position j goes 1->0 (b-a = -1):
        mask_js = (j - s >= 0) & (j - s < n)
        mask_js2 = (j + s >= 0) & (j + s < n)
        delta[mask_js] -= (1.0 - b[j - s[mask_js]])
        delta[mask_js2] += b[j + s[mask_js2]]
        
        new_corr = corr + delta
        new_max = np.max(new_corr)
        
        if new_max <= curr_max or (T > 1e-12 and rng.random() < np.exp(-(new_max - curr_max) / T)):
            # Accept
            b[i] = 1
            b[j] = 0
            corr = new_corr
            curr_max = new_max
            n_acc += 1
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
        else:
            # Reject
            pass
    
    return best_b, best_max * dx
```

```python
    dx = 2.0 / n

    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    return np.sum(x) - len(x) / 2.0
```
