    target_sum = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    prev_h = h.copy()
    prev_g = None
    alpha = 0.1
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        max_idx = np.argmax(corr)
        g = subgrad_at_max(h, max_idx)
        
        # BB step size
        if prev_g is not None:
            s = h - prev_h
            y = g - prev_g
            ys = np.dot(y, s)
            ss = np.dot(s, s)
            if ss > 1e-20 and abs(ys) > 1e-20:
                alpha = ss / ys if ys > 0 else 2 * ss / (ys + ss)
                alpha = np.clip(alpha, 0.01, 1.0)
        
        prev_h = h.copy()
        prev_g = g.copy()
        
        h_new = project_box_sum(h - alpha * g * dx, target_sum)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

Let me code this up properly. The key is the analytical gradient for the smooth max.

For np.correlate(h, v, mode='full') where v = 1-h:
The result c has length 2n-1.
c[k] = sum_j h[j + k - (n-1)] * v[j] for valid j

Actually, the scipy documentation says:
c_k = sum_n a_{n+k} * conj(v_n)

For real signals: c_k = sum_n a_{n+k} * v_n

where the sum is over all n where both a_{n+k} and v_n are defined.

So corr[k] = sum_n h[n+k] * (1-h[n])

The valid range of n depends on k:
- We need 0 <= n < n (for h[n])
- We need 0 <= n+k < n (for h[n+k]), i.e., -k <= n <= n-1-k

So for k >= 0: n ranges from 0 to n-1-k
For k < 0: n ranges from -k to n-1

The gradient d(corr[k])/dh[i]:
- h[i] appears in h[n+k] when n+k = i, i.e., n = i-k: contributes (1-h[n]) = (1-h[i-k])
  Valid when 0 <= i-k and i-k is in valid range for n
- h[i] appears in (1-h[n]) when n = i: contributes -h[n+k] = -h[i+k]
  Valid when 0 <= i+k < n

For k = 0:
d/dh[i] = (1-h[i]) - h[i] = 1 - 2h[i]

For k > 0:
d/dh[i] = (1-h[i-k]) * [i >= k] - h[i+k] * [i < n-k]

For k < 0 (let m = -k > 0):
d/dh[i] = (1-h[i+m]) * [i < n-m] - h[i-m] * [i >= m]

This matches my earlier derivation. Good.

Now, the soft max gradient:
The smooth max is: S(x) = (1/T) * log(sum_k exp(T * f_k(x)))
dS/dx_i = sum_k (exp(T*f_k) / sum_j exp(T*f_j)) * df_k/dx_i
        = sum_k w_k * df_k/dx_i

where w_k = exp(T*(f_k - f_max)) / sum_j exp(T*(f_j - f_max))

For large T, w concentrates on the argmax. But for numerical stability, I'll use the shifted version.

```python
from scipy.optimize import minimize as scminimize

def lbfgsb_run(h, n, T, mu, maxiter, ftol):
    target_sum = n / 2.0
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        max_s = np.max(shifted)
        log_sm = mc + (max_s + np.log(np.sum(np.exp(shifted - max_s)))) / T
        return log_sm * dx + mu * (np.sum(x) - target_sum)**2
    
    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        max_s = np.max(shifted)
        e = np.exp(shifted - max_s)
        w = e / e.sum()
        
        g = np.zeros(n)
        # Only compute for shifts with significant weight
        thresh = w.max() * 1e-6
        active = np.where(w > thresh)[0]
        for k in active:
            shift = int(k) - (n - 1)
            if shift == 0:
                sg = 1.0 - 2.0 * x
            elif shift > 0:
                sg = np.zeros(n)
                sg[shift:] += (1.0 - x[:n-shift])
                sg[:n-shift] -= x[shift:]
            else:
                s = -shift
                sg = np.zeros(n)
                sg[:n-s] += (1.0 - x[s:])
                sg[s:] -= x[:n-s]
            g += w[k] * sg
        g *= dx
        g += 2 * mu * (np.sum(x) - target_sum) * np.ones(n)
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = scminimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                     bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
    return res.x, res.fun
```

import numpy as np
from scipy.optimize import minimize as scminimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def subgrad_at_shift(h, shift):
    n = len(h)
    g = np.zeros(n)
    if shift == 0:
        g[:] = 1.0 - 2.0 * h
    elif shift > 0:
        g[shift:] += (1.0 - h[:n - shift])
        g[:n - shift] -= h[shift:]
    else:
        s = -shift
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def smooth_max_and_weights(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    max_s = np.max(shifted)
    e = np.exp(shifted - max_s)
    w = e / e.sum()
    val = mc + (max_s + np.log(e.sum())) / T
    return val, w, mc

def lbfgsb_smooth(h, n, T, mu, maxiter, ftol, start_time, budget_s):
    target_sum = n / 2.0
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        val, _, _ = smooth_max_and_weights(corr, T)
        return val * dx + mu * (np.sum(x) - target_sum) ** 2

    def grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        _, w, _ = smooth_max_and_weights(corr, T)
        g = np.zeros(n)
        thresh = w.max() * 1e-8
        active = np.where(w > thresh)[0]
        for idx in active:
            shift = int(idx) - (n - 1)
            g += w[idx] * subgrad_at_shift(x, shift)
        g *= dx
        g += 2.0 * mu * (np.sum(x) - target_sum)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = scminimize(obj, h.copy(), jac=grad, method='L-BFGS-B',
                         bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-12})
        return res.x
    except Exception:
        return h

def binary_sa(h, n, max_iters, T0, T1, rng):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[rng.randint(len(ones))]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[rng.randint(len(zeros))]
        b[idx] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** (1.0 - (1.0 - T1 / T0) * frac) if T0 > 0 else T1
        if it % 2 == 0:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0, 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i], b[j] = 1, 0
        else:
            bs = 1 + rng.randint(0, 5)
            if bs >= n:
                bs = 1
            start = rng.randint(0, n - bs)
            seg = b[start:start + bs]
            ones_in = np.where(seg == 1)[0]
            zeros_in = np.where(seg == 0)[0]
            if len(ones_in) == 0 or len(zeros_in) == 0:
                continue
            sc = min(len(ones_in), len(zeros_in))
            co = rng.choice(ones_in, size=sc, replace=False)
            cz = rng.choice(zeros_in, size=sc, replace=False)
            b[start + co] = 0
            b[start + cz] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[start + co] = 1
                b[start + cz] = 0
    return best_b, best_c5

def proj_subgrad(h, n, steps, step_size, start_time, budget_s):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(steps):
        if time.time() - start_time > budget_s - 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - (n - 1)
        g = subgrad_at_shift(h, shift)
        h = project_box_sum(h - step_size * g, target_sum)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    return project_box_sum(h, n / 2.0)

def coord_descent(h, n, max_rounds, start_time, budget_s):
    target_sum = n / 2.0
    best_c5 = compute_c5(h)
    h = h.copy()
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        if time.time() - start_time > budget_s - 20:
            break
        for j in range(n):
            if time.time() - start_time > budget_s - 15:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = project_box_sum(trial, target_sum)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = project_box_sum(h, target_sum)
                best_c5 = best_c
                improved = True
    return h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
