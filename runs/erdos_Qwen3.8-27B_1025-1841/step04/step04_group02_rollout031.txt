import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    size = 2 * n
    bh = 1.0 - h
    Fh = np.fft.rfft(h, size)
    Fbh = np.fft.rfft(bh, size)
    r = np.fft.irfft(Fh * np.conj(Fbh))
    max_val = max(np.max(r[:n]), np.max(r[n+1:]))
    return float(max_val * dx)

def compute_c5_corr(h):
    """Returns (c5_value, correlation_array) for use in continuous optimization"""
    n = len(h)
    dx = 2.0 / n
    size = 2 * n
    bh = 1.0 - h
    Fh = np.fft.rfft(h, size)
    Fbh = np.fft.rfft(bh, size)
    r = np.fft.irfft(Fh * np.conj(Fbh))
    # Valid lags: 0 to n-1 and -(n-1) to -1
    # r[0:n] for positive lags, r[n+1:size] for negative lags
    valid = np.concatenate([r[:n], r[n+1:]])
    return valid, dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa_fast(h, n, max_iters, T0, T1, rng, budget_s, start_time):
    """Binary SA with FFT-based C5 evaluation"""
    b = to_binary(h, n)
    curr_c5 = compute_c5_fast(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for it in range(max_iters):
        if it % 10000 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 10:
                break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5_fast(b)
        
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    
    return best_b, best_c5

The key question is: what's the best strategy to beat 0.38085?

The previous solution got 0.380934. The gap is 0.000084. That's small but achievable.

With FFT, we can do much more SA iterations. The main improvements should come from:

Let me also consider: should I use block moves instead of single-point swaps? Block moves (swapping a contiguous block of ones with an adjacent block of zeros) can be more effective at large-scale rearrangements. But they're more expensive to evaluate (need to recompute C5 from scratch anyway with FFT).

Actually with FFT, the cost of evaluation doesn't depend on how many bits we change, so block moves are free! Let me add block moves.

For block moves:
- Pick a contiguous segment of length L (say 3-10)
- If the segment has more ones than zeros, swap some ones to adjacent zeros
- Or: pick two segments and swap their content

Actually, the simplest block move: pick two positions i < j, and swap the values at i and j (which is what we're already doing for binary). A more powerful move: pick a block [a, b) and shift it by some amount.

Let me keep it simple: single-bit swaps are fine with enough iterations. The reheat mechanism handles getting stuck.

Let me also think about the SLSQP step. The log-sum-exp smoothing parameter T controls how "sharp" the approximation to the max is. Large T → closer to true max. But large T makes the problem harder for the optimizer (more peaked).

I'll use a geometric cooling of T: start with small T (smooth) and increase to large T (sharp).

Actually, for SLSQP, I think using a moderate T (like 50-200) and then doing a final "projection" step (just computing the actual max) is better. The SLSQP finds a good minimum of the smoothed objective, which approximates the true objective.

import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    size = 2 * n
    bh = 1.0 - h
    Fh = np.fft.rfft(h, size)
    Fbh = np.fft.rfft(bh, size)
    r = np.fft.irfft(Fh * np.conj(Fbh))
    max_val = max(np.max(r[:n]), np.max(r[n+1:]))
    return float(max_val * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def golden_ratio_init(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    positions = np.array([int((k * phi) % 1.0 * n) for k in range(n // 2)])
    b = np.zeros(n)
    b[positions] = 1.0
    return b

def _sl_obj(x):
    n = _SL_N
    T = _SL_T
    size = 2 * n
    bh = 1.0 - x
    Fh = np.fft.rfft(x, size)
    Fbh = np.fft.rfft(bh, size)
    r = np.fft.irfft(Fh * np.conj(Fbh))
    valid = np.concatenate([r[:n], r[n+1:]])
    mc = np.max(valid)
    shifted = valid - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)

def _sl_con(x):
    return np.sum(x) - _SL_N / 2.0

def _sl_con_jac(x):
    return np.ones(_SL_N)

def slsqp_polish(h, n, T, maxiter, start_time, budget_s):
    global _SL_N, _SL_T
    remaining = budget_s - (time.time() - start_time)
    if remaining < 30:
        return h
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sl_con, 'jac': _sl_con_jac}]
    try:
        res = minimize(_sl_obj, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h

def binary_sa_fast(start_b, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = start_b.copy()
    curr_c5 = compute_c5_fast(b)
    best_c5 = curr_c5
    best_b = b.copy()
    it_since_improve = 0
    cur_T0 = T0

    ones_list = np.where(b == 1)[0]
    zeros_list = np.where(b == 0)[0]

    for it in range(max_iters):
        if it % 20000 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 10:
                break
            if it_since_improve > 150000:
                cur_T0 = min(cur_T0 * 2.0, 0.05)
                it_since_improve = 0

        frac = float(it) / max_iters
        T = cur_T0 * (T1 / cur_T0) ** frac

        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]

        b[i] = 0.0
        b[j] = 1.0
        new_c5 = compute_c5_fast(b)

        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            it_since_improve = 0
            ones_list = np.where(b == 1)[0]
            zeros_list = np.where(b == 0)[0]
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0

    return best_b, best_c5

def binary_sa_blocks(start_b, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = start_b.copy()
    curr_c5 = compute_c5_fast(b)
    best_c5 = curr_c5
    best_b = b.copy()

    for it in range(max_iters):
        if it % 20000 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 10:
                break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac

        block_size = rng.randint(2, 8)
        start = rng.randint(0, n - block_size)
        end = start + block_size
        seg = b[start:end]
        ones_in = np.where(seg == 1)[0]
        zeros_in = np.where(seg == 0)[0]
        if len(ones_in) == 0 or len(zeros_in) == 0:
            continue
        k = min(len(ones_in), len(zeros_in))
        chosen_ones = rng.choice(ones_in, size=k, replace=False)
        chosen_zeros = rng.choice(zeros_in, size=k, replace=False)
        b[start + chosen_ones] = 0.0
        b[start + chosen_zeros] = 1.0
        new_c5 = compute_c5_fast(b)

        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + chosen_ones] = 1.0
            b[start + chosen_zeros] = 0.0

    return best_b, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_descent_fast(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5_fast(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5_fast(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5
