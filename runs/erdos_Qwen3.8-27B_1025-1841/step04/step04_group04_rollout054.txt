        else:

        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = R.max()
    shifted = T * (R - mc)
    # For numerical stability
    obj = mc + np.log(np.sum(np.exp(shifted))) / T
    return obj

def softmax_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = R.max()
    shifted = T * (R - mc)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    
    # Compute gradient
    i_idx = np.arange(n)[:, None]
    k_idx = np.arange(-(n-1), n)[None, :]
    j1 = i_idx + k_idx
    j2 = i_idx - k_idx
    mask1 = (j1 >= 0) & (j1 < n)
    mask2 = (j2 >= 0) & (j2 < n)
    
    # term1[i] = sum_k w[k] * (1-h[i+k]) * mask
    G = np.zeros((n, 2*n-1))
    # For efficiency, only fill valid entries
    flat_j1 = j1[mask1]
    flat_i1 = np.where(mask1)
    G[flat_i1] = 0  # already zero
    G[flat_i1[0], flat_i1[1]] = 1.0 - h[flat_j1]
    G *= w[None, :]
    
    H = np.zeros((n, 2*n-1))
    flat_j2 = j2[mask2]
    flat_i2 = np.where(mask2)
    H[flat_i2[0], flat_i2[1]] = h[flat_j2]
    H *= w[None, :]
    
    grad = dx * (G.sum(axis=1) - H.sum(axis=1))
    return grad

np.correlate(h, 1-h, 'full')[n-1] = sum_i h[i]*(1-h[i]) = sum h[i] - sum h[i]² = n/2 - sum h[i]²

The integral approximation: (n/2 - sum h[i]²) * dx = 1 - sum h[i]² * dx

For h = 0.5 everywhere: sum h[i]² * dx = n * 0.25 * dx = 0.5. So corr_0 * dx = 0.5.

For binary h (half 0, half 1): sum h[i]² * dx = (n/2) * 1 * dx = 1. So corr_0 * dx = 0.

The minimum of corr_0 * dx is 0 (achieved by binary solutions). But we also need to control all other lags.

So there's a trade-off: making h more binary reduces lag-0 correlation but might increase other lags.

This suggests the optimal solution is near-binary! The binary_snap step in the previous code makes sense.

Let me think about a different approach: **start from a binary solution and refine**.

A binary solution with sum = n/2 has exactly n/2 ones and n/2 zeros. The correlation at lag k is:
R(k) = number of i where h[i]=1 and h[i+k]=0 (approximately, with the full correlation formula)

Actually for binary h:
R(k) * dx = (number of (i, i+k) pairs where h[i]=1, h[i+k]=0) * dx

This is related to the "transition count" in the sequence.

For a binary sequence, the number of 1→0 transitions at lag k is:
T(k) = sum_i h[i]*(1-h[i+k])

We want to minimize max_k T(k) * dx.

This is related to the "cross-correlation" of the binary sequence with itself, which connects to combinatorial designs and difference sets.

For n/2 = m ones in a sequence of length n=2m, we want to arrange the 1s so that for every shift k, the number of 1s that are followed (k steps later) by a 0 is minimized.

This is essentially a variant of the "minimum discrepancy" problem for subsets of Z_n.

    So term2[i] = np.correlate(w_rev, h, 'full')[n-1-i]
    
    And term1[i] = np.correlate(w, 1-h, 'full')[n-1-i]
    
    So: grad[i] = dx * (term1[i] - term2[i])
    = dx * (correlate(w, 1-h, 'full')[n-1-i] - correlate(w_rev, h, 'full')[n-1-i])
    
    Let me define: C1 = np.correlate(w, 1-h, 'full') (length 3n-2)
    C2 = np.correlate(w_rev, h, 'full') (length 3n-2)
    
    grad[i] = dx * (C1[n-1-i] - C2[n-1-i]) for i=0..n-1
    
    So grad = dx * (C1[n-1:n-1-n:-1][:n] - C2[n-1:n-1-n:-1][:n])
    = dx * (C1[n-1::-1][:n] - C2[n-1::-1][:n])
    
    Wait, C1 has length (2n-1) + n - 1 = 3n-2.
    C1[n-1] is at index n-1. C1[n-1-i] for i=0..n-1 gives indices n-1, n-2, ..., 0.
    
    So the relevant part is C1[0:n] reversed, i.e., C1[n-1::-1][:n].
    
    Actually: C1[n-1-i] for i=0 gives C1[n-1], for i=1 gives C1[n-2], ..., for i=n-1 gives C1[0].
    So the gradient vector is dx * (C1[n-1::-1][:n] - C2[n-1::-1][:n]).
    
    But C1[0:n] reversed = C1[n-1::-1][:n]... let me just write it as:
    idx = np.arange(n-1, -1, -1)  # [n-1, n-2, ..., 0]
    grad = dx * (C1[idx] - C2[idx])
    
    This is O(n) after computing the two correlations, which are O(n²) each (or O(n log n) with FFT).
    
    For n=960, n² = 921600, which is fast.
    
    Great, so the gradient computation is:
    1. Compute R = correlate(h, 1-h, 'full') * dx (O(n²))
    2. Compute softmax weights w (O(n))
    3. Compute C1 = correlate(w, 1-h, 'full') (O(n²))
    4. Compute C2 = correlate(w_rev, h, 'full') (O(n²))
    5. grad = dx * (C1[idx] - C2[idx]) (O(n))
    
    Total: O(n²) per gradient evaluation. For n=960, that's about 3 * 960² ≈ 2.8M operations. Very fast!

    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        if diff > 0:

        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
