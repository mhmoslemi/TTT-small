def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def to_binary(h, n):
    hb = (h > 0.5).astype(float)
    n_ones = int(np.sum(hb))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(hb == 1.0)[0]
        rng = np.random.RandomState(int(np.sum(h * 1000) % 100000))
        rng.shuffle(one_idx)
        for k in range(n_ones - target):
            hb[one_idx[k]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(hb == 0.0)[0]
        rng = np.random.RandomState(int(np.sum(h * 1000) % 100000) + 7)
        rng.shuffle(zero_idx)
        for k in range(target - n_ones):
            hb[zero_idx[k]] = 1.0
    return hb

def simulated_annealing(h_bin, n, rng, T_start, T_end, n_steps, kick_interval=5000):
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    n_cool = max(1, n_steps - 1)
    for step in range(n_steps):
        T = T_start * (T_end / T_start) ** (step / n_cool)
        if step > 0 and step % kick_interval == 0:
            n_kick = max(1, n // 50)
            kick_idx = rng.choice(n, size=2 * n_kick, replace=False)
            for ki in range(0, 2 * n_kick, 2):
                if h[kick_idx[ki]] != h[kick_idx[ki+1]]:
                    h[kick_idx[ki]], h[kick_idx[ki+1]] = h[kick_idx[ki+1]], h[kick_idx[ki]]
            current_c5 = compute_c5(h)
            ones = np.where(h > 0.5)[0].copy()
            zeros = np.where(h <= 0.5)[0].copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            continue
        if len(ones) == 0 or len(zeros) == 0:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0

        delta = c - current_c5

def block_moves(h_bin, n, rng, n_rounds=200):
    h = h_bin.copy()

        improved = False

            blen = rng.randint(2, min(8, n // 10) + 1)
            start = rng.randint(0, n - blen + 1)
            block = h[start:start+blen].copy()
            n_block_ones = int(np.sum(block))
            if n_block_ones == 0 or n_block_ones == blen:
                continue
            new_start = rng.randint(0, n - blen + 1)
            if new_start == start or abs(new_start - start) < blen:
                continue
            target_region = h[new_start:new_start+blen].copy()
            n_target_zeros = int(np.sum(1.0 - target_region))
            if n_target_zeros != n_block_ones:
                continue
            h[start:start+blen] = target_region
            h[new_start:new_start+blen] = block
            c = compute_c5(h)

                current_c5 = c
                improved = True

                    best_c5 = c
                    best_h = h.copy()
            else:
                h[start:start+blen] = block
                h[new_start:new_start+blen] = target_region
        if not improved:
            break
    return best_h, best_c5
