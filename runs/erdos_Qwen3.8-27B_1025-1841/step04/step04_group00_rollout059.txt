    w = np.exp(T * cx)
    w_sum = np.sum(w)

    # grad[j] = sum_s w[n-1+s] * [(1-h[j+s])*I(0<=j+s<n) - h[j-s]*I(0<=j-s<n)]

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0, 1, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        np.clip(h, 0, 1, out=h)
    return h

def grad_full(h, n, T=None):
    """Compute gradient of max_k corr[k] (or smoothed version)."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    if T is None or T <= 0:
        # Subgradient: just the argmax shift
        idx = np.argmax(corr)
        w = np.zeros(len(corr))
        w[idx] = 1.0
    else:
        cx = corr - corr.max()
        ex = np.exp(T * cx)
        w = ex / ex.sum()
    
    grad = np.zeros(n)
    # Only iterate over shifts with significant weight
    for idx in np.where(w > 1e-10)[0]:
        wi = w[idx]
        s = idx - (n - 1)
        j1, j2 = max(0, -s), min(n, n - s)
        if j2 > j1:
            grad[j1:j2] += wi * (1.0 - h[j1+s:j2+s])
        j12, j22 = max(0, s), min(n, n + s)
        if j22 > j12:
            grad[j12:j22] -= wi * h[j12-s:j22-s]
    return grad * (2.0 / n)

    """Projected gradient descent with backtracking line search."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    velocity = np.zeros(n)
    momentum = 0.8
    it = 0
    stall = 0
    
    while time.time() - start_time < budget:
        it += 1
        c5_cur = compute_c5(h)
        if c5_cur < best_c5 - 1e-13:
            best_h = h.copy()
            best_c5 = c5_cur
            stall = 0
        else:
            stall += 1
            if stall > 50:
                lr *= 0.5
                if lr < 1e-8:
                    break
                stall = 0
                momentum = 0.5
        
        grad = grad_full(h, n, T)
        
        # Nesterov-accelerated step
        y = h - momentum * velocity
        y = np.clip(y, 0, 1)
        grad_y = grad_full(y, n, T)
        velocity = momentum * velocity - lr * grad_y
        h_new = h + velocity
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new <= c5_cur + 1e-12:
            h = h_new
            # Occasionally boost step
            if it % 100 == 0 and lr < 0.05:

        else:

            c5_new2 = compute_c5(h_new2)

    if c5 < best_c5:

    return best_h, best_c5

def slsqp_step(h0, n, T, top_k, maxit):
    """One SLSQP optimization phase."""
    def fun(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1-h, mode='full')
        if top_k is not None:
            ti = np.argsort(corr)[-top_k:]
            tc = corr[ti]
            mc = tc.max()
            return (mc + np.log(np.sum(np.exp(T*(tc-mc))))/T) * (2.0/n)
        else:
            return (corr.max() + np.log(np.sum(np.exp(T*(corr-corr.max()))))/T) * (2.0/n)
    
    def jac(x):
        h = np.clip(x, 0, 1)
        if top_k is not None:
            corr = np.correlate(h, 1-h, mode='full')
            ti = np.argsort(corr)[-top_k:]
            w_local = np.zeros(len(corr))

        else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = float(mask.sum())
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-11:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        np.clip(h, 0.0, 1.0, out=h)
    return h

def weighted_grad(h, n, w):
    grad = np.zeros(n)
    nz = np.where(w > 1e-12)[0]
    for idx in nz:
        wi = w[idx]
        s = idx - (n - 1)
        j1, j2 = max(0, -s), min(n, n - s)
        if j2 > j1:
            grad[j1:j2] += wi * (1.0 - h[j1+s:j2+s])
        j12, j22 = max(0, s), min(n, n + s)
        if j22 > j12:
            grad[j12:j22] -= wi * h[j12-s:j22-s]
    return grad * (2.0 / n)

def softmax_weights(corr, T, top_k):
    if top_k is not None and top_k < len(corr):
        ti = np.argsort(corr)[-top_k:]
        tc = corr[ti]
        ex = np.exp(T * (tc - tc.max()))
        w = np.zeros(len(corr))
        w[ti] = ex / ex.sum()
        return w
    else:
        cx = corr - corr.max()
        ex = np.exp(T * cx)
        return ex / ex.sum()

def pgd_optimize(h0, n, budget, t0, T=50.0, lr=0.01, top_k=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    mom = 0.8
    it = 0
    stall = 0
    while time.time() - t0 < budget:
        it += 1
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_h, best_c5, stall = h.copy(), c5, 0
        else:
            stall += 1
            if stall > 40:
                lr *= 0.5
                mom = 0.5
                if lr < 1e-7:
                    break
                stall = 0
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = softmax_weights(corr, T, top_k)
        g = weighted_grad(h, n, w)
        y = h - mom * vel
        y = np.clip(y, 0, 1)
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        w_y = softmax_weights(corr_y, T, top_k)
        g_y = weighted_grad(y, n, w_y)
        vel = mom * vel - lr * g_y
        hn = make_feasible(h + vel, n)
        c5n = compute_c5(hn)
        if c5n <= c5 + 1e-11:
            h = hn
            if it % 100 == 0 and lr < 0.04:
                lr *= 1.15
        else:
            hn2 = make_feasible(h - lr * g, n)
            c5n2 = compute_c5(hn2)
            if c5n2 <= c5:
                h = hn2
                vel *= 0.3
            else:
                lr *= 0.65
                vel *= 0.2
    h = make_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h, best_c5 = h, c5
    return best_h, best_c5

def slsqp_step(h0, n, T, top_k, maxit):
    def fun(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        if top_k is not None:
            ti = np.argsort(corr)[-top_k:]
            tc = corr[ti]
            mc = tc.max()
            return (mc + np.log(np.sum(np.exp(T*(tc-mc))))/T) * (2.0/n)
        mc = corr.max()
        return (mc + np.log(np.sum(np.exp(T*(corr-mc))))/T) * (2.0/n)
    def jac(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = softmax_weights(corr, T, top_k)
        return weighted_grad(h, n, w)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxit, 'ftol': 1e-14})
        ho = make_feasible(res.x, n)
        return ho, compute_c5(ho)
    except Exception:
        return h0, compute_c5(h0)

def coord_descent(h0, n, budget, t0, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001]
    h = h0.copy()
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() - t0 < budget:
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 2:
                return h, best_c5
            old = h[j]
            bv = old
            bc = best_c5
            for sp in steps:
                for sg in (-1.0, 1.0):
                    v = old + sg * sp
                    if 0.0 <= v <= 1.0:
                        tr = h.copy()
                        tr[j] = v
                        tr = make_feasible(tr, n)
                        c = compute_c5(tr)
                        if c < bc - 1e-13:
                            bc, bv = c, v
            for v in (0.0, 1.0):
                if abs(v - old) > 1e-5:
                    tr = h.copy()
                    tr[j] = v
                    tr = make_feasible(tr, n)
                    c = compute_c5(tr)
                    if c < bc - 1e-13:
                        bc, bv = c, v
            if bv != old:
                h[j] = bv
                h = make_feasible(h, n)
                best_c5 = bc
                improved = True
    return h, best_c5
