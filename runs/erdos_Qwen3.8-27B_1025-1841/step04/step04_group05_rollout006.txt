import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def grad_active(h, n):
    """Gradient of the active (max) correlation term."""
    v = 1.0 - h
    R = np.correlate(h, v, mode='full')
    dx = 2.0 / n
    k_star = int(np.argmax(R))
    k = k_star - (n - 1)
    
    lo = max(0, -k)
    hi = min(n - 1, n - 1 - k)
    
    grad = np.zeros(n)
    
    if lo <= hi:
        idx1 = np.arange(lo, hi + 1)
        idx1_k = idx1 + k
        valid1 = (idx1_k >= 0) & (idx1_k < n)
        if np.any(valid1):
            grad[idx1[valid1]] += (1.0 - h[idx1_k[valid1]]) * dx
    
    p_lo = max(0, lo + k)
    p_hi = min(n - 1, hi + k)
    if p_lo <= p_hi:
        idx2 = np.arange(p_lo, p_hi + 1)
        idx2_m = idx2 - k
        grad[idx2] -= h[idx2_m] * dx
    
    return grad, k_star

def subgradient_step(h, n, lr, max_iter):
    """Iterative subgradient descent on the max correlation."""
    h_curr = h.copy()
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    
    for it in range(max_iter):
        grad, _ = grad_active(h_curr, n)
        h_new = h_curr - lr * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
        # But also check if we should use best
        if c5_new > best_c5 + 0.001:
            h_curr = best_h.copy()
    
    return best_h, best_c5

def binary_local_search(h, n, max_iter, rng, time_fn, budget_check):
    """Local search on binary sequences via swaps."""
    # Round to binary
    h_bin = (h > 0.5).astype(float)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    # Ensure exactly n/2 ones
    while len(ones) < n // 2:
        z = rng.randint(len(zeros))
        ones = np.append(ones, zeros[z])
        zeros = np.delete(zeros, z)
    while len(ones) > n // 2:
        z = rng.randint(len(ones))
        zeros = np.append(zeros, ones[z])
        ones = np.delete(ones, z)
    h_bin = np.zeros(n)
    h_bin[ones] = 1.0
    h_bin[zeros] = 0.0
    h_bin = make_feasible(h_bin, n)
    
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    no_improve = 0
    
    for it in range(max_iter):
        if budget_check():
            break
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        ones[ones == i] = j  # update tracking
        zeros[zeros == j] = i
        # Actually let's just recompute
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_bin.copy()
            no_improve = 0
        else:
            # Revert
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
            no_improve += 1
        if no_improve > 500:
            break
    
    return best_h, best_c5

def frank_wolfe_step(h, n):
    """One Frank-Wolfe step."""
    grad, _ = grad_active(h, n)
    # LP: minimize grad · z subject to 0≤z≤1, Σz = n/2
    # Solution: z_i = 1 for the n/2 indices with smallest grad_i, 0 otherwise
    idx_sorted = np.argsort(grad)
    z = np.zeros(n)
    z[idx_sorted[:n//2]] = 1.0
    
    return z, grad

3. **Multi-restart**: Try different initial directions.

Let me also add a "coordinate descent" step that's very effective for this type of problem:
- For each coordinate i, fix all others and find the best value of h[i]
- But we need to maintain the sum constraint, so: fix all others, then h[i] is determined by the sum constraint
- Actually this doesn't work directly. Instead: pick a pair (i,j), fix all others, and optimize h[i] and h[j] jointly (with h[i]+h[j] = constant to maintain sum)

This is a 2D subproblem: minimize max_k c_k(h) over h[i]+h[j] = const, 0≤h[i],h[j]≤1.

For a fixed pair, c_k is linear in both h[i] and h[j] (since the bilinear term h[i]*h[j] only appears if i and j are paired in the k-th correlation). So the 2D subproblem is still manageable.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def grad_active(h, n):
    v = 1.0 - h
    R = np.correlate(h, v, mode='full')
    dx = 2.0 / n
    k_star = int(np.argmax(R))
    k = k_star - (n - 1)
    lo = max(0, -k)
    hi = min(n - 1, n - 1 - k)
    grad = np.zeros(n)
    if lo <= hi:
        i_idx = np.arange(lo, hi + 1)
        j_idx = i_idx + k
        valid_j = (j_idx >= 0) & (j_idx < n)
        if np.any(valid_j):
            grad[i_idx[valid_j]] += (1.0 - h[j_idx[valid_j]]) * dx
        p_lo = max(0, lo + k)
        p_hi = min(n - 1, hi + k)
        if p_lo <= p_hi:
            p_idx = np.arange(p_lo, p_hi + 1)
            i_check = p_idx - k
            valid_i = (i_check >= lo) & (i_check <= hi)
            if np.any(valid_i):
                grad[p_idx[valid_i]] -= h[i_check[valid_i]] * dx
    return grad

def fw_optimize(h, n, max_iter, time_limit_fn):
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr)
    best_c5 = c5_curr
    best_h = h_curr.copy()
    for t in range(max_iter):
        if time_limit_fn() < 25:
            break
        grad = grad_active(h_curr, n)
        idx_sorted = np.argsort(grad)
        z = np.zeros(n)
        z[idx_sorted[:n // 2]] = 1.0
        gamma = 2.0 / (t + 2.0)
        h_new = (1.0 - gamma) * h_curr + gamma * z
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < c5_curr:
            h_curr = h_new
            c5_curr = c5_new
        else:
            h_curr = h_new
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_curr.copy()
    return best_h, best_c5

def fw_linesearch(h, n, max_iter, time_limit_fn):
    h_curr = h.copy()
    c5_curr = compute_c5(h_curr)
    best_c5 = c5_curr
    best_h = h_curr.copy()
    for t in range(max_iter):
        if time_limit_fn() < 25:
            break
        grad = grad_active(h_curr, n)
        idx_sorted = np.argsort(grad)
        z = np.zeros(n)
        z[idx_sorted[:n // 2]] = 1.0
        direction = z - h_curr
        best_gamma = 0.0
        best_c = c5_curr
        for g in [0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
            h_try = h_curr + g * direction
            h_try = make_feasible(h_try, n)
            c_try = compute_c5(h_try)
            if c_try < best_c:
                best_c = c_try
                best_gamma = g
        if best_gamma > 0:
            h_curr = make_feasible(h_curr + best_gamma * direction, n)
            c5_curr = compute_c5(h_curr)
        else:
            h_curr = z.copy()
            c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
    return best_h, best_c5

    T = 0.05

            break
        i = ones_list[int(rng.randint(len(ones_list)))]
        j = zeros_list[int(rng.randint(len(zeros_list)))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        delta = c5_new - best_c5
        if delta < -1e-12 or (T > 1e-6 and rng.random() < np.exp(-delta / T)):
            ones_list[ones_list.index(i)] = j
            zeros_list[zeros_list.index(j)] = i
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        T *= 0.998
        if T < 0.002:
            T = 0.002
        if no_improve > 2000:
            T = 0.03
            no_improve = 0
    return best_h, best_c5

def pair_swap_optimize(h, n, max_iter, time_limit_fn, rng):
    h_curr = (h > 0.5).astype(float).copy()
    n_half = n // 2
    ones_list = list(np.where(h_curr == 1.0)[0])
    zeros_list = list(np.where(h_curr == 0.0)[0])
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    no_improve = 0
    for it in range(max_iter):
        if time_limit_fn() < 15:
            break
        pair_count = 50 if it % 100 == 0 else 200
        improved = False
        for _ in range(pair_count):
            if time_limit_fn() < 15:
                break
            i = ones_list[int(rng.randint(len(ones_list)))]
            j = zeros_list[int(rng.randint(len(zeros_list)))]
            h_curr[i] = 0.0
            h_curr[j] = 1.0
            c5_new = compute_c5(h_curr)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_curr.copy()
                ones_list[ones_list.index(i)] = j
                zeros_list[zeros_list.index(j)] = i
                improved = True
            else:
                h_curr[i] = 1.0
                h_curr[j] = 0.0
        if not improved:
            no_improve += 1
        else:
            no_improve = 0
        if no_improve > 50:
            break
    return best_h, best_c5
