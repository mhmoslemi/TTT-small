```python
def softmax_obj_and_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    c = np.correlate(h, v, mode='full')
    mc = np.max(c)
    exp_T = np.exp(T * (c - mc))
    S = np.sum(exp_T)
    p = exp_T / S
    
    # Objective
    obj = (mc + np.log(S) / T) * dx + mu * (np.sum(h) - n/2.0)**2
    
    # Gradient
    # term1[j] = sum_k p[k]*v[j-k+n-1] = (p*v)[j+n-1]
    # term2[j] = sum_k p[k]*h[j+k-(n-1)] = (p_rev*h)[j+n-1]
    
    fft_size = 1
    while fft_size < 3*n:
        fft_size <<= 1
    
    P = np.fft.rfft(p, fft_size)
    V = np.fft.rfft(v, fft_size)
    H = np.fft.rfft(h, fft_size)
    
    conv_pv = np.fft.irfft(P * V, size=fft_size)
    
    p_rev = p[::-1].copy()
    Pr = np.fft.rfft(p_rev, fft_size)
    conv_prh = np.fft.irfft(Pr * H, size=fft_size)
    
    idx_start = n - 1
    idx_end = n - 1 + n  # exclusive
    term1 = conv_pv[idx_start:idx_end]
    term2 = conv_prh[idx_start:idx_end]
    
    grad = dx * (term1 - term2) + 2.0 * mu * (np.sum(h) - n/2.0)
    
    return obj, grad
```

```python
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _softmax_obj(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(c)
    exp_T = np.exp(T * (c - mc))
    S = np.sum(exp_T)
    return (mc + np.log(S) / T) * dx + mu * (np.sum(x) - n / 2.0) ** 2

def _softmax_grad(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    v = 1.0 - x
    c = np.correlate(x, v, mode='full')
    mc = np.max(c)
    exp_T = np.exp(T * (c - mc))
    S = np.sum(exp_T)
    p = exp_T / S

    fft_size = 1
    while fft_size < 3 * n + 10:
        fft_size <<= 1

    P = np.fft.rfft(p, fft_size)
    V = np.fft.rfft(v, fft_size)
    H = np.fft.rfft(x, fft_size)

    conv_pv = np.fft.irfft(P * V, size=fft_size)
    p_rev = p[::-1].copy()
    Pr = np.fft.rfft(p_rev, fft_size)
    conv_prh = np.fft.irfft(Pr * H, size=fft_size)

    s = n - 1
    term1 = conv_pv[s:s + n]
    term2 = conv_prh[s:s + n]

    grad = dx * (term1 - term2) + 2.0 * mu * (np.sum(x) - n / 2.0)
    return grad

def _pnorm_obj(x, p, mu):
    n = len(x)
    dx = 2.0 / n
    c = np.correlate(x, 1.0 - x, mode='full')
    c_safe = np.maximum(c, 1e-12)
    val = (np.sum(c_safe ** p)) ** (1.0 / p)
    return val * dx + mu * (np.sum(x) - n / 2.0) ** 2

def _act_obj(x, weights, mu, n):
    c = np.correlate(x, 1.0 - x, mode='full')
    return float(np.dot(weights, c)) * (2.0 / n) + mu * (np.sum(x) - n / 2.0) ** 2

def opt_soft(h0, n, T, mu, max_iter=150):
    res = minimize(_softmax_obj, h0, args=(T, mu), jac=_softmax_grad,
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxfun': max_iter * 30})
    return make_feasible(res.x, n)

def opt_pnorm(h0, n, p, mu, max_iter=150):
    res = minimize(_pnorm_obj, h0, args=(p, mu),
                   method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9, 'maxfun': max_iter * 20})
    return make_feasible(res.x, n)

def active_set_opt(h0, n, n_iters=25, active_frac=0.10, mu=500.0, max_inner=80):
    h = h0.copy()
    c_best = compute_c5(h)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_c = len(corr)
        k = max(10, int(active_frac * n_c))
        idx = np.argpartition(corr, -k)[-k:]
        w_raw = np.exp(80.0 * (corr[idx] - np.max(corr)))
        weights = np.zeros(n_c)
        weights[idx] = w_raw
        weights /= weights.sum()
        res = minimize(_act_obj, h, args=(weights, mu, n),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_inner, 'ftol': 1e-14, 'gtol': 1e-9, 'maxfun': max_inner * 20})
        h_new = make_feasible(res.x, n)
        c_new = compute_c5(h_new)
        if c_new <= c_best + 1e-12:
            h = h_new
            c_best = c_new
    return h

def sa_binary(h_bin, n, rng, n_rounds=40000, T_start=0.008, T_end=0.00005):
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    best_c5 = cur_c5
    best_h = h.copy()
    for rnd in range(n_rounds):
        T = T_start * (T_end / T_start) ** (rnd / max(1, n_rounds - 1))
        if not ones or not zeros:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-9 and rng.rand() < np.exp(min(0, -delta / T))):
            cur_c5 = new_c5
            ones[ii] = oj
            zeros[jj] = oi
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best_c5

def coord_descent(h0, n, max_rounds=12, step_list=None):
    if step_list is None:
        step_list = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        order = np.argsort(-np.abs(h - 0.5)) if rnd % 3 == 0 else np.arange(n)
        for j in order:
            ov = h[j]
            for d in step_list:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = len(h0)

    # Phase 1: Annealing at parent resolution
    n1 = best_n
    h1 = h0.copy()
    T_schedule = [50, 200, 800, 3000, 10000, 30000, 80000]
    for T in T_schedule:
        if rem() < 25:
            break
        h1 = opt_soft(h1, n1, T, mu=600.0, max_iter=120)
        c = compute_c5(h1)
        if c < best_c5:
            best_c5, best_h = c, h1.copy()

    # Phase 1b: p-norm refinement
    if rem() > 40:
        for p in [8, 20, 50]:
            if rem() < 20:
                break
            h1 = opt_pnorm(best_h, n1, p, mu=600.0, max_iter=100)
            c = compute_c5(h1)
            if c < best_c5:
                best_c5, best_h = c, h1.copy()

    # Phase 2: Active set at parent resolution
    if rem() > 50:
        h2 = active_set_opt(best_h, n1, n_iters=20, active_frac=0.08, mu=600.0, max_inner=60)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

    # Phase 3: Upscale to higher resolution
    for n3 in [1280, 2560]:
        if rem() < 70:
            break
        h3 = upsample_h(best_h, n3)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()

        for T in [200, 1000, 5000, 20000]:
            if rem() < 25:
                break
            h3 = opt_soft(h3, n3, T, mu=400.0, max_iter=100)
            c = compute_c5(h3)
            if c < best_c5:
                best_c5, best_h = c, h3.copy()

        if rem() > 40:
            h3 = active_set_opt(best_h, n3, n_iters=15, active_frac=0.06, mu=400.0, max_inner=50)
            c = compute_c5(h3)
            if c < best_c5:
                best_c5, best_h = c, h3.copy()

    # Phase 4: Binary SA at moderate resolution
    if rem() > 80:
        n_bin = min(n1, 1024)
        h_b = upsample_h(best_h, n_bin) if n_bin < len(best_h) else best_h.copy()
        h_bin = (h_b > 0.4).astype(float)
        target = n_bin // 2
        cnt = int(h_bin.sum())
        if cnt > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:cnt - target]] = 0.0
        elif cnt < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target - cnt]] = 1.0
        h_bs, c_bs = sa_binary(h_bin, n_bin, rng, n_rounds=50000, T_start=0.006, T_end=0.00003)
        if c_bs < best_c5:
            best_c5, best_h = c_bs, h_bs.copy()

    # Phase 5: Multi-start at best resolution
    n_ms = len(best_h)
    if rem() > 50:
        for restart in range(5):
            if rem() < 20:
                break
            h_r = best_h.copy() + 0.06 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            for T in [500, 5000, 50000]:
                if rem() < 15:
                    break
                h_r = opt_soft(h_r, n_ms, T, mu=500.0, max_iter=80)
                c = compute_c5(h_r)
                if c < best_c5:
                    best_c5, best_h = c, h_r.copy()

    # Phase 6: Final active set + coord descent
    if rem() > 40:
        nf = min(len(best_h), 2048)
        hf = upsample_h(best_h, nf) if nf < len(best_h) else best_h.copy()
        hf = active_set_opt(hf, nf, n_iters=15, active_frac=0.05, mu=500.0, max_inner=50)
        c = compute_c5(hf)
        if c < best_c5:
            best_c5, best_h = c, hf.copy()

    if rem() > 25:
        nc = min(len(best_h), 1536)
        hc = upsample_h(best_h, nc) if nc < len(best_h) else best_h.copy()
        hc, cc = coord_descent(hc, nc, max_rounds=10)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
