import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = float(np.sum(mask))
        if cnt < 1e-14:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idxs = np.argsort(h) if diff > 0 else np.argsort(-h)
        for idx in idxs:
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=500, ftol=1e-18, gtol=1e-14):
    dx = 2.0 / n
    
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -100, 100)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                            'maxls': 50, 'maxfun': 30000})
    return make_feasible(res.x, n)

class FastBinarySearch:
    def __init__(self, h_bin, n):
        self.h = h_bin.astype(float).copy()
        self.n = n
        self.corr = compute_corr_full(self.h).astype(float)
        self.dx = 2.0 / n
        self.current = float(np.max(self.corr) * self.dx)
        self.ones = np.where(self.h > 0.5)[0].copy()
        self.zeros = np.where(self.h <= 0.5)[0].copy()
    
    def try_swap(self, oi, oj):
        n = self.n
        # Compute change in correlation
        delta = np.zeros(2*n - 1)
        
        # h[oi]: 1 -> 0
        # h[i] term: -(1-h[oi+m]) for m in [-oi, n-1-oi]

        if m_lo <= m_hi:
            k_lo = m_lo + (n-1)
            k_hi = m_hi + (n-1)
            ms = np.arange(m_lo, m_hi+1)
            delta[k_lo:k_hi+1] -= (1.0 - self.h[oi + ms])
        
        # (1-h[i+m]) term: +h[oi-m] for m in [oi-(n-1), oi]
        m_lo2 = max(oi-(n-1), -(n-1))
        m_hi2 = min(oi, n-1)
        if m_lo2 <= m_hi2:
            k_lo2 = m_lo2 + (n-1)
            k_hi2 = m_hi2 + (n-1)
            ms2 = np.arange(m_lo2, m_hi2+1)
            delta[k_lo2:k_hi2+1] += self.h[oi - ms2]
        
        # h[oj]: 0 -> 1
        # h[i] term: +(1-h[oj+m]) for m in [-oj, n-1-oj]
        m_lo3 = max(-oj, -(n-1))
        m_hi3 = min(n-1-oj, n-1)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = float(np.sum(mask))
        if cnt < 1e-14:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idxs = np.argsort(h) if diff > 0 else np.argsort(-h)
        for idx in idxs:
            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
            diff = target - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=400, ftol=1e-18, gtol=1e-14):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -100, 100)

        viol = np.sum(h) - n / 2.0

    bounds = [(0.0, 1.0)] * n

                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': gtol,
                            'maxls': 50, 'maxfun': 50000})
    return make_feasible(res.x, n)

def swap_delta(corr_old, h, oi, oj, n):
    """Compute delta in full correlation when h[oi]:1->0, h[oj]:0->1."""

    m_hi = min(n - 1 - oi, n - 1)

        delta[k_lo:k_hi + 1] -= (1.0 - h[oi + ms])
    m_lo2 = max(oi - (n - 1), -(n - 1))

def fast_binary_search(h_bin, n, rng, max_rounds=200, deadline=0.0):
    h = h_bin.astype(float).copy()
    corr = compute_corr_full(h).astype(float)
    dx = 2.0 / n
    current = float(np.max(corr) * dx)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        n_tries = min(4000, len(ones) * len(zeros) * 20)
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            dl = swap_delta(corr, h, oi, oj, n)
            new_max = float(np.max(corr + dl) * dx)
            if new_max < current - 1e-15:
                corr += dl
                current = new_max
                h[oi] = 0.0
                h[oj] = 1.0
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
        if not improved:
            break
    return h, current

    n_ones = int(np.sum(h))
    if n_ones > target:

        for k in range(target - n_ones):
            h[idx[k]] = 1.0
    return h
