            p = get_active_weights(corr, width=0.02)
        else:
            max_lag = np.argmax(corr)
            new_active = set(int(i) for i in np.where(corr > corr.max() - 1e-6)[0])
            active_set = new_active
            p = np.zeros(len(corr))
            for i in np.where(corr > corr.max() * 0.999 - 1e-8)[0]:
                p[i] = 1.0
            p /= p.sum()

        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(p, c)) * dx + 0.5 * mu2 * (np.sum(x) - n/2.0)**2
        mu2 = 200.0
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': inner_iters, 'ftol': 1e-14, 'gtol': 1e-9})
        h = make_feasible(res.x, n)
    return h, compute_c5(h)

def binary_local_search(h_bin, n, rng, max_rounds=100, tries_per_round=8000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries_per_round, max(len(ones) * len(zeros), 1))
        for t in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def simulated_annealing_binary(h_bin, n, rng, start_T=0.5, end_T=0.01, n_steps=50000):
    h = h_bin.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for step in range(n_steps):
        T = start_T * (end_T / start_T) ** (step / n_steps)
        if not ones or not zeros:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        c_new = compute_c5(h)
        delta = c_new - cur
        if delta < 0 or rng.random() < np.exp(-n * delta / (2.0 * T + 1e-10)):
            cur = c_new
            ones[ii] = oj
            zeros[jj] = oi
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best_c

def multi_scale_optimize(h0, n0, n_target, T_list, t_start, t_budget):
    h = h0
    n = n0
    best_c = compute_c5(h)
    best_h = h.copy()
    scales = [n0]
    cur = n0
    while cur < n_target:
        cur = min(cur * 2, n_target)
        scales.append(cur)
    for ns in scales[1:]:
        if time.time() - t_start > t_budget:
            break
        h = upsample_h(best_h, ns)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
        for T in T_list:
            if time.time() - t_start > t_budget:
                break
            h_opt = lbgfs_optimize(h, ns, T, mu=400, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                h = h_opt
            else:
                h = h_opt
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n_init = 640
        h0 = np.zeros(n_init)
        h0[:n_init//2] = 1.0

    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n0

    # Phase 1: Optimize at parent resolution with various temperatures
    T_schedule_1 = [500, 1000, 2000, 5000, 10000]
    for T in T_schedule_1:
        if rem() < 40:
            break
        h_opt = lbgfs_optimize(best_h, best_n, T, mu=400, max_iter=300)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5 - 1e-12:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Also try complement
    h_comp = 1.0 - best_h
    h_comp = make_feasible(h_comp, best_n)
    c_comp = compute_c5(h_comp)
    if c_comp < best_c5:
        best_c5 = c_comp
        best_h = h_comp.copy()

    # Phase 2: Multi-scale optimization
    if rem() > 150:
        for n_target in [960, 1280, 1920]:
            if rem() < 80:
                break
            h_up = upsample_h(best_h, n_target)
            c_up = compute_c5(h_up)
            if c_up < best_c5:
                best_c5 = c_up
                best_h = h_up.copy()
                best_n = n_target
            for T in [800, 2000, 8000]:
                if rem() < 40:
                    break
                h_opt = lbgfs_optimize(h_up, n_target, T, mu=300, max_iter=250)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5 = c_opt
                    best_h = h_opt.copy()
                    best_n = n_target
                    h_up = h_opt
                else:
                    h_up = h_opt

    # Phase 3: Constraint generation at best resolution
    if rem() > 120:
        n_cg = min(best_n, 1280)
        h_cg = upsample_h(best_h, n_cg) if best_n != n_cg else best_h.copy()
        h_cg2, c_cg = constraint_gen_optimize(h_cg, n_cg, max_rounds=10, T=3000, inner_iters=15)
        if c_cg < best_c5:
            best_c5 = c_cg
            best_h = h_cg2.copy()
            best_n = n_cg

    # Phase 4: Binary local search at multiple resolutions
    for n_bs in [960, 1280, 1600, 2048]:
        if rem() < 100:
            break
        h_b = upsample_h(best_h, n_bs) if n_bs != best_n else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        target_ones = n_bs // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target_ones - n_ones]] = 1.0
        h_bs, c_bs = binary_local_search(h_bin, n_bs, rng, max_rounds=80, tries_per_round=10000)
        if c_bs < best_c5:
            best_c5 = c_bs
            best_h = h_bs.copy()
            best_n = n_bs
            # Continue from improved binary
            h_fin = (h_bs > 0.5).astype(float)
            h_fin2, c_fin = binary_local_search(h_fin, n_bs, rng, max_rounds=40, tries_per_round=12000)
            if c_fin < best_c5:
                best_c5 = c_fin
                best_h = h_fin2.copy()

    # Phase 5: Simulated annealing on best binary
    if rem() > 120:
        n_sa = min(best_n, 1600)
        h_b = upsample_h(best_h, n_sa) if abs(best_n - n_sa) > 0 else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        target_ones = n_sa // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target_ones - n_ones]] = 1.0
        h_sa, c_sa = simulated_annealing_binary(h_bin, n_sa, rng, start_T=0.3, end_T=0.005, n_steps=40000)
        if c_sa < best_c5:
            best_c5 = c_sa
            best_h = h_sa.copy()
            best_n = n_sa

    # Phase 6: Final continuous polish on the best solution
    if rem() > 50:
        n_f = best_n
        for T in [3000, 10000, 30000]:
            if rem() < 20:
                break
            h_opt = lbgfs_optimize(best_h, n_f, T, mu=1000, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 7: Try SLSQP with tight tolerance
    if rem() > 30:
        n_sp = min(best_n, 1024)
        h_sp = upsample_h(best_h, n_sp) if best_n != n_sp else best_h.copy()
        dx = 2.0 / n_sp
        def obj_s(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return soft_max_corrs(corr, 5000) * dx
        def con_eq(x):
            return np.sum(x) - n_sp / 2.0
        res = minimize(obj_s, h_sp, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n_sp,
                       constraints=[{'type': 'eq', 'fun': con_eq, 'jac': lambda x: np.ones(n_sp)}],
                       options={'maxiter': 200, 'ftol': 1e-12})
        h_sp_opt = make_feasible(res.x, n_sp)
        c_sp = compute_c5(h_sp_opt)
        if c_sp < best_c5:
            best_c5 = c_sp
            best_h = h_sp_opt.copy()
            best_n = n_sp

    # Phase 8: Final binary polish
    if rem() > 30:
        n_fb = min(best_n, 1280)
        h_fb = upsample_h(best_h, n_fb) if best_n != n_fb else best_h.copy()
        h_bin = (h_fb > 0.5).astype(float)
        target_ones = n_fb // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target_ones - n_ones]] = 1.0
        h_pol, c_pol = binary_local_search(h_bin, n_fb, rng, max_rounds=50, tries_per_round=15000)
        if c_pol < best_c5:
            best_c5 = c_pol
            best_h = h_pol.copy()
            best_n = n_fb

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    if time.time() - t0 > budget_s:
        return best_h, best_c5, len(best_h)
    return best_h, best_c5, len(best_h)
