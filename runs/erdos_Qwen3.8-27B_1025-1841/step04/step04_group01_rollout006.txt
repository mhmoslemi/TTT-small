        if c_up2 < best_c5:
            best_c5 = c_up2
            best_h = make_feasible(x_up, n2)
        elif c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()
        # More search at new resolution
        if _time_left() > 100:
            best_h, best_c5 = _pushdown_search(best_h, best_c5, n2, max_iters=1500)
        if _time_left() > 50:
            _G_T = 800.0
            x_res = _slsqp(best_h, maxiter=300)
            c_res = compute_c5(x_res)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = make_feasible(x_res, n2)

    # Phase 8: Random restarts
    for rs in range(5):
        if _time_left() < 50:
            break
        n_pts = len(best_h)
        _G_N = n_pts
        _G_DX = 2.0 / n_pts
        scale = 0.02 if rs < 3 else 0.04
        h_try = make_feasible(best_h + np.random.randn(n_pts) * scale, n_pts)
        _G_T = 400.0
        x_res = _slsqp(h_try, maxiter=250)
        c_res = compute_c5(x_res)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = make_feasible(x_res, n_pts)

    # Final: coordinate descent polish
    if _time_left() > 10:
        n_pts = len(best_h)
        for j in range(n_pts):
            if _time_left() < 3:
                break
            old_val = best_h[j]
            best_val = old_val
            best_c = best_c5
            for step in [0.0003, 0.0005, 0.001, 0.002]:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * step
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c - 1e-16:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
