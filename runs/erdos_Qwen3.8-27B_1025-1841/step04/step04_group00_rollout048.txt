        grad = np.zeros(n)
        for k in top_idx:

            for j in range(n):
                v = 0.0
                js = j + s
                if 0 <= js < n:
                    v += (1.0 - h_cur[j])
                jm = j - s
                if 0 <= jm < n:
                    v -= h_cur[jm]
                grad[j] += v
        grad /= len(top_idx)

        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur:
            h_cur = new_h
        else:
            new_h2 = make_feasible(h_cur - step_size * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_cur:
                h_cur = new_h2
            else:
                step_size *= 0.7
                vel *= 0.5
            if step_size < 1e-8:
                break

    c5_final = compute_c5(h_final)
    if c5_final < best_c5:
        best_h, best_c5 = h_final, c5_final
    return best_h, best_c5
