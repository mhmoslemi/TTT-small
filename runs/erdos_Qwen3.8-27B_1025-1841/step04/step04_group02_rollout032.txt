    a[:n] = h
    b = np.zeros(n_fft)
    b[:n] = (1.0 - h)[::-1]
    corr = np.fft.ifft(np.fft.fft(a) * np.fft.fft(b)).real
    return float(np.max(corr[:2 * n - 1]) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(np.float64)
    target = n // 2
    if b.sum() != target:
        if b.sum() > target:
            order = np.argsort(h)
            for idx in order:
                if b[idx] == 1:
                    b[idx] = 0
                    if b.sum() == target:
                        break
        else:
            order = np.argsort(h)[::-1]
            for idx in order:
                if b[idx] == 0:
                    b[idx] = 1
                    if b.sum() == target:
                        break
    return b

def sa_core(b, n, deadline, T0, T1, rng, total_expected_iters=500000):
    c5_fn = compute_c5_fft
    b = b.copy()
    curr_c5 = c5_fn(b)
    best_c5 = curr_c5
    best_b = b.copy()
    it = 0
    while time.time() < deadline:
        it += 1
        frac = min(float(it) / total_expected_iters, 1.0)
        T = T0 * (T1 / T0) ** frac
        r = rng.random()
        if r < 0.6:
            # Single swap
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or b[i] == b[j]:
                continue
            b[i] = 0.0
            b[j] = 1.0
            new_c5 = c5_fn(b)
            accept = new_c5 <= curr_c5 or (T > 1e-15 and np.log(rng.random() + 1e-30) < -(new_c5 - curr_c5) / T)
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0
        elif r < 0.85:
            # Double swap
            i1 = rng.randint(n)
            i2 = rng.randint(n)
            j1 = rng.randint(n)
            j2 = rng.randint(n)
            if len(set([i1, i2, j1, j2])) < 4:
                continue
            if b[i1] + b[i2] != 2.0 or b[j1] + b[j2] != 0.0:
                continue
            b[i1] = 0.0; b[i2] = 0.0
            b[j1] = 1.0; b[j2] = 1.0
            new_c5 = c5_fn(b)
            accept = new_c5 <= curr_c5 or (T > 1e-15 and np.log(rng.random() + 1e-30) < -(new_c5 - curr_c5) / T)
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i1] = 1.0; b[i2] = 1.0
                b[j1] = 0.0; b[j2] = 0.0
        else:
            # Segment swap: swap two contiguous segments of equal length and equal #ones
            seg_len = rng.randint(2, min(12, n // 30 + 1))
            p1 = rng.randint(0, n - seg_len)
            p2 = rng.randint(0, n - seg_len)
            if abs(p1 - p2) < seg_len:
                continue
            s1 = b[p1:p1 + seg_len].sum()
            s2 = b[p2:p2 + seg_len].sum()
            if s1 != s2 or s1 == 0.0 or s1 == float(seg_len):
                continue
            b[p1:p1 + seg_len], b[p2:p2 + seg_len] = b[p2:p2 + seg_len].copy(), b[p1:p1 + seg_len].copy()
            new_c5 = c5_fn(b)
            accept = new_c5 <= curr_c5 or (T > 1e-15 and np.log(rng.random() + 1e-30) < -(new_c5 - curr_c5) / T)
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[p1:p1 + seg_len], b[p2:p2 + seg_len] = b[p2:p2 + seg_len].copy(), b[p1:p1 + seg_len].copy()
    return best_b, best_c5

_N_G = 0
_T_G = 0.0

def _slsqp_obj(x):
    n = _N_G
    T = _T_G
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted)) + 1e-30) / T) * dx

def _slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def _slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _N_G, _T_G
    _N_G = n
    _T_G = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_descent(h, n, deadline):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    step_sizes = [0.0, 1.0, 0.3, 0.7, 0.1, 0.9]
    while improved and time.time() < deadline:
        improved = False
        rnd += 1
        for j in range(n):
            if time.time() > deadline:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in step_sizes:
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Long SA at native resolution
    deadline = start_time + budget_s * 0.30
    b1, c5_1 = sa_core(h_init, n0, deadline, 0.008, 0.00001, rng, 800000)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()

    # Phase 2: SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        try:
            x_res = slsqp_run(best_h, n0, 150.0, 250, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 3: SA from continuous solution
    deadline = start_time + budget_s * 0.55
    if time.time() < deadline:
        b3, c5_3 = sa_core(best_h, n0, deadline, 0.005, 0.000005, rng, 600000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()

    # Phase 4: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        n_hi = n0 + 480
        h_hi = upsample(best_h, n_hi)
        b_hi, c5_hi = sa_core(h_hi, n_hi, start_time + budget_s * 0.75, 0.005, 0.00001, rng, 500000)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = b_hi.copy()
        # SLSQP at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 150 and c5_hi < best_c5 + 0.001:
            try:
                x_res = slsqp_run(best_h, n_hi, 200.0, 200, 1e-13)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n_hi)
            except Exception:
                pass

    # Phase 5: Try lower resolution for structural insight
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n_lo = max(n0 // 2, 480)
        h_lo = downsample(best_h, n_lo)
        b_lo, c5_lo = sa_core(h_lo, n_lo, start_time + budget_s * 0.88, 0.006, 0.00001, rng, 400000)
        # Upsample back and compare
        if c5_lo < 0.382:
            h_lo_up = upsample(b_lo, n0)
            b_lo_up, c5_lo_up = sa_core(h_lo_up, n0, start_time + budget_s * 0.93, 0.003, 0.00001, rng, 300000)
            if c5_lo_up < best_c5:
                best_c5 = c5_lo_up
                best_h = b_lo_up.copy()

    # Phase 6: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 400.0, 150, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 7: Coordinate descent polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, start_time + remaining * 0.7)

    # Phase 8: Final short SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        b_f, c5_f = sa_core(best_h, n_pts, start_time + remaining - 2, 0.002, 0.000001, rng, 100000)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = b_f.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
