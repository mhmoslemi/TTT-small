def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    exps = np.exp(T * (corr - mx))
    w = exps / np.sum(exps)
    
    # First sum: A[j] = sum_{l=j}^{n-1} w[n-1+l-j] * (1-h[l])
    # Second sum: B[j] = sum_{l=0}^{j} w[n-1+j-l] * h[l]
    
    # Use broadcasting
    # A[j] = sum over l from j to n-1 of w[n-1+l-j] * (1-h[l])
    # Let's create arrays
    j_idx = np.arange(n)
    l_idx = np.arange(n)
    # diff = l - j, from -j to n-1-j for each j
    # For A: diff >= 0, so l >= j
    # A[j] = sum_{diff=0}^{n-1-j} w[n-1+diff] * (1-h[j+diff])
    
    # I'll compute this with a loop (n=960, each iteration is O(n) numpy op)
    A = np.zeros(n)

    return dx * (A - B)
