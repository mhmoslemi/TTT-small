def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    target = 2 * n - 1
    fsize = 1
    while fsize < target:
        fsize *= 2
    H = np.fft.rfft(h, fsize)
    O = np.fft.rfft(1.0 - h, fsize)
    corr = np.fft.irfft(H * np.conj(O), fsize)
    return float(np.max(corr) * dx)

def slsqp_run(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def con_sum(x):
        return np.sum(x) - len(x) / 2.0
    
    def con_jac(x):
        return np.ones(len(x))
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    target = 2 * n - 1
    fsize = 1
    while fsize < target:
        fsize *= 2
    H = np.fft.rfft(h, fsize)
    O = np.fft.rfft(1.0 - h, fsize)
    corr = np.fft.irfft(H * np.conj(O), fsize)
    return float(np.max(corr) * dx)

def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        order = np.argsort(-h)
        for idx in order:
            if diff <= 0:
                break
            if b[idx] == 0:
                b[idx] = 1
                diff -= 1
    elif diff < 0:
        order = np.argsort(h)
        for idx in order:
            if diff >= 0:
                break
            if b[idx] == 1:
                b[idx] = 0
                diff += 1
    return b

def upsample_linear(h_old, n_new):
    n_old = len(h_old)

        return make_feasible(h_old.copy(), n_new)

def sa_binary(b, n, n_iters, T0, T1, max_swap, rng, start_time, budget_s):
    b = b.copy()
    curr_c5 = compute_c5_fft(b)
    best_c5 = curr_c5
    best_b = b.copy()

    check_interval = 200

            if remaining < 15:
                break
        frac = float(it) / max(n_iters, 1)

        if max_swap > 1 and rng.random() < 0.35:
            k = rng.randint(1, max_swap + 1)
        else:
            k = 1

        zeros = np.where(b == 0)[0]
        if len(ones) < k or len(zeros) < k:
            k = min(len(ones), len(zeros))
            if k == 0:
                it += 1
                continue
        si = rng.choice(ones, size=k, replace=False)
        sj = rng.choice(zeros, size=k, replace=False)
        b[si] = 0
        b[sj] = 1
        new_c5 = compute_c5_fft(b)
        if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):

def slsqp_objective(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(T * shifted)
    wsum = np.sum(weights)
    ew = weights / wsum
    grad = np.zeros(2 * n - 1)

            i_lo = 0
            i_hi = n - 1 - offset
            grad[k] = (ew[k] * T * (corr[k] - mc) / T + (1.0 / T) * (ew[k])) * 0
    # Use numerical gradient as fallback - SLSQP will do its own
    return None

def slsqp_run(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con_sum(x):
        return np.sum(x) - len(x) / 2.0
    def con_jac(x):
        return np.ones(len(x))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
