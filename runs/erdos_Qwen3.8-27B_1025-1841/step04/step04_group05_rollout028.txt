def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:

        if cnt == 0:
            break

        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    """Compute np.correlate(a, v, 'full') using FFT."""
    n = len(a)
    m = len(v)
    # Zero-pad to length >= n + m - 1
    N = 1
    while N < n + m - 1:
        N *= 2
    A = rfft(a, N)
    V = rfft(v, N)

    corr = irfft(A * np.conj(V), N)
