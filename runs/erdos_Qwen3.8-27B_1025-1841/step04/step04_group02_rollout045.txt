        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100:
            x_res2 = run_slsqp(best_h, n0, [50.0, 200.0, 800.0], 300, 1e-14)
            c5_res2 = compute_c5(x_res2)
            if c5_res2 < best_c5:
                best_c5 = c5_res2
                best_h = make_feasible(x_res2, n0)

    # Phase 5: Try different resolution (higher)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 400:
        n1 = n0 + 256
        h_up = upsample_h(best_h, n1)
        iters3 = min(60000, int(remaining * 0.3 * 700))
        b3, c5_3 = binary_sa(h_up, n1, max(iters3, 20000), 0.006, 0.00003, rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 150:
            x_res3 = run_slsqp(best_h, n1, [30.0, 150.0, 600.0], 300, 1e-14)
            c5_res3 = compute_c5(x_res3)
            if c5_res3 < best_c5:
                best_c5 = c5_res3
                best_h = make_feasible(x_res3, n1)

    # Phase 6: Try lower resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        n2 = max(n0 - 128, 480)
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x2 = np.linspace(0.0, 2.0, n2)
        h_lo = np.interp(new_x2, old_x, best_h)
        h_lo = make_feasible(h_lo, n2)
        iters4 = min(80000, int(remaining * 0.4 * 800))
        b4, c5_4 = binary_sa(h_lo, n2, max(iters4, 30000), 0.007, 0.00003, rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = b4.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100:
            x_res4 = run_slsqp(best_h, n2, [40.0, 200.0, 700.0], 250, 1e-14)
            c5_res4 = compute_c5(x_res4)
            if c5_res4 < best_c5:
                best_c5 = c5_res4
                best_h = make_feasible(x_res4, n2)

    # Phase 7: Coordinate descent on best
    n_pts = len(best_h)
    best_h, best_c5 = coord_descend(best_h.copy(), n_pts, start_time,
                                    start_time + budget_s - 30, 12, 0.01)

    # Phase 8: Final SLSQP with very high T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        x_final = run_slsqp(best_h, n_pts, [500.0, 2000.0], 250, 1e-15)
        c5_final = compute_c5(x_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = make_feasible(x_final, n_pts)

    # Phase 9: Final targeted search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        best_h, best_c5 = shift_targeted_search(best_h.copy(), n_pts, start_time,
                                                start_time + remaining - 5, 15)

    # Phase 10: One last SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        n_pts = len(best_h)
        x_last = run_slsqp(best_h, n_pts, [1500.0, 5000.0], 200, 1e-15)
        c5_last = compute_c5(x_last)
        if c5_last < best_c5:
            best_c5 = c5_last
            best_h = make_feasible(x_last, n_pts)

    # Final: ensure feasibility
    n_pts = len(best_h)
    best_h = make_feasible(best_h, n_pts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_pts
