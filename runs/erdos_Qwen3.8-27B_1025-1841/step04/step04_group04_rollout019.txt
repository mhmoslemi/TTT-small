import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max_obj(h, n, T, p=1.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Log-sum-exp for numerical stability
    val = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    return val

def obj_lf(bx, n, T):
    # bx are the free variables (n-1 of them), last is determined
    h = np.empty(n)
    h[:n-1] = np.clip(bx, 0, 1)
    h[n-1] = n/2.0 - np.sum(h[:n-1])
    if h[n-1] < 0 or h[n-1] > 1:

        h = make_feasible(h, n)
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = corr - mc

from scipy.optimize import minimize

_G_n = 480
_G_T = 100.0
_G_lam = 0.0

def _gobj(x):
    dx = 2.0 / _G_n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    sh = corr - mc
    T = _G_T
    mx = np.clip(T * sh, -400, 400)
    val = (mc + np.log(np.sum(np.exp(mx))) / T) * dx
    if _G_lam > 0:
        s = np.sum(x) - _G_n / 2.0
        val += _G_lam * s * s
    return val

def _geq(x):
    return np.sum(x) - _G_n / 2.0

def _geqj(x):
    return np.ones(_G_n)

def _pf(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    t = n / 2.0
    for _ in range(50):
        d = t - np.sum(h)
        if abs(d) < 1e-15:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        if not np.any(m):
            break
        h[m] += d / np.sum(m)
        h = np.clip(h, 0.0, 1.0)
    return h
