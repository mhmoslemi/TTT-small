import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def subgradient_step(h, n, alpha, top_k=5):
    """One subgradient step targeting the top-k worst lags."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -top_k)[-top_k:]
    # Compute gradient as average over top-k lags
    grad = np.zeros(n)
    for m in idx:
        k = m - (n - 1)
        if k >= 0:
            # c[n-1+k] = sum_{i=k}^{n-1} h[i](1-h[i-k])

            for j in range(k, n):

            for j in range(0, n - k):
                grad[j] -= h[j + k]
        else:

            for j in range(kp, n):

    h_new = h - alpha * grad

    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_avg = np.zeros(n)
    for t in range(1, n_iter + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Find top_k lag indices
        idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(n)
        for m in idx:
            k = int(m) - (n - 1)
            if k >= 0:
                grad[k:] += (1.0 - h[:n-k])
                grad[:n-k] -= h[k:]
            else:
                kp = -k
                grad[:n-kp] += (1.0 - h[kp:])
                grad[kp:] -= h[:n-kp]
        grad /= float(top_k)
        alpha = c0 / np.sqrt(t)
        h_new = h - alpha * grad
        h = make_feasible(h_new, n)
        h_avg = (t-1)/t * h_avg + (1.0/t) * h

        if c5 < best_c5:

def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):

        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            flat = np.argwhere(h < 1.0)
            if len(flat) > 0:
                idx = flat[np.argmin(h[flat[:,0]])][0]

def upsample_h(h, new_n):
    old_n = len(h)

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def slsqp_polish(h_start, n, T, max_iter=100):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -700, 700)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    def con(h):
        return np.sum(h) - n / 2.0
    def jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14, 'iprint': 0})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

        viol = np.sum(h) - n / 2.0

        h1, c5_sg2 = subgrad_optimize(best_h, n1, n_iter=5000, top_k=3, c0=0.06, time_budget=80.0)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h1.copy()

    # Phase 2: L-BFGS-B smooth at n=2048 with multiple temperatures
    for T in [100, 500, 2000]:
        if tl() < 60:
            break
        h_opt = lbgfb_smooth(best_h, n1, T, mu=300.0, max_iter=150)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 3: Binary local search at n=2048
    if tl() > 100:
        h_bin, c5_bin = targeted_lag_search(best_h, n1, rng, n_rounds=50, time_budget=100.0)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    if tl() > 80:
        h_bs, c5_bs = swap_local_search(best_h, n1, n_swaps=80000, rng=rng, time_budget=80.0)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 4: Go to n=4096
    n2 = 4096
    if tl() > 200:
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

        h2, c5_sg2 = subgrad_optimize(h2, n2, n_iter=5000, top_k=5, c0=0.03, time_budget=80.0)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h2.copy()

        if tl() > 100:
            h_opt = lbgfb_smooth(best_h, n2, 500, mu=200.0, max_iter=100)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

        if tl() > 120:
            h_bin2, c5_bin2 = targeted_lag_search(best_h, n2, rng, n_rounds=40, time_budget=80.0)
            if c5_bin2 < best_c5:
                best_c5 = c5_bin2
                best_h = h_bin2.copy()

        if tl() > 80:
            h_bs2, c5_bs2 = swap_local_search(best_h, n2, n_swaps=40000, rng=rng, time_budget=80.0)
            if c5_bs2 < best_c5:
                best_c5 = c5_bs2
                best_h = h_bs2.copy()

    # Phase 5: Alternative starting points at n=2048
    if tl() > 150:
        # Periodic pattern with period 7
        for period in [7, 11, 13]:
            if tl() < 60:
                break
            half = (period + 1) // 2
            pat = np.zeros(period)
            pat[:half] = 1.0
            rep = np.tile(pat, n1 // period + 1)[:n1]
            rep = make_feasible(rep, n1)
            # Add noise and optimize
            rep_noisy = make_feasible(rep + 0.3 * rng.randn(n1), n1)
            h_alt, c5_alt = subgrad_optimize(rep_noisy, n1, n_iter=3000, top_k=5, c0=0.05, time_budget=30.0)
            if c5_alt < best_c5:
                best_c5 = c5_alt
                best_h = h_alt.copy()

    # Phase 6: SLSQP polish
    if tl() > 40:
        n_s = min(len(best_h), 1024)
        h_s = upsample_h(best_h, n_s) if n_s < len(best_h) else best_h.copy()
        h_opt = slsqp_polish(h_s, n_s, T=1000, max_iter=100)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 7: Coordinate descent final polish
    if tl() > 20:
        n_cd = min(len(best_h), 1024)
        h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        h_cd, c5_cd = coord_descent(h_cd, n_cd, time_budget=25.0)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Final: upsample best to max resolution and verify
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_final
