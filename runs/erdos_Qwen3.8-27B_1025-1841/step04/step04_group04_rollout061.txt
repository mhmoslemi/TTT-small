def compute_gradient(h, T):
    """Compute gradient of soft-max C5 objective w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(T * shifted)
    weights /= weights.sum()
    
    # w[s] for s = -(n-1), ..., 0, ..., (n-1), indexed as w[s + n - 1]
    # corr index k corresponds to shift s = k - (n-1)
    # grad_h[j] = sum_s w[s] * (1-h[j+s])*[0<=j+s<n] - sum_s w[s]*h[j-s]*[0<=j-s<n]
    
    grad_h = np.zeros(n)
    w = weights  # w[k] where k = 0..2n-2, shift s = k-(n-1)
    
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)
        # First term: (1-h[j+s]) for 0<=j<n, 0<=j+s<n
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_lo < j_hi:
            grad_h[j_lo:j_hi] += wk * (1.0 - h[j_lo + s:j_hi + s])
        # Second term: -h[j-s] for 0<=j<n, 0<=j-s<n
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_lo2 < j_hi2:
            grad_h[j_lo2:j_hi2] -= wk * h[j_lo2 - s:j_hi2 - s]
    
    return grad_h * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_feasible(x, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(x - lam, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_sh = np.exp(T * shifted)
    weights = exp_sh / exp_sh.sum()
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        wk = weights[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j_lo = max(0, -s)
        j_hi = min(n, n - s)
        if j_lo < j_hi:
            grad_h[j_lo:j_hi] += wk * (1.0 - h[j_lo + s:j_hi + s])
        j_lo2 = max(0, s)
        j_hi2 = min(n, n + s)
        if j_lo2 < j_hi2:
            grad_h[j_lo2:j_hi2] -= wk * h[j_lo2 - s:j_hi2 - s]
    return grad_h * dx

def pgd_nesterov(h, n, T_sched, lr_init, iters_per_T, start_time, time_limit):
    h = make_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    beta = 0.8
    momentum = np.zeros(n)
    for T in T_sched:
        if time.time() - start_time > time_limit - 30:
            break
        lr = lr_init
        y_next = h.copy()
        for it in range(iters_per_T):
            if time.time() - start_time > time_limit - 30:
                break
            grad = compute_gradient(y_next, T)
            momentum = beta * momentum - lr * grad
            y_next = h + (beta + 1.0) * momentum / (1.0 + 1e-30) if False else h + momentum
            h_candidate = project_onto_feasible(h + momentum, n)
            c5_cand = compute_c5(h_candidate)
            if c5_cand < best_c5 + 1e-13:
                h = h_candidate
                if c5_cand < best_c5:
                    best_c5 = c5_cand
                    best_h = h.copy()
                lr = min(lr * 1.03, 0.05)
            else:
                lr *= 0.6
                h_candidate = project_onto_feasible(h - lr * grad, n)
                c5_cand2 = compute_c5(h_candidate)
                if c5_cand2 < best_c5:
                    h = h_candidate
                    best_c5 = c5_cand2
                    best_h = h.copy()
                momentum *= 0.5
                if lr < 1e-10:
                    break
    return best_h, best_c5

def _slsqp_obj(x, T_opt):
    n = T_opt[0]
    T = T_opt[1]
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_con(x, n_opt):
    return np.sum(x) - n_opt / 2.0

def _slsqp_jac(x, n_opt):
    return np.ones(n_opt)

def slsqp_polish(h, n, T, maxiter, start_time, time_limit):
    if time.time() - start_time > time_limit - 60:
        return h, compute_c5(h)
    T_opt = (n, T)
    n_opt = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        def obj_wrapped(x):
            dx = 2.0 / n
            corr = np.correlate(x, 1.0 - x, mode='full')
            mc = np.max(corr)
            shifted = corr - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        res = minimize(obj_wrapped, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'iprint': 0})
        h_res = make_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        return h_res, c5_res
    except Exception:
        return h, compute_c5(h)

def local_search_binary(h, n, start_time, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for j in range(n):
        if time.time() - start_time > time_limit - 15:
            break
        for val in [0.0, 1.0]:
            trial = best_h.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-13:
                best_c5 = c
                best_h = trial
    return best_h, best_c5

def local_search_coord(h, n, start_time, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(50):
        if time.time() - start_time > time_limit - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 5:
                break
            old = best_h[j]
            cands = [0.0, 1.0, max(0.0, old - 0.02), max(0.0, old - 0.01),
                     min(1.0, old + 0.01), min(1.0, old + 0.02), 0.5]
            best_c = best_c5
            best_val = old
            for cand in cands:
                if abs(cand - old) < 1e-14:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_val = cand
            if best_val != old:
                best_h = best_h.copy()
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return best_h, best_c5
