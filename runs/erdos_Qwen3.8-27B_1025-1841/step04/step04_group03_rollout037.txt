def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)

            step = max(step, -min_step) if min_step > 0 else step

def obj_slsqp(x):
    n = _GL_N
    T = _GL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _GL_N, _GL_T
    _GL_N = n
    _GL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def upsample(h_old, n_old, n_new):
    old_x = np.linspace(0.0, 2.0, n_old)

def update_corr(corr, b, i, j, n):
    """Compute the change in correlation when b[i] goes 1->0 and b[j] goes 0->1."""
    delta = np.zeros(2*n-1)
    
    # i (1->0), as first arg: for m in [i, n-1+i]
    # delta[m] -= (1 - b[i+n-1-m])
    # Let p = i+n-1-m, so m = i+n-1-p, p in [i, n-1]
    m_start = i
    m_end = n-1+i  # inclusive
    p_vals = i + n - 1 - np.arange(m_start, m_end+1)  # p from n-1 down to i
    delta[m_start:m_end+1] -= (1 - b[p_vals])
    
    # i (1->0), as second arg: for m in [n-1-i, 2n-2-i]
    # delta[m] += b[i-n+1+m]
    # Let q = i-n+1+m, m = q+n-1-i, q in [0, n-1]
    m_start2 = n-1-i
    m_end2 = 2*n-2-i  # inclusive
    q_vals = i - n + 1 + np.arange(m_start2, m_end2+1)  # q from 0 to n-1
    delta[m_start2:m_end2+1] += b[q_vals]
    
    # j (0->1), as first arg: for m in [j, n-1+j]
    # delta[m] += (1 - b[j+n-1-m])
    m_start3 = j
    m_end3 = n-1+j
    p_vals3 = j + n - 1 - np.arange(m_start3, m_end3+1)
    delta[m_start3:m_end3+1] += (1 - b[p_vals3])
    
    # j (0->1), as second arg: for m in [n-1-j, 2n-2-j]
    # delta[m] -= b[j-n+1+m]
    m_start4 = n-1-j
    m_end4 = 2*n-2-j
    q_vals4 = j - n + 1 + np.arange(m_start4, m_end4+1)
    delta[m_start4:m_end4+1] -= b[q_vals4]
    
    return delta

def obj_slsqp(x):
