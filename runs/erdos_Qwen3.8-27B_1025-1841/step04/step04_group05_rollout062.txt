def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        if step > 0.5:
            step = 0.5
        elif step < -0.5:
            step = -0.5
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)[-abs(int(diff)):]
            for i in idx:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
        else:
            idx = np.argsort(h)[:abs(int(-diff)):]
            for i in idx:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
    return h

def project(h, n):
    """Project onto {0<=h<=1, sum(h)=n/2} using bisection."""
    h = np.asarray(h, dtype=float)
    target = n / 2.0
    
    def sum_at_tau(tau):
        return np.sum(np.clip(h - tau, 0.0, 1.0))
    
    lo, hi = -1.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = sum_at_tau(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    h_proj = np.clip(h - tau, 0.0, 1.0)
    # Fix tiny numerical issues
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h_proj)
        else:
            idx = np.argmax(h_proj)
        h_proj[idx] = np.clip(h_proj[idx] + diff, 0.0, 1.0)
    return h_proj

    if new_n == old_n:

    old_x = np.arange(old_n) * (2.0 / old_n)

def projected_gradient_step(h, n, T, step_size, time_limit):
    """One batch of projected gradient descent iterations."""
    start_t = time.time()
    dx = 2.0 / n
    best_h = h.copy()
    best_cpp = float('inf')
    
    for it in range(500):
        if time.time() - start_t > time_limit:
            break
        
        g = 1.0 - h
        corr = np.correlate(h, g, mode='full')
        c = corr * dx
        
        # Smooth max
        max_c = np.max(c)
        shifted = T * (c - max_c)
        exp_vals = np.exp(shifted)
        sum_exp = np.sum(exp_vals)
        w = exp_vals / sum_exp
        smooth_val = (max_c + np.log(sum_exp) / T)
        
        if smooth_val < best_cpp:
            best_cpp = smooth_val
            best_h = h.copy()
        
        # Gradient
        grad = compute_gradient(h, w, dx)

        alpha = step_size
        direction = -grad
        for _ in range(20):
            trial = h + alpha * direction
            trial = project(trial, n)
            g2 = 1.0 - trial
            corr2 = np.correlate(trial, g2, mode='full')
            c2 = corr2 * dx
            max_c2 = np.max(c2)
            shifted2 = T * (c2 - max_c2)
            exp2 = np.exp(shifted2)
            sum_exp2 = np.sum(exp2)
            smooth2 = max_c2 + np.log(sum_exp2) / T
            
            if smooth2 < smooth_val - 1e-10 * abs(grad).max() * alpha:
                h = trial
                break
            alpha *= 0.5
        
        # Check convergence
        if it > 10 and abs(smooth_val - best_cpp) < 1e-12:
            # Stagnant
            pass

        size *= 2

    # Actually: np.correlate(a, b, 'full') = IFFT(FFT(a_padded) * conj(FFT(b_padded)))... 
    # No, that's convolution. Correlation is: IFFT(conj(FFT(a)) * FFT(b))

    # B = convolve(h, flip(w_shift_rev)) ... wait I need to be more careful.

    # = (flip(w) * (1-h))[i] (convolution where flip(w)[k] = w[M-1-k])
    # In FFT: A_full = IFFT(FFT(flip(w)) * FFT(1-h))
    # A = A_full[0:N] (the first N elements, for the linear convolution)
    
    # Wait, I need to be more careful about which elements I extract.
    # The full linear convolution of a (length N) and b (length M) has length N+M-1.
    # conv[k] = sum_j a[j] * b[k-j] for valid j, k-j
    # A[i] = sum_l w[l] * f[i+l] where f=1-h
    # Let's reindex: let k = i+l, so l = k-i
    # A[i] = sum_k f[k] * w[k-i] for valid k, k-i
    # = sum_j f[j] * w[j-i] where j=k (just renaming)
    # This is (f ⋆ w)[i] = cross-correlation of f and w at lag i
    # = conv(f, flip(w))[i + M - 1] (if we define convolution standardly)

    # = sum_k h[k] * w[i-k]
    # = conv(h, w)[i] (standard convolution!)
    # = IFFT(FFT(h) * FFT(w))[i]
    
    # So: B = IFFT(FFT(h, size) * FFT(w, size))[0:N]
    
    f = 1.0 - h
    f_rev = f  # not reversing f
    w_flip = w[::-1]
    
    F = np.fft.rfft(f, size)

    Hf = np.fft.rfft(h, size)
    Wt = np.fft.rfft(w, size)
    
    A_full = np.fft.irfft(F * Wf, size)
    B_full = np.fft.irfft(Hf * Wt, size)
    
    A = A_full[2*N-2 : 3*N-2]
    B = B_full[0:N]
