from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def slsqp_optimize(h, n, T, maxiter, top_idx=None):
    dx = 2.0 / n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    if top_idx is not None:
        ti = top_idx
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')[ti]
            mc = np.max(c)
            s = c - mc
            return (mc + np.log(np.sum(np.exp(T * s))) / T) * dx
    else:
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            mc = np.max(c)
            s = c - mc

                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descend(h, n, steps=None, max_rounds=3, elite_only=True):
    if steps is None:
        steps = [0.005, 0.01, 0.02, 0.05]
    best_c = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        if elite_only:
            active = np.where((h > 0.02) & (h < 0.98))[0]
        else:
            active = np.arange(n)
        for j in active:
            old = h[j]
            bv, bc = old, best_c
            for s in steps:
                for sg in (-1, 1):
                    v = old + sg * s
                    if v < 0 or v > 1:
                        continue
                    h[j] = v
                    h = make_feasible(h, n)
                    c = compute_c5(h)
                    if c < bc:
                        bc, bv = c, v
            if bv != old:
                h[j] = bv
                h = make_feasible(h, n)
                best_c = bc
                improved = True
        if not improved:
            break
    return h, best_c
