                a, b = ones[i1], ones[i2]
                c1, c2 = zeros[j1], zeros[j2]
                if a == c1 or a == c2 or b == c1 or b == c2:
                    continue
                h[a] = 0.0; h[b] = 0.0
                h[c1] = 1.0; h[c2] = 1.0
                evals += 1
                cv, _ = compute_c5_and_corr(h)
                if cv < cur - 1e-15:
                    cur = cv
                    ones[i1] = c1; ones[i2] = c2
                    zeros[j1] = a; zeros[j2] = b
                    improved_this_round = True
                else:
                    h[a] = 1.0; h[b] = 1.0
                    h[c1] = 0.0; h[c2] = 0.0
            if not improved_this_round:
                break
        elif not two_stage and not improved_this_round:
            break
    return h, cur

def simulated_annealing_binary(h_bin, n, rng, T_start=0.008, T_end=0.0003, n_steps=80000):
    h = h_bin.copy()
    cur, _ = compute_c5_and_corr(h)
    best = cur
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for step in range(n_steps):
        T = T_start * (T_end / T_start) ** (step / max(1, n_steps - 1))
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        c, _ = compute_c5_and_corr(h)
        delta = c - cur
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-12)):
            cur = c
            ones[ii] = oj
            zeros[jj] = oi
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best

def block_move_search(h_bin, n, rng, max_evals=50000):
    """Try moving blocks of consecutive 1s to new locations."""
    h = h_bin.copy()
    cur, _ = compute_c5_and_corr(h)
    evals = 0
    for _ in range(2000):
        if evals >= max_evals:
            break
        # Find a block of 1s
        ones_mask = h > 0.5
        transitions = np.diff(ones_mask.astype(int))
        starts = list(np.where(transitions == 1)[0] + 1)
        if ones_mask[0]:
            starts.append(0)
        if not starts:
            break
        ss = rng.choice(starts)
        # Find end of this block
        ee = ss
        while ee < n - 1 and h[ee + 1] > 0.5:
            ee += 1
        blen = ee - ss + 1
        if blen > n // 4 or blen < 2:
            continue
        # Choose random destination
        dest = rng.randint(0, n - blen)
        if dest <= ee + 1 and dest + blen >= ss:
            continue
        # Check if destination is clear
        if np.any(h[dest:dest + blen] > 0.5):
            continue
        # Perform the move
        h[ss:ee + 1] = 0.0
        h[dest:dest + blen] = 1.0
        evals += 1
        c, _ = compute_c5_and_corr(h)
        if c < cur - 1e-15:
            cur = c
        else:
            h[ss:ee + 1] = 1.0
            h[dest:dest + blen] = 0.0
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 640
        h0 = np.zeros(n0)
        for i in range(n0 // 2):
            h0[(i * 2) % n0] = 1.0

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = len(h0)

    # Phase 1: Quick continuous optimization (120s)
    if rem() > 100:
        n1 = best_n
        h_cur = h0.copy()
        for T in [300, 1000, 4000]:
            if rem() < 15:
                break
            h_opt = lbgfb_optimize(h_cur, n1, T, mu=500.0, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                h_cur = h_opt.copy()

    # Phase 2: Binary local search at multiple resolutions (700s)
    binary_resolutions = [best_n, 480, 320, 960]
    for n_b in binary_resolutions:
        if rem() < 100:
            break
        if n_b % 2 != 0:
            n_b += 1
        if n_b < 128:
            continue
        h_b = interp_periodic(best_h, n_b)
        h_bin = to_binary(h_b, n_b)

        # Hill climb
        h_ls, c_ls = binary_hill_climb(h_bin, n_b, rng, max_evals=200000, two_stage=True)
        if c_ls < best_c5:
            best_c5 = c_ls
            best_h = h_ls.copy()

        # Simulated annealing from current best
        if rem() > 80:
            n_sa_evals = min(60000, max(5000, int(rem() * 40)))
            h_sa, c_sa = simulated_annealing_binary(h_ls, n_b, rng, T_start=0.006, T_end=0.0002, n_steps=n_sa_evals)
            if c_sa < best_c5:
                best_c5 = c_sa
                best_h = h_sa.copy()
                h_ls = h_sa.copy()

        # Another hill climb round
        if rem() > 60:
            h_ls2, c_ls2 = binary_hill_climb(h_ls, n_b, rng, max_evals=100000, two_stage=False)
            if c_ls2 < best_c5:
                best_c5 = c_ls2
                best_h = h_ls2.copy()

        # Block moves
        if rem() > 40:
            h_bm, c_bm = block_move_search(h_ls, n_b, rng, max_evals=30000)
            if c_bm < best_c5:
                best_c5 = c_bm
                best_h = h_bm.copy()

    # Phase 3: Higher resolution continuous optimization (200s)
    if rem() > 100:
        for n2 in [1280, 1920]:
            if rem() < 80:
                break
            h2 = interp_periodic(best_h, n2)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()
            for T in [400, 2000]:
                if rem() < 25:
                    break
                h_opt = lbgfb_optimize(h2, n2, T, mu=300.0, max_iter=180)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5 = c_opt
                    best_h = h_opt.copy()
                    h2 = h_opt.copy()

    # Phase 4: Multi-start (80s)
    if rem() > 60:
        n_ms = best_n
        for restart in range(3):
            if rem() < 20:
                break
            h_r = interp_periodic(best_h, n_ms)
            h_r += 0.12 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            h_opt = lbgfb_optimize(h_r, n_ms, 600, mu=400, max_iter=150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 5: Final binary polish at best resolution (100s)
    if rem() > 50:
        n_fin = len(best_h)
        h_fin = to_binary(best_h, n_fin)
        max_ev = min(150000, max(2000, int(rem() * 150)))
        h_ps, c_ps = binary_hill_climb(h_fin, n_fin, rng, max_evals=max_ev, two_stage=True)
        if c_ps < best_c5:
            best_c5 = c_ps
            best_h = h_ps.copy()

    # Phase 6: Coordinate descent polish
    if rem() > 25:
        n_cd = len(best_h)
        h = best_h.copy()
        cur = compute_c5(h)
        steps = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
        for rnd in range(4):
            if rem() < 5:
                break
            improved = False
            for j in range(n_cd):
                for d in steps:
                    nv = h[j] + d
                    if nv < 0 or nv > 1:
                        continue
                    trial = h.copy()
                    trial[j] = nv
                    trial = make_feasible(trial, n_cd)
                    c = compute_c5(trial)
                    if c < cur - 1e-14:
                        cur = c
                        h = trial
                        improved = True
                        break
            if not improved:
                break
        if cur < best_c5:
            best_c5 = cur
            best_h = h.copy()

    # Final check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
