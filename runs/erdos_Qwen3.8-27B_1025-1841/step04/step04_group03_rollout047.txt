    """One subgradient step targeting shift s_star."""
    n = len(h)

    shift = s_star - (n - 1)

        gm = j - shift
        val = 0.0
        if 0 <= gp < n:
            val += (1.0 - h[gp])
        if 0 <= gm < n:
            val -= h[gm]
        g[j] = val
    h_new = h - alpha * g
    return make_feasible(h_new, n)

    """Subgradient method with decreasing step size."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha0 = 0.1
    for it in range(max_iters):
        if time.time() - start_time > start_time + time_limit:
            break
        corr = compute_corr(h)
        s_star = np.argmax(corr)
        alpha = alpha0 / np.sqrt(it + 1)
        h_new = subgradient_step(h, s_star, alpha)
        c5_new = np.max(corr) * dx  # Use the same corr for now (slight approximation)

        if it % 10 == 0:
            c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new

def compute_delta_flip(h, j, flip_to_0, n):
    """Compute change in all correlation values when h[j] is flipped."""
    # k ranges from -(n-1) to (n-1)
    # For flip to 0: ΔC[k] = -(1-h[j+k])*[valid] + h[j-k]*[valid]
    # For flip to 1: ΔC[k] = (1-h[j+k])*[valid] - h[j-k]*[valid]
    
    k = np.arange(-(n-1), n)  # shape (2n-1,)
    jk = j + k  # j+k for each k
    jm = j - k  # j-k for each k
    
    h_jk = np.where((jk >= 0) & (jk < n), h[jk[(jk >= 0) & (jk < n)]], 0.0)
    h_jm = np.where((jm >= 0) & (jm < n), h[jm[(jm >= 0) & (jm < n)]], 0.0)
    
    if flip_to_0:
        delta = -(1.0 - h_jk) + h_jm
    else:
        delta = (1.0 - h_jk) - h_jm
    
    return delta

```python
def delta_flip(h, j, n):
    """Compute delta in correlation array when h[j] is flipped (before the flip)."""
    k_arr = np.arange(-(n-1), n)
    jk = j + k_arr
    jm = j - k_arr
    
    mask_jk = (jk >= 0) & (jk < n)
    h_jk = np.zeros(2*n-1)
    h_jk[mask_jk] = h[jk[mask_jk]]
    
    mask_jm = (jm >= 0) & (jm < n)
    h_jm = np.zeros(2*n-1)
    h_jm[mask_jm] = h[jm[mask_jm]]
    
    if h[j] > 0.5:  # flip 1 -> 0
        return -(1.0 - h_jk) + h_jm
    else:  # flip 0 -> 1
        return (1.0 - h_jk) - h_jm
```
