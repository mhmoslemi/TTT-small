def binary_sa(h_start, n, n_iters, seed, t_init=0.01, t_final=0.001):
    """Simulated annealing with swap moves on binary sequences."""
    rng = np.random.RandomState(seed)
    h = (h_start > 0.5).astype(float).copy()
    # Ensure exactly n/2 ones
    target = n // 2
    ones = np.where(h == 1)[0]
    while len(ones) > target:
        idx = ones[rng.randint(len(ones))]
        h[idx] = 0
        ones = np.where(h == 1)[0]
    while len(ones) < target:
        zeros = np.where(h == 0)[0]
        h[zeros[rng.randint(len(zeros))]] = 1
        ones = np.where(h == 1)[0]
    
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    total_iters = n_iters
    for it in range(total_iters):
        # Temperature schedule
        frac = it / max(1, total_iters - 1)
        t = t_init * (t_final / t_init) ** frac
        
        # Pick swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        
        h[i1] = 0
        h[i0] = 1
        new_c5 = compute_c5(h)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i1] = 1
            h[i0] = 0
    
    return best_h, best_c5
