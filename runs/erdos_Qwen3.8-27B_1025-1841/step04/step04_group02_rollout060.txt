def update_corr_fast(c, h, p, delta, n):
    """Update correlation c after h[p] changes by delta. h is NOT yet updated."""
    nn1 = n - 1
    # a[p] change: affects c[k] for k where 0 <= p+k-(n-1) < n
    # i.e., n-1-p <= k <= n-1+p
    lo_a = max(0, nn1 - p)
    hi_a = min(2*nn1, nn1 + p)
    if lo_a <= hi_a:
        js = p + np.arange(lo_a, hi_a+1) - nn1  # b-indices
        mask = (js >= 0) & (js < n)
        c[lo_a:hi_a+1][mask] += delta * (1 - h[js[mask]])
    
    # b[p] change: affects c[k] for k where 0 <= p-k+(n-1) < n
    # i.e., p <= k <= p+n-1 => p <= k <= p+nn1
    lo_b = max(0, p)

    if lo_a <= nn1 <= hi_a and lo_b <= nn1 <= hi_b:

def apply_delta(c, b, p, delta, n):
    nn1 = n - 1
    total_corr_len = 2 * nn1
    lo_a = max(0, nn1 - p)
    hi_a = min(total_corr_len, 2*nn1 - p)

        js = p + np.arange(lo_a, hi_a+1) - nn1
        c[lo_a:hi_a+1] += delta * (1.0 - b[js])

    hi_b = min(total_corr_len, p + nn1)

        js = p - np.arange(lo_b, hi_b+1) + nn1
        c[lo_b:hi_b+1] += (-delta) * b[js]
