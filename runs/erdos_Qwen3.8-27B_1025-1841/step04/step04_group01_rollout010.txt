    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_coeffs(h, j, n):
    """Compute partial corr[m] / partial h[j] for all m = 0, ..., 2n-2."""
    a = np.zeros(2 * n - 1)
    a[0] = 1.0 - 2.0 * h[j]
    
    # m = 1 to n-1
    m_arr = np.arange(1, n, dtype=int)
    # term1: (1-h[j+m]) for m where j+m < n
    jm = j + m_arr  # array of indices j+1, ..., j+n-1
    term1 = np.zeros(n - 1)
    v1 = jm < n
    if np.any(v1):
        term1[v1] = 1.0 - h[jm[v1]]
    # term2: h[j-m] for m where j-m >= 0
    jm2 = j - m_arr  # array of indices j-1, ..., j-(n-1)
    term2 = np.zeros(n - 1)
    v2 = jm2 >= 0
    if np.any(v2):
        term2[v2] = h[jm2[v2]]
    a[1:n] = term1 - term2
    
    # m = n to 2n-2, r = 2n-2-m (r goes from n-2 down to 0)
    r = np.arange(n - 2, -1, -1, dtype=int)  # n-2, n-3, ..., 0
    # term1: (1-h[j-r]) for r where 2r >= j and r <= j (i.e., ceil(j/2) <= r <= j)
    term1b = np.zeros(n - 1)
    v1b = (r <= j) & (2 * r >= j)
    if np.any(v1b):
        term1b[v1b] = 1.0 - h[j - r[v1b]]
    # term2: h[j+r] for r where j+r < n
    term2b = np.zeros(n - 1)
    jr = j + r
    v2b = jr < n
    if np.any(v2b):
        term2b[v2b] = h[jr[v2b]]
    a[n:] = term1b - term2b
    
    return a

def solve_1d_minimax(corr, a, old_val):
    """Minimize max_m(corr[m] + a[m]*t) over t in [-old_val, 1-old_val]."""
    lo = -old_val
    hi = 1.0 - old_val
    if hi - lo < 1e-14:
        return 0.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = corr + a * mid
        max_val = np.max(vals)
        active = np.where(vals >= max_val - 1e-14)[0]
        if len(active) == 0:
            active = np.array([np.argmax(vals)])
        max_slope = a[active].max()
        min_slope = a[active].min()
        if max_slope > 1e-12:
            hi = mid
        elif min_slope < -1e-12:
            lo = mid
        else:
            break
    
    t_opt = (lo + hi) / 2.0
    f_opt = np.max(corr + a * t_opt)
    f_lo = np.max(corr + a * lo)
    f_hi = np.max(corr + a * hi)
    if f_lo < f_opt - 1e-14:
        t_opt = lo
    if f_hi < f_opt - 1e-14:
        t_opt = hi
    return t_opt

def exact_coord_sweep(h, n, max_passes=100, time_limit=None, start_time=None):
    """Run exact coordinate descent with 1D minimax subproblem."""
    best_c5 = compute_c5(h)
    corr = get_corr(h).copy()
    
    for sweep in range(max_passes):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        for j in range(n):
            old = h[j]
            a = compute_coeffs(h, j, n)
            t_opt = solve_1d_minimax(corr, a, old)
            if abs(t_opt) > 1e-10:
                new_val = old + t_opt
                new_val = np.clip(new_val, 0.0, 1.0)
                actual_t = new_val - old
                if abs(actual_t) > 1e-12:
                    h[j] = new_val
                    corr += a * actual_t
                    improved = True
        
        # Re-feasibilize
        h = make_feasible(h, n)
        corr = get_corr(h).copy()
        c5_new = float(np.max(corr) * (2.0 / n))
        
        if c5_new < best_c5:
            best_c5 = c5_new
        
        if not improved:
            break
    
    return h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
