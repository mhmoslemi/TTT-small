def compute_all_overlaps(h):

def obj_slsqp(x):

    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)

    shifted = corr - mc
    # Clip to avoid underflow

    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def obj_and_grad(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    G = np.max(corr)
    shifted = T * (corr - G)
    shifted = np.clip(shifted, -50.0, 0.0)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Compute gradient: dx * sum_k w[k] * dcorr[k]/dx
    # dcorr[k]/dx[m] = (1 - x[m+k-(n-1)]) [if valid] - x[m-k+(n-1)] [if valid]
    grad = np.zeros(n)
    offset = n - 1
    for k in range(2*n - 1):
        if weights[k] < 1e-10:
            continue
        lag = k - offset
        # First term: (1 - x[m+lag]) for m where 0<=m<n and 0<=m+lag<n
        m_min = max(0, -lag)
        m_max = min(n, n - lag) if lag >= 0 else n
        # Actually: 0 <= m < n and 0 <= m+lag < n
        # So: max(0, -lag) <= m < min(n, n-lag)
        Lo = max(0, -lag)
        Hi = min(n, n - lag)
        if Hi > Lo:
            grad[Lo:Hi] += weights[k] * (1.0 - x[Lo+lag:Hi+lag])
        # Second term: -x[m-lag] for m where 0<=m<n and 0<=m-lag<n
        # i.e., lag <= m < n+lag, intersect with [0, n)
        Lo2 = max(0, lag)
        Hi2 = min(n, n + lag)
        if Hi2 > Lo2:
            grad[Lo2:Hi2] -= weights[k] * x[Lo2-lag:Hi2-lag]
    
    val = (G + np.log(np.sum(weights)) / T) * dx
    return val, dx * grad

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_and_grad(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    G = np.max(corr)
    shifted = T * (corr - G)
    shifted = np.clip(shifted, -40.0, 0.0)
    exp_s = np.exp(shifted)
    s = np.sum(exp_s)
    w = exp_s / s
    grad = np.zeros(n)
    offset = n - 1
    sig = np.where(w > 1e-12)[0]
    for k in sig:
        lag = k - offset
        Lo = max(0, -lag)
        Hi = min(n, n - lag)
        if Hi > Lo:
            grad[Lo:Hi] += w[k] * (1.0 - x[Lo + lag:Hi + lag])
        Lo2 = max(0, lag)
        Hi2 = min(n, n + lag)
        if Hi2 > Lo2:
            grad[Lo2:Hi2] -= w[k] * x[Lo2 - lag:Hi2 - lag]
    val = (G + np.log(s) / T) * dx

        else:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            if 'start' in dir() and bsize >= 2 and rng.random() < 0.3:
                b[start + co] = 1
                b[start + cz] = 0
            else:
                try:
                    b[i] = 1
                    b[j] = 0
                except Exception:
                    pass
    if curr_c5 < best_c5:
        best_c5 = curr_c5
        best_b = b.copy()
    return best_b, best_c5

def fourier_perturb(h, n, rng, n_modes=3, amp=0.03):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 8)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP with analytical gradients - fast
    for T in [100.0, 500.0, 2000.0, 8000.0, 30000.0]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 15:
            break
        try:
            x_res = slsqp_opt(best_h, n0, T, maxiter=400, ftol=1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 2: Multi-restart SLSQP with perturbations
    for _ in range(3):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 60:
            break
        try:
            hp = fourier_perturb(best_h, n0, rng, n_modes=3, amp=0.02)
            for T in [2000.0, 10000.0]:
                remaining = budget_s - (time.time() - start_time)
                if remaining < 20:
                    break
                x_res = slsqp_opt(hp, n0, T, maxiter=300, ftol=1e-15)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n0)
                    hp = fourier_perturb(best_h, n0, rng, n_modes=2, amp=0.015)
        except Exception:
            pass

    # Phase 3: Binary targeted local search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        tl = min(remaining - 30, 80)
        b, c5_b = binary_targeted_search(best_h, n0, 5000, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase 4: SLSQP after binary search
    for T in [2000.0, 10000.0]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 15:
            break
        try:
            x_res = slsqp_opt(best_h, n0, T, maxiter=400, ftol=1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 5: General SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 40, 120)
        b, c5_b = general_sa(best_h, n0, 300000, 0.004, 0.000003, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 40:
                try:
                    x_res = slsqp_opt(best_h, n0, 5000.0, maxiter=300, ftol=1e-15)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(x_res, n0)
                except Exception:
                    pass

    # Phase 6: Another targeted search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        tl = min(remaining - 20, 50)
        b, c5_b = binary_targeted_search(best_h, n0, 4000, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase 7: SLSQP refinement
    for T in [5000.0, 20000.0, 80000.0]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 15:
            break
        try:
            x_res = slsqp_opt(best_h, n0, T, maxiter=400, ftol=1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 8: Higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        for T in [3000.0, 10000.0, 40000.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break
            try:
                x_res = slsqp_opt(h_up, n1, T, maxiter=300, ftol=1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n1)
            except Exception:
                pass
        remaining = budget_s - (time.time() - start_time)
        if remaining > 50:
            b, c5_b = binary_targeted_search(best_h, n1, 3000, rng, min(remaining - 20, 40), start_time)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = b.copy()
                remaining = budget_s - (time.time() - start_time)
                if remaining > 30:
                    try:
                        x_res = slsqp_opt(best_h, n1, 10000.0, maxiter=200, ftol=1e-14)
                        c5_res = compute_c5(x_res)
                        if c5_res < best_c5:
                            best_c5 = c5_res
                            best_h = make_feasible(x_res, n1)
                    except Exception:
                        pass

    # Phase 9: Even higher resolution if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        n2 = n0 + 960
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, n2)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n2)
        for T in [5000.0, 20000.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break
            try:
                x_res = slsqp_opt(h_up, n2, T, maxiter=200, ftol=1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n2)
            except Exception:
                pass

    # Phase 10: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        for T in [20000.0, 100000.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 10:
                break
            try:
                x_res = slsqp_opt(best_h, n_pts, T, maxiter=300, ftol=1e-15)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n_pts)
            except Exception:
                pass

    # Phase 11: Final binary check
    remaining = budget_s - (time.time() - start_time)
    if remaining > 10:
        n_pts = len(best_h)
        b = (best_h >= 0.5).astype(float)
        target = n_pts // 2
        while int(np.sum(b)) > target:
            ones = np.where(b == 1)[0]
            if len(ones) == 0:
                break
            b[ones[rng.randint(len(ones))]] = 0
        while int(np.sum(b)) < target:
            zeros = np.where(b == 0)[0]
            if len(zeros) == 0:
                break
            b[zeros[rng.randint(len(zeros))]] = 1
        b_c5 = compute_c5(b)
        if b_c5 < best_c5:
            best_c5 = b_c5
            best_h = b.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
