        b_a, c5_a = sa_mixed(b_alt, n_alt, 200000, 0.012, 0.00001, rng, t_limit)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = b_a.copy()
        remaining = budget_s - (time.time() - start_time)
        if c5_a < best_c5 + 0.002 and remaining > 100:
            try:
                x_res = slsqp_run(best_h, n_alt, 120.0, 300, 1e-14)
                x_res = make_feasible(x_res, n_alt)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = x_res.copy()
            except Exception:
            pass
        remaining = budget_s - (time.time() - start_time)

    # Phase 5: Second SA pass on best (with different temp schedule)
    remaining = budget_s - (time.time() - start_time)
    n_pts = len(best_h)
    if remaining > 150:
        b_cur = to_binary(best_h, n_pts)
        t_limit = min(180, remaining - 60)
        b5, c5_5 = sa_mixed(b_cur, n_pts, 400000, 0.008, 0.000005, rng, t_limit)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = b5.copy()

    # Phase 6: SLSQP with higher T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 200.0, 400, 1e-15)
            x_res = make_feasible(x_res, n_pts)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass

    # Phase 7: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 20, start_time, budget_s)

    # Phase 8: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 500.0, 300, 1e-16)
            x_res = make_feasible(x_res, n_pts)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass

    # Phase 9: Final binary SA (short)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        n_pts = len(best_h)
        b_final = to_binary(best_h, n_pts)
        t_limit = min(50, remaining - 20)
        b_f, c5_f = sa_mixed(b_final, n_pts, 100000, 0.003, 0.000001, rng, t_limit)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = b_f.copy()

    # Final safety
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 5:
        try:
            evaluate_erdos_solution(best_h, len(best_h))
        except Exception:
            pass
    return best_h, best_c5, len(best_h)
