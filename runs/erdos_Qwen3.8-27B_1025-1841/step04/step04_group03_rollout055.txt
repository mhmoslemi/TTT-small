    # Phase 2: Enhanced binary SA (main search)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        tl = min(remaining - 50, 350)
        b, c5b = enhanced_sa(best_h, n0, 300000, 0.006, 0.000003, rng, tl, start_time)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()

    # Phase 3: Lag-targeted refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = lag_targeted(best_h, n0, rng, min(remaining - 20, 80), start_time)

    # Phase 4: SLSQP refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        for T in [200.0, 1000.0, 5000.0, 20000.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break
            try:
                xr = slsqp_opt(best_h, n0, T, 150, 1e-14)
                c5r = compute_c5(xr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = make_feasible(xr, n0)
            except Exception:
                pass

    # Phase 5: Another SA round with tighter parameters
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 40, 200)
        b, c5b = enhanced_sa(best_h, n0, 200000, 0.003, 0.000001, rng, tl, start_time)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()

    # Phase 6: Perturbation + SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        for _ in range(4):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 50:
                break
            try:
                hp = fourier_perturb(best_h, n0, rng, n_modes=3, amp=0.025)
                xr = slsqp_opt(hp, n0, 1000.0, 100, 1e-13)
                c5r = compute_c5(xr)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = make_feasible(xr, n0)
            except Exception:
                pass

    # Phase 7: Multiple diverse starts at lower resolution, upscale
    remaining = budget_s - (time.time() - start_time)
    if remaining > 180:
        n_coarse = 120
        for period, ones_p in [(8, 4), (12, 6), (15, 7), (16, 8), (20, 10), (24, 12), (10, 5)]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
            pat = periodic_pattern(n_coarse, period, ones_p, seed + period)
            # Quick SLSQP at coarse level
            try:
                xr = slsqp_opt(pat, n_coarse, 500.0, 80, 1e-12)
                c5c = compute_c5(xr)
                # Upsample to n0
                old_x = np.linspace(0.0, 2.0, n_coarse)
                new_x = np.linspace(0.0, 2.0, n0)
                h_up = np.interp(new_x, old_x, xr)
                h_up = make_feasible(h_up, n0)
                # Refine at full resolution
                xr2 = slsqp_opt(h_up, n0, 2000.0, 80, 1e-13)
                c5r = compute_c5(xr2)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = make_feasible(xr2, n0)
            except Exception:
                pass

    # Phase 8: Final refinements
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        best_h, best_c5 = lag_targeted(best_h, n0, rng, min(remaining - 20, 50), start_time)

    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        try:
            xr = slsqp_opt(best_h, n0, 30000.0, 200, 1e-15)
            c5r = compute_c5(xr)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = make_feasible(xr, n0)
        except Exception:
            pass

    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, best_c5 = coord_descent(best_h.copy(), n0, 10, start_time, budget_s)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
