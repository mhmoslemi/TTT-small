        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    mask = (h < 1.0) if diff > 0 else (h > 0.0)
    cnt = float(np.sum(mask))
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_subgrad(h, n, active_shifts):
    grad = np.zeros(n)
    for s in active_shifts:
        if s > 0:
            grad[:n - s] -= (1.0 - h[s:n])
            grad[s:n] -= h[0:n - s]
        elif s < 0:
            sp = -s
            grad[sp:n] -= (1.0 - h[0:n - sp])
            grad[0:n - sp] -= h[sp:n]
        else:
            grad[:] -= (1.0 - h)
            grad[:] -= h
    return grad

def pgd_vectorized(h, n, budget, start_time, top_m=5, thresh=0.90):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha = 0.008
    vel = np.zeros(n)
    mom = 0.6
    it = 0
    patience = 0
    max_pat = 60

    while time.time() - start_time < budget:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = np.max(corr) * (2.0 / n)

        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_pat:
                alpha *= 0.6
                if alpha < 1e-9:
                    break
                patience = 0

        max_corr = np.max(corr)
        thr = max_corr * thresh
        active = np.where(corr >= thr)[0]
        if len(active) > top_m:
            active = active[np.argsort(corr[active])[-top_m:]]
        shifts = active - (n - 1)
        grad = compute_subgrad(h, n, shifts)

        y = np.clip(h - mom * vel, 0, 1)
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        max_y = np.max(corr_y)
        thr_y = max_y * thresh
        active_y = np.where(corr_y >= thr_y)[0]
        if len(active_y) > top_m:
            active_y = active_y[np.argsort(corr_y[active_y])[-top_m:]]
        shifts_y = active_y - (n - 1)
        grad_y = compute_subgrad(y, n, shifts_y)

        vel = mom * vel - alpha * grad_y
        new_h = project_feasible(h + vel, n)
        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur:
            h = new_h
        else:
            new_h2 = project_feasible(h - alpha * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_cur:
                h = new_h2
            else:
                alpha *= 0.85
                vel *= 0.5

        if it % 80 == 0 and alpha < 0.015:
            alpha *= 1.25
            mom = min(0.80, mom + 0.01)

    h = project_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h = h
        best_c5 = c5_h
    return best_h, best_c5

def binary_local_search(h_bin, n, budget, start_time, rng):
    best_h = h_bin.copy()
    best_c5 = compute_c5(best_h)
    ones = np.where(best_h == 1)[0].copy()
    zeros = np.where(best_h == 0)[0].copy()
    stuck = 0

    while time.time() - start_time < budget:
        improved = False
        n_try = 30 + stuck * 5
        for _ in range(n_try):
            if time.time() - start_time > budget - 1:
                break
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            oi, oz = int(ones[i]), int(zeros[j])
            best_h[oi] = 0.0
            best_h[oz] = 1.0
            c5_try = compute_c5(best_h)
            if c5_try < best_c5 - 1e-15:
                best_c5 = c5_try
                ones[i] = oz
                zeros[j] = oi
                improved = True
                stuck = 0
            else:
                best_h[oi] = 1.0
                best_h[oz] = 0.0
        if not improved:
            stuck += 1
            if stuck > 10:
                break
    return best_h, best_c5

def slsqp_polish(h, n, budget, start_time, T=2000.0):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-18})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h):
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h, compute_c5(h)

def coord_descent_fine(h, n, budget, start_time):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    steps = [0.002, 0.001, 0.0005, 0.0002]
    improved = True
    while improved and time.time() - start_time < budget:
        improved = False
        for j in range(n):
            if time.time() - start_time > budget - 1:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in steps:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 < v < 1.0 and abs(v - old_val) > 1e-10:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Vectorized PGD (main)
    if remaining() > 300:
        h, c5 = pgd_vectorized(best_h, n, min(300, remaining() - 40), start_time, top_m=5, thresh=0.88)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: SLSQP high-T
    if remaining() > 180:
        h, c5 = slsqp_polish(best_h, n, min(150, remaining() - 30), start_time, T=3000.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: PGD with tighter top set
    if remaining() > 180:
        h, c5 = pgd_vectorized(best_h, n, min(150, remaining() - 30), start_time, top_m=3, thresh=0.85)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Binary local search
    if remaining() > 180:
        h_bin = (best_h > 0.5).astype(float)
        target = n // 2
        ones = np.where(h_bin == 1)[0]
        if len(ones) > target:
            order = np.argsort(best_h[ones])[:len(ones) - target]
            h_bin[ones[order]] = 0.0
        elif len(ones) < target:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros])[:target - len(ones)]
            h_bin[zeros[order]] = 1.0
        h_bin, c5_bin = binary_local_search(h_bin, n, min(120, remaining() - 30), start_time, rng)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Phase 5: Fine coordinate descent
    if remaining() > 120:
        h, c5 = coord_descent_fine(best_h, n, min(100, remaining() - 20), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: SLSQP ultra-high T
    if remaining() > 100:
        h, c5 = slsqp_polish(best_h, n, min(80, remaining() - 20), start_time, T=8000.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: PGD again with different params
    if remaining() > 120:
        h, c5 = pgd_vectorized(best_h, n, min(100, remaining() - 30), start_time, top_m=4, thresh=0.92)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 8: Random perturbations + refine
    for restart in range(5):
        if remaining() < 50:
            break
        scale = 0.004 + 0.004 * restart
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        h2, c5_2 = pgd_vectorized(h_try, n, min(35, remaining() - 25), start_time, top_m=4, thresh=0.88)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 9: Another binary search from current best
    if remaining() > 100:
        h_bin = (best_h > 0.5).astype(float)
        target = n // 2
        ones = np.where(h_bin == 1)[0]
        if len(ones) > target:
            order = np.argsort(best_h[ones])[:len(ones) - target]
            h_bin[ones[order]] = 0.0
        elif len(ones) < target:
            zeros = np.where(h_bin == 0)[0]
            order = np.argsort(-best_h[zeros])[:target - len(ones)]
            h_bin[zeros[order]] = 1.0
        h_bin, c5_bin = binary_local_search(h_bin, n, min(80, remaining() - 25), start_time, rng)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Phase 10: Final SLSQP
    if remaining() > 30:
        h, c5 = slsqp_polish(best_h, n, min(25, remaining() - 8), start_time, T=10000.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 11: Final ultra-fine coord descent
    if remaining() > 15:
        h, c5 = coord_descent_fine(best_h, n, min(10, remaining() - 5), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
