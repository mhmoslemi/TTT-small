_SL_T = 0.0

def obj_slsqp(x):

def slsqp_optimize(h, n, T, maxiter, ftol):

    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,

If we change h[i] from 1 to 0: corr[k] changes by -(1-h[i+k]) for all k where i+k is valid, and by -h[i-k] for all k where i-k is valid (from the (1-h) side).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_G_N = 0
_G_T = 0.0

def obj_slsqp(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Log-sum-exp for numerical stability
    if T < 1e-10:
        return mc * dx
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp(h, n, T, maxiter=100, ftol=1e-13):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def sa_binary(h, n, max_iters, T0, T1, rng, time_limit, start_time, block_size=1):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        b[zeros[rng.randint(len(zeros))]] = 1
    
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        
        if block_size == 1:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
            new_c = compute_c5(b)
            if new_c <= curr or (T > 1e-12 and rng.random() < np.exp(-(new_c - curr) / T)):
                curr = new_c
                if new_c < best:
                    best = new_c
                    best_b = b.copy()
            else:
                b[i] = 1
                b[j] = 0
        else:
            start = rng.randint(0, n - block_size)
            seg = b[start:start+block_size]
            oi = np.where(seg == 1)[0]
            zi = np.where(seg == 0)[0]
            if len(oi) == 0 or len(zi) == 0:
                continue
            sc = min(len(oi), len(zi))
            co = rng.choice(oi, size=sc, replace=False)
            cz = rng.choice(zi, size=sc, replace=False)
            b[start+co] = 0
            b[start+cz] = 1
            new_c = compute_c5(b)
            if new_c <= curr or (T > 1e-12 and rng.random() < np.exp(-(new_c - curr) / T)):
                curr = new_c
                if new_c < best:
                    best = new_c
                    best_b = b.copy()
            else:
                b[start+co] = 1
                b[start+cz] = 0
    
    return best_b, best

def perturb(h, n, rng, mode='fourier'):

        h_new = h.copy()
        for _ in range(rng.randint(3, 8)):

            phase = rng.uniform(0, 2 * np.pi)
            h_new += np.sin(2 * np.pi * freq * x / n + phase) * rng.uniform(0.02, 0.08)
        return make_feasible(h_new, n)
    elif mode == 'segment':
        h_new = h.copy()
        for _ in range(rng.randint(2, 5)):
            length = rng.randint(n // 8, n // 3)
            start = rng.randint(0, max(1, n - length))
            if rng.random() < 0.5:
                h_new[start:start+length] = 1.0 - h_new[start:start+length]
            else:
                h_new[start:start+length] = rng.uniform(0, 1, size=length)
        return make_feasible(h_new, n)
    elif mode == 'gaussian':
        h_new = h + rng.normal(0, 0.05, size=n)
        return make_feasible(h_new, n)
    else:
        return make_feasible(h.copy() + rng.uniform(-0.1, 0.1, size=n), n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_G_N = 0
_G_T = 0.0

def obj_slsqp(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    if T < 1e-10:
        return mc * dx
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

    return np.ones(len(x))

    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]

        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
