```python
        corr = np.correlate(h, 1.0 - h, mode='full')
        self.weights = np.exp(self.T * (corr - np.max(corr)))
        self.weights /= self.weights.sum()
        # Normalize
        self.eq_max = np.max(corr)
    
    def value(self, h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(self.T * (corr - mc)))) / self.T) * self.dx
    
    def gradient(self, h):
        # Compute weights based on current corr
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        w = np.exp(self.T * (corr - mc))
        w /= w.sum()
        # Gradient: (1/T) * sum_k w[k] * [(1-h[i+k]) - h[i-k]]
        # This is approximately (1/T) * cross_correlation(w, 1-2h)
        # But we need to be careful about the full correlation indexing
        
        # corr has length 2n-1, index k corresponds to lag = k - (n-1)
        # w has the same indexing
        # For position i, the gradient is:
        # (1/T) * sum_k w[k] * [(1-h[i+lag_k]) * 1_{0≤i+lag_k<n} - h[i-lag_k] * 1_{0≤i-lag_k<n}]
        
        # Let's compute this directly for now (O(n²) but n=960 is manageable)
        n = self.n
        grad = np.zeros(n)
        for i in range(n):
            s = 0.0
            for k in range(2*n-1):
                lag = k - (n-1)
                ip = i + lag
                im = i - lag
                term = 0.0
                if 0 <= ip < n:
                    term += (1 - h[ip])
                if 0 <= im < n:
                    term -= h[im]
                s += w[k] * term
            grad[i] = s / self.T
        return grad
```

```python
def smooth_grad(h, T):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    w = np.exp(T * (corr - mc))
    w /= w.sum()
    
    # w has length 2n-1, index k corresponds to lag = k - (n-1)
    # lags = np.arange(2*n-1) - (n-1)
    
    grad = np.zeros(n)
    # For each position i, grad[i] = (1/T) * sum_k w[k] * [(1-h[i+lag_k])*1_{valid} - h[i-lag_k]*1_{valid}]
    
    # Vectorized:
    # First term: sum_k w[k] * (1-h[i+lag_k]) for valid i+lag_k
    # Second term: sum_k w[k] * h[i-lag_k] for valid i-lag_k
    
    # We can compute these as convolutions
    
    # Let's create a matrix approach:
    # For position i, the valid lags for the first term are those where 0 <= i + (k-(n-1)) < n
    # i.e., -(i-(n-1)) <= k <= n-1-i+(n-1) = 2n-2-i
    # Wait, let me just do it with numpy broadcasting for small n
    
    # Actually for n=960, let me just loop over i
    lags = np.arange(2*n-1) - (n-1)  # lags from -(n-1) to n-1
    
    for i in range(n):
        # First term: (1-h[i+lag]) for valid i+lag
        idx1 = i + lags
        mask1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.sum(w * np.where(mask1, 1.0 - h[idx1[mask1]], 0.0))
        
        # Second term: h[i-lag] for valid i-lag
        idx2 = i - lags
        mask2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.sum(w * np.where(mask2, h[idx2[mask2]], 0.0))
        
        grad[i] = (term1 - term2) / T
    
    return grad
```

```python
import numpy as np
from scipy.fft import fft, ifft, next_fast_len
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng, time_limit=None, start_time=None):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time_limit and time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters

        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def smooth_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    w = np.exp(T * (corr - mc))
    w /= w.sum()
    
    lags = np.arange(2*n-1) - (n-1)
    
    # Vectorized gradient computation
    idx1 = np.arange(n)[:, None] + lags[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(mask1, h[np.clip(idx1, 0, n-1)], 0.0)
    term1 = np.sum(w[None, :] * (1.0 - h1), axis=1)
    
    idx2 = np.arange(n)[:, None] - lags[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    term2 = np.sum(w[None, :] * h2, axis=1)
    
    grad = (term1 - term2) * dx / T
    return grad

def grad_slqp(h, n, T, maxiter, ftol, start_time, budget_s):
    def obj(x):
        return smooth_max_obj(x, T)
    
    def obj_grad(x):
        return smooth_max_grad(x, T)
    
    def con_eq(x):
        return np.sum(x) - n / 2.0
    
    def con_eq_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_eq_jac}]
    
    res = minimize(obj, h.copy(), jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.fun

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.1), max(0.0, old - 0.01),
                     min(1.0, old + 0.01), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5
```

```python
def smooth_max_grad_fast(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    w = np.exp(T * (corr - mc))
    w /= w.sum()
    # w has length 2n-1, index k corresponds to lag = k - (n-1)
    
    N = next_fast_len(max(2*n - 1, 2 * len(w) + n - 1))
    # Actually, for cross-correlation of two signals of lengths n and 2n-1:
    # We need N >= n + (2n-1) - 1 = 3n - 2
    N = next_fast_len(3 * n - 2)
    
    # A_i = cross_correlation(1-h, w)[i] = sum_j (1-h[j]) * w[j-i+(n-1)]
    # Using FFT: C = ifft(fft(a, N) * conj(fft(b, N)))
    # C[k] = sum_n a[(n+k) mod N] * conj(b[n])
    # For zero-padded a (length n) and b (length 2n-1), with N >= n + 2n-1 - 1:
    # C[k] for k=0..n-2 gives the "positive shift" part
    
    # Hmm, this is getting complicated with the indexing. Let me just use the 
    # O(n²) vectorized approach which is fast enough for n<=1920.
    
    lags = np.arange(2*n-1) - (n-1)
    
    idx1 = np.arange(n)[:, None] + lags[None, :]
    mask1 = (idx1 >= 0) & (idx1 < n)
    h1 = np.where(mask1, h[np.clip(idx1, 0, n-1)], 0.0)
    term1 = np.sum(w[None, :] * (1.0 - h1), axis=1)
    
    idx2 = np.arange(n)[:, None] - lags[None, :]
    mask2 = (idx2 >= 0) & (idx2 < n)
    h2 = np.where(mask2, h[np.clip(idx2, 0, n-1)], 0.0)
    term2 = np.sum(w[None, :] * h2, axis=1)
    
    grad = (term1 - term2) * dx / T
    return grad
```

```python
def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)

    return make_feasible(h_new, new_n)

    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0

        zeros = np.where(b == 0)[0]

def smooth_max_grad_fn(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    w = np.exp(T * (corr - mc))
    w /= w.sum()
    lags = np.arange(2 * n - 1) - (n - 1)

    def obj_grad(x):
        return smooth_max_grad_fn(x, T)

        return np.sum(x) - n / 2.0

        return np.ones(n)

                        c5_res = compute_c5(x_res)
                        if c5_res < best_c5:
                            best_c5 = c5_res
                            best_h = x_res
                    except Exception:
                        pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
