import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fft_correlation(h):
    """Compute np.correlate(h, 1-h, 'full') using FFT."""
    n = len(h)
    N = 2 * n
    g = 1.0 - h
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    r = np.fft.irfft(H * np.conj(G), N)
    return np.concatenate([r[n+1:], r[:n]])

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_smooth(h0, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def obj(x):
        corr = fft_correlation(x)
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def mirror_descent(h0, n, n_iters=500, eta=0.1, T_active=50):
    """Mirror descent for min-max with multiplicative weight updates."""
    dx = 2.0 / n
    h = h0.copy()
    # p is the distribution over lags (length 2n-1)
    p = np.ones(2*n-1) / (2*n-1)
    
    for t in range(n_iters):
        corr = fft_correlation(h)
        # Update p (multiplicative weights)
        p = p * np.exp(eta * corr / (dx + 1e-10))
        p = np.clip(p, 1e-300, None)
        p /= p.sum()
        
        # Compute subgradient g[j] = dx * sum_s p[s_idx] * dcorr[s]/dh[j]
        # For efficiency, only use active lags
        max_corr = np.max(corr)
        active = np.where(corr > max_corr - 0.01 / dx)[0]
        if len(active) > 50:
            active = np.where(corr > max_corr - 0.05 / dx)[0]
        
        g = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)  # shift
            w = p[idx]
            if s >= 0:
                # dcorr[idx]/dh[j] = (1-h[j+s])*1_{j+s<n} - h[j-s]*1_{j-s>=0}
                # First term: (1-h[j+s]) for j = 0, ..., n-1-s
                # Second term: -h[j-s] for j = s, ..., n-1

                g[:n-s] += w * first
                if s > 0:

            else:

                g[:n-m] += w * first

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fft_correlation(h):
    n = len(h)
    N = 2 * n
    g = 1.0 - h
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(g, N)
    r = np.fft.irfft(H * np.conj(G), N)
    return np.concatenate([r[n+1:], r[:n]])

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -50, 50)
    return mc + np.log(np.sum(np.exp(shifted))) / T

    def obj(x):
        corr = fft_correlation(x)
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': gtol})
    return make_feasible(res.x, n)

def mirror_descent_run(h0, n, n_iters=2000, eta_scale=0.5, active_thresh=0.002):
    dx = 2.0 / n
    h = h0.copy()
    p = np.ones(2 * n - 1) / (2 * n - 1)
    
    for t in range(n_iters):
        corr = fft_correlation(h)
        max_corr = np.max(corr)
        
        # Update p
        p = p * np.exp(eta_scale * (corr - max_corr) * 10.0)
        p = np.clip(p, 1e-300, None)
        p /= p.sum()
        
        # Compute gradient from active lags
        active = np.where(corr > max_corr - active_thresh / dx)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        if len(active) > 30:
            active = np.where(corr > max_corr - 5 * active_thresh / dx)[0]
        
        g = np.zeros(n)
        for idx in active:
            s = idx - (n - 1)
            w = p[idx]
            if s >= 0:
                m = s
                g[:n - m] += w * (1.0 - h[m:])
                if m > 0:
                    g[m:] -= w * h[:n - m]
            else:
                m = -s
                g[:n - m] += w * (1.0 - h[m:])
                g[m:] -= w * h[:n - m]
        
        g *= dx
        
        # Update

        eta = eta_scale / max_g
        h = h * np.exp(-eta * g)
        h = make_feasible(h, n)
    
    return h
