def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0.0, 1.0))
    return h

def compute_smooth_grad(h, T, top_m=0):
    """Compute log-sum-exp smoothed objective value and gradient.
    If top_m > 0, use only top-m correlation values."""
    n = len(h)
    dx = 2.0 / n
    cvals = np.correlate(h, 1.0 - h, mode='full')
    nc = len(cvals)
    
    if top_m > 0 and top_m < nc:
        top_idx = np.argpartition(cvals, -top_m)[-top_m:]
    else:
        top_idx = np.arange(nc)
    
    tv = cvals[top_idx]
    mc = np.max(tv)
    exps = np.exp(T * (tv - mc))
    se = np.sum(exps)
    obj = (mc + np.log(se) / T) * dx
    wts = exps / se
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        w = wts[i]

        if s >= 0:
