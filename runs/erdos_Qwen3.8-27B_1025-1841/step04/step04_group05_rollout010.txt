Now, for the main optimization, let me think about the gradient computation.

Given h of length n, the 'full' correlation has 2n-1 entries indexed 0 to 2n-2.
For entry m (0 ≤ m ≤ 2n-2), let s = m - (n-1) (so s ranges from -(n-1) to n-1).

corr[m] = ∑_{i=max(0,-s)}^{min(n-1, n-1-s)} h[i] · (1-h[i+s])

The gradient of corr[m] with respect to h[j]:
- From the h[i] term (i=j): (1-h[j+s]) if 0 ≤ j+s ≤ n-1 and j is in the valid range for i
- From the (1-h[i+s]) term (i+s=j, so i=j-s): -h[j-s] if 0 ≤ j-s ≤ n-1 and (j-s) is in the valid range for i

So:
grad[j] = (1-h[j+s]) · 1_{0≤j≤min(n-1, n-1-s) and 0≤j+s≤n-1} 
         - h[j-s] · 1_{max(0,-s)≤j-s≤min(n-1, n-1-s) and 0≤j≤n-1}

Simplifying:
- First term: j+s is in [0, n-1] AND j is in [max(0,-s), min(n-1,n-1-s)]
  Since j+s in [0,n-1] means j in [-s, n-1-s], and j in [max(0,-s), min(n-1,n-1-s)]:
  The intersection is j in [max(0,-s), min(n-1, n-1-s)]
  
- Second term: j-s is in [0, n-1] AND j-s is in [max(0,-s), min(n-1,n-1-s)]
  j-s in [0,n-1] means j in [s, s+n-1]
  j-s in [max(0,-s), min(n-1,n-1-s)] means j in [s+max(0,-s), s+min(n-1,n-1-s)]
  Intersection with j in [0, n-1]:
  j in [max(0, s, s+max(0,-s)), min(n-1, s+n-1, s+min(n-1,n-1-s))]
  
  If s ≥ 0: j in [s, min(n-1, n-1)] = [s, n-1] (since s+max(0,-s) = s, s+min(n-1,n-1-s) = s+n-1-s = n-1)
  If s < 0: j in [max(0, s, 0), min(n-1, s+n-1, n-1-s+s)] = [0, min(n-1, n-1+s)] = [0, n-1+s]

So:
- If s ≥ 0:
  grad[j] = (1-h[j+s]) for j in [0, n-1-s]  (first term)
         - h[j-s] for j in [s, n-1]          (second term)
  
- If s < 0 (let t = -s > 0):
  grad[j] = (1-h[j-t]) for j in [t, n-1]     (first term)
         - h[j+t] for j in [0, n-1-t]       (second term)

∂corr[4]/∂h[0] = (1-h[1])

∂corr[4]/∂h[2] = (1-h[3]) - h[1]
∂corr[4]/∂h[3] = -h[2]

grad[1] = (1-h[2]) - h[0]

```python
def soft_max_grad(corr, h, T, M=50):
    """Compute gradient of (1/T)*log(sum(exp(T*corr))) for top M lags."""
    n = len(h)
    # Find top M lags
    if M < len(corr):
        top_idx = np.argpartition(corr, -M)[-M:]
    else:
        top_idx = np.arange(len(corr))
    top_corr = corr[top_idx]
    # Softmax weights
    exp_vals = np.exp(T * top_corr - T * top_corr.max())
    weights = exp_vals / exp_vals.sum()
    # Gradient = weighted average of individual gradients
    grad = np.zeros(n)
    for mi, m in enumerate(top_idx):
        s = m - n + 1
        w = weights[mi]
        if s >= 0:
            # First term: (1-h[j+s]) for j in [0, n-1-s]
            if n-1-s >= 0:
                grad[0:n-s] += w * (1.0 - h[s:n])
            # Second term: -h[j-s] for j in [s, n-1]
            grad[s:n] -= w * h[0:n-s]
        else:
            t = -s
            # First term: (1-h[j-t]) for j in [t, n-1]
            if n-1-t >= 0:
                grad[t:n] += w * (1.0 - h[0:n-t])
            # Second term: -h[j+t] for j in [0, n-1-t]
            grad[0:n-t] -= w * h[t:n]
    return grad
```

            alpha = 1.0 / (T * np.max(np.abs(grad)) + 1)

                if c5_new < c5:
                    h = h_new
                    break
                alpha *= 0.5
            # If no improvement after backtracking, break
            if h is best_h:  # unchanged
                break
    return best_h, best_c5

In the standard MW approach for minimax:
- We have f_k(x) for k=1,...,K
- We want min_x max_k f_k(x)
- Algorithm:
  - p_1,...,p_K = 1/K each
  - for t = 1,2,...:
    - x_t = argmin_x ∑_k p_k f_k(x)  (or a step towards it)
    - p_k ← p_k · exp(η f_k(x_t))
    - normalize p
  - Return the best x_t found

In our case, K = 2n-1 ≈ 8000, which is a lot. But we can keep p as a sparse structure (only store the top M with significant weight).

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0-h, mode='full')) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            h[np.argmin(h)] += diff
        else:
            h[np.argmax(h)] += diff
    np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    # Periodic interpolation
    old_x = np.arange(old_n) * 2.0 / old_n
    new_x = np.arange(new_n) * 2.0 / new_n
    # Extend for periodicity
    old_x_ext = np.concatenate([old_x - 2.0, old_x, old_x + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_new = np.interp(new_x, old_x_ext, h_ext)
    return make_feasible(h_new, new_n)

def get_top_lags(corr, M):
    K = len(corr)
    if M >= K:
        return np.arange(K)
    idx = np.argpartition(corr, -M)[-M:]
    return idx

def compute_grad_topM(h, top_idx, weights):
    """Compute weighted gradient of correlations for top M lags."""
    n = len(h)
    grad = np.zeros(n)
    for i, m in enumerate(top_idx):
        s = m - n + 1
        w = weights[i]
        if w < 1e-15:
            continue
        if s >= 0:
            ns = n - s
            if ns > 0:
                grad[:ns] += w * (1.0 - h[s:])
                grad[s:] -= w * h[:ns]
        else:
            t = -s
            nt = n - t
            if nt > 0:
                grad[t:] += w * (1.0 - h[:nt])
                grad[:nt] -= w * h[t:]
    return grad

def projected_grad_step(h, grad, alpha, n):
    h_new = h - alpha * grad
    return make_feasible(h_new, n)

def optimize_multiscale(h, n, budget_frac, rng, T_schedule, steps_per_T, topM):
    """Main optimization: projected gradient with adaptive T."""
    start = time.time()
    budget = budget_frac * 1300  # approximate
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for T in T_schedule:
        if time.time() - start > budget * 0.95:
            break
        h_curr = h_curr.copy() if 'h_curr' in dir() else h.copy()
        for step in range(steps_per_T):
            if time.time() - start > budget:
                break
            corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
            # Find top M
            top_idx = get_top_lags(corr, topM)
            top_corr = corr[top_idx]
            # Softmax weights (temperature T applied to normalized values)
            shifted = top_corr - top_corr.max()
            exp_vals = np.exp(T * shifted)
            w_sum = exp_vals.sum()
            if w_sum < 1e-300:
                weights = np.ones(len(top_idx)) / len(top_idx)
            else:
                weights = exp_vals / w_sum
            # Compute gradient
            grad = compute_grad_topM(h_curr, top_idx, weights)
            grad_norm = np.linalg.norm(grad)
            if grad_norm < 1e-15:
                break
            # Adaptive step size
            alpha = 1.0 / (grad_norm + 1e-10)
            # Backtracking
            c5_curr = np.max(corr) * dx
            improved = False
            for bt in range(15):
                h_new = make_feasible(h_curr - alpha * grad, n)
                corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
                c5_new = np.max(corr_new) * dx
                if c5_new < c5_curr - 1e-15:
                    h_curr = h_new
                    c5_curr = c5_new
                    improved = True
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    break
                alpha *= 0.5
            if not improved:
                # Try a smaller step or break
                break
        # Shake between temperature regimes
        h_curr[h_curr > 0.01] += rng.normal(0, 0.02, n) * (h_curr > 0.01)
        h_curr[h_curr < 0.99] += rng.normal(0, 0.02, n) * (h_curr < 0.99)
        h_curr = make_feasible(h_curr, n)
    
    return best_h, best_c5
