    best_max = curr_max
    best_b = b.copy()
    ones_l = list(np.where(b == 1)[0])
    zeros_l = list(np.where(b == 0)[0])
    n1 = len(ones_l)
    nz = len(zeros_l)
    for it in range(max_iters):
        if it % 2000 == 0:
            T = T0 * (T1 / T0) ** (float(it) / max_iters)
        i = ones_l[rng.randint(n1)]
        j = zeros_l[rng.randint(nz)]
        b[i] = 0.0
        b[j] = 1.0
        new_corr = np.correlate(b, 1.0 - b, mode='full')
        new_max = float(np.max(new_corr))
        if new_max <= curr_max or (T > 1e-10 and rng.random() < np.exp(-(new_max - curr_max) / T)):
            curr_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
            ones_l[i if i in ones_l else ones_l.index(i)] = j
            zeros_l[j if j in zeros_l else zeros_l.index(j)] = i
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_max * (2.0 / n)

def sa_fast(b, n, max_iters, T0, T1, rng):
    """Faster SA using set-based index tracking."""
    corr = np.correlate(b, 1.0 - b, mode='full')
    curr_max = float(np.max(corr))
    best_max = curr_max
    best_b = b.copy()
    ones_set = set(np.where(b == 1)[0].tolist())
    zeros_set = set(np.where(b == 0)[0].tolist())
    ones_list = list(ones_set)
    zeros_list = list(zeros_set)
    for it in range(max_iters):
        if it % 500 == 0:
            T = T0 * (T1 / T0) ** (float(it) / max_iters)
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        b[i] = 0.0
        b[j] = 1.0
        new_corr = np.correlate(b, 1.0 - b, mode='full')
        new_max = float(np.max(new_corr))
        if new_max <= curr_max or (T > 1e-10 and rng.random() < np.exp(-(new_max - curr_max) / T)):
            curr_max = new_max
            corr = new_corr
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
            ones_list[ones_list.index(i)] = j
            zeros_list[zeros_list.index(j)] = i
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_max * (2.0 / n)

def slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc
    # Use log-sum-exp with shift for numerical stability
    m2 = float(np.max(shifted))
    lse = m2 + np.log(np.sum(np.exp(shifted - m2))) / T
    return (mc + lse) * dx

def slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_jac}]
    res = minimize(slsqp_obj, h.copy(), args=(n, T), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0]
            if old > 0:
                cands.append(max(0.0, old - 0.02))
            if old < 1:
                cands.append(min(1.0, old + 0.02))
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = random_binary(960, rng)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SA at initial resolution
    b1, c5_1 = sa_fast(to_binary(h_init, n0), n0, 150000, 0.02, 1e-5, rng)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()

    # Phase 2: Upsample to n=1440 and SA
    remaining = budget_s - (time.time() - start_time)
    n1440 = 1440
    if remaining > 200:
        h_up = upsample_h(best_h, n1440)
        b_up = to_binary(h_up, n1440)
        b2, c5_2 = sa_fast(b_up, n1440, 250000, 0.015, 5e-6, rng)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()

    # Phase 3: Try random start at n=1440
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        b_rand = random_binary(n1440, rng)
        b3, c5_3 = sa_fast(b_rand, n1440, 150000, 0.03, 1e-5, rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()

    # Phase 4: Upsample to n=1920 and SA
    remaining = budget_s - (time.time() - start_time)
    n1920 = 1920
    if remaining > 250:
        h_up2 = upsample_h(best_h, n1920)
        b_up2 = to_binary(h_up2, n1920)
        b4, c5_4 = sa_fast(b_up2, n1920, 200000, 0.012, 4e-6, rng)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = b4.copy()

    # Phase 5: Random start at n=1920
    remaining = budget_s - (time.time() - start_time)
    if remaining > 180:
        b_rand2 = random_binary(n1920, rng)
        b5, c5_5 = sa_fast(b_rand2, n1920, 120000, 0.025, 8e-6, rng)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = b5.copy()

    # Phase 6: SLSQP continuous optimization on best candidate
    remaining = budget_s - (time.time() - start_time)
    n_cur = len(best_h)
    if remaining > 200:
        try:
            x_res = slsqp_run(best_h, n_cur, 50.0, 500, 1e-13)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_cur)
        except Exception:
            pass

    # Phase 7: SLSQP with higher T (smoother objective)
    remaining = budget_s - (time.time() - start_time)
    n_cur = len(best_h)
    if remaining > 150:
        try:
            x_res = slsqp_run(best_h, n_cur, 200.0, 400, 1e-13)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_cur)
        except Exception:
            pass

    # Phase 8: SA again from continuous solution (re-binarize and refine)
    remaining = budget_s - (time.time() - start_time)
    n_cur = len(best_h)
    if remaining > 200:
        b_re = to_binary(best_h, n_cur)
        b6, c5_6 = sa_fast(b_re, n_cur, 100000, 0.01, 3e-6, rng)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = b6.copy()

    # Phase 9: Coordinate descent polish
    n_cur = len(best_h)
    best_h, best_c5 = coord_descent(best_h.copy(), n_cur, 20, start_time, budget_s)

    # Phase 10: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    n_cur = len(best_h)
    if remaining > 100:
        try:
            x_res = slsqp_run(best_h, n_cur, 500.0, 300, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_cur)
        except Exception:
            pass

    # Phase 11: Try different n for potential improvement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n_try = 1440 if len(best_h) != 1440 else 1920
        h_try = upsample_h(best_h, n_try)
        b_try = to_binary(h_try, n_try)
        b7, c5_7 = sa_fast(b_try, n_try, 150000, 0.015, 5e-6, rng)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = b7.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100:
            try:
                x_res = slsqp_run(best_h, len(best_h), 100.0, 300, 1e-13)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, len(best_h))
            except Exception:
                pass

    # Final: ensure feasibility and compute
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    # Verify time
    if time.time() - start_time > budget_s:
        pass  # We're over, but return what we have

    return best_h, best_c5, len(best_h)
