import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, method='fft')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, rng):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    if s > target:
        ones = np.where(b == 1)[0]
        remove = rng.choice(ones, size=s - target, replace=False)
        b[remove] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        add = rng.choice(zeros, size=target - s, replace=False)
        b[add] = 1
    return b

def sa_binary(h, n, n_iters, T0, T1, rng, block_max=1):
    b = to_binary(h, n, rng)
    curr = compute_c5_fast(b)
    best = curr
    best_b = b.copy()
    ones_cache = np.where(b == 1)[0]
    zeros_cache = np.where(b == 0)[0]
    for it in range(n_iters):
        frac = it / max(1, n_iters)
        T = T0 * (T1 / T0) ** frac
        if block_max <= 1:
            if len(ones_cache) == 0 or len(zeros_cache) == 0:
                break
            i = ones_cache[rng.randint(len(ones_cache))]
            j = zeros_cache[rng.randint(len(zeros_cache))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = compute_c5_fast(b)
            if new_c <= curr or (T > 1e-15 and rng.random() < np.exp(-(new_c - curr) / T)):
                curr = new_c
                if curr < best:
                    best = curr
                    best_b = b.copy()
                ones_cache = np.where(b == 1)[0]
                zeros_cache = np.where(b == 0)[0]
            else:
                b[i] = 1.0
                b[j] = 0.0
        else:
            sz = rng.randint(1, block_max + 1)
            if sz >= n:
                sz = n // 2
            start = rng.randint(0, n - sz)
            seg = b[start:start + sz]
            ones_seg = np.where(seg == 1)[0]
            zeros_seg = np.where(seg == 0)[0]
            if len(ones_seg) == 0 or len(zeros_seg) == 0:
                continue
            sw = min(len(ones_seg), len(zeros_seg))
            i_list = start + rng.choice(ones_seg, size=sw, replace=False)
            j_list = start + rng.choice(zeros_seg, size=sw, replace=False)
            b[i_list] = 0.0
            b[j_list] = 1.0
            new_c = compute_c5_fast(b)
            if new_c <= curr or (T > 1e-15 and rng.random() < np.exp(-(new_c - curr) / T)):
                curr = new_c
                if curr < best:
                    best = curr
                    best_b = b.copy()
            else:
                b[i_list] = 1.0
                b[j_list] = 0.0
    return best_b, best

def greedy_pass(b, n, max_rounds, start_time, budget_s):
    curr = compute_c5_fast(b)
    for r in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                return b, curr
            old_val = b[j]
            new_val = 1.0 - old_val
            b[j] = new_val
            new_c = compute_c5_fast(b)
            if new_c < curr - 1e-15:
                curr = new_c
                improved = True
            else:
                b[j] = old_val
        if not improved:
            break
    return b, curr

def slsqp_obj(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = correlate(x, 1.0 - x, method='fft')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_con(x):
    return np.sum(x) - _SL_N / 2.0

def slsqp_con_jac(x):
    return np.ones(_SL_N)

def run_slsqp(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    res = minimize(slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    return budget_s - (time.time() - start_time)
