def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break

        h = np.clip(h, 0.0, 1.0)

def slsqp_opt(h, n, T, maxiter=200, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = np.max(corr)
        sh = corr - mc
        return mc + np.log(np.sum(np.exp(T * sh))) / T

        return np.sum(x) - n / 2.0

    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h, n, T, maxiter=200, ftol=1e-14):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        mc = np.max(corr)
        sh = corr - mc
        return mc + np.log(np.sum(np.exp(T * sh))) / T
    def con(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def targeted_refine(h, n, rng, start_time, time_limit):
    h = h.copy()
    curr = compute_c5(h)
    dx = 2.0 / n
    while time.time() - start_time < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        improved = False
        for _ in range(80):
            if time.time() - start_time > time_limit:
                break
            if lag >= 0:
                valid = np.arange(n - lag)
                if len(valid) == 0: break
                weights = h[valid + lag] * (1.0 - h[valid])
                tw = np.sum(weights)
                if tw < 1e-8: break
                probs = weights / tw
                idx = int(rng.choice(valid, p=probs))
                eps = rng.uniform(0.01, 0.05)
                h[idx + lag] -= eps
                h[idx] += eps
                h = np.clip(h, 0, 1)
                h = make_feasible(h, n)
                nc = compute_c5(h)
                if nc < curr:
                    curr = nc
                    improved = True
                else:
                    h[idx + lag] += eps
                    h[idx] -= eps
                    h = np.clip(h, 0, 1)
                    h = make_feasible(h, n)
            else:
                m = -lag
                valid = np.arange(m, n)
                if len(valid) == 0: break
                weights = h[valid - m] * (1.0 - h[valid])
                tw = np.sum(weights)
                if tw < 1e-8: break
                probs = weights / tw
                idx = int(rng.choice(valid, p=probs))
                eps = rng.uniform(0.01, 0.05)
                h[idx - m] -= eps
                h[idx] += eps
                h = np.clip(h, 0, 1)
                h = make_feasible(h, n)
                nc = compute_c5(h)
                if nc < curr:
                    curr = nc
                    improved = True
                else:
                    h[idx - m] += eps
                    h[idx] -= eps
                    h = np.clip(h, 0, 1)
                    h = make_feasible(h, n)
        if not improved:
            break
    return h, curr

def coord_descent_fine(h, n, start_time, time_limit):
    h = h.copy()
    curr = compute_c5(h)
    for _ in range(50):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                break
            old = h[j]
            best_c = curr
            best_val = old
            for cand in [0.0, 1.0, max(0, old - 0.005), max(0, old - 0.001),
                         min(1, old + 0.001), min(1, old + 0.005)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                curr = best_c
                improved = True
        if not improved:
            break
    return h, curr
