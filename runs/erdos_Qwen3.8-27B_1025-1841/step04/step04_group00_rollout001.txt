import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-16:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgrad_at_shift(h, s, n):
    grad = np.zeros(n)
    for j in range(n):
        g = 0.0
        jp = j + s
        jm = j - s
        if 0 <= jp < n:
            g += 1.0 - h[jp]
        if 0 <= jm < n:
            g -= h[jm]
        grad[j] = g
    return grad

def pgd_top1(h, n, budget_s, start_time, step_init=0.015):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = step_init
    mom = 0.8
    vel = np.zeros(n)
    stall = 0
    dx = 2.0 / n

    while time.time() - start_time < budget_s - 5:
        corr = corr_vec(h)
        c5_cur = np.max(corr) * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 200:
                break

        top_k = np.argmax(corr)
        s = top_k - (n - 1)
        grad = subgrad_at_shift(h, s, n)

        y = h - mom * vel
        y = np.clip(y, 0, 1)
        corr_y = corr_vec(y)
        top_k_y = np.argmax(corr_y)
        s_y = top_k_y - (n - 1)
        grad_y = subgrad_at_shift(y, s_y, n)
        vel = mom * vel - step * grad_y
        new_h = make_feasible(h + vel, n)
        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur + 1e-15:
            h = new_h
        else:
            new_h2 = make_feasible(h - step * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_cur + 1e-15:
                h = new_h2
                vel *= 0.5
            else:
                step *= 0.65
                vel *= 0.3
            if step < 1e-9:
                break
        if step < 0.03 and stall % 80 == 0:
            step = min(0.03, step * 1.4)

    return best_h, best_c5

def coord_descent_exact(h, n, budget_s, start_time, steps_list=None):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    if steps_list is None:
        steps_list = [0.01, 0.005, 0.002, 0.001, 0.0005]

    improved = True
    passes = 0
    while improved and time.time() - start_time < budget_s - 10:
        improved = False
        passes += 1
        corr = corr_vec(best_h)
        max_k = np.argmax(corr)
        max_s = max_k - (n - 1)

        impacts = np.zeros(n)
        for j in range(n):
            imp = 0.0
            if 0 <= j + max_s < n:
                imp += (1.0 - best_h[j + max_s])
            if 0 <= j - max_s < n:
                imp += best_h[j - max_s]
            impacts[j] = abs(imp)

        order = np.argsort(-impacts)
        for j in order:
            if time.time() - start_time > budget_s - 15:
                break
            old = best_h[j]
            best_val = old
            best_c = best_c5
            for delta in steps_list:
                for sign in [-1.0, 1.0]:
                    v = old + sign * delta
                    if 0.0 <= v <= 1.0 and abs(v - old) > 1e-12:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            for v in [0.0, 1.0]:
                if abs(v - old) > 0.005:
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_val = v
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def binary_swap_search(h_start, n, budget_s, start_time, seed=42):
    h_bin = (h_start >= 0.5).astype(float)
    diff = n / 2.0 - np.sum(h_bin)
    if diff != 0:
        if diff > 0:
            order = np.where(h_bin == 0)[0]
            for i in range(int(abs(diff))):
                h_bin[order[i]] = 1.0
        else:
            order = np.where(h_bin == 1)[0]
            for i in range(int(abs(diff))):
                h_bin[order[i]] = 0.0
    h_bin = make_feasible(h_bin, n)

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    stall = 0

    while time.time() - start_time < budget_s - 15 and stall < 50000:
        ones = np.where(best_h == 1.0)[0]
        zeros = np.where(best_h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        best_h[i] = 1.0
        best_h[j] = 0.0
        c = compute_c5(best_h)
        if c < best_c5 - 1e-14:
            best_c5 = c
            stall = 0
        else:
            best_h[i] = 0.0
            best_h[j] = 1.0
            stall += 1
    return best_h, best_c5

def slsqp_exact(h, n, budget_s, start_time):
    dx = 2.0 / n
    def obj(x):
        c = corr_vec(x)
        return np.max(c) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 500, 'ftol': 1e-18})
        if res.x is not None:
            res.x = make_feasible(res.x, n)
            c_res = compute_c5(res.x)
            if c_res < compute_c5(h):
                return res.x, c_res
    except Exception:
        pass
    return h, compute_c5(h)

def perturbed_restart(h, n, budget_s, start_time, rng, scale=0.005):

        c3 = compute_c5(h2)
        return h2, c3
    return h, compute_c5(h)
