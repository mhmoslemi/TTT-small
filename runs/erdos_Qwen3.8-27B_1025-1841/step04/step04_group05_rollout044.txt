def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def grad_topk(h, top_k_indices, n):
    """Compute gradient of sum of top-K correlations w.r.t. h."""
    grad = np.zeros(n)
    ih = 1.0 - h
    for k_idx in top_k_indices:
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[:n - lag] += ih[lag:]
            grad[lag:] += -h[:n - lag]
        else:
            m = -lag
            grad[m:] += ih[:n - m]
            grad[:n - m] += -h[m:]
    return grad

def gradient_descent(h, n, max_iter, K_init, K_max, lr_init, decay):
    """Adaptive-K gradient descent on the max correlation."""
    h = h.copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for it in range(max_iter):
        # Adaptive K: start small, grow as we converge
        K = min(K_max, K_init + it // 50)
        
        # Find top K lags
        if K < len(corr):
            top_k = np.argpartition(corr, -K)[-K:]
        else:
            top_k = np.arange(len(corr))
        
        grad = grad_topk(h, top_k, n)
        
        # Adaptive learning rate
        lr = lr_init / (1.0 + decay * it)
        
        # Gradient step
        h_new = h - lr * grad / K
        h_new = make_feasible(h_new, n)
        
        # Check if improvement
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        if np.max(corr_new) < np.max(corr) + 1e-12:
            h = h_new
            corr = corr_new
        else:
            # Halve learning rate
            lr *= 0.5
            h_new = h - lr * grad / K
            h_new = make_feasible(h_new, n)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            if np.max(corr_new) < np.max(corr) + 1e-12:
                h = h_new
                corr = corr_new
    
    return h

        delta_corr[n-1-ms_pos] += delta * (1.0 - h[j + ms_pos])
    
    if j >= 1:
        # For m = 1..j: h[j-m] is valid
        max_m_neg = j
        ms_neg = np.arange(1, max_m_neg + 1)
        delta_corr[n-1-ms_neg] += delta * (-h[j - ms_neg])
    
    return delta_corr

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def grad_topk(h, top_k_indices, n):
    grad = np.zeros(n)
    ih = 1.0 - h
    for k_idx in top_k_indices:
        lag = k_idx - (n - 1)
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            if n - lag > 0:
                grad[:n - lag] += ih[lag:]
            if lag > 0:
                grad[lag:] += -h[:n - lag]
        else:
            m = -lag
            if n - m > 0:
                grad[m:] += ih[:n - m]
                grad[:n - m] += -h[m:]
    return grad

def gradient_descent(h, n, max_iter, K_init, K_max, lr_init, decay, time_fn, time_limit):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(corr)
    stuck_count = 0
    
    for it in range(max_iter):
        if time_fn() < time_limit:
            break
        K = min(K_max, K_init + it // 80)
        K = min(K, len(corr))
        top_k = np.argpartition(corr, -K)[-K:]
        grad = grad_topk(h, top_k, n)
        lr = lr_init / (1.0 + decay * it)
        
        for attempt in range(3):
            h_new = h - lr * grad / K
            h_new = make_feasible(h_new, n)
            corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
            new_max = np.max(corr_new)
            if new_max < best_max - 1e-14:
                h = h_new
                corr = corr_new
                best_max = new_max
                stuck_count = 0
                break
            else:
                lr *= 0.4
        else:
            stuck_count += 1
            if stuck_count > 200:
                break
    return h

def coord_delta(h, j, delta, n):
    dc = np.zeros(2 * n - 1)
    if abs(delta) < 1e-15:
        return dc
    ih = 1.0 - h
    # lag > 0: indices n to 2n-2
    max_lp = min(j, n - 1)
    if max_lp >= 1:
        lp = np.arange(1, max_lp + 1)
        dc[n - 1 + lp] += delta * ih[j - lp]
    max_ln = n - 1 - j
    if max_ln >= 1:
        ln = np.arange(1, max_ln + 1)
        dc[n - 1 + ln] += -delta * h[j + ln]
    # lag = 0
    dc[n - 1] += delta * (1.0 - 2.0 * h[j])
    # lag < 0: indices 0 to n-2
    max_mp = n - 1 - j
    if max_mp >= 1:
        mp = np.arange(1, max_mp + 1)
        dc[n - 1 - mp] += delta * ih[j + mp]
    max_mn = j
    if max_mn >= 1:
        mn = np.arange(1, max_mn + 1)
        dc[n - 1 - mn] += -delta * h[j - mn]
    return dc

def coordinate_descent(h, n, max_rounds, time_fn, time_limit):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(corr)
    target = n / 2.0
    step_sizes = [0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01]
    
    for rnd in range(max_rounds):
        if time_fn() < time_limit:
            break
        improved = False
        for j in range(n):
            if time_fn() < time_limit:
                break
            old = h[j]
            best_c = best_max
            best_v = old
            for sv in step_sizes:
                v = old + sv
                if v < 0.0 or v > 1.0:
                    continue
                delta = v - old
                d_corr = coord_delta(h, j, delta, n)
                new_corr_vals = corr + d_corr
                new_max = np.max(new_corr_vals)
                if new_max < best_c - 1e-12:
                    # Check if we can fix the sum
                    sum_diff = target - (np.sum(h) + delta)
                    if abs(sum_diff) < 1e-10:
                        best_c = new_max
                        best_v = v
                        corr = new_corr_vals
                    else:
                        # Find compensating index
                        if sum_diff > 0:
                            comp = int(np.argmin(h))
                            if comp != j and h[comp] + sum_diff <= 1.0:
                                best_c = new_max
                                best_v = v
                        elif sum_diff < 0:
                            comp = int(np.argmax(h))
                            if comp != j and h[comp] + sum_diff >= 0.0:
                                best_c = new_max
                                best_v = v
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                cm = np.max(corr)
                if cm < best_max:
                    best_max = cm
                improved = True
        if not improved:
            break
    return h

def binary_swap_search(h, n, rng, max_swaps, time_fn, time_limit):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(corr)
    
    for _ in range(max_swaps):
        if time_fn() < time_limit:
            break
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        
        h[i], h[j] = h[j], h[i]
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(corr_new)
        if new_max < best_max - 1e-13:
            best_max = new_max
            corr = corr_new
        else:
            h[i], h[j] = h[j], h[i]
    return h, best_max
