    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    top5 = np.sort(corr)[::-1][:5]
    return mc * dx + 0.05 * (top5 ** 6).sum() * dx * 1e-6

def do_slsqp(h, n, T, maxiter, max_fun=10000):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    def obj_local(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    try:
        res = minimize(obj_local, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14, 'maxfun': max_fun})
        return make_feasible(res.x, n)
    except Exception:
        return h

def do_slsqp_targeted(h, n, maxiter=150):
    global _G_N
    _G_N = n
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_lags = np.argsort(corr)[::-1][:8]
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    def obj_local(x):
        dx2 = 2.0 / n
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(c)
        tv = c[top_lags]
        tm = np.max(tv)
        return mc * dx2 + 0.1 * (tv - tm + 0.01) ** 4 * dx2
    try:
        res = minimize(obj_local, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'maxfun': 5000})
        return make_feasible(res.x, n)
    except Exception:
        return h

def binary_snap_fast(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        for val in (0.0, 1.0):
            if val == best[j]:
                continue
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def fine_coord_descent(h, n, n_rounds=3):
    current_c5 = compute_c5(h)
    for _ in range(n_rounds):
        improved = False
        for j in range(n):
            cur = h[j]
            best_c = current_c5
            best_val = cur
            cands = [0.0, 1.0, max(0.0, cur - 0.03), max(0.0, cur - 0.005),
                     min(1.0, cur + 0.03), min(1.0, cur + 0.005)]
            for cand in cands:
                if abs(cand - cur) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != cur:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                current_c5 = best_c
                improved = True
        if not improved:
            break
    return h, current_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Quick SLSQP at native resolution with varied temperatures
    for T in [60.0, 150.0, 300.0]:
        if remaining() < 50:
            break
        h_new = do_slsqp(best_h, n_init, T, maxiter=200)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Upsample to n=1920 and optimize
    if remaining() > 100:
        n_up = 1920
        h_up = upsample_h(best_h, n_up)
        for T in [80.0, 250.0]:
            if remaining() < 80:
                break
            h_new = do_slsqp(h_up, n_up, T, maxiter=150, max_fun=6000)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
                h_up = h_new.copy()
        # Targeted optimization at high res
        if remaining() > 60:
            h_new = do_slsqp_targeted(h_up, n_up, maxiter=100)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 3: Also try n=768 for a different resolution perspective
    if remaining() > 120:
        n_alt = 768
        h_alt = upsample_h(best_h, n_alt)
        for T in [100.0, 300.0]:
            if remaining() < 80:
                break
            h_new = do_slsqp(h_alt, n_alt, T, maxiter=200)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: Try perturbed restarts at best resolution
    n_best = len(best_h)
    for restart in range(3):
        if remaining() < 60:
            break
        h_pert = best_h + rng.randn(n_best) * 0.02
        h_pert = make_feasible(h_pert, n_best)
        T_r = 200.0 + restart * 100.0
        h_new = do_slsqp(h_pert, n_best, T_r, maxiter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 5: Binary snap (exploit near-binary structure)
    if remaining() > 40:
        best_h, best_c5 = binary_snap_fast(best_h, n_best)

    # Phase 6: SLSQP polish after binary snap
    if remaining() > 40:
        h_new = do_slsqp(best_h, n_best, 500.0, maxiter=150)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 7: Fine coordinate descent
    if remaining() > 30:
        best_h, best_c5 = fine_coord_descent(best_h, n_best, n_rounds=2)

    # Phase 8: Final targeted SLSQP
    if remaining() > 35:
        h_new = do_slsqp_targeted(best_h, n_best, maxiter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 9: Another binary snap round
    if remaining() > 40:
        best_h, best_c5 = binary_snap_fast(best_h, n_best)

    # Phase 10: Final polish
    if remaining() > 25:
        h_new = do_slsqp(best_h, n_best, 800.0, maxiter=100)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        best_h, best_c5 = fine_coord_descent(best_h, n_best, n_rounds=1)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
