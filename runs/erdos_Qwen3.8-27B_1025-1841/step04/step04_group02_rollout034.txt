```python
    # grad[j] = sum_k w_k * (A_k[j] - B_k[j])

    # and B_k[j] = h[j-s_k] if j-s_k >= 0 else 0

    # sum_k w_k * A_k[j] = sum_k w_k * (1-h[j+s_k]) * 1[j+s_k < n]
    # = sum_k w_k * 1[j+s_k < n] - sum_k w_k * h[j+s_k] * 1[j+s_k < n]

    # and W_minus[j] = sum_{k: j-s_k >= 0} w_k

    # and most weights will be small

    # and w_k = w_{k} (the softmax weight at correlation index k)

    # w(s) = weights[k] where k = s + (n-1)

    # Then the second term is a cross-correlation of h and W
    # And the first term is a "cumulative sum" type operation

    active = weights > thresh
    if np.sum(active) == 0:
        return obj_val, grad
    
    active_weights = weights[active]
    active_shifts = shifts[active]

        s = active_shifts[idx]

            # For s > 0:
            # d(corr_k)/dh[j] where s>0:
            #   h[j] appears as first factor when i=j, need j+s < n: contribution (1-h[j+s])
            #   h[j] appears as second factor (1-h[i+s]) when i+s=j, i=j-s, need j-s>=0: contribution -h[j-s]

            # For j in [0, s): just (1-h[j+s])  [since j-s < 0]
            # For j in [n-s, n): just -h[j-s]  [since j+s >= n]
            # Wait, that's not right either. Let me be very careful.
            #
            # corr_k = sum_i h[i] * (1-h[i+s]) where s = k-(n-1) > 0
            # Valid i: 0 <= i < n AND 0 <= i+s < n, so 0 <= i < n-s

            # d(corr_k)/dh[j]:

            #   Contribution: (1-h[j+s])
            # Case 2: h[j] is inside (1-h[i+s]), i.e., i+s=j, so i=j-s. Valid if 0 <= j-s < n-s, i.e., s <= j < n.

                grad[:] += w * (1.0 - 2.0 * h)

                if s < n:
                    grad[:s] += w * (1.0 - h[s:s + 0])  # This should be (1-h[j+s]) for j=0..s-1
                # Let me just do it properly
                # j in [0, min(s, n-s)): (1-h[j+s])
                # j in [max(0,s), min(n-s, n)): (1-h[j+s]) - h[j-s]  
                # j in [max(n-s, s), n): -h[j-s]
                
                # Simpler: just loop
                for j in range(n):

                    if j < n - s:  # first factor valid
                        val += (1.0 - h[j + s])
                    if j >= s:  # second factor valid (j-s >= 0) and (j-s) < n-s i.e. j < n
                        val -= h[j - s]

        elif s < 0:

            # Valid i: 0 <= i < n AND 0 <= i-ss < n, so ss <= i < n
            #
            # d/dh[j]:
            # Case 1: i=j (first factor), valid if ss <= j < n: contribution (1-h[j-ss])

            for j in range(n):

                if j >= ss and j < n:  # first factor
                    val += (1.0 - h[j - ss])
                if j + ss < n:  # second factor
                    val -= h[j + ss]
                grad[j] += w * val
    
    grad *= dx
    
    return obj_val, grad
```

Hmm, this O(n²) approach with the inner loop is going to be too slow for n=960 (that's ~921K operations per gradient evaluation, and we might need hundreds of evaluations).

Let me use FFT instead:

For the term `sum_s w(s) * (1-h[j+s]) * 1(0<=j+s<n)`:
This is a convolution of the sequence (1-h) with the weights w, evaluated at specific points.

Let me define:

- For j: sum_s w_ext[s] * (1-h[j+s]) * 1(0<=j+s<n)

This is (part of) the cross-correlation of (1-h) with w_ext.

Similarly for the h[j-s] term.

Let me use FFT to compute these:

```python
def c5_obj_grad_fft(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_corr = len(corr)  # 2n-1
    mc = np.max(corr)
    shifted = corr - mc
    
    # Avoid overflow
    max_shifted = np.max(shifted)
    exp_shifted = np.exp(T * (shifted - max_shifted))
    sum_exp = np.sum(exp_shifted)
    log_sum_exp = max_shifted + np.log(sum_exp) / T
    obj_val = (mc + log_sum_exp) * dx
    
    # Softmax weights
    w = exp_shifted / sum_exp  # These sum to 1
    # Actually the gradient of the smooth max is:
    # d/dh = dx * sum_k w_k * d(corr_k)/dh
    # where w_k = exp(T*(corr_k - mc)) / sum_m exp(T*(corr_m - mc))
    
    shifts = np.arange(n_corr) - (n - 1)  # -(n-1), ..., 0, ..., (n-1)
    
    # We need: grad[j] = dx * sum_s w(s) * [(1-h[j+s])*1(0<=j+s<n) - h[j-s]*1(0<=j-s<n)]
    # where w(s) = w[s + (n-1)]
    
    # First term: sum_s w(s) * (1-h[j+s]) * 1(0<=j+s<n)
    # Let's define u[j] = 1-h[j] for j in [0,n), 0 otherwise
    # And v[s] = w(s) for s in [-(n-1), n-1], 0 otherwise
    # Then the first term at position j is: sum_s v[s] * u[j+s] (with u=0 outside [0,n))
    # This is the cross-correlation of u and v, evaluated at j (with sign convention)
    
    # Using FFT:
    N_fft = 2 * n  # or next power of 2 >= 2n-1
    N_fft = int(2**np.ceil(np.log2(2*n-1)))
    
    # Pad u to length N_fft
    u = np.zeros(N_fft)
    u[:n] = 1.0 - h
    
    # Pad v to length N_fft (v is defined on shifts, centered at 0)
    # v[s] for s in [-(n-1), n-1]
    # In the FFT array, we place v[s] at position: if s >= 0, position s; if s < 0, position N_fft + s
    v = np.zeros(N_fft)
    for s_idx in range(n_corr):
        s = shifts[s_idx]
        w_val = w[s_idx]
        if w_val < 1e-15:
            continue
        pos = s if s >= 0 else N_fft + s
        v[pos] = w_val
    
    # Cross-correlation: sum_s v[s] * u[j+s]
    # = ifft(fft(u) * conj(fft(v)))[j]  ... let me verify
    # Actually: cross_corr[j] = sum_s u[j+s] * v[s]
    # In terms of FFT: this is ifft(FFT(u) * conj(FFT(v))) 
    # Wait, let me be more careful.
    # 
    # Standard cross-correlation: (u ⋆ v)[j] = sum_s u[j+s] * conj(v[s])
    # In frequency domain: FFT(u ⋆ v) = conj(FFT(u)) * FFT(v)
    # So u ⋆ v = ifft(conj(FFT(u)) * FFT(v))
    
    # But we want sum_s v[s] * u[j+s] which is the same as u ⋆ conj(v) (if v is real)
    # = ifft(conj(FFT(u)) * FFT(v))  [since v is real, conj(v) = v]
    
    first_term = np.fft.irfft(np.conj(np.fft.rfft(u)) * np.fft.rfft(v), N_fft)
    
    # Second term: sum_s w(s) * h[j-s] * 1(0<=j-s<n)
    # = sum_s v[s] * h[j-s] (with h=0 outside [0,n))
    # This is the convolution of h with v (flipped): sum_s h[s] * v[j-s] ... 
    # Actually: sum_s v[s] * h[j-s] = (v * h)[j] where * is convolution
    # Convolution: (v * h)[j] = sum_s v[s] * h[j-s]
    # In FFT: FFT(v * h) = FFT(v) * FFT(h)
    # So: second_term = ifft(FFT(v) * FFT(h))
    
    h_padded = np.zeros(N_fft)
    h_padded[:n] = h
    second_term = np.fft.irfft(np.fft.rfft(v) * np.fft.rfft(h_padded), N_fft)
    
    grad = (first_term[:n] - second_term[:n]) * dx
    
    return obj_val, grad
```

- Do local search (SA) until convergence

- Do local search again

Let me write the final algorithm combining these ideas:

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_val(h):
    """Just the max correlation value (before multiplying by dx)"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(min(diff, len(order))):
            b[zeros[order[i]]] = 1
    elif diff < 0:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for i in range(min(-diff, len(order))):
            b[ones[order[i]]] = 0
    return b

def sa_mixed(h, n, max_iters, T0, T1, rng, time_start, budget_s, check_interval=100):
    """SA with mixed move types: single swap, long-range swap, block swap"""
    b = to_binary(h, n)
    curr_c = compute_c5_val(b)
    best_c = curr_c
    best_b = b.copy()
    
    for it in range(max_iters):
        if it % check_interval == 0:
            if (time.time() - time_start) > budget_s - 30:
                break
        
        frac = min(1.0, float(it) / max_iters)
        # Adaptive cooling: faster initially, slower at the end
        T = T0 * (T1 / T0) ** (frac ** 0.5)
        
        # Choose move type
        move_type = rng.random()
        
        if move_type < 0.5:
            # Single swap (local)
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            # Try to pick j close to i (local move)
            if rng.random() < 0.7:
                dist = rng.randint(1, n // 6)
                j_cand = (i + dist) % n
                if b[j_cand] == 0:
                    j = j_cand
                j_cand = (i - dist) % n
                if b[j_cand] == 0:
                    j = j_cand
        elif move_type < 0.8:
            # Long-range swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        else:
            # Block swap: pick a segment and swap some bits within
            seg_len = rng.randint(3, min(12, n // 10))
            start = rng.randint(0, n - seg_len)
            ones_in = np.where(b[start:start+seg_len] == 1)[0]
            zeros_in = np.where(b[start:start+seg_len] == 0)[0]
            if len(ones_in) == 0 or len(zeros_in) == 0:
                continue
            k = min(len(ones_in), len(zeros_in), rng.randint(1, 3))
            chosen_o = rng.choice(ones_in, size=k, replace=False)
            chosen_z = rng.choice(zeros_in, size=k, replace=False)
            b[start + chosen_o] = 0
            b[start + chosen_z] = 1
            new_c = compute_c5_val(b)
            if new_c <= curr_c or (T > 1e-12 and rng.random() < np.exp(-(new_c - curr_c) / T)):
                curr_c = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_b = b.copy()
            else:
                b[start + chosen_o] = 1
                b[start + chosen_z] = 0
            continue
        
        b[i] = 0
        b[j] = 1
        new_c = compute_c5_val(b)
        if new_c <= curr_c or (T > 1e-12 and rng.random() < np.exp(-(new_c - curr_c) / T)):
            curr_c = new_c
            if new_c < best_c:
                best_c = new_c
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    
    return best_b, best_c * (2.0 / n)

def ills(h, n, n_restarts, sa_iters, T0, T1, perturbation_size, rng, time_start, budget_s):
    """Iterated Local Search"""
    best_b = to_binary(h, n)
    best_c = compute_c5(best_b)
    
    for r in range(n_restarts):
        if (time.time() - time_start) > budget_s - 60:
            break
        
        # Local search
        b, c = sa_mixed(best_b, n, sa_iters, T0, T1, rng, time_start, budget_s)
        if c < best_c:
            best_c = c
            best_b = b
        
        # Perturbation: flip k random bits (maintaining balance)
        k = perturbation_size
        for _ in range(k):
            ones = np.where(best_b == 1)[0]
            zeros = np.where(best_b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_b[i] = 0
            best_b[j] = 1
    
    return best_b, best_c

def fft_gradient_step(h, n, T, lr, time_start, budget_s, max_steps=500):
    """Gradient descent on the smoothed C₅ objective using FFT for gradient computation"""
    h = h.copy().astype(float)
    N_fft = int(2 ** np.ceil(np.log2(2 * n - 1)))
    dx = 2.0 / n
    n_corr = 2 * n - 1
    shifts = np.arange(n_corr) - (n - 1)
    
    # Precompute FFT of v (the weight kernel changes each step, so we can't precompute)
    best_c = compute_c5(h)
    best_h = h.copy()
    
    for step in range(max_steps):
        if step % 50 == 0:
            if (time.time() - time_start) > budget_s - 30:
                break
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])
        for i in range(min(diff, len(order))):
            b[zeros[order[i]]] = 1
    elif diff < 0:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])
        for i in range(min(-diff, len(order))):
            b[ones[order[i]]] = 0
    return b

                break

        T = T0 * (T1 / T0) ** (frac ** 0.6)

        chosen_zeros = rng.choice(zeros, size=k, replace=False)
        
        b[chosen_ones] = 0
        b[chosen_zeros] = 1
        new_c = compute_c5(b)

                best_b = b.copy()
        else:
            b[chosen_ones] = 1
            b[chosen_zeros] = 0

def sa_ilS(h, n, n_restarts, sa_iters_per_restart, T0, T1, perturbation, rng, time_start, budget_s):
    """Iterated Local Search with multi-point SA"""
    best_b = to_binary(h, n)
    best_c = compute_c5(best_b)
    
    for r in range(n_restarts):
        if (time.time() - time_start) > budget_s - 60:
            break
        
        # Local search from current best
        b, c = sa_multi_point(best_b, n, sa_iters_per_restart, T0, T1, rng, time_start, budget_s)
        if c < best_c:
            best_c = c
            best_b = b

        k = perturbation
        for _ in range(k):

            zeros = np.where(best_b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            best_b[i] = 0
            best_b[j] = 1
    
    return best_b, best_c

    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def con_sum(x):
        return np.sum(x) - n / 2.0

        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
