```python
    # So np.correlate(w, v, 'full')[k] = sum_i w[i] * v[i + k - n + 1]
    # We want sum_i w[i] * v[i + (j) ] ... 
    # If k - n + 1 = j, then k = j + n - 1
    # But i + k - n + 1 = i + j should be in [0, n-1]
    # And i should be in [0, 2n-2]
    # So: np.correlate(w, v, 'full')[j + n - 1] = sum_i w[i] * v[i+j] (valid terms)
    # Hmm, but w has length 2n-1 and v has length n
    # np.correlate(w, v, 'full') has length (2n-1) + n - 1 = 3n-2
    
    # Actually I realize this is getting too complicated. Let me just use the direct formula
    # with numpy broadcasting for moderate n.
    
    # For n up to 2048, a (2n-1) x n matrix is 2048*4095 ≈ 8M elements. Manageable.
    
    # A[j] = sum_m w[m] * v[(j + m - (n-1)) % n] -- NO, it's not periodic, it's linear
    # A[j] = sum_m w[m] * v[j+m-n+1] where 0 <= j+m-n+1 < n
    
    # Let me just loop over j (it's only n iterations) and do a vectorized inner product
    grad_first = np.zeros(n)
    grad_second = np.zeros(n)
    m_idx = np.arange(2*n - 1)
    
    for j in range(n):
        # First term: v[j + m - (n-1)] valid when 0 <= j+m-n+1 < n
        # i.e., n-1-j <= m <= n-1+j
        lo = max(0, n-1-j)
        hi = min(2*n-2, n-1+j)
        if lo <= hi:
            v_idx = j + m_idx[lo:hi+1] - (n-1)
            grad_first[j] = np.dot(w[lo:hi+1], v[v_idx])
        
        # Second term: h[j - m + (n-1)] valid when 0 <= j-m+n-1 < n
        # i.e., j-n+1 <= m <= j+n-1
        lo2 = max(0, j-n+1)
        hi2 = min(2*n-2, j+n-1)
        if lo2 <= hi2:
            h_idx = j - m_idx[lo2:hi2+1] + (n-1)
            grad_second[j] = np.dot(w[lo2:hi2+1], h[h_idx])
    
    return dx * (grad_first - grad_second)
```

This is O(n²) but with good numpy vectorization (each inner dot product is O(n) in C). For n=1024, each j takes O(n) = 1024 ops, total O(n²) = 1M ops. Very fast!

Actually wait, the loop over j is n=1024 iterations, each doing a dot product of length up to n=1024. So total is n * n = n² = 1M multiply-adds. In numpy, this is essentially instantaneous (< 1ms).

But Python loop overhead: 1024 iterations of Python code. Each iteration has some overhead (~1μs). So total loop overhead is ~1ms. Plus the actual computation ~1ms. Total: < 5ms per gradient evaluation.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)

def smooth_corr_obj(h, T):
    """Compute smooth approximation of max correlation."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -50, 50)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    val = (mc + np.log(sum_exp) / T)
    weights = exp_s / sum_exp
    return val, weights, corr

    dx = 2.0 / n
    v = 1.0 - h
    val, weights, corr = smooth_corr_obj(h, T)
    soft_val = val * dx
    viol = np.sum(h) - n / 2.0
    f = soft_val + mu * viol * viol
    
    # Compute gradient analytically
    m_idx = np.arange(2 * n - 1)
    grad = np.zeros(n)
    
    for j in range(n):
        # First term: sum_m w[m] * v[j+m-n+1] (valid where 0 <= j+m-n+1 < n)
        lo = max(0, n - 1 - j)
        hi = min(2 * n - 2, n - 1 + j)
        if lo <= hi:
            v_idx = j + m_idx[lo:hi+1] - (n - 1)
            grad[j] += np.dot(weights[lo:hi+1], v[v_idx])
        
        # Second term: -sum_m w[m] * h[j-m+n-1] (valid where 0 <= j-m+n-1 < n)
        lo2 = max(0, j - n + 1)
        hi2 = min(2 * n - 2, j + n - 1)

            h_idx = j - m_idx[lo2:hi2+1] + (n - 1)
            grad[j] -= np.dot(weights[lo2:hi2+1], h[h_idx])
    
    grad *= dx
    grad += 2.0 * mu * viol
    
    return f, grad

def lbgfb_with_grad(h_start, n, T, mu=500.0, max_iter=200):
    bounds = [(0.0, 1.0)] * n
    
    def f(h):
        val, _, _ = smooth_corr_obj(h, T)
        viol = np.sum(h) - n / 2.0
        return val * (2.0/n) + mu * viol * viol

        _, g = smooth_obj_and_grad(h, T, mu)
        return g
