def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def soft_max_obj_and_grad(h, T, lam=0.0):
    """Compute soft-max objective and gradient with optional sum penalty."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    exp_shifted = np.exp(T * (corr - m))
    log_sum = np.log(np.sum(exp_shifted))
    F = (m + log_sum / T) * dx
    
    # Softmax weights
    p = exp_shifted / np.sum(exp_shifted)
    
    # Gradient of soft-max part
    grad = np.zeros(n)
    # Only iterate over significant p values
    thresh = max(1e-12, np.max(p) * 1e-8)
    sig_idx = np.where(p > thresh)[0]
    
    for k in sig_idx:
        pk = p[k]
        shift = k - (n - 1)
        # d(corr[k])/d(h[j]) = (1-h[j-shift]) - h[j+shift] (where indices valid)
        # Term 1: (1-h[j-shift]) valid when 0 <= j-shift < n
        t1_lo = max(0, shift)
        t1_hi = min(n - 1, shift + n - 1)
        if t1_lo <= t1_hi:
            idx = t1_lo
            h_idx = idx - shift
            grad[t1_lo:t1_hi+1] += pk * (1.0 - h[h_idx:h_idx + (t1_hi - t1_lo + 1)])
        
        # Term 2: -h[j+shift] valid when 0 <= j+shift < n
        t2_lo = max(0, -shift)
        t2_hi = min(n - 1, n - 1 - shift)
        if t2_lo <= t2_hi:
            idx = t2_lo
            h_idx = idx + shift
            grad[t2_lo:t2_hi+1] -= pk * h[h_idx:h_idx + (t2_hi - t2_lo + 1)]
    
    grad *= dx
    
    # Add penalty gradient
    if lam > 0:
        sum_dev = np.sum(h) - n / 2.0
        F += lam * sum_dev ** 2
        grad += 2.0 * lam * sum_dev
    
    return F, grad

    log_sum = np.log(np.sum(exp_shifted))

    # Softmax weights over top shifts only (but we need the full corr for gradient)
    p_top = exp_shifted / np.sum(exp_shifted)

        t2_lo = max(0, -shift)
        t2_hi = min(n - 1, n - 1 - shift)
        if t2_lo <= t2_hi:
            h_idx = np.arange(t2_lo, t2_hi + 1) + shift
            grad[t2_lo:t2_hi+1] -= pk * h[h_idx]
    
    grad *= dx
    
    if lam > 0:
        sum_dev = np.sum(h) - n / 2.0
        F += lam * sum_dev ** 2
        grad += 2.0 * lam * sum_dev
    
    return F, grad
