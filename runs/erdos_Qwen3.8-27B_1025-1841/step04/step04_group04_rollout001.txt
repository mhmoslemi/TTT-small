    best_c = compute_c5(h)

            break
        improved = False
        for i in range(n):

                continue
            old_i = h[i]
            candidates = [0.0, 1.0, 0.5,
                         max(0.0, old_i - 0.1), max(0.0, old_i - 0.05),
                         max(0.0, old_i - 0.01), max(0.0, old_i - 0.001),
                         min(1.0, old_i + 0.001), min(1.0, old_i + 0.01),
                         min(1.0, old_i + 0.05), min(1.0, old_i + 0.1)]
            best_v = old_i
            best_c_i = best_c

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argsort(np.abs(h - 0.5))[:10]

            else:
                h[k] += max(diff, -h[k])
            diff = target - np.sum(h)
            if abs(diff) < 1e-15:
                break
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max(corr, T):

    if T <= 1.0:
        return mc
    shifted = corr - mc
    max_shift = np.max(shifted)
    if max_shift * T > 700:
        mask = shifted > (max_shift - 30.0 / T)
        return mc + (max_shift + np.log(np.sum(np.exp(T * shifted[mask])) ) - T * max_shift) / T
    return mc + np.log(np.sum(np.exp(T * shifted))) / T

                v = old_i + d
                if 0.0 <= v <= 1.0 and abs(v - old_i) > 1e-14:
                    cands.append(v)
            best_v = old_i
            best_c_i = best_c
            for v in cands:
                delta = old_i - v
                new_p = h[partner] + delta
                if new_p < -1e-12 or new_p > 1.0 + 1e-12:
                    continue
