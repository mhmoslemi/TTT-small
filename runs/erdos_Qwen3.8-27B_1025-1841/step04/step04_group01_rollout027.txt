        h = np.clip(h, 0.0, 1.0)
    # Final fix
    d = target - h.sum()
    if abs(d) > 1e-12:
        if d > 0:
            m = h < 1.0
            if np.any(m):
                h[m[np.argmax(1-h[m])]] += d
        else:
            m = h > 0
            if np.any(m):
                h[m[np.argmax(h[m])]] += d
    return h

def _slsqp_obj(x, n, T):
    c = compute_corr(x)
    mc = np.max(c)
    shifted = c - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)

def _con_fun(x, n):
    return x.sum() - n / 2.0

def _con_jac(x, n):
    return np.ones(n)

            # Compute gradient for top shifts
            # g_k(j) = (1 - h[j-s_k]) - h[j+s_k]
            # where s_k = top_idx[k] - n + 1
            g_top = np.empty(40)
            for ki in range(40):
                s = top_idx[ki] - n + 1
                js = j + s
