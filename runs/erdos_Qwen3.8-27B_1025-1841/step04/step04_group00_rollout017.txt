where delta_h[i] = h_new[i] - h_old[i] (= -1 at i=a, +1 at i=b, 0 elsewhere).

This is exact ONLY when no single term has both delta_h[i] ≠ 0 AND delta_h[i-L] ≠ 0. When both are nonzero (i.e., i and i-L are both in {a,b}), the linearization misses the second-order term delta_h[i] * delta_h[i-L].

The second-order term for element e in the product h[i]*(1-h[i-L]) = h[i] - h[i]*h[i-L]:
if both h[i] and h[i-L] change, the second-order term is -delta_h[i]*delta_h[i-L].

For i=b, L=b-a: h[i]=h[b] changes by +1, h[i-L]=h[a] changes by -1.
Second-order: -(+1)*(-1) = +1.

For i=a, L=a-b: h[i]=h[a] changes by -1, h[i-L]=h[b] changes by +1.
Second-order: -(-1)*(+1) = +1.

So the correction is: add -delta_h[i]*delta_h[i-L] for each i where both change.

The i values where both h[i] and h[i-L] change: i ∈ {a,b} and i-L ∈ {a,b}.
- i=a, i-L=a: L=0 (trivial, no effect since h[i]*(1-h[i])=0 for binary)
- i=a, i-L=b: L=a-b
- i=b, i-L=a: L=b-a
- i=b, i-L=b: L=0 (trivial)

So corrections at L=a-b and L=b-a, each contributing +1.

In m-index: m_ab = n1 + a - b, m_ba = n1 + b - a.

Great, my earlier derivation was correct. Add +1 at those two m values.

import numpy as np
import time

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def threshold_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def swap_delta(h, a, b, n):
    """O(n) delta for full correlation when h[a]:1->0, h[b]:0->1. Assumes h[a]=1, h[b]=0."""
    n1 = n - 1
    ns = 2*n1 + 1  # length of full correlation
    delta = np.zeros(ns)
    
    # a-part: h[a] 1->0
    # i=a term: -(1-h[a-L]) for valid a-L ∈ [0,n)
    # a-L valid: L ∈ [a-n1, a] => m ∈ [a, a+n1] (clipped to [0, ns-1])
    lo_a1 = max(a, 0)
    hi_a1 = min(a + n1, ns - 1)
    if lo_a1 <= hi_a1:
        m_vals = np.arange(lo_a1, hi_a1 + 1)
        a_minus_L = a + n1 - m_vals  # = a - (m - n1)
        delta[lo_a1:hi_a1+1] -= (1.0 - h[a_minus_L])
    
    # i=a+L term: +h[a+L] for valid a+L ∈ [0,n)
    # a+L valid: L ∈ [-a, n1-a] => m ∈ [n1-a, n1+n1-a] = [n1-a, 2n1-a]
    lo_a2 = max(n1 - a, 0)
    hi_a2 = min(2*n1 - a, ns - 1)
    if lo_a2 <= hi_a2:
        m_vals = np.arange(lo_a2, hi_a2 + 1)
        a_plus_L = m_vals - n1 + a
        delta[lo_a2:hi_a2+1] += h[a_plus_L]
    
    # b-part: h[b] 0->1
    # i=b term: +(1-h[b-L]) for valid b-L ∈ [0,n)
    lo_b1 = max(b, 0)
    hi_b1 = min(b + n1, ns - 1)
    if lo_b1 <= hi_b1:
        m_vals = np.arange(lo_b1, hi_b1 + 1)
        b_minus_L = b + n1 - m_vals
        delta[lo_b1:hi_b1+1] += (1.0 - h[b_minus_L])
    
    # i=b+L term: -h[b+L] for valid b+L ∈ [0,n)
    lo_b2 = max(n1 - b, 0)
    hi_b2 = min(2*n1 - b, ns - 1)
    if lo_b2 <= hi_b2:
        m_vals = np.arange(lo_b2, hi_b2 + 1)
        b_plus_L = m_vals - n1 + b
        delta[lo_b2:hi_b2+1] -= h[b_plus_L]
    
    # Second-order corrections
    m_ab = n1 + a - b
    if 0 <= m_ab < ns:
        delta[m_ab] += 1.0
    m_ba = n1 + b - a
    if 0 <= m_ba < ns:
        delta[m_ba] += 1.0
    
    return delta

def binary_local_search(h_bin, n, time_budget, start_time, rng, top_m=15, max_stall=200):
    """Efficient binary local search with incremental correlation updates."""
    dx = 2.0 / n
    n1 = n - 1
    ns = 2*n1 + 1
    
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full').astype(float)
    c5 = float(np.max(corr) * dx)
    best_h = h_bin.copy()
    best_c5 = c5
    best_corr = corr.copy()
    
    ones = np.where(h_bin == 1)[0]
    zeros = np.where(h_bin == 0)[0]
    
    stall = 0
    T = 0.5  # annealing temperature (in units of corr value)
    
    while time.time() - start_time < time_budget - 10:
        # Find top shifts
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        max_val = np.max(corr)
        
        # Generate candidate swaps focused on top shifts
        best_move = None
        best_new_c5 = c5
        trials = min(500, max(100, n // 2))
        
        # Strategy: for each top shift, find positions that contribute and try to fix them
        # Also try random swaps
        candidates = []
        
        # Random candidates
        for _ in range(trials // 2):
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            if a != b:
                candidates.append((a, b))
        
        # Targeted candidates: for top shifts, find 1s that have 0 at distance L
        for m in top_idx[:5]:
            L = m - n1
            # Find i where h[i]=1 and h[i-L]=0 (i-L valid)
            if L >= 0:
                i_range = np.arange(L, n)
            else:
                i_range = np.arange(0, n + L)
            if len(i_range) == 0:
                continue
            # Get the ones in this range
            ones_in_range = ones[(ones >= i_range[0]) & (ones < i_range[-1]+1)] if len(ones) > 0 else np.array([])
            # Try moving some of these ones
            for _ in range(10):
                if len(ones_in_range) == 0 or len(zeros) == 0:
                    break
                a = ones_in_range[rng.randint(len(ones_in_range))]
                b = zeros[rng.randint(len(zeros))]
                if a != b:
                    candidates.append((a, b))
        
        # Evaluate candidates
        improved = False
        for a, b in candidates:
            if h_bin[a] != 1 or h_bin[b] != 0:
                continue
            d = swap_delta(h_bin, a, b, n)
            new_corr = corr + d
            new_max = float(np.max(new_corr))
            new_c5 = new_max * dx
            
            if new_c5 < best_c5 - 1e-15:
                # Accept and update
                h_bin[a] = 0
                h_bin[b] = 1
                corr = new_corr
                c5 = new_c5
                best_h = h_bin.copy()
                best_c5 = c5
                best_corr = corr.copy()
                ones = np.where(h_bin == 1)[0]
                zeros = np.where(h_bin == 0)[0]
                improved = True
                stall = 0
                break
            elif new_c5 < c5 + T * max_val * 0.1:
                # Annealing: accept worse move with some probability
                if rng.random() < np.exp(-(new_c5 - c5) / (T * dx)):
                    h_bin[a] = 0
                    h_bin[b] = 1
                    corr = new_corr
                    c5 = new_c5
                    ones = np.where(h_bin == 1)[0]
                    zeros = np.where(h_bin == 0)[0]
                    stall += 1
                    # Don't break, try more
        
        if improved:
            stall = 0
        else:
            stall += 1
            # Cool down
            if stall > 50:
                T *= 0.8
                if T < 0.01:
                    T = 0.01
            # Periodic restart: random perturbation
            if stall > max_stall:
                # Do 10 random swaps
                for _ in range(10):
                    a = ones[rng.randint(len(ones))]
                    b = zeros[rng.randint(len(zeros))]
                    if a != b:
                        d = swap_delta(h_bin, a, b, n)
                        h_bin[a] = 0
                        h_bin[b] = 1
                        corr += d
                        ones = np.where(h_bin == 1)[0]
                        zeros = np.where(h_bin == 0)[0]
                stall = 0
                T = 0.3
    
    return best_h, best_c5

The correction was derived for the case where a+L = b (L = b-a) or b+L = a (L = a-b). At these lags, the "i=a+L" and "i=b" terms (or "i=b+L" and "i=a" terms) refer to the same index, and the linearization has a second-order error.

At L = b-a: the overlapping index is i = b = a+L.
- a-part at i=a+L=b: +h[a+L] = +h[b] = +0 (original h[b]=0)
- b-part at i=b: +(1-h[b-L]) = +(1-h[a]) = +(1-1) = +0
- True change at i=b: new h[b]*(1-new h[a]) - old h[b]*(1-old h[a]) = 1*1 - 0*0 = 1
- Naive: 0 + 0 = 0
- Correction: +1 ✓

At L = 0 with a and b arbitrary: the overlap is at i=a (where i=a and i=a+L=a coincide) AND at i=b (where i=b and i=b+L=b coincide).
- At i=a: a-part i=a gives -(1-h[a]) = 0, a-part i=a+L=a gives +h[a] = +1. Naive: 1. True: 0.
- At i=b: b-part i=b gives +(1-h[b]) = +1, b-part i=b+L=b gives -h[b] = 0. Naive: 1. True: 0.
- Total naive: 2. Total true: 0. Error: +2.

So at L=0, the error is +2, not +1. The correction formula I had (add +1 at m_ab and m_ba) doesn't handle L=0 correctly because the overlap is DIFFERENT (it's a+L=a and b+L=b, not a+L=b).
