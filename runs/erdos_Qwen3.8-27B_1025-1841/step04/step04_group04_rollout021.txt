    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n, rng=None):
    if rng is not None:
        h = h + rng.randn(n) * 0.01
    sorted_idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[sorted_idx[:n // 2]] = 1.0
    return binary

def build_delta(h, n):
    lags = np.arange(-(n - 1), n)
    p = np.arange(n)[:, None]
    l = lags[None, :]
    p_plus_l = p + l
    p_minus_l = p - l
    valid_plus = (p_plus_l >= 0) & (p_plus_l < n)
    valid_minus = (p_minus_l >= 0) & (p_minus_l < n)
    ppc = np.clip(p_plus_l, 0, n - 1)
    pmc = np.clip(p_minus_l, 0, n - 1)
    direct = (1.0 - h[ppc]) * valid_plus
    indirect = h[pmc] * valid_minus
    return direct - indirect

def binary_local_search(h, n, max_passes=200, time_limit=None, start_time=None):
    h = to_binary(h, n).copy()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    M = float(np.max(corr))
    delta = build_delta(h, n)
    for _pass in range(max_passes):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        slack = M - corr
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        best_new_M = M
        best_pair = None
        for j in ones_idx:
            upper = delta[j] + slack
            viol = (delta[zeros_idx] - upper[None, :]).max(axis=1)
            good = np.where(viol < 0)[0]
            if len(good) > 0:
                cand_delta = delta[zeros_idx[good]] - delta[j]
                new_Ms = (corr[None, :] + cand_delta).max(axis=1)
                bl = new_Ms.argmin()
                if new_Ms[bl] < best_new_M:
                    best_new_M = new_Ms[bl]
                    best_pair = (j, zeros_idx[good][bl])
        if best_pair is None:
            break
        j, jp = best_pair
        h[j] = 0.0
        h[jp] = 1.0
        corr = corr + delta[jp] - delta[j]
        M = best_new_M
        delta = build_delta(h, n)
    return h, M * dx

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_N, _OPT_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(320, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binary local search at initial resolution with multiple starts
    n1 = n_init
    for attempt in range(5):
        if remaining() < 120:
            break
        if attempt == 0:
            h_try = h_init.copy()
        else:
            h_try = h_init + rng.randn(n1) * rng.uniform(0.02, 0.08)
            h_try = make_feasible(h_try, n1)
        h_bin, c5_bin = binary_local_search(h_try, n1, max_passes=200,
                                            time_limit=min(remaining() - 30, 200),
                                            start_time=start_time)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Phase 2: Iterated local search with kicks on the best binary
    if remaining() > 100:
        n2 = len(best_h)
        h_cur = best_h.copy()
        c5_cur = best_c5
        for kick_round in range(10):
            if remaining() < 60:
                break
            # Random kick: flip 3-8 random pairs
            n_kicks = rng.randint(3, 9)
            h_kicked = h_cur.copy()
            ones = np.where(h_kicked == 1.0)[0]
            zeros = np.where(h_kicked == 0.0)[0]
            for _ in range(n_kicks):
                if len(ones) == 0 or len(zeros) == 0:
                    break
                j = ones[rng.randint(len(ones))]
                jp = zeros[rng.randint(len(zeros))]
                h_kicked[j] = 0.0
                h_kicked[jp] = 1.0
                ones = np.where(h_kicked == 1.0)[0]
                zeros = np.where(h_kicked == 0.0)[0]
            h_new, c5_new = binary_local_search(h_kicked, n2, max_passes=150,
                                                time_limit=min(remaining() - 20, 120),
                                                start_time=start_time)
            if c5_new < c5_cur:
                c5_cur = c5_new
                h_cur = h_new.copy()
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()

    # Phase 3: Continuous SLSQP at higher resolution
    n3_list = [int(n1 * 1.5), int(n1 * 2)]
    for n3 in n3_list:
        if remaining() < 120:
            break
        h_start = upsample_h(best_h, n3)
        _OPT_N = n3
        for T in [80.0, 200.0, 400.0]:
            if remaining() < 60:
                break
            _OPT_T = T
            bounds = [(0.0, 1.0)] * n3
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_obj, h_start, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-12})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n3)
                    h_start = best_h.copy()
            except Exception:
                pass

    # Phase 4: Binary search at the current best resolution (if reasonable)
    n4 = len(best_h)
    if n4 <= 960 and remaining() > 60:
        h_bin4, c5_bin4 = binary_local_search(best_h, n4, max_passes=150,
                                              time_limit=min(remaining() - 20, 200),
                                              start_time=start_time)
        if c5_bin4 < best_c5:
            best_c5 = c5_bin4
            best_h = h_bin4.copy()

    # Phase 5: Final SLSQP polish
    if remaining() > 50:
        n5 = len(best_h)
        _OPT_N = n5
        for T_final in [300.0, 600.0]:
            if remaining() < 40:
                break
            _OPT_T = T_final
            bounds = [(0.0, 1.0)] * n5
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_obj, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-13})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n5)
            except Exception:
                pass

    # Phase 6: Final coordinate descent for near-binary fine-tuning
    n6 = len(best_h)
    for _round in range(30):
        if remaining() < 10:
            break
        improved = False
        for j in range(n6):
            if remaining() < 5:
                break
            old = best_h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.05), max(0.0, old - 0.01),
                         min(1.0, old + 0.01), min(1.0, old + 0.05)]:
                if abs(cand - old) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n6)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_val = cand
            if best_val != old:
                best_h = best_h.copy()
                best_h[j] = best_val
                best_h = make_feasible(best_h, n6)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
