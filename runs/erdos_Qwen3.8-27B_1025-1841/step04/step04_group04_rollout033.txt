def subgradient_step(h, alpha, n):
    """One projected subgradient step."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    
    # Subgradient of c[k_star] with respect to h
    # c[k] = sum_i h[i] * (1 - h[i+k-n+1]) for the 'full' mode
    # Actually, let me just use the formula:
    # For full correlation, result[k] = sum_i a[i+k] * v[i] where a=h, v=1-h
    # So result[k] = sum_i h[i+k] * (1-h[i])
    # d/dh[j] result[k] = (1-h[j]) * delta_{j+k, valid} + sum_i h[i+k] * (-delta_{i, j})
    # = (1-h[j]) if shift is valid - h[j-k] if j-k is valid
    
    # For the 'full' mode with length 2n-1, the lag s = k - (n-1)
    # result[n-1+s] = sum_i h[i+s] * (1-h[i]) for valid i
    # d/dh[j] result[n-1+s] = (1-h[j-s]) - h[j+s] (with bounds checking)
    
    s = k_star - (n - 1)
    sg = np.zeros(n)
    
    # First term: (1-h[j-s]) for valid j-s
    if s >= 0:
        sg[:n-s] += (1.0 - h[s:])  # h[j-s] for j in [s, n-1] → sg[j] for j in [s, n-1]
        # Wait, I need to be more careful
        # result[n-1+s] = sum_i h[i+s]*(1-h[i]) for i=0 to n-1-s-1
        # d/dh[j] = (1-h[j]) if j+s <= n-1 (i.e., j <= n-1-s) [from the h[i+s] term with i=j-s... no]
        # Let me redo:
        # result[n-1+s] = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
        # d/dh[j] = derivative of h[j] in this sum
        # h[j] appears as h[i+s] when i = j-s (if 0 <= j-s <= n-1-s, i.e., s <= j <= n-1)
        #         and as h[i] when i = j (if 0 <= j <= n-1-s)
        # So: d/dh[j] = (1-h[j-s])*[s <= j <= n-1] - h[j+s]*[0 <= j <= n-1-s]
    
    # Let me just compute it numerically for clarity
    # For each j, the contribution is:
    # (1-h[j-s]) if s <= j < n AND 0 <= j-s < n-s... 
    # Actually this is getting confusing. Let me use a different approach.
    
    # Simpler: just compute the gradient numerically or use the correlation structure
    # For the 'full' correlation:
    # corr_full[k] = sum_i h[i] * (1-h)[i + k - (n-1)]
    # where the sum is over valid i
    # Let's define shift = k - (n-1) = s
    # corr_full[n-1+s] = sum_i h[i] * (1-h)[i+s] (valid i)
    # = sum_i h[i] * (1-h[i+s])
    
    # d/dh[j] = (1-h[j+s]) [if i=j is valid, i.e., 0<=j<n and 0<=j+s<n] 
    #           - h[j-s] [if i+s=j is valid, i.e., 0<=j-s<n and 0<=j<n]
    
    if s >= 0:
        # (1-h[j+s]) for j = 0 to n-1-s-1
        sg[:n-s] += (1.0 - h[s:n])
        # -h[j-s] for j = s to n-1
        sg[s:] -= h[:n-s]
    else:
        s_abs = -s
        # (1-h[j+s]) = (1-h[j-s_abs]) for j = s_abs to n-1
        sg[s_abs:] += (1.0 - h[:-s_abs])
        # -h[j-s] = -h[j+s_abs] for j = 0 to n-1-s_abs-1
        sg[:n-s_abs] -= h[s_abs:n]
    
    # Projected gradient step
    h_new = h - alpha * sg
    h_new = make_feasible(h_new, n)
    return h_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Get initial solution
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h_init = np.full(480, 0.5)
    
    best_h = make_feasible(h_init, len(h_init))
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Subgradient method at moderate resolution for exploration
    n1 = 960
    h_sg = upsample_h(best_h, n1) if len(best_h) != n1 else best_h.copy()
    c5_sg = compute_c5(h_sg)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()
    
    # Run subgradient method
    dx = 2.0 / n1
    n_iter = 5000
    for t in range(n_iter):
        if remaining() < 300:
            break
        alpha = 0.5 / (1 + 0.01 * t)
        h_sg = subgradient_step(h_sg, alpha, n1)
        if t % 100 == 0:
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
    
    # Phase 2: SLSQP at higher resolution
    for n_pts in [1440, 1920]:
        if remaining() < 150:
            break
        
        h_work = upsample_h(best_h, n_pts)
        c5_work = compute_c5(h_work)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
        
        dx = 2.0 / n_pts
        
        def obj_slsqp(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            mc = np.max(corr)
            T = 300.0
            shifted = corr - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        
        def con_fun(x):
            return np.sum(x) - n_pts / 2.0
        
        def con_jac(x):
            return np.ones(n_pts)
        
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        
        # Start from current best
        for s in range(3):
            if remaining() < 80:
                break
            if s == 0:
                x0 = h_work.copy()
            elif s == 1:
                x0 = h_work + rng.randn(n_pts) * 0.015
                x0 = make_feasible(x0, n_pts)
            else:
                x0 = (h_work > 0.5).astype(float)
                x0 = make_feasible(x0, n_pts)
            
            try:
                res = minimize(obj_slsqp, x0, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 250, 'ftol': 1e-12})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_pts)
                    h_work = best_h.copy()
            except Exception:
                pass
        
        # Coordinate descent at this resolution
        n = len(best_h)
        for cd_round in range(15):
            if remaining() < 30:
                break
            improved = False
            for j in range(n):
                if remaining() < 10:
                    break
                old = best_h[j]
                best_c = best_c5
                best_val = old
                for cand in [0.0, 1.0, 0.5,
                             max(0.0, old-0.02), max(0.0, old-0.005),
                             min(1.0, old+0.005), min(1.0, old+0.02)]:
                    if abs(cand - old) < 1e-15:
                        continue
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_val = cand
                if best_val != old:
                    best_h[j] = best_val
                    best_h = make_feasible(best_h, n)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break
    
    # Phase 3: Binary local search with simulated annealing
    if remaining() > 80:
        n = len(best_h)
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        
        h_sa = best_h.copy()
        c5_sa = compute_c5(h_sa)
        T_sa = 0.005
        ones = np.where(h_sa > 0.5)[0]
        zeros = np.where(h_sa < 0.5)[0]
        
        n_sa_steps = min(3000, int(remaining() / 0.02))
        for sa_iter in range(n_sa_steps):
            if remaining() < 20:
                break
            if sa_iter % 300 == 299:
                T_sa *= 0.9
            
            i_idx = rng.randint(len(ones))
            j_idx = rng.randint(len(zeros))
            
            trial = h_sa.copy()
            trial[ones[i_idx]] = 0.0
            trial[zeros[j_idx]] = 1.0
            
            c5_trial = compute_c5(trial)
            delta = c5_trial - c5_sa
            
            if delta < 0 or (T_sa > 1e-8 and rng.rand() < np.exp(-delta / T_sa)):
                h_sa = trial
                c5_sa = c5_trial
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
                # Update ones/zeros
                old_one = ones[i_idx]
                old_zero = zeros[j_idx]
                # Replace in arrays
                ones_arr = list(ones)
                zeros_arr = list(zeros)
                ones_arr[i_idx] = old_zero
                zeros_arr[j_idx] = old_one
                ones = np.array(ones_arr)
                zeros = np.array(zeros_arr)
    
    # Phase 4: Final SLSQP polish
    if remaining() > 50:
        n = len(best_h)
        dx = 2.0 / n
        
        def obj_final(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            mc = np.max(corr)
            T = 500.0
            shifted = corr - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        
        def con_fun3(x):
            return np.sum(x) - n / 2.0
        
        def con_jac3(x):
            return np.ones(n)
        
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun3, 'jac': con_jac3}]
        
        try:
            res = minimize(obj_final, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-13})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n)
        except Exception:
            pass
