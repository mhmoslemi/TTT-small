```python
import time

def fft_corr(h):
    """Compute circular correlation of h with 1-h using FFT."""
    n = len(h)
    H = np.fft.rfft(h)
    H1 = np.fft.rfft(1.0 - h)
    return np.real(np.fft.irfft(H * np.conj(H1)))

def compute_c5_exact(h):
    """Compute C5 using exact np.correlate as specified."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_fft(h):
    """Compute C5 using FFT circular correlation (upper bound on exact)."""
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)
```

```python
def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def fft_corr(h):
    n = len(h)
    H = np.fft.rfft(h)
    H1 = np.fft.rfft(1.0 - h)
    return np.real(np.fft.irfft(H * np.conj(H1)))

def c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(fft_corr(h)) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = h < 1.0 - 1e-12 if diff > 0 else h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    ox = np.linspace(0, 2, old_n, endpoint=False)
    nx = np.linspace(0, 2, new_n, endpoint=False)
    return make_feasible(np.interp(nx, ox, h, period=2.0), new_n)

def obj_grad(h, T, mu):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    mc = np.max(corr)
    sh = np.clip(T * (corr - mc), -80, 80)
    w = np.exp(sh)
    ws = np.sum(w)
    obj = (mc + np.log(ws) / T) * dx + mu * (np.sum(h) - n/2.0)**2
    
    W = np.fft.rfft(w)
    H = np.fft.rfft(h)
    S1 = np.real(np.fft.irfft(W * H))
    S2 = np.real(np.fft.irfft(np.conj(W) * H))
    cg = ws - S1 - S2
    viol = np.sum(h) - n / 2.0
    grad = dx * cg / (T * ws) + 2.0 * mu * viol
    return obj, grad

def lbfgsb(h0, T, mu, maxiter, budget_check):
    n = len(h0)
    def f(x, jac=False):
        o, g = obj_grad(x, T, mu)
        return o if not jac else (o, g)
    res = minimize(f, h0, method='L-BFGS-B',
                   bounds=[(0,1)]*n, options={'maxiter': maxiter, 'ftol': 1e-18, 'gtol': 1e-14}, jac=True)
    return make_feasible(res.x, n)

def binary_search(h_bin, n, rng, max_rounds, budget_check):
    h = h_bin.copy()
    cur = c5_fft(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for r in range(max_rounds):
        if budget_check() < 30:
            break
        improved = False
        for _ in range(min(3000, len(ones) * len(zeros))):
            if budget_check() < 30:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            if oi == oj:
                continue
            h[oi], h[oj] = 0.0, 1.0
            c = c5_fft(h)
            if c < cur - 1e-12:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi], h[oj] = 1.0, 0.0
        if not improved:
            break
    return h, cur
```
