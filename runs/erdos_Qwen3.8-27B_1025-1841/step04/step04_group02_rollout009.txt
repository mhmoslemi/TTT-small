```python
def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def eval_swap_delta(h, p, q):
    """Compute delta in correlation when h[p] 1->0 and h[q] 0->1"""
    n = len(h)
    L = np.arange(-(n-1), n)  # all lags
    # delta(L) = h[p+L] - h[q+L] + h[p-L] - h[q-L]
    # Need to handle boundaries
    delta = np.zeros(2*n - 1)
    
    # h[p+L] is valid when 0 <= p+L <= n-1, i.e., -p <= L <= n-1-p
    mask1 = (p + L >= 0) & (p + L < n)
    delta[mask1] += h[p + L[mask1]]
    
    mask2 = (q + L >= 0) & (q + L < n)
    delta[mask2] -= h[q + L[mask2]]
    
    mask3 = (p - L >= 0) & (p - L < n)
    delta[mask3] += h[p - L[mask3]]
    
    mask4 = (q - L >= 0) & (q - L < n)
    delta[mask4] -= h[q - L[mask4]]
    
    return delta
```
