```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    """Compute full correlation vector."""
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    """Project h onto feasible set: 0<=h<=1, sum(h)=n/2."""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Distribute to multiple points to stay in [0,1]
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_gradient(h, corr, t, top_m=10):
    """Compute gradient of log-sum-exp approximation."""
    n = len(h)
    dx = 2.0 / n
    # corr has 2n-1 entries, centered at index n-1
    # corr[i] corresponds to lag i-(n-1)
    
    # Compute softmax weights
    scaled = t * corr
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)
    p = exp_scaled / np.sum(exp_scaled)
    
    # Use only top-m lags
    top_idx = np.argsort(p)[-top_m:]
    
    grad = np.zeros(n)
    for idx in top_idx:
        ps = p[idx]
        if ps < 1e-15:
            continue
        s = idx - (n - 1)  # actual lag
        if s >= 0:
            # Positive lag s
            # Contribution: ps * dx * [(1-h[j+s])*1{0<=j<n-s} - h[j-s]*1{s<=j<n}]
            if s < n:
                # First part: (1-h[j+s]) for j in [0, n-s-1]
                if n - s > 0:
                    grad[:n-s] += ps * dx * (1.0 - h[s:n])
                # Second part: -h[j-s] for j in [s, n-1]
                if s > 0:
                    grad[s:] -= ps * dx * h[:n-s]
        else:
            # Negative lag, s' = -s > 0
            sp = -s
            if sp < n:
                # (1-h[j-sp])*1{sp<=j<n}
                if n - sp > 0:
                    grad[sp:] += ps * dx * (1.0 - h[:n-sp])
                # -h[j+sp]*1{0<=j<n-sp}
                if sp > 0:
                    grad[:n-sp] -= ps * dx * h[sp:n]
    
    return grad

def lse_pgd(h, n, start_time, budget_s, t_schedule=None, top_m=10, 
            step_size=0.05, momentum=0.9, max_iter=50000):
    """Log-sum-exp projected gradient descent."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    no_improve = 0
    max_no_improve = 100
    
    if t_schedule is None:
        t_schedule = [50, 100, 200, 500, 1000, 2000, 5000]
    
    it = 0
    while time.time() - start_time < budget_s and it < max_iter:
        it += 1
        # Adaptive t
        t_idx = min(it // (max_iter // len(t_schedule)), len(t_schedule) - 1)
        t = t_schedule[t_idx]
        
        corr = compute_corr(h)
        c5_cur = np.max(corr) * (2.0 / n)
        
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step_size *= 0.5
                if step_size < 1e-8:
                    break
                no_improve = 0
        
        # Nesterov lookahead
        y = h - momentum * velocity
        y = np.clip(y, 0, 1)
        corr_y = compute_corr(y)
        grad_y = lse_gradient(y, corr_y, t, top_m)
        
        velocity = momentum * velocity - step_size * grad_y
        new_h = h + velocity
        new_h = make_feasible(new_h, n)
        
        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur + 1e-14:
            h = new_h
        else:
            # Try without momentum
            grad = lse_gradient(h, corr, t, top_m)
            new_h2 = make_feasible(h - step_size * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_cur + 1e-14:
                h = new_h2
                velocity = np.zeros(n)
            else:
                step_size *= 0.7
                velocity *= 0.3
            if step_size < 1e-9:
                break
        
        if it % 100 == 0 and step_size < 0.03:
            step_size = min(0.05, step_size * 1.5)
    
    # Final check
    h = make_feasible(h, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_h = h
        best_c5 = c5_final
    return best_h, best_c5

def coord_descent_fast(h, n, start_time, budget_s, steps=None):
    """Fast coordinate descent."""
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() - start_time < budget_s:
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 3:
                return best_h, best_c5
            old_val = best_h[j]
            best_val = old_val
            best_c = best_c5
            for s in steps:
                for delta in [-s, s]:
                    v = old_val + delta
                    if 0.0 < v < 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def slsqp_smooth(h, n, start_time, budget_s, T=100.0, top_m=None):
    """SLSQP with smooth max objective."""
    def obj(x):
        corr = compute_corr(x)
        if top_m is not None:
            top_idx = np.argsort(corr)[-top_m:]
            mc = np.max(corr[top_idx])
            shifted = corr[top_idx] - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h):
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h, compute_c5(h)

def random_01_start(n, rng):

def binary_search_coord(h, n, j, lo, hi, tol=1e-6):
    """Binary search for optimal value of h[j] given others fixed."""
    best_val = h[j]
    best_c = compute_c5(h)
    for _ in range(30):
        mid = (lo + hi) / 2
        if mid <= lo + tol or mid >= hi - tol:
            break
        trial = h.copy()
        trial[j] = mid
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c:
```

```python
def slp_step(h, n, active_lags, max_iter=50):
    """Sequential Linear Programming step."""
    dx = 2.0 / n
    h_current = h.copy()
    
    for _ in range(max_iter):
        corr = compute_corr(h_current)
        # Get gradient for each active lag
        # Solve LP
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2.0
        ]
        for lag_idx in active_lags:
            s = lag_idx - (n - 1)
            # Linearized correlation
            g = np.zeros(n)
            if s >= 0:
                g[:n-s] += (1.0 - h_current[s:n])
                g[s:] -= h_current[:n-s]
            else:
                sp = -s
                g[:n-sp] += (1.0 - h_current[sp:n])
                g[sp:] -= h_current[:n-sp]
            # corr_s_linear = corr[lag_idx] + g · (h_var - h_current)
            constraints.append(corr[lag_idx] + g @ (h_var - h_current) <= t / dx)
        
        obj = cp.Minimize(t)
        prob = cp.Problem(obj, constraints)
        try:

        if np.allclose(h_new, h_current, atol=1e-8):
            break
        h_current = h_new
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def get_lag_gradient(h, n, s):
    """Get gradient of correlation at lag s."""
    g = np.zeros(n)
    if s >= 0:
        if s < n and n - s > 0:
            g[:n-s] += (1.0 - h[s:n])
            g[s:] -= h[:n-s]
    else:
        sp = -s
        if sp < n and n - sp > 0:
            g[:n-sp] += (1.0 - h[sp:n])
            g[sp:] -= h[:n-sp]
    return g

def slp_optimize(h, n, start_time, budget_s, top_m=5, max_outer=20):
    """Sequential Linear Programming optimization."""
    h_current = h.copy()
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)
    
    for outer in range(max_outer):
        if time.time() - start_time > budget_s - 5:
            break
        
        corr = compute_corr(h_current)
        c5_cur = np.max(corr) * (2.0 / n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def get_lag_gradient(h, n, s):
    g = np.zeros(n)
    if s >= 0:
        if s < n and n - s > 0:
            g[:n - s] += (1.0 - h[s:n])
            g[s:] -= h[:n - s]
    else:
        sp = -s
        if sp < n and n - sp > 0:
            g[:n - sp] += (1.0 - h[sp:n])
            g[sp:] -= h[:n - sp]
    return g

def slp_optimize(h, n, start_time, budget_s, top_m=5, max_outer=30, damping=0.8):
    h_current = h.copy()
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)
    dx = 2.0 / n

    for outer in range(max_outer):
        if time.time() - start_time > budget_s - 5:
            break
        corr = compute_corr(h_current)
        c5_cur = np.max(corr) * dx
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h_current.copy()

        top_idx = np.argsort(corr)[-top_m:]

        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
        for idx in top_idx:
            s = idx - (n - 1)
            g = get_lag_gradient(h_current, n, s)
            constraints.append(corr[idx] + g @ (h_var - h_current) <= t / dx)

        prob = cp.Problem(cp.Minimize(t), constraints)
        solved = False
        for solver in [cp.GLOP, cp.CBC, cp.CLARABEL]:
            try:
                prob.solve(solver=solver)
                solved = True
                break
            except Exception:
                continue
        if not solved or h_var.value is None:
            break

        h_lp = np.clip(h_var.value, 0, 1)
        h_lp = make_feasible(h_lp, n)

        alpha = damping
        for _ in range(10):
            h_try = make_feasible((1 - alpha) * h_current + alpha * h_lp, n)
            c5_try = compute_c5(h_try)
            if c5_try <= c5_cur + 1e-14:
                break
            alpha *= 0.5
        if alpha < 0.01:
            break
        h_current = make_feasible((1 - alpha) * h_current + alpha * h_lp, n)
        c5_new = compute_c5(h_current)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_current.copy()
        if abs(c5_new - c5_cur) < 1e-13:
            break

    return best_h, best_c5

def lse_pgd(h, n, start_time, budget_s, t_schedule=None, top_m=10,
            step_size=0.04, momentum=0.9):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    dx = 2.0 / n
    no_improve = 0
    if t_schedule is None:
        t_schedule = [50, 100, 200, 500, 1000, 2000, 5000]
    total_iters = 5000
    it = 0

    while time.time() - start_time < budget_s and it < total_iters:
        it += 1
        t = t_schedule[min(it // (total_iters // len(t_schedule)), len(t_schedule) - 1)]
        corr = compute_corr(h)
        c5_cur = np.max(corr) * dx
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                step_size *= 0.5
                if step_size < 1e-9:
                    break
                no_improve = 0

        y = np.clip(h - momentum * velocity, 0, 1)
        corr_y = compute_corr(y)
        scaled = t * corr_y
        mx = np.max(scaled)
        exp_s = np.exp(scaled - mx)
        p = exp_s / np.sum(exp_s)
        top_idx = np.argsort(p)[-top_m:]
        grad = np.zeros(n)
        for idx in top_idx:
            ps = p[idx]
            if ps < 1e-12:

        steps = [0.01, 0.005, 0.002, 0.001]
    best_h = h.copy()

        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 3:
                return best_h, best_c5
            ov = best_h[j]
            bv, bc = ov, best_c5
            for st in steps:
                for d in [-st, st]:
                    v = ov + d
                    if 0.0 < v < 1.0:
                        t = best_h.copy()
                        t[j] = v
                        t = make_feasible(t, n)
                        c = compute_c5(t)
                        if c < bc - 1e-14:

            if bv != ov:
                best_h[j] = bv
                best_h = make_feasible(best_h, n)
                best_c5 = bc
                improved = True
    return best_h, best_c5

def slsqp_smooth(h, n, start_time, budget_s, T=200.0, top_m=None):
    def obj(x):
        corr = compute_corr(np.clip(x, 0, 1))
        if top_m is not None:
            ti = np.argsort(corr)[-top_m:]
            mc = np.max(corr[ti])
            sh = corr[ti] - mc
            return (mc + np.log(np.sum(np.exp(T * sh))) / T) * (2.0 / n)
        mc = np.max(corr)
        sh = corr - mc
        return (mc + np.log(np.sum(np.exp(T * sh))) / T) * (2.0 / n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
        c5_r = compute_c5(res.x)
        if c5_r < compute_c5(h):
            return make_feasible(res.x, n), c5_r
    except Exception:
        pass
    return h, compute_c5(h)

def sharpen(h, n, thr=0.25):
    h2 = h.copy()
    h2[h2 < thr] = 0.0
    h2[h2 > 1.0 - thr] = 1.0
```
