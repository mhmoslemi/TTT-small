def optimize_lbgfs(h_start, n, T, mu, max_iter, time_limit):
    dx = 2.0 / n
    
    def fun_and_grad(x):
        return softmax_obj_and_grad(x, n, T, mu)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun_and_grad, h_start, jac=True, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 50})
    return make_feasible(res.x, n)

def efficient_binary_search(h_bin, n, rng, max_rounds, time_limit):
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current = float(np.max(corr) * (2.0/n))
    dx = 2.0 / n
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    n_ones = len(ones)
    
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        n_tries = min(5000, max(100, len(ones) * len(zeros) // 10))
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            # Compute delta in corr when flipping oi: 1->0, oj: 0->1
            # dc[k]/dh[oi] for the flip: h[oi] changes by -1
            # dc[k]/dh[oj] for the flip: h[oj] changes by +1
            # Total delta: -dc/k/dh[oi] + dc[k]/dh[oj] (each is a vector)
            # dc[k]/dh[i] = (1-h[k-i+n-1]) - h[i+k-n+1] (with zero ext)
            
            # For position oi (changing by -1):
            # delta_oi[k] = -(1-h[oi_k]) + h[oi_k2] where oi_k = k-oi+n-1, oi_k2 = oi+k-n+1
            # For position oj (changing by +1):
            # delta_oj[k] = (1-h[oj_k]) - h[oj_k2] where oj_k = k-oj+n-1, oj_k2 = oj+k-n+1
            
            # This is O(n) per swap. For 5000 swaps * n=640: 3.2M ops. Fine.
            
            # Actually let's just do it simply:
            h[oi] = 0.0
            h[oj] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(new_corr)) * dx
            if new_max < current:
                current = new_max
                corr = new_corr
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def simulated_annealing(h_bin, n, rng, time_limit, T_start=1.0, T_end=0.01, max_rounds=2000):
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current = float(np.max(corr) * (2.0/n))
    dx = 2.0 / n
    best_c = current
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    deadline = time_limit
    total_rounds = max_rounds
    for rnd in range(total_rounds):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (rnd / total_rounds)
        n_tries = 100
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_c = float(np.max(new_corr)) * dx
            delta = new_c - current
            if delta < 0 or rng.rand() < np.exp(-delta / (T * dx)):
                current = new_c
                corr = new_corr
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                if current < best_c:
                    best_c = current
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oj] = 0.0
    return best_h, best_c
