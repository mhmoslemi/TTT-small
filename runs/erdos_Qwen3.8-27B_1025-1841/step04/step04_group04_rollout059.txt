    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    w = exp_s / exp_s.sum()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max_val(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def compute_weight(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    return exp_s / exp_s.sum()

def lbg_obj(x, n, T, lam):
    val = smooth_max_val(x, n, T)
    if lam > 0:
        val += lam * (np.sum(x) - n / 2.0) ** 2
    return val

def lbg_grad(x, n, T, lam):
    dx = 2.0 / n
    w = compute_weight(x, n, T)
    term1 = np.correlate(1.0 - x, w, mode='full')[n - 1: 2 * n - 1]
    term2 = np.convolve(w, x, mode='full')[n - 1: 2 * n - 1]
    g = (term1 - term2) * dx
    if lam > 0:
        g += 2.0 * lam * (np.sum(x) - n / 2.0)
    return g

def slsqp_obj(x, n, T):
    return smooth_max_val(x, n, T)

def slsqp_grad(x, n, T):
    dx = 2.0 / n
    w = compute_weight(x, n, T)
    term1 = np.correlate(1.0 - x, w, mode='full')[n - 1: 2 * n - 1]
    term2 = np.convolve(w, x, mode='full')[n - 1: 2 * n - 1]
    return (term1 - term2) * dx

def slsqp_con(x, n):
    return np.sum(x) - n / 2.0

def slsqp_con_jac(x, n):
    return np.ones(n)

def binary_nudge(h, n, strength=0.3):
    h2 = h.copy()
    mask_high = h2 > 0.5
    mask_low = h2 < 0.5
    h2[mask_high] += strength * (1.0 - h2[mask_high])
    h2[mask_low] -= strength * h2[mask_low]
    return make_feasible(h2, n)
