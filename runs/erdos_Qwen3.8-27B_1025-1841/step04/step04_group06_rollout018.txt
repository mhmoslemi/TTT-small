import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def binarize(h, n, rng):
    """Convert to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    n_target = n // 2
    if n_ones > n_target:
        one_idx = np.where(h_bin == 1)[0]
        rng.shuffle(one_idx)
        for k in range(n_ones - n_target):
            h_bin[one_idx[k]] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(h_bin == 0)[0]
        rng.shuffle(zero_idx)
        for k in range(n_target - n_ones):
            h_bin[zero_idx[k]] = 1.0
    return h_bin

def binary_sa(h_bin, n, rng, time_limit, T_start=0.5, T_end=0.01):
    """Simulated annealing on binary configurations."""
    dx = 2.0 / n
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full').astype(float)
    current_max = float(np.max(corr))
    best_max = current_max
    best_corr = corr.copy()
    best_h = h_bin.copy()
    
    ones = np.where(h_bin == 1)[0].copy()
    zeros = np.where(h_bin == 0)[0].copy()
    n_ones = len(ones)
    n_zeros = len(zeros)
    
    start_t = time.time()
    total_iters = 0
    n_lags = 2 * n - 1
    
    # Precompute lag array
    lags = np.arange(-(n-1), n)
    
    it = 0
    while time.time() - start_t < time_limit:
        it += 1
        # Temperature
        frac = min(1.0, (time.time() - start_t) / time_limit)
        T = T_start * (T_end / T_start) ** frac
        
        # Pick random swap
        ii = rng.randint(n_ones)
        jj = rng.randint(n_zeros)
        p = ones[ii]  # position with 1, will become 0
        q = zeros[jj]  # position with 0, will become 1
        
        # Compute delta
        # Δc[lag] = -(1-h[p-lag]) + h[lag+p] + (1-h[q-lag]) - h[lag+q]
        # (only for valid indices)
        delta = np.zeros(n_lags)
        
        # Term 1: -(1-h[p-lag]) for valid p-lag
        idx1 = p - lags
        m1 = (idx1 >= 0) & (idx1 < n)
        delta[m1] -= (1.0 - h_bin[idx1[m1]])
        
        # Term 2: +h[lag+p] for valid lag+p
        idx2 = lags + p
        m2 = (idx2 >= 0) & (idx2 < n)
        delta[m2] += h_bin[idx2[m2]]
        
        # Term 3: +(1-h[q-lag]) for valid q-lag
        idx3 = q - lags
        m3 = (idx3 >= 0) & (idx3 < n)
        delta[m3] += (1.0 - h_bin[idx3[m3]])
        
        # Term 4: -h[lag+q] for valid lag+q
        idx4 = lags + q
        m4 = (idx4 >= 0) & (idx4 < n)
        delta[m4] -= h_bin[idx4[m4]]
        
        new_corr = corr + delta
        new_max = float(np.max(new_corr))
        
        # Accept/reject
        if new_max <= current_max or rng.random() < np.exp(-(new_max - current_max) / T):
            # Accept
            h_bin[p] = 0.0
            h_bin[q] = 1.0
            corr = new_corr
            current_max = new_max
            
            # Update ones/zeros arrays
            ones[ii] = q
            zeros[jj] = p
            
            if current_max < best_max:
                best_max = current_max
                best_corr = corr.copy()
                best_h = h_bin.copy()
        else:
            pass  # reject, no change
        
        # Periodic checkpoint
        if it % 10000 == 0:
            # Verify
            pass
    
    return best_h, best_max * dx
