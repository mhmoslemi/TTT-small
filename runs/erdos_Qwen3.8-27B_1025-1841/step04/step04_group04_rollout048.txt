import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 960
_OPT_T = 200.0

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)

    n_ones = int(round(np.sum(h)))
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n_ones]] = 1.0
    return b

def binary_swap_search(h_bin, n, n_trials, rng, time_fn):
    """Try random swap moves on a binary solution."""
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    corr = np.correlate(best, 1.0 - best, mode='full')
    best_corr_max = np.max(corr)
    
    ones = np.where(best == 1)[0]
    zeros = np.where(best == 0)[0]
    
    stalls = 0
    trials_per_iter = max(50, n_trials // 10)
    
    for it in range(n_trials):
        if time_fn() < 5:
            break
        found = False
        for _ in range(trials_per_iter):
            if time_fn() < 3:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            trial = best.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            c = compute_c5(trial)
            if c < best_c5 - 1e-14:
                best = trial
                best_c5 = c
                ones = np.where(best == 1)[0]
                zeros = np.where(best == 0)[0]
                found = True
                break
        if not found:
            stalls += 1
        else:
            stalls = 0
        if stalls >= 3:
            break
    return best, best_c5
