from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = np.max(C)
    exp_C = np.exp(T * (C - max_C))
    w = exp_C / np.sum(exp_C)

    obj = (max_C + np.log(np.sum(exp_C)) / T) * dx
    
    # Gradient of smoothed max

    one_minus_h = 1.0 - h
    
    S1_full = np.correlate(W, one_minus_h, mode='full')
    S2_full = np.correlate(W[::-1], h, mode='full')
    
    S1 = S1_full[n-1::-1][:n]
    S2 = S2_full[n-1::-1][:n]

    sum_diff = np.sum(h) - n / 2.0
    grad += 2.0 * sum_penalty * sum_diff * np.ones(n)
    obj += sum_penalty * sum_diff ** 2
    
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)

def chase_worst(h, n, n_steps=50, step_size=0.01):

    h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')

    return make_feasible(res.x, n)

    dx = 2.0 / n
    step = base_step
    for _ in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        C = corr * dx
        max_C = np.max(C)
        # Active set: shifts within 0.001 of max
        active = np.where(C > max_C - 0.001)[0]
        # Average gradient over active shifts
        grad = np.zeros(n)
        for k in active:
            s = k - (n - 1)
            lo1 = max(0, -s)
            hi1 = min(n - 1, n - 1 - s)
            if lo1 <= hi1:

            hi2 = min(n - 1, n - 1 + s)
