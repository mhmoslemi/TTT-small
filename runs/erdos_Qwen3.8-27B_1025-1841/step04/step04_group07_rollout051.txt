def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):

    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(80):

        if np.clip(h + mid, 0, 1).sum() > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.any():

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def obj_and_grad(x, T, dx, mu):

    corr = compute_corr(x) * dx

    shifted = np.clip(T * (corr - mc), -50, 50)

    pesos = exps / exps.sum()
    val = mc + np.log(exps.sum()) / T
    val += mu * (np.sum(x) - n / 2.0) ** 2
    # Gradient

def greedy_polish(h0, n, max_rounds=50, step=0.01):
    """Coordinate descent with the actual max gradient"""
    h = h0.copy()
    cur = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        corr = compute_corr(h) * dx
        max_val = np.max(corr)
        # Find active lags (within 1% of max)
        active = corr > max_val * 0.99
        # Compute gradient from active lags
        g = np.zeros(n)
        shifts_arr = np.arange(2*n - 1) - (n - 1)
        one_minus = 1.0 - h
        for idx in np.where(active)[0]:
            s = shifts_arr[idx]
            wi = 1.0 / active.sum()  # equal weight
            lo1 = max(0, -s)
            hi1 = min(n, n - s) if s > 0 else n
            if hi1 > lo1:
                g[lo1:hi1] += wi * one_minus[lo1 + s: hi1 + s]
            lo2 = max(0, s)
            hi2 = min(n, n + s) if s < 0 else n
            if hi2 > lo2:
                g[lo2:hi2] -= wi * h[lo2 - s: hi2 - s]
        # Try to reduce h where g > 0, increase where g < 0
        # Subject to 0 <= h <= 1 and sum = n/2
        improved = False
        # Sort by gradient magnitude
        order = np.argsort(-np.abs(g))[::1]  # largest |g| first
        for j in order:
            if g[j] > 0:
                delta = -step
            elif g[j] < 0:
                delta = step
            else:
                continue
            if h[j] + delta < 0 or h[j] + delta > 1:
                continue
            h[j] += delta
            h = make_feasible(h, n)
            c = compute_c5(h)
            if c < cur:
                cur = c
                improved = True
            else:
                h[j] -= delta
                h = make_feasible(h, n)
        if not improved:
            step *= 0.7
            if step < 0.001:
                break
    return h, cur

```python
def projected_grad_descent(h0, n, n_iters=500, lr=None, topk=10):
    h = h0.copy()
    dx = 2.0 / n
    if lr is None:
        lr = 0.1 / n
    cur = compute_c5(h)
    for it in range(n_iters):
        corr = compute_corr(h) * dx
        # Top-k weights
        k = min(topk, len(corr))
        top_indices = np.argpartition(corr, -k)[-k:]
        w = np.zeros(len(corr))
        w[top_indices] = 1.0 / k
        # Compute gradient
        g = np.zeros(n)
        shifts_arr = np.arange(2*n-1) - (n-1)
        one_minus = 1.0 - h
        for idx in top_indices:
            s = shifts_arr[idx]
            lo1 = max(0, -s); hi1 = min(n, n-s) if s > 0 else n
            if hi1 > lo1:
                g[lo1:hi1] += (1.0/k) * one_minus[lo1+s:hi1+s]
            lo2 = max(0, s); hi2 = min(n, n+s) if s < 0 else n
            if hi2 > lo2:
                g[lo2:hi2] -= (1.0/k) * h[lo2-s:hi2-s]
        # Gradient step
        h_new = h - lr * g
        h_new = make_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < cur:
            h = h_new
            cur = c
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    return h, cur
```
