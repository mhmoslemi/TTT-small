    order = np.argsort(-h)
    h_bin = np.zeros(n)

    curr_corr = get_correlation(curr_h)
    curr_max = best_max_corr

    zeros_idx = np.where(curr_h == 0)[0]
    
    n_swaps_est = max(100000, int(max_time * 50000))  # rough estimate

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

        return np.zeros(n)

        return np.sum(x) - n / 2.0

        return np.ones(n)
