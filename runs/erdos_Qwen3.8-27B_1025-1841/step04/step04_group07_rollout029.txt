        hs2, cs2 = sim_anneal(hb, n0, rng, n_iter=25000, T0=0.0008, T1=5e-8)
        if cs2 < best_c5:
            best_c5, best_h = cs2, hs2.copy()
            hb = hs2.copy()

    # Phase 5: Higher resolutions
    for n2 in [800, 960, 1200, 1600, 2000]:
        if rem() < 100:
            break
        h2 = upsample(best_h, n2)
        hb2 = to_binary(h2, n2)
        c2 = compute_c5(hb2)
        if c2 < best_c5:
            best_c5, best_h = c2, hb2.copy()
        hg2, cg2 = greedy_search(hb2, n2, rng, max_rounds=30, tries=2000)
        if cg2 < best_c5:
            best_c5, best_h = cg2, hg2.copy()
            hb2 = hg2.copy()
        if rem() > 70:
            hs3, cs3 = sim_anneal(hb2, n2, rng, n_iter=15000, T0=0.001, T1=1e-7)
            if cs3 < best_c5:
                best_c5, best_h = cs3, hs3.copy()

    # Phase 6: Active-lag min-max continuous
    if rem() > 60:
        for n3 in [n0, 800]:
            if rem() < 30:
                break
            h3 = upsample(best_h, n3)
            h_am = active_lag_minmax(h3, n3, n_iters=35, kfrac=0.12, max_inner=30)
            c_am = compute_c5(h_am)
            if c_am < best_c5:
                best_c5, best_h = c_am, h_am.copy()
                h3 = h_am.copy()
            for T in [300, 800, 2000]:
                if rem() < 20:
                    break
                ho = lbgfb_soft(h3, n3, T, mu=500.0, max_iter=120)
                co = compute_c5(ho)
                if co < best_c5:
                    best_c5, best_h = co, ho.copy()
                    h3 = ho.copy()

    # Phase 7: Final binary polish
    if rem() > 20:
        nf = len(best_h)
        hf = to_binary(best_h, nf)
        hp, cp = greedy_search(hf, nf, rng, max_rounds=25, tries=2500)
        if cp < best_c5:
            best_c5, best_h = cp, hp.copy()
        if rem() > 15:
            hs4, cs4 = sim_anneal(hp, nf, rng, n_iter=10000, T0=0.0005, T1=1e-8)
            if cs4 < best_c5:
                best_c5, best_h = cs4, hs4.copy()

    best_h = make_feasible(best_h, len(best_h))
    fc = compute_c5(best_h)
    if fc < best_c5:
        best_c5 = fc

    return best_h, best_c5, len(best_h)
