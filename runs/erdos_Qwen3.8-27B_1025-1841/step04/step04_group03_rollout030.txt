    except Exception:
        h_init = np.full(960, 0.5)
    
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    temps = [50.0, 200.0, 1000.0, 5000.0, 20000.0, 80000.0]
    
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 60:
            break
        maxiter = min(100, int(max(20, remaining / 3.0)))
        x_res = slsqp_lse(best_h, T, maxiter, 1e-14)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = make_feasible(x_res, n0)
        
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            h_pg, c5_pg = projected_gradient(best_h, T, start_time, 
                                              min(budget_s - 30, remaining - 20),
                                              n_steps=5000, lr_init=n0 * 0.3)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
    
    # Perturbation-based restarts
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        for ptype in range(4):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 80:
                break
            if ptype == 0:
                hp = fourier_perturb(best_h, n0, rng, n_modes=3, amp=0.02)
            elif ptype == 1:
                hp = segment_flip(best_h, n0, rng)
            else:
                hp = fourier_perturb(best_h, n0, rng, n_modes=5, amp=0.05)
            
            for T in [500.0, 5000.0]:
                remaining = budget_s - (time.time() - start_time)
                if remaining < 40:
                    break
                x_res = slsqp_lse(hp, T, 80, 1e-13)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n0)
                    break
    
    # Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        n1 = n0 * 2
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        
        for T in [200.0, 2000.0, 10000.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 50:
                break
            x_res = slsqp_lse(h_up, T, 60, 1e-13)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n1)
        
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            h_pg, c5_pg = projected_gradient(best_h, 5000.0, start_time,
                                              min(budget_s - 30, remaining - 20),
                                              n_steps=3000, lr_init=len(best_h) * 0.3)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
    
    # Coordinate descent polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        max_rounds = min(20, max(3, int(remaining / 15)))
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n_pts, start_time, budget_s, max_rounds)
    
    # Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        x_res = slsqp_lse(best_h, 50000.0, min(80, int(remaining / 2.5)), 1e-15)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = make_feasible(x_res, n_pts)
    
    # Final coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n_pts, start_time, budget_s, 5)
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
