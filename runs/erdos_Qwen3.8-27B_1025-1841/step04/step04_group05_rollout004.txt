    if T > 1e8:
        return mc * dx
    sh = T * (c - mc)
    sh -= np.max(sh)
    lse = np.log(np.sum(np.exp(sh)))
    return (mc + lse / T) * dx

def _sm_grad(h, n, T):
    c = compute_corr(h)
    mc = np.max(c)
    dx = 2.0 / n
    sh = T * (c - mc)
    sh -= np.max(sh)
    z = np.exp(sh)
    z /= z.sum()
    grad = np.zeros(n)
    thresh = np.max(z) * 1e-8
    active = np.where(z > thresh)[0]
    for ki in active:
        w = z[ki]
        lag = ki - n + 1
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += w * (1.0 - h[lag:])
                grad[lag:] -= w * h[:m]
        else:
            kl = -lag
            m = n - kl
            if m > 0:
                grad[kl:] += w * (1.0 - h[:m])
                grad[:m] -= w * h[kl:]
    grad *= dx
    return grad

def _slbfgs(h0, n, T, max_iter=300):
    def obj(x):
        return _sm_obj(x, n, T)
    def grad(x):
        return _sm_grad(x, n, T)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj, h0, jac=grad, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': 1e-10, 'factr': 1e6})
    return make_feasible(res.x, n)

def _subgrad_step(h, n, n_lags=5, step_size=None):
    c = compute_corr(h)
    mc = np.max(c)
    dx = 2.0 / n
    top = np.argsort(c)[-n_lags:][::-1]
    grad = np.zeros(n)
    for ki in top:
        lag = ki - n + 1
        if lag >= 0:
            m = n - lag
            if m > 0:
                grad[:m] += (1.0 - h[lag:])
                grad[lag:] -= h[:m]
        else:
            kl = -lag
            m = n - kl
            if m > 0:
                grad[kl:] += (1.0 - h[:m])
                grad[:m] -= h[kl:]
    grad /= n_lags
    grad *= dx
    if step_size is None:
        gnorm = np.linalg.norm(grad)
        step_size = 0.5 / (gnorm + 1e-12)
    h_new = h - step_size * grad
    return make_feasible(h_new, n), mc * dx

def binary_project(h, n):
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n % 2 == 1:
        target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = np.argsort(h[one_idx])
        rem = one_idx[order[:n_ones - target]]
        h_bin[rem] = 0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        add = zero_idx[order[:target - n_ones]]
        h_bin[add] = 1
    return h_bin

def binary_local_search(h_bin, n, time_limit, start_time):
    c5 = compute_c5(h_bin)
    c = compute_corr(h_bin)
    max_c = np.max(c)
    ones = set(np.where(h_bin == 1)[0].tolist())
    zeros = set(np.where(h_bin == 0)[0].tolist())
    
    for it in range(20000):
        if time.time() - start_time > time_limit:
            break
        worst_lag_idx = np.argmax(c)
        worst_lag = worst_lag_idx - n + 1
        
        # Find bad pairs at worst lag
        if worst_lag >= 0:
            m = n - worst_lag
            bad_positions = [p for p in range(m) if h_bin[p] == 1 and h_bin[p + worst_lag] == 0]
        else:
            kl = -worst_lag
            bad_positions = [p for p in range(kl, n) if h_bin[p] == 1 and h_bin[p - kl] == 0]
        
        if not bad_positions:
            # Try random perturbation
            if len(ones) > 0 and len(zeros) > 0:
                p = list(ones)[np.random.randint(len(ones))]
                q = list(zeros)[np.random.randint(len(zeros))]
                h_bin[p], h_bin[q] = 0.0, 1.0
                new_c5 = compute_c5(h_bin)
                if new_c5 < c5:
                    c5 = new_c5
                    c = compute_corr(h_bin)
                    ones = set(np.where(h_bin == 1)[0].tolist())
                    zeros = set(np.where(h_bin == 0)[0].tolist())
                else:
                    h_bin[p], h_bin[q] = 1.0, 0.0
            break
        
        improved = False
        # Try fixing worst bad pairs by swapping
        for p in bad_positions[:30]:
            if time.time() - start_time > time_limit:
                break
            # Try swapping h[p] with a random zero
            z_list = list(zeros)
            np.random.shuffle(z_list)
            for q in z_list[:20]:
                if q == p:
                    continue
                h_bin[p] = 0.0
                h_bin[q] = 1.0
                new_c5 = compute_c5(h_bin)
                if new_c5 < c5 - 1e-14:
                    c5 = new_c5
                    c = compute_corr(h_bin)
                    ones.discard(p)
                    ones.add(q)
                    zeros.add(p)
                    zeros.discard(q)
                    improved = True
                    break
                else:
                    h_bin[p] = 1.0
                    h_bin[q] = 0.0
            if improved:
                break
        if not improved:
            break
    return h_bin, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Load parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)
    
    n0 = len(h0)
    best_h = make_feasible(h0, n0)
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Optimize at parent resolution with L-BFGS-B
    if time_left() > 60:
        for T in [15.0, 8.0, 3.0]:
            if time_left() < 30:
                break
            h_try = _slbfgs(best_h, n0, T, max_iter=200)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
    
    # Phase 2: Go to 2560 resolution
    n2 = 2560
    if time_left() > 200:
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
        
        # L-BFGS-B with varying T
        for T in [30.0, 10.0, 4.0, 1.5]:
            if time_left() < 60:
                break
            h_try = _slbfgs(h2, n2, T, max_iter=250)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
                h2 = h_try.copy()
            else:
                h2 = h_try.copy()
        
        # Subgradient polish
        if time_left() > 80:
            h_sg = h2.copy()
            c_sg = compute_c5(h_sg)
            step_mult = 1.0
            for it in range(500):
                if time_left() < 10:
                    break
                h_new, _ = _subgrad_step(h_sg, n2, n_lags=5)
                c_new = compute_c5(h_new)
                if c_new < c_sg - 1e-12:
                    h_sg = h_new
                    c_sg = c_new
                    step_mult = min(step_mult * 1.1, 5.0)
                else:
                    step_mult = max(step_mult * 0.5, 0.1)
            if c_sg < best_c5:
                best_c5 = c_sg
                best_h = h_sg.copy()
        
        # Binary projection + local search
        if time_left() > 100:
            h_bin = binary_project(best_h, n2)
            h_bin, c_bin = binary_local_search(h_bin, n2, time_limit=time_left() - 50,
                                               start_time=start_time)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()
    
    # Phase 3: Go to 4096 resolution
    n3 = 4096
    if time_left() > 250:
        h3 = upsample_h(best_h, n3)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
        
        for T in [25.0, 8.0, 3.0]:
            if time_left() < 60:
                break
            h_try = _slbfgs(h3, n3, T, max_iter=200)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try.copy()
                h3 = h_try.copy()
            else:
                h3 = h_try.copy()
        
        if time_left() > 80:
            h_sg = h3.copy()
            c_sg = compute_c5(h_sg)
            step_mult = 1.0
            for it in range(300):
                if time_left() < 10:
                    break
                h_new, _ = _subgrad_step(h_sg, n3, n_lags=5)
                c_new = compute_c5(h_new)
                if c_new < c_sg - 1e-12:
                    h_sg = h_new
                    c_sg = c_new
                    step_mult = min(step_mult * 1.1, 5.0)
                else:
                    step_mult = max(step_mult * 0.5, 0.1)
            if c_sg < best_c5:
                best_c5 = c_sg
                best_h = h_sg.copy()
    
    # Phase 4: Targeted coordinate descent at best resolution
    n_cd = len(best_h)
    if time_left() > 60:
        h_cd = best_h.copy()
        c_cd = compute_c5(h_cd)
        c_corr = compute_corr(h_cd)
        dx = 2.0 / n_cd
        # Focus on variables that affect the worst lags
        top_lags = np.argsort(c_corr)[-10:][::-1]
        affected = set()
        for ki in top_lags:
            lag = ki - n_cd + 1
            if lag >= 0:
                affected.update(range(n_cd - lag))
                affected.update(range(lag, n_cd))
            else:
                kl = -lag
                affected.update(range(n_cd - kl))
                affected.update(range(kl, n_cd))
        
        for cd_round in range(5):
            if time_left() < 15:
                break
            improved = False
            for j in list(affected)[:min(len(affected), 200)]:
                if time_left() < 15:
                    break
                for delta in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02, 0.2, -0.2]:
                    trial = h_cd.copy()
                    trial[j] = np.clip(trial[j] + delta, 0.0, 1.0)
                    trial = make_feasible(trial, n_cd)
                    c = compute_c5(trial)
                    if c < c_cd - 1e-13:
                        c_cd = c
                        h_cd = trial
                        improved = True
                        break
                # Try setting to 0 or 1
                for bv in [0.0, 1.0]:
                    if abs(h_cd[j] - bv) < 1e-8:
                        continue
                    trial = h_cd.copy()
                    trial[j] = bv
                    trial = make_feasible(trial, n_cd)
                    c = compute_c5(trial)
                    if c < c_cd - 1e-13:
                        c_cd = c
                        h_cd = trial
                        improved = True
                        break
            if not improved:
                break
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()
    
    # Phase 5: Final L-BFGS-B polish
    if time_left() > 30:
        n_f = len(best_h)
        try:
            h_f = _slbfgs(best_h, n_f, T=1.0, max_iter=100)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = h_f.copy()
        except Exception:
            pass
    
    # Phase 6: Small random perturbations
    if time_left() > 20:
        n_p = len(best_h)
        for restart in range(5):
            if time_left() < 5:
                break
            h_p = best_h + 0.05 * rng.randn(n_p)
            h_p = make_feasible(h_p, n_p)
            try:
                h_opt = _slbfgs(h_p, n_p, T=5.0, max_iter=80)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5 = c_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
