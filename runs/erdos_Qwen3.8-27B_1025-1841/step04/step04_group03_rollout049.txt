def coord_descent_fast(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        improved = False
        for j in range(0, n, 2):  # Every other point for speed
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                # Quick feasibility: adjust neighbor
                diff = (n/2.0) - trial.sum()
                if abs(diff) > 1e-10:
                    # Adjust a nearby point
                    j2 = (j+1) % n
                    trial[j2] += diff
                    if trial[j2] < 0 or trial[j2] > 1:
                        continue
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                # Fix sum
                diff = (n/2.0) - h.sum()
                if abs(diff) > 1e-10:
                    h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_optimize(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con_sum(x):
        return np.sum(x) - n / 2.0
    def con_sum_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def bottleneck_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time, top_m=5):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    recompute_interval = 40
    last_recompute = -recompute_interval
    bottleneck_shifts = np.array([0, n//4, n//2, 3*n//4])
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = min(float(it) / max_iters, 1.0)
        T = T0 * (T1 / T0) ** frac if T0 > T1 else T0
        if it - last_recompute >= recompute_interval:
            corr = np.correlate(b, 1.0 - b, mode='full')
            top_idx = np.argsort(corr)[-top_m:]
            bottleneck_shifts = ((top_idx - (n - 1)) % n).astype(int)
            last_recompute = it
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if rng.random() < 0.45:
            s = bottleneck_shifts[rng.randint(len(bottleneck_shifts))]
            i = ones[rng.randint(len(ones))]
            j_target = (i + s) % n
            dists = np.abs(zeros - j_target)
            dists = np.minimum(dists, n - dists)
            j = zeros[np.argmin(dists)]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        if i == j:

        b[j] = 1

                best_c5 = new_c5
                best_b = b.copy()

    return best_b, best_c5

def subgradient_step(h, n, T, step_size, max_iter_grad, top_k=15):

        top_corr = corr[top_idx]
        mc = np.max(top_corr)
        shifted = top_corr - mc
        weights = np.exp(T * shifted)
        weights /= np.sum(weights)

def fourier_perturb(h, n, rng, n_modes=4, amp=0.04):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 8)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def segment_flip(h, n, rng):
    h_new = h.copy()
    n_segs = rng.randint(1, 3)
    for _ in range(n_segs):
        length = rng.randint(n // 15, n // 5)
        start = rng.randint(0, max(1, n - length))
        h_new[start:start+length] = 1.0 - h_new[start:start+length]
    return make_feasible(h_new, n)

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        improved = False
        step = max(1, n // 200)
        for j in range(0, n, step):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                diff = (n / 2.0) - trial.sum()
                if abs(diff) > 1e-12:
                    j2 = (j + 1) % n
                    trial[j2] += diff
                    if trial[j2] < 0 or trial[j2] > 1:
                        trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                if abs(h.sum() - n/2.0) > 1e-10:
                    h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

                        best_h = make_feasible(x_res, n0)
                except Exception:
                    pass

    # Phase 3: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 30, 120)
        b, c5_b = block_sa(best_h, n0, 150000, 0.002, 0.000005, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 40:
                try:
                    x_res = slsqp_optimize(best_h, n0, 3000.0, 80, 1e-14)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(x_res, n0)
                except Exception:
                    pass

    # Phase 4: Subgradient
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        hl, cl = subgradient_step(best_h, n0, T=500.0, step_size=0.003, max_iter_grad=300, top_k=20)
        if cl < best_c5:
            best_c5 = cl
            best_h = hl.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            hl, cl = subgradient_step(best_h, n0, T=2000.0, step_size=0.001, max_iter_grad=200, top_k=15)
            if cl < best_c5:
                best_c5 = cl
                best_h = hl.copy()

    # Phase 5: Multi-temp SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        for T in [100.0, 500.0, 2000.0, 10000.0, 50000.0]:
            if time.time() - start_time > budget_s - 20:
                break
            try:
                x_res = slsqp_optimize(best_h, n0, T, 100, 1e-15)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n0)
            except Exception:
                pass

    # Phase 6: Perturbations + re-optimize
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        for pf, amp in [(fourier_perturb, 0.03), (segment_flip, 0.0), (fourier_perturb, 0.02)]:
            if time.time() - start_time > budget_s - 20:
                break
            try:
                if pf == fourier_perturb:
                    hp = fourier_perturb(best_h, n0, rng, n_modes=3, amp=amp)
                else:
                    hp = segment_flip(best_h, n0, rng)
                x_res = slsqp_optimize(hp, n0, 5000.0, 80, 1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n0)
            except Exception:
                pass

    # Phase 7: Higher resolution (1200)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n1 = 1200
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        for T in [1000.0, 10000.0, 50000.0]:
            if time.time() - start_time > budget_s - 30:
                break
            try:
                x_res = slsqp_optimize(h_up, n1, T, 80, 1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n1)
                    h_up = best_h.copy()
            except Exception:
                pass
        remaining = budget_s - (time.time() - start_time)
        if remaining > 80:
            tl = min(remaining - 20, 60)
            b, c5_b = bottleneck_sa(best_h, n1, 150000, 0.002, 0.000005, rng, tl, start_time, top_m=5)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = b.copy()
                remaining = budget_s - (time.time() - start_time)
                if remaining > 40:
                    try:
                        x_res = slsqp_optimize(best_h, n1, 10000.0, 60, 1e-14)
                        c5_res = compute_c5(x_res)
                        if c5_res < best_c5:
                            best_c5 = c5_res
                            best_h = make_feasible(x_res, n1)
                    except Exception:
                        pass

    # Phase 8: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 25:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 15, start_time, budget_s)

    # Phase 9: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        try:
            x_res = slsqp_optimize(best_h, n_pts, 100000.0, 60, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
