import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_fft(h, n):
    """Compute full cross-correlation of h and (1-h) using FFT."""
    g = 1.0 - h
    size = 2 * n - 1
    # Use next power of 2 for FFT efficiency
    fsize = 1
    while fsize < size:
        fsize *= 2
    H = np.fft.rfft(h, fsize)
    G = np.fft.rfft(g, fsize)
    corr = np.fft.irfft(H * np.conj(G), fsize)[:size]
    return corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Water-filling style projection
    for _ in range(50):
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
    # Exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_gradient(h, T, n):

    corr = compute_corr_fft(h, n)
    # Stable softmax

    p = exp_c / np.sum(exp_c)

    B_full = np.fft.irfft(np.fft.rfft(p[::-1], 4*n) * np.fft.rfft(h, 4*n), 4*n)
    
    offset = n - 1
    A = A_full[offset:offset+n]
    B = B_full[offset:offset+n]
    
    grad = 1.0 - A - B

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fft_corr(h, fsize):
    """Full cross-correlation of h and (1-h) via FFT."""
    g = 1.0 - h
    H = np.fft.rfft(h, fsize)
    G = np.fft.rfft(g, fsize)
    return np.fft.irfft(H * np.conj(G), fsize)

def fft_grad(h, T, fsize, n, conv_fsize):
    """Gradient of smooth max via FFT."""
    corr = fft_corr(h, fsize)[:2*n-1]
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    p = exp_c / np.sum(exp_c)
    
    # A[x] = convolve(p, h)[n-1+x]
    A_full = np.fft.irfft(np.fft.rfft(p, conv_fsize) * np.fft.rfft(h, conv_fsize), conv_fsize)
    # B[x] = convolve(h, p_rev)[n-1+x]
    p_rev = p[::-1].copy()
    B_full = np.fft.irfft(np.fft.rfft(h, conv_fsize) * np.fft.rfft(p_rev, conv_fsize), conv_fsize)
    
    offset = n - 1
    A = A_full[offset:offset+n]
    B = B_full[offset:offset+n]
    
    return 1.0 - A - B

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(30):
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def fft_gd(h, n, budget_s, start_time, T_start=50.0, T_end=3000.0, iters_per_phase=None):
    """FFT-accelerated smooth max gradient descent."""
    fsize = 1
    while fsize < 2*n-1:
        fsize *= 2
    conv_fsize = 1
    while conv_fsize < 3*n-2:
        conv_fsize *= 2
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    mu = 0.9  # momentum
    alpha = 1.0  # step size
    it = 0
    no_improve = 0
    total_budget = budget_s
    
    while time.time() - start_time < total_budget - 3:

        y = np.clip(y, 0.0, 1.0)
        y = project_feasible(y, n)

        h_new = h + velocity
        h_new = project_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0

            no_improve += 1

            alpha *= 0.75
            velocity *= 0.3
            no_improve = 0
            if alpha < 1e-8:
                break
        elif no_improve == 0 and it % 100 == 0 and alpha < 2.0:
            alpha *= 1.05
    
    return best_h, best_c5

    vel = np.zeros(n)
    alpha = alpha_init
    it = 0
    no_imp = 0

        T = T_lo + (T_hi - T_lo) * min(1.0, it / 30000.0)

        ec = np.exp(T * (corr - mc))
        p = ec / ec.sum()

        Hh = np.fft.rfft(h_arr, csize)
        A = np.fft.irfft(Hp * Hh, csize)[n-1:n-1+n]
        B = np.fft.irfft(Hh * np.fft.rfft(p[::-1], csize), csize)[n-1:n-1+n]

        h_new = project_feasible(h_arr + vel, n)
        
        # Evaluate

            best_h = h_new.copy()
            no_imp = 0
        else:

            no_imp = 0
