def binary_flip_sa(h, n, rng, start_time, time_limit, T0=0.004, T1=1e-6, max_iters=300000):
    """SA on binary flips with pair moves."""
    b = np.where(h >= 0.5, 1, 0).astype(np.int32)
    target = n // 2
    s = b.sum()
    while s > target:
        idx = rng.choice(np.where(b == 1)[0])
        b[idx] = 0; s -= 1
    while s < target:
        idx = rng.choice(np.where(b == 0)[0])
        b[idx] = 1; s += 1
    
    curr = compute_c5(b.astype(float))
    best = curr
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = min(1.0, it / max_iters)
        T = T0 * (T1 / T0) ** frac
        
        r = rng.random()
        if r < 0.7:
            # Single pair swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0; b[j] = 1
            new_c = compute_c5(b.astype(float))
            if new_c <= curr or rng.random() < np.exp(-(new_c - curr) / max(T, 1e-15)):
                curr = new_c
                if new_c < best:
                    best = new_c
                    best_b = b.copy()
            else:
                b[i] = 1; b[j] = 0
        else:
            # Multi-pair swap (2 or 3 pairs)
            k = rng.randint(1, 4)
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if len(ones) < k or len(zeros) < k:
                continue
            pick_o = rng.choice(ones, size=k, replace=False)
            pick_z = rng.choice(zeros, size=k, replace=False)
            old_o = b[pick_o].copy()
            old_z = b[pick_z].copy()
            b[pick_o] = 0
            b[pick_z] = 1
            new_c = compute_c5(b.astype(float))
            if new_c <= curr or rng.random() < np.exp(-(new_c - curr) / max(T, 1e-15)):
                curr = new_c
                if new_c < best:
                    best = new_c
                    best_b = b.copy()
            else:
                b[pick_o] = old_o
                b[pick_z] = old_z
    
    return best_b.astype(float), best

def exact_binary_improve(b, n, max_rounds=50):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_sl_n = 0
_sl_t = 1.0

def _obj(x):
    n = _sl_n
    T = _sl_t
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    sh = np.clip((corr - mc) * T, -60, 60)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def _eq(x):
    return np.sum(x) - _sl_n / 2.0

def _eq_jac(x):
    return np.ones(_sl_n)

def _slsqp(h, T, maxiter, ftol):
    global _sl_n, _sl_t
    _sl_n = len(h)
    _sl_t = T
    bounds = [(0.0, 1.0)] * _sl_n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eq_jac}]
    try:
        r = minimize(_obj, h.copy(), method='SLSQP',
                     bounds=bounds, constraints=cons,
                     options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(r.x, _sl_n)
    except Exception:
        return h.copy()

def _sa(h, n, rng, t_start, t_deadline, T0, T1):
    b = (h >= 0.5).astype(np.int8)
    tgt = n // 2
    while b.sum() > tgt:

    while b.sum() < tgt:
        b[rng.choice(np.where(b == 0)[0])] = 1
    cur = compute_c5(b.astype(float))
    best = cur
    bb = b.copy()
    while time.time() < t_deadline:
        frac = min(1.0, (time.time() - t_start) / max(1e-6, t_deadline - t_start))
        T = T0 * (T1 / T0) ** frac
        r = rng.random()

            zi = np.where(b == 0)[0]
            i = oi[rng.randint(len(oi))]
            j = zi[rng.randint(len(zi))]
            b[i] = 0; b[j] = 1
            nc = compute_c5(b.astype(float))
            if nc <= cur or rng.random() < np.exp(-(nc - cur) / max(T, 1e-15)):
                cur = nc
                if nc < best:

            else:

        elif r < 0.85:

def _greedy_binary(b, n, rng, n_passes=30):
    """Greedy: try each random flip, keep if improves."""
    cur = compute_c5(b.astype(float))
    for _ in range(n_passes):
        improved = False
        for _ in range(300):
            oi = np.where(b == 1)[0]
            zi = np.where(b == 0)[0]
            i = oi[rng.randint(len(oi))]
            j = zi[rng.randint(len(zi))]
            b[i] = 0; b[j] = 1
            nc = compute_c5(b.astype(float))
            if nc < cur - 1e-15:
                cur = nc; improved = True
            else:
                b[i] = 1; b[j] = 0
        if not improved:
            break
    return b.astype(float), cur

def _periodic(n, p, duty=0.5):
    h = np.zeros(n)
    pp = max(2, int(round(p * n / 2.0)))
    w = max(1, int(round(duty * pp)))
    for i in range(n):
        if i % pp < w:
            h[i] = 1.0
    return make_feasible(h, n)

def _coord(h, n, t_deadline, n_rounds=8):
    best = compute_c5(h)
    for _ in range(n_rounds):
        if time.time() > t_deadline:
            break
        imp = False
        for j in range(n):
            if time.time() > t_deadline:
                break
            old = h[j]
            for cand in [0.0, 1.0, max(0.0, old - 0.04), max(0.0, old - 0.01),
                         min(1.0, old + 0.01), min(1.0, old + 0.04)]:
                if abs(cand - old) < 1e-10:

                tr = h.copy()
                tr[j] = cand
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < best - 1e-13:
                    best = c; h = tr; imp = True; break
        if not imp:
            break
    return h, best
