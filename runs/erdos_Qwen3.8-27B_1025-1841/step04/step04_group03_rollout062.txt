def coord_descent(h, n, corr, max_sweeps=50, start_time=0, time_limit=60):
    """Coordinate descent with exact 1D min-max solve."""
    best_c5 = float(np.max(corr) * 2.0 / n)

        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            k_idx = np.arange(2 * n - 1)

            vm = (k_idx >= j) & (k_idx <= j + n - 1)
            # vp: h[j+k] valid, where j+k = j + k_idx - n + 1
            vp = (k_idx >= n - 1 - j) & (k_idx <= 2 * n - j - 2)
            
            # Compute indices into h
            idx_m = np.clip(j - k_idx + n - 1, 0, n - 1)
            idx_p = np.clip(j + k_idx - n + 1, 0, n - 1)
            
            hp_vals = np.where(vp, h[idx_p], 0.0)
            hm_vals = np.where(vm, h[idx_m], 0.0)
            
            B = vm.astype(float) * (1.0 - hm_vals) - vp.astype(float) * hp_vals
            A = corr - h[j] * B
            
            # Minimize max(A + B*v) for v in [0,1]
            # Use fine grid + refinement
            v_grid = np.linspace(0, 1, 41)
            f_vals = np.max(A[:, None] + B[:, None] * v_grid[None, :], axis=0)
            best_i = np.argmin(f_vals)
            v_best = v_grid[best_i]
            f_best = f_vals[best_i]
            
            # Check boundaries
            v_best = max(0.0, min(1.0, v_best))
            
            if abs(v_best - h[j]) > 1e-8:
                new_corr = A + B * v_best
                new_c5 = float(np.max(new_corr) * 2.0 / n)
                if new_c5 < best_c5 - 1e-14:
                    h[j] = v_best
                    corr = new_corr
                    best_c5 = new_c5
                    improved = True
        if not improved:
            break
    
    return h, corr, best_c5

Actually, for the 1D minimax, there's a more efficient algorithm:
1. The function f(v) = max_k (A_k + B_k v) is convex and piecewise linear.
2. Its minimum on [0,1] is either at v=0, v=1, or at a "vertex" where two lines cross.
3. The vertex of interest is where the upper envelope changes from decreasing to increasing.
4. I can find this by:
   a. Compute f(0) and the argmax k₀ = argmax A_k
   b. Compute f(1) and the argmax k₁ = argmax (A_k + B_k)
   c. If k₀ has B ≤ 0 and k₁ has B ≥ 0, the minimum is at their intersection (if in [0,1])
   d. Otherwise, the minimum is at v=0 or v=1

Let me implement this more efficient approach:

```python
def solve_1d_minmax(A, B):
    """Minimize max_i (A[i] + B[i]*v) over v in [0,1]."""
    # f(0) = max A
    f0 = np.max(A)
    idx0 = np.argmax(A)
    # f(1) = max (A+B)
    f1 = np.max(A + B)
    idx1 = np.argmax(A + B)
    
    # If the max at v=0 has B<=0 and max at v=1 has B>=0, 
    # the minimum might be at their intersection
    if B[idx0] <= 0 and B[idx1] >= 0 and idx0 != idx1:
        # Intersection: A[idx0] + B[idx0]*v = A[idx1] + B[idx1]*v
        denom = B[idx0] - B[idx1]
        if abs(denom) > 1e-12:
            v_cross = (A[idx1] - A[idx0]) / denom
            if 0 <= v_cross <= 1:
                f_cross = np.max(A + B * v_cross)
                # Check if this is actually the min
                # The two "dominant" lines should both achieve f_cross
                if f_cross <= f0 + 1e-12 and f_cross <= f1 + 1e-12:
                    return v_cross, f_cross
    # Fall back to grid search
    v_grid = np.linspace(0, 1, 21)
    f_vals = np.array([np.max(A + B * v) for v in v_grid])
    best_i = np.argmin(f_vals)
    # Binary refinement
    lo = v_grid[max(0, best_i - 1)]
    hi = v_grid[min(len(v_grid)-1, best_i + 1)]
    for _ in range(20):
        mid = (lo + hi) / 2
        f_lo = np.max(A + B * lo)
        f_mid = np.max(A + B * mid)
        f_hi = np.max(A + B * hi)
        if f_mid <= f_lo and f_mid <= f_hi:
            return mid, f_mid
        elif f_mid < f_hi:
            hi = mid
        else:
            lo = mid
    return v_grid[best_i], f_vals[best_i]
```

This is much faster: O(n) per coordinate instead of O(n*41).

Total per sweep: n * O(n) = O(n²) = 960² ≈ 1M operations. Very fast!
