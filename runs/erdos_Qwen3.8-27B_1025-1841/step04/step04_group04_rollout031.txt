import numpy as np
from scipy.optimize import minimize
import time

_N_G = 0
_T_G = 0

def _slsqp_obj(x):
    n = _N_G
    T = _T_G
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + (1.0 / T) * np.log(np.sum(np.exp(T * shifted)))) * dx

def _slsqp_con(x):
    return np.sum(x) - _N_G / 2.0

def _slsqp_jac(x):
    return np.ones(_N_G)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def run_slsqp(h, T, maxiter, ftol):
    global _N_G, _T_G
    _N_G = len(h)
    _T_G = T
    bounds = [(0.0, 1.0)] * _N_G
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
    try:
        return minimize(_slsqp_obj, h.copy(), method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return None

def binary_snap(h, n, time_limit):
    start = time.time()
    best_c = compute_c5(h)
    for j in range(n):
        if time.time() - start > time_limit:
            break
        for val in (0.0, 1.0):
            if abs(h[j] - val) < 1e-8:
                continue
            trial = h.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                h = trial
                best_c = c
                break
    return h, best_c

def fine_cd(h, n, time_limit):
    start = time.time()
    best_c = compute_c5(h)
    order = np.argsort(-np.abs(h - 0.5))
    for j in order:
        if time.time() - start > time_limit:
            break
        v = h[j]
        if v < 0.003 or v > 0.997:
            continue
        local_best = v
        local_c = best_c
        cands = [0.0, 1.0, 0.25, 0.5, 0.75,
                 max(0.0, v - 0.1), max(0.0, v - 0.04), max(0.0, v - 0.015),
                 min(1.0, v + 0.015), min(1.0, v + 0.04), min(1.0, v + 0.1)]

                local_best = cand
