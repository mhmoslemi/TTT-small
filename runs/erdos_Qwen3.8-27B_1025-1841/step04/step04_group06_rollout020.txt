def make_feasible(h, n):

    target = n / 2.0

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)

        idx = np.argmin(h) if diff > 0 else np.argmax(h)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    def f(h):

                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})

            for k in range(n_target - n_ones):
                h_bin[zero_idx[k]] = 1.0
        h_bs2, c5_bs2 = binary_local_search(h_bin, n_bin, rng, max_rounds=100, n_tries=2000)
        if c5_bs2 < best_c5:
            best_c5, best_h = c5_bs2, h_bs2.copy()

    # Phase 7: Final coordinate descent polish on continuous
    n_cd = min(len(best_h), 1024)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    if tl() > 20:
        deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.03, -0.03]
        for rnd in range(6):
            if tl() < 10:
                break
            improved = False
            for j in range(n_cd):
                if tl() < 8:
                    break
                ov = h_cd[j]
                for d in deltas:
                    nv = ov + d
                    if nv < 0 or nv > 1:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5_exact(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        improved = True
                        break
                for bv in (0.0, 1.0):
                    if abs(bv - h_cd[j]) < 1e-9:
                        continue
                    tr = h_cd.copy()
                    tr[j] = bv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5_exact(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        improved = True
            if not improved:
                break
    c5_cd = compute_c5_exact(h_cd)
    if c5_cd < best_c5:
        best_c5, best_h = c5_cd, h_cd.copy()

    # Final active-lag squeeze
    if tl() > 15:
        n_f = len(best_h)
        h_al_f, c5_al_f = active_lag_optimize(best_h, n_f, max_iter=200, lr=0.03)
        c5_al_fe = compute_c5_exact(h_al_f)
        if c5_al_fe < best_c5:
            best_c5, best_h = c5_al_fe, h_al_f.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5_exact(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
