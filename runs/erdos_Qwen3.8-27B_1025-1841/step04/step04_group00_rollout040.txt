sum_{x: x and x+k both in [0,2]} h(x) * (1-h(x+k)) * dx

If we interpret h as being 0 outside [0,2], then:
sum_{all x} h(x) * (1-h(x+k)) * dx = sum_x h(x)*dx - sum_x h(x)*h(x+k)*dx = 1 - autocorr_k

This is NOT the same as sum_j h[j] * (1-h[j+k]) for the full range!

In the continuous sense:
∫_{-∞}^{∞} h(x)(1-h(x+k)) dx = ∫_0^2 h(x)(1-h(x+k)) dx (since h=0 outside [0,2])

But np.correlate(h, 1-h, mode='full') at the corresponding index gives:

These are NOT the same in general! They differ because the overlap region is the same (j and j+s are both in [0,n-1]) but the roles of h and (1-h) are swapped.

C₅ = max_k np.correlate(h, 1-h, mode='full')[k] * dx

Which is the maximum over all lags of the cross-correlation of h with (1-h).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(h)
        else:
            order = np.argsort(-h)
        for i in order:
            room = (1.0 - h[i]) if diff > 0 else h[i]
            if room > 1e-15:
                adj = min(room, abs(diff))
                h[i] += adj if diff > 0 else -adj
                diff = diff - adj if diff > 0 else diff + adj
            if abs(diff) < 1e-13:
                break
    return h

def compute_gradient_softmax(h, top_m, T):
    """Compute gradient of soft max of top_m correlations."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < len(corr):
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        top_vals = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        top_vals = corr
    
    mc = np.max(top_vals)
    exps = np.exp(T * (top_vals - mc))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        lag = k - (n - 1)
        w = weights[i]
        # d/dh[j] c[k] = (1-h[j-lag]) * I(0<=j-lag<n) - h[j+lag] * I(0<=j+lag<n)
        if lag >= 0:
            # (1-h[j-lag]) valid for j >= lag
            if lag < n:
                grad[lag:] += w * (1.0 - h[:n-lag])
            # -h[j+lag] valid for j+lag < n, i.e., j < n-lag
            if lag < n:
                grad[:n-lag] -= w * h[lag:]
        else:
            lag_abs = -lag
            # (1-h[j+lag_abs]) valid for j+lag_abs < n, i.e., j < n-lag_abs
            if lag_abs < n:
                grad[:n-lag_abs] += w * (1.0 - h[lag_abs:])
            # -h[j-lag_abs] valid for j-lag_abs >= 0, i.e., j >= lag_abs
            if lag_abs < n:
                grad[lag_abs:] -= w * h[:n-lag_abs]
    
    return grad * (2.0 / n)

def smooth_objective_and_grad(h, top_m, T):
    """Smooth approximation to max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < len(corr):
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        top_vals = corr[top_idx]
    else:
        top_vals = corr
    mc = np.max(top_vals)
    log_sum_exp = mc + np.log(np.sum(np.exp(T * (top_vals - mc))))
    obj = log_sum_exp / T * dx
    grad = compute_gradient_softmax(h, top_m, T)
    return obj, grad

def pgd_smooth(h, n, budget_s, start_time, top_m=20, T_init=20.0, T_final=500.0, n_iters=None):
    """PGD with smooth max, adaptive step size, and annealing."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    T = T_init
    step = 0.5
    momentum = 0.8
    velocity = np.zeros(n)
    no_improve = 0
    max_no_improve = 50
    
    it = 0
    while time.time() - start_time < budget_s:
        it += 1
        if n_iters is not None and it > n_iters:
            break
        
        # Anneal T
        if n_iters is not None:
            frac = it / n_iters
            T = T_init * (T_final / T_init) ** frac
        else:
            T = min(T * 1.02, T_final)
        
        obj, grad = smooth_objective_and_grad(h, top_m, T)
        c5_cur = compute_c5(h)
        
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step *= 0.5
                if step < 1e-5:
                    break
                no_improve = 0
        
        # Nesterov
        y = h - momentum * velocity
        y = np.clip(y, 0, 1)
        _, grad_y = smooth_objective_and_grad(y, top_m, T)
        velocity = momentum * velocity - step * grad_y
        new_h = h + velocity
        new_h = make_feasible(new_h, n)
        
        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur + 1e-12:
            h = new_h
        else:
            # Simple step
            new_h = make_feasible(h - step * grad, n)
            c5_new = compute_c5(new_h)
            if c5_new <= c5_cur + 1e-12:
                h = new_h
            else:
                step *= 0.7
                velocity *= 0.5
            if step < 1e-6:
                break
        
        if it % 100 == 0 and step < 0.1:
            step = min(step * 1.2, 0.5)
    
    h = make_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h = h
        best_c5 = c5_h
    return best_h, best_c5

def smooth_max_grad(h, top_m, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < len(corr):
        top_idx = np.argpartition(corr, -top_m)[-top_m:]
        top_vals = corr[top_idx].copy()
    else:
        top_idx = np.arange(len(corr))
        top_vals = corr.copy()
    
    mc = np.max(top_vals)
    exps = np.exp(T * (top_vals - mc))
    wsum = np.sum(exps)
    weights = exps / wsum
    
    log_sum_exp = mc + np.log(wsum)
    obj = log_sum_exp / T * dx
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        k = top_idx[i]
        lag = k - (n - 1)

            la = -lag
            if la < n:
                grad[:n-la] += w * (1.0 - h[la:])
                grad[la:] -= w * h[:n-la]

        return g

        return np.sum(x) - n / 2.0

        return np.ones(n)
    
    try:

                       bounds=[(0.0, 1.0)] * n,

For the binary local search, I'll use a simpler approach:
- Maintain the correlation array
- Try random swaps (1↔0 at two positions)
- Check if the max correlation decreases
- Use the incremental update formula

For the incremental update: when we swap positions i (1→0) and j (0→1):
The change in c[k] (where k is the index in the full correlation array):
Let s = k - (n-1) be the lag.

Original c[k] = ∑_m h[m+s]*(1-h[m]) for valid m, m+s
- h[i] = 1, h[j] = 0 before swap
- h[i] = 0, h[j] = 1 after swap

Change in c[k]:
- From the change at i (1→0):
  - i appears as "m" (the (1-h[m]) factor): h[m+s]*(1-h[m]) → h[i+s]*(1-0) = h[i+s] instead of h[i+s]*(1-1) = 0
    Change: +h[i+s] if i+s is valid
  - i appears as "m+s" (the h[m+s] factor): h[i]*(1-h[i-s]) → 0 instead of 1*(1-h[i-s])
    Change: -(1-h[i-s]) if i-s is valid

- From the change at j (0→1):
  - j appears as "m": h[j+s]*(1-h[j]) → h[j+s]*(1-1) = 0 instead of h[j+s]*(1-0) = h[j+s]
    Change: -h[j+s] if j+s is valid
  - j appears as "m+s": h[j]*(1-h[j-s]) → 1*(1-h[j-s]) = (1-h[j-s]) instead of 0
    Change: +(1-h[j-s]) if j-s is valid

Total change in c[k]:
Δc[k] = h[i+s]*I(valid) - (1-h[i-s])*I(valid) - h[j+s]*I(valid) + (1-h[j-s])*I(valid)

where s = k - (n-1), and "valid" means the index is in [0, n-1].

This is O(1) per k, so O(n) total for all k. Then we need to check if max(c[k] + Δc[k]) < max(c[k]), which is O(n).

So each swap evaluation is O(n) = O(960), which is fast! If we try 1000 swaps per pass, that's about 1M operations per pass.

def binary_local_search(h_init, n, budget_s, start_time):
    """Local search over binary sequences."""
    dx = 2.0 / n
    target = n // 2
    
    # Round to binary: pick top n/2 values
    order = np.argsort(-h_init)
    h = np.zeros(n)
    h[order[:target]] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    
    rng = np.random.RandomState(42)
    max_iter = 500000
    
    it = 0
    while it < max_iter and time.time() - start_time < budget_s:

        # Try a random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]

        # Δc[k] for k in range(2n-1)
        # s = k - (n-1)
        delta = np.zeros(2*n-1)
        for k in range(2*n-1):
            s = k - (n - 1)
            d = 0.0
            # i as "m" in (1-h[m]): contributes h[i+s] if 0 <= i+s < n
            if 0 <= i + s < n:
                d += h[i + s]
            # i as "m+s": removes h[i]*(1-h[i-s]) = (1-h[i-s]) if 0 <= i-s < n
            if 0 <= i - s < n:
