def project(h, n):
    """Project h onto {x : 0 ≤ x ≤ 1, sum(x) = n/2}."""
    h = h.copy()
    target = n / 2.0
    # First clip
    h = np.clip(h, 0, 1)
    # Then adjust sum
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            # Add to variables not at 1
            mask = h < 1.0
            if not mask.any():
                break
            # Add equally to non-saturated variables
            step = diff / mask.sum()
            h[mask] = np.minimum(h[mask] + step, 1.0)
        else:
            # Remove from variables not at 0
            mask = h > 0.0
            if not mask.any():
                break
            step = diff / mask.sum()
            h[mask] = np.maximum(h[mask] + step, 0.0)
    # Fix any remaining error
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h
