To minimize max a(d), we want R(d) to be as large as possible for all d (positive autocorrelation everywhere).

                i2 = int(ones[rng.randint(len(ones))])
                while i2 == i1:
                    i2 = int(ones[rng.randint(len(ones))])
                j2 = int(zeros[rng.randint(len(zeros))])
                while j2 == j1 or j2 == i1:
                    j2 = int(zeros[rng.randint(len(zeros))])

        if do_2swap:
            old_vals = (b[i1], b[i2], b[j1], b[j2])
            b[i1] = 0; b[i2] = 0; b[j1] = 1; b[j2] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                b[i1], b[i2], b[j1], b[j2] = old_vals
                stall_count += 1
        else:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = int(ones[rng.randint(len(ones))])
            j = int(zeros[rng.randint(len(zeros))])
            b[i] = 0; b[j] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
                    stall_count = 0
                else:
                    stall_count += 1
            else:
                b[i] = 1; b[j] = 0
                stall_count += 1

        if stall_count > 8000:
            for _ in range(rng.randint(3, 8)):
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                pi = int(ones[rng.randint(len(ones))])
                pj = int(zeros[rng.randint(len(zeros))])
                b[pi] = 0; b[pj] = 1
            curr_c5 = compute_c5(b)
            stall_count = 0

    return best_b, best_c5

def slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    res = minimize(slsqp_obj, h.copy(), args=(n, T), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s, obj_T=50.0):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.1), max(0.0, old - 0.02),
                     min(1.0, old + 0.02), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def golden_init(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    positions = (np.arange(n) * phi) % 1.0
    rng.shuffle(positions)
    sorted_pos = np.argsort(positions)
    target = n // 2
    for idx in sorted_pos[:target]:
        h[idx] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = golden_init(960, rng)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binary SA at initial resolution
    b1, c5_1 = binary_sa(h_init, n0, 100000, 0.012, 0.00001, rng, two_swap_frac=0.3)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()

    # Phase 2: Restart SA from perturbed best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        perturbed = best_h.copy()
        n_p = len(perturbed)
        np_ones = np.where(perturbed == 1)[0] if np.allclose(perturbed, to_binary(perturbed, n_p)) else np.where(perturbed >= 0.5)[0]
        np_zeros = np.where(perturbed < 0.5)[0] if np.allclose(perturbed, to_binary(perturbed, n_p)) else np.where(perturbed < 0.5)[0]
        for _ in range(15):
            if len(np_ones) == 0 or len(np_zeros) == 0:
                break
            pi = int(np_ones[rng.randint(len(np_ones))])
            pj = int(np_zeros[rng.randint(len(np_zeros))])
            perturbed[pi] = 0.0
            perturbed[pj] = 1.0
        b2, c5_2 = binary_sa(perturbed, n0, 80000, 0.008, 0.00001, rng, two_swap_frac=0.3)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()

    # Phase 3: Try golden ratio initialization
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        h_gold = golden_init(n0, rng)
        b3, c5_3 = binary_sa(h_gold, n0, 80000, 0.015, 0.00001, rng, two_swap_frac=0.3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = b3.copy()

    # Phase 4: SLSQP polish at current resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        try:
            x_res = slsqp_run(best_h, n0, 50.0, 300, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 5: Upsample to higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 400:
        n1 = min(n0 + 480, 1440)
        h_up = upsample_h(best_h, n1)
        b4, c5_4 = binary_sa(h_up, n1, 70000, 0.010, 0.00001, rng, two_swap_frac=0.3)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = b4.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 150:
            try:
                x_res = slsqp_run(best_h, n1, 40.0, 250, 1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n1)
            except Exception:
                pass

    # Phase 6: Try n=1920
    remaining = budget_s - (time.time() - start_time)
    if remaining > 350:
        n2 = 1920
        h_up2 = upsample_h(best_h, n2)
        b5, c5_5 = binary_sa(h_up2, n2, 50000, 0.008, 0.00001, rng, two_swap_frac=0.25)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = b5.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 120:
            try:
                x_res = slsqp_run(best_h, n2, 30.0, 200, 1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n2)
            except Exception:
                pass

    # Phase 7: Coordinate descent on best solution
    n_pts = len(best_h)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 15, start_time, budget_s)

    # Phase 8: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    n_pts = len(best_h)
    if remaining > 80:
        try:
            x_res = slsqp_run(best_h, n_pts, 20.0, 200, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 9: One more binary SA if time permits
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100 and len(best_h) <= 1000:
        n_pts = len(best_h)
        b_fin, c5_fin = binary_sa(best_h, n_pts, 40000, 0.005, 0.000005, rng, two_swap_frac=0.4)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = b_fin.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
