```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def subgrad_at_lag(h, lag):
    n = len(h)
    grad = np.ones(n)
    if lag > 0:
        grad[:n-lag] -= h[lag:]
        grad[lag:] -= h[:n-lag]
    elif lag < 0:
        ll = -lag
        grad[ll:] -= h[:n-ll]
        grad[:n-ll] -= h[ll:]
    else:
        grad -= 2.0 * h
    return grad

def projected_gradient_step(h, lag_weights, lr, n):
    """Take a projected gradient step using weighted subgradients."""
    grad = np.zeros(n)
    for lag, w in lag_weights:
        grad += w * subgrad_at_lag(h, lag)
    
    h_new = h - lr * grad
    h_new = make_feasible(h_new, n)
    return h_new

def pgd_optimize(h, n, max_iter, start_lr, end_lr, n_lags=5, time_limit=30):
    """Project gradient descent with adaptive step size."""
    start_time = time.time()
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    center = n - 1
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        
        # Find top lags
        top_idx = np.argsort(corr)[-n_lags:]
        lags = top_idx - center
        weights = np.exp(50.0 * (corr[top_idx] - max_val))  # softmax-like
        weights /= np.sum(weights)
        
        # Adaptive step size
        t_frac = it / max_iter
        lr = start_lr * (end_lr / start_lr) ** t_frac
        
        # Try the step
        h_trial = h.copy()
        grad = np.zeros(n)
        for lag, w in zip(lags, weights):
            grad += w * subgrad_at_lag(h, int(lag))
        
        h_trial = h - lr * grad
        h_trial = make_feasible(h_trial, n)
        c_trial = compute_c5(h_trial)
        
        if c_trial < best_c - 1e-15:
            best_h = h_trial.copy()
            best_c = c_trial
            h = h_trial
        else:
            # Halve step size and retry
            h_trial = h - lr/2 * grad
            h_trial = make_feasible(h_trial, n)
            c_trial = compute_c5(h_trial)
            if c_trial < best_c - 1e-15:
                best_h = h_trial.copy()
                best_c = c_trial
                h = h_trial
            else:
                # No improvement, shrink lr
                pass
    
    return best_h, best_c
```

Actually, coordinate descent is slow because of the correlate calls. Let me optimize: instead of full correlate for each candidate, I can update the correlation incrementally.

When h[j] changes by δ, the correlation changes by the delta vector I computed. So for coordinate descent:
- Precompute corr
- For each j, for each candidate value v:
  - δ = v - h[j]
  - Compute delta vector (O(n) using the formula above)
  - new_corr = corr + δ_adjusted (accounting for non-linear change)
  
  Wait, the delta formula assumes a small change or binary flip. For a general change δ, the formula is:
  Δc_ℓ = δ(1-h[j+ℓ]) - h[j-ℓ](1-δ)... 
  
  Hmm, actually the exact change when h[j] → h[j]+δ is:
  Δc_ℓ = δ(1-h[j+ℓ]) + h[j-ℓ](1-(h[j]+δ)) - h[j-ℓ](1-h[j])
  = δ(1-h[j+ℓ]) - δ·h[j-ℓ]
  = δ[(1-h[j+ℓ]) - h[j-ℓ]]
  
  Wait, that's the same linear formula! Because c_ℓ is bilinear in h (it's a sum of products h[i]·h[i+ℓ]), the change when one variable changes is linear in the change. So the delta formula is exact for any δ, not just for binary flips!

  Great! So:
  Δc_ℓ = δ · [(1-h[j+ℓ])_{valid} - h[j-ℓ]_{valid}]
  
  This means I can compute the new correlation for any candidate value in O(n) using the delta vector, without recomputing the full correlation.

Let me exploit this for coordinate descent:

```python
def coord_descent_fast(h, n, max_rounds=30, candidates_per_j=5, time_limit=100):
    """Fast coordinate descent using incremental correlation updates."""
    h = h.copy()
    start = time.time()
    dx = 2.0 / n
    center = n - 1
    els = np.arange(1, n)  # lags 1..n-1
    
    best_c = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[j]
            best_val = old_val
            best_mc = np.max(corr)
            
            # Candidate values
            cands = [0.0, 1.0, 0.5, max(0, old_val-0.1), min(1, old_val+0.1)]
            cands = [max(0.0, min(1.0, c)) for c in cands]
            
            # Precompute h_plus and h_minus for this j
            h_plus = np.zeros(n-1)
            h_minus = np.zeros(n-1)
            if j + 1 < n:
                h_plus[:n-1-j] = h[j+1:n]
            if j >= 1:
                h_minus[:j] = h[j-1:-1:-1]
            
            base_pos = (1.0 - h_plus) - h_minus
            base_neg = (1.0 - h_minus) - h_plus
            
            for v in set(cands):
                if v == old_val:
                    continue
                delta = v - old_val
                # Compute new corr
                new_corr = corr.copy()
                new_corr[center+1:] += delta * base_pos
                new_corr[:n-1] += delta * base_neg[::-1]
                new_corr[center] += delta * (1.0 - 2.0*old_val)
                
                # Check bounds: h[j] = v must be in [0,1] (already ensured)
                # Also need to maintain sum constraint - skip for now, fix later
                
                new_max = np.max(new_corr)
                if new_max < best_mc - 1e-14:
                    best_mc = new_max
                    best_val = v
            
            if best_val != old_val:
                h[j] = best_val
                delta = best_val - old_val
                corr[center+1:] += delta * base_pos
                corr[:n-1] += delta * base_neg[::-1]
                corr[center] += delta * (1.0 - 2.0*old_val)
                h = make_feasible(h, n)
                # After make_feasible, corr might be slightly off
                # Recheck
                actual_c = compute_c5(h)
                if actual_c < best_c - 1e-14:
                    best_c = actual_c
                    corr = np.correlate(h, 1.0 - h, mode='full')
                    improved = True
                else:
                    # Revert
                    h[j] = old_val
                    h = make_feasible(h, n)
                    corr = np.correlate(h, 1.0 - h, mode='full')
        
        if not improved:
            break
    
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    return h, best_c
```

Wait, there's an issue with make_feasible. When I change h[j] by δ, the sum changes by δ. make_feasible will redistribute this across all other h[i] values, which means the correlation changes in a way I can't easily track. So I should either:
1. Not use make_feasible during the loop (allow sum to drift), and fix at the end
2. Use a different constraint enforcement that only affects h[j]

Actually, option 1 is fine if I track the cumulative sum drift and fix it at the end of each round. Or I can simply not enforce the constraint during coordinate descent and enforce it only when evaluating the final C₅.

But wait, the C₅ evaluation needs the constraint to be satisfied. If sum ≠ n/2 during the descent, the C₅ I'm computing is for an infeasible point.

Alternative: I'll enforce the constraint by making a "virtual" adjustment. When h[j] changes by δ, I'll also add -δ/n to all other h[i] (n-1 of them). But this makes the delta computation more complex.

Simplest approach: just do the coordinate descent without the sum constraint (allow sum to drift), and after each round, call make_feasible and recompute. The drift over one round (at most n small changes) should be small.

Actually, even simpler: since each step changes h[j] by a small amount (at most 0.1 or so), and there are n=960 steps per round, the total sum drift per round is at most 960 × 0.1 = 96, which is way too much. So I do need to handle the constraint.

Best approach: after each individual coordinate change, immediately fix the constraint by distributing the excess across all other coordinates. The effect on the correlation is:
- h[j] changes by δ
- All other h[i] (i≠j) change by -δ/(n-1)
- The total effect on c_ℓ is: δ·[base_pos/n-1 stuff] + sum of the -δ/(n-1) contributions

This is getting complicated. Let me just use the simpler approach:
- Do coordinate descent allowing sum to drift
- Every 10 coordinate changes, call make_feasible and recompute corr
- This keeps the drift bounded

Or, even simpler: just do the full approach with recompute every step. At n=960, correlate takes ~1ms, and I'm doing 960×5 = 4800 evaluations per round. That's 4.8s per round. With 20 rounds, that's 96s. Acceptable.

Let me simplify and just recompute the correlation from scratch for each candidate:

```python
def coord_descent(h, n, max_rounds=20, time_limit=80):
    h = h.copy()

    best_c = compute_c5(h)
    
    for rnd in range(max_rounds):

            break
        improved = False
        for j in range(n):

            best_c_local = best_c
            
            for v in [0.0, 1.0, 0.5, max(0, old_val-0.1), min(1, old_val+0.1)]:
                if abs(v - old_val) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_val = v
            
            if best_val != old_val:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = compute_c5(h)
                improved = True
        
        if not improved:
            break
    
    h = make_feasible(h, n)
    best_c = compute_c5(h)
    return h, best_c
```

This is O(n × num_cands × T_corr) per round. At n=960, 5 cands, T_corr≈1ms: 960×5×1ms = 4.8s per round. 20 rounds = 96s. OK.

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        else:

        if cnt == 0:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
