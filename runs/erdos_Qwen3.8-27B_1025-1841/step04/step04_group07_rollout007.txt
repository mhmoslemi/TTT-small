def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -700, 700)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_run(h0, n, T, mu, max_iter=150, gtol=1e-11):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_obj(corr, T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': gtol})
    return make_feasible(res.x, n)

def active_set_run(h0, n, n_rounds=40, topk_frac=0.12, mu=500):
    dx = 2.0 / n
    h = h0.copy()
    for rnd in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_c = len(corr)
        k = max(8, int(topk_frac * n_c))
        idx = np.argpartition(corr, -k)[-k:]
        # Weight by how close to max
        max_c = np.max(corr)
        w = np.zeros(n_c)
        w[idx] = 1.0
        # Soft weighting within active set
        w[idx] = np.exp(-0.5 * (max_c - corr[idx]) / (max_c - np.min(corr[idx]) + 1e-10))
        w /= w.sum()
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(w, c)) * dx + mu * (x.sum() - n / 2.0) ** 2
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 25, 'ftol': 1e-15, 'gtol': 1e-10})
        h_new = make_feasible(res.x, n)
        if compute_c5(h_new) <= compute_c5(h) + 1e-13:
            h = h_new
    return h

def binary_local(h_bin, n, rng, max_rounds=80, tries=4000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_t = min(tries, len(ones) * len(zeros))
        for _ in range(n_t):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def coord_polish(h0, n, max_rounds=6):
    steps = [0.04, -0.04, 0.015, -0.015, 0.005, -0.005, 0.002, -0.002]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Load initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)
    n0 = len(h0)
    h0 = make_feasible(h0, n0)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = n0
    
    # Phase 1: Quick L-BFGS at parent resolution
    if rem() > 100:
        for T in [300, 1000]:
            if rem() < 30:
                break
            h_opt = lbgfb_run(best_h, best_n, T, 600, max_iter=100)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt
    
    # Phase 2: Upscale to n=960 and optimize
    n_up = 960
    if rem() > 200:
        h_up = upsample_h(best_h, n_up)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5, best_h, best_n = c_up, h_up, n_up
        
        for T in [200, 800, 2000]:
            if rem() < 30:
                break
            h_opt = lbgfb_run(h_up, n_up, T, 400, max_iter=120)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h, best_n = c_opt, h_opt.copy(), n_up
                h_up = h_opt.copy()
    
    # Phase 3: Active set at best resolution
    if rem() > 150:
        h_as = active_set_run(best_h, best_n, n_rounds=35, topk_frac=0.12)
        c_as = compute_c5(h_as)
        if c_as < best_c5:
            best_c5, best_h = c_as, h_as.copy()
    
    # Phase 4: Binary local search at n0
    if rem() > 150:
        n_bin = n0  # Use original resolution for binary search
        h_b = upsample_h(best_h, n_bin) if n_bin != best_n else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        # Ensure exact count
        target = n_bin // 2
        n_ones = int(h_bin.sum())
        if n_ones != target:
            if n_ones > target:
                idx = np.where(h_bin == 1)[0]
                # Remove the ones that contribute most to max correlation
                h_bin[idx[:n_ones - target]] = 0.0
            else:
                idx = np.where(h_bin == 0)[0]
                h_bin[idx[:target - n_ones]] = 1.0
        h_bs, c_bs = binary_local(h_bin, n_bin, rng, max_rounds=60, tries=5000)
        if c_bs < best_c5:
            best_c5, best_h, best_n = c_bs, h_bs.copy(), n_bin
    
    # Phase 5: Upscale to higher resolution for final optimization
    n_hi = 1280
    if rem() > 150:
        h_hi = upsample_h(best_h, n_hi)
        c_hi = compute_c5(h_hi)
        if c_hi < best_c5:
            best_c5, best_h, best_n = c_hi, h_hi.copy(), n_hi
        
        for T in [500, 1500, 4000]:
            if rem() < 25:
                break
            h_opt = lbgfb_run(h_hi, n_hi, T, 300, max_iter=100)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h, best_n = c_opt, h_opt.copy(), n_hi
                h_hi = h_opt.copy()
        
        if rem() > 50:
            h_as2 = active_set_run(h_hi, n_hi, n_rounds=25, topk_frac=0.10)
