8. **Subgradient / cutting plane approach**: Since C₅(h) = max_k f_k(h) where each f_k is a (bilinear) function, at any given point h, the active set is S = {k : f_k(h) ≈ C₅(h)}. The subgradient direction is the average of gradients of active functions. We can do:
   - Compute active set
   - Compute gradient of average active function
   - Take a step in -gradient direction
   - Project back to feasible set
   
   This is essentially projected subgradient or a variant of the cutting plane method.

```python
import numpy as np
from scipy.optimize import minimize
import time

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def exact_coord_min(h, j, corr, n, v):
    """Find optimal value for h[j] given current corr, minimizing max(corr + delta*coeff)."""
    # coeff_j,m = change in corr[m] per unit change in h[j]
    # For m in 0..N-1: corr[m] = sum_{i=0}^{N-1-m} h[i+m]*(1-h[i])
    # For m in N..2N-2: corr[m] = sum_{j2=0}^{N-2-(m-N)} h[j2+m-N+1]*(1-h[j2])
    
    # Build coefficient array for h[j]
    # coeff[m] = d(corr[m])/d(h[j])
    
    coeff = np.zeros(2 * n - 1)
    
    # Case 1: m < N, corr[m] = sum_{i=0}^{N-1-m} h[i+m]*(1-h[i])
    # h[j] as h[i+m]: i = j-m, valid if 0<=j-m<=N-1-m → m<=j<=N-1 → m in [0, min(j, N-1)]
    for m in range(0, min(j, n - 1) + 1):
        i = j - m
        coeff[m] += (1.0 - h[i])
    
    # h[j] in (1-h[i]) where i=j: valid if j <= N-1-m → m <= N-1-j
    if j < n:
        for m in range(0, min(n - 1 - j, n - 1) + 1):
            if j + m < n:
                coeff[m] -= h[j + m]
    
    # Case 2: m = N+s, 0<=s<=N-2
    # corr[N+s] = sum_{j2=0}^{N-2-s} h[j2+s+1]*(1-h[j2])
    # h[j] as h[j2+s+1]: j2 = j-s-1, valid if 0<=j-s-1<=N-2-s → s+1<=j<=N-1, s in [max(0,j-N+1), min(j-1, N-2)]
    for s in range(max(0, j - n + 1), min(j - 1, n - 2) + 1):
        m = n + s
        j2 = j - s - 1
        if 0 <= j2 < n:
            coeff[m] += (1.0 - h[j2])
    
    # h[j] in (1-h[j2]) where j2=j: valid if j <= N-2-s → s <= N-2-j
    for s in range(0, min(n - 2 - j, n - 2) + 1):
        m = n + s
        if j + s + 1 < n:
            coeff[m] -= h[j + s + 1]
    
    # Now: new_corr[m] = corr[m] + (v' - v) * coeff[m]
    # Minimize max_m new_corr[m] over v' in [0,1]
    # g(v') = max_m (corr[m] + (v'-v)*coeff[m])
    #       = max_m (corr[m] - v*coeff[m] + v'*coeff[m])
    
    base = corr - v * coeff  # B_m = corr[m] - v*coeff[m]
    # g(v') = max_m (base[m] + v'*coeff[m])
    
    # Evaluate at a grid of points
    n_grid = 25
    vs = np.linspace(0, 1, n_grid)
    # g(v') = max_m (base + v'*coeff) = max over m
    # Compute efficiently: for each v', g = np.max(base + v'*coeff)
    gs = np.array([np.max(base + vi * coeff) for vi in vs])
    
    # Find the best grid point
    best_idx = np.argmin(gs)
    v_opt = vs[best_idx]
    
    # Refine: if interior, do finer search
    if 0 < best_idx < n_grid - 1:
        lo = vs[best_idx] - (vs[1] - vs[0])
        hi = vs[best_idx] + (vs[1] - vs[0])
        lo = max(0.0, lo)
        hi = min(1.0, hi)
        vs2 = np.linspace(lo, hi, 25)
        gs2 = np.array([np.max(base + vi * coeff) for vi in vs2])
        v_opt = vs2[np.argmin(gs2)]
    
    return v_opt

def coord_descent_pass(h, n, max_iter=30):
    """One full pass of coordinate descent."""
    corr = compute_corr(h)
    best_c5 = np.max(corr)
    
    for _ in range(max_iter):
        improved = False
        for j in range(n):
            v_opt = exact_coord_min(h, j, corr, n, h[j])
            if abs(v_opt - h[j]) > 1e-10:
                # Update h[j] and corr
                delta = v_opt - h[j]
                # Instead of recomputing full corr, update incrementally
                # But for safety, just set and recompute
                h[j] = v_opt
                h = make_feasible(h, n)
                corr = compute_corr(h)
                new_max = np.max(corr)
                if new_max < best_c5 - 1e-14:
                    best_c5 = new_max
                    improved = True
        if not improved:
            break
    
    return h, float(best_c5 * 2.0 / n)
```

For a given j:

  For m=0: -= h[j], for m=1: -= h[j+1], ..., for m=n-1-j: -= h[j+n-1-j] = h[n-1]
  So coeff[0:n-j] -= h[j:n] (since n-1-j+1 = n-j elements)

```python
    """Compute d(corr[m])/d(h[j]) for all m."""
    coeff = np.zeros(2*n - 1)

    # Case A2: m in [0, n-j-1], coeff[m] -= h[j+m]

    if k2 >= 0:
        coeff[0:k2+1] -= h[j:j+k2+1]
    
    # Case B1: s in [max(0, j-n+1), min(j-1, n-2)], m=n+s, coeff[m] += (1-h[j-s-1])
    s_lo = max(0, j - n + 1)
    s_hi = min(j - 1, n - 2)
    if s_lo <= s_hi:
        # j2 = j-s-1, ranges from j-s_lo-1 down to j-s_hi-1
        # h[j2] for j2 = j-s_lo-1, j-s_lo-2, ..., j-s_hi-1

        coeff[n+s_lo : n+s_hi+1] += 1.0 - h[j2_lo:j2_hi+1][::-1]
    
    # Case B2: s in [0, n-2-j] (where j+s+1 < n, i.e., s < n-1-j, s <= n-2-j), m=n+s

        coeff[n : n+s_hi2+1] -= h[j+1:j+s_hi2+2]
    
    return coeff
```
