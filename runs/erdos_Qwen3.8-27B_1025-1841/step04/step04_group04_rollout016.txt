import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

_OPT_N = 0
_OPT_T = 0.0

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    Nft = 2 * n
    H = fft(x, Nft)
    G = fft(1.0 - x, Nft)
    corr = ifft(H * np.conj(G)).real[:2 * n - 1]
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        per = diff / cnt
        h[mask] += per
        h = np.clip(h, 0.0, 1.0)
    return h

def _upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return _make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return _make_feasible(h_new, new_n)

def _slsqp(h_init, n, T, maxiter, ftol=1e-13):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init

def _coord_descent(h, n, max_rounds=15, tol=1e-14):
    best_c = _compute_c5(h)
    deltas = [-0.1, -0.05, -0.02, -0.005, 0.005, 0.02, 0.05, 0.1]
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        improved = False
        rnd += 1
        for j in range(n):
            old = h[j]
            best_c_j = best_c
            best_trial = old
            for d in deltas:
                cand = max(0.0, min(1.0, old + d))
                if d == 0:
                    cand = old
                trial = h.copy()
                trial[j] = cand
                trial = _make_feasible(trial, n)
                c = _compute_c5(trial)
                if c < best_c_j - tol:
                    best_c_j = c
                    best_trial = cand
            if best_trial != old:
                h = h.copy()
                h[j] = best_trial
                h = _make_feasible(h, n)
                best_c = best_c_j
                improved = True
    return h, best_c

def _binary_snap(h, n, max_passes=3):
    best_c = _compute_c5(h)
    improved = True
    passes = 0
    while improved and passes < max_passes:
        improved = False
        passes += 1
        for j in range(n):
            for val in [0.0, 1.0]:
                if val == h[j]:
                    continue
                trial = h.copy()
                trial[j] = val
                trial = _make_feasible(trial, n)
                c = _compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    h = trial.copy()
                    improved = True
                    break
    return h, best_c
