    corr = np.correlate(h_bin.astype(float), 1.0 - h_bin.astype(float), mode='full')

    bounds = list(zip(-b, 1.0 - b)) + [(0, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')

    # Re-project to maintain sum

    return h_new, res.fun
    
    return None, None

def solve_lp(b, n):
