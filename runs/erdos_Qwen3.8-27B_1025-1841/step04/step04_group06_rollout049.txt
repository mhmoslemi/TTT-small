    for T in [30, 100, 500, 1500]:
        if remaining() < 80:
            break
        try:
            h_opt = lbgfb_refine(h1, n1, T, mu=1000.0, max_iter=350)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5, best_h = c, h_opt.copy()
            h1 = h_opt.copy()
        except Exception:
            pass

    # Phase 2: Fourier global search at n=2048
    if remaining() > 150:
        M_f = 35
        params_f, cos_f, sin_f, x_f = fourier_init(h1, n1, M_f)
        for T_f in [200, 800, 3000]:
            if remaining() < 60:
                break
            try:
                res = minimize(
                    lambda p: fourier_obj(p, n1, x_f, cos_f, sin_f, M_f, T_f, 800.0),
                    params_f, method='L-BFGS-B',
                    options={'maxiter': 600, 'ftol': 1e-14, 'gtol': 1e-10})
                h_four = fourier_to_h(res.x, n1, M_f, cos_f, sin_f)
                c4 = compute_c5(h_four)
                if c4 < best_c5:
                    best_c5, best_h = c4, h_four.copy()
                    h1 = h_four.copy()
                    params_f, _, _, _ = fourier_init(h1, n1, M_f)
            except Exception:
                pass

    # Phase 3: Binary SA at n=2048
    if remaining() > 200:
        h_bin = (h1 > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_t = n1 // 2
        if n_ones > n_t:
            idx = np.where(h_bin == 1)[0]
            for k in range(n_ones - n_t):
                h_bin[idx[k]] = 0.0
        elif n_ones < n_t:
            idx = np.where(h_bin == 0)[0]
            for k in range(n_t - n_ones):
                h_bin[idx[k]] = 1.0
        n_sa = min(6000, max(1500, int(remaining() * 20)))
        h_sa, c_sa = binary_sa(h_bin, n1, rng, n_sa, 0.4, 0.003)
        if c_sa < best_c5:
            best_c5, best_h = c_sa, h_sa.copy()
            h1 = h_sa.copy()

    # Phase 4: Upsample to 4096, L-BFGS-B
    if remaining() > 120:
        n2 = 4096
        h2 = upsample_h(h1, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        for T in [100, 500, 2000]:
            if remaining() < 80:
                break
            try:
                h_opt = lbgfb_refine(h2, n2, T, mu=1500.0, max_iter=250)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
                h2 = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Binary SA at n=4096
    if remaining() > 150:
        h_bin2 = (h2 > 0.5).astype(float)
        n_o2 = int(np.sum(h_bin2))
        n_t2 = n2 // 2
        if n_o2 > n_t2:
            idx = np.where(h_bin2 == 1)[0]
            for k in range(n_o2 - n_t2):
                h_bin2[idx[k]] = 0.0
        elif n_o2 < n_t2:
            idx = np.where(h_bin2 == 0)[0]
            for k in range(n_t2 - n_o2):
                h_bin2[idx[k]] = 1.0
        n_sa2 = min(4000, max(800, int(remaining() * 15)))
        h_sa2, c_sa2 = binary_sa(h_bin2, n2, rng, n_sa2, 0.3, 0.002)
        if c_sa2 < best_c5:
            best_c5, best_h = c_sa2, h_sa2.copy()
            h2 = h_sa2.copy()

    # Phase 6: L-BFGS-B polish at 4096
    if remaining() > 60:
        for T in [500, 2000, 8000]:
            if remaining() < 25:
                break
            try:
                h_opt = lbgfb_refine(h2, n2, T, mu=3000.0, max_iter=200)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
                h2 = h_opt.copy()
            except Exception:
                pass

    # Phase 7: Coordinate descent polish
    if remaining() > 30:
        h_cd, c_cd = coord_descend(h2, n2, rng, max_rounds=3,
                                   time_limit=budget_s - 20, start_time=start_time)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 8: Final L-BFGS-B squeeze
    if remaining() > 15:
        n_f = len(best_h)
        try:
            h_final = lbgfb_refine(best_h, n_f, 10000, mu=5000.0, max_iter=80)
            c_final = compute_c5(h_final)
            if c_final < best_c5:
                best_c5, best_h = c_final, h_final.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
