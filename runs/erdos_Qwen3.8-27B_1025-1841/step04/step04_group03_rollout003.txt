    max_c = np.max(corr)
    shifted = corr - max_c  # all <= 0
    # For T very large, T*shifted can be very negative, underflowing
    # Only the terms close to max_c contribute
    threshold = max_c - 50.0/T  # exp(T*(-50/T)) = exp(-50) ≈ 2e-22, negligible
    mask = corr >= threshold
    if not np.any(mask):
        return max_c, np.zeros_like(corr)
    active = corr[mask]
    exps = np.exp(T * (active - max_c))
    sum_exps = np.sum(exps)
    lse_val = max_c + np.log(sum_exps) / T
    # Gradient weights
    weights = np.zeros_like(corr)
    weights[mask] = exps / sum_exps
    return lse_val, weights

    mc = np.max(corr)
    shifted = T * (corr - mc)
    # Safe log-sum-exp
    ms = np.min(shifted)
    if ms < -700:
        return mc * dx
    exps = np.exp(shifted - ms)
    s = np.sum(exps)
    if s < 1e-300:
        return mc * dx
    return (mc + (ms + np.log(s)) / T) * dx

def lse_obj_and_wrapper(x, T_val, n_val):
    return lse_objective(x, T_val, n_val)

def slsqp_solve(h, n, T, maxiter=200):
    def obj(x):
        return lse_objective(x, T, n)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h.copy()

def top_k_objective(x, n, K, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top = np.sort(corr)[-K:]
    mc = np.max(top)
    shifted = T * (top - mc)
    ms = np.min(shifted)
    if ms < -700:
        return mc * dx
    return (mc + (ms + np.log(np.sum(np.exp(shifted - ms)))) / T) * dx

def top_k_slqp(h, n, K, T, maxiter=100):
    def obj(x):
        return top_k_objective(x, n, K, T)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h.copy()

def active_lag_slqp(h, n, max_cycles=10, iters_per_lag=80, time_limit=60):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for cyc in range(max_cycles):
        if time.time() - start > time_limit:
            break
        cur_h = best_h.copy()
        dx = 2.0 / n
        corr = np.correlate(cur_h, 1.0 - cur_h, mode='full')
        k_star = int(np.argmax(corr))
        def obj_al(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return c[k_star] * dx
        def con_fun(x):
            return np.sum(x) - n / 2.0
        def con_jac(x):
            return np.ones(n)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
        try:
            res = minimize(obj_al, cur_h.copy(), method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': iters_per_lag, 'ftol': 1e-14})
            c5_new = compute_c5(res.x)
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = make_feasible(res.x, n)
        except Exception:
            pass
    return best_h, best_c5

def coord_descent_fin(h, n, max_rounds, start_time, budget_s, step_sizes=None):
    best_c5 = compute_c5(h)
    if step_sizes is None:
        step_sizes = [0.0, 1.0, -0.05, 0.05, -0.02, 0.02, -0.008, 0.008, -0.003, 0.003]
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for ss in step_sizes:
                if abs(ss) == 0:
                    cand = 0.0 if ss == 0.0 else 1.0
                    if ss == -0.0:
                        continue
                else:
                    cand = old + ss
                if cand < 0 or cand > 1:
                    continue
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def perturb_fourier(h, n, rng, n_modes=3, amp=0.03):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def perturb_local(h, n, rng, width_frac=0.1):
    h_new = h.copy()
    w = max(2, int(n * width_frac * rng.uniform(0.5, 1.5)))
    start = rng.randint(0, n - w)
    h_new[start:start+w] += rng.normal(0, 0.05, w)
    return make_feasible(h_new, n)

def perturb_swap(h, n, rng, n_swaps=5):
    h_new = h.copy()
    target = n // 2
    for _ in range(n_swaps):
        ones = np.where(h_new >= 0.5)[0]
        zeros = np.where(h_new < 0.5)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h_new[i], h_new[j] = h_new[j], h_new[i]
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP with increasing temperature
    temps = [100, 400, 1500, 6000, 25000, 100000]
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        mi = 200 if T < 1000 else 150
        h_res = slsqp_solve(best_h, n0, T, mi)
        h_res = make_feasible(h_res, n0)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    # Phase 2: Active lag cycling
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        tl = min(remaining - 20, 100)
        h_al, c5_al = active_lag_slqp(best_h, n0, max_cycles=12, iters_per_lag=100, time_limit=tl)
        if c5_al < best_c5:
            best_c5 = c5_al
            best_h = h_al
            # Re-polish with SLSQP
            remaining = budget_s - (time.time() - start_time)
            if remaining > 40:
                h_res = slsqp_solve(best_h, n0, 50000, 150)
                h_res = make_feasible(h_res, n0)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res

    # Phase 3: Top-K SLSQP (focus on top correlations)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        for K in [5, 10, 20]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break
            h_res = top_k_slqp(best_h, n0, K, 500.0, 100)
            h_res = make_feasible(h_res, n0)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res

    # Phase 4: Perturbation + re-optimize
    perts = [perturb_fourier, perturb_local, perturb_swap, perturb_fourier, perturb_local]
    for pf in perts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 80:
            break
        try:
            hp = pf(best_h, n0, rng)
            # Quick SLSQP
            h_res = slsqp_solve(hp, n0, 2000, 100)
            h_res = make_feasible(h_res, n0)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
                remaining = budget_s - (time.time() - start_time)
                if remaining > 50:
                    h_res2 = slsqp_solve(best_h, n0, 50000, 150)
                    h_res2 = make_feasible(h_res2, n0)
                    c5_res2 = compute_c5(h_res2)
                    if c5_res2 < best_c5:
                        best_c5 = c5_res2
                        best_h = h_res2
        except Exception:
            pass

    # Phase 5: Coordinate descent (fine steps)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        ss_fine = [0.0, 1.0, -0.1, 0.1, -0.04, 0.04, -0.015, 0.015, -0.005, 0.005, -0.001, 0.001]
        max_rd = min(40, int((remaining - 10) / max(1, n_pts * 12 * 0.001)))
        best_h, best_c5 = coord_descent_fin(best_h, n_pts, max_rd, start_time, budget_s, ss_fine)

    # Phase 6: Higher resolution attempt
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n1 = n0 * 2
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        # Optimize at higher res
        for T in [500, 5000, 50000]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 40:
                break
            h_res = slsqp_solve(h_up, n1, T, 100)
            h_res = make_feasible(h_res, n1)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
        # Coord descent at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            n_pts = len(best_h)
            ss_fine = [-0.05, 0.05, -0.01, 0.01, -0.003, 0.003]
            max_rd = min(10, int((remaining - 10) / max(1, n_pts * 6 * 0.0005)))
            best_h, best_c5 = coord_descent_fin(best_h, n_pts, max_rd, start_time, budget_s, ss_fine)

    # Phase 7: Final polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        h_res = slsqp_solve(best_h, n_pts, 200000, 100)
        h_res = make_feasible(h_res, n_pts)
        c5_res = compute_c5(h_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = h_res

    remaining = budget_s - (time.time() - start_time)
    if remaining > 15:
        n_pts = len(best_h)
        ss_final = [-0.02, 0.02, -0.005, 0.005, -0.001, 0.001]
        best_h, best_c5 = coord_descent_fin(best_h, n_pts, 5, start_time, budget_s, ss_final)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
