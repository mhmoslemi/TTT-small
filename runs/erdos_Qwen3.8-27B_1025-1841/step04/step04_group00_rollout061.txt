def make_feasible(h, n):

        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)

def pair_flip_delta(h, i, j, n):
    """Δcorr when h[i]: 0→1, h[j]: 1→0."""
    k_arr = np.arange(2 * n - 1)
    d = np.zeros(2 * n - 1)
    idx1 = i + k_arr - (n - 1)
    v1 = (idx1 >= 0) & (idx1 < n)
    d[v1] += (1.0 - h[idx1[v1]])
    idx2 = i - k_arr + (n - 1)
    v2 = (idx2 >= 0) & (idx2 < n)
    d[v2] -= h[idx2[v2]]
    idx3 = j + k_arr - (n - 1)
    v3 = (idx3 >= 0) & (idx3 < n)
    d[v3] -= (1.0 - h[idx3[v3]])
    idx4 = j - k_arr + (n - 1)
    v4 = (idx4 >= 0) & (idx4 < n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    idx_sorted = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx_sorted[:n // 2]] = 1.0
    return b

def pair_flip_delta(h, i, j, n):
    k_arr = np.arange(2 * n - 1)
    d = np.zeros(2 * n - 1)
    idx1 = i + k_arr - (n - 1)
    v1 = (idx1 >= 0) & (idx1 < n)
    d[v1] += (1.0 - h[idx1[v1]])
    idx2 = i - k_arr + (n - 1)
    v2 = (idx2 >= 0) & (idx2 < n)
    d[v2] -= h[idx2[v2]]
    idx3 = j + k_arr - (n - 1)
    v3 = (idx3 >= 0) & (idx3 < n)
    d[v3] -= (1.0 - h[idx3[v3]])
    idx4 = j - k_arr + (n - 1)
    v4 = (idx4 >= 0) & (idx4 < n)
    d[v4] += h[idx4[v4]]
    return d

def grad_at_shift(h, s, n):
    grad = np.zeros(n)
    i_min = max(0, -s)
    i_max = min(n - 1, n - 1 - s)
    if i_min > i_max:
        return grad
    js = np.arange(i_min, i_max + 1) + s
    valid_js = (js >= 0) & (js < n)
    indices = np.arange(i_min, i_max + 1)
    grad[indices[valid_js]] += (1.0 - h[js[valid_js]])
    jm_start = max(0, i_min + s)
    jm_end = min(n - 1, i_max + s)
    if jm_start <= jm_end:
        jm = np.arange(jm_start, jm_end + 1)
        jms = jm - s
        valid_jms = (jms >= 0) & (jms < n)
        grad[jm[valid_jms]] -= h[jms[valid_jms]]
    return grad

def subgradient(h, n, top_m=5):

def slsqp_polish(h, n, budget_s, start_time, T=3000.0, top_m=None):
    dx = 2.0 / n
    corr = compute_corr(h)
    sorted_corr = np.sort(corr)[::-1]
    gap = max(sorted_corr[0] - sorted_corr[1], 1e-6)
    T_eff = max(T, 5.0 / gap)
    
    def obj(x):
        c = compute_corr(x)
        if top_m is not None and top_m < len(c):
            ti = np.argsort(c)[-top_m:]
            mc = np.max(c[ti])
            shifted = c[ti] - mc
            return (mc + np.log(np.sum(np.exp(T_eff * shifted))) / T_eff) * dx
        mc = np.max(c)
        shifted = c - mc
        return (mc + np.log(np.sum(np.exp(T_eff * shifted))) / T_eff) * dx
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-18})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h) - 1e-15:
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h, compute_c5(h)

            if c5_up < best_c5 + 0.003:
                if remaining() > 130:
                    h_up2, c5_up2 = pgd_optimize(h_up, n2, min(100, remaining() - 40), start_time, top_m=5)
                    if c5_up2 < best_c5:
                        best_h, best_c5 = h_up2, c5_up2
                if remaining() > 60:
                    h_up3, c5_up3 = slsqp_polish(best_h.copy(), len(best_h), min(50, remaining() - 20), start_time, T=3000)
                    if c5_up3 < best_c5:
                        best_h, best_c5 = h_up3, c5_up3
    
    # Phase 9: Final PGD + SLSQP
    if remaining() > 60:
        h9, c5_9 = pgd_optimize(best_h.copy(), len(best_h), min(40, remaining() - 20), start_time, top_m=3)
        if c5_9 < best_c5:
            best_h, best_c5 = h9, c5_9
    
    if remaining() > 30:
        h10, c5_10 = slsqp_polish(best_h.copy(), len(best_h), min(25, remaining() - 5), start_time, T=15000, top_m=3)
        if c5_10 < best_c5:
            best_h, best_c5 = h10, c5_10
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
