            break

        if directed and rng.random() < 0.6:
            max_idx = int(np.argmax(corr))
            max_k = max_idx - half
            i = ones[rng.randint(len(ones))]
            found = False
            for dk in [max_k, -max_k]:
                jj = i + dk
                if 0 <= jj < n and b[jj] == 0:
                    j = jj
                    found = True
                    break
            if not found:
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        if i == j:
            continue

        b[i] = 0.0
        b[j] = 1.0
        fast_update_corr(corr, b, i, j, n)
        new_val = float(np.max(corr) * dx)

        if new_val <= curr or (t > 1e-15 and rng.random() < np.exp(-(new_val - curr) / t)):
            curr = new_val
            if new_val < best:
                best = new_val
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
            fast_update_corr(corr, b, j, i, n)

    return best_b, best

def sa_block_fast(h, n, n_iter, T0, T1, rng, deadline):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    while s > target:
        idx = np.where(b == 1)[0]
        if len(idx) == 0: break
        b[idx[rng.randint(len(idx))]] = 0
        s -= 1
    while s < target:
        idx = np.where(b == 0)[0]
        if len(idx) == 0: break
        b[idx[rng.randint(len(idx))]] = 1
        s += 1

    dx = 2.0 / n
    corr = np.correlate(b, 1.0 - b, mode='full').astype(float)
    curr = float(np.max(corr) * dx)
    best = curr
    best_b = b.copy()

    for it in range(n_iter):
        if time.time() >= deadline:
            break
        frac = it / max(n_iter, 1)
        t = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, 9)
        start = rng.randint(0, n - bsize)
        end = start + bsize
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi), rng.randint(1, 4))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        old_vals = b[start:end].copy()
        half = n - 1
        for ci, cj in zip(co, cz):
            gi = start + ci
            gj = start + cj
            b[gi] = 0.0
            b[gj] = 1.0
            fast_update_corr(corr, b, gi, gj, n)
        new_val = float(np.max(corr) * dx)
        if new_val <= curr or (t > 1e-15 and rng.random() < np.exp(-(new_val - curr) / t)):
            curr = new_val
            if new_val < best:
                best = new_val
                best_b = b.copy()
        else:
            b[start:end] = old_vals
            for ci, cj in zip(co, cz):
                gi = start + ci
                gj = start + cj
                b[gj] = 0.0
                b[gi] = 1.0
                fast_update_corr(corr, b, gj, gi, n)

    return best_b, best

def coord_refine(h, n, max_rounds, deadline):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() >= deadline:
            break
        improved = False
        for j in range(n):
            if time.time() >= deadline - 5:
                break
            old = h[j]
            if old <= 0.01 or old >= 0.99:
                continue
            best_c = best_c5
            best_v = old
            for dv in [-0.05, -0.02, -0.005, 0.005, 0.02, 0.05]:
                cand = old + dv
                if cand < 0 or cand > 1:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_v = cand
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def multi_res_slsqp(h, n, start_t, budget_s):
    temps = [100.0, 400.0, 1500.0, 6000.0, 25000.0, 80000.0]
    best_h = h.copy()
    best_c = compute_c5(h)
    for T in temps:
        rem = budget_s - (time.time() - start_t)
        if rem < 35:
            break
        try:
            x = do_slsqp(best_h, n, T, maxiter=200)
            c = compute_c5(x)
            if c < best_c:
                best_c = c
                best_h = make_feasible(x, n)
        except Exception:
            pass
    return best_h, best_c

def perturb_fourier(h, n, rng, n_modes=3, amp=0.04):
    x = np.arange(n)
    hnew = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 5)
        phase = rng.uniform(0, 2 * np.pi)
        hnew += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(hnew, n)

def perturb_segment(h, n, rng):
    hnew = h.copy()
    nseg = rng.randint(1, 4)
    for _ in range(nseg):
        length = rng.randint(n // 15, n // 4)
        start = rng.randint(0, max(1, n - length))
        if rng.random() < 0.5:
            hnew[start:start+length] = 1.0 - hnew[start:start+length]
        else:
            hnew[start:start+length] = rng.uniform(0.2, 0.8)
    return make_feasible(hnew, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Multi-temperature SLSQP
    rem = budget_s - (time.time() - start_time)
    if rem > 150:
        best_h, best_c5 = multi_res_slsqp(best_h, n0, start_time, budget_s)

    # Phase 2: Fast binary SA with directed moves
    rem = budget_s - (time.time() - start_time)
    if rem > 120:
        dl = start_time + min(rem - 20, 100)
        b, c5b = sa_fast(best_h, n0, 500000, 0.004, 0.000005, rng, dl, directed=True)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()
            rem = budget_s - (time.time() - start_time)
            if rem > 60:
                best_h, best_c5 = multi_res_slsqp(best_h, n0, start_time, budget_s)

    # Phase 3: Perturbations + SLSQP
    perts = [perturb_fourier, perturb_segment]
    for pf in perts:
        rem = budget_s - (time.time() - start_time)
        if rem < 80:
            break
        for _ in range(2):
            rem = budget_s - (time.time() - start_time)
            if rem < 50:
                break
            try:
                hp = pf(best_h, n0, rng)
                hr, cr = multi_res_slsqp(hp, n0, start_time, budget_s)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 4: Another round of SA
    rem = budget_s - (time.time() - start_time)
    if rem > 100:
        dl = start_time + min(rem - 20, 90)
        b, c5b = sa_fast(best_h, n0, 400000, 0.003, 0.000003, rng, dl, directed=True)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()
            rem = budget_s - (time.time() - start_time)
            if rem > 50:
                best_h, best_c5 = multi_res_slsqp(best_h, n0, start_time, budget_s)

    # Phase 5: Block SA
    rem = budget_s - (time.time() - start_time)
    if rem > 80:
        dl = start_time + min(rem - 15, 70)
        b, c5b = sa_block_fast(best_h, n0, 200000, 0.003, 0.00001, rng, dl)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()
            rem = budget_s - (time.time() - start_time)
            if rem > 50:
                best_h, best_c5 = multi_res_slsqp(best_h, n0, start_time, budget_s)

    # Phase 6: Try higher resolution
    rem = budget_s - (time.time() - start_time)
    if rem > 250:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        rem = budget_s - (time.time() - start_time)
        if rem > 100:
            h_up, c5_up = multi_res_slsqp(h_up, n1, start_time, budget_s)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
            rem = budget_s - (time.time() - start_time)
            if rem > 60:
                dl = start_time + min(rem - 10, 50)
                b2, c5b2 = sa_fast(best_h, len(best_h), 300000, 0.003, 0.000005, rng, dl, directed=True)
                if c5b2 < best_c5:
                    best_c5 = c5b2
                    best_h = b2.copy()

    # Phase 7: Coordinate descent on fractional values
    rem = budget_s - (time.time() - start_time)
    if rem > 30:
        n_pts = len(best_h)
        dln = start_time + min(rem - 5, 25)
        best_h, best_c5 = coord_refine(best_h.copy(), n_pts, 10, dln)

    # Phase 8: Final SLSQP polish
    rem = budget_s - (time.time() - start_time)
    if rem > 40:
        n_pts = len(best_h)
        try:
            x = do_slsqp(best_h, n_pts, 50000.0, maxiter=200)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(x, n_pts)
        except Exception:
            pass

    # Final check
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
