```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    # LogSumExp
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    p = exp_corr / np.sum(exp_corr)
```

```python
    """
    h: array of length n
    p: probability weights, length 2n-1, p[idx] corresponds to shift idx-(n-1)
    Returns gradient of the LogSumExp objective (without the 1/beta factor)
    """

    grad = np.zeros(n)
    
    # For each shift s, the contribution to grad[j] is -p_s * h[j+s] (if valid)
    # Combined with the symmetric term, it's -(p_s + p_{-s}) * h[j+s]
    
    # Build the combined kernel
    # kernel[s] = p[s+(n-1)] + p[-s+(n-1)] = p[n-1+s] + p[n-1-s]
    # For s = 0: kernel[0] = 2*p[n-1]
    # For s > 0: kernel[s] = p[n-1+s] + p[n-1-s]
    
    # Then grad[j] = 1 - sum_{s=-n+1}^{n-1} kernel[s] * h[j+s] * 1_{valid}
    # Due to symmetry of kernel: = 1 - kernel[0]*h[j] - sum_{s=1}^{n-1} kernel[s]*(h[j+s]*1_{j+s<n} + h[j-s]*1_{j-s>=0})
    
    kernel = np.zeros(n)  # kernel[s] for s=0,...,n-1
    for s in range(n):
        kernel[s] = p[n-1+s] + p[n-1-s]
    
    # Now compute result[j] = sum_{s=0}^{n-1} kernel[s] * h[j+s] * 1_{valid}
    # For s=0: contributes kernel[0] * h[j] to all j
    # For s=1: contributes kernel[1] * h[j+1] to j=0,...,n-2
    # For s=k: contributes kernel[k] * h[j+k] to j=0,...,n-k
    # Also symmetric: kernel[s] * h[j-s] to j=s,...,n-1
    
    result = np.zeros(n)
    result += kernel[0] * h
    for s in range(1, n):
        if kernel[s] < 1e-15:
            continue
        result[:n-s] += kernel[s] * h[s:]
        result[s:] += kernel[s] * h[:n-s]
    
    grad = 1.0 - result
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {x : 0<=x_i<=1, sum(x)=n/2}"""
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0.0, 1.0)
    return x

def make_feasible(h, n):
    """Ensure h satisfies constraints exactly (up to floating point)"""
    h = project_feasible(h, n)
    # Final fix
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def logsumexp_corr(h, beta):
    """Compute logsumexp of correlation values"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    val = (max_c + np.log(np.sum(np.exp(beta * (corr - max_c)))) / beta)
    return val

def gradient_logsumexp(h, beta):
    """Compute gradient of logsumexp(corr) * dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    p = exp_c / exp_c.sum()
    
    # Build kernel K of length 2n-1
    # K[n-1+s] = p[n-1+s] + p[n-1-s]
    K = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        K[idx] = p[idx] + p[2 * (n - 1) - idx]
    
    # result[j] = sum_s K[n-1+s] * h[j+s] * 1_valid
    # = (K * h)[n-1+j] (cross-correlation via convolution)
    # Using FFT for speed
    size = 1
    while size < n + 2 * n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    Kf = np.fft.rfft(K, size)
    conv = np.fft.irfft(H * Kf, size)
    result = conv[n - 1:n - 1 + n]
    
    grad = (1.0 - result) * dx
    return grad

        # Penalty for sum constraint
        penalty = 1e8 * (np.sum(x) - n / 2.0) ** 2

        return g

    res = minimize(obj, h, method='L-BFGS-B', jac=grad, bounds=bounds,
                   options={'maxiter': 1000, 'ftol': 1e-18, 'gtol': 1e-12,
                            'maxfun': 5000, 'disp': False})

    c5 = compute_c5(h_out)
    return h_out, c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-10) & (x < 1.0 - 1e-10)
        cnt = np.sum(mask)
        if cnt > 0:
            x[mask] += diff / cnt
            x = np.clip(x, 0.0, 1.0)
    return x

    h = project_feasible(h, n)
    diff = n / 2.0 - np.sum(h)

        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_corr_value(h, beta):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return max_c + np.log(np.sum(np.exp(beta * (corr - max_c)))) / beta

def lse_gradient(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(beta * (corr - max_c))
    p = exp_c / exp_c.sum()
    K = p + p[::-1]
    size = 1
    while size < n + 2 * n - 1:
        size <<= 1
    H = np.fft.rfft(h, size)
    Kf = np.fft.rfft(K, size)

    result = conv[n - 1:n - 1 + n]
    return (1.0 - result) * dx

    g = lse_gradient(x, beta)
    g += lam * 2.0 * (np.sum(x) - n / 2.0) * np.ones(n)
    return g

                       method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-18,
                                'gtol': 1e-12, 'maxfun': max_iter * 3})
        h_out = make_feasible(res.x, n)
        c5 = compute_c5(h_out)
        return h_out, c5
    except Exception:
        return h, compute_c5(h)
```
