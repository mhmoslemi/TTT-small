def subgradient_optimize(h, n, budget_s, start_time, temp_init=30.0, temp_end=200.0):
    """Subgradient method with Nesterov acceleration and adaptive stepping."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05
    momentum = 0.7
    velocity = np.zeros(n)
    no_improve = 0
    max_no_improve = 50
    it = 0

        it += 1
        corr = compute_corr(h)
        c5_cur = np.max(corr) * (2.0/n)
        
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step *= 0.5
                if step < 1e-7:
                    break
                no_improve = 0
                momentum = max(0.3, momentum * 0.8)

        progress = min(1.0, it / 500.0)
        temp = temp_init + (temp_end - temp_init) * progress

        y = np.clip(y, 0, 1)

        velocity = momentum * velocity - step * grad_y

        new_h = project_feasible(new_h, n)

        if c5_new <= c5_cur + 1e-15:

        else:

            else:

            if step < 1e-8:
                break

    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-15:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_subgradient(h, corr, n, temp=50.0):
    max_c = np.max(corr)
    diff = np.clip(corr - max_c, -100.0 / temp, 0.0)
    w = np.exp(diff * temp)
    total_w = np.sum(w)
    if total_w < 1e-30:
        active = np.where(corr >= max_c - 1e-10)[0]
        w = np.zeros(len(corr))
        w[active] = 1.0 / len(active)
    else:
        w /= total_w
    grad = np.zeros(n)
    n_half = n - 1
    threshold = 1e-7
    active_k = np.where(w > threshold)[0]
    for k in active_k:
        s = k - n_half
        wk = w[k]
        if s >= 0:
            grad[0:n-s] += wk * (1.0 - h[s:n])
            grad[s:n]   -= wk * h[0:n-s]
        else:
            t = -s
            grad[t:n]     += wk * (1.0 - h[0:n-t])
            grad[0:n-t]  -= wk * h[t:n]
    return grad

def subgradient_optimize(h, n, budget_s, start_time, temp_init=30.0, temp_end=200.0):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.04
    momentum = 0.6
    velocity = np.zeros(n)
    no_improve = 0
    max_no_improve = 60
    it = 0
    while time.time() - start_time < budget_s:
        it += 1
        corr = compute_corr(h)
        c5_cur = float(np.max(corr) * (2.0 / n))
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step *= 0.5
                if step < 1e-7:
                    break
                no_improve = 0
                momentum = max(0.3, momentum * 0.85)
        progress = min(1.0, it / 400.0)
        temp = temp_init + (temp_end - temp_init) * progress
        grad = soft_subgradient(h, corr, n, temp)
        y = h - momentum * velocity
        y = np.clip(y, 0, 1)
        corr_y = compute_corr(y)
        grad_y = soft_subgradient(y, corr_y, n, temp)
        velocity = momentum * velocity - step * grad_y
        new_h = project_feasible(h + velocity, n)
        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur + 1e-14:
            h = new_h
        else:
            new_h2 = project_feasible(h - step * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_cur + 1e-14:
                h = new_h2
                velocity *= 0.4
            else:
                step *= 0.7
                velocity *= 0.3
            if step < 1e-8:
                break
        if it % 80 == 0 and step < 0.03:
            step = min(0.05, step * 1.4)
        if time.time() - start_time > budget_s - 2:
            break
    h = project_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h, best_c5 = h, c5_h
    return best_h, best_c5

def targeted_local_search(h, n, budget_s, start_time, max_rounds=500):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng_local = np.random.RandomState(12345)
    rounds = 0
    while rounds < max_rounds and time.time() - start_time < budget_s:
        rounds += 1
        corr = compute_corr(best_h)
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        improved = False
        for _ in range(30):
            if time.time() - start_time > budget_s - 1:
                break
            if s >= 0:
                n_valid = n - s
                candidates = np.where((best_h[0:n_valid] > 0.3) & (best_h[s:n] < 0.7))[0]
                if len(candidates) == 0:
                    break
                j = candidates[rng_local.randint(len(candidates))]
                trial = best_h.copy()
                amt = min(best_h[j] * 0.8, (1.0 - best_h[j + s]) * 0.8)
                if amt < 0.005:
                    continue
                trial[j] -= amt
                trial[j + s] += amt
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c
                    improved = True
            else:
                t = -s
                candidates = np.where((best_h[t:n] > 0.3) & (best_h[0:n-t] < 0.7))[0]
                if len(candidates) == 0:
                    break
                j = candidates[rng_local.randint(len(candidates))]
                trial = best_h.copy()
                amt = min(best_h[j] * 0.8, (1.0 - best_h[j - t]) * 0.8)
                if amt < 0.005:
                    continue
                trial[j] -= amt
                trial[j - t] += amt
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c
                    improved = True
        if not improved and rounds > 50:
            break
    return best_h, best_c5

def slsqp_logsumexp(h, n, T, maxiter=200, budget_s=120, start_time=None):
    if start_time is None:
        start_time = time.time()
    dx = 2.0 / n
    def obj(x):
        corr_x = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr_x)
        shifted = corr_x - mc
        safe = np.clip(shifted, -100.0/T, 0.0)
        return (mc + np.log(np.sum(np.exp(T * safe))) / T) * dx
    def con(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'maxls': 40})
        c5_res = compute_c5(res.x)
        if c5_res < best_c5:
            best_h = make_feasible(res.x, n)
            best_c5 = compute_c5(best_h)
    except Exception:
        pass
    if time.time() - start_time > budget_s:
        return best_h, best_c5
    return best_h, best_c5

def try_binary_snap(h, n):
    thresh = 0.5
    b = (h >= thresh).astype(float)
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        order = np.argsort(h)
        to_flip = [i for i in order if b[i] == 1][:n_ones - target]
        for i in to_flip:
            b[i] = 0
    elif n_ones < target:
        order = np.argsort(-h)
        to_flip = [i for i in order if b[i] == 0][:target - n_ones]
        for i in to_flip:
            b[i] = 1
    b = make_feasible(b, n)
    return b, compute_c5(b)

def upsample_optimize(h, n, n2, budget_s, start_time):
    old_x = np.linspace(0.0, 2.0, n)
    new_x = np.linspace(0.0, 2.0, n2)
    h_up = np.interp(new_x, old_x, h)
    h_up = make_feasible(h_up, n2)
    c5_up = compute_c5(h_up)
    if budget_s > 60:
        h_up, c5_up = subgradient_optimize(h_up, n2, min(80, budget_s), start_time,
                                            temp_init=40, temp_end=150)
    if budget_s > 40:
        h_up, c5_up = slsqp_logsumexp(h_up, n2, T=800, maxiter=100,
                                       budget_s=budget_s, start_time=start_time)
    return h_up, c5_up

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Subgradient with low-to-mid temperature
    if remaining() > 200:
        h, c5 = subgradient_optimize(best_h, n, min(220, remaining() - 40), start_time,
                                      temp_init=25.0, temp_end=120.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Targeted local search on worst shifts
    if remaining() > 100:
        h, c5 = targeted_local_search(best_h, n, min(120, remaining() - 30), start_time, max_rounds=300)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: SLSQP with log-sum-exp (moderate T)
    if remaining() > 100:
        h, c5 = slsqp_logsumexp(best_h, n, T=400, maxiter=150,
                                 budget_s=min(100, remaining()-20), start_time=start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Subgradient with high temperature (balance all shifts)
    if remaining() > 120:
        h, c5 = subgradient_optimize(best_h, n, min(140, remaining() - 30), start_time,
                                      temp_init=100.0, temp_end=400.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Targeted local search again
    if remaining() > 80:
        h, c5 = targeted_local_search(best_h, n, min(80, remaining() - 20), start_time, max_rounds=200)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: SLSQP with high T
    if remaining() > 80:
        h, c5 = slsqp_logsumexp(best_h, n, T=1000, maxiter=150,
                                 budget_s=min(80, remaining()-20), start_time=start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Try upsample to 2n
    if remaining() > 350:
        n2 = n * 2
        h_up, c5_up = upsample_optimize(best_h, n, n2, min(remaining()-60, 350), start_time)
        if c5_up < best_c5:
            best_h, best_c5 = h_up, c5_up

    # Phase 8: Random restarts with perturbation
    for restart in range(4):
        if remaining() < 70:
            break
        scale = 0.015 * (restart + 1)
        n_pts = len(best_h)
        h_try = make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
        h2, c5_2 = subgradient_optimize(h_try, n_pts, min(50, remaining()-30), start_time,
                                         temp_init=30, temp_end=150)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        h3, c5_3 = targeted_local_search(best_h, len(best_h), min(20, remaining()-20),
                                          start_time, max_rounds=50)
        if c5_3 < best_c5:
            best_h, best_c5 = h3, c5_3

    # Phase 9: Final SLSQP polish
    if remaining() > 40:
        h, c5 = slsqp_logsumexp(best_h, len(best_h), T=2000, maxiter=150,
                                 budget_s=min(40, remaining()-10), start_time=start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 10: Final subgradient + local search
    if remaining() > 30:
        n_pts = len(best_h)
        h, c5 = subgradient_optimize(best_h, n_pts, min(25, remaining()-15), start_time,
                                      temp_init=20.0, temp_end=200.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        h, c5 = targeted_local_search(best_h, n_pts, min(10, remaining()-5), start_time, max_rounds=50)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Try binary snap as a last resort
    b_snap, c5_snap = try_binary_snap(best_h, len(best_h))
    if c5_snap < best_c5:
        best_h, best_c5 = b_snap, c5_snap

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
