import numpy as np
from scipy.optimize import minimize

def compute_c5_exact(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def circular_corr_fft(h, b):
    """Compute C[k] = sum_i h[i] * b[(i+k) mod n] via FFT."""
    n = len(h)
    H = np.fft.rfft(h)
    B = np.fft.rfft(b)
    return np.fft.irfft(H * np.conj(B), n)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def nesterov_smoothed_subgrad(h0, n, T, max_iters, lr_base, momentum, budget_fn):
    """Optimize using Nesterov-accelerated subgradient on smoothed objective."""
    h = h0.copy().astype(float)
    v = np.zeros(n)  # momentum
    ones = np.ones(n)
    
    def project(x):
        x = np.clip(x, 0.0, 1.0)
        diff = n/2.0 - x.sum()
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = x < 1.0 - 1e-8
            else:
                mask = x > 1e-8
            if mask.any():
                x[mask] = np.clip(x[mask] + diff/mask.sum(), 0.0, 1.0)
        return x
    
    best_c5 = compute_c5_exact(h)
    
    for it in range(max_iters):
        if budget_fn() < 5:
            break
        
        # Compute g_k values via FFT
        b = 1.0 - h
        H = np.fft.rfft(h)
        B = np.fft.rfft(b)
        g = np.fft.irfft(H * np.conj(B), n)
        
        # Smoothing
        g_max = np.max(g)
        logits = T * (g - g_max)
        logits = np.clip(logits, -700, 700)
        log_sum = g_max * T + np.log(np.sum(np.exp(logits)))
        S = log_sum / T  # smoothed objective value (before * dx)
        
        # Softmax weights
        p = np.exp(logits - np.log(np.sum(np.exp(logits))))
        
        # Gradient computation via FFT
        P = np.fft.rfft(p)
        # Term 1: sum_k p[k] * h[(j+k) mod n] = ifft(fft(p) * conj(fft(h)))
        term1 = np.fft.irfft(P * np.conj(H), n)
        # Term 2: sum_k p[k] * h[(j-k) mod n] = ifft(fft(p) * fft(h)) = circular convolution
        term2 = np.fft.irfft(P * H, n)
        
        grad = ones - term1 - term2
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Step size
        lr = lr_base / np.sqrt(it + 1)
        
        # Nesterov step
        h_corr = h - momentum * v  # look-ahead
        b_corr = 1.0 - h_corr
        H_c = np.fft.rfft(h_corr)
        B_c = np.fft.rfft(b_corr)
        g_c = np.fft.irfft(H_c * np.conj(B_c), n)

        term1_c = np.fft.irfft(P_c * np.conj(H_c), n)
        term2_c = np.fft.irfft(P_c * H_c, n)
        grad_c = ones - term1_c - term2_c
        
        v = momentum * v - lr * grad_c
        h = project(h + v)
        
        # Update best
        c5 = compute_c5_exact(h)
        if c5 < best_c5:
            best_c5 = c5
    
    return make_feasible(h, n), best_c5

        h_y = np.clip(h_y, 0.0, 1.0)
        b_y = 1.0 - h_y
        Hf = np.fft.rfft(h_y)
        Bf = np.fft.rfft(b_y)
        g = np.fft.irfft(Hf * np.conj(Bf), n)
        g_max = np.max(g)
        logits = T * (g - g_max)
        logits = np.clip(logits, -700, 700)
        log_sum = np.log(np.sum(np.exp(logits)))
        p = np.exp(logits - log_sum)
        Pf = np.fft.rfft(p)
        term1 = np.fft.irfft(Pf * np.conj(Hf), n)
        term2 = np.fft.irfft(Pf * Hf, n)
        grad = ones_n - term1 - term2
        v = momentum * v - lr * grad
        h = project_feasible(h + v, n)
        if (it + 1) % check_interval == 0:
            c5 = compute_c5_exact(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
                if no_improve > 15:
                    break
    c5 = compute_c5_exact(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return make_feasible(best_h, n), best_c5

def lbgfb_smoothed(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -80, 80)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=40, tries=4000, rem_fn=None):
    h = h_bin.copy()
    cur = compute_c5_exact(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        if rem_fn is not None and rem_fn() < 10:
            break
        improved = False
        n_tries = min(tries, max(1, len(ones) * len(zeros)))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5_exact(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_h = h0.copy()
    best_c5 = compute_c5_exact(h0)
    best_n = len(h0)

    # Phase 1: FFT subgradient at parent resolution
    n1 = best_n
    h_curr = best_h.copy()
    for T in [100, 300, 1000, 3000, 10000]:
        if rem() < 30:
            break
        lr_val = 0.8 / T
        h_new, c5_new = fft_subgrad_optimize(
            h_curr, n1, T=T, max_iters=3000, lr=lr_val, momentum=0.9,
            rem_fn=rem, check_interval=100)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new.copy()

    # L-BFGS-B polish at parent resolution
    if rem() > 40:
        for T in [500, 2000]:
            if rem() < 20:
                break
            h_opt = lbgfb_smoothed(best_h, n1, T, mu=1000.0, max_iter=150)
            c_opt = compute_c5_exact(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 2: Higher resolution with FFT subgradient
    for n2 in [1280, 2560]:
        if rem() < 100:
            break
        h_up = upsample_h(best_h, n2)
        c5_up = compute_c5_exact(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        h_curr = h_up.copy()
        for T in [300, 1000, 5000, 20000]:
            if rem() < 40:
                break
            lr_val = 0.8 / T
            h_new, c5_new = fft_subgrad_optimize(
                h_curr, n2, T=T, max_iters=2000, lr=lr_val, momentum=0.9,
                rem_fn=rem, check_interval=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new.copy()

    # Phase 3: N=4096 if time permits
    if rem() > 120:
        n3 = 4096
        h_up = upsample_h(best_h, n3)
        c5_up = compute_c5_exact(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        h_curr = h_up.copy()
        for T in [500, 3000, 10000]:
            if rem() < 50:
                break
            lr_val = 0.8 / T
            h_new, c5_new = fft_subgrad_optimize(
                h_curr, n3, T=T, max_iters=1500, lr=lr_val, momentum=0.9,
                rem_fn=rem, check_interval=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new.copy()

    # Phase 4: Multi-start at n=1280
    if rem() > 60:
        n_ms = 1280
        h_base = upsample_h(best_h, n_ms)
        for restart in range(5):
            if rem() < 25:
                break
            noise_scale = 0.06 + 0.03 * restart
            h_r = h_base + noise_scale * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            T = 300 * (1 + restart * 2)
            h_new, c5_new = fft_subgrad_optimize(
                h_r, n_ms, T=T, max_iters=1500, lr=0.8/T, momentum=0.9,
                rem_fn=rem, check_interval=100)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    # Phase 5: Binary local search
    if rem() > 50:
        n_bin = min(len(best_h), 1024)
        h_b = upsample_h(best_h, n_bin) if n_bin < len(best_h) else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        target_ones = n_bin // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target_ones - n_ones]] = 1.0
        h_bs, c_bs = binary_local_search(h_bin, n_bin, rng, max_rounds=30, tries=5000, rem_fn=rem)
        if c_bs < best_c5:
            best_c5 = c_bs
            best_h = h_bs.copy()

    # Phase 6: Final L-BFGS-B polish at moderate n
    if rem() > 25:
        n_fin = min(len(best_h), 1280)
        h_fin = upsample_h(best_h, n_fin) if n_fin < len(best_h) else best_h.copy()
        for T in [2000, 10000]:
            if rem() < 12:
                break
            h_opt = lbgfb_smoothed(h_fin, n_fin, T, mu=800.0, max_iter=120)
            c_opt = compute_c5_exact(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 7: One more FFT round at final best
    if rem() > 15:
        n_last = len(best_h)
        h_new, c5_new = fft_subgrad_optimize(
            best_h, n_last, T=50000, max_iters=500, lr=0.5/50000.0,
            momentum=0.85, rem_fn=rem, check_interval=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5_exact(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
