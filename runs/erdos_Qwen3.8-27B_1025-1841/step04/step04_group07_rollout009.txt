def lbgfb_smooth(h0, n, T, mu=500.0, max_iter=300):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -60, 60)
        return float(mc + np.log(np.sum(np.exp(shifted))) / T) * dx + mu * (x.sum() - n / 2.0) ** 2

    try:
        res = minimize(obj, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
        return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=40, tries=6000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries, max(1, len(ones) * len(zeros)))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

        return float(mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    def con_eq(x):
        return np.sum(x) - n / 2.0

        return np.ones(n)
