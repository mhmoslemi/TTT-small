```python
    h = h.copy()
    t0 = time.time()
```

```python
def compute_c5_fast_binary(b, n):
    """Fast C5 computation for binary b using FFT autocorrelation."""
    dx = 2.0 / n
    # Autocorrelation via FFT
    # A(k) = sum_i b[i]*b[i+k] for valid i
    # This is the autocorrelation of b
    N = 2 * n  # FFT size
    B = np.fft.rfft(b, N)
    auto = np.fft.irfft(B * np.conj(B), N)[:n]
    # auto[k] = sum_i b[i]*b[i-k] for k=0..n-1
    # We need A(k) for k = -(n-1)..(n-1)
    # A(k) for k>=0 is auto[k], A(k) for k<0 is auto[-k] (by symmetry)
    # H(k) for k>=0: sum of b[0:n-k], for k<0: sum of b[-k:n]
    H = np.zeros(2*n-1)
    ps = np.concatenate(([0], np.cumsum(b)))
    for k in range(n):
        H[n-1+k] = ps[n-k] - ps[0]  # sum b[0:n-k]
        if k > 0:
            H[n-1-k] = ps[n] - ps[k]  # sum b[k:n]
    # corr[n-1+k] = H[n-1+k] - auto[abs(k)]
    corr = H - np.concatenate([auto[n-1:0:-1], auto])
    return np.max(corr) * dx
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

        j = zeros[rng.randint(len(zeros))]
```

```python
def subgrad_refine(h, n, max_iters, alpha0, time_limit, top_k=5, T=50):
    h = h.copy()
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top K lags
        top_idx = np.argsort(corr)[-top_k:][::-1]
        top_corr = corr[top_idx]
        # Weights
        w = np.exp(T * (top_corr - top_corr[0]))
        w /= np.sum(w)
        # Gradient
        g = np.zeros(n)
        for wf, k_full in zip(w, top_idx):
            k = k_full - (n - 1)
            if k >= 0:
                g[:n-k] += wf * (1.0 - h[k:])
                g[k:] -= wf * h[:n-k]
            else:
                kp = -k
                g[kp:] += wf * (1.0 - h[:n-kp])
                g[:n-kp] -= wf * h[kp:]
        alpha = alpha0 / np.sqrt(1.0 + it)
        h_new = h - alpha * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5
```
