from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = h.copy()
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def gradient_top_shifts(h, top_idx, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        grad += _shift_grad(h, s, n)
    return (grad / len(top_idx)) * dx

def _shift_grad(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        # (1-h[j-s]) for j in [s, n-1]
        g[s:] += (1.0 - h[:n-s])
        # -h[j+s] for j in [0, n-1-s]
        g[:n-s] -= h[s:]
    else:
        ss = -s
        # (1-h[j-s]) = (1-h[j+ss]) for j in [0, n-1-ss]
        g[:n-ss] += (1.0 - h[ss:])
        # -h[j+s] = -h[j-ss] for j in [ss, n-1]
        g[ss:] -= h[:n-ss]
    return g

        else:

def slsqp_smooth(h, n, t0, budget, T=1000.0, top_m=8):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m and top_m < len(corr):
            idx = np.argsort(corr)[-top_m:]
            mc = np.max(corr[idx])
            sh = corr[idx] - mc
            return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx
        mc = np.max(corr)
        sh = corr - mc
        sh = np.clip(sh, -100.0, 0.0)
        return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx
    def con_f(x):
        return np.sum(x) - n / 2.0
    def con_j(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-18})
        h_res = project_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        if c5_res < compute_c5(h):
            return h_res, c5_res
    except Exception:
        pass
    return h, compute_c5(h)

def fine_coord(h, n, t0, budget):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    steps = [0.005, 0.002, 0.001, 0.0005]
    improved = True
    while improved and time.time() - t0 < budget:
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 2:
                break
            old = best_h[j]
            best_c = best_c5
            best_v = old
            for st in steps:
                for sgn in [-1.0, 1.0]:
                    v = old + sgn * st
                    if 0.0 <= v <= 1.0 and abs(v - old) > 1e-12:
                        tr = best_h.copy()
                        tr[j] = v
                        tr = project_feasible(tr, n)
                        c = compute_c5(tr)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_v = v
            for v_ext in [0.0, 1.0]:
                if abs(v_ext - old) > 1e-6:
                    tr = best_h.copy()
                    tr[j] = v_ext
                    tr = project_feasible(tr, n)

                        best_v = v_ext
            if best_v != old:
                best_h[j] = best_v
                best_h = project_feasible(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5
