    bounds = [(0.0, 1.0)] * b
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,
             'jac': lambda x: np.ones(b)}]
    
    try:
        res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or res.fun < obj(x0):
            return res.x
    except Exception:
        pass
    return x0

def binary_pair_flip(h, n):
    """Try all pairs of binary flips to reduce C5."""
    # This only works well if h is close to binary
    # Find positions to try flipping
    zeros = np.where(h < 0.5)[0]
    ones = np.where(h >= 0.5)[0]
    
    if len(zeros) == 0 or len(ones) == 0:
        return h.copy(), compute_c5(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # For each zero position, try flipping to 1 and flipping a one to 0
    # But this is O(n^2), so we need to be selective
    # Focus on the worst shifts
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    dx = 2.0 / n
    
    # Find the worst shift
    worst_idx = np.argmax(corr)
    worst_lag = worst_idx - (n - 1)
    
    # For the worst shift, we want to reduce corr[worst_idx]
    # corr[worst_idx] = Σ_i h[i+lag] * (1-h[i])
    # To reduce it, we can:
    # 1. Set h[i+lag] = 0 for some i where h[i+lag]=1 and h[i]=0
    # 2. Set h[i] = 1 for some i where h[i]=0 and h[i+lag]=1
    
    # Strategy: for the worst shift, find a position i where h[i+lag]=1 and h[i]=0
    # Flip h[i+lag] to 0 (reduces corr at worst shift)
    # Find a position j where h[j]=0 and h[j-lag]=0 (to compensate sum)
    # Flip h[j] to 1
    
    # This is getting complicated. Let me just do a simpler local search.
    return best_h, best_c5
