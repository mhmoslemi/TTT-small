def adam_soft_max_optimize(h, n, T_start, T_end, n_iters, lr, start_time, budget_s):
    """Adam optimization of soft-max objective."""
    dx = 2.0 / n
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        if time.time() - start_time > budget_s - 30:
            break
        
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters - 1))
        
        # Compute shifts and weights
        shifts = compute_all_shifts(h)
        max_s = np.max(shifts)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_shifts(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, T):
    """Compute gradient of soft-max objective."""
    n = len(h)
    dx = 2.0 / n
    shifts = compute_shifts(h)
    max_s = np.max(shifts)
    exp_s = np.exp(T * (shifts - max_s))
    w = exp_s / np.sum(exp_s)
    
    # S1[j] = np.correlate(w, 1-h, 'full')[2n-2-j]
    u = 1.0 - h
    corr1 = np.correlate(w, u, 'full')
    S1 = corr1[2*n-2 : n-1 : -1]  # indices 2n-2, 2n-3, ..., n-1
    
    # S2[j] = np.convolve(h, w, 'full')[n-1+j]
    conv2 = np.convolve(h, w, 'full')
    S2 = conv2[n-1 : 2*n-1]  # indices n-1, n, ..., 2n-2
    
    grad = dx * (S1 - S2)
    # Project to maintain sum constraint
    grad = grad - np.mean(grad)
    return grad, float(np.max(shifts))

def adam_optimize(h, n, T_schedule, n_iters, lr, start_time, budget_s):
    """Adam optimization with temperature schedule."""
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    best_c5_patience = best_c5
    
    for it in range(n_iters):
        if time.time() - start_time > budget_s - 30:
            break
        
        T = T_schedule[min(it, len(T_schedule)-1)]
        
        grad, c5 = compute_grad(h, T)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
        
        # Adaptive learning rate: reduce if no improvement
        if patience > 200:
            lr *= 0.5
            patience = 0
            if lr < 1e-6:
                break
        
        h = h_new
    
    return best_h, best_c5
