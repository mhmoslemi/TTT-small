        h[mask] = np.clip(h[mask] + diff / np.sum(mask), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

# Global vars for nested functions
_G = {}

def _smooth_obj(h):
    n = _G['n']
    T = _G['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Clip for numerical stability
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (mc + log_sum / T) * dx

def _smooth_grad(h):
    n = _G['n']
    T = _G['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -500, 500)
    exp_vals = np.exp(shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    # Gradient of corr[j] w.r.t. h[k]
    # corr[j] = sum_i h[i] * (1-h[i+j-(n-1)]) for valid i
    # Let p_j = j - (n-1)
    p = np.arange(2*n-1) - (n-1)
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    for j in np.where(weights > 1e-8)[0]:
        w = weights[j]
        lag = p[j]
        k = np.arange(n)
        mask1 = (k - lag >= 0) & (k - lag < n)
        mask2 = (k + lag >= 0) & (k + lag < n)
        grad[mask1] += w * (1.0 - h[k[mask1] - lag])
        grad[mask2] -= w * h[k[mask2] + lag]
    grad *= dx
    return grad

def _cons_eq(h):
    return np.sum(h) - _G['n'] / 2.0

def _cons_jac(h):
    return np.ones(_G['n'])

def slsqp_opt(h0, n, T, max_iter=200):
    _G['n'] = n
    _G['T'] = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    res = minimize(_smooth_obj, h0, jac=_smooth_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

_G = {}

def _obj(h):
    n, T = _G['n'], _G['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = np.clip(T * (corr - mc), -500.0, 0.0)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

def _grad(h):
    n, T = _G['n'], _G['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = np.clip(T * (corr - mc), -500.0, 0.0)
    ew = np.exp(sh)
    w = ew / ew.sum()
    p = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    nw = min(60, int(np.sum(w > 1e-12)))
    top = np.argsort(w)[::-1][:nw]
    for j in top:
        a = w[j]
        lag = int(p[j])
        k = np.arange(n)
        m1 = (k - lag >= 0) & (k - lag < n)
        m2 = (k + lag >= 0) & (k + lag < n)
        if np.any(m1):
            grad[m1] += a * (1.0 - h[k[m1] - lag])
        if np.any(m2):
            grad[m2] -= a * h[k[m2] + lag]
    return grad * dx

def _ceq(h):
    return np.sum(h) - _G['n'] / 2.0

def _cjac(h):
    return np.ones(_G['n'])

def opt_slsqp(h0, n, T, max_iter):
    _G['n'], _G['T'] = n, T
    try:
        res = minimize(_obj, h0, jac=_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': _ceq, 'jac': _cjac}],
                       options={'maxiter': max_iter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h0, n)

def binary_snap(h, n, threshold=0.1):
    h = h.copy()
    improved = True
    rounds = 0
    while improved and rounds < 3:
        improved = False
        rounds += 1
        c_cur = compute_c5(h)
        for j in range(n):
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < threshold:
                    continue
                trial = h.copy()
                trial[j] = bv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < c_cur - 1e-12:
                    h = trial
                    c_cur = c
                    improved = True
                    break
    return h
