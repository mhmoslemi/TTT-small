```python
def grad_active(h, active_idx):
    n = len(h)
    grad = np.zeros(n)
    one_minus_2h = 1.0 - 2.0 * h
    for j in active_idx:
        # grad[k] += (1-2*h[j-k]) for 0<=j-k<=n-1
        # i.e., k ranges from max(0, j-n+1) to min(n-1, j)
        k_lo = max(0, j - n + 1)
        k_hi = min(n - 1, j)
        if k_lo <= k_hi:
            indices = np.arange(k_lo, k_hi + 1)
            grad[indices] += one_minus_2h[j - indices]
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        room = 1.0 - h if diff > 0 else h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_c(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def grad_corr_active(h, active_idx):
    """Gradient of sum of correlations at active_idx w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    om2h = 1.0 - 2.0 * h
    for j in active_idx:
        k_lo = max(0, j - n + 1)
        k_hi = min(n - 1, j)
        if k_lo <= k_hi:
            idx = np.arange(k_lo, k_hi + 1)
            grad[idx] += om2h[j - idx]
    return grad

def projected_gradient_minmax(h, n, n_iters=200, n_active=40, base_lr=0.01):
    """Minimize max correlation via projected gradient on active set."""
    h = h.copy()
    best = compute_c5(h)
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        # Active set: lags within some fraction of max, plus top-k
        thresh = mc - 0.05 * mc  # within 5% of max
        active = np.where(corr >= thresh)[0]
        k = min(n_active, len(corr))
        topk = np.argpartition(corr, -k)[-k:]
        active = np.unique(np.concatenate([active, topk]))
        
        grad = grad_corr_active(h, active)
        # Backtracking line search
        lr = base_lr
        best_trial = best
        for bt in range(10):
            h_trial = np.clip(h - lr * grad, 0.0, 1.0)
            h_trial = make_feasible(h_trial, n)
            c_trial = compute_c5(h_trial)
            if c_trial < best:
                best_trial = c_trial
                h = h_trial
                best = c_trial
                break
            lr *= 0.5
        # Anneal learning rate
        base_lr *= (1.0 - it / (n_iters * 2.0))
    return h, best

def lbgfb_softmax(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_c(corr, T) * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=60, tries=3000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries, len(ones) * len(zeros))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        # Also try segment reversal
        if len(ones) >= 2:
            for _ in range(200):
                a, b = sorted(rng.choice(n, 2, replace=False))
                if b - a < 3 or b - a > n // 3:
                    continue
                # Check that segment has equal-ish ones/zeros for feasibility
                seg = h[a:b+1].copy()
                h[a:b+1] = seg[::-1]
                # Fix feasibility
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < cur - 1e-15:
                    cur = c
                    # Update lists
                    ones = np.where(h > 0.5)[0].tolist()
                    zeros = np.where(h <= 0.5)[0].tolist()
                    improved = True
                else:
                    h[a:b+1] = seg
                    h = make_feasible(h, n)
        if not improved:
            break
    return h, cur

def coord_descent(h0, n, max_rounds=5):
    h = h0.copy()
    cur = compute_c5(h)
    steps = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur
