def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-14
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def downsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_val(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -100, 100)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def soft_max_obj(h, T, mu=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    val = soft_max_val(corr, T) * dx
    if mu > 0:
        val += mu * (np.sum(h) - n / 2.0) ** 2
    return float(val)

def soft_max_grad(h, T, mu=0.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -100, 100)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / exp_shifted.sum()
    s = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-17:
            continue
        sk = s[k]
        if sk == 0:
            grad += w[k] * (1.0 - 2.0 * h)
        else:
            term1 = np.zeros(n)
            if sk > 0:
                term1[:n - sk] = 1.0 - h[sk:]
            else:
                term1[-sk:] = 1.0 - h[:n + sk]
            grad += w[k] * term1
            term2 = np.zeros(n)
            if sk > 0:
                term2[sk:] = h[:n - sk]
            else:
                term2[:n + sk] = h[-sk:]
            grad -= w[k] * term2
    grad *= dx
    if mu > 0:
        grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return grad

def anneal_lbfgsb(h0, n, T_start, T_end, n_stages, mu, max_iter_per_stage, t0, budget):
    h = h0.copy()
    for st in range(n_stages):
        T = T_start * (T_end / T_start) ** (st / max(1, n_stages - 1))
        T = max(T, 1.0)
        
        def obj_f(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            sm = soft_max_val(c, T) * (2.0 / n)
            if mu > 0:
                sm += mu * (np.sum(x) - n / 2.0) ** 2
            return float(sm)
        
        def grad_f(x):
            return soft_max_grad(x, T, mu)
        
        try:
            res = minimize(obj_f, h, jac=grad_f, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_iter_per_stage, 'ftol': 1e-16, 'gtol': 1e-10})
            h_new = make_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                h = h_new
        except Exception:
            pass
        if time.time() - t0 > budget - 30:
            break
    return h

def focused_optimize(h0, n, n_rounds, t0, budget):
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() - t0 > budget - 20:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        frac = 0.08
        k = max(3, int(frac * len(corr)))
        top_idx = np.argpartition(corr, -k)[-k:]
        penalty = np.zeros(len(corr))
        penalty[top_idx] = 1.0
        penalty /= penalty.sum()
        T = 200.0
        mu_p = 0.3
        
        def obj_focus(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            base = np.max(c)
            pen = np.dot(penalty, c)
            val = (base + mu_p * pen) * (2.0 / n)
            val += 500.0 * (np.sum(x) - n / 2.0) ** 2
            return float(val)
        
        def grad_focus(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            smc = np.max(c)
            sh = np.clip(300.0 * (c - smc), -80, 80)
            we = np.exp(sh)
            we /= we.sum()
            wc = we + mu_p * penalty
            s_arr = np.arange(2 * n - 1) - (n - 1)
            g = np.zeros(n)
            for kk in range(2 * n - 1):
                if wc[kk] < 1e-17:
                    continue
                sk = s_arr[kk]
                if sk == 0:
                    g += wc[kk] * (1.0 - 2.0 * x)
                else:
                    t1 = np.zeros(n)
                    if sk > 0:
                        t1[:n - sk] = 1.0 - x[sk:]
                    else:
                        t1[-sk:] = 1.0 - x[:n + sk]
                    g += wc[kk] * t1
                    t2 = np.zeros(n)
                    if sk > 0:
                        t2[sk:] = x[:n - sk]
                    else:
                        t2[:n + sk] = x[-sk:]
                    g -= wc[kk] * t2
            g *= (2.0 / n)
            g += 1000.0 * (np.sum(x) - n / 2.0)
            return g
        
        try:
            res = minimize(obj_focus, h, jac=grad_focus, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 60, 'ftol': 1e-15})
            h_new = make_feasible(res.x, n)
            if compute_c5(h_new) < cur:
                h = h_new
                cur = compute_c5(h)
        except Exception:
            pass
    return h, cur

def pair_swap_local_search(h0, n, rng, max_rounds=80, tries_per_round=3000, t0=None, budget=None):
    h = (h0 > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target - n_ones]] = 1.0
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        if t0 is not None and time.time() - t0 > budget - 15:
            break
        improved = False
        n_tries = min(tries_per_round, len(ones) * len(zeros))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def coord_descent_polish(h0, n, max_rounds=8, step_list=None, t0=None, budget=None):
    if step_list is None:
        step_list = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        if t0 is not None and time.time() - t0 > budget - 10:
            break
        improved = False
        for j in range(n):
            ov = h[j]
            for d in step_list:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def multi_start(h0, n, rng, n_starts, t0, budget):
    best_h = h0.copy()
    best_c = compute_c5(h0)
    for s in range(n_starts):
        if time.time() - t0 > budget - 30:
            break
        noise_scale = 0.08 + 0.04 * s
        h_r = h0.copy() + noise_scale * rng.randn(n)
        h_r = make_feasible(h_r, n)
        h_opt = anneal_lbfgsb(h_r, n, 1000, 100, 3, 500.0, 100, t0, budget)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    return best_h, best_c
