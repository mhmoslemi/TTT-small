def corr_and_grad(h, T):
    n = len(h)
    n_corr = 2 * n - 1
    dx = 2.0 / n
    
    # Compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    max_c = np.max(corr)
    exp_t = np.exp(T * (corr - max_c))
    exp_t = np.clip(exp_t, 1e-300, None)
    p = exp_t / np.sum(exp_t)
    
    # Objective value
    obj = (max_c + np.log(np.sum(exp_t)) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    for k in range(n_corr):
        if p[k] < 1e-12:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo >= hi:
            continue
        # For j in [lo, hi-1]:
        # grad[j] += p[k] * (1 - h[j+s])
        # If j-s is valid (0 <= j-s < n): grad[j] -= p[k] * h[j-s]
        j_arr = np.arange(lo, hi)
        grad[j_arr] += p[k] * (1.0 - h[j_arr + s])
        # For the second term: j-s in [0, n-1] means j in [s, s+n-1]
        # Intersected with [lo, hi-1]:
        j2_lo = max(lo, s)
        j2_hi = min(hi, s + n)
        if j2_lo < j2_hi:
            j2_arr = np.arange(j2_lo, j2_hi)
            grad[j2_arr] -= p[k] * h[j2_arr - s]
    
    grad *= dx
    return obj, grad

        return val

        _, g = lse_objective_and_grad(x, T)
        return g

        return np.sum(x) - target

        return np.ones(n)

Alternative: use the fact that the constraint is linear. I can use the method of Lagrange multipliers:
minimize f(h) subject to sum(h) = n/2
Lagrangian: f(h) - λ(sum(h) - n/2)
Optimality: ∇f(h) = λ·1

So at the optimum, the gradient is constant (all components equal). I can use this as a stopping criterion and adjust the step size accordingly.

Actually, the simplest approach is: use SLSQP with the analytical gradient. SLSQP handles linear equality constraints very efficiently (it just adjusts the step to maintain feasibility).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0 - 1e-10) if diff > 0 else (h > 1e-10)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = T * (corr - max_c)
    shifted = np.clip(shifted, -700, 0)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    obj = (max_c + np.log(sum_exp) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    n_corr = 2 * n - 1
    for k in range(n_corr):
        pk = p[k]
        if pk < 1e-12:
            continue
        s = k - (n - 1)
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo >= hi:
            continue
        j = np.arange(lo, hi)
        grad[j] += pk * (1.0 - h[j + s])
        # Second part: subtract h[j-s] where j-s is in [lo, hi-1]
        # j-s in [lo, hi-1] means j in [lo+s, hi-1+s]
        # Also need j in [lo, hi-1] (already guaranteed since we're in that range)
        # And j-s in [0, n-1]
        j2_lo = max(lo, s)  # j >= s for j-s >= 0
        j2_hi = min(hi, s + n)  # j < s+n for j-s < n
        if j2_lo < j2_hi:
            j2 = np.arange(j2_lo, j2_hi)
            grad[j2] -= pk * h[j2 - s]
    grad *= dx
    return obj, grad

def slsqp_solve(h, T, maxiter=300):
    n = len(h)
    target = n / 2.0
    
    def obj(x):
        return lse_obj_and_grad(x, T)[0]
    
    def grad(x):
        return lse_obj_and_grad(x, T)[1]
    
    def con_f(x):
        return np.sum(x) - target
    
    def con_j(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    
    res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15, 'disp': False})
    return res.x

def upsample(h, n_new):
    n = len(h)
    x_old = np.linspace(0, 2.0, n)
    x_new = np.linspace(0, 2.0, n_new)
    return np.interp(x_new, x_old, h)

def coord_descent_fine(h, n, budget_s, start_time):
    """Coordinate descent focusing on non-trivial coordinates."""
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    target = n / 2.0
    
    # Focus on coordinates where 0 < h < 1
    active = np.where((best_h > 0.01) & (best_h < 0.99))[0]
    # Also include boundaries of the support
    for j in np.where((best_h > 0.5) & (best_h < 1.0))[0]:
        if j not in active:
            active = np.append(active, j)
    
    if len(active) == 0:
        active = np.arange(n)
    
    steps = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002, 1.0, -1.0]  # 1.0 and -1.0 are "set to boundary"
    
    while time.time() - start_time < budget_s:
        improved = False
        for j in active:
            if time.time() - start_time > budget_s - 3:
                return best_h, best_c5
            old = best_h[j]
            for s in steps:
                if s == 1.0:
                    v = 1.0
                elif s == -1.0:
                    v = 0.0
                else:
                    v = old + s
                if v < -0.001 or v > 1.001 or abs(v - old) < 1e-8:
                    continue
                v = np.clip(v, 0, 1)
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_h = trial
                    best_c5 = c
                    old = v
                    improved = True
    return best_h, best_c5
