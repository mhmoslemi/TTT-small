import numpy as np
from scipy.optimize import minimize
import time

_gs = {'n': 960, 'T': 100.0}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = float(np.sum(mask))
        if cnt < 1e-10:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj(x):
    n = _gs['n']
    T = _gs['T']
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    sw = np.sum(np.exp(T * shifted))
    return (mc + np.log(sw) / T) * (2.0 / n)

def _grad(x):
    n = _gs['n']
    T = _gs['T']
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    ew = np.exp(T * shifted)
    sw = np.sum(ew)
    if sw < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = ew / sw
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, s + n)
        if lo1 < hi1:
            grad[lo1:hi1] += wk * (1.0 - x[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= wk * x[lo2 + s:hi2 + s]
    grad *= (2.0 / n)
    return grad

def _con_fun(x):
    return np.sum(x) - _gs['n'] / 2.0

def _con_jac(x):
    return np.ones(_gs['n'])

def slsqp_optimize(h, n, T, maxiter=400):
    _gs['n'] = n
    _gs['T'] = T
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h.copy(), jac=_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-20})
        if res.x is not None and np.all(np.isfinite(res.x)):
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h.copy(), compute_c5(h)

def _subgrad_top(h, top_idx, n):
    grad = np.zeros(n)
    for k_idx in top_idx:
        s = k_idx - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, s + n)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2 + s:hi2 + s]
    return grad / len(top_idx)

            c5_up = compute_c5(h_up)
            if c5_up < best_c5 + 0.003:
                for T2 in [100.0, 500.0, 3000.0]:
                    if rem() < 30:
                        break
                    h_u, c5_u = slsqp_optimize(h_up, n2, T2, maxiter=250)
                    if c5_u < best_c5 + 0.003:
                        h_up = h_u
                if rem() > 50:
                    h_u, c5_u = pgd_optimize(h_up, n2, min(50, rem() - 20), t0, k_top=3)
                    if c5_u < best_c5 + 0.003:
                        h_up = h_u
                h_up = make_feasible(h_up, n2)
                c5_final_up = compute_c5(h_up)
                if c5_final_up < best_c5:
                    best_h, best_c5 = h_up, c5_final_up

    # Phase 7: Final PGD + SLSQP at best resolution
    n_best = len(best_h)
    if rem() > 40:
        h_new, c5_new = pgd_optimize(best_h, n_best, min(35, rem() - 10), t0, k_top=3)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new
    if rem() > 20:
        h_new, c5_new = slsqp_optimize(best_h, len(best_h), 30000.0, maxiter=200)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 8: Final coordinate descent
    if rem() > 15:
        h_new, c5_new = coord_descent(best_h, len(best_h), min(12, rem() - 5), t0)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 9: Ultra-final SLSQP
    if rem() > 12:
        h_new, c5_new = slsqp_optimize(best_h, len(best_h), 100000.0, maxiter=150)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
