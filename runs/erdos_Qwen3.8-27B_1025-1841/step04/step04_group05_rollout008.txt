        wn = weights[n - 1 - lag]
        if wp > 1e-14:
            grad[:n - lag] += wp * (1.0 - x[lag:n])
            grad[lag:] += wp * (-x[:n - lag])
        if wn > 1e-14:
            grad[lag:] += wn * (1.0 - x[:n - lag])
            grad[:n - lag] += wn * (-x[lag:n])
    grad *= dx
    sd = np.sum(x) - n / 2.0
    obj += pen * sd * sd
    grad += 2.0 * pen * sd
    return obj, grad

def _run_lbfgs(h_start, n, T, max_iter, penalty=50.0):
    global _G_N, _G_T, _G_PEN
    _G_N = n
    _G_T = T
    _G_PEN = penalty
    try:
        res = minimize(_lbg_obj_grad, h_start, method='L-BFGS-B',
                       jac=True, bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16,
                                'gtol': 1e-10, 'maxls': 50, 'maxcor': 20})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def _local_search(h, n, max_sweeps, seed_vals, start_c5):
    rng = np.random.RandomState(seed_vals)
    h = h.copy()
    best_c5 = start_c5
    best_h = h.copy()
    for sweep in range(max_sweeps):
        ta = 0.005 * (0.97 ** sweep)
        order = rng.permutation(n)
        improved = False
        for j in order:
            for cand in (0.0, 1.0, h[j] + 0.02, h[j] - 0.02):
                if cand < 0.0 or cand > 1.0:
                    continue
                if abs(cand - h[j]) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = trial.copy()
                    h = trial.copy()
                    improved = True
                    break
                elif c < best_c5 + ta:
                    if rng.rand() < np.exp(-(c - best_c5) / ta):
                        h = trial.copy()
        if not improved and sweep > 10:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T, _G_PEN
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Optimize at N=640
    n1 = len(h0)
    if n1 != 640:
        h0 = upsample_h(h0, 640)
    h_curr = h0.copy()

    T_schedule_640 = [20, 50, 100, 200, 400, 800, 1500]
    for T in T_schedule_640:
        if time_left() < 15:
            break
        max_it = 300 if T < 200 else 200
        h_opt = _run_lbfgs(h_curr, 640, T, max_it, penalty=50.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h_curr = h_opt.copy()

    # Random restarts at 640
    for ri in range(3):
        if time_left() < 20:
            break
        if ri == 0:
            h_r = np.clip(0.5 + 0.4 * rng.randn(640), 0, 1)
        elif ri == 1:
            h_r = (rng.rand(640) < 0.5).astype(float)
        else:
            h_r = 0.5 * best_h + 0.5 * rng.rand(640)
        h_r = make_feasible(h_r, 640)
        for T in [80, 400, 1000]:
            if time_left() < 10:
                break
            h_opt = _run_lbfgs(h_r, 640, T, 250, penalty=50.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_r = h_opt.copy()

    # Phase 2: Upsample to 1280 and optimize
    n2 = 1280
    if time_left() > 120:
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        h_curr2 = h2.copy()

        T_schedule_1280 = [20, 50, 100, 200, 400, 800, 1500, 2500]
        for T in T_schedule_1280:
            if time_left() < 15:
                break
            max_it = 250 if T < 200 else 150
            h_opt = _run_lbfgs(h_curr2, n2, T, max_it, penalty=50.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_curr2 = h_opt.copy()

        # Restart at 1280
        for ri in range(2):
            if time_left() < 20:
                break
            if ri == 0:
                h_r2 = np.clip(0.5 + 0.35 * rng.randn(n2), 0, 1)
            else:
                h_r2 = 0.6 * best_h + 0.4 * rng.rand(n2)
            h_r2 = make_feasible(h_r2, n2)
            for T in [50, 200, 800]:
                if time_left() < 10:
                    break
                h_opt = _run_lbfgs(h_r2, n2, T, 200, penalty=50.0)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_r2 = h_opt.copy()

    # Phase 3: Try N=2560 briefly
    n3 = 2560
    if time_left() > 80:
        h3 = upsample_h(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        h_curr3 = h3.copy()
        for T in [50, 200, 600]:
            if time_left() < 15:
                break
            h_opt = _run_lbfgs(h_curr3, n3, T, 150, penalty=50.0)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_curr3 = h_opt.copy()

    # Phase 4: Local search / coordinate descent
    n_search = len(best_h)
    if n_search > 2560:
        h_search = upsample_h(best_h, 1280)
        n_search = 1280
    else:
        h_search = best_h.copy()
    c5_search = compute_c5(h_search)

    if time_left() > 60:
        h_search, c5_search = _local_search(h_search, n_search, 200, seed, c5_search)
        if c5_search < best_c5:
            best_c5 = c5_search
            best_h = h_search.copy()

    # Phase 5: One more L-BFGS-B polish from best
    if time_left() > 20:
        n_final = len(best_h)
        if n_final > 2560:
            best_h = upsample_h(best_h, 1280)
            n_final = 1280
            best_c5 = compute_c5(best_h)
        h_opt = _run_lbfgs(best_h, n_final, 1000, 150, penalty=200.0)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
