```python
def project_feasible(h, n):
    """Project h onto {x: 0<=x<=1, sum(x)=n/2} using bisection."""
    target = n / 2.0
    # First clip to [0,1]
    h = np.clip(h, 0.0, 1.0)
    # Now find tau such that sum(clip(h - tau, 0, 1)) = target
    # If sum(h) == target, we're done
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return h
    # Bisection on tau
    lo, hi = -1.0, 1.0
    s_target = target
    for _ in range(60):
        mid = (lo + hi) / 2
        h_trial = np.clip(h - mid, 0.0, 1.0)
        s_trial = np.sum(h_trial)
        if s_trial > s_target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Final fixup
    diff = target - np.sum(h)

        cnt = np.sum(mask)
        if cnt > 0:
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        h_t = np.clip(h - mid, 0.0, 1.0)
        if np.sum(h_t) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    if new_n == old_n:
        return project_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
```

Let me stick with the smooth max approach but with adaptive T.

**Key insight for improvement**: The previous approach used SLSQP which is slower than L-BFGS-B for large n. By switching to L-BFGS-B with bounds, we can:
1. Handle the box constraints [0,1] natively
2. Use the exact gradient (not finite differences)
3. Potentially converge faster

The penalty for the equality constraint (sum = n/2) adds a quadratic term. With large enough λ, the iterates stay close to feasible, and the final projection gives the feasible solution.

Let me also add a "peak targeting" step: after optimizing, identify the top few lags and do a focused optimization to reduce them specifically.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    s = np.sum(h)
    if abs(s - target) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s_mid = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_mid > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
def upsample_h(h, new_n):
```
