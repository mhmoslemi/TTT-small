def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj_closure(h, T, dx, topK=200):
    """Objective using top-K soft-max"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topK < len(corr):
        # Get top K values
        idx = np.argpartition(corr, -topK)[-topK:]
        top_vals = corr[idx]
        mc = np.max(top_vals)
        shifted = T * (top_vals - mc)
        shifted = np.clip(shifted, -80, 80)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    else:
        mc = np.max(corr)
        shifted = T * (corr - mc)

        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def lbgfb_smooth(h_start, n, T, mu, max_iter=300, topK=200):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if topK < len(corr):
            idx = np.argpartition(corr, -topK)[-topK:]
            top_vals = corr[idx]
        else:
            top_vals = corr
        mc = np.max(top_vals)
        shifted = T * (top_vals - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_sa(h_bin, n, rng, n_steps, T0, cooling, time_limit):
    """Simulated annealing on binary solution"""
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    best_c5 = current_c5
    best_h = h.copy()
    
    T = T0
    step = 0
    t_start = time.time()
    while step < n_steps:
        if time.time() - t_start > time_limit:
            break
        # Pick random 1 and 0 to swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        # Swap
        h[oi] = 0.0
        h[oj] = 1.0
        ones[ii] = oj
        zeros[jj] = oi
        # Evaluate
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[oi] = 1.0
            h[oj] = 0.0
            ones[ii] = oi
            zeros[jj] = oj
        step += 1
        T *= cooling
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter, topK):
    dx = 2.0 / n
    total_lags = 2 * n - 1
    use_topK = min(topK, total_lags)

    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if use_topK < total_lags:
            idx = np.argpartition(corr, -use_topK)[-use_topK:]
            top_vals = corr[idx]
        else:
            top_vals = corr
        mc = np.max(top_vals)
        shifted = T * (top_vals - mc)
        shifted = np.clip(shifted, -80.0, 80.0)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def slsqp_polish(h_start, n, T, max_iter):
    dx = 2.0 / n
    total_lags = 2 * n - 1
    topK = min(200, total_lags)

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        if topK < total_lags:
            idx = np.argpartition(corr, -topK)[-topK:]
            top_vals = corr[idx]
        else:
            top_vals = corr
        mc = np.max(top_vals)
        shifted = T * (top_vals - mc)
        shifted = np.clip(shifted, -80.0, 80.0)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    def con(h):
        return np.sum(h) - n / 2.0

    def jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def binary_sa(h_bin, n, rng, time_limit, T0, cooling):
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    T = T0
    t_start = time.time()
    noop_streak = 0
    while True:
        if time.time() - t_start > time_limit:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        ones[ii] = oj
        zeros[jj] = oi
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < -1e-15 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                noop_streak = 0
            else:
                noop_streak += 1
        else:
            h[oi] = 1.0
            h[oj] = 0.0
            ones[ii] = oi
            zeros[jj] = oj
            noop_streak += 1
        T *= cooling
        if T < 1e-12:
            break
    return best_h, best_c5

def make_periodic_pattern(period, n, asym=0):
    if period % 2 == 0:
        half = period // 2
    else:
        half = (period + 1) // 2
    pat = np.zeros(period)
    pat[:half] = 1.0
    if asym > 0 and period > 2 * asym:
        pat[half - 1] = 0.0
        pat[0] = 1.0
    rep = np.tile(pat, n // period + 1)[:n]
    return rep
