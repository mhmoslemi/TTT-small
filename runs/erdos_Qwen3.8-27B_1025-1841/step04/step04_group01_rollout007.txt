def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T, _G_TOP_IDX, _G_PENALTY
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: L-BFGS-B with progressive T at initial resolution
    for T_val in [40.0, 100.0, 250.0, 500.0, 1000.0]:
        if remaining() < 80:
            break
        n = len(best_h)
        top_idx = None
        if T_val >= 250:
            top_idx = find_top_lags(best_h, frac=0.015, max_k=30)
        x_new = lbgfs_step(best_h, n, T_val, top_idx=top_idx, penalty=8000.0, maxiter=400, ftol=1e-16)
        x_new = make_feasible(x_new, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new

    # Phase 2: Top-lag focused L-BFGS-B (multiple rounds)
    for rnd in range(5):
        if remaining() < 60:
            break
        n = len(best_h)
        top_idx = find_top_lags(best_h, frac=0.008, max_k=20)
        T_val = 600.0 + rnd * 200.0
        x_new = lbgfs_step(best_h, n, T_val, top_idx=top_idx, penalty=10000.0, maxiter=300, ftol=1e-16)
        x_new = make_feasible(x_new, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new

    # Phase 3: Coordinate descent
    if remaining() > 40:
        n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n, max_rounds=3,
                                        step_sizes=[0.003, 0.007, 0.015, 0.03],
                                        time_budget=min(120, remaining() - 30),
                                        start_time=start_time)

    # Phase 4: Pair perturbation for diversity
    if remaining() > 50:
        n = len(best_h)
        best_h, best_c5 = pair_perturb(best_h, n, rng, n_iter=500,
                                       time_budget=min(60, remaining() - 30),
                                       start_time=start_time)

    # Phase 5: Another L-BFGS-B round after perturbation
    if remaining() > 80:
        n = len(best_h)
        top_idx = find_top_lags(best_h, frac=0.01, max_k=25)
        x_new = lbgfs_step(best_h, n, 400.0, top_idx=top_idx, penalty=10000.0, maxiter=300, ftol=1e-16)
        x_new = make_feasible(x_new, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new

    # Phase 6: Try upsample to 2x resolution
    if remaining() > 300:
        n_up = len(best_h) * 2
        if n_up <= 1920:
            h_up = upsample_h(best_h, n_up)
            c5_up = compute_c5(h_up)
            if c5_up <= best_c5 + 0.002:
                if remaining() > 150:
                    T_val = 200.0
                    x_new = lbgfs_step(h_up, n_up, T_val, top_idx=None, penalty=8000.0, maxiter=200, ftol=1e-16)
                    x_new = make_feasible(x_new, n_up)
                    c5_new = compute_c5(x_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = x_new
                if remaining() > 100:
                    n = len(best_h)
                    top_idx = find_top_lags(best_h, frac=0.01, max_k=30)
                    x_new = lbgfs_step(best_h, n, 600.0, top_idx=top_idx, penalty=12000.0, maxiter=250, ftol=1e-16)
                    x_new = make_feasible(x_new, n)
                    c5_new = compute_c5(x_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = x_new
                if remaining() > 50:
                    n = len(best_h)
                    best_h, best_c5 = coord_descent(best_h, n, max_rounds=2,
                                                    step_sizes=[0.002, 0.005, 0.01],
                                                    time_budget=min(60, remaining() - 20),
                                                    start_time=start_time)

    # Phase 7: Random restarts with L-BFGS-B
    for restart in range(6):
        if remaining() < 70:
            break
        n = len(best_h)
        scale = 0.01 if restart < 3 else 0.025
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        top_idx = None
        if restart >= 3:
            top_idx = find_top_lags(h_try, frac=0.02, max_k=20)
        x_new = lbgfs_step(h_try, n, 300.0, top_idx=top_idx, penalty=8000.0, maxiter=250, ftol=1e-15)
        x_new = make_feasible(x_new, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new

    # Phase 8: Final targeted optimization
    if remaining() > 50:
        n = len(best_h)
        top_idx = find_top_lags(best_h, frac=0.005, max_k=15)
        x_new = lbgfs_step(best_h, n, 1200.0, top_idx=top_idx, penalty=15000.0, maxiter=200, ftol=1e-16)
        x_new = make_feasible(x_new, n)
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new

    # Phase 9: Ultra-fine coordinate descent
    if remaining() > 20:
        n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n, max_rounds=2,
                                        step_sizes=[0.0005, 0.001, 0.002, 0.004],
                                        time_budget=min(30, remaining() - 5),
                                        start_time=start_time)

    # Phase 10: Final binary snap attempt
    if remaining() > 10:
        n = len(best_h)
        for j in range(n):
            if remaining() < 5:
                break
            for val in (0.0, 1.0):
                if abs(best_h[j] - val) < 1e-6:
                    continue
                trial = best_h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = trial

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
