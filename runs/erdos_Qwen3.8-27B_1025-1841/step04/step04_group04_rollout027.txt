        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]

        contrib_j = []

            if 0 <= j2 < n_total and h[j2] == 0.0:
                contrib_j.append(j)
        
        if not contrib_j:
            contrib_j = list(ones)
        
        # Try moving a contributing 1 to a random zero position
        i = contrib_j[np.random.randint(len(contrib_j))]
        j_new = zeros[np.random.randint(len(zeros))]
        if i == j_new:
            continue
        
        h[i] = 0.0
        h[j_new] = 1.0
        c = compute_c5(h)

            best_c = c
        else:
            h[i] = 1.0
            h[j_new] = 0.0
    
    return h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_project(h, n):
    n_ones = n // 2
    idx = np.argsort(-h, kind='stable')
    result = np.zeros(n)
    result[idx[:n_ones]] = 1.0
    return result

def _softmax_obj(x, T, n_pts):
    dx = 2.0 / n_pts
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _sum_obj(x, n_pts):
    return np.sum(x) - n_pts / 2.0

def _jac_obj(x, n_pts):
    return np.ones(n_pts)

def _slsqp_optimize(h_start, n_pts, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': _sum_obj, 'jac': _jac_obj, 'args': (n_pts,)}]
    res = minimize(_softmax_obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, args=(T, n_pts),
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def _targeted_binary_search(h, n, max_iters, timeout, start_time):
    h = binary_project(h, n)
    best_c = compute_c5(h)
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() - start_time > timeout:
            break
        if not ones_set or not zeros_set:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = int(np.argmax(corr))
        shift = max_idx - n + 1
        contrib = []
        ones_list = list(ones_set)
        for j in range(n):
            if h[j] == 1.0:
                j2 = j + shift
                if 0 <= j2 < n and h[j2] == 0.0:
                    contrib.append(j)
        if not contrib:
            contrib = ones_list
        i = contrib[it % len(contrib)] if len(contrib) else ones_list[0]
        zeros_list = list(zeros_set)
        j_new = zeros_list[it % len(zeros_list)] if zeros_list else 0
        if i == j_new:
            continue
        h[i] = 0.0
        h[j_new] = 1.0
        c = compute_c5(h)
        if c < best_c - 1e-15:
            best_c = c
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j_new)
            ones_set.add(j_new)
        else:
            h[i] = 1.0
            h[j_new] = 0.0
    return h, best_c

def _simulated_annealing_binary(h, n, max_iters, timeout, start_time, T_start=0.008, T_end=0.0005):
    h = binary_project(h, n)
    best_c = compute_c5(h)
    best_h = h.copy()
    current_c = best_c
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    for it in range(max_iters):
        if time.time() - start_time > timeout:
            break
        if not ones_set or not zeros_set:
            break
        T = T_start * (T_end / T_start) ** (it / max_iters)
        ones_list = list(ones_set)
        zeros_list = list(zeros_set)
        i = ones_list[it % len(ones_list)]
        j = zeros_list[(it * 7 + 3) % len(zeros_list)]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        delta = c - current_c
        accept = False
        if delta < 0:
            accept = True
        elif T > 1e-12:
            accept = (np.random.random() < np.exp(-delta / T))
        if accept:
            current_c = c
            if c < best_c - 1e-15:
                best_c = c
                best_h = h.copy()
            ones_set.remove(i)
            zeros_set.add(i)
            zeros_set.remove(j)
            ones_set.add(j)
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def _coordinate_descent(h, n, rounds, timeout, start_time):
    best_c = compute_c5(h)
    for rd in range(rounds):
        if time.time() - start_time > timeout:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > timeout:
                break
            old = h[j]
            best_local_c = best_c
            best_local_val = old
            cands = [0.0, 1.0, 0.5, max(0.0, old - 0.05), min(1.0, old + 0.05),
                     max(0.0, old - 0.02), min(1.0, old + 0.02)]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_local_c - 1e-14:
                    best_local_c = c
                    best_local_val = cand
            if best_local_val != old:
                h[j] = best_local_val
                h = make_feasible(h, n)
                best_c = best_local_c
                improved = True
        if not improved:
            break
    return h, best_c
