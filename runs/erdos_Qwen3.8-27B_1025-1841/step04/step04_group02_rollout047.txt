import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    """Compute C5 using FFT for speed."""
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    N = 1 << (2 * n - 1).bit_length()  # next power of 2 >= 2n-1
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(v, N)
    corr = np.real(np.fft.irfft(H * np.conj(V), N))
    full_corr = np.concatenate([corr[N - n + 1:], corr[:n]])
    return float(np.max(full_corr) * dx)

def compute_c5_direct(h):
    """Compute C5 using direct correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

        h = np.clip(h, 0.0, 1.0)

        b[idx] = 0

        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1

    h = np.zeros(n)
    for i in range(n):
        pos_in_period = i % period
        if pos_in_period < duty:
            h[i] = 1.0
    return h

            chosen_ones = rng.choice(ones_in_seg, size=swap_count, replace=False)
            chosen_zeros = rng.choice(zeros_in_seg, size=swap_count, replace=False)
            old_vals_ones = b[start + chosen_ones].copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    v = 1.0 - h
    N = 1 << (2 * n - 1).bit_length()
    H = np.fft.rfft(h, N)
    V = np.fft.rfft(v, N)
    corr = np.real(np.fft.irfft(H * np.conj(V), N))
    full_corr = np.concatenate([corr[N - n + 1:], corr[:n]])
    return float(np.max(full_corr) * dx)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)

For a single bin change h[j] → h[j] + δ:
The change in correlation c[k] (at full correlation index k) is:

Actually, let me be more precise. The full correlation is:
c[m] = ∑_i h[i + (m-(n-1))] * (1-h[i]) for valid i

If we change h[j] by δ:
- h[j] appears as the "a[m+n] = h[m+(n-1)+(m-(n-1))]" term... this is getting confusing.

Let me just use the direct formula:
c[m] = ∑_i a[i+m] * b[i] where a=h, b=1-h, and m ranges from -(n-1) to (n-1).

If we change h[j] to h[j]+δ, then:
- a[j] changes to a[j]+δ
- b[j] changes to b[j]-δ

The effect on c[m]:
c[m] involves terms a[i+m]*b[i] for all valid i.
- If i+m = j (i.e., i = j-m): term becomes (a[j]+δ)*b[j-m] instead of a[j]*b[j-m]. Change: δ*b[j-m]
- If i = j: term becomes a[j+m]*(b[j]-δ) instead of a[j+m]*b[j]. Change: -δ*a[j+m]

So Δc[m] = δ * (b[j-m] - a[j+m]) = δ * (1-h[j-m] - h[j+m]) (for valid indices)

If indices are invalid, the contribution is 0.

This gives us an O(1) update per correlation bin, or O(n) to compute all bins' changes. Then we can find if the max decreases.
