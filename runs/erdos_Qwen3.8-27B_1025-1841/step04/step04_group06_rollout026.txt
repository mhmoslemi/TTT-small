        exp_s = np.exp(shifted)
        w = exp_s / np.sum(exp_s)
        
        W = np.fft.rfft(w, n)
        F1h = np.fft.rfft(1.0 - h, n)
        term1 = np.fft.irfft(np.conj(W) * F1h, n)
        GH = np.fft.rfft(h, n)
        term2 = np.fft.irfft(W * GH, n)
        
        grad = dx * (term1 - term2)
        viol = np.sum(h) - n / 2.0
        grad += 2.0 * mu * viol / n
        
        decay = 1.0 - step / max(n_steps, 1)
        lr = lr_init * decay * decay
        lr = max(lr, lr_init * 0.005)
        
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = make_feasible(h, n)
    
    return h

def peak_shave(h, n, rng, n_rounds=200):
    """Direct peak reduction by adjusting individual elements"""
    dx = 2.0 / n
    h = h.copy()
    current = compute_c5(h)
    
    for rnd in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        peak_idx = np.argmax(corr)
        
        improved = False
        attempts = 300
        for _ in range(attempts):
            j = rng.randint(n)
            old = h[j]
            # Try several new values
            for frac in [0.0, 1.0, old + 0.03, old - 0.03, old + 0.07, old - 0.07, 0.5]:
                nv = frac
                if not (0.0 <= nv <= 1.0) or abs(nv - old) < 1e-9:
                    continue
                h[j] = nv
                h = make_feasible(h, n)
                nc = compute_c5(h)
                if nc < current - 1e-14:
                    current = nc
                    improved = True
                    break
                h[j] = old
                h = make_feasible(h, n)
            if improved:
                break
        
        if not improved:
            break
    
    return h, current
