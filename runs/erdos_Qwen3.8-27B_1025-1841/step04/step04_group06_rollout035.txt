    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def to_binary_smart(h, n):
    """Convert to binary, choosing which to flip based on h values."""
    h = np.asarray(h, dtype=float).copy()
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1)[0]
        order = one_idx[np.argsort(h[one_idx])]  # flip lowest-valued ones
        for i in range(n_ones - target):
            h_bin[order[i]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = zero_idx[np.argsort(-h[zero_idx])]  # flip highest-valued zeros
        for i in range(target - n_ones):
            h_bin[order[i]] = 1.0
    return h_bin

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def binary_search_vectorized(h, n, max_rounds=30, time_limit=60.0):
    """Vectorized best pair-exchange local search."""
    h = h.copy()
    current = compute_c5(h)
    base_corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / n
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    lags = np.arange(-(n-1), n)
    t0 = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        best_c = current
        best_oi, best_zi = -1, -1
        n_o, n_z = len(ones), len(zeros)
        
        for i in range(n_o):
            p = ones[i]
            # delta when h[p]: 1->0
            p_plus = p + lags
            p_minus = p - lags
            v1 = (p_plus >= 0) & (p_plus < n)
            v2 = (p_minus >= 0) & (p_minus < n)
            delta_p = np.zeros(2*n-1)
            pp_safe = np.clip(p_plus, 0, n-1)
            pm_safe = np.clip(p_minus, 0, n-1)
            delta_p[v1] -= (1.0 - h[pp_safe[v1]])
            delta_p[v2] -= h[pm_safe[v2]]
            
            # delta for all zeros at once
            q_idx = zeros
            q_plus = q_idx[:, None] + lags[None, :]
            q_minus = q_idx[:, None] - lags[None, :]
            vq1 = (q_plus >= 0) & (q_plus < n)
            vq2 = (q_minus >= 0) & (q_minus < n)
            qp_safe = np.clip(q_plus, 0, n-1)
            qm_safe = np.clip(q_minus, 0, n-1)
            delta_q = np.zeros((n_z, 2*n-1))
            delta_q[vq1] += (1.0 - h[qp_safe[vq1]])
            delta_q[vq2] += h[qm_safe[vq2]]
            
            new_max = np.max(base_corr[None, :] + delta_p[None, :] + delta_q, axis=1)
            new_c5 = new_max * dx
            li = np.argmin(new_c5)
            if new_c5[li] < best_c:
                best_c = new_c5[li]
                best_oi, best_zi = i, li
        
        if best_oi < 0:
            break
        
        p = ones[best_oi]
        q = zeros[best_zi]
        h[p] = 0.0
        h[q] = 1.0
        current = best_c
        base_corr = np.correlate(h, 1.0 - h, mode='full')
        ones = np.delete(ones, best_oi)
        zeros = np.delete(zeros, best_zi)
        ones = np.append(ones, q)
        zeros = np.append(zeros, p)
    
    return h, current

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)
