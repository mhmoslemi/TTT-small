def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff))+1:]
        else:
            idx = np.argsort(h)[:int(abs(diff))+1:]
        for i in idx:
            adj = min(diff if diff > 0 else -diff, 1.0 - h[i] if diff > 0 else h[i])
            h[i] += adj if diff > 0 else -adj
    return h

            block_j = np.arange(start_j, min(start_j + bs, n)) % n

            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones[ii] = oj
                zeros[jj] = oi
                # Sort to maintain consistency
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current
