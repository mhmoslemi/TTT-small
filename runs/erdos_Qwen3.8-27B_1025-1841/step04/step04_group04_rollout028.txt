import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)

import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 960
_OPT_T = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def pairwise_swap_search(h, deadline, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    min_diff = 0.4
    stall_count = 0

    while time.time() < deadline:
        improved = False
        attempts = 0
        tabu = set()
        max_attempts = 5000

        while attempts < max_attempts and time.time() < deadline:
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j or abs(h[i] - h[j]) < min_diff:
                attempts += 1
                continue
            key = (min(i, j), max(i, j))
            if key in tabu:
                attempts += 1
                continue
            hi, hj = h[i], h[j]
            h[i], h[j] = hj, hi
            new_c5 = compute_c5(h)
            if new_c5 < c5 - 1e-15:
                c5 = new_c5
                improved = True
                tabu.add(key)
            else:
                h[i], h[j] = hi, hj
            attempts += 1

        if not improved:
            stall_count += 1
            min_diff *= 0.7
            if min_diff < 0.05 or stall_count > 8:
                break
        else:
            stall_count = 0

    return h, c5

def iterated_perturb_search(h, deadline, rng):
    n = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_h = h.copy()
    curr_c5 = best_c5

    while time.time() < deadline - 30:
        h_pert = curr_h.copy()
        n_swaps = rng.randint(3, 12)
        for _ in range(n_swaps):
            i = rng.randint(n)
            j = rng.randint(n)
            if i != j and abs(h_pert[i] - h_pert[j]) > 0.3:
                h_pert[i], h_pert[j] = h_pert[j], h_pert[i]

        sub_deadline = min(deadline - 30, time.time() + 40)
        h_ls, c5_ls = pairwise_swap_search(h_pert, sub_deadline, rng)

        if c5_ls < curr_c5 or rng.rand() < 0.25:
            curr_h = h_ls
            curr_c5 = c5_ls

        if curr_c5 < best_c5 - 1e-15:
            best_c5 = curr_c5
            best_h = curr_h.copy()

    return best_h, best_c5

def coord_descent_fine(h, deadline, rng):
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)

    while time.time() < deadline:
        improved = False
        for j in range(n):
            if time.time() >= deadline:
                break
            old = h[j]
            best_c = c5
            best_trial = None
            cands = [0.0, 1.0, max(0.0, old - 0.05), max(0.0, old - 0.01),
                     min(1.0, old + 0.05), min(1.0, old + 0.01)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = trial.copy()
            if best_trial is not None:
                h = best_trial
                c5 = best_c
                improved = True
        if not improved:
            break

    return h, c5
