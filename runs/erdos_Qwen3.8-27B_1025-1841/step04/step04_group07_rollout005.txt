import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / max(mask.sum(), 1)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.sum(np.exp(shifted)) / T

def grad_corr_lag(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m > 0:
        grad[:n - m] += -h[m:n]
        grad[m:n] += (1.0 - h[:n - m])
    elif m < 0:
        m2 = -m
        grad[:n - m2] += (1.0 - h[m2:n])
        grad[m2:n] += -h[:n - m2]
    else:
        grad = 1.0 - 2.0 * h
    return grad

def subgradient_polish(h0, n, n_iters=8000, top_k=5, lr_init=0.005, decay=0.9995):
    h = h0.copy()
    target = n / 2.0
    best = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argpartition(corr, -top_k)[-top_k:]
        for k in idx:
            m = int(k) - (n - 1)
            g = grad_corr_lag(h, m)
            h = h - lr * g
            h = np.clip(h, 0.0, 1.0)
            diff = target - h.sum()
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
        lr *= decay
        c = compute_c5(h)
        if c < best:
            best = c
            best_h = h.copy()
    return best_h, best

def binary_sa(h_bin, n, rng, T_init=0.006, T_final=0.0001, n_iters=60000):

    best = cur
    best_h = h.copy()

    zeros_list = np.where(h <= 0.5)[0].tolist()

        ii = rng.randint(len(ones_list))
        jj = rng.randint(len(zeros_list))

        h[oi] = 0.0
        h[oj] = 1.0
        new_c = compute_c5(h)
        if new_c < cur:

            ones_list[ii] = oj
            zeros_list[jj] = oi
            if cur < best:

                best_h = h.copy()
        elif rng.random() < np.exp(-(new_c - cur) / max(T, 1e-10)):
            cur = new_c
            ones_list[ii] = oj
            zeros_list[jj] = oi
        else:

def lbgfb_optimize(h0, n, T, mu, max_iter=250):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def active_lag_opt(h0, n, n_active=30, max_iter=200):
    corr = np.correlate(h0, 1.0 - h0, mode='full')
    k = min(n_active, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    active = np.zeros(len(corr))
    active[idx] = 1.0 / k
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return float(np.dot(active, c)) * dx + 500.0 * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11})
    return make_feasible(res.x, n)

        h_sg, c_sg = subgradient_polish(h_sg, n1, n_iters=10000, top_k=5, lr_init=0.004, decay=0.9997)
        if c_sg < best_c5:
            best_c5, best_h = c_sg, h_sg.copy()

    if rem() > 30:
        h_sg, c_sg = subgradient_polish(h_sg, n1, n_iters=10000, top_k=3, lr_init=0.002, decay=0.9998)
        if c_sg < best_c5:
            best_c5, best_h = c_sg, h_sg.copy()

    # Phase 3: L-BFGS-B with multiple temperatures
    h_soft = best_h + 0.03 * rng.randn(n1)
    h_soft = make_feasible(h_soft, n1)
    for T in [80, 400, 2000]:
        if rem() < 25:
            break
        h_opt = lbgfb_optimize(h_soft, n1, T, mu=600.0, max_iter=200)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()
            h_soft = h_opt.copy()

    # Phase 4: Active lag focused optimization
    if rem() > 35:
        h_al = active_lag_opt(h_soft, n1, n_active=20, max_iter=250)
        c_al = compute_c5(h_al)
        if c_al < best_c5:
            best_c5, best_h = c_al, h_al.copy()
        h_soft = best_h.copy()
    if rem() > 25:
        h_al = active_lag_opt(h_soft, n1, n_active=8, max_iter=250)
        c_al = compute_c5(h_al)
        if c_al < best_c5:
            best_c5, best_h = c_al, h_al.copy()

    # Phase 5: Another binary round from current best
    if rem() > 80:
        h_bin2 = (best_h > 0.5).astype(float)
        n_o2 = int(h_bin2.sum())
        if n_o2 > target_ones:
            idx = np.where(h_bin2 == 1)[0]
            h_bin2[idx[:n_o2 - target_ones]] = 0.0
        elif n_o2 < target_ones:
            idx = np.where(h_bin2 == 0)[0]
            h_bin2[idx[:target_ones - n_o2]] = 1.0
        h_bin2, c_b2 = greedy_binary(h_bin2, n1, rng, max_rounds=200, tries=3000)
        h_bin2, c_b2 = binary_sa(h_bin2, n1, rng, T_init=0.005, T_final=0.0002, n_iters=30000)
        if c_b2 < best_c5:
            best_c5, best_h = c_b2, h_bin2.copy()
        # Subgradient from binary
        h_sg2, c_sg2 = subgradient_polish(best_h.copy(), n1, n_iters=8000, top_k=5, lr_init=0.003, decay=0.9998)
        if c_sg2 < best_c5:
            best_c5, best_h = c_sg2, h_sg2.copy()

    # Phase 6: Higher resolution attempt
    n2 = min(n1 * 2, 1280)
    if rem() > 100:
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        if rem() > 50:
            h2_sg, c2s = subgradient_polish(h2, n2, n_iters=8000, top_k=5, lr_init=0.003, decay=0.9998)
            if c2s < best_c5:
                best_c5, best_h = c2s, h2_sg.copy()
        if rem() > 30:
            h2_opt = lbgfb_optimize(best_h, n2, 500, mu=400, max_iter=150)
            c2o = compute_c5(h2_opt)
            if c2o < best_c5:
                best_c5, best_h = c2o, h2_opt.copy()

    # Phase 7: Final coordinate descent polish
    if rem() > 25:
        n_cd = len(best_h)
        h_cd, c_cd = coord_descent(best_h, n_cd, max_rounds=5,
                                   steps=[0.015, -0.015, 0.006, -0.006, 0.002, -0.002])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Final subgradient micro-polish
    if rem() > 10:
        h_fin, c_fin = subgradient_polish(best_h, len(best_h), n_iters=5000, top_k=3, lr_init=0.001, decay=0.9999)
        if c_fin < best_c5:
            best_c5, best_h = c_fin, h_fin.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
