   w_k ← w_k * exp(f_k(h)/τ) / Σ_j w_j * exp(f_j(h)/τ)
4. Repeat

This is essentially a multiplicative weights / exponential weights approach to min-max. It's related to the "boosting" approach and to centroid methods for min-max.

The advantage: at each step, we minimize a single smooth function (not a max), which L-BFGS-B handles well. The weights adapt to focus on the hard lags.

Let me implement this:

```python
def minmax_reweight(h_start, n, max_rounds=20, tau=0.01, max_iter_lbfgs=50):
    dx = 2.0 / n
    h = h_start.copy()
    weights = np.ones(2*n-1) / (2*n-1)
    
    for r in range(max_rounds):
        # Define weighted objective
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            weighted = np.sum(weights * corr)
            return weighted * dx
        # ... compute gradient analytically?
        # For now, use L-BFGS-B with numerical gradient
        bounds = [(0.0, 1.0)] * n
        res = minimize(obj, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter_lbfgs, 'ftol': 1e-14})
        h = make_feasible(res.x, n)
        
        # Update weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        exp_w = weights * np.exp(corr / tau)
        weights = exp_w / np.sum(exp_w)
    
    return h, compute_c5(h)
```

This is nice but the inner optimization is expensive (50 L-BFGS iterations × 20 rounds = 1000 evaluations of the full correlation). At n=512, each evaluation is ~262K ops, so total is ~262M ops. Fast.

But the gradient of the weighted objective is:
∂/∂h[i] Σ_k w_k corr[k]

where corr[k] = Σ_i h[i+k](1-h[i]) (for k≥0) or similar.

This is actually a quadratic form in h, so the gradient is:
∂corr[k]/∂h[i] involves h[i+k] and h[i] terms.

Let me derive it:
corr[k] for k in [0, n-1]: Σ_{i=0}^{n-1-k} h[i+k] - h[i+k]*h[i]
∂corr[k]/∂h[j] = -h[j-k] (if j≥k, from the h[i+k]*h[i] term with i=j-k) 
               - h[j+k] (if j+k < n, from the h[i+k]*h[i] term with i=k-j... wait)

This is getting complicated. Let me just use numerical gradients or the automatic approach of L-BFGS-B.

Actually, for the weighted sum:
Σ_k w_k corr[k] = Σ_k w_k Σ_i h[i+k](1-h[i])

This is a quadratic form: h^T A h + b^T h (where A is a symmetric matrix). L-BFGS-B can handle this with numerical gradients fine.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_weighted(h_start, n, weights, max_iter=200):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        weighted = np.dot(weights, corr) * dx
        viol = np.sum(h) - n / 2.0
        return weighted + 2000.0 * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def lbgfb_smooth(h_start, n, T, mu=1000.0, max_iter=250):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def minimax_reweight(h_start, n, max_rounds=15, tau_init=0.02, max_iter_lbfgs=80, check_time=None):
    dx = 2.0 / n
    n_lags = 2 * n - 1
    h = h_start.copy()
    weights = np.ones(n_lags) / n_lags
    tau = tau_init
    for r in range(max_rounds):
        if check_time is not None and check_time() < 15:
            break
        h = lbgfb_weighted(h, n, weights, max_iter=max_iter_lbfgs)
        corr = np.correlate(h, 1.0 - h, mode='full')
        exp_w = weights * np.exp(corr / tau)
        s = np.sum(exp_w)
        if s > 0:
            weights = exp_w / s
        tau = max(0.001, tau * 0.7)
    return h

def binary_sa(h_bin, n, rng, max_iters=40000, T0=0.004, T1=0.00005, block_size=3):
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    n_iters = max_iters
    for it in range(n_iters):
        T = T0 * (T1 / T0) ** (it / n_iters)
        if it % 1000 == 0:
            ones = np.where(h > 0.5)[0].copy()
            zeros = np.where(h <= 0.5)[0].copy()
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Mix of single swaps and block moves
        if rng.random() < 0.7:
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                current_c5 = new_c5
                ones[ii] = oj
                zeros[jj] = oi
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        else:
            # Block move: swap a block of 1s with a block of 0s
            bs = rng.randint(1, block_size + 1)
            oi = rng.randint(0, max(1, len(ones) - bs))
            jj = rng.randint(0, max(1, len(zeros) - bs))
            block_ones = ones[oi:oi+bs]
            block_zeros = zeros[jj:jj+bs]
            if len(block_ones) != bs or len(block_zeros) != bs:
                continue
            h[block_ones] = 0.0
            h[block_zeros] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                current_c5 = new_c5
                ones[oi:oi+bs] = block_zeros
                zeros[jj:jj+bs] = block_ones
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            else:
                h[block_ones] = 1.0
                h[block_zeros] = 0.0
    return best_h, best_c5

def subgradient_polish(h_start, n, max_rounds=50, lr=0.02, top_k=20):
    dx = 2.0 / n
    h = h_start.copy()
    current = compute_c5(h)
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        threshold = mc - 0.05 * mc
        active = np.where(corr >= threshold)[0]
        if len(active) == 0:

            # Let's just use finite differences
            eps = 1e-5

                kk = active[attempt]

                pass
        # Simpler: just do a finite-difference subgradient on the max
        eps = 1e-6
        grad_max = np.zeros(n)
        for j in range(0, n, max(1, n // (2 * top_k))):
            hp = h.copy()
            hp[j] = min(1.0, h[j] + eps)
            grad_max[j] = (compute_c5(hp) - current) / eps
        # Scale and apply
        g_norm = np.linalg.norm(grad_max)
        if g_norm > 1e-12:

            h_new = make_feasible(h_new, n)

            if new_c5 < current - 1e-12:
```
