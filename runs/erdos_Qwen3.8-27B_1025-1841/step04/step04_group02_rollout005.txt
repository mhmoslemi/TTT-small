def soft_max_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc

    s = np.sum(exps)
    p = exps / s
    c5 = (mc + np.log(s) / T) * dx
    
    # Gradient
    shifts_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)

        pk = p[k_idx]
        if pk < 1e-16:

        if j_hi > j_lo:
            idx = np.arange(j_lo, j_hi)
            grad[idx] += pk * (1.0 - h[idx + sk])

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_c5_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exps = np.exp(T * shifted)
    s = np.sum(exps)
    p = exps / s
    c5 = (mc + np.log(s) / T) * dx
    shifts_arr = np.arange(2 * n - 1) - (n - 1)
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        pk = p[k_idx]
        if pk < 1e-16:
            continue
        sk = shifts_arr[k_idx]
        jj = 0
        if sk >= 0:
            jj = 0
            jhi = n - sk
        else:
            jj = -sk
            jhi = n
        if jhi > jj:
            idx = np.arange(jj, jhi)
            grad[idx] += pk * (1.0 - h[idx + sk])
        if sk >= 0:
            jj2 = sk
            jhi2 = n
        else:
            jj2 = 0
            jhi2 = n + sk
        if jhi2 > jj2:
            idx2 = np.arange(jj2, jhi2)
            grad[idx2] -= pk * h[idx2 - sk]
    grad *= dx
    return c5, grad

    momentum = np.zeros(n)
    lr = lr_init
    best_local = h_curr.copy()
    best_local_c5 = compute_c5(h_curr)
    stall_count = 0
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        T = T_lo * (T_hi / T_lo) ** frac
        c5_curr, grad = soft_max_c5_grad(h_curr, T)
        if c5_curr < best_local_c5 - 1e-15:
            best_local_c5 = c5_curr
            best_local = h_curr.copy()
            stall_count = 0
        else:
            stall_count += 1
        momentum = mom_decay * momentum - lr * grad
        h_new = h_curr + momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_new)
        if diff > 0:
            mask = h_new < 1.0
        else:
            mask = h_new > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h_new[mask] += diff / cnt
            h_new = np.clip(h_new, 0.0, 1.0)
        c5_new = compute_c5(h_new)

def coord_descent_fast(h, n, max_rounds, start_time, budget_s):

    best_c5 = compute_c5(h)
    dx = 2.0 / n
    shifts_arr = np.arange(2 * n - 1) - (n - 1)

                    val += 1.0 - h[jp]
                jm = j - sk
                if 0 <= jm < n:
                    val += h[jm]
                w[k_idx] = val
            best_b = old
            best_val = best_c5
            cands = [0.0, 1.0]
            top_idx = np.argsort(corr)[-4:]
            for ii in range(len(top_idx)):

                    denom = w[top_idx[ii]] - w[top_idx[jj]]

                        b_try = old - (corr[top_idx[ii]] - corr[top_idx[jj]]) / denom
                        cands.append(float(np.clip(b_try, 0.0, 1.0)))

                h = make_feasible(h, n)
                best_c5 = compute_c5(h)
                corr = np.correlate(h, 1.0 - h, mode='full')
                improved = True
        if not improved:
            break
    return h, best_c5

def binary_sa(h, n, max_iters, T0, T1, rng):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = it / max(max_iters - 1, 1)
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def slsqp_run(h, n, T, maxiter, ftol):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol, 'maxls': 30})
    return res.x
