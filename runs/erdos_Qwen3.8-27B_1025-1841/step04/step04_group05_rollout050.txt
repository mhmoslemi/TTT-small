def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj_and_grad(h, n, T):

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def sm_obj_grad(h, n, T):
    dx = 2.0 / n
    h1 = 1.0 - h
    corr = np.correlate(h, h1, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -600, 600)
    exp_s = np.exp(shifted)
    s_exp = np.sum(exp_s)
    log_sum = np.log(s_exp)
    val = (mc + log_sum / T) * dx
    p = exp_s / s_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        pm = p[m]
        if pm < 1e-13:
            continue
        k = m - (n - 1)
        if k >= 0:

            if n - k > 0:

        else:

            if n - kp > 0:
                grad[:n - kp] += pm * h1[kp:]
            if kp < n:
                grad[kp:] -= pm * h[:n - kp]
    return val, dx * grad

                   bounds=bounds,
                   options={'maxiter': max_iter, 'gtol': 1e-12, 'maxls': 20})

            if c5_n < c5_c - 1e-14:
                h_curr, c5_c = h_new, c5_n
                if c5_c < best_c5:
                    best_c5, best_h = c5_c, h_curr.copy()
                stall = 0
                alpha = min(alpha * 1.005, 1.5)
            else:
                stall += 1
                alpha = max(alpha * 0.98, 0.05)
            if stall > 500:
                break
        if tl() > 25:
            try:
                h_opt = lbfgsb_opt(h_curr, n1, 300, 150)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
            except Exception:
                pass

    # Phase 3: L-BFGS-B at n=2048
    if tl() > 180:
        n2 = 2048
        h_curr = upsample(best_h, n2)
        c = compute_c5(h_curr)
        if c < best_c5:
            best_c5, best_h = c, h_curr.copy()
        for T in [60, 150, 400]:
            if tl() < 40:
                break
            try:
                h_opt = lbfgsb_opt(h_curr, n2, T, 200)
                c = compute_c5(h_opt)
                h_curr = h_opt
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
            except Exception:
                pass

        # Subgradient at n=2048
        if tl() > 60:
            alpha = 0.3
            c5_c = compute_c5(h_curr)
            stall = 0
            for it in range(4000):
                if tl() < 15:
                    break
                h_new, _ = subgrad_step(h_curr, n2, alpha)
                h_new = make_feasible(h_new, n2)
                c5_n = compute_c5(h_new)
                if c5_n < c5_c - 1e-14:
                    h_curr, c5_c = h_new, c5_n
                    if c5_c < best_c5:
                        best_c5, best_h = c5_c, h_curr.copy()
                    stall = 0
                    alpha = min(alpha * 1.005, 1.2)
                else:
                    stall += 1
                    alpha = max(alpha * 0.98, 0.03)
                if stall > 400:
                    break

    # Phase 4: Binary local search at n=1024
    if tl() > 60:
        n_b = 1024
        h_bin_start = upsample(best_h, n_b)
        h_bin, c5_btn = binary_local_search(h_bin_start, n_b, rng, 1500)
        if c5_btn < best_c5:
            best_c5 = c5_btn
            best_h = h_bin.copy()
        # Also try from a random binary init
        h_rand_bin = (rng.rand(n_b) < 0.5).astype(float)
        h_rand_bin = make_feasible(h_rand_bin, n_b)
        h_rb, c5_rb = binary_local_search(h_rand_bin, n_b, rng, 800)
        if c5_rb < best_c5:
            best_c5, best_h = c5_rb, h_rb.copy()

    # Phase 5: Coordinate descent
    if tl() > 40:
        n_cd = min(len(best_h), 1024)
        h_cd_start = upsample(best_h, n_cd)
        h_cd, c5_cd = coord_descent(h_cd_start, n_cd, rng, 4)
        if c5_cd < best_c5:
            if n_cd < len(best_h):
                best_h = upsample(h_cd, len(best_h))
                best_c5 = compute_c5(best_h)
            else:
                best_h, best_c5 = h_cd, c5_cd

    # Phase 6: Final L-BFGS-B polish at high resolution
    if tl() > 50:
        n_f = min(len(best_h), 2048)
        h_fs = upsample(best_h, n_f)
        for T in [200, 600]:
            if tl() < 20:
                break
            try:
                h_opt = lbfgsb_opt(h_fs, n_f, T, 150)
                c = compute_c5(h_opt)
                h_fs = h_opt
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
            except Exception:
                pass

    # Phase 7: Try n=4096 if time permits
    if tl() > 120:
        n3 = 4096
        h_curr = upsample(best_h, n3)
        c = compute_c5(h_curr)
        if c < best_c5:
            best_c5, best_h = c, h_curr.copy()
        for T in [200, 500]:
            if tl() < 30:
                break
            try:
                h_opt = lbfgsb_opt(h_curr, n3, T, 150)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5, best_h = c, h_opt.copy()
            except Exception:
                pass
        if tl() > 40:
            alpha = 0.2
            c5_c = compute_c5(h_curr)
            for it in range(2000):
                if tl() < 10:
                    break
                h_new, _ = subgrad_step(h_curr, n3, alpha)
                h_new = make_feasible(h_new, n3)
                c5_n = compute_c5(h_new)
                if c5_n < c5_c - 1e-14:
                    h_curr, c5_c = h_new, c5_n
                    if c5_c < best_c5:
                        best_c5, best_h = c5_c, h_curr.copy()
                    alpha = min(alpha * 1.005, 1.0)
                else:
                    alpha = max(alpha * 0.98, 0.02)

    # Final
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
