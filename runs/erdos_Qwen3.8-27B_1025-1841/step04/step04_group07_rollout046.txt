        return float(active @ c) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def softmax_opt(h, n, T, mu, max_iter):
    dx = 2.0 / n

    def obj(x):
        c = all_corrs(x)
        return softmax_max(c, T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def flatten_opt(h, n, topk, mu, max_iter):
    dx = 2.0 / n
    corr = all_corrs(h)
    thresh = np.sort(corr)[-topk].min()

    def obj(x):
        c = all_corrs(x)
        cmax = np.max(c)
        excess = np.maximum(c - thresh, 0.0)
        return cmax * dx + 0.05 * (excess ** 2).sum() * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_ls(h_bin, n, rng, max_rounds, tries, use_2swap):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()

    def eval_c(h_):
        return compute_c5(h_)

    for rnd in range(max_rounds):
        improved = False
        for _ in range(tries):
            if not ones or not zeros:
                break
            if use_2swap and rnd % 3 != 0 and len(ones) >= 2 and len(zeros) >= 2:
                i1, i2 = rng.randint(len(ones), 2)
                j1, j2 = rng.randint(len(zeros), 2)
                if i1 == i2 or j1 == j2:
                    continue
                oi1, oi2, oj1, oj2 = ones[i1], ones[i2], zeros[j1], zeros[j2]
                h[oi1] = h[oi2] = 0.0
                h[oj1] = h[oj2] = 1.0
                c = eval_c(h)
                if c < cur - 1e-15:
                    cur = c
                    ones[i1], ones[i2] = oj1, oj2
                    zeros[j1], zeros[j2] = oi1, oi2
                    improved = True
                else:
                    h[oi1] = h[oi2] = 1.0
                    h[oj1] = h[oj2] = 0.0
            else:
                ii, jj = rng.randint(len(ones)), rng.randint(len(zeros))
                oi, oj = ones[ii], zeros[jj]
                h[oi] = 0.0
                h[oj] = 1.0
                c = eval_c(h)
                if c < cur - 1e-15:
                    cur = c
                    ones[ii] = oj
                    zeros[jj] = oi
                    improved = True
                else:
                    h[oi] = 1.0
                    h[oj] = 0.0
        if not improved:
            break
    return h, cur

def subgradient_polish(h, n, steps, step_size):
    dx = 2.0 / n
    h = h.copy()
    cur = compute_c5(h)
    for s in range(steps):
        corr = all_corrs(h)
        kmax = np.argmax(corr)
        # gradient of corr[kmax] w.r.t. h[j]
        # corr[kmax] = sum_i h[i]*(1-h[i+kmax-n+1])
        # grad[j] = (1-h[j+kmax-n+1]) - h[j-kmax+n-1] (periodic)
        lag = kmax - (n - 1)
        fwd = np.roll(h, -lag)
        bwd = np.roll(h, lag)
        grad = (1.0 - fwd) - bwd
        # subgradient step
        h_new = h - step_size * grad
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < cur - 1e-16:
            h = h_new
            cur = c_new
        else:
            step_size *= 0.7
            if step_size < 1e-6:
                break
    return h, cur

def coord_polish(h, n, max_rounds, steps_list):
    h = h.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        any_imp = False
        for j in range(n):
            ov = h[j]
            for d in steps_list:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    any_imp = True
                    break
        if not any_imp:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Upsample to 1280 and optimize
    n1 = 1280
    h1 = upsample(best_h, n1)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1.copy()

    # Active set optimization (top 3)
    for _ in range(4):
        if rem() < 60:
            break
        h1 = active_set_opt(h1, n1, topk=3, mu=500, max_iter=50)
        c = compute_c5(h1)
        if c < best_c5:
            best_c5, best_h = c, h1.copy()

    # Softmax refinement
    for T in [5000, 20000]:
        if rem() < 40:
            break
        h_new = softmax_opt(h1, n1, T, mu=500, max_iter=150)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new.copy()
            h1 = h_new.copy()

    # Flattening
    if rem() > 40:
        h_new = flatten_opt(h1, n1, topk=8, mu=500, max_iter=100)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new.copy()
            h1 = h_new.copy()

    # Another active set round with top 5
    for _ in range(3):
        if rem() < 50:
            break
        h1 = active_set_opt(h1, n1, topk=5, mu=400, max_iter=40)
        c = compute_c5(h1)
        if c < best_c5:
            best_c5, best_h = c, h1.copy()

    # Subgradient polish at n=1280
    if rem() > 30:
        h1, c1 = subgradient_polish(h1, n1, steps=200, step_size=0.005)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

    # Phase 2: Binary local search at n=1280 with multi-swap
    if rem() > 120:
        h_b = (h1 > 0.5).astype(float)
        target = n1 // 2
        n_ones = int(h_b.sum())
        if n_ones > target:
            idx = np.where(h_b == 1)[0]
            h_b[idx[:n_ones - target]] = 0
        elif n_ones < target:
            idx = np.where(h_b == 0)[0]
            h_b[idx[:target - n_ones]] = 1
        h_b, c_b = binary_ls(h_b, n1, rng, max_rounds=40, tries=6000, use_2swap=True)
        if c_b < best_c5:
            best_c5, best_h = c_b, h_b.copy()

    # Phase 3: Try n=2560 if time allows
    if rem() > 250:
        n2 = 2560
        h2 = upsample(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

        if rem() > 100:
            h2 = active_set_opt(h2, n2, topk=3, mu=400, max_iter=40)
            c = compute_c5(h2)
            if c < best_c5:
                best_c5, best_h = c, h2.copy()

        if rem() > 80:
            h2 = softmax_opt(h2, n2, T=10000, mu=400, max_iter=100)
            c = compute_c5(h2)
            if c < best_c5:
                best_c5, best_h = c, h2.copy()

        if rem() > 60:
            h2, c2 = subgradient_polish(h2, n2, steps=150, step_size=0.003)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 4: Binary local search at a good resolution
    if rem() > 150:
        n_bs = 1280
        h_bs = upsample(best_h, n_bs) if len(best_h) != n_bs else best_h.copy()
        h_bin = (h_bs > 0.5).astype(float)
        target = n_bs // 2
        n_ones = int(h_bin.sum())
        if n_ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target]] = 0
        elif n_ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target - n_ones]] = 1
        h_bin, c_bin = binary_ls(h_bin, n_bs, rng, max_rounds=35, tries=8000, use_2swap=True)
        if c_bin < best_c5:
            best_c5, best_h = c_bin, h_bin.copy()

    # Phase 5: Final subgradient + coordinate polish
    if rem() > 20:
        n_f = len(best_h)
        h_f, c_f = subgradient_polish(best_h, n_f, steps=300, step_size=0.003)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    if rem() > 15:
        h_f, c_f = coord_polish(best_h, len(best_h), max_rounds=4,
                                steps_list=[0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005])
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
