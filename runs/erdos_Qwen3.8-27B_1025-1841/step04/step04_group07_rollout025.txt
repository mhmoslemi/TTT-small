import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-10
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def softmax_obj(corr_vals, T):
    mc = np.max(corr_vals)
    shifted = np.clip(T * (corr_vals - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def adaptive_weight_optimize(h0, n, n_rounds=10, T_base=50, top_frac=0.1, mu=500):
    """Optimize with adaptive weights focusing on worst lags."""
    dx = 2.0 / n
    h = h0.copy()
    best_c = compute_c5(h)
    
    for rnd in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        mc = np.max(c_vals)
        
        # Create adaptive weights: higher weight for larger correlation values
        # Use exponential weighting
        T_w = T_base
        weights = np.exp(T_w * (c_vals - mc))
        weights = np.clip(weights, 1e-10, 1.0)
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full') * dx
            return float(np.dot(weights, c)) + mu * (x.sum() - n/2.0)**2
        
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12})
        h_new = make_feasible(res.x, n)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            h = h_new.copy()
    
    return h, best_c

def binary_sa(h0, n, rng, t_start=1.0, t_end=0.01, n_sweeps=200):
    """Simulated annealing for binary problem."""
    h = (h0 > 0.5).astype(float).copy()
    # Ensure exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        h[idx[rng.randint(len(idx), size=n_ones-target)]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        h[idx[rng.randint(len(idx), size=target-n_ones)]] = 1
    
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for sweep in range(n_sweeps):
        T = t_start * (t_end/t_start) ** (sweep / n_sweeps)
        n_tries = n // 2  # try n/2 swaps per sweep
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            new_c = compute_c5(h)
            delta = new_c - cur_c
            if delta < 0 or rng.random() < np.exp(-delta/T):
                cur_c = new_c
                ones[ii] = oj
                zeros[jj] = oi
                if cur_c < best_c:
                    best_c = cur_c
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oj] = 0.0
    
    return best_h, best_c

def refine_binary_to_continuous(h_bin, n, mu=1000):
    """Refine a binary solution to continuous."""
    h = h_bin.astype(float).copy()
    # Add small perturbation to break symmetry
    h = np.clip(h + 0.01 * (0.5 - h), 0, 1)
    h = make_feasible(h, n)
    
    dx = 2.0/n
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        c = corr * dx
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(np.clip(200*(c-mc), -80, 80))))/200) + mu*(x.sum()-n/2)**2
    
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0,1)]*n,
                   options={'maxiter': 500, 'ftol': 1e-16})
    h_new = make_feasible(res.x, n)
    
    # Second pass with higher T
    def obj2(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        c = corr * dx
        mc = np.max(c)
        return (mc + np.log(np.sum(np.exp(np.clip(500*(c-mc), -80, 80))))/500) + mu*(x.sum()-n/2)**2

def coord_descent(h0, n, max_rounds=20, steps=None):
    """Coordinate descent with adaptive steps."""
    if steps is None:
        steps = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            best_v = h[j]
            best_c = cur
            for d in steps:
                nv = h[j] + d
                if nv < -0.001 or nv > 1.001:
                    continue
                trial = h.copy()
                trial[j] = np.clip(nv, 0, 1)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = trial[j]
            if best_v != h[j]:
                h = make_feasible(h.copy(), n)
                h[j] = best_v
                h = make_feasible(h, n)
                cur = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, cur

- For k = 0: sum_t a[t-(n-1)] * v[t] → only t = n-1 gives a valid index for a (a[0]), so sum = a[0]*v[n-1]
  
Hmm, that doesn't seem right. Let me look up the actual definition.

Actually, from the scipy docs:
"correlate(a, v) = sum_n a[n+k] * conj(v[n])"

For mode='full', k ranges from -(len(v)-1) to (len(a)-1), and the output is arranged with k = -(len(v)-1) first.

So output[k + (len(v)-1)] = sum_n a[n+k] * conj(v[n])

With a=h, v=1-h, both of length n:
output[k + n - 1] = sum_n h[n+k] * (1-h[n])

where the sum is over n such that 0 ≤ n < n and 0 ≤ n+k < n.

So for shift k (mathematical), the output index is k + n - 1.

For k = 0 (zero shift): output[n-1] = sum_n h[n] * (1-h[n]) = sum h[n] - sum h[n]² = n/2 - sum h[n]²

For k > 0: output[n-1+k] = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
For k < 0: output[n-1+k] = sum_{n=-k}^{n-1} h[n+k] * (1-h[n])

So in the `grad_single_lag` function, when I want the gradient of c[k] (mathematical shift k), I should use the formula I derived above, and the corresponding index in the full correlation output is n-1+k.

Let me fix the code:

def subgradient_descent(h0, n, n_iters=10000, lr=0.005, top_k=5, T=50):
    h = h0.copy()

    best_c = compute_c5(h)
    best_h = h.copy()

        corr = np.correlate(h, 1.0-h, mode='full')
        c_vals = corr * dx
        mc = np.max(c_vals)
        
        # Top-k lags (in output index)
        k_sel = min(top_k, len(c_vals))
        top_idx = np.argpartition(c_vals, -k_sel)[-k_sel:]
        
        # Weights
        w = np.exp(T * (c_vals[top_idx] - mc))
        w /= w.sum()
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = idx - (n - 1)  # mathematical shift
            grad += wi * grad_single_lag(h, k, n)
        
        # Gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-10
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def grad_corr_lag(h, k, n):
    """Gradient of c[k] (mathematical shift k) w.r.t. h."""
    g = np.zeros(n)
    omh = 1.0 - h
    if k >= 0:
        g[k:] = omh[:n-k]
        g[:n-k] -= h[k:]
    else:
        m = -k
        g[:n-m] = omh[m:]
        g[m:] -= h[:n-m]
    return g

def subgrad_method(h0, n, n_iters=5000, lr_init=0.002, top_k=10, T=100, decay=0.999):
    """Subgradient method with adaptive weights."""
    h = h0.copy()
    dx = 2.0 / n
    best_c = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0-h, mode='full')
        c_vals = corr * dx
        mc = np.max(c_vals)
        
        k_sel = min(top_k, len(c_vals))
        top_idx = np.argpartition(c_vals, -k_sel)[-k_sel:]
        
        w = np.exp(T * (c_vals[top_idx] - mc))
        w /= w.sum()
        
        grad = np.zeros(n)
        for wi, idx in zip(w, top_idx):
            k = idx - (n-1)
            grad += wi * grad_corr_lag(h, k, n)
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
        
        h = h_new
        lr *= decay
    
    return best_h, best_c

def lbgfb_topk(h0, n, k=20, mu=500, max_iter=200):
    """L-BFGS-B minimizing the max of top-k correlation values (via softmax)."""
    dx = 2.0 / n
    h_init = h0.copy()
    
    # Precompute which lags to focus on (recomputed each call)
    def obj(x):
        corr = np.correlate(x, 1.0-x, mode='full')
        c = corr * dx
        # Softmax over all lags
        mc = np.max(c)
        T = 300.0
        shifted = np.clip(T * (c - mc), -80, 80)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) + mu * (x.sum()-n/2)**2
    
    res = minimize(obj, h_init, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)]*n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_sa_fast(h0, n, rng, n_sweeps=500, t0=0.3, t1=0.005):
    """Fast binary SA using precomputed increment."""
    h = (h0 > 0.5).astype(float).copy()
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        idx = np.where(h==1)[0]
        h[idx[rng.choice(len(idx), size=n_ones-target, replace=False)]] = 0
    elif n_ones < target:
        idx = np.where(h==0)[0]
        h[idx[rng.choice(len(idx), size=target-n_ones, replace=False)]] = 1
    
    cur_c = compute_c5(h)
    best_h = h.copy()
    best_c = cur_c
    ones = np.where(h>0.5)[0].tolist()
    zeros = np.where(h<=0.5)[0].tolist()
    
    for sweep in range(n_sweeps):
        T = t0 * (t1/t0)**(sweep/n_sweeps)
        for _ in range(n//4):  # fewer tries per sweep for speed
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0; h[oj] = 1
            new_c = compute_c5(h)
            d = new_c - cur_c
            if d < 0 or rng.random() < np.exp(-d/T):
                cur_c = new_c
                ones[ii] = oj; zeros[jj] = oi
                if cur_c < best_c:
                    best_c = cur_c; best_h = h.copy()
            else:
                h[oi] = 1; h[oj] = 0
    
    return best_h, best_c
