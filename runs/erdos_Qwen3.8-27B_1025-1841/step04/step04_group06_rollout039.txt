def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def softmax_gradient(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -80, 80)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        m = k - (n - 1)
        a = max(0, -m)
        b = min(n-1, n-1-m)
        if b < a:
            continue
        # Term 1: (1-h[j+m]) for j in [a, b]
        # j ranges from a to b, h[j+m] where j+m ranges from a+m to b+m
        # a+m = max(0,-m)+m = max(m, 0)
        # b+m = min(n-1, n-1-m)+m = min(n-1+m, n-1)
        j1s = a
        j1e = b
        i1s = a + m  # = max(m, 0)
        i1e = b + m  # = min(n-1+m, n-1)
        grad[j1s:j1e+1] += wk * (1.0 - h[i1s:i1e+1])
        
        # Term 2: -h[j-m] for j-m in [a, b], i.e., j in [a+m, b+m]
        j2s = a + m   # = max(m, 0)
        j2e = b + m   # = min(n-1+m, n-1)
        i2s = a
        i2e = b
        # But we need j2s, j2e in [0, n-1]
        j2s_c = max(0, j2s)
        j2e_c = min(n-1, j2e)
        if j2s_c <= j2e_c:
            i2s_c = j2s_c + m  # wait, j-m = i, so i = j-m
            # When j = j2s_c, i = j2s_c - m = a (by construction)
            # When j = j2e_c, i = j2e_c - m
            i_start = j2s_c - m
            i_end = j2e_c - m
            grad[j2s_c:j2e_c+1] += wk * (-h[i_start:i_end+1])
    
    return grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)[-3:]
        else:
            idx = np.argsort(h)[:3]
        for i in idx:
            adj = np.clip(diff, -(h[i]), 1.0-h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def softmax_obj_grad(h, n, T, mu=100.0):
    """ Computes objective and gradient for soft-max surrogate. """
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -80, 80)
    exp_s = np.exp(shifted)
    w = exp_s / exp_s.sum()
    soft_val = (mc + np.log(exp_s.sum()) / T) * dx
    viol = np.sum(h) - n / 2.0
    val = soft_val + mu * viol * viol
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k] * dx
        if wk < 1e-12:
            continue
        m = k - (n - 1)
        a = max(0, -m)
        b = min(n-1, n-1-m)
        if b < a:
            continue
        # Term 1: (1-h[j+m]) for j in [a, b]
        i1s = a + m
        i1e = b + m
        grad[a:b+1] += wk * (1.0 - h[i1s:i1e+1])
        # Term 2: -h[j-m] for j in [a+m, b+m] (clipped)
        j2s = a + m
        j2e = b + m
        j2s_c = max(0, j2s)
        j2e_c = min(n-1, j2e)
        if j2s_c <= j2e_c:
            i_s = j2s_c - m
            i_e = j2e_c - m
            grad[j2s_c:j2e_c+1] += wk * (-h[i_s:i_e+1])
    grad += 2.0 * mu * viol * np.ones(n)
    return val, grad

def lbgfb_with_grad(h_start, n, T_schedule, mu=100.0, max_iter=200):
    h = h_start.copy()
    for T in T_schedule:
        def f(hv):
            val, _ = softmax_obj_grad(hv, n, T, mu)
            return val
        def g(hv):
            _, grad = softmax_obj_grad(hv, n, T, mu)
            return grad
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(f, h, jac=g, method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-9, 'maxls': 20})
            h = make_feasible(res.x, n)
        except Exception:
            pass
    return h

def subgradient_descent(h_start, n, lr_init, n_iter=2000, T=200, momentum=0.9):
    """Subgradient descent with momentum and adaptive step size."""
    h = h_start.copy()
    v = np.zeros(n)
    lr = lr_init
    dx = 2.0 / n
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        exp_s = np.exp(shifted)
        w = exp_s / exp_s.sum()
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2*n - 1):
            wk = w[k]
            if wk < 1e-10:
                continue
            m = k - (n - 1)
            a = max(0, -m)
            b = min(n-1, n-1-m)
            if b < a:
                continue
            i1s = a + m
            i1e = b + m
            grad[a:b+1] += wk * (1.0 - h[i1s:i1e+1])
            j2s = a + m
            j2e = b + m
            j2s_c = max(0, j2s)
            j2e_c = min(n-1, j2e)
            if j2s_c <= j2e_c:
                i_s = j2s_c - m
                i_e = j2e_c - m
                grad[j2s_c:j2e_c+1] += wk * (-h[i_s:i_e+1])
        
        # Nesterov momentum
        v = momentum * v - lr * grad
        h_new = h + v
        h = make_feasible(h_new, n)
        
        # Adaptive step size
        if it % 100 == 99:
            # Halve step if oscillating
            lr *= 0.95
    
    return h

def coord_descent_binary(h_start, n, rng, max_rounds=50):
    """Coordinate descent with binary flips and small adjustments."""
    h = h_start.copy()
    current_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        # Try binary flips
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        n_tries = min(3000, max(1, len(ones)*len(zeros)))
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi], h[oj] = 0.0, 1.0
            c = compute_c5(h)
            if c < current_c5 - 1e-15:
                current_c5 = c
                ones = np.append(np.delete(ones, ii), oj)
                zeros = np.append(np.delete(zeros, jj), oi)
                improved = True
            else:
                h[oi], h[oj] = 1.0, 0.0
        if not improved:
            break
    return h, current_c5

        return budget_s - (time.time() - start_time)
