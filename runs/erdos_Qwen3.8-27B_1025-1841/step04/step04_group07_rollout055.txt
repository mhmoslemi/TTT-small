def compute_topk_grad(h, k):
    n = len(h)
    omh = 1.0 - h
    corr = np.correlate(h, omh, mode='full')
    # Get top-k indices
    top_idx = np.argpartition(corr, -k)[-k:]
    grad = np.zeros(n)
    for m in top_idx:
        s = m - (n - 1)
        if s == 0:
            grad += 1.0 - 2.0 * h
        elif s > 0:
            grad[:n-s] += (1.0 - h[s:])  # (1-h[j+s]) for j=0..n-1-s
            grad[s:] += -h[:n-s]          # -h[j-s] for j=s..n-1
        else:
            t = -s
            grad[t:] += (1.0 - h[:n-t])   # (1-h[j-t]) for j=t..n-1
            grad[:n-t] += -h[t:]           # -h[j+t] for j=0..n-1-t
    grad /= k
    return corr, top_idx, grad
```

def nesterov_subgrad(h0, n, k_active, max_iters, t0_budget):
    h = h0.copy()
    h_prev = h.copy()
    y = h.copy()  # Nesterov momentum point
    alpha = 0.5 / n  # initial step
    beta = 0.9  # momentum
    best = (compute_c5(h), h.copy())
    
    for it in range(max_iters):
        if time.time() > t0_budget:
            break
        corr, top_idx, grad = compute_topk_grad(y, k_active)
        c = np.max(corr) * (2.0/n)
        if c < best[0]:
            best = (c, y.copy())
        
        # Nesterov step
        y = h + beta * (y - h_prev) - alpha * grad
        y = np.clip(y, 0, 1)
        # Enforce sum constraint
        y = make_feasible(y, n)
        
        h_prev = h
        h = y
        # Decay alpha
        alpha = max(0.1/n, alpha * 0.995)
    
    return best
```

        h_b = (upsample_h(best_h, n1) > 0.5).astype(float)
        target = n1 // 2
        n_ones = int(h_b.sum())
        if n_ones > target:
            idx = np.where(h_b == 1)[0]
            h_b[idx[:n_ones - target]] = 0.0
        elif n_ones < target:
            idx = np.where(h_b == 0)[0]
            h_b[idx[:target - n_ones]] = 1.0
        h_bs, c_bs = directed_binary_search(h_b, n1, rng, t0 + rem() - 10, max_rounds=200)
        if c_bs < best_c5:
            best_c5, best_h, best_n = c_bs, h_bs.copy(), n1

    # Phase 4: Higher resolution n=2560
    n2 = 2560
    if rem() > 200:
        h3 = upsample_h(best_h, n2)
        h3, c3 = momentum_subgrad(h3, n2, k_active=12, max_iters=3000, deadline=t0 + rem() - 60)
        if c3 < best_c5:
            best_c5, best_h, best_n = c3, h3.copy(), n2
        if rem() > 60:
            h3, c3 = lbgfb_soft(h3, n2, 500, 300.0, 200, t0 + rem() - 10)
            if c3 < best_c5:
                best_c5, best_h, best_n = c3, h3.copy(), n2

    # Phase 5: Periodic initializations
    if rem() > 120:
        n_p = 1280
        for period in [7, 5, 9]:
            if rem() < 40:
                break
            h_p = periodic_init(n_p, period)
            h_p, c_p = momentum_subgrad(h_p, n_p, k_active=15, max_iters=2000, deadline=t0 + rem() - 15)
            if c_p < best_c5:
                best_c5, best_h, best_n = c_p, h_p.copy(), n_p

    # Phase 6: Binary search on best at its resolution
    if rem() > 60:
        n_f = min(best_n, 1280)
        h_fb = (upsample_h(best_h, n_f) > 0.5).astype(float)
        target = n_f // 2
        n_ones = int(h_fb.sum())
        if n_ones > target:
            idx = np.where(h_fb == 1)[0]
            h_fb[idx[:n_ones - target]] = 0.0
        elif n_ones < target:
            idx = np.where(h_fb == 0)[0]
            h_fb[idx[:target - n_ones]] = 1.0
        h_fs, c_fs = directed_binary_search(h_fb, n_f, rng, t0 + rem() - 5, max_rounds=150)
        if c_fs < best_c5:
            best_c5, best_h, best_n = c_fs, h_fs.copy(), n_f

    # Phase 7: Final subgradient + L-BFGS at current best
    if rem() > 30:
        n_fin = best_n
        if n_fin > 2048:
            n_fin = 1280
        h_fin = upsample_h(best_h, n_fin)
        h_fin, c_fin = momentum_subgrad(h_fin, n_fin, k_active=15, max_iters=1500, deadline=t0 + rem() - 10)
        if c_fin < best_c5:
            best_c5, best_h, best_n = c_fin, h_fin.copy(), n_fin

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
