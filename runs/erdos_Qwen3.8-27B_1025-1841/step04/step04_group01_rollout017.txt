    # Phase 4: Coordinate descent (targeted)
    if remaining() > 40:
        for cd_pass in range(4):
            if remaining() < 20:
                break
            improved = False
            # Focus on variables that are near the active shifts
            corr = get_corr(best_h)
            mc = np.max(corr)
            active_k = np.where(corr >= mc - 0.02 * mc)[0]
            for j in range(n):
                if remaining() < 10:
                    break
                old = best_h[j]
                best_local = best_c5
                best_v = old
                for step in [0.001, 0.002, 0.005, 0.01, 0.02]:
                    for sign in [-1.0, 1.0]:
                        v = old + sign * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n)
                            c = compute_c5(trial)
                            if c < best_local - 1e-14:
                                best_local = c
                                best_v = v
                if best_v != old:
                    best_h[j] = best_v
                    best_h = make_feasible(best_h, n)
                    best_c5 = best_local
                    improved = True
            if not improved:
                break

    # Phase 5: More subgradient refinement
    if remaining() > 50:
        lr = 0.01
        stagnation = 0
        for it in range(300):
            if remaining() < 15:
                break
            sg, _ = exact_subgrad(best_h)
            h_try = best_h - lr * sg
            h_try = make_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
                best_h = h_try
                stagnation = 0
            else:
                stagnation += 1
                lr *= 0.97
            if stagnation > 40:
                break

    # Phase 6: Random restarts + L-BFGS-B
    for restart in range(4):
        if remaining() < 50:
            break
        scale = 0.008 * (1.0 + 0.5 * restart)
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        try:
            res = minimize(obj_fn, h_try, jac=grad_fn, args=(1000.0, 5000.0),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 250, 'ftol': 1e-16, 'gtol': 1e-12})
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    # Phase 7: Ultra-fine coordinate descent
    if remaining() > 25:
        for _ in range(2):
            if remaining() < 15:
                break
            for j in range(n):
                if remaining() < 5:
                    break
                old = best_h[j]
                for step in [0.0003, 0.0005, 0.001]:
                    for sign in [-1.0, 1.0]:
                        v = old + sign * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n)
                            c = compute_c5(trial)
                            if c < best_c5 - 1e-15:
                                best_c5 = c
                                best_h = trial
                                old = v
                                break

    # Phase 8: Final L-BFGS-B
    if remaining() > 20:
        try:
            res = minimize(obj_fn, best_h, jac=grad_fn, args=(8000.0, 50000.0),
                         method='L-BFGS-B', bounds=bounds,
                         options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-14})
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
