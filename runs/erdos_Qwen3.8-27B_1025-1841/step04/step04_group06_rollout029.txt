def fft_soft_max_c5_and_grad(h, n, T):
    """Compute soft-max C5 and gradient using FFT for efficiency."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -700, 700)
    exp_shifted = np.exp(shifted)
    log_sum_exp = mc + np.log(np.sum(exp_shifted)) / T
    c5_smooth = log_sum_exp * dx
    weights = exp_shifted / np.sum(exp_shifted)
    
    # Gradient computation (O(n^2) but vectorized)
    grad = np.zeros(n)
    for m in range(-(n-1), n):
        wm = weights[m + n - 1]
        if wm < 1e-15:
            continue
        j_start = max(0, m)
        j_end = min(n, m + n)
        if j_start < j_end:
            grad[j_start:j_end] += wm * (1.0 - h[j_start - m: j_end - m])
        j_start2 = max(0, -m)
        j_end2 = min(n, n - m)
        if j_start2 < j_end2:
            grad[j_start2:j_end2] -= wm * h[j_start2 + m: j_end2 + m]
    
    return c5_smooth, grad * dx


def projected_gradient_descent(h_init, n, T_init, T_final, n_iter, lr_init, lr_final, start_time, budget_s):
    """Projected gradient descent on the soft-max C5 objective."""
    h = h_init.copy()
    c5_best = compute_c5(h)
    h_best = h.copy()
    
    for it in range(n_iter):
        if time.time() - start_time > budget_s - 10:
            break
        T = T_init * (T_final / T_init) ** (it / max(n_iter - 1, 1))
        lr = lr_init * (lr_final / lr_init) ** (it / max(n_iter - 1, 1))
        
        c5_s, grad = fft_soft_max_c5_and_grad(h, n, T)
        h = h - lr * grad
        h = project_onto_simplex(h, n)
        
        c5 = compute_c5(h)
        if c5 < c5_best:
            c5_best = c5
            h_best = h.copy()
    
    return h_best, c5_best
