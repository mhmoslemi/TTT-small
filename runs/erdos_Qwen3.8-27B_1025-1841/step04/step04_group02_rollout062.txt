def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):

        if abs(diff) < 1e-14:

        if diff > 0:

        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:

        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5

**Alternative: Formulate as a linear program with binary variables**

For binary h with sum = n/2:
C₅ = (2/n) * max_m (n/2 - A[m]) where A[m] = sum_i h[i]*h[i+m-n+1]

We want to minimize C₅, i.e., maximize min_m A[m].

Let t = min_m A[m]. We want to maximize t s.t.:
- A[m] >= t for all m
- sum h[i] = n/2
- h[i] ∈ {0,1}

This is a 0-1 quadratic program. We can linearize A[m] = sum_i h[i]*h[i+m-n+1] by introducing y[i,m] = h[i]*h[i+m-n+1] with constraints:
- y[i,m] ≤ h[i]
- y[i,m] ≤ h[i+m-n+1]
- y[i,m] ≥ h[i] + h[i+m-n+1] - 1
- y[i,m] ≥ 0

But the number of (i,m) pairs is O(n²), and the number of binary variables is n + O(n²). For n=960, that's about 960 + 2*960*960 ≈ 1.8M binary variables. Way too many for any solver.

So LP is out. Let me stick with heuristics.

**Another idea: Use the "worst lag" to guide the search**

At each step of the greedy descent, instead of trying random swaps, try swaps that specifically improve the worst lag.

For the worst lag m*, we want to increase A[m*] = sum_i h[i]*h[i+m*-n+1].
To increase this, we want to create new pairs (i, i+m*-n+1) where both are 1.
If h[i]=0 and h[i+m*-n+1]=1, flipping h[i] to 1 would help (but we need to unflip something else to maintain the sum).

So: find i where h[i]=0 and h[i+m*-n+1]=1, and j where h[j]=1 and h[j+m*-n+1]=0 (or where the flip doesn't hurt as much). Try the swap h[i]=1, h[j]=0.

This is a targeted search that focuses on the bottleneck lag.

        b, c = sa_binary(best_h, len(best_h), 100000, 0.008, 0.000003, rng, 6)
        if c < best_c5:
            best_c5 = c
            best_h = b.copy()

    # Phase 5: Greedy descent again
    if rem() > 120:
        b, c = greedy_descent(best_h.copy(), len(best_h), 50, 25000, rng, t0, budget_s)
        if c < best_c5:
            best_c5 = c
            best_h = b.copy()

    # Phase 6: SLSQP with higher T
    if rem() > 80:
        try:
            x = slsqp_polish(best_h, len(best_h), 200.0, 400, 1e-14)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(x, len(best_h))
        except Exception:
            pass

    # Phase 7: Try higher resolution
    if rem() > 300:
        n_up = n0 + 480
        old_x = np.linspace(0, 2, len(best_h))
        new_x = np.linspace(0, 2, n_up)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n_up)
        b, c = sa_binary(h_up, n_up, 50000, 0.01, 0.00001, rng, 3)
        if c < best_c5:
            best_c5 = c
            best_h = b.copy()
        if rem() > 100:
            b, c = greedy_descent(best_h.copy(), len(best_h), 20, 15000, rng, t0, budget_s)
            if c < best_c5:
                best_c5 = c
                best_h = b.copy()
        if rem() > 80:
            try:
                x = slsqp_polish(best_h, len(best_h), 100.0, 300, 1e-13)
                c = compute_c5(x)
                if c < best_c5:
                    best_c5 = c
                    best_h = make_feasible(x, len(best_h))
            except Exception:
                pass

    # Phase 8: Try lower resolution
    if rem() > 250 and n0 > 480:
        n_dn = n0 // 2
        old_x = np.linspace(0, 2, len(best_h))
        new_x = np.linspace(0, 2, n_dn)
        h_dn = np.interp(new_x, old_x, best_h)
        h_dn = make_feasible(h_dn, n_dn)
        b, c = sa_binary(h_dn, n_dn, 40000, 0.015, 0.00001, rng, 3)
        if c < best_c5:
            best_c5 = c
            best_h = b.copy()
        if rem() > 100:
            b, c = greedy_descent(best_h.copy(), len(best_h), 30, 20000, rng, t0, budget_s)
            if c < best_c5:
                best_c5 = c
                best_h = b.copy()
        if rem() > 60:
            try:
                x = slsqp_polish(best_h, len(best_h), 100.0, 200, 1e-13)
                c = compute_c5(x)
                if c < best_c5:
                    best_c5 = c
                    best_h = make_feasible(x, len(best_h))
            except Exception:
                pass

    # Phase 9: Random restart SA (try escaping local optimum)
    if rem() > 250:
        h_rand = rng.rand(n0)
        h_rand = to_binary(h_rand, n0)
        b, c = sa_binary(h_rand, n0, 60000, 0.05, 0.00001, rng, 4)
        if c < best_c5:
            best_c5 = c
            best_h = b.copy()
            b, c = greedy_descent(best_h.copy(), n0, 30, 15000, rng, t0, budget_s)
            if c < best_c5:
                best_c5 = c
                best_h = b.copy()

    # Phase 10: Final greedy + SLSQP
    if rem() > 60:
        b, c = greedy_descent(best_h.copy(), len(best_h), 20, 15000, rng, t0, budget_s)
        if c < best_c5:
            best_c5 = c
            best_h = b.copy()

    if rem() > 60:
        try:
            x = slsqp_polish(best_h, len(best_h), 400.0, 300, 1e-15)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(x, len(best_h))
        except Exception:
            pass

    if rem() > 30:
        b, c = greedy_descent(best_h.copy(), len(best_h), 10, 10000, rng, t0, budget_s)
        if c < best_c5:
            best_c5 = c
            best_h = b.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
