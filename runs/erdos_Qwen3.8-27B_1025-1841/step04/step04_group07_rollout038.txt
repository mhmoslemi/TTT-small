    mc = np.max(corr)
    shifted = np.clip((corr - mc) * T, -50, 50)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_opt(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return softmax_obj(corr, T) * dx + mu * (x.sum() - n/2.0)**2
    
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def slsqp_opt(h0, n, T, max_iter):

        return np.sum(x) - n/2.0

        return np.ones(n)
    
    res = minimize(obj, h0, method='SLSQP',

def gradient_peak_shave(h0, n, n_iters=2000, rem_fn=None):

    target = n / 2.0
    lr = 0.005  # small learning rate

        if rem_fn is not None and rem_fn() < 10:
            break

        topk = 20
        top_idx = np.argpartition(corr, -topk)[-topk:]

        # Exponential weights
        w = np.exp(top_corr * 10 - np.max(top_corr) * 10)
        w /= w.sum()
        
        # Compute gradient of weighted sum
        g = np.zeros(n)
        shift_base = n - 1
        for i, m in enumerate(top_idx):
            if w[i] < 1e-6:
                continue
            shift = m - shift_base
            # ∂corr[m]/∂h[j] = (1-h[j-shift])*[0<=j-shift<n] - h[shift+j]*[0<=shift+j<n]

        h_new = h - lr * g
        h_new = np.clip(h_new, 0, 1)
        h_new = make_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def softmax_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip((corr - mc) * T, -50, 50)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def slsqp_opt(h0, n, T, max_iter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return softmax_obj(corr, T) * dx
    def con_eq(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def lbgfb_opt(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return softmax_obj(corr, T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def grad_corr_shift(h, k_full, n):
    shift = k_full - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, shift)
    hi1 = min(n, n + shift)
    if lo1 < hi1:
        g[lo1:hi1] += 1.0 - h[lo1 - shift:hi1 - shift]
    lo2 = max(0, -shift)
    hi2 = min(n, n - shift)
    if lo2 < hi2:
        g[lo2:hi2] -= h[shift + lo2:shift + hi2]
    return g

def gradient_peak_shave(h0, n, max_iters, time_limit, t_start):
    h = h0.copy()
    target = n / 2.0
    lr = 0.003
    decay = 0.999
    for t in range(max_iters):
        if time.time() - t_start > time_limit:
            break
        corr = get_corr(h)
        topk = 15
        top_idx = np.argpartition(corr, -topk)[-topk:]
        top_vals = corr[top_idx]
        w = np.exp((top_vals - np.max(top_vals)) * 15.0)
        w /= w.sum()
        g = np.zeros(n)
        for i, m in enumerate(top_idx):
            if w[i] < 1e-8:
                continue
            g += w[i] * grad_corr_shift(h, m, n)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        h_new += diff / n
        h_new = np.clip(h_new, 0.0, 1.0)
        diff = target - h_new.sum()
        h_new += diff / n
        h = h_new
        lr *= decay
    return h

def coord_descent(h0, n, max_rounds, time_limit, t_start):
    h = h0.copy()
    cur = compute_c5(h)
    steps = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t_start > time_limit:
                return h
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:

                trial = h.copy()
                trial[j] = nv

                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h

            if rem() < 50:
                break
            h_r = best_h + 0.05 * rng.randn(n)
            h_r = make_feasible(h_r, n)
            h_opt = slsqp_opt(h_r, n, T=300, max_iter=200)
            h_opt2 = slsqp_opt(h_opt, n, T=50, max_iter=150)
            c = compute_c5(h_opt2)
            if c < best_c5 - 1e-15:
                best_c5 = c
                best_h = h_opt2.copy()

    # Phase 6: Try different resolutions
    for n2 in [800, 1024]:
        if rem() < 120:
            break
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5 - 1e-15:
            best_c5 = c2
            best_h = h2.copy()
        for T in [500, 200, 50]:
            if rem() < 60:
                break
            h_opt = slsqp_opt(h2, n2, T, max_iter=150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5 - 1e-15:
                best_c5 = c_opt
                best_h = h_opt.copy()
                h2 = h_opt.copy()
        if rem() > 80:
            h_g2 = gradient_peak_shave(h2, n2, max_iters=2000,
                                       time_limit=rem() - 40, t_start=t0)
            c_g2 = compute_c5(h_g2)
            if c_g2 < best_c5 - 1e-15:
                best_c5 = c_g2
                best_h = h_g2.copy()

    # Phase 7: Binary local search
    if rem() > 60:
        n_bin = len(best_h)
        h_b = (best_h > 0.5).astype(float)
        target_ones = n_bin // 2
        n_ones = int(h_b.sum())
        if n_ones > target_ones:
            idx = np.where(h_b == 1)[0]
            h_b[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_b == 0)[0]
            h_b[idx[:target_ones - n_ones]] = 1.0
        h_bs = pair_swap_search(h_b, n_bin, rng, time_limit=min(rem() - 30, 120), t_start=t0)
        c_bs = compute_c5(h_bs)
        if c_bs < best_c5 - 1e-15:
            best_c5 = c_bs
            best_h = h_bs.copy()

    # Phase 8: Coordinate descent polish
    if rem() > 40:
        n_cd = min(len(best_h), 1024)
        h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        h_cd = coord_descent(h_cd, n_cd, max_rounds=8,
                             time_limit=rem() - 20, t_start=t0)
        c_cd = compute_c5(h_cd)
        if c_cd < best_c5 - 1e-15:
            best_c5 = c_cd
            best_h = h_cd.copy()

    # Phase 9: Final SLSQP
    if rem() > 20:
        h_f = slsqp_opt(best_h, len(best_h), T=30, max_iter=100)
        c_f = compute_c5(h_f)
        if c_f < best_c5 - 1e-15:
            best_c5 = c_f
            best_h = h_f.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
