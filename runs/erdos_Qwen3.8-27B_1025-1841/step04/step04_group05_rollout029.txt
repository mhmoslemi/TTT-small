```python
def binary_local_search(h, n, max_iter=10000, T_start=10.0, T_end=0.1):
    """Local search in the binary space with pair flips"""
    h = h.copy()
    # First, threshold to binary
    h = (h > 0.5).astype(float)
    h = make_feasible(h, n)
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        T = T_start * (T_end/T_start) ** (it / max_iter)
        # Pick a random 1 and a random 0 to flip
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        
        if i == j:
            continue
        
        trial = h.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        
        c = compute_c5(trial)
        delta = c - best_c5
        
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            h = trial
            # Update ones and zeros
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    
    return best_h, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        # Distribute to coordinates with most room
        if diff > 0:
            room = 1.0 - h[mask]
            order = np.argsort(-room)
        else:
            room = h[mask]
            order = np.argsort(-room)
        # Apply to top coordinates first
        step = diff / cnt
        h[new_mask] = np.clip(h[new_mask] + step, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def smoothed_obj_and_grad(h, T):
    """Compute smoothed max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exp_vals = np.exp(T * (corr - mc))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    
    obj = (mc + np.log(sum_exp) / T) * dx

    first_term = np.correlate(w, 1.0 - h, mode='valid')
    second_term = np.correlate(w_rev, h, mode='valid')[::-1]
    grad = (first_term - second_term) * dx
    
    return obj, grad
```

Let me also add a "structured perturbation" idea:
- Identify the active lags (those near the maximum)
- For each active lag k, the overlap is sum_i h[i]*(1-h[i+k])
- To reduce this, we want to make h[i] and h[i+k] more similar (both high or both low)
- This suggests a "smoothing" direction: for active lag k, push h[i] toward h[i+k]

```python
def lag_targeted_step(h, n, alpha=0.01):
    """Push h toward reducing overlap at active lags"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Active lags: those with corr > 0.95 * max
    active_mask = corr > 0.95 * max_val
    lags = np.arange(2*n-1) - (n-1)
    active_lags = lags[active_mask]
    
    # For each active lag k, add alpha * (h[i+k] - h[i]) to h[i]
    # (push h[i] toward h[i+k])
    grad = np.zeros(n)
    for k in active_lags:
        if k >= 0:
            # h[i+k] - h[i] for i = 0 to n-1-k
            grad[:n-k] += h[k:] - h[:n-k]
        else:
            # k < 0, let k' = -k
            k2 = -k
            # h[i-k'] - h[i] for i = k' to n-1
            grad[k2:] += h[:n-k2] - h[k2:]
    
    h_new = h + alpha * grad
    h_new = make_feasible(h_new, n)
    return h_new
```

```python
        temp_schedule_2 = [
            (60.0, 200),
            (30.0, 300),
            (15.0, 350),
            (8.0, 400),
            (4.0, 500),
        ]
        for T, max_it in temp_schedule_2:
            if _time_left() < 35:
                break
            try:
                h_opt = slsqp_opt(h_curr2, n2, T, max_it)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()
                    h_curr2 = h_opt.copy()
                else:
                    h_curr2 = h_opt.copy()
            except Exception:
                pass

    # Phase 6: Try even higher resolution if time allows
    n3 = 5120
    if _time_left() > 120:
        h_curr3 = upsample_h(best_h, n3)
        c = compute_c5(h_curr3)
        if c < best_c5:
            best_c5, best_h = c, h_curr3.copy()
        for T, max_it in [(20.0, 150), (10.0, 200), (5.0, 250)]:
            if _time_left() < 25:
                break
            try:
                h_opt = slsqp_opt(h_curr3, n3, T, max_it)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()
                    h_curr3 = h_opt.copy()
            except Exception:
                pass

    # Phase 7: Coordinate descent polish at best resolution
    n_cd = len(best_h)
    h_cd = best_h.copy()
    for _round in range(8):
        if _time_left() < 12:
            break
        improved = False
        for j in range(n_cd):
            if _time_left() < 8:
                break
            for bv in (0.0, 1.0):
                if abs(h_cd[j] - bv) < 1e-9:
                    continue
                trial = h_cd.copy()
                trial[j] = bv
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h_cd = trial
                    improved = True
                    break
        if not improved:
            break

    if compute_c5(h_cd) < best_c5:
        best_c5 = compute_c5(h_cd)
        best_h = h_cd.copy()

    # Phase 8: Final SLSQP pass
    if _time_left() > 25:
        n_final = len(best_h)
        try:
            h_opt = slsqp_opt(best_h, n_final, 15.0, 150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
        except Exception:
            pass
        if _time_left() > 15:
            try:
                h_opt = slsqp_opt(best_h, n_final, 5.0, 100)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()
            except Exception:
                pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
