def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def fix_binary_sum(h, n, rng):
    """Fix a binary vector to have exactly n/2 ones."""
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    n_ones = len(ones)
    if n_ones > target:
        # Remove excess ones
        excess = n_ones - target
        to_flip = rng.choice(ones, size=excess, replace=False)
        h[to_flip] = 0
    elif n_ones < target:
        # Add ones
        deficit = target - n_ones
        to_flip = rng.choice(zeros, size=deficit, replace=False)
        h[to_flip] = 1
    return h
