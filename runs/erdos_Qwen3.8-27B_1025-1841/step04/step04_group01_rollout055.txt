def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)

    global _G_N, _G_T, _G_TOP_IDX
    _G_N = n
    _G_T = T_val
    _G_TOP_IDX = top_idx
    obj = _obj_top if top_idx is not None else _obj_full
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def get_top_idx(h, frac=0.02, max_count=25):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    threshold = mc - frac * mc
    top = np.where(corr >= threshold)[0]
    if len(top) > max_count:
        tv = corr[top]
        so = np.argsort(-tv)
        top = top[so[:max_count]]
    elif len(top) == 0:
        top = np.array([int(np.argmax(corr))])
    return top

def binary_local_search(h_bin, n, rng, time_limit_s, max_passes=200):
    """Intensive binary local search with gradient guidance and SA."""
    h_cur = h_bin.copy()
    c5_cur = compute_c5(h_cur)
    best_local = c5_cur
    dx = 2.0 / n
    start = time.time()
    
    improved_streak = 0
    for p in range(max_passes):
        if time.time() - start > time_limit_s:
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
        mc = np.max(corr)
        # Active shifts
        thr = mc * 0.92
        active = np.where(corr >= thr)[0]
        
        # Compute cost for each position (contribution to active correlations)
        # Position i=1 contributes to corr[k] where k corresponds to shift involving i
        # Simplified: for each active shift, find contributing 1-positions
        cost = np.zeros(n)
        for ak in active:
            s = ak - (n - 1)
            if s >= 0:
                # corr[ak] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
                # 1-positions contributing: i+s where h[i+s]=1, h[i]=0
                seg_h = h_cur[s:]
                seg_1h = 1.0 - h_cur[:-s] if s > 0 else 1.0 - h_cur
                contrib = seg_h * seg_1h
                cost[s:] += contrib
            else:
                off = -s
                seg_h = h_cur[:-off] if off > 0 else h_cur
                seg_1h = 1.0 - h_cur[off:]
                contrib = seg_h * seg_1h
                cost[:n-off] += contrib
        
        # Greedy: try to flip highest-cost 1 to a 0, and lowest-cost 0 stays
        improved = False
        for _ in range(100):
            if time.time() - start > time_limit_s:
                break
            ones = np.where(h_cur == 1.0)[0]
            zeros = np.where(h_cur == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            # Pick highest cost 1
            a = ones[np.argmax(cost[ones])]
            # Pick from zeros: prefer those with low cost (less disruption)
            # Or random for diversity
            if rng.rand() < 0.7:
                b = zeros[np.argmin(cost[zeros])]
            else:
                b = rng.choice(zeros)
            
            h_cur[a] = 0.0
            h_cur[b] = 1.0
            new_c5 = compute_c5(h_cur)
            if new_c5 < c5_cur - 1e-15:
                c5_cur = new_c5
                improved = True
                if c5_cur < best_local:
                    best_local = c5_cur
                improved_streak = 0
            else:
                h_cur[a] = 1.0
                h_cur[b] = 0.0
        
        # Simulated annealing phase when stuck
        if not improved:
            improved_streak += 1
            temp = 0.003 / (1 + 0.1 * improved_streak)
            for _ in range(200):
                if time.time() - start > time_limit_s:
                    break
                ones = np.where(h_cur == 1.0)[0]
                zeros = np.where(h_cur == 0.0)[0]
                a = rng.choice(ones)
                b = rng.choice(zeros)
                h_cur[a] = 0.0
                h_cur[b] = 1.0
                new_c5 = compute_c5(h_cur)
                delta = new_c5 - c5_cur
                if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-8)):
                    c5_cur = new_c5
                    if c5_cur < best_local:
                        best_local = c5_cur
                        improved_streak = 0
                else:
                    h_cur[a] = 1.0
                    h_cur[b] = 0.0
    
    return h_cur, best_local

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    # Initialize
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Phase 1: SLSQP to get good continuous solution
    for T_val in [200.0, 500.0]:
        if remaining() < 80:
            break
        try:
            res_x = slsqp_optimize(best_h, n, T_val, None, maxiter=250)
            c5_res = compute_c5(res_x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res_x, n)
        except Exception:
            pass
    
    # Phase 2: SLSQP with top-shift focus
    if remaining() > 60:
        top = get_top_idx(best_h, 0.02, 20)
        try:
            res_x = slsqp_optimize(best_h, n, 400.0, top, maxiter=200)
            c5_res = compute_c5(res_x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res_x, n)
        except Exception:
            pass
    
    # Phase 3: Binary local search (main workhorse)
    h_bin = binary_project(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    time_for_ls = min(remaining() * 0.45, 400)
    h_bin, c5_bin = binary_local_search(h_bin, n, rng, time_for_ls)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Phase 4: SLSQP from best binary (continuous relaxation)
    if remaining() > 80:
        top = get_top_idx(best_h, 0.01, 20)
        try:
            res_x = slsqp_optimize(best_h, n, 600.0, top, maxiter=250)
            c5_res = compute_c5(res_x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res_x, n)
        except Exception:
            pass
        if remaining() > 40:
            top = get_top_idx(best_h, 0.01, 20)
            try:
                res_x = slsqp_optimize(best_h, n, 800.0, top, maxiter=200)
                c5_res = compute_c5(res_x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res_x, n)
            except Exception:
                pass
    
    # Phase 5: Binary local search again from improved continuous
    h_bin = binary_project(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    time_for_ls2 = min(remaining() * 0.3, 200)
    if time_for_ls2 > 30:
        h_bin, c5_bin = binary_local_search(h_bin, n, rng, time_for_ls2)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
    
    # Phase 6: Multi-start binary search
    if remaining() > 60:
        for restart in range(3):
            if remaining() < 30:
                break
            # Perturb the best binary
            h_pert = h_bin.copy()
            n_flip = range(5, 15)
            flips = rng.randint(n_flip[0], n_flip[1] + 1)
            ones = np.where(h_pert == 1.0)[0]
            zeros = np.where(h_pert == 0.0)[0]
            for _ in range(flips):
                a = rng.choice(ones)
                b = rng.choice(zeros)
                h_pert[a] = 0.0
                h_pert[b] = 1.0
                ones = np.where(h_pert == 1.0)[0]
                zeros = np.where(h_pert == 0.0)[0]
            
            h_pert, c5_pert = binary_local_search(h_pert, n, rng, min(remaining() * 0.2, 30))
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
    
    # Phase 7: Final SLSQP polish
    if remaining() > 50:
        top = get_top_idx(best_h, 0.005, 15)
        try:
            res_x = slsqp_optimize(best_h, n, 1000.0, top, maxiter=300, ftol=1e-15)
            c5_res = compute_c5(res_x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res_x, n)
        except Exception:
            pass
    
    # Phase 8: Try half resolution
    if remaining() > 120:
        n_half = n // 2
        h_half = np.zeros(n_half)
        for i in range(n_half):
            h_half[i] = (best_h[2*i] + best_h[2*i+1]) / 2.0
        h_half = make_feasible(h_half, n_half)
        c5_half = compute_c5(h_half)
        if c5_half < best_c5:
            try:
                res_x = slsqp_optimize(h_half, n_half, 300.0, None, maxiter=200)
                c5_res = compute_c5(res_x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res_x, n_half)
                    n = n_half
            except Exception:
                best_c5 = c5_half
                best_h = h_half
                n = n_half
    
    # Phase 9: Try double resolution
    if remaining() > 200 and n <= 960:
        n2 = n * 2
        h2 = np.interp(np.linspace(0.0, 2.0, n2), np.linspace(0.0, 2.0, n), best_h)
        h2 = make_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5 + 0.001:
            try:
                res_x = slsqp_optimize(h2, n2, 300.0, None, maxiter=200)
                c5_res = compute_c5(res_x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res_x, n2)
            except Exception:
                pass
            # Quick binary search at higher res
            if remaining() > 60 and len(best_h) == n2:
                h_bin2 = binary_project(best_h, n2)
                c5_b2 = compute_c5(h_bin2)
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_bin2
                h_bin2, c5_b2 = binary_local_search(h_bin2, n2, rng, min(remaining() * 0.25, 60))
                if c5_b2 < best_c5:
                    best_c5 = c5_b2
                    best_h = h_bin2
    
    # Phase 10: Ultra-fine final adjustments
    if remaining() > 20:
        n_pts = len(best_h)
        # Try snapping near-0 or near-1 values exactly
        for j in range(n_pts):
            if remaining() < 5:
                break
            for val in [0.0, 1.0]:
                if abs(best_h[j] - val) < 0.1 and abs(best_h[j] - val) > 1e-6:
                    trial = best_h.copy()
                    trial[j] = val
                    trial = make_feasible(trial, n_pts)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_h = trial
                        break
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
