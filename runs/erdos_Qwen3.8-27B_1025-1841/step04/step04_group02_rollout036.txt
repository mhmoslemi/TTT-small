```python
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(weights)):
        wi = weights[i]
        if wi < 1e-12:
            continue
        k = i - (n - 1)
        if k > 0:
            # (1-h[p-k]) valid for p >= k: positions k to n-1
            grad[k:] += wi * one_minus_h[:n-k]
            # -h[p+k] valid for p+k < n: positions 0 to n-k-1
            grad[:n-k] -= wi * h[k:]
        elif k < 0:
```

```python
def project_feasible(x, n):
    """Project x onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(x - mid, 0, 1)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo+hi)/2, 0, 1)
    # Fix any remaining error
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.any():
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0, 1)
    return h
```

1. Start from initial_h_values
2. Run subgradient method with decreasing T (from low to high, to gradually focus on critical lags)
3. Run SLSQP with the critical lags identified
4. Iterate between subgradient and SLSQP
5. Also try binary SA for structure discovery
6. Try different resolutions

Let me also think about what T values to use:
- Small T (e.g., 5-10): weights are spread out, gradient is a smooth average of all lags
- Large T (e.g., 100-1000): weights concentrate on the top few lags, gradient focuses on critical ones

A good strategy:
- Start with T=10 (smooth), run subgradient for a while
- Increase to T=50, run more
- Increase to T=200, run more
- Then switch to SLSQP with critical lags

Let me also think about the learning rate. For subgradient methods, a decreasing learning rate like lr = c / sqrt(it+1) or lr = c / (it+1) is standard. The constant c needs to be tuned.

Given that the correlation values are around n/4 = 240 (for n=960) and we're trying to reduce them by maybe 1-2 units, the "distance" to the optimum in h-space is small. So a small learning rate like 0.01-0.1 might work.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(x, n):
    target = n / 2.0
    lo, hi = x.min() - 2.0, x.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, n, weights):
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    for i in range(len(weights)):
        wi = weights[i]
        if wi < 1e-12:
            continue
        k = i - (n - 1)
        if k > 0:
            grad[k:] += wi * one_minus_h[:n-k]
            grad[:n-k] -= wi * h[k:]
        elif k < 0:
            m = -k
            grad[:n-m] += wi * one_minus_h[m:]
            grad[m:] -= wi * h[:n-m]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad

def subgradient_optimize(h, n, n_iter, T, lr0, start_time, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for it in range(n_iter):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        weights = np.exp(T * (corr - max_corr))
        weights /= np.sum(weights)
        
        grad = compute_grad(h, n, weights)
        
        # Decreasing learning rate
        lr = lr0 / (1.0 + 0.01 * it)
        
        h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def slsqp_obj(x, n, T, critical_lags=None):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if critical_lags is not None:
        sub = corr[critical_lags]
        mc = np.max(sub)
        shifted = sub - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    else:
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_solve(h, n, T, maxiter, ftol, critical_lags=None):
    bounds = [(0.0, 1.0)] * n
```

```python
def compute_grad(h, n, weights):
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    # Only process lags with significant weight
    significant = np.where(weights > 1e-8)[0]
    for i in significant:
        wi = weights[i]
        k = i - (n - 1)
        if k > 0:
            grad[k:] += wi * one_minus_h[:n-k]
            grad[:n-k] -= wi * h[k:]
        elif k < 0:
            m = -k
            grad[:n-m] += wi * one_minus_h[m:]
            grad[m:] -= wi * h[:n-m]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad
```

```python
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(x, n):
    target = n / 2.0
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(x - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, n, weights):
    grad = np.zeros(n)
    one_minus_h = 1.0 - h
    significant = np.where(weights > 1e-10)[0]
    for i in significant:
        wi = weights[i]
        k = i - (n - 1)
        if k > 0:
            grad[k:] += wi * one_minus_h[:n - k]
            grad[:n - k] -= wi * h[k:]
        elif k < 0:
            m = -k
            grad[:n - m] += wi * one_minus_h[m:]
            grad[m:] -= wi * h[:n - m]
        else:
            grad += wi * (1.0 - 2.0 * h)
    return grad

def subgradient_optimize(h, n, n_iter, T, lr0, start_time, budget_s, use_momentum=True):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    v = np.zeros(n)
    mom = 0.85 if use_momentum else 0.0
    for it in range(n_iter):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_corr = np.max(corr)
        weights = np.exp(T * (corr - max_corr))
        sw = np.sum(weights)
        if sw < 1e-300:
            break
        weights /= sw
        grad = compute_grad(h, n, weights)
        lr = lr0 / (1.0 + 0.003 * it)
        if use_momentum:
            v = mom * v - lr * grad
            h_new = h + v
        else:
            h_new = h - lr * grad
        h_new = project_feasible(h_new, n)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _con_sum(x):
    return np.sum(x) - len(x) / 2.0

def _con_sum_jac(x):
    return np.ones(len(x))

def slsqp_solve(h, n, T, maxiter, ftol, critical_lags=None):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        if critical_lags is not None:
            sub = corr[critical_lags]
            mc = np.max(sub)
            shifted = sub - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        else:
            mc = np.max(corr)
            shifted = corr - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_sum_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def binary_sa(h, n, max_iters, T0, T1, rng):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)
    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Subgradient with increasing T (smooth to sharp)
    for T in [5.0, 15.0, 40.0, 100.0]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        n_iter = min(int(remaining * 0.12), 2500)
        lr = 0.04 if T < 30 else 0.02
        h_sub, c5_sub = subgradient_optimize(best_h, n, n_iter, T, lr, start_time, budget_s)
        if c5_sub < best_c5 - 1e-9:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 2: SLSQP on top critical lags
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(corr)[-15:]
        try:
            x_res = slsqp_solve(best_h, n, 50.0, 400, 1e-14, sorted(top_idx))
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n)
        except Exception:
            pass

    # Phase 3: Sharp subgradient (focus on max lags)
    for T in [200.0, 500.0, 1500.0]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 25:
            break
        n_iter = min(int(remaining * 0.1), 2000)
        h_sub, c5_sub = subgradient_optimize(best_h, n, n_iter, T, 0.01, start_time, budget_s)
        if c5_sub < best_c5 - 1e-9:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 4: SLSQP full (all lags, moderate T)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        try:
            x_res = slsqp_solve(best_h, n, 80.0, 500, 1e-15, None)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n)
        except Exception:
            pass

    # Phase 5: Binary SA from current best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n_sa = min(int(remaining * 0.25), 120000)
        b, c5_b = binary_sa(best_h, n, n_sa, 0.006, 0.00001, rng)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase 6: SLSQP polish after binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        try:
            x_res = slsqp_solve(best_h, n, 200.0, 400, 1e-15, None)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n)
        except Exception:
            pass

    # Phase 7: One more round of subgradient + SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        h_sub, c5_sub = subgradient_optimize(best_h, n, 1000, 500.0, 0.008, start_time, budget_s)
        if c5_sub < best_c5 - 1e-9:
            best_c5 = c5_sub
            best_h = h_sub.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            try:
                x_res = slsqp_solve(best_h, n, 400.0, 300, 1e-15, None)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n)
            except Exception:
                pass

    # Phase 8: Higher resolution attempt
    remaining = budget_s - (time.time() - start_time)
    if remaining > 400 and n < 1440:
        n2 = min(n + 480, 1440)
        h_up = upsample_h(best_h, n2)
        for T in [30.0, 100.0, 400.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break
            n_iter = min(int(remaining * 0.08), 1500)
            h_sub, c5_sub = subgradient_optimize(h_up, n2, n_iter, T, 0.02, start_time, budget_s)
            if c5_sub < best_c5:
                best_c5 = c5_sub
                best_h = h_sub.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100:
            try:
                x_res = slsqp_solve(best_h, n2, 100.0, 300, 1e-14, None)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n2)
            except Exception:
                pass
        remaining = budget_s - (time.time() - start_time)
        if remaining > 200:
            n_sa2 = min(int(remaining * 0.3), 80000)
            b2, c5_b2 = binary_sa(best_h, n2, n_sa2, 0.004, 0.00001, rng)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = b2.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 80:
                try:
                    x_res = slsqp_solve(best_h, n2, 200.0, 300, 1e-15, None)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(x_res, n2)
                except Exception:
                    pass

    # Phase 9: Final polish at current resolution
    remaining = budget_s - (time.time() - start_time)
    n_pts = len(best_h)
    if remaining > 40:
        try:
            x_res = slsqp_solve(best_h, n_pts, 800.0, 200, 1e-16, None)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
