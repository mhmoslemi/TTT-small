    return np.correlate(h, 1.0 - h, mode='full')

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.array(h, dtype=float)
    # Threshold
    b = (h >= 0.5).astype(float)
    cnt = int(np.sum(b))
    target = n // 2
    if cnt > target:
        # Remove excess ones: flip the ones with lowest h value
        ones_idx = np.where(b == 1)[0]
        order = np.argsort(h[ones_idx])  # lowest h values first
        for i in range(cnt - target):
            b[ones_idx[order[i]]] = 0
    elif cnt < target:

    # Changing h[j] from 0 to 1: affects all shifts s where j+s is valid
    #   - increases corr by (1-h[j+s]) for each valid s
    # Also the "other direction" shifts

    new_corr = corr.copy()
    if h[i] == 1 and h[j] == 0:

        # h[i] appears as h[m+s] when m=i-s: contribution was h[i-s]*(1-1)=0, now h[i-s]*(1-0)=h[i-s]
        # h[j] appears as h[m] when m=j: contribution was 0, now 1*(1-h[j+s])
        # h[j] appears as h[m+s] when m=j-s: contribution was h[j-s]*(1-0)=h[j-s], now h[j-s]*(1-1)=0

            if 0 <= i+s < n:
                delta -= (1 - h[i+s])  # was 1*(1-h[i+s]), now 0
            # h[i] as h[m+s] with m=i-s, need i-s in [0,n)
            if 0 <= i-s < n:
                delta += h[i-s]  # was 0, now h[i-s]*1
            # h[j] as h[m] with m=j, need j+s in [0,n)
            if 0 <= j+s < n:
                delta += (1 - h[j+s])  # was 0, now 1*(1-h[j+s])
            # h[j] as h[m+s] with m=j-s, need j-s in [0,n)
            if 0 <= j-s < n:
                delta -= h[j-s]  # was h[j-s]*1, now 0
            new_corr[idx] += delta
    elif h[i] == 0 and h[j] == 1:

        for s in range(-(n-1), n):
            idx = s + n - 1
            delta = 0
            if 0 <= i+s < n:
                delta += (1 - h[i+s])
            if 0 <= i-s < n:
                delta -= h[i-s]
            if 0 <= j+s < n:
                delta -= (1 - h[j+s])
            if 0 <= j-s < n:
                delta += h[j-s]
            new_corr[idx] += delta
    return np.max(new_corr) - np.max(corr), new_corr

Key improvement idea: **Adaptive candidate generation**. Instead of purely random swaps, bias the candidates toward swaps that are most likely to reduce the max overlap.

For the current max at shift s*, the overlap is:
O(s*) = sum_i h[i] * (1-h[i+s*]) (for valid i)

To reduce O(s*), I want to:
- Set h[i]=0 for some i where h[i]=1 and h[i+s*]=0 (removes a contributing term)
- Set h[i+s*]=1 for some i where h[i]=1 and h[i+s*]=0 (removes a contributing term)
- But do so in a way that doesn't increase other shifts too much

So a good candidate swap is:
- i: a 1 that has a 0 at position i+s* (contributes to O(s*))
- j: a 0 that, when set to 1, doesn't create new contributions to other high shifts
