_g_n = 0
_g_T = 1.0

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    T = _g_T
    if T < 1e-6:
        return mc * (2.0 / _g_n)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / _g_n)

def _sum_con(x):
    return np.sum(x) - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _g_n, _g_T
    _g_n = n
    _g_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
    res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    T = _g_T
    if T < 1e-6:
        return mc * (2.0 / _g_n)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / _g_n)

def _sum_con(x):
    return np.sum(x) - len(x) / 2.0

def _sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _g_n, _g_T
    _g_n = n
    _g_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
    res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def bump_perturb(h, n, rng):
    h_new = h.copy()
    center = rng.uniform(0, n)
    width = rng.uniform(n / 15, n / 4)
    amplitude = rng.uniform(-0.4, 0.4)
    x = np.arange(n)
    bump = amplitude * np.exp(-0.5 * ((x - center) / width) ** 2)
    h_new += bump
    return make_feasible(h_new, n)

def fourier_perturb(h, n, rng, n_modes=3, amp=0.05):
    h_new = h.copy()
    x = np.arange(n)
    for _ in range(n_modes):
        freq = rng.randint(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += amp * rng.uniform(0.5, 1.5) * np.sin(2 * np.pi * freq * x / n + phase)
    return make_feasible(h_new, n)

def segment_flip_perturb(h, n, rng):
    h_new = h.copy()
    L = rng.randint(n // 10, n // 3)
    s = rng.randint(0, n - L)
    h_new[s:s + L] = 1.0 - h_new[s:s + L]
    return make_feasible(h_new, n)

def pairwise_swap_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    curr = h.copy()
    curr_c5 = compute_c5(curr)
    best_c5 = curr_c5
    best_h = curr.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        curr[i], curr[j] = curr[j], curr[i]
        new_c5 = compute_c5(curr)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr.copy()
        else:
            curr[i], curr[j] = curr[j], curr[i]
    return best_h, best_c5

def local_swap_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    curr = h.copy()
    curr_c5 = compute_c5(curr)
    best_c5 = curr_c5
    best_h = curr.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(n)
        dist = rng.randint(1, min(n, 20))
        j = (i + dist) % n
        if i == j:
            continue
        curr[i], curr[j] = curr[j], curr[i]
        new_c5 = compute_c5(curr)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = curr.copy()
        else:
            curr[i], curr[j] = curr[j], curr[i]
    return best_h, best_c5

def coord_descent_fast(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            candidates = [0.0, 1.0, max(0.0, old - 0.05), min(1.0, old + 0.05)]
            for cand in candidates:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def slsqp_refine(h, n, start_time, budget_s, temps=(1000.0, 5000.0, 20000.0)):
    best = h.copy()
    best_c = compute_c5(best)
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 50:
            break
        try:
            x_res = slsqp_run(best, n, T, 200, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c:
                best_c = c5_res
                best = make_feasible(x_res, n)
        except Exception:
            pass
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Quick SLSQP sweep
    best_h, best_c5 = slsqp_refine(best_h, n0, start_time, budget_s, (50.0, 500.0, 5000.0))

    # Phase 2: Long global pairwise swap SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        tl = min(remaining - 80, 350)
        b, c5_b = pairwise_swap_sa(best_h, n0, 800000, 0.002, 0.000001, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
        best_h, best_c5 = slsqp_refine(best_h, n0, start_time, budget_s, (2000.0, 10000.0))

    # Phase 3: Perturbation rounds
    perts = [bump_perturb, fourier_perturb, segment_flip_perturb]
    for pf in perts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 150:
            break
        for _ in range(2):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 80:
                break
            try:
                hp = pf(best_h, n0, rng)
                hr, cr = slsqp_refine(hp, n0, start_time, budget_s, (1000.0, 5000.0))
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 4: Local (nearby) swap SA for fine-tuning
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 50, 200)
        b, c5_b = local_swap_sa(best_h, n0, 500000, 0.0005, 0.0000005, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
        best_h, best_c5 = slsqp_refine(best_h, n0, start_time, budget_s, (5000.0, 30000.0))

    # Phase 5: Another global swap SA round
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 50, 180)
        b, c5_b = pairwise_swap_sa(best_h, n0, 400000, 0.001, 0.0000005, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
        best_h, best_c5 = slsqp_refine(best_h, n0, start_time, budget_s, (2000.0, 20000.0))

    # Phase 6: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        hu, cu = slsqp_refine(h_up, n1, start_time, budget_s, (1000.0, 5000.0, 20000.0))
        if cu < best_c5:
            best_c5 = cu
            best_h = hu.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100:
            b, c5_b = local_swap_sa(best_h, len(best_h), 200000, 0.0003, 0.0000003, rng, min(remaining - 40, 100), start_time)
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = b.copy()

    # Phase 7: Final coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n_pts, 3, start_time, budget_s)

    # Phase 8: Final high-T SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 50000.0, 250, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 9: Final short local swap
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        b, c5_b = local_swap_sa(best_h, len(best_h), 100000, 0.0001, 0.0000001, rng, min(remaining - 5, 15), start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
