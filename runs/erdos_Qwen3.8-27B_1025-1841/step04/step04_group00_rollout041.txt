```python
def solve_lp(h_current, active_shifts, n, solver='CLARABEL'):
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for s_idx in active_shifts:
        s = s_idx - (n - 1)  # actual shift
        w = cp.Variable(n)  # w[j] represents h[j] * (1-h[j+s])
        
        # McCormick envelopes for w[j] = h[j] * (1-h[j+s])
        # Let a = h[j], b = 1-h[j+s], w = a*b
        # w <= a, w <= b, w >= a+b-1, w >= 0
        for j in range(n):
            valid_j = 0 <= j < n
            valid_js = 0 <= j + s < n
            if valid_j and valid_js:
                constraints.append(w[j] <= h_var[j])
                constraints.append(w[j] <= 1 - h_var[j + s])
                constraints.append(w[j] >= h_var[j] - h_var[j + s])
                constraints.append(w[j] >= 0)
            else:
                constraints.append(w[j] == 0)
        
        # Correlation constraint
        constraints.append(cp.sum(w) * (2.0/n) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return h_current, np.inf
    
    if h_var.value is None or np.any(np.isnan(h_var.value)):
        return h_current, np.inf
    
    h_new = np.clip(h_var.value, 0, 1)
    h_new = make_feasible(h_new, n)
    c5_val = compute_c5(h_new)
    return h_new, c5_val
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_overlaps(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    if abs(diff) > 1e-13:

        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    if s >= 0:
        return [(j, j + s) for j in range(n - s)]
    else:
        return [(j, j + s) for j in range(-s, n)]

def solve_lp_mccormick(h0, active_shifts, n, solver=None):
    """Solve LP with McCormick envelopes for the given active shifts."""
    dx = 2.0 / n
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0

    for s in active_shifts:

        m_pairs = len(pairs)

        for idx, (j, js) in enumerate(pairs):

                w[idx] <= h_var[j],
                w[idx] <= 1.0 - h_var[js],
                w[idx] >= h_var[j] - h_var[js],
                w[idx] >= 0

            break

    h_new = make_feasible(h_new, n)
    c5 = compute_c5(h_new)
    return h_new, c5
```

```python
def binary_projection(h, n):
    """Project to binary (0/1) while maintaining sum = n/2."""
    h_bin = (h >= 0.5).astype(float)
    # Adjust to get exactly n/2 ones
    num_ones = int(np.sum(h_bin))
    if num_ones > n // 2:
        # Need to flip some 1s to 0s
        excess = num_ones - n // 2
        # Flip the ones with lowest h values
        one_indices = np.where(h_bin == 1.0)[0]
        one_indices = one_indices[np.argsort(h[one_indices])]
        h_bin[one_indices[:excess]] = 0.0
    elif num_ones < n // 2:
        deficit = n // 2 - num_ones
        zero_indices = np.where(h_bin == 0.0)[0]
        zero_indices = zero_indices[np.argsort(h[zero_indices])[::-1]]
        h_bin[zero_indices[:deficit]] = 1.0
    return h_bin
```

```python
def compute_c5(h):

    return float(np.max(corr) * dx)

def compute_overlaps(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def binary_projection(h, n):
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    num_ones = int(np.sum(h_bin))
    if num_ones > target:
        excess = num_ones - target
        one_idx = np.where(h_bin == 1.0)[0]
        one_idx = one_idx[np.argsort(h[one_idx])]
        h_bin[one_idx[:excess]] = 0.0
    elif num_ones < target:
        deficit = target - num_ones
        zero_idx = np.where(h_bin == 0.0)[0]
        zero_idx = zero_idx[np.argsort(h[zero_idx])[::-1]]
        h_bin[zero_idx[:deficit]] = 1.0

        n_r = len(best_h)
        scale = 0.02 if restart < 4 else 0.04
        h_try = make_feasible(best_h + rng.randn(n_r) * scale, n_r)
        h_r, c5_r = pgd_optimize(h_try, n_r, min(25, remaining() - 20), start_time, top_m=5)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 7: Final PGD
    if remaining() > 30:
        h_fin, c5_fin = pgd_optimize(best_h, len(best_h), min(25, remaining() - 5), start_time, top_m=3)
        if c5_fin < best_c5:
            best_h, best_c5 = h_fin, c5_fin

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
