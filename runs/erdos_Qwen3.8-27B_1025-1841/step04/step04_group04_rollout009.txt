    v = h.copy()  # momentum position

    corr = np.correlate(h, hm, mode='full')

    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _remaining(budget_s, start_time):
    return budget_s - (time.time() - start_time)

def projected_subgradient(h, time_budget, topk=3):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = compute_c5(h)
    y = h.copy()
    v = h.copy()
    t = 1.0
    start = time.time()
    for it in range(20000):
        if time.time() - start > time_budget:
            break
        K_eff = max(1, topk - it // 3000)
        c5, grad = compute_c5_grad(y, K=K_eff)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-14:
            break
        alpha = 0.4 / gnorm
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        z = y + beta * (y - v)
        y_new = z - alpha * grad
        y_new = np.clip(y_new, 0.0, 1.0)
        diff = target - np.sum(y_new)
        if abs(diff) > 1e-13:
            if diff > 0:
                mask = y_new < 1.0
            else:
                mask = y_new > 0.0
            if np.any(mask):
                y_new[mask] += diff / np.sum(mask)
                y_new = np.clip(y_new, 0.0, 1.0)
        c_new = compute_c5(y_new)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = y_new.copy()
        v = y
        y = y_new
        t = t_new
    return best_h, best_c

def coordinate_descent(h, time_budget, n_rounds=40):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    deltas = [-0.15, -0.1, -0.05, -0.02, -0.01, 0.0, 0.01, 0.02, 0.05, 0.1, 0.15]
    for rnd in range(n_rounds):
        if time.time() - start > time_budget:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_budget:
                break
            old = best_h[j]
            best_trial = old
            bc = best_c
            cands = [0.0, 1.0, 0.5]
            for d in deltas:
                cands.append(max(0.0, min(1.0, old + d)))
            for cand in cands:
                if abs(cand - old) < 1e-15:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-14:
                    bc = c
                    best_trial = cand
            if best_trial != old:
                best_h = best_h.copy()
                best_h[j] = best_trial
                best_h = make_feasible(best_h, n)
                best_c = bc
                improved = True
        if not improved:
            break
    return best_h, best_c

def binary_local_search(h, time_budget, rng):
    n = len(h)
    best_h = h.copy()
    best_c = compute_c5(h)
    start = time.time()
    bh = np.where(best_h > 0.5, 1.0, 0.0)
    bh = make_feasible(bh, n)
    bc = compute_c5(bh)
    if bc < best_c:
        best_c = bc
        best_h = bh
    ones = list(np.where(best_h > 0.5)[0])
    zeros = list(np.where(best_h < 0.5)[0])
    for _ in range(200):
        if time.time() - start > time_budget:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        ntries = min(300, 10 * (len(ones) + len(zeros)))
        for _t in range(ntries):
            if time.time() - start > time_budget:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            trial = best_h.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best_c = c
                best_h = trial
                improved = True
                ones = list(np.where(best_h > 0.5)[0])
                zeros = list(np.where(best_h < 0.5)[0])
                if len(ones) == 0 or len(zeros) == 0:
                    break
        if not improved:
            break
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_N, _OPT_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP moderate T
    if _remaining(budget_s, start_time) > 100:
        _OPT_N = n_init
        _OPT_T = 40.0
        bounds = [(0.0, 1.0)] * n_init
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 250, 'ftol': 1e-12})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, n_init)
        except Exception:
            pass

    # Phase 2: Projected subgradient on true max
    if _remaining(budget_s, start_time) > 200:
        sg_budget = min(160.0, _remaining(budget_s, start_time) - 40.0)
        h_sg, c_sg = projected_subgradient(best_h, sg_budget, topk=3)
        if c_sg < best_c5:
            best_c5 = c_sg
            best_h = h_sg

    # Phase 3: SLSQP with increasing T (annealing)
    for T in [80, 200, 400, 800]:
        if _remaining(budget_s, start_time) < 50:
            break
        _OPT_N = n_init
        _OPT_T = float(T)
        bounds = [(0.0, 1.0)] * n_init
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 150, 'ftol': 1e-13})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, n_init)
        except Exception:
            pass

    # Phase 4: Binary local search
    if _remaining(budget_s, start_time) > 60:
        bls_budget = min(50.0, _remaining(budget_s, start_time) - 20.0)
        best_h, best_c5 = binary_local_search(best_h, bls_budget, rng)

    # Phase 5: Coordinate descent
    if _remaining(budget_s, start_time) > 80:
        cd_budget = min(120.0, _remaining(budget_s, start_time) - 20.0)
        best_h, best_c5 = coordinate_descent(best_h, cd_budget, n_rounds=30)

    # Phase 6: Final SLSQP polish
    if _remaining(budget_s, start_time) > 50:
        _OPT_N = n_init
        _OPT_T = 1000.0
        bounds = [(0.0, 1.0)] * n_init
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-14})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, n_init)
        except Exception:
            pass

    # Phase 7: Short coordinate descent
    if _remaining(budget_s, start_time) > 30:
        cd2_budget = min(30.0, _remaining(budget_s, start_time) - 10.0)
        best_h, best_c5 = coordinate_descent(best_h, cd2_budget, n_rounds=15)

    # Phase 8: Try n=480 (lower resolution might find simpler optimum)
    if _remaining(budget_s, start_time) > 50:
        h_480 = upsample_h(best_h, 480)
        _OPT_N = 480
        _OPT_T = 500.0
        bounds = [(0.0, 1.0)] * 480
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h_480, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 100, 'ftol': 1e-13})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, 480)
        except Exception:
            pass

    # Phase 9: Try n=1920 (higher resolution)
    if _remaining(budget_s, start_time) > 60:
        h_1920 = upsample_h(best_h, 1920)
        _OPT_N = 1920
        _OPT_T = 300.0
        bounds = [(0.0, 1.0)] * 1920
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h_1920, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 80, 'ftol': 1e-13})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, 1920)
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
