import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def active_gradient(h, active_shifts, n):
    """Gradient of sum of overlaps at given shifts (integer shifts, not full-correlate indices)."""
    grad = np.zeros(n)
    for k in active_shifts:
        if k == 0:
            grad += 2.0 * h - 1.0
        elif k > 0:
            grad[k:] += (1.0 - h[:-k])
            grad[:-k] -= h[k:]
        else:
            s = -k
            grad[:-s] += (1.0 - h[s:])
            grad[s:] -= h[:-s]
    return grad

def get_active_shifts(h, n, top_frac=0.98, top_count=30):
    """Return integer shifts corresponding to near-max correlation values."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    threshold = mx * top_frac
    idx = np.where(corr >= threshold)[0]
    if len(idx) > top_count:
        order = np.argsort(corr[idx])[::-1][:top_count]
        idx = idx[order]
    shifts = idx - (n - 1)
    return shifts

def targeted_binary_sa(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    """Binary SA with moves targeted at worst lags."""
    b = (h >= 0.5).astype(float)
    target_count = n // 2
    while np.sum(b) > target_count:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[rng.randint(len(ones))]
        b[idx] = 0
    while np.sum(b) < target_count:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[rng.argmax(h[zeros])]
        b[idx] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 25:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        corr = np.correlate(b, 1.0 - b, mode='full')
        mx = np.max(corr)
        thresh = mx * 0.97
        active_idx = np.where(corr >= thresh)[0]
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Pick a random active lag and do a targeted swap
        if len(active_idx) > 0:
            k_idx = active_idx[rng.randint(len(active_idx))]
            shift = k_idx - (n - 1)
        else:
            shift = rng.randint(1, n - 1)
        # Find helpful positions for this shift
        if shift > 0:
            hi = np.where((b[:-shift] == 1) & (b[shift:] == 0))[0]
            hj = np.where((b[:-shift] == 0) & (b[shift:] == 1))[0] + shift
        elif shift < 0:
            s = -shift
            hi = np.where((b[s:] == 1) & (b[:-s] == 0))[0] + s
            hj = np.where((b[s:] == 0) & (b[:-s] == 1))[0]
        else:
            hi = ones
            hj = zeros
        if len(hi) > 0 and len(hj) > 0:
            i = hi[rng.randint(len(hi))]
            j = hj[rng.randint(len(hj))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def block_targeted_sa(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    """Block moves with lag-targeting."""
    b = (h >= 0.5).astype(float)
    target_count = n // 2
    while np.sum(b) > target_count:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target_count:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 25:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        blk = min(rng.randint(2, 12), n // 4)
        start = rng.randint(0, n - blk)
        seg = b[start:start + blk]
        ones_seg = np.where(seg == 1)[0]
        zeros_seg = np.where(seg == 0)[0]
        if len(ones_seg) == 0 or len(zeros_seg) == 0:
            continue
        swap_cnt = min(rng.randint(1, len(ones_seg) + 1), len(zeros_seg))
        chosen_o = rng.choice(ones_seg, size=swap_cnt, replace=False)
        chosen_z = rng.choice(zeros_seg, size=swap_cnt, replace=False)
        b[start + chosen_o] = 0
        b[start + chosen_z] = 1
        new_c5 = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + chosen_o] = 1
            b[start + chosen_z] = 0
    return best_b, best_c5

def active_gradient_descent(h, n, steps, lr, start_time, budget_s):
    """Continuous gradient descent on the active set of lags."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for step in range(steps):
        if time.time() - start_time > budget_s - 20:
            break
        shifts = get_active_shifts(h, n, top_frac=0.95, top_count=40)
        grad = active_gradient(h, shifts, n)
        grad -= np.mean(grad)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        if c_new < compute_c5(h) + 1e-15:
            h = h_new
            if lr > 0.001:
                lr *= 0.95
        else:
            lr *= 0.7
            if lr < 1e-6:
                break
    return best_h, best_c5

def _slsqp_obj(x):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    T = _SL_T
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / _SL_N)

def _slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def _slsqp_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
    res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        if time.time() - start_time > budget_s - 15:
            break
        for j in range(n):
            if time.time() - start_time > budget_s - 10:
                return h, best_c5
            old = h[j]
            best_val = old
            best_c = best_c5
            for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5
