import numpy as np
from scipy.optimize import minimize
import time

_S_N = 0
_S_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = int(np.sum(b))
    while s > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
        s -= 1
    while s < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
        s += 1
    return b

def _sobj(x):
    n = _S_N
    T = _S_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _scon(x):
    return np.sum(x) - _S_N / 2.0

def _scon_jac(x):
    return np.ones(_S_N)

def slsqp(h, n, T, maxiter, ftol):
    global _S_N, _S_T
    _S_N = n
    _S_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _scon, 'jac': _scon_jac}]
    res = minimize(_sobj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def binary_sa(h, n, max_iters, T0, T1, rng):
    b = to_binary(h, n)
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    best_frac = 0.0
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        i = rng.randint(n)
        while b[i] < 0.5:
            i = rng.randint(n)
        j = rng.randint(n)
        while b[j] > 0.5:
            j = rng.randint(n)
        b[i] = 0.0
        b[j] = 1.0
        new = compute_c5(b)
        if new <= curr or (T > 1e-14 and rng.random() < np.exp(-(new - curr) / T)):
            curr = new
            if new < best:
                best = new
                best_b = b.copy()
                best_frac = frac
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best

def binary_sa_block(h, n, max_iters, T0, T1, rng, max_blk=4):
    b = to_binary(h, n)
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        blk = rng.randint(1, max_blk + 1)
        s1 = rng.randint(0, n - blk)
        s2 = rng.randint(0, n - blk)
        if abs(s1 - s2) < blk:
            continue
        b[s1:s1+blk] = 1.0 - b[s1:s1+blk]
        b[s2:s2+blk] = 1.0 - b[s2:s2+blk]
        new = compute_c5(b)
        if new <= curr or (T > 1e-14 and rng.random() < np.exp(-(new - curr) / T)):
            curr = new
            if new < best:
                best = new
                best_b = b.copy()
        else:
            b[s1:s1+blk] = 1.0 - b[s1:s1+blk]
            b[s2:s2+blk] = 1.0 - b[s2:s2+blk]
    return best_b, best

def coord_desc(h, n, max_rounds, start_time, budget_s):

                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    return t
