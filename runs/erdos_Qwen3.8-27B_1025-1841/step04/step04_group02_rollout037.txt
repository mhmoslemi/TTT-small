            continue
        chosen_ones = rng.choice(ones_in_seg, size=swap_count, replace=False)
        chosen_zeros = rng.choice(zeros_in_seg, size=swap_count, replace=False)
        b[start + chosen_ones] = 0
        b[start + chosen_zeros] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + chosen_ones] = 1
            b[start + chosen_zeros] = 0
    return best_b, best_c5

def exact_coord_descent(h, n, max_passes, start_time, budget_s):
    best_c5 = compute_c5(h)
    h = h.copy()
    no_improve_count = 0
    for p in range(max_passes):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 8:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        pass_improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break
            g = np.zeros(2 * n - 1)
            m_lo = max(0, n - 1 - j)
            m_hi = min(2 * n - 2, n - 1 + j)
            if m_lo <= m_hi:
                h_idx = j + np.arange(m_lo, m_hi + 1) - n + 1
                g[m_lo:m_hi + 1] += 1.0 - h[h_idx]
            m_lo2 = max(0, j - n + 1)
            m_hi2 = min(2 * n - 2, j + n - 1)
            if m_lo2 <= m_hi2:
                h_idx2 = j - np.arange(m_lo2, m_hi2 + 1) + n - 1
                g[m_lo2:m_hi2 + 1] -= h[h_idx2]
            delta_min = -h[j]
            delta_max = 1.0 - h[j]
            if delta_max - delta_min < 1e-14:
                continue
            f_curr = np.max(c)
            best_f = f_curr
            best_delta = 0.0
            f_lo = np.max(c + delta_min * g)
            if f_lo < best_f:
                best_f = f_lo
                best_delta = delta_min
            f_hi = np.max(c + delta_max * g)
            if f_hi < best_f:
                best_f = f_hi
                best_delta = delta_max
            ta, tb = delta_min, delta_max
            for _ in range(55):
                tm1 = ta + (tb - ta) / 3.0
                tm2 = tb - (tb - ta) / 3.0
                fm1 = np.max(c + tm1 * g)
                fm2 = np.max(c + tm2 * g)
                if fm1 < fm2:
                    tb = tm2
                else:
                    ta = tm1
            tmid = (ta + tb) / 2.0
            fmid = np.max(c + tmid * g)
            if fmid < best_f:
                best_f = fmid
                best_delta = tmid
            if abs(best_delta) > 1e-13:
                h[j] = np.clip(h[j] + best_delta, 0.0, 1.0)
                c += best_delta * g
                pass_improved = True
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            no_improve_count = 0
        else:
            no_improve_count += 1
        if no_improve_count >= 3 and not pass_improved:
            break
    return h, best_c5

def targeted_swap_sa(h, n, max_iters, rng, start_time, budget_s):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    c = np.correlate(b, 1.0 - b, mode='full')
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        active_k = int(np.argmax(c))
        shift = active_k - (n - 1)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        if shift >= 0:
            cand_ones = [i for i in ones if i + shift < n and b[i + shift] == 0]
            cand_zeros = [j for j in zeros if 0 <= j - shift and b[j - shift] == 1]
        else:
            s = -shift
            cand_ones = [i for i in ones if i - s >= 0 and b[i - s] == 0]
            cand_zeros = [j for j in zeros if j + s < n and b[j + s] == 1]
        if len(cand_ones) > 0 and len(cand_zeros) > 0:
            i = cand_ones[rng.randint(len(cand_ones))]
            j = cand_zeros[rng.randint(len(cand_zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 < curr_c5:
            curr_c5 = new_c5
            c = np.correlate(b, 1.0 - b, mode='full')
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def slsqp_run(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con_sum(x):
        return np.sum(x) - len(x) / 2.0
    def con_sum_jac(x):
        return np.ones(len(x))
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    n0 = len(h_init)
    h = make_feasible(h_init, n0)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: Exact coordinate descent (fast, many passes)
    h1, c5_1 = exact_coord_descent(h, n0, 400, start_time, budget_s)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: Targeted swap SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        b2, c5_2 = targeted_swap_sa(best_h, n0, 60000, rng, start_time, budget_s)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = b2.copy()

    # Phase 3: More coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        h3, c5_3 = exact_coord_descent(best_h, n0, 400, start_time, budget_s)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        b4, c5_4 = binary_sa_block(best_h, n0, 50000, 0.005, 0.00005, rng, start_time, budget_s)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = b4.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 50:
            h4, c5_4b = exact_coord_descent(best_h, n0, 300, start_time, budget_s)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4.copy()

    # Phase 5: SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        try:
            x_res = slsqp_run(best_h, n0, 200.0, 300, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass
        remaining = budget_s - (time.time() - start_time)
        if remaining > 30:
            h5, c5_5 = exact_coord_descent(best_h, n0, 200, start_time, budget_s)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()

    # Phase 6: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 350:
        n1 = min(n0 + 480, 1440)
        h_up = upsample_h(best_h, n1)
        h_up, c5_up = exact_coord_descent(h_up, n1, 200, start_time, budget_s)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 150:
            b_up, c5_up2 = targeted_swap_sa(best_h, n1, 40000, rng, start_time, budget_s)
            if c5_up2 < best_c5:
                best_c5 = c5_up2
                best_h = b_up.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 30:
                h_up2, c5_up3 = exact_coord_descent(best_h, n1, 200, start_time, budget_s)
                if c5_up3 < best_c5:
                    best_c5 = c5_up3
                    best_h = h_up2.copy()

    # Phase 7: Try lower resolution for better binary structure
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250 and n0 > 600:
        n2 = max(n0 // 2, 480)
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x2 = np.linspace(0.0, 2.0, n2)
        h_lo = np.interp(new_x2, old_x, best_h)
        h_lo = make_feasible(h_lo, n2)
        b_lo, c5_lo = binary_sa(h_lo, n2, 50000, 0.01, 0.0001, rng, start_time, budget_s)
        if c5_lo < best_c5:
            best_c5 = c5_lo
            best_h = b_lo.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 30:
            h_lo2, c5_lo2 = exact_coord_descent(best_h, len(best_h), 300, start_time, budget_s)
            if c5_lo2 < best_c5:
                best_c5 = c5_lo2
                best_h = h_lo2.copy()

    # Phase 8: Final targeted SA + coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        n_pts = len(best_h)
        b8, c5_8 = targeted_swap_sa(best_h, n_pts, 40000, rng, start_time, budget_s)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = b8.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 30:
            h8, c5_8b = exact_coord_descent(best_h, n_pts, 300, start_time, budget_s)
            if c5_8b < best_c5:
                best_c5 = c5_8b
                best_h = h8.copy()

    # Phase 9: Final SLSQP with high T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 500.0, 200, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 10: Final coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        h10, c5_10 = exact_coord_descent(best_h, len(best_h), 500, start_time, budget_s)
        if c5_10 < best_c5:
            best_c5 = c5_10
            best_h = h10.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
