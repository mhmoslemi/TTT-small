import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_binary(h, n, max_iters, T0, T1, rng, block_max=5):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5

        T = T0 * (T1 / T0) ** frac

            zeros = np.where(b == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0, 1
            nc = compute_c5(b)
            if nc <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(nc - curr_c5) / T)):
                curr_c5 = nc
                if nc < best_c5:

            else:

                    curr_c5 = nc

    T = _sl_T

    mc = np.max(corr)
    sh = corr - mc
    dx = 2.0 / len(x)

    return np.sum(x) - len(x) / 2.0

    return np.ones(len(x))

    _sl_n = n
    _sl_T = T
    bounds = [(0.0, 1.0)] * n

    res = minimize(_sl_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
