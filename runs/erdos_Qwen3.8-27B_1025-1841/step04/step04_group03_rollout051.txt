def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

                softmax_grad[il:n] += w * (one_minus[:n - il])
                softmax_grad[:n - il] -= w * (one_minus[:n - il])
        penalty_grad = 2.0 * lam * (np.sum(x) - _g_n / 2.0) * np.ones(n)
        return softmax_grad + penalty_grad
    try:
        res = minimize(obj, h.copy(), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
        return make_feasible(res.x, n)
    except Exception:
        return h.copy()

def multi_temp_slsqp(h, n, t0, budget, temps=None):
    if temps is None:
        temps = [30, 60, 150, 400, 1000, 3000, 10000, 40000, 150000]
    best_h = h.copy()
    best_c = compute_c5(h)
    for T in temps:
        if time.time() - t0 > budget - 15:
            break
        x = do_slsqp(best_h, n, T, 200, 1e-14)
        c = compute_c5(x)
        if c < best_c - 1e-15:
            best_c = c
            best_h = x
    return best_h, best_c

def cont_sa(h, n, iters, T0, T1, rng, t0, budget, step0=0.06, step1=0.003):
    b = h.copy()
    curr = compute_c5(b)
    best = curr
    bb = b.copy()
    for it in range(iters):
        if time.time() - t0 > budget - 5:
            break
        f = it / max(1, iters)
        T = T0 * (T1 / T0) ** f
        step = step0 * (step1 / step0) ** f
        npert = rng.randint(1, 4)
        saved = b.copy()
        for _ in range(npert):
            i = rng.randint(n)
            b[i] = np.clip(b[i] + rng.uniform(-step, step), 0, 1)
        b = make_feasible(b, n)
        nc = compute_c5(b)
        if nc <= curr or (T > 1e-15 and rng.random() < np.exp(-(nc - curr) / T)):
            curr = nc
            if nc < best:
                best = nc
                bb = b.copy()
        else:
            b = saved.copy()
    return bb, best

def coord_finestep(h, n, max_rounds, t0, budget):
    b = h.copy()
    best_c = compute_c5(b)
    for rnd in range(max_rounds):
        if time.time() - t0 > budget - 3:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 2:
                break
            old = b[j]
            bc = best_c
            bv = old
            for cand in [0.0, 1.0, max(0.0, old - 0.003), max(0.0, old - 0.0005),
                         min(1.0, old + 0.0005), min(1.0, old + 0.003)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = b.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-14:
                    bc = c
                    bv = cand
            if bv != old:
                b[j] = bv
                b = make_feasible(b, n)
                best_c = bc
                improved = True
        if not improved:
            break
    return b, best_c

def targeted_perturb(h, n, rng, n_active=10, amp=0.02):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    thresh = mc - 0.002 / dx
    active_idx = np.where(corr >= thresh)[0]
    h_new = h.copy()
    for k_idx in active_idx[:n_active]:
        lag = k_idx - (n - 1)
        for _ in range(6):
            i = rng.randint(n)
            j = i + lag
            if 0 <= j < n:
                if h_new[j] < 0.4:
                    h_new[i] = max(0.0, h_new[i] - amp * rng.uniform(0.5, 1.5))
                    h_new[j] = min(1.0, h_new[j] + amp * rng.uniform(0.3, 1.0))
                elif h_new[j] > 0.6:
                    h_new[i] = min(1.0, h_new[i] + amp * rng.uniform(0.3, 1.0))
                    h_new[j] = max(0.0, h_new[j] - amp * rng.uniform(0.5, 1.5))
    return make_feasible(h_new, n)

def fourier_perturb(h, n, rng, n_modes=4, amp=0.04):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 10))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.3, 1.5)
    return make_feasible(h_new, n)

def random_perturb(h, n, rng, scale=0.04):
    h_new = h + rng.uniform(-scale, scale, size=n)
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(480, 0.5)
    n0 = len(h0)
    h0 = make_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Multi-temp SLSQP
    if budget_s - (time.time() - t0) > 280:
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, t0, budget_s)

    # Phase 2: L-BFGS-B
    if budget_s - (time.time() - t0) > 150:
        for T in [200, 2000, 20000]:
            if time.time() - t0 > budget_s - 40:
                break
            x = do_lbfgsb(best_h, n0, T, 200)
            c = compute_c5(x)
            if c < best_c5 - 1e-15:
                best_c5 = c
                best_h = x

    # Phase 3: Continuous SA
    if budget_s - (time.time() - t0) > 200:
        best_h, best_c5 = cont_sa(best_h, n0, 400000, 0.004, 0.00002, rng, t0, budget_s,
                                  step0=0.07, step1=0.002)

    # Phase 4: SLSQP after SA
    if budget_s - (time.time() - t0) > 160:
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, t0, budget_s,
                                           temps=[100, 500, 2000, 8000, 30000])

    # Phase 5: Perturb + optimize cycles
    perts = [targeted_perturb, fourier_perturb, random_perturb, targeted_perturb]
    for pf in perts:
        if time.time() - t0 > budget_s - 100:
            break
        hp = pf(best_h, n0, rng)
        hr, cr = multi_temp_slsqp(hp, n0, t0, budget_s, temps=[200, 1000, 5000])
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()

    # Phase 6: More SA with different params
    if budget_s - (time.time() - t0) > 120:
        best_h, best_c5 = cont_sa(best_h, n0, 200000, 0.002, 0.00001, rng, t0, budget_s,
                                  step0=0.04, step1=0.001)

    # Phase 7: Fine coordinate descent
    if budget_s - (time.time() - t0) > 60:
        best_h, best_c5 = coord_finestep(best_h, n0, 25, t0, budget_s)

    # Phase 8: Higher resolution
    if budget_s - (time.time() - t0) > 250:
        n1 = n0 + 480
        x_old = np.linspace(0, 2, n0, endpoint=False)
        x_new = np.linspace(0, 2, n1, endpoint=False)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = make_feasible(h_up, n1)
        h_up, c_up = multi_temp_slsqp(h_up, n1, t0, budget_s,
                                      temps=[100, 500, 2000, 10000])
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up
        if budget_s - (time.time() - t0) > 60:
            best_h, best_c5 = coord_finestep(best_h, len(best_h), 8, t0, budget_s)

    # Phase 9: Final SLSQP + coord
    if budget_s - (time.time() - t0) > 40:
        x = do_slsqp(best_h, len(best_h), 200000, 200, 1e-15)
        c = compute_c5(x)
        if c < best_c5:
            best_c5 = c
            best_h = x
    if budget_s - (time.time() - t0) > 15:
        best_h, best_c5 = coord_finestep(best_h, len(best_h), 5, t0, budget_s)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
