import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def slsqp_epigraph(h, n, top_k=10, max_iter=200, ftol=1e-14):
    """SLSQP with epigraph: minimize t s.t. corr_k*dx <= t for top_k shifts."""
    dx = 2.0 / n
    
    def get_top_idx(corr):
        return np.argsort(corr)[-top_k:]
    
    corr = compute_corr(h)
    top_idx = get_top_idx(corr)
    
    def obj_t(x):
        return x[n]  # minimize t
    
    def obj_t_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    # SLSQP with changing active set
    x0 = np.concatenate([h, [np.max(corr[top_idx]) * dx]])
    
    for round in range(5):
        corr_h = compute_corr(x0[:n])
        top_idx = get_top_idx(corr_h)
        
        def constraint_fun(x):
            h_x = x[:n]
            t = x[n]
            corr_x = compute_corr(h_x)
            vals = corr_x[top_idx] * dx - t
            return vals
        
        def constraint_grad(x):
            h_x = x[:n]
            J = np.zeros((top_k, n + 1))
            for ii, s_idx in enumerate(top_idx):
                s = s_idx - (n - 1)
                for j in range(n):
                    val = 0.0
                    if 0 <= j + s < n:
                        val += (1.0 - h_x[j + s])
                    if 0 <= j - s < n:
                        val += -h_x[j - s]
                    J[ii, j] = val * dx
                J[ii, n] = -1.0
            return J
        
        cons = [{'type': 'ineq', 'fun': constraint_fun, 'jac': constraint_grad}]
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        
        try:
            res = minimize(obj_t, x0, method='SLSQP', jac=obj_t_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': max_iter, 'ftol': ftol})
            if res.success or res.fun < obj_t(x0):
                x0 = make_feasible(np.clip(res.x[:n], 0, 1), n)
                x0_t = np.max(compute_corr(x0)[top_idx]) * dx
                x0 = np.concatenate([x0, [x0_t]])
        except Exception:
            break
    
    h_final = make_feasible(x0[:n], n)
    return h_final, compute_c5(h_final)

def paired_coord_descent(h, n, budget_s, start_time, top_m=10):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    while time.time() - start_time < budget_s:
        corr = compute_corr(best_h)
        top_idx = np.argsort(corr)[-top_m:]
        c5_cur = best_c5
        
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            # Find best k to pair with j
            # Try a few random k's
            for _ in range(3):
                k = int(np.random.randint(n))
                if k == j:
                    continue
                
                # Compute linear coefficients for each top shift
                lin = np.zeros(top_m)
                for ii, s_idx in enumerate(top_idx):
                    s = s_idx - (n - 1)
                    # Coefficient for h[j]
                    cj = 0.0
                    if 0 <= j + s < n:
                        cj += (1.0 - best_h[j + s])
                    if 0 <= j - s < n:
                        cj += -best_h[j - s]
                    # Coefficient for h[k]
                    ck = 0.0
                    if 0 <= k + s < n:
                        ck += (1.0 - best_h[k + s])
                    if 0 <= k - s < n:
                        ck += -best_h[k - s]
                    lin[ii] = cj - ck  # change in corr_s for δ at j, -δ at k
                
                # corr_s + δ * lin_s, find δ that minimizes max
                # Subject to: max(-h[j], h[k]-1) ≤ δ ≤ min(1-h[j], h[k])
                delta_lo = max(-best_h[j], best_h[k] - 1)
                delta_hi = min(1 - best_h[j], best_h[k])
                if delta_hi - delta_lo < 1e-12:
                    continue
                
                # Binary search for optimal δ
                best_delta = 0.0
                best_val = np.max(corr[top_idx])
                for _ in range(30):  # bisection
                    dm = (delta_lo + delta_hi) / 2
                    vals = corr[top_idx] + dm * lin
                    vm = np.max(vals)
                    if vm < best_val:
                        best_val = vm
                        best_delta = dm
                    # Subgradient: if any lin[ii] > 0 for the argmax, decrease δ
                    argmax_mask = np.abs(vals - vm) < 1e-10
                    if np.sum(lin[argmax_mask]) > 0:
                        delta_hi = dm
                    else:
                        delta_lo = dm

                    if c_trial < best_c5 - 1e-14:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0, 1))
    return h

def corr_shift_grad(h, s):
    """Gradient of correlation at shift s with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    t = abs(s)
    if t == 0:
        return 1.0 - 2.0 * h
    if s > 0:
        # corr = sum h[i+s]*(1-h[i]) for i in [0, n-1-s]
        # d/dh[j] = (1-h[j-s]) for j in [s, n-1] - h[j+s] for j in [0, n-1-s]
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] += -h[s:]
    else:
        # corr = sum h[i]*(1-h[i+t]) for i in [0, n-1-t]
        # d/dh[j] = (1-h[j+t]) for j in [0, n-1-t] - h[j-t] for j in [t, n-1]
        grad[:n-t] += (1.0 - h[t:])
        grad[t:] += -h[:n-t]
    return grad

def slsqp_epigraph(h, n, top_k, maxiter, start_time, budget_s):
    dx = 2.0 / n
    
    for rnd in range(8):
        if time.time() - start_time > budget_s - 30:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        
        def obj(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def con_fun(x):
            hx = x[:n]
            t_val = x[n]
            c = compute_corr(hx)
            return c[top_idx] * dx - t_val
        
        def con_jac(x):
            hx = x[:n]
            J = np.zeros((top_k, n + 1))
            for ii, si in enumerate(top_idx):
                s = si - (n - 1)
                J[ii, :n] = corr_shift_grad(hx, s) * dx
                J[ii, n] = -1.0
            return J
        
        # Sum constraint
        def sum_fun(x):
            return np.sum(x[:n]) - n / 2.0
        
        def sum_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
                {'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        
        t0 = np.max(corr[top_idx]) * dx
        x0 = np.concatenate([h, [t0]])
        
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-16})
            if res.x[n] < t0:
                h_new = make_feasible(np.clip(res.x[:n], 0, 1), n)
                c5_new = compute_c5(h_new)
                if c5_new < compute_c5(h):
                    h = h_new
        except Exception:
            pass
    
    return h, compute_c5(h)

def pgd_fine(h, n, start_time, budget_s, top_m=5):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    velocity = np.zeros(n)

    momentum = 0.85

    no_improve = 0

        it += 1
        corr = compute_corr(best_h)
        c5_cur = np.max(corr) * dx
        
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:

                    break
                no_improve = 0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = float(np.clip(h[idx] + diff, 0, 1))
    return h

def corr_shift_grad(h, s):
    n = len(h)
    grad = np.zeros(n)
    t = abs(s)
    if t == 0:
        return 1.0 - 2.0 * h
    if s > 0:
        grad[s:] += (1.0 - h[:n - t])
        grad[:n - t] += -h[s:]
    else:
        grad[:n - t] += (1.0 - h[t:])
        grad[t:] += -h[:n - t]
    return grad

def compute_top_corr(h, top_idx, n):
    vals = np.zeros(len(top_idx))
    for ii, si in enumerate(top_idx):
        s = si - (n - 1)
        t = abs(s)
        if t == 0:
            vals[ii] = np.sum(h * (1.0 - h))
        else:
            vals[ii] = np.sum(h[:n - t] * (1.0 - h[t:]))
    return vals

def slsqp_epigraph(h, n, top_k, maxiter, start_time, budget_s):
    dx = 2.0 / n
    for rnd in range(10):
        if time.time() - start_time > budget_s - 20:
            break
        corr = compute_corr(h)
        top_idx = np.argsort(corr)[-top_k:]
        
        def obj(x):
            return x[n]
        
        def obj_grad(x):
            g = np.zeros(n + 1)
            g[n] = 1.0
            return g
        
        def con_fun(x):
            hx = x[:n]
            t_val = x[n]
            return compute_top_corr(hx, top_idx, n) * dx - t_val
        
        def con_jac(x):
            hx = x[:n]
            J = np.zeros((top_k, n + 1))
            for ii, si in enumerate(top_idx):
                s = si - (n - 1)
                J[ii, :n] = corr_shift_grad(hx, s) * dx
                J[ii, n] = -1.0
            return J
        
        def sum_fun(x):
            return np.sum(x[:n]) - n / 2.0
        
        def sum_jac(x):
            g = np.zeros(n + 1)
            g[:n] = 1.0
            return g
        
        cons = [{'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
                {'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
        bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
        
        t0 = np.max(compute_top_corr(h, top_idx, n)) * dx
        x0 = np.concatenate([h, [t0]])
        
        try:
            res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                          bounds=bounds, constraints=cons,
                          options={'maxiter': maxiter, 'ftol': 1e-18})
            h_new = make_feasible(np.clip(res.x[:n], 0, 1), n)
            c5_new = compute_c5(h_new)
            c5_old = compute_c5(h)
            if c5_new < c5_old - 1e-15:
                h = h_new
            else:
                break
        except Exception:
            break
    return h, compute_c5(h)
