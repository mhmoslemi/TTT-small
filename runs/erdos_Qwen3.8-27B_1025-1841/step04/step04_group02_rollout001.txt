def binary_sa_fast(h, n, max_iters, T0, T1, rng, time_limit):

            j = rng.randint(n)

def slsqp_objective(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_sum(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_sum_jac(x):
    return np.ones(len(x))

def slsqp_polish(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum, 'jac': slsqp_sum_jac}]
    res = minimize(slsqp_objective, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   args=(n, T),
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

        improved = False
        if time.time() > budget_s:
            break
        for j in range(n):
            if time.time() > budget_s:
                break
            old = h[j]
            best_val = old
            best_c = best_c5
            for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:

                trial[j] = cand
                trial = make_feasible(trial, n)

                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

    if time.time() < hard_limit - 80:
        n_pts = len(best_h)
        b_fin = to_binary(best_h, n_pts)
        t_limit = min(time.time() + 40, hard_limit - 30)
        bf, cf = binary_sa_swap(b_fin, n_pts, 500000, 0.003, 0.000001, rng, t_limit)
        cf_real = compute_c5(bf)
        if cf_real < best_c5:
            best_c5 = cf_real
            best_h = bf.copy()

    # Phase 9: Coordinate descent
    if time.time() < hard_limit - 30:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 15, start_time, hard_limit)

    # Final SLSQP if time
    if time.time() < hard_limit - 10:
        n_pts = len(best_h)
        try:
            x_res = slsqp_polish(best_h, n_pts, 500.0, 200, 1e-15)
            x_res = make_feasible(x_res, n_pts)
            c_res = compute_c5(x_res)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = x_res
        except Exception:
            pass

    # Final verification with exact computation
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
