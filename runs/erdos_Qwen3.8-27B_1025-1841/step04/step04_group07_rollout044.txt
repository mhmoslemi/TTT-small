    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_weighted, h0, args=(n, weights, mu), method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds, tries_single, tries_double):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones_list = np.where(h > 0.5)[0].tolist()
    zeros_list = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        # Single swaps
        ns = min(tries_single, max(1, len(ones_list) * len(zeros_list)))
        for _ in range(ns):
            if not ones_list or not zeros_list:
                break
            ii = rng.randint(len(ones_list))
            jj = rng.randint(len(zeros_list))
            oi, oj = ones_list[ii], zeros_list[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-14:
                cur = c
                ones_list[ii] = oj
                zeros_list[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        # Double swaps
        if len(ones_list) >= 2 and len(zeros_list) >= 2:
            nd = min(tries_double, max(1, (len(ones_list) * (len(ones_list) - 1)) // 2))
            for _ in range(nd):
                if rnd > 80:
                    break
                i1 = rng.randint(len(ones_list))
                i2 = rng.randint(len(ones_list))
                j1 = rng.randint(len(zeros_list))
                j2 = rng.randint(len(zeros_list))
                if i1 == i2 or j1 == j2:
                    continue
                o1, o2 = ones_list[i1], ones_list[i2]
                z1, z2 = zeros_list[j1], zeros_list[j2]
                h[o1] = 0.0; h[o2] = 0.0
                h[z1] = 1.0; h[z2] = 1.0
                c = compute_c5(h)
                if c < cur - 1e-14:
                    cur = c
                    ones_list[i1] = z1; ones_list[i2] = z2
                    zeros_list[j1] = o1; zeros_list[j2] = o2
                    improved = True
                else:
                    h[o1] = 1.0; h[o2] = 1.0
                    h[z1] = 0.0; h[z2] = 0.0
        if not improved and rnd > 30:
            break
    return h, cur

def coord_polish(h0, n, max_rounds, steps):
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-10 or nv > 1.0 + 1e-10:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    n_base = len(h0)

    # Phase 1: L-BFGS-B at base resolution with adaptive temperatures
    h_work = h0.copy()
    for T in [20, 60, 200, 600, 30, 100]:
        if rem() < 25:
            break
        h_new = run_lbgfb(h_work, n_base, T, 600.0, 180)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new.copy()
            h_work = h_new.copy()
        elif c_new < compute_c5(h_work):
            h_work = h_new.copy()

    # Phase 2: Alternating top-k weighted optimization at base
    if rem() > 50:
        for topk_frac in [0.04, 0.08, 0.15, 0.25]:
            if rem() < 15:
                break
            for it in range(5):
                if rem() < 8:
                    break
                w = get_topk_weights(h_work, n_base, topk_frac)
                h_new = run_weighted_lbgfb(h_work, n_base, w, 800.0, 35)
                c_new = compute_c5(h_new)
                if c_new <= best_c5 + 5e-6:
                    h_work = h_new
                    if c_new < best_c5:
                        best_c5, best_h = c_new, h_work.copy()

    # Phase 3: Upsample and optimize at higher resolutions
    for n_hi in [n_base * 2, n_base * 4]:
        if rem() < 100:
            break
        if n_hi > 4096:
            break
        h_hi = upsample_h(best_h, n_hi)
        c_hi = compute_c5(h_hi)
        if c_hi < best_c5:
            best_c5, best_h = c_hi, h_hi.copy()

        for T in [40, 150, 500]:
            if rem() < 35:
                break
            h_new = run_lbgfb(h_hi, n_hi, T, 400.0, 120)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new.copy()
                h_hi = h_new.copy()
            elif c_new < c_hi:
                h_hi = h_new.copy()
                c_hi = c_new

        if rem() > 30:
            w = get_topk_weights(h_hi, n_hi, 0.06)
            h_new = run_weighted_lbgfb(h_hi, n_hi, w, 500.0, 50)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new.copy()
            elif c_new < c_hi:
                h_hi = h_new.copy()

    # Phase 4: Binary local search with 2-swap moves
    if rem() > 100:
        n_bin = min(n_base, 1024)
        if n_bin % 2 != 0:
            n_bin -= 1
        h_b = upsample_h(best_h, n_bin) if n_bin < n_base else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        target_ones = n_bin // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            perm = rng.permutation(len(idx))
            for i in range(n_ones - target_ones):
                h_bin[idx[perm[i]]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            perm = rng.permutation(len(idx))
            for i in range(target_ones - n_ones):
                h_bin[idx[perm[i]]] = 1.0

        h_bs, c_bs = binary_local_search(h_bin, n_bin, rng, max_rounds=150,
                                         tries_single=6000, tries_double=2000)
        if c_bs < best_c5:
            best_c5, best_h = c_bs, h_bs.copy()

    # Phase 5: Multi-start perturbations at base resolution
    n_ms = n_base
    for restart in range(4):
        if rem() < 30:
            break
        noise_scale = 0.08 + 0.04 * restart
        h_r = best_h + noise_scale * rng.randn(n_ms)
        h_r = make_feasible(h_r, n_ms)
        for T in [80, 300]:
            if rem() < 18:
                break
            h_opt = run_lbgfb(h_r, n_ms, T, 500.0, 150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h_r = h_opt.copy()

    # Phase 6: Final high-res L-BFGS-B if time permits
    if rem() > 80:
        n_fin = len(best_h)
        if n_fin < 2560:
            n_fin = min(2560, n_base * 4)
        h_fin = upsample_h(best_h, n_fin)
        c_fin = compute_c5(h_fin)
        if c_fin < best_c5:
            best_c5, best_h = c_fin, h_fin.copy()
        for T in [50, 200]:
            if rem() < 30:
                break
            h_new = run_lbgfb(h_fin, n_fin, T, 350.0, 100)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5, best_h = c_new, h_new.copy()
                h_fin = h_new.copy()

    # Phase 7: Coordinate descent final polish
    if rem() > 25:
        n_cd = len(best_h)
        h_cd, c_cd = coord_polish(best_h, n_cd, max_rounds=6,
                                  steps=[0.05, -0.05, 0.02, -0.02, 0.008, -0.008, 0.003, -0.003])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
