import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_smooth(h0, n, T, mu, max_iter=200):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

    return make_feasible(h_new, new_n)

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        order = idx[np.argsort(h[idx])]
        h_bin[order[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        order = idx[np.argsort(-h[idx])]
        h_bin[order[:target - n_ones]] = 1.0
    return h_bin

def local_search_single(h_bin, n, rng, deadline, tries_limit=200000):
    h = h_bin.copy()
    cur_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    dx = 2.0 / n
    tries = 0
    while time.time() < deadline and tries < tries_limit:
        batch = min(5000, tries_limit - tries)
        for _ in range(batch):
            if time.time() > deadline:
                break
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            new_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if new_max < cur_max:
                cur_max = new_max
                ones[ii] = oj
                zeros[jj] = oi
            else:
                h[oi] = 1.0
                h[oj] = 0.0
            tries += 1
        if tries >= tries_limit:
            break
    return h, cur_max * dx

def iterated_local_search(h_bin, n, rng, deadline, n_restarts=50):
    dx = 2.0 / n
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    h = h_bin.copy()
    cur_max = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    per_restart = max(20, min(40000, 800000 // max(1, n_restarts)))
    
    for restart in range(n_restarts):
        if time.time() > deadline:
            break
        h, cur_c5 = local_search_single(h, n, rng, deadline, tries_limit=per_restart)
        c5_val = compute_c5(h)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if time.time() > deadline:
            break
        
        n_kicks = rng.randint(8, 25)
        ones = np.where(h > 0.5)[0].tolist()
        zeros = np.where(h <= 0.5)[0].tolist()
        for _ in range(n_kicks):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            h[ones[ii]] = 0.0
            h[zeros[jj]] = 1.0
            ones[ii] = zeros[jj]
            zeros[jj] = ones[ii - 1] if ii > 0 else zeros[jj]
            # Fix the lists properly
        # Rebuild lists after kicks
        ones = np.where(h > 0.5)[0].tolist()
        zeros = np.where(h <= 0.5)[0].tolist()
        # Ensure correct count
        target = n // 2
        n_ones = len(ones)
        if n_ones > target:
            for _ in range(n_ones - target):
                idx = rng.randint(len(ones))
                zeros.append(ones.pop(idx))
        elif n_ones < target:
            for _ in range(target - n_ones):
                idx = rng.randint(len(zeros))
                ones.append(zeros.pop(idx))
    
    return best_h, best_c5

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_optimize(h0, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def alternating_minmax(h0, n, n_iters=40, topk_frac=0.12, max_inner=35, deadline=None):
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_c = len(corr)
        k = max(5, int(topk_frac * n_c))
        idx = np.argpartition(corr, -k)[-k:]
        active = np.zeros(n_c)
        active[idx] = 1.0
        active /= active.sum()
        mu = 500.0
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(active, c)) * dx + mu * (x.sum() - n / 2.0) ** 2
        try:
            res = minimize(obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = make_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new <= best_c5 + 1e-10:
                h = h_new
                best_c5 = min(best_c5, c_new)
        except Exception:
            pass
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: Intensive binary search at parent resolution (45% budget)
    n_p = len(h0)
    h_bin = to_binary(h0, n_p)
    c_bin = compute_c5(h_bin)
    if c_bin < best_c5:
        best_c5, best_h = c_bin, h_bin.copy()
    
    deadline_p1 = t0 + budget_s * 0.45
    h_p1, c_p1 = iterated_local_search(h_bin, n_p, rng, deadline_p1, n_restarts=100)
    if c_p1 < best_c5:
        best_c5, best_h = c_p1, h_p1.copy()
    
    # Phase 2: Higher resolutions (35% budget)
    deadline_p2 = t0 + budget_s * 0.80
    for n2 in [960, 1280]:
        if rem() < 120 or time.time() > deadline_p2:
            break
        h2 = upsample_h(best_h, n2)
        h2_bin = to_binary(h2, n2)
        c2 = compute_c5(h2_bin)
        if c2 < best_c5:
            best_c5, best_h = c2, h2_bin.copy()
        
        sub_deadline = min(t0 + budget_s * 0.80, time.time() + rem() * 0.4)
        h_bs, c_bs = iterated_local_search(h2_bin, n2, rng, sub_deadline, n_restarts=40)
        if c_bs < best_c5:
            best_c5, best_h = c_bs, h_bs.copy()
    
    # Phase 3: Continuous optimization (15% budget)
    if rem() > 60:
        n_c = len(best_h)
        h_c = best_h.copy()
        cont_deadline = t0 + budget_s * 0.95
        for T in [300, 1500, 6000]:
            if time.time() > cont_deadline or rem() < 30:
                break
            h_opt = lbgfb_optimize(h_c, n_c, T, mu=1000.0, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h_c = h_opt.copy()
    
    # Phase 4: Alternating min-max
    if rem() > 40:
        n_am = len(best_h)
        h_am, c_am = alternating_minmax(best_h, n_am, n_iters=30, topk_frac=0.10,
                                         max_inner=30, deadline=t0 + budget_s - 20)
        if c_am < best_c5:
            best_c5, best_h = c_am, h_am.copy()
    
    # Phase 5: Final binary polish
    if rem() > 30:
        n_f = len(best_h)
        h_f_bin = to_binary(best_h, n_f)
        h_f, c_f = iterated_local_search(h_f_bin, n_f, rng, t0 + budget_s - 5, n_restarts=30)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
    
    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    return best_h, best_c5, len(best_h)
