    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Fast subgradient with iterate averaging
    if remaining() > 150:
        t_alloc = min(150, remaining() - 30)
        max_it = int(t_alloc / 0.05)  # rough estimate
        h, c5 = fast_subgradient_optimize(best_h, n, max_iter=max_it)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 2: SLSQP with smooth max
    if remaining() > 100:
        dx = 2.0 / n
        def obj(x):
            return slsqp_smooth(x, n, T=1000.0)
        def obj_grad(x):
            # Numeric gradient
            eps = 1e-5
            g = np.zeros(n)
            f0 = obj(x)
            for i in range(n):
                xp = x.copy()
                xp[i] += eps
                xp = make_feasible(xp, n)
                g[i] = (obj(xp) - f0) / eps
            return g
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
                 'jac': lambda x: np.ones(n)}]
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 200, 'ftol': 1e-16})
            c5_res = compute_c5(make_feasible(res.x, n))
            if c5_res < best_c5:
                best_h = make_feasible(res.x, n)
                best_c5 = c5_res
        except Exception:
            pass
    
    # Phase 3: Coordinate descent with binary search
    if remaining() > 120:
        n_pts = len(best_h)
        dx = 2.0 / n_pts
        improved = True
        passes = 0
        while improved and remaining() > 30 and passes < 20:
            improved = False
            passes += 1
            for j in range(n_pts):
                if remaining() < 35:
                    break
                old_val = best_h[j]
                lo, hi = 0.0, 1.0
                for _ in range(25):
                    mid = (lo + hi) / 2.0
                    trial = best_h.copy()
                    trial[j] = mid
                    trial = make_feasible(trial, n_pts)
                    c_trial = compute_c5(trial)
                    # Compare with value at (lo+hi)/2 + small perturbation
                    # Use subgradient sign: if increasing h[j] decreases c5, go right
                    trial2 = best_h.copy()
                    trial2[j] = min(1.0, mid + 0.01)
                    trial2 = make_feasible(trial2, n_pts)
                    c_trial2 = compute_c5(trial2)
                    if c_trial2 < c_trial:
                        lo = mid
                    else:
                        hi = mid
                # Evaluate at best
                v_lo = best_h.copy()
                v_lo[j] = lo
                v_lo = make_feasible(v_lo, n_pts)
                v_hi = best_h.copy()
                v_hi[j] = hi
                v_hi = make_feasible(v_hi, n_pts)
                v_mid = best_h.copy()
                v_mid[j] = (lo+hi)/2
                v_mid = make_feasible(v_mid, n_pts)
                c_lo = compute_c5(v_lo)
                c_hi = compute_c5(v_hi)
                c_mid = compute_c5(v_mid)
                best_v = min(c_lo, c_hi, c_mid)
                if c_lo == best_v:
                    best_h = v_lo
                elif c_hi == best_v:
                    best_h = v_hi
                else:
                    best_h = v_mid
                best_c5 = compute_c5(best_h)
                if best_c5 < best_c5 + 1e-14:  # always true, just for clarity
                    improved = True
                if abs(best_h[j] - old_val) > 1e-6:
                    improved = True
    
    # Phase 4: SLSQP top-3 focus
    if remaining() > 80:
        def obj_top3(x):
            return slsqp_smooth(x, len(x), T=3000.0, top_m=3)
        bounds = [(0.0, 1.0)] * len(best_h)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2.0,
                 'jac': lambda x: np.ones(len(x))}]
        try:
            res = minimize(obj_top3, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
            c5_res = compute_c5(make_feasible(res.x, len(best_h)))
            if c5_res < best_c5:
                best_h = make_feasible(res.x, len(best_h))
                best_c5 = c5_res
        except Exception:
            pass
    
    # Phase 5: Random restarts
    for restart in range(4):
        if remaining() < 60:
            break
        scale = 0.005 + 0.005 * restart
        h_try = make_feasible(best_h + rng.randn(len(best_h)) * scale, len(best_h))
        # Quick subgradient
        h2, c5_2 = fast_subgradient_optimize(h_try, len(best_h), max_iter=200)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        # Quick SLSQP
        if remaining() > 40:
            try:
                def obj_r(x):
                    return slsqp_smooth(x, len(x), T=2000.0)
                bounds_r = [(0.0, 1.0)] * len(x)
                cons_r = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2.0,
                           'jac': lambda x: np.ones(len(x))}]
                res = minimize(obj_r, best_h, method='SLSQP', bounds=bounds_r,
                              constraints=cons_r, options={'maxiter': 100, 'ftol': 1e-15})
                c5_res = compute_c5(make_feasible(res.x, len(res.x)))
                if c5_res < best_c5:
                    best_h = make_feasible(res.x, len(res.x))
                    best_c5 = c5_res
            except Exception:
                pass
    
    # Phase 6: Try lower resolution (downsample)
    if remaining() > 100:
        n_down = len(best_h) // 2
        if n_down >= 240:
            h_su = np.zeros(n_down)
            block = len(best_h) // n_down
            for i in range(n_down):
                h_su[i] = np.mean(best_h[i*block:(i+1)*block])
            h_su = make_feasible(h_su, n_down)
            c5_su = compute_c5(h_su)
            if c5_su < best_c5 + 0.001:
                h2, c5_2 = fast_subgradient_optimize(h_su, n_down, max_iter=500)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2
    
    # Phase 7: Final polish
    if remaining() > 30:
        try:
            def obj_final(x):
                return slsqp_smooth(x, len(x), T=5000.0, top_m=3)
            bounds_f = [(0.0, 1.0)] * len(best_h)
            cons_f = [{'type': 'eq', 'fun': lambda x: np.sum(x) - len(x)/2.0,
                       'jac': lambda x: np.ones(len(x))}]
            res = minimize(obj_final, best_h, method='SLSQP', bounds=bounds_f,
                          constraints=cons_f, options={'maxiter': 200, 'ftol': 1e-17})
            c5_res = compute_c5(make_feasible(res.x, len(res.x)))
            if c5_res < best_c5:
                best_h, best_c5 = make_feasible(res.x, len(res.x)), c5_res
        except Exception:
            pass
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
