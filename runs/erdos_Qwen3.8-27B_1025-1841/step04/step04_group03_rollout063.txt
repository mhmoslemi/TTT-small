```python
def projected_subgradient(h, N, start_time, budget, lr_init=0.01, beta_init=500, rng=None):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    lr = lr_init
    beta = beta_init
    
    it = 0
    while time.time() - start_time < budget:
        # Compute correlation
        corr = np.correlate(h, 1.0-h, mode='full')
        mc = np.max(corr)
        
        # Soft-max weights
        shifted = corr - mc
        W = np.exp(beta * shifted)
        W_sum = np.sum(W)
        if W_sum > 0:
            W /= W_sum
        else:
            W = np.zeros_like(W)
            W[np.argmax(corr)] = 1.0
        
        # W_full: W is already indexed 0 to 2N-2 (corr indices)
        # W_full[k] = W[k] for k=0..2N-2
        W_full = W  # Already the right shape!
        
        # Compute gradient
        partA = np.correlate(W_full, 1.0-h, mode='full')[:N]
        partB = np.correlate(W_full, h, mode='full')[2*N-2 : 2*N-2+N][::-1]
        # Wait, partB[j] = np.correlate(W_full, h, 'full')[2N-2-j]
        # So partB = np.correlate(W_full, h, 'full')[2N-2 : 2N-2-N : -1]  -- no, that's not right
        # partB[j] for j=0..N-1 = result[2N-2-j] 
        # = result[2N-2], result[2N-3], ..., result[N-2]
        # So partB = result[N-2 : 2N-1][::-1] = result[N-2:2N-1] reversed
        
        corr_B = np.correlate(W_full, h, mode='full')
        partB = corr_B[N-2 : 2*N-1][::-1]  # This gives [result[2N-2], result[2N-3], ..., result[N-2]]
        
        grad = partA - partB
        
        # Gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, N)
        
        new_c5 = compute_c5(h_new)
        
        # Accept/reject (simple: always accept if improving, otherwise with SA probability)
        T = max(0.0001, 0.001 * (0.999 ** it))
        if new_c5 < curr_c5 or (rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            h = h_new
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        
        # Decay
        if it % 100 == 0:
            lr *= 0.99
            beta *= 1.001
        
        it += 1
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, W, n):
    """Compute subgradient of max-correlation w.r.t. h."""
    W_rev = W[::-1]
    partA = np.correlate(W_rev, 1.0 - h, mode='full')[n-1 : 2*n-1][::-1]
    partB = np.correlate(W, h, mode='full')[n-1 : 2*n-1][::-1]
    return partA - partB

def soft_max_weights(corr, beta):
    mc = np.max(corr)
    shifted = corr - mc
    W = np.exp(beta * shifted)
    s = np.sum(W)
    if s > 0:
        W /= s
    else:
        W = np.zeros_like(corr)
        W[np.argmax(corr)] = 1.0
    return W

def subgrad_descent(h, n, start_time, time_limit, rng, 
                    lr_init=0.01, lr_decay=0.9998, beta_init=200, beta_growth=1.0005,
                    sa_T0=0.002, sa_T1=1e-5):
    """Projected subgradient with soft-max focusing and SA."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    lr = lr_init
    beta = beta_init
    
    it = 0
    n_total = 200000  # max iterations as safety
    
    while it < n_total:
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        W = soft_max_weights(corr, beta)
        grad = compute_grad(h, W, n)
        
        # Step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        new_c5 = compute_c5(h_new)
        
        # SA acceptance
        frac = min(1.0, it / n_total)
        T = sa_T0 * (sa_T1 / sa_T0) ** frac
        if new_c5 <= curr_c5:
            accept = True
        else:
            delta = new_c5 - curr_c5
            accept = T > 1e-12 and rng.random() < np.exp(-delta / T)
        
        if accept:
            h = h_new
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        
        # Decay
        if it % 50 == 0:
            lr *= 0.995
            beta *= 1.002
            if beta > 5000:
                beta = 5000
        
        it += 1
    
    return best_h, best_c5
