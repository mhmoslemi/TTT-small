import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse_binary(n):
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    t = t.astype(float)
    # Balance: need exactly n//2 ones
    target = n // 2
    num_ones = int(np.sum(t))
    if num_ones > target:
        ones = np.where(t == 1)[0]
        excess = num_ones - target
        flip = ones[:excess]
        t[flip] = 0
    elif num_ones < target:
        zeros = np.where(t == 0)[0]
        deficit = target - num_ones
        flip = zeros[:deficit]
        t[flip] = 1
    return t

def upsine_init(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2.0, n)
    h = np.zeros(n)
    ncomp = rng.randint(2, 5)
    for _ in range(ncomp):
        f = rng.uniform(1, 6)
        phase = rng.uniform(0, 2*np.pi)
        amp = rng.uniform(0.3, 1.0)
        h += amp * np.sin(np.pi * f * x + phase)
    h = np.clip(h, 0, 1)
    h = make_feasible(h, n)
    return h

def update_corr(c, h, i0, delta, n):
    """Update correlation vector when h[i0] changes by delta. Returns updated c (in-place modification)."""
    # First case: m = i0, k from max(0, n-1-i0) to min(2n-2, 2n-2-i0)
    k_start1 = max(0, n - 1 - i0)
    k_end1 = min(2 * n - 2, 2 * n - 2 - i0)
    for k in range(k_start1, k_end1 + 1):
        j = i0 + k - n + 1
        if 0 <= j < n:
            c[k] += delta * (1.0 - h[j])
    # Second case: k from i0 to min(2n-2, i0+n-1)
    k_start2 = i0
    k_end2 = min(2 * n - 2, i0 + n - 1)
    for k in range(k_start2, k_end2 + 1):
        j = i0 - k + n - 1
        if 0 <= j < n:
            c[k] -= delta * h[j]
    return c

def exact_coord_descent(h, n, max_sweeps, start_time, budget_s, rng):
    """Coordinate descent with golden section search per coordinate."""
    best_c5 = compute_c5(h)
    c = compute_corr(h).copy()
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    
    for sweep in range(max_sweeps):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 30:
            break
        improved = False
        perm = rng.permutation(n)
        for i0 in perm:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 10:
                break
            old = h[i0]
            if old < 1e-12 or old > 1.0 - 1e-12:
                # Still allow moving from boundary
                pass
            # Compute update vector u
            u = np.zeros(2 * n - 1)
            k_start1 = max(0, n - 1 - i0)
            k_end1 = min(2 * n - 2, 2 * n - 2 - i0)
            for k in range(k_start1, k_end1 + 1):
                j = i0 + k - n + 1
                if 0 <= j < n:
                    u[k] += (1.0 - h[j])
            k_start2 = i0
            k_end2 = min(2 * n - 2, i0 + n - 1)
            for k in range(k_start2, k_end2 + 1):
                j = i0 - k + n - 1
                if 0 <= j < n:
                    u[k] -= h[j]
            
            # Golden section on d in [-old, 1-old]
            d_lo = -old
            d_hi = 1.0 - old
            if d_hi - d_lo < 1e-14:
                continue
            
            a = d_lo
            b = d_hi
            x1 = b - gr * (b - a)
            x2 = a + gr * (b - a)
            f1 = np.max(c + x1 * u)
            f2 = np.max(c + x2 * u)
            for _ in range(35):
                if f1 < f2:
                    b = x2
                    x2 = x1
                    f2 = f1
                    x1 = b - gr * (b - a)
                    f1 = np.max(c + x1 * u)
                else:
                    a = x1
                    x1 = x2
                    f1 = f2
                    x2 = a + gr * (b - a)
                    f2 = np.max(c + x2 * u)
            
            d_opt = (a + b) / 2.0
            new_val = old + d_opt
            new_val = np.clip(new_val, 0.0, 1.0)
            
            if abs(new_val - old) > 1e-9:
                delta = new_val - old
                c_new = c + delta * u
                new_max = np.max(c_new) * (2.0 / n)
                # Also check feasibility: after this change, sum changes by delta
                # We'll handle feasibility after the full sweep
                if new_max < best_c5 - 1e-12:
                    h[i0] = new_val
                    c = c_new
                    best_c5 = new_max
                    improved = True
        
        # After each sweep, enforce feasibility
        h = make_feasible(h, n)
        c = compute_corr(h)
        actual_c5 = np.max(c) * (2.0 / n)
        if actual_c5 < best_c5:
            best_c5 = actual_c5
        else:
            best_c5 = actual_c5
            
        if not improved and sweep > 2:
            break
    
    return h, best_c5

def binary_sa_adaptive(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    """Binary SA with adaptive temperature (reheating)."""
    b = (h >= 0.5).astype(float)
    target = n // 2
    # Ensure exactly target ones
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        idx = ones[rng.randint(len(ones))]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        idx = zeros[rng.randint(len(zeros))]
        b[idx] = 1
    
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    stuck_count = 0
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 60:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        if stuck_count > 2000:
            T = max(T, 0.005)  # Reheat
            stuck_count = 0
        
        # Random move: swap one 1 with one 0
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        
        if new_c5 <= curr_c5 or rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-15)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                best_b = b.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            b[i] = 1
            b[j] = 0
            stuck_count += 1
    
    return best_b, best_c5

def binary_sa_multiswap(h, n, max_iters, T0, T1, rng, start_time, budget_s):

            b[chosen_ones] = 1
            b[chosen_zeros] = 0
            stuck_count += 1
    
    return best_b, best_c5

def slsqp_smooth(h_init, n, T, maxiter, start_time, budget_s):
    """SLSQP with log-sum-exp smoothed max."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        # Clip to avoid overflow
        shifted = np.clip(shifted, -50.0/T, 0.0)
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx / T * T
    
    def obj_simple(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        val = mc + np.log(np.sum(np.exp(np.clip(T * shifted, -100, 0)))) / T
        return val * dx
    
    def con_sum(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_jac}]
    
    try:
        res = minimize(obj_simple, h_init.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return h_init

def to_binary_balanced(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b
