import numpy as np
from scipy.optimize import minimize
import time

# Pre-computed constants
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

# SLSQP with log-sum-exp
_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

# Fast subgradient with momentum
def subgradient_optimize(h, n, max_iters, start_time, time_limit):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.5
    step = 0.01
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        c5 = max_val * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        # Active lags
        tol = max(1e-4, max_val * 0.001)
        active_mask = corr >= max_val - tol
        active_indices = np.where(active_mask)[0]
        if len(active_indices) == 0:
            active_indices = np.array([np.argmax(corr)])
        # Compute subgradient
        grad = np.zeros(n)
        for l in active_indices:
            k = l - (n - 1)
            if k >= 0:
                grad[:n-k] += (1.0 - h[k:n])
                if k > 0:
                    grad[k:] -= h[:n-k]
            else:
                k_abs = -k
                grad[k_abs:] += (1.0 - h[:n-k_abs])
                grad[:n-k_abs] -= h[k_abs:]
        grad /= len(active_indices)
        # Momentum step
        v = beta * v - step * grad
        h_new = h + v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 <= c5:
            h = h_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.9
            v *= 0.5
        if step < 1e-12:
            break
    return best_h, best_c5

# Coordinate descent with adaptive steps
def coord_descent(h, n, max_rounds, start_time, time_limit):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(0, n, 1):  # every point
            if time.time() - start_time > time_limit:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            # Try a few candidate values
            for frac in [0.0, 0.25, 0.5, 0.75, 1.0]:
                # Interpolate between 0 and 1 based on current value
                cand = old + (frac - 0.5) * 0.02
                cand = np.clip(cand, 0.0, 1.0)
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                # Adjust volume: redistribute the change to other points
                delta = (n/2.0) - np.sum(trial)
                if abs(delta) > 1e-14:
                    # Distribute to a random subset
                    trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_val = cand
            if abs(best_val - old) > 1e-10:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    one_h = 1.0 - x
    corr = np.correlate(x, one_h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

    _SL_T = T

                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

        return h.copy()

                best_c5 = best_c
                improved = True
        if not improved:

def cd_fast_round(h, n, corr):
    """One round of fast coordinate descent. Returns improved h and corr."""
    dx = 2.0 / n
    improved = False
    for j in range(n):
        old = h[j]
        # Compute sensitivity a_k
        a = np.zeros(2*n-1)
        # j as first index: k from -j to n-1-j
        ks1 = np.arange(-j, n-j)
        a[ks1 + n-1] += (1.0 - h[j + ks1])
        # j as second index: k from j-n+1 to j
        ks2 = np.arange(j-n+1, j+1)
        a[ks2 + n-1] -= h[j - ks2]
        
        # Try candidate t values
        best_c = np.max(corr) * dx
        best_t = old
        for t in [0.0, 1.0, max(0,old-0.02), max(0,old-0.005), min(1,old+0.005), min(1,old+0.02)]:
            if abs(t - old) < 1e-10:
                continue
            delta = t - old
            new_max = np.max(corr + delta * a) * dx
            # Small correction for volume redistribution
            # (negligible, skip for speed)
            if new_max < best_c - 1e-14:
                best_c = new_max
                best_t = t
        if abs(best_t - old) > 1e-10:
            h[j] = best_t
            # Fix volume
            diff = n/2.0 - h.sum()
            h = h + diff/n
            h = np.clip(h, 0.0, 1.0)
            corr = np.correlate(h, 1.0-h, mode='full')
            improved = True
    return h, corr, improved

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = int(np.argmin(np.abs(h - 0.5)))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, n, T, maxiter=100, ftol=1e-14):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(res.x, n)
    except Exception:
        return h.copy()

def perturb_fourier(h, n, rng, n_modes=4, amp=0.03):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 6))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.3, 1.2)
    return make_feasible(h_new, n)

def perturb_block(h, n, rng):
    h_new = h.copy()
    for _ in range(rng.randint(2, 5)):
        length = rng.randint(n // 16, n // 4)
        start = rng.randint(0, max(1, n - length))
        if rng.random() < 0.5:
            h_new[start:start+length] = 1.0 - h_new[start:start+length]
        else:
            h_new[start:start+length] += rng.uniform(-0.08, 0.08)
    return make_feasible(h_new, n)

def perturb_blend(h, n, rng):
    s = rng.randint(1, max(2, n // 8))
    hs = np.roll(h, s)
    alpha = rng.uniform(0.3, 0.7)
    return make_feasible(alpha * h + (1-alpha) * hs, n)

def upsample(h, n_old, n_new):
    x_old = np.linspace(0, 2, n_old)
    x_new = np.linspace(0, 2, n_new)
    return make_feasible(np.interp(x_new, x_old, h), n_new)

def cd_fast(h, n, start_time, time_limit, max_rounds=200):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit - 5:
            break
        h_cand = h.copy()
        any_change = False
        for j in range(n):
            if time.time() - start_time > time_limit - 5:
                break
            old = h[j]
            a = np.zeros(2 * n - 1)
            ks1 = np.arange(-j, n - j)
            a[ks1 + n - 1] += (1.0 - h[j + ks1])
            ks2 = np.arange(j - n + 1, j + 1)
            a[ks2 + n - 1] -= h[j - ks2]
            cur_max = np.max(corr)
            best_t = old
            best_mc = cur_max
            for t in (0.0, 1.0, old - 0.03, old - 0.01, old - 0.003,
                      old + 0.003, old + 0.01, old + 0.03):
                t = min(1.0, max(0.0, t))
                if abs(t - old) < 1e-10:
                    continue
                delta = t - old
                mc = np.max(corr + delta * a)
                if mc < best_mc - 1e-14:
                    best_mc = mc
                    best_t = t
            if abs(best_t - old) > 1e-10:

        if not any_change:

        h_cand = make_feasible(h_cand, n)

        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_cand.copy()
        h = h_cand
    return best_h, best_c5

def multi_slsqp(h, n, start_time, time_limit):
    best_c5 = compute_c5(h)
    for T in [30.0, 100.0, 400.0, 1500.0, 6000.0, 20000.0]:
        if time.time() - start_time > time_limit - 15:
            break
        hr = slsqp_opt(h, n, T, 80, 1e-14)
        cr = compute_c5(hr)
        if cr < best_c5:
            best_c5 = cr
            h = hr
    return h, best_c5

def binary_sa(h, n, start_time, time_limit, max_iters=50000):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while b.sum() > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[int(np.random.randint(len(ones)))]] = 0
    while b.sum() < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[int(np.random.randint(len(zeros)))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit - 5:
            break
        T = 0.005 * (0.0001 / 0.005) ** (it / max_iters)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[int(np.random.randint(len(ones)))]
        j = zeros[int(np.random.randint(len(zeros)))]
        b[i] = 0
        b[j] = 1
        nc5 = compute_c5(b)
        if nc5 <= curr_c5 or (T > 1e-12 and np.random.random() < np.exp(-(nc5 - curr_c5) / T)):
            curr_c5 = nc5
            if nc5 < best_c5:
                best_c5 = nc5
                best_b = b.copy()

            b[j] = 0
