    # Fix any remaining tiny infeasibility
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def subgradient_optimize(h, n, n_iters=300, topk_frac=0.1, max_step=0.05):
    """Subgradient optimization with backtracking line search"""
    dx = 2.0 / n
    h_cur = h.copy()
    c_cur = compute_c5(h_cur)
    corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    
    for it in range(n_iters):
        n_corr = len(corr)
        k_active = max(10, int(topk_frac * n_corr))
        top_idx = np.argpartition(corr, -k_active)[-k_active:]
        
        # Compute subgradient
        grad = np.zeros(n)
        one_minus_h = 1.0 - h_cur
        offset = n - 1
        for m in top_idx:
            s = m - offset
            # Term 1: (1-h[j+s]) for valid j
            lo = max(0, -s)
            hi = min(n, n - s)
            if hi > lo:
                grad[lo:hi] += one_minus_h[lo+s:hi+s]
            # Term 2: -h[j-s] for valid j  
            lo2 = max(0, s)
            hi2 = min(n, n + s)
            if hi2 > lo2:
                grad[lo2:hi2] -= h_cur[lo2-s:hi2-s]
        
        # Adaptive step size with backtracking
        base_step = max_step / (1 + 0.05 * it)
        step = base_step
        improved = False
        for ls in range(5):
            h_try = make_feasible(h_cur - step * grad, n)
            c_try = compute_c5(h_try)
            if c_try < c_cur - 1e-15:
                h_cur = h_try
                c_cur = c_try
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try smaller perturbation
            break
        
        corr = np.correlate(h_cur, 1.0 - h_cur, mode='full')
    
    return h_cur, c_cur

def simulated_annealing_bin(h_bin, n, rng, n_iters=50000, t_start=0.5, t_end=0.01):
    """Simulated annealing for binary sequences"""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for it in range(n_iters):
        T = t_start * (t_end / t_start) ** (it / float(n_iters))
        i = rng.randint(len(ones))
        j = rng.randint(len(zeros))
        oi, oj = ones[i], zeros[j]
        h[oi] = 0.0
        h[oj] = 1.0
        c_new = compute_c5(h)
        delta = c_new - cur_c5
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur_c5 = c_new
            ones[i] = oj
            zeros[j] = oi
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
        if it % 10000 == 0 and it > 0:
            # Cooling is already handled by the formula
            pass
    
    return best_h, best_c5

def lbgfb_optimize(h0, n, T=1000, mu=500, max_iter=400):
    """L-BFGS-B with softmax objective"""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sm = mc + np.log(np.sum(np.exp(np.clip(T * (corr - mc), -80, 80)))) / T
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def pair_swap_with_neighborhood(h_bin, n, rng, max_rounds=200, n_tries=3000, neighbor_range=5):
    """Local search with neighbor-restricted swaps"""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            i = rng.randint(len(ones))
            oi = ones[i]
            # Try neighbors of oi first, then random
            candidates = []
            for d in range(1, neighbor_range+1):
                n1 = (oi + d) % n
                n2 = (oi - d) % n
                if h[n1] <= 0.5:
                    candidates.append(n1)
                if h[n2] <= 0.5:
                    candidates.append(n2)
            if candidates:
                oj = candidates[rng.randint(len(candidates))]
            else:
                j = rng.randint(len(zeros))
                oj = zeros[j]
            
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur_c5 - 1e-15:
                cur_c5 = c
                ones[i] = oj
                # Update zeros list
                zeros.remove(oi) if oi in zeros else None
                zeros.append(oi)
                if oj in zeros:
                    zeros.remove(oj)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur_c5
