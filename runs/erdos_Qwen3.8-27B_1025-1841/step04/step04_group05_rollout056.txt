def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    old_x_ext = np.concatenate([old_x - 2.0, old_x, old_x + 2.0])

    return make_feasible(h_new, new_n)

            break
        if s > target:
            lo = mid
        else:

    weights = np.zeros(len(top_k))
    for wi, t in enumerate(top_k):
        weights[wi] = max(0, corr[t] - max_corr + 1e-4)  # only active
    if np.sum(weights) < 1e-10:
        weights = np.ones(len(top_k))
    weights /= np.sum(weights)

            continue

    return np.sum(h) - _GLOBAL_N / 2.0

1. Sort z
2. For the sorted z, find the "knee" where the function transitions from "all above" to "all below"
3. This is equivalent to finding the τ such that exactly (n - n_above - n_below) elements are in (0,1)

Actually, there's a simpler approach: since clip(z-τ, 0, 1) = max(0, min(1, z-τ)):
- Elements with z-τ > 1 (i.e., z > τ+1): contribute 1
- Elements with z-τ < 0 (i.e., z < τ): contribute 0
- Elements with 0 ≤ z-τ ≤ 1 (i.e., τ ≤ z ≤ τ+1): contribute z-τ

So sum = n_above + sum_{middle}(z_i - τ) = n_above + sum_{middle}(z_i) - n_middle * τ

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_ext = np.concatenate([h, h, h])
    old_x_ext = np.concatenate([old_x - 2.0, old_x, old_x + 2.0])
    h_new = np.interp(new_x, old_x_ext, h_ext)
    return make_feasible(h_new, new_n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(z, n):
    target = n / 2.0
    lo = z.min() - 1.0
    hi = z.max()
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if abs(s - target) < 1e-13:
            break
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    return np.clip(z - tau, 0.0, 1.0)

def subgradient_optimize(h, n, max_time, rng, K=8, momentum=0.5):
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    alpha0 = 0.4
    alpha_min = 1e-5
    vel = np.zeros(n)
    stuck_count = 0
    
    while time.time() - start < max_time:
        corr = compute_corr(h)
        max_val = corr.max()
        
        # Top-K lags with weight by proximity to max
        idx_sorted = np.argsort(corr)[::-1][:K]
        weights = np.zeros(K)
        threshold = max_val - 0.05  # lags within 5% of max
        for wi, t in enumerate(idx_sorted):
            weights[wi] = max(0.0, corr[t] - threshold)
        wsum = weights.sum()
        if wsum < 1e-10:
            weights[:] = 1.0
            wsum = K
        else:
            weights /= wsum
        
        # Compute subgradient
        grad = np.zeros(n)
        n1 = n - 1
        for wi, t in enumerate(idx_sorted):
            w = weights[wi]
            if w < 1e-10:
                continue
            i_lo = max(0, t - n1)
            i_hi = min(n - 1, t)
            # First term: h[i]*(1-h[i-t+n1]), derivative w.r.t. h[j] where i=j
            j1 = np.arange(i_lo, i_hi + 1)
            s1 = j1 - t + n1
            v1 = (s1 >= 0) & (s1 < n)
            grad[j1[v1]] += w * (1.0 - h[s1[v1]])
            # Second term: -h[i-t+n1], derivative w.r.t. h[j] where i-t+n1=j
            j2_start = max(0, i_lo - t + n1)
            j2_end = min(n - 1, i_hi - t + n1)
            if j2_start <= j2_end:
                j2 = np.arange(j2_start, j2_end + 1)
                s2 = j2 + t - n1
                v2 = (s2 >= 0) & (s2 < n)
                grad[j2[v2]] -= w * h[s2[v2]]
        
        # Momentum
        vel = momentum * vel + (1.0 - momentum) * grad
        norm_vel = np.linalg.norm(vel)
        if norm_vel < 1e-15:
            break
        
        # Annealed step
        elapsed = time.time() - start
        frac = min(1.0, elapsed / max_time)
        alpha = alpha0 * (0.05 + 0.95 * (1.0 - frac)) + alpha_min
        alpha = min(alpha, 0.3 * n / (n + elapsed))
        
        z = h - alpha * vel
        h_new = project_feasible(z, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            stuck_count = 0
        else:
            h = h_new
            stuck_count += 1
        
        # Perturbation if stuck
        if stuck_count > 500:
            h = best_h + 0.03 * rng.randn(n)
            h = make_feasible(h, n)
            vel[:] = 0.0
            stuck_count = 0
    
    return best_h, best_c5

def coordinate_descent(h, n, max_time):
    start = time.time()
    best_c5 = compute_c5(h)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005]
    
    for rnd in range(20):
        if time.time() - start > max_time:
            break
        improved = False
        for j in range(n):
            if time.time() - start > max_time:
                break
            for delta in deltas:
                v = h[j] + delta
                if v < 0.0 or v > 1.0:
                    continue
                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial
                    improved = True
                    break
            else:
                for bv in (0.0, 1.0):
                    if abs(h[j] - bv) < 1e-9:
                        continue
                    trial = h.copy()
                    trial[j] = bv
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-12:
                        best_c5 = c
                        h = trial
                        improved = True
                        break
        if not improved:
            break
    return h, best_c5

_GLOBAL_N = 0
_GLOBAL_T = 0

def _soft_obj(h):
    n = _GLOBAL_N
    T = _GLOBAL_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(np.clip(T * shifted, -500, 500))))
    return (mc + log_sum / T) * dx

def _cons_eq(h):
    return np.sum(h) - _GLOBAL_N / 2.0

def _cons_eq_jac(h):
    return np.ones(_GLOBAL_N)

def slsqp_optimize(h, n, T, max_iter):
    global _GLOBAL_N, _GLOBAL_T
    _GLOBAL_N = n
    _GLOBAL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    res = minimize(_soft_obj, h, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)
