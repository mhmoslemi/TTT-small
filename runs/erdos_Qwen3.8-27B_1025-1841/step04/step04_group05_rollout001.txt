def slsqp_optimize(h_start, n, T, max_iter):
    dx = 2.0 / n
    
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        # Use top 50 for stability
        top_k = min(50, len(corr))
        sorted_idx = np.argsort(corr)[-top_k:]
        top = corr[sorted_idx] - mc
        lse = mc + np.log(np.sum(np.exp(T * top))) / T
        return lse * dx
    
    def con_f(hv):
        return np.sum(hv) - n / 2.0
    
    def con_j(hv):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n

                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})

    # SLSQP round 1 at n=1024
    if tleft() > 120:
        h_opt = slsqp_opt(h1, n1, 50.0, 250)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h1 = h_opt
    
    # CD round 1 at n=1024
    if tleft() > 80:
        h1, c5_1 = coord_descent(h1, n1, 40, 40, start_time, budget_s)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # SLSQP round 2 at n=1024 (higher T)
    if tleft() > 80:
        h_opt = slsqp_opt(h1, n1, 300.0, 150)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h1 = h_opt
    
    # CD round 2
    if tleft() > 60:
        h1, c5_1 = coord_descent(h1, n1, 25, 25, start_time, budget_s)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Phase 2: Upsample to n=2048
    n2 = 2048
    h2 = upsample_h(h1, n2)
    c5_2 = compute_c5(h2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    # CD at n=2048 (fewer rounds, but each coord is more impactful)
    if tleft() > 100:
        h2, c5_2 = coord_descent(h2, n2, 15, 15, start_time, budget_s)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # SLSQP at n=2048
    if tleft() > 90:
        h_opt = slsqp_opt(h2, n2, 200.0, 120)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h2 = h_opt
    
    # Final CD at n=2048
    if tleft() > 60:
        h2, c5_2 = coord_descent(h2, n2, 10, 10, start_time, budget_s)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Phase 3: Random restarts at n=1024
    for i in range(4):
        if tleft() < 50:
            break
        h_r = 0.7 * upsample_h(best_h, 1024) + 0.3 * rng.rand(1024)
        h_r = make_feasible(h_r, 1024)
        
        # Also try binary-ish restart
        if i % 2 == 1:
            h_bin = (rng.rand(1024) < 0.5).astype(float)
            h_bin = make_feasible(h_bin, 1024)
            h_r = 0.5 * upsample_h(best_h, 1024) + 0.5 * h_bin
            h_r = make_feasible(h_r, 1024)
        
        if tleft() > 40:
            h_opt = slsqp_opt(h_r, 1024, 80.0, 150)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h_r = h_opt
        
        if tleft() > 30:
            h_r, c5_r = coord_descent(h_r, 1024, 15, 15, start_time, budget_s)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()
    
    # Phase 4: Final polish - upsample best and one more SLSQP
    n_final = 2048
    h_f = upsample_h(best_h, n_final)
    c5_f = compute_c5(h_f)
    if c5_f < best_c5:
        best_c5 = c5_f
        best_h = h_f.copy()
    
    if tleft() > 50:
        h_opt = slsqp_opt(h_f, n_final, 500.0, 80)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    if tleft() > 30:
        h_f, c5_f = coord_descent(h_f, n_final, 8, 8, start_time, budget_s)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
