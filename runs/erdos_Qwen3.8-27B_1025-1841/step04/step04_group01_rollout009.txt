import numpy as np
from scipy.optimize import minimize
import time

_G_N = 960
_G_T = 100.0
_G_TOP_K = 50
_G_DX = 2.0 / 960
_G_T0 = 0.0
_G_BUDGET = 1300.0


def _rem():
    return _G_BUDGET - (time.time() - _G_T0)


def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)


def _feas(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = h < 1.0 if d > 0 else h > 0.0
        c = float(np.sum(m))
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    d = target - np.sum(h)
    if abs(d) > 1e-14:
        i = int(np.argmin(h)) if d > 0 else int(np.argmax(h))
        h[i] = np.clip(h[i] + d, 0.0, 1.0)
    return h


def _soft_og(h, T, dx, top_k):
    n = len(h)
    c = _corr(h)
    if top_k < len(c):
        ti = np.argsort(c)[-top_k:]
    else:
        ti = np.arange(len(c))
    cs = c[ti]
    mx = np.max(cs)
    e = np.exp(T * (cs - mx))
    sw = np.sum(e)
    w = e / sw
    val = (mx + np.log(sw) / T) * dx
    g = np.zeros(n)
    for idx, s in enumerate(ti):
        wi = w[idx]
        if wi < 1e-10:
            continue
        k = s - (n - 1)
        if k == 0:
            g += wi * (1.0 - 2.0 * h)
        elif k > 0:
            g[:n - k] += wi * (1.0 - h[k:])
            g[k:] -= wi * h[:n - k]
        else:
            ka = -k
            g[:n - ka] += wi * (1.0 - h[ka:])
            g[ka:] -= wi * h[:n - ka]
    return val, dx * g


def _obj(x):
    val, _ = _soft_og(x, _G_T, _G_DX, _G_TOP_K)
    return val + 1e4 * (np.sum(x) - _G_N / 2.0) ** 2


def _obj_g(x):
    val, g = _soft_og(x, _G_T, _G_DX, _G_TOP_K)
    return val, g + 2e4 * (np.sum(x) - _G_N / 2.0) * np.ones(_G_N)


def _cd(h, n, steps, deadline):
    best_c = _c5(h)
    for j in range(n):
        if time.time() > deadline:
            break
        ov = h[j]
        bv, bc = ov, best_c
        for v in [0.0, 1.0]:
            if abs(v - ov) < 1e-10:
                continue
            t = h.copy()
            t[j] = v
            t = _feas(t, n)
            c = _c5(t)
            if c < bc - 1e-15:
                bc, bv = c, v
        for d in steps:
            for sgn in [-1, 1]:
                v = ov + sgn * d
                if 0.0 <= v <= 1.0:
                    t = h.copy()
                    t[j] = v
                    t = _feas(t, n)
                    c = _c5(t)
                    if c < bc - 1e-15:
                        bc, bv = c, v
        if bv != ov:
            h[j] = bv
            h = _feas(h, n)
            best_c = bc
    return h, best_c


def _ups(h, nn):
    on = len(h)
    if nn == on:
        return _feas(h.copy(), nn)
    ox = np.linspace(0.0, 2.0, on)
    nx = np.linspace(0.0, 2.0, nn)
    return _feas(np.interp(nx, ox, h), nn)


def _lbfbs(h, T, top_k, maxiter):
    n = len(h)
    _G_N, _G_T, _G_TOP_K, _G_DX = n, float(T), int(top_k), 2.0 / n
    try:
        res = minimize(_obj, h, jac=_obj_g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-14, 'ftol': 1e-16})
        return _feas(res.x, n)
    except Exception:
        return h
