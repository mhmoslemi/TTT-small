The key new idea is the **exact 1D minimax coordinate descent**:

For each coordinate j, given all other coordinates fixed, find the optimal h[j]:
- The objective is max_m (c[m] + delta * d[m]) where delta = h[j] - h[j]_fixed
- d[m] is the derivative of c[m] with respect to h[j]
- This is a 1D problem: minimize max_m (a_m + delta * b_m) over delta in [-(h[j]), 1-h[j]]

The minimum of a max of linear functions in 1D can be found by:
- Binary search on the optimal value t
- For a given t, check if there exists delta such that a_m + delta * b_m ≤ t for all m
  This is: delta ≥ (t - a_m)/b_m for b_m > 0, and delta ≤ (t - a_m)/b_m for b_m < 0
  So we need: max_{b_m>0} (t-a_m)/b_m ≤ min_{b_m<0} (t-a_m)/b_m
  And also delta in [lower, upper]

Actually, this is a linear feasibility problem in 1D! We can solve it by:
1. Compute L(t) = max({(t-a_m)/b_m : b_m > 0} ∪ {-inf}) 
   and U(t) = min({(t-a_m)/b_m : b_m < 0} ∪ {+inf})
2. We need L(t) ≤ U(t) and L(t) ≤ upper and U(t) ≥ lower
3. Binary search on t to find the minimum feasible t

This gives the EXACT optimal value for h[j] (given all others fixed), without any trial steps!

def coord_optimal_1d(corr, j, n, h_lower=0.0, h_upper=1.0):
    """Find optimal h[j] given corr and all other h fixed.
    
    corr is the current correlation array (full mode, length 2n-1).
    d[m] = change in corr[m] per unit change in h[j].
    """
    # Compute d[m] = d(corr[m])/d(h[j])
    # corr[m] changes when h[j] changes.
    # From the correlation formula, h[j] appears in:
    # - As h[n+k] when n+k = j, i.e., k = j-n, contributing (1-h[n]) = (1-h[0...n-1])
    # - As (1-h[n]) when n = j, contributing -h[n+k]... 
    # 
    # Actually, let me compute d[m] directly:
    # corr[m] = sum_n h[n+k] * (1-h[n]) where k = m-(n-1)
    # d(corr[m])/d(h[j]):
    #   Term 1: h[n+k]=h[j] when n+k=j, i.e., n=j-k=j-m+(n-1)
    #           Contribution: (1-h[j-k]) = (1-h[j-m+(n-1)]) if valid
    #   Term 2: (1-h[n]) when n=j, contribution is -h[j+k] = -h[j+m-(n-1)] if valid
    #
    # So d[m] = (1-h[j-m+(n-1)]) * [0<=j-m+(n-1)<n] - h[j+m-(n-1)] * [0<=j+m-(n-1)<n]
    
    d = np.zeros(2*n-1)
    # Term 1: n1 = j - m + (n-1), valid when 0 <= j-m+(n-1) <= n-1
    # i.e., j-(n-1) <= m <= j
    m1_lo = max(0, j - (n-1))
    m1_hi = min(2*n-2, j)
    n1 = j - np.arange(m1_lo, m1_hi+1) + (n-1)
    d[m1_lo:m1_hi+1] += (1.0 - h[n1])
    
    # Term 2: n2 = j + m - (n-1), valid when 0 <= j+m-(n-1) <= n-1
    # i.e., (n-1)-j <= m <= n-1+(n-1)-j = 2(n-1)-j
    m2_lo = max(0, (n-1) - j)
    m2_hi = min(2*n-2, 2*(n-1) - j)
    n2 = j + np.arange(m2_lo, m2_hi+1) - (n-1)
    d[m2_lo:m2_hi+1] -= h[n2]
    
    # Now solve: min_delta max_m (corr[m] + delta * d[m])
    # subject to h_lower <= h[j] + delta <= h_upper
    # i.e., delta in [h_lower - h[j], h_upper - h[j]]
    
    delta_lo = h_lower - h[j]
    delta_hi = h_upper - h[j]
    
    if delta_lo > delta_hi:
        return h[j]  # infeasible, shouldn't happen
    
    # Binary search on t
    # Check feasibility of t: exists delta in [delta_lo, delta_hi] such that
    # corr[m] + delta * d[m] <= t for all m
    # For d[m] > 0: delta <= (t - corr[m]) / d[m]
    # For d[m] < 0: delta >= (t - corr[m]) / d[m]
    # For d[m] = 0: need corr[m] <= t
    
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    zero_mask = ~pos_mask & ~neg_mask
    
    def feasible(t):
        if np.max(corr[zero_mask]) > t:
            return False
        if np.any(pos_mask):
            upper = np.min((t - corr[pos_mask]) / d[pos_mask])
        else:
            upper = delta_hi
        if np.any(neg_mask):
            lower = np.max((t - corr[neg_mask]) / d[neg_mask])
        else:
            lower = delta_lo
        return lower <= upper and lower <= delta_hi and upper >= delta_lo
    
    # Find min t
    lo = np.max(corr)  # can't do better than current max
    hi = np.max(corr + delta_hi * d) if delta_hi >= 0 else np.max(corr + delta_lo * d)
    if hi < lo:
        hi = lo
    
    for _ in range(50):
        mid = (lo + hi) / 2
        if feasible(mid):
            hi = mid
        else:
            lo = mid
    
    t_opt = hi
    # Find optimal delta
    if np.any(pos_mask):
        upper = np.min((t_opt - corr[pos_mask]) / d[pos_mask])
    else:
        upper = delta_hi
    if np.any(neg_mask):
        lower = np.max((t_opt - corr[neg_mask]) / d[neg_mask])
    else:
        lower = delta_lo
    
    # Pick delta in [max(lower, delta_lo), min(upper, delta_hi)]
    d_lo = max(lower, delta_lo)
    d_hi = min(upper, delta_hi)
    if d_lo > d_hi:
        # Infeasible at t_opt (numerical), try slightly larger t
        delta = (delta_lo + delta_hi) / 2
    else:
        delta = (d_lo + d_hi) / 2
    
    return h[j] + delta

def alternating_lp(h, n, max_iters=50, tol=1e-10):
    """Alternating LP: alternate between fixing v=1-h and solving LP for h."""
    v = 1.0 - h
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        # Build LP: min t s.t. sum_i h[i]*v[i+k] <= t, sum(h)=n/2, 0<=h<=1
        # Variables: h[0], ..., h[n-1], t
        # Inequality constraints: for each k, sum_i A[k,i] * h[i] - t <= 0
        # where A[k,i] = v[i+k] (with appropriate indexing for full correlation)
        
        # The full correlation at output index m (corresponding to lag k = m-(n-1)):
        # c[m] = sum_n h[n+k] * v[n] where k = m-(n-1)
        # = sum_n h[n+m-(n-1)] * v[n]
        
        # So A[m,n] = v[n] (for the h side) and the coefficient on h[n+m-(n-1)]
        # Actually: c[m] = sum over valid n of h[n + m - (n-1)] * v[n]
        # Let me reindex: let p = n + m - (n-1) = n + k where k = m-(n-1)
        # So c[m] = sum_p h[p] * v[p - k] = sum_p h[p] * v[p - m + (n-1)]
        # A[m, p] = v[p - m + (n-1)] if 0 <= p-m+(n-1) < n, else 0
        
        # This is a (2n-1) x n matrix. For n=960, that's 1919 x 960 ≈ 1.8M entries.
        # Most are zero (each row has at most n non-zeros, but actually each row has
        # min(n, m+1, 2n-1-m) non-zeros).
        
        # Using scipy.sparse:
        from scipy.sparse import csr_matrix
        import scipy.optimize as opt
        
        # Build A matrix (2n-1) x (n+1) [last column is for t]
        # For row m: sum_p A[m,p] * h[p] - t <= 0
        # A[m,p] = v[p - m + n - 1] if valid, else 0
        # A[m, n] = -1 (for the t variable)
        
        rows = []
        cols = []
        vals = []
        for m in range(2*n-1):
            # h coefficients
            p_lo = max(0, m - (n-1))
            p_hi = min(n-1, 2*(n-1) - m)
            for p in range(p_lo, p_hi+1):
                v_idx = p - m + (n-1)
                if 0 <= v_idx < n:
                    rows.append(m)
                    cols.append(p)
                    vals.append(v[v_idx])
            # t coefficient
            rows.append(m)
            cols.append(n)
            vals.append(-1.0)
        
        A_ub = csr_matrix((vals, (rows, cols)), shape=(2*n-1, n+1))
        b_ub = np.zeros(2*n-1)
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n+1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n/2.0])
        
        # Objective: minimize t (last variable)
        c_obj = np.zeros(n+1)
        c_obj[n] = 1.0
        
        # Bounds: 0 <= h<=1, t unbounded (or t >= 0)
        bounds = [(0,1)]*n + [(0, None)]
        
        res = opt.linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        
        if res.success:
            h_new = res.x[:n]
            h_new = np.clip(h_new, 0, 1)
            h_new = make_feasible(h_new, n)
            new_c5 = compute_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                h = h_new
                v = 1.0 - h
            # Check convergence
            if new_c5 < best_c5 + tol:  # always true since we only update on improvement
                pass
        else:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize, linprog
from scipy.sparse import csr_matrix
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def compute_d_j(h, j, N):
    d = np.zeros(2 * N - 1)
    m_lo = max(0, j - (N - 1))
    m_hi = min(2 * N - 2, j)
    n_idx = j - np.arange(m_lo, m_hi + 1) + (N - 1)
    d[m_lo:m_hi + 1] += (1.0 - h[n_idx])
    m_lo2 = max(0, (N - 1) - j)
    m_hi2 = min(2 * N - 2, 2 * (N - 1) - j)
    n_idx2 = j + np.arange(m_lo2, m_hi2 + 1) - (N - 1)
    d[m_lo2:m_hi2 + 1] -= h[n_idx2]
    return d

def solve_1d_minimax(a, b, d_lo, d_hi):
    if d_hi - d_lo < 1e-15:
        return (d_lo + d_hi) / 2
    pos = b > 1e-14
    neg = b < -1e-14
    zero = ~pos & ~neg
    zero_max = np.max(a[zero]) if np.any(zero) else -np.inf

    def check(t):
        if zero_max > t + 1e-13:
            return None
        upper = np.min((t - a[pos]) / b[pos]) if np.any(pos) else d_hi
        lower = np.max((t - a[neg]) / b[neg]) if np.any(neg) else d_lo
        lo_b = max(lower, d_lo)
        hi_b = min(upper, d_hi)
        if lo_b <= hi_b:
            return (lo_b + hi_b) / 2
        return None

    t_lo = max(np.max(np.where(b > 0, a + d_lo * b, a + d_hi * b)), zero_max)
    t_hi = min(np.max(a + d_lo * b), np.max(a + d_hi * b))
    if t_hi < t_lo:
        t_hi = t_lo + 1e-10
    best_delta = (d_lo + d_hi) / 2
    for _ in range(50):
        t_mid = (t_lo + t_hi) / 2.0
        res = check(t_mid)
        if res is not None:
            t_hi = t_mid
            best_delta = res
        else:
            t_lo = t_mid
    return best_delta

def binary_local_search(h_bin, N, max_iters, rng, time_limit, start_t):

    corr = get_corr(h_bin)
    current_max = np.max(corr)
    best_c5 = current_max * dx
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]

def alternating_lp(h, N, max_iters, time_limit, start_t):
    v = 1.0 - h
    dx = 2.0 / N
    best_c5 = compute_c5(h)
    row_idx = []
    col_idx = []
    is_t = []
    for m in range(2 * N - 1):
        k = m - (N - 1)
        p_lo = max(0, k)
        p_hi = min(N - 1, N - 1 + k)
        for p in range(p_lo, p_hi + 1):
            row_idx.append(m)
            col_idx.append(p)
            is_t.append(0)
        row_idx.append(m)
        col_idx.append(N)
        is_t.append(1)
    row_idx = np.array(row_idx, dtype=np.int32)
    col_idx = np.array(col_idx, dtype=np.int32)
    is_t = np.array(is_t, dtype=np.int32)
    h_mask = is_t == 0
    t_mask = is_t == 1
    v_indices = col_idx[h_mask] - (row_idx[h_mask] - (N - 1))
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([float(N / 2)])
    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0
    bounds = [(0, 1)] * N + [(0, None)]
    b_ub = np.zeros(2 * N - 1)
    for it in range(max_iters):
        if time.time() - start_t > time_limit:
            break
        vals = np.empty(len(row_idx))
        vals[h_mask] = v[v_indices]
        vals[t_mask] = -1.0
        A_ub = csr_matrix((vals, (row_idx, col_idx)), shape=(2 * N - 1, N + 1))
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        except Exception:
            break
        if res.success:

            new_c5 = compute_c5(h_new)

                best_c5 = new_c5
                h = h_new.copy()
                v = 1.0 - h
            else:
                break
        else:
            break
    return h, best_c5

def cd_1d_minimax(h, N, max_passes, time_limit, start_t):
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        if time.time() - start_t > time_limit:
            break
        corr = get_corr(h)
        improved = False
        order = np.random.RandomState(p * 7 + 3).permutation(N)
        for j in order:
            if time.time() - start_t > time_limit:
                break
            d = compute_d_j(h, j, N)
            d_lo = -h[j]
            d_hi = 1.0 - h[j]
            if d_hi - d_lo < 1e-14:
                continue
            opt_delta = solve_1d_minimax(corr, d, d_lo, d_hi)
            if abs(opt_delta) > 1e-10:
                h[j] += opt_delta
                h[j] = np.clip(h[j], 0.0, 1.0)
                corr = corr + opt_delta * d
                improved = True
        if improved:
            h = make_feasible(h, N)
            new_c5 = compute_c5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
            else:
                break
    return h, best_c5

def pair_1d_minimax(h, N, max_passes, n_pairs_per_pass, time_limit, start_t, rng):

    for p in range(max_passes):

def slsqp_polish(h, N, T, max_iter, time_limit, start_t):
    dx = 2.0 / N
    bounds = [(0.0, 1.0)] * N
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - N / 2.0,
             'jac': lambda x: np.ones(N)}]

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': max_iter, 'ftol': 1e-16})
        if res.success or res.fun < obj(h):
            h_new = make_feasible(res.x, N)
            c5_new = compute_c5(h_new)
            return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)
