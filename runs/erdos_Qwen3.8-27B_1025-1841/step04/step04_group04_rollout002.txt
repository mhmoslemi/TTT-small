def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_constraint(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Bisection on Lagrange multiplier
    lo, hi = -2.0, 2.0
    for _ in range(60):

        h_proj = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h + (lo + hi) / 2, 0.0, 1.0)

        else:

import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 480
_OPT_T = 50.0

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_polytope(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hp) > target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(hp)
    if diff > 1e-12:
        mask = hp < 1.0
    elif diff < -1e-12:
        mask = hp > 0.0
    else:
        return hp
    if np.sum(mask) > 0:
        hp[mask] += diff / np.sum(mask)
        hp = np.clip(hp, 0.0, 1.0)
    return hp

def subgrad_max_corr(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s = k_star - (n - 1)
    g = np.zeros(n)
    if s >= 0:
        idx1 = np.arange(0, n - s)
        idx2 = idx1 + s
        g[idx1] += (1.0 - h[idx2])
        g[idx2] += -h[idx1]
    else:
        sp = -s
        idx1 = np.arange(sp, n)
        idx2 = idx1 - sp
        g[idx2] += (1.0 - h[idx1])
        g[idx1] += -h[idx2]
    return g

def pgd_optimize(h, n, n_iter=3000, lr_init=0.5, time_limit=180):
    global _OPT_N
    t0 = time.time()
    _OPT_N = n
    dx = 2.0 / n
    best_h = h.copy()
    best_c = compute_c5(h)
    h_cur = h.copy()
    lr = lr_init
    no_improve = 0
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        g = subgrad_max_corr(h_cur)
        improved = False
        for _ in range(15):
            h_try = project_onto_polytope(h_cur - lr * g * dx, n)
            c_try = compute_c5(h_try)
            if c_try < best_c - 1e-15:
                best_h = h_try.copy()
                best_c = c_try
                h_cur = h_try.copy()
                lr = min(lr * 1.2, 5.0)
                improved = True
                no_improve = 0
                break
            lr *= 0.5
        if not improved:
            no_improve += 1
            lr = lr_init * 0.5
            h_cur = best_h + np.random.randn(n) * 0.005
            h_cur = make_feasible(h_cur, n)
            if no_improve > 20:
                break
    return best_h, best_c

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_snap(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        for val in (0.0, 1.0):
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def coord_descent(h, n, rounds=30, time_limit=180):
    t0 = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    for r in range(rounds):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = best_h[j]
            best_trial = old
            best_c_local = best_c
            cands = [0.0, 1.0, 0.5]
            for d in (-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1):
                cands.append(max(0.0, min(1.0, old + d)))
            for cand in cands:
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_trial = cand
            if best_trial != old:
                best_h = best_h.copy()
                best_h[j] = best_trial
                best_h = make_feasible(best_h, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return best_h, best_c

def slsqp_step(h, n, T, maxiter, ftol):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_obj, h, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(res.x, n)
