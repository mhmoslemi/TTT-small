def softmax_objective_and_grad(h, n, T, lambda_pen):
    """Objective and gradient for softmax approximation + penalty."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx  # c[k] is the integral at each lag
    
    # Softmax approximation of max
    c_max = np.max(c)
    shifted = T * (c - c_max)
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    p = exp_shifted / sum_exp  # softmax weights
    
    # Objective: c_max + (1/T) * log(sum_exp) + penalty
    obj = c_max + np.log(sum_exp) / T
    # Add penalty for sum constraint
    sum_h = np.sum(h)
    target = n / 2.0
    penalty = lambda_pen * (sum_h - target) ** 2
    obj += penalty
    
    # Gradient
    # d/dh of c_k for each k:
    # c_k = dx * corr[k] where corr is from np.correlate(h, 1-h, 'full')
    # The gradient of corr[k] with respect to h is computed as before
    # But we need the weighted sum: sum_k p_k * grad_c_k
    
    # Compute gradient as weighted sum of individual gradients
    grad = np.zeros(n)
    
    # For efficiency, only compute gradients for lags with significant weight
    threshold = 1e-6
    active = p > threshold
    active_indices = np.where(active)[0]
    
    for idx in active_indices:
        k = idx - (n - 1)  # actual lag
        w = p[idx]
        g_k = np.zeros(n)
        if k == 0:
            g_k = dx * (1.0 - 2.0 * h)
        elif k > 0:
            g_k[k:] += dx * (1.0 - h[:n-k])
            g_k[:n-k] += -dx * h[k:]
        else:
            p2 = -k
            g_k[:n-p2] += dx * (1.0 - h[p2:])
            g_k[p2:] += -dx * h[:n-p2]
        grad += w * g_k
    
    # Add penalty gradient
    grad += 2 * lambda_pen * (sum_h - target)
    
    return obj, grad
