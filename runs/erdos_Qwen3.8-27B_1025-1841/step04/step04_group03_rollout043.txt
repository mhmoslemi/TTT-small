def make_feasible(h, n):

        else:

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    norm = np.sum(exps)

    # Gradient
    corr_hw = np.correlate(h, weights, 'full')
    conv_hw = np.convolve(h, weights, 'full')

    conv_B = conv_hw[n-1:2*n-1]
    grad = dx * (1.0 - conv_A - conv_B)
    
    return f, grad

def targeted_swap(h, n, max_iters, start_time, budget_s):
    """Targeted swaps that specifically reduce the max correlation."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Find the maximizing index
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        
        # Shift = max_idx - (n-1)
        shift = max_idx - (n-1)
        
        # Find positions i where h[i]=1 and h[i+shift]=0 (valid)
        # These are the "contributing" positions for this shift
        ones = np.where(h == 1)[0]
        # For each one at position i, check if i+shift is in [0,n) and h[i+shift]=0
        valid_i = []
        for i in ones:
            j = i + shift
            if 0 <= j < n and h[j] == 0:
                valid_i.append(i)
        
        if len(valid_i) > 0:

            i = valid_i[rng.randint(len(valid_i))]
            j = i + shift
            # Swap
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = np.max(new_corr) * (2.0/n)
            
            if new_c5 <= best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                corr = new_corr
            else:
                # Revert
                h[i] = 1
                h[j] = 0
                # Accept with probability based on "temperature"
                # (simple: just reject)
        else:
            # No directed swap possible for this shift, try random swap
            ones = np.where(h == 1)[0]

    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: L-BFGS-B sweep
    temps = [200.0, 100.0, 50.0, 20.0, 10.0, 5.0, 2.0, 1.0]
    for T in temps:
        if time.time() > deadline - 500:
            break
        try:
            x = lbfgs_run(best_h, n0, T, 300, start_time, deadline)
            x = make_feasible(x, n0)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x.copy()
        except Exception:
            pass

    # Phase 2: Targeted binary SA
    if time.time() < deadline - 300:
        dl = min(deadline - 300, start_time + 250)
        b, c5b = targeted_sa(best_h, n0, 200000, 0.004, 0.000005, start_time, dl, rng)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()

    # Phase 3: L-BFGS-B refinement after SA
    for T in [10.0, 5.0, 2.0, 1.0]:
        if time.time() > deadline - 250:
            break
        try:
            x = lbfgs_run(best_h, n0, T, 250, start_time, deadline)
            x = make_feasible(x, n0)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x.copy()
        except Exception:
            pass

    # Phase 4: Block swap SA
    if time.time() < deadline - 150:
        dl = min(deadline - 150, start_time + 150)
        b, c5b = block_swap_sa(best_h, n0, 80000, 0.003, 0.00001, start_time, dl, rng)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()
            if time.time() < deadline - 80:
                try:
                    x = lbfgs_run(best_h, n0, 5.0, 200, start_time, deadline)
                    x = make_feasible(x, n0)
                    c = compute_c5(x)
                    if c < best_c5:
                        best_c5 = c
                        best_h = x.copy()
                except Exception:
                    pass

    # Phase 5: Coordinate descent
    if time.time() < deadline - 50:
        best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 15, start_time, deadline)

    # Phase 6: Higher resolution
    if time.time() < deadline - 200:
        n1 = n0 * 2
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        for T in [50.0, 10.0, 3.0]:
            if time.time() > deadline - 80:
                break
            try:
                x = lbfgs_run(h_up, n1, T, 200, start_time, deadline)
                x = make_feasible(x, n1)
                c = compute_c5(x)
                if c < compute_c5(h_up):
                    h_up = x.copy()
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5 = c_up
            best_h = h_up.copy()
        if time.time() < deadline - 30:
            best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 5, start_time, deadline)

    # Phase 7: Final perturb + refine
    if time.time() < deadline - 60:
        hp = fourier_perturb(best_h, len(best_h), rng, n_modes=3, amp=0.03)
        try:
            x = lbfgs_run(hp, len(best_h), 5.0, 200, start_time, deadline)
            x = make_feasible(x, len(best_h))
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x.copy()
        except Exception:
            pass
        if time.time() < deadline - 10:
            best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 5, start_time, deadline)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
