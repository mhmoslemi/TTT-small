```python
    m_lo2 = max(0, d)
    m_hi2 = min(n-1, n-1+d)
    if m_lo2 <= m_hi2:
        grad[m_lo2:m_hi2+1] -= h[m_lo2-d:m_hi2-d+1] * dx
    
    return grad
```

```python
    dx = 2.0 / n
    x = cp.Variable(n)
    t = cp.Variable()
    constraints = [x >= 0, x <= 1, cp.sum(x) * dx == 1]
    
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            m_lo, m_hi = 0, n - 1 - d
        else:
            m_lo, m_hi = -d, n - 1
        m_idx = np.arange(m_lo, m_hi + 1)
        coeff = (1.0 - h_lin[m_idx + d]) * dx
        constraints.append(cp.sum(x[m_idx] * coeff) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=10000)
        if x.value is not None:
            return np.array(x.value)
    except Exception:
        pass
    return h_lin.copy()
```

```python
import time

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

        else:

        cnt = np.sum(mask)
        if cnt == 0:

    for _ in range(200):

    dx = 2.0 / n
    grad = np.zeros(n)

    if m_lo > m_hi:
        return grad
    m_idx = np.arange(m_lo, m_hi + 1)
    grad[m_idx] += (1.0 - h[m_idx + d]) * dx
    grad[m_idx + d] -= h[m_idx] * dx
    return grad

def subgradient_opt(h, n, max_iter, time_limit, start_time, use_stochastic=True):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.02
    dx = 2.0 / n
    
    for t in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_vals = corr * dx
        k_max = np.argmax(c_vals)

            thresh = c_vals[k_max] - 0.002

        else:
            k = k_max
        
        g = grad_corr_k(h, n, k)
        h_new = project_simplex(h - lr * g, n)
        
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        
        h = h_new
        lr = max(0.002, lr * 0.9995)
    
    return best_h, best_c5

def dca_iteration(h, n, alpha_list=(1.0, 0.7, 0.5, 0.3, 0.1)):
    import cvxpy as cp
    dx = 2.0 / n
    h_lin = h.copy()
    
    x = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [x >= 0, x <= 1, cp.sum(x) * dx == 1]
    
    for k in range(2 * n - 1):
        d = k - (n - 1)
        if d >= 0:
            m_lo, m_hi = 0, n - 1 - d
        else:
            m_lo, m_hi = -d, n - 1
        if m_lo > m_hi:
            continue
        m_idx = np.arange(m_lo, m_hi + 1)
        coeff = (1.0 - h_lin[m_idx + d]) * dx
        constraints.append(cp.sum(cp.multiply(x[m_idx], coeff)) <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=cp.ECOS, max_iters=20000, abstol=1e-8, reltol=1e-8)
    except Exception:
        try:
            prob.solve(solver=cp.SCS)
        except Exception:
            return h, compute_c5(h)
    
    if x.value is None:
        return h, compute_c5(h)
    
    x_val = np.array(x.value).flatten()
    c5_lin = compute_c5(h)
    best_h = h
    best_c5 = c5_lin
    
    for alpha in alpha_list:
        h_try = (1.0 - alpha) * h + alpha * x_val
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
    
    return best_h, best_c5

def coord_desc(h, n, max_rounds, start_time, time_limit):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 2:
                break
            old = h[j]
            candidates = [0.0, 1.0, old - 0.02, old - 0.005, old + 0.005, old + 0.02]
            best_c = best_c5
            best_v = old
            for cv in candidates:
                if cv < -0.01 or cv > 1.01:
                    continue
                if abs(cv - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = cv
            if best_v != old:
                h = make_feasible(h.copy(), n)
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def swap_sa(h, n, max_iters, T0, T1, start_time, time_limit, rng):
    b = (h >= 0.5).astype(float)
    target = n // 2
    # Adjust to exact target
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
```
