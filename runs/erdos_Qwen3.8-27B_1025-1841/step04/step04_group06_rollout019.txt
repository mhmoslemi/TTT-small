```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)
```

  When h[i]: 1→0, h[j]: 0→1:

  - Terms involving h[i] in the second factor: l=i: contribution h[i+k]*(1-h[i]) = h[i+k]*0 = 0 → h[i+k]*1

  - Terms involving h[j] in the first factor: l+k=j, so l=j-k: contribution h[j]*(1-h[j-k]) = 0 → 1*(1-h[j-k])
    Change: +(1-h[j-k]) if j-k is in [0,n-1]
  - Terms involving h[j] in the second factor: l=j: contribution h[j+k]*(1-h[j]) = h[j+k]*1 → h[j+k]*0
    Change: -h[j+k] if j+k is in [0,n-1]
