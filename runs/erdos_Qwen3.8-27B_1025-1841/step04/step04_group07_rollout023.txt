def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    # Final adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n_ones]] = 1.0

    return best_h, float(best_c5 * 2.0 / n)

def active_set_refine(h0, n, max_rounds=500, topk_frac=0.08, lr=0.002):
    dx = 2.0 / n
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    target = n / 2.0
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_corr = len(corr)
        k = max(3, int(topk_frac * n_corr))
        top_idx = np.argpartition(corr, -k)[-k:]
        active_corr = corr[top_idx]
        alpha = np.exp(50.0 * active_corr)
        alpha /= alpha.sum()
        grad = np.zeros(n)
        for w, idx in zip(alpha, top_idx):
            s = int(idx) - (n - 1)
            g = np.ones(n)
            if s >= 0:
                for j in range(n - s):
                    g[j] -= h[j + s]
                for j in range(s, n):
                    g[j] -= h[j - s]
            else:
                t = -s
                for j in range(t, n):
                    g[j] -= h[j + s]
                for j in range(n - t):
                    g[j] -= h[j - s]
            grad += w * g
        grad -= grad.mean()
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5 + 1e-12:
            h = h_new
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
                lr = min(lr * 1.2, 0.05)
            else:
                lr *= 0.8
        else:
            lr *= 0.7
            if lr < 1e-5:
                break
    return best_h, best_c5

def hill_climb_binary(h_init, n, rng, n_iter=200000):
    h = h_init.copy()
    cur = compute_c5(h)
    ones = np.where(h == 1)[0].tolist()
    zeros = np.where(h == 0)[0].tolist()
    for it in range(n_iter):
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a, b = ones[ia], zeros[ib]
        h[a] = 0.0
        h[b] = 1.0
        c_new = compute_c5(h)
        if c_new < cur - 1e-14:
            cur = c_new
            ones[ia] = b
            zeros[ib] = a
        else:
            h[a] = 1.0
            h[b] = 0.0
    return h, cur

def l2_smooth_minimize(h0, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -50, 50)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def multi_lag_gradient(h0, n, max_rounds=300, n_active=10, lr_init=0.003):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argpartition(corr, -n_active)[-n_active:]
        active_vals = corr[top_k]
        weights = np.exp(100.0 * (active_vals - active_vals.max()))
        weights += 1e-10
        weights /= weights.sum()
        grad = np.zeros(n)
        for w, idx in zip(weights, top_k):
            s = int(idx) - (n - 1)
            g = np.ones(n)
            if s >= 0:
                for j in range(n - s):
                    g[j] -= h[j + s]
                for j in range(s, n):
                    g[j] -= h[j - s]
            else:
                t = -s
                for j in range(t, n):
                    g[j] -= h[j + s]
                for j in range(n - t):
                    g[j] -= h[j - s]
            grad += w * g
        grad -= grad.mean()
        h_new = make_feasible(h - lr * grad, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.05, 0.03)
        else:
            lr *= 0.85
            if lr < 1e-6:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = len(h0)

    # Phase 1: Active-set refinement at parent resolution
    if rem() > 100:
        h_ast, c_ast = active_set_refine(h0, best_n, max_rounds=300, topk_frac=0.06, lr=0.002)
        if c_ast < best_c5:
            best_c5, best_h = c_ast, h_ast.copy()

    # Phase 2: Binary optimization at parent resolution
    if rem() > 200:
        n_b = best_n
        h_bin = to_binary(best_h, n_b)
        h_sa, c_sa = binary_sa(h_bin, n_b, rng, n_iter=500000, T0=0.6, T1=0.01)
        if c_sa < best_c5:
            best_c5, best_h = c_sa, h_sa.copy()
            best_n = n_b
        # Hill climb on result
        h_hc, c_hc = hill_climb_binary(h_sa, n_b, rng, n_iter=300000)
        if c_hc < best_c5:
            best_c5, best_h = c_hc, h_hc.copy()

    # Phase 3: Try higher resolutions
    for n_try in [800, 1024, 1280]:
        if rem() < 150:
            break
        h_up = upsample_h(best_h, n_try)
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up.copy()
            best_n = n_try

        # Active set refinement at new resolution
        if rem() > 120:
            h_ast, c_ast = active_set_refine(h_up, n_try, max_rounds=250, topk_frac=0.05, lr=0.0015)
            if c_ast < best_c5:
                best_c5, best_h = c_ast, h_ast.copy()
                h_up = h_ast.copy()

        # Binary SA at new resolution
        if rem() > 100:
            h_bin2 = to_binary(h_up, n_try)
            h_sa2, c_sa2 = binary_sa(h_bin2, n_try, rng, n_iter=400000, T0=0.5, T1=0.008)
            if c_sa2 < best_c5:
                best_c5, best_h = c_sa2, h_sa2.copy()
                best_n = n_try

        # L-BFGS smooth at new resolution
        if rem() > 60:
            for T in [200, 800]:
                if rem() < 30:
                    break
                h_opt = l2_smooth_minimize(h_up, n_try, T, mu=500.0, max_iter=150)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()
                    h_up = h_opt.copy()

    # Phase 4: Multi-lag gradient at best resolution
    if rem() > 80:
        n_ml = min(best_n, 1024)
        h_ml = upsample_h(best_h, n_ml) if n_ml < best_n else best_h.copy()
        h_opt, c_opt = multi_lag_gradient(h_ml, n_ml, max_rounds=400, n_active=15, lr_init=0.002)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()
            best_n = n_ml

    # Phase 5: Final binary polish at best resolution
    if rem() > 60:
        n_fin = best_n
        h_bin_f = to_binary(best_h, n_fin)
        # Quick SA
        h_res, c_res = binary_sa(h_bin_f, n_fin, rng, n_iter=300000, T0=0.3, T1=0.002, t_start=0.5)
        if c_res < best_c5:
            best_c5, best_h = c_res, h_res.copy()
        # Hill climb
        h_hc2, c_hc2 = hill_climb_binary(h_res, n_fin, rng, n_iter=200000)
        if c_hc2 < best_c5:
            best_c5, best_h = c_hc2, h_hc2.copy()

    # Phase 6: Final continuous refinement
    if rem() > 30:
        h_final, c_final = active_set_refine(best_h, best_n, max_rounds=200, topk_frac=0.04, lr=0.001)
        if c_final < best_c5:
            best_c5, best_h = c_final, h_final.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
