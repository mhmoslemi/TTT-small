_G_N = 0
_G_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _soft_obj(x):
    dx = 2.0 / _G_N
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    s = corr - mc
    return (mc + np.log(np.sum(np.exp(_G_T * s))) / _G_T) * dx

def _eq_con(x):
    return np.sum(x) - x.size / 2.0

def _eq_jac(x):
    return np.ones(x.size)

def slsqp_opt(h, n, T, maxiter=200, ftol=1e-14):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bnds = [(0.0, 1.0)] * n
    cns = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}]
    r = minimize(_soft_obj, h.copy(), method='SLSQP',
                 bounds=bnds, constraints=cns,
                 options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(r.x, n)

def worst_shift_lgbfsb(h, n, maxiter=500, gtol=1e-12, mu=1e6):
    """Minimize worst shift using L-BFGS-B with penalty for sum constraint."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    km = int(np.argmax(corr))
    d = km - (n - 1)
    i_lo = max(0, -d)
    i_hi = min(n, n - d)
    idx1 = np.arange(i_lo, i_hi)
    idx2 = idx1 + d

    def fun(x):
        val = np.sum(x[idx1] * (1.0 - x[idx2])) * dx
        s = np.sum(x) - n / 2.0
        return val + mu * s * s

    def jac(x):
        grad = np.zeros(n)
        grad[idx1] += (1.0 - x[idx2]) * dx
        grad[idx2] -= x[idx1] * dx
        s = np.sum(x) - n / 2.0
        grad += 2.0 * mu * s
        return grad

    bnds = [(0.0, 1.0)] * n
    x0 = h.copy()
    try:
        res = minimize(fun, x0, method='L-BFGS-B', jac=jac, bounds=bnds,
                       options={'maxiter': maxiter, 'gtol': gtol})
        return make_feasible(res.x, n)
    except Exception:
        return h

def worst_shift_slsqp(h, n, rounds=30, start_time=None, budget=1300):
    """Iteratively minimize the single worst shift with SLSQP."""
    best = compute_c5(h)
    for _ in range(rounds):
        if start_time is not None and time.time() - start_time > budget - 5:
            break
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        km = int(np.argmax(corr))
        d = km - (n - 1)
        i_lo = max(0, -d)
        i_hi = min(n, n - d)
        idx1 = np.arange(i_lo, i_hi)
        idx2 = idx1 + d

        def f(w):
            return np.sum(w[idx1] * (1.0 - w[idx2])) * dx

        def jf(w):
            g = np.zeros(n)
            g[idx1] += (1.0 - w[idx2]) * dx
            g[idx2] -= w[idx1] * dx
            return g

        bnds = [(0.0, 1.0)] * n
        cns = [{'type': 'eq', 'fun': _eq_con, 'jac': _eq_jac}]
        try:
            r = minimize(f, h.copy(), jac=jf, method='SLSQP',
                         bounds=bnds, constraints=cns,
                         options={'maxiter': 80, 'ftol': 1e-15})
            x = make_feasible(r.x, n)
            c = compute_c5(x)
            if c < best - 1e-16:
                h = x
                best = c
            else:
                break
        except Exception:
            break
    return h, best

def binary_sa(h, n, niter, T0, T1, rng, tlim, t0):
    b = (h >= 0.5).astype(float)
    tgt = n // 2
    while b.sum() > tgt:
        on = np.where(b == 1)[0]
        if len(on) == 0:
            break
        b[on[rng.randint(len(on))]] = 0
    while b.sum() < tgt:
        zo = np.where(b == 0)[0]
        if len(zo) == 0:
            break
        b[zo[rng.randint(len(zo))]] = 1
    cur = compute_c5(b)
    bst = cur
    bb = b.copy()
    for it in range(niter):
        if time.time() - t0 > tlim:
            break
        T = T0 * (T1 / T0) ** (float(it) / niter)
        mt = rng.randint(4)
        if mt == 0:
            on = np.where(b == 1)[0]
            zo = np.where(b == 0)[0]
            i = on[rng.randint(len(on))]
            j = zo[rng.randint(len(zo))]
            b[i] = 0; b[j] = 1
            nc = compute_c5(b)
            if nc <= cur or (T > 1e-12 and rng.random() < np.exp(-(nc - cur) / T)):
                cur = nc
                if nc < bst:
                    bst = nc; bb = b.copy()
            else:
                b[i] = 1; b[j] = 0
        elif mt == 1:
            # Pair swap with distance
            on = np.where(b == 1)[0]
            zo = np.where(b == 0)[0]
            i = on[rng.randint(len(on))]
            dist = rng.randint(1, max(2, n // 10))
            j2 = (i + dist) % n
            if b[j2] == 0:
                b[i] = 0; b[j2] = 1
                nc = compute_c5(b)
                if nc <= cur or (T > 1e-12 and rng.random() < np.exp(-(nc - cur) / T)):
                    cur = nc
                    if nc < bst:
                        bst = nc; bb = b.copy()
                else:
                    b[i] = 1; b[j2] = 0
        elif mt == 2:
            # Reversal
            sz = rng.randint(2, max(3, n // 18))
            s = rng.randint(0, n - sz)
            seg = b[s:s + sz].copy()
            b[s:s + sz] = seg[::-1]
            nc = compute_c5(b)
            if nc <= cur or (T > 1e-12 and rng.random() < np.exp(-(nc - cur) / T)):
                cur = nc
                if nc < bst:
                    bst = nc; bb = b.copy()
            else:
                b[s:s + sz] = seg
        else:
            # 2-swap (swap two 1s with two 0s)
            on = np.where(b == 1)[0]
            zo = np.where(b == 0)[0]
            if len(on) < 2 or len(zo) < 2:
                continue
            i1, i2 = rng.choice(on, 2, replace=False)
            j1, j2 = rng.choice(zo, 2, replace=False)
            b[i1] = 0; b[i2] = 0; b[j1] = 1; b[j2] = 1
            nc = compute_c5(b)
            if nc <= cur or (T > 1e-12 and rng.random() < np.exp(-(nc - cur) / T)):
                cur = nc
                if nc < bst:
                    bst = nc; bb = b.copy()
            else:
                b[i1] = 1; b[i2] = 1; b[j1] = 0; b[j2] = 0
    return bb, bst

def coord_descent(h, n, rounds=15, start_time=None, budget=1300):
    best = compute_c5(h)
    for r in range(rounds):
        if start_time and time.time() - start_time > budget - 3:
            break
        imp = False
        for j in range(n):
            if start_time and time.time() - start_time > budget - 2:
                break
            old = h[j]
            bc, bv = best, old
            for sv in [old - 0.02, old - 0.005, old - 0.001,
                       old + 0.001, old + 0.005, old + 0.02]:
                sv = max(0.0, min(1.0, sv))
                if abs(sv - old) < 1e-14:
                    continue
                t = h.copy()
                t[j] = sv
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < bc:
                    bc, bv = c, sv
            if bv != old:
                h[j] = bv
                h = make_feasible(h, n)
                best = bc
                imp = True
        if not imp:
            break
    return h, best

def fourier_perturb(h, n, rng, n_modes=4, amp=0.05):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(960, 0.5)

    n0 = len(h0)
    h0 = make_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Aggressive SLSQP at native resolution
    for T in [50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0, 500000.0]:
        rem = budget_s - (time.time() - t0)
        if rem < 20:
            break
        try:
            x = slsqp_opt(best_h, n0, T, 250, 1e-15)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x.copy()
        except Exception:
            pass

    # Phase 2: L-BFGS-B on worst shift (with penalty for sum)
    rem = budget_s - (time.time() - t0)
    if rem > 30:
        for mu in [1e5, 1e7]:
            rem = budget_s - (time.time() - t0)
            if rem < 15:
                break
            global mu
            x = worst_shift_lgbfsb(best_h, n0, 300, 1e-12, mu)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x.copy()

    # Phase 3: Worst-shift SLSQP with analytical gradient
    rem = budget_s - (time.time() - t0)
    if rem > 40:
        best_h, best_c5 = worst_shift_slsqp(best_h, n0, 30, t0, budget_s)

    # Phase 4: Binary SA with diverse moves
    rem = budget_s - (time.time() - t0)
    if rem > 120:
        tl = min(rem - 10, 100)
        b, cb = binary_sa(best_h, n0, 120000, 0.006, 0.000005, rng, tl, t0)
        if cb < best_c5:
            best_c5 = cb
            best_h = b.copy()
            for T in [2000.0, 20000.0, 200000.0]:
                rem = budget_s - (time.time() - t0)
                if rem < 20:
                    break
                try:
                    x = slsqp_opt(best_h, n0, T, 150, 1e-14)
                    c = compute_c5(x)
                    if c < best_c5:
                        best_c5 = c
                        best_h = x.copy()
                except Exception:
                    pass
            # Another round of worst-shift refinement
            rem = budget_s - (time.time() - t0)
            if rem > 30:
                best_h, best_c5 = worst_shift_slsqp(best_h, n0, 20, t0, budget_s)

    # Phase 5: Perturbation + re-optimize cycle
    for _ in range(3):
        rem = budget_s - (time.time() - t0)
        if rem < 60:
            break
        hp = fourier_perturb(best_h, n0, rng, n_modes=3, amp=0.03)
        try:
            x = slsqp_opt(hp, n0, 10000.0, 200, 1e-14)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = x.copy()
        except Exception:
            pass
        # Quick SA after perturbation
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            tl = min(rem - 5, 25)
            b, cb = binary_sa(best_h, n0, 40000, 0.003, 0.00001, rng, tl, t0)
            if cb < best_c5:
                best_c5 = cb
                best_h = b.copy()
                for T in [5000.0, 50000.0]:
                    rem = budget_s - (time.time() - t0)
                    if rem < 15:
                        break
                    try:
                        x = slsqp_opt(best_h, n0, T, 120, 1e-14)
                        c = compute_c5(x)
                        if c < best_c5:
                            best_c5 = c
                            best_h = x.copy()
                    except Exception:
                        pass

    # Phase 6: Upscale to higher resolution
    rem = budget_s - (time.time() - t0)
    if rem > 180:
        n1 = 1440
        ox = np.linspace(0.0, 2.0, n0)
        nx = np.linspace(0.0, 2.0, n1)
        hu = np.interp(nx, ox, best_h)
        hu = make_feasible(hu, n1)
        for T in [500.0, 5000.0, 50000.0, 500000.0]:
            rem = budget_s - (time.time() - t0)
            if rem < 25:
                break
            try:
                x = slsqp_opt(hu, n1, T, 150, 1e-14)
                c = compute_c5(x)
                if c < best_c5:
                    best_c5 = c
                    best_h = x.copy()
            except Exception:
                pass
        # Worst shift at higher res
        rem = budget_s - (time.time() - t0)
        if rem > 30:
            best_h, best_c5 = worst_shift_slsqp(best_h, n1, 20, t0, budget_s)
        # SA at higher res
        rem = budget_s - (time.time() - t0)
        if rem > 50:
            tl = min(rem - 10, 40)
            b, cb = binary_sa(best_h, n1, 60000, 0.004, 0.00001, rng, tl, t0)
            if cb < best_c5:
                best_c5 = cb
                best_h = b.copy()
            rem = budget_s - (time.time() - t0)
            if rem > 25:
                for T in [5000.0, 50000.0]:
                    rem = budget_s - (time.time() - t0)
                    if rem < 15:
                        break
                    try:
                        x = slsqp_opt(best_h, n1, T, 120, 1e-14)
                        c = compute_c5(x)
                        if c < best_c5:
                            best_c5 = c
                            best_h = x.copy()
                    except Exception:
                        pass

    # Phase 7: Final coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 20:
        best_h, best_c5 = coord_descent(best_h, len(best_h), 12, t0, budget_s)

    # Phase 8: Final worst-shift refinement
    rem = budget_s - (time.time() - t0)
    if rem > 15:
        best_h, best_c5 = worst_shift_slsqp(best_h, len(best_h), 15, t0, budget_s)

    # Phase 9: Very fine coordinate descent
    rem = budget_s - (time.time() - t0)
    if rem > 10:
        best = compute_c5(best_h)
        for j in range(len(best_h)):
            if time.time() - t0 > budget_s - 3:
                break
            old = best_h[j]
            bc, bv = best, old
            for sv in [max(0.0, old - 0.0005), max(0.0, old - 0.0001),
                       min(1.0, old + 0.0001), min(1.0, old + 0.0005)]:
                if abs(sv - old) < 1e-15:
                    continue
                t = best_h.copy()
                t[j] = sv
                t = make_feasible(t, len(t))
                c = compute_c5(t)
                if c < bc:
                    bc, bv = c, sv
            if bv != old:
                best_h[j] = bv
                best_h = make_feasible(best_h, len(best_h))
                best = bc

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
