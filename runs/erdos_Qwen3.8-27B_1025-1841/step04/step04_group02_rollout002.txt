To minimize max_k corr_k, we want to maximize min_k |S ∩ (S-k)| over k, i.e., make all autocorrelations as large and equal as possible.

This is a well-studied problem related to "perfect sequences" or "Golay sequences" in signal processing.

For a binary sequence of length n with n/2 ones, the autocorrelation A(k) = # {i : h[i]=1, h[i+k]=1} for shift k.

We want to minimize max_k (n/2 - A(k)) * 2/n = (1/n) * max_k (n - 2*A(k))

So we want to maximize min_k A(k), or equivalently minimize max_k (n - 2*A(k)).

This is equivalent to finding a binary sequence with as flat a (non-periodic) autocorrelation as possible.

For the continuous version, h[i] can be in [0,1], giving more flexibility.

import time

_NS = 0
_NT = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

    n = _NS
    T = _NT
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - _NS / 2.0

def con_jac(x):
    return np.ones(_NS)

    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    for it in range(n_iters):

        T = T0 * (T1 / T0) ** frac

        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        # Choose move type
        if rng.random() < 0.3:
            # Single swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0, 1
            new = compute_c5(b)
            if new <= curr or (T > 1e-12 and rng.random() < np.exp(-(new - curr) / T)):
                curr = new
                if new < best:
                    best = new
                    best_b = b.copy()
            else:
                b[i], b[j] = 1, 0
        else:

            blk = rng.randint(2, block_max + 1)
            s = rng.randint(0, n - blk)
            seg = b[s:s+blk]
            oi = np.where(seg == 1)[0]
            zi = np.where(seg == 0)[0]
            if len(oi) == 0 or len(zi) == 0:

            cz = rng.choice(zi, size=sc, replace=False)

            new = compute_c5(b)
            if new <= curr or (T > 1e-12 and rng.random() < np.exp(-(new - curr) / T)):
                curr = new
                if new < best:
                    best = new
                    best_b = b.copy()

    return best_b, best

    _NT = T

                   options={'maxiter': maxiter, 'ftol': ftol})

            bc, cc = sa_binary(h_rand, nc, 60000, 0.02, 0.00001, rng, block_max=6, restart_interval=5000)
            if cc < best_c:
                best_c = cc
                best_bc = bc.copy()
        # Upsample best coarse to n0
        if best_c < best_c5 + 0.001:
            h_up = np.interp(old_x, new_x, best_bc)
            h_up = make_feasible(h_up, n0)
            b_up, c5_up = sa_binary(h_up, n0, 30000, 0.003, 0.000002, rng, block_max=2)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = b_up.copy()
            elif compute_c5(h_up) < best_c5:
                best_c5 = compute_c5(h_up)
                best_h = h_up.copy()

    # Phase 6: Very low temp SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        n_it6 = min(40000, int(remaining * 30))
        b6, c56 = sa_binary(best_h, n0, n_it6, 0.002, 0.0000005, rng, block_max=2, restart_interval=5000)
        if c56 < best_c5:
            best_c5 = c56
            best_h = b6.copy()

    # Phase 7: SLSQP very high T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        try:
            x = slsqp_optimize(best_h, n0, 500.0, 300, 1e-16)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(x, n0)
        except Exception:
            pass

    # Phase 8: Coordinate descent refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = coord_refine(best_h.copy(), n0, 15, start_time, budget_s)

    # Phase 9: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        try:
            x = slsqp_optimize(best_h, n0, 1000.0, 200, 1e-16)
            c = compute_c5(x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(x, n0)
        except Exception:
            pass

    # Phase 10: One more quick SA from final
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        b10, c510 = sa_binary(best_h, n0, 15000, 0.001, 0.0000001, rng, block_max=2)
        if c510 < best_c5:
            best_c5 = c510
            best_h = b10.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
