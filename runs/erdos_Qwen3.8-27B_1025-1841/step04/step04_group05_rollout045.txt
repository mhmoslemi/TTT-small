import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def _upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return _make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return _make_feasible(h_new, new_n)

def _soft_max_val(h, n, T, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = (corr - mc) / T
    e = np.exp(shifted)
    soft = mc + T * np.log(e.sum())
    pen = lam * (h.sum() - n / 2.0) ** 2
    return (soft + pen) * dx

def _soft_max_grad(h, n, T, lam):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = (corr - mc) / T
    e = np.exp(shifted)
    s = e / e.sum()
    
    term1 = np.convolve(1.0 - h, s, mode='full')
    part1 = term1[n-1 : 2*n-1]
    
    term2 = np.correlate(s, h, mode='full')
    part2 = term2[2*n-2 : n-2 : -1]
    
    grad = dx * (part1 - part2)
    grad += 2.0 * lam * (h.sum() - n / 2.0)
    return grad

def _lbfgs_optimize(h0, n, T, lam, maxiter=200):
    def fun(x):
        return _soft_max_val(x, n, T, lam)
    def gfun(x):
        return _soft_max_grad(x, n, T, lam)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun, h0, jac=gfun, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return _make_feasible(res.x, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def _time_left():
        return budget_s - (time.time() - start_time)
    
    best_c5 = float('inf')
    best_h = None
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Multi-resolution optimization
    for n_pts in [1024, 2048, 3072]:
        if _time_left() < 60:
            break
        
        h_curr = _upsample(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()
        
        # Temperature schedule
        temps = [30.0, 15.0, 8.0, 5.0, 3.0, 2.0, 1.5]
        lam = 5000.0
        
        for T in temps:
            if _time_left() < 25:
                break
            try:
                h_opt = _lbfgs_optimize(h_curr, n_pts, T, lam, maxiter=300)
                c5_opt = compute_c5(h_opt)
                h_curr = h_opt.copy()
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
        
        # Random restart
        if _time_left() > 40:
            h_rand = 0.5 * h_curr + 0.5 * rng.rand(n_pts)
            h_rand = _make_feasible(h_rand, n_pts)
            try:
                h_opt = _lbfgs_optimize(h_rand, n_pts, 8.0, lam, maxiter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
        
        # Binary-like restart
        if _time_left() > 35:
            h_bin = (rng.rand(n_pts) < 0.5).astype(float)
            h_bin = _make_feasible(h_bin, n_pts)
            h_blend = 0.4 * h_curr + 0.6 * h_bin
            h_blend = _make_feasible(h_blend, n_pts)
            try:
                h_opt = _lbfgs_optimize(h_blend, n_pts, 10.0, lam, maxiter=200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
    
    # Coordinate descent polish
    n_pts = len(best_h)
    cd_n = min(n_pts, 1024)
    if cd_n < n_pts:
        h_cd = _upsample(best_h, cd_n)
    else:
        h_cd = best_h.copy()
    
    best_cd_c5 = compute_c5(h_cd)
    if best_cd_c5 < best_c5:
        best_c5 = best_cd_c5
    
    for _round in range(5):
        if _time_left() < 20:
            break
        improved = False
        for j in range(cd_n):
            if _time_left() < 10:
                break
            old = h_cd[j]
            for cand in [0.0, 1.0, old * 0.5, min(1.0, old + 0.15), max(0.0, old - 0.15)]:
                if abs(cand - old) < 1e-9:
                    continue
                trial = h_cd.copy()
                trial[j] = cand
                trial = _make_feasible(trial, cd_n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-10:
                    best_c5 = c
                    h_cd = trial
                    old = cand
                    improved = True
        if not improved:
            break
    
    # Final: project back and verify
    final_n = len(best_h)
    if cd_n < final_n and compute_c5(h_cd) < best_c5:
        best_c5 = compute_c5(h_cd)
        best_h = _upsample(h_cd, final_n)
    elif cd_n == final_n:
        best_h = h_cd
        best_c5 = min(best_c5, compute_c5(h_cd))
    
    # One final L-BFGS-B pass at the best resolution if time allows
    if _time_left() > 30:
        try:
            h_opt = _lbfgs_optimize(best_h, len(best_h), 2.0, 5000.0, maxiter=100)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
        except Exception:
            pass
    
    best_h = _make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    else:
        # If make_feasible changed things and made it worse, use pre-feasible
        pass
    
    return best_h, best_c5, len(best_h)
