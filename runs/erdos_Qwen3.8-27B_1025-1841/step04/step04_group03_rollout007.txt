import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.999
        else:
            mask = h > 0.001
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_and_weights(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exps = np.exp(T * shifted)
    logsum = np.log(np.sum(exps)) / T
    val = (max_c + logsum) * dx
    weights = exps / np.sum(exps)
    return val, weights, corr

def grad_smooth_max(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-10:
            continue
        s = m - (n - 1)
        if s >= 0:
            if n - s > 0:
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] -= w * h[s:]
        else:
            sp = -s
            if n - sp > 0:
                grad[:n-sp] += w * (1.0 - h[sp:])
                grad[sp:] -= w * h[:n-sp]
    return grad * dx

def smooth_max_obj(x, T):
    val, _, _ = smooth_max_and_weights(x, T)
    return val

def smooth_max_grad_full(x, T):
    val, weights, _ = smooth_max_and_weights(x, T)
    g = grad_smooth_max(x, weights, len(x))
    return val, g

def slsqp_smooth(x0, n, T, maxiter, ftol, start_time, budget_s):
    def obj(x):
        return smooth_max_obj(x, T)
    def grad(x):
        _, g = smooth_max_grad_full(x, T)
        return g
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    remaining = budget_s - (time.time() - start_time)
    if remaining < 10:
        return x0
    res = minimize(obj, x0.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if time.time() - start_time > budget_s - 5:
        return x0
    if res is not None and hasattr(res, 'x'):
        return res.x
    return x0

def gradient_descent(h, n, T_init, T_end, lr_init, n_iters, momentum, start_time, budget_s):
    velocity = np.zeros(n)
    T_curr = T_init
    for it in range(n_iters):
        if time.time() - start_time > budget_s - 2:
            break
        frac = it / max(n_iters, 1)
        T = T_init * (T_end / T_init) ** frac
        lr = lr_init * (0.98 ** (it // 100))
        
        val, weights, _ = smooth_max_and_weights(h, T)
        g = grad_smooth_max(h, weights, n)

        h_new = h + velocity
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        h = h_new
    return h

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 0.9999
        else:
            mask = h > 0.0001
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def smooth_max_and_weights(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    exps = np.exp(T * shifted)
    ssum = np.sum(exps)
    val = (max_c + np.log(ssum) / T) * dx
    weights = exps / ssum
    return val, weights, corr


def grad_single_lag(h, s, n):
    g = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            g[s:] += 1.0 - h[:n - s]
            g[:n - s] -= h[s:]
    else:
        sp = -s
        if n - sp > 0:
            g[:n - sp] += 1.0 - h[sp:]
            g[sp:] -= h[:n - sp]
    return g


def grad_smooth_max(h, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for m in range(len(weights)):
        w = weights[m]
        if w < 1e-12:
            continue
        s = m - (n - 1)
        grad += w * grad_single_lag(h, s, n)
    return grad * dx


def slsqp_smooth(x0, n, T, maxiter, ftol, start_time, budget_s):
    def obj(x):
        val, _, _ = smooth_max_and_weights(x, T)
        return val

    def grad(x):
        _, w, _ = smooth_max_and_weights(x, T)
        return grad_smooth_max(x, w, n)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    if budget_s - (time.time() - start_time) < 15:
        return x0
    try:
        res = minimize(obj, x0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res is not None and hasattr(res, 'x') and np.all(np.isfinite(res.x)):
            return res.x
    except Exception:
        pass
    return x0

    v = np.zeros(n)

    best_val = compute_c5(h)

            break
        t += 1
        frac = it / max(n_iters - 1, 1)
        T = T_init * (T_end / T_init) ** frac

        g = grad_smooth_max(h, weights, n)

def fourier_perturb(h, n, rng, n_modes=3, amp=0.03):

            best_h = h_sg.copy()

    # Phase 4: Adam GD at high T (fine-tuning)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        h_gd, c5_gd = adam_gd(best_h, n, 3000.0, 50000.0, 0.001, 20000,
                              (0.9, 0.999), 1e-8, start_time, budget_s)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()

    # Phase 5: SLSQP at very high T (near non-smooth max)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 50:
        x_res = slsqp_smooth(best_h, n, 30000.0, 300, 1e-13, start_time, budget_s)
        x_res = make_feasible(x_res, n)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = x_res.copy()

    # Phase 6: Perturbation + refinement (escape local min)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        hp = fourier_perturb(best_h, n, rng, n_modes=3, amp=0.02)
        h_gd, c5_gd = adam_gd(hp, n, 2000.0, 30000.0, 0.002, 8000,
                              (0.9, 0.999), 1e-8, start_time, budget_s)
        if c5_gd < best_c5:
            best_c5 = c5_gd
            best_h = h_gd.copy()

    # Phase 7: Coordinate descent polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n, start_time, budget_s)

    # Phase 8: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        x_res = slsqp_smooth(best_h, n, 20000.0, 200, 1e-13, start_time, budget_s)
        x_res = make_feasible(x_res, n)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = x_res.copy()

    # Phase 9: Final coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n, start_time,
                                             min(budget_s, start_time + remaining - 2))

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
