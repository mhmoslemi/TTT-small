def fourier_to_h(coeffs, n, K):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    val = np.full(n, 0.5, dtype=float)

        val += coeffs[2*(k-1)+1] * np.sin(k * np.pi * x)

def active_set_lp(h, n, top_k_frac=0.1):
    """Solve a local QP on the active set using projected gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_c = len(corr)
    k_active = max(10, int(top_k_frac * n_c))
    idx = np.argpartition(corr, -k_active)[-k_active:]
    
    # Weight by how close to max
    vals = corr[idx]
    max_v = vals.max()
    weights = np.exp(-5 * (max_v - vals) / (max_v + 1e-10))
    weights /= weights.sum()
    
    # Gradient of weighted sum of correlations w.r.t. h
    # corr[k] = sum_j h[j] * (1 - h[j + k - n + 1])
    # d corr[k] / d h[i] = (1 - h[i + k - n + 1]) - h[i - k + n - 1]
    # (where indices are valid)
    
    grad = np.zeros(n)
    for t, ki in enumerate(idx):
        shift = ki - (n - 1)
        w = weights[t]
        # Contribution to gradient from corr[ki]
        for j in range(n):
            j2 = j + shift
            if 0 <= j2 < n:
                grad[j] += w * (1.0 - h[j2])
            j3 = j - shift
            if 0 <= j3 < n:
                grad[j] -= w * h[j3]
    
    # Projected gradient step
    # Minimize: 0.5*h'Ah - b'h (quadratic approximation)
    # Just do a projected gradient step
    alpha = 0.01
    h_new = h - alpha * grad * dx
    h_new = np.clip(h_new, 0, 1)
    h_new = make_feasible(h_new, n)
    return h_new

def binary_sa(h_bin, n, rng, t_rem, max_rounds=200):
    """Simulated annealing on binary solution."""
    h = h_bin.copy()
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    T = 0.01
    T_min = 0.0001
    
    for rnd in range(max_rounds):
        if t_rem() < 5:
            break
        n_tries = min(3000, len(ones) * len(zeros))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            
            h[oi] = 0.0
            h[oj] = 1.0
            new_c = compute_c5(h)
            delta = new_c - cur_c
            
            if delta < 0 or rng.random() < np.exp(-delta / T):
                cur_c = new_c
                if cur_c < best_c:
                    best_c, best_h = cur_c, h.copy()
                ones[ii] = oj
                zeros[jj] = oi
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        
        T *= 0.98
        if T < T_min:
            T = T_min
    
    return best_h, best_c

def coord_polish(h0, n, t_rem):
    """Coordinate descent with binary swap pairs."""
    h = h0.copy()
    cur_c = compute_c5(h)
    steps = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
    
    for rnd in range(5):
        if t_rem() < 10:
            break
        improved = False
        for j in range(n):
            if t_rem() < 5:
                break
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur_c - 1e-14:
                    cur_c = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur_c

            x_f = np.linspace(0.0, 2.0, n_f, endpoint=False)
            cfs = np.cos(np.outer(np.arange(1, K + 1), np.pi * x_f))
            cfs = np.hstack([cfs, np.sin(np.outer(np.arange(1, K + 1), np.pi * x_f))])
            # cfs shape: (n_f, 2K)
            for T in [150, 800, 4000]:
                if rem() < 30:
                    break
                def obj_four(c):
                    h_tmp = 0.5 + cfs.dot(c)
                    h_tmp = np.clip(h_tmp, 0.0, 1.0)
                    dx_f = 2.0 / n_f
                    corr = np.correlate(h_tmp, 1.0 - h_tmp, mode='full')
                    mc = np.max(corr)
                    shifted = np.clip(T * (corr - mc), -80, 80)
                    sm = mc + np.log(np.sum(np.exp(shifted))) / T
                    ipen = 800.0 * (np.sum(h_tmp) - n_f / 2.0) ** 2
                    return sm * dx_f + ipen
                try:
                    res = minimize(obj_four, coeffs, method='L-BFGS-B',
                                   options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-10})
                    h_opt = np.clip(0.5 + cfs.dot(res.x), 0.0, 1.0)
                    h_opt = make_feasible(h_opt, n_f)
                    c_opt = compute_c5(h_opt)
                    if c_opt < best_c5:
                        best_c5, best_h = c_opt, h_opt.copy()
                        h_f = h_opt.copy()
                    coeffs = res.x
                except Exception:
                    pass

    # Phase 2: Direct L-BFGS-B with adaptive temperature at multiple resolutions
    for n_try in [best_n, best_n * 2, best_n * 3]:
        if rem() < 120:
            break
        h_n = upsample_h(best_h, n_try)
        c_n = compute_c5(h_n)
        if c_n < best_c5:
            best_c5, best_h = c_n, h_n.copy()
        dx_n = 2.0 / n_try
        for T in [80, 400, 2000, 8000]:
            if rem() < 30:
                break
            def obj_direct(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                mc = np.max(corr)
                shifted = np.clip(T * (corr - mc), -80, 80)
                sm = mc + np.log(np.sum(np.exp(shifted))) / T
                return sm * dx_n + 500.0 * (x.sum() - n_try / 2.0) ** 2
            try:
                res = minimize(obj_direct, h_n, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n_try,
                               options={'maxiter': 250, 'ftol': 1e-16, 'gtol': 1e-11})
                h_opt = make_feasible(res.x, n_try)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()
                    h_n = h_opt.copy()
            except Exception:
                pass

    # Phase 3: Alternating active-set optimization
    for n_alt in [best_n, min(best_n * 2, 1920)]:
        if rem() < 80:
            break
        h_a = upsample_h(best_h, n_alt)
        dx_a = 2.0 / n_alt
        for it in range(50):
            if rem() < 15:
                break
            corr = np.correlate(h_a, 1.0 - h_a, mode='full')
            mc = np.max(corr)
            n_c = len(corr)
            topk = max(20, int(0.12 * n_c))
            idx_top = np.argpartition(corr, -topk)[-topk:]
            vals_top = corr[idx_top]
            w = np.exp(-3.0 * (mc - vals_top) / (mc + 1e-10))
            w /= w.sum()
            def obj_active(x):
                c = np.correlate(x, 1.0 - x, mode='full')
                return float(np.dot(w, c[idx_top])) * dx_a + 400.0 * (x.sum() - n_alt / 2.0) ** 2
            try:
                res = minimize(obj_active, h_a, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n_alt,
                               options={'maxiter': 60, 'ftol': 1e-15, 'gtol': 1e-10})
                h_new = make_feasible(res.x, n_alt)
                if compute_c5(h_new) <= compute_c5(h_a) + 1e-12:
                    h_a = h_new
            except Exception:
                pass
        c_a = compute_c5(h_a)
        if c_a < best_c5:
            best_c5, best_h = c_a, h_a.copy()

    # Phase 4: Binary simulated annealing
    if rem() > 80:
        n_bin = min(best_n, 1024)
        h_b = upsample_h(best_h, n_bin) if n_bin < best_n else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        target_ones = n_bin // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            rng.shuffle(idx)
            h_bin[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            rng.shuffle(idx)
            h_bin[idx[:target_ones - n_ones]] = 1.0
        h_sa = h_bin.copy()
        cur_c = compute_c5(h_sa)
        best_sa_c = cur_c
        best_sa_h = h_sa.copy()
        ones = np.where(h_sa > 0.5)[0].tolist()
        zeros = np.where(h_sa <= 0.5)[0].tolist()
        T_sa = 0.008
        for rnd in range(150):
            if rem() < 5:
                break
            n_tries = min(4000, len(ones) * len(zeros))
            for _ in range(n_tries):
                if not ones or not zeros:
                    break
                ii = rng.randint(len(ones))
                jj = rng.randint(len(zeros))
                oi, oj = ones[ii], zeros[jj]
                h_sa[oi] = 0.0
                h_sa[oj] = 1.0
                new_c = compute_c5(h_sa)
                delta = new_c - cur_c
                if delta < 0 or rng.random() < np.exp(-delta / max(T_sa, 1e-8)):
                    cur_c = new_c
                    if cur_c < best_sa_c:
                        best_sa_c, best_sa_h = cur_c, h_sa.copy()
                    ones[ii] = oj
                    zeros[jj] = oi
                else:
                    h_sa[oi] = 1.0
                    h_sa[oj] = 0.0
            T_sa *= 0.97
            if T_sa < 0.0003:
                T_sa = 0.0003
        if best_sa_c < best_c5:
            best_c5, best_h = best_sa_c, best_sa_h.copy()

    # Phase 5: Coordinate descent polish
    if rem() > 25:
        h_cd = best_h.copy()
        cur_cd = compute_c5(h_cd)
        n_cd = len(h_cd)
        steps = [0.06, -0.06, 0.03, -0.03, 0.015, -0.015, 0.008, -0.008]
        for rnd in range(8):
            if rem() < 8:
                break
            improved = False
            for j in range(n_cd):
                if rem() < 4:
                    break
                ov = h_cd[j]
                for d in steps:
                    nv = ov + d
                    if nv < -1e-12 or nv > 1.0 + 1e-12:
                        continue
                    trial = h_cd.copy()
                    trial[j] = nv
                    trial = make_feasible(trial, n_cd)
                    c_try = compute_c5(trial)
                    if c_try < cur_cd - 1e-14:
                        cur_cd = c_try
                        h_cd = trial
                        ov = nv
                        improved = True
                        break
            if not improved:
                break
        if cur_cd < best_c5:
            best_c5, best_h = cur_cd, h_cd.copy()

    # Phase 6: Final high-T L-BFGS polish
    if rem() > 20:
        n_fin = len(best_h)
        dx_fin = 2.0 / n_fin
        for T_final in [6000, 20000]:
            if rem() < 12:
                break
            def obj_fin(x):
                corr = np.correlate(x, 1.0 - x, mode='full')
                mc = np.max(corr)
                shifted = np.clip(T_final * (corr - mc), -50, 50)
                sm = mc + np.log(np.sum(np.exp(shifted))) / T_final
                return sm * dx_fin + 1000.0 * (x.sum() - n_fin / 2.0) ** 2
            try:
                res = minimize(obj_fin, best_h, method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n_fin,
                               options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12})
                h_fin = make_feasible(res.x, n_fin)
                c_fin = compute_c5(h_fin)
                if c_fin < best_c5:
                    best_c5, best_h = c_fin, h_fin.copy()
            except Exception:
                pass

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
