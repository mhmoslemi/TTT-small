3. Identify the peaks of C(k) (or equivalently, the troughs of R(k))
4. Adjust h to reduce the peaks: for each peak at shift k*, apply a correction that reduces the overlap at that shift

The correction could be: for shift k*, the overlap is ∑_j h[j](1-h[j+k*]). To reduce this, we want to make h[j] and h[j+k*] more "aligned" (both high or both low). A simple approach:
- For j where h[j] is high and h[j+k*] is low (or vice versa), try to swap or adjust.

This is essentially what the gradient descent is doing, but a more structured version might converge faster.

Let me also consider: **what if we use a different parameterization?**

Instead of optimizing h directly, we could parameterize h = sigmoid(2*α) (or similar), which automatically satisfies 0<h<1. Then we optimize α with the constraint ∫sigmoid(2α)dx = 1. But this makes the constraint nonlinear and harder.

Another parameterization: h = 0.5 + 0.5*g where ∫g dx = 0 and |g| ≤ 1. This centers the problem around 0.

1. Use SLSQP with a very good smooth max (high T, top-5)
2. Follow with very fine coordinate descent (tries many small perturbations)
3. Try upscaling to higher resolution
4. Multi-start

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)[-1]
            room = 1.0 - h[idx]
            h[idx] += min(diff, room)
        else:
            idx = np.argsort(h)[0]
            h[idx] += max(diff, -h[idx])
    return h

def project_simple(h, n):
    """Fast projection: clip and redistribute."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:

            cnt = np.sum(mask)

def slsqp_smooth_max(h, n, T, top_m, maxiter=300):
    """SLSQP with log-sum-exp on top-m correlations."""
    dx = 2.0 / n
    offsets = np.arange(-(n-1), n)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-top_m:]
        c_sel = corr[idx]
        mc = c_sel[0]  # max
        # Log-sum-exp for numerical stability
        shifted = T * (c_sel - mc)
        shifted = np.minimum(shifted, 50.0)
        lse = mc + np.log(np.sum(np.exp(shifted))) / T
        return lse * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-top_m:]
        c_sel = corr[idx]
        mc = c_sel[0]
        shifted = T * (c_sel - mc)
        shifted = np.minimum(shifted, 50.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for i, k in enumerate(idx):
            s = offsets[k]
            w = weights[i]
            # d corr[k]/dh[j] = (1-h[j+s])*I[0<=j+s<n] - h[j-s]*I[0<=j-s<n]
            # First term: (1-x) shifted by -s, zero-padded
            # Second term: x shifted by s, zero-padded
            if s >= 0:
                # (1-x[j+s]) for j=0..n-1-s
                grad[0:n-s] += w * (1.0 - x[s:n])
                # -x[j-s] for j=s..n-1
                grad[s:n] -= w * x[0:n-s]
            else:
                ss = -s
                # (1-x[j+s]) = (1-x[j-ss]) for j=ss..n-1
                grad[ss:n] += w * (1.0 - x[0:n-ss])
                # -x[j-s] = -x[j+ss] for j=0..n-1-ss
                grad[0:n-ss] -= w * x[ss:n]
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                  constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def coord_descent_fine(h, n, budget_s, start_time, base_step=0.001):
    """Fine coordinate descent with adaptive step sizes."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    steps = [base_step, base_step*0.5, base_step*0.25, base_step*0.1]

                return best_h, best_c5
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in steps:

                    if 0.0 < v < 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = project_simple(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = project_simple(best_h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            # Try smaller steps
            base_step *= 0.5
            if base_step < 1e-5:
                break
            steps = [base_step, base_step*0.5, base_step*0.25, base_step*0.1]
    return best_h, best_c5

def pgd_nesterov(h, n, budget_s, start_time, top_m=5, lr=0.01):
    """Projected gradient descent with Nesterov momentum."""
    dx = 2.0 / n
    offsets = np.arange(-(n-1), n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    momentum = 0.9
    current_lr = lr
    no_improve = 0
    
    while time.time() - start_time < budget_s - 3:
        # Nesterov lookahead
        y = h - momentum * velocity
        y = np.clip(y, 0, 1)
        
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        idx_top = np.argsort(corr_y)[-top_m:]
        c_sel = corr_y[idx_top]
        mc = c_sel[0]
        shifted = 100.0 * (c_sel - mc)
        shifted = np.minimum(shifted, 50.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for i, k in enumerate(idx_top):
            s = offsets[k]
            w = weights[i]
            if s >= 0:
                grad[0:n-s] += w * (1.0 - y[s:n])
                grad[s:n] -= w * y[0:n-s]
            else:
                ss = -s
                grad[ss:n] += w * (1.0 - y[0:n-ss])
                grad[0:n-ss] -= w * y[ss:n]
        
        velocity = momentum * velocity - current_lr * grad
        new_h = h + velocity
        new_h = project_simple(new_h, n)
        
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            no_improve = 0
        elif c5_new <= compute_c5(h) + 1e-12:
            no_improve = 0
        else:
            no_improve += 1
        
        h = new_h
        
        if no_improve > 50:
            current_lr *= 0.5
            velocity *= 0.5
            no_improve = 0
            if current_lr < 1e-7:
                break
        
        if no_improve == 0 and current_lr < 0.05:
            current_lr *= 1.1
    
    return best_h, best_c5

def random_perturb(h, n, rng, scale=0.02):
    """Add random perturbation and project."""
    h_new = h + rng.randn(n) * scale
    return make_feasible(h_new, n)

def upscale(h, n, n_new):
    """Upscale h from n points to n_new points."""
    old_x = np.linspace(0, 2, n, endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, n_new)

**Idea: Use the "greedy peak reduction" approach**

1. Compute correlation, find the peak at shift k*
2. The overlap at k* is ∑_j h[j](1-h[j+k*])
3. To reduce this, we want to make h[j] and h[j+k*] more similar
4. For each j where h[j] is significantly different from h[j+k*], move them toward each other
5. This is like an "iterative water-filling" on the correlation

More specifically:
- At shift k*, the contribution to the overlap from position j is h[j](1-h[j+k*])
- If we decrease h[j] by δ (when h[j] > h[j+k*]), the overlap decreases by approximately (1-h[j+k*])*δ
- But this affects other shifts too!

So a local move might help one shift but hurt another. The key is to find moves that help the worst shifts without significantly hurting others.

This is what the gradient-based methods do, but a more targeted approach might work:

**Targeted move**: For the worst shift k*, find pairs (j, j+k*) where h[j] and h[j+k*] are most "misaligned" (one high, one low). Swap or adjust them.

        if s >= 0:
            grad[0:n-s] += w * (1.0 - h[s:n])
            grad[s:n] -= w * h[0:n-s]
        else:
            ss = -s
            grad[ss:n] += w * (1.0 - h[0:n-ss])
            grad[0:n-ss] -= w * h[ss:n]
    return grad * dx, corr

def slsqp_optimize(h, n, T, top_m, maxiter):
    """SLSQP with log-sum-exp smoothing."""
    dx = 2.0 / n
    offsets = np.arange(-(n-1), n)
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-top_m:]
        c_sel = corr[idx]
        mc = c_sel[0]
        shifted = np.clip(T * (c_sel - mc), None, 50.0)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argsort(corr)[-top_m:]
        c_sel = corr[idx]
        mc = c_sel[0]
        shifted = np.clip(T * (c_sel - mc), None, 50.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        
        grad = np.zeros(n)
        for i, k in enumerate(idx):
            s = offsets[k]
            w = weights[i]
            if s >= 0:
                grad[0:n-s] += w * (1.0 - x[s:n])
                grad[s:n] -= w * x[0:n-s]
            else:
                ss = -s
                grad[ss:n] += w * (1.0 - x[0:n-ss])
                grad[0:n-ss] -= w * x[ss:n]
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                  constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
    return res.x

def coord_descent(h, n, budget_s, start_time, base_step=0.001, max_rounds=5):
    """Coordinate descent with binary search refinement."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = base_step
    
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 3:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 3:
                return best_h, best_c5
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in [step, step*0.5, step*0.1]:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 < v < 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = project_fast(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = project_fast(best_h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            step *= 0.5
            if step < 1e-5:
                break
    
    return best_h, best_c5

def pgd_optimize(h, n, budget_s, start_time, top_m=5, lr=0.005):
    """Nesterov PGD."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    mu = 0.85
    current_lr = lr
    no_imp = 0
    
    while time.time() - start_time < budget_s - 2:
        y = np.clip(h - mu * velocity, 0, 1)
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        idx_top = np.argsort(corr_y)[-top_m:]
        c_sel = corr_y[idx_top]
        mc = c_sel[0]
        shifted = np.clip(100.0 * (c_sel - mc), None, 50.0)
        weights = np.exp(shifted)
        weights /= np.sum(weights)
        offsets = np.arange(-(n-1), n)
        
        grad = np.zeros(n)
        for i, k in enumerate(idx_top):
            s = offsets[k]
            w = weights[i]
            if s >= 0:
                grad[0:n-s] += w * (1.0 - y[s:n])
                grad[s:n] -= w * y[0:n-s]
            else:
                ss = -s
                grad[ss:n] += w * (1.0 - y[0:n-ss])
                grad[0:n-ss] -= w * y[ss:n]
        
        velocity = mu * velocity - current_lr * grad
        new_h = project_fast(h + velocity, n)
        c5_new = compute_c5(new_h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
            no_imp = 0
            if current_lr < 0.02:
                current_lr *= 1.05
        else:
            no_imp += 1
        
        h = new_h
        
        if no_imp > 100:
            current_lr *= 0.5
            velocity *= 0.3
            no_imp = 0
            if current_lr < 1e-6:
                break
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
