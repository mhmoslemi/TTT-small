So I need to compute a weighted sum of n gradient vectors, one for each lag. That's O(n * (2n-1)) = O(n²) in the worst case. For n=1280, that's about 3.3M operations. In numpy with vectorization, this should be fast.

```python
def softmax_obj_grad(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    
    # Softmax
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -100, 100)
    exps = np.exp(shifted)
    sum_exp = np.sum(exps)
    p = exps / sum_exp  # weights for each lag
    
    # Objective value
    obj_val = (mc + np.log(sum_exp) / T) * dx + mu * (x.sum() - n/2)**2
    
    # Gradient
    # For each lag m (0 to 2n-2), lag = m - n + 1
    # ∂corr_m/∂x[j] = (1-x[j+lag])*I(0≤j+lag<n) - x[j-lag]*I(0≤j-lag<n)
    
    # Vectorized computation:
    # g[j] = sum_m p[m] * [(1-x[j+lag_m])*I(j+lag_m in [0,n)) - x[j-lag_m]*I(j-lag_m in [0,n))]
    
    # Active lags only (where p[m] > threshold)
    active = p > 1e-12
    active_idx = np.where(active)[0]
    active_p = p[active_idx]
    active_lags = active_idx - n + 1
    
    g = np.zeros(n)
    for i in range(len(active_idx)):
        lag = active_lags[i]
        w = active_p[i]
        # First term: (1-x[j+lag]) for valid j
        js = max(0, -lag)
        je = min(n, n - lag)
        if js < je:
            g[js:je] += w * (1.0 - x[js+lag:je+lag])
        # Second term: -x[j-lag] for valid j
        js2 = max(0, lag)
        je2 = min(n, n + lag)
        if js2 < je2:
            g[js2:je2] -= w * x[js2-lag:je2-lag]
    
    g *= dx
    g += 2 * mu * (x.sum() - n/2)
    
    return obj_val, g
```

The inner loop is O(k_active * n) where k_active is the number of significant lags. With T=300 and 2559 lags, most weights will be negligible, so k_active might be 50-200. For n=1280, that's 64K-256K operations per gradient. Very fast.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_feasible(x, n):
    target = n / 2.0
    for _ in range(200):
        x = np.clip(x, 0.0, 1.0)
        diff = target - np.sum(x)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        if not mask.any():
            break
        x[mask] += diff / mask.sum()
    return np.clip(x, 0.0, 1.0)

def softmax_obj_grad(x, T, mu):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -100, 100)
    exps = np.exp(shifted)
    sum_exp = np.sum(exps)
    p = exps / sum_exp
    obj_val = (mc + np.log(sum_exp) / T) * dx + mu * (x.sum() - n / 2.0) ** 2

    active = p > 1e-10
    active_idx = np.where(active)[0]
    active_p = p[active_idx]

    g = np.zeros(n)
    for i in range(len(active_idx)):
        lag = int(active_idx[i]) - n + 1
        w = active_p[i]
        # First term: (1-h[j-lag]) for j in [max(0,lag), min(n, n+lag))
        js = max(0, lag)
        je = min(n, n + lag)
        if js < je:
            g[js:je] += w * (1.0 - x[js - lag:je - lag])
        # Second term: -h[j+lag] for j in [max(0,-lag), min(n, n-lag))
        js2 = max(0, -lag)
        je2 = min(n, n - lag)
        if js2 < je2:
            g[js2:je2] -= w * x[js2 + lag:je2 + lag]

    g *= dx
    g += 2.0 * mu * (x.sum() - n / 2.0)
    return obj_val, g

def subgradient_max(h0, n, n_iters=4000, topk=40):
    dx = 2.0 / n
    h = h0.copy()
    h_avg = h.copy()
    best_h = h0.copy()
    best_c5 = compute_c5(h)
    target = best_c5 - 0.004

    for t in range(1, n_iters + 1):
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_c5 = float(np.max(corr) * dx)

        k = min(topk, len(corr))
        idx = np.argpartition(corr, -k)[-k:]
        vals = corr[idx]
        shifted = 10.0 * (vals - np.max(vals))
        weights = np.exp(shifted)
        weights /= np.sum(weights)

        g = np.zeros(n)
        for i in range(k):
            lag = int(idx[i]) - n + 1
            w = weights[i]
            js = max(0, lag)
            je = min(n, n + lag)
            if js < je:
                g[js:je] += w * (1.0 - h[js - lag:je - lag])
            js2 = max(0, -lag)
            je2 = min(n, n - lag)
            if js2 < je2:
                g[js2:je2] -= w * h[js2 + lag:je2 + lag]

        gnorm_sq = float(np.sum(g ** 2)) + 1e-10
        gap = cur_c5 - target
        if gap < 0:
            gap = 0.001
        alpha = gap / (gnorm_sq * dx * dx + 1e-10)
        alpha = min(alpha, 3.0 / (t ** 0.55))
        alpha = max(alpha, 1e-4)

        h = project_feasible(h - alpha * g, n)
        h_avg = ((t - 1) / t) * h_avg + (1.0 / t) * h

        if t % 500 == 0:
            c5_avg = compute_c5(h_avg)
            c5_cur = compute_c5(h)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg.copy()
                target = best_c5 - 0.0015
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
                target = best_c5 - 0.0015

    c5_avg = compute_c5(h_avg)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = h_avg.copy()
    return best_h, best_c5

def lbgfb_with_grad(h0, n, T, mu, max_iter=300):
    def fun(x):
        return softmax_obj_grad(x, T, mu)[0]

    def grad(x):
        return softmax_obj_grad(x, T, mu)[1]

    res = minimize(fun, h0, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxfun': 5000})
    return make_feasible(res.x, n)

def coord_descent_polish(h0, n, max_rounds=8, steps=None):
    if steps is None:
        steps = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def slsqp_polish(h0, n, T, max_iter=80):
    def fun(x):
        return softmax_obj_grad(x, T, 0.0)[0]

    def grad(x):
        return softmax_obj_grad(x, T, 0.0)[1]

    def con_eq(x):
        return np.sum(x) - n / 2.0

    res = minimize(fun, h0, jac=grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq}],
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)
