    M = len(corr)

    top_idx = np.argsort(corr)[-K:]
    top_vals = corr[top_idx]
    weights = top_vals / np.sum(top_vals)

    grad = np.zeros(n)

            if 0 <= j - s < n:
                grad[j] -= wt * h[j-s]
    grad *= dx
    
    # Step

    return h_new
