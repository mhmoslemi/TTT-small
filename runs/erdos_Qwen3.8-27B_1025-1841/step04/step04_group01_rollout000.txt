import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0
_G_TOP_IDX = None

def _obj_full(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _obj_top_shifts(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top = corr[_G_TOP_IDX]
    mc = np.max(top)
    shifted = top - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def binary_local_search(h, max_iters=2000, time_limit=None, start_time=None):
    n = len(h)
    h = np.round(h)
    h = make_feasible(h, n)
    ones = np.where(h > 0.5)[0]

        ones = np.where(h > 0.5)[0]
        zeros = np.where(h < 0.5)[0]
    corr = get_corr(h)
    best_c5 = float(np.max(corr) * (2.0 / n))
    rng = np.random.RandomState(42)
    it = 0
    while it < max_iters:
        if time_limit is not None:
            elapsed = time.time() - start_time
            if elapsed > time_limit:
                break
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_trials = min(n, max(50, n_ones * n_zeros // 200))
        for _ in range(n_trials):
            a = ones[rng.randint(n_ones)]
            b = zeros[rng.randint(n_zeros)]
            h[a] = 0.0
            h[b] = 1.0
            corr_new = get_corr(h)
            c5_new = float(np.max(corr_new) * (2.0 / n))
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                corr = corr_new
                improved = True
                # Update sets
                ones = np.where(h > 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                n_ones = len(ones)
                n_zeros = len(zeros)
            else:
                h[a] = 1.0
                h[b] = 0.0
        it += 1
        if not improved and it > 20:

                    b1 = zeros[rng.randint(n_zeros)]
                    b2 = zeros[rng.randint(n_zeros)]

                        corr = get_corr(h)
                        ones = np.where(h > 0.5)[0]
                        zeros = np.where(h < 0.5)[0]
                        n_ones = len(ones)

def coord_descent_fine(h, n_steps=3, time_limit=None, start_time=None):
    n = len(h)
    best_c5 = compute_c5(h)
    steps_list = [0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    for _ in range(n_steps):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        improved_any = False
        for j in range(n):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            old_val = h[j]
            best_val = old_val
            best_c = best_c5
            for step in steps_list:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * step
                    if 0.0 < v < 1.0:
                        trial = h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved_any = True
        if not improved_any:
            break
    return h, best_c5
