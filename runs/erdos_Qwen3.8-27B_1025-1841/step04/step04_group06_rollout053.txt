def lbgfb_optimize(h_start, n, T, mu, max_iter=200, deadline_fn=None):
    dx = 2.0 / n

    def f(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(hv) - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def fractional_polish(h_bin, n, deadline_fn=None):
    """Allow small fractional deviations at transition points."""
    h = h_bin.copy().astype(float)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    trans = set()
    for i in ones:
        if i > 0 and h[i-1] < 0.5:
            trans.add(i)
            trans.add(i-1)
        if i < n-1 and h[i+1] < 0.5:
            trans.add(i)
            trans.add(i+1)
    trans = sorted(trans)
    if len(trans) == 0:
        trans = list(range(0, n, max(1, n // 50)))
    
    current_c5 = compute_c5(h)
    for step_size in [0.1, 0.05, 0.02, 0.01, 0.005]:
        if deadline_fn is not None and deadline_fn() < 5:
            break
        improved = True
        while improved:
            if deadline_fn is not None and deadline_fn() < 3:
                break
            improved = False
            for idx in trans:
                if deadline_fn is not None and deadline_fn() < 3:
                    break
                for d in [step_size, -step_size]:
                    nv = h[idx] + d
                    if nv < 0 or nv > 1:

                    tr = h.copy()
                    tr[idx] = nv
                    tr = make_feasible(tr, n)
                    c = compute_c5(tr)
                    if c < current_c5 - 1e-13:
                        current_c5 = c
                        h = tr
                        improved = True
                        break
    return h, current_c5

        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = np.array([1.0 if (i % 2 == 0) else 0.0 for i in range(640)])

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Get binary solution at n=1024
    n1 = 1024
    h_up = upsample_h(best_h, n1)
    c5u = compute_c5(h_up)
    if c5u < best_c5:
        best_c5, best_h = c5u, h_up.copy()
    h_bin = to_binary_fixed(h_up, n1, rng)
    c5b = compute_c5(h_bin)
    if c5b < best_c5:
        best_c5, best_h = c5b, h_bin.copy()

    # Phase 2: Intensive greedy binary search
    if tl() > 200:
        h_bin2, c5b2 = greedy_binary(h_bin, rng, n_tries=500, max_rounds=400, deadline_fn=tl)
        if c5b2 < best_c5:
            best_c5, best_h = c5b2, h_bin2.copy()
        h_bin = h_bin2

    # Phase 3: Simulated annealing from binary
    if tl() > 150:
        h_sa, c5sa = sa_binary(h_bin, rng, n_steps=15000, T_start=1.0, T_end=0.003, deadline_fn=tl)
        if c5sa < best_c5:
            best_c5, best_h = c5sa, h_sa.copy()
        h_bin = h_sa.copy()
        h_bin3, c5b3 = greedy_binary(h_bin, rng, n_tries=400, max_rounds=200, deadline_fn=tl)
        if c5b3 < best_c5:
            best_c5, best_h = c5b3, h_bin3.copy()
        h_bin = h_bin3

    # Phase 4: Iterated local search
    if tl() > 120:
        h_ils, c5ils = iterated_local_search(h_bin, rng, n_iter=8, n_flip=3, deadline_fn=tl)
        if c5ils < best_c5:
            best_c5, best_h = c5ils, h_ils.copy()
        h_bin = h_ils.copy()

    # Phase 5: More SA + greedy cycles
    if tl() > 80:
        for cycle in range(3):
            if tl() < 30:
                break
            h_c, c5c = sa_binary(h_bin, rng, n_steps=8000, T_start=0.5, T_end=0.005, deadline_fn=tl)
            h_c, c5c2 = greedy_binary(h_c, rng, n_tries=400, max_rounds=150, deadline_fn=tl)
            if c5c2 < best_c5:
                best_c5, best_h = c5c2, h_c.copy()
            h_bin = h_c

    # Phase 6: Fractional polish at n=1024
    if tl() > 40:
        h_fp, c5fp = fractional_polish(h_bin, n1, deadline_fn=tl)
        if c5fp < best_c5:
            best_c5, best_h = c5fp, h_fp.copy()

    # Phase 7: L-BFGS-B soft-max optimization
    if tl() > 60:
        h_work = h_bin.astype(float).copy()
        for T in [100, 500, 2000]:
            if tl() < 20:
                break
            try:
                h_opt = lbgfb_optimize(h_work, n1, T, mu=500.0, max_iter=150, deadline_fn=tl)
                c5o = compute_c5(h_opt)
                if c5o < best_c5:
                    best_c5, best_h = c5o, h_opt.copy()
                h_work = h_opt.copy()
            except Exception:
                pass
        # Fractional polish after L-BFGS-B
        if tl() > 15:
            h_fp2, c5fp2 = fractional_polish(h_work, n1, deadline_fn=tl)
            if c5fp2 < best_c5:
                best_c5, best_h = c5fp2, h_fp2.copy()

    # Phase 8: Try at higher resolution n=2048
    if tl() > 100:
        n2 = 2048
        h_hi = upsample_h(best_h, n2)
        c5hi = compute_c5(h_hi)
        if c5hi < best_c5:
            best_c5, best_h = c5hi, h_hi.copy()
        h_hi_bin = to_binary_fixed(h_hi, n2, rng)
        c5hbi = compute_c5(h_hi_bin)
        if c5hbi < best_c5:
            best_c5, best_h = c5hbi, h_hi_bin.copy()
        if tl() > 50:
            h_hi2, c5hi2 = greedy_binary(h_hi_bin, rng, n_tries=300, max_rounds=150, deadline_fn=tl)
            if c5hi2 < best_c5:
                best_c5, best_h = c5hi2, h_hi2.copy()
        if tl() > 30:
            try:
                h_opt = lbgfb_optimize(h_hi_bin.astype(float), n2, 500, mu=300.0, max_iter=100, deadline_fn=tl)
                c5o = compute_c5(h_opt)
                if c5o < best_c5:
                    best_c5, best_h = c5o, h_opt.copy()
            except Exception:
                pass

    # Phase 9: Try original resolution n=640 (parent)
    if tl() > 40:
        n3 = len(h0)
        h_640 = to_binary_fixed(h0, n3, rng)
        c5_640 = compute_c5(h_640)
        if c5_640 > best_c5:
            h_640 = upsample_h(best_h, n3)
            h_640 = to_binary_fixed(h_640, n3, rng)
            c5_640 = compute_c5(h_640)
        if c5_640 < best_c5:
            best_c5, best_h = c5_640, h_640.copy()
        if tl() > 20:
            h_640b, c5_640b = greedy_binary(h_640, rng, n_tries=400, max_rounds=200, deadline_fn=tl)
            if c5_640b < best_c5:
                best_c5, best_h = c5_640b, h_640b.copy()

    # Phase 10: Final squeeze - try a few more SA restarts at best resolution
    if tl() > 30:
        n_best = len(best_h)
        h_try = to_binary_fixed(best_h, n_best, rng)
        for _ in range(3):
            if tl() < 10:
                break
            h_try, c5_try = sa_binary(h_try, rng, n_steps=5000, T_start=0.4, T_end=0.005, deadline_fn=tl)
            h_try, c5_try2 = greedy_binary(h_try, rng, n_tries=300, max_rounds=100, deadline_fn=tl)
            if c5_try2 < best_c5:
                best_c5, best_h = c5_try2, h_try.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
