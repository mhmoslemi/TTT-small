    return float(np.max(corr) * dx)

def compute_c5_fft(h):

    H = np.fft.rfft(h)
    ac = np.fft.irfft(H * np.conj(H), n=n)

    if n_new == n_old:
        return make_feasible(h_old.copy(), n_new)
    # Linear interpolation with periodic wrapping
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0

    corr = get_corr(h)
    dx = 2.0 / len(h)
    overlaps = corr * dx
    
    if top_k is not None:
        # Use only top_k largest overlaps
        idx = np.argsort(overlaps)[::-1][:top_k]
        vals = overlaps[idx]

  s = 0: full[j] = sum_i h[i]*(1-h[i])
  s < 0: let t = -s: full[j] = sum_{i=0}^{n-1-t} h[i+t]*(1-h[i])

  The max C₅ = dx * max(full[j] for j=0..2n-2)
  
  The "dangerous" lags are those with small |s| (i.e., j near n-1), because they have the most terms.

Now for the gradient:
∂full[j]/∂h[p] for s = n-1-j > 0:
full[j] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
∂/∂h[p] = sum_{i=0}^{n-1-s} [δ_{i,p}*(1-h[i+s]) + h[i]*(-δ_{i+s,p})]
= (1-h[p+s]) * [0≤p≤n-1-s] - h[p] * [s≤p≤n-1]
  (second term: i+s=p => i=p-s, valid if 0≤p-s≤n-1-s => s≤p≤n-1)

For s < 0, t = -s:
full[j] = sum_{i=0}^{n-1-t} h[i+t]*(1-h[i])
∂/∂h[p] = sum_{i=0}^{n-1-t} [δ_{i+t,p}*(1-h[i]) + h[i+t]*(-δ_{i,p})]
= (1-h[p-t]) * [0≤p-t≤n-1-t => t≤p≤n-1] - h[p-t] * [0≤p≤n-1-t]
= (1-h[p-t]) * [t≤p] - h[p-t] * [p≤n-1-t]

So for each lag j (or shift s), the gradient with respect to h[p] is:
- For s > 0: (1-h[p+s]) * [p ≤ n-1-s] - h[p] * [p ≥ s]
- For s = 0: 1 - 2*h[p]
- For s < 0 (t=-s): (1-h[p-t]) * [p ≥ t] - h[p-t] * [p ≤ n-1-t]

This is O(n) to compute for a single lag. For K lags, it's O(K*n).

Now, the soft-max objective:
L = (1/T) * log(sum_j exp(T * dx * full[j]))
∂L/∂h[p] = sum_j w_j * dx * ∂full[j]/∂h[p]
where w_j = exp(T*dx*full[j]) / sum_k exp(T*dx*full[k])

If I restrict to top-K lags:
∂L/∂h[p] ≈ sum_{j in top-K} w_j * dx * ∂full[j]/∂h[p]

This is very efficient!

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return make_feasible(h_old.copy(), n_new)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return make_feasible(h_new, n_new)

def lag_gradient(h, s):
    """Gradient of full[n-1-s] (= overlap at shift s) w.r.t. h. O(n)."""
    n = len(h)
    grad = np.zeros(n)
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    elif s > 0:
        # full[n-1-s] = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        # d/dh[p] = (1-h[p+s])*[p<=n-1-s] - h[p]*[p>=s]
        grad[0:n-s] += 1.0 - h[s:n]
        grad[s:] -= h[0:n-s]
    else:
        t = -s
        # full[n-1+s] = sum_{i=0}^{n-1-t} h[i+t]*(1-h[i])
        # d/dh[p] = (1-h[p-t])*[p>=t] - h[p-t]*[p<=n-1-t]
        grad[t:] += 1.0 - h[0:n-t]
        grad[0:n-t] -= h[t:n]
    return grad

def topk_gradient_descent(h, n, lr=0.005, top_k=15, n_iters=20000, T_sched=None, verbose=False):
    """Gradient descent on top-K overlaps with adaptive step size."""
    h = h.copy()
    h = make_feasible(h, n)
    dx = 2.0 / n
    current_c5 = compute_c5(h)
    lr_cur = lr
    stall_count = 0
    
    for it in range(n_iters):
        corr = get_corr(h)
        overlaps = corr * dx
        # Get top-K lag indices (in the full correlation array)
        sorted_idx = np.argsort(overlaps)[::-1][:top_k]
        top_vals = overlaps[sorted_idx]
        
        # Softmax weights over top-K
        tv = top_vals / max(T_sched, 1e-8)
        tv -= np.max(tv)
        w = np.exp(tv)
        w /= np.sum(w)
        
        # Compute gradient: sum of w_j * dx * grad_lag(j)
        grad = np.zeros(n)
        for ki, j in enumerate(sorted_idx):
            s = (n - 1) - j  # shift
            lg = lag_gradient(h, s)
            grad += w[ki] * dx * lg
        
        # Project to maintain sum constraint
        grad -= np.mean(grad)
        
        # Take step
        h_new = h + lr_cur * grad
        h_new = make_feasible(h_new, n)
        
        new_c5 = compute_c5(h_new)
        if new_c5 < current_c5 - 1e-15:
            h = h_new
            current_c5 = new_c5
            stall_count = 0
            lr_cur = min(lr_cur * 1.02, 0.02)
        else:
            stall_count += 1
            lr_cur *= 0.92
            if lr_cur < 1e-6:
                break
    
    return h, current_c5

def soft_max_lbgfb(h_start, n, T=100.0, mu=500.0, max_iter=300):
    dx = 2.0 / n
    def f(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(hv) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_swap_search(h, n, rng, n_rounds=50, n_tries_per_round=1500):
    """Random swap local search on binary h."""
    h = h.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(n_rounds):
        improved = False
        for _ in range(n_tries_per_round):
            if len(ones) == 0 or len(zeros) == 0:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def coordinate_polish(h, n, n_rounds=20, step_sizes=None):
    """Coordinate descent: try small changes to each h[i]."""
    if step_sizes is None:
        step_sizes = [0.01, -0.01, 0.005, -0.005, 0.002, -0.002]
    h = h.copy()
    current = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in step_sizes:
                nv = ov + d
                if nv < -0.01 or nv > 1.01:
                    continue
                tr = h.copy()
                tr[j] = np.clip(nv, 0.0, 1.0)
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-13:
                    current = c
                    h = tr
                    ov = h[j]
                    improved = True
                    break
            # Try binary
            bv = 1.0 - h[j]
            if abs(bv - h[j]) > 0.01:
                tr = h.copy()
                tr[j] = bv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-13:
                    current = c
                    h = tr
                    improved = True
        if not improved:
            break
    return h, current

def focused_lag_fix(h, n, rng, n_iters=3000):
    """Identify the worst lag and do targeted gradient steps on it."""
    h = h.copy()
    dx = 2.0 / n
    current = compute_c5(h)
    
    for it in range(n_iters):
        corr = get_corr(h)
        overlaps = corr * dx
        k_max = np.argmax(overlaps)
        s = (n - 1) - k_max
        # Gradient for this single lag
        grad = lag_gradient(h, s) * dx
        # Project
        grad -= np.mean(grad)
        # Small step
        lr = 0.003 * (1.0 + 0.5 * rng.rand())
        h_new = np.clip(h + lr * grad, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < current - 1e-15:
            h = h_new
            current = c_new
    return h, current
