    """Binary simulated annealing."""
    h = h.copy()
    # Project to binary
    sorted_idx = np.argsort(-h)
    h = np.zeros(n)

            break
        T = T_start * (T_end / T_start) ** (it / max(n_iters, 1))

        j_idx = rng.randint(len(zeros))

        j = zeros[j_idx]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def compute_subgrad(h, n, top_m):
    corr = get_corr(h)
    top_idx = np.argsort(-corr)[:top_m]
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)
        if s >= 0:
            if s < n:
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] -= h[s:]
        else:
            sp = -s
            if sp < n:
                grad[:n - sp] += (1.0 - h[sp:])
                grad[sp:] -= h[:n - sp]
    grad /= float(top_m)
    return grad, corr

        i = ones[i_idx]
        j = zeros[j_idx]
        cur[i] = 0.0
        cur[j] = 1.0
        new_c5 = compute_c5(cur)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur_c5 = new_c5
            ones[i_idx] = j
            zeros[j_idx] = i
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur.copy()
        else:
            cur[i] = 1.0
            cur[j] = 0.0
    return best_h, best_c5

def subgrad_optimize(h, n, time_limit, start_step, top_m, restart_every=5000):
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        step_size = start_step / (1.0 + 0.005 * (it % restart_every))
        grad, corr = compute_subgrad(h, n, top_m)
        h = h - step_size * grad
        h = make_feasible(h, n)
        c5_val = float(np.max(corr) * dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        it += 1
    return best_h, best_c5

_G_N = 0

    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
