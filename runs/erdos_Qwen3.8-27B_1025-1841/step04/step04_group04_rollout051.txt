def _obj_and_grad(x):
    n = _N
    dx = 2.0 / n
    T = _T
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    w = np.exp(T * shifted)
    sw = w.sum()
    w = w / sw
    fval = (mc + np.log(sw) / T) * dx
    
    grad = np.zeros(n)
    # s = k - (n-1), k from 0 to 2n-2, s from -(n-1) to (n-1)
    for k in range(2*n-1):
        if w[k] < 1e-12:
            continue
        s = k - (n-1)
        # First term: (1-h[i+s]) for i where 0<=i+s<n and 0<=i<n
        # i range: max(0,-s) <= i < min(n, n-s)
        lo = max(0, -s)
        hi = min(n, n - s)
        if lo < hi:
            # i goes from lo to hi-1
            # i+s goes from lo+s to hi-1+s
            grad[lo:hi] += w[k] * (1.0 - x[lo+s:hi+s])
        
        # Second term: -h[i-s] for i where 0<=i-s<n and 0<=i<n
        # i range: max(0, s) <= i < min(n, s+n)
        lo2 = max(0, s)
        hi2 = min(n, s + n)
        if lo2 < hi2:
            # i goes from lo2 to hi2-1
            # i-s goes from lo2-s to hi2-1-s
            grad[lo2:hi2] -= w[k] * x[lo2-s:hi2-s]
    
    grad *= dx
    return fval, grad
