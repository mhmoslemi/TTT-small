        if abs(diff) < 1e-15:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-14
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)

        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def smooth_max_val(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -60, 60)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_run(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = smooth_max_val(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10,
                            'maxls': 50})
    return make_feasible(res.x, n)

def topk_minimax(h0, n, n_iters, topk_frac, regime_iters):
    """Alternating: optimize a weighted average of top-k correlations"""
    dx = 2.0 / n
    h = h0.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = max(3, int(topk_frac * len(corr)))
        idx = np.argpartition(corr, -k)[-k:]
        weights = np.zeros(len(corr))
        weights[idx] = 1.0 / k
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(weights, c)) * dx + 500.0 * (x.sum() - n / 2.0) ** 2
        
        try:
            res = minimize(obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': regime_iters, 'ftol': 1e-15, 'gtol': 1e-9})
            h_new = make_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h):
                h = h_new
        except:
            pass
    return h

def binary_local_search(h_bin, n, rng, max_rounds, tries_per_round):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries_per_round, max(1, len(ones) * len(zeros)))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def coord_descent(h0, n, max_rounds):
    h = h0.copy()
    cur = compute_c5(h)
    steps = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    improved = True
                    break
        if not improved:
            break
    return h, cur
