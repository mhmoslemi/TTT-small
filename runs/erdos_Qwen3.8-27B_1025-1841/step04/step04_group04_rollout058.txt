    _G_N = n
    _G_T = T
    _G_DX = 2.0 / n
    _G_TARGET = n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(obj_func, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        h_new = make_feasible(res.x, n)
        return h_new
    except Exception:
        return h

def coord_descend(h, n, t_limit, max_rounds):
    c5_cur = compute_c5(h)
    for _r in range(max_rounds):
        if time.time() > t_limit:
            break
        improved = False
        for j in range(n):
            if time.time() > t_limit:
                break
            old = h[j]
            best_val = old
            best_c = c5_cur
            cands = [0.0, 1.0]
            if 0.02 < old < 0.98:
                cands += [max(0.0, old - 0.15), max(0.0, old - 0.05),
                          min(1.0, old + 0.05), min(1.0, old + 0.15)]
            for cand in cands:
                if abs(cand - old) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                c5_cur = best_c
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Low-resolution exploration at n=128
    n_low = 128
    h_low = np.full(n_low, 0.5)
    best_low_c5 = compute_c5(h_low)
    best_low_h = h_low.copy()

    low_starts = [h_low.copy()]
    for period in [3, 4, 5, 6]:
        k = period // 2
        pat = np.zeros(period)
        pat[:k] = 1.0
        h_t = np.tile(pat, n_low // period + 1)[:n_low]
        low_starts.append(make_feasible(h_t, n_low))
    for _ in range(4):
        h_r = (rng.rand(n_low) < 0.5).astype(float)
        low_starts.append(make_feasible(h_r, n_low))

    for hs in low_starts:
        if time.time() - t0 > budget_s * 0.15:
            break
        hw = hs.copy()
        for T in [20.0, 60.0]:
            hw = slsqp_optimize(hw, n_low, T, 200, _obj_full)
        c5_l = compute_c5(hw)
        if c5_l < best_low_c5:
            best_low_c5 = c5_l
            best_low_h = hw.copy()

    h_from_low = upsample_h(best_low_h, n)

    # Phase 2: Collect diverse high-res starting points
    hi_starts = [best_h.copy(), h_from_low.copy()]

    for period in [3, 4, 5, 6, 7, 8, 12]:
        for k_frac in [2, 3]:
            k = min(k_frac, period - 1)
            pat = np.zeros(period)
            pat[:k] = 1.0
            h_t = np.tile(pat, n // period + 1)[:n]
            h_t = make_feasible(h_t, n)
            hi_starts.append(h_t)

    for _ in range(4):
        h_r = (rng.rand(n) < 0.5).astype(float)
        hi_starts.append(make_feasible(h_r, n))

    for sc in [0.04, 0.08, 0.15, 0.3]:
        hi_starts.append(make_feasible(best_h + rng.randn(n) * sc, n))

    # Evaluate and sort by quality
    c5_list = []
    for hs in hi_starts:
        c5_list.append(compute_c5(hs))
    order = np.argsort(c5_list)

    # Phase 3: Optimize the best high-res starting points
    n_opt = min(8, len(hi_starts))
    for idx in range(n_opt):
        si = int(order[idx])
        if time.time() - t0 > budget_s - 250:
            break
        if c5_list[si] > best_c5 + 0.04:
            break

        hw = hi_starts[si].copy()

        # Multi-temperature SLSQP
        for T, mi in [(25, 250), (75, 220), (200, 180), (500, 150)]:
            if time.time() - t0 > budget_s - 120:
                break
            hw = slsqp_optimize(hw, n, T, mi, _obj_full)

        # Power mean refinement
        if time.time() - t0 < budget_s - 60:
            hw = slsqp_optimize(hw, n, 100, 150, _obj_pmean)

        # Coordinate descent
        hw, c5_cd = coord_descend(hw, n, budget_s - 40, max_rounds=8)

        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = hw.copy()

    # Phase 4: Pairwise swap refinement
    t_limit_swap = budget_s - 40
    for _r in range(30):
        if time.time() > t_limit_swap:
            break
        improved = False
        for _ in range(500):
            if time.time() > t_limit_swap:
                break
            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            if abs(best_h[i] - best_h[j]) < 1e-6:
                continue
            trial = best_h.copy()
            trial[i], trial[j] = best_h[j], best_h[i]
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-13:
                best_h = trial
                best_c5 = c
                improved = True
        # Single coordinate passes
        for j in range(n):
            if time.time() > t_limit_swap:
                break
            old = best_h[j]
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-8:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_h = trial
                    best_c5 = c
                    improved = True
        if not improved:
            break

    # Phase 5: Final SLSQP polish
    if time.time() < budget_s - 15:
        best_h = slsqp_optimize(best_h, n, 400, 120, _obj_full)
        c5_polish = compute_c5(best_h)
        if c5_polish < best_c5:
            best_c5 = c5_polish

    # Phase 6: Ultra-fine coordinate descent
    if time.time() < budget_s - 5:
        best_h, best_c5 = coord_descend(best_h, n, budget_s - 2, max_rounds=3)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
