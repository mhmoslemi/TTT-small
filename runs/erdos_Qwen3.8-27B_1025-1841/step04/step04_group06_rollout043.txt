```python
def fast_correlate(h):
    n = len(h)
    # Linear correlation via FFT
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fsize = 1 << (size - 1).bit_length()
    H = np.fft.rfft(h, fsize)
    V = np.fft.rfft(1 - h, fsize)
    # Correlation: ifft(fft(a) * conj(fft(v)))
    corr = np.fft.irfft(H * np.conj(V), fsize)[:size]
    return corr
```

```python
def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

    h_new = np.interp(new_x, old_x, h, period=2.0)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def make_binary_feasible(h, n):
    """Make h binary with exactly n/2 ones."""
    h = (h > 0.5).astype(float)
    n_target = n // 2
    n_ones = int(h.sum())
    if n_ones > n_target:
        idx = np.where(h == 1)[0]
        # Remove (n_ones - n_target) ones, preferring those with lower "importance"
        h[idx[:n_ones - n_target]] = 0
    elif n_ones < n_target:
        idx = np.where(h == 0)[0]
        h[idx[:n_target - n_ones]] = 1
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_optimize(h_start, n, T, mu, max_iter, time_limit, start_time):
    dx = 2.0 / n
    
    def f(h):
        if time.time() - start_time > time_limit:
            return 1e10
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        return make_feasible(res.x, n)
    except:
        return h_start
```

```python
    # Contribution from n=i (h changes)
    # Valid shifts: 0 <= i+shift < n, i.e., -i <= shift <= n-1-i
    s_min = max(-i, -(n-1))
    s_max = min(n-1-i, n-1)
    for s in range(s_min, s_max + 1):
        delta[s + n - 1] += h[i+s] - 1  # = -(1-h[i+s])
    
    # Contribution from n=j (h changes)
    s_min = max(-j, -(n-1))
    s_max = min(n-1-j, n-1)
    for s in range(s_min, s_max + 1):
        delta[s + n - 1] += (1 - h[j+s])
    
    # Contribution from n=i-s ((1-h) changes at position i)
    # Valid: 0 <= i-s < n and 0 <= i < n (always true)
    s_min = max(i - (n-1), -(n-1))
    s_max = min(i, n-1)
    for s in range(s_min, s_max + 1):
        delta[s + n - 1] += h[i-s]
    
    # Contribution from n=j-s ((1-h) changes at position j)
    s_min = max(j - (n-1), -(n-1))
    s_max = min(j, n-1)
    for s in range(s_min, s_max + 1):
        delta[s + n - 1] += -(h[j-s])
    
    # Overlap corrections
    # At shift=0: n=i appears in both h-role and (1-h)-role (as n=i-0=i)
    #   h-role: -(1-h[i]) = h[i]-1 = 0 (since h[i]=1)
    #   (1-h)-role: h[i-0] = h[i] = 1
    #   Sum: 0 + 1 = 1
    #   Correct: h'[i]*(1-h'[i]) - h[i]*(1-h[i]) = 0*1 - 1*0 = 0
    #   So we added 1 but should add 0. Correction: -1
    # Wait, let me recheck.
    #   The term at n=i, shift=0 is h[i]*(1-h[i]). Before: 1*0=0. After: 0*1=0. Change: 0.
    #   My formula gives: (h[i]-1) + h[i] = 0 + 1 = 1. But correct is 0. So correction = -1.
    
    #   Similarly for n=j, shift=0: h[j]*(1-h[j]). Before: 0*1=0. After: 1*0=0. Change: 0.
    #   My formula gives: (1-h[j]) + (-h[j]) = 1 + 0 = 1. But correct is 0. Correction: -1.
    
    delta[0 + n - 1] += -1  # correction for n=i at shift=0
    delta[0 + n - 1] += -1  # correction for n=j at shift=0
```
