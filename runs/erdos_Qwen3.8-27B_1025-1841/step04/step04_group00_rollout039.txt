pp_polish(h, n, T, maxiter=200):
    """SLSQP with log-sum-exp smooth max."""
    dx = 2.0 / n
    
    def obj(x):
        if np.any(x < -0.1) or np.any(x > 1.1):
            return 1.0  # penalty
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def obj_grad(x):
        xc = np.clip(x, 0, 1)
        corr = np.correlate(xc, 1.0 - xc, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        weights = np.exp(T * shifted)
        weights /= np.sum(weights)
        # Gradient of smooth max = weighted average of individual gradients
        grad = np.zeros(n)
        for idx, w in enumerate(weights):
            if w < 1e-15:
                continue
            l = idx - (n - 1)
            if l == 0:
                grad += w * (1 - 2 * xc)
            elif l > 0:
                j_range = np.arange(n - l)
                grad[j_range] += w * (1 - xc[j_range + l])
                grad[np.arange(l, n)] -= w * xc[np.arange(l, n) - l]
            else:
                l2 = -l
                grad[np.arange(l2, n)] += w * (1 - xc[np.arange(l2, n) - l2])
                grad[np.arange(n - l2)] -= w * xc[np.arange(n - l2) + l2]
        return grad * dx
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad, bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h):
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h, compute_c5(h)
