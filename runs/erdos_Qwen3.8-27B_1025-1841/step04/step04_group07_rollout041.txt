def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):

    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max(corr, T):
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(np.clip(T * (corr - mc), -50, 50)))) / T

def active_set_optimize(h0, n, n_iters=30, T_init=200, T_decay=0.7):
    """Optimize using active set + L-BFGS-B."""
    h = h0.copy()
    dx = 2.0 / n
    T = T_init
    for it in range(n_iters):
        if T < 1:
            T = 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        # Active set: lags where corr is in the top 20%
        thresh = mc - 0.15 * mc  # or some threshold
        active = (corr >= thresh).astype(float)
        active /= active.sum()
        mu = 1000.0

            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(active, c)) * dx + mu * (x.sum() - n/2.0)**2

                       bounds=[(0,1)]*n,
                       options={'maxiter': 50, 'ftol': 1e-14})
        h_new = make_feasible(res.x, n)
        if compute_c5(h_new) < compute_c5(h):
            h = h_new
        T *= T_decay
    return h

def binary_local_search(h_bin, n, rng, max_rounds=40, tries=3000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):

                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0

            break

def fast_correlate(a, b):
    """Compute linear cross-correlation using FFT. Matches np.correlate(a, b, 'full')."""
    L = len(a)  # assume len(a) == len(b)
    N = 1
    while N < 2*L - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    C = np.fft.irfft(fa * np.conj(fb), N)
    return np.concatenate([C[N-L+1:], C[:L]])
