def softmax_obj_grad(x):
    n = _G_N
    dx = _G_DX
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_TOP_K > 0:
        idx = np.argsort(-corr)[:_G_TOP_K]
        corr_use = corr[idx]
    else:
        corr_use = corr
        idx = np.arange(len(corr))
    mc = np.max(corr_use)
    shifted = corr_use - mc
    T = _G_T
    # Soft-max weights
    max_s = T * np.max(shifted)
    exps = np.exp(T * shifted - max_s)
    weights = exps / np.sum(exps)  # These are the soft-max weights (sum to 1)
    # d obj / d h[j] = dx * sum_k weights_k * d corr[idx_k]/d h[j]
    grad = np.zeros(n)
    # For each shift k in idx, compute d corr[k]/d h
    # d corr[k]/d h[j] = (1-h[j+shift_k])*I(j in S_k) - h[j-shift_k]*I(j-shift_k valid for S_k)
    js = np.arange(n)
    for ii in range(len(idx)):
        k = idx[ii]
        shift_k = k - (n - 1)
        w_k = weights[ii]
        # j in S_k: 0<=j<n and 0<=j+shift_k<n
        j_in_S = (js + shift_k >= 0) & (js + shift_k < n)
        # j-shift_k in S_k: 0<=j-shift_k<n (and j is already in [0,n), so j-shift_k+shift_k=j is valid)
        jm = js - shift_k
        jm_in_S = (jm >= 0) & (jm < n)
        grad[j_in_S] += w_k * (1.0 - x[js[j_in_S] + shift_k])
        grad[jm_in_S] -= w_k * x[jm[jm_in_S]]
    grad *= dx
    return grad

    return np.sum(x) - _G_N / 2.0

    return np.ones(_G_N)

                              options={'maxiter': 250, 'ftol': 1e-15})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_pts)
            except Exception:
                pass

    # Phase 7: Subgradient at current best
    if rem() > 120:
        n_cur = len(best_h)
        _G_N = n_cur
        h_sg = best_h.copy()
        step_sz = 0.01
        for it in range(4000):
            if rem() < 10:
                break
            corr = get_corr(h_sg)
            k_star = int(np.argmax(corr))
            c5_old = compute_c5(h_sg)
            h_new = subgradient_step(h_sg, k_star, n_cur, step_sz)
            h_new = make_feasible(h_new, n_cur)
            c5_new = compute_c5(h_new)
            if c5_new < c5_old:
                h_sg = h_new
                step_sz = min(step_sz * 1.02, 0.06)
            else:
                step_sz *= 0.95
        c5_sg = compute_c5(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 8: Ultra-fine coordinate descent
    if rem() > 20:
        n_cur = len(best_h)
        for rd in range(3):
            for j in range(n_cur):
                if rem() < 3:
                    break
                old = best_h[j]
                best_v = old
                best_c = best_c5
                for step in [0.0001, 0.0003, 0.0005, 0.001]:
                    for s in [-1.0, 1.0]:
                        v = old + s * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n_cur)
                            c = compute_c5(trial)
                            if c < best_c - 1e-16:
                                best_c = c
                                best_v = v
                if best_v != old:
                    best_h[j] = best_v
                    best_h = make_feasible(best_h, n_cur)
                    best_c5 = best_c

    # Phase 9: Final binary check
    if rem() > 15:
        n_cur = len(best_h)
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, n_cur)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h = h_bin
            best_c5 = c5_bin

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
