        h_r, c5_r = adam_phase(h_try, n, start_time, t_alloc, 0.03, 200.0, 1000.0, 4, 60, seed + restart * 100)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r
    
    # Phase 7: Ultra-fine coord descent
    if remaining() > 30:
        t_alloc = min(25, remaining() - 5)
        h, c5 = coord_descent_fast(best_h, n, start_time, t_alloc,
                                   [0.001, 0.0005, 0.0002])
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 8: Final SLSQP
    if remaining() > 40:
        try:
            _N = n
            from scipy.optimize import minimize
            bounds = [(0.0, 1.0)] * n
            res = minimize(_slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': _slsqp_con}],
                           options={'maxiter': 500, 'ftol': 1e-20})
            h_res = project_feasible(res.x, n)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_h, best_c5 = h_res, c5_res
        except Exception:
            pass
    
    # Final projection and evaluation
    best_h = project_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
