    h = np.clip(h.astype(float), 0.0, 1.0)

        if cnt == 0:

                    b[i], b[j] = 1, 0
                no_improve += 1
                continue
            
            # Find i where b[i]=0, b[i+k]=1 (setting b[i]=1 reduces f(k))
            mask_a = (b[valid] == 0) & (b[valid + k] == 1)
            cands_a = valid[mask_a]
            # Find i where b[i]=1, b[i+k]=1 (setting b[i+k]=0 reduces f(k))
            mask_b = (b[valid] == 1) & (b[valid + k] == 1)
            cands_b = valid[mask_b]
            
            if len(cands_a) + len(cands_b) == 0:
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0, 1
                new_c5 = float(np.max(get_corr(b)) * dx)
                if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                    curr_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_b = b.copy()
                        no_improve = 0
                else:
                    b[i], b[j] = 1, 0
                no_improve += 1
                continue
            
            if len(cands_a) > 0 and (len(cands_b) == 0 or rng.random() < 0.5):
                i = cands_a[rng.randint(len(cands_a))]
                # Set b[i]=1, need to undo a 1 elsewhere
                ones = np.where(b == 1)[0]
                ones = ones[ones != i + k]  # avoid undoing the one we're using
                if len(ones) == 0:
                    ones = np.where(b == 1)[0]
                j = ones[rng.randint(len(ones))]
                if j == i:
                    j = zeros[rng.randint(len(zeros))]
                b[i] = 1
                b[j] = 0
            else:
                i = cands_b[rng.randint(len(cands_b))]
                # Set b[i+k]=0, need to set another 0 to 1
                zeros = np.where(b == 0)[0]
                zeros = zeros[zeros != i]
                if len(zeros) == 0:
                    zeros = np.where(b == 0)[0]
                j = zeros[rng.randint(len(zeros))]
                b[i + k] = 0
                b[j] = 1
        elif move_type < 0.8:
            # Block swap
            bsize = rng.randint(2, max(3, n // 40))
            start = rng.randint(0, n - bsize)
            seg = b[start:start + bsize]
            no_ones = np.sum(seg)
            no_zeros = bsize - no_ones
            if no_ones == 0 or no_zeros == 0:
                no_improve += 1
                continue
            n_swap = min(no_ones, no_zeros, max(1, rng.randint(1, max(2, bsize // 3))))
            oi = np.where(seg == 1)[0]
            zi = np.where(seg == 0)[0]
            co = rng.choice(oi, size=n_swap, replace=False)
            cz = rng.choice(zi, size=n_swap, replace=False)
            b[start + co] = 0
            b[start + cz] = 1
            new_c5 = float(np.max(get_corr(b)) * dx)
            if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[start + co] = 1
                b[start + cz] = 0
                no_improve += 1
        else:
            # Two-pair swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            idx1 = rng.choice(len(ones), size=2, replace=False)
            idx2 = rng.choice(len(zeros), size=2, replace=False)
            i1, i2 = ones[idx1[0]], ones[idx1[1]]
            j1, j2 = zeros[idx2[0]], zeros[idx2[1]]
            if len({i1, i2, j1, j2}) < 4:
                no_improve += 1
                continue
            b[i1] = 0
            b[i2] = 0
            b[j1] = 1
            b[j2] = 1
            new_c5 = float(np.max(get_corr(b)) * dx)
            if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                b[i1] = 1
                b[i2] = 1
                b[j1] = 0
                b[j2] = 0
                no_improve += 1
        
        if no_improve > 4000:
            b = best_b.copy()
            curr_c5 = best_c5
            no_improve = 0
            npert = rng.randint(3, max(4, int(0.02 * n)))
            for _ in range(npert):
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i], b[j] = 0, 1
    
    return best_b, best_c5

def _SL_N = None
def _SL_T = None

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter=200, ftol=1e-14):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def multi_temp_slsqp(h, n, start_time, budget_s):
    for T in [50.0, 200.0, 800.0, 3000.0, 10000.0, 50000.0]:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        try:
            x_res = slsqp_optimize(h, n, T, 200, 1e-15)
            c5_res = compute_c5(x_res)
            cur_c5 = compute_c5(h)
            if c5_res < cur_c5:
                h = make_feasible(x_res, n)
        except Exception:
            pass
    return h, compute_c5(h)

def gen_thue_morse(n, target):
    b = np.zeros(n, dtype=float)
    for i in range(n):
        b[i] = bin(i).count('1') % 2
    # Adjust to have exactly target ones
    ones = np.sum(b)
    if ones > target:
        excess = ones - target
        idx = np.where(b == 1)[0]
        to_flip = idx[:int(excess)]
        b[to_flip] = 0
    elif ones < target:
        deficit = target - ones
        idx = np.where(b == 0)[0]
        to_flip = idx[:int(deficit)]
        b[to_flip] = 1
    return b

def gen_periodic(n, target, period):
    b = np.zeros(n, dtype=float)
    half = period // 2
    for i in range(n):
        b[i] = 1.0 if (i % period) < half else 0.0
    # Adjust
    diff = target - np.sum(b)
    if diff > 0:
        zeros = np.where(b == 0)[0]
        b[zeros[:int(diff)]] = 1
    elif diff < 0:
        ones = np.where(b == 1)[0]
        b[ones[:int(-diff)]] = 0
    return b

def gen_beatty(n, target, alpha):
    b = np.zeros(n, dtype=float)
    positions = np.floor(np.arange(target) * alpha + 0.5).astype(int)
    positions = positions[positions < n]
    b[positions] = 1.0
    diff = target - np.sum(b)
    if diff > 0:
        zeros = np.where(b == 0)[0]
        b[zeros[:int(diff)]] = 1
    elif diff < 0:
        ones = np.where(b == 1)[0]
        b[ones[:int(-diff)]] = 0
    return b

def fourier_perturb(h, n, rng, n_modes=4, amp=0.05):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n0 = len(h_init)
    n0 = 960
    target = n0 // 2
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Generate diverse initial binary solutions
    b_parent = (h_init >= 0.5).astype(float)
    diff = target - np.sum(b_parent)
    if diff > 0:
        zeros = np.where(b_parent == 0)[0]
        b_parent[zeros[:int(diff)]] = 1
    elif diff < 0:
        ones = np.where(b_parent == 1)[0]
        b_parent[ones[:int(-diff)]] = 0
    
    initials = [b_parent.copy()]
    initials.append(gen_thue_morse(n0, target))
    for period in [16, 20, 24, 30, 32, 40, 48]:
        initials.append(gen_periodic(n0, target, period))
    for alpha in [2.0, 2.5, 3.0, 1 + np.sqrt(5)]:
        initials.append(gen_beatty(n0, target, alpha))
    # Random structured
    for _ in range(3):
        perm = rng.permutation(n0)
        b_rand = np.zeros(n0, dtype=float)
        b_rand[perm[:target]] = 1.0
        initials.append(b_rand)
    
    # Evaluate all initials
    best_init_b = None
    best_init_c5 = float('inf')
    for b in initials:
        c = float(np.max(get_corr(b)) * (2.0 / n0))
        if c < best_init_c5:
            best_init_c5 = c
            best_init_b = b.copy()
    
    if best_init_c5 < best_c5:
        best_c5 = best_init_c5
        best_h = best_init_b.astype(float).copy()
    
    # Phase 2: Lag-aware SA on best initial, then others
    remaining = budget_s - (time.time() - start_time)
    sa_budget_each = remaining * 0.7 / len(initials)
    
    sa_starts = sorted(initials, key=lambda b: float(np.max(get_corr(b)) * (2.0/n0)))
    
    for b_init in sa_starts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 60:
            break
        tl = min(sa_budget_each, remaining - 60)
        if tl < 20:
            tl = 20
        b_res, c5_res = sa_lag_aware(b_init.copy(), n0, 200000, 0.004, 0.000005,
                                      rng, tl, start_time)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.astype(float).copy()
    
    # Phase 3: SLSQP polish on best binary solution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        h_cont = best_h.astype(float).copy()
        h_cont, c5_polish = multi_temp_slsqp(h_cont, n0, start_time, budget_s)
        if c5_polish < best_c5:
            best_c5 = c5_polish
            best_h = h_cont.copy()
    
    # Phase 4: More SA on the polished solution (rounded)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        b_polished = (best_h >= 0.5).astype(float)
        diff = target - np.sum(b_polished)
        if diff > 0:
            zeros = np.where(b_polished == 0)[0]
            b_polished[zeros[:int(diff)]] = 1
        elif diff < 0:
            ones = np.where(b_polished == 1)[0]
            b_polished[ones[:int(-diff)]] = 0
        tl = min(remaining - 40, 200)
        if tl > 30:
            b_res, c5_res = sa_lag_aware(b_polished.copy(), n0, 300000, 0.003, 0.000003,
                                          rng, tl, start_time)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = b_res.astype(float).copy()
    
    # Phase 5: Final SLSQP + coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        try:
            global _SL_N, _SL_T
            _SL_N = n0
            _SL_T = 20000.0
            x_res = slsqp_optimize(best_h, n0, 20000.0, 300, 1e-16)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass
    
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        # Coordinate descent
        h_cd = best_h.astype(float).copy()
        dx = 2.0 / n0
        for rnd in range(10):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break
            improved = False
            for j in range(n0):
                if (time.time() - start_time) > budget_s - 5:
                    break
                old = h_cd[j]
                best_val = old
                best_c = compute_c5(h_cd)
                for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:
                    if abs(cand - old) < 1e-10:
                        continue
                    trial = h_cd.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n0)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_val = cand
                if best_val != old:
                    h_cd[j] = best_val
                    h_cd = make_feasible(h_cd, n0)
                    improved = True
            if not improved:
                break
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
