    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -70, 70)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_opt(h0, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_obj(corr, T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-13, 'maxls': 40})
    return make_feasible(res.x, n)

def coord_descent_fine(h0, n, max_rounds=8, steps=None):
    if steps is None:
        steps = [0.1, -0.1, 0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-15:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def two_swap_search(h0, n, rng, max_trials=50000):
    h = h0.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.7)[0].tolist()
    zeros = np.where(h < 0.3)[0].tolist()
    if len(ones) < 2 or len(zeros) < 2:
        return h, cur
    trials = 0
    while trials < max_trials:
        i1 = rng.randint(len(ones))
        i2 = rng.randint(len(ones))
        if i1 == i2:
            continue
        j1 = rng.randint(len(zeros))
        j2 = rng.randint(len(zeros))
        if j1 == j2:
            continue
        a, b = ones[i1], ones[i2]
        c, d = zeros[j1], zeros[j2]
        if a == c or a == d or b == c or b == d:
            continue
        h[a], h[b] = h[a] - 1.0, h[b] - 1.0
        h[c], h[d] = h[c] + 1.0, h[d] + 1.0
        h = np.clip(h, 0.0, 1.0)
        c_new = compute_c5(h)
        if c_new < cur - 1e-15:
            cur = c_new
            ones[i1] = c
            ones[i2] = d
            zeros[j1] = a
            zeros[j2] = b
        else:
            h[a], h[b] = h[a] + 1.0, h[b] + 1.0
            h[c], h[d] = h[c] - 1.0, h[d] - 1.0
            h = np.clip(h, 0.0, 1.0)
        trials += 1
    return h, cur

def one_swap_search(h0, n, rng, max_trials=80000):
    h = h0.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    if not ones or not zeros:
        return h, cur
    trials = 0
    while trials < max_trials and len(ones) > 0 and len(zeros) > 0:
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        c = compute_c5(h)
        if c < cur - 1e-15:
            cur = c
            ones[ii] = oj
            zeros[jj] = oi
        else:
            h[oi] = 1.0
            h[oj] = 0.0
        trials += 1
    return h, cur

def targeted_local_search(h0, n, rng, max_rounds=30):
    """Focus on the shifts that achieve the max, try to reduce them."""
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        threshold = mc - 0.01 * mc
        active_shifts = np.where(corr > threshold)[0]
        improved = False
        for _ in range(200):
            j = rng.randint(n)
            d = rng.choice([0.05, -0.05, 0.1, -0.1, 0.02, -0.02])
            nv = h[j] + d
            if nv < 0.0 or nv > 1.0:
                continue
            trial = h.copy()
            trial[j] = nv
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < cur - 1e-16:
                cur = c
                h = trial
                improved = True
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = len(h0)

    # Phase 1: Tight L-BFGS-B at initial resolution
    n1 = best_n
    h1 = best_h.copy()
    for T in [30, 80, 200, 500]:
        if rem() < 40:
            break
        h1 = lbgfb_opt(h1, n1, T, mu=1500, max_iter=400)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()

    # Phase 2: Upsample to 1280 and optimize
    if rem() > 120:
        n2 = 1280
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        for T in [50, 150, 500, 1200]:
            if rem() < 60:
                break
            h2 = lbgfb_opt(h2, n2, T, mu=800, max_iter=350)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h = c2, h2.copy()

    # Phase 3: Upsample to 2560
    if rem() > 150:
        n3 = 2560
        h3 = upsample_h(best_h, n3)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
        for T in [100, 400, 1000]:
            if rem() < 60:
                break
            h3 = lbgfb_opt(h3, n3, T, mu=500, max_iter=250)
            c3 = compute_c5(h3)
            if c3 < best_c5:
                best_c5, best_h = c3, h3.copy()

    # Phase 4: Coordinate descent with fine steps on best resolution
    if rem() > 60:
        nc = len(best_h)
        h_cd, c_cd = coord_descent_fine(best_h, nc, max_rounds=6,
                                         steps=[0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 5: One-swap and two-swap local search
    if rem() > 60:
        ns = len(best_h)
        # Work with a slightly binarized version for swap search
        h_bin = (best_h > 0.5).astype(float)
        target_ones = ns // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            order = np.argsort(h_bin[idx])
            h_bin[idx[order[:n_ones - target_ones]]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            order = np.argsort(-h_bin[idx])
            h_bin[idx[order[:target_ones - n_ones]]] = 1.0
        h_s1, c_s1 = one_swap_search(h_bin, ns, rng, max_trials=100000)
        if c_s1 < best_c5:
            best_c5, best_h = c_s1, h_s1.copy()
        if rem() > 40:
            h_s2, c_s2 = two_swap_search(best_h, ns, rng, max_trials=60000)
            if c_s2 < best_c5:
                best_c5, best_h = c_s2, h_s2.copy()

    # Phase 6: Targeted local search on the final best
    if rem() > 40:
        nt = len(best_h)
        h_t, c_t = targeted_local_search(best_h, nt, rng, max_rounds=20)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t.copy()

    # Phase 7: Final fine coordinate descent
    if rem() > 20:
        nf = len(best_h)
        h_f, c_f = coord_descent_fine(best_h, nf, max_rounds=4,
                                       steps=[0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.002, -0.002])
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
