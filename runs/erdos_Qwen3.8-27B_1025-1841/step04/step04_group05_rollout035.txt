import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-14:
        return np.clip(h, 0, 1)
    # Use bisection on lambda
    lo, hi = -2.0, 2.0

        if cnt > 0:
            result[mask] = np.clip(result[mask] + diff/cnt, 0, 1)
    return result

def adam_optimize(h0, n, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, 
                  n_steps=3000, time_budget=None, start_time=None):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for t in range(1, n_steps + 1):
        if time_budget is not None and start_time is not None:
            if time.time() - start_time > time_budget:
                break
        # Compute subgradient
        v_arr = 1.0 - h
        corr = np.correlate(h, v_arr, mode='full')
        k_star = np.argmax(corr)

        offset = k_star - (n - 1)
        grad = np.zeros(n)

        j_min = max(0, -offset)
        j_max = min(n - 1, n - 1 - offset)

        i_twos = np.arange(max(0, j_min + offset), min(n - 1, j_max + offset) + 1)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    result = np.clip(h, 0.0, 1.0)
    s = np.sum(result)
    if abs(s - target) < 1e-13:
        return result
    # Bisection
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_p = np.clip(h + mid, 0.0, 1.0)
        if np.sum(h_p) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += diff / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def subgrad_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    grad = np.zeros(n)

    n_max = min(n - 1, n - 1 + lag)

        idx2 = np.arange(i_min2, i_max2 + 1)
        shifted2 = idx2 + lag
        grad[idx2] -= h[shifted2]
    
    return grad * dx

def adam_optimize(h0, n, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,
                  n_steps=5000, deadline=None):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    patience = 0
    patience_limit = 1000
    
    for t in range(1, n_steps + 1):
        if deadline is not None and time.time() > deadline:
            break
        # Decay learning rate
        lr_t = lr * (0.9995 ** (t // 100))
        
        grad = subgrad_c5(h, n)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                break
    
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return project_feasible(h_new, new_n)

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    result = np.clip(h, 0.0, 1.0)
    s = np.sum(result)
    if abs(s - target) < 1e-13:
        return result
    lo, hi = -1.5, 1.5
    for _ in range(70):
        mid = (lo + hi) / 2.0
        hp = np.clip(h + mid, 0.0, 1.0)
        if np.sum(hp) > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] = np.clip(result[mask] + diff / cnt, 0.0, 1.0)
    return result

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return project_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return project_feasible(h_new, new_n)

def softmax_gradient(h, n, T, K=30):
    """Compute gradient of soft-max smoothed C5 using top-K lags."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top K indices
    if K >= len(corr):
        top_idx = np.argsort(corr)[::-1]
    else:
        top_idx = np.argpartition(corr, -K)[-K:]
        top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]
    # Softmax weights over top K
    c_top = corr[top_idx]
    c_max = c_top[0]
    exp_vals = np.exp(T * (c_top - c_max))
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for k_idx, w in zip(top_idx, weights):
        lag = int(k_idx) - (n - 1)
        n_min = max(0, -lag)
        n_max = min(n - 1, n - 1 - lag)
        if n_min > n_max:
            continue
        # Term 1: (1-h[i-lag]) for i in [n_min, n_max] with i-lag in [0,n-1]
        idx1 = np.arange(n_min, n_max + 1)
        s1 = idx1 - lag
        v1 = (s1 >= 0) & (s1 < n)
        if np.any(v1):
            grad[idx1[v1]] += w * (1.0 - h[s1[v1]])
        # Term 2: -h[i+lag] for i where i+lag in [n_min, n_max] and 0<=i+lag<n
        i_lo = max(0, n_min - lag)
        i_hi = min(n - 1, n_max - lag)
        if i_lo <= i_hi:
            idx2 = np.arange(i_lo, i_hi + 1)
            s2 = idx2 + lag
            grad[idx2] -= w * h[s2]
    return grad * dx

def adam_softmax(h0, n, lr=0.03, T=10.0, K=30, beta1=0.9, beta2=0.999,
                 eps=1e-8, n_steps=5000, deadline=None, T_decay=1.0):
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for t in range(1, n_steps + 1):
        if deadline is not None and time.time() > deadline:
            break
        T_t = T * (T_decay ** (t / 500.0))
        lr_t = lr * (0.999 ** (t // 200))
        
        grad = softmax_gradient(h, n, T_t, K)
        
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h_new, n)
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def subgrad_adam(h0, n, lr=0.05, beta1=0.9, beta2=0.999, eps=1e-8,

    best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        
        grad = np.zeros(n)
        n_min = max(0, -lag)
        n_max = min(n - 1, n - 1 - lag)
        
        idx1 = np.arange(n_min, n_max + 1)

        v1 = (s1 >= 0) & (s1 < n)

        i_lo = max(0, n_min - lag)
        i_hi = min(n - 1, n_max - lag)

            s2 = idx2 + lag
            grad[idx2] -= h[s2]

        m_hat = m / (1.0 - beta1 ** t)

        c5 = compute_c5(h)
