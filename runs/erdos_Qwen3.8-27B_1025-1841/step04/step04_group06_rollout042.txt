def make_feasible(h, n):

    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n

    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=200, n_tries=2000):
    h = h_bin.copy().astype(float)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    no_improve = 0

    for rnd in range(max_rounds):
        improved = False
        nt = min(n_tries, max(1, len(ones) * len(zeros)))
        for _ in range(nt):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current_c5 - 1e-14:
                current_c5 = c
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
                no_improve = 0
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            no_improve += 1
            if no_improve >= 20:
                break
    return best_h, best_c5

def simulated_annealing_binary(h_bin, n, rng, T_start=1.0, T_end=0.01, max_rounds=500):
    """Binary simulated annealing to escape local optima."""
    h = h_bin.copy().astype(float)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()

    for rnd in range(max_rounds):
        T = T_start * (T_end / T_start) ** (rnd / max_rounds)
        n_tries = 1000
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            delta = c - current_c5
            if delta < 0 or rng.rand() < np.exp(-delta / (T * dx_scale)):
                current_c5 = c
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
            else:
                h[oi] = 1.0
                h[oj] = 0.0
    return best_h, best_c5

def slsqp_polish(h_start, n, T, max_iter):
    dx = 2.0 / n

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    def con(h):
        return np.sum(h) - n / 2.0

    def jac_con(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac_con}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def coordinate_descent_polish(h, n, max_rounds=10):
    """Coordinate descent with binary and small perturbations."""
    h = h.copy()
    current = compute_c5(h)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 1.0, 0.0]
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < -1e-10 or nv > 1 + 1e-10:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-13:
                    h = tr
                    current = c
                    improved = True
                    break
        if not improved:
            break
    return h, current

def exp_weights_minmax(h_start, n, eta, n_outer=30, n_inner_iters=20):
    """
    Exponential weights method for the min-max problem.
    Maintains weights over correlation lags, alternates:
    1. Minimize weighted sum of correlations (via gradient steps)
    2. Update weights via exponential rule
    """
    dx = 2.0 / n
    m = 2 * n - 1  # number of lags
    w = np.ones(m) / m
    h = h_start.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(n_outer):
        # Step 1: minimize weighted objective using a few gradient steps
        for _ in range(n_inner_iters):
            corr = np.correlate(h, 1.0 - h, mode='full')
            # Compute gradient of weighted sum: sum_k w[k] * corr[k]
            # corr[k] = sum_i h[i] * (1 - h[i+k-n+1])
            # d/dh[j] corr[k] = (1-h[j+k-n+1]) - h[j-k+n-1] (with bounds checking)
            # Do it via: gradient[j] = sum_k w[k] * [(1-h[j+k-n+1]) - h[j-k+n-1]]
            # This is a convolution-like operation
            g = np.zeros(n)
            # For each j, g[j] = sum_k w[k] * (1-h[(j+k-n+1)%n]) - sum_k w[k] * h[(j-k+n-1)%n]
            # But we need to be careful with the "full" mode indexing
            # Let's just use the formula: g = w_convolved_with_1h - w_convolved_with_h (shifted)
            # Actually for full mode: corr[m] = sum_i h[i] * g[i+m-n+1] where g=1-h, valid when 0<=i+m-n+1<n
            # d corr[m]/d h[j] = g[j+m-n+1] (if valid) + h[j-m+n-1] * (-1) (if valid, from the h[j] term in another sum)
            # This is getting complex. Let me just use finite differences for the gradient.
            # For efficiency, use the analytical form:
            # Let's define: for the "full" correlation of a and b:
            # corr_full[m] = sum_{i} a[i] * b[i + m - (len(b)-1)]
            # Here a=h, b=1-h, both length n
            # corr_full[m] = sum_i h[i] * (1-h[i+m-n+1])
            # d/dh[j]: term 1: (1-h[j+m-n+1]) for each m where 0<=j+m-n+1<n
            #           term 2: -h[j] * (d/dh[j])(1-h[i+m-n+1]) for each i where i+m-n+1=j, i.e., m = j-i+n-1
            #                   = -sum_i [i+m-n+1==j] * (contribution) 
            # This is O(n*m) = O(n^2) which is the same as correlate. Let's just do it.
            
            # Simpler: use the fact that the gradient of sum_k w[k]*corr[k] 
            # can be computed as a cross-correlation
            # grad = correlate(w, 1-h) - correlate(w, h) (with appropriate shifts)
            # Actually: sum_k w[k] * corr[k] = sum_k w[k] * sum_i h[i]*(1-h[i+k-n+1])
            # = sum_i h[i] * sum_k w[k]*(1-h[i+k-n+1])
            # = sum_i h[i] * (sum_k w[k] - sum_k w[k]*h[i+k-n+1])
            # The gradient w.r.t h[j] involves:
            # From the direct term: sum_k w[k]*(1-h[j+k-n+1])
            # From the "h in (1-h[...])" term: for each i, when i+k-n+1 = j, we get -h[i]*w[k]
            #   i.e., sum_{i: i+k-n+1=j for some k} -h[i]*w[j-i+n-1] = sum_i -h[i]*w[j-i+n-1]
            
            # Let me just compute it numerically with a small step for simplicity
            pass
        
        # For simplicity in this version, just use L-BFGS-B on the weighted objective
        def weighted_obj(hv):
            corr = np.correlate(hv, 1.0 - hv, mode='full')
            return float(np.dot(w, corr) * dx)
        
        def weighted_grad(hv):
            # Gradient via cross-correlation
            # d/dh[j] [sum_m w[m] * sum_i h[i]*(1-h[i+m-n+1])]
            # = sum_m w[m]*(1-h[j+m-n+1]) - sum_m w[m]*h[j]*[j appears in h[i+m-n+1]]
            # = conv(w, 1-h)[j] - h[j] * conv(w, 1)[j] (approximately, with boundary handling)
            # For full mode: let's define shifted_w to handle boundaries
            # Actually, the cleanest way: 
            # sum_m w[m] * h[j] * (1-h[j+m-n+1]) contributes w[m]*(1-h[j+m-n+1]) to grad[j]
            # sum_m w[m] * h[i] * (1-h[i+m-n+1]) where i+m-n+1 = j
            #   => i = j-m+n-1, so m = j-i+n-1
            #   => sum_i h[i] * w[j-i+n-1] * (-1)
            # So grad[j] = sum_{valid m} w[m]*(1-h[j+m-n+1]) - sum_{valid i} h[i]*w[j-i+n-1]
            
            grad = np.zeros(n)
            # First term: for each j, sum over m where 0<=j+m-n+1<n => max(0,n-1-j) <= m <= min(2n-2, 2n-2-j)
            #   Actually j+m-n+1 in [0,n-1] => n-1-j <= m <= n-1-j+n-1 => m in [max(0,n-1-j), min(2n-2, 2n-2-j)]
            # This is just a convolution of w with (1-h), restricted to valid range
            # Let's use np.correlate for this
            # sum_m w[m]*(1-h[j+m-n+1]) is like a cross-correlation
            # Define: for the "full" correlation of w (length m=2n-1) and (1-h) (length n):
            #   This doesn't directly work since they have different lengths
            
            # Let me just do it with FFT or direct O(n^2) for now
            # For n up to 2048, O(n^2) is ~4M which is fine
            one_minus_h = 1.0 - hv
            # First term: grad_first[j] = sum_m w[m] * one_minus_h[(j+m-n+1) % n] but only valid indices
            # Actually let's be more careful. In full mode:
            # corr[m] = sum_{i: 0<=i<n, 0<=i+m-n+1<n} h[i] * (1-h[i+m-n+1])
            # d dm corr[m] / d h[j] has two contributions:
            # (a) h[j] appears as the first factor: (1-h[j+m-n+1]) if 0<=j+m-n+1<n
            # (b) h[j] appears in (1-h[i+m-n+1]) when i+m-n+1=j, i.e., m=j-i+n-1: contribution -h[i]
            
            # (a): sum_{m: 0<=j+m-n+1<n, 0<=m<2n-1} w[m] * (1-h[j+m-n+1])
            #     Let k = j+m-n+1, so m = k-j+n-1, k ranges over [0,n-1]
            #     m valid: 0<=k-j+n-1<2n-1 => j-n+1<=k<=j+n-2, intersect [0,n-1]
            #     = sum_{k in [max(0,j-n+1), min(n-1,j+n-2)]} w[k-j+n-1] * (1-h[k])
            # (b): sum_{i: 0<=j-i+n-1<2n-1, 0<=i<n} w[j-i+n-1] * h[i] * (-1)
            #     m = j-i+n-1, valid: 0<=j-i+n-1<2n-1 => i-j-1 < n-1 and i > j-n+1
            #     i in [max(0, j-n+2), min(n-1, j+n-2)]... let me just bound by [0,n-1]
            #     = -sum_{i in [0,n-1]} w[j-i+n-1] * h[i] (where w index is valid)
            
            # For simplicity and correctness, let me just use a different approach:
            # Compute the gradient by noting that:
            # L = sum_m w[m] * corr[m] = sum_m w[m] * sum_i h[i]*(1-h[i+m-n+1])
            # Let's change variable: let p = i+m-n+1 (so m = p-i+n-1)
            # L = sum_p sum_i h[i]*(1-h[p]) * w[p-i+n-1] (with valid ranges)
            #   = sum_p (1-h[p]) * (sum_i h[i] * w[p-i+n-1])
            # = sum_p (1-h[p]) * (h conv w)[p]  (with appropriate shift)
            
            # So dL/dh[j] = from the (1-h[p]) term: -(h conv w)[j]
            #              from the h[i] term: sum_p (1-h[p]) * w[p-j+n-1]
            
            # (h conv w)[j] = sum_i h[i] * w[j-i+n-1] (with valid w indices)
            # (1-h conv w)[j] = sum_p (1-h[p]) * w[j-p+n-1] = sum_p (1-h[p]) * w[p-j+n-1]... 
            # wait, I need to be careful about the convolution direction.
            
            # Let me define: (f * w)[j] = sum_k f[k] * w[j-k] (standard convolution)
            # But our w has length 2n-1 and f has length n, so this is valid.
            
            # From the formula: dL/dh[j] = (1-h) * w convolved at j - h * w convolved at j
            # where the convolution is: (f*w)[j] = sum_{valid k} f[k]*w[j-k]
            
            # Let's just compute it directly:
            # term1[j] = sum_p (1-h[p]) * w[p-j+n-1] for valid p and valid w-index
            # term2[j] = sum_i h[i] * w[j-i+n-1] for valid i and valid w-index
            # grad[j] = term1[j] - term2[j]
            
            # Using np.convolve or manual:
            # term1 = correlate(1-h, w) ... but w is length 2n-1, 1-h is length n
            # np.correlate(1-h, w, 'full') would give length n+(2n-1)-1 = 3n-2, not what we want
            # Let me just do it with a loop for correctness (n is at most 2048, so n^2 = 4M is fine)
            
            # Actually, let me use a cleaner formulation:
            # L = sum_m w[m] * corr[m]
            # corr[m] = sum_i h[i] * (1-h[i+m-n+1]) (full mode)
            # Let's define the "effective" gradient by linearity:
            # L = sum_m w[m] * sum_i h[i] * (1-h[i+m-n+1])
            # = sum_i h[i] * sum_m w[m] * (1-h[i+m-n+1])
            # Let S[i] = sum_m w[m] * (1-h[i+m-n+1]) (sum over valid m)
            # Then L = sum_i h[i] * S[i]
            # But S[i] also depends on h! So:
            # dL/dh[j] = S[j] + sum_i h[i] * dS[i]/dh[j]
            # dS[i]/dh[j] = -sum_m w[m] * [i+m-n+1 == j] = -w[j-i+n-1] (if valid)
            # So dL/dh[j] = S[j] - sum_i h[i] * w[j-i+n-1]
            
            # S[j] = sum_{m: 0<=j+m-n+1<n} w[m] * (1-h[j+m-n+1])
            #      = sum_{k: 0<=k<n, 0<=k-j+n-1<2n-1} w[k-j+n-1] * (1-h[k])
            # Let k' = k, then m = k-j+n-1
            # Valid m: 0<=k-j+n-1<2n-1 => j-n+1<=k<=j+n-1, intersect [0,n-1]
            
            # And sum_i h[i]*w[j-i+n-1]: let i'=i, m=j-i+n-1
            # Valid m: 0<=j-i+n-1<2n-1 => i-n+1<j-i+n-1<... => i in [j-n+2, j+n] intersect [0,n-1]
            # Wait: 0<=j-i+n-1 => i<=j+n-1; j-i+n-1<2n-1 => i>j-n => i>=j-n+1
            # So i in [max(0, j-n+1), min(n-1, j+n-1)]
            
            grad = np.zeros(n)
            for j in range(n):
                t1 = 0.0
                t2 = 0.0
                lo = max(0, j - n + 1)
                hi = min(n - 1, j + n - 2)
                for k in range(lo, hi + 1):
                    m_idx = k - j + n - 1
                    if 0 <= m_idx < m:
                        t1 += w[m_idx] * (1.0 - hv[k])
                        t2 += hv[k] * w[m_idx]
                grad[j] = t1 - t2
            return grad
        
        # This O(n^2) gradient is too slow for n=2048 with many iterations
        # Let's skip this and just use the simpler approaches
        # Update weights
        corr = np.correlate(h, 1.0 - h, mode='full')
        w = w * np.exp(eta * corr)
        w = w / w.sum()
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.zeros(640)
        h0[::2] = 1.0

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Upsample and L-BFGS-B optimization at moderate resolution
    n1 = 1024
    h_w = upsample_h(best_h, n1)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()

    # Multiple temperature L-BFGS-B runs
    temp_list = [30, 80, 200, 500, 1200]
    for T in temp_list:
        if tl() < 30:
            break
        h_init = h_w.copy()
        try:
            h_opt = lbgfb_smooth(h_init, n1, T, mu=500.0, max_iter=300)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h_w = h_opt.copy()
        except Exception:
            pass

    # Random restart
    if tl() > 60:
        h_r = h_w + 0.3 * rng.randn(n1)
        h_r = make_feasible(h_r, n1)
        try:
            h_opt = lbgfb_smooth(h_r, n1, 200, mu=500.0, max_iter=250)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass

    # Phase 2: Binary local search
    if tl() > 60:
        h_bin = (h_w > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_target = n1 // 2
        if n_ones > n_target:
            one_idx = np.where(h_bin == 1)[0]
            order = rng.permutation(len(one_idx))
            for k in range(n_ones - n_target):
                h_bin[one_idx[order[k]]] = 0.0
        elif n_ones < n_target:
            zero_idx = np.where(h_bin == 0)[0]
            order = rng.permutation(len(zero_idx))
            for k in range(n_target - n_ones):
                h_bin[zero_idx[order[k]]] = 1.0

        h_bs, c5_bs = binary_local_search(h_bin, n1, rng, max_rounds=150, n_tries=3000)
        if c5_bs < best_c5:
            best_c5, best_h = c5_bs, h_bs.copy()

    # Phase 3: Try higher resolution
    for n_pts in [1536, 2048]:
        if tl() < 80:
            break
        h_c = upsample_h(best_h, n_pts)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()

        for T in [50, 200, 600, 2000]:
            if tl() < 30:
                break
            try:
                h_opt = lbgfb_smooth(h_c, n_pts, T, mu=200.0, max_iter=200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass

        # Binary search at this resolution if time permits
        if tl() > 80:
            h_bin2 = (h_c > 0.5).astype(float)
            n_ones2 = int(np.sum(h_bin2))
            n_target2 = n_pts // 2
            if n_ones2 > n_target2:
                oi = np.where(h_bin2 == 1)[0]
                o = rng.permutation(len(oi))
                for k in range(n_ones2 - n_target2):
                    h_bin2[oi[o[k]]] = 0.0
            elif n_ones2 < n_target2:
                zi = np.where(h_bin2 == 0)[0]
                o = rng.permutation(len(zi))
                for k in range(n_target2 - n_ones2):
                    h_bin2[zi[o[k]]] = 1.0
            h_b2, c5_b2 = binary_local_search(h_bin2, n_pts, rng, max_rounds=80, n_tries=2000)
            if c5_b2 < best_c5:
                best_c5, best_h = c5_b2, h_b2.copy()

    # Phase 4: SLSQP polish
    if tl() > 30:
        n_s = min(len(best_h), 1024)
        h_s = upsample_h(best_h, n_s) if n_s < len(best_h) else best_h.copy()
        for T in [500, 3000]:
            if tl() < 15:
                break
            try:
                h_opt = slsqp_polish(h_s, n_s, T, max_iter=100)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_s = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Coordinate descent
    if tl() > 20:
        n_cd = min(len(best_h), 1024)
        h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        h_cd, c5_cd = coordinate_descent_polish(h_cd, n_cd, max_rounds=5)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 6: One more L-BFGS-B at best resolution with high temperature
    if tl() > 20:
        n_f = len(best_h)
        try:
            h_opt = lbgfb_smooth(best_h, n_f, 5000, mu=500.0, max_iter=200)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass

    # Phase 7: Final binary search at best resolution
    if tl() > 30:
        n_bf = min(len(best_h), 2048)
        h_bf = upsample_h(best_h, n_bf)
        h_bf_bin = (h_bf > 0.5).astype(float)
        n_ones_f = int(np.sum(h_bf_bin))
        n_tgt_f = n_bf // 2
        if n_ones_f > n_tgt_f:
            oi = np.where(h_bf_bin == 1)[0]
            o = rng.permutation(len(oi))
            for k in range(n_ones_f - n_tgt_f):
                h_bf_bin[oi[o[k]]] = 0.0
        elif n_ones_f < n_tgt_f:
            zi = np.where(h_bf_bin == 0)[0]
            o = rng.permutation(len(zi))
            for k in range(n_tgt_f - n_ones_f):
                h_bf_bin[zi[o[k]]] = 1.0
        h_bf_bin, c5_bf = binary_local_search(h_bf_bin, n_bf, rng, max_rounds=50, n_tries=2000)
        if c5_bf < best_c5:
            best_c5, best_h = c5_bf, h_bf_bin.copy()

    # Final SLSQP squeeze
    if tl() > 10:
        n_ff = len(best_h)
        try:
            h_opt = slsqp_polish(best_h, n_ff, 5000, max_iter=60)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
