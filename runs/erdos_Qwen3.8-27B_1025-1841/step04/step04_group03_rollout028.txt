```python
import numpy as np
from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, T):
    """Log-sum-exp smoothing of the max correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def lbgfb_optimize(h, n, T, maxiter=200):
    """L-BFGS-B optimization of the smoothed objective."""
    def obj(x):
        return soft_max_obj(x, T)
    res = minimize(obj, h.copy(), method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-12})
    return res.x

def korihara_harvey(h, n, max_rounds, start_time, budget_s):
    """Greedy coordinate descent focusing on max-reducing moves."""
    best_c5 = compute_c5(h)
    h = h.copy()
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        # Find current max lag
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Try flipping individual positions
        for j in range(n):
            if time.time() - start_time > budget_s - 10:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            # Try a few candidates
            for cand in [0.0, 1.0]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                # Adjust for feasibility: redistribute the difference
                diff = old - cand
                if diff > 0:
                    mask = trial < 1.0
                    mask[j] = False
                else:
                    mask = trial > 0.0
                    mask[j] = False
                cnt = np.sum(mask)
                if cnt == 0:
                    continue
                trial[mask] += diff / cnt
                trial = np.clip(trial, 0.0, 1.0)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
                    best_trial = trial
            if best_val != old:
                h = best_trial
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5
```

```python
def fast_binary_swap(h, a, b, n):
    """Compute the change in correlation when h[a]: 1->0, h[b]: 0->1."""
    # Δcorr[m] = 2*(h[a+m-n+1] - h[b+m-n+1]) for valid indices
    # m ranges from 0 to 2n-2
    # a+m-n+1 is valid when 0 <= a+m-n+1 <= n-1, i.e., n-1-a <= m <= n-1+a
    # b+m-n+1 is valid when 0 <= b+m-n+1 <= n-1, i.e., n-1-b <= m <= n-1+b
    
    # Create the delta array
    delta = np.zeros(2*n-1)
    # Contribution from removing a (1->0): 2*h[a+m-n+1] for valid m
    lo_a = max(0, n-1-a)
    hi_a = min(2*n-2, n-1+a)
    idx_a = np.arange(lo_a, hi_a+1) + n - 1 - a  # these are the indices into h
    delta[lo_a:hi_a+1] += 2 * h[idx_a]
    
    # Contribution from adding b (0->1): -2*h[b+m-n+1] for valid m
    lo_b = max(0, n-1-b)
    hi_b = min(2*n-2, n-1+b)
    idx_b = np.arange(lo_b, hi_b+1) + n - 1 - b
    delta[lo_b:hi_b+1] -= 2 * h[idx_b]
    
    return delta
```

```python
def smooth_obj_and_grad(x, n, T, K=20):
    """Objective and approximate gradient using top-K correlations."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    obj = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    # Approximate gradient using top-K correlations
    top_k = np.argsort(corr)[-K:]
    w = np.exp(T * (corr[top_k] - mc))
    w /= np.sum(w)
    
    grad = np.zeros(n)

        offset = k - (n - 1)
        j_min = max(0, -offset)
        j_max = min(n, n - offset)
        if j_min >= j_max:
            continue
        idx = np.arange(j_min, j_max) + offset
        grad[j_min:j_max] += w[t] * dx * (1 - 2 * x[idx])
    
    return obj, grad

        _, g = smooth_obj_and_grad(x, n, T, K)
        return g
```

```python
def gradient_step(h, n, T, K, start_time, deadline, max_steps=500):
    """Subgradient steps with line search."""
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / n
    m = 2*n - 1
    
    for step in range(max_steps):
        if time.time() > deadline - 10:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute top-K subgradient
        top_idx = np.argsort(corr)[-K:]
        w = np.exp(T * (corr[top_idx] - corr[top_idx[-1]]))
        w /= np.sum(w)
        
        grad = np.zeros(n)
        for t in range(K):
            k = top_idx[t]
            offset = k - (n - 1)
            j_min = max(0, -offset)
            j_max = min(n, n - offset)
            if j_min >= j_max:
                continue
            idx = np.arange(j_min, j_max) + offset
            grad[j_min:j_max] += w[t] * (1 - 2 * h[idx])
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Try several step sizes
        best_trial = h.copy()
        best_trial_c5 = c5
        for alpha in [0.5/gnorm, 0.1/gnorm, 0.01/gnorm, 0.001/gnorm]:
            trial = h - alpha * grad
            trial = np.clip(trial, 0, 1)
            trial = make_feasible(trial, n)
            tc5 = compute_c5(trial)
            if tc5 < best_trial_c5:
                best_trial_c5 = tc5
                best_trial = trial
        h = best_trial
    
    return best_h, best_c5
```
