```python
def subgradient_step(h, n, alpha):
    """One subgradient step for min max_k corr[k]*dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    j_star = np.argmax(corr)
    d = j_star - (n - 1)  # the lag
    
    # Compute subgradient of corr[j_star] w.r.t. h
    grad = np.zeros(n)
    
    if d >= 0:
        i_start, i_end = d, n - 1
    else:
        i_start, i_end = 0, n - 1 + d
    
    for i in range(i_start, i_end + 1):
        val = 1.0
        if 0 <= i - d < n:
            val -= h[i - d]
        if 0 <= i + d < n:
            val -= h[i + d]
        grad[i] = val
    
    grad *= dx
    
    # Subgradient step
    h_new = h - alpha * grad
    # Project onto constraint
    h_new = make_feasible(h_new, n)
    return h_new, j_star
```

def binary_swap_search(h_bin, n, rng, max_rounds=50, n_tries=1000):
    """Local search with single swaps on binary vectors."""
    h = h_bin.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
