def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)

def subgrad_balance(h, n, step_size, num_active=10):
    corr = compute_corr(h)
    center = n - 1
    active = np.argsort(corr)[-num_active:]
    grad = np.zeros(n)
    for k in active:
        lag = k - center
        if lag > 0:
            grad[:n-lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n-lag]
