import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp_active(h, n, n_active=80):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_idx = np.argsort(corr)[::-1]
    active = sorted_idx[:n_active]
    
    A = np.zeros((len(active), n))
    b_const = np.zeros(len(active))
    
    for idx, k in enumerate(active):
        i_min = max(0, n - 1 - k)
        i_max = min(n - 1, 2 * n - 2 - k)
        if i_min > i_max:
            continue
        ii = np.arange(i_min, i_max + 1)
        jj = ii + k - (n - 1)
        A[idx, ii] += (1.0 - h[jj])
        A[idx, jj] += (-h[ii])
        b_const[idx] = np.sum(h[ii] * h[jj])
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0,
        A @ h_var + b_const <= t_var * dx
    ]
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    for solver in [cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if h_var.value is not None:
                return np.clip(h_var.value, 0.0, 1.0), float(t_var.value) if t_var.value is not None else None
        except Exception:
            continue
    return None, None

def slsqp_smooth(h, n, T, maxiter=100):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mx = np.max(corr)
        shifted = corr - mx
        return (mx + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def con(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': lambda x: np.ones(len(x))}]
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
    return np.clip(res.x, 0.0, 1.0)

def binary_sa(h, n, time_limit, start_time, rng, T0=0.004, T1=0.00005, max_iters=80000):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        b[ones[rng.randint(len(ones))]] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        b[zeros[rng.randint(len(zeros))]] = 1
    
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = min(1.0, float(it) / max(1, max_iters))
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def block_sa(h, n, time_limit, start_time, rng, T0=0.003, T1=0.00002, max_iters=40000):
    b = (h >= 0.5).astype(float)
    target = n // 2

    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = min(1.0, float(it) / max(1, max_iters))
        T = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, min(10, n // 10))
        start = rng.randint(0, max(1, n - bsize))
        seg = b[start:start + bsize]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)

                best_c5 = new_c5
                best_b = b.copy()
        else:

            b[start + cz] = 0
    return best_b, best_c5

def fourier_perturb(h, n, rng, n_modes=4, amp=0.04):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 5)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def segment_perturb(h, n, rng):
    h_new = h.copy()
    n_segs = rng.randint(1, 4)
    for _ in range(n_segs):
        length = rng.randint(n // 15, n // 5)
        start = rng.randint(0, max(1, n - length))
        if rng.random() < 0.5:
            h_new[start:start + length] = 1.0 - h_new[start:start + length]
        else:
            h_new[start:start + length] = rng.uniform(0, 1, length)
    return make_feasible(h_new, n)

def upscale(h, old_n, new_n):
    old_x = np.linspace(0, 2.0, old_n)
    new_x = np.linspace(0, 2.0, new_n)
    h_up = np.interp(new_x, old_x, h)
    return make_feasible(h_up, new_n)

def lp_refine(h, n, start_time, budget_s, max_iters=40, step_frac=1.0):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 10:
            break
        n_act = min(80 + it * 5, 200)
        h_new, _ = solve_lp_active(best_h, n, n_active=n_act)
        if h_new is None:
            break
        h_new = make_feasible(h_new, n)
        if step_frac < 1.0:
            h_trial = make_feasible(best_h + step_frac * (h_new - best_h), n)
        else:
            h_trial = h_new
        c5_new = compute_c5(h_trial)
        if c5_new < best_c5 - 1e-9:
            best_c5 = c5_new
            best_h = h_trial
        else:
            if step_frac > 0.1:
                step_frac *= 0.5
                continue
            break
    return best_h, best_c5

def slsqp_multi(h, n, start_time, budget_s):
    temps = [200.0, 1000.0, 5000.0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for T in temps:
        if time.time() - start_time > budget_s - 30:
            break
        try:
            x_res = slsqp_smooth(best_h, n, T, maxiter=80)
            x_res = make_feasible(x_res, n)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res
        except Exception:
            pass
    return best_h, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 5:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 3:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5
