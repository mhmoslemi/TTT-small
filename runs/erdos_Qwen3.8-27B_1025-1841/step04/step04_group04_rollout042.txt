    delta[valid2] += h[i - k[valid2]]

    valid3 = (k >= -j) & (k <= n-1-j)
    delta[valid3] += (1 - h[j + k[valid3]])

    valid4 = (k >= j-(n-1)) & (k <= j)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    if len(h) != n:
        # Upsample or downsample
        old_n = len(h)
        old_x = np.linspace(0, 2, old_n)
        new_x = np.linspace(0, 2, n)
        h = np.interp(new_x, old_x, h)
    
    # Threshold at 0.5
    binary = (h >= 0.5).astype(float)
    count = binary.sum()
    target = n // 2
    
    if count > target:
        # Need to turn some 1s to 0s
        ones = np.where(binary == 1)[0]
        # Turn off the ones with lowest h values
        order = np.argsort(h[ones])[::-1]  # descending h values
        for idx in order[count - target]:
            binary[ones[idx]] = 0
    elif count < target:
        # Need to turn some 0s to 1s
        zeros = np.where(binary == 0)[0]
        order = np.argsort(h[zeros])[::-1]  # descending h values
        for idx in order[target - count]:
            binary[zeros[idx]] = 1
    
    return binary

    n = len(h)
    one_minus_h = 1.0 - h
    D = np.correlate(h, one_minus_h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_binary(h, n):
    if len(h) == n:
        h_interp = h.astype(float)
    else:
        old_n = len(h)
        old_x = np.linspace(0, 2, old_n)
        new_x = np.linspace(0, 2, n)
        h_interp = np.interp(new_x, old_x, h.astype(float))
    
    target = n // 2
    # Sort by value, pick top n/2
    thresh = np.sort(h_interp)[::-1][target - 1] if target > 0 else 1.0
    binary = (h_interp > thresh).astype(float)
    
    # Handle ties
    count = int(binary.sum())
    if count > target:
        # Too many ones, turn off some
        ones = np.where(binary == 1)[0]
        # Turn off the ones with lowest h_interp values
        order = ones[np.argsort(h_interp[ones])]
        for idx in order[:count - target]:
            binary[idx] = 0
    elif count < target:
        zeros = np.where(binary == 0)[0]
        order = zeros[np.argsort(h_interp[zeros])[::-1]]
        for idx in order[:target - count]:
            binary[idx] = 1
    
    return binary

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    if len(h) == n:
        h_interp = h.astype(float)
    else:
        old_n = len(h)
        old_x = np.linspace(0, 2, old_n)
        new_x = np.linspace(0, 2, n)
        h_interp = np.interp(new_x, old_x, h.astype(float))
    target = n // 2
    sorted_h = np.sort(h_interp)[::-1]
    thresh = sorted_h[target - 1]
    binary = (h_interp > thresh).astype(np.float64)
    count = int(binary.sum())
    if count > target:
        ones = np.where(binary == 1)[0]
        order = ones[np.argsort(h_interp[ones])]
        for idx in order[:count - target]:
            binary[idx] = 0.0
    elif count < target:
        zeros = np.where(binary == 0)[0]
        order = zeros[np.argsort(h_interp[zeros])[::-1]]
        for idx in order[:target - count]:
            binary[idx] = 1.0
    return binary

def sa_optimize(h0, n, time_limit, rng, T_init=0.05):
    D = np.correlate(h0, 1.0 - h0, mode='full')
    cur_max = float(D.max())
    h0 = h0.copy()
    omh = 1.0 - h0
    rev_h = h0[::-1].copy()
    delta = np.zeros(2 * n - 1)
    T = T_init
    T_min = 0.0003
    cooling = 0.999995
    no_improve = 0
    best_val = cur_max
    while time.time() < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h0[i] == h0[j]:
            continue
        if h0[i] == 0:
            i, j = j, i
        delta.fill(0.0)
        delta[n-1-i:2*n-1-i] -= omh[:n-i]
        delta[i:i+n] += rev_h
        delta[n-1-j:2*n-1-j] += omh[:n-j]
        delta[j:j+n] -= rev_h
        new_max = (D + delta).max()
        dc = new_max - cur_max
        dc5 = dc * 2.0 / n
        accept = dc < 0
        if not accept and T > T_min:
            if rng.random() < np.exp(-dc5 / T):
                accept = True
        if accept:
            h0[i] = 0.0
            h0[j] = 1.0
            D += delta
            cur_max = new_max
            omh[i] = 1.0
            omh[j] = 0.0
            rev_h[n-1-i] = 0.0
            rev_h[n-1-j] = 1.0
            no_improve = 0
            if cur_max < best_val:
                best_val = cur_max
        else:
            no_improve += 1
        T *= cooling
        if T < T_min:
            T = T_min
        if no_improve > 80000 and T == T_min:
            T = 0.02
            no_improve = 0
    return h0, cur_max * 2.0 / n

def greedy_polish(h0, n, time_limit, rng, n_rounds=50):
    D = np.correlate(h0, 1.0 - h0, mode='full')
    cur_max = float(D.max())
    h0 = h0.copy()
    omh = 1.0 - h0
    rev_h = h0[::-1].copy()
    delta = np.zeros(2 * n - 1)
    for rnd in range(n_rounds):
        if time.time() > time_limit:
            break
        improved = False
        for _ in range(n):
            if time.time() > time_limit:
                break
            i = rng.randint(n)
            for _ in range(30):
                j = rng.randint(n)
                if i == j or h0[i] == h0[j]:
                    continue
                if h0[i] == 0:
                    i, j = j, i
                delta.fill(0.0)
                delta[n-1-i:2*n-1-i] -= omh[:n-i]
                delta[i:i+n] += rev_h
                delta[n-1-j:2*n-1-j] += omh[:n-j]
                delta[j:j+n] -= rev_h
                new_max = (D + delta).max()
                if new_max < cur_max - 1e-15:
                    h0[i] = 0.0
                    h0[j] = 1.0
                    D += delta
                    cur_max = new_max
                    omh[i] = 1.0
                    omh[j] = 0.0
                    rev_h[n-1-i] = 0.0
                    rev_h[n-1-j] = 1.0
                    improved = True
                break
        if not improved:
            break
    return h0, cur_max * 2.0 / n
