def exact_coord_solve_fast(h, j, active_idx, n):
    """Fast version using direct formula for the change in correlations."""
    # For shift s (in correlation index m, where s = m - (n-1)):
    # c(s) = sum_{i: valid} h[i] * (1-h[i+s])
    # Change when h[j] goes from 0 to 1:
    #   i=j term: +1 * (1-h[j+s]) = 1-h[j+s] (if j+s in [0,n))
    #   i+k=j term: h[j-s] * (1-1) - h[j-s] * (1-0) = -h[j-s] (if j-s in [0,n))
    # So Δc(s) = (1-h[j+s])_valid - h[j-s]_valid
    
    # active_idx are indices into the full correlation array
    # shift s = active_idx - (n-1)
    shifts = active_idx - (n - 1)
    
    b = np.zeros(len(shifts))
    a = np.zeros(len(shifts))  # c(k) when h[j]=0
    
    # Current c(k) values
    corr_full = get_corr(h)
    a_full = corr_full[active_idx].copy()  # c(k) with current h[j]
    
    # b_k = c(k|h[j]=1) - c(k|h[j]=0)
    # c(k|h[j]=1) = c(k) - h[j]*(1-h[j+s]) + h[j]*h[j-s] + (1-h[j+s])_if_valid
    # Hmm, this is getting complicated. Let me just use the formula directly.
    
    # c(s) with h[j] = v:
    # = c(s) - h[j]*b_k + v*b_k  where b_k = (1-h[j+s])_valid - h[j-s]_valid
    # So a_k = c(s) - h[j]*b_k, and the slope is b_k
    
    for i, s in enumerate(shifts):
        bk = 0.0
        if 0 <= j + s < n:
            bk += (1.0 - h[j + s])
        if 0 <= j - s < n:
            bk -= h[j - s]
        b[i] = bk
        a[i] = a_full[i] - h[j] * bk
    
    # Find v in [0,1] minimizing max_k (a[k] + b[k] * v)
    candidates = [0.0, 1.0]
    for i in range(len(a)):
        for j2 in range(i + 1, len(a)):
            bi = b[i]
            bj = b[j2]
            if abs(bi - bj) > 1e-15:
                v = (a[j2] - a[i]) / (bi - bj)
                if -1e-10 <= v <= 1.0 + 1e-10:
                    candidates.append(np.clip(v, 0.0, 1.0))
    
    best_v = h[j]
    best_val = np.max(a + b * h[j])
    for v in candidates:
        val = np.max(a + b * v)
        if val < best_val - 1e-15:
            best_val = val
            best_v = v
    
    return best_v

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0
_G_TOP_IDX = None

def _obj_smooth(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_TOP_IDX is not None:
        corr = corr[_G_TOP_IDX]
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def active_set(h, top_frac=0.003, max_count=20):
    corr = get_corr(h)
    mc = np.max(corr)
    threshold = mc - top_frac * mc
    idx = np.where(corr >= threshold)[0]
    if len(idx) > max_count:
        vals = corr[idx]
        order = np.argsort(-vals)
        idx = idx[order[:max_count]]
    elif len(idx) == 0:
        idx = np.array([int(np.argmax(corr))])
    return idx

def exact_coord_solve(h, j, active_idx, n):
    corr_full = get_corr(h)
    a_full = corr_full[active_idx].copy()
    shifts = active_idx - (n - 1)
    b = np.zeros(len(shifts))
    for i in range(len(shifts)):
        s = shifts[i]
        bk = 0.0
        if 0 <= j + s < n:
            bk += (1.0 - h[j + s])
        if 0 <= j - s < n:
            bk -= h[j - s]
        b[i] = bk
    a = a_full - h[j] * b
    candidates = [0.0, 1.0]
    for i in range(len(a)):
        for j2 in range(i + 1, len(a)):
            bi = b[i]
            bj = b[j2]
            if abs(bi - bj) > 1e-14:
                v = (a[j2] - a[i]) / (bi - bj)
                if -1e-10 <= v <= 1.0 + 1e-10:
                    candidates.append(float(np.clip(v, 0.0, 1.0)))
    best_v = h[j]
    best_val = np.max(a + b * h[j])
    for v in candidates:
        val = np.max(a + b * v)
        if val < best_val - 1e-15:
            best_val = val
            best_v = v
    return best_v

def exact_cd_pass(h, n, active_idx, max_passes=1, time_limit=None, start_time=None):
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit is not None and (time.time() - start_time) > time_limit:
                break
            v = exact_coord_solve(h, j, active_idx, n)
            if abs(v - h[j]) > 1e-12:
                old = h[j]
                h[j] = v
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < best_c5 - 1e-16:
                    best_c5 = c
                    improved = True
                else:
                    h[j] = old
                    h = make_feasible(h, n)
        if not improved:
            break
    return h, best_c5
