result[lag + n - 1] = ∑_i h[i] * (1-h[i+lag])

That's useful - it means the max is always achieved at a non-zero lag.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h = np.clip(h, 0, 1)
    target = n // 2
    # Sort by h value to decide which to round up/down
    order = np.argsort(-h)  # descending
    binary = np.zeros(n)
    binary[order[:target]] = 1.0
    return binary

def sa_binary_search(h, n, rng, time_limit, T_start=0.005, alpha=0.9999):
    """Simulated annealing on binary h via swap moves."""
    dx = 2.0 / n
    h = h.astype(float).copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = float(np.max(corr))
    best_c5 = best_max * dx
    best_h = h.copy()
    
    T = T_start
    start = time.time()
    
    while time.time() - start < time_limit:
        # Random swap
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        
        # Perform swap
        h[i] = 0.0
        h[j] = 1.0
        
        # Reindex
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = float(np.max(corr))
        
        if new_max <= best_max:
            best_max = new_max
            best_c5 = new_max * dx
            best_h = h.copy()
        elif rng.random() < np.exp(-(new_max - best_max) * dx / T):
            # Accept worse solution
            pass
        else:
            # Revert
            h[i] = 1.0
            h[j] = 0.0
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = float(np.max(corr))
        
        T *= alpha
        if T < 1e-6:
            T = 1e-6
    
    return best_h, best_c5


def _obj_smooth(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _opt(x, T):
    return _obj_smooth(x, T)

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def continuous_refine(h, n, T, maxiter, time_limit):
    """SLSQP refinement with smooth max."""
    current = h.copy()
    start = time.time()
    try:
        res = minimize(_opt, current, args=(T,), method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}],
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() - start < time_limit and compute_c5(res.x) < compute_c5(current):
            return make_feasible(res.x, n)
    except Exception:
        pass
    return current

def coord_descent(h, n, time_limit):
    """Coordinate descent on individual values."""
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    cands = [0.0, 1.0, 0.5, 0.2, 0.8]
    
    improved = True
    while improved and time.time() - start < time_limit:
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old = h[j]
            best_trial = old
            best_c = best_c5
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                h[j] = best_trial
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                h[j] = best_trial
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binary SA from initial at n=960 (~300s)
    if remaining() > 300:
        h_bin = to_binary(h_init, n_init)
        h_bin, c5_1 = sa_search(h_bin, n_init, rng, time_limit=min(280, remaining()-10), T_start=0.012, alpha=0.99995)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h_bin.copy()

    # Phase 2: SA from Thue-Morse starting point (~150s)
    if remaining() > 150:
        n = len(best_h)
        tm = thue_morse(n)
        tm_bin = to_binary(tm + 0.5, n)
        tm_bin, c5_2 = sa_search(tm_bin, n, rng, time_limit=min(130, remaining()-10), T_start=0.015, alpha=0.9999)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = tm_bin.copy()

    # Phase 3: Second SA round from best, different params (~150s)
    if remaining() > 150:
        n = len(best_h)
        h_bin = to_binary(best_h, n)
        h_bin, c5_3 = sa_search(h_bin, n, rng, time_limit=min(130, remaining()-10), T_start=0.008, alpha=0.99997)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h_bin.copy()

    # Phase 4: Upsample to n=1280 and SA (~300s)
    if remaining() > 300:
        n_large = 1280
        h_large = upsample_h(best_h, n_large)
        h_large_bin = to_binary(h_large, n_large)
        h_large_bin, c5_4 = sa_search(h_large_bin, n_large, rng, time_limit=min(270, remaining()-20), T_start=0.009, alpha=0.99995)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h_large_bin.copy()

    # Phase 5: SA at n=1280 from Thue-Morse (~150s)
    if remaining() > 150:
        n_large = 1280
        tm = thue_morse(n_large)
        tm_bin = to_binary(tm + 0.5, n_large)
        tm_bin, c5_5 = sa_search(tm_bin, n_large, rng, time_limit=min(130, remaining()-10), T_start=0.012, alpha=0.9999)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = tm_bin.copy()

    # Phase 6: Continuous SLSQP refinement
    if remaining() > 80:
        n = len(best_h)
        for T in [300.0, 800.0]:
            if remaining() < 40:
                break
            h_ref = slsqp_refine(best_h, n, T, maxiter=200)
            c5_ref = compute_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref

    # Phase 7: Coordinate descent
    if remaining() > 60:
        n = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n, time_limit=min(50, remaining()-10))

    # Phase 8: Final binary snap and short SA
    if remaining() > 30:
        n = len(best_h)
        h_bin = to_binary(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin
        h_bin2, c5_8 = sa_search(best_h, n, rng, time_limit=min(20, remaining()-5), T_start=0.004, alpha=0.9999)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h_bin2

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
