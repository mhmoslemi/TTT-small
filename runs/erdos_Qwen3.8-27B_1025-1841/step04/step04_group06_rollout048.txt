def sa_binary(h_bin, n, rng, T0=0.001, alpha=0.9997, max_iters=50000, patience=5000):
    """Simulated annealing on binary configurations."""
    dx = 2.0 / n
    h = h_bin.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    corr = compute_corr(h)
    current_c = float(np.max(corr) * dx)
    best_h = h.copy()
    best_c = current_c
    
    no_improve = 0
    for it in range(max_iters):
        T = T0 * (alpha ** it)
        
        # Random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = compute_corr(h)
        new_c = float(np.max(new_corr) * dx)
        
        if new_c < current_c or rng.rand() < np.exp(-(new_c - current_c) / max(T, 1e-10)):
            current_c = new_c
            corr = new_corr
            # Update ones and zeros
            ones = np.append(np.delete(ones, np.where(ones == i)[0][0]), j)
            zeros = np.append(np.delete(zeros, np.where(zeros == j)[0][0]), i)
            
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        if no_improve > patience and T < T0 * 0.01:
            break
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    """Compute C5 using full correlation (exact)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def circ_corr_max(h):
    """Compute max of circular cross-correlation of h with 1-h."""
    n = len(h)
    H = np.fft.rfft(h)
    OH = np.fft.rfft(1.0 - h)
    corr = np.fft.irfft(H * np.conj(OH))
    return float(np.max(corr)), corr

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binarize(h, n):
    """Convert to binary with exactly n//2 ones."""
    sorted_idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n//2]] = 1.0
    return h_bin

def sa_binary(h_bin, n, rng, T0, n_cool, patience, max_time):
    """
    Simulated annealing on binary configurations using circular correlation.
    """
    dx = 2.0 / n
    h = h_bin.copy()
    ones_idx = np.where(h > 0.5)[0].copy()
    zeros_idx = np.where(h <= 0.5)[0].copy()
    
    _, current_corr = circ_corr_max(h)
    current_val = float(np.max(current_corr) * dx)
    best_h = h.copy()
    best_val = current_val
    
    n_ones = n // 2
    no_improve = 0
    t0 = time.time()
    
    for it in range(max_time[0]):  # max_time is a tuple (max_iters, deadline)
        if time.time() > max_time[1]:
            break
        
        T = T0 * (0.9998 ** it)
        
        # Choose move: 70% random, 30% directed
        if rng.rand() < 0.7:
            pi = rng.randint(len(ones_idx))
            pj = rng.randint(len(zeros_idx))
        else:
            # Directed: try to reduce the max lag
            max_k = int(np.argmax(current_corr))
            # Find positions contributing to high correlation at max_k
            # h[i]=1 and h[(i+max_k)%n]=0 contribute 1 to corr at max_k
            contrib = (h > 0.5) & (np.roll(h, -max_k) <= 0.5)
            contrib_ones = np.where(contrib)[0]
            if len(contrib_ones) > 0:
                pi = rng.randint(len(ones_idx))
                # Prefer zeros that would reduce corr at max_k
                helpful_zeros = (zeros_idx[np.where(np.roll(h, -max_k, zeros_idx) > 0.5)[0]])
                if len(helpful_zeros) > 0:
                    pj = int(np.where(zeros_idx == helpful_zeros[rng.randint(len(helpful_zeros))])[0][0])
                else:
                    pj = rng.randint(len(zeros_idx))
            else:
                pi = rng.randint(len(ones_idx))
                pj = rng.randint(len(zeros_idx))
        
        i = ones_idx[pi]
        j = zeros_idx[pj]
        
        # Try swap
        h[i] = 0.0
        h[j] = 1.0
        _, new_corr = circ_corr_max(h)
        new_val = float(np.max(new_corr) * dx)
        
        accept = new_val < current_val or (T > 1e-10 and rng.rand() < np.exp(-(new_val - current_val) / T))
        
        if accept:
            current_val = new_val
            current_corr = new_corr
            ones_idx = np.append(np.delete(ones_idx, pi), j)
            zeros_idx = np.append(np.delete(zeros_idx, pj), i)
            if new_val < best_val - 1e-12:
                best_val = new_val
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
        
        if no_improve > patience:
            break
    
    return best_h, best_val
