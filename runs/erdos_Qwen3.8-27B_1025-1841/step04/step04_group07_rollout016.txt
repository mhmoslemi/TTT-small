import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step_ = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step_, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_val(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_opt(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_val(c, T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def topk_opt(h0, n, n_iters, k_frac, max_inner):
    dx = 2.0 / n
    h = h0.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_c = len(corr)
        k = max(10, int(k_frac * n_c))
        idx = np.argpartition(corr, -k)[-k:]
        active = np.zeros(n_c)
        m = corr[idx]
        rng_ = m.max() - m.min() + 1e-10
        w = np.exp(3.0 * (m - m.max()) / rng_)
        active[idx] = w
        active /= active.sum()
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(active, c)) * dx + 500.0 * (x.sum() - n / 2.0) ** 2
        try:
            res = minimize(obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = make_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                h = h_new
        except Exception:
            pass
    return h

def binary_sa(h_bin, n, rng, max_rounds, t0, cool, max_tries):
    h = h_bin.copy()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    temp = t0
    for rnd in range(max_rounds):
        improved = False
        nt = min(max_tries, max(1, len(ones) * len(zeros)))
        for _ in range(nt):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            dc = c - cur
            if dc < -1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
                if c < best:
                    best = c
                    best_h = h.copy()
            elif rng.rand() < np.exp(-dc / max(temp, 1e-10)):
                ones[ii] = oj
                zeros[jj] = oi
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        temp *= cool
        if not improved and temp < 0.02:
            break
    return best_h, best

def greedy_refine(h_bin, n, rng, max_rounds, max_tries):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        nt = min(max_tries, max(1, len(ones) * len(zeros)))
        for _ in range(nt):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def to_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    target = n // 2
    no = int(h_b.sum())
    if no > target:
        idx = np.where(h_b == 1)[0]
        rng.shuffle(idx)
        h_b[idx[:no - target]] = 0.0
    elif no < target:
        idx = np.where(h_b == 0)[0]
        rng.shuffle(idx)
        h_b[idx[:target - no]] = 1.0
    return h_b

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = len(h0)

    # Phase 1: Smooth opt at parent res with progressive T
    n1 = best_n
    h1 = best_h.copy()
    for T in [300, 2000, 10000, 50000]:
        if rem() < 30: break
        h_opt = lbgfb_opt(h1, n1, T, 800, 200)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()
            h1 = h_opt.copy()

    # Top-k minmax
    if rem() > 50:
        h_tk = topk_opt(h1, n1, 40, 0.08, 35)
        c_tk = compute_c5(h_tk)
        if c_tk < best_c5:
            best_c5, best_h = c_tk, h_tk.copy()
            h1 = h_tk.copy()

    # Phase 2: Upsample + smooth
    for n2 in [960, 1280]:
        if rem() < 100: break
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        for T in [1000, 5000, 30000]:
            if rem() < 30: break
            h_opt = lbgfb_opt(h2, n2, T, 400, 200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h2 = h_opt.copy()
        if rem() > 50:
            h_tk = topk_opt(h2, n2, 25, 0.08, 30)
            c_tk = compute_c5(h_tk)
            if c_tk < best_c5:
                best_c5, best_h = c_tk, h_tk.copy()

    # Phase 3: Binary SA
    for n_b in [640, 960]:
        if rem() < 80: break
        h_b = upsample_h(best_h, n_b) if n_b < len(best_h) else best_h.copy()
        h_bin = to_binary(h_b, n_b, rng)
        for r in range(3):
            if rem() < 30: break
            h_sa, c_sa = binary_sa(h_bin, n_b, rng, 60, 0.8 + 0.4*r, 0.96, 800)
            if c_sa < best_c5:
                best_c5, best_h = c_sa, h_sa.copy()
            h_g, c_g = greedy_refine(h_sa, n_b, rng, 30, 1500)
            if c_g < best_c5:
                best_c5, best_h = c_g, h_g.copy()

    # Phase 4: Multi-start
    n_ms = len(best_h)
    if rem() > 50:
        for r in range(3):
            if rem() < 20: break
            h_r = best_h.copy() + 0.15 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            for T in [2000, 10000]:
                if rem() < 15: break
                h_opt = lbgfb_opt(h_r, n_ms, T, 600, 150)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()
                    h_r = h_opt.copy()

    # Phase 5: Final binary greedy at best resolution
    if rem() > 30:
        n_f = min(len(best_h), 1024)
        h_f = upsample_h(best_h, n_f) if n_f < len(best_h) else best_h.copy()
        h_fb = to_binary(h_f, n_f, rng)
        h_fg, c_fg = greedy_refine(h_fb, n_f, rng, 40, 2000)
        if c_fg < best_c5:

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    h = h0.copy()
    for it in range(ni):
        c = np.correlate(h, 1.0 - h, mode='full')
        nc = len(c)
        k = max(10, int(kf * nc))
        ix = np.argpartition(c, -k)[-k:]
        act = np.zeros(nc)
        m = c[ix]
        w = np.exp(3.0 * (m - m.max()) / (m.max() - m.min() + 1e-10))
        act[ix] = w
        act /= act.sum()
        def obj(x):
            cc = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(act, cc)) * dx + 500.0 * (x.sum() - n / 2.0) ** 2
        try:
            r = minimize(obj, h, method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                         options={'maxiter': mi, 'ftol': 1e-15, 'gtol': 1e-10})
            hn = make_feasible(r.x, n)
            if compute_c5(hn) <= compute_c5(h) + 1e-10:
                h = hn
        except Exception:
            pass
    return h

def bin_sa(hb, n, rng, mr, t_init, cool, mt):
    h = hb.copy()
    cur = compute_c5(h)
    best = cur
    bh = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    t = t_init
    for rd in range(mr):
        imp = False
        nt = min(mt, max(1, len(ones) * len(zeros)))
        for _ in range(nt):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0; h[oj] = 1.0
            c = compute_c5(h)
            dc = c - cur
            if dc < -1e-15:
                cur = c; ones[ii] = oj; zeros[jj] = oi; imp = True
                if c < best: best = c; bh = h.copy()
            elif rng.rand() < np.exp(-dc / max(t, 1e-10)):
                ones[ii] = oj; zeros[jj] = oi
            else:
                h[oi] = 1.0; h[oj] = 0.0
        t *= cool
        if not imp and t < 0.02:
            break
    return bh, best

def to_bin(h, n, rng):
    hb = (h > 0.5).astype(float)
    tgt = n // 2
    no = int(hb.sum())
    if no > tgt:
        ix = np.where(hb == 1)[0]
        rng.shuffle(ix)
        hb[ix[:no - tgt]] = 0.0
    elif no < tgt:
        ix = np.where(hb == 0)[0]
        rng.shuffle(ix)
        hb[ix[:tgt - no]] = 1.0
    return hb

def thue_morse(n):
    h = np.zeros(n)
    for i in range(n):
        v = i
        c = 0
        while v:
            c ^= 1
            v >>= 1
        h[i] = float(c)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        room = (1.0 - h) if diff > 0 else h
        mask = room > 1e-13
        if not mask.any():
            break
        h[mask] = np.clip(h[mask] + diff / mask.sum(), 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    ox = np.linspace(0.0, 2.0, old_n, endpoint=False)
    nx = np.linspace(0.0, 2.0, new_n, endpoint=False)
    return make_feasible(np.interp(nx, ox, h, period=2.0), new_n)

def soft_max_val(c, T):
    mc = c.max()
    return mc + np.log(np.sum(np.exp(np.clip(T * (c - mc), -80, 80)))) / T

def lbgfb(h0, n, T, mu, mi):
    dx = 2.0 / n
    def obj(x):
        return soft_max_val(np.correlate(x, 1.0-x, 'full'), T) * dx + mu*(x.sum()-n/2.0)**2
    r = minimize(obj, h0, method='L-BFGS-B', bounds=[(0.,1.)]*n,
                 options={'maxiter':mi,'ftol':1e-16,'gtol':1e-11})
    return make_feasible(r.x, n)

def topk(h0, n, ni, kf, mi):
    dx = 2.0/n
    h = h0.copy()
    for _ in range(ni):
        c = np.correlate(h, 1.-h, 'full')
        nc = len(c)
        k = max(10, int(kf*nc))
        ix = np.argpartition(c, -k)[-k:]
        act = np.zeros(nc)
        m = c[ix]
        act[ix] = np.exp(3*(m-m.max())/(m.max()-m.min()+1e-10))
        act /= act.sum()
        def obj(x):
            return float(np.dot(act, np.correlate(x,1.-x,'full')))*dx + 500*(x.sum()-n/2.)**2
        try:
            r = minimize(obj, h, method='L-BFGS-B', bounds=[(0.,1.)]*n,
                         options={'maxiter':mi,'ftol':1e-15,'gtol':1e-10})
            hn = make_feasible(r.x, n)
            if compute_c5(hn) <= compute_c5(h)+1e-10:
                h = hn
        except:
            pass
    return h

def bsa(hb, n, rng, mr, ti, cl, mt, t_end):
    h = hb.copy()
    cur = compute_c5(h); best = cur; bh = h.copy()
    ones = np.where(h>.5)[0].tolist()
    zeros = np.where(h<=.5)[0].tolist()
    t = ti
    for _ in range(mr):
        imp = False
        nt = min(mt, max(1, len(ones)*len(zeros)))
        for _ in range(nt):
            if not ones or not zeros: break
            ii = rng.randint(len(ones)); jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi]=0; h[oj]=1
            c = compute_c5(h); dc = c-cur
            if dc < -1e-15:
                cur=c; ones[ii]=oj; zeros[jj]=oi; imp=True
                if c<best: best=c; bh=h.copy()
            elif rng.rand() < np.exp(-dc/max(t,1e-10)):
                ones[ii]=oj; zeros[jj]=oi
            else:
                h[oi]=1; h[oj]=0
        t *= cl
        if (not imp and t<.02) or time.time()>t_end:
            break
    return bh, best

def gre(hb, n, rng, mr, mt, t_end):
    h = hb.copy()
    cur = compute_c5(h)
    ones = np.where(h>.5)[0].tolist()
    zeros = np.where(h<=.5)[0].tolist()
    for _ in range(mr):
        imp = False
        nt = min(mt, max(1, len(ones)*len(zeros)))
        for _ in range(nt):
            if not ones or not zeros: break
            if time.time()>t_end: break
            ii = rng.randint(len(ones)); jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi]=0; h[oj]=1
            c = compute_c5(h)
            if c < cur-1e-15:
                cur=c; ones[ii]=oj; zeros[jj]=oi; imp=True
            else:
                h[oi]=1; h[oj]=0
        if not imp: break
    return h, cur

def tbin(h, n, rng):
    hb = (h>.5).astype(float)
    tgt = n//2
    no = int(hb.sum())
    if no > tgt:
        ix = np.where(hb==1)[0]; rng.shuffle(ix); hb[ix[:no-tgt]]=0
    elif no < tgt:
        ix = np.where(hb==0)[0]; rng.shuffle(ix); hb[ix[:tgt-no]]=1
    return hb
