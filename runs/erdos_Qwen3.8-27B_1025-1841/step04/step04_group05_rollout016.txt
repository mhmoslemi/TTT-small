                term_a[valid_a] = 1.0 - h[idx_a[valid_a]]
                term_b = np.zeros(2 * n - 1)
                idx_b = j + shifts
                valid_b = (idx_b >= 0) & (idx_b < n)
                term_b[valid_b] = h[idx_b[valid_b]]
                new_corr = corr + delta * (term_a - term_b)
                new_max = float(np.max(new_corr))
                if new_max < best_c:
                    best_c = new_max
                    best_delta = delta
            # Also try setting to 0 or 1
            for bv in (0.0, 1.0):
                new_val = bv
                delta = new_val - old_val
                if abs(delta) < 1e-15:
                    continue
                shifts = np.arange(2 * n - 1) - (n - 1)
                term_a = np.zeros(2 * n - 1)
                idx_a = j - shifts
                valid_a = (idx_a >= 0) & (idx_a < n)
                term_a[valid_a] = 1.0 - h[idx_a[valid_a]]
                term_b = np.zeros(2 * n - 1)
                idx_b = j + shifts
                valid_b = (idx_b >= 0) & (idx_b < n)
                term_b[valid_b] = h[idx_b[valid_b]]
                new_corr = corr + delta * (term_a - term_b)
                new_max = float(np.max(new_corr))
                if new_max < best_c:
                    best_c = new_max
                    best_delta = delta
            if best_delta != 0.0:
                h[j] = old_val + best_delta
                shifts = np.arange(2 * n - 1) - (n - 1)
                term_a = np.zeros(2 * n - 1)
                idx_a = j - shifts
                valid_a = (idx_a >= 0) & (idx_a < n)
                term_a[valid_a] = 1.0 - h[idx_a[valid_a]]
                term_b = np.zeros(2 * n - 1)
                idx_b = j + shifts
                valid_b = (idx_b >= 0) & (idx_b < n)
                term_b[valid_b] = h[idx_b[valid_b]]
                corr = corr + best_delta * (term_a - term_b)
                max_corr = best_c
                improved = True
                if best_c * dx < best_c5:
                    best_c5 = best_c * dx
        if not improved:
            break
    return h, best_c5

def coord_descent_fast(h, n, max_rounds, dx):
    """Faster CD: precompute shift vectors, use vectorized updates."""
    corr = compute_corr(h)
    max_corr = float(np.max(corr))
    best_c5 = max_corr * dx
    shifts = np.arange(2 * n - 1) - (n - 1)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            # Precompute index arrays for this j
            idx_a = j - shifts
            valid_a = (idx_a >= 0) & (idx_a < n)
            idx_b = j + shifts
            valid_b = (idx_b >= 0) & (idx_b < n)
            # Term vectors: (1-h[j-s]) and h[j+s]
            ta = np.zeros(2 * n - 1)
            ta[valid_a] = 1.0 - h[idx_a[valid_a]]
            tb = np.zeros(2 * n - 1)
            tb[valid_b] = h[idx_b[valid_b]]
            diff_vec = ta - tb
            old_val = h[j]
            best_delta = 0.0
            best_max = max_corr
            for delta in (0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.05, -0.05, 0.001, -0.001):
                nv = old_val + delta
                if nv < 0.0 or nv > 1.0:
                    continue
                new_max = float(np.max(corr + delta * diff_vec))
                if new_max < best_max:
                    best_max = new_max
                    best_delta = delta
            for bv in (0.0, 1.0):
                if abs(bv - old_val) < 1e-15:
                    continue
                d = bv - old_val
                new_max = float(np.max(corr + d * diff_vec))
                if new_max < best_max:
                    best_max = new_max
                    best_delta = d
            if best_delta != 0.0:
                h[j] = old_val + best_delta
                corr = corr + best_delta * diff_vec
                max_corr = best_max
                improved = True
                if max_corr * dx < best_c5:
                    best_c5 = max_corr * dx
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_N, _OPT_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - start_time)

    # Initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 160)
    h0 = make_feasible(h0, len(h0))
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: SLSQP at progressive resolutions
    resolutions = [960, 1600, 2400]
    for n_pts in resolutions:
        if _tl() < 120:
            break
        h_curr = upsample_h(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        temps = [(30, 300), (80, 250), (200, 200), (500, 150)]
        for T, miter in temps:
            if _tl() < 40:
                break
            try:
                h_opt = _slsqp_optimize(h_curr, n_pts, T, miter)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_curr = h_opt.copy()
                else:
                    h_curr = h_opt.copy()
            except Exception:
                pass

    # Phase 2: L-BFGS-B with smooth max (no equality constraint, use penalty)
    if _tl() > 200:
        n_lb = len(best_h)
        if n_lb > 2000:
            n_lb = 2000
            h_curr = upsample_h(best_h, n_lb)
        else:
            h_curr = best_h.copy()

        lam_vals = [1e2, 1e3, 1e4]
        for lam in lam_vals:
            if _tl() < 100:
                break
            for T in [100, 300, 1000]:
                if _tl() < 60:
                    break
                try:
                    h_opt = _lbfgsb_optimize(h_curr, n_lb, T, lam, 200)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        h_curr = h_opt.copy()
                    else:
                        h_curr = h_opt.copy()
                except Exception:
                    pass

    # Phase 3: Up to high resolution for CD
    if _tl() > 150:
        n_cd = min(len(best_h), 2048)
        if n_cd < 2048 and _tl() > 250:
            n_cd = 2048
        h_cd = upsample_h(best_h, n_cd)
        dx_cd = 2.0 / n_cd
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

        # Coordinate descent with early stopping
        if _tl() > 100:
            h_cd, c5_cd = coord_descent_fast(h_cd, n_cd, 8, dx_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd.copy()

    # Phase 4: Random restarts at current best resolution
    n_curr = len(best_h)
    for restart in range(3):
        if _tl() < 80:
            break
        h_rand = 0.5 * best_h + 0.5 * rng.rand(n_curr)
        h_rand = make_feasible(h_rand, n_curr)
        # Binary perturbation
        mask = rng.rand(n_curr) < 0.1
        h_rand[mask] = rng.rand(int(np.sum(mask)))
        h_rand = make_feasible(h_rand, n_curr)
        try:
            T_use = 100.0
            h_opt = _slsqp_optimize(h_rand, n_curr, T_use, 150)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 5: Final SLSQP squeeze at very high T
    if _tl() > 30:
        n_fin = len(best_h)
        try:
            h_opt = _slsqp_optimize(best_h, n_fin, 3000, 80)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 6: Final CD polish
    if _tl() > 30:
        n_pol = len(best_h)
        dx_pol = 2.0 / n_pol
        try:
            h_pol, c5_pol = coord_descent_fast(best_h.copy(), n_pol, 4, dx_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
