```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _softmax_objgrad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    exp_s = np.exp(shifted)
    sum_exp = np.sum(exp_s)
    p = exp_s / sum_exp
    f = (mc + np.log(sum_exp) / T) * dx + lam * (np.sum(h) - n/2.0)**2
    
    # Gradient using FFT-based computation
    # A[j] = sum_m p[m]*h[j+m-(n-1)] → from correlate
    # B[j] = sum_m p[m]*h[j-m+(n-1)] → from convolve
    full_corr = np.correlate(h, p, mode='full')
    full_conv = np.convolve(p, h, mode='full')
    A = full_corr[3*n-3 : 2*n-4 : -1]
    B = full_conv[n-1 : 2*n-1]
    
    grad = dx * (np.sum(p) - A - B)
    grad += 2.0 * lam * (np.sum(h) - n / 2.0)
    
    return f, grad

def lbfgsb_optimize(h_start, n, T, lam, max_iter, time_limit):
    res = minimize(_softmax_objgrad, h_start, method='L-BFGS-B',
                   jac=True,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-8,
                            'maxfun': 10000})
    return make_feasible(res.x, n)

def binary_local_search(h, n, max_sweeps=20):
    """Try flipping pairs of binary values to reduce max correlation."""
    dx = 2.0 / n
    h = np.clip(h, 0, 1).copy()
    # Project to binary first
    h_bin = (h > 0.5).astype(float)
    h_bin = make_feasible(h_bin, n)
    best = compute_c5(h_bin)
    
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    
    for sweep in range(max_sweeps):
        improved = False
        for i in range(n):
            for j in range(i+1, n):
                if h_bin[i] == h_bin[j]:
```
