def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:

        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break

        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:

            idx = np.argmin(h)

            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)

def upsample_h(h, new_n):
    old_n = len(h)

        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_lbgfb(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n

                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})

def binary_local_search_deterministic(h_bin, n, max_sweeps=50):
    """Deterministic local search: try swapping each 1 with each nearby 0."""
    h = h_bin.copy()
    current = compute_c5(h)
    window = 8  # Look within this window for swaps
    
    for sweep in range(max_sweeps):
        improved = False
        ones = list(np.where(h > 0.5)[0])
        zeros = list(np.where(h <= 0.5)[0])
        
        for i in range(len(ones)):
            oi = ones[i]
            best_gain = 0
            best_j = -1
            # Try swapping with nearby zeros
            for j in zeros:
                oj = j
                dist = min(abs(oi - oj), n - abs(oi - oj))  # circular distance
                if dist > window:
                    continue
                h[oi] = 0.0
                h[oj] = 1.0
                c = compute_c5(h)
                h[oi] = 1.0
                h[oj] = 0.0

                ones[i] = best_j

(the sum is over valid i)

So result[k] corresponds to shift (k - n + 1), ranging from -(n-1) to n-1.

The C₅ is max of this array times dx.

In my compute_c5, I do exactly this. Good.

Let me also think about whether the "shift" interpretation matters. For the Erdős problem, we're looking at h(x)(1-h(x+k)) integrated over x where both are defined. For the discretized version, this corresponds to the linear correlation (not circular).

For shift k (in the range -(n-1) to n-1):
overlap(k) = dx * Σ_{i=max(0,-k)}^{min(n-1, n-1-k)} h[i] * (1-h[i+k])

The maximum over all these shifts is C₅.

This is exactly what np.correlate(h, 1-h, mode='full') * dx gives.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def binarize_feasible(h, n):
    h_b = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_b))
    target = n // 2
    rng = np.random.RandomState(0)
    if n_ones > target:
        idx = np.where(h_b == 1)[0]
        rng.shuffle(idx)
        for k in range(n_ones - target):
            h_b[idx[k]] = 0.0
    elif n_ones < target:
        idx = np.where(h_b == 0)[0]
        rng.shuffle(idx)
        for k in range(target - n_ones):
            h_b[idx[k]] = 1.0
    return h_b

def soft_lbgfb(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def topk_objective(h, K, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argsort(corr)[::-1][:K]
    return np.mean(corr[idx]) * (2.0 / n)

def local_search_singleswap(h_bin, n, rng, n_iter=20000, window=None):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    accepted = 0
    for it in range(n_iter):
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        if window is not None:
            dist = abs(oi - oj)
            if dist > window and dist < n - window:
                continue
        h[oi] = 0.0
        h[oj] = 1.0
        c = compute_c5(h)
        if c < current - 1e-15:
            current = c
            ones[ii] = oj
            zeros[jj] = oi
            accepted += 1
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return h, current

def local_search_pairswap(h_bin, n, rng, n_iter=20000):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for it in range(n_iter):
        i1 = rng.randint(len(ones))
        i2 = rng.randint(len(ones))
        j1 = rng.randint(len(zeros))
        j2 = rng.randint(len(zeros))
        if i1 == i2 or j1 == j2:
            continue
        o1, o2 = ones[i1], ones[i2]
        z1, z2 = zeros[j1], zeros[j2]
        h[o1] = 0.0; h[o2] = 0.0
        h[z1] = 1.0; h[z2] = 1.0
        c = compute_c5(h)
        if c < current - 1e-15:
            current = c
            ones[i1], ones[i2] = z1, z2
            zeros[j1], zeros[j2] = o1, o2
        else:
            h[o1] = 1.0; h[o2] = 1.0
            h[z1] = 0.0; h[z2] = 0.0
    return h, current

def simulated_annealing(h_bin, n, rng, T_start=0.3, T_end=0.003, n_iter=40000):
    h = h_bin.copy()
    current = compute_c5(h)
    best = h.copy()
    best_c = current
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (float(it) / n_iter)
        use_pair = (rng.random() < 0.3)
        if use_pair and len(ones) >= 2 and len(zeros) >= 2:
            i1 = rng.randint(len(ones))
            i2 = rng.randint(len(ones))
            j1 = rng.randint(len(zeros))
            j2 = rng.randint(len(zeros))
            if i1 == i2 or j1 == j2:
                continue
            o1, o2 = ones[i1], ones[i2]
            z1, z2 = zeros[j1], zeros[j2]
            h[o1]=0; h[o2]=0; h[z1]=1; h[z2]=1
            c = compute_c5(h)
            delta = c - current
            if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta/T)):
                current = c
                ones[i1], ones[i2] = z1, z2
                zeros[j1], zeros[j2] = o1, o2
                if current < best_c:
                    best_c = current
                    best = h.copy()
            else:
                h[o1]=1; h[o2]=1; h[z1]=0; h[z2]=0
        else:
            i = rng.randint(len(ones))
            j = rng.randint(len(zeros))
            o, z = ones[i], zeros[j]
            h[o] = 0.0; h[z] = 1.0
            c = compute_c5(h)
            delta = c - current
            if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta/T)):
                current = c
                ones[i] = z
                zeros[j] = o
                if current < best_c:
                    best_c = current
                    best = h.copy()
            else:
                h[o] = 1.0; h[z] = 0.0
    return best, best_c

def fourier_restart(h_bin, n, rng, cutoff):
    H = np.fft.fft(h_bin)
    hz = np.zeros_like(H)
    n_keep = max(1, int(n * cutoff / 2))
    hz[:n_keep+1] = H[:n_keep+1]
    hz[n-n_keep:] = H[n-n_keep:]
    h_smooth = np.real(np.fft.ifft(hz))
    h_thresh = (h_smooth > 0.5).astype(float)
    return binarize_feasible(h_thresh, n)

    rng = np.random.RandomState(seed)
    def tl():
