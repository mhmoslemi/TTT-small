import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return np.clip(h, 0.0, 1.0)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def subgrad_at_max(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = int(np.argmax(corr))
    j = m - (n - 1)
    g = np.zeros(n)
    if j == 0:
        g[:] = 1.0 - 2.0 * h
    elif j > 0:
        g[:n - j] += (1.0 - h[j:])
        g[j:] += -h[:n - j]
    else:
        jj = -j
        g[jj:] += (1.0 - h[:n - jj])
        g[:n - jj] += -h[jj:]
    return g, m

def subgrad_top3(h, n):
    corr = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(corr, -3)[-3:]
    g = np.zeros(n)
    for m in idx:
        j = int(m) - (n - 1)
        if j == 0:
            g += (1.0 - 2.0 * h)
        elif j > 0:
            g[:n - j] += (1.0 - h[j:])
            g[j:] += -h[:n - j]
        else:
            jj = -j
            g[jj:] += (1.0 - h[:n - jj])
            g[:n - jj] += -h[jj:]
    g /= 3.0

def lbgfb_softmax(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -80, 80)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def subgrad_nesterov(h0, n, n_iters, deadline, lr_init=0.01, mom_init=0.88):

    mom = mom_init

            for m in idx3:
                j = int(m) - (n_try - 1)
                if j == 0:
                    g += (1.0 - 2.0 * h_s)
                elif j > 0:
                    g[:n_try - j] += (1.0 - h_s[j:])
                    g[j:] += -h_s[:n_try - j]
                else:
                    jj = -j
                    g[jj:] += (1.0 - h_s[:n_try - jj])
                    g[:n_try - jj] += -h_s[jj:]
            g /= 3.0
            # Small step in -g direction
            for lr in (0.008, 0.004, 0.002):
                trial = make_feasible(h_s - lr * g, n_try)
                c_t = compute_c5(trial)
                if c_t < c_max:
                    h_s = trial
                    c_max = c_t
                    break

        c_s = compute_c5(h_s)
        if c_s < best_c5:
            best_c5, best_h = c_s, h_s.copy()

    # Phase 4: Local search (~250s)
    dl4 = t0 + 1050
    n_ls = min(len(best_h), 1024)
    h_ls = upsample(best_h, n_ls) if n_ls < len(best_h) else best_h.copy()
    dl4a = t0 + 950
    h_ls, c_ls = local_search(h_ls, n_ls, dl4a, rng, 20000,
                               [0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.004, -0.004])
    if c_ls < best_c5:
        best_c5, best_h = c_ls, h_ls.copy()
    # Finer local search
    if rem() > 80:
        h_f, c_f = local_search(best_h[:n_ls], n_ls, dl4, rng, 15000,
                                 [0.01, -0.01, 0.005, -0.005, 0.002, -0.002])
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    # Phase 5: SLSQP active-set polish (~100s)
    if rem() > 60:
        n_sp = min(len(best_h), 960)
        h_sp = upsample(best_h, n_sp) if n_sp < len(best_h) else best_h.copy()
        dl5 = t0 + budget_s - 20
        for frac in (0.03, 0.06, 0.10):
            if time.time() > dl5:
                break
            h_o = slsqp_active(h_sp, n_sp, frac, 80)
            c_o = compute_c5(h_o)
            if c_o < best_c5:
                best_c5, best_h = c_o, h_o.copy()
                h_sp = h_o.copy()

    # Phase 6: Final tiny local search
    if rem() > 10:
        n_f = len(best_h)
        h_f, c_f = local_search(best_h, n_f, t0 + budget_s - 2, rng, 3000,
                                 [0.005, -0.005, 0.002, -0.002, 0.001, -0.001])
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
