from scipy.optimize import minimize
import time

_N = 0
_ACTIVE_IDX = None

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def epigraph_obj(x):
    t = x[_N]
    return t

def epigraph_constr(x):
    n = _N
    h = x[:n]
    t = x[n]
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[_ACTIVE_IDX] * dx - t

def epigraph_constr_jac(x):
    n = _N
    h = x[:n]
    jac = np.zeros((len(_ACTIVE_IDX), n + 1))
    for row, k in enumerate(_ACTIVE_IDX):
        m = k - n + 1
        for j in range(n):
            val = 1.0
            if 0 <= j + m < n:
                val -= h[j + m]
            if 0 <= j - m < n:
                val -= h[j - m]
            jac[row, j] = val
        jac[row, n] = -1.0
    return jac

def subgradient_step(h, alpha, rng):
    n = len(h)
    dx = 2.0 / n
    corr = get_corr(h)
    cmax = np.max(corr)
    # Find all shifts within 1% of max (active set)
    active = np.where(corr >= cmax * 0.99)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    k_star = active[rng.randint(len(active))]
    m = k_star - n + 1
    g = np.ones(n)
    for j in range(n):
        if 0 <= j + m < n:
            g[j] -= h[j + m]
        if 0 <= j - m < n:
            g[j] -= h[j - m]
    # Projected gradient step
    y = h - alpha * g * dx
    return make_feasible(y, n)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
