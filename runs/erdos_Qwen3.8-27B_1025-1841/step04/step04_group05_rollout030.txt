                    best_c5 = c5
                    improved = True
                    old_val = new_val
                else:
                    h[j] = old_val
                    h = make_feasible(h, N)
            for bv in [0.0, 1.0]:
                if abs(bv - h[j]) < 1e-9:
                    continue
                h[j] = bv
                h = make_feasible(h, N)
                c5 = compute_c5(h)
                if c5 < best_c5:
                    best_c5 = c5
                    improved = True
                else:
                    h[j] = old_val
                    h = make_feasible(h, N)
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    N = 4096
    rho = 50.0

    if h0 is not None:
        best_h = upsample_h(h0, N)
    else:
        best_h = upsample_h(random_binary(N, rng), N)

    best_c5 = compute_c5(best_h)

    # Phase 1: Multi-temperature L-BFGS-B
    temp_schedule = [(200, 200), (100, 200), (50, 200), (20, 200), (10, 200), (5, 200), (3, 150)]
    for T, max_it in temp_schedule:
        if time_left() < 40:
            break
        h_opt = lbfgs_run(best_h, N, T, rho, max_iter=max_it, top_k=40)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        else:
            # Still use the optimized point (may be at a different local min)
            best_h = h_opt

    # Phase 2: Subgradient polish
    if time_left() > 60:
        h_sg, c5_sg = subgradient_optimize(best_h, N, n_iter=2000, init_step=0.03)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    # Phase 3: Another L-BFGS-B round from the subgradient result
    if time_left() > 40:
        h_opt = lbfgs_run(best_h, N, 10, rho, max_iter=200, top_k=40)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Random binary restarts
    for restart in range(3):
        if time_left() < 80:
            break
        h_bin = random_binary(N, rng)
        # Blend with current best
        h_blend = 0.5 * best_h + 0.5 * h_bin
        h_blend = make_feasible(h_blend, N)
        h_opt = lbfgs_run(h_blend, N, 50, rho, max_iter=200, top_k=40)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 5: Coordinate descent at reduced resolution
    if time_left() > 40:
        cd_N = 1024
        h_cd = upsample_h(best_h, cd_N)
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = upsample_h(h_cd, N)
        h_cd2, c5_cd2 = cd_polish(h_cd, cd_N, n_rounds=3, remaining_fn=time_left)
        if c5_cd2 < best_c5:
            best_c5 = c5_cd2
            best_h = upsample_h(h_cd2, N)
        elif c5_cd < best_c5:
            best_h = upsample_h(h_cd, N)

    # Phase 6: Final polish at full resolution
    if time_left() > 30:
        h_opt = lbfgs_run(best_h, N, 5, rho, max_iter=150, top_k=50)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final subgradient if time
    if time_left() > 30:
        h_sg, c5_sg = subgradient_optimize(best_h, N, n_iter=500, init_step=0.01)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
