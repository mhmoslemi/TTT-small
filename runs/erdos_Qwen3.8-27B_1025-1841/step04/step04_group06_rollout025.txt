Let me verify:

So I can compute the correlation via FFT-based convolution:
```python
def fast_correlate(a, b):
    fft_size = 1
    size = len(a) + len(b) - 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b[::-1], fft_size)
    result = np.fft.irfft(fa * fb, fft_size)[:size][::-1]
    return result
```

Let me also think about the local search more carefully.

For the local search, the key question is: which positions to modify?

If the maximum correlation is at lag k*, then the correlation at lag k* is:
c(k*) = Σ_{i=0}^{n-k*-1} h[i] * (1-h[i+k*])

To reduce c(k*), we want to either:
- Decrease h[i] where h[i+k*] = 0 (reduces the product)
- Increase h[i+k*] where h[i] = 1 (reduces the (1-h) factor)

But we also need to not increase other correlations above c(k*).

A smart local search would:
1. Find the top few correlations (say top 5)
2. For each, identify the "contributing" positions
3. Try moves that reduce the top correlations without increasing others

- Cool T over time

def fft_max_corr(h):
    """Fast computation of max(correlation) * dx using FFT"""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    size = 2 * n - 1
    fft_size = 1
    while fft_size < size:

    fa = np.fft.rfft(h, fft_size)
    fb = np.fft.rfft(b[::-1], fft_size)
    corr = np.fft.irfft(fa * fb, fft_size)[:size][::-1] * dx
    return float(np.max(corr))

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def fft_max_corr(h):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    size = 2 * n - 1
    fs = 1
    while fs < size:
        fs <<= 1
    fa = np.fft.rfft(h, fs)
    fb = np.fft.rfft(b[::-1], fs)
    corr = np.fft.irfft(fa * fb, fs)[:size][::-1] * dx
    return float(np.max(corr))

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    m = len(h_vec)
    dx = 2.0 / m
    b = 1.0 - h_vec
    size = 2 * m - 1
    fs = 1
    while fs < size:

    corr = np.fft.irfft(fa * fb, fs)[:size][::-1] * dx
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -700, 700)
    soft = mc + np.log(np.sum(np.exp(shifted))) / T
    viol = np.sum(h_vec) - target
    return soft + mu * viol * viol

                                       min(150000, int(remaining() * 6000)), remaining, best_ref)
            if c5_sahi < best_c5:
                best_c5 = c5_sahi
                best_h = hb_hi.copy()
                best_ref = [best_c5, best_h.copy()]

    # Phase: Continuous coordinate descent on best
    if remaining() > 15:
        n_cd = len(best_h)
        h_cd = best_h.copy()
        c5_cd = fft_max_corr(h_cd)
        for rnd in range(5):
            if remaining() < 8:
                break
            improved = False
            step_size = 2 if n_cd > 1500 else 1
            for j in range(0, n_cd, step_size):
                if remaining() < 4:
                    break
                ov = h_cd[j]
                for d in [0.05, -0.05, 0.1, -0.1, 0.02, -0.02]:
                    nv = ov + d
                    if nv < 0 or nv > 1:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_cd)
                    c = fft_max_corr(tr)
                    if c < c5_cd - 1e-13:
                        c5_cd = c
                        h_cd = tr
                        improved = True
                        if c5_cd < best_c5:
                            best_c5 = c5_cd
                            best_h = h_cd.copy()
                            best_ref = [best_c5, best_h.copy()]
                        break
            if not improved:
                break

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
