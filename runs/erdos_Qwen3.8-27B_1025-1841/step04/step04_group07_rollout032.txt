    weights has same length as corr (2n-1).
    corr[k] = sum_n h[n+k] * (1-h[n]) for valid indices (numpy correlate 'full' convention)

def lbgfb_with_grad(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    
    def obj_and_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -80, 80)
        obj = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx + mu * (x.sum() - n/2.0)**2
        
        # Gradient of soft-max part
        weights = np.exp(shifted) / np.sum(np.exp(shifted))
        grad_sm = _grad_weighted_corr(x, weights) * dx
        
        # Gradient of penalty
        grad_pen = 2 * mu * (x.sum() - n/2.0)
        
        return obj, grad_sm + grad_pen
    
    res = minimize(obj_and_grad, h0, method='L-BFGS-B',
                   jac=True,
                   bounds=[(0.0, 1.0)] * n,

        h_new = make_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < cur - 1e-15:
            cur = c
            h = h_new
            if c < best:
                best = c
                best_h = h.copy()
    return best_h, best

def coord_descent(h0, n, steps, max_rounds=3):
    h = h0.copy()
    cur = compute_c5(h)
    for _ in range(max_rounds):
        improved = False
        for j in range(n):
            for d in steps:
                h_try = h.copy()
                h_try[j] = np.clip(h_try[j] + d, 0.0, 1.0)
                h_try = make_feasible(h_try, n)
                c_try = compute_c5(h_try)
                if c_try < cur - 1e-14:
                    cur = c_try
                    h = h_try
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = len(h0)

    # Phase 1: Screen resolutions
    n_list = sorted(set([best_n, 800, 1000, 1280, 1600, 2000, 2560, 3200]))
    candidates = []
    for n_try in n_list:
        if rem() < 250:
            break
        h_t = upsample_h(best_h, n_try)
        c_t = compute_c5(h_t)
        candidates.append((c_t, h_t, n_try))

    # Phase 2: L-BFGS-B with analytical gradients
    for c_init, h_cand, n_cand in sorted(candidates, key=lambda x: x[0])[:5]:
        if rem() < 80:
            break
        for T in [200, 1000, 5000]:
            if rem() < 40:
                break
            h_opt = lbgfb_optimize(h_cand, n_cand, T, mu=5000.0, max_iter=300)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
                best_n = n_cand
            h_cand = h_opt

    # Phase 3: Projected gradient on best
    if rem() > 80:
        h_pg, c_pg = proj_grad_opt(best_h, best_n, n_iter=400, T=500, topk_frac=0.1, lr_init=0.01)
        if c_pg < best_c5:
            best_c5 = c_pg
            best_h = h_pg.copy()

    # Phase 4: Higher-T L-BFGS-B for sharper max
    for T in [10000, 50000]:
        if rem() < 40:
            break
        h_opt = lbgfb_optimize(best_h, best_n, T, mu=10000.0, max_iter=250)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Phase 5: Upscale to higher resolution
    n_up = min(best_n * 2, 5000)
    if n_up > best_n and rem() > 100:
        h_u = upsample_h(best_h, n_up)
        c_u = compute_c5(h_u)
        if c_u < best_c5:
            best_c5 = c_u
            best_h = h_u.copy()
            best_n = n_up
        if rem() > 60:
            h_opt = lbgfb_optimize(best_h, best_n, 2000, mu=5000.0, max_iter=250)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()
        if rem() > 40:
            h_pg2, c_pg2 = proj_grad_opt(best_h, best_n, n_iter=200, T=1000, topk_frac=0.08, lr_init=0.008)
            if c_pg2 < best_c5:
                best_c5 = c_pg2
                best_h = h_pg2.copy()

    # Phase 6: Local search
    if rem() > 50:
        h_ls, c_ls = local_search(best_h, best_n, rng, n_iter=4000, max_step=0.05)
        if c_ls < best_c5:
            best_c5 = c_ls
            best_h = h_ls.copy()

    # Phase 7: Final L-BFGS-B polish
    if rem() > 35:
        h_opt = lbgfb_optimize(best_h, best_n, 20000, mu=20000.0, max_iter=300)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Phase 8: Coordinate descent
    if rem() > 20:
        h_cd, c_cd = coord_descent(best_h, best_n,
                                   steps=[0.01, -0.01, 0.005, -0.005, 0.002, -0.002],
                                   max_rounds=3)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
