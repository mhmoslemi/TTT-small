            js = np.arange(0, n - s)

            # -h[j-s] for j in [s, n-1]
            js2 = np.arange(s, n)

        else:

            js = np.arange(sp, n)

            # -h[j+sp] for j in [0, n-1-sp]
            js2 = np.arange(0, n - sp)
            grad[js2] -= w * h[js2 + sp]
    
    return obj, grad

    return res_x, compute_c5(res_x)

def fourier_perturb(h, n, rng, n_modes=4, amp=0.05):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 5)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.3, 1.5)
    return make_feasible(h_new, n)

def block_perturb(h, n, rng):
    h_new = h.copy()
    n_segs = rng.randint(1, 4)
    for _ in range(n_segs):
        length = rng.randint(n // 16, n // 6)
        start = rng.randint(0, max(1, n - length))
        if rng.random() < 0.4:
            h_new[start:start + length] = 1.0 - h_new[start:start + length]
        elif rng.random() < 0.5:
            h_new[start:start + length] = rng.uniform(0.2, 0.8)
        else:
            h_new[start:start + length] = rng.uniform(0, 1)
    return make_feasible(h_new, n)

def shift_blend(h, n, rng):
    s = rng.randint(1, n // 6 + 1)
    h_shift = np.roll(h, s)
    alpha = rng.uniform(0.35, 0.65)
    h_new = alpha * h + (1.0 - alpha) * h_shift
    return make_feasible(h_new, n)

def binary_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if it % 500 == 0:
            frac = float(it) / max_iters
            T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def block_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while int(np.sum(b)) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        if it % 500 == 0:
            frac = float(it) / max_iters
            T = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, 10)
        start = rng.randint(0, max(1, n - bsize))
        end = start + bsize
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        step_sizes = [0.0, 1.0, 0.05, 0.02, 0.005, -0.02, -0.05]
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for ss in step_sizes:
                cand = old + ss
                if cand <= 0.0:
                    cand = 0.0
                elif cand >= 1.0:
                    cand = 1.0
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(960, 0.5)

    n = len(h0)
    h0 = make_feasible(h0, n)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Multi-temperature L-BFGS-B from parent
    temps = [80.0, 300.0, 1000.0, 5000.0, 20000.0]
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        try:
            h_r, c_r = lbfgsb_opt(best_h, n, T, maxiter=250, ftol=1e-14)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()
        except Exception:
            pass

    # Phase 2: Perturbation + reoptimization
    perts = [fourier_perturb, block_perturb, shift_blend]
    for pf in perts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 100:
            break
        for _ in range(3):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 50:
                break
            try:
                hp = pf(best_h, n, rng)
                hr, cr = lbfgsb_opt(hp, n, 8000.0, maxiter=200, ftol=1e-14)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 3: Binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        tl = min(remaining - 50, 150)
        b, c5_b = binary_sa(best_h, n, 200000, 0.005, 0.000005, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                hr, cr = lbfgsb_opt(best_h, n, 10000.0, maxiter=200)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()

    # Phase 4: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        tl = min(remaining - 40, 120)
        b, c5_b = block_sa(best_h, n, 100000, 0.004, 0.00002, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 50:
                hr, cr = lbfgsb_opt(best_h, n, 15000.0, maxiter=200)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()

    # Phase 5: High-T L-BFGS-B polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        try:
            hr, cr = lbfgsb_opt(best_h, n, 50000.0, maxiter=400, ftol=1e-15)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()
        except Exception:
            pass
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            try:
                hr, cr = lbfgsb_opt(best_h, n, 100000.0, maxiter=300, ftol=1e-15)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 6: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        best_h, best_c5 = coord_descent(best_h.copy(), n, 25, start_time, budget_s)

    # Phase 7: Higher resolution attempt
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        n1 = n + 480
        old_x = np.linspace(0.0, 2.0, n)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        try:
            h_up, c5_up = lbfgsb_opt(h_up, n1, 5000.0, maxiter=200)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 100:
                h_up, c5_up = lbfgsb_opt(best_h, n1, 50000.0, maxiter=300)
                if c5_up < best_c5:
                    best_c5 = c5_up
                    best_h = h_up.copy()
        except Exception:
            pass

    # Phase 8: Final coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 10, start_time, budget_s)

    # Phase 9: Final L-BFGS-B polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 25:
        n_pts = len(best_h)
        try:
            hr, cr = lbfgsb_opt(best_h, n_pts, 200000.0, maxiter=200, ftol=1e-15)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
