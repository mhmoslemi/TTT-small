            block_len = rng.randint(2, min(12, n // 4))
            start = rng.randint(0, n - block_len)
            seg_end = start + block_len
            seg_ones = np.where(b[start:seg_end] == 1)[0]
            seg_zeros = np.where(b[start:seg_end] == 0)[0]
            if len(seg_ones) == 0 or len(seg_zeros) == 0:
                stall += 1
                continue
            n_swap = min(len(seg_ones), len(seg_zeros))
            if n_swap == 0:
                stall += 1
                continue
            chosen_o = rng.choice(seg_ones, n_swap, replace=False)
            chosen_z = rng.choice(seg_zeros, n_swap, replace=False)
            b[start + chosen_o] = 0
            b[start + chosen_z] = 1
            new_corr = np.correlate(b, 1.0 - b, mode='full')
            new_c5 = float(np.max(new_corr)) * dx
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                corr = new_corr
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                b[start + chosen_o] = 1
                b[start + chosen_z] = 0
                stall += 1
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            b[i] = 0
            b[j] = 1
            new_corr = np.correlate(b, 1.0 - b, mode='full')
            new_c5 = float(np.max(new_corr)) * dx
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                corr = new_corr
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                b[i] = 1
                b[j] = 0
                stall += 1
        if stall > stall_limit:
            break
    return best_b, best_c5

def softobj(x):
    n = _N_G
    T = _T_G
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _N_G, _T_G
    _N_G = n
    _T_G = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(softobj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, deadline):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        if time.time() > deadline - 3:
            break
        for j in range(n):
            if time.time() > deadline - 2:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01),
                     max(0.0, old - 0.003), min(1.0, old + 0.003)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def random_binary(n, rng):
    b = np.zeros(n)
    half = n // 2
    idx = rng.choice(n, half, replace=False)
    b[idx] = 1.0
    return b

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)
    
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Phase 0: Quick multi-start comparison
    deadline_p0 = start_time + 0.08 * budget_s
    for trial_seed in [0, 1, 2]:
        if time.time() > deadline_p0:
            break
        b_rand = random_binary(n0, np.random.RandomState(seed + 1000 + trial_seed))
        b_test, c5_test = targeted_sa(b_rand, n0, 30000, 0.02, 0.001, rng, start_time, time.time() + 10)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = b_test.copy()
    
    # Phase 1: Long targeted SA from best so far
    deadline_p1 = start_time + 0.30 * budget_s
    b1, c5_1 = targeted_sa(best_h, n0, 800000, 0.012, 0.00001, rng, start_time, deadline_p1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()
    
    # Phase 2: SLSQP continuation (decreasing T)
    deadline_p2 = start_time + 0.50 * budget_s
    for T in [300.0, 100.0, 50.0, 20.0, 10.0, 5.0, 3.0]:
        if time.time() > deadline_p2:
            break
        try:
            x_res = slsqp_run(best_h, n0, T, 250, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass
    
    # Phase 3: SA again from polished solution
    deadline_p3 = start_time + 0.65 * budget_s
    b3, c5_3 = targeted_sa(best_h, n0, 400000, 0.008, 0.000003, rng, start_time, deadline_p3)
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = b3.copy()
    
    # Phase 4: Final SLSQP
    deadline_p4 = start_time + 0.78 * budget_s
    for T in [8.0, 4.0, 2.0, 1.0]:
        if time.time() > deadline_p4:
            break
        try:
            x_res = slsqp_run(best_h, n0, T, 400, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass
    
    # Phase 5: Coordinate descent
    deadline_p5 = start_time + 0.90 * budget_s
    best_h, best_c5 = coord_descent(best_h.copy(), n0, 40, start_time, deadline_p5)
    
    # Phase 6: One more SLSQP
    if time.time() < start_time + budget_s - 40:
        try:
            x_res = slsqp_run(best_h, n0, 2.0, 300, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass
    
    # Phase 7: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150 and n0 >= 480:
        n1 = n0 * 2
        h_up = upsample_h(best_h, n1)
        deadline_p7 = start_time + budget_s - 25
        b_up, c5_up = targeted_sa(h_up, n1, 150000, 0.01, 0.00002, rng, start_time, deadline_p7)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = b_up.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 50:
            try:
                x_res = slsqp_run(best_h, len(best_h), 10.0, 200, 1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, len(best_h))
            except Exception:
                pass
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
