def thue_morse(n):
    """Generate first n bits of Thue-Morse sequence."""
    tm = np.zeros(n, dtype=float)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm

def thue_morse_init(n):
    """Create a binary initialization from Thue-Morse with exactly n/2 ones."""
    tm = thue_morse(n)
    target = n // 2
    current_ones = int(np.sum(tm))
    if current_ones > target:
        # Flip some 1s to 0s (prefer those with most 1-neighbor conflicts)
        ones = np.where(tm == 1)[0]
        for idx in ones:
            if current_ones == target:
                break
            tm[idx] = 0
            current_ones -= 1
    elif current_ones < target:
        zeros = np.where(tm == 0)[0]
        for idx in zeros:
            if current_ones == target:

            tm[idx] = 1

    #   So: 0 <= j-t+n-1 <= 2n-2 → 1-n+j <= t <= j+n-1
    #   Combined with 0 <= t <= n-1: max(0, 1-n+j) <= t <= min(n-1, j+n-1)
    
    # For term 2: grad[j] -= sum_{s: 0<=j+s<n} w[s_idx] * x[j+s]
    #   Let t = j+s, so s = t-j, s_idx = t-j+n-1
    #   t ranges over [j, n-1] (since 0<=j+s<n → t>=j and t<n)
    #   And s_idx must be in [0, 2n-2]: 0 <= t-j+n-1 <= 2n-2 → 1-j-n+1 <= t <= 2n-2+j-n+1 = n-1+j
    #   Combined: max(j, -j) <= t <= min(n-1, n-1+j) → j <= t <= n-1 (since j >= 0)
    
    # Let me just do the loop - it's O(n^2) but with numpy inner ops
    for j in range(n):
        # Term 1
        # s_idx range: max(0, j+n-1-(n-1)) to min(2n-2, j+n-1) = max(0,j) to min(2n-2, j+n-1)
        lo1 = max(0, j)
        hi1 = min(2*n-1, j + n)  # exclusive
        if lo1 < hi1:
            s_idxs = np.arange(lo1, hi1)
            t = j + n - 1 - s_idxs  # = j - s = position in x for (1-x[.])
            valid = (t >= 0) & (t < n)
            if np.any(valid):
                grad[j] += np.dot(w[s_idxs[valid]], (1.0 - x[t[valid]]))
        
        # Term 2
        # s_idx range: max(0, n-1-j) to min(2n-2, 2n-2-j) inclusive → [max(0,n-1-j), min(2n-2,2n-2-j)+1)
        lo2 = max(0, n - 1 - j)
        hi2 = min(2*n - 1, 2*n - j)  # exclusive, since we want s_idx <= 2n-2-j → s_idx < 2n-1-j
        if lo2 < hi2:
            s_idxs = np.arange(lo2, hi2)
            t = j + s_idxs - (n - 1)  # = j + s = position in x for x[.]
            valid = (t >= 0) & (t < n)
            if np.any(valid):
                grad[j] -= np.dot(w[s_idxs[valid]], x[t[valid]])
    
    grad *= dx
    return obj, grad

def lbfghb_run(h, T, maxiter, factr):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    
    def fun(x):
        o, _ = logsumexp_obj_grad(x, T)
        return o
    
    def fprime(x):
        _, g = logsumexp_obj_grad(x, T)
        return g
    
    try:
        res = minimize(fun, h, jac=fprime, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'factr': factr, 'disp': False})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h, n)

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    if target * 2 != n:
        target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        # Remove the one with lowest h value
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def sa_binary(h, n, max_iters, T0, T1, rng, time_limit):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    start = time.time()
    stagger = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        
        # Every 1000 iterations, try a bigger move
        big_move = (it % 2000 < 100)
        
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        if big_move:
            # Block move: swap two blocks of equal number of ones
            blen = rng.randint(4, 16)
            s1 = rng.randint(0, n - blen)
            s2 = rng.randint(0, n - blen)
            if abs(s1 - s2) < blen + 2:
                continue
            b1 = b[s1:s1+blen].copy()
            b2 = b[s2:s2+blen].copy()
            if np.sum(b1) != np.sum(b2):
                continue
            b[s1:s1+blen] = b2
            b[s2:s2+blen] = b1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_b = b.copy()
            else:
                b[s1:s1+blen] = b1
                b[s2:s2+blen] = b2
        else:
            # Single point swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if curr_c5 < best_c5:
                    best_c5 = curr_c5
                    best_b = b.copy()
            else:
                b[i] = 1
                b[j] = 0
    
    return best_b, best_c5

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
