def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def active_set_step(h, n, dx, top_k=5, alpha=0.01):
    """Take a subgradient step focusing on top-k worst lags."""
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Get top-k indices
    top_idx = np.argsort(corr)[-top_k:]
    # Compute weighted subgradient (equal weight on top-k)
    grad = np.zeros(n)
    for idx in top_idx:
        s = idx - (n - 1)  # shift amount
        # dC(s)/dh[j] = dx * [(1-h[j-s])_valid - h[j+s]_valid]
        if s >= 0:
            # First term: j in [s, n-1], (1-h[j-s])

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final fix
    if abs(np.sum(h) - target) > 1e-12:
        # Distribute remaining error
        remaining = target - np.sum(h)
        if remaining > 0:
            # Add to smallest values
            order = np.argsort(h)
            for idx in order:
                add = min(remaining, 1.0 - h[idx])
                h[idx] += add
                remaining -= add

    return corr  # length 2n-1, index i corresponds to lag i-(n-1)

    """Compute dC(s)/dh for a given shift s."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        # dC/dh[j] = dx * [(1-h[j+s]) for j=0..n-1-s] - dx * [h[j-s] for j=s..n-1]
        grad[:n-s] += (1.0 - h[s:n])  # first term, j from 0 to n-1-s
        grad[s:] -= h[:n-s]            # second term, j from s to n-1
    else:
        t = -s
        # dC/dh[j] = dx * [(1-h[j-t]) for j=t..n-1] - dx * [h[j+t] for j=0..n-1-t]

        grad[:n-t] -= h[t:n]           # second term, j from 0 to n-1-t

        else:
            alpha *= 0.8
            if alpha < 1e-6:
                break

        alpha = max(alpha, 1e-7)
