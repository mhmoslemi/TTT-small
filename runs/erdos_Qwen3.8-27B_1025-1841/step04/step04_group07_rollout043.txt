def simulated_annealing_binary(h_bin, n, rng, max_iters=20000, T_start=1.0, T_end=0.01):
    """Simulated annealing on binary solutions."""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T_start
    cool_rate = (T_end / T_start) ** (1.0 / max_iters)
    
    for it in range(max_iters):
        T *= cool_rate
        ii = rng.randint(n)
        jj = rng.randint(n)
        if h[ii] == h[jj]:
            continue
        h[ii], h[jj] = h[jj], h[ii]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[ii], h[jj] = h[jj], h[ii]
    return best_h, best_c5

def alternating_minmax_v2(h0, n, n_iters=60, topk_frac=0.15, max_inner=40):

def slsqp_polish(h0, n, T, max_iter=100):

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_obj(corr, T) * dx

        return np.sum(x) - n / 2.0

        return np.ones(n)

                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],

    return make_feasible(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):

    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    return mc + np.log(np.sum(np.exp(shifted))) / T

def subgradient_descent(h0, n, n_iters=500, lr0=0.05, decay=1.05):
    """Subgradient descent on the max correlation."""
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()

    lr = lr0

    def obj(x):

        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

        for _ in range(n_tries):

                h[oj] = 1.0

            else:
                # Position shift: move a 1 by a small distance
                ii = rng.randint(len(ones))
                oi = ones[ii]
                shift = rng.randint(-12, 13)
                if shift == 0:
                    continue
                oj = (oi + shift) % n
                if h[oj] > 0.5:
                    continue
                h[oi] = 0.0
                h[oj] = 1.0
                c = compute_c5(h)
                if c < cur - 1e-15:
                    cur = c
                    ones[ii] = oj
                    zeros.append(oi)
                    # Remove oj from zeros if present
                    try:
                        zeros.remove(oi)
                    except ValueError:
                        pass
                    improved = True
                else:
                    h[oi] = 1.0
                    h[oj] = 0.0
        if not improved:
            break
    return h, cur

def simulated_annealing_binary(h_bin, n, rng, max_iters=15000, T_start=0.5, T_end=0.003):
    """Simulated annealing on binary solutions with swap moves."""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    T = T_start
    cool_rate = (T_end / T_start) ** (1.0 / max_iters)
    for it in range(max_iters):
        T *= cool_rate
        ii = rng.randint(n)
        jj = rng.randint(n)
        if h[ii] == h[jj]:
            continue
        h[ii], h[jj] = h[jj], h[ii]
        new_c5 = compute_c5(h)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-12)):
            cur_c5 = new_c5
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = h.copy()
        else:
            h[ii], h[jj] = h[jj], h[ii]
    return best_h, best_c5

def targeted_binary_search(h_bin, n, rng, max_rounds=60, tries_per_round=5000):
    """Binary search that targets the worst lags."""
    h = h_bin.copy()
    cur = compute_c5(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    for rnd in range(max_rounds):
        # Re-identify worst lags
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        max_lag = int(np.argmax(corr))
        improved = False
        for _ in range(tries_per_round):
            # Try to reduce correlation at the worst lag
            # Random swap that might help
            ii = rng.randint(n)
            jj = rng.randint(n)
            if h[ii] == h[jj]:
                continue
            h[ii], h[jj] = h[jj], h[ii]
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                improved = True
            else:
                h[ii], h[jj] = h[jj], h[ii]
        if not improved:
            break
    return h, cur

def slsqp_polish(h0, n, T, max_iter=100):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_obj(corr, T) * dx
    def con_eq(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    res = minimize(obj, h0, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def coord_descent_polish(h0, n, max_rounds=10, step_list=None):
    if step_list is None:
        step_list = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in step_list:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def alternating_minmax_v2(h0, n, n_iters=50, max_inner=35):
    dx = 2.0 / n
    h = h0.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = corr.max()
        # Weight by proximity to max
        weights = np.exp(np.clip(50.0 * (corr - mc), -30, 0))
        if weights.sum() < 1e-10:
            weights = np.zeros(len(corr))
            weights[np.argmax(corr)] = 1.0
        weights /= weights.sum()

        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(weights, c)) * dx + 500.0 * (x.sum() - n / 2.0) ** 2

        try:
            res = minimize(obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = make_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                h = h_new
        except Exception:
            pass
    return h

def make_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        order = np.argsort(-h[idx])  # remove those closest to 0.5 first
        idx = idx[order]
        h_bin[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        h_bin[idx[:target - n_ones]] = 1.0
    return h_bin

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = len(h0)

    # Phase 1: Smooth optimization with continuation at parent resolution
    n1 = best_n
    if rem() > 120:
        h_cont = continuation_opt(h0, n1,
                                  T_list=[30, 100, 400, 1500, 5000],
                                  mu_list=[2000, 1000, 600, 400, 300],
                                  max_iter_each=150)
        c_cont = compute_c5(h_cont)
        if c_cont < best_c5:
            best_c5, best_h = c_cont, h_cont.copy()

    # Subgradient descent at parent resolution
    if rem() > 80:
        h_sg, c_sg = subgradient_descent(best_h, n1, n_iters=300, lr0=0.03, decay=1.03)
        if c_sg < best_c5:
            best_c5, best_h = c_sg, h_sg.copy()

    # Alternating min-max
    if rem() > 60:
        h_alt = alternating_minmax_v2(best_h, n1, n_iters=40, max_inner=30)
        c_alt = compute_c5(h_alt)
        if c_alt < best_c5:
            best_c5, best_h = c_alt, h_alt.copy()

    # Phase 2: Binary search at multiple resolutions
    for n_bin in [640, 800, 1000]:
        if rem() < 120:
            break
        h_b = upsample_h(best_h, n_bin)
        h_bin = make_binary(h_b, n_bin)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5:
            best_c5, best_h = c_bin, h_bin.copy()

        # Enhanced local search
        h_ls, c_ls = binary_local_search_v2(h_bin, n_bin, rng, max_rounds=60, tries_per_round=6000)
        if c_ls < best_c5:
            best_c5, best_h = c_ls, h_ls.copy()
            h_bin = h_ls.copy()

        # Simulated annealing
        if rem() > 60:
            n_sa = min(10000, max(3000, n_bin * 10))
            h_sa, c_sa = simulated_annealing_binary(h_bin, n_bin, rng,
                                                     max_iters=n_sa, T_start=0.3, T_end=0.002)
            if c_sa < best_c5:
                best_c5, best_h = c_sa, h_sa.copy()

    # Phase 3: Higher resolution smooth optimization
    for n2 in [1000, 1280, 1600]:
        if rem() < 100:
            break
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

        if rem() > 60:
            h_opt = continuation_opt(h2, n2,
                                     T_list=[50, 200, 800, 3000],
                                     mu_list=[800, 600, 400, 300],
                                     max_iter_each=120)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt

        if rem() > 40:
            h_sg2, c_sg2 = subgradient_descent(best_h, n2, n_iters=200, lr0=0.02, decay=1.04)
            if c_sg2 < best_c5:
                best_c5, best_h = c_sg2, h_sg2

    # Phase 4: Multi-start with diverse inits
    if rem() > 80:
        n_ms = min(len(best_h), 800)
        h_ref = upsample_h(best_h, n_ms) if n_ms < len(best_h) else best_h.copy()
        
        # Start 1: noisy version
        h_r1 = make_feasible(h_ref + 0.15 * rng.randn(n_ms), n_ms)
        if rem() > 40:
            h_o1 = lbgfb_smooth(h_r1, n_ms, 1000, mu=600, max_iter=200)
            c_o1 = compute_c5(h_o1)
            if c_o1 < best_c5:
                best_c5, best_h = c_o1, h_o1.copy()

        # Start 2: threshold + subgradient
        h_r2 = make_binary(h_ref, n_ms)
        if rem() > 40:
            h_sg_r, c_sg_r = subgradient_descent(h_r2, n_ms, n_iters=200, lr0=0.02, decay=1.03)
            if c_sg_r < best_c5:
                best_c5, best_h = c_sg_r, h_sg_r.copy()

        # Start 3: smooth from random binary
        h_r3 = make_binary(rng.rand(n_ms), n_ms)
        if rem() > 40:
            h_o3 = lbgfb_smooth(h_r3, n_ms, 500, mu=500, max_iter=200)
            c_o3 = compute_c5(h_o3)
            if c_o3 < best_c5:
                best_c5, best_h = c_o3, h_o3.copy()

    # Phase 5: SLSQP polish
    if rem() > 30:
        n_sp = min(len(best_h), 1000)
        h_sp = upsample_h(best_h, n_sp) if n_sp < len(best_h) else best_h.copy()
        for T in [2000, 8000]:
            if rem() < 15:
                break
            h_opt = slsqp_polish(h_sp, n_sp, T, max_iter=80)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h_sp = h_opt.copy()

    # Phase 6: Coordinate descent
    if rem() > 25:
        n_cd = min(len(best_h), 1000)
        h_cd_in = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        h_cd, c_cd = coord_descent_polish(h_cd_in, n_cd, max_rounds=8,
                                          step_list=[0.03, -0.03, 0.015, -0.015, 0.008, -0.008, 0.003, -0.003])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 7: Final binary refinement
    if rem() > 40:
        n_fin = min(len(best_h), 800)
        h_fin = upsample_h(best_h, n_fin) if n_fin < len(best_h) else best_h.copy()
        h_bin_fin = make_binary(h_fin, n_fin)
        h_ls_fin, c_ls_fin = binary_local_search_v2(h_bin_fin, n_fin, rng, max_rounds=50, tries_per_round=5000)
        if c_ls_fin < best_c5:
            best_c5, best_h = c_ls_fin, h_ls_fin.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
