def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, corr, active_idx, step_size):
    """Take a subgradient step to reduce the max correlation."""
    n = len(h)
    dx = 2.0 / n
    # Subgradient of max(corr[k]*dx) at h is:
    # Average of gradients of active c_k's * dx
    # c_k = sum_i h[i] * (1-h[i+lag_k]) where lag_k = k - (n-1)
    # dc_k/dh[j] = (1-h[j+lag_k]) - h[j-lag_k] (for valid indices)
    
    grad = np.zeros(n)
    for k in active_idx:
        lag = k - (n - 1)  # positive or negative shift in units of dx
        for j in range(n):
            # h[j] appears as h[i] when i=j, contributing (1-h[j+lag])
            i1 = j + lag
            if 0 <= i1 < n:

            # h[j] appears as h[i+lag] when i+lag=j, i.e., i=j-lag, contributing -h[j-lag]
            i2 = j - lag
            if 0 <= i2 < n:
