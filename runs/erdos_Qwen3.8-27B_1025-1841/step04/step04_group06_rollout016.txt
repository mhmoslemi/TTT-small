import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        # Find position with most room
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    b = (h > 0.5).astype(float)
    n_ones = int(np.sum(b))
    n_target = n // 2  # n must be even
    if n_ones > n_target:
        # Need to flip some 1s to 0s: pick the ones with lowest h value
        one_idx = np.where(b == 1)[0]
        h_one = h[one_idx]
        order = np.argsort(h_one)  # lowest first
        for k in range(n_ones - n_target):
            b[one_idx[order[k]]] = 0.0
    elif n_ones < n_target:
        zero_idx = np.where(b == 0)[0]
        h_zero = h[zero_idx]
        order = np.argsort(-h_zero)  # highest first
        for k in range(n_target - n_ones):
            b[zero_idx[order[k]]] = 1.0
    return b

def binary_sa(h_bin, n, rng, n_iter, T0, T1):
    """Simulated annealing on binary sequence."""
    h = h_bin.copy()
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    for it in range(n_iter):
        T = T0 * (T1 / T0) ** (it / max(1, n_iter - 1))
        # Propose swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        
        h[oi] = 0.0
        h[oj] = 1.0
        c_new = compute_c5(h)
        
        if c_new < current_c5 or rng.rand() < np.exp(-(c_new - current_c5) / max(T, 1e-10)):

            ones = np.delete(ones, ii)
            ones = np.append(ones, oj)
            zeros = np.delete(zeros, jj)
            zeros = np.append(zeros, oi)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    
    return best_h, best_c5

def greedy_local_search(h_bin, n, rng, max_rounds=50):

    ones = np.where(h > 0.5)[0].copy()

    #   sum_m a[m] * b[m - p + n - 1]
    # Or equivalently: let k = p - (n-1), then sum_m a[m+k] * b[m]

    # sum_m (-1)*a_unit[m] ... wait.

    # Actually: convolve(delta_a, flip(b))
    # 
    # np.correlate(a, v) = np.convolve(a, v[::-1])
    # So np.correlate(delta_a, b) = np.convolve(delta_a, b[::-1])
    # = -convolve(unit_i, b[::-1]) + convolve(unit_j, b[::-1])
    # = -b[::-1] shifted by i + b[::-1] shifted by j
    
    # In the full output (length 2n-1):
    # b[::-1] is at positions [i, i+n-1] and [j, j+n-1] (clipped to [0, 2n-2])
    
    br = b[::-1]  # b reversed
    # Contribution from -delta_a[i]*unit: -br placed starting at index i
    # br has length n, so it occupies [i, i+n-1] in the 2n-1 output
    lo1 = max(0, i)
    hi1 = min(2*n-1, i+n)
    new_corr[lo1:hi1] -= br[lo1-i:hi1-i]
    
    lo2 = max(0, j)
    hi2 = min(2*n-1, j+n)
    new_corr[lo2:hi2] += br[lo2-j:hi2-j]
    
    # Now corr(h, delta_b, 'full') where delta_b has +1 at i, -1 at j:
    # = convolve(h, (-1+delta_b)...) 
    # = convolve(h, flip(delta_b))
    # flip(delta_b) has +1 at n-1-i, -1 at n-1-j
    # = h shifted by (n-1-i) - h shifted by (n-1-j)
    # In the 2n-1 output:
    lo3 = max(0, n-1-i)
    hi3 = min(2*n-1, n-1-i+n)
    new_corr[lo3:hi3] += h[lo3-(n-1-i):hi3-(n-1-i)]
    
    lo4 = max(0, n-1-j)
    hi4 = min(2*n-1, n-1-j+n)
    new_corr[lo4:hi4] -= h[lo4-(n-1-j):hi4-(n-1-j)]
    
    # corr(delta_a, delta_b, 'full'):
    # delta_a: -1 at i, +1 at j
    # delta_b: +1 at i, -1 at j
    # = -delta_b convolved... 
    # np.correlate(delta_a, delta_b) = convolve(delta_a, delta_b[::-1])
    # delta_b[::-1]: +1 at n-1-i, -1 at n-1-j
    # convolve(delta_a, delta_b[::-1]):
    # = -1 * (unit at n-1-i) shifted by i + 1 * (unit at n-1-i) shifted by j
    #   + 1 * (unit at n-1-j) shifted by i + (-1) * (unit at n-1-j) shifted by j
    # Hmm, let me just compute it directly since it's tiny.
    # Actually: delta_a has 2 entries, delta_b has 2 entries, so the correlation has at most 4 non-zero entries.
    # corr(delta_a, delta_b, 'full')[p] = sum_m delta_a[m] * delta_b[m - p + n - 1]
    # Non-zero when m-p+n-1 is in {i, j} and m is in {i, j}
    # m=i: delta_a[i]=-1, need i-p+n-1 in {i,j} → p in {n-1, n-1+i-j}
    # m=j: delta_a[j]=+1, need j-p+n-1 in {i,j} → p in {n-1+i-j, n-1}
    
    # This is getting complex. Let me just compute it directly for the 4 combinations:
    for m in [i, j]:
        va = -1.0 if m == i else 1.0
        fornb in [i, j]:
            vb = 1.0 if fnb == i else -1.0
            idx = m + (n-1-fnb)
            if 0 <= idx < 2*n-1:
                new_corr[idx] += va * vb
    
    return new_corr

            if tl() < 10:
                break
            improved = False
            for j in range(n_cd):
                if tl() < 5:
                    break
                ov = h_cd[j]
                for d in deltas:
                    nv = ov + d
                    if nv < -1e-9 or nv > 1 + 1e-9:
                        continue
                    tr = h_cd.copy()
                    tr[j] = nv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        ov = nv
                        improved = True
                        break
                for bv in (0.0, 1.0):
                    if abs(bv - h_cd[j]) < 1e-9:
                        continue
                    tr = h_cd.copy()
                    tr[j] = bv
                    tr = make_feasible(tr, n_cd)
                    c = compute_c5(tr)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h_cd = tr
                        improved = True
            if not improved:
                break
        c5_cd = compute_c5(h_cd)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Final SLSQP-style polish
    if tl() > 15:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        def f_final(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            mc = float(np.max(corr))
            shifted = 5000.0 * (corr - mc)
            shifted = np.clip(shifted, -50, 50)
            return float((mc + np.log(np.sum(np.exp(shifted))) / 5000.0) * dx_f)
        bounds = [(0.0, 1.0)] * n_f
        try:
            res = minimize(f_final, best_h, method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 100, 'ftol': 1e-16, 'gtol': 1e-12})
            res_x = make_feasible(res.x, n_f)
            c5_f = compute_c5(res_x)
            if c5_f < best_c5:
                best_c5, best_h = c5_f, res_x.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
