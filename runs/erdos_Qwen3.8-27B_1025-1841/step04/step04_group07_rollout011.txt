import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def grad_corr_k(h, m):
    n = len(h)
    s = m - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if lo1 < hi1:
        g[lo1:hi1] += 1.0 - h[lo1+s:hi1+s]
    lo2 = max(0, s)
    hi2 = min(n, n + s)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2-s:hi2-s]
    return g

def topk_gradient(h, k=12, T=50.0):
    corr = compute_corr(h)
    n = len(h)
    idx = np.argpartition(corr, -k)[-k:]
    mc = corr.max()
    w = np.exp(T * (corr[idx] - mc))
    w /= w.sum()
    g = np.zeros(n)
    for i in range(len(idx)):
        if w[i] > 1e-8:
            g += w[i] * grad_corr_k(h, int(idx[i]))
    return g

def project_onto_box_sum(v, target):
    lo_lam = v.min() - 1.0
    hi_lam = v.max() + 1.0
    for _ in range(80):

        x = np.clip(v - lam, 0.0, 1.0)
        s = x.sum()
        if s > target:
            lo_lam = lam
        else:
            hi_lam = lam
    return np.clip(v - (lo_lam + hi_lam) / 2.0, 0.0, 1.0)

def lbgfsb_topk(h0, n, topk_frac=0.08, T=200.0, max_iter=250):
    dx = 2.0 / n
    n_c = 2 * n - 1
    k_act = max(5, int(topk_frac * n_c))

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        idx = np.argpartition(corr, -k_act)[-k_act:]
        topk = corr[idx]
        mc = topk.max()
        shifted = np.clip(T * (topk - mc), -50, 50)
        smax = mc + np.log(np.sum(np.exp(shifted))) / T
        pen = max(0.0, float(corr.max()) - smax) ** 2
        return smax * dx + 100.0 * pen + 800.0 * (x.sum() - n / 2.0) ** 2

    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

        h_sa, c_sa = sim_anneal_binary(h_b, n_bin, rng, n_rounds=50, tries_per_round=5000, T0=0.035)
        if c_sa < best_c5:
            best_c5, best_h = c_sa, h_sa.copy()

    # Phase 4: Continue annealing with lower temperature
    if rem() > 150:
        n_bin = len(best_h)
        h_sa2, c_sa2 = sim_anneal_binary(best_h, n_bin, rng, n_rounds=30, tries_per_round=4000, T0=0.015)
        if c_sa2 < best_c5:
            best_c5, best_h = c_sa2, h_sa2.copy()

    # Phase 5: Pairwise continuous swaps
    if rem() > 100:
        h_ps, c_ps = pairwise_swap(best_h, len(best_h), rng, max_rounds=20, tries=8000)
        if c_ps < best_c5:
            best_c5, best_h = c_ps, h_ps.copy()

    # Phase 6: Fine coordinate descent
    if rem() > 60:
        n_cd = len(best_h)
        h_cd, c_cd = fine_coord_descent(best_h, n_cd, max_rounds=12,
                                         steps=[0.004, -0.004, 0.002, -0.002, 0.001, -0.001])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 7: Subgradient polish
    if rem() > 60:
        h_sg2, c_sg2 = momentum_subgrad(best_h, len(best_h), n_iters=2000, lr0=0.001, momentum=0.9, topk=15)
        if c_sg2 < best_c5:
            best_c5, best_h = c_sg2, h_sg2.copy()

    # Phase 8: Multi-resolution: upsample best to 3x and refine
    if rem() > 100:
        n_hi = 3 * n0
        h_hi = upsample_h(best_h, n_hi)
        c_hi = compute_c5(h_hi)
        if c_hi < best_c5:
            best_c5, best_h = c_hi, h_hi.copy()
        if rem() > 50:
            h_opt = lbgfsb_topk(h_hi, n_hi, topk_frac=0.06, T=500, max_iter=150)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()

    # Phase 9: Final fine coordinate descent with very small steps
    if rem() > 25:
        h_cd2, c_cd2 = fine_coord_descent(best_h, len(best_h), max_rounds=8,
                                           steps=[0.002, -0.002, 0.001, -0.001, 0.0005, -0.0005])
        if c_cd2 < best_c5:
            best_c5, best_h = c_cd2, h_cd2.copy()

    # Phase 10: Final pairwise + coord
    if rem() > 20:
        h_ps2, c_ps2 = pairwise_swap(best_h, len(best_h), rng, max_rounds=10, tries=5000)
        if c_ps2 < best_c5:
            best_c5, best_h = c_ps2, h_ps2.copy()
        if rem() > 10:
            h_cd3, c_cd3 = fine_coord_descent(best_h, len(best_h), max_rounds=5,
                                               steps=[0.001, -0.001, 0.0005, -0.0005])
            if c_cd3 < best_c5:
                best_c5, best_h = c_cd3, h_cd3.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
