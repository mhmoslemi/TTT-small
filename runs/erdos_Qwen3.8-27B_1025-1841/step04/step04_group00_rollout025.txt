```python
def subgradient_binary_step(h, v, step_size, top_m=5):
    n = len(h)
    # Compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Find top shifts
    top_idx = np.argsort(corr)[-top_m:]
    # Compute subgradient (average over top shifts)
    grad = np.zeros(n)
    for k in top_idx:
        L = k - (n - 1)  # actual lag
        for i in range(n):
            # dh[i+L] term
            if 0 <= i + L < n:
                grad[i] += (1.0 - h[i + L])
            # -h[i-L] term
            if 0 <= i - L < n:
                grad[i] -= h[i - L]
    grad /= len(top_idx)
    # Update v
    v = v - step_size * grad
    # Project to binary: keep top n/2 as 1
    threshold = np.partition(v, n//2)[n//2]
    h_new = (v >= threshold).astype(float)
    # Ensure exactly n/2 ones
    while h_new.sum() > n/2:
        # Find a 1 with the lowest v value and make it 0
        ones = np.where(h_new == 1)[0]
        min_idx = ones[np.argmin(v[ones])]
        h_new[min_idx] = 0
    while h_new.sum() < n/2:
        zeros = np.where(h_new == 0)[0]
        max_idx = zeros[np.argmax(v[zeros])]
        h_new[max_idx] = 1
    return h_new, v
```

The gradient ∂result[m]/∂h[i]:

- h[i] appears as h[k+L] when k+L=i, i.e., k=i-L: contribution is h[i-L]*(-1) if 0≤i-L<n (because derivative of (1-h[i]) is -1)

For the vectorized update for a given L:
- If L ≥ 0:
  - First term: (1-h[i+L]) for i=0..n-1-L → grad[0:n-L] += (1-h[L:n])
  - Second term: -h[i-L] for i=L..n-1 → grad[L:n] -= h[0:n-L]
- If L < 0 (let L' = -L > 0):

  - Second term: -h[i-L] = -h[i+L'] for i=0..n-1-L' → grad[0:n-L'] -= h[L':n]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def to_binary(h, n):
    """Project real-valued h to binary with exactly n/2 ones."""
    v = h.copy()
    threshold = np.partition(v, n//2 - 1)[n//2 - 1]
    h_bin = (v >= threshold).astype(float)
    # Fix count if needed
    count = int(h_bin.sum())
    if count > n // 2:
        ones = np.where(h_bin == 1)[0]
        excess = count - n // 2
        idx = ones[np.argsort(v[ones])[:excess]]
        h_bin[idx] = 0
    elif count < n // 2:
        zeros = np.where(h_bin == 0)[0]
        deficit = n // 2 - count
        idx = zeros[np.argsort(v[zeros])[-deficit:]]
        h_bin[idx] = 1
    return h_bin

def subgradient_step(h, n, top_m=5, T=100.0):
    """Compute subgradient of max-k correlation for binary h."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights (focus on top shifts)
    max_c = np.max(corr)
    weights = np.exp(T * (corr - max_c))
    total_w = np.sum(weights)
    
    # For efficiency, only use top_m
    top_idx = np.argsort(corr)[-top_m:]
    weights = np.zeros_like(corr)
    for idx in top_idx:
        weights[idx] = np.exp(T * (corr[idx] - max_c))
    total_w = np.sum(weights)
    if total_w == 0:
        weights[top_idx] = 1.0 / top_m
        total_w = 1.0
    else:
        weights /= total_w
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for idx in top_idx:
        if weights[idx] < 1e-10:
            continue
        L = idx - (n - 1)
        w = weights[idx]
        if L >= 0:
            # First term: (1-h[i+L]) for i=0..n-1-L
            # grad[i] += w * (1-h[i+L])
            grad[:n-L] += w * (1 - h[L:])
            # Second term: -h[i-L] for i=L..n-1
            # grad[i] += w * (-h[i-L])
            grad[L:] -= w * h[:n-L]
        else:

            # First term: (1-h[i-Lp]) for i=Lp..n-1
            grad[Lp:] += w * (1 - h[:n-Lp])
            # Second term: -h[i+Lp] for i=0..n-1-Lp
            grad[:n-Lp] -= w * h[Lp:]
    
    return grad, corr

def swap_delta_c5(h, c, i, j, n):
    """Compute new C5 after swapping h[i]:1→0 and h[j]:0→1.
    c is the current correlation vector."""
    # Δc[L] = h[i+L] + h[i-L] - h[j+L] - h[j-L]
    # We need to find max_L (c[L] + Δc[L]) * dx
    # For efficiency, only check shifts where Δc[L] ≠ 0
    
    # Δc[L] ≠ 0 only when at least one of i+L, i-L, j+L, j-L is in [0, n-1]
    # and the value differs from the "out of bounds" default (0)
    
    # Actually, Δc[L] = h[i+L] + h[i-L] - h[j+L] - h[j-L]
    # This is non-zero for L in range [-max(i,j,n-1-i,n-1-j), max(i,j,n-1-i,n-1-j)]
    # which is at most n-1 in each direction, so up to 2n-1 shifts.
