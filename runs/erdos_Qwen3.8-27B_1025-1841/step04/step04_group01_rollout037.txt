```python
def make_feasible(h, n):

def build_grad_matrix(h0, n):
    """Build the gradient matrix A where A[k][j] = df_{s_k}/dh[j] at h0.
    k indexes the shifts: k = s + (n-1), s ranges from -(n-1) to (n-1)."""
    A = np.zeros((2*n-1, n))
    for s in range(-(n-1), n):
        k = s + n - 1
        # S_s = {j : max(0,-s) <= j <= min(n-1, n-1-s)}
        j_start = max(0, -s)
        j_end = min(n, n - s)  # exclusive
        if j_end > j_start:
            # First term: (1 - h0[j+s]) for j in [j_start, j_end)
            idx = np.arange(j_start, j_end)
            A[k, idx] += (1.0 - h0[idx + s])
        
        # Second term: -h0[j-s] for j in S_{-s} = [max(0,s), min(n, n+s))
        j2_start = max(0, s)
        j2_end = min(n, n + s)  # exclusive
        if j2_end > j2_start:
            idx2 = np.arange(j2_start, j2_end)
            A[k, idx2] -= h0[idx2 - s]
    return A
```

```python
def sca_iteration_selective(h0, n, top_k=50, solver='SCIP'):
    """SCA iteration using only the top-k active shifts."""
    corr_vals = np.correlate(h0, 1.0 - h0, mode='full')
    max_corr = np.max(corr_vals)
    # Select top k shifts
    top_idx = np.argsort(-corr_vals)[:top_k]
    
    # Autocorrelation
    autocorr = np.correlate(h0, h0, mode='full')
    
    # Build gradient matrix for selected shifts only
    A_sel = np.zeros((top_k, n))
    for i, k in enumerate(top_idx):
        s = k - (n - 1)
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_end > j_start:
            idx = np.arange(j_start, j_end)
            A_sel[i, idx] += (1.0 - h0[idx + s])
        j2_start = max(0, s)
        j2_end = min(n, n + s)
        if j2_end > j2_start:
            idx2 = np.arange(j2_start, j2_end)
            A_sel[i, idx2] -= h0[idx2 - s]
    
    # LP
    h = cp.Variable(n)
    t = cp.Variable()
    constraints = [
        A_sel @ h + autocorr[top_idx] <= t,
        cp.sum(h) == n / 2.0,
        h >= 0,
        h <= 1
    ]
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    try:
        prob.solve(solver=solver)
    except Exception:
        try:
            prob.solve(solver='ECOS')
        except Exception:
            return h0, None
    
    if h.value is None:
        return h0, None
    
    h_sol = make_feasible(np.array(h.value).flatten(), n)
    return h_sol, t.value
```

```python
from scipy.optimize import linprog

def solve_sca_lp(h_t, n, active_idx):
    """Solve the SCA LP using linprog."""
    autocorr = np.correlate(h_t, h_t, mode='full')
    K = len(active_idx)
    
    # Build A_sel (K × n)
    A_sel = np.zeros((K, n))
    b_const = np.zeros(K)
    for i, k in enumerate(active_idx):
        s = k - (n - 1)
        b_const[i] = autocorr[k]
        j_start = max(0, -s)
        j_end = min(n, n - s)
        if j_end > j_start:
            idx = np.arange(j_start, j_end)
            A_sel[i, idx] += (1.0 - h_t[idx + s])
        j2_start = max(0, s)
        j2_end = min(n, n + s)
        if j2_end > j2_start:
            idx2 = np.arange(j2_start, j2_end)
            A_sel[i, idx2] -= h_t[idx2 - s]
    
    # Full LP matrix: (K+1) × (n+1)
    # Constraints: A_sel @ h - t ≤ -b_const  (K constraints)
    #              -t ≤ 0                    (1 constraint)
    A_ub = np.zeros((K+1, n+1))
    A_ub[:K, :n] = A_sel
    A_ub[:K, n] = -1.0
    A_ub[K, n] = -1.0
    
    b_ub = np.zeros(K+1)
    b_ub[:K] = -b_const
    b_ub[K] = 0.0
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n/2.0])
    
    # Objective: minimize t
    c = np.zeros(n+1)
    c[n] = 1.0
    
    # Bounds: 0 ≤ h_i ≤ 1, t ≥ 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_sol = res.x[:n]
        t_sol = res.x[n]
        h_sol = make_feasible(h_sol, n)
        return h_sol, t_sol
    return h_t, None
```
