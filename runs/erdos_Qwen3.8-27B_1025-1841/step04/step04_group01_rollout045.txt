    log_sum = np.log(np.sum(exp_shifted)) + shifted_max
    lse_val = (mc + log_sum / T) * dx
    # Penalty
    pen = mu * (np.sum(h) - n / 2.0) ** 2
    return lse_val + pen, corr, top_idx

def lse_gradient(h, T, top_k, mu):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    if top_k >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    top_vals = corr[top_idx]
    mc = top_vals.max()
    shifted = T * (top_vals - mc)
    shifted_max = shifted.max()
    exp_shifted = np.exp(shifted - shifted_max)
    w = exp_shifted / exp_shifted.sum()
    
    grad = np.zeros(n)
    shifts = top_idx - (n - 1)
    
    for i in range(len(shifts)):
        s = int(shifts[i])
        wi = w[i]
        # d corr[n-1+s] / d h[j] = (1-h[j-s])*[0<=j-s<n] - h[j+s]*[0<=j+s<n]
        if s >= 0:
            # (1-h[j-s]) for j in [s, n+s) -> j in [s, n) (since j<n)
            j1_start, j1_end = s, n
            term1 = np.ones(j1_end - j1_start) - h[:j1_end - s]
            grad[j1_start:j1_end] += wi * term1
            # -h[j+s] for j in [0, n-s)
            j2_end = n - s
            grad[:j2_end] -= wi * h[s:s + j2_end]
        else:
            # (1-h[j-s]) for j in [0, n+s) -> j in [0, n) since -s < n
            j1_end = n + s  # = n - |s|
            term1 = np.ones(j1_end) - h[-s:]
            grad[:j1_end] += wi * term1
            # -h[j+s] for j in [-s, n) -> j in [-s, n)
            j2_start = -s
            grad[j2_start:] -= wi * h[:n - j2_start]
    
    grad *= T * dx
    # Penalty gradient
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def fft_correlate(a, b):
    n = len(a)
    m = len(b)
    out_len = n + m - 1
    size = 1
    while size < out_len:
        size *= 2
    A = np.fft.rfft(a, size)
    B = np.fft.rfft(b[::-1], size)
    c = np.fft.irfft(A * B, size)
    return c[:out_len]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def lse_val_grad(h, T, top_k, mu):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate(h, 1.0 - h)
    total = len(corr)
    k = min(top_k, total)
    top_idx = np.argpartition(corr, -k)[-k:]
    top_vals = corr[top_idx]
    mc = top_vals.max()
    shifted = T * (top_vals - mc)
    sm = shifted.max()
    es = np.exp(shifted - sm)
    w = es / es.sum()
    lse_val = (mc + np.log(np.sum(es)) + sm) / T * dx
    pen = mu * (np.sum(h) - n / 2.0) ** 2
    grad = np.zeros(n)
    shifts = top_idx - (n - 1)
    for i in range(k):
        s = int(shifts[i])
        wi = w[i]
        if s >= 0:
            j1s, j1e = s, n
            grad[j1s:j1e] += wi * (1.0 - h[:j1e - s])
            j2e = n - s
            if j2e > 0:
                grad[:j2e] -= wi * h[s:s + j2e]
        else:
            j1e = n + s
            if j1e > 0:
                grad[:j1e] += wi * (1.0 - h[-s:])
            j2s = -s
            j2e2 = n - j2s
            if j2e2 > 0:
                grad[j2s:] -= wi * h[:j2e2]
    grad *= T * dx
    grad += 2.0 * mu * (np.sum(h) - n / 2.0)
    obj = lse_val + pen
    return obj, grad

def coordinate_descent(h, n, steps, passes, time_limit_fn):
    best_c5 = compute_c5(h)
    for p in range(passes):
        if time_limit_fn() < 10:
            break
        improved = False
        for j in range(n):
            if time_limit_fn() < 5:
                break
            old = h[j]
            best_val = old
            best_c = best_c5
            for st in steps:
                for sg in [-1.0, 1.0]:
                    v = old + sg * st
                    if v < 0.0 or v > 1.0:
                        continue
                    h[j] = v
                    h = make_feasible(h, n)
                    c = compute_c5(h)
                    if c < best_c:
                        best_c = c
                        best_val = v
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
            else:
                h[j] = old
                h = make_feasible(h, n)
        if not improved and p >= 2:
            break
    return h, best_c5
