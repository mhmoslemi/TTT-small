    """Coordinate descent with exact 1D subproblem solving"""
    h = h.astype(float).copy()

    for _ in range(n_passes):
        improved = False
        for j in range(n):
            old_val = h[j]
            
            # Compute corr for current h
            corr = np.correlate(h, 1.0 - h, mode='full')
            
            # For each shift k, compute a_k (coefficient of h[j]) and b_k (intercept)
            # corr_k = a_k * h[j] + b_k
            # a_k = (1-h[j-k]) - h[j+k] (with valid index checks)
            # Let's compute this for all k at once
            
            # The shift d = k - (n-1) ranges from -(n-1) to n-1
            # a_k involves h[j-d] and h[j+d]
            # h[j-d] is valid when 0 <= j-d < n, i.e., j-(n-1) <= d <= j
            # h[j+d] is valid when 0 <= j+d < n, i.e., -j <= d <= n-1-j
            
            d_range = np.arange(-(n-1), n)  # length 2n-1
            # a_k = (1 - h[j-d]) * valid1 - h[j+d] * valid2
            # where valid1 = (0 <= j-d < n) and valid2 = (0 <= j+d < n)
            
            j_minus_d = j - d_range
            j_plus_d = j + d_range
            
            valid1 = (j_minus_d >= 0) & (j_minus_d < n)
            valid2 = (j_plus_d >= 0) & (j_plus_d < n)
            
            # Clip indices for safe access
            j_minus_d_clipped = np.clip(j_minus_d, 0, n-1)
            j_plus_d_clipped = np.clip(j_plus_d, 0, n-1)
            
            a = np.zeros(2*n-1)
            a[valid1] += (1.0 - h[j_minus_d_clipped[valid1]])
            a[valid2] -= h[j_plus_d_clipped[valid2]]

            b = corr - a * old_val

            # Candidate points: x=0, x=1, and intersections

            f0 = np.max(b)
            f1 = np.max(a + b)

            best_f = min(f0, f1)

            k_max_1 = int(np.argmax(a + b))  # k that achieves max at x=1

            # This is O(n) but let's just check a few key ones

            # Alternative: parametric search

            # Its minimum is where the subdifferential contains 0, i.e., where
            # there exist k1, k2 with a_k1 < 0 < a_k2 (or a_k=0) such that

            best_f_grid = f_grid[best_idx]
