  - Second factor: h[i-l]*(1-h[i]) → change: h[i-l] if i-l is valid (since (1-h[i]) goes from 0 to 1)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    c5 = compute_c5(h)
    corr = get_corr(h)
    stall = 0
    max_stall = 40

    for it in range(max_iters):
        if time_fn is not None and (time_fn() - min_time < 0):
            break
        if stall >= max_stall:
            break

        l_star = int(np.argmax(corr))
        l_shift = l_star - (n - 1)

        score = np.zeros(n)
        v1 = np.arange(n) + l_shift
        m1 = (v1 >= 0) & (v1 < n)
        if np.any(m1):
            score[m1] += h[v1[m1]]
        v2 = np.arange(n) - l_shift
        m2 = (v2 >= 0) & (v2 < n)
        if np.any(m2):
            score[m2] += h[v2[m2]]

        on_mask = h > 0.5
        off_mask = h < 0.5
        if not np.any(on_mask) or not np.any(off_mask):
            break

        on_idx = np.where(on_mask)[0]
        off_idx = np.where(off_mask)[0]
        on_s = score[on_idx]
        off_s = score[off_idx]

        if on_s.max() <= off_s.min() + 1e-12:
            stall += 1
            # Try next few worst shifts
            top = np.argsort(-corr)[:5]
            found = False
            for lt in top:
                ls = lt - (n - 1)
                if ls == l_shift:
                    continue
                sc = np.zeros(n)
                v1t = np.arange(n) + ls
                m1t = (v1t >= 0) & (v1t < n)
                if np.any(m1t):
                    sc[m1t] += h[v1t[m1t]]
                v2t = np.arange(n) - ls
                m2t = (v2t >= 0) & (v2t < n)
                if np.any(m2t):
                    sc[m2t] += h[v2t[m2t]]
                os_ = sc[on_idx]
                ofs_ = sc[off_idx]
                if os_.max() > ofs_.min() + 1e-12:
                    i_b = on_idx[np.argmax(os_)]
                    j_b = off_idx[np.argmin(ofs_)]
                    hd = h.copy()
                    hd[i_b] = 0.0
                    hd[j_b] = 1.0
                    hd = make_feasible(hd, n)
                    cd = compute_c5(hd)
                    if cd < c5 - 1e-15:
                        h = hd
                        c5 = cd
                        corr = get_corr(h)
                        stall = 0
                        found = True
                        break
            if not found:
                stall += 1
            continue

        i_best = on_idx[np.argmax(on_s)]
        j_best = off_idx[np.argmin(off_s)]
        hd = h.copy()
        hd[i_best] = 0.0
        hd[j_best] = 1.0
        hd = make_feasible(hd, n)
        cd = compute_c5(hd)

        if cd < c5 - 1e-15:
            h = hd
            c5 = cd
            corr = get_corr(h)
            stall = 0
        else:
            stall += 1

    return h, c5

def swap_descent_kick(h, n, rng, max_iters=2000, time_fn=None, min_time=5.0):
    """Swap descent with random kicks to escape local optima."""
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stall = 0

    for it in range(max_iters):
        if time_fn is not None and time_fn() - min_time < 0:
            break
        if stall >= 30:
            # Random kick
            n_kick = rng.randint(3, 8)
            on_idx = np.where(h > 0.5)[0]
            off_idx = np.where(h < 0.5)[0]
            if len(on_idx) < n_kick or len(off_idx) < n_kick:
                break
            ko = rng.choice(on_idx, n_kick, replace=False)
            kf = rng.choice(off_idx, n_kick, replace=False)
            hd = h.copy()
            hd[ko] = 0.0
            hd[kf] = 1.0
            hd = make_feasible(hd, n)
            cd = compute_c5(hd)
            if cd < c5 + 0.003:
                h = hd
                c5 = cd
            stall = 0
            continue

        l_star = int(np.argmax(get_corr(h)))
        l_shift = l_star - (n - 1)
        score = np.zeros(n)
        v1 = np.arange(n) + l_shift
        m1 = (v1 >= 0) & (v1 < n)
        if np.any(m1):
            score[m1] += h[v1[m1]]
        v2 = np.arange(n) - l_shift
        m2 = (v2 >= 0) & (v2 < n)
        if np.any(m2):
            score[m2] += h[v2[m2]]

        on_mask = h > 0.5
        off_mask = h < 0.5
        if not np.any(on_mask) or not np.any(off_mask):
            break
        on_idx = np.where(on_mask)[0]
        off_idx = np.where(off_mask)[0]
        on_s = score[on_idx]
        off_s = score[off_idx]

        if on_s.max() <= off_s.min() + 1e-12:
            stall += 1
            continue

        i_b = on_idx[np.argmax(on_s)]
        j_b = off_idx[np.argmin(off_s)]
        hd = h.copy()
        hd[i_b] = 0.0
        hd[j_b] = 1.0
        hd = make_feasible(hd, n)
        cd = compute_c5(hd)

        if cd < c5 - 1e-15:
            h = hd
            c5 = cd
            stall = 0
            if cd < best_c5:
                best_c5 = cd
                best_h = h.copy()
        else:
            stall += 1

    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T, _G_TOP, _G_DX
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    # Get initial solution
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)

    n = len(h)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

    # Phase 1: SLSQP quick polish on parent
    if remaining() > 100:
        corr = get_corr(h)
        mc = np.max(corr)
        thr = mc - 0.02 * mc
        top_idx = np.where(corr >= thr)[0]
        if len(top_idx) > 20:
            top_idx = top_idx[np.argsort(-corr[top_idx])[:20]]
        _G_N = n
        _G_T = 400.0
        _G_TOP = top_idx
        _G_DX = 2.0 / n
        try:
            res = minimize(_obj_top, h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _con_f, 'jac': _con_j}],
                          options={'maxiter': 200, 'ftol': 1e-15})
            cr = compute_c5(res.x)
            if cr < best_c5:
                best_c5 = cr
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 2: Binary snap + swap descent
    h_bin = best_h.copy()
    h_bin[h_bin > 0.5] = 1.0
    h_bin[h_bin < 0.5] = 0.0
    h_bin = make_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    # Swap descent from best
    h_work = best_h.copy()
    h_work, c5_sd = swap_descent(h_work, n, max_iters=5000, time_fn=remaining, min_time=60.0)
    if c5_sd < best_c5:
        best_c5 = c5_sd
        best_h = h_work.copy()

    # Phase 3: SLSQP polish after swap descent
    if remaining() > 80:
        _G_N = n
        _G_T = 600.0
        corr = get_corr(best_h)
        mc = np.max(corr)
        thr = mc - 0.015 * mc
        top_idx = np.where(corr >= thr)[0]
        if len(top_idx) > 15:
            top_idx = top_idx[np.argsort(-corr[top_idx])[:15]]
        _G_TOP = top_idx
        _G_DX = 2.0 / n
        try:
            res = minimize(_obj_top, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _con_f, 'jac': _con_j}],
                          options={'maxiter': 300, 'ftol': 1e-16})
            cr = compute_c5(res.x)
            if cr < best_c5:
                best_c5 = cr
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 4: Swap descent with kicks (escape local optima)
    h_work = best_h.copy()
    h_work[h_work > 0.5] = 1.0
    h_work[h_work < 0.5] = 0.0
    h_work = make_feasible(h_work, n)
    h_work, c5_k = swap_descent_kick(h_work, n, rng, max_iters=3000, time_fn=remaining, min_time=60.0)
    if c5_k < best_c5:
        best_c5 = c5_k
        best_h = h_work.copy()

    # Phase 5: Another SLSQP round
    if remaining() > 70:
        _G_N = n
        _G_T = 800.0
        corr = get_corr(best_h)
        mc = np.max(corr)
        thr = mc - 0.01 * mc
        top_idx = np.where(corr >= thr)[0]
        if len(top_idx) > 12:
            top_idx = top_idx[np.argsort(-corr[top_idx])[:12]]
        _G_TOP = top_idx
        _G_DX = 2.0 / n
        try:
            res = minimize(_obj_top, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _con_f, 'jac': _con_j}],
                          options={'maxiter': 250, 'ftol': 1e-16})
            cr = compute_c5(res.x)
            if cr < best_c5:
                best_c5 = cr
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 6: Final swap descent
    if remaining() > 40:
        h_work2 = best_h.copy()
        h_work2[h_work2 > 0.5] = 1.0
        h_work2[h_work2 < 0.5] = 0.0
        h_work2 = make_feasible(h_work2, n)
        h_work2, c5_f = swap_descent(h_work2, n, max_iters=3000, time_fn=remaining, min_time=20.0)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_work2.copy()

    # Phase 7: Try upsample to 2n if time allows
    if remaining() > 250:
        n_up = n * 2
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up <= best_c5 + 0.002:
            _G_N = n_up
            _G_T = 300.0
            _G_TOP = None
            _G_DX = 2.0 / n_up
            try:
                res = minimize(_obj_top, h_up, method='SLSQP',
                              bounds=[(0.0, 1.0)] * n_up,
                              constraints=[{'type': 'eq', 'fun': _con_f, 'jac': _con_j}],
                              options={'maxiter': 200, 'ftol': 1e-14})
                cr = compute_c5(res.x)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = make_feasible(res.x, n_up)
                    n = n_up
            except Exception:
                pass
            # Swap descent at higher res
            if remaining() > 80:
                h_up2 = best_h.copy()
                h_up2[h_up2 > 0.5] = 1.0
                h_up2[h_up2 < 0.5] = 0.0
                h_up2 = make_feasible(h_up2, n)
                h_up2, c5_up2 = swap_descent(h_up2, n, max_iters=2000, time_fn=remaining, min_time=50.0)
                if c5_up2 < best_c5:
                    best_c5 = c5_up2
                    best_h = h_up2.copy()

    # Phase 8: Final SLSQP polish
    if remaining() > 40:
        n_pts = len(best_h)
        _G_N = n_pts
        _G_T = 1000.0
        corr = get_corr(best_h)
        mc = np.max(corr)
        thr = mc - 0.005 * mc
        top_idx = np.where(corr >= thr)[0]
        if len(top_idx) > 10:
            top_idx = top_idx[np.argsort(-corr[top_idx])[:10]]
        _G_TOP = top_idx
        _G_DX = 2.0 / n_pts
        try:
            res = minimize(_obj_top, best_h, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_pts,
                          constraints=[{'type': 'eq', 'fun': _con_f, 'jac': _con_j}],
                          options={'maxiter': 150, 'ftol': 1e-16})
            cr = compute_c5(res.x)
            if cr < best_c5:
                best_c5 = cr
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Final verification
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
