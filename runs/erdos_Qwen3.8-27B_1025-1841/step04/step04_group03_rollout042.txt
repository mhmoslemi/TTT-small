import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smoothed_gradient(h, T):

    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)

    n = len(h)
    current_obj = compute_c5(h)  # Use exact for tracking
    for step in range(n_steps):
        if time.time() - start_time > time_limit:
            break
        grad = smoothed_gradient(h, T)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        new_obj = compute_c5(h_new)
        if new_obj < current_obj:
            h = h_new
            current_obj = new_obj
        else:

            h_new = np.clip(h_new, 0.0, 1.0)
            h_new = make_feasible(h_new, n)
            new_obj = compute_c5(h_new)
            if new_obj < current_obj:

                current_obj = new_obj
                lr *= 0.5
    return h, current_obj

def smooth_obj(h, T):

    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    return (max_c + np.log(np.sum(np.exp(T * (corr - max_c)))) / T) * dx

from scipy.optimize import minimize

_G_N = 0
_G_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _sf(x):
    n = _G_N; T = _G_T; dx = 2.0/n
    corr = np.correlate(x, 1.0-x, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T*(corr-mc))))/T) * dx

def _sg(x):
    n = _G_N; T = _G_T; dx = 2.0/n
    corr = np.correlate(x, 1.0-x, mode='full')
    mc = np.max(corr)
    et = np.exp(T*(corr-mc))
    p = et / np.sum(et)
    return dx * np.correlate(1.0-2.0*x, p, mode='full')[n-1:2*n-1]

def _sfg(x):
    return _sf(x), _sg(x)

def _cef(x):
    return np.sum(x) - len(x)/2.0

def _cej(x):
    return np.ones(len(x))

def slsqp_opt(h, n, T, maxiter=200, ftol=1e-14):
    global _G_N, _G_T
    _G_N = n; _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type':'eq', 'fun':_cef, 'jac':_cej}]
    try:
        res = minimize(_sfg, h.copy(), method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter':maxiter, 'ftol':ftol, 'iprint':0})
        if compute_c5(res.x) < compute_c5(h):
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h
