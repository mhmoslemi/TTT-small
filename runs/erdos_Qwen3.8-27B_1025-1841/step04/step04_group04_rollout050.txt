```python
    u -= np.max(u)
    w = np.exp(u)
    w /= np.sum(w)
    
    # Use only top-k for speed (others have negligible weight for large T)
    top_idx = np.argsort(w)[-k:]
    w_top = np.zeros_like(w)
    w_top[top_idx] = w[top_idx]
    w_top /= np.sum(w_top)  # renormalize
    
    # Compute gradient: grad[i] = dx * sum_j w_top[j] * ((1-h[i+s_j])*1_{valid} - h[i-s_j]*1_{valid})
    grad = np.zeros(n)
    for j in top_idx:
        s = j - n + 1
        wj = w_top[j]
        if wj < 1e-15:
            continue
        # (1-h[i+s]) for valid i: i in [max(0,-s), min(n, n-s))
        lo_a = max(0, -s)
        hi_a = min(n, n - s)
        if hi_a > lo_a:
            grad[lo_a:hi_a] += wj * (1 - h[lo_a+s:hi_a+s])
        # -h[i-s] for valid i: i in [max(0,s), min(n, n+s))
        lo_b = max(0, s)
        hi_b = min(n, n + s)
        if hi_b > lo_b:
            grad[lo_b:hi_b] -= wj * h[lo_b-s:hi_b-s]
    
    return dx * grad
```

This is O(k*n) per gradient evaluation, which is very fast!

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-8
        else:
            mask = h > 1e-8
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def project_feasible(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-10:
        mask = (h_proj > 1e-12) & (h_proj < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            h_proj[mask] += diff / cnt
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def smooth_obj_val(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_hessian_placeholder(x):
    return None

def slsqp_obj(x, T, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_con(x, n):
    return np.sum(x) - n / 2.0

def slsqp_con_jac(x, n):
    return np.ones(n)

def compute_grad_topk(h, T, k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    u = T * (corr - mc)
    u -= np.max(u)
    w = np.exp(u)
    w_sum = np.sum(w)
    w /= w_sum
    top_idx = np.argsort(w)[-k:]
    w_top = np.zeros_like(w)
    w_top[top_idx] = w[top_idx]
    w_top /= np.sum(w_top)
    grad = np.zeros(n)
    for j in top_idx:
        wj = w_top[j]
        if wj < 1e-15:
            continue
        s = j - n + 1
        if s >= 0:
            hi_a = n - s
            if hi_a > 0:
                grad[:hi_a] += wj * (1.0 - h[s:hi_a + s])
            if s < n:
                grad[s:] -= wj * h[:n - s]
        else:
            sp = -s
            hi_b = n - sp
            if hi_b > 0:
                grad[:, ] # placeholder
                grad[:] += 0
                # A term: (1-h[i+s]) for i in [sp, n)
                grad[sp:] += wj * (1.0 - h[:hi_b])
                # B term: -h[i-s] for i in [0, sp)
                grad[:sp] -= wj * h[sp:n]
    return dx * grad

def nesterov_phase(h0, n_steps, lr0, T, beta, time_limit):
    n = len(h0)
    start = time.time()
    best_h = h0.copy()
    best_f = compute_c5(h0)
    y = h0.copy()
    v = np.zeros(n)
    for t in range(n_steps):
        if time.time() - start > time_limit:
            break
        lr = lr0 / (1.0 + 0.005 * t)
        grad = compute_grad_topk(y, T, k=5)
        v = beta * v + grad
        y_new = project_feasible(y - lr * v)
        f_new = compute_c5(y_new)
        if f_new < best_f:
            best_f = f_new
            best_h = y_new.copy()
        y = y_new
    return best_h, best_f

def block_cd(h, n_blocks, block_size, n_rounds, time_limit):
    n = len(h)
    start = time.time()
    best_c = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for b in range(n_blocks):
            if time.time() - start > time_limit:
                break
            idx = slice(b * block_size, (b + 1) * block_size)
            old_block = h[idx].copy()
            # Try shifting block values
            for shift in [-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1]:
                trial = h.copy()
                trial[idx] = np.clip(trial[idx] + shift, 0.0, 1.0)
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    h = trial
                    best_c = c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c

def fine_cd(h, n_cands_per, n_rounds, time_limit):
    n = len(h)
    start = time.time()
    best_c = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old = h[j]
            cands = [0.0, 1.0, 0.5]
            cands += np.concatenate([np.linspace(0, 0.5, n_cands_per),
                                     np.linspace(0.5, 1.0, n_cands_per)]).tolist()
            best_trial = old
            best_c_local = best_c
            for cand in cands:
                if abs(cand - old) < 1e-8:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_trial = cand
            if best_trial != old:
                h[j] = best_trial
                h = make_feasible(h, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return h, best_c

    for T, maxiter in [(60, 300), (150, 250), (400, 200)]:
```
