def subgradient_descent(h0, n, T, n_iters, lr, decay):
    h = h0.copy()
    dx = 2.0 / n
    for i in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        # Subgradient: sum of gradients of active constraints
        active = corr >= mc - 1e-10  # lags near the max
        K = active.sum()
        # Gradient of corr[k] wrt h
        # corr_full[k] for k in [-(n-1), ..., (n-1)]
        # For the full correlation at offset k:
        #   if k >= 0: sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
        #   if k < 0: sum_{i=|k|}^{n-1} h[i] * (1-h[i-|k|])
        # Gradient wrt h[j]:
        #   For k >= 0, j appears as h[j] (when j <= n-1-k) with coeff (1-h[j+k])
        #             and as (1-h[j]) (when j >= k, i.e., j = i+k, so i=j-k) with coeff h[j-k]
        #   This is getting complex. Let me use numerical gradient instead.
        
        # Actually, let me compute the gradient of the smooth approximation
        # and use that as a proxy
        if K < 3:
            K = 3
        weights = np.zeros(len(corr))
        weights[active] = 1.0 / K
        
        # Gradient of weighted sum of correlations
        # corr_full[m] for m = 0, ..., 2n-2 corresponds to lag m - (n-1)
        # For lag l = m - (n-1):
        #   if l >= 0: corr = sum_{i=0}^{n-1-l} h[i]*(1-h[i+l])
        #              d(corr)/dh[j] = (1-h[j+l]) if j <= n-1-l AND h[j] term
        #                            - h[j-l] if j >= l AND (1-h[j]) term
        #   if l < 0: let l' = -l, corr = sum_{i=l'}^{n-1} h[i]*(1-h[i-l'])
        #              d(corr)/dh[j] = (1-h[j-l']) if j >= l' AND h[j] term
        #                            - h[j+l'] if j <= n-1-l' AND (1-h[j]) term
        
        # This is O(n) per active lag, O(K*n) total
        # For n=640, K~50, that's 32K operations - fine
        
        grad = np.zeros(n)
        for m in range(len(corr)):
            if weights[m] == 0:
                continue
            l = m - (n - 1)
            w = weights[m]
            if l >= 0:
                # corr = sum_{i=0}^{n-1-l} h[i]*(1-h[i+l])
                for j in range(n - l):
                    # h[j] appears as first factor: derivative = (1-h[j+l])
                    grad[j] += w * (1.0 - h[j + l])
                for j in range(l, n):
                    # h[j] appears as (1-h[j]) in second factor (i = j-l): derivative = -h[j-l]
                    grad[j] += w * (-h[j - l])
            else:
                lp = -l
                # corr = sum_{i=lp}^{n-1} h[i]*(1-h[i-lp])
                for j in range(lp, n):
                    grad[j] += w * (1.0 - h[j - lp])
                for j in range(n - lp):
                    grad[j] += w * (-h[j + lp])
        
        step = lr * (1.0 - i / n_iters)
        h = h - step * grad * dx
        h = make_feasible(h, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        if not mask.any():
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary_exact(h, n):
    hb = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(hb.sum())
    if n_ones > target:
        idx = np.where(hb == 1)[0]
        hb[idx[:n_ones - target]] = 0.0
    elif n_ones < target:
        idx = np.where(hb == 0)[0]
        hb[idx[:target - n_ones]] = 1.0
    return hb

def softmax_obj_val(corr, T):
    mc = np.max(corr)
    if T < 1.0:
        T = 1.0
    shifted = np.clip(T * (corr - mc), -50, 50)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def subgrad_h(h, n, active_mask):
    grad = np.zeros(n)
    active_ms = np.where(active_mask)[0]
    for m in active_ms:
        l = m - (n - 1)
        if l >= 0:
            grad[:n - l] += (1.0 - h[l:])
            grad[l:] += -h[:n - l]
        else:
            lp = -l
            grad[lp:] += (1.0 - h[:n - lp])
            grad[:n - lp] += -h[lp:]
    return grad

def subgrad_descent(h0, n, tol_eps, n_iters, lr_init, lr_decay, time_limit):
    t_start = time.time()
    h = h0.copy()
    dx = 2.0 / n
    for i in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        active = corr >= mc - tol_eps * n / 2.0
        K = max(1, int(active.sum()))
        w = active.astype(float) / K
        grad = subgrad_h(h, n, w)
        lr = lr_init * (lr_decay ** i)
        h = h - lr * grad * dx
        h = make_feasible(h, n)
    return h

def lbgfb_optimize(h0, n, T, mu, max_iter, time_limit):
    t_start = time.time()
    dx = 2.0 / n
    target = n / 2.0

    def obj(x):

            return 1e10
        corr = np.correlate(x, 1.0 - x, mode='full')
        return softmax_obj_val(corr, T) * dx + mu * (x.sum() - target) ** 2

        h_f, c_f = binary_sa(h_f, best_n, rng, min(rem() - 10, 35),
                             n_11=3000, n_22=1500, T0=0.15, cool=0.99)
        if c_f < best_c5:
            best_c5, best_h, best_n = c_f, h_f.copy(), best_n

    # Phase 7: Continuous polish on final
    if rem() > 25:
        n_p = min(best_n, 960)
        h_p = upsample_h(best_h, n_p)
        h_p = subgrad_descent(h_p, n_p, tol_eps=0.005, n_iters=2000,
                               lr_init=0.1, lr_decay=0.999,
                               time_limit=min(rem() - 10, 20))
        c_p = compute_c5(h_p)
        if c_p < best_c5:
            best_c5, best_h, best_n = c_p, h_p.copy(), n_p
        if rem() > 15:
            h_opt = lbgfb_optimize(h_p, n_p, 2000, mu=400.0, max_iter=200,
                                   time_limit=min(rem() - 8, 12))
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h, best_n = c_opt, h_opt.copy(), n_p

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
