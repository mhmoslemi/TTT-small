        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = h < 1.0 if diff > 0 else h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

# SLSQP with smooth max
_G_N = 0
_G_T = 1.0

def _obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_eq(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, n, T, maxiter=200, ftol=1e-14):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def rich_sa(b, n, max_iters, T0, T1, rng, time_limit, start_time):
    target = n // 2
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = min(1.0, (it + 1) / max_iters)
        T = T0 * (T1 / T0) ** frac
        
        r = rng.random()
        b_t = b.copy()
        moved = False
        
        if r < 0.35:
            # Single swap
            oi = np.where(b == 1)[0]
            zi = np.where(b == 0)[0]
            i = oi[rng.randint(len(oi))]
            j = zi[rng.randint(len(zi))]
            b_t[i], b_t[j] = 0, 1
            moved = True
        elif r < 0.6:
            # Block exchange
            blen = rng.randint(2, max(3, n // 12))
            s1 = rng.randint(0, n - blen)
            s2 = rng.randint(0, n - blen)
            if abs(s1 - s2) >= blen:
                b_t[s1:s1+blen], b_t[s2:s2+blen] = b[s2:s2+blen], b[s1:s1+blen]
                moved = True
        elif r < 0.8:
            # Reversal
            slen = rng.randint(3, max(4, n // 6))
            s = rng.randint(0, n - slen)
            b_t[s:s+slen] = b[s:s+slen][::-1]
            moved = True
        else:
            # Double swap
            oi = np.where(b == 1)[0]
            zi = np.where(b == 0)[0]
            if len(oi) >= 2 and len(zi) >= 2:
                i1, i2 = rng.choice(oi, 2, replace=False)
                j1, j2 = rng.choice(zi, 2, replace=False)
                b_t[i1], b_t[j1] = 0, 1
                b_t[i2], b_t[j2] = 0, 1
                moved = True
        
        if not moved:
            continue
        
        new_c5 = compute_c5(b_t)
        delta = new_c5 - curr_c5
        if delta <= 0 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
            b = b_t
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
    
    return best_b, best_c5

def perturb_fourier(h, n, rng, n_modes=4, amp=0.05):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def perturb_segment(h, n, rng):
    h_new = h.copy()
    for _ in range(rng.randint(1, 3)):
        length = rng.randint(n // 15, n // 5)
        start = rng.randint(0, max(1, n - length))
        if rng.random() < 0.5:
            h_new[start:start+length] = 1.0 - h_new[start:start+length]
        else:
            h_new[start:start+length] = rng.uniform(0, 1, length)
    return make_feasible(h_new, n)

def perturb_roll(h, n, rng):
    s = rng.randint(1, n // 10)
    h_shift = np.roll(h, s)
    alpha = rng.uniform(0.3, 0.7)
    return make_feasible(alpha * h + (1 - alpha) * h_shift, n)

def coord_descent(h, n, start_time, budget_s, max_rounds=15):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_v = old
            for cand in [0.0, 1.0, max(0.0, old - 0.02), max(0.0, old - 0.005),
                         min(1.0, old + 0.005), min(1.0, old + 0.02)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = cand
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)
    
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    def elapsed():
        return time.time() - start_time
    
    def remaining():
        return budget_s - elapsed()
    
    # Phase 1: Rich binary SA on current solution
    b0 = (h_init >= 0.5).astype(float)
    # Fix sum
    target = n0 // 2
    while np.sum(b0) > target:
        oi = np.where(b0 == 1)[0]
        b0[oi[rng.randint(len(oi))]] = 0
    while np.sum(b0) < target:
        zi = np.where(b0 == 0)[0]
        b0[zi[rng.randint(len(zi))]] = 1
    
    # Multiple SA runs with different parameters
    sa_configs = [
        (80000, 0.008, 0.000005),
        (60000, 0.005, 0.00001),
        (50000, 0.012, 0.000008),
    ]
    
    for maxit, T0, T1 in sa_configs:
        rem = remaining()
        if rem < 60:
            break
        tl = min(rem - 20, 300)
        b, c5 = rich_sa(b0.copy(), n0, maxit, T0, T1, rng, tl, start_time)
        if c5 < best_c5:
            best_c5 = c5
            best_h = b.copy()
            b0 = b.copy()
    
    # Phase 2: SLSQP refinement from best binary
    rem = remaining()
    if rem > 100:
        for T in [100.0, 500.0, 2000.0, 10000.0, 50000.0]:
            rem = remaining()
            if rem < 30:
                break
            x = slsqp_opt(best_h, n0, T, 200, 1e-14)

            if c5 < best_c5:

                best_h = make_feasible(x, n0)

                break
