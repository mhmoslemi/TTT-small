import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def thue_morse(n):
    tm = np.zeros(n)
    for i in range(n):
        tm[i] = bin(i).count('1') % 2
    return tm.astype(float)

def m_sequence(n, rng):
    """Generate a balanced binary sequence with good spreading."""
    h = np.zeros(n)
    half = n // 2
    positions = set(rng.choice(n, size=half, replace=False).tolist())
    for i in positions:
        h[i] = 1.0
    return h

def greedy_binary(h_bin, n, rng, n_rounds=100, n_tries=50000, t_limit=None, t0=None):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for rnd in range(n_rounds):
        if t_limit is not None and (time.time() - t0) > t_limit:
            break
        improved = False
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            new_c = compute_c5(h)
            if new_c < cur - 1e-15:
                cur = new_c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def sa_binary(h_bin, n, rng, t_start=0.05, t_end=0.0005, n_iter=200000, t_limit=None, t0=None):
    h = h_bin.copy()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for it in range(n_iter):
        if t_limit is not None and (time.time() - t0) > t_limit:
            break
        T = t_start * (t_end / t_start) ** (it / n_iter)
        if not ones or not zeros:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur
        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            cur = new_c
            ones[ii] = oj
            zeros[jj] = oi
            if cur < best:
                best = cur
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best

def sa_2swap(h_bin, n, rng, t_start=0.03, t_end=0.001, n_iter=100000, t_limit=None, t0=None):
    """SA with 2-swap moves: flip two 1s to 0 and two 0s to 1."""
    h = h_bin.copy()
    cur = compute_c5(h)
    best = cur
    best_h = h.copy()
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for it in range(n_iter):

            break
        i1 = rng.randint(len(ones))
        i2 = rng.randint(len(ones) - 1)
        if i2 >= i1:
            i2 += 1
        j1 = rng.randint(len(zeros))
        j2 = rng.randint(len(zeros) - 1)
        if j2 >= j1:
            j2 += 1
        o1, o2 = ones[i1], ones[i2]
        z1, z2 = zeros[j1], zeros[j2]
        if o1 == o2 or o1 == z1 or o1 == z2 or o2 == z1 or o2 == z2:
            continue
        h[o1] = 0.0
        h[o2] = 0.0
        h[z1] = 1.0
        h[z2] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur
        if delta < 0 or (T > 1e-12 and rng.rand() < np.exp(-delta / T)):
            cur = new_c
            ones[i1] = z1
            ones[i2] = z2
            zeros[j1] = o1
            zeros[j2] = o2

        else:
            h[o1] = 1.0
            h[o2] = 1.0

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_smooth(h0, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    try:
        res = minimize(obj, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h0.copy(), n)

def alternate_minmax(h0, n, n_iters=40, topk_frac=0.15, max_inner=30):
    dx = 2.0 / n
    h = h0.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        n_c = len(corr)
        k = max(5, int(topk_frac * n_c))
        idx = np.argpartition(corr, -k)[-k:]
        active = np.zeros(n_c)
        active[idx] = 1.0
        active /= active.sum()
        mu = 500.0
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(active, c)) * dx + mu * (x.sum() - n / 2.0) ** 2
        try:
            res = minimize(obj, h, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': max_inner, 'ftol': 1e-15, 'gtol': 1e-10})
            h_new = make_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-10:
                h = h_new
        except Exception:
            pass
    return h

def blocks_to_h(widths, heights, n):
    total_width = np.sum(widths)
    if total_width < 1e-10:
        return make_feasible(np.zeros(n), n)
    widths = widths / total_width * 2.0
    h = np.zeros(n)
    start = 0.0
    for w, a in zip(widths, heights):
        s_idx = int(round(start / 2.0 * n))
        e_idx = int(round((start + w) / 2.0 * n))
        e_idx = min(e_idx, n)
        s_idx = max(s_idx, 0)
        if e_idx > s_idx:
            h[s_idx:e_idx] = a
        start += w
    return make_feasible(h, n)

            best_n = n_try

        h_b2 = (h_up > 0.5).astype(float)
        target = n_try // 2
        n_ones2 = int(h_b2.sum())
        if n_ones2 > target:
            idx2 = np.where(h_b2 == 1)[0]
            h_b2[idx2[:n_ones2 - target]] = 0.0
        elif n_ones2 < target:
            idx2 = np.where(h_b2 == 0)[0]
            h_b2[idx2[:target - n_ones2]] = 1.0

        if rem() > 40:
            h_g3, c_g3 = greedy_binary(h_b2, n_try, rng, n_rounds=40, n_tries=30000,
                                        t_limit=budget_s * 0.6, t0=t0)
            if c_g3 < best_c5:
                best_c5 = c_g3
                best_h = h_g3.copy()
                best_n = n_try

        if rem() > 30:
            h_sa3, c_sa3 = sa_binary(h_b2, n_try, rng, t_start=0.02, t_end=0.0005,
                                      n_iter=200000, t_limit=budget_s * 0.7, t0=t0)
            if c_sa3 < best_c5:
                best_c5 = c_sa3
                best_h = h_sa3.copy()
                best_n = n_try

    # Phase 3: Continuous refinement with L-BFGS-B
    if rem() > 50:
        n_ref = len(best_h)
        for T in [200, 800, 3000]:
            if rem() < 25:
                break
            h_opt = lbgfb_smooth(best_h, n_ref, T, mu=600.0, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 4: Alternating min-max
    if rem() > 40:
        h_alt = alternate_minmax(best_h, len(best_h), n_iters=30, topk_frac=0.12, max_inner=25)
        c_alt = compute_c5(h_alt)
        if c_alt < best_c5:
            best_c5 = c_alt
            best_h = h_alt.copy()

    # Phase 5: Multi-start from diverse points at best resolution
    if rem() > 80:
        n_ms = len(best_h)
        for restart in range(5):
            if rem() < 30:
                break
            h_r = best_h + 0.1 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            # Quick greedy on binary version
            h_rb = (h_r > 0.5).astype(float)
            target = n_ms // 2
            n_o = int(h_rb.sum())
            if n_o > target:
                idx_r = np.where(h_rb == 1)[0]
                h_rb[idx_r[:n_o - target]] = 0.0
            elif n_o < target:
                idx_r = np.where(h_rb == 0)[0]
                h_rb[idx_r[:target - n_o]] = 1.0
            h_gr, c_gr = greedy_binary(h_rb, n_ms, rng, n_rounds=30, n_tries=20000,
                                        t_limit=budget_s * 0.9, t0=t0)
            if c_gr < best_c5:
                best_c5 = c_gr
                best_h = h_gr.copy()

    # Phase 6: Final polish
    if rem() > 20:
        n_fin = len(best_h)
        h_fin = (best_h > 0.5).astype(float)
        target = n_fin // 2
        n_o = int(h_fin.sum())
        if n_o > target:
            idx_f = np.where(h_fin == 1)[0]
            h_fin[idx_f[:n_o - target]] = 0.0
        elif n_o < target:
            idx_f = np.where(h_fin == 0)[0]
            h_fin[idx_f[:target - n_o]] = 1.0
        h_fp, c_fp = greedy_binary(h_fin, n_fin, rng, n_rounds=50, n_tries=40000)
        if c_fp < best_c5:
            best_c5 = c_fp
            best_h = h_fp.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
