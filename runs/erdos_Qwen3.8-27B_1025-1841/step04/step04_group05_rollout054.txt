    """One subgradient step targeting the max correlation lag."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    
    # Compute gradient of corr at this lag w.r.t. h
    g = np.zeros(n)
    if lag >= 0:
        # corr = sum_{i=0}^{n-1-lag} h[i] * (1-h[i+lag])
        # d/dh[j]: from first sum (i=j) if j <= n-1-lag: (1-h[j+lag])
        #          from second sum (i+lag=j, i=j-lag) if lag <= j <= n-1: -h[j-lag]
        valid1 = np.arange(0, n - lag)
        g[valid1] += (1.0 - h[valid1 + lag])
        valid2 = np.arange(lag, n)
        g[valid2] += (-h[valid2 - lag])
    else:

        # corr = sum_{i=0}^{n-1-k} h[i+k] * (1-h[i])
        # d/dh[j]: from first sum (i+k=j, i=j-k) if k <= j: (1-h[j-k])
        #          from second sum (i=j) if j <= n-1-k: -h[j+k]
        valid1 = np.arange(k, n)
        g[valid1] += (1.0 - h[valid1 - k])
        valid2 = np.arange(0, n - k)
        g[valid2] += (-h[valid2 + k])
    
    # Projected gradient step
    h_new = h - alpha * g
    h_new = make_feasible(h_new, n)
    return h_new, corr[max_idx] * dx

            for d in deltas:
                for bv in [0.0, 1.0]:
                    cand_val = old + d
                    if cand_val < -1e-9 or cand_val > 1 + 1e-9:
                        continue
                    trial = h.copy()
                    trial[j] = np.clip(cand_val, 0.0, 1.0)
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        h = trial
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _ON, _OT
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    def el():
        return time.time() - t0 - (budget_s - tl())

    def elapsed_fn():
        return time.time() - t0 - (budget_s - tl())

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = (rng.rand(640) < 0.5).astype(float)
        h0 = make_feasible(h0, len(h0))

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    n = len(h0)
    for T in [50.0, 200.0, 500.0]:
        if tl() < 40:
            break
        try:
            ho = slsqp_pass(best_h, n, T, 250)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho.copy()
        except Exception:
            pass

    for n2 in [1280, 2560, 4096]:
        if tl() < 250:
            break
        hc = upsample_h(best_h, n2)
        cc = compute_c5(hc)
        if cc < best_c5:
            best_c5, best_h = cc, hc.copy()

        for T in [100.0, 400.0, 1500.0]:
            if tl() < 60:
                break
            try:
                ho = slsqp_pass(hc, n2, T, 300)
                co = compute_c5(ho)
                if co < best_c5:
                    best_c5, best_h = co, ho.copy()
                    hc = ho.copy()
            except Exception:
                pass

        if tl() > 120:
            hsg, csg = wsubgrad(hc, n2, 5000, 0.004, 5, elapsed_fn)
            if csg < best_c5:
                best_c5, best_h = csg, hsg.copy()

    nf = len(best_h)
    if tl() > 60:
        try:
            ho = slsqp_pass(best_h, nf, 3000.0, 200)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho.copy()
        except Exception:
            pass

    if tl() > 80:
        hsg, csg = wsubgrad(best_h, nf, 8000, 0.002, 5, elapsed_fn)
        if csg < best_c5:
            best_c5, best_h = csg, hsg.copy()

    if tl() > 60:
        hc2, cc2 = coord_descent(best_h, nf, 20, elapsed_fn)
        if cc2 < best_c5:
            best_c5, best_h = cc2, hc2.copy()

    if tl() > 30:
        try:
            ho = slsqp_pass(best_h, nf, 5000.0, 150)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho.copy()
        except Exception:
            pass

    for _ in range(3):
        if tl() < 15:
            break
        try:
            ho = slsqp_pass(best_h, nf, 2000.0, 100)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5, best_h = co, ho.copy()
        except Exception:
            break

    best_h = make_feasible(best_h, len(best_h))
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
