    shifted = T * (corr - M)
    smax = np.max(shifted)
    exp_s = np.exp(shifted - smax)
    w = exp_s / np.sum(exp_s)
    log_sum = np.log(np.sum(exp_s)) + smax
    obj = (M + log_sum / T) * dx + P * (np.sum(h) - n / 2.0) ** 2
    s = w
    s_rev = s[::-1]
    N = 3 * n - 2
    Nfft = 1 << (N - 1).bit_length()
    fa = np.fft.rfft(s, Nfft)
    fa_rev = np.fft.rfft(s_rev, Nfft)
    fb1 = np.fft.rfft(1.0 - h, Nfft)
    fb2 = np.fft.rfft(h, Nfft)
    first_sum = np.fft.irfft(fa * fb1, Nfft)[n - 1:2 * n - 1]
    second_sum = np.fft.irfft(fa_rev * fb2, Nfft)[n - 1:2 * n - 1]
    sum_diff = np.sum(h) - n / 2.0
    grad = dx * (first_sum - second_sum) + 2.0 * P * sum_diff
    return obj, grad

def _lbfgs_obj(h):
    return lse_obj_and_grad(h, _G_T, _G_P)[0]

def _lbfgs_grad(h):
    return lse_obj_and_grad(h, _G_T, _G_P)[1]

def _slsqp_obj(h):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    M = np.max(corr)
    shifted = _OPT_T * (corr - M)
    smax = np.max(shifted)
    log_sum = np.log(np.sum(np.exp(shifted - smax))) + smax
    return (M + log_sum / _OPT_T) * dx

def _slsqp_cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _slsqp_cons_jac(h):
    return np.ones(_OPT_N)

def _slsqp_optimize(h_start, n, T, max_iter):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
    res = minimize(_slsqp_obj, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def _lbfgs_run(h_start, n, T, P, max_iter):
    global _G_T, _G_P
    _G_T = T
    _G_P = P
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_lbfgs_obj, h_start, jac=_lbfgs_grad, method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': max_iter, 'ftol': 1e-14,
                                'gtol': 1e-10, 'maxfun': max_iter * 5})
        return res.x
    except Exception:
        return h_start

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    resolutions = [1280, 2560, 5120, 10240]

    for n_pts in resolutions:
        if time_left() < 80:
            break

        h_curr = upsample_h(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        schedules = [
            (50.0, 5000.0, 400),
            (25.0, 10000.0, 400),
            (10.0, 20000.0, 400),
            (5.0, 50000.0, 350),
            (2.0, 100000.0, 300),
            (1.0, 200000.0, 250),
        ]

        for T, P, max_it in schedules:
            if time_left() < 40:
                break
            h_opt = _lbfgs_run(h_curr, n_pts, T, P, max_it)
            h_opt = make_feasible(h_opt, n_pts)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_curr = h_opt.copy()

        # Perturbation restarts at this resolution
        if time_left() > 100:
            for restart in range(2):
                if time_left() < 50:
                    break
                if restart == 0:
                    h_pert = h_curr + 0.08 * rng.randn(n_pts)
                else:
                    block_size = max(20, n_pts // 40)
                    start = rng.randint(0, max(1, n_pts - block_size))
                    h_pert = h_curr.copy()
                    h_pert[start:start + block_size] = 1.0 - h_pert[start:start + block_size]
                h_pert = make_feasible(h_pert, n_pts)
                h_opt = _lbfgs_run(h_pert, n_pts, 10.0, 50000.0, 300)
                h_opt = make_feasible(h_opt, n_pts)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_curr = h_opt.copy()

    # SLSQP polish at manageable resolution
    n_polish = min(len(best_h), 2560)
    h_polish = upsample_h(best_h, n_polish) if n_polish < len(best_h) else best_h.copy()

    for T in [20, 100, 500]:
        if time_left() < 25:
            break
        h_opt = _slsqp_optimize(h_polish, n_polish, T, 150)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            if n_polish < len(best_h):
                best_h = upsample_h(h_opt, len(best_h))
            else:
                best_h = h_opt.copy()
            h_polish = h_opt.copy()

    # Coordinate descent at lower resolution
    n_cd = min(len(best_h), 1280)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    cd_deltas = [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]

    for cd_round in range(10):
        if time_left() < 15:
            break
        improved = False
        for j in range(n_cd):
            if time_left() < 15:
                break
            old_val = h_cd[j]
            for d in cd_deltas:
                new_val = old_val + d
                if new_val < 0 or new_val > 1:
                    continue
                trial = h_cd.copy()
                trial[j] = new_val
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < best_c5 - 1e-10:
                    best_c5 = c
                    h_cd = trial
                    old_val = new_val
                    improved = True
            for bv in (0.0, 1.0):
                if abs(bv - h_cd[j]) < 1e-9:
                    continue
                trial = h_cd.copy()
                trial[j] = bv
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < best_c5 - 1e-10:
                    best_c5 = c
                    h_cd = trial
                    improved = True
        if not improved:
            break

    # Project CD result back
    if n_cd < len(best_h):
        h_full = upsample_h(h_cd, len(best_h))
        c_full = compute_c5(h_full)
        if c_full < best_c5:
            best_c5 = c_full
            best_h = h_full
    else:
        c_cd = compute_c5(h_cd)
        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd

    # Final SLSQP at full resolution
    n_final = len(best_h)
    if time_left() > 20 and n_final <= 5120:
        h_opt = _slsqp_optimize(best_h, n_final, 200.0, 100)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
