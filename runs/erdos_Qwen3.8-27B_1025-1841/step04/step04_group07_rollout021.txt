```python
def topk_focus_optimize(h0, n, n_iters, k_frac, mu):

def swap_local_search(h0, n, rng, max_rounds, tries_per_round):
    """Local search with swap moves for near-binary solutions."""
    h = h0.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries_per_round, max(1, len(ones)) * max(1, len(zeros)))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi], h[oj] = h[oj], h[oi]  # swap values
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                improved = True
                ones[ii], zeros[jj] = zeros[jj], ones[ii]
            else:
                h[oi], h[oj] = h[oj], h[oi]  # undo
        if not improved:
            break
    return h, cur

def guided_swap_search(h0, n, rng, max_rounds, n_worst):
    """Guided local search: focus swaps on worst overlaps."""
    h = h0.copy()
    cur = compute_c5(h)
    dx = 2.0 / n
    corr_vec = np.correlate(h, 1.0 - h, mode='full') * dx
    
    for rnd in range(max_rounds):
        improved = False
        # Find worst shifts
        worst_idx = np.argsort(corr_vec)[-n_worst:]
        
        for _ in range(2000):
            # Pick a random worst shift
            k_idx = rng.choice(worst_idx)
            k = k_idx - (n - 1)  # actual shift
            
            # Find a position i where h[i] is high and h[i+k] is low
            # (this makes the product h[i]*(1-h[i+k]) large)
            # Try to swap h[i] with some h[j] to reduce this
            if k >= 0:
                valid_i = range(n - k)
            else:
                valid_i = range(-k, n)
            
            i = rng.choice(list(valid_i))
            # Find j to swap with
            j = rng.randint(n)
            if j == i:
                continue
            
            h[i], h[j] = h[j], h[i]
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                improved = True
                corr_vec = np.correlate(h, 1.0 - h, mode='full') * dx
                worst_idx = np.argsort(corr_vec)[-n_worst:]
            else:
                h[i], h[j] = h[j], h[i]
        
        if not improved:
            break
    return h, cur
```

```python
def project_to_constraint(h, n):
    """Project h onto {x: 0 <= x <= 1, sum(x) = n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = x.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(h - (lo + hi) / 2, 0, 1)
    # Fix small numerical errors
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = (x < 1.0) & (x > 0.0)
        else:
            mask = (x > 0.0) & (x < 1.0)
        if mask.sum() > 0:
            x[mask] += diff / mask.sum()
            x = np.clip(x, 0, 1)
    return x
```

```python
        mc = np.max(corr)
        w = np.exp(np.clip(T * (corr - mc), -100, 100))
        w /= w.sum()

        # For efficiency, use the fact that the gradient is a weighted sum
        # of the gradients of individual overlaps
        # grad[f] = sum_k w[k] * grad[overlap_k]
        # grad[overlap_k][j] = dx * ((1-h[(j+k)%n]) - h[(j-k)%n]) (circular approx)
        
        # For the non-circular case, it's more complex. Let me use a simpler approach:
        # finite difference on a few coordinates to estimate the gradient direction
        # Or just use the circular approximation which should be close for large n.

        # Use top weighted shifts for efficiency
        n_active = min(100, len(corr))
        top_w = np.argpartition(w, -n_active)[-n_active:]
        for m in top_w:
            k = m - (n - 1)
            idx_pos = (np.arange(n) + k) % n
            idx_neg = (np.arange(n) - k) % n
            grad += w[m] * ((1 - y[idx_pos]) - y[idx_neg])
        
        # Nesterov step
        y_new = project_to_constraint(y - lr * grad, n)
        t_new = (1 + np.sqrt(1 + 4 * t * t)) / 2
        y = y_new + ((t - 1) / t_new) * (y_new - h)
        y = np.clip(y, 0, 1)
        h = y_new
        t = t_new
        
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    return best_h, best_c
```

Hmm, but the circular approximation for the gradient might not be accurate enough. The problem is that the "full" correlation doesn't wrap around, so the gradient at the edges is different.

Actually, for the "full" mode correlation:
np.correlate(a, v, mode='full')[m] = sum_i a[min(i+m, ...)] * v[i]

More precisely, for arrays of length n:
full_correlation[m] = sum_{i} a[i] * v[i - (m - n + 1)]
where the sum is over valid i and i-(m-n+1).

Let k = m - (n-1). Then:
full_correlation[m] = sum_i a[i] * v[i-k]

For this to be valid, we need 0 ≤ i < n and 0 ≤ i-k < n.
So i ranges from max(0, k) to min(n-1, n-1+k).

The gradient of full_correlation[m] with respect to a[j]:
- a[j] appears in the term where i = j: contributes v[j-k] (if valid)
- a[j] also appears if j = i-k for some i, i.e., i = j+k: contributes v[j+k]... wait no.

Let me be more careful. Let's say a = h, v = 1-h.
full[m] = sum_i h[i] * (1-h[i-k]) where k = m-(n-1)

d/dh[j]:
- h[j] appears directly: if 0 ≤ j-k < n, then term is (1-h[j-k])
- h[j] appears in (1-h[i-k]) when i-k = j, i.e., i = j+k: if 0 ≤ j+k < n, then term is -h[j+k]

So d full[m] / d h[j] = (1-h[j-k]) * 1[j-k ∈ [0,n-1]] - h[j+k] * 1[j+k ∈ [0,n-1]]

This is the exact (non-circular) gradient. Let me implement this properly.

```python
def compute_grad(corr_weights, h, n):
    """Compute gradient of weighted sum of overlaps."""
    # corr_weights[m] for m in range(2n-1)
    # k = m - (n-1)
    # d full[m] / d h[j] = (1-h[j-k]) * 1[0<=j-k<n] - h[j+k] * 1[0<=j+k<n]
    grad = np.zeros(n)
    js = np.arange(n)
    for m in np.where(corr_weights > 1e-15)[0]:
        k = m - (n - 1)
        # First term: (1-h[j-k]) for 0 <= j-k < n, i.e., k <= j < k+n
        mask1 = (js - k >= 0) & (js - k < n)
        # Second term: -h[j+k] for 0 <= j+k < n, i.e., -k <= j < n-k
        mask2 = (js + k >= 0) & (js + k < n)
        grad[mask1] += corr_weights[m] * (1 - h[js[mask1] - k])
        grad[mask2] -= corr_weights[m] * h[js[mask2] + k]
    return grad
```

    return best_h, best_c

def lbgfb_smooth(h0, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full') * dx
        return float(smooth_max(corr, T)) + mu * (x.sum() - n/2.0)**2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def topk_focus(h0, n, n_iters=30, k_frac=0.08, mu=500):
    h = h0.copy()
    dx = 2.0 / n
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = max(10, int(k_frac * len(corr)))
        idx = np.argpartition(corr, -k)[-k:]
        active = np.zeros(len(corr))
        active[idx] = 1.0 / k
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(active, c)) * dx + mu * (x.sum() - n/2.0)**2
        
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 25, 'ftol': 1e-16, 'gtol': 1e-10})
        h_new = make_feasible(res.x, n)
        if compute_c5(h_new) <= compute_c5(h) + 1e-10:
            h = h_new
    return h

def swap_search(h0, n, rng, max_rounds=30, tries=3000):
    h = h0.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tr = min(tries, max(1, len(ones)) * max(1, len(zeros)))
        for _ in range(n_tr):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi], h[oj] = h[oj], h[oi]
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii], zeros[jj] = zeros[jj], ones[ii]
                improved = True
            else:
                h[oi], h[oj] = h[oj], h[oi]
        if not improved:
            break
    return h, cur

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)
