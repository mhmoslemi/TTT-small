def proj_subgrad_optimize(h, n, step_init=0.005, top_k=30, n_iters=20000, decay=0.9999):
    dx = 2.0 / n
    target = n / 2.0
    h = h.astype(float).copy()
    best_c5 = float('inf')
    best_h = h.copy()
    step = step_init
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        curr_c5 = np.max(corr) * dx
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h.copy()
        
        # Top K lags
        top_idx = np.argsort(corr)[-top_k:]
        
        # Compute gradient
        grad = np.zeros(n)
        for k in top_idx:
            # d(corr[k])/d(h[j]) = (1-h[j+k-n+1]) - h[j-k+n-1] (with range checks)
            # First term: for j where j+k-n+1 in [0,n-1]
            # j in [max(0,n-1-k), min(n-1, 2n-2-k)]
            j_min = max(0, n - 1 - k)
            j_max = min(n - 1, 2 * n - 2 - k)
            
            # First term: (1-h[j+k-n+1]) for j in [j_min, j_max]
            if j_min <= j_max:
                idx = j_min + np.arange(j_max - j_min + 1)
                idx2 = idx + k - n + 1
                grad[idx] += (1.0 - h[idx2])
                
            # Second term: -h[j-k+n-1] for j where j-k+n-1 in [0,n-1]
            # j in [max(0, k-n+1), min(n-1, k)]
            j_min2 = max(0, k - n + 1)
            j_max2 = min(n - 1, k)
            if j_min2 <= j_max2:
                idx = j_min2 + np.arange(j_max2 - j_min2 + 1)
                idx2 = idx - k + n - 1
                grad[idx] -= h[idx2]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Gradient descent step
        h_new = h - step * grad / grad_norm
        
        # Project to [0,1] and match sum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        # Check if improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < curr_c5:
            h = h_new
            step *= 1.1  # increase step
        else:
            step *= 0.7  # decrease step
            step = max(step, 1e-8)
        
        # Decay
        step *= decay
    
    return best_h, best_c5
