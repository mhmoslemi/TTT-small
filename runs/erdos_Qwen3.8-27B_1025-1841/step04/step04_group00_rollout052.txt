def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize_with_count(h, n, rng):
    """Binarize h to have exactly n//2 ones."""
    v = (h >= 0.5).astype(float)
    target = n // 2
    num_ones = int(np.sum(v))
    if num_ones > target:
        idx = np.where(v == 1)[0]
        order = rng.permutation(len(idx))

    elif num_ones < target:

        v[idx[order[:target - num_ones]]] = 1
    return v

def sa_binary(v, n, dx, rng, time_limit, T_init, T_min, cool_rate):

    best_v = v.copy()
    
    T = T_init
    it = 0
    while time.time() - start < time_limit and T > T_min:
        # Pick random positions
        p = int(ones_list[rng.integers(len(ones_list))])
        q = int(zeros_list[rng.integers(len(zeros_list))])

        v[p] = 0

            ones_list.append(q)
            zeros_list.remove(q)

            v[p] = 1
            v[q] = 0

        it += 1

        n_tries = 8000
        for _ in range(n_tries):
            if time.time() - start > time_limit - 1:
                break
            ones_idx = np.flatnonzero(v)
            zeros_idx = np.flatnonzero(1 - v)
            p = int(ones_idx[rng.integers(len(ones_idx))])
            q = int(zeros_idx[rng.integers(len(zeros_idx))])
            v[p] = 0.0
            v[q] = 1.0
            corr = np.correlate(v, 1.0 - v, mode='full')
            new_max = float(np.max(corr))
            if new_max < cur_max - 1e-12:
                cur_max = new_max
                if cur_max * dx < best_c5:
                    best_c5 = cur_max * dx
                    best_v = v.copy()
                improved = True
            else:
                v[p] = 1.0
                v[q] = 0.0
        if not improved:
            break
    return best_v, best_c5

def slsqp_smooth(v, n, dx, time_limit, T_smooth=300.0, top_m=None):
    from scipy.optimize import minimize
    h0 = v.astype(np.float64).copy()

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None and top_m < len(corr):
            top_idx = np.argsort(corr)[-top_m:]
            mc = np.max(corr[top_idx])
            sub = corr[top_idx] - mc
            return (mc + np.log(np.sum(np.exp(T_smooth * sub))) / T_smooth) * dx
        mc = np.max(corr)
        sub = corr - mc
        return (mc + np.log(np.sum(np.exp(T_smooth * sub))) / T_smooth) * dx

    def con_eq(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq}]
    try:
        end_t = start_time_global + time_limit
        res = minimize(obj, h0, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 500, 'ftol': 1e-18})
        x = make_feasible(res.x, n)
        c5 = compute_c5(x)
        return x, c5
    except Exception:
        return h0, compute_c5(h0)

start_time_global = 0.0

def run(seed=42, budget_s=1300, **kwargs):
    global start_time_global
    start_time_global = time.time()
    rng = np.random.default_rng(seed)

    def elapsed():
        return time.time() - start_time_global

    def remaining():
        return budget_s - elapsed()

    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_orig = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Binary SA at original resolution
    n = n_orig
    dx = 2.0 / n
    v1 = binarize_with_count(h_init, n, rng)
    c5_v1 = compute_c5(v1)
    if c5_v1 < best_c5:
        best_c5 = c5_v1
        best_h = v1.copy()

    t1 = min(500, remaining() - 60)
    if t1 > 30:
        v1, c5_1 = sa_binary(v1, n, dx, rng, t1, T_init=2.0, T_min=0.02, cool_rate=0.999985)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = v1.copy()

    # Phase 2: Greedy local search
    t2 = min(120, remaining() - 40)
    if t2 > 20:
        v_ls = best_h.copy()
        v_ls, c5_ls = greedy_ls(v_ls, n, dx, rng, t2, max_rounds=20)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = v_ls.copy()

    # Phase 3: SA restarts with perturbation
    for restart in range(4):
        if remaining() < 150:
            break
        v_r = best_h.copy()
        k_pert = int(rng.integers(4, 10))
        ones_i = np.flatnonzero(v_r)
        zeros_i = np.flatnonzero(1 - v_r)
        for _ in range(k_pert):
            p = int(ones_i[rng.integers(len(ones_i))])
            q = int(zeros_i[rng.integers(len(zeros_i))])
            v_r[p], v_r[q] = v_r[q], v_r[p]
        t_r = min(80, remaining() - 50)
        if t_r < 15:
            break
        v_r, c5_r = sa_binary(v_r, n, dx, rng, t_r, T_init=1.5, T_min=0.03, cool_rate=0.99998)
        t_g = min(30, remaining() - 30)
        if t_g > 10:
            v_r, c5_r = greedy_ls(v_r, n, dx, rng, t_g, max_rounds=8)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = v_r.copy()

    # Phase 4: Try n*2 resolution
    n2 = 2 * n
    if remaining() > 300 and n2 <= 3840:
        dx2 = 2.0 / n2
        v_up = np.repeat(best_h.astype(np.float64), 2)
        v_up = (v_up >= 0.5).astype(np.float64)
        target2 = n2 // 2
        num1 = int(np.sum(v_up))
        if num1 > target2:
            idx = np.where(v_up == 1)[0]
            rng.shuffle(idx)
            v_up[idx[:num1 - target2]] = 0
        elif num1 < target2:
            idx = np.where(v_up == 0)[0]
            rng.shuffle(idx)
            v_up[idx[:target2 - num1]] = 1
        t_up = min(200, remaining() - 80)
        if t_up > 30:
            v_up, c5_up = sa_binary(v_up, n2, dx2, rng, t_up, T_init=2.0, T_min=0.02, cool_rate=0.99999)
            t_g2 = min(60, remaining() - 40)
            if t_g2 > 15:
                v_up, c5_up = greedy_ls(v_up, n2, dx2, rng, t_g2, max_rounds=10)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = v_up.copy()
                n = n2

    # Phase 5: Continuous SLSQP polish
    if remaining() > 60:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        t_sl = min(remaining() - 20, 120)
        x_sl, c5_sl = slsqp_smooth(best_h, n_f, dx_f, t_sl, T_smooth=400.0)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = x_sl

    # Phase 6: Second SLSQP with different T
    if remaining() > 40:
        n_f = len(best_h)
        dx_f = 2.0 / n_f
        t_sl2 = min(remaining() - 10, 60)
        x_sl2, c5_sl2 = slsqp_smooth(best_h, n_f, dx_f, t_sl2, T_smooth=800.0, top_m=7)
        if c5_sl2 < best_c5:
            best_c5 = c5_sl2
            best_h = x_sl2

    # Phase 7: Micro coordinate adjustments at max-shift positions
    if remaining() > 20:
        n_f = len(best_h)
        h_m = best_h.copy()
        for _ in range(200):
            if time.time() - start_time_global > budget_s - 3:
                break
            j = int(rng.integers(n_f))
            improved_here = False
            for d in [0.01, -0.01, 0.005, -0.005, 0.02, -0.02, 0.003, -0.003]:
                trial = h_m.copy()
                trial[j] = np.clip(trial[j] + d, 0.0, 1.0)
                trial = make_feasible(trial, n_f)
                c5_t = compute_c5(trial)
                if c5_t < best_c5 - 1e-15:
                    best_c5 = c5_t
                    best_h = trial
                    h_m = trial
                    improved_here = True
                    break
            if not improved_here:
                continue

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
