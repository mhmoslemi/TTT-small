def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def topk_weights(corr, k=20):
    n_c = len(corr)
    k = min(k, n_c)
    idx = np.argpartition(corr, -k)[-k:]
    w = np.zeros(n_c)
    w[idx] = 1.0
    w /= w.sum()
    return w

def lbgfb_topk(h0, n, k, mu, max_iter=200):
    dx = 2.0 / n
    best_h = h0.copy()

        w = topk_weights(corr, k)

            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(w, c)) * dx + mu * (x.sum() - n / 2.0) ** 2

        h_bp2, c_bp2 = binary_swaps(h_bp, n_bp, rng, max_tries=min(30000, int(rem()//0.002)))
        c_bp2_dx = c_bp2 * (2.0/n_bp)
        if c_bp2_dx < best_c5:
            best_c5 = c_bp2_dx
            best_h = h_bp2.copy()
    
    # Phase 7: Final gradient nudge
    if rem() > 10:
        h_n = make_feasible(best_h, len(best_h))
        h_opt = lbgfb_softmax(h_n, len(best_h), 50, mu=200, max_iter=50)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()
    
    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    return best_h, best_c5, len(best_h)
