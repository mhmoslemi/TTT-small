import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def subgradient_pg(h, n, n_iters=200, topk_frac=0.08, lr_init=None, time_limit=60):
    """Subgradient projected gradient with backtracking."""
    if lr_init is None:
        lr_init = 0.5 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t_start = time.time()
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = float(np.max(corr)) * (2.0/n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        n_corr = len(corr)
        k = max(5, int(topk_frac * n_corr))
        top_idx = np.argpartition(corr, -k)[-k:]
        
        # Compute subgradient
        grad = np.zeros(n)
        for m in top_idx:
            lag = int(m) - (n - 1)
            if lag >= 0:
                grad[:n-lag] += (1.0 - h[lag:])
                grad[lag:] += (-h[:n-lag])
            else:
                lp = -lag
                grad[:n-lp] += (-h[lp:])
                grad[lp:] += (1.0 - h[:n-lp])
        grad /= k
        
        # Backtracking line search
        lr = lr_init * (0.95 ** (it // 20))
        for bt in range(10):
            trial = h - lr * grad
            trial = np.clip(trial, 0, 1)
            trial = make_feasible(trial, n)
            c_trial = compute_c5(trial)
            if c_trial < c5 - 1e-15:
                h = trial
                break
            lr *= 0.8
        else:
            # No improvement, try smaller lr
            lr *= 0.5
            lr_init = lr
    
    return best_h, best_c5

def swap_local_search(h, n, rng, max_rounds=100, time_limit=60):
    """Binary swap local search."""
    h_bin = (h > 0.5).astype(float)
    h_bin = make_feasible(h_bin, n)
    cur = compute_c5(h_bin)
    ones = np.where(h_bin > 0.5)[0].tolist()
    zeros = np.where(h_bin <= 0.5)[0].tolist()
    
    t_start = time.time()
    for rnd in range(max_rounds):
        if time.time() - t_start > time_limit:
            break
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        n_tries = min(3000, n_ones * n_zeros)
        
        for _ in range(n_tries):
            ii = rng.randint(n_ones)
            jj = rng.randint(n_zeros)
            oi, oj = ones[ii], zeros[jj]
            h_bin[oi] = 0.0
            h_bin[oj] = 1.0
            c = compute_c5(h_bin)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h_bin[oi] = 1.0
                h_bin[oj] = 0.0
        if not improved:
            break
    
    return h_bin, cur

def soft_max_smooth(h, n, T, max_iter=200, mu=500.0, time_limit=60):
    """L-BFGS-B with soft-max objective."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -50, 50)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx + mu * (x.sum() - n/2.0)**2
    
    res = minimize(obj, h, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)
