If m < 0: valid j are -m to n-1, so j+m ranges from 0 to n-1+m.
Sum = sum(h[0:n+m]) = sum(h[0:n+|m|])

So correlate(h, 1-h, 'full')[k] = S_overlap(m) - A(m) where:
- For m ≥ 0: S_overlap = sum(h[m:]) = S_total - cumsum_h[m]
- For m < 0: S_overlap = sum(h[:n+m]) = cumsum_h[n+m]

where cumsum_h is the cumulative sum with cumsum_h[0] = 0, cumsum_h[i] = sum(h[0:i]).

And A(m) = sum_{j: j+m valid} h[j]*h[j+m] is the autocorrelation at lag m.

So C₅ = max_m (S_overlap(m) - A(m)) * dx

To compute this efficiently:
1. Compute A(m) for all m via FFT (autocorrelation)
2. Compute S_overlap(m) via cumulative sum
3. Take max of (S_overlap - A)

def compute_c5_fast(h):
    n = len(h)
    dx = 2.0 / n
    # Autocorrelation via FFT
    Nfft = 1 << (2*n - 1 - 1).bit_length()
    Fh = np.fft.rfft(h, Nfft)
    acorr = np.fft.irfft(Fh * Fh.conj(), Nfft)
    # acorr[m] for m = 0, ..., n-1 gives A(m) (positive lags)
    # acorr[Nfft-m] gives A(-m) = A(m)
    
    # Cumulative sum for S_overlap
    cs = np.concatenate([[0.0], np.cumsum(h)])
    S_total = cs[n]
    
    # For m = 0 to n-1: S_overlap = S_total - cs[m]
    # For m = -(n-1) to -1: S_overlap = cs[n+m] = cs[n-|m|]
    # For m = 0: S_overlap = S_total
    
    # Correlation values:
    # For m = 0 to n-1: val(m) = (S_total - cs[m]) - acorr[m]
    # For m = 0 to n-1: val(-m) = cs[n-m] - acorr[m]
    
    vals = np.empty(2*n - 1)
    # m = 0
    vals[n-1] = S_total - acorr[0]  # cs[0] = 0, so S_total - cs[0] = S_total
    # m = 1 to n-1
    for m in range(1, n):
        vals[n-1 + m] = (S_total - cs[m]) - acorr[m]
        vals[n-1 - m] = cs[n-m] - acorr[m]
    
    return float(np.max(vals) * dx)

import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    Nfft = 1 << (2 * n - 2).bit_length()
    Fh = np.fft.rfft(h, Nfft)
    acorr = np.fft.irfft(Fh * Fh.conj(), Nfft)[:n]
    cs = np.concatenate([[0.0], np.cumsum(h)])
    S_total = cs[n]
    pos_vals = (S_total - cs[:n]) - acorr
    neg_vals = cs[n-1:0:-1] - acorr[1:n]
    return float(np.max(np.concatenate([neg_vals, pos_vals])) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def fast_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = np.sum(b)
    while int(s) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[rng.randint(len(ones))]
        b[idx] = 0
    s = np.sum(b)
    while int(s) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[rng.randint(len(zeros))]
        b[idx] = 1
    curr_c5 = compute_c5_fft(b)
    best_c5 = curr_c5
    best_b = b.copy()
    ones_set = list(np.where(b == 1)[0])
    zeros_set = list(np.where(b == 0)[0])
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        i = ones_set[rng.randint(len(ones_set))]
        j = zeros_set[rng.randint(len(zeros_set))]
        b[i] = 0.0
        b[j] = 1.0
        new_c5 = compute_c5_fft(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-12))):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
            ones_set[i] = j
            zeros_set[j] = i
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c5

def block_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = np.sum(b)
    while int(s) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[rng.randint(len(ones))]
        b[idx] = 0
    s = np.sum(b)
    while int(s) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[rng.randint(len(zeros))]
        b[idx] = 1
    curr_c5 = compute_c5_fft(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, min(12, n // 2))
        start = rng.randint(0, n - bsize)
        end = start + bsize
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0.0
        b[start + cz] = 1.0
        new_c5 = compute_c5_fft(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / max(T, 1e-12))):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1.0
            b[start + cz] = 0.0
    return best_b, best_c5

def targeted_local_search(h, n, max_rounds, start_time, budget_s):
    b = (h >= 0.5).astype(float)
    target = n // 2
    s = np.sum(b)
    while int(s) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        idx = ones[rng.randint(len(ones))]
        b[idx] = 0
    s = np.sum(b)
    while int(s) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        idx = zeros[rng.randint(len(zeros))]
        b[idx] = 1
    best_c5 = compute_c5_fft(b)
    best_b = b.copy()
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(b, 1.0 - b, mode='full')
        mc = np.max(corr)
        lag_idx = np.argmax(corr)
        m = lag_idx - (n - 1)
        improved = False
        n_try = min(4000, max(500, n * 4))
        for _ in range(n_try):
            if time.time() - start_time > budget_s - 10:
                break
            a = np.random.randint(0, n)
            c2 = np.random.randint(0, n)
            if b[a] == b[c2]:
                continue
            b[a] = 1.0 - b[a]
            b[c2] = 1.0 - b[c2]
            new_c5 = compute_c5_fft(b)
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_b = b.copy()
                improved = True
            else:
                b[a] = 1.0 - b[a]
                b[c2] = 1.0 - b[c2]
        if not improved:
            break
    return best_b, best_c5

def multi_temp_slsqp(h, n, start_time, budget_s):
    temps = [50.0, 200.0, 800.0, 3000.0, 10000.0, 40000.0]
    best = h.copy()
    best_c = compute_c5(h)
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        try:
            x_res = slsqp_optimize(best, n, T, 120, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c:
                best_c = c5_res
                best = make_feasible(x_res, n)
        except Exception:
            pass
    return best, best_c

def fourier_perturb(h, n, rng, n_modes=4, amp=0.04):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def random_binary(n, rng):
    b = rng.randint(0, 2, size=n).astype(float)
    target = n // 2
    s = int(np.sum(b))
    if s > target:
        ones = np.where(b == 1)[0]
        excess = s - target
        idx = rng.choice(ones, size=excess, replace=False)
        b[idx] = 0
    elif s < target:
        zeros = np.where(b == 0)[0]
        deficit = target - s
        idx = rng.choice(zeros, size=deficit, replace=False)
        b[idx] = 1
    return b
