- Minimize C5_bound subject to these K quadratic constraints + sum(h) = n/2 + 0<=h<=1
- This is a QCQP (quadratically constrained quadratic program)
- CVXPY with a QCQP solver might handle this

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    top_k = min(80, len(corr))
    idx = np.argsort(corr)[::-1][:top_k]
    top_vals = corr[idx]
    logsum = np.log(np.sum(np.exp(T * (top_vals - mc)))) / T
    return (mc + logsum) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s, n_trial=9):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 5:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            old = h[j]
            if old <= 1e-10:
                cands = [1.0, 0.05, 0.1, 0.2]
            elif old >= 1.0 - 1e-10:
                cands = [0.0, 0.9, 0.95, 0.8]
            else:
                cands = [0.0, 1.0]
                cands += list(np.linspace(0.0, 1.0, n_trial))
                if old > 0.1:
                    cands.append(old - 0.01)
                if old < 0.9:
                    cands.append(old + 0.01)
                cands = sorted(set(max(0.0, min(1.0, c)) for c in cands))
            best_c = best_c5
            best_val = old
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def pairwise_cd(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    rng_local = np.random.RandomState(12345)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 5:
            break
        improved = False
        for _ in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            i, j = rng_local.choice(n, size=2, replace=False)
            if h[i] <= 0 and h[j] <= 0:
                continue
            if h[i] >= 1 and h[j] >= 1:
                continue
            s = h[i] + h[j]
            cands = np.linspace(0.0, s, min(13, max(7, int(s * 10) + 1)))
            cands = np.clip(cands, 0, 1)
            cands = np.unique(cands)
            best_c = best_c5
            best_vi = h[i]
            for vi in cands:
                vj = s - vi
                if vj < -1e-12 or vj > 1 + 1e-12:
                    continue
                trial = h.copy()
                trial[i] = max(0.0, vi)
                trial[j] = max(0.0, min(1.0, vj))
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_vi = max(0.0, min(1.0, vi))
            if best_vi != h[i]:
                h[i] = best_vi
                h[j] = max(0.0, min(1.0, s - best_vi))
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def spsa_optimize(h, n, start_time, budget_s, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    x = h.copy()
    gamma = 0.04
    alpha = 0.6
    c = 0.008
    steps = 0
    while time.time() - start_time < budget_s:
        steps += 1
        d = rng.choice([-1.0, 1.0], size=n)
        x_plus = np.clip(x + c * d, 0, 1)
        x_plus = make_feasible(x_plus, n)
        x_minus = np.clip(x - c * d, 0, 1)
        x_minus = make_feasible(x_minus, n)
        f_plus = compute_c5(x_plus)
        f_minus = compute_c5(x_minus)
        g = (f_plus - f_minus) / (2.0 * c)
        a = gamma / (1 + steps / 200) ** alpha
        x = x - a * g * d
        x = np.clip(x, 0, 1)
        x = make_feasible(x, n)
        fx = compute_c5(x)
        if fx < best_c5:
            best_c5 = fx
            best_h = x.copy()
    return best_h, best_c5

def binary_sa(h, n, time_limit, start_time, rng):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    T0, T1 = 0.005, 0.00001
    while time.time() - start_time < time_limit:
        frac = min(1.0, (time.time() - start_time) / max(time_limit - start_time, 1))
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5
