def project_onto_constraint(y, n, target=None):
    """Project y onto {h: 0<=h<=1, sum(h)=target}"""
    if target is None:
        target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 1.0  # range for lambda
    # Actually, y_i - lambda in [0,1] means lambda in [y_i - 1, y_i]
    # So lambda in [min(y) - 1, max(y)]
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0
    
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid  # need to increase lambda (subtract more)
        else:
            hi = mid
    return np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)

    if target is None:
        target = n / 2.0
    lo = np.min(y) - 1.0
    hi = np.max(y) + 1.0

        if np.sum(h) > target:

        else:

def compute_c5(h):

def smooth_max_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full')
    out_max = np.max(out)
    exp_out = np.exp(T * (out - out_max))
    sum_exp = np.sum(exp_out)
    val = (out_max + np.log(sum_exp) / T) * dx
    weights = exp_out / sum_exp
    # Gradient
    w = weights  # shape (2n-1,)
    # term1[j] = correlate(h, w, 'full')[3n-3-j]
    corr_full = np.correlate(h, w, 'full')  # length n + (2n-1) - 1 = 3n-2
    term1 = corr_full[2*n-2 : 3*n-2][::-1]  # shape (n,)
    # term2[j] = convolve(w, h, 'full')[j+1]
    conv_full = np.convolve(w, h, 'full')  # length (2n-1) + n - 1 = 3n-2
    term2 = conv_full[1:n+1]  # shape (n,)
    grad = dx * (1.0 - term1 - term2)
    return val, grad

def fista(h_init, n, T, max_iters, lr=None):
    dx = 2.0 / n
    h = h_init.copy()
    y = h.copy()
    t_k = 1.0
    # Lipschitz estimate: need L such that ||grad(f(x)) - grad(f(y))|| <= L||x-y||
    # For our function, L is proportional to T * (2n-1) * dx or so
    # Let's estimate: each out[k] has at most n terms, each term contributes at most 1 to the gradient
    # The gradient is dx * (1 - term1 - term2), and term1, term2 are bounded by 1
    # So ||grad|| <= dx * 3 = 6/n
    # Lipschitz constant is harder to bound. Let's use adaptive step size.
    
    if lr is None:

        # The Hessian of out[k] has entries at most 1, so ||Hessian|| <= 2n roughly
        # With the smooth max, L ~ T * 2n * dx
        lr = 1.0 / (T * n * dx * 2)  # conservative
    
    best_val = smooth_max_and_grad(h, T)[0]
    best_h = h.copy()
    
    for it in range(max_iters):
        val, grad = smooth_max_and_grad(y, T)
        
        # Gradient step
        h_new = y - lr * grad
        
        # Backtracking line search
        while True:
            h_proj = project_constraint(h_new, n)
            val_proj, _ = smooth_max_and_grad(h_proj, T)
            if val_proj <= val + 0.5 * lr * np.dot(grad, h_proj - y):
                break
            lr *= 0.5
            h_new = y - lr * grad
        
        # FISTA update
        t_next = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        y = h_proj + (t_k - 1.0) / t_next * (h_proj - h)
        h = h_proj
        t_k = t_next
        
        val, _ = smooth_max_and_grad(h, T)
        if val < best_val:
            best_val = val
            best_h = h.copy()
    
    return best_h, best_val

def project_constraint(y, n, target=None):
    if target is None:
        target = n / 2.0
    lo = float(np.min(y)) - 1.0
    hi = float(np.max(y)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        h = np.clip(y - mid, 0.0, 1.0)
        if np.sum(h) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    out = np.correlate(h, 1.0 - h, mode='full')
    out_max = np.max(out)
    exp_out = np.exp(T * (out - out_max))
    sum_exp = np.sum(exp_out)
    val = (out_max + np.log(sum_exp) / T) * dx
    weights = exp_out / sum_exp
    w = weights
    corr_full = np.correlate(h, w, 'full')
    term1 = corr_full[2 * n - 2: 3 * n - 2][::-1]
    conv_full = np.convolve(w, h, 'full')
    term2 = conv_full[1: n + 1]
    grad = dx * (1.0 - term1 - term2)
    return val, grad

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)

        else:

                best_alpha = alpha
        if best_c_alpha < best_c5 - 1e-15:
            best_h = (1 - best_alpha) * best_h + best_alpha * h_lp
            best_h = project_constraint(best_h, n)
            best_c5 = best_c_alpha

    # Phase 8: Try upsample if time allows
    if remaining() > 200:
        n_up = n * 2
        h_up = upsample_h(best_h, n_up)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.002:
            for T_val in [200.0, 800.0]:
                if remaining() < 50:
                    break
                step_up = 1.0 / (T_val * 1.5)
                budget_up = min(80.0, remaining() * 0.25)
                h_up_new, c5_up_new = fista_optimize(h_up, T_val, 3000, step_up, budget_up)
                if c5_up_new < c5_up:
                    h_up = h_up_new
                    c5_up = c5_up_new
            # LP at higher res
            for lp_i in range(8):
                if remaining() < 40:
                    break
                h_lp2, _ = linearized_lp_refine(h_up, K=10)
                if h_lp2 is None:
                    break
                h_lp2 = project_constraint(h_lp2, n_up)
                best_a = 0.2
                best_ca = compute_c5(h_up)
                for a in [0.1, 0.2, 0.3, 0.5]:
                    ht = (1 - a) * h_up + a * h_lp2
                    ht = project_constraint(ht, n_up)
                    ct = compute_c5(ht)
                    if ct < best_ca:
                        best_ca = ct
                        best_a = a
                if best_ca < compute_c5(h_up) - 1e-15:
                    h_up = (1 - best_a) * h_up + best_a * h_lp2
                    h_up = project_constraint(h_up, n_up)
            c5_up_final = compute_c5(h_up)
            if c5_up_final < best_c5:
                best_c5 = c5_up_final
                best_h = h_up
                n = n_up

    # Phase 9: Ultra-fine coordinate descent
    if remaining() > 20:
        cd_time2 = min(remaining() * 0.4, 100.0)
        best_h = coord_descent(best_h, len(best_h), max_passes=3, time_limit=cd_time2)
        c5_final_cd = compute_c5(best_h)
        if c5_final_cd < best_c5:
            best_c5 = c5_final_cd

    # Phase 10: Binary snap for near-binary values
    if remaining() > 10:
        n_f = len(best_h)
        changed = False
        for j in range(n_f):
            if remaining() < 5:
                break
            for bv in [0.0, 1.0]:
                if abs(best_h[j] - bv) > 0.05:
                    continue
                trial = best_h.copy()
                trial[j] = bv
                trial = project_constraint(trial, n_f)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial
                    changed = True
        if not changed:
            best_c5 = compute_c5(best_h)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
