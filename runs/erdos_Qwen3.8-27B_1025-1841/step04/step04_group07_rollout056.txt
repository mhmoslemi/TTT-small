def fft_correlate_fast(a, b):
    """Fast full cross-correlation using FFT."""
    n = len(a)
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    A = np.fft.rfft(a, N)
    B_rev = b[::-1]
    B = np.fft.rfft(B_rev, N)
    corr = np.fft.irfft(A * B, N)[:size]
    return corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = fft_correlate_fast(h, 1.0 - h)
    return float(np.max(corr) * dx)

def project_to_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = h.copy()
    target = n / 2.0
    # Bisection on alpha
    lo, hi = -max(h.max(), 0) - 1, 1 - min(h.min(), 1) + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0, 1)
        s = h_proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix any remaining discrepancy
    diff = target - h_proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h_proj < 1.0 - 1e-12)[0]
        else:
            idx = np.where(h_proj > 1e-12)[0]
        if len(idx) > 0:
            adj = diff / len(idx)
            h_proj[idx] = np.clip(h_proj[idx] + adj, 0, 1)
    return h_proj

def active_gradient(h, corr, n, top_k=None):
    """Compute weighted subgradient of max(corr) w.r.t. h."""
    size = len(corr)
    if top_k is None:
        top_k = max(5, size // 10)
    
    # Get top-k correlations
    sorted_idx = np.argpartition(corr, -top_k)[-top_k:]
    vals = corr[sorted_idx]
    
    # Softmax weights
    T = max(1.0, top_k / (vals.max() - vals.min() + 1e-12) * 2)
    weights = np.exp(T * (vals - vals.max()))
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for r, w in zip(sorted_idx, weights):
        if w < 1e-10:
            continue
        s = int(r) - n + 1
        # (1-h[j-s]) * I(0<=j-s<n)
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:])
        else:
            grad[-s:] += w * (1.0 - h[:n+s])
        # -h[j+s] * I(0<=j+s<n)
        if s >= 0:
            grad[s:] -= w * h[:n-s]
        else:
            grad[:n+s] -= w * h[-s:]
    return grad

    def obj(x):
        if time.time() - t_start > max_time:
            raise RuntimeError("timeout")
        corr = fft_correlate_fast(x, 1.0 - x)
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -50, 50)
        val = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        penalty = mu * (x.sum() - n / 2.0) ** 2
        return val + penalty
    try:
        res = minimize(obj, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-14, 'gtol': 1e-7, 'maxfun': 400})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h0, n)

def binary_local_search(h_bin, n, rng, max_rounds=40, tries=3000):
    h = h_bin.copy().astype(float)
    cur = compute_c5_fast(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        nt = min(tries, max(1, len(ones) * len(zeros)))
        for _ in range(nt):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5_fast(h)
            if c < cur - 1e-14:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def coord_polish(h0, n, max_rounds=6, steps=None):
    if steps is None:
        steps = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5_fast(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            for d in steps:
                nv = h[j] + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                c = compute_c5_fast(trial)
                if c < cur - 1e-14:
                    h = make_feasible(trial, n)
                    cur = compute_c5_fast(h)
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def pair_swap_local_search(h0, n, rng, max_rounds=30, tries=5000):
    h = h0.copy().astype(float)
    cur = compute_c5_fast(h)
    for rnd in range(max_rounds):
        improved = False
        for _ in range(tries):
            i, j = rng.randint(n), rng.randint(n)
            if i == j:
                continue
            h[i], h[j] = h[j], h[i]
            h[i] = np.clip(h[i], 0, 1)
            h[j] = np.clip(h[j], 0, 1)
            c = compute_c5_fast(h)
            if c < cur - 1e-14:
                cur = c
                improved = True
            else:
                h[i], h[j] = h[j], h[i]
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.zeros(640)
        h0[::2] = 1.0

    best_c5 = compute_c5_fast(h0)
    best_h = h0.copy()
    best_n = len(h0)

    n1 = 2048
    h1 = upsample_h(best_h, n1)
    c1 = compute_c5_fast(h1)
    if c1 < best_c5:
        best_c5, best_h, best_n = c1, h1.copy(), n1

    if rem() > 100:
        tl = min(rem() * 0.25, 250)
        h_adam, c_adam = adam_optimize(h1, n1, max_time=tl, lr=0.02, top_k=50)
        if c_adam < best_c5:
            best_c5, best_h, best_n = c_adam, h_adam.copy(), n1
            h1 = h_adam.copy()

    for T in [100, 500, 2000]:
        if rem() < 30:
            break
        h_opt = lbgfb_smooth(h1, n1, T, 500.0, 250, rem() - 5)
        c_opt = compute_c5_fast(h_opt)
        if c_opt < best_c5:
            best_c5, best_h, best_n = c_opt, h_opt.copy(), n1
            h1 = h_opt.copy()

    if rem() > 80:
        tl = min(rem() * 0.2, 180)
        h_adam2, c_adam2 = adam_optimize(h1, n1, max_time=tl, lr=0.008, top_k=20, beta1=0.95)
        if c_adam2 < best_c5:
            best_c5, best_h, best_n = c_adam2, h_adam2.copy(), n1
            h1 = h_adam2.copy()

    for T in [200, 1000]:
        if rem() < 30:
            break
        h_opt = lbgfb_smooth(h1, n1, T, 800.0, 300, rem() - 5)
        c_opt = compute_c5_fast(h_opt)
        if c_opt < best_c5:
            best_c5, best_h, best_n = c_opt, h_opt.copy(), n1
            h1 = h_opt.copy()

    n2 = 4096
    if rem() > 100:
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5_fast(h2)
        if c2 < best_c5:
            best_c5, best_h, best_n = c2, h2.copy(), n2
        for T in [200, 800]:
            if rem() < 30:
                break
            h_opt = lbgfb_smooth(h2, n2, T, 300.0, 150, rem() - 5)
            c_opt = compute_c5_fast(h_opt)
            if c_opt < best_c5:
                best_c5, best_h, best_n = c_opt, h_opt.copy(), n2
                h2 = h_opt.copy()
        if rem() > 50:
            tl = min(rem() * 0.4, 120)
            h_adam3, c_adam3 = adam_optimize(h2, n2, max_time=tl, lr=0.01, top_k=30)
            if c_adam3 < best_c5:
                best_c5, best_h, best_n = c_adam3, h_adam3.copy(), n2

    if rem() > 50:
        n_bin = 1024
        h_b = upsample_h(best_h, n_bin) if n_bin < best_n else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        h_bs, c_bs = binary_local_search(h_bin, n_bin, rng, max_rounds=30, tries=4000)
        if c_bs < best_c5:
            best_c5, best_h, best_n = c_bs, h_bs.copy(), n_bin

    if rem() > 30:
        n_cd = min(best_n, 2048)
        h_cd = best_h.copy() if best_n == n_cd else upsample_h(best_h, n_cd)
        h_cd, c_cd = coord_polish(h_cd, n_cd, max_rounds=5)
        if c_cd < best_c5:
            best_c5, best_h, best_n = c_cd, h_cd.copy(), n_cd

    if rem() > 20:
        n_ps = min(best_n, 2048)
        h_ps = best_h.copy() if best_n == n_ps else upsample_h(best_h, n_ps)
        h_ps, c_ps = pair_swap_local_search(h_ps, n_ps, rng, max_rounds=15, tries=3000)
        if c_ps < best_c5:
            best_c5, best_h, best_n = c_ps, h_ps.copy(), n_ps

    if rem() > 15:
        n_f = best_n
        h_f = best_h.copy()
        for T in [500, 3000]:
            if rem() < 8:
                break
            h_opt = lbgfb_smooth(h_f, n_f, T, 1500.0, 200, rem() - 3)
            c_opt = compute_c5_fast(h_opt)
            if c_opt < best_c5:
                best_c5, best_h, best_n = c_opt, h_opt.copy(), n_f
                h_f = h_opt.copy()

    best_h = make_feasible(best_h, best_n)
    final_c5 = compute_c5_fast(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, best_n
