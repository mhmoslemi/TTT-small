```python
    """p-norm smooth max approximation"""
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Use top-k elements for efficiency
    k = max(1, n // 4)
    topk = np.partition(corr, -k)[-k:]
    val = np.sum(topk ** p) ** (1.0/p)
    return val * dx

def slsqp_obj(x, n, T, dx, target_lag=None, lag_weight=1.0):
    """Log-sum-exp with optional worst-lag targeting"""
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    obj = mc + np.log(np.sum(np.exp(T * shifted))) / T
    
    # Add penalty for specific target lag
    if target_lag is not None and lag_weight > 1.0:
        # corr indices: 0 to 2n-2, lag 0 is at index n-1
        lag_idx = n - 1 + target_lag  # need to handle range
        if 0 <= lag_idx < len(corr):
            obj += (lag_weight - 1.0) * max(0, corr[lag_idx] - mc)
    
    return obj * dx
```

```python
def c5_delta(h, n, j, delta, corr, max_corr, top_k_lags):
    """Estimate change in C5 when h[j] changes by delta.
    corr: current full correlation array
    max_corr: current max of corr
    top_k_lags: indices of top K lags
    """
    # Compute delta corr for all k efficiently
    # h[j] changes by delta
    # For lag k (0-indexed in full output, length 2n-1):
    # corr[k] = sum_i h[i] * (1-h[i+k-n+1])
    # 
    # Terms involving j:
    # 1. i = j: needs 0 <= j+k-n+1 < n, i.e., n-1-j <= k <= n-1+j
    #    contribution: delta * (1-h[j+k-n+1])
    # 2. i+k-n+1 = j, i.e., i = j-k+n-1: needs 0 <= j-k+n-1 < n, i.e., j-n+1 <= k <= j+n-1
    #    contribution: -delta * h[j-k+n+1-1] = -delta * h[j-k+n-1]
    
    # This is O(n) to compute the full delta_corr
    # But we can shortcut: only check if max changes
    
    k_lo1 = max(0, n - 1 - j)
    k_hi1 = min(2*n-2, n - 1 + j)
    k_lo2 = max(0, j - n + 1)  
    k_hi2 = min(2*n-2, j + n - 1)
    
    # For all k in [k_lo1, k_hi1]: delta_corr[k] += delta * (1 - h[j+k-n+1])
    # For all k in [k_lo2, k_hi2]: delta_corr[k] += -delta * h[j-k+n-1]
    # (with overlap region having both contributions)
    
    # Quick check: will the current max decrease?
    k_max = np.argmax(corr)
    in_range1 = k_lo1 <= k_max <= k_hi1
    in_range2 = k_lo2 <= k_max <= k_hi2
    
    delta_max = 0.0
    if in_range1:
        idx = j + k_max - n + 1
        if 0 <= idx < n:
            delta_max += delta * (1 - h[idx])
    if in_range2:
        idx2 = j - k_max + n - 1
        if 0 <= idx2 < n:
            delta_max += -delta * h[idx2]
    
    new_max_estimate = max_corr + delta_max
    
    # Also check: could another lag surpass?
    # For the top K lags, check if they increase more
    for k in top_k_lags:
        d_k = 0.0
        if k_lo1 <= k <= k_hi1:
            idx = j + k - n + 1
            if 0 <= idx < n:
                d_k += delta * (1 - h[idx])
        if k_lo2 <= k <= k_hi2:
            idx2 = j - k + n - 1
            if 0 <= idx2 < n:
                d_k += -delta * h[idx2]
        new_max_estimate = max(new_max_estimate, corr[k] + d_k)
    
    return new_max_estimate
```

```python
    old_val = h[j]

    if abs(delta) < 1e-15:
        return

    # Contribution 1: terms where i=j (first factor)
    #   Valid for k where 0 <= j+k-n+1 < n, i.e., n-1-j <= k <= n-1+j
    #   Change: delta * (1 - h[j+k-n+1])
```
