    res = minimize(slsqp_obj, h.copy(), jac=slsqp_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def fourier_perturb(h, n, rng, n_modes=4, amp=0.03):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def random_binary(n, rng):
    b = np.zeros(n)
    idx = rng.choice(n, size=n//2, replace=False)
    b[idx] = 1.0
    return b

def gradient_step(h, n, T, alpha, start_time, time_limit):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exps = np.exp(T * shifted)
    w = exps / np.sum(exps)
    neg_grad = np.zeros(n)
    for d in range(n):
        k = n - 1 + d
        wi = w[k]
        if wi < 1e-14:
            continue
        if d == 0:
            neg_grad[:] += wi * (1.0 - 2.0 * h)
        else:
            neg_grad[d:] += wi * (1.0 - h[:-d])
            neg_grad[:n-d] += -wi * h[d:]
    for d in range(1, n):
        k = n - 1 - d
        wi = w[k]
        if wi < 1e-14:
            continue
        neg_grad[:n-d] += wi * (1.0 - h[d:])
        neg_grad[d:] += -wi * h[:-d]
    h_new = h - alpha * dx * neg_grad
    h_new = make_feasible(h_new, n)
    return h_new

def projected_grad_descent(h, n, max_iters, T, start_time, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_cur = h.copy()
    alpha = 0.02
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        h_cur = gradient_step(h_cur, n, T, alpha, start_time, time_limit)
        c = compute_c5(h_cur)
        if c < best_c5:
            best_c5 = c
            best_h = h_cur.copy()
            alpha = min(alpha * 1.1, 0.05)
        else:
            alpha *= 0.7
            h_cur = make_feasible(0.5 * h_cur + 0.5 * best_h, n)
            if alpha < 0.001:
                break
    return best_h, best_c5

def worst_shift_swap(b, n, rng, max_iters, time_limit, start_time):
    target = n // 2
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    dx = 2.0 / n
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(b, 1.0 - b, mode='full')
        worst_k = np.argmax(corr)
        d = worst_k - (n - 1)
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        best_improve = 1e-30
        best_swap = None
        n_tries = min(20, len(ones), len(zeros))
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            b[i] = 0
            b[j] = 1
            nc = np.max(np.correlate(b, 1.0 - b, mode='full')) * dx
            b[i] = 1
            b[j] = 0
            if curr_c5 - nc > best_improve:
                best_improve = curr_c5 - nc
                best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            b[i] = 0
            b[j] = 1
            curr_c5 = compute_c5(b)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_b = b.copy()
    return best_b, best_c5

def block_swap_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, 10)
        start = rng.randint(0, max(1, n - bsize))
        end = start + bsize
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def coord_descent_fast(h, n, max_rounds, start_time, time_limit):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = time_limit - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = time_limit - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.005), min(1.0, old + 0.005)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def elapsed():
        return time.time() - start_time

    # Phase 1: SLSQP from parent with escalating T
    for T in [50000, 200000, 800000]:
        rem = budget_s - elapsed()
        if rem < 60:
            break
        try:
            x_res = slsqp_optimize(best_h, n0, T, 250, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 2: Projected gradient refinement
    rem = budget_s - elapsed()
    if rem > 120:
        tl = time.time() + min(rem - 20, 100)
        best_h, best_c5 = projected_grad_descent(best_h, n0, 5000, 500000, start_time, tl)

    # Phase 3: Multiple perturbations + SLSQP
    n_perturb = 6
    for i in range(n_perturb):
        rem = budget_s - elapsed()
        if rem < 80:
            break
        h_p = fourier_perturb(best_h, n0, rng, n_modes=3 + i, amp=0.02 + 0.01*i)
        try:
            x_res = slsqp_optimize(h_p, n0, 300000, 200, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 4: Random binary starts
    for i in range(4):
        rem = budget_s - elapsed()
        if rem < 100:
            break
        b_rand = random_binary(n0, rng)
        try:
            x_res = slsqp_optimize(b_rand, n0, 100000, 200, 1e-13)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 5: Worst-shift binary swap search
    rem = budget_s - elapsed()
    if rem > 120:
        tl = min(rem - 30, 100)
        b_cur = (best_h >= 0.5).astype(float)
        b_res, c5_res = worst_shift_swap(b_cur, n0, rng, 50000, tl, start_time)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()
            rem = budget_s - elapsed()
            if rem > 60:
                try:
                    x_res = slsqp_optimize(best_h, n0, 300000, 200, 1e-14)
                    c5_r = compute_c5(x_res)
                    if c5_r < best_c5:
                        best_c5 = c5_r
                        best_h = make_feasible(x_res, n0)
                except Exception:
                    pass

    # Phase 6: Block swap SA
    rem = budget_s - elapsed()
    if rem > 120:
        tl = min(rem - 30, 100)
        b_res, c5_res = block_swap_sa(best_h, n0, 40000, 0.004, 0.00001, rng, tl, start_time)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = b_res.copy()
            rem = budget_s - elapsed()
            if rem > 50:
                try:
                    x_res = slsqp_optimize(best_h, n0, 500000, 200, 1e-14)
                    c5_r = compute_c5(x_res)
                    if c5_r < best_c5:
                        best_c5 = c5_r
                        best_h = make_feasible(x_res, n0)
                except Exception:
                    pass

    # Phase 7: Try different resolution
    rem = budget_s - elapsed()
    if rem > 300:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0, endpoint=False) + 1.0/n0
        new_x = np.linspace(0.0, 2.0, n1, endpoint=False) + 1.0/n1
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
        rem2 = budget_s - elapsed()
        if rem2 > 100 and len(best_h) == n1:
            try:
                x_res = slsqp_optimize(best_h, n1, 300000, 200, 1e-14)
                c5_r = compute_c5(x_res)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(x_res, n1)
            except Exception:
                pass
        # Try from parent at new resolution
        rem2 = budget_s - elapsed()
        if rem2 > 100:
            h_up2 = np.interp(new_x, old_x, h_init)
            h_up2 = make_feasible(h_up2, n1)
            try:
                x_res = slsqp_optimize(h_up2, n1, 200000, 150, 1e-13)
                c5_r = compute_c5(x_res)
                if c5_r < best_c5:
                    best_c5 = c5_r
                    best_h = make_feasible(x_res, n1)
            except Exception:
                pass

    # Phase 8: Final SLSQP push
    rem = budget_s - elapsed()
    if rem > 40:
        n_pts = len(best_h)
        for T in [1000000, 500000]:
            rem = budget_s - elapsed()
            if rem < 30:
                break
            try:
                x_res = slsqp_optimize(best_h, n_pts, T, 200, 1e-15)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(x_res, n_pts)
            except Exception:
                pass

    # Phase 9: Coordinate descent
    rem = budget_s - elapsed()
    if rem > 30:
        n_pts = len(best_h)
        tl = time.time() + min(rem - 10, 40)
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n_pts, 15, start_time, tl)

    # Final check
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
