```python
        #   first part: (1-h[j-d]) for j in [max(0,d), min(n, n+d)) 
        #   second part: -h[j+d] for j in [max(0,-d), min(n, n-d))
        
        # Let's build the gradient matrix G where G[d_idx, j] = ∂c(offset_d)/∂h[j]
        # offset_d = d_idx - (n-1)
        
        G = np.zeros((2*n-1, n))
        for d_idx in range(2*n-1):
            d = d_idx - (n + 1)  # offset
            # First part: (1-h[j-d]) for valid j
            j_start = max(0, d)
            j_end = min(n, n + d)
            if j_start < j_end:
                G[d_idx, j_start:j_end] = 1.0 - h[j_start - d : j_end - d]
            # Second part: -h[j+d] for valid j
            j_start2 = max(0, -d)
            j_end2 = min(n, n - d)
            if j_start2 < j_end2:
                G[d_idx, j_start2:j_end2] -= h[j_start2 + d : j_end2 + d]
        
        # LP:
        # min t
        # s.t. G @ h_new - t <= b  where b[d] = -c_vals[d] + G[d] @ h
        #      sum(h_new) = n/2
        #      0 <= h_new <= 1
        
        b = -c_vals + G @ h
        
        constraints = [
            G @ h_var - t_var <= b,
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1
        ]
        obj = cp.Minimize(t_var)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.GLOP)  # or another solver
        except:
            try:
                prob.solve(solver=cp.CBC)
            except:
                try:
                    prob.solve(solver=cp.GLPK)
                except:
                    break
        
        if prob.status not in ['optimal', 'optimal_inaccurate']:
            break
        
        h_new = h_var.value
        if h_new is None:
            break
        
        h_new = np.clip(h_new, 0, 1)
        # Ensure exact sum
        diff = n/2.0 - np.sum(h_new)
        if abs(diff) > 1e-10:
            # Adjust
            h_new = h_new + diff / n  # crude adjustment
            h_new = np.clip(h_new, 0, 1)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5 - 1e-12:
            h = h_new
        else:
            # Convergence
            break
    
    return h, compute_c5(h)
```
