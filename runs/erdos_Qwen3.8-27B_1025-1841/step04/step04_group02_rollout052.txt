from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    k_star = np.argmax(corr)
    # Compute subgradient at k_star
    # corr[k_star] = sum_i h[i] * (1-h[i + k_star - (n-1)])
    # (using the convention that the shift is k_star - (n-1))

    # d/dh[j] of sum_i h[i]*(1-h[i+shift]):

        if 0 <= jm < n:

    h_new = make_feasible(h_new, n)

import numpy as np
from scipy.optimize import minimize
import time

_SG_N = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def sa_binary(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    it = 0
    while it < max_iters:
        rem = budget_s - (time.time() - start_time)
        if rem < 10:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        r = rng.random()
        if r < 0.5:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0
        elif r < 0.8:
            bs = rng.randint(1, 7)
            s1 = rng.randint(0, n - bs)
            s2 = rng.randint(0, n - bs)
            if abs(s1 - s2) < bs:
                it += 1
                continue
            seg1 = b[s1:s1+bs].copy()
            seg2 = b[s2:s2+bs].copy()
            ones1 = np.where(seg1 == 1)[0]
            if len(ones1) == 0 or len(ones1) == bs:
                it += 1
                continue
            b[s1:s1+bs] = seg2
            b[s2:s2+bs] = seg1
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[s1:s1+bs] = seg1
                b[s2:s2+bs] = seg2
        else:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i1 = ones[rng.randint(len(ones))]
            i2 = ones[rng.randint(len(ones))]
            j1 = zeros[rng.randint(len(zeros))]
            j2 = zeros[rng.randint(len(zeros))]
            if i1 == i2 or j1 == j2:
                it += 1
                continue
            b[i1] = 0.0; b[i2] = 0.0; b[j1] = 1.0; b[j2] = 1.0
            new_c5 = compute_c5(b)
            if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):

                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i1] = 1.0; b[i2] = 1.0; b[j1] = 0.0; b[j2] = 0.0
        it += 1
    return best_b, best_c5

def subgradient_descend(h, n, max_iters, start_time, budget_s):
    h = h.copy().astype(float)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr0 = 0.02
    for it in range(max_iters):
        rem = budget_s - (time.time() - start_time)
        if rem < 5:
            break
        one_minus_h = 1.0 - h
        corr = np.correlate(h, one_minus_h, mode='full')
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        grad = np.zeros(n)
        if shift >= 0:
            grad[:n-shift] += (1.0 - h[shift:n])
            grad[shift:] -= h[:n-shift]
        else:
            s2 = -shift
            grad[s2:] += (1.0 - h[:n-s2])
            grad[:n-s2] -= h[s2:]
        lr = lr0 * (1.0 / (1.0 + it * 0.001))
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def obj_slsqp(x):
    n = _SG_N
    T = 200.0
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _SG_N
    _SG_N = n
    global _SG_T
    _SG_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

_SG_T = 200.0

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        rem = budget_s - (time.time() - start_time)
        if rem < 5:
            break
        for j in range(n):
            rem = budget_s - (time.time() - start_time)
            if rem < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.02), max(0.0, old - 0.005),
                     min(1.0, old + 0.005), min(1.0, old + 0.02)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def golden_ratio_pattern(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    half = n // 2
    for i in range(half):
        pos = int((i * phi * phi) % n)
        h[pos] = 1.0
    return make_feasible(h, n)

def thue_morse_pattern(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = tm.astype(float)
    ones = h.sum()
    if int(ones) != n // 2:
        h = to_binary(h, n)
    return h
