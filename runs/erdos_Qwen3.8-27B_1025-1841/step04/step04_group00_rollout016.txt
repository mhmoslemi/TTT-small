For s ≥ 0:
corr(s) = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])

∂corr(s)/∂h[j]:
- j appears in h[i] when i=j, valid if 0 ≤ j ≤ n-1-s: term is (1-h[j+s])
- j appears in h[i+s] when i=j-s, valid if 0 ≤ j-s ≤ n-1-s, i.e., s ≤ j ≤ n-1: term is -h[j-s]

So ∂corr(s)/∂h[j] = (1-h[j+s]) * 1[j ≤ n-1-s] - h[j-s] * 1[s ≤ j ≤ n-1]

For s < 0, let s' = -s > 0:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def vec_gradient_for_shift(h, s):
    """Vectorized gradient of corr(s) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        L = n - s
        if L > 0:
            grad[:L] += (1.0 - h[s:])
            grad[s:] -= h[:L]
    else:
        s_abs = -s
        L = n - s_abs
        if L > 0:
            grad[s_abs:] += (1.0 - h[:L])
            grad[:L] -= h[s_abs:]
    return grad

def smooth_obj_grad(h, T, top_m):
    """Compute smooth max objective and gradient using top_m shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    
    # Get top_m indices
    top_idx = np.argpartition(corr, -top_m)[-top_m:]
    corr_top = corr[top_idx]
    
    # Softmax weights
    c_max = np.max(corr_top)
    exps = np.exp(T * (corr_top - c_max))
    weights = exps / np.sum(exps)
    
    # Smooth max value
    smax = c_max + np.log(np.sum(exps)) / T
    obj = smax * dx
    
    # Gradient
    grad = np.zeros(n)
    for m in range(top_m):
        s = top_idx[m] - (n - 1)
        grad += weights[m] * vec_gradient_for_shift(h, s)
    grad *= dx
    
    return obj, grad

def pgd_smooth(h, n, budget_s, start_time, T_init=50.0, top_m=10, lr_init=0.01):
    """Projected gradient descent with smooth max, adaptive T and step size."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    T = T_init
    lr = lr_init
    it = 0
    no_improve = 0
    vel = np.zeros(n)
    momentum = 0.7
    
    while time.time() - start_time < budget_s:
        it += 1
        obj, grad = smooth_obj_grad(h, T, top_m)
        c5_exact = compute_c5(h)
        
        if c5_exact < best_c5 - 1e-14:
            best_c5 = c5_exact
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Adaptive T: increase over time for sharper max
        T = T_init * (1 + 0.1 * it)
        if T > 200:
            T = 200
        
        # Nesterov momentum step
        y = h - momentum * vel
        y_clipped = np.clip(y, 0, 1)
        obj_y, grad_y = smooth_obj_grad(y_clipped, T, top_m)
        vel = momentum * vel - lr * grad_y
        new_h = h + vel
        new_h = project_feasible(new_h, n)
        
        c5_new = compute_c5(new_h)
        if c5_new <= c5_exact + 1e-15:
            h = new_h
        else:
            # Backtrack
            new_h2 = project_feasible(h - lr * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_exact + 1e-15:
                h = new_h2
            else:
                lr *= 0.5
                vel *= 0.5
            if lr < 1e-8:
                break
        
        # Periodically increase lr
        if no_improve % 30 == 0 and lr < lr_init:
            lr = min(lr_init, lr * 1.5)
            T = max(T_init, T * 0.8)
        
        if no_improve > 100:
            break
        
        if time.time() - start_time > budget_s - 2:
            break
    
    h = project_feasible(h, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_h, best_c5 = h, c5_final
    return best_h, best_c5

        else:
            critical[-s:] = True
            critical[:n-(-s)] = True

    rng = np.random.RandomState(42)

    while improvement and time.time() - start_time < budget_s:
        improvement = False
        for _ in range(n):  # Try n random directions
            if time.time() - start_time > budget_s - 3:

            # Random subset of critical variables
            n_perturb = rng.randint(3, max(4, n // 20))
            idx = np.where(critical)[0]
            pert_idx = rng.choice(idx, size=min(n_perturb, len(idx)), replace=False)

            c5_trial = compute_c5(trial)
            if c5_trial < best_c5 - 1e-14:

    return best_h, best_c5

def coordinate_descent_fine(h, n, budget_s, start_time):
    """Fine coordinate descent on a subset of variables."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(best_h)
    top_idx = np.argsort(corr)[-8:]
    dx = 2.0 / n

    for k in top_idx:
        s = abs(k - (n - 1))
        # Variables around the shift boundary
        lo = max(0, s - 5)
        hi = min(n, n - s + 5)
        for j in range(lo, hi):
            if time.time() - start_time > budget_s - 2:
                return best_h, best_c5
            for delta in [0.001, -0.001, 0.005, -0.005, 0.01, -0.01]:
                trial = best_h.copy()
                trial[j] += delta
                trial = project_feasible(trial, n)
                c = compute_c5(trial)

                    best_c5 = c
    return best_h, best_c5

def upsample(h, n_new):
    """Upsample h to n_new points using linear interpolation."""
    old_x = np.linspace(0, 2, len(h), endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)  # periodic interpolation
    return h_new

subject to 0 ≤ h ≤ 1, sum h = n/2

This is a QP (quadratic programming) if the weights w_m are fixed! The objective is:
sum_m w_m * sum_i h[i]*(1-h[i+k_m])
= sum_m w_m * (sum_i h[i] - sum_i h[i]*h[i+k_m])

The first part (sum_i h[i]) is linear and constant (equals n/2).
The second part is quadratic: -sum_m w_m * sum_i h[i]*h[i+k_m]

So the objective is: const - sum_m w_m * sum_i h[i]*h[i+k_m]

This is a quadratic form: -h^T Q h where Q depends on the weights and shifts.
Maximizing this (which is what we want, since we're minimizing the negative) is a QP with the constraint set {0≤h≤1, sum h = n/2}.

We can solve this iteratively:
1. Compute current correlations
2. Compute softmax weights
3. Solve the QP with those weights
4. Repeat

This is an alternating optimization / majorization-minimization scheme!

The QP can be solved with CVXPY. Let me set it up:

minimize -sum_m w_m * sum_i h[i]*h[i+k_m]
subject to 0 ≤ h ≤ 1, sum h = n/2

In CVXPY:
- Objective: -sum_m w_m * (h * shift_k(m)(h))
  This is -h^T Q h / 2 (for symmetric Q) or we can write it as a quadratic form.

Actually, sum_i h[i]*h[i+k] for a fixed k can be written as h^T S_k h where S_k is a shift matrix. So the total objective is -h^T (sum_m w_m S_k(m)) h.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt == 0:
                break
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def vec_grad_shift(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        L = n - s
        if L > 0:
            grad[:L] += (1.0 - h[s:])
            grad[s:] -= h[:L]
    else:
        sa = -s
        L = n - sa
        if L > 0:
            grad[sa:] += (1.0 - h[:L])
            grad[:L] -= h[sa:]
    return grad

def smooth_obj_grad(h, T, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    top_idx = np.argpartition(corr, -top_m)[-top_m:]
    corr_top = corr[top_idx]
    c_max = np.max(corr_top)
    exps = np.exp(T * (corr_top - c_max))
    w = exps / np.sum(exps)
    smax = c_max + np.log(np.sum(exps)) / T
    obj = smax * dx
    grad = np.zeros(n)
    for m in range(top_m):
        s = top_idx[m] - (n - 1)
        grad += w[m] * vec_grad_shift(h, s)
    grad *= dx
    return obj, grad

def pgd_smooth(h, n, budget_s, start_time, T_init=30, T_max=150, top_m=10, lr_init=0.005):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    T = T_init
    lr = lr_init
    lr_max = lr_init * 2
    it = 0
    no_improve = 0
    vel = np.zeros(n)
    mu = 0.7
    while time.time() - start_time < budget_s:
        it += 1
        T = min(T_max, T_init + 0.5 * it)
        if it % 100 < 50:
            T = T_init + 0.3 * it
        corr = compute_corr(h)
        c5_exact = float(np.max(corr) * (2.0 / n))
        if c5_exact < best_c5 - 1e-15:
            best_c5 = c5_exact
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > 200:
            lr *= 0.7
            mu = max(0.5, mu - 0.05)
            T = max(T_init, T * 0.7)
            no_improve = 0
            if lr < 1e-9:
                break
        obj, grad = smooth_obj_grad(h, T, top_m)
        y = h - mu * vel
        y = np.clip(y, 0, 1)
        obj_y, grad_y = smooth_obj_grad(y, T, top_m)
        vel = mu * vel - lr * grad_y
        new_h = h + vel
        new_h = project_feasible(new_h, n)
        c5_new = compute_c5(new_h)
        if c5_new <= c5_exact + 1e-15:
            h = new_h
        else:
            new_h2 = project_feasible(h - lr * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_exact + 1e-15:
                h = new_h2
            else:
                lr *= 0.5
                vel *= 0.3
            if lr < 1e-9:
                break
        if no_improve % 50 == 0 and lr < lr_max:
            lr = min(lr_max, lr * 1.4)
    h = project_feasible(h, n)
    c5_f = compute_c5(h)
    if c5_f < best_c5:
        best_h, best_c5 = h, c5_f
    return best_h, best_c5

def targeted_descent(h, n, budget_s, start_time, top_m=10):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(123)
    it = 0
    while time.time() - start_time < budget_s:
        it += 1
        corr = compute_corr(best_h)
        top_idx = np.argsort(corr)[-top_m:]
        for _ in range(30):
            if time.time() - start_time > budget_s - 2:
                return best_h, best_c5
            n_p = rng.randint(2, max(3, n // 30))
            cands = []
            for k in top_idx:
                s = abs(k - (n - 1))
                lo = max(0, s - 3)
                hi = min(n, n - s + 3)
                cands.extend(range(lo, hi))
            if not cands:
                cands = list(range(n))
            pert_idx = rng.choice(cands, size=min(n_p, len(cands)), replace=False)
            scale = 0.005 + 0.01 * (1.0 - min(1.0, it / 100.0))
            delta = rng.randn(len(pert_idx)) * scale
            trial = best_h.copy()
            trial[pert_idx] += delta
            trial = project_feasible(trial, n)
            c5_t = compute_c5(trial)
            if c5_t < best_c5 - 1e-15:
                best_h = trial
                best_c5 = c5_t
    return best_h, best_c5

def coord_fine(h, n, budget_s, start_time, top_m=6):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    corr = compute_corr(best_h)
    top_idx = np.argsort(corr)[-top_m:]
    deltas = [0.001, -0.001, 0.003, -0.003, 0.008, -0.008]
    for pass_i in range(20):
        if time.time() - start_time > budget_s - 2:
            break
        corr = compute_corr(best_h)
        top_idx = np.argsort(corr)[-top_m:]
        for k in top_idx:
            s = abs(k - (n - 1))
            lo = max(0, s - 4)
            hi = min(n, n - s + 4)
            for j in range(lo, hi):
                if time.time() - start_time > budget_s - 2:
                    return best_h, best_c5
                for d in deltas:
                    trial = best_h.copy()
                    trial[j] += d
                    trial = project_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        best_h = trial
                        best_c5 = c
    return best_h, best_c5

def slsqp_smooth(h, n, budget_s, start_time, T=100.0, top_m=8):
    def obj_fn(x):
        return smooth_obj_grad(x, T, top_m)[0]
    def obj_grad(x):
        return smooth_obj_grad(x, T, top_m)[1]
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-14, 'iprint': 0})
        rx = project_feasible(res.x, n)
        c5_r = compute_c5(rx)
        if c5_r < compute_c5(h):
            return rx, c5_r
    except Exception:
        pass
    return h, compute_c5(h)

def upsample_fn(h, n_new):
    old_x = np.linspace(0, 2, len(h), endpoint=False)
    new_x = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return h_new

def make_initial_gauss(n, center, width, rng):
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.exp(-((x - center) ** 2) / (2 * width ** 2))
    h += 0.3 * np.exp(-((x - (center + 1.0) % 2.0) ** 2) / (2 * (width * 0.7) ** 2))
    h = h / np.sum(h) * (n / 2.0)
    h = project_feasible(h, n)
    return h
