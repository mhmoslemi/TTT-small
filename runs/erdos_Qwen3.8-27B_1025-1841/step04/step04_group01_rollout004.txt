    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Soft max
    shifted = corr - mc
    softmax = mc + np.log(np.sum(np.exp(T * shifted))) / T
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return softmax * dx + penalty

    top = corr[top_idx]

    shifted = top - mc

def single_shift_heuristic(h, n, k_star, steps=100):
    # For the worst shift, try to reduce its correlation
    # by adjusting h values
    shift = k_star - (n - 1)  # physical shift index
    for step in range(steps):
        # Find h[i] that most contributes to corr[k_star]
        # corr[k_star] = sum_i h[i] * (1-h[i+shift])
        # The "damage" at position i is h[i] * (1-h[i+shift])
        damage = np.zeros(n)
        for i in range(n):
            j = i + shift
            if 0 <= j < n:
                damage[i] = h[i] * (1 - h[j])
            else:
                damage[i] = 0
        
        # Also consider the reverse: h[i-shift] * (1-h[i])
        # Total contribution of h[i] to corr[k_star]:
        # From h[i] as first factor: h[i] * (1-h[i+shift])
        # From h[i] as second factor (i.e., (1-h[i]) in h[i-shift]*(1-h[i])):
        #   -h[i-shift] * h[i]
        
        total_contrib = np.zeros(n)
        for i in range(n):
            contrib = h[i] * (1 - 0)  # h[i] term, ignoring the h[i+shift] for now
            # Actually let me just compute the gradient of corr[k_star] w.r.t. h[i]
            # d/dh[i] sum_j h[j]*(1-h[j+shift])
            # = (1-h[i+shift]) - h[i-shift]  (for valid indices)
            grad = 0.0
            if 0 <= i + shift < n:
                grad += (1 - h[i + shift])
            if 0 <= i - shift < n:
                grad -= h[i - shift]
            total_contrib[i] = grad
        
        # Find the h[i] where decreasing h[i] most reduces corr[k_star]
        # (i.e., grad > 0, so decreasing h[i] decreases corr)
        # Or increasing h[i] where grad < 0
        if np.any(total_contrib > 0.1):
            i = int(np.argmax(total_contrib))
            h[i] = max(0, h[i] - 0.1)
        elif np.any(total_contrib < -0.1):
            i = int(np.argmin(total_contrib))
            h[i] = min(1, h[i] + 0.1)
        
        h = make_feasible(h, n)
    return h

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
