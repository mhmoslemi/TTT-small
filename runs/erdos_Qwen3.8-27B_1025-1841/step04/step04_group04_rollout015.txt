    mc = np.max(corr)

    exp_vals = np.exp(_OPT_T * shifted)
    log_sum = np.log(np.sum(exp_vals))
    return (mc + log_sum / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_b(h, j, n):
    s = np.arange(-(n - 1), n)
    idx1 = j + s
    idx2 = j - s
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    b = np.zeros(2 * n - 1)
    if np.any(valid1):
        b[valid1] += (1.0 - h[idx1[valid1]])
    if np.any(valid2):
        b[valid2] -= h[idx2[valid2]]
    return b

def coord_descent(h, n, max_sweeps=5000, stop_time=None):
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_c = np.max(corr) * dx
    stagnant = 0
    shift_arr = np.arange(-(n - 1), n)

    for sweep in range(max_sweeps):
        if stop_time is not None and time.time() > stop_time:
            break
        improved = False
        for j in range(n):
            idx1 = j + shift_arr
            idx2 = j - shift_arr
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            b = np.zeros(2 * n - 1)
            if np.any(valid1):
                b[valid1] = (1.0 - h[idx1[valid1]])
            if np.any(valid2):
                b[valid2] -= h[idx2[valid2]]
            a = corr - b * h[j]

            lo, hi = 0.0, 1.0
            for _ in range(20):
                mid = (lo + hi) * 0.5
                vals = a + b * mid
                mv = np.max(vals)
                close = vals >= mv - 1e-7
                bmin = np.min(b[close])
                bmax = np.max(b[close])
                if bmin > 1e-9:
                    hi = mid
                elif bmax < -1e-9:
                    lo = mid
                else:
                    break
            t_new = (lo + hi) * 0.5

            f_new = np.max(a + b * t_new)
            f_old = np.max(corr)
            if f_new < f_old - 1e-11:
                corr += (t_new - h[j]) * b
                h[j] = t_new
                improved = True

        target_sum = n / 2.0
        diff = target_sum - np.sum(h)
        if abs(diff) > 1e-11:
            if diff > 0:
                mask = h < 1.0 - 1e-12
            else:
                mask = h > 1e-12
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0, 1)
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float)

        c_now = np.max(corr) * dx
        if c_now < best_c - 1e-14:
            best_c = c_now
            stagnant = 0
        else:
            stagnant += 1
        if not improved:
            break
        if stagnant > 50:
            break

    return h, best_c

def binary_swap_search(h, n, max_iters=80000, stop_time=None, rng=None):
    if rng is None:
        rng = np.random.RandomState(0)
    dx = 2.0 / n
    h_bin = (h > 0.5).astype(float)
    ones = int(np.sum(h_bin))
    if ones > n // 2:
        one_idx = np.where(h_bin == 1)[0]
        rng.shuffle(one_idx)
        h_bin[one_idx[:ones - n // 2]] = 0
    elif ones < n // 2:
        zero_idx = np.where(h_bin == 0)[0]
        rng.shuffle(zero_idx)
        h_bin[zero_idx[:n // 2 - ones]] = 1

    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    best_c = np.max(corr) * dx

    for it in range(max_iters):
        if stop_time is not None and time.time() > stop_time:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if h_bin[i] == h_bin[j]:
            continue
        h_bin[i], h_bin[j] = h_bin[j], h_bin[i]
        new_corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
        new_c = np.max(new_corr) * dx
        if new_c < best_c - 1e-14:
            best_c = new_c
            corr = new_corr
        else:
            h_bin[i], h_bin[j] = h_bin[j], h_bin[i]

    return h_bin, best_c

def slsqp_opt(h, n, T, maxiter):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        hx = make_feasible(res.x, n)
        return hx, compute_c5(hx)
    except Exception:
        return h, compute_c5(h)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP convergence
    if remaining() > 200:
        h_new, c_new = slsqp_opt(best_h, n0, 80.0, 120)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    if remaining() > 180:
        h_new, c_new = slsqp_opt(best_h, n0, 300.0, 100)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 2: Coordinate descent (primary engine)
    stop_t = start_time + budget_s - 500
    h_new, c_new = coord_descent(best_h, n0, max_sweeps=8000, stop_time=stop_t)
    if c_new < best_c5:
        best_c5 = c_new
        best_h = h_new

    # Phase 3: Binary local search
    if remaining() > 300:
        stop_t2 = start_time + budget_s - 250
        h_bin, c_bin = binary_swap_search(best_h, n0, max_iters=100000,
                                          stop_time=stop_t2, rng=rng)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin

    # Phase 4: Coordinate descent from best (or binary)
    if remaining() > 200:
        stop_t3 = start_time + budget_s - 150
        h_new, c_new = coord_descent(best_h, n0, max_sweeps=6000, stop_time=stop_t3)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 5: SLSQP high-T polish
    if remaining() > 80:
        h_new, c_new = slsqp_opt(best_h, n0, 800.0, 80)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 6: Final coordinate descent
    if remaining() > 50:
        stop_t4 = start_time + budget_s - 20
        h_new, c_new = coord_descent(best_h, n0, max_sweeps=4000, stop_time=stop_t4)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 7: Higher resolution attempt
    if remaining() > 250:
        n_high = 1920
        h_high = upsample_h(best_h, n_high)
        c_high = compute_c5(h_high)
        if remaining() > 200:
            h_high, c_high = slsqp_opt(h_high, n_high, 100.0, 60)
        if remaining() > 100:
            stop_t5 = start_time + budget_s - 30
            h_high, c_high = coord_descent(h_high, n_high, max_sweeps=2000, stop_time=stop_t5)
        if c_high < best_c5:
            best_c5 = c_high
            best_h = h_high

    # Phase 8: Final tiny polish
    if remaining() > 20:
        h_new, c_new = coord_descent(best_h, len(best_h), max_sweeps=1000,
                                     stop_time=start_time + budget_s - 5)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
