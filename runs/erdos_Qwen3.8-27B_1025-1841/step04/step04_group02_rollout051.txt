def compute_grad(h, softmax_weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # For s from -(n-1) to (n-1):
    # grad[j] += w * (h[j+s] if 0<=j+s<n else 0 + h[j-s] if 0<=j-s<n else 0)
    for k in range(len(softmax_weights)):
        w = softmax_weights[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += 2.0 * w * h
        elif s > 0:
            # h[j+s]: j in [0, n-1-s]
            grad[:n-s] += w * h[s:]
            # h[j-s]: j in [s, n-1]
            grad[s:] += w * h[:n-s]
        else:
            sp = -s
            # h[j+s] = h[j-sp]: j in [sp, n-1]
            if n - sp > 0:
                grad[sp:] += w * h[:n-sp]
            # h[j-s] = h[j+sp]: j in [0, n-1-sp]
            if n - sp > 0:
                grad[:n-sp] += w * h[sp:]
    grad *= dx
    return grad

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    expo = np.exp(T * shifted)
    expo_sum = np.sum(expo)
    softmax = expo / expo_sum
    F = (max_c + np.log(expo_sum) / T) * dx

    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c

    expo_sum = np.sum(expo)

    F = (max_c + np.log(expo_sum) / T) * dx
    
    grad = np.zeros(n)
    for k in range(len(softmax)):
        w = softmax[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            c = np.ones(n)  # (1 - h[j+s]) starts as all 1s (h=0 out of bounds)
            # Subtract h[j+s] where j+s is in [0, n-1]
            if s > 0:
                # j+s in [0, n-1] ⟺ j in [0, n-1-s]
                c[:n-s] -= h[s:]
                # Subtract h[j-s] where j-s is in [0, n-1]
                # j-s in [0, n-1] ⟺ j in [s, n-1]
                c[s:] -= h[:n-s]
            else:
                sp = -s
                # j+s = j-sp in [0, n-1] ⟺ j in [sp, n-1]
                c[sp:] -= h[:n-sp]
                # j-s = j+sp in [0, n-1] ⟺ j in [0, n-1-sp]
                c[:n-sp] -= h[sp:]
            grad += w * c
    grad *= dx
    return F, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    expo = np.exp(T * shifted)
    expo_sum = np.sum(expo)
    softmax = expo / expo_sum
    F = (max_c + np.log(expo_sum) / T) * dx
    grad = np.zeros(n)
    for k in range(len(softmax)):
        w = softmax[k]
        if w < 1e-14:
            continue
        s = k - (n - 1)
        if s == 0:
            grad += w * (1.0 - 2.0 * h)
        elif s > 0:
            c = np.ones(n)
            c[:n - s] -= h[s:]
            c[s:] -= h[:n - s]
            grad += w * c
        else:
            sp = -s
            c = np.ones(n)
            c[sp:] -= h[:n - sp]
            c[:n - sp] -= h[sp:]
            grad += w * c
    grad *= dx
    return F, grad

def slsqp_smooth(h_init, T, maxiter, tol):
    n = len(h_init)
    bounds = [(0.0, 1.0)] * n
    cons_kwargs = {'type': 'eq',
                   'fun': lambda x: np.sum(x) - n / 2.0,
                   'jac': lambda x: np.ones(n)}

    def obj(x):
        return smooth_obj_grad(x, T)[0]

    def objg(x):
        return smooth_obj_grad(x, T)[1]

    res = minimize(obj, h_init, jac=objg, method='SLSQP',
                   bounds=bounds, constraints=[cons_kwargs],
                   options={'maxiter': maxiter, 'ftol': tol})
    return res.x

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h_init, n, max_iters, T0, T1, rng):
    b = to_binary(h_init, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def progressive_slsqp(h_init, n, T_list, maxiter_per, start_time, budget_s):
    x = h_init.copy()
    for T_val in T_list:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 15:
            break
        iters = maxiter_per
        try:
            x = slsqp_smooth(x, T_val, iters, 1e-13)
        except Exception:
            break
        x = make_feasible(x, n)
    return x

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01),
                     max(0.0, old - 0.003), min(1.0, old + 0.003)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5
