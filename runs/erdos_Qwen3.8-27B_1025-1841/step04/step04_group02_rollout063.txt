_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

    _SL_N = n
    _SL_T = T

    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,

    return res.x

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    for _ in range(300):
        d = n / 2.0 - h.sum()
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    t = n // 2
    while b.sum() > t:
        o = np.where(b == 1)[0]
        if len(o) == 0:
            break
        b[o[np.argmin(h[o])]] = 0
    while b.sum() < t:
        z = np.where(b == 0)[0]
        if len(z) == 0:
            break
        b[z[np.argmax(h[z])]] = 1
    return b

def resample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    return make_feasible(
        np.interp(np.linspace(0, 2, new_n), np.linspace(0, 2, old_n), h), new_n)

_SN = 0
_ST = 0.0

def _obj(x):
    dx = 2.0 / _SN
    c = np.correlate(x, 1.0 - x, mode='full')
    m = c.max()
    return (m + np.log(np.exp((c - m) * _ST).sum()) / _ST) * dx

def _con(x):
    return x.sum() - _SN / 2.0

def _con_j(x):
    return np.ones(_SN)

def slsqp(h, n, T, maxiter=300, ftol=1e-14):
    global _SN, _ST
    _SN = n; _ST = T
    r = minimize(_obj, h.copy(), method='SLSQP',
                 bounds=[(0.0, 1.0)] * n,
                 constraints=[{'type': 'eq', 'fun': _con, 'jac': _con_j}],
                 options={'maxiter': maxiter, 'ftol': ftol})
    return r.x

                best_c5 = bc; imp = True
        if not imp:
            break
    return h, best_c5
