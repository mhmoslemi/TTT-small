import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def to_binary(h, n, rng):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_sa(h_bin, n, rng, T0, T1, alpha, time_limit, start_time, multi_swap=1):

    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    cur = h_bin.copy()
    cur_c5 = best_c5
    ones = np.where(cur == 1)[0]
    zeros = np.where(cur == 0)[0]
    visited = 0
    while T > T1 and time.time() - start_time < time_limit:
        ns = multi_swap
        if ns > 1:
            i_idx = ones[rng.randint(len(ones), size=ns)]
            j_idx = zeros[rng.randint(len(zeros), size=ns)]
        else:
            i_idx = ones[rng.randint(len(ones))]
            j_idx = zeros[rng.randint(len(zeros))]
        cur[i_idx] = 0
        cur[j_idx] = 1

            cur_c5 = new_c5

            alpha_adj = T1 / T
            alpha = alpha_adj ** (1.0 / max(1, 5000))

def slsqp_smooth(h, n, T, top_m, maxiter):
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None and top_m < len(corr):
            top = corr[np.argsort(corr)[-top_m:]]
            mc = np.max(top)
            return (mc + np.log(np.sum(np.exp(T * (top - mc)))) / T) * (2.0 / n)
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0 / n)
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h) - 1e-15:
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h.copy(), compute_c5(h)

def targeted_perturb(h, n, rng, time_limit, start_time, scale=0.005):
    best_h, best_c5 = h.copy(), compute_c5(h)

def coord_refine(h, n, time_limit, start_time, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() - start_time < time_limit:
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 2:
                break
            old = best_h[j]
            for s in steps:
                for sign in (-1.0, 1.0):
                    v = old + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-15:
                            best_c5 = c
                            best_h = trial
            for v in (0.0, 1.0):
                if abs(v - old) > 1e-4:
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        best_c5 = c
                        best_h = trial
            if best_h[j] != old:
                improved = True
    return best_h, best_c5
