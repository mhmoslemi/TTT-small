```python
def swap_delta(b, i, j, n):
    """Compute change in correlation vector when swapping b[i]:1->0 and b[j]:0->1"""
    # b is the current binary array
    # For shift k (corresponding to array index p = k + n - 1):
    # delta[p] = (-1 + b[i+k])*I(0<=i+k<n) + (1-b[j+k])*I(0<=j+k<n) + b[i-k]*I(0<=i-k<n) - b[j-k]*I(0<=j-k<n)
    
    # Vectorized computation
    p = np.arange(2*n - 1)
    k = p - (n - 1)  # k ranges from -(n-1) to n-1
    
    delta = np.zeros(2*n - 1)
    
    # Term 1: (-1 + b[i+k]) where 0 <= i+k < n
    idx1 = i + k
    mask1 = (idx1 >= 0) & (idx1 < n)
    delta[mask1] += -1 + b[idx1[mask1]]
    
    # Term 2: (1 - b[j+k]) where 0 <= j+k < n
    idx2 = j + k
    mask2 = (idx2 >= 0) & (idx2 < n)
    delta[mask2] += 1 - b[idx2[mask2]]
    
    # Term 3: b[i-k] where 0 <= i-k < n
    idx3 = i - k
    mask3 = (idx3 >= 0) & (idx3 < n)
    delta[mask3] += b[idx3[mask3]]
    
    # Term 4: -b[j-k] where 0 <= j-k < n
    idx4 = j - k
    mask4 = (idx4 >= 0) & (idx4 < n)
    delta[mask4] -= b[idx4[mask4]]
    
    return delta
```

```python
def softmax_grad(h, n, T):
    """Compute gradient of soft-max approximation of C5"""
    dx = 2.0 / n
    corr = compute_corr(h)  # length 2n-1
    mc = np.max(corr)
    
    if T > 1e4:
        # Just use the max component's gradient
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        probs = np.zeros(len(corr))
        probs[max_idx] = 1.0
    else:
        shifted = corr - mc
        exp_shifted = np.exp(T * shifted)
        probs = exp_shifted / np.sum(exp_shifted)
    
    grad = np.zeros(n)
    
    # For each shift k with non-zero probability:
    for p in range(len(corr)):
        pk = probs[p]
        if pk < 1e-10:
            continue
        k = p - (n - 1)
        
        if k >= 0:
            # Forward: m in [0, n-1-k], contribution: pk * (1-h[m+k])
            m_range = np.arange(0, n - k)
            grad[m_range] += pk * (1.0 - h[m_range + k])
            # Backward: m in [k, n-1], contribution: pk * (-h[m-k])
            m_range = np.arange(k, n)
            grad[m_range] += -pk * h[m_range - k]
        else:
            k_abs = -k
            # Forward: m in [k_abs, n-1], contribution: pk * (1-h[m-k_abs]) = (1-h[m+k])
            m_range = np.arange(k_abs, n)
            grad[m_range] += pk * (1.0 - h[m_range + k])
            # Backward: m in [0, n-1-k_abs], contribution: pk * (-h[m+k_abs]) = -h[m+k]
            m_range = np.arange(0, n - k_abs)
            grad[m_range] += -pk * h[m_range - k]
    
    return dx * T * grad
```
