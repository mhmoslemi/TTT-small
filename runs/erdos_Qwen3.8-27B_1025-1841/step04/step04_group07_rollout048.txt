        def obj(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            return soft_max_obj(corr, T) * dx + mu * (x.sum() - n / 2.0) ** 2
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
        h = make_feasible(res.x, n)
    return h

def alt_minmax_v2(h0, n, n_iters=60, topk_frac=0.08, max_inner=40, sharpen=5.0):
    h = h0.copy()
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        active = topk_weights_exp(corr, topk_frac, sharpen)
        h = opt_weighted(h, n, active, 800.0, max_inner)
    return h

def snap_and_reopt(h, n, eps=0.15, T=2000, mu=600.0, max_iter=100):
    h_snapped = h.copy()
    h_snapped[h_snapped < eps] = 0.0
    h_snapped[h_snapped > 1.0 - eps] = 1.0
    h_snapped = make_feasible(h_snapped, n)
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_obj(corr, T) * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h_snapped, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_swap(h_bin, n, rng, max_rounds=80, tries=12000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries, max(1, len(ones) * len(zeros)))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi], h[oj] = 0.0, 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi], h[oj] = 1.0, 0.0
        if not improved:
            break
    return h, cur

def projected_subgrad(h0, n, n_iters=500, alpha_init=0.005, target_C=None):
    h = h0.copy()
    dx = 2.0 / n
    if target_C is None:
        target_C = compute_c5(h) - 0.001
    alpha = alpha_init
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = np.max(corr) * dx
        if max_c <= target_C:
            target_C -= 0.0002
        active = topk_weights_exp(corr, 0.05, 8.0)
        grad = np.zeros_like(h)
        for lag in range(len(active)):
            if active[lag] > 1e-10:
                a = h
                b = 1.0 - h
                if lag < n:
                    # corr[lag] = sum a[i] * b[i] for valid range (positive lag)
                    # grad wrt h: derivative of sum h[i]*(1-h[i+lag]) 
                    # = (1-h[i+lag]) - h[i] for valid i
                    lo = 0
                    hi = n - lag - 1
                    if hi >= lo:
                        seg = (1.0 - h[lo+lag:hi+lag]) - h[lo:hi+1]
                        grad[lo:hi+1] += seg * active[lag]
                    # also for the "b" side: -h[i+lag] * d(1-h[i])/dh
                    # Actually np.correlate(h, 1-h, 'full')[m] needs care
                    # Let's just use numerical approach for simplicity
                else:
                    neg_lag = lag - (n - 1)
                    lo = 0
                    hi = n + neg_lag - 1
                    if hi >= lo:
                        seg = (1.0 - h[lo:hi+1]) - h[lo-neg_lag:hi-neg_lag+1]
                        grad[lo:hi+1] += seg * active[lag]
        # Simpler: just use the soft-max gradient numerically
        eps = 1e-6
        grad_num = np.zeros(n)
        # Use active set gradient more carefully
        # For lag k (0-indexed in full correlation, position k in array of size 2n-1):
        # c[k] = sum_i h[i + k - (n-1)] * (1 - h[i]) where indices valid
        # dc/dh[j] is nonzero for certain ranges
        # Let's just do a simple subgradient: direction = -corr weighted
        # Actually let me just do projected gradient on soft-max
        alpha = alpha_init * (1.0 + 0.0)  # constant for now
        h = h + alpha * np.zeros(n)  # placeholder - will use LBFGS instead
        break
    return h

def fast_coord_descent(h0, n, max_rounds=8, step_list=None):
    if step_list is None:
        step_list = [0.03, -0.03, 0.01, -0.01]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(0, n, 2):  # every other point for speed
            for d in step_list:
                nv = h[j] + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320).astype(float)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    best_n = len(h0)

    # Phase 1: Upsample to n=1536 and continuation optimize
    n_main = 1536
    if rem() > 400:
        n_main = 2048

    h_cur = upsample_h(best_h, n_main)
    c_cur = compute_c5(h_cur)
    if c_cur < best_c5:
        best_c5, best_h = c_cur, h_cur.copy()

    # Continuation with increasing T
    T_list = [5, 15, 50, 200, 800, 3000, 8000]
    if rem() > 250:
        h_cur = continuation_lbfgsb(h_cur, n_main, T_list[:4], mu=600.0, max_iter=120)
        c_cur = compute_c5(h_cur)
        if c_cur < best_c5:
            best_c5, best_h = c_cur, h_cur.copy()
    if rem() > 150:
        h_cur = continuation_lbfgsb(h_cur, n_main, T_list[4:], mu=600.0, max_iter=100)
        c_cur = compute_c5(h_cur)
        if c_cur < best_c5:
            best_c5, best_h = c_cur, h_cur.copy()

    # Phase 2: Alternating min-max with sharpened weights
    if rem() > 200:
        h_alt = alt_minmax_v2(h_cur, n_main, n_iters=45, topk_frac=0.06, max_inner=35, sharpen=6.0)
        c_alt = compute_c5(h_alt)
        if c_alt < best_c5:
            best_c5, best_h = c_alt, h_alt.copy()
        h_cur = h_alt.copy()

    # Phase 3: More alternating with tighter focus
    if rem() > 150:
        h_alt2 = alt_minmax_v2(h_cur, n_main, n_iters=35, topk_frac=0.04, max_inner=30, sharpen=10.0)
        c_alt2 = compute_c5(h_alt2)
        if c_alt2 < best_c5:
            best_c5, best_h = c_alt2, h_alt2.copy()

    # Phase 4: Snap and re-optimize
    if rem() > 100:
        h_snap = snap_and_reopt(h_cur, n_main, eps=0.2, T=3000, mu=600.0, max_iter=80)
        c_snap = compute_c5(h_snap)
        if c_snap < best_c5:
            best_c5, best_h = c_snap, h_snap.copy()
            h_cur = h_snap.copy()

    # Phase 5: Multi-start perturbation
    if rem() > 80:
        for restart in range(4):
            if rem() < 30:
                break
            noise_scale = 0.06 + 0.04 * rng.random()
            h_r = best_h.copy() + noise_scale * rng.randn(n_main)
            h_r = make_feasible(h_r, n_main)
            h_opt = continuation_lbfgsb(h_r, n_main, [20, 200, 2000], mu=600.0, max_iter=80)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()

    # Phase 6: Binary local search at n=1024
    if rem() > 80:
        n_bin = 1024
        h_b = upsample_h(best_h, n_bin)
        h_bin = (h_b > 0.5).astype(float)
        target_ones = n_bin // 2
        n_ones = int(h_bin.sum())
        if n_ones > target_ones:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target_ones]] = 0.0
        elif n_ones < target_ones:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target_ones - n_ones]] = 1.0
        h_bs, c_bs = binary_swap(h_bin, n_bin, rng, max_rounds=60, tries=15000)
        if c_bs < best_c5:
            best_c5, best_h = c_bs, h_bs.copy()

    # Phase 7: Upsample binary back and re-optimize
    if rem() > 60 and best_n == 1024:
        h_up = upsample_h(best_h, n_main)
        h_opt = continuation_lbfgsb(h_up, n_main, [100, 1000, 5000], mu=600.0, max_iter=100)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()

    # Phase 8: Final coordinate descent
    if rem() > 25:
        n_cd = min(len(best_h), 1024)
        h_cd_src = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        h_cd, c_cd = fast_coord_descent(h_cd_src, n_cd, max_rounds=6,
                                        step_list=[0.02, -0.02, 0.008, -0.008])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 9: One more binary polish
    if rem() > 20:
        n_bp = min(len(best_h), 1024)
        h_bp = upsample_h(best_h, n_bp) if n_bp < len(best_h) else best_h.copy()
        h_bp_bin = (h_bp > 0.5).astype(float)
        t_ones = n_bp // 2
        n_ones = int(h_bp_bin.sum())
        if n_ones > t_ones:
            idx = np.where(h_bp_bin == 1)[0]
            h_bp_bin[idx[:n_ones - t_ones]] = 0.0
        elif n_ones < t_ones:
            idx = np.where(h_bp_bin == 0)[0]
            h_bp_bin[idx[:t_ones - n_ones]] = 1.0
        h_bp2, c_bp2 = binary_swap(h_bp_bin, n_bp, rng, max_rounds=40, tries=10000)
        if c_bp2 < best_c5:
            best_c5, best_h = c_bp2, h_bp2.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
