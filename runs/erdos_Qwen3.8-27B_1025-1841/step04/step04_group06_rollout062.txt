    # Compute h[i+lags] and h[i-lags]

    im = i - lags
    jp = j + lags

        h_bin2 = np.where(h2 > 0.5, 1.0, 0.0)
        n_ones2 = int(np.sum(h_bin2))
        if n_ones2 > n2 // 2:
            oi = np.where(h_bin2 == 1)[0]
            for k in range(n_ones2 - n2 // 2):
                h_bin2[oi[k]] = 0.0
        elif n_ones2 < n2 // 2:
            zi = np.where(h_bin2 == 0)[0]
            for k in range(n2 // 2 - n_ones2):
                h_bin2[zi[k]] = 1.0
        h_opt2, c5_opt2 = binary_local_search(h_bin2, n2, rng, time_limit=min(tl() - 20, 300), batch_size=5000)
        if c5_opt2 < best_c5:
            best_c5, best_h = c5_opt2, h_opt2.copy()
    
    # Final: ensure feasibility and recompute
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
