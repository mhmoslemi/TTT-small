from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0 - 1e-14
    else:
        mask = h > 1e-14
    cnt = mask.sum()
    if cnt == 0:
        return h
    delta = diff / cnt
    h[mask] += delta
    return np.clip(h, 0.0, 1.0)

def make_feasible_iter(h, n, iters=100):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(iters):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        delta = diff / cnt
        h[mask] = np.clip(h[mask] + delta, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = clip_safe(h[idx] + diff)
    return h

def clip_safe(v):
    return max(0.0, min(1.0, v))

def upsample_h(h, new_n):

    if new_n == old_n:
        return make_feasible_iter(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible_iter(h_new, new_n)

        viol = np.sum(h) - n / 2.0

def make_feasible_iter(h, n, iters=100):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(iters):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break

        else:
            mask = h > 1e-14

        if cnt == 0:

    best_h = make_feasible_iter(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
