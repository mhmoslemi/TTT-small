            grad[i - lag_pos] += (-h[i]) * dx

    return make_feasible(h_new, n)

            grad[i + lag] += (-h[i]) * dx
    else:
        lp = -lag
        for i in range(lp, n):
            grad[i] += (1.0 - h[i - lp]) * dx
            grad[i - lp] += (-h[i]) * dx
    return grad
