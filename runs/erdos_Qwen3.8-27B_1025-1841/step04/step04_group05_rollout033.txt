def project_to_simplex_with_bounds(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    n = len(h)
    # Use bisection on the dual variable
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0, 1)
    # Final adjustment for exactness
    diff = target_sum - np.sum(projected)
    if abs(diff) > 1e-15:
        # Adjust the element closest to its bound
        if diff > 0:
            idx = np.argmax(projected < 1)
            # Actually, distribute among non-bound elements
            mask = projected < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                projected[mask] = np.clip(projected[mask] + diff/cnt, 0, 1)
        else:
            mask = projected > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                projected[mask] = np.clip(projected[mask] + diff/cnt, 0, 1)
    return projected
```

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        mask = h < 1.0 - 1e-10 if diff > 0 else h > 1e-10
        cnt = np.sum(mask)

            break

    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            # Find indices where h < 1
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] = np.clip(h[idx[0]] + diff, 0.0, 1.0)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] = np.clip(h[idx[0]] + diff, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def softmax_obj_and_grad(h, T):
    """Compute soft-max approximation of max(corr)*dx and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    shifted = corr - max_corr
    # For numerical stability
    shifted = np.minimum(shifted, 0.0)
    exp_shifted = np.exp(T * shifted)
    sum_exp = np.sum(exp_shifted)
    weights = exp_shifted / sum_exp
    
    obj = (max_corr + np.log(sum_exp) / T) * dx
    
    # Gradient computation
    # g[j] = dx * [sum_{m=-j}^{n-1-j} w[m+(n-1)] * (1-h[j+m]) - sum_{m=j-(n-1)}^{j} w[m+(n-1)] * h[j-m]]
    g = np.zeros(n)
    for j in range(n):
        # First sum
        m_vals = np.arange(-j, n - j)  # -j to n-1-j
        k_vals = m_vals + (n - 1)
        h_idx1 = j + m_vals  # j+m, ranges from 0 to n-1
        s1 = np.sum(weights[k_vals] * (1.0 - h[h_idx1]))
        
        # Second sum
        m_vals2 = np.arange(j - (n-1), j + 1)  # j-(n-1) to j
        k_vals2 = m_vals2 + (n - 1)
        h_idx2 = j - m_vals2  # j-m, ranges from 0 to n-1
        s2 = np.sum(weights[k_vals2] * h[h_idx2])
        
        g[j] = dx * (s1 - s2)
    
    return obj, g

def lbfgs_optimize(h_start, n, T, max_iter=200, gtol=1e-10):
    """L-BFGS-B optimization with soft-max objective."""
    best_h = h_start.copy()
    best_obj = compute_c5(h_start)
    
    def obj_fn(h):
        o, _ = softmax_obj_and_grad(h, T)
        # Add penalty for sum constraint violation
        penalty = 1e6 * (np.sum(h) - n/2.0)**2
        return o + penalty
    
    def grad_fn(h):
        _, g = softmax_obj_and_grad(h, T)
        g += 2e6 * (np.sum(h) - n/2.0) / n  # derivative of penalty
        return g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h_start, method='L-BFGS-B', jac=grad_fn,
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': gtol, 'maxfun': 5000})
    
    h_opt = make_feasible(res.x, n)
    c5 = compute_c5(h_opt)
    if c5 < best_obj:
        best_h = h_opt
        best_obj = c5
    return best_h, best_obj

def subgradient_method(h_start, n, n_iter=200, step_init=0.1):
    """Subgradient method for the actual max objective."""
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(corr)
        m = max_idx - (n - 1)
        
        # Subgradient of corr[max_idx] w.r.t. h
        g = np.zeros(n)
        a = max(0, -m)
        b = min(n-1, n-1-m)
        # d(corr)/d(h[j]) = (1-h[j+m])*1[j in [a,b]] - h[j-m]*1[j-m in [a,b]]
        for j in range(a, b+1):
            g[j] += (1.0 - h[j+m])
        j_min2 = max(0, a + m)
        j_max2 = min(n-1, b + m)
        for j in range(j_min2, j_max2+1):
            g[j] -= h[j-m]
        
        # Also include lags close to the max
        sorted_idx = np.argsort(corr)[::-1]
        for idx in sorted_idx[1:5]:
            if corr[idx] < max(corr) - 0.005:
                break
            m2 = idx - (n - 1)
            a2 = max(0, -m2)
            b2 = min(n-1, n-1-m2)
            for j in range(a2, b2+1):
                g[j] += 0.5 * (1.0 - h[j+m2])
            j_min2_2 = max(0, a2 + m2)
            j_max2_2 = min(n-1, b2 + m2)
            for j in range(j_min2_2, j_max2_2+1):
                g[j] -= 0.5 * h[j-m2]
        
        # Step size (decreasing)
        step = step_init / (1.0 + 0.01 * it)
        
        h_new = h - step * g
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

Actually, for the subgradient, I can compute it more efficiently:
- For lag m, the subgradient is: g_m[j] = (1-h[j+m])*1[j∈R(m)] - h[j-m]*1[j-m∈R(m)]
- This can be computed as: g_m = (1-h_shifted_up) * mask1 - h_shifted_down * mask2

def subgradient_for_lag(h, m):
    n = len(h)
    g = np.zeros(n)
    a = max(0, -m)
    b = min(n-1, n-1-m)
    
    # First term: (1-h[j+m]) for j in [a, b]
    j_range = np.arange(a, b+1)
    h_shifted = h[j_range + m]  # h[j+m] for j in [a,b]
    g[j_range] += (1.0 - h_shifted)
    
    # Second term: -h[j-m] for j-m in [a, b], i.e., j in [a+m, b+m]
    j_range2_start = max(0, a + m)
    j_range2_end = min(n-1, b + m)
    if j_range2_start <= j_range2_end:
        j_range2 = np.arange(j_range2_start, j_range2_end + 1)
        g[j_range2] -= h[j_range2 - m]
    
    return g

This is related to the concept of "almost periodic" functions or functions with small discrepancy.

One approach: think of the set A = {x: h(x) = 1} with |A| = 1. We want to minimize max_k |A \ (A-k)| / 1 = max_k (1 - |A ∩ (A-k)|).

For a convex set (like an interval), |A ∩ (A-k)| = 1-|k| for |k|≤1, which gives min = 0 (at k=1 or k=2).

For a more "spread out" set, we can do better. A union of small intervals might work.

If A is n intervals of length 1/n evenly spaced, then for small k, most intervals overlap, but for k near the spacing, they don't overlap at all.

This suggests that the optimal structure might be related to a "balanced" distribution of the 1-set.

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            candidates = np.where(h < 1.0)[0]
            if len(candidates) > 0:
                h[candidates[0]] = min(1.0, h[candidates[0]] + diff)
        else:
            candidates = np.where(h > 0.0)[0]
            if len(candidates) > 0:
                h[candidates[0]] = max(0.0, h[candidates[0]] + diff)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def softobj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    shifted = np.minimum(corr - max_c, 0.0)
    exp_v = np.exp(T * shifted)
    s = np.sum(exp_v)
    return (max_c + np.log(s) / T) * dx

def softgrad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    shifted = np.minimum(corr - max_c, 0.0)
    exp_v = np.exp(T * shifted)
    s = np.sum(exp_v)
    w = exp_v / s
    
    omh = 1.0 - h
    g = np.zeros(n)
    for j in range(n):
        # First sum: m from -j to n-1-j
        ms = np.arange(-j, n - j)
        g[j] += np.dot(w[ms + n - 1], omh[j + ms])
        # Second sum: m from j-(n-1) to j
        ms2 = np.arange(j - n + 1, j + 1)
        g[j] -= np.dot(w[ms2 + n - 1], h[j - ms2])
    g *= dx
    return g

    # ac_circ[m] = sum_{i=0}^{size-1} h_padded[i] * h_padded[(i+m) % size]
    # For m in [0, n-1]: this equals sum_{i=0}^{n-1-m} h[i]*h[i+m] + sum_{i=size-n+m}^{size-1} h[i]*h_padded[(i+m)%size]
    # Since h_padded is zero outside [0,n-1]:
    # = sum_{i=0}^{n-1-m} h[i]*h[i+m] + sum_{i=size-n+m}^{size-1} 0 * ...
    # Hmm, this isn't quite right.
    
    # Let me just use np.correlate for simplicity (it's O(n²) but fine for n ≤ 2560)
    corr = np.correlate(h_bin, 1.0 - h_bin, mode='full')
    return float(np.max(corr) * dx)
```

Actually, for the binary case, I can compute the correlation more efficiently:
corr[k] = sum_i h[i] * (1-h[i+m]) = (n/2) - sum_i h[i]*h[i+m] (approximately, ignoring boundary effects)

The full correlation with boundary effects is:
corr_full[k] = sum_{i in valid range} h[i] * (1-h[i+m])

For the binary case, this is:
= (number of valid i with h[i]=1) - (number of valid i with h[i]=1 and h[i+m]=1)

The first term depends on m (it's the number of 1s in positions i where both i and i+m are valid). For m > 0: i ranges from 0 to n-1-m, so it's the number of 1s in h[0:n-m]. For m < 0: i ranges from -m to n-1, so it's the number of 1s in h[-m:n].

This makes the boundary effects matter. For large n, the boundary effects are small (O(n) out of n/2 total ones), so the approximation corr[k] ≈ n/2 - autocorr[m] is reasonable.

For the local search, I'll just use the exact computation via np.correlate (it's fast enough for n ≤ 2560).

Let me now write the complete solution. I'll combine:
1. L-BFGS-B with analytical gradient (for continuous optimization)
2. Binary local search (for discrete optimization)
3. Multi-resolution strategy
4. Time management

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 0
_G_PEN = 1e8
_SG_START = 0.0
_SG_BUDGET = 60.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            cands = np.where(h < 1.0)[0]
            if len(cands) > 0:
                h[int(cands[0])] = min(1.0, h[int(cands[0])] + diff)
        else:
            cands = np.where(h > 0.0)[0]
            if len(cands) > 0:
                h[int(cands[0])] = max(0.0, h[int(cands[0])] + diff)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def binary_project(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def softobj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    shifted = np.minimum(corr - max_c, 0.0)
    exp_v = np.exp(T * shifted)
    s = float(np.sum(exp_v))
    return (max_c + np.log(s) / T) * dx

def softgrad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(np.max(corr))
    shifted = np.minimum(corr - max_c, 0.0)
    exp_v = np.exp(T * shifted)
    s = float(np.sum(exp_v))
    w = exp_v / s
    omh = 1.0 - h
    g = np.zeros(n)
    for j in range(n):
        ms1 = np.arange(-j, n - j)
        g[j] += np.dot(w[ms1 + n - 1], omh[j + ms1])
        ms2 = np.arange(j - n + 1, j + 1)
        g[j] -= np.dot(w[ms2 + n - 1], h[j - ms2])
    g *= dx
    return g

def _obj_func(h):
    o = softobj(h, _G_T)
    pen = _G_PEN * (np.sum(h) - _G_N / 2.0) ** 2
    return o + pen

def _grad_func(h):
    g = softgrad(h, _G_T)
    g += 2.0 * _G_PEN * (np.sum(h) - _G_N / 2.0) / _G_N
    return g

def _lbfgs_step(h_start, n, T, max_iter=300, gtol=1e-8, maxfun=8000):
    global _G_N, _G_T, _G_PEN
    _G_N = n
    _G_T = T
    _G_PEN = 1e8
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_func, h_start, method='L-BFGS-B', jac=_grad_func,
                   bounds=bounds, options={'maxiter': max_iter, 'gtol': gtol, 'maxfun': maxfun})
    return make_feasible(res.x, n)

def _subgrad_polish(h_start, n, n_iter=300, budget=60.0):
    global _SG_START, _SG_BUDGET
    _SG_START = time.time()
    _SG_BUDGET = budget
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_init = 0.03
    for it in range(n_iter):
        if time.time() - _SG_START > budget:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(5, len(corr))
        sorted_idx = np.argsort(corr)[::-1][:top_k]
        g = np.zeros(n)
        for idx in sorted_idx:
            m = int(idx) - (n - 1)
            a = max(0, -m)
            b = min(n - 1, n - 1 - m)
            j_range = np.arange(a, b + 1)
            g[j_range] += (1.0 - h[j_range + m]) / top_k
            j2s = max(0, a + m)
            j2e = min(n - 1, b + m)
            if j2s <= j2e:
                j2r = np.arange(j2s, j2e + 1)
                g[j2r] -= h[j2r - m] / top_k
        step = step_init / (1.0 + 0.15 * it)
        h_new = make_feasible(h - step * g, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h

def _binary_local_search(h_bin, n, n_iter=20000, budget=30.0):
    start_t = time.time()
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    rng = np.random.RandomState(12345)
    for it in range(n_iter):
        if time.time() - start_t > budget:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c5

def _nb_local_search(h_bin, n, n_iter=10000, budget=30.0):
    start_t = time.time()
    h = h_bin.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    ones = np.where(h == 1.0)[0].tolist()
    zeros = np.where(h == 0.0)[0].tolist()
    improved = True
    while improved and time.time() - start_t < budget:
        improved = False
        for i in ones:
            if time.time() - start_t > budget:
                break
            for j in zeros:
                if hue_diff := abs(i - j) > n // 2:
                    hue_diff = n - abs(i - j)
                else:
                    hue_diff = abs(i - j)
                if hue_diff > n // 2:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-15:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                    ones = np.where(h == 1.0)[0].tolist()
                    zeros = np.where(h == 0.0)[0].tolist()
                    break
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:

    for n_pts in [1024, 1536, 2048, 2560]:
        if _tl() < 100:
            break
        h_curr = upsample_h(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        for T in [30, 80, 200, 500, 1200]:
            if _tl() < 25:
                break
            try:
                h_opt = _lbfgs_step(h_curr, n_pts, T, max_iter=250, gtol=1e-9)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_curr = h_opt.copy()
            except Exception:
                pass

        # Subgradient polish
        if _tl() > 50:
            h_sg = _subgrad_polish(h_curr, n_pts, n_iter=200, budget=min(45, _tl() - 20))
            c5_sg = compute_c5(h_sg)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h_curr = h_sg.copy()

        # Binary projection + local search
        if _tl() > 40:
            h_bin = binary_project(h_curr, n_pts)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
            # Binary local search
            h_bl, c5_bl = _binary_local_search(h_bin, n_pts, n_iter=15000, budget=min(30, _tl() - 15))
            if c5_bl < best_c5:
                best_c5 = c5_bl
                best_h = h_bl.copy()

    # Phase 2: Higher resolution L-BFGS-B from best
    n_final = max(len(best_h), 2048)
    if _tl() > 80:
        h_f = upsample_h(best_h, n_final)
        if compute_c5(h_f) < best_c5:
            best_c5 = compute_c5(h_f)
            best_h = h_f.copy()
        for T in [100, 400, 1000, 3000]:
            if _tl() < 25:
                break
            try:
                h_opt = _lbfgs_step(h_f, n_final, T, max_iter=200, gtol=1e-9)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                h_f = h_opt.copy()
            except Exception:
                pass
        h_sg = _subgrad_polish(h_f, n_final, n_iter=150, budget=min(40, _tl() - 15))
        c5_sg = compute_c5(h_sg)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()

    # Phase 3: Multiple restarts
    n_r = len(best_h)
    for trial in range(5):
        if _tl() < 25:
            break
        if trial == 0:
            h_init = binary_project(best_h, n_r)
        elif trial == 1:
            h_init = (rng.rand(n_r) < 0.5).astype(float)
        elif trial == 2:
            h_init = 0.6 * best_h + 0.4 * (rng.rand(n_r) < 0.5).astype(float)
        elif trial == 3:
            period = max(2, n_r // 8)
            h_init = np.zeros(n_r)
            for s in range(4):
                start = s * period + trial * 2
                h_init[start:start+period//2] = 1.0
        else:
            h_init = best_h + 0.1 * rng.randn(n_r)
        h_init = make_feasible(h_init, n_r)
        for T in [100, 500]:
            if _tl() < 20:
                break
            try:
                h_opt = _lbfgs_step(h_init, n_r, T, max_iter=150, gtol=1e-8)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_init = h_opt.copy()
            except Exception:
                pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
