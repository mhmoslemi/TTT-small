def simulated_annealing_binary(h, n, start_time, budget_s, rng, 
                                T_start=1.0, T_end=0.01, n_iter=50000):
    """Simulated annealing on binary space with swap moves."""
    h = h.copy()
    h = np.where(h > 0.5, 1.0, 0.0)
    # Fix balance
    n_ones = int(np.sum(h == 1.0))
    target = n // 2
    if n_ones > target:
        idx_ones = np.where(h == 1.0)[0]
        rng.shuffle(idx_ones)
        for i in range(n_ones - target):
            h[idx_ones[i]] = 0.0
    elif n_ones < target:
        idx_zeros = np.where(h == 0.0)[0]
        rng.shuffle(idx_zeros)
        for i in range(target - n_ones):
            h[idx_zeros[i]] = 1.0
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    curr_c5 = best_c5
    
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    
    for it in range(n_iter):
        if time.time() - start_time > budget_s - 30:
            break
        
        # Annealing schedule
        progress = it / n_iter
        T_sa = T_start * (T_end / T_start) ** progress
        
        # Random swap
        one_idx = list(ones_set)[rng.randint(len(ones_set))]
        zero_idx = list(zeros_set)[rng.randint(len(zeros_set))]
        
        # Make swap
        h[one_idx] = 0.0
        h[zero_idx] = 1.0
        new_c5 = compute_c5(h)
        
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / (T_sa * 0.001)):
            # Accept
            curr_c5 = new_c5
            ones_set.remove(one_idx)
            ones_set.add(zero_idx)
            zeros_set.remove(zero_idx)
            zeros_set.add(one_idx)
            
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
        else:
            # Reject
            h[one_idx] = 1.0
            h[zero_idx] = 0.0
    
    return best_h, best_c5

            break
        h_low = upsample_h(best_h, n_try)
        for T in [30.0, 100.0, 250.0]:
            if _tl() < 60:
                break
            h_res = do_slsqp(h_low, n_try, T, maxiter=200, top_k=15)
            c5r = compute_c5(h_res)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = h_res

    # Phase 3: Try higher resolution
    if _tl() > 150:
        n_hi = min(1920, n_init * 2)
        if n_hi % 2 == 0:
            h_hi = upsample_h(best_h, n_hi)
            for T in [50.0, 150.0]:
                if _tl() < 80:
                    break
                h_res = do_slsqp(h_hi, n_hi, T, maxiter=150, top_k=15)
                c5r = compute_c5(h_res)
                if c5r < best_c5:
                    best_c5 = c5r
                    best_h = h_res

    # Phase 4: Coordinate descent at best resolution
    n_best = len(best_h)
    if _tl() > 80:
        best_h, best_c5 = coord_descent(best_h, n_best, start_time, budget_s,
                                        n_rounds=5, candidate_fn=cands_medium)

    # Phase 5: Binary search at multiple resolutions
    for n_bin in [int(n_best * 0.5), int(n_best * 0.67), n_best]:
        if _tl() < 50:
            break
        n_bin = max(160, n_bin)
        if n_bin % 2 != 0:
            n_bin -= 1
        h_bin = to_binary(upsample_h(best_h, n_bin), n_bin, rng)
        h_bin, c5_bin = binary_anneal(h_bin, n_bin, start_time, budget_s, rng,
                                      n_iter=25000 if n_bin <= 640 else 15000)
        h_bin, c5_bin = binary_sweep(h_bin, n_bin, start_time, budget_s, rng)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin

    # Phase 6: SLSQP polish
    if _tl() > 60:
        n_pol = len(best_h)
        h_res = do_slsqp(best_h, n_pol, T=300.0, maxiter=250, top_k=10)
        c5r = compute_c5(h_res)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_res

    # Phase 7: Final fine coordinate descent
    if _tl() > 30:
        n_fin = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n_fin, start_time, budget_s,
                                        n_rounds=4, candidate_fn=cands_fine)

    # Phase 8: One more SLSQP if time permits
    if _tl() > 40:
        n_
pol2 = len(best_h)
        h_res = do_slsqp(best_h, n_pol2, T=500.0, maxiter=200, top_k=8)
        c5r = compute_c5(h_res)
        if c5r < best_c5:
            best_c5 = c5r
            best_h = h_res

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
