        h_sl = make_feasible(h_sl, n)
        c5_sl = compute_c5(h_sl)
        if c5_sl < best_c5:
            best_c5 = c5_sl
            best_h = h_sl
    
    # Phase 4: CD again after SLSQP
    if remaining() > 150:
        h = best_h.copy()
        corr = get_corr(h)
        for cd_pass in range(200):
            if remaining() < 100:
                break
            h, corr, improved = fast_cd_pass(h, corr, n)
            if not improved:
                break
            h = make_feasible(h, n)
            corr = get_corr(h)
            c5 = np.max(corr) * (2.0 / n)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Phase 5: Try higher resolution
    if remaining() > 500:
        n2 = n * 2
        if n2 <= 3840:
            h2 = upsample_h(best_h, n2)
            c5_2 = compute_c5(h2)
            if c5_2 <= best_c5 + 0.003:
                corr2 = get_corr(h2)
                for cd_pass in range(300):
                    if remaining() < 250:
                        break
                    h2, corr2, improved = fast_cd_pass(h2, corr2, n2)
                    if not improved:
                        break
                    h2 = make_feasible(h2, n2)
                    corr2 = get_corr(h2)
                    c5_2 = np.max(corr2) * (2.0 / n2)
                    if c5_2 < best_c5:
                        best_c5 = c5_2
                        best_h = h2.copy()
                
                # SLSQP at higher res
                if remaining() > 150:
                    n_cur = len(best_h)
                    top_corr = get_corr(best_h)
                    mc = np.max(top_corr)
                    thr = mc - 0.005 * mc
                    top_idx = np.where(top_corr >= thr)[0]
                    if len(top_idx) > 15:
                        tv = top_corr[top_idx]
                        so = np.argsort(-tv)
                        top_idx = top_idx[so[:15]]
                    h2_sl = slsqp_optimize(best_h, n_cur, 400.0, top_idx, maxiter=200)
                    h2_sl = make_feasible(h2_sl, n_cur)
                    c5_2sl = compute_c5(h2_sl)
                    if c5_2sl < best_c5:
                        best_c5 = c5_2sl
                        best_h = h2_sl
                    
                    # More CD at higher res
                    if remaining() > 80:
                        h2 = best_h.copy()
                        corr2 = get_corr(h2)
                        for cd_pass in range(100):
                            if remaining() < 40:
                                break
                            h2, corr2, improved = fast_cd_pass(h2, corr2, n_cur)
                            if not improved:
                                break
                            h2 = make_feasible(h2, n_cur)
                            corr2 = get_corr(h2)
                            c5_2 = np.max(corr2) * (2.0 / n_cur)
                            if c5_2 < best_c5:
                                best_c5 = c5_2
                                best_h = h2.copy()
    
    # Phase 6: Try structured initializations at current resolution
    if remaining() > 200:
        n_cur = len(best_h)
        for ptype in ['sawtooth', 'checker']:
            if remaining() < 100:
                break
            h_struct = make_structured_initial(n_cur, ptype)
            h_s = slsqp_optimize(h_struct, n_cur, 200.0, maxiter=150)
            h_s = make_feasible(h_s, n_cur)
            c5_s = compute_c5(h_s)
            if c5_s < best_c5:
                best_c5 = c5_s
                best_h = h_s
                # CD polish
                h_s2 = best_h.copy()
                corr_s2 = get_corr(h_s2)
                for _ in range(50):
                    if remaining() < 60:
                        break
                    h_s2, corr_s2, imp = fast_cd_pass(h_s2, corr_s2, n_cur)
                    if not imp:
                        break
                    h_s2 = make_feasible(h_s2, n_cur)
                    corr_s2 = get_corr(h_s2)
                    c5_s2 = np.max(corr_s2) * (2.0 / n_cur)
                    if c5_s2 < best_c5:
                        best_c5 = c5_s2
                        best_h = h_s2.copy()
    
    # Phase 7: Random perturbation restarts
    for restart in range(5):
        if remaining() < 80:
            break
        n_cur = len(best_h)
        scales = [0.01, 0.02, 0.04, 0.02, 0.05]
        h_try = make_feasible(best_h + rng.randn(n_cur) * scales[restart], n_cur)
        corr_try = get_corr(h_try)
        for cd_pass in range(80):
            if remaining() < 50:
                break
            h_try, corr_try, improved = fast_cd_pass(h_try, corr_try, n_cur)
            if not improved:
                break
            h_try = make_feasible(h_try, n_cur)
            corr_try = get_corr(h_try)
            c5_try = np.max(corr_try) * (2.0 / n_cur)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
    
    # Phase 8: Final SLSQP with very high T on top shifts
    if remaining() > 60:
        n_cur = len(best_h)
        top_corr = get_corr(best_h)
        mc = np.max(top_corr)
        thr = mc - 0.003 * mc
        top_idx = np.where(top_corr >= thr)[0]
        if len(top_idx) > 10:
            tv = top_corr[top_idx]
            so = np.argsort(-tv)
            top_idx = top_idx[so[:10]]
        h_fin = slsqp_optimize(best_h, n_cur, 1000.0, top_idx, maxiter=300)
        h_fin = make_feasible(h_fin, n_cur)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin
    
    # Phase 9: Final binary CD sweep
    if remaining() > 20:
        n_cur = len(best_h)
        h = best_h.copy()
        corr = get_corr(h)
        for cd_pass in range(50):
            if remaining() < 10:
                break
            h, corr, improved = fast_cd_pass(h, corr, n_cur)
            if not improved:
                break
            h = make_feasible(h, n_cur)
            corr = get_corr(h)
            c5 = np.max(corr) * (2.0 / n_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
