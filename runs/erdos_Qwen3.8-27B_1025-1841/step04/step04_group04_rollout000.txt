        frac = it / max(n_iter - 1, 1)
        temp = t_start * (t_end / t_start) ** frac
        one_idx = np.where(h == 1.0)[0]
        zero_idx = np.where(h == 0.0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            continue
        i = rng.choice(one_idx)
        j = rng.choice(zero_idx)
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(temp, 1e-12)):
            cur_c = new_c
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def coord_descent(h, n, candidates, max_rounds, time_limit):
    start = time.time()
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old = h[j]
            best_trial = old
            bc = best_c
            for cand in candidates:
                val = max(0.0, min(1.0, cand))
                trial = h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-13:
                    bc = c
                    best_trial = val
            if best_trial != old:
                h = h.copy()
                h[j] = best_trial
                h = make_feasible(h, n)
                best_c = bc
                improved = True
        if not improved:
            break
    return h, best_c

def multi_swap_search(h, n, n_iters, seed, time_limit):
    rng = np.random.RandomState(seed)
    h = fix_binary_sum(binary_snapshot(h), n)
    cur_c = compute_c5(h)
    best_c = cur_c
    best_h = h.copy()
    start = time.time()
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        one_idx = np.where(h == 1.0)[0]
        zero_idx = np.where(h == 0.0)[0]
        if len(one_idx) == 0 or len(zero_idx) == 0:
            break
        i = rng.choice(one_idx)
        j = rng.choice(zero_idx)
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        if new_c < cur_c:
            cur_c = new_c
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def remaining():
        return budget_s - elapsed()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)

    n_parent = len(h_init)
    h_init = make_feasible(h_init, n_parent)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: SLSQP at parent resolution with multiple T
    n1 = n_parent
    for T, maxit, ftol in [(60.0, 400, 1e-13), (150.0, 350, 1e-13), (400.0, 300, 1e-14)]:
        if remaining() < 120:
            break
        h_try = run_slsqp(best_h.copy(), n1, T, maxit, ftol)
        h_try = make_feasible(h_try, n1)
        c = compute_c5(h_try)
        if c < best_c5 - 1e-14:
            best_c5 = c
            best_h = h_try.copy()

    # Phase 2: Simulated annealing at parent resolution
    if remaining() > 200:
        h_sa, c_sa = simulated_annealing(best_h, n1, 15000, 0.003, 0.0001, seed)
        if c_sa < best_c5 - 1e-14:
            best_c5 = c_sa
            best_h = h_sa.copy()

    # Phase 3: Coordinate descent at parent resolution
    if remaining() > 100:
        cands = [0.0, 1.0]
        best_h, best_c5 = coord_descent(best_h, n1, cands, 6, remaining() - 80)

    # Phase 4: Upsample to higher resolution
    n2 = 1920
    if remaining() > 250:
        h_up = upsample_h(best_h, n2)
        for T, maxit, ftol in [(80.0, 300, 1e-13), (200.0, 250, 1e-13)]:
            if remaining() < 100:
                break
            h_try = run_slsqp(h_up.copy(), n2, T, maxit, ftol)
            h_try = make_feasible(h_try, n2)
            c = compute_c5(h_try)
            if c < best_c5 - 1e-14:
                best_c5 = c
                best_h = h_try.copy()
                h_up = best_h.copy()
        if remaining() > 100:
            h_sa2, c_sa2 = simulated_annealing(best_h, n2, 8000, 0.002, 0.00005, seed + 1)
            if c_sa2 < best_c5 - 1e-14:
                best_c5 = c_sa2
                best_h = h_sa2.copy()
        if remaining() > 60:
            best_h, best_c5 = coord_descent(best_h, n2, [0.0, 1.0], 4, remaining() - 40)

    # Phase 5: Multi-swap local search at best resolution
    n_best = len(best_h)
    if remaining() > 80:
        h_ms, c_ms = multi_swap_search(best_h, n_best, 20000, seed + 2, remaining() - 50)
        if c_ms < best_c5 - 1e-14:
            best_c5 = c_ms
            best_h = h_ms.copy()

    # Phase 6: Fine coordinate descent with intermediate values
    if remaining() > 40:
        cands_fine = [0.0, 1.0]
        for d in [0.1, 0.2, 0.3, 0.7, 0.8, 0.9]:
            cands_fine.append(d)
        best_h, best_c5 = coord_descent(best_h, len(best_h), cands_fine, 3, remaining() - 20)

    # Phase 7: Final SLSQP polish
    if remaining() > 50:
        n_f = len(best_h)
        h_pol = run_slsqp(best_h.copy(), n_f, 500.0, 200, 1e-14)
        h_pol = make_feasible(h_pol, n_f)
        c_pol = compute_c5(h_pol)
        if c_pol < best_c5 - 1e-14:
            best_c5 = c_pol
            best_h = h_pol.copy()

    # Phase 8: Final binary CD
    if remaining() > 15:
        best_h, best_c5 = coord_descent(best_h, len(best_h), [0.0, 1.0], 2, remaining() - 5)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
