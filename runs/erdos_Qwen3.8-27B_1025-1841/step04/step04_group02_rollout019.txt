Hmm, let me just think of it as: the output at index m (0 to 2N-2) represents the correlation at lag (m - (N-1)).

For the correlation at lag L (where L can be negative):
corr[L] = sum_i h[i] * (1-h[i+L]) for valid i (where both i and i+L are in [0, N-1])

∂corr[L]/∂h[j]:
- h[j] appears in h[i] when i = j: contributes (1-h[j+L])
- h[j] appears in h[i+L] when i+L = j, i.e., i = j-L: contributes -h[j-L]

This is the subgradient for the active lag L*.

For the max, the subgradient is the gradient of the active lag (or convex combination of active lags).

h[j] ← clip(h[j] - η * g[j], 0, 1)

import numpy as np
from scipy.optimize import minimize
import time

_G = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        mask = h < 1.0 if diff > 0 else h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_fn(x):
    n = _G['n']; T = _G['T']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    return float((mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx)

def con_fn(x):
    return np.sum(x) - len(x) / 2.0

def con_jac(x):
    return np.ones(len(x))

def do_slsqp(h, T, maxiter, ftol):
    n = len(h)
    _G['n'] = n; _G['T'] = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fn, 'jac': con_jac}]
    res = minimize(obj_fn, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    b = (h >= 0.5).astype(float)
    t = n // 2

        oi = np.where(b == 1)[0]
        if len(oi) == 0: break
        b[oi[np.argmin(h[oi])]] = 0

        if len(zi) == 0: break
        b[zi[np.argmax(h[zi])]] = 1
    return b

            h_hi = make_feasible(np.interp(nx, ox, bh), n_hi)
            b_hi = to_bin(h_hi, n_hi)
            b_hi, c = sa_search(b_hi, n_hi, 150000, 0.008, 1e-5, rng, st, budget_s)
            if c < bc:
                bc = c; bh = b_hi.copy()
            if _rem(st, budget_s) > 150:
                try:
                    x = do_slsqp(bh, 200, 400, 1e-15)
                    c = compute_c5(x)
                    if c < bc: bc = c; bh = make_feasible(x, len(bh))
                except: pass

    # Phase 8: Back to n0 resolution with best structure
    if _rem(st, budget_s) > 300 and len(bh) != n0:
        ox = np.linspace(0, 2, len(bh))
        nx = np.linspace(0, 2, n0)
        h_mid = make_feasible(np.interp(nx, ox, bh), n0)
        b_mid = to_bin(h_mid, n0)
        b_mid, c = targeted_sa(b_mid, n0, 300000, 0.006, 1e-6, rng, st, budget_s)
        if c < bc:
            bc = c; bh = b_mid.copy()
        if _rem(st, budget_s) > 120:
            try:
                x = do_slsqp(bh, 500, 400, 1e-15)
                c = compute_c5(x)
                if c < bc: bc = c; bh = make_feasible(x, n0)
            except: pass

    # Phase 9: Coordinate descent
    if _rem(st, budget_s) > 80:
        bh, bc = coord_search(bh.copy(), len(bh), 50, st, budget_s)

    # Phase 10: Final SLSQP
    if _rem(st, budget_s) > 80:
        try:
            x = do_slsqp(bh, 800, 400, 1e-16)
            c = compute_c5(x)
            if c < bc: bc = c; bh = make_feasible(x, len(bh))
        except: pass

    # Phase 11: Final targeted SA
    if _rem(st, budget_s) > 100:
        b = to_bin(bh, len(bh))
        b, c = targeted_sa(b, len(b), 200000, 0.003, 1e-7, rng, st, budget_s)
        if c < bc: bc = c; bh = b.copy()
        if _rem(st, budget_s) > 60:
            try:
                x = do_slsqp(bh, 1000, 300, 1e-16)
                c = compute_c5(x)
                if c < bc: bc = c; bh = make_feasible(x, len(bh))
            except: pass

    # Phase 12: Final coordinate descent
    if _rem(st, budget_s) > 20:
        bh, bc = coord_search(bh.copy(), len(bh), 20, st, budget_s)

    bh = make_feasible(bh, len(bh))
    bc = compute_c5(bh)
    return bh, bc, len(bh)
