import numpy as np
from scipy.optimize import minimize
import time

_SSL_N = 0
_SSL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    h[np.sum(h) < target] = np.minimum(h[np.sum(h) < target], 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_sa(h, n, max_iters, T0, T1, rng, block_sizes=None):
    if block_sizes is None:
        block_sizes = [1, 2, 3, 4, 6, 10, 16]
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        if T < 1e-11:
            T = 1e-11
        r = rng.random()
        if r < 0.45:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0; b[j] = 1
            new_c5 = compute_c5(b)
            accept = new_c5 <= curr_c5 or (rng.random() < np.exp(-(new_c5 - curr_c5) / T))
            if accept:
                curr_c5 = new_c5
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_b = b.copy()
            else:
                b[i] = 1; b[j] = 0
        else:
            bs = block_sizes[rng.randint(len(block_sizes))]
            if bs >= n:
                bs = n // 2
            start = rng.randint(0, n - bs + 1)
            end = start + bs
            seg = b[start:end]
            oi = np.where(seg == 1)[0]
            zi = np.where(seg == 0)[0]
            if len(oi) == 0 or len(zi) == 0:
                continue
            sw = min(len(oi), len(zi), bs // 2)
            if sw == 0:
                continue

            cz = rng.choice(zi, size=sw, replace=False)

            else:
                b[start + co] = 1
                b[start + cz] = 0
    return best_b, best_c5

def soft_obj(x):
    n = _SSL_N
    T = _SSL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_opt(h, n, T, maxiter, ftol):
    global _SSL_N, _SSL_T
    _SSL_N = n
    _SSL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(soft_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def targeted_local_search(h, n, start_time, budget_s, max_sweeps=50):
    """Target the worst lags and try to reduce them via local moves."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for sweep in range(max_sweeps):
        if time.time() - start_time > budget_s - 15:
            break
        n_cur = len(best_h)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        # Find the lags with highest corr
        top_lags = np.argsort(corr)[-20:]
        improved = False
        for lag in top_lags:
            if time.time() - start_time > budget_s - 10:
                break
            lag_idx = int(lag)
            # lag in full correlation: index lag_idx corresponds to shift (lag_idx - (n-1))
            shift = lag_idx - (n_cur - 1)
            if shift <= 0 or shift >= n_cur:
                continue
            # Try to reduce overlap at this shift
            # Find positions i where h[i]=1 and h[i+shift]=0
            valid_i = np.arange(n_cur - shift)
            overlap_positions = valid_i[(best_h[valid_i] > 0.5) & (best_h[valid_i + shift] < 0.5)]
            if len(overlap_positions) == 0:
                continue
            # Try moving one of these
            for i in overlap_positions[:5]:
                if time.time() - start_time > budget_s - 10:
                    break
                # Try setting h[i] toward 0
                trial = best_h.copy()
                trial[i] = 0.0
                trial = make_feasible(trial, n_cur)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    break
                # Try moving a 1 from position i to somewhere else
                candidates = np.where(best_h < 0.5)[0]
                if len(candidates) == 0:
                    break
                for j in candidates[rng.randint(0, len(candidates), size=min(3, len(candidates)))]:
                    trial2 = best_h.copy()
                    trial2[i] = 0.0
                    trial2[j] = 1.0
                    trial2 = make_feasible(trial2, n_cur)
                    c2 = compute_c5(trial2)
                    if c2 < best_c5 - 1e-15:
                        best_c5 = c2
                        best_h = trial2
                        improved = True
                        break
                if improved:
                    break
        if not improved:
            break
    return best_h, best_c5
