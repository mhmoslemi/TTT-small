def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0

    h = h_bin.copy()

    zeros = np.where(h <= 0.5)[0].copy()

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def lbgfb_smooth(h0, n, T, mu, lam, max_iter):
    dx = 2.0 / n
    def f(h):
        c = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(c)
        s = np.clip(T * (c - mc), -100, 100)
        soft = mc + np.log(np.sum(np.exp(s))) / T
        v = np.sum(h) - n / 2.0
        bp = lam * np.sum(h * (1.0 - h))
        return (soft + mu * v * v + bp) * dx
    bnds = [(0.0, 1.0)] * n
    r = minimize(f, h0, method='L-BFGS-B', bounds=bnds,
                 options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14, 'maxls': 40})
    return make_feasible(r.x, n)

def bin_search(hb, n, rng, rounds, tries):
    h = hb.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for r in range(rounds):
        imp = False
        for _ in range(tries):
            i1 = rng.randint(len(ones))
            i0 = rng.randint(len(zeros))
            o1, o0 = ones[i1], zeros[i0]
            h[o1] = 0.0; h[o0] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones = np.delete(ones, i1); zeros = np.delete(zeros, i0)
                ones = np.append(ones, o0); zeros = np.append(zeros, o1)
                imp = True
            else:
                h[o1] = 1.0; h[o0] = 0.0
        if not imp:
            break
    return h, cur
