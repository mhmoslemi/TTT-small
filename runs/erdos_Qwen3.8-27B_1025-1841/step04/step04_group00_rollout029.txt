- If max(active_slopes) < 0: f is decreasing at t, move lo up

```python
def minimax_1d(a, b, lo=0.0, hi=1.0, tol=1e-14):
    """Minimize max(a + b*t) over t in [lo, hi]. Returns (t_opt, f_opt)."""
    for _ in range(100):
        t = (lo + hi) / 2.0
        vals = a + b * t
        f_val = np.max(vals)
        # Active set: indices where vals is close to f_val
        active = vals >= f_val - 1e-12
        active_slopes = b[active]
        if len(active_slopes) == 0:
            break
        min_slope = np.min(active_slopes)
        max_slope = np.max(active_slopes)
        if min_slope <= 0 <= max_slope:
            break
        elif min_slope > 0:
            hi = t
        else:
            lo = t
        if hi - lo < tol:
            break
    t_opt = (lo + hi) / 2.0
    f_opt = float(np.max(a + b * t_opt))
    return t_opt, f_opt
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(np.float64).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = int(np.argmax(1.0 - h))
        else:
            idx = int(np.argmin(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def minimax_1d(a, b):
    """Minimize max(a + b*t) over t in [0,1] using subgradient binary search."""
    lo = 0.0
    hi = 1.0
    for _ in range(80):
        t = (lo + hi) * 0.5
        vals = a + b * t
        f_val = np.max(vals)
        active = vals >= f_val - 1e-13
        if not np.any(active):
            break
        slopes = b[active]
        smin = np.min(slopes)
        smax = np.max(slopes)
        if smin <= 0.0 <= smax:
            break
        elif smin > 0.0:
            hi = t
        else:
            lo = t
        if hi - lo < 1e-15:
            break
    t_opt = (lo + hi) * 0.5
    f_opt = float(np.max(a + b * t_opt))
    return t_opt, f_opt

def coord_search(h, n, max_time, start_time, n_sweeps_limit=20000):
    """Fast coordinate descent with exact 1D minimax per coordinate."""
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    shifts = np.arange(1 - n, n, dtype=np.int64)
    best_c5 = float(np.max(corr)) * (2.0 / n)
    best_h = h.copy()

    sweep = 0
    while sweep < n_sweeps_limit:
        if time.time() - start_time > max_time - 15:
            break
        sweep += 1
        # Recompute correlation periodically to avoid drift
        if sweep % 20 == 0:
            corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

        improved = False
        cur_max = float(np.max(corr))
        order = np.arange(n) if sweep % 2 == 1 else np.arange(n - 1, -1, -1)

        for j in order:
            jp = j + shifts
            jm = j - shifts
            delta_corr = np.zeros(m, dtype=np.float64)
            mp = (jp >= 0) & (jp < n)
            mm = (jm >= 0) & (jm < n)
            delta_corr[mp] -= h[jp[mp]]
            delta_corr[mm] += 1.0 - h[jm[mm]]

            old_val = h[j]
            a = corr - old_val * delta_corr
            b = delta_corr

            t_opt, f_opt = minimax_1d(a, b)

            if f_opt < cur_max - 1e-13:
                delta = t_opt - old_val
                corr += delta * b
                h[j] = t_opt
                cur_max = f_opt
                improved = True

        new_c5 = float(np.max(corr)) * (2.0 / n)
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            best_h = h.copy()

        if not improved:
            break

    h = make_feasible(h, n)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
```
