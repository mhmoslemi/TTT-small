import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n//2]] = 1.0
    return h_bin

def efficient_binary_search(h_bin, n, rng, max_trials=100000):
    """Random pair swap local search with O(n) correlation update."""
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr))
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    offset = n - 1
    k_all = np.arange(-(n-1), n)
    
    accepted = 0
    for trial in range(max_trials):
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        i = ones[ii]
        j = zeros[jj]
        
        # Compute delta
        m1 = (i - k_all >= 0) & (i - k_all < n)
        m2 = (i + k_all >= 0) & (i + k_all < n)
        m3 = (j - k_all >= 0) & (j - k_all < n)
        m4 = (j + k_all >= 0) & (j + k_all < n)
        
        delta = np.zeros(2*n-1)
        if m1.any():
            delta[k_all[m1] + offset] -= 1.0 - h[i - k_all[m1]]
        if m2.any():
            delta[k_all[m2] + offset] += h[i + k_all[m2]]
        if m3.any():
            delta[k_all[m3] + offset] += 1.0 - h[j - k_all[m3]]
        if m4.any():
            delta[k_all[m4] + offset] -= h[j + k_all[m4]]
        
        new_max = np.max(corr + delta)
        if new_max < current_max - 1e-15:
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + delta
            current_max = new_max
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            accepted += 1
        # No else: h unchanged, corr unchanged
    
    return h, current_max
