def slsqp_softmax(h, n, T, maxiter, ftol, start_time, time_limit):
    """SLSQP with soft-max objective."""
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    
    def obj(x):
        if time.time() - start_time > time_limit:
            return np.inf
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def con_sum(x):
        return np.sum(x) - n / 2.0
    
    def con_sum_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

    step = step_init
    ave_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = get_corr(h)
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)  # the lag
        
        # Subgradient of corr_k
        grad = np.zeros(n)
        if k >= 0:
            # d(corr_k)/d(h[j]) = (1-h[j+k]) for j in [0, n-1-k]
            grad[:n-k] += (1 - h[k:])
            # d(corr_k)/d(h[j]) -= h[j-k] for j in [k, n-1]
            grad[k:] -= h[:n-k]
        else:

            # d(corr_k)/d(h[j]) = (1-h[j-kk]) for j in [kk, n-1]  (i.e., j+(-kk) in range)
            # Wait, for negative lag k = -kk:

            #   If j <= n-1-kk: contributes -h[j+kk] from the second term (h[j] appears as h[i-kk] when i = j+kk)
            grad[kk:] += (1 - h[:n-kk])
            grad[:n-kk] -= h[kk:]

        h = h - step * dx * grad
        # Project: clip to [0,1] and adjust sum
        h = np.clip(h, 0.0, 1.0)

        # Averaging
        ave_h = (it * ave_h + h) / (it + 1)

            best_h = h.copy()

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):

        else:

        if cnt == 0:

def subgrad_polish(h, n, max_iters, start_time, time_limit):
    """Subgradient method for continuous h."""
    h = h.copy().astype(float)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 0.1
    
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        
        corr = get_corr(h)
        max_idx = np.argmax(corr)
        k = max_idx - (n - 1)
        
        # Subgradient
        grad = np.zeros(n)
        if k >= 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:n-k]
        else:
            kk = -k
            grad[kk:] += (1.0 - h[:n-kk])
            grad[:n-kk] -= h[kk:]
        
        # Step
        h = h - step * grad
        h = np.clip(h, 0.0, 1.0)
        # Fix sum
        diff = n/2.0 - np.sum(h)
        if abs(diff) > 1e-10:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
        
        c_curr = compute_c5(h)
        if c_curr < best_c5:
            best_c5 = c_curr
            best_h = h.copy()
        
        step = 0.1 / (1.0 + 0.005 * it)
    
    return best_h, best_c5
