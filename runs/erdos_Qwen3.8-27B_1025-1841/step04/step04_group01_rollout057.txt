    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = h.min() - 1.0, h.max() + 1.0

∂c_{-s'}/∂h[k]:
- If k in [s', n-1]: contributes (1-h[k-s'])
- If k+s' in [s', n-1], i.e., k in [0, n-1-s']: contributes -h[k+s']

So:
∂c_{-s'}/∂h[k] = (1-h[k-s']) * [k ≥ s'] - h[k+s'] * [k ≤ n-1-s']

OK so in general, the gradient of c_s w.r.t. h is a shifted version of (1-h) minus a shifted version of h. This can be computed in O(n).

Now, for the full soft-max gradient:
∂F/∂h[k] = dx * Σ_s w_s * ∂c_s/∂h[k]

where w_s are the softmax weights.

If I only use the top-K shifts (top_idx), then:
∂F/∂h[k] = dx * Σ_{s ∈ top_idx} w_s * ∂c_s/∂h[k]

This is O(K*n) which for K=20, n=960 is ~20,000 operations. Very fast!

So I can do projected gradient descent with analytical gradients! This should be much faster than SLSQP.

**Algorithm: Projected Gradient Descent with Soft-Max**

```
for iteration:
    1. Compute corr, find top-K shifts
    2. Compute softmax weights w_s
    3. Compute gradient: g = Σ_s w_s * grad_c_s
    4. Line search: find α such that F(h - α*g) < F(h)
    5. Project: h = project_box_hyp(h - α*g)
    6. Check convergence
```

    target = n / 2.0
    lo = h.min() - 1.0

import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

        if mask.any():

    if s >= 0:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    else:
        sp = -s
        g[sp:] += (1.0 - h[:n - sp])
        g[:n - sp] -= h[sp:]
    return g

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
    return h

def pgd_optimize(h, n, T, top_K, max_iters, lr_init, remaining_fn):
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    best_local = compute_c5(h)
    best_local_h = h.copy()
    no_improve = 0
    
    for it in range(max_iters):
        if remaining_fn() < 10:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        if top_K < 2 * n - 1:
            top_idx = np.argsort(corr)[-top_K:]
        else:
            top_idx = np.arange(2 * n - 1)

        mc = top_corr.max()
        exp_c = np.exp(T * (top_corr - mc))
        sum_exp = exp_c.sum()
        weights = exp_c / sum_exp
        fval = (mc + np.log(sum_exp) / T) * dx

        g = np.zeros(n)
        for k in range(len(top_idx)):
            s = int(top_idx[k]) - (n - 1)
            g += weights[k] * grad_corr_h(h, s, n)
        g *= dx

        v = beta * v - lr * g
        h_candidate = project_box_hyp(h + v, n)
        
        # Check improvement
        f_candidate = compute_c5(h_candidate)
        if f_candidate < best_local - 1e-15:
            h = h_candidate
            best_local = f_candidate
            best_local_h = h_candidate.copy()
            no_improve = 0
        else:
            no_improve += 1
            # Backtrack
            v *= 0.5
            if no_improve > 20:
                lr *= 0.7
                v = -lr * g
                no_improve = 0
            # Still try the step

            f_candidate = compute_c5(h_candidate)
            if f_candidate < best_local:

                best_local = f_candidate
                best_local_h = h_candidate.copy()

def slsqp_phase(h, n, T, top_idx, max_iter, remaining_fn):
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c = corr[top_idx]
        mc = c.max()
        shifted = c - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.success or res.fun < obj(h):
            res.x = make_feasible(res.x, n)
            return res.x
    except Exception:
        pass
    return h

def coord_descent(h, n, steps_list, remaining_fn):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    
    for j in range(n):
        if remaining_fn() < 5:
            break
        old_val = h[j]
        best_val = old_val
        best_c = best_c5
        
        for step in steps_list:
            for sign in [-1.0, 1.0]:
                v = old_val + sign * step
                if 0.0 <= v <= 1.0:
                    trial = h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-16:
                        best_c = c
                        best_val = v
        if best_val != old_val:
            h[j] = best_val
            h = make_feasible(h, n)
            best_c5 = best_c
    
    return h, best_c5

def pair_swap_optimize(h, n, remaining_fn, max_rounds=5):
    """Try pair transfers to reduce C5"""
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(h)
    eps_vals = [0.001, 0.002, 0.005, 0.01, 0.02]
    
    for rd in range(max_rounds):
        if remaining_fn() < 10:
            break
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Focus on indices where h is "extreme" (close to 0 or 1)
        # and also mixed
        for j in range(n):
            if remaining_fn() < 5:
                break
            h_j = h[j]
            # Find candidate partners
            for eps in eps_vals:
                # Transfer from j to another index
                # Try a few random partners
                for _ in range(3):
                    k = int(np.random.randint(n))
                    if k == j:
                        continue
                    trial = h.copy()
                    if h_j > 0.5 and h[k] < 0.5:
                        # Move mass from j to k
                        d = min(eps, h_j - 0.0, 1.0 - h[k])
                        trial[j] = h_j - d
                        trial[k] = h[k] + d
                    elif h_j < 0.5 and h[k] > 0.5:
                        d = min(eps, 1.0 - h_j, h[k])
                        trial[j] = h_j + d
                        trial[k] = h[k] - d
                    else:
                        continue
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        h = trial
                        best_c5 = c
                        improved = True
