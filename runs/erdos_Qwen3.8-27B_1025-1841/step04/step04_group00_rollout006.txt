def make_feasible(h, n):

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

    """Returns (smooth_max_value * dx, gradient)."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    w = np.exp(T * (corr - cmax))
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.zeros_like(corr)
        w[int(np.argmax(corr))] = 1.0
    else:
        w /= wsum

    """L-BFGS-B with penalty on integral constraint."""
    def obj(x):

    return h, compute_c5(h)

                        trial[j] = v

                        c = compute_c5(trial)
                        if c < best_c - 1e-13:
                            best_c = c
                            best_val = v
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Nesterov PGD moderate T
    if rem() > 200:
        t_alloc = min(180, rem() - 20)
        h, c5 = nesterov_pgd(best_h, n, 50, t_alloc, start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: L-BFGS-B
    if rem() > 150:
        t_alloc = min(120, rem() - 20)
        h, c5 = lbfgsb_opt(best_h, len(best_h), 100, 50.0, t_alloc, start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Nesterov higher T
    if rem() > 180:
        t_alloc = min(160, rem() - 20)
        h, c5 = nesterov_pgd(best_h, len(best_h), 200, t_alloc, start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: L-BFGS-B again with higher T
    if rem() > 120:
        t_alloc = min(100, rem() - 15)
        h, c5 = lbfgsb_opt(best_h, len(best_h), 300, 100.0, t_alloc, start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Upsample to 2n
    n2 = n * 2
    if rem() > 350 and n2 <= 2560:
        old_x = np.linspace(0, 2, n, endpoint=False)
        new_x = np.linspace(0, 2, n2, endpoint=False)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n2)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.003:
            if rem() > 150:
                h2, c5_2 = nesterov_pgd(h_up, n2, 50, min(120, rem() - 20), start_time)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2
            if rem() > 100:
                h2, c5_2 = lbfgsb_opt(best_h, len(best_h), 100, 50, min(80, rem() - 15), start_time)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2
            if rem() > 80:
                h2, c5_2 = nesterov_pgd(best_h, len(best_h), 200, min(60, rem() - 15), start_time)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2
            n = len(best_h)

    # Phase 6: Coordinate descent polish
    if rem() > 100:
        h, c5 = coord_polish(best_h, len(best_h), min(80, rem() - 10), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Another Nesterov round
    if rem() > 100:
        h, c5 = nesterov_pgd(best_h, len(best_h), 400, min(80, rem() - 10), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 8: L-BFGS-B final
    if rem() > 60:
        h, c5 = lbfgsb_opt(best_h, len(best_h), 500, 200.0, min(50, rem() - 10), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 9: Random perturbations + refine
    for i in range(4):
        if rem() < 40:
            break
        n_pts = len(best_h)
        scale = 0.008 if i < 2 else 0.015
        h_try = make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
        h2, c5_2 = nesterov_pgd(h_try, n_pts, 100, min(25, rem() - 10), start_time)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 10: Final coord polish
    if rem() > 30:
        h, c5 = coord_polish(best_h, len(best_h), min(25, rem() - 5), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 11: Final Nesterov
    if rem() > 25:
        h, c5 = nesterov_pgd(best_h, len(best_h), 500, rem() - 5, start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
