        return h_new, c5_new
    except Exception:
        return h, compute_c5(h)

def subgrad_hf(h, n, start_time, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    step = 0.003
    it = 0
    while time.time() - start_time < budget_s:
        progress = min(1.0, it / 3000)
        T = 50.0 + 3000.0 * progress
        step = max(0.001, 0.003 * (1.0 - progress))
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        cx = np.max(corr)
        exp_v = np.exp((corr - cx) * T)
        weights = exp_v / np.sum(exp_v)
        
        # Focus on top shifts
        threshold = np.max(weights) * 0.005
        active = np.where(weights > threshold)[0]
        if len(active) == 0:
            active = np.array([np.argmax(weights)])
        elif len(active) > 50:
            active = np.argsort(weights)[-50:]
        
        w_sub = weights[active]
        w_sub = w_sub / np.sum(w_sub)
        
        grad = np.zeros(n)
        omh = 1.0 - h
        for m in range(len(active)):
            w = w_sub[m]
            if w < 1e-7:
                continue
            k = active[m]
            s = k - (n - 1)
            if s >= 0:
                idx1 = np.arange(s, n)
                grad[idx1] += w * omh[idx1 - s]
                idx2 = np.arange(0, n - s)
                grad[idx2] -= w * h[idx2 + s]
            else:
                sp = -s
                idx1 = np.arange(0, n - sp)
                grad[idx1] += w * omh[idx1 + sp]
                idx2 = np.arange(sp, n)
                grad[idx2] -= w * h[idx2 - sp]
        
        h_new = h - step * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if c5_new <= c5_cur + 1e-15:
            h = h_new
            c5_cur = c5_new
        else:
            step *= 0.95
            if step < 5e-4:
                step = 0.001
        
        it += 1
    
    return best_h, best_c5

def sa_hf(h, n, start_time, budget_s, seed=42):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_cur = best_c5
    rng = np.random.RandomState(seed)
    T = 0.002
    cooling = 0.99998
    min_T = 1e-9
    
    it = 0
    stall = 0
    while time.time() - start_time < budget_s:
        it += 1
        if stall > 20000:
            break
        
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        r = rng.rand()
        if r < 0.5:
            delta = rng.uniform(0.0005, 0.005)
        elif r < 0.8:
            delta = rng.uniform(0.003, 0.02)
        else:
            delta = rng.uniform(0.01, 0.05)
        
        if rng.rand() < 0.5:
            di, dj = delta, -delta
        else:
            di, dj = -delta, delta
        
        vi = h[i] + di
        vj = h[j] + dj
        if not (0.0 <= vi <= 1.0 and 0.0 <= vj <= 1.0):
            continue
        
        h[i] = vi
        h[j] = vj
        c5_new = compute_c5(h)
        dc = c5_new - c5_cur
        
        if dc < 0:
            c5_cur = c5_new
            stall = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            if T > min_T and rng.rand() < np.exp(-dc / T):
                c5_cur = c5_new
                stall += 1
            else:
                h[i] -= di
                h[j] -= dj
        
        T = max(min_T, T * cooling)
    
    return best_h, best_c5

def coord_hf(h, n, start_time, budget_s):
    """Coordinate descent with binary search."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved_total = True
    while improved_total and time.time() - start_time < budget_s:
        improved_total = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                return best_h, best_c5
            old = best_h[j]
            best_v = old
            best_c = best_c5
            # Try perturbations
            for d in [0.005, 0.002, 0.001, 0.0005]:
                for s in [-1, 1]:
                    v = old + s * d
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_v = v
            # Try extreme
            for v in [0.0, 1.0]:
                if abs(v - old) > 0.01:
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_v = v
            if best_v != old:
                best_h = make_feasible(best_h.copy(), n)
                best_h[j] = best_v
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved_total = True
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    def rem():
        return budget_s - (time.time() - t0)
    
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)
    n = len(h)
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Quick SLSQP
    if rem() > 80:
        h, c5 = slsqp_smooth(h, n, T=100, maxiter=80)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 2: Subgradient
    if rem() > 250:
        h, c5 = subgrad_hf(best_h, n, t0, min(250, rem() - 10))
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 3: SLSQP with higher T
    if rem() > 80:
        h, c5 = slsqp_smooth(best_h, n, T=300, maxiter=100)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 4: SA (main)
    if rem() > 300:
        h, c5 = sa_hf(best_h, n, t0, min(350, rem() - 10), seed=seed)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 5: Coord descent
    if rem() > 80:
        h, c5 = coord_hf(best_h, n, t0, min(60, rem() - 10))
        if c5 < best_c5:
            best_h, best_c5 = h, c5
    
    # Phase 6: SA round 2
    if rem() > 150:
        h, c5 = sa_hf(best_h, n
