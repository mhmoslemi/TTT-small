    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def get_marginal_effects(h, s, n):
    """Compute marginal effect m[j] on corr[s] when changing h[j] by delta=+1."""
    m = np.zeros(n)
    for j in range(n):
        # h[j] appears as first factor: n+s=j => n=j-s, need 0<=j-s<n
        # h[j] appears as (1-h[n]) factor: n=j, need 0<=j+s<n (contribution is -h[j+s])
        effect = 0.0
        if 0 <= j - s < n:
            effect += (1.0 - h[j - s])
        if 0 <= j + s < n:
            effect += (-h[j + s])
        m[j] = effect
    return m

def directed_local_search(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_idx = int(np.argmax(corr))
        s = worst_idx - (n - 1)
        m = get_marginal_effects(h, s, n)
        # Best j: if m[j]>0, decrease h[j]; if m[j]<0, increase h[j]
        # Score: |m[j]| * avail_slack in that direction
        scores = np.zeros(n)
        for j in range(n):
            if m[j] > 0:
                if h[j] > 1e-6:
                    scores[j] = m[j] * h[j]
            else:
                if h[j] < 1.0 - 1e-6:
                    scores[j] = -m[j] * (1.0 - h[j])
        if np.max(scores) < 1e-8:
            break
        trial_best = best_c5
        trial_h = h.copy()
        # Try top candidates
        top_j = np.argsort(scores)[::-1][:min(20, n)]
        for j in top_j:
            if trial_best >= best_c5 - 1e-12:
                if m[j] > 0:
                    delta = -min(h[j], 0.05)
                else:
                    delta = min(1.0 - h[j], 0.05)
                trial = h.copy()
                trial[j] += delta
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < trial_best:
                    trial_best = c
                    trial_h = trial.copy()
        if trial_best < best_c5 - 1e-12:
            h = trial_h
            best_c5 = trial_best
        else:
            # Try smaller steps
            improved = False
            for j in top_j[:10]:
                if m[j] > 0:
                    delta = -min(h[j] * 0.3, 0.01)
                else:
                    delta = min((1.0 - h[j]) * 0.3, 0.01)
                trial = h.copy()
                trial[j] += delta
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    h = trial
                    best_c5 = c
                    improved = True
                    break
            if not improved:
                break
    return h, best_c5

def pair_swap_local_search(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        worst_idx = int(np.argmax(corr))
        s = worst_idx - (n - 1)
        # Find pairs (i, j) where swapping values would help
        # Try: for the worst shift, find i where h[i]*(1-h[i+s]) is large
        # and j where h[j]*(1-h[j+s]) is small, then swap their "excess"
        improved = False
        for _ in range(5):
            # Random pair swap
            i = np.random.randint(0, n)
            j = np.random.randint(0, n)
            if i == j:
                continue
            di = np.random.choice([-0.1, -0.05, 0.05, 0.1])
            dj = -di  # keep sum roughly constant
            trial = h.copy()
            trial[i] = np.clip(trial[i] + di, 0, 1)
            trial[j] = np.clip(trial[j] + dj, 0, 1)
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-12:
                h = trial
                best_c5 = c
                improved = True
        if not improved and rnd > 10:
            break
    return h, best_c5

def diverse_perturb(h, n, rng, method='mixed'):
    h_new = h.copy()
    if method == 'fourier':
        x = np.arange(n)
        n_modes = rng.randint(2, 6)
        for _ in range(n_modes):
            freq = rng.randint(1, n // 8)
            phase = rng.uniform(0, 2 * np.pi)
            amp = rng.uniform(0.02, 0.08)
            h_new += amp * np.sin(2 * np.pi * freq * x / n + phase)
    elif method == 'block':
        n_blocks = rng.randint(1, 5)
        for _ in range(n_blocks):
            blen = rng.randint(n // 20, n // 5)
            start = rng.randint(0, n - blen)
            if rng.random() < 0.4:
                h_new[start:start+blen] = 1.0 - h_new[start:start+blen]
            else:
                h_new[start:start+blen] += rng.uniform(-0.15, 0.15)
    elif method == 'blink':
        n_blinks = rng.randint(5, 30)
        for _ in range(n_blinks):
            idx = rng.randint(0, n)
            h_new[idx] = 1.0 - h_new[idx]
    elif method == 'gaussian':
        sigma = rng.uniform(0.03, 0.12)
        h_new += rng.normal(0, sigma, size=n)
    elif method == 'mixed':
        r = rng.random()
        if r < 0.3:
            return diverse_perturb(h, n, rng, 'fourier')
        elif r < 0.6:
            return diverse_perturb(h, n, rng, 'block')
        elif r < 0.8:
            return diverse_perturb(h, n, rng, 'blink')
        else:
            return diverse_perturb(h, n, rng, 'gaussian')
    return make_feasible(h_new, n)

def multi_temp_slsqp(h, n, start_time, budget_s, temps=None):
    if temps is None:
        temps = [50.0, 200.0, 800.0, 3000.0, 10000.0, 50000.0]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        maxiter = min(200, int(remaining / 0.5))
        if maxiter < 20:
            break
        x_res = slsqp_optimize(best_h, n, T, maxiter, 1e-14)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = make_feasible(x_res, n)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Aggressive multi-start SLSQP at n0
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s - 150)

    # Phase 2: Directed local search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        tl = min(remaining - 50, 300)
        best_h, c5_d = directed_local_search(best_h, n0, tl, start_time, budget_s - 50)
        if c5_d < best_c5:
            best_c5 = c5_d

    # Phase 3: Multiple diverse restarts
    methods = ['fourier', 'block', 'blink', 'gaussian', 'fourier', 'block']
    for idx, m in enumerate(methods):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 120:
            break
        hp = diverse_perturb(best_h, n0, rng, method=m)
        hr, cr = multi_temp_slsqp(hp, n0, start_time, budget_s - 60,
                                   temps=[100.0, 500.0, 2000.0, 8000.0])
        if cr < best_c5:
            best_c5 = cr
            best_h = hr.copy()
            # Follow up with directed search
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                best_h, c5_d = directed_local_search(best_h, n0, 100,
                                                      start_time, budget_s - 30)
                if c5_d < best_c5:
                    best_c5 = c5_d

    # Phase 4: Upward migration to higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n1 = n0 * 2
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        # Quick SLSQP at higher res with fewer temps
        temps_up = [200.0, 1000.0, 5000.0]
        h_up, c5_up = multi_temp_slsqp(h_up, n1, start_time, budget_s - 80, temps=temps_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            # Directed search at higher res
            remaining = budget_s - (time.time() - start_time)
            if remaining > 80:
                best_h, c5_d = directed_local_search(best_h, n1, 80,
                                                      start_time, budget_s - 30)
                if c5_d < best_c5:
                    best_c5 = c5_d
        # Also try binary at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 150:
            b = (best_h >= 0.5).astype(float)
            b = make_feasible(b, n1)
            c5_b = compute_c5(b)
            # Pair swap SA on binary
            T0, T1 = 0.003, 0.00001
            curr_c5 = c5_b
            best_b = b.copy()
            max_iters = int(min(remaining - 40, 20000))
            for it in range(max_iters):
                if time.time() - start_time > budget_s - 40:
                    break
                frac = float(it) / max_iters
                T = T0 * (T1 / T0) ** frac
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                if len(ones) == 0 or len(zeros) == 0:
                    break
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                b[i] = 0
                b[j] = 1
                new_c5 = compute_c5(b)
                if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
                    curr_c5 = new_c5
                    if new_c5 < c5_b:
                        c5_b = new_c5
                        best_b = b.copy()
                else:
                    b[i] = 1
                    b[j] = 0
            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = best_b.copy()
                # Refine with SLSQP
                remaining = budget_s - (time.time() - start_time)
                if remaining > 60:
                    best_h, c5_r = multi_temp_slsqp(best_h, n1, start_time,
                                                     budget_s - 20, temps=[100.0, 500.0, 2000.0])
                    if c5_r < best_c5:
                        best_c5 = c5_r

    # Phase 5: Final intensive directed + SLSQP at best resolution
    remaining = budget_s - (time.time() - start_time)
    n_final = len(best_h)
    if remaining > 60:
        best_h, c5_d = directed_local_search(best_h, n_final,
                                              int(min(remaining - 20, 200)),
                                              start_time, budget_s - 20)
        if c5_d < best_c5:
            best_c5 = c5_d

    # Phase 6: Final SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_final = len(best_h)
        x_res = slsqp_optimize(best_h, n_final, 20000.0, 150, 1e-15)
        c5_res = compute_c5(x_res)
        if c5_res < best_c5:
            best_c5 = c5_res
            best_h = make_feasible(x_res, n_final)

    # Phase 7: Final directed search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_final = len(best_h)
        best_h, c5_d = directed_local_search(best_h, n_final,
                                              int(min(remaining - 5, 100)),
                                              start_time, budget_s - 5)
        if c5_d < best_c5:
            best_c5 = c5_d

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
