output[m+(n-1)] = Σ_n h[n+m] * (1-h[n])

This is the cross-correlation at lag m. For m ≥ 0:
output[m+(n-1)] = Σ_{n=0}^{n-1-m} h[n+m] * (1-h[n])

For m < 0, let m' = -m:
output[(n-1)-m'] = Σ_{n=m'}^{n-1} h[n] * (1-h[n-m'])

OK so the key point is: for lag m (m from -(n-1) to n-1), the correlation is:
corr[m] = Σ_n h[n+m] * (1-h[n]) (for valid n)

And ∂corr[m]/∂h[j]:
- h[j] appears as h[n+m] when n+m=j, i.e., n=j-m: contributes (1-h[j-m])
- h[j] appears in (1-h[n]) when n=j: contributes -h[j+m]

So ∂corr[m]/∂h[j] = (1-h[j-m]) * 𝟙[0≤j-m<n] + (-h[j+m]) * 𝟙[0≤j+m<n]
                  = (1-h[j-m]) * 𝟙[0≤j-m<n] - h[j+m] * 𝟙[0≤j+m<n]

And the objective is max_m corr[m] * dx.

For the softmax smooth:
f(h) = (max_m corr[m]*dx + log(Σ_m exp(T*(corr[m]*dx - max_m corr[m]*dx)))/T)

∂f/∂h[j] = (1/T) * Σ_m exp(T*corr[m]*dx) * ∂(corr[m]*dx)/∂h[j] / Σ_m exp(T*corr[m]*dx)

where p[m] = exp(T*corr[m]*dx) / Σ_m exp(T*corr[m]*dx)

import time

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_grad(h, n, T):
    """Compute gradient of softmax-smoothed C5."""
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    dx = 2.0 / n
    corr_s = corr * dx
    max_c = np.max(corr_s)
    ex = T * (corr_s - max_c)
    ex -= np.max(ex)  # numerical stability
    p = np.exp(ex)
    p_sum = np.sum(p)
    p /= p_sum
    
    grad = np.ones(n)
    r = n - 1
    for m in range(2*n - 1):
        if p[m] < 1e-16:
            continue
        k = m - r
        # -p[m] * h[j-k] where 0 <= j-k < n => max(0,k) <= j < min(n, n+k)
        lo = max(0, k)
        hi = min(n, n + k)
        if lo < hi:
            grad[lo:hi] -= p[m] * h[lo-k:hi-k]
        # -p[m] * h[j+k] where 0 <= j+k < n => max(0,-k) <= j < min(n, n-k)
        lo2 = max(0, -k)
        hi2 = min(n, n - k)
        if lo2 < hi2:
            grad[lo2:hi2] -= p[m] * h[lo2+k:hi2+k]
    grad *= dx
    return grad

def subgrad_optimize(h, n, T, max_steps, start_time, time_limit, eta0=0.01):
    """Sub-gradient descent with adaptive step size."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    momentum = np.zeros(n)
    beta = 0.9
    for step in range(max_steps):
        if time.time() - start_time > time_limit:
            break
        grad = compute_grad(h_curr, n, T)
        momentum = beta * momentum + (1-beta) * grad
        # Adaptive step size
        eta = eta0 / (1 + step * 0.001)
        h_new = h_curr - eta * momentum
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
    return best_h, best_c5

            lags = None
            if focused:
                lags = get_worst_lags(best_h, n, 15)
            x_res = slsqp_optimize(best_h, n, T, 200, 1e-14, lags)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5 - 1e-15:
                best_h = make_feasible(x_res, n)
                best_c5 = c5_res
        except Exception:
            pass
    return best_h, best_c5

def focused_slsqp_rounds(h, n, n_rounds, start_time, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(n_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        lags = get_worst_lags(best_h, n, 12)
        for T in [100.0, 1000.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 20:
                break
            try:
                x_res = slsqp_optimize(best_h, n, T, 150, 1e-14, lags)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5 - 1e-15:
                    best_h = make_feasible(x_res, n)
                    best_c5 = c5_res
            except Exception:
                pass
    return best_h, best_c5

def try_resolution(h_best, n_best, n_try, start_time, budget_s, rng):
    if n_try <= n_best:
        old_x = np.linspace(0.0, 2.0, n_best)
        new_x = np.linspace(0.0, 2.0, n_try)
        h_try = np.interp(new_x, old_x, h_best)
    else:
        old_x = np.linspace(0.0, 2.0, n_best)
        new_x = np.linspace(0.0, 2.0, n_try)
        h_try = np.interp(new_x, old_x, h_best)
    h_try = make_feasible(h_try, n_try)
    c5_try = compute_c5(h_try)
    remaining = budget_s - (time.time() - start_time)
    if remaining < 30:
        return h_try, c5_try
    h_try, c5_try = multi_temp_slsqp(h_try, n_try, start_time, budget_s,
                                     temps=[200.0, 2000.0, 15000.0])
    return h_try, c5_try

def coord_descent_fine(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    trial_vals = [0.0, 1.0, 0.1, 0.25, 0.5, 0.75, 0.9]
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 10:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 5:
                break
            old = best_h[j]
            best_local_c = best_c5
            best_local_v = old
            for cand in trial_vals:
                if abs(cand - old) < 1e-9:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_local_c - 1e-14:
                    best_local_c = c
                    best_local_v = cand
            if best_local_v != old:
                best_h[j] = best_local_v
                best_h = make_feasible(best_h, n)
                best_c5 = best_local_c
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Sub-gradient warm-up
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = subgrad_optimize(best_h, n0, 500.0, 3000, start_time,
                                           min(remaining - 20, 50), eta0=0.003)

    # Phase 2: Focused SLSQP (top lags only)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        best_h, best_c5 = focused_slsqp_rounds(best_h, n0, 6, start_time, budget_s)

    # Phase 3: Full-softmax SLSQP multi-temp
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s,
                                           temps=[100.0, 500.0, 5000.0, 50000.0])

    # Phase 4: Targeted perturbations + SLSQP
    perts = [targeted_perturb, fourier_perturb, segment_flip_perturb,
             blend_perturb, smooth_perturb]
    for pf in perts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 60:
            break
        for _ in range(2):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 40:
                break
            try:
                if pf == targeted_perturb:
                    hp = pf(best_h, n0, rng, top_k=12)
                else:
                    hp = pf(best_h, n0, rng)
                hr, cr = multi_temp_slsqp(hp, n0, start_time, budget_s,
                                          temps=[500.0, 5000.0], focused=True)
                if cr < best_c5 - 1e-15:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 5: Binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        tl = min(remaining - 30, 100)
        b, c5_b = binary_sa(best_h, n0, 80000, 0.005, 0.00002, rng, tl, start_time)
        if c5_b < best_c5 - 1e-15:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 50:
                best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s,
                                                   temps=[500.0, 5000.0], focused=True)

    # Phase 6: Resolution sweep
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        res_list = [n0 - 80, n0 - 40, n0 + 40, n0 + 80, n0 + 160, n0 + 320]
        for n_try in res_list:
            if n_try < 100 or n_try % 2 != 0:
                continue
            remaining = budget_s - (time.time() - start_time)
            if remaining < 40:
                break
            try:
                h_try, c5_try = try_resolution(best_h, len(best_h), n_try,
                                               start_time, budget_s, rng)
                if c5_try < best_c5 - 1e-15:
                    best_c5 = c5_try
                    best_h = h_try.copy()
            except Exception:
                pass

    # Phase 7: More sub-gradient at best resolution
    remaining = budget_s - (time.time() - start_time)
    n_cur = len(best_h)
    if remaining > 40:
        best_h, best_c5 = subgrad_optimize(best_h, n_cur, 1000.0, 2000, start_time,
                                           min(remaining - 10, 30), eta0=0.002)

    # Phase 8: Final focused SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        best_h, best_c5 = focused_slsqp_rounds(best_h, n_cur, 4, start_time, budget_s)

    # Phase 9: High-temp SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        try:
            x_res = slsqp_optimize(best_h, n_cur, 80000.0, 250, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5 - 1e-15:
                best_h = make_feasible(x_res, n_cur)
                best_c5 = c5_res
        except Exception:
            pass

    # Phase 10: Coordinate descent fine-tune
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        best_h, best_c5 = coord_descent_fine(best_h, n_cur, 5, start_time, budget_s)

    # Final check
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
