import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    size = 1
    while size < 2 * n:
        size *= 2
    f = np.fft.rfft(h, size)
    g = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(np.conj(f) * g)
    return float(np.max(corr) * dx)

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_sa_fast(h, n, max_iters, T0, T1, rng, start_time, budget_s):
    """Fast binary SA using FFT-based C5 computation."""
    b = (h >= 0.5).astype(float)
    # Ensure exactly n/2 ones
    target = n // 2
    n_ones = int(np.sum(b == 1))
    if n_ones > target:
        ones = np.where(b == 1)[0]
        extra = n_ones - target
        to_flip = rng.choice(ones, size=extra, replace=False)
        b[to_flip] = 0
    elif n_ones < target:
        zeros = np.where(b == 0)[0]
        need = target - n_ones
        to_flip = rng.choice(zeros, size=need, replace=False)
        b[to_flip] = 1
    
    curr_c5 = compute_c5_fft(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    # Precompute for speed
    ones_idx = np.where(b == 1)[0]
    zeros_idx = np.where(b == 0)[0]
    
    for it in range(max_iters):
        if it % 1000 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 30:
                break
        
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        b[i] = 0
        b[j] = 1

        ones_idx = np.delete(ones_idx, np.where(ones_idx == i)[0][0])
        ones_idx = np.append(ones_idx, j)
        zeros_idx = np.delete(zeros_idx, np.where(zeros_idx == j)[0][0])
        zeros_idx = np.append(zeros_idx, i)
        
        new_c5 = compute_c5_fft(b)

```python
def c5_gradient(h, T):
    """Compute gradient of smoothed C5 = dx/T * log(sum exp(T*C)) using FFT."""
    n = len(h)
    dx = 2.0 / n
    size = 1
    while size < 2 * n:
        size *= 2
    
    # Compute correlation
    f = np.fft.rfft(h, size)
    g = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(np.conj(f) * g)
    # corr has 'size' elements, but only first 2n-1 are meaningful
    C = corr[:2*n-1]
    
    # Softmax weights
    T_C = T * C
    max_TC = np.max(T_C)
    exp_TC = np.exp(T_C - max_TC)
    w = exp_TC / np.sum(exp_TC)
    
    # w has length 2n-1, corresponds to lags L = -(n-1), ..., n-1
    # W_full[k] = w[k] for k = 0, ..., 2n-2 (W_full[n-1+L] = w[n-1+L] = w for lag L)
    W_full = w.copy()
    
    # T1[m] = conv(W_full, 1-h)[m+n-1] for m=0,...,n-1
    H1 = np.zeros(size)
    H1[:n] = 1.0 - h
    T1_full = np.fft.irfft(np.fft.rfft(W_full, size) * np.fft.rfft(H1, size))
    T1 = T1_full[n-1:2*n-1]
    
    # T2[m] = conv(W2, h)[m+n-1] for m=0,...,n-1
    W2 = W_full[::-1]
    H2 = np.zeros(size)
    H2[:n] = h
    T2_full = np.fft.irfft(np.fft.rfft(W2, size) * np.fft.rfft(H2, size))
    T2 = T2_full[n-1:2*n-1]
    
    grad = dx * (T1 - T2)
    return grad
```

def smoothed_c5_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    size = 1
    while size < 2 * n:
        size *= 2
    
    f = np.fft.rfft(h, size)
    g = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(np.conj(f) * g)
    C = corr[:2*n-1]
    
    TC = T * C
    max_TC = np.max(TC)
    exp_TC = np.exp(TC - max_TC)
    w = exp_TC / np.sum(exp_TC)
    
    # Objective value
    val = dx * (np.log(np.sum(exp_TC)) + max_TC) / T
    
    # Gradient
    conv_size = 1
    while conv_size < 3*n:
        conv_size *= 2
    
    W_full = w.copy()
    H1 = np.zeros(conv_size)
    H1[:n] = 1.0 - h
    T1_full = np.fft.irfft(np.fft.rfft(W_full, conv_size) * np.fft.rfft(H1, conv_size))
    T1 = T1_full[n-1:2*n-1]
    
    W2 = W_full[::-1]
    H2 = np.zeros(conv_size)
    H2[:n] = h
    T2_full = np.fft.irfft(np.fft.rfft(W2, conv_size) * np.fft.rfft(H2, conv_size))
    T2 = T2_full[n-1:2*n-1]
    
    grad = dx * (T1 - T2)
    return val, grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5_fft(h):
    n = len(h)
    dx = 2.0 / n
    size = 1
    while size < 2 * n:
        size <<= 1
    f = np.fft.rfft(h, size)
    g = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(np.conj(f) * g)
    return float(np.max(corr) * dx)

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_c5_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    size = 1
    while size < 2 * n:
        size <<= 1
    
    f = np.fft.rfft(h, size)
    g = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(np.conj(f) * g)
    C = corr[:2 * n - 1]
    
    TC = T * C
    max_TC = np.max(TC)
    exp_TC = np.exp(TC - max_TC)
    w = exp_TC / np.sum(exp_TC)
    
    val = dx * (np.log(np.sum(exp_TC)) + max_TC) / T
