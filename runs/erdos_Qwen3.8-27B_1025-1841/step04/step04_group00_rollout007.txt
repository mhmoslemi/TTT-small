                    if 0.0 <= v <= 1.0 and abs(v - old) > 1e-12:
                        tr = best_h.copy()
                        tr[j] = v
                        tr = make_feasible(tr, n)
                        c = compute_c5(tr)
                        if c < bc - 1e-14:
                            bc = c
                            bv = v
            if bv != old:
                best_h[j] = bv
                best_h = make_feasible(best_h, n)
                best_c5 = bc
                improved = True
        passed += 1
        if not improved:
            if steps[0] > 5e-5:
                steps = [s * 0.5 for s in steps]
            else:
                break
    return best_h, best_c5

def random_init_rng(rng, n, scale):
    h = np.clip(0.5 + rng.randn(n) * scale, 0.0, 1.0)
    return make_feasible(h, n)

def bimod_init(h, n):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = make_feasible(h_bin, n)
    h_smooth = 0.15 + 0.7 * h_bin
    return make_feasible(h_smooth, n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: PGD moderate T, top-5
    if deadline - time.time() > 150:
        h, c5 = pgd_opt(best_h, n, deadline - 200, top_m=5, T_start=20, T_end=1500, step_init=0.04)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: SLSQP T=300, top-5
    if deadline - time.time() > 120:
        h, c5 = slsqp_opt(best_h, n, T=300, top_m=5, maxiter=120, deadline=deadline - 20)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: PGD high T, top-3
    if deadline - time.time() > 120:
        h, c5 = pgd_opt(best_h, n, deadline - 120, top_m=3, T_start=200, T_end=8000, step_init=0.02, mom_init=0.7)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Fine coord descent
    if deadline - time.time() > 80:
        h, c5 = coord_fine(best_h, n, deadline - 80, steps=[0.002, 0.001, 0.0005, 0.0002])
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: SLSQP T=3000, top-3
    if deadline - time.time() > 80:
        h, c5 = slsqp_opt(best_h, n, T=3000, top_m=3, maxiter=100, deadline=deadline - 20)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: PGD very high T, top-2
    if deadline - time.time() > 60:
        h, c5 = pgd_opt(best_h, n, deadline - 70, top_m=2, T_start=1000, T_end=20000, step_init=0.01, mom_init=0.5)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Ultra-fine coord descent
    if deadline - time.time() > 40:
        h, c5 = coord_fine(best_h, n, deadline - 40, steps=[0.0005, 0.0002, 0.0001, 0.00005])
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 8: Random restarts (small perturbations + local opt)
    for ri in range(5):
        if deadline - time.time() < 60:
            break
        scale = 0.008 + 0.004 * ri
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        h2, c52 = pgd_opt(h_try, n, deadline - 60, top_m=4, T_start=100, T_end=3000, step_init=0.02)
        if c52 < best_c5:
            best_h, best_c5 = h2, c52
        # Quick coord descent on the perturbed
        if deadline - time.time() > 30:
            h3, c53 = coord_fine(h2, n, deadline - 30, steps=[0.001, 0.0005, 0.0002])
            if c53 < best_c5:
                best_h, best_c5 = h3, c53

    # Phase 9: Bimodal restart
    if deadline - time.time() > 80:
        h_bim = bimod_init(best_h, n)
        h2, c52 = pgd_opt(h_bim, n, deadline - 80, top_m=5, T_start=50, T_end=2000, step_init=0.03)
        if c52 < best_c5:
            best_h, best_c5 = h2, c52
        if deadline - time.time() > 40:
            h3, c53 = coord_fine(h2, n, deadline - 40, steps=[0.001, 0.0005, 0.0002])
            if c53 < best_c5:
                best_h, best_c5 = h3, c53

    # Phase 10: Final SLSQP polish
    if deadline - time.time() > 30:
        h, c5 = slsqp_opt(best_h, n, T=5000, top_m=3, maxiter=80, deadline=deadline - 5)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Final feasibility check
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
