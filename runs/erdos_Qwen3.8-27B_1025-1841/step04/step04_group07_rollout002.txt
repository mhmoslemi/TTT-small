    size = n + m - 1

import numpy as np
from scipy.optimize import minimize
import time

def full_corr_fft(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    n = len(h)
    b = 1.0 - h
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft(b, L)
    R = np.fft.irfft(np.conj(fa) * fb)
    return np.concatenate([R[L-n+1:L], R[0:n]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr_fft(h)
    return float(np.max(corr)) * dx

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h.sum())
    if n_ones > target:
        # Remove some ones (those with lowest "contribution")
        idx = np.where(h == 1)[0]
        # Remove the ones that contribute most to max overlap
        # Simple: remove random ones
        rng_order = np.random.permutation(len(idx))
        for i in rng_order:
            if n_ones == target:
                break
            h[idx[i]] = 0.0
            n_ones -= 1
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng_order = np.random.permutation(len(idx))
        for i in rng_order:
            if n_ones == target:
                break
            h[idx[i]] = 1.0
            n_ones += 1
    return h

def binary_local_search(h_bin, n, rng, max_rounds=200, tries_per_round=2000):
    """Aggressive binary local search using FFT evaluation."""
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        for _ in range(tries_per_round):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-14:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def binary_simulated_annealing(h_bin, n, rng, max_rounds=50, tries_per_round=3000, T_start=0.02, T_end=0.001):
    """Simulated annealing on binary space."""
    h = h_bin.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for rnd in range(max_rounds):
        T = T_start * (T_end / T_start) ** (rnd / max_rounds)
        accepted = 0
        for _ in range(tries_per_round):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            delta = c - cur
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                accepted += 1
                if cur < best_c - 1e-14:
                    best_c = cur
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if accepted == 0:
            break
    return best_h, best_c

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_smooth(h0, n, T, mu, max_iter=200):
    dx = 2.0 / n
    def obj(x):
        corr = full_corr_fft(x)
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    try:
        res = minimize(obj, h0, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
        return make_feasible(res.x, n)
    except:
        return h0.copy()

def alternating_topk(h0, n, n_iters=50, topk_frac=0.10, max_inner=30):

    h = h0.copy()
    best_c = compute_c5(h)
    for it in range(n_iters):
        corr = full_corr_fft(h)
        k = max(5, int(topk_frac * len(corr)))
        idx = np.argpartition(corr, -k)[-k:]
        active = np.zeros(len(corr))
        active[idx] = 1.0
        active /= active.sum()
        mu = 500.0
        def obj(x):
            c = full_corr_fft(x)
            return float(np.dot(active, c)) * dx + mu * (x.sum() - n/2.0)**2
        try:

                           bounds=[(0.0, 1.0)] * n,

            h_new = make_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c + 1e-12:
                h = h_new
                best_c = c_new
        except:
            pass

def block_swap_local_search(h_bin, n, rng, max_rounds=50, block_sizes=(1,2,3,4)):
    """Try swapping contiguous blocks of different sizes."""
    h = h_bin.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for bs in block_sizes:
            n_tries = 500
            for _ in range(n_tries):
                if len(h) < 2*bs:
                    break
                i = rng.randint(0, len(h) - 2*bs + 1)
                # Swap block [i, i+bs) with block [i+bs, i+2*bs)
                # But only make sense if one is 1s and other is 0s
                a = h[i:i+bs]
                b = h[i+bs:i+2*bs]
                if a.sum() == b.sum():
                    continue  # No change in count
                # After swap: a becomes b, b becomes a
                # Change in sum: -sum(a) + sum(b) for first block, -sum(b) + sum(a) for second
                # Net change: 0! So block swaps preserve the total.
                h[i:i+bs] = b.copy()
                h[i+bs:i+2*bs] = a.copy()
                c = compute_c5(h)
                if c < cur - 1e-14:
                    cur = c
                    improved = True
                else:
                    h[i:i+bs] = a.copy()
                    h[i+bs:i+2*bs] = b.copy()
        if not improved:
            break
    return h, cur

import numpy as np
from scipy.optimize import minimize
import time

# FFT-based full correlation
def full_corr_fft(h):
    n = len(h)
    b = 1.0 - h
    size = 2 * n - 1
    L = 1 << (size - 1).bit_length()
    fa = np.fft.rfft(h, L)
    fb = np.fft.rfft(b, L)
    R = np.fft.irfft(np.conj(fa) * fb)
    return np.concatenate([R[L-n+1:L], R[0:n]])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = full_corr_fft(h)
    return float(np.max(corr) * dx)
