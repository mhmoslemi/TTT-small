    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break

        else:

        if cnt == 0:

        np.clip(h, 0.0, 1.0, out=h)
    diff = target - np.sum(h)

    old_n = len(h)

        return make_feasible(h.copy(), new_n)

    h = h0.copy()
    best_local = compute_c5(h)
    
    for T in T_seq:

        params.T = T
        params.penalty = penalty

            return get_objective(x, params.T, params.penalty)

                                               'ftol': 1e-16, 'gtol': 1e-12})
        h = make_feasible(res.x, n)
        c = compute_c5(h)
        if c < best_local:
            best_local = c
    
    return h, best_local
