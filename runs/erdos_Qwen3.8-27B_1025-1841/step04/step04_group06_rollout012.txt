    return float(np.max(corr) * dx)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def get_lag_gradient(h, n, k):
    grad = np.zeros(n)
    if k > 0:
        i = np.arange(0, n - k)
        grad[i] += (1.0 - h[i + k])
        grad[i + k] += (-h[i])
    elif k < 0:
        kp = -k
        i = np.arange(kp, n)
        grad[i] += (1.0 - h[i - kp])
        grad[i - kp] += (-h[i])
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def linearized_minmax(h0, n, max_iter=50, damp=0.8, top_frac=1.0, time_limit=None, t0=None):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    n_lags = len(corr)
    n_top = max(10, int(top_frac * n_lags))

    for it in range(max_iter):
        if time_limit is not None:
            if time.time() - t0 > time_limit:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idxs = np.argsort(corr)[-n_top:]

        c_vec = np.zeros(n + 1)
        c_vec[n] = 1.0
        A_rows = []
        b_rows = []
        for idx in top_idxs:
            k = int(idx) - (n - 1)
            grad = get_lag_gradient(h, n, k)
            gtk = corr[idx]
            b_k = gtk - np.dot(grad, h)
            row = np.zeros(n + 1)
            row[:n] = grad
            row[n] = -1.0
            A_rows.append(row)
            b_rows.append(-b_k)

        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n + [(0.0, None)]

        res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if not res.success:
            break

        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        h_lp = make_feasible(h_lp, n)
        c5_lp = compute_c5(h_lp)

        h_new = (1.0 - damp) * h + damp * h_lp
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp.copy()
            if c5_lp < c5_new:
                h = h_lp.copy()
            else:
                h = h_new
        elif c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            h = h_new
            damp = max(0.3, damp * 0.9)

    return best_h, best_c5

def lbgfb_smooth(h_start, n, T, mu, max_iter=250):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=60):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(1500, len(ones) * len(zeros))
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current:
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current
