            #                - delta * h[j + s_k] for valid j+s_k

            #   i-s = j (i = j+s): h[j+s]*(1-h[j]) changes by -delta*h[j+s]

            effect = np.zeros(2*n-1)

            for cand in [0.0, 1.0, max(0.0, old-0.01), max(0.0, old-0.003),

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(x, T, lam, n):
    dx = 2.0 / n
    corr = compute_corr(x)
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = mc + np.log(np.sum(np.exp(T * shifted))) / T
    val = log_sum * dx
    penalty = lam * (np.sum(x) - n / 2.0) ** 2
    return val + penalty

def lgbfgs_optimize(h, T, lam, maxiter=200, ftol=1e-12):
    n = len(h)
    res = minimize(softmax_obj, h.copy(), args=(T, lam, n),
                   method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-10})
    return res.x

def slsqp_optimize(h, n, T, maxiter, ftol, lam=1e6):
    def obj(x):
        return softmax_obj(x, T, lam, n)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def fourier_perturb(h, n, rng, n_modes, amp):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 4))
        phase = rng.uniform(0, 2 * np.pi)
        amp_i = amp * rng.uniform(0.5, 2.0)
        if rng.random() < 0.5:
            h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp_i
        else:
            h_new += np.cos(2 * np.pi * freq * x / n + phase) * amp_i
    return make_feasible(h_new, n)

def segment_perturb(h, n, rng):
    h_new = h.copy()
    n_segs = rng.randint(1, 5)
    for _ in range(n_segs):
        length = rng.randint(max(2, n // 25), n // 4)
        start = rng.randint(0, max(1, n - length))
        r = rng.random()
        if r < 0.3:
            h_new[start:start + length] = 1.0 - h_new[start:start + length]
        elif r < 0.6:
            h_new[start:start + length] = rng.uniform(0, 1)
        else:
            shift = rng.randint(-length, length + 1)
            h_new[start:start + length] = np.roll(h[start:start + length], shift)
    return make_feasible(h_new, n)

def blend_perturb(h, n, rng):
    s = rng.randint(1, n // 5)
    h_shift = np.roll(h, s)
    alpha = rng.uniform(0.2, 0.8)
    h_new = alpha * h + (1 - alpha) * h_shift
    return make_feasible(h_new, n)

def active_set_optimize(h, n, T, lam, focus_frac, maxiter, start_time, budget_s):
    corr = compute_corr(h)
    mc = np.max(corr)
    threshold = mc - focus_frac * mc
    active = np.where(corr > threshold)[0]
    
    def focused_obj(x):
        dx = 2.0 / n
        c = compute_corr(x)
        mc_x = np.max(c)
        shifted = c - mc_x
        base = (mc_x + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        active_pen = 0.0
        for k in active:
            if 0 <= k < len(c):
                active_pen += max(0, c[k] - mc_x + 1e-10)
        penalty = lam * (np.sum(x) - n / 2.0) ** 2
        return base + 0.5 * active_pen * dx + penalty
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(focused_obj, h.copy(), method='L-BFGS-B',
                       bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x
    except Exception:
        return h

def binary_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    patience = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-15 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_b = b.copy()
                patience = 0
            else:
                patience += 1
        else:
            b[i] = 1
            b[j] = 0
            patience += 1
    return best_b, best_c5

def block_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, min(12, n // 10))
        start = rng.randint(0, max(1, n - bsize))
        end = start + bsize
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi), rng.randint(1, max(2, bsize // 3)))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-15 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def coord_descent_fast(h, n, max_rounds, start_time, budget_s):
    dx = 2.0 / n
    corr = compute_corr(h).copy()
    best_c5 = np.max(corr) * dx
    target = n / 2.0
    s_arr = np.arange(2 * n - 1) - (n - 1)
    
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 15:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 10:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            effect = np.zeros(2 * n - 1)
            idx1 = j - s_arr
            mask1 = (idx1 >= 0) & (idx1 < n)
            effect[mask1] += 1.0 - h[idx1[mask1]]
            idx2 = j + s_arr
            mask2 = (idx2 >= 0) & (idx2 < n)
            effect[mask2] -= h[idx2[mask2]]
            
            cands = [0.0, 1.0]
            if old > 0.01:
                cands.append(max(0.0, old - 0.01))
                cands.append(max(0.0, old - 0.003))
            if old < 0.99:
                cands.append(min(1.0, old + 0.01))
                cands.append(min(1.0, old + 0.003))
            cands.append(old - old * 0.1 if old > 0 else 0.0)
            cands.append(old + (1 - old) * 0.1 if old < 1 else 1.0)
            
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                cand = max(0.0, min(1.0, cand))
                delta = cand - old
                new_corr = corr + delta * effect
                c = np.max(new_corr) * dx
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            
            if abs(best_val - old) > 1e-12:
                delta = best_val - old
                corr += delta * effect
                h[j] = best_val
                best_c5 = best_c
                improved = True
        
        if not improved:
            break
    
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    return h, best_c5

def refine_lgbfgs(h, n, start_time, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    temps = [100.0, 400.0, 1500.0, 6000.0, 20000.0, 60000.0]
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 35:
            break
        try:
            x_res = lgbfgs_optimize(best_h, T, 5000.0, 150, 1e-13)
            x_res = make_feasible(x_res, n)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5 - 1e-14:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass
    return best_h, best_c5

def refine_slsqp(h, n, T, maxiter, start_time, budget_s):
    remaining = budget_s - (time.time() - start_time)
    if remaining < 30:
        return h, compute_c5(h)
    try:
        x_res = slsqp_optimize(h, n, T, maxiter, 1e-13)
        x_res = make_feasible(x_res, n)
        c5_res = compute_c5(x_res)
        if c5_res < compute_c5(h):
            return x_res, c5_res
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    lam = 5000.0
    
    # Phase 1: Multi-temperature L-BFGS-B refinement from start
    best_h, best_c5 = refine_lgbfgs(best_h, n0, start_time, budget_s)
    
    # Phase 2: SLSQP at moderate temperature
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        best_h, best_c5 = refine_slsqp(best_h, n0, 2000.0, 200, start_time, budget_s)
    
    # Phase 3: Structured perturbations + refinement
    perts = [fourier_perturb, segment_perturb, blend_perturb]
    for pf in perts:
        for attempt in range(3):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
            try:
                if pf == fourier_perturb:
                    hp = pf(best_h, n0, rng, n_modes=4, amp=0.06)
                else:
                    hp = pf(best_h, n0, rng)
                hr, cr = refine_lgbfgs(hp, n0, start_time, budget_s)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
                remaining = budget_s - (time.time() - start_time)
                if remaining > 40 and cr >= best_c5:
                    hr2, cr2 = refine_slsqp(hp, n0, 3000.0, 150, start_time, budget_s)
                    if cr2 < best_c5:
                        best_c5 = cr2
                        best_h = hr2.copy()
            except Exception:
                pass
        remaining = budget_s - (time.time() - start_time)
        if remaining < 60:
            break
    
    # Phase 4: Active-set focused optimization
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        try:
            x_f = active_set_optimize(best_h, n0, 5000.0, lam, 0.05, 200, start_time, budget_s)
            x_f = make_feasible(x_f, n0)
            c_f = compute_c5(x_f)
            if c_f < best_c5:
                best_c5 = c_f
                best_h = x_f.copy()
        except Exception:
            pass
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            try:
                x_f = active_set_optimize(best_h, n0, 15000.0, lam, 0.02, 200, start_time, budget_s)
                x_f = make_feasible(x_f, n0)
                c_f = compute_c5(x_f)
                if c_f < best_c5:
                    best_c5 = c_f
                    best_h = x_f.copy()
            except Exception:
                pass
    
    # Phase 5: Binary SA (structural exploration)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 180:
        tl = min(remaining - 60, 140)
        b, c5_b = binary_sa(best_h, n0, 200000, 0.004, 0.000003, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
        # Refine
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            best_h, best_c5 = refine_lgbfgs(best_h, n0, start_time, budget_s)
    
    # Phase 6: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        tl = min(remaining - 40, 100)
        b, c5_b = block_sa(best_h, n0, 80000, 0.004, 0.00002, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 50:
            best_h, best_c5 = refine_lgbfgs(best_h, n0, start_time, budget_s)
    
    # Phase 7: Coordinate descent polishing
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n0, 25, start_time, budget_s)
    
    # Phase 8: Higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 350:
        n1 = n0 + 240
        old_x = np.linspace(0.0, 2.0, n0 + 1)
        new_x = np.linspace(0.0, 2.0, n1 + 1)
        h_up = np.interp(new_x[:-1], old_x[:-1], best_h)
        h_up = make_feasible(h_up, n1)
        h_up, c5_up = refine_lgbfgs(h_up, n1, start_time, budget_s)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        # Additional perturb at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 80:
            try:
                hp = fourier_perturb(best_h, len(best_h), rng, n_modes=3, amp=0.04)
                hr, cr = refine_lgbfgs(hp, len(best_h), start_time, budget_s)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass
        remaining = budget_s - (time.time() - start_time)
        if remaining > 30:
            n_pts = len(best_h)
            best_h, best_c5 = coord_descent_fast(best_h.copy(), n_pts, 15, start_time, budget_s)
    
    # Phase 9: Final SLSQP refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        n_pts = len(best_h)
        best_h, best_c5 = refine_slsqp(best_h, n_pts, 10000.0, 250, start_time, budget_s)
    
    # Phase 10: Final L-BFGS-B at very high T
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        try:
            x_res = lgbfgs_optimize(best_h, 100000.0, lam, 300, 1e-14)
            x_res = make_feasible(x_res, n_pts)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass
    
    # Phase 11: Final coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent_fast(best_h.copy(), n_pts, 10, start_time, budget_s)
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
