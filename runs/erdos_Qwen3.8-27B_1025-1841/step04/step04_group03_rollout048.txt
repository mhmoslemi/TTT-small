```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, step_size, max_iters=50):
    """Frank-Wolfe style subgradient descent on max correlation."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Subgradient of max correlation
        k_max = np.argmax(corr)
        # d/dh[i] of correlate(h, 1-h)[k_max]
        # correlate at lag s = k_max - (n-1):
        #   if s >= 0: sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
        #     grad: for i in 0..n-1-s: (1-h[i+s]) - h[i]*(1-h[i]) ... 
        #     Actually d/dh[i] [h[i]*(1-h[i+s])] = 1-h[i+s]
        #     d/dh[i+s] [h[i]*(1-h[i+s])] = -h[i]
        #   if s < 0: let s' = -s, sum_{i=s'}^{n-1} h[i]*(1-h[i-s'])
        #     similar

        s = k_max - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            for i in range(n - s):
                grad[i] += (1.0 - h[i + s])
                grad[i + s] += (-h[i])
        else:
            sp = -s
            for i in range(sp, n):
                grad[i] += (1.0 - h[i - sp])
                grad[i - sp] += (-h[i])
        
        # Step: h_new = h - step_size * grad, project
        h_new = np.clip(h - step_size * grad, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    return best_h, best_c5
```

```python
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ss = step_size
    for it in range(max_iters):
        if time_limit and time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        grad = np.zeros(n)

            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            sp = -s
            grad[sp:] += (1.0 - h[:n-sp])
            grad[:n-sp] += (-h[sp:])

            ss = min(ss * 1.1, 0.05)  # slightly increase step
        else:
            ss *= 0.5
            if ss < 1e-10:
```

```python
                    found = True
                    break

                    found = True

                continue
```

```python
import numpy as np

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_descent(h, n, max_iters, start_time, time_limit, step0=0.01):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    ss = step0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            sp = -s
            grad[sp:] += (1.0 - h[:n-sp])
            grad[:n-sp] += (-h[sp:])
        h_new = np.clip(h - ss * grad, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            ss = min(ss * 1.05, 0.04)
        else:
            ss *= 0.5
            if ss < 1e-10:
                break
    return best_h, best_c5

_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    global _SL_N, _SL_T
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def multi_slsqp(h, n, start_time, budget_s):

    best_h = h.copy()

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        # Summary descent at higher res
        h_up, c5_up = subgrad_descent(h_up, n1, 200, start_time, remaining - 100)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up
        # Binary SA at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 80:
            b_up = (best_h >= 0.5).astype(float)
            t1 = n1 // 2
            while np.sum(b_up) > t1:
                ones = np.where(b_up == 1)[0]
                b_up[rng.randint(len(ones))] = 0
            while np.sum(b_up) < t1:
                zeros = np.where(b_up == 0)[0]
                b_up[rng.randint(len(zeros))] = 1
            tl = min(remaining - 40, 100)
            sb, sc = binary_sa(b_up, n1, 80000, 0.003, 0.000001, rng, tl, start_time)
            if sc < best_c5:
                best_c5 = sc
                best_h = sb.astype(float)
        # SLSQP at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            best_h, best_c5 = multi_slsqp(best_h, n1, start_time, budget_s)

    # Phase 7: Coordinate descent polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 15, start_time, budget_s)

    # Phase 8: Final high-T SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        try:
            _SL_N = n_pts
            _SL_T = 100000.0
            x_res = slsqp_optimize(best_h, n_pts, 100000.0, 150, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Final subgradient polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        best_h, best_c5 = subgrad_descent(best_h, len(best_h), 50, start_time, remaining - 2)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
