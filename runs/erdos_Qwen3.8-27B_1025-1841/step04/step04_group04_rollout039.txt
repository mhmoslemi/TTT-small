    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), float(np.max(corr))

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)

        else:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_step(h, n, alpha, momentum_v, beta, window=5):
    """One subgradient step with momentum."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find active set (within window of max)
    threshold = max_val - 1e-10 * max_val
    active = np.where(corr >= threshold)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    # Average gradient over active set
    grad = np.zeros(n)
    for m in active:
        s = m - (n - 1)
        # grad[i] += (1-h[i+s]) - h[i-s]
        # First term: sum of (1-h[i+s]) for valid i
        # i+s valid when 0 <= i+s < n, i.e., -s <= i < n-s
        lo1 = max(0, -s)
        hi1 = min(n, n - s)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
        # Second term: -h[i-s] for valid i
        # i-s valid when 0 <= i-s < n, i.e., s <= i < n+s
        lo2 = max(0, s)
        hi2 = min(n, n + s)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2-s:hi2-s]
    
    grad /= len(active)
    
    # Momentum update
    momentum_v[:] = beta * momentum_v - alpha * grad
    h_new = h + momentum_v
    h_new = make_feasible(h_new, n)
    return h_new, corr, max_val

def softmax_objective(x, n, T, top_k=None):

    corr = np.correlate(x, 1.0 - x, mode='full')

        c = corr[idx]
    else:
        c = corr
    mc = np.max(c)
    shifted = c - mc
    logsum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + logsum / T) * dx

        if lo2 < hi2:

        grad *= 1.0  # will be scaled by weight below
        # Actually need to weight by weights[k_idx]

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def corr_gradient(h, n, m):
    """Gradient of corr[m] w.r.t. h for full correlate(h, 1-h)."""
    s = m - (n - 1)
    grad = np.zeros(n)
    lo1 = max(0, -s)
    hi1 = min(n, n - s)
    if lo1 < hi1:
        grad[lo1:hi1] += (1.0 - h[lo1+s:hi1+s])
    lo2 = max(0, s)
    hi2 = min(n, n + s)
    if lo2 < hi2:
        grad[lo2:hi2] -= h[lo2-s:hi2-s]
    return grad

def softmax_obj(x, n, T, top_k=None):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
    else:
        c = corr
        idx = np.arange(len(corr))
    mc = np.max(c)
    shifted = c - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def softmax_obj_grad(x, n, T, top_k=None):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        c = corr[idx]
    else:
        c = corr
        idx = np.arange(len(corr))
    mc = np.max(c)
    shifted = c - mc
    exps = np.exp(T * shifted)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for ki in range(len(idx)):
        w = weights[ki] * T / T  # weight for soft max
        if w < 1e-12:
            continue
        g = corr_gradient(x, n, idx[ki])
        grad += w * g
    return grad * dx

def slsqp_optimize(h, n, T, top_k, maxiter, ftol):
    def obj(x):
        return softmax_obj(x, n, T, top_k)
    def obj_grad(x):
        return softmax_obj_grad(x, n, T, top_k)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
