    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(500):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def subgradient_one_step(h, n, lr, top_L):
    corr = compute_corr(h)
    # Find top L indices
    top_idx = np.argsort(corr)[::-1][:top_L]
    # Average gradients
    grad = np.zeros(n)
    for K in top_idx:
        k = K - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n-k])
            grad[:n-k] -= h[k:]
        else:
            s = -k
            grad[:n-s] += (1.0 - h[s:n])
            grad[s:n] -= h[:n-s]
    grad /= top_L
    h_new = h - lr * grad
    return h_new, corr

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    for _ in range(500):
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
        diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def get_subgrad(h, n, top_L=5):
    corr = compute_corr(h)
    top_idx = np.argsort(corr)[::-1][:top_L]
    grad = np.zeros(n)
    for K in top_idx:
        k = int(K) - (n - 1)
        if k >= 0:
            grad[k:] += (1.0 - h[:n - k])
            grad[:n - k] -= h[k:]
        else:
            s = -k
            grad[:n - s] += (1.0 - h[s:n])
            grad[s:n] -= h[:n - s]
    grad /= top_L
    return grad, corr

            break

        shift = k_star - (n - 1)
        n_valid = n - abs(shift)
        if shift >= 0:
            h_a = h[shift:shift + n_valid]
            h_b = 1.0 - h[:n_valid]
        else:
            s = -shift
            h_a = h[:n_valid]
            h_b = 1.0 - h[s:s + n_valid]
        contrib = (h_a > 0.5) & (h_b > 0.5)
        n_contrib = int(np.sum(contrib))
        if n_contrib == 0:

        best_p0 = -1

def subgradient_descent(h, n, time_fn, min_time=30, n_iters=50000, lr0=0.1):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for t in range(n_iters):
        if time_fn() < min_time:
            break
        lr = lr0 / (1.0 + t * 0.001)
        grad, corr = get_subgrad(h, n, top_L=5)
        h = h - lr * grad
        h = np.clip(h, 0.0, 1.0)
        h = make_feasible(h, n)
        if t % 200 == 0:
            c5_val = compute_c5(h)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
    return best_h, best_c5

def smooth_optimize(h, n, T, max_iter, time_fn, min_time=20):
    dx = 2.0 / n
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        log_sum = np.log(np.sum(np.exp(T * shifted)))
        return (mc + log_sum / T) * dx
    def cons_fun(hv):
        return np.sum(hv) - n / 2.0
    def cons_jac(hv):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def coordinate_descent(h, n, time_fn, min_time=15, max_rounds=5):
    h = h.copy()
    curr_c5 = compute_c5(h)
    deltas = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05, 0.005, -0.005]
    for rnd in range(max_rounds):
        if time_fn() < min_time:
            break
        improved = False
        for j in range(n):
            if time_fn() < min_time:
                break

                nv = old + d

                    curr_c5 = c

                if c < curr_c5 - 1e-12:
                    h = trial
                    curr_c5 = c
                    improved = True
        if not improved:
            break
    return h, curr_c5
