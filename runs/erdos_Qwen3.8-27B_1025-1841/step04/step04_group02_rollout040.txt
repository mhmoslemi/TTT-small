import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diffs = h - 0.5
    order = np.argsort(diffs)
    for idx in order:
        if np.sum(b) == target:
            break
        if b[idx] == 0 and diffs[idx] > 0:
            b[idx] = 1
        elif b[idx] == 1 and diffs[idx] < 0:
            b[idx] = 0
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h, n, max_iters, T0, T1, rng, start_time=None, budget_frac=1.0, budget_s=1300):

    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if start_time is not None and (time.time() - start_time) > budget_s * budget_frac:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def slsqp_obj_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted_max = np.max(shifted)
    exp_s = np.exp(shifted - shifted_max)
    w = exp_s / np.sum(exp_s)
    f = (mc + (shifted_max + np.log(np.sum(exp_s))) / T) * dx
    # Gradient: grad[i] = (1/T)*dx * sum_k w[k] * ((1-h[i-k])*[0<=i-k<n] - h[i+k]*[0<=i+k<n])
    grad = np.zeros(n)
    for i in range(n):
        # First term: sum over j where k=i-j, 0<=k<=n-1 => 0<=i-j<n
        j_start = max(0, i - (n - 1))
        j_end = min(i, n - 1)
        if j_start <= j_end:
            js = np.arange(j_start, j_end + 1)
            ks = i - js
            s1 = np.dot(1.0 - x[js], w[ks + n - 1])
        else:
            s1 = 0.0
        # Second term: sum over m where k=m-i, 0<=k<=n-1
        m_start = max(i, 0)
        m_end = min(i + n - 1, n - 1)
        if m_start <= m_end:
            ms = np.arange(m_start, m_end + 1)
            ks2 = ms - i
            s2 = np.dot(x[ms], w[ks2 + n - 1])
        else:
            s2 = 0.0
        grad[i] = (s1 - s2) * dx / T
    return f, grad

def slsqp_run(h_init, n, T, maxiter=300, ftol=1e-15):
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        return slsqp_obj_grad(x, n, T)[0]

    def grad(x):
        return slsqp_obj_grad(x, n, T)[1]

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(obj, h_init.copy(), jac=grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def coord_descent_pw(h, n, max_rounds, start_time, budget_s, stop_frac=0.92):
    """Coordinate descent exploiting piecewise-linear structure in each coordinate."""
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    stop_time = start_time + budget_s * stop_frac
    for rnd in range(max_rounds):
        if time.time() > stop_time:
            break
        improved = False
        for j in range(n):
            if time.time() > stop_time:
                break
            old = h[j]
            # Compute a_k and b_k for all lags
            # c_k = a_k * t + b_k where t = h[j]
            # a_k = (1-h[j-k])*[0<=j-k<n] - h[j+k]*[0<=j+k<n]
            # b_k = c_k - a_k * old
            k_arr = np.arange(-(n-1), n)
            # a_k computation
            a_k = np.zeros(2 * n - 1)
            # First part: (1-h[j-k]) for 0<=j-k<n
            jk = j - k_arr
            mask1 = (jk >= 0) & (jk < n)
            a_k[mask1] += (1.0 - h[jk[mask1]])
            # Second part: -h[j+k] for 0<=j+k<n
            jk2 = j + k_arr
            mask2 = (jk2 >= 0) & (jk2 < n)
            a_k[mask2] -= h[jk2[mask2]]
            # Current corr
            corr = np.correlate(h, 1.0 - h, mode='full')
            b_k = corr - a_k * old
            # Find t in [0,1] minimizing max_k (a_k*t + b_k)
            # Use ternary search
            lo, hi = 0.0, 1.0
            for _ in range(40):
                m1 = lo + (hi - lo) / 3.0
                m2 = hi - (hi - lo) / 3.0
                f1 = np.max(a_k * m1 + b_k)
                f2 = np.max(a_k * m2 + b_k)
                if f1 < f2:
                    hi = m2
                else:
                    lo = m1
            t_opt = (lo + hi) / 2.0
            # Check boundaries too
            f0 = np.max(b_k)
            f1 = np.max(a_k + b_k)
            fopt = np.max(a_k * t_opt + b_k)
            if f0 < fopt:
                t_opt = 0.0
            elif f1 < fopt:

            if abs(t_opt - old) > 1e-10:
                trial = h.copy()
                trial[j] = t_opt
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    h = trial
                    best_c5 = c
                    improved = True
    return h, best_c5

                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
            # Also try small perturbations
            for delta in [0.01, -0.01, 0.005, -0.005]:
                cand = old + delta
                if cand < 0 or cand > 1:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Binary SA for good structure
    b0, c5_b0 = binary_sa(h_init, n, 80000, 0.01, 0.00001, rng,
                          start_time, 0.15, budget_s)
    if c5_b0 < best_c5:
        best_c5 = c5_b0
        best_h = b0.copy()

    # Phase 2: Block SA
    remaining_frac = 0.25
    b1, c5_b1 = binary_sa_block(best_h, n, 40000, 0.006, 0.00003, rng,
                                start_time, remaining_frac, budget_s)
    if c5_b1 < best_c5:
        best_c5 = c5_b1
        best_h = b1.copy()

    # Phase 3: Smoothing annealing SLSQP with analytical gradient
    T_schedule = [40.0, 20.0, 10.0, 5.0, 2.5, 1.5, 1.0]
    for T in T_schedule:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.70:
            break
        try:
            x_res = slsqp_run(best_h, n, T, maxiter=250, ftol=1e-16)
            x_res = make_feasible(x_res, n)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res
        except Exception:
            pass

    # Phase 4: Coordinate descent (piecewise-linear)
    best_h, best_c5 = coord_descent_pw(best_h.copy(), n, 15, start_time, budget_s, 0.88)

    # Phase 5: Another SLSQP round with low T
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.90:
        try:
            x_res = slsqp_run(best_h, n, 0.8, maxiter=200, ftol=1e-16)
            x_res = make_feasible(x_res, n)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res
        except Exception:
            pass

    # Phase 6: Fine coordinate descent
    best_h, best_c5 = coord_descent_fine(best_h.copy(), n, 10, start_time, budget_s, 0.97)

    # Phase 7: Final SLSQP squeeze
    elapsed = time.time() - start_time
    if elapsed < budget_s * 0.985:
        try:
            x_res = slsqp_run(best_h, n, 0.5, maxiter=100, ftol=1e-17)
            x_res = make_feasible(x_res, n)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res
        except Exception:
            pass

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
