    """Local search: try all single-swap moves, accept improving ones."""
    dx = 2.0 / n
    best = h_bin.copy()
    best_c5 = compute_c5(best)
    curr = h_bin.copy()
    curr_c5 = best_c5
    ones_list = list(np.flatnonzero(curr))
    zeros_list = list(np.where(curr == 0)[0])
    
    st = time.time()
    while time.time() - st < time_limit:
        improved = False
        # Try random swaps (biased toward changing the worst shift)
        n_try = 500
        for _ in range(n_try):
            if time.time() - st > time_limit:
                break
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            curr[i] = 0
            curr[j] = 1
            nc5 = compute_c5(curr)
            if nc5 < curr_c5 - 1e-14:
                curr_c5 = nc5
                improved = True
                if nc5 < best_c5:
                    best_c5 = nc5
                    best = curr.copy()
                # Update lists
                idx_i = ones_list.index(i)
                ones_list[idx_i] = j
                idx_j = zeros_list.index(j)
                zeros_list[idx_j] = i
            else:
                curr[i] = 1
                curr[j] = 0
        if not improved:
            break
    return best, best_c5
