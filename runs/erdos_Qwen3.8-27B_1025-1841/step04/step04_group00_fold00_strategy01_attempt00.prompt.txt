def top_shift_gradient(h, top_idx, n):

            if 0 <= j + s < n:

def slsqp_polish(h, n, budget_s, start_time, T=1000.0, top_m=None):
    """SLSQP optimization with smooth max."""
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None and top_m < len(corr):
            top_idx = np.argsort(corr)[-top_m:]
            mc = np.max(corr[top_idx])
            shifted = corr[top_idx] - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
