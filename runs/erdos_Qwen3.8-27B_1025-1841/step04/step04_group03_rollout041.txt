```python
def swap_corr(corr, h, n, i, j):
    """Compute the change in corr after swapping h[i]:1->0 and h[j]:0->1.
    Returns the delta vector (change to add to corr)."""
    ni = n - 1
    delta = np.zeros(2*n - 1)
    
    # Remove i (was 1, now 0)
    ks = range(ni - i, 2*ni - i + 1)  # k values where i contributes
    q_idx = i + np.array(ks) - ni  # corresponding q = i + k - ni
    # contribution was (1 - h[q]), now removed
    rem = 1.0 - h[q_idx]
    # If q == j, h[j] = 0 (old value), so 1-h[j] = 1. This is correct since j is still 0 at this step.
    delta[ks] -= rem
    
    # Add j (was 0, now 1)
    js = range(ni - j, 2*ni - j + 1)
    q_idx2 = j + np.array(js) - ni
    add = 1.0 - h[q_idx2]
    # If q == i, h[i] is now 0 (after step 1), so 1-h[i] = 1
    # But in our h array, h[i] is still 1. So we need to fix:
    mask_i = (q_idx2 == i)
    if np.any(mask_i):
        add = add.copy()
        add[mask_i] = 1.0
    delta[js] += add
    
    return delta
```

```python
def fast_binary_sa(b, n, rng, time_limit, start_time, T0=3.0, T1=0.005):
    dx = 2.0 / n
    ni = n - 1
    size = 2 * n - 1
    
    corr = np.correlate(b, 1.0 - b, mode='full').astype(np.float64)
    curr_max = float(np.max(corr))
    best_max = curr_max
    best_b = b.copy()
    
    ones_arr = np.where(b == 1)[0].copy()
    zeros_arr = np.where(b == 0)[0].copy()
    half_n = n // 2
    
    delta = np.zeros(size)
    
    while True:
        t_now = time.time()
        if t_now - start_time > time_limit:
            break
        
        elapsed = t_now - start_time
        frac = min(1.0, elapsed / time_limit)
        T = T0 * (T1 / T0) ** frac
        
        idx_i = rng.randint(half_n)
        idx_j = rng.randint(half_n)
        i = ones_arr[idx_i]
        j = zeros_arr[idx_j]
        
        # rem = 1 - b (entire vector)
        rem = 1.0 - b
        
        # add = 1 - b, but add[i] = 1 (since b[i] becomes 0)
        # We need a modified version
        # Instead of copying, we can handle it specially:
        add = rem  # same as 1-b
        # But we need add[i] = 1.0
        # rem[i] = 1 - b[i] = 1 - 1 = 0
        # So we need to add 1 at position i in the add vector
        # The add vector is applied at positions [j, j+ni-1] in corr
        # Position i in the add vector corresponds to corr position j + i
        # So we add 1 to delta[j+i]
        
        delta.fill(0)
        delta[i:i+n] -= rem  # remove i's contribution
        delta[j:j+n] += rem  # add j's contribution (using 1-b)
        # Fix: at position i in b, h[i] becomes 0, so (1-h[i]) = 1
        # This affects the 'add' part at corr position j + i
        delta[j + i] += 1.0  # correction: was 0 (1-b[i]=0), should be 1
        
        new_max = float(np.max(corr + delta))
        
        if new_max < curr_max or (T > 1e-10 and rng.random() < np.exp(-(new_max - curr_max) / T)):
            curr_max = new_max
            corr += delta
            b[i] = 0.0
            b[j] = 1.0
            ones_arr[idx_i] = j
            zeros_arr[idx_j] = i
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
    
    return best_b, best_max * dx
```

This is very clean! The key steps per iteration:
1. `rem = 1.0 - b` (n ops)
2. `delta.fill(0)` (2n ops)
3. `delta[i:i+n] -= rem` (n ops)
4. `delta[j:j+n] += rem` (n ops)
5. `delta[j+i] += 1.0` (1 op)
6. `np.max(corr + delta)` (2n ops)
7. Maybe `corr += delta` (2n ops)

Total: ~8n = ~8000 operations. In numpy, each vector operation on 960 elements takes ~1-2µs. So total per iteration: ~5-10µs.

In 800 seconds: 80-160 million iterations.

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-6
        else:
            mask = h > 1e-6
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binarize(h, n, rng):
    b = (h >= 0.5).astype(np.float64)
    target = n // 2
    s = int(b.sum())
    if s > target:
        ones = np.where(b == 1.0)[0]
        to_remove = rng.choice(ones, size=s - target, replace=False)
        b[to_remove] = 0.0
    elif s < target:
        zeros = np.where(b == 0.0)[0]
        to_add = rng.choice(zeros, size=target - s, replace=False)
        b[to_add] = 1.0
    return b

def fast_binary_sa(b, n, rng, time_limit, start_time, T0=3.0, T1=0.003):

    ni = n - 1
    size = 2 * n - 1

    best_max = curr_max
    best_b = b.copy()
    
    ones_arr = np.where(b == 1.0)[0].copy()
    zeros_arr = np.where(b == 0.0)[0].copy()
    half_n = n // 2

        if t_now - start_time > time_limit:

def _slsqp_obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc

    return np.ones(_G_N)

def slsqp_refine(h, n, time_limit, start_time, budget_s):
    global _G_N, _G_T
    _G_N = n
    best = h.copy()
    best_c5 = compute_c5(h)
    
    temps = [50.0, 200.0, 1000.0, 5000.0, 20000.0]

def coord_descent_quick(h, n, start_time, time_limit):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(50):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit - 2:
                break
            old = h[j]
            candidates = [0.0, 1.0]
            if old > 0:
                candidates.append(old * 0.5)
            if old < 1:
                candidates.append(1.0 - old * 0.5)
            best_val = old
            best_c = best_c5
            for cand in candidates:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def perturb_binary(b, n, rng, n_swaps):
    """Perturb a binary solution by doing n_swaps random swaps."""
    b = b.copy()
    ones = np.where(b == 1.0)[0]
    zeros = np.where(b == 0.0)[0]
    for _ in range(n_swaps):
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0.0
        b[j] = 1.0
        # Update arrays
        idx_i = np.where(ones == i)[0][0]
        idx_j = np.where(zeros == j)[0][0]
        ones[idx_i] = j
        zeros[idx_j] = i
    return b
```
