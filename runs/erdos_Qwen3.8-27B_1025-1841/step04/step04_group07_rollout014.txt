def make_feasible(h, n):

    target = n / 2.0

        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:

        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

                   bounds=[(0.0, 1.0)] * n,

            h[oj] = 0.0
            no_improve += 1
        if no_improve > patience:

Total: A + C + B + D = n-k, and C ≈ B ≈ n/2 - A.

To minimize C(k), we want to maximize A(k) (the autocorrelation). But we can't maximize A(k) for all k simultaneously.

The sum of A(k) over all k is fixed: sum_k A(k) = sum_i h[i] * sum_i h[i] - sum_i h[i]^2 ... no, that's for circular. For linear:
sum_{k=0}^{n-1} A(k) = sum_{k=0}^{n-1} sum_{i=0}^{n-1-k} h[i]h[i+k]

This equals sum_i h[i] * (sum_{j>=i} h[j]) = sum_i h[i] * (n/2 - sum_{j<i} h[j]) for binary h with n/2 ones.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
