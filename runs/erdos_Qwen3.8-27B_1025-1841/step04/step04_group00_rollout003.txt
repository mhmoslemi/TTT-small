from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_shifts(h):
    """Compute all shift values O(k) = ∫h(x)(1-h(x+k))dx discretized."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # array of 2n-1 values

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def solve_slp_iteration(h0, n, active_set, max_iter=50):
    """
    Solve SLP iteration: linearize the overlap constraints around h0,
    solve the resulting LP.
    """
    dx = 2.0 / n
    # Variables: [h[0], ..., h[n-1], t]
    # Minimize t: c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    A_ub = []
    b_ub = []
    
    for k_idx in active_set:
        # shift = k_idx - (n-1)
        shift = k_idx - (n - 1)
        # h[i] * (1 - h[i+shift]) for valid i
        # Valid i: 0 <= i < n and 0 <= i+shift < n
        if shift >= 0:
            i_start = 0
            i_end = n - shift
        else:
            i_start = -shift
            i_end = n
        
        row = np.zeros(n + 1)
        for i in range(i_start, i_end):
            j = i + shift
            # Linearized: (1-h0[j])*h[i] <= t - h0[i]*h0[j] (times dx)
            row[i] = (1.0 - h0[j]) * dx
        row[-1] = -1.0  # -t
        A_ub.append(row)
        # RHS: t - sum h0[i]*h0[j]*dx <= 0 => sum h0[i](1-h0[i])... 
        # Actually: sum_i [(1-h0[j])*h[i] - h0[i]*h0[j]] * dx <= t
        # sum_i (1-h0[j])*h[i]*dx - t <= sum_i h0[i]*h0[j]*dx
        rhs = np.sum(h0[i_start:i_end] * h0[i_start+shift:i_end+shift]) * dx
        b_ub.append(rhs)
    
    # Equality: sum(h) * dx = 1 => sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [n / 2.0]
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    try:
        res = linprog(c, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            h_new = res.x[:n]
            t_new = res.x[-1]
            return h_new, t_new
    except Exception:
        pass
    return h0, np.max(get_shifts(h0))

def slp_optimize(h, n, budget_s, start_time, tol=1e-6, max_outer=100):
    """Sequential Linear Programming for the overlap minimization."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for outer in range(max_outer):
        if time.time() - start_time > budget_s - 5:
            break
        
        shifts = get_shifts(h)
        c5_cur = np.max(shifts)
        
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Find active set: shifts where O(k) > best_c5 - tol
        # Or top-m shifts
        threshold = c5_cur - tol
        active = np.where(shifts >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(shifts)[-3:]
        # Limit active set size for LP efficiency
        if len(active) > 30:
            active = np.argsort(shifts)[-30:]
        
        h_new, t_new = solve_slp_iteration(h, n, active)
        
        if np.allclose(h_new, h, atol=1e-10):
            break
        
        # Relaxation: move partially toward h_new to avoid cycling
        alpha = 1.0
        for _ in range(10):
            h_test = h + alpha * (h_new - h)
            h_test = np.clip(h_test, 0, 1)
            h_test = make_feasible(h_test, n)
            c5_test = compute_c5(h_test)
            if c5_test <= c5_cur + 0.01 * tol:
                h = h_test
                break
            alpha *= 0.5
        else:
            h = h_new
            h = make_feasible(h, n)
        
        if time.time() - start_time > budget_s - 5:
            break
    
    h = make_feasible(h, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_h = h
        best_c5 = c5_final
    return best_h, best_c5

def nesterov_momentum(h, n, budget_s, start_time, top_m=5):
    """Nesterov-accelerated gradient descent on the smooth max."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    momentum = 0.9
    step = 0.05
    it = 0
    
    while time.time() - start_time < budget_s:
        it += 1
        y = h - momentum * velocity
        y_clip = np.clip(y, 0, 1)
        
        corr = np.correlate(y_clip, 1.0 - y_clip, mode='full')
        top_idx = np.argsort(corr)[-top_m:]
        
        # Gradient of sum of top-m overlaps (linearized)
        grad = np.zeros(n)
        for k_idx in top_idx:
            shift = k_idx - (n - 1)
            if shift >= 0:
                i_start, i_end = 0, n - shift
            else:
                i_start, i_end = -shift, n
            for i in range(i_start, i_end):
                j = i + shift
                grad[i] += (1.0 - y_clip[j])
        grad /= len(top_idx)
        
        velocity = momentum * velocity - step * grad
        h_new = np.clip(h + velocity, 0, 1)
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new <= compute_c5(h) + 1e-12:
            h = h_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
        else:
            step *= 0.8
            velocity *= 0.5
            if step < 1e-8:
                break
        
        if it % 100 == 0 and step < 0.03:
            step = min(0.05, step * 1.3)
    
    h = make_feasible(h, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_h = h
        best_c5 = c5_final
    return best_h, best_c5

def pn_smooth_optimize(h, n, budget_s, start_time, p=20, top_m=10):
    """Optimize p-norm of top-m overlaps using SLSQP."""
    dx = 2.0 / n
    
    def objective(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        vals = np.sort(corr)[-top_m:]
        return np.sum(vals**p) / top_m * dx**p  # p-norm scaled
    
    def objective_grad(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        top_thresh = np.sort(corr)[-top_m]
        mask = corr >= top_thresh - 1e-10
        # grad of sum(corr[k]^p) w.r.t. x[j]
        # corr[k] = sum_i x[i]*(1-x[i+shift_k]) * dx
        # d(corr[k])/dx[j] = dx * sum over k where j is in range: (1-x[j+shift]) - x[j-shift]
        grad = np.zeros(n)
        for k_idx in np.where(mask)[0]:
            shift = k_idx - (n - 1)
            if shift >= 0:
                i_start, i_end = 0, n - shift
            else:
                i_start, i_end = -shift, n
            ck = corr[k_idx]
            coeff = p * (ck ** (p-1)) * dx
            for i in range(i_start, i_end):
                j = i + shift
                grad[i] += coeff * dx * (1.0 - x[j])
        return grad
    
    def constraint(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint}]
    
    try:
        res = minimize(objective, h, jac=objective_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-18})
        if res.success or compute_c5(res.x) < compute_c5(h):
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h):
                return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)

import numpy as np
from scipy.optimize import minimize, linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_shifts_array(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def slp_single_lp(h0, n, active_indices, trust_radius=0.3):
    """Solve one SLP LP subproblem. Returns (h_new, t_new) or None on failure."""
    dx = 2.0 / n
    nv = n + 1
    c = np.zeros(nv)
    c[-1] = 1.0

    rows = []
    rhs = []
    for k_idx in active_indices:
        shift = k_idx - (n - 1)
        if shift >= 0:
            i_start, i_end = 0, n - shift
        else:
            i_start, i_end = -shift, n
        if i_end <= i_start:
            continue
        row = np.zeros(nv)
        for i in range(i_start, i_end):
            j = i + shift
            row[i] = (1.0 - h0[j]) * dx
        row[-1] = -1.0
        rows.append(row)
        rhs.append(np.sum(h0[i_start:i_end] * h0[i_start + shift:i_end + shift]) * dx)

    if len(rows) == 0:
        return None

    A_ub = np.array(rows)
    b_ub = np.array(rhs)

    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = []
    for i in range(n):
        lo = max(0.0, h0[i] - trust_radius)
        hi = min(1.0, h0[i] + trust_radius)
        bounds.append((lo, hi))
    bounds.append((0.0, None))

    try:
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'presolve': True, 'dual_feasibility_tolerance': 1e-10})
        if res.success:
            return res.x[:n], res.x[-1]
    except Exception:
        pass
    return None

def slp_optimize(h, n, time_budget, start_time, max_iter=500, init_trust=0.5):
    """Sequential Linear Programming with trust region."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    trust = init_trust
    stall = 0

    for it in range(max_iter):
        if time.time() - start_time > time_budget - 5:
            break

        shifts = get_shifts_array(h)
        c5_cur = float(np.max(shifts))
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 20:
                trust *= 0.5
                if trust < 1e-8:
                    break
                stall = 0

        thresh = best_c5 - max(1e-7, 0.01 * best_c5)
        active = np.where(shifts >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(shifts)[-5:]
        if len(active) > 25:
            active = np.argsort(shifts)[-25:]

        sol = slp_single_lp(h, n, active, trust)
        if sol is None:
            trust *= 0.5
            continue
        h_new, t_new = sol

        if np.max(np.abs(h_new - h)) < 1e-12:
            trust *= 0.7
            continue

        # Relaxation to avoid cycling
        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_try = h + alpha * (h_new - h)
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = make_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try <= c5_cur + 1e-14:
                h = h_try
                accepted = True
                break
        if not accepted:
            h = h_new
            h = make_feasible(h, n)

        if c5_try is not None and c5_try < best_c5 - 1e-10:
            trust = min(1.0, trust * 1.1)

    h = make_feasible(h, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_h = h.copy()
        best_c5 = c5_final
    return best_h, best_c5

def nesterov_pgd(h, n, time_budget, start_time, top_m=5):
    """Nesterov momentum PGD on top-m shifts."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    mu = 0.85
    step = 0.04
    stall = 0

    while time.time() - start_time < time_budget - 3:
        y = np.clip(h - mu * velocity, 0, 1)
        corr = np.correlate(y, 1.0 - y, mode='full')
        top_idx = np.argsort(corr)[-top_m:]

        grad = np.zeros(n)
        for k_idx in top_idx:
            shift = k_idx - (n - 1)
            if shift >= 0:
                i_s, i_e = 0, n - shift
            else:
                i_s, i_e = -shift, n
            for i in range(i_s, i_e):
                grad[i] += (1.0 - y[i + shift])
        grad /= len(top_idx) * dx

        velocity = mu * velocity - step * grad
        h_new = np.clip(h + velocity, 0, 1)
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)

        if c5_new <= compute_c5(h) + 1e-13:
            h = h_new
            if c5_new < best_c5 - 1e-15:
                best_h = h.copy()
                best_c5 = c5_new
                stall = 0
            else:
                stall += 1
        else:
            step *= 0.75
            velocity *= 0.3
            stall += 1

        if stall > 40:
            if step > 1e-6:
                step *= 0.5
            else:
                break
            stall = 0
        if (it if 'it' in dir() else 0) % 200 == 0 and step < 0.03:
            step = min(0.05, step * 1.2)

    h = make_feasible(h, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_h, best_c5 = h, c5_final
    return best_h, best_c5

def smooth_max_slsqp(h, n, time_budget, start_time, T=2000.0, top_m=None):
    """SLSQP with log-sum-exp smooth max."""
    dx = 2.0 / n

    def obj(x):
        x = np.clip(x, 0, 1)
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None:
            top = np.sort(corr)[-top_m:]
            mc = top[-1]
            return (mc + np.log(np.sum(np.exp(T * (top - mc)))) / T) * dx
        mc = np.max(corr)
        return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

    def con(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-18})
        hx = make_feasible(res.x, n)
        cx = compute_c5(hx)
        if cx < compute_c5(h):
            return hx, cx
    except Exception:
        pass
    return h, compute_c5(h)

def coord_descent_fine(h, n, time_budget, start_time, steps=None):
    """Fine coordinate descent."""
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() - start_time < time_budget - 3:
        improved = False
        for j in range(n):
            if time.time() - start_time > time_budget - 3:
                break
            old = best_h[j]
            bc, bv = best_c5, old
            for s in steps:
                for sg in [-1.0, 1.0]:
                    v = old + sg * s
                    if 0 <= v <= 1 and abs(v - old) > 1e-12:
                        t = best_h.copy()
                        t[j] = v
                        t = make_feasible(t, n)
                        c = compute_c5(t)
                        if c < bc - 1e-15:
                            bc, bv = c, v
            for v in [0.0, 1.0]:
                if abs(v - old) > 1e-6:
                    t = best_h.copy()
                    t[j] = v
                    t = make_feasible(t, n)
                    c = compute_c5(t)
                    if c < bc - 1e-15:
                        bc, bv = c, v
            if bv != old:
                best_h[j] = bv
                best_h = make_feasible(best_h, n)
                best_c5 = bc
                improved = True
    return best_h, best_c5
