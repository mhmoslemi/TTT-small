        b[idx[rng.randint(len(idx))]] = 1
    dx = 2.0 / n
    curr = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
    best = curr
    best_b = b.copy()
    for it in range(iters):
        if time.time() > deadline:
            break
        frac = min(1.0, it / max(1, iters))
        T = T0 * (T1 / T0) ** frac
        blen = rng.randint(3, 20)
        start = rng.randint(0, n - blen)
        seg = b[start:start + blen]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi), rng.randint(1, 6))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        c = float(np.max(np.correlate(b, 1.0 - b, mode='full')) * dx)
        if c <= curr or (T > 1e-12 and rng.random() < np.exp(-(c - curr) / T)):
            curr = c
            if c < best:
                best = c
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best

def subgrad_step(h, n, alpha, T, n_active):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    weights = np.exp(T * (corr - mc))
    weights /= np.sum(weights)
    g = np.zeros(n)
    for k in range(len(corr)):
        if weights[k] < 1e-8:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                g[i + lag] += (1.0 - h[i])
                g[i + lag] -= h[i]
                g[i] -= h[i + lag]
        else:
            j = -lag
            for i in range(j, n):
                g[i] += (1.0 - h[i - j])
                g[i] -= h[i - j]
                g[i - j] -= h[i]
    g *= weights[np.arange(len(corr)) - (n - 1)] if False else 1.0
    # Simpler: just use the top active constraint subgradient
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            g[i + lag] += (1.0 - h[i])
            g[i + lag] -= h[i]
            g[i] -= h[i + lag]
    else:
        j = -lag
        for i in range(j, n):
            g[i] += (1.0 - h[i - j])
            g[i] -= h[i - j]
            g[i - j] -= h[i]
    h_new = h - alpha * g
    return make_feasible(h_new, n)

def multi_slq(h, n, deadline, min_time=5):
    best = h.copy()
    best_c5 = compute_c5(h)
    configs = [(300, 0.0, 150, 1e-14), (1000, 0.0, 150, 1e-14),
               (1000, 0.3, 150, 1e-14), (3000, 0.0, 100, 1e-14),
               (3000, 0.5, 100, 1e-14), (10000, 0.0, 80, 1e-14)]
    for T, lam, mi, ft in configs:
        if time.time() > deadline - min_time:
            break
        x = slsqp_opt(best, n, T, lam, mi, ft)
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best = make_feasible(x, n)
    return best, best_c5

def perturb_h(h, n, rng, frac=0.05):
    h_new = h.copy()
    n_flip = max(1, int(n * frac))
    idx = rng.choice(n, size=n_flip, replace=False)
    h_new[idx] = 1.0 - h_new[idx]
    return make_feasible(h_new, n)

def upsample(h, n_old, n_new):
    old_x = np.linspace(0.0, 2.0, n_old)
    new_x = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)
    deadline = start + budget_s - 3

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)
    n = len(h)
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SLSQP with structured objective from parent
    best_h, best_c5 = multi_slq(best_h, n, deadline, min_time=budget_s * 0.3)

    # Phase 2: Binary SA (long run)
    if time.time() < deadline - budget_s * 0.25:
        dl = deadline - budget_s * 0.35
        b, c5b = binary_sa(best_h, n, 5000000, 0.005, 0.0000005, rng, dl)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()

    # Phase 3: SLSQP from binary result
    if time.time() < deadline - budget_s * 0.2:
        best_h, best_c5 = multi_slq(best_h, len(best_h), deadline, min_time=budget_s * 0.15)

    # Phase 4: Block SA
    if time.time() < deadline - budget_s * 0.15:
        dl = deadline - budget_s * 0.2
        b, c5b = block_sa(best_h, len(best_h), 2000000, 0.004, 0.000001, rng, dl)
        if c5b < best_c5:
            best_c5 = c5b
            best_h = b.copy()
            if time.time() < deadline - budget_s * 0.1:
                best_h, best_c5 = multi_slq(best_h, len(best_h), deadline, min_time=budget_s * 0.08)

    # Phase 5: Try 2x resolution
    n_cur = len(best_h)
    if time.time() < deadline - budget_s * 0.2:
        n2 = n_cur * 2
        h2 = upsample(best_h, n_cur, n2)
        h2, c5_2 = multi_slq(h2, n2, min(deadline, start + budget_s * 0.55), min_time=30)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
            dl2 = deadline - budget_s * 0.1
            b2, c5b2 = binary_sa(best_h, n2, 1000000, 0.004, 0.000001, rng, dl2)
            if c5b2 < best_c5:
                best_c5 = c5b2
                best_h = b2.copy()
                best_h, best_c5 = multi_slq(best_h, n2, deadline, min_time=20)

    # Phase 6: ILS from best
    n_cur = len(best_h)
    if time.time() < deadline - budget_s * 0.1:
        for kick in range(5):
            if time.time() > deadline - 30:
                break
            hk = perturb_h(best_h, n_cur, rng, frac=0.03)
            hk, ck = multi_slq(hk, n_cur, min(deadline, time.time() + budget_s * 0.03), min_time=10)
            if ck < best_c5:
                best_c5 = ck
                best_h = hk.copy()

    # Phase 7: Final fine-tune SLSQP
    n_cur = len(best_h)
    if time.time() < deadline - 5:
        for T, lam in [(100, 0.0), (500, 0.2), (2000, 0.0), (2000, 0.4), (8000, 0.6)]:
            if time.time() > deadline - 3:
                break
            x = slsqp_opt(best_h, n_cur, T, lam, 80, 1e-15)
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = make_feasible(x, n_cur)

    # Final feasibility fix
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
