import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
    cnt = mask.sum()
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argmax(1.0 - h)
        else:
            idx = np.argmin(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, T, top_k=None):
    """Smooth max of correlation values using log-sum-exp."""
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argsort(corr)[-top_k:]
        vals = corr[idx]
    else:
        vals = corr
    m = vals.max()
    return (m + np.log(np.sum(np.exp(T * (vals - m)))) / T) * (2.0 / n)

def lnbfgs_optimize(h, n, budget_s, start_time, T=500.0, top_k=30):
    """L-BFGS-B with smooth max objective."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    def obj(x):
        return smooth_max_obj(x, n, T, top_k)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
        if time.time() - start_time > budget_s - 10:
            return best_h, best_c5
        c5_res = compute_c5(res.x)
        if c5_res < best_c5:
            best_h, best_c5 = res.x, c5_res
    except:
        pass
    return best_h, best_c5
