def compute_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)

    """Compute the vector of changes in correlations when swapping h[p](1→0) and h[q](0→1)."""

    changes = np.zeros(2 * n - 1)

        delta = 0.0
        if L >= 0:
            if p + L < n:
                delta += (-1.0 + h[p + L])
            if p >= L:
                delta += h[p - L]
            if q + L < n:
                delta += (1.0 - h[q + L])

                delta += h[p + M]
            if q >= M:
                delta += (1.0 - h[q - M])
            if q + M < n:
                delta -= h[q + M]
        changes[j] = delta
    return changes

def get_corr_delta(h, p, q):
    """O(n) computation of correlation changes from swapping h[p](1→0) and h[q](0→1)."""
    n = len(h)
    delta = np.zeros(2 * n - 1)
    # For L >= 0 (indices n-1 to 2n-2):
    L_range = np.arange(0, n)  # L = 0, 1, ..., n-1
    idx = L_range + (n - 1)  # indices in full corr
    
    # Term 1: (-1+h[p+L]) for p+L < n
    valid1 = L_range <= (n - 1 - p)
    delta[idx[valid1]] += (-1.0 + np.where(valid1, h[p + L_range[valid1]], 0))
    
    # Term 2: h[p-L] for p >= L
    valid2 = L_range <= p
    delta[idx[valid2]] += np.where(valid2, h[p - L_range[valid2]], 0)
    
    # Term 3: (1-h[q+L]) for q+L < n
    valid3 = L_range <= (n - 1 - q)
    delta[idx[valid3]] += np.where(valid3, (1.0 - h[q + L_range[valid3]]), 0)
    
    # Term 4: -h[q-L] for q >= L
    valid4 = L_range <= q
    delta[idx[valid4]] -= np.where(valid4, h[q - L_range[valid4]], 0)
    
    # For L < 0, i.e., negative lags (indices 0 to n-2):
    M_range = np.arange(1, n)  # M = 1, 2, ..., n-1
    idx_neg = (n - 1) - M_range  # indices in full corr for negative lags
    
    # Term 1: (-1+h[p-M]) for p >= M
    valid1 = M_range <= p
    delta[idx_neg[valid1]] += (-1.0 + np.where(valid1, h[p - M_range[valid1]], 0))
    
    # Term 2: h[p+M] for p+M < n
    valid2 = M_range <= (n - 1 - p)
    delta[idx_neg[valid2]] += np.where(valid2, h[p + M_range[valid2]], 0)
    
    # Term 3: (1-h[q-M]) for q >= M
    valid3 = M_range <= q
    delta[idx_neg[valid3]] += np.where(valid3, (1.0 - h[q - M_range[valid3]]), 0)
    
    # Term 4: -h[q+M] for q+M < n
    valid4 = M_range <= (n - 1 - q)
    delta[idx_neg[valid4]] -= np.where(valid4, h[q + M_range[valid4]], 0)
    
    return delta

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        room = (1.0 - h) if diff > 0 else h
        mask = room > 1e-14
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_opt(h0, n, T, mu, max_iter=300):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -60, 60)
        sm = mc + np.log(np.sum(np.exp(sh))) / T
        return sm * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-13})
    return make_feasible(res.x, n)

def active_set_opt(h0, n, n_outer=30, top_k=8, mu=1000.0):
    dx = 2.0 / n
    target = n / 2.0
    h = h0.copy()
    for _ in range(n_outer):
        corr = get_corr(h)
        idx = np.argpartition(corr, -top_k)[-top_k:]
        def obj(x):
            c = get_corr(x)
            return float(np.mean(c[idx])) * dx + mu * (x.sum() - target) ** 2
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 40, 'ftol': 1e-15})
        h_new = make_feasible(res.x, n)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            h = h_new
    return h

        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()

        for T in [300, 1500, 8000, 40000]:
            if rem() < 35:
                break
            h_opt = lbgfb_opt(h2, n2, T, mu=600, max_iter=300)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h2 = h_opt.copy()

        if rem() > 50:
            h_act = active_set_opt(h2, n2, n_outer=25, top_k=10, mu=1200)
            c_act = compute_c5(h_act)
            if c_act < best_c5:
                best_c5, best_h = c_act, h_act.copy()
                h2 = h_act.copy()

    # Phase 3: High-resolution binary local search
    if rem() > 150:
        n_bin = min(best_n, 1280)
        h_b = upsample_h(best_h, n_bin) if n_bin != best_n else best_h.copy()
        h_bin = (h_b > 0.5).astype(float)
        target = n_bin // 2
        n_ones = int(h_bin.sum())
        if n_ones > target:
            idx = np.where(h_bin == 1)[0]
            h_bin[idx[:n_ones - target]] = 0.0
        elif n_ones < target:
            idx = np.where(h_bin == 0)[0]
            h_bin[idx[:target - n_ones]] = 1.0
        h_bs, c_bs = efficient_binary_search(h_bin, n_bin, rng, max_rounds=50, tries=25000)
        if c_bs < best_c5:
            best_c5, best_h = c_bs, h_bs.copy()
            best_n = n_bin

    # Phase 4: Targeted search at higher res
    if rem() > 120:
        n_t = min(best_n, 1600)
        h_t = upsample_h(best_h, n_t) if n_t != best_n else best_h.copy()
        h_t_bin = (h_t > 0.5).astype(float)
        target = n_t // 2
        n_ones = int(h_t_bin.sum())
        if n_ones > target:
            idx = np.where(h_t_bin == 1)[0]
            h_t_bin[idx[:n_ones - target]] = 0.0
        elif n_ones < target:
            idx = np.where(h_t_bin == 0)[0]
            h_t_bin[idx[:target - n_ones]] = 1.0
        h_ts, c_ts = targeted_binary_search(h_t_bin, n_t, rng, max_rounds=25, tries=8000)
        if c_ts < best_c5:
            best_c5, best_h = c_ts, h_ts.copy()
            best_n = n_t

    # Phase 5: Multi-start
    n_ms = len(best_h)
    if rem() > 80:
        for restart in range(4):
            if rem() < 25:
                break
            h_r = upsample_h(best_h, n_ms) + 0.12 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            for T in [2000, 10000]:
                if rem() < 15:
                    break
                h_opt = lbgfb_opt(h_r, n_ms, T, mu=1000, max_iter=250)
                c_opt = compute_c5(h_opt)
                if c_opt < best_c5:
                    best_c5, best_h = c_opt, h_opt.copy()
                    h_r = h_opt.copy()

    # Phase 6: Final polish with very high T
    if rem() > 25:
        n_fin = len(best_h)
        h_fin = best_h.copy()
        for T in [20000, 80000, 200000]:
            if rem() < 12:
                break
            h_opt = lbgfb_opt(h_fin, n_fin, T, mu=3000, max_iter=300)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h = c_opt, h_opt.copy()
                h_fin = h_opt.copy()

    # Phase 7: One more binary search pass on best
    if rem() > 40:
        n_final = len(best_h)
        h_fb = (best_h > 0.5).astype(float)
        target = n_final // 2
        n_ones = int(h_fb.sum())
        if n_ones > target:
            idx = np.where(h_fb == 1)[0]
            h_fb[idx[:n_ones - target]] = 0.0
        elif n_ones < target:
            idx = np.where(h_fb == 0)[0]
            h_fb[idx[:target - n_ones]] = 1.0
        h_bs, c_bs = efficient_binary_search(h_fb, n_final, rng, max_rounds=20, tries=15000)
        if c_bs < best_c5:
            best_c5, best_h = c_bs, h_bs.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
