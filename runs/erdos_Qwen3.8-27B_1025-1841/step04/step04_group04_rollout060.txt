        b_plus = b + lags
        b_minus = b - lags
        
        va_p = (a_plus >= 0) & (a_plus < n)
        va_m = (a_minus >= 0) & (a_minus < n)
        vb_p = (b_plus >= 0) & (b_plus < n)
        vb_m = (b_minus >= 0) & (b_minus < n)
        
        delta = np.zeros(2 * n - 1)
        if np.any(va_p):
            delta[va_p] += h_bin[a_plus[va_p]]
        if np.any(va_m):
            delta[va_m] += h_bin[a_minus[va_m]]
        if np.any(vb_p):
            delta[vb_p] -= h_bin[b_plus[vb_p]]
        if np.any(vb_m):
            delta[vb_m] -= h_bin[b_minus[vb_m]]
        
        new_corr = corr + delta
        new_max = np.max(new_corr)
        new_c5 = float(new_max * (2.0 / n))
        delta_c5 = new_c5 - curr_c5
        
        # Acceptance
        accept = False
        if delta_c5 <= 0:
            accept = True
        else:
            if T > 1e-10:
                prob = np.exp(-delta_c5 / (T * (2.0 / n)))
                accept = rng.random() < prob
        
        if accept:
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            corr = new_corr
            curr_c5 = new_c5
            # Update position lists
            one_pos[ia] = b
            zero_pos[ib] = a
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_h = h_bin.copy()
                stuck_count = 0
            else:
                stuck_count += 1
        else:
            stuck_count += 1
        
        it += 1
        # Cool
        T *= alpha
        if T < t_end:
            T = t_end
        
        # If stuck for too long at very low T, warm restart
        if stuck_count > 5000 and T < t_end * 10:
            T = t_start * 0.1
            stuck_count = 0
    
    return best_h, best_c5

def multi_resolve_slsqp(h_start, n_pts, T, maxiter, time_fn):
    if time_fn() < 30:
        return None
    bounds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': lambda x: slsqp_con(x, n_pts),
             'jac': lambda x: slsqp_con_jac(x, n_pts)}]
    try:
        res = minimize(lambda x: slsqp_obj(x, n_pts, T), h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return res.x
    except Exception:
        return None

def coord_descent(h, c5_cur, n, time_fn, max_rounds=10):
    """Coordinate descent with several candidate values per coordinate."""
    for rnd in range(max_rounds):
        if time_fn() < 5:
            break
        improved = False
        for j in range(n):
            if time_fn() < 3:
                break
            old = h[j]
            best_c = c5_cur
            best_val = old
            cands = [0.0, 1.0, 0.5]
            cands += [max(0.0, min(1.0, old + d)) for d in [-0.2, -0.1, -0.05, 0.05, 0.1, 0.2]]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                c5_cur = best_c
                improved = True
        if not improved:
            break
    return h, c5_cur

def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values  # pre-imported
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP at initial resolution
    if remaining() > 120:
        for T_val in [80.0, 200.0]:
            if remaining() < 40:
                break
            x0 = best_h.copy()
            x0 = make_feasible(x0, n0)
            res = multi_resolve_slsqp(x0, n0, T_val, 200, remaining)
            if res is not None:
                c5_try = compute_c5(res)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = make_feasible(res, n0)

    # Phase 2: Binary SA at n0
    if remaining() > 200:
        h_bin = binary_project(best_h, rng)
        c5_bin = compute_c5(h_bin)
        sa_best_h, sa_best_c5 = sa_binary(
            h_bin.copy(), n0,
            min_iters=20000,
            max_iters=500000,
            t_start=0.02,
            t_end=0.0005,
            alpha=0.9993,
            rng=rng,
            time_fn=remaining
        )
        if sa_best_c5 < best_c5:
            best_c5 = sa_best_c5
            best_h = sa_best_h.copy()

    # Phase 3: SLSQP polish on binary result
    if remaining() > 80:
        for T_val in [300.0, 800.0]:
            if remaining() < 30:
                break
            res = multi_resolve_slsqp(best_h.copy(), n0, T_val, 150, remaining)
            if res is not None:
                c5_try = compute_c5(res)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = make_feasible(res, n0)

    # Phase 4: Binary SA round 2 (shorter)
    if remaining() > 120:
        h_bin2 = binary_project(best_h, rng)
        sa_best_h2, sa_best_c5_2 = sa_binary(
            h_bin2.copy(), n0,
            min_iters=10000,
            max_iters=300000,
            t_start=0.008,
            t_end=0.0002,
            alpha=0.9995,
            rng=rng,
            time_fn=remaining
        )
        if sa_best_c5_2 < best_c5:
            best_c5 = sa_best_c5_2
            best_h = sa_best_h2.copy()

    # Phase 5: Try higher resolution
    if remaining() > 120:
        n_new = n0 * 2
        x_grid_old = np.linspace(0.0, 2.0, n0)
        x_grid_new = np.linspace(0.0, 2.0, n_new)
        h_up = np.interp(x_grid_new, x_grid_old, best_h)
        h_up = make_feasible(h_up, n_new)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        if remaining() > 60:
            res = multi_resolve_slsqp(h_up.copy(), n_new, 400.0, 150, remaining)
            if res is not None:
                c5_try = compute_c5(res)
                if c5_try < best_c5:
                    best_c5 = c5_try
                    best_h = make_feasible(res, n_new)

    # Phase 6: Coordinate descent
    if remaining() > 30:
        best_h, best_c5 = coord_descent(best_h, best_c5, len(best_h), remaining, max_rounds=8)

    # Phase 7: Final SLSQP polish
    if remaining() > 40:
        n_final = len(best_h)
        res = multi_resolve_slsqp(best_h.copy(), n_final, 1000.0, 100, remaining)
        if res is not None:
            c5_try = compute_c5(res)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = make_feasible(res, n_final)

    # Final: short coord descent
    if remaining() > 5:
        best_h, best_c5 = coord_descent(best_h, best_c5, len(best_h), remaining, max_rounds=3)

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
