import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_run(h_start, n, T, mu, max_iter):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -80, 80)
        val = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return val + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-17, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def binary_search_focused(h_bin, n, rng, max_rounds):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = list(np.where(h > 0.5)[0])
    zeros = list(np.where(h <= 0.5)[0])
    for rnd in range(max_rounds):
        improved = False
        n_t = min(2000, len(ones) * len(zeros))
        for _ in range(n_t):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def block_swap_search(h, n, rng, max_rounds, block_len=2):
    h = h.copy()
    current = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for _ in range(1000):
            a = rng.randint(0, n - block_len + 1)
            if rng.random() < 0.5:
                # Try swapping block of 1s with block of 0s
                seg_a = h[a:a+block_len]
                if np.mean(seg_a) > 0.5:
                    b = rng.randint(0, n - block_len + 1)
                    if np.mean(h[b:b+block_len]) < 0.5:
                        h[a:a+block_len], h[b:b+block_len] = h[b:b+block_len].copy(), seg_a.copy()
                        c = compute_c5(h)
                        if c < current - 1e-15:
                            current = c
                            improved = True
                        else:
                            h[a:a+block_len], h[b:b+block_len] = seg_a.copy(), h[a:a+block_len].copy()
            else:
                # Single swap
                ones_idx = np.where(h > 0.5)[0]
                zeros_idx = np.where(h <= 0.5)[0]
                if len(ones_idx) == 0 or len(zeros_idx) == 0:
                    break
                oi = ones_idx[rng.randint(len(ones_idx))]
                oj = zeros_idx[rng.randint(len(zeros_idx))]
                h[oi] = 0.0
                h[oj] = 1.0
                c = compute_c5(h)
                if c < current - 1e-15:
                    current = c
                    improved = True
                else:
                    h[oi] = 1.0
                    h[oj] = 0.0
        if not improved:
            break
    return h, current

def make_periodic_pat(n, period, n_ones_in_period, offset=0):
    pat = np.zeros(period)
    pat[:n_ones_in_period] = 1.0
    rep = np.tile(pat, n // period + 1)[:n]
    rep = np.roll(rep, offset)
    return make_feasible(rep, n)
