def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        room = (1.0 - h) if diff > 0 else h
        mask = room > 1e-14
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

                break
            ho = lbgfb_smooth(hc, nc, T, 400.0, 80)
            co = compute_c5(ho)
            if co < best_c5:
                best_c5 = co
                best_h = ho.copy()
                hc = ho.copy()

    # Phase 4: Final binary SA on polished solution
    if rem() > 30:
        nf = min(len(best_h), 2048)
        hf = to_binary(best_h, nf)
        dl4 = time.time() + min(rem() * 0.45, 70)
        hsa, csa = sa_binary(hf, nf, rng, dl4)
        if csa < best_c5:
            best_c5 = csa
            best_h = hsa.copy()

        if rem() > 20:
            gdl = time.time() + min(rem() * 0.3, 25)
            hgp, cgp = greedy_polish(hsa, nf, gdl)
            if cgp < best_c5:
                best_c5 = cgp
                best_h = hgp.copy()

    best_h = make_feasible(best_h, len(best_h))
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
