import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_opt(h0, n, T, mu, max_iter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -50, 50)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def lp_norm_opt(h0, n, p, mu, max_iter):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        val = float(np.sum(np.abs(corr) ** p) ** (1.0 / p))
        return val * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def binary_anneal(h_bin, n, rng, time_limit, T_init=8e-4, T_final=1e-6):
    h = h_bin.copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c = cur
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    t_start = time.time()
    while time.time() - t_start < time_limit and ones and zeros:
        elapsed = time.time() - t_start
        frac = elapsed / time_limit
        T = T_init * (T_final / T_init) ** frac
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi], h[oj] = 0.0, 1.0
        c = compute_c5(h)
        delta = c - cur
        if delta < 0 or (T > 1e-8 and rng.rand() < np.exp(-delta / T)):
            cur = c
            ones[ii], zeros[jj] = oj, oi
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[oi], h[oj] = 1.0, 0.0
    return best_h, best_c

def binary_hill(h_bin, n, rng, time_limit):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    t_start = time.time()
    while time.time() - t_start < time_limit and ones and zeros:
        improved = False
        for _ in range(2000):
            if time.time() - t_start >= time_limit:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi], h[oj] = 0.0, 1.0
            c = compute_c5(h)
            if c < cur - 1e-14:
                cur = c
                ones[ii], zeros[jj] = oj, oi
                improved = True
            else:
                h[oi], h[oj] = 1.0, 0.0
        if not improved:
            break
    return h, cur

def to_binary(h, n):
    h_bin = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target:
        idx = np.where(h_bin == 1)[0]
        for i in range(n_ones - target):
            h_bin[idx[i]] = 0.0
    elif n_ones < target:
        idx = np.where(h_bin == 0)[0]
        for i in range(target - n_ones):
            h_bin[idx[i]] = 1.0
    return h_bin

def coord_polish(h0, n, max_rounds=4):
    h = h0.copy()
    cur = compute_c5(h)
    steps = [0.08, -0.08, 0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                h[j] = nv
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < cur - 1e-13:
                    cur = c
                    ov = nv
                    improved = True
                    break
            h[j] = ov
        if not improved:
            break
    return h, cur
