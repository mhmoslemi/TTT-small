def coord_polish(h0, n, max_rounds=5, steps=None):

            ov = h[j]

                    ov = nv

import cvxpy as cp

def mccormick_lp(h_init, n, active_shifts, n_iters=20):
    """Solve McCormick LP relaxation iteratively with Benders-style cuts."""
    dx = 2.0 / n
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    z_vars = {}  # z_vars[s] is an array of variables for shift s
    
    # McCormick constraints for each active shift
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    for s in active_shifts:
        z_vars[s] = cp.Variable(max(0, n - abs(s)))
        m = max(0, n - abs(s))
        if m == 0:
            continue
        if s >= 0:
            h1 = h_var[:m]
            h2 = h_var[s:s+m]
        else:
            h1 = h_var[(-s):]
            h2 = h_var[:m]
        
        z = z_vars[s]
        constraints += [z >= h1 + h2 - 1, z <= h1, z <= h2, z >= 0]
        # Constraint: sum(h1 - z) * dx <= t
        constraints.append(cp.sum(h1 - z) * dx <= t_var)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    # Try different solvers
    best_val = np.inf
    best_h = None
    for solver in [cp.GLOP, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status == 'optimal' and h_var.value is not None:
                if h_var.value.sum() * dx < 2.0:  # sanity check
                    h_sol = np.clip(h_var.value, 0, 1)
                    h_sol = make_feasible(h_sol, n)
                    c5 = compute_c5(h_sol)
                    if c5 < best_val:
                        best_val = c5
                        best_h = h_sol
        except Exception:
            continue
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_val(corr, T):
    mc = np.max(corr)
    if mc < 1e-15:
        return 0.0
    shifted = np.clip(T * (corr - mc), -50, 50)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_softmax(h0, n, T, mu, max_iter=300):
    dx = 2.0 / n
    target_sum = n / 2.0
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_val(corr, T)
        return sm * dx + mu * (x.sum() - target_sum) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def boosting_opt(h0, n, n_rounds=80, max_inner=30):
    """Iteratively reweighted: focus on worst lags."""
    dx = 2.0 / n
    target_sum = n / 2.0
    n_c = 2 * n - 1
    w = np.ones(n_c) / n_c
    h = h0.copy()
    
    for rnd in range(n_rounds):
        corr = compute_corr(h)
        # Update weights: boost the worst lags
        mc = np.max(corr)
        if mc > 1e-15:
            w_new = np.exp(3.0 * (corr - mc) / mc)  # shifts gap to [-3, 0]
            w_new = np.clip(w_new, 1e-3, None)
            w_new /= w_new.sum()
            w = w_new
        
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return float(np.dot(w, c)) * dx + 800.0 * (x.sum() - target_sum) ** 2
        
        try:
            res = minimize(obj, h, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': max_inner, 'ftol': 1e-13, 'gtol': 1e-9})
            h_new = make_feasible(res.x, n)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                h = h_new
        except Exception:
            pass
    
    return h

def binary_hill_climb(h_bin, n, rng, max_rounds=300, tries_per_round=5000):
    dx = 2.0 / n
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries_per_round, max(1, len(ones) * len(zeros)))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur_c5 - 1e-15:
                cur_c5 = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur_c5

def binary_sa(h_bin, n, rng, max_iters=5000, T_init=0.3, T_end=0.01):
    dx = 2.0 / n
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    cur_corr = compute_corr(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    
    for it in range(max_iters):
        T = T_init * (T_end / T_init) ** (it / max(1, max_iters - 1))
        if not ones or not zeros:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        new_corr = compute_corr(h)
        new_c5 = float(np.max(new_corr) * dx)
        delta = new_c5 - cur_c5
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            cur_c5 = new_c5
            cur_corr = new_corr
            ones[ii] = oj
            zeros[jj] = oi
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return h, cur_c5

def coord_polish(h0, n, max_rounds=4, steps=None):
    if steps is None:
        steps = [0.03, -0.03, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur

def slsqp_polish(h0, n, T, max_iter=100):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return soft_max_val(corr, T) * dx
    def con_eq(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq, 'jac': con_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        return make_feasible(res.x, n)
    except Exception:
        return h0
