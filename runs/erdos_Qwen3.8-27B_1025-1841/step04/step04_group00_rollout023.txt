        top_idx = np.arange(len(corr))
        c = corr
    
    Tc = T * c
    Tc_shifted = Tc - np.max(Tc)
    exps = np.exp(Tc_shifted)
    Z = np.sum(exps)
    p = exps / Z  # softmax weights
    
    obj = (np.max(Tc) + np.log(Z)) / T * dx

    for i, k in enumerate(top_idx):

        if pi < 1e-12:
            continue

            grad[0:n-s] += pi * (1.0 - h[s:n])
            grad[s:n] += pi * (-h[0:n-s])
        else:
            t = -s
            grad[t:n] += pi * (1.0 - h[0:n-t])
            grad[0:n-t] += pi * (-h[t:n])
    
    grad *= dx
    return obj, grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def corr_gradient(h, T, top_m):
    """Compute smooth max of correlation values and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        c = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        c = corr
    Tc = T * c
    Tc_max = np.max(Tc)
    exps = np.exp(Tc - Tc_max)
    Z = np.sum(exps)
    p = exps / Z
    obj = (Tc_max + np.log(Z)) / T * dx
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        pi = p[i]
        if pi < 1e-12:

        s = k - (n - 1)

            t = -s
            grad[t:n] += pi * (1.0 - h[0:n-t])
            grad[0:n-t] += pi * (-h[t:n])
    grad *= dx
    return obj, grad

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(fun_and_grad, h, method='SLSQP', jac=True,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        h_new = project_feasible(res.x, n)
        c5 = compute_c5(h_new)
        return h_new, c5
    except Exception:
        return h, compute_c5(h)

def binary_local_search(h, n, max_rounds=5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        conf = np.abs(best_h - 0.5)
        order = np.argsort(-conf)
        for j in order:
            ov = best_h[j]
            if ov > 0.05:
                tr = best_h.copy()
                tr[j] = 0.0
                tr = project_feasible(tr, n)
                c = compute_c5(tr)
                if c < best_c5 - 1e-12:
                    best_h, best_c5 = tr, c
                    improved = True
                    continue
            if ov < 0.95:
                tr = best_h.copy()
                tr[j] = 1.0
                tr = project_feasible(tr, n)
                c = compute_c5(tr)
                if c < best_c5 - 1e-12:
                    best_h, best_c5 = tr, c
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def fourier_init(n, K, seed):
    """Generate a Fourier-based initialization."""
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n)
    freqs = np.arange(1, K + 1)
    coeffs = rng.randn(2 * K) * 0.3
    h = np.full(n, 0.5)
    for k in range(K):
        h += coeffs[2*k] * np.cos(2*np.pi*freqs[k]*x/2.0)
        h += coeffs[2*k+1] * np.sin(2*np.pi*freqs[k]*x/2.0)
    h = project_feasible(h, n)
    return h

def pipeline(h, n, start_time, budget_s, label=""):
    """Run optimization pipeline on h at resolution n."""
    def rem():
        return budget_s - (time.time() - start_time)
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # Stage 1: broad exploration
    if rem() > 50:
        h2, c5 = lbfgs_optimize(best_h, n, T=50, top_m=80, penalty=500, maxiter=200)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    # Stage 2: narrows
    if rem() > 40:
        h2, c5 = lbfgs_optimize(best_h, n, T=200, top_m=30, penalty=1000, maxiter=250)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    # Stage 3: focus
    if rem() > 35:
        h2, c5 = slsqp_optimize(best_h, n, T=1000, top_m=10, maxiter=250)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    # Stage 4: binary push
    if rem() > 25:
        h2, c5 = binary_local_search(best_h, n, max_rounds=4)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    # Stage 5: high-T polish
    if rem() > 30:
        h2, c5 = slsqp_optimize(best_h, n, T=5000, top_m=5, maxiter=300)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    # Stage 6: more binary
    if rem() > 20:
        h2, c5 = binary_local_search(best_h, n, max_rounds=5)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    # Stage 7: final SLSQP
    if rem() > 25:
        h2, c5 = slsqp_optimize(best_h, n, T=10000, top_m=3, maxiter=300)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - start_time)

    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = project_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Run main pipeline at original resolution
    if rem() > 250:
        h2, c5 = pipeline(best_h, n, start_time, budget_s, "orig")
        if c5 < best_c5:
            best_h, best_c5 = h2, c5

    # Try Fourier-based initialization
    for K in [10, 15, 20]:
        if rem() < 250:
            break
        h_f = fourier_init(n, K, seed + K)
        h2, c5 = pipeline(h_f, n, start_time, budget_s, "fourier")
        if c5 < best_c5:
            best_h, best_c5 = h2, c5

    # Try 2x resolution
    if rem() > 300:
        n2 = n * 2
        if n2 <= 30000:
            old_x = np.linspace(0.0, 2.0, n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = project_feasible(h_up, n2)
            h2, c5 = pipeline(h_up, n2, start_time, budget_s, "2x")
            if c5 < best_c5:
                best_h, best_c5 = h2, c5

    # Random restarts at best resolution
    n_best = len(best_h)
    for restart in range(4):
        if rem() < 100:
            break
        h_try = project_feasible(best_h + rng.randn(n_best) * 0.015, n_best)
        h2, c5 = pipeline(h_try, n_best, start_time, budget_s, "restart")
        if c5 < best_c5:
            best_h, best_c5 = h2, c5

    # Final polish
    if rem() > 30:
        h2, c5 = slsqp_optimize(best_h, len(best_h), T=10000, top_m=3, maxiter=300)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5
    if rem() > 15:
        h2, c5 = binary_local_search(best_h, len(best_h), max_rounds=5)
        if c5 < best_c5:
            best_h, best_c5 = h2, c5

    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
