    coeffs = [a1, b1, a2, b2, ..., aK, bK]
    h(x) = 1/2 + sum_j (a_j cos(pi*j*x) + b_j sin(pi*j*x))
    where x = 2*i/n for i=0,...,n-1

        h += coeffs[2*(j-1)] * np.cos(np.pi * j * x)
        h += coeffs[2*(j-1)+1] * np.sin(np.pi * j * x)
    return h

        coeffs[2*(j-1)+1] = (2.0/n) * np.sum((h - 0.5) * np.sin(np.pi * j * x))

def augmented_lagrangian(h, n, budget_s, start_time, T=2000, outer_iters=20, inner_iters=200):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lam = 0.0
    rho = 100.0
    
    for outer in range(outer_iters):
        if time.time() - start_time > budget_s - 10:
            break
        
        def obj_aug(x):
            corr = np.correlate(x, 1.0-x, mode='full')
            mc = np.max(corr)
            shifted = corr - mc
            f = (mc + np.log(np.sum(np.exp(T*shifted)))/T) * (2.0/n)
            c = np.sum(x) - n/2.0
            return f + lam * c + 0.5 * rho * c * c
        
        res = minimize(obj_aug, h, method='L-BFGS-B',
                      bounds=[(0,1)]*n,
                      options={'maxiter': inner_iters, 'ftol': 1e-14, 'gtol': 1e-10})
        h_new = res.x
        c_new = np.sum(h_new) - n/2.0
        
        if abs(c_new) < 1e-6:
            # Constraint satisfied, check objective
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h, best_c5 = h_new, c5_new
            h = h_new
        else:
            lam = lam + rho * c_new
            rho = rho * 2
            h = h_new
    
    return best_h, best_c5

    g = 1.0 - h

    f_max = np.max(f_k)
    exp_f = np.exp(T * (f_k - f_max))
    log_sum = f_max + np.log(np.sum(exp_f)) / T
    w = exp_f / np.sum(exp_f)  # softmax weights

    # corr_k = sum_i h[i] * (1-h[i+lag_k]) where lag_k = k - (n-1)
    # d corr_k / d h[j] = (1-h[j+lag_k])*[valid] - h[j-lag_k]*[valid]

        # j - lag in [0, n-1] means lag in [j-(n-1), j]

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):

def smooth_max_obj_grad(h, T):
    """Compute smooth max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')

    # where lag_k = k - (n-1)

    lags = np.arange(n_c) - (n - 1)
    
    grad = np.zeros(n)
    for j in range(n):

        p_idx = j + lags

        c = np.sum(x) - n / 2.0
        return grad_f + (lam + rho * c) * np.ones(n)
    
    for outer in range(50):

            break
        
        res = minimize(obj_aug, h_cur, jac=jac_aug, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,

        h_cur = np.clip(res.x, 0.0, 1.0)

        if c5_new < best_c5:

            best_c5 = c5_new
    
    return best_h, best_c5

            nl = -lag
            if nl < n:
                grad[nl:] += wi * (1.0 - h[:n - nl])
                grad[:n - nl] -= wi * h[nl:]
    grad *= dx
    return val, grad

def aug_lag_lbfgsb(h, n, budget_s, start_time, T=5000, max_outer=30):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    lam = 0.0
    rho = 500.0
    h_cur = h.copy()
    for outer in range(max_outer):
        if time.time() - start_time > budget_s - 15:
            break
        lam_local = lam
        rho_local = rho
        def obj_a(x):
            val, _ = smooth_max_fg(x, T)
            c = np.sum(x) - n / 2.0
            return val + lam_local * c + 0.5 * rho_local * c * c
        def jac_a(x):
            _, gf = smooth_max_fg(x, T)
            c = np.sum(x) - n / 2.0
            return gf + (lam_local + rho_local * c)
        try:
            res = minimize(obj_a, h_cur, jac=jac_a, method='L-BFGS-B',
                          bounds=[(0.0, 1.0)] * n,
                          options={'maxiter': 200, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 50})
            h_cur = np.clip(res.x, 0.0, 1.0)
        except Exception:
            break
        c_val = np.sum(h_cur) - n / 2.0
        lam = lam + rho * c_val
        if abs(c_val) < 1e-8:
            rho *= 0.7
        elif abs(c_val) > 0.5:
            rho *= 2.0
        else:
            rho *= 1.3
        h_feas = make_feasible(h_cur, n)
        c5_new = compute_c5(h_feas)
        if c5_new < best_c5:
            best_h = h_feas.copy()
            best_c5 = c5_new
        h_cur = h_feas
    return best_h, best_c5

def fourier_opt(h, n, K, budget_s, start_time, T=2000):
    x = 2.0 * np.arange(n) / n
    h_centered = h - 0.5
    coeffs = np.zeros(2 * K)
    for j in range(1, K + 1):
        coeffs[2*(j-1)] = (2.0/n) * np.sum(h_centered * np.cos(np.pi * j * x))
        coeffs[2*(j-1)+1] = (2.0/n) * np.sum(h_centered * np.sin(np.pi * j * x))
    best_c5 = compute_c5(h)
    best_coeffs = coeffs.copy()
    for T_try in [500, 1500, 3000]:
        if time.time() - start_time > budget_s - 10:
            break
        def f_obj(c):
            hx = 0.5 * np.ones(n)
            for j in range(1, K+1):
                hx += c[2*(j-1)] * np.cos(np.pi * j * x)
                hx += c[2*(j-1)+1] * np.sin(np.pi * j * x)
            hx = np.clip(hx, 0.0, 1.0)
            corr = np.correlate(hx, 1.0 - hx, mode='full')
            f_k = corr * (2.0/n)
            fm = np.max(f_k)
            return fm + np.log(np.sum(np.exp(T_try * (f_k - fm)))) / T_try
        try:
            res = minimize(f_obj, best_coeffs, method='SLSQP',
                          bounds=[(-0.45, 0.45)] * (2*K),
                          options={'maxiter': 500, 'ftol': 1e-13})
            hx = 0.5 * np.ones(n)
            for j in range(1, K+1):
                hx += res.x[2*(j-1)] * np.cos(np.pi * j * x)
                hx += res.x[2*(j-1)+1] * np.sin(np.pi * j * x)
            hx = make_feasible(hx, n)
            c5_new = compute_c5(hx)
            if c5_new < best_c5:
                best_c5 = c5_new
                return hx, best_c5
        except Exception:
            pass
    return h, best_c5

def pgd_topm(h, n, budget_s, start_time, top_m=5, step_init=0.3):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    step = step_init
    vel = np.zeros(n)
    mom = 0.8
    no_impr = 0
    while time.time() - start_time < budget_s - 5:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        f_k = corr * (2.0/n)
        top_idx = np.argsort(f_k)[-top_m:]
        n_c = 2*n - 1
        lags = np.arange(n_c) - (n-1)
        grad = np.zeros(n)
        for k in top_idx:
            lag = lags[k]
            if lag >= 0:
                if lag < n:
                    grad[:n-lag] += (1.0 - best_h[lag:n])
                    grad[lag:] -= best_h[:n-lag]
            else:
                nl = -lag
                if nl < n:
                    grad[nl:] += (1.0 - best_h[:n-nl])
                    grad[:n-nl] -= best_h[nl:]
        grad *= (2.0/n) / top_m
        y = best_h - mom * vel
        y = np.clip(y, 0.0, 1.0)
        vel = mom * vel - step * grad
        new_h = best_h + vel
        new_h = make_feasible(new_h, n)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            no_impr = 0
        else:
            new_h2 = make_feasible(best_h - step * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 < best_c5 - 1e-15:
                best_c5 = c5_new2
                vel *= 0.3
            else:
                no_impr += 1
                step *= 0.6
                vel *= 0.3
        best_h = new_h if c5_new <= (best_c5 if c5_new < best_c5 else compute_c5(new_h)) else best_h
        best_h = new_h
        new_h = make_feasible(new_h, n)
        c5_check = compute_c5(new_h)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = new_h
        else:
            best_h = best_h
        if no_impr > 50:
            step *= 0.5
            if step < 1e-6:
                break
            no_impr = 0
        if no_impr % 100 == 0 and step < 0.1:
            step = min(0.3, step * 1.5)
    return best_h, best_c5

def coord_descent_fine(h, n, budget_s, start_time, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    improved = True
    while improved and time.time() - start_time < budget_s - 3:
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in steps:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            for v in [0.0, 1.0]:
                if abs(v - old_val) > 0.01:
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Fourier domain optimization (K=15 modes)
    if remaining() > 200:
        h_f, c5_f = fourier_opt(best_h, n, 15, min(80, remaining() - 20), start_time)
        if c5_f < best_c5:
            best_h, best_c5 = h_f, c5_f

    # Phase 2: Augmented Lagrangian + L-BFGS-B (main optimization)
    if remaining() > 300:
        h_al, c5_al = aug_lag_lbfgsb(best_h, n, min(250, remaining() - 30), start_time, T=5000)
        if c5_al < best_c5:
            best_h, best_c5 = h_al, c5_al

    # Phase 3: PGD with top-5 correlations
    if remaining() > 150:
        h_p, c5_p = pgd_topm(best_h, n, min(120, remaining() - 20), start_time, top_m=5, step_init=0.2)
        if c5_p < best_c5:
            best_h, best_c5 = h_p, c5_p

    # Phase 4: Higher T refinement with Aug Lag
    if remaining() > 200:
        h_al2, c5_al2 = aug_lag_lbfgsb(best_h, n, min(150, remaining() - 20), start_time, T=10000)
        if c5_al2 < best_c5:
            best_h, best_c5 = h_al2, c5_al2

    # Phase 5: PGD with top-3 (sharper focus)
    if remaining() > 100:
        h_p3, c5_p3 = pgd_topm(best_h, n, min(80, remaining() - 15), start_time, top_m=3, step_init=0.1)
        if c5_p3 < best_c5:
            best_h, best_c5 = h_p3, c5_p3

    # Phase 6: Coordinate descent fine polish
    if remaining() > 80:
        h_cd, c5_cd = coord_descent_fine(best_h, n, min(60, remaining() - 15), start_time)
        if c5_cd < best_c5:
            best_h, best_c5 = h_cd, c5_cd

    # Phase 7: Another Aug Lag round
    if remaining() > 120:
        h_al3, c5_al3 = aug_lag_lbfgsb(best_h, n, min(100, remaining() - 20), start_time, T=8000)
        if c5_al3 < best_c5:
            best_h, best_c5 = h_al3, c5_al3

    # Phase 8: Try upsample to 2n
    if remaining() > 250:
        n2 = n * 2
        if n2 <= 2400:
            old_x = np.linspace(0.0, 2.0, n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n2, endpoint=False)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = make_feasible(h_up, n2)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5 + 0.005:
                if remaining() > 150:
                    h_u2, c5_u2 = aug_lag_lbfgsb(h_up, n2, min(120, remaining() - 30), start_time, T=5000)
                    if c5_u2 < best_c5:
                        best_h, best_c5 = h_u2, c5_u2
                if remaining() > 80:
                    h_u3, c5_u3 = pgd_topm(best_h, len(best_h), min(60, remaining() - 20), start_time, top_m=5, step_init=0.15)
                    if c5_u3 < best_c5:
                        best_h, best_c5 = h_u3, c5_u3

    # Phase 9: Final PGD rounds with different top_m
    for tm in [4, 6]:
        if remaining() < 40:
            break
        h_pf, c5_pf = pgd_topm(best_h, len(best_h), min(30, remaining() - 10), start_time, top_m=tm, step_init=0.08)
        if c5_pf < best_c5:
            best_h, best_c5 = h_pf, c5_pf

    # Phase 10: Ultra-fine coordinate descent
    if remaining() > 25:
        h_cf, c5_cf = coord_descent_fine(best_h, len(best_h), min(20, remaining() - 5), start_time,
                                         steps=[0.002, 0.001, 0.0005, 0.0002])
        if c5_cf < best_c5:
            best_h, best_c5 = h_cf, c5_cf

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
