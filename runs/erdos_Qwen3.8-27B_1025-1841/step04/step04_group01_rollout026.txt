   - Adjust h to increase q_τ for bottleneck shifts while maintaining the sum constraint

**Key optimization**: When we change h[j] from old to new, the correlation changes as:
corr_new[tau] = corr_old[tau] + A_tau * (new - old)

where A_tau = (1 - h[j+tau]) - h[j-tau] (as derived above).

So we can update the correlation in O(n) time (for all 2n-1 shifts) instead of O(n²).

Then to find the best new value of h[j], we need to minimize max_tau (corr[tau] + A_tau * delta) over delta in some range.

This is still the "minimize max of linear functions" problem, but now we can do it in O(n log n) using the bisection approach.

Total per coordinate: O(n) for A computation + O(n log n) for bisection = O(n log n)
Total per pass: O(n² log n)
For n=960: ~960 * 960 * 10 = ~9.2M operations per pass. That's fast!

```python
    best_c5 = compute_c5(h)
    corr = get_corr(h).copy()  # current correlation
    
    for p in range(max_passes):
        improved = False
        for j in range(n):
            old = h[j]
            
            # Compute A[tau] for all shifts
            # corr[tau + n - 1] = sum_i h[i]*(1-h[i+tau])
            # When h[j] changes by delta:
            #   delta * [(1-h[j+tau]) if 0<=j+tau<n] - delta * [h[j-tau] if 0<=j-tau<n]
            A = np.zeros(2*n - 1)
            shift = np.arange(2*n-1) - (n-1)  # shifts from -(n-1) to (n-1)
            # h[j+shift] valid where 0 <= j+shift < n
            idx1 = j + shift
            valid1 = (idx1 >= 0) & (idx1 < n)
            A[valid1] += (1.0 - h[idx1[valid1]])
            # h[j-shift] valid where 0 <= j-shift < n  
            idx2 = j - shift
            valid2 = (idx2 >= 0) & (idx2 < n)
            A[valid2] -= h[idx2[valid2]]
            
            # f(delta) = max_tau (corr[tau] + A[tau] * delta)
            # where delta = new_h[j] - old
            # We want delta in range such that new_h[j] = old + delta in [0, 1]
            # i.e., delta in [-old, 1-old]
            # Also need to maintain sum: sum(new_h) = n/2
            # If we only change h[j], sum changes by delta
            # We need to compensate: distribute -delta among other entries
            # For simplicity, distribute evenly among all other entries that can absorb it
            
            # For now, let's find the unconstrained optimum of f(delta)
            # then handle the sum constraint separately
            
            lo = -old
            hi = 1.0 - old
            
            # Bisection on subgradient
            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = corr + A * mid
                max_val = np.max(vals)
                active_mask = vals >= max_val - 1e-12
                min_A = A[active_mask].min()
                max_A = A[active_mask].max()
                if min_A >= -1e-14:
                    hi = mid
                elif max_A <= 1e-14:
                    lo = mid
                else:
                    break
            
            delta_opt = (lo + hi) / 2.0
            new_val = old + delta_opt
            
            if abs(new_val - old) > 1e-8:
                # Try this change
                h[j] = new_val
                corr = corr + A * delta_opt  # update correlation
                h = make_feasible(h, n)
                # Recompute corr after make_feasible (since it changed other entries)
                corr = get_corr(h)
                c_test = compute_c5(h)
                if c_test < best_c5 - 1e-15:
                    best_c5 = c_test
                    improved = True
                else:
                    h[j] = old
                    corr = get_corr(h)  # revert
                    h = make_feasible(h, n)
                    corr = get_corr(h)
        
        if not improved:
            break
    
    return h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
    return h

def autocorr_fft(h):
    n = len(h)
    N = 1
    while N < 2*n:
        N *= 2
    F = np.fft.rfft(h, N)
    acf = np.fft.irfft(F * np.conj(F), N)[:n]
    return acf

def coord_descent(h, n, max_passes=10, tol=1e-12):
    """Coordinate descent with exact 1D subgradient optimization"""
    best_c5 = compute_c5(h)
    
    for p in range(max_passes):
        improved = False
        for j in range(n):
            old = h[j]
            if old <= 1e-10 or old >= 1.0 - 1e-10:
                # Try to move away from boundary
                candidates = [0.0 if old > 0.5 else 1.0, 0.5]
                best_t = old
                best_c = best_c5
                for t in candidates:
                    trial = h.copy()
                    trial[j] = t
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c:
                        best_c = c
                        best_t = t
                if best_t != old:
                    h = h.copy()
                    h[j] = best_t
                    h = make_feasible(h, n)
                    best_c5 = best_c
                    improved = True
                continue
            
            # Compute A[tau]
            shifts = np.arange(2*n-1) - (n-1)
            A = np.zeros(2*n - 1)
            idx1 = j + shifts
            v1 = (idx1 >= 0) & (idx1 < n)
            A[v1] += (1.0 - h[idx1[v1]])
            idx2 = j - shifts
            v2 = (idx2 >= 0) & (idx2 < n)
            A[v2] -= h[idx2[v2]]
            
            # Current correlation
            corr = get_corr(h)
            
            # Bisection to find optimal delta
            lo, hi = -old, 1.0 - old
            for _ in range(40):
                mid = (lo + hi) / 2.0
                vals = corr + A * mid
                mv = vals.max()
                am = vals >= mv - tol
                if am.any():
                    minA = A[am].min()
                    maxA = A[am].max()
                    if minA >= -1e-14:
                        hi = mid
                    elif maxA <= 1e-14:
                        lo = mid
                    else:
                        break
            
            delta = (lo + hi) / 2.0
            new_val = old + delta
            
            if abs(new_val - old) > 1e-8:
                trial = h.copy()
                trial[j] = new_val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    h = trial
                    best_c5 = c
                    improved = True
        
        if not improved:
            break
    
    return h, best_c5

def subgrad_optimize(h, n, steps=2000, lr_init=0.02, decay=0.9995, top_frac=0.15):
    """Subgradient method to maximize min autocorrelation"""
    best = h.copy()
    best_c5 = compute_c5(h)
    cur = h.copy()
    lr = lr_init
    
    for s in range(steps):
        acf = autocorr_fft(cur)
        min_q = acf.min()
        max_q = acf.max()
        spread = max_q - min_q
        if spread < 1e-14:
            break
        threshold = min_q + top_frac * spread
        active = np.where(acf <= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmin(acf))])
        
        # Gradient of min q = average of grad q_k over active
        grad = np.zeros(n)
        for k in active:
            if k == 0:
                grad += 2.0 * cur
            else:
                grad[:n-k] += cur[k:]
                grad[k:] += cur[:n-k]
        grad /= len(active)
        
        cur = cur + lr * grad
        cur = make_feasible(cur, n)
        
        c5 = compute_c5(cur)
        if c5 < best_c5:
            best_c5 = c5
            best = cur.copy()
        
        lr *= decay
    
    return best, best_c5

def slsqp_polish(h, n, T=300.0, maxiter=300):
    """SLSQP with smooth max approximation"""
    def obj(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    def obj_grad(x):
        dx = 2.0 / n
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        weights = np.exp(T * shifted)
        weights /= weights.sum()
        # Gradient of mc + log(sum(exp(T*(corr-mc))))/T with respect to x
        # d(corr[tau])/dx[j] = (1-x[j+shift]) - x[j-shift] (as before)
        # grad = sum_tau weights[tau] * d(corr[tau])/dx[j] * dx
        grad = np.zeros(n)
        for tau in range(2*n-1):
            w = weights[tau]
            if w < 1e-15:
                continue
            shift = tau - (n-1)
            idx1 = np.arange(n) + shift
            v1 = (idx1 >= 0) & (idx1 < n)
            # d corr[tau] / dx[j] for j where j+shift is valid: (1 - x[j+shift])
            # for j where j-shift is valid: -x[j-shift]
            g = np.zeros(n)
            g[v1] += (1.0 - x[idx1[v1]])
            idx2 = np.arange(n) - shift
            v2 = (idx2 >= 0) & (idx2 < n)
            g[v2] -= x[idx2[v2]]
            grad += w * g
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', jac=obj_grad,
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-16})
        c5 = compute_c5(res.x)
        return make_feasible(res.x, n), c5
    except Exception:
        return h.copy(), compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)
    
    n = len(h)
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Phase 1: Subgradient method
    if remaining() > 300:
        h1, c5_1 = subgrad_optimize(best_h, n, steps=3000, lr_init=0.03, decay=0.9998, top_frac=0.2)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1
    
    # Phase 2: Coordinate descent
    if remaining() > 200:
        h2, c5_2 = coord_descent(best_h, n, max_passes=8, tol=1e-10)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
    
    # Phase 3: SLSQP polish
    if remaining() > 100:
        h3, c5_3 = slsqp_polish(best_h, n, T=400.0, maxiter=500)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
    
    # Phase 4: More coordinate descent
    if remaining() > 100:
        h4, c5_4 = coord_descent(best_h, n, max_passes=10, tol=1e-12)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    
    # Phase 5: Random perturbations + re-optimize
    for restart in range(6):
        if remaining() < 80:
            break
        scale = 0.01 if restart < 3 else 0.02
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        
        # Quick subgradient
        h_try, c5_try = subgrad_optimize(h_try, n, steps=500, lr_init=0.02, decay=0.999, top_frac=0.15)
        # Coordinate descent
        if remaining() > 40:
            h_try, c5_try = coord_descent(h_try, n, max_passes=5, tol=1e-11)
        
        if c5_try < best_c5:

            best_h = h_try
    
    # Phase 6: Final SLSQP
```
