def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    return np.correlate(h, 1.0 - h, mode='full')

    return idx[idx[::-1]]

        corr = compute_corr(best_h)
        top_idx = top_k(corr, top_m)
        worst_si = top_idx[0]
        s = worst_si - (n - 1)
        if s >= 0:
            valid_i = np.arange(0, n - s)
        else:
            valid_i = np.arange(-s, n)
        contrib = valid_i[(best_h[valid_i] == 1.0) & (best_h[valid_i + s] == 0.0)]
        if len(contrib) == 0:
            patience += 1
            if patience > 3000:
                break
            continue
        i_move = int(contrib[rng.randint(len(contrib))])
        zeros = np.where(best_h == 0.0)[0]
        n_samples = min(30, len(zeros))
        cands = zeros[rng.choice(len(zeros), size=n_samples, replace=False)]
        best_target = -1
        best_score = np.inf
        for i0 in cands:
            if i0 == i_move:
                continue
            trial = best_h.copy()
            trial[i_move] = 0.0
            trial[i0] = 1.0
            vals = eval_at_shifts(trial, top_idx, n)
            sc = float(np.max(vals))
            if sc < best_score:
                best_score = sc
                best_target = int(i0)
        if best_target >= 0:
            best_h[i_move] = 0.0
            best_h[best_target] = 1.0
            patience = 0
        else:
            patience += 1
        if it % 30 == 0:
            c = compute_corr(best_h)
            c5 = float(np.max(c) * dx)
            if c5 < best_c5:
                best_c5 = c5
                patience = 0
        if patience > 5000:
            break
    corr = compute_corr(best_h)
    best_c5 = float(np.max(corr) * dx)
    return best_h, best_c5

def slsqp_refine(h, n, budget_s, start_time, T=800.0):
    from scipy.optimize import minimize
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(c)
        shifted = c - mc
        smax = mc + np.log(np.sum(np.exp(T * shifted))) / T
        return smax * (2.0 / n)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 300, 'ftol': 1e-16})
        c5 = compute_c5(res.x)
        if c5 < compute_c5(h):
            return make_feasible(res.x, n), c5
    except Exception:
        pass
    return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_parent = np.full(960, 0.5)

    n_parent = len(h_parent)
    h_parent = make_feasible(h_parent, n_parent)
    c5_parent = compute_c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = c5_parent

    # Phase 1: Binary local search at parent resolution
    h_bin = to_binary(h_parent, n_parent)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin

    if remaining() > 180:
        h_b = to_binary(best_h, n_parent) if not np.all((best_h == 0) | (best_h == 1)) else best_h.copy()
        h, c5 = local_search(h_b, n_parent, min(250, remaining() - 40), start_time, rng, top_m=20, T_init=1.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: Directed search
    if remaining() > 150:
        h_b = to_binary(best_h, n_parent) if not np.all((best_h == 0) | (best_h == 1)) else best_h.copy()
        h, c5 = directed_search(h_b, n_parent, min(180, remaining() - 40), start_time, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: Another local search round
    if remaining() > 120:
        h_b = to_binary(best_h, n_parent) if not np.all((best_h == 0) | (best_h == 1)) else best_h.copy()
        h, c5 = local_search(h_b, n_parent, min(150, remaining() - 30), start_time, rng, top_m=25, T_init=0.5)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Try different resolutions
    for n_try in [480, 256, 200]:
        if remaining() < 100:
            break
        x_old = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_try)
        h_down = np.interp(x_new, x_old, best_h)
        h_down = make_feasible(h_down, n_try)
        h_db = to_binary(h_down, n_try)
        if remaining() > 60:
            h2, c5_2 = local_search(h_db, n_try, min(50, remaining() - 30), start_time, rng, top_m=15, T_init=2.0)
            if c5_2 < best_c5:
                best_h, best_c5 = h2, c5_2

    # Phase 5: Perturbation + local search (multiple restarts)
    n_cur = len(best_h)
    for restart in range(10):
        if remaining() < 50:
            break
        h_b = to_binary(best_h, n_cur)
        k_flip = 3 + restart * 2
        ones = np.where(h_b == 1.0)[0]
        zeros = np.where(h_b == 0.0)[0]
        if len(ones) < k_flip or len(zeros) < k_flip:
            break
        ff = ones[rng.choice(len(ones), size=k_flip, replace=False)]
        ft = zeros[rng.choice(len(zeros), size=k_flip, replace=False)]
        h_b[ff] = 0.0
        h_b[ft] = 1.0
        h, c5 = local_search(h_b, n_cur, min(35, remaining() - 20), start_time, rng, top_m=18, T_init=0.8)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: Real-valued SLSQP refinement
    if remaining() > 60:
        h_r = make_feasible(best_h + rng.randn(len(best_h)) * 0.005, len(best_h))
        h_r, c5_r = slsqp_refine(h_r, len(best_h), remaining() - 20, start_time, T=1000.0)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    # Phase 7: Final binary local search
    if remaining() > 30:
        n_f = len(best_h)
        h_b = to_binary(best_h, n_f) if not np.all((best_h == 0) | (best_h == 1)) else best_h.copy()
        h, c5 = local_search(h_b, n_f, min(30, remaining() - 5), start_time, rng, top_m=15, T_init=0.3)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 8: Final SLSQP
    if remaining() > 25:
        h_r, c5_r = slsqp_refine(best_h, len(best_h), remaining() - 5, start_time, T=2000.0)
        if c5_r < best_c5:
            best_h, best_c5 = h_r, c5_r

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
