- The entry at index n-1+j corresponds to lag j>0: sum_{i=0}^{n-1-j} h[i]*(1-h[i+j])
- The entry at index n-1-j corresponds to lag -j: sum_{i=j}^{n-1} h[i]*(1-h[i-j])

So the "max over k" includes all lags from -(n-1) to (n-1), but the integral interpretation is:
∫₀² h(x)(1-h(x+k)) dx ≈ sum_{i: x_i and x_i+k both in [0,2]} h[i]*dx*(1-h[i+k_idx])

For the full correlation, the effective domain shrinks as |k| increases.

            break

                h[j] = old

            break

    best_c5 = compute_c5(h)

      = m'(v) + Σ f_j'(v) * p_j - m'(v)
      = Σ f_j'(v) * p_j
      = Σ a_j * p_j

where p_j = exp(T(f_j - m)) / Σ exp(T(f_i - m))

So the derivative is a weighted average of the a_j's. For a linear function f_j = a_j * v + b_j, f_j' = a_j.

The optimum satisfies Σ a_j * p_j = 0.

For our problem, when h[j] changes by δ:
- corr_full[m] for various m change linearly in δ
- Each corr_full[m] = Σ_i h[i]*(1-h[i+m-n+1])
- Terms involving h[j]:
  1. i=j: term = h[j]*(1-h[j+m-n+1]) → derivative w.r.t. h[j] is (1-h[j+m-n+1])
  2. i+m-n+1=j, i.e., i = j+m-n+1: term = h[i]*(1-h[j]) → derivative is -h[i]

So for each m (corresponding to lag k = m-(n-1)):
a_m = (1-h[j+k])_valid - h[j-k]_valid (with appropriate boundary handling)

This is computable in O(n) for a given j. Then I can solve for the optimal h[j] using the weighted equation.

import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(60):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def sa_run(h_init, n, max_iters, T0, T1, rng, block_prob=0.25, max_block=8):
    b = to_binary(h_init, n)
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        if T < 1e-15:
            T = 1e-15
        if rng.random() < block_prob:
            blen = rng.randint(2, min(max_block, n // 6) + 1)
            start = rng.randint(0, n - blen)
            old_seg = b[start:start + blen].copy()
            b[start:start + blen] = 1.0 - b[start:start + blen]
            target = n // 2
            diff = target - int(np.sum(b))
            if diff > 0:
                idxs = np.where(b == 0)[0]
                for _ in range(min(diff, len(idxs))):
                    p = idxs[rng.randint(len(idxs))]
                    b[p] = 1.0
            elif diff < 0:
                idxs = np.where(b == 1)[0]
                for _ in range(min(-diff, len(idxs))):
                    p = idxs[rng.randint(len(idxs))]
                    b[p] = 0.0
            new_c = compute_c5(b)
            if new_c <= curr or (T > 1e-15 and rng.random() < np.exp(-(new_c - curr) / T)):
                curr = new_c
                if new_c < best:
                    best = new_c
                    best_b = b.copy()
            else:
                b[start:start + blen] = old_seg
        else:
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                continue
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0.0
            b[j] = 1.0
            new_c = compute_c5(b)
            if new_c <= curr or (T > 1e-15 and rng.random() < np.exp(-(new_c - curr) / T)):
                curr = new_c
                if new_c < best:
                    best = new_c
                    best_b = b.copy()
            else:
                b[i] = 1.0
                b[j] = 0.0
    return best_b, best

def obj_sm(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    s = np.sum(np.exp(T * (corr - mc)))
    return (mc + np.log(s) / T) * dx

def con_s(x):
    return np.sum(x) - len(x) / 2.0

def con_sj(x):
    return np.ones(len(x))

def slsqp(h, n, T, maxit, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_s, 'jac': con_sj}]
    res = minimize(obj_sm, h.copy(), method='SLSQP', bounds=bnds,
                   constraints=cons, options={'maxiter': maxit, 'ftol': ftol})
    return res.x

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.1), max(0.0, old - 0.03),
                     min(1.0, old + 0.03), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def targeted_local(h, n, rng, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    b = to_binary(h, n)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 3:
            break
        improved = False
        corr = np.correlate(b, 1.0 - b, mode='full')
        worst_idx = int(np.argmax(corr))
        lag = worst_idx - (n - 1)
        if lag <= 0:
            lag = -lag
        if lag >= n:
            break
        nm = n - abs(lag)
        if lag > 0:
            contrib = (b[:nm] == 1) & (b[lag:n] == 0)
        else:
            contrib = (b[-lag:] == 1) & (b[:nm] == 0)
        contrib_idx = np.where(contrib)[0]
        if len(contrib_idx) == 0:
            break
        n_try = min(30, len(contrib_idx))
        for _ in range(n_try):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 2:
                break
            i = contrib_idx[rng.randint(len(contrib_idx))]
            if lag > 0:
                pos_a = i
                pos_b = i + lag
            else:
                pos_a = i - lag
                pos_b = i
            b[pos_a] = 0.0
            best_j_c5 = best_c5
            best_j = -1
            zeros = np.where(b == 0)[0]
            n_sample = min(15, len(zeros))
            if n_sample > 0:
                sample = zeros[rng.choice(len(zeros), size=n_sample, replace=False)]
                for j in sample:
                    b[j] = 1.0
                    c = compute_c5(b)
                    b[j] = 0.0
                    if c < best_j_c5 - 1e-15:
                        best_j_c5 = c
                        best_j = j
            if best_j >= 0:
                b[pos_a] = 0.0
                b[best_j] = 1.0
                best_c5 = best_j_c5
                improved = True
                break
            else:
                b[pos_a] = 1.0
        if not improved:
            break
    return b, best_c5

def perturb_reopt(h, n, rng, T, maxit):

        x2 = make_feasible(x2, n)
        c2 = compute_c5(x2)
        return x2, c2

        return h2, compute_c5(h2)
