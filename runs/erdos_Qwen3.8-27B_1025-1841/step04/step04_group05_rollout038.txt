import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def project_feasible(h_temp, n):
    """Project h_temp onto {h : 0 ≤ h ≤ 1, sum(h) = n/2}"""
    target = n / 2.0
    # Bisection on lambda: find lambda such that sum(clip(h_temp - lambda, 0, 1)) = target
    lo = h_temp.min() - 1.0
    hi = h_temp.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h_temp - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h_temp - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        if mask.sum() > 0:
            proj[mask] += diff / mask.sum()
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def compute_gradient(h, n, mu):
    """Compute the mu-smoothed gradient of max_k corr[k]*dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    corr_shifted = corr - np.max(corr)
    w = np.exp(corr_shifted / mu)
    w /= w.sum()
    
    # Compute weighted sum of gradients
    # For lag k (0-indexed in full correlation), actual lag = k - (n-1)
    grad = np.zeros(n)
    for idx in range(len(w)):
        if w[idx] < 1e-10:
            continue
        k = idx - (n - 1)  # actual lag
        if k == 0:
            grad += w[idx] * (1.0 - 2.0 * h)
        elif k > 0:
            # ∂c_k/∂h[i] = (1-h[i-k]) * 1[k ≤ i ≤ n-1] - h[i+k] * 1[0 ≤ i ≤ n-1-k]
            if k < n:
                # First term: i from k to n-1
                grad[k:] += w[idx] * (1.0 - h[:n-k])
                # Second term: i from 0 to n-1-k
                grad[:n-k] -= w[idx] * h[k:]
        else:
            m = -k
            # ∂c_{-m}/∂h[i] = (1-h[i+m]) * 1[0 ≤ i ≤ n-1-m] - h[i-m] * 1[m ≤ i ≤ n-1]
            if m < n:
                grad[:n-m] += w[idx] * (1.0 - h[m:])
                grad[m:] -= w[idx] * h[:n-m]
    
    return grad * dx

def gradient_descent_step(h, n, mu, alpha, max_ls=20):
    """One projected gradient descent step with backtracking line search"""
    grad = compute_gradient(h, n, mu)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for ls in range(max_ls):
        a = alpha * (0.5 ** ls)
        h_new = h - a * grad
        h_new = project_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        else:
            break
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def project_feasible(h_temp, n):
    target = n / 2.0
    lo = h_temp.min() - 1.0
    hi = h_temp.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        proj = np.clip(h_temp - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h_temp - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - proj.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = proj < 1.0 - 1e-10
        else:
            mask = proj > 1e-10
        cnt = mask.sum()
        if cnt > 0:
            proj[mask] = np.clip(proj[mask] + diff / cnt, 0.0, 1.0)
    return proj

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def compute_smoothed_gradient(h, n, mu):
    """Compute gradient of mu-smoothed max of correlation integrals."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    cmax = np.max(corr)
    exp_corr = np.exp((corr - cmax) / mu)
    w = exp_corr / np.sum(exp_corr)
    
    # Define weight vectors
    # w_pos[m] = w[n-1+m] for m=0,...,n-1 (non-negative lags)
    w_pos = w[n-1:]  # length n, indices n-1 to 2n-2
    
    # w_neg_full[m] = w[n-1-m] for m=1,...,n-1, w_neg_full[0]=0 (negative lags by |m|)
    w_neg_full = np.zeros(n)
    w_neg_full[1:] = w[0:n-1][::-1]  # w[n-2], w[n-3], ..., w[0]
    # w_neg_full[m] = w[n-1-m] for m=1,...,n-1
    # w_neg_full[1] = w[n-2], w_neg_full[2] = w[n-3], ..., w_neg_full[n-1] = w[0]
    # So w_neg_full[1:n] = w[0:n-1][::-1] ✓
    
    # Compute the four components
    one_minus_h = 1.0 - h
    
    # A[i] = sum_{m=0}^{i} w_pos[m] * (1-h[i-m]) for i=0,...,n-1
    A = np.convolve(w_pos, one_minus_h, mode='full')[:n]
    
    # B[i] = sum_{m=0}^{n-1-i} w_pos[m] * h[i+m] for i=0,...,n-1
    B = np.correlate(w_pos, h, mode='full')[n-1:]
    
    # C[i] = sum_{m=1}^{n-1-i} w_neg_full[m] * (1-h[i+m]) for i=0,...,n-1
    C = np.correlate(w_neg_full, one_minus_h, mode='full')[n-1:]
    
    # D[i] = sum_{m=1}^{i} w_neg_full[m] * h[i-m] for i=0,...,n-1
    D = np.convolve(w_neg_full, h, mode='full')[:n]
    
    # k=0 contribution
    k0 = w[n-1] * (1.0 - 2.0 * h)
    
    grad = dx * (A - B + C - D + k0)
    return grad, corr

def projected_gradient_descent(h, n, initial_mu=1.0, final_mu=0.01, 
                                 n_iter=5000, initial_alpha=10.0,
                                 time_limit=1200, start_time=None):
    """Run projected gradient descent with decreasing mu."""
    if start_time is None:
        start_time = time.time()
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mu = initial_mu
    alpha = initial_alpha
    mu_decay = (final_mu / initial_mu) ** (1.0 / n_iter)
    
    for it in range(n_iter):
        if time.time() - start_time > time_limit:
            break
        if it % 50 == 0:
            mu = max(final_mu, mu * mu_decay ** 50)
        
        grad, corr = compute_smoothed_gradient(h, n, mu)
        
        # Backtracking line search
        best_alpha = alpha
        best_h_new = h
        best_obj = best_c5
        
        for ls in range(15):
            a = alpha * (0.5 ** ls)
            h_trial = h - a * grad
            h_trial = project_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_obj:
                best_obj = c5_trial
                best_h_new = h_trial
                best_alpha = a
            else:
                break
        
        h = best_h_new
        if best_obj < best_c5:
            best_c5 = best_obj
            best_h = h.copy()
        
        # Adjust alpha
        if best_alpha > alpha * 0.7:
            alpha = min(alpha * 1.1, 100.0)
        elif best_alpha < alpha * 0.3:
            alpha = max(alpha * 0.7, 0.1)
    
    return best_h, best_c5
