import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # For numerical stability (shifted <= 0, so T*shifted <= 0)
    exp_vals = np.exp(T * shifted)
    log_sum = np.log(np.sum(exp_vals))
    obj = (mc + log_sum / T) * dx
    return obj

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # Gradient computation using FFT
    size = 1
    while size < 4 * n:
        size *= 2
    
    h_flip = h[::-1]
    W = np.fft.rfft(w, size)
    H = np.fft.rfft(h, size)
    Hf = np.fft.rfft(h_flip, size)
    
    conv_w_h = np.fft.irfft(W * H, size)  # linear conv of w and h
    conv_w_hf = np.fft.irfft(W * Hf, size)  # linear conv of w and h_flip
    
    # A[j] = conv_w_hf[2n-2-j]
    # B[j] = conv_w_h[j+n-1]
    A = conv_w_hf[2*n-2 : 2*n-2-n : -1]  # indices 2n-2, 2n-3, ..., n-1
    B = conv_w_h[n-1 : 2*n-1]  # indices n-1, n, ..., 2n-2
    
    grad = dx * (1.0 - A - B)
    return grad

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
        # Fix remaining
        while abs(np.sum(h) - target) > 1e-13:
            diff2 = target - np.sum(h)
            if diff2 > 0:
                candidates = np.where(h < 1.0)[0]
            else:
                candidates = np.where(h > 0.0)[0]
            if len(candidates) == 0:
                break
            h[candidates[0]] = np.clip(h[candidates[0]] + diff2, 0.0, 1.0)
    return h

Now, the optimization strategy. Let me think about what will be most effective:

1. The soft-max with L-BFGS-B: L-BFGS-B handles bounds natively and is much faster than SLSQP for 960 variables. But it doesn't handle the equality constraint sum(h) = n/2 directly. I can handle this by:
   - Using a penalty for the constraint
   - Or projecting back after each step
   - Or using a Lagrangian approach

Actually, for L-BFGS-B, I can incorporate the equality constraint into the objective as a large penalty:
obj = soft_max + lambda * (sum(h) - n/2)^2

With a large enough lambda, this effectively enforces the constraint.

Or better: I can use the projection onto the affine constraint {h: sum(h) = n/2, 0≤h≤1}. The projection is:
- Start with h' = h - (sum(h) - n/2)/n
- Clip to [0,1]
- Iterate: adjust the free variables to make sum correct

This is what make_feasible does.

For L-BFGS-B, I'll add the constraint as a penalty in the objective.

2. For the coordinate descent, I can be much faster by computing the effect of changing one coordinate analytically.

When I change h[j] from v to v', the correlation changes as:
corr_new[k] = corr_old[k] + (v'-v) * [(1-h[j+r_k]) - h[j-r_k]]

Wait, that's not quite right because changing h[j] affects multiple correlation terms. Let me think...

corr[k] = ∑_m h[m] * (1-h[m+r_k])

If I change h[j] from v to v':
- In the h[m] factor: when m=j, corr changes by (v'-v)*(1-h[j+r_k])
- In the (1-h[m+r]) factor: when m+r_k=j, i.e., m=j-r_k, corr changes by -(v'-v)*h[j-r_k]

So: corr_new[k] = corr[k] + (v'-v) * [(1-h[j+r_k]) - h[j-r_k]]

This means I can compute the effect of changing one coordinate in O(2n-1) time (for all shifts k), and the max in O(2n-1). With precomputation, this is very fast per coordinate.

Actually, for the full coordinate descent with binary search on the optimal value, I need to find v' that minimizes max_k corr_new[k]. Since corr_new[k] is affine in v' (for fixed j), the optimal v' is where the max is minimized, which can be found by binary search.

```python
def fast_coord_descent(h, n, dx, max_iter=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    for iteration in range(max_iter):
        improved = False
        for j in range(n):
            old_v = h[j]
            # Compute delta[k] = (1-h[j+r_k]) - h[j-r_k] for all k
            r = np.arange(2*n-1) - (n-1)  # shifts
            # h[j+r] and h[j-r] (with boundary handling)
            jr = j + r
            jr_valid = (jr >= 0) & (jr < n)
            jrm = j - r
            jrm_valid = (jrm >= 0) & (jrm < n)
            
            h_jr = np.zeros(2*n-1)
            h_jr[jr_valid] = h[jr[jr_valid]]
            h_jrm = np.zeros(2*n-1)
            h_jrm[jrm_valid] = h[jrm[jrm_valid]]
            
            delta = (1.0 - h_jr) - h_jrm
            
            # corr_new(v) = corr + (v - old_v) * delta
            # We want to minimize max_k (corr[k] + (v-old_v)*delta[k])
            # subject to 0 <= v <= 1
            
            # This is a linear program in 1D: minimize t s.t. 
            #   corr[k] + (v-old_v)*delta[k] <= t for all k
            #   0 <= v <= 1
            
            # For a given v, the max is max_k (corr[k] + (v-old_v)*delta[k])
            # This is a piecewise linear function of (v-old_v).

        else:
            adj = max(adj, h[mask].min() + 1e-15)
        h[mask] += adj

    diff = target - np.sum(h)

            else:

    H = np.fft.rfft(h, size)

    B = conv_w_h[n-1 : 2*n-1]

= (b-a)*[h[j+r_k] - h[i+r_k] + h[i-r_k] - h[j-r_k]]

This is O(1) per k, so O(n) total for all k.

def grad_fn(x):
    n = len(x)
    dx = 2.0 / n
    m = 2 * n - 1  # number of shifts
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(_G_T * shifted)
    w = exp_vals / np.sum(exp_vals)
    
    # r_k = k - (n-1) for k=0,...,2n-2
    r = np.arange(m) - (n - 1)
    
    # For each j, g[j] = dx * sum_k w[k] * ((1-h[j+r_k]) - h[j-r_k])
    # Vectorized: create matrix of j+r and j-r
    # j: 0..n-1, r: -(n-1)..n-1
    # j+r: -(n-1)..2n-2, valid where 0<=j+r<=n-1
    # j-r: 0-(n-1)..n-1-(-(n-1)) = -(n-1)..2n-2, same range
    
    js = np.arange(n)[:, None]  # (n, 1)
    jr = js + r[None, :]  # (n, 2n-1)
    jrm = js - r[None, :]  # (n, 2n-1)
    
    h_jr = np.where((jr >= 0) & (jr < n), x[np.clip(jr, 0, n-1)], 0.0)
    h_jrm = np.where((jrm >= 0) & (jrm < n), x[np.clip(jrm, 0, n-1)], 0.0)
    
    delta = (1.0 - h_jr) - h_jrm  # (n, 2n-1)
    g = dx * (w[None, :] * delta).sum(axis=1)
    
    s = np.sum(x) - n / 2.0
    g += 2.0 * _G_LAM * s
    
    return g
```
