```python
def compute_swap_delta(h, oi, oj, n):
    """Compute delta for all correlation values when swapping h[oi]: 1->0, h[oj]: 0->1"""
    m = np.arange(-(n-1), n)  # 2n-1 values, indexed 0..2n-2
    delta = np.zeros(2*n-1)
    
    for idx, mval in enumerate(m):
        # Unique i values: oi, oi-mval, oj, oj-mval (within [0,n))
        is_oi = oi  # i where h(i) changes
        is_oi_m = oi - mval  # i where h(i+m) = h(oi) changes
        is_oj = oj
        is_oj_m = oj - mval
        
        # For each unique i, compute before and after
        i_vals = set()
        for iv in [is_oi, is_oi_m, is_oj, is_oj_m]:
            if 0 <= iv < n:
                i_vals.add(iv)
        
        for iv in i_vals:
            # h(iv) and h(iv+mval)
            h_iv = h[iv]
            im = iv + mval
            h_im = h[im] if 0 <= im < n else 0.0
            
            # After swap
            h_iv_new = 1.0 - h_iv  # only if iv is oi or oj
            if iv == oi:
                h_iv_new = 0.0
            elif iv == oj:
                h_iv_new = 1.0
            else:
                h_iv_new = h_iv
            
            h_im_new = h_im  # only changes if im is oi or oj
            if im == oi:
                h_im_new = 0.0
            elif im == oj:
                h_im_new = 1.0
            
            before = h_iv * (1.0 - h_im)
            after = h_iv_new * (1.0 - h_im_new)
            delta[idx] += (after - before)
    
    return delta
```
