def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        if diff > 0:
            mask = h < 1.0
        else:

        cnt = np.sum(mask)
        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

def obj_smooth(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_smooth(h, n, T, maxiter=200, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_smooth, h.copy(), args=(n, T), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x, res.success

    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr[active_idx] * dx

def make_con_active(idx, n):
    dx = 2.0 / n
    def con(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return corr[idx] * dx
    def con_jac(x):

        # corr[idx] = sum_i x[i] * (1-x[i+l]) where l = idx - (n-1)
        l = idx - (n-1)

                if i + l < n:

                if i + l >= 0:

    # For numerical stability
    shifted = T * (corr - mc)
    # Clip for stability
    shifted = np.clip(shifted, -500, 500)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    return np.ones(len(x))

def subgradient_step(h, n, step_size, top_k=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Find indices where corr is close to max (within tolerance)
    tol = 1e-6
    active = np.where(corr >= max_val - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    
    # Compute gradient of each active corr
    grads = []
    for idx in active:
        l = idx - (n - 1)
        g = np.zeros(n)
        if l >= 0 and l < n:
            g[:n-l] += (1.0 - h[l:])
            g[l:] += (-h[:n-l])
        elif l < 0:
            lp = -l
            if lp < n:
                g[lp:] += (-h[:n-lp])
                g[:n-lp] += (1.0 - h[lp:])

            continue
        sc = min(len(oi), len(zi), 3)
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_v = old
            for cand in [0.0, 1.0, max(0.0, old - 0.01), max(0.0, old - 0.002),
                         min(1.0, old + 0.002), min(1.0, old + 0.01)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = cand
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)

    n = len(h)
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Continuation SLSQP with analytical gradients
    temps = [100, 300, 1000, 3000, 8000, 20000, 50000, 100000]
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        best_h, best_c5 = continuation_slsqp(best_h, n, temps, 250, start_time, budget_s)

    # Phase 2: Additional SLSQP passes with perturbations
    pert_fns = [fourier_perturb, grad_perturb, shift_perturb, segment_flip_perturb]
    for pf in pert_fns:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 100:
            break
        for trial in range(3):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
            try:
                if pf == fourier_perturb:
                    hp = pf(best_h, rng, n_modes=rng.randint(2, 6), amp=rng.uniform(0.02, 0.06))
                elif pf == grad_perturb:
                    hp = pf(best_h, n, rng, amp=rng.uniform(0.01, 0.04))
                elif pf == shift_perturb:
                    hp = pf(best_h, n, rng)
                else:
                    hp = pf(best_h, n, rng)
                hp = make_feasible(hp, n)
                hp2 = slsqp_solve(hp, n, 5000, maxiter=200)
                hp2 = make_feasible(hp2, n)
                c5 = compute_c5(hp2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hp2
                    hp3 = slsqp_solve(hp2, n, 50000, maxiter=200)
                    hp3 = make_feasible(hp3, n)
                    c5_3 = compute_c5(hp3)
                    if c5_3 < best_c5:
                        best_c5 = c5_3
                        best_h = hp3
            except Exception:
                pass

    # Phase 3: Binary SA for exploration
    remaining = budget_s - (time.time() - start_time)
    if remaining > 180:
        tl = min(remaining - 30, 120)
        b, c5_b = binary_sa(best_h, n, rng, 80000, 0.007, 0.00001, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                best_h2 = slsqp_solve(best_h, n, 10000, maxiter=250)
                best_h2 = make_feasible(best_h2, n)
                c5_2 = compute_c5(best_h2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = best_h2

    # Phase 4: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        tl = min(remaining - 20, 80)
        b, c5_b = block_sa(best_h, n, rng, 40000, 0.005, 0.00003, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                best_h2 = slsqp_solve(best_h, n, 10000, maxiter=200)
                best_h2 = make_feasible(best_h2, n)
                c5_2 = compute_c5(best_h2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = best_h2

    # Phase 5: Higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n1 = n + 480
        old_x = np.linspace(0.0, 2.0, n, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n1, endpoint=False)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        c5_up = compute_c5(h_up)
        remaining = budget_s - (time.time() - start_time)
        if remaining > 80:
            h_up2 = slsqp_solve(h_up, n1, 1000, maxiter=150)
            h_up2 = make_feasible(h_up2, n1)
            c5_up2 = compute_c5(h_up2)
            if c5_up2 < c5_up:
                h_up = h_up2
                c5_up = c5_up2
        remaining = budget_s - (time.time() - start_time)
        if remaining > 40:
            h_up3 = slsqp_solve(h_up, n1, 20000, maxiter=200)
            h_up3 = make_feasible(h_up3, n1)
            c5_up3 = compute_c5(h_up3)
            if c5_up3 < c5_up:
                h_up = h_up3
                c5_up = c5_up3
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up

    # Phase 6: Final high-T SLSQP refinement
    remaining = budget_s - (time.time() - start_time)
    n = len(best_h)
    if remaining > 40:
        for T in [50000, 100000, 200000]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 20:
                break
            x = slsqp_solve(best_h, n, T, maxiter=200)
            x = make_feasible(x, n)
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x

    # Phase 7: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    n = len(best_h)
    if remaining > 20:
        best_h, best_c5 = coord_descent(best_h.copy(), n, 10, start_time, budget_s)

    # Phase 8: One final SLSQP
    remaining = budget_s - (time.time() - start_time)
    n = len(best_h)
    if remaining > 15:
        x = slsqp_solve(best_h, n, 100000, maxiter=150)
        x = make_feasible(x, n)
        c5 = compute_c5(x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = x

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
