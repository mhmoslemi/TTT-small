def structured_binary_search(h, n, rng, time_budget):
    """
    Local search on binary sequences.
    h: binary array of length n with exactly n/2 ones
    """
    t_start = time.time()
    ones = set(np.where(h > 0.5)[0])
    zeros = set(np.where(h <= 0.5)[0])
    
    def evaluate(h):
        corr = np.correlate(h, 1.0-h, mode='full')
        return float(np.max(corr))
    
    cur_max = evaluate(h)
    no_improve = 0
    
    while time.time() - t_start < time_budget:
        # Try targeted swaps
        improved = False
        
        # Strategy 1: Random single swap
        for _ in range(2000):
            i = rng.choice(list(ones))
            j = rng.choice(list(zeros))
            h[i] = 0; h[j] = 1
            new_max = evaluate(h)
            if new_max < cur_max:
                cur_max = new_max
                ones.remove(i); ones.add(j)
                zeros.add(i); zeros.remove(j)
                improved = True
                no_improve = 0
            else:
                h[i] = 1; h[j] = 0
                no_improve += 1
        
        if not improved:
            # Try 2-swap: exchange two 1s with two 0s
            ss = list(ones)
            zz = list(zeros)
            for _ in range(1000):
                i1 = ss[rng.randint(len(ss))]
                i2 = ss[rng.randint(len(ss))]
                if i1 == i2: continue
                j1 = zz[rng.randint(len(zz))]
                j2 = zz[rng.randint(len(zz))]
                h[i1]=0; h[i2]=0; h[j1]=1; h[j2]=1
                new_max = evaluate(h)
                if new_max < cur_max:
                    cur_max = new_max
                    ones.discard(i1); ones.discard(i2)
                    ones.add(j1); ones.add(j2)
                    zeros.discard(j1); zeros.discard(j2)
                    zeros.add(i1); zeros.add(i2)
                    improved = True
                    no_improve = 0
                else:
                    h[i1]=1; h[i2]=1; h[j1]=0; h[j2]=0
        
        if not improved:
            # Cooler: accept worse solutions (simulated annealing)

For a swap (i: 1→0, j: 0→1), the effect on the correlation can be computed analytically.

Let's define the "forward" correlation: F[k] = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k]) for k = 0, ..., n-1
And the "backward" correlation: B[k] = sum_{i=k}^{n-1} h[i] * (1-h[i-k]) for k = 0, ..., n-1

The full correlation array (length 2n-1) combines these.

When we swap h[i] (1→0) and h[j] (0→1):
- F[k] changes:
  - At position i: was h[i]*(1-h[i+k]) = 1*(1-h[i+k]), now 0 → change = -(1-h[i+k]) for k where i+k < n
  - At position j: was h[j]*(1-h[j+k]) = 0, now 1*(1-h[j+k]) → change = +(1-h[j+k]) for k where j+k < n
  - If i+k = j for some k (i.e., k = j-i > 0 and j < n): the term at i in F[j-i] was 1*(1-h[j]) = 1*1 = 1, now 0 → additional change of -1
    Wait, this is already accounted for: when i is removed, F[j-i] loses the term h[i]*(1-h[j]) = 1*1 = 1.
    And when j is added, F[j+k] gains the term h[j]*(1-h[j+k]).
    But also, if j+k = i, then F[k] for k = i-j (if i > j) gains h[j]*(1-h[i]) = 1*(1-0) = 1.

One more idea: **Use the structure of the "worst lag" to guide swaps.**

If lag k* is the one achieving the maximum, then:
- F[k*] = sum_{i=0}^{n-1-k*} h[i]*(1-h[i+k*])
- To reduce F[k*], I want to either:
  - Change h[i] from 1 to 0 where h[i+k*] = 0 (removes a contributing term)
  - Change h[i+k*] from 0 to 1 where h[i] = 1 (removes a contributing term)
  
So the "natural" swaps that reduce F[k*] are:
- (i, i+k*): swap 1 at i with 0 at i+k*
- (i+k*, i): same thing

These are "adjacent" swaps at distance k*. If I do this for the top few lags, I should be able to reduce the max quickly.

def continuous_polish(h0, n, T, mu, max_iter):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        return smooth_max_obj(corr, T) * dx + mu * (x.sum() - n / 2.0) ** 2

    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})

        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = thue_morse(640)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    best_n = len(h0)

    # Phase 1: Binary search at parent resolution
    n1 = best_n
    h_bin = to_binary(h0, n1)
    c1 = compute_c5(h_bin)
    if c1 < best_c5:
        best_c5, best_h, best_n = c1, h_bin.copy(), n1

    if rem() > 300:
        h_ref, c_ref, _ = targeted_refine(h_bin.copy(), n1, rng, min(150, rem() - 100))
        if c_ref < best_c5:
            best_c5, best_h, best_n = c_ref, h_ref.copy(), n1

    if rem() > 300:
        h_ls, c_ls, _ = binary_ls(h_ref if 'h_ref' in dir() and c_ref < best_c5 else to_binary(best_h, n1), n1, rng, min(200, rem() - 100))
        if c_ls < best_c5:
            best_c5, best_h, best_n = c_ls, h_ls.copy(), n1

    # Phase 2: Binary search at higher resolutions
    for n2 in [1024, 1536, 2048]:
        if rem() < 250:
            break
        h2 = to_binary(upsample(best_h, n2), n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h, best_n = c2, h2.copy(), n2

        tier = min(180, rem() - 80)
        if tier > 30:
            h_t, c_t, _ = targeted_refine(h2.copy(), n2, rng, tier // 2)
            if c_t < best_c5:
                best_c5, best_h, best_n = c_t, h_t.copy(), n2
                h2 = h_t.copy()

        if rem() > 80:
            h_b, c_b, _ = binary_ls(h2.copy(), n2, rng, min(120, rem() - 60))
            if c_b < best_c5:
                best_c5, best_h, best_n = c_b, h_b.copy(), n2

    # Phase 3: Try different starting patterns at best resolution
    if rem() > 150:
        n_try = best_n
        patterns = [
            to_binary(upsample(best_h, n_try), n_try),
            thue_morse(n_try),
            beatty_seq(n_try, (1.0 + np.sqrt(5.0)) / 2.0),
            beatty_seq(n_try, np.sqrt(2.0)),
        ]
        rng_pat = np.zeros(n_try)
        half = n_try // 2
        rng_pat[rng.choice(n_try, size=half, replace=False)] = 1.0
        patterns.append(rng_pat)

        for pat in patterns:
            if rem() < 60:
                break
            c_p = compute_c5(pat)
            if c_p < best_c5:
                best_c5, best_h, best_n = c_p, pat.copy(), n_try
            if rem() > 50:
                h_r, c_r, _ = binary_ls(pat.copy(), n_try, rng, min(40, rem() - 30))
                if c_r < best_c5:
                    best_c5, best_h, best_n = c_r, h_r.copy(), n_try

    # Phase 4: Continuous relaxation polish on best binary
    if rem() > 60:
        n_cp = min(best_n, 1280)
        h_cp = upsample(best_h, n_cp) if n_cp < best_n else best_h.copy()
        for T in [500, 2000]:
            if rem() < 20:
                break
            h_opt = continuous_polish(h_cp.copy(), n_cp, T, mu=1000.0, max_iter=200)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5, best_h, best_n = c_opt, h_opt.copy(), n_cp
                h_cp = h_opt.copy()

    # Phase 5: Final targeted refinement at best resolution
    if rem() > 40:
        n_final = best_n
        h_f = to_binary(upsample(best_h, n_final), n_final)
        h_f2, c_f, _ = targeted_refine(h_f, n_final, rng, min(30, rem() - 10))
        if c_f < best_c5:
            best_c5, best_h, best_n = c_f, h_f2.copy(), n_final

    # Final check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
