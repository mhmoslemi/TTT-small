def opt_lbgfb(h_start, n, T, mu, max_iter, time_limit):
    """L-BFGS-B optimization with FFT-based gradient"""
    bounds = [(0.0, 1.0)] * n
    
    def func(h):
        corr = compute_corr_full(h)
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -50, 50)
        soft = mc + np.log(np.sum(np.exp(shifted))) / T
        viol = np.sum(h) - n / 2.0
        return (soft + mu * viol * viol) * (2.0 / n)

    t0 = time.time()

                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10,
                            'maxfun': 50000})
    # Check time (L-BFGS-B doesn't have a callback for time easily)
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, max_rounds=50):
    h = h_bin.copy()
    current = compute_c5_fast(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(3000, max(100, len(ones) * len(zeros) // 4))
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5_fast(h)
            if c < current:
                current = c
                ones = np.concatenate([ones[:ii], ones[ii+1:], [oj]])
                zeros = np.concatenate([zeros[:jj], zeros[jj+1:], [oi]])
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, current

def coordinate_descent(h, n, rng, max_rounds=5):
    """Coordinate descent with variable steps"""
    current = compute_c5_fast(h)
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        improved = False
        # Random order
        order = rng.permutation(n)
        for j in order:
            orig = best_h[j]
            # Try several values
            candidates = [0.0, 1.0, 0.1, 0.5, 0.9,
                         max(0, orig-0.05), min(1, orig+0.05),
                         max(0, orig-0.02), min(1, orig+0.02)]
            for val in candidates:
                if abs(val - orig) < 1e-10:
                    continue
                tr = best_h.copy()
                tr[j] = val
                tr = make_feasible(tr, n)
                c = compute_c5_fast(tr)
                if c < current - 1e-12:
                    current = c
                    best_h = tr
                    improved = True
                    break
        if not improved:
            break
    return best_h, current
