def coord_descent_fast(h, n, budget_s, start_time):
    """Faster coordinate descent focusing on high-gradient coordinates."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    steps_cd = [0.02, 0.01, 0.005, 0.002, 0.001]
    
    improved = True
    while improved and time.time() - start_time < budget_s - 3:
        improved = False
        n_pts = len(best_h)
        # Process in random order
        order = np.random.permutation(n_pts)
        for j in order:
            if time.time() - start_time > budget_s - 5:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in steps_cd:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:

                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def slsqp_smooth(h, n, budget_s, start_time, T=1000.0, top_m=8):
    """SLSQP with log-sum-exp smooth max."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None and top_m < len(corr):
            top_idx = np.argsort(corr)[-top_m:]
            mc = np.max(corr[top_idx])
            shifted = T * (corr[top_idx] - mc)
            return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        mc = np.max(corr)
        shifted = T * (corr - mc)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 500, 'ftol': 1e-18})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h):
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h, compute_c5(h)

def binary_project(h, n):
    """Project to nearest binary (with exactly n/2 ones)."""
    target = n // 2
    thresh = np.sort(h)[::-1][target - 1] if target > 0 else 1.0
    binary = (h >= thresh).astype(float)
    if np.sum(binary) != target:
        # Fix count
        if np.sum(binary) > target:
            # Turn off some 1s with lowest h
            ones = np.where(binary == 1)[0]
            # Sort by h value
            ones_sorted = ones[np.argsort(h[ones])]
            for i in ones_sorted:
                if np.sum(binary) == target:
                    break
                binary[i] = 0
        else:
            zeros = np.where(binary == 0)[0]
            zeros_sorted = zeros[np.argsort(h[zeros])[::-1]]
            for i in zeros_sorted:
                if np.sum(binary) == target:
                    break
                binary[i] = 1
    return binary

            mask = h > 1e-12

        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:

        else:
            order = np.argsort(h)
        for i in order:
            if diff > 0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
            else:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-14:
                break

    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        if s < n:
            g[:n-s] += (1.0 - h[s:])
            g[s:] -= h[:n-s]
    else:
        s = -s
        if s < n:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] -= h[s:]
    return g

                    break

        if no_improve > max_no_improve:
            step = max(0.1, step * 2.0)
            beta = 0.8
            momentum = np.zeros(n)
            no_improve = 0
            perturb = 0.005 * np.random.randn(n)
            h = make_feasible(h + perturb, n)

        if it % 100 == 0:
            beta = min(0.95, beta + 0.005)

    final_c5 = compute_c5(h)
    if final_c5 < best_c5:
        best_h = h
        best_c5 = final_c5
    return best_h, best_c5

def slsqp_smooth(h, n, budget_s, start_time, T=800.0, top_m=8):
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None and top_m < len(corr):
            top_idx = np.argsort(corr)[-top_m:]
            mc = np.max(corr[top_idx])
            shifted = T * (corr[top_idx] - mc)
            return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        mc = np.max(corr)
        shifted = T * (corr - mc)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 600, 'ftol': 1e-18})
        c5_res = compute_c5(res.x)
        if c5_res < compute_c5(h):
            return make_feasible(res.x, n), c5_res
    except Exception:
        pass
    return h, compute_c5(h)

def binary_project(h, n):
    target = n // 2
    sorted_vals = np.sort(h)[::-1]
    thresh = sorted_vals[target - 1]
    binary = (h >= thresh).astype(float)
    cnt = int(np.sum(binary))
    if cnt > target:
        ones = np.where(binary == 1.0)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        for i in ones_sorted[:cnt - target]:
            binary[i] = 0.0
    elif cnt < target:
        zeros = np.where(binary == 0.0)[0]
        zeros_sorted = zeros[np.argsort(h[zeros])[::-1]]
        for i in zeros_sorted[:target - cnt]:
            binary[i] = 1.0
    return binary

def binary_local_search(h_bin, n, budget_s, start_time, rng):
    best_h = h_bin.copy()
    best_c5 = compute_c5(h_bin)
    improved = True
    while improved and time.time() - start_time < budget_s - 3:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() - start_time > budget_s - 5:
                break
            if best_h[j] == 1.0:
                zeros = np.where(best_h == 0.0)[0]
                if len(zeros) == 0:
                    continue
                ntry = min(8, len(zeros))
                candidates = zeros[rng.choice(len(zeros), ntry, replace=False)]
                for k in candidates:
                    trial = best_h.copy()
                    trial[j] = 0.0
                    trial[k] = 1.0
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_h = trial
                        best_c5 = c
                        improved = True
                        break
            else:
                ones = np.where(best_h == 1.0)[0]
                if len(ones) == 0:
                    continue
                ntry = min(8, len(ones))
                candidates = ones[rng.choice(len(ones), ntry, replace=False)]
                for k in candidates:
                    trial = best_h.copy()
                    trial[j] = 1.0
                    trial[k] = 0.0
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_h = trial
                        best_c5 = c
                        improved = True
                        break
    return best_h, best_c5

def coord_descent_focused(h, n, budget_s, start_time, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    mid_mask = (best_h > 0.05) & (best_h < 0.95)
    mid_idx = np.where(mid_mask)[0]
    if len(mid_idx) == 0:
        mid_idx = np.arange(n)
    steps_cd = [0.01, 0.005, 0.002, 0.001]
    improved = True
    while improved and time.time() - start_time < budget_s - 3:
        improved = False
        order = rng.permutation(mid_idx)
        for j in order:
            if time.time() - start_time > budget_s - 5:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in steps_cd:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Main ADGD
    if remaining() > 180:
        h, c5 = adgd_optimize(best_h, n, min(150, remaining() - 30), start_time, K=10, T_init=100.0, T_end=2000.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: SLSQP polish
    if remaining() > 120:
        h, c5 = slsqp_smooth(best_h, n, min(90, remaining() - 30), start_time, T=500.0, top_m=10)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: ADGD with tighter focus
    if remaining() > 100:
        h, c5 = adgd_optimize(best_h, n, min(80, remaining() - 20), start_time, K=5, T_init=500.0, T_end=4000.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: SLSQP with higher T
    if remaining() > 80:
        h, c5 = slsqp_smooth(best_h, n, min(60, remaining() - 20), start_time, T=1500.0, top_m=6)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Coordinate descent on mid-range values
    if remaining() > 70:
        h, c5 = coord_descent_focused(best_h, n, min(50, remaining() - 20), start_time, rng)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: Binary exploration
    if remaining() > 120:
        h_bin = binary_project(best_h, n)
        c5_bin = compute_c5(h_bin)
        h_bin, c5_bin = binary_local_search(h_bin, n, min(40, remaining() - 60), start_time, rng)
        # Smooth from best binary
        h_s = h_bin + 0.03 * rng.randn(n)
        h_s = make_feasible(h_s, n)
        h_s, c5_s = adgd_optimize(h_s, n, min(50, remaining() - 30), start_time, K=8, T_init=80.0, T_end=1200.0)
        if c5_s < best_c5:
            best_h, best_c5 = h_s, c5_s
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin

    # Phase 7: Random restarts
    for restart in range(5):
        if remaining() < 50:
            break
        scale = 0.02 if restart < 2 else 0.03
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        h2, c5_2 = adgd_optimize(h_try, n, min(35, remaining() - 30), start_time, K=7, T_init=100.0, T_end=1000.0)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
            if remaining() > 40:
                h2, c5_2 = slsqp_smooth(best_h, n, min(20, remaining() - 15), start_time, T=800.0, top_m=6)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2

    # Phase 8: Final tight ADGD
    if remaining() > 50:
        h, c5 = adgd_optimize(best_h, n, min(35, remaining() - 15), start_time, K=3, T_init=3000.0, T_end=6000.0)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 9: Final SLSQP
    if remaining() > 25:
        h, c5 = slsqp_smooth(best_h, n, min(20, remaining() - 8), start_time, T=3000.0, top_m=3)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 10: Ultra-fine local search on critical coordinates
    if remaining() > 12:
        n_pts = len(best_h)
        mid_mask = (best_h > 0.15) & (best_h < 0.85)
        mid_idx = np.where(mid_mask)[0]
        for j in mid_idx[:300]:
            if time.time() - start_time > budget_s - 2:
                break
            old_val = best_h[j]
            for s in [0.001, 0.0005, 0.0002]:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-14:
                            best_h = trial
                            best_c5 = c
                            old_val = v
                            break

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
