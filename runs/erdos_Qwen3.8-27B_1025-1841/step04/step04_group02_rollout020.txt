def slsqp_run(h, n, T, maxiter, ftol):

    _SL_N = n
    _SL_T = T
    
    def obj(x):
        dx = 2.0 / n
        L = 2 * n - 1
        X = np.fft.rfft(x, L)
        B = np.fft.rfft(1.0 - x, L)
        corr = np.fft.irfft(np.conj(X) * B, L)
        mc = np.max(corr)
        # Smooth max approximation
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

        return np.sum(x) - n / 2.0

        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n

    res = minimize(obj, h.copy(), method='SLSQP',

import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5_fft(h):
    n = len(h)
    L = 2 * n - 1
    H = np.fft.rfft(h, L)
    B = np.fft.rfft(1.0 - h, L)
    corr = np.fft.irfft(np.conj(H) * B, L)
    return float(np.max(corr) * (2.0 / n))

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

            b[iz] = 1

            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    L = 2 * n - 1
    X = np.fft.rfft(x, L)
    B = np.fft.rfft(1.0 - x, L)
    corr = np.fft.irfft(np.conj(X) * B, L)
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

            trial = b.copy()

            c = compute_c5_fft(trial)
            if c < best_c5 - 1e-15:
                b = trial
                best_c5 = c
                improved = True
        if not improved:
            break
    return b, best_c5
