from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def softmax_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def topk_mean_obj(corr, k):
    """Mean of top-k correlation values"""
    n_c = len(corr)
    if k >= n_c:
        return np.max(corr)
    topk = np.sort(corr)[-k:]
    return np.sum(topk) / k

def lbgfb_optimize(h0, n, T, k_top, mu, max_iter):
    """L-BFGS-B optimization with combined softmax + top-k objective"""
    dx = 2.0 / n

        tk = topk_mean_obj(corr, k_top)

    n_c = len(corr)
    mc = np.max(corr)

    # df/dh[j] = dx * sum_k p[k]*((1-h[j+l_k]) - h[j-l_k]) + 2*mu*(sum(h)-n/2)

    # = sum_m p[m+n-1] * h[j+m] for m = -(n-1),...,(n-1)

    # sum_k p[k] * h[j + k - (n-1)] = sum_m p[m] * h[j + m - (n-1)]
    # Let i = j + m - (n-1), so m = i - j + (n-1)
    # = sum_i p[i-j+n-1] * h[i] (for valid i and m)
    # This is (p reversed) * h (convolution) ...

    # Contribution to grad[j] from lag l: weight * ((1-h[j+l]) - h[j-l]) for valid indices

    #   j+l in [0,n-1] AND j in [0,n-1] for the first term
    #   j-l in [0,n-1] AND j in [0,n-1] for the second term
    
    grad = np.zeros(n)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)

def obj_grad_softmax(h, n, T, mu):
    """Objective and gradient for softmax-smoothed max correlation."""
    dx = 2.0 / n
    h = np.asarray(h, dtype=float)
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    n_c = len(corr)
    mc = np.max(corr)
    lw = np.clip(T * (corr - mc), -80, 80)
    exp_lw = np.exp(lw)
    sum_exp = exp_lw.sum()
    p = exp_lw / sum_exp  # softmax weights
    
    obj = (mc + np.log(sum_exp) / T) * dx + mu * (h.sum() - n/2.0)**2
    
    # Gradient
    # d(corr[k])/dh[j] = (1-h[j+l]) - h[j-l] where l = k-(n-1)
    # grad[j] = dx * sum_k p[k]*((1-h[j+l_k]) - h[j-l_k]) + 2*mu*(sum(h)-n/2)
    
    grad = np.zeros(n)
    for k in range(n_c):
        w = p[k]
        if w < 1e-14:
            continue
        l = k - (n - 1)
        # (1-h[j+l]) term: valid when 0<=j<n and 0<=j+l<n
        jlo = max(0, -l)
        jhi = min(n-1, n-1-l)
        if jhi >= jlo:
            grad[jlo:jhi+1] += w * one_minus_h[jlo+l:jhi+l+1]
        # -h[j-l] term: valid when 0<=j<n and 0<=j-l<n
        jlo2 = max(0, l)
        jhi2 = min(n-1, n-1+l)
        if jhi2 >= jlo2:
            grad[jlo2:jhi2+1] -= w * h[jlo2-l:jhi2-l+1]
    
    grad = dx * grad + 2.0 * mu * (h.sum() - n/2.0)
    return obj, grad

def lbgfb_with_grad(h0, n, T, mu, max_iter=300):
    """L-BFGS-B with analytical gradient."""
    def fun(x):
        return obj_grad_softmax(x, n, T, mu)[0]
    
    def jac(x):
        return obj_grad_softmax(x, n, T, mu)[1]
    
    res = minimize(fun, h0, method='L-BFGS-B', jac=jac,
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-12,
                            'maxfun': 5000})
    return make_feasible(res.x, n)

def fine_coord_descent(h0, n, max_rounds=10, rng=None):
    """Very fine coordinate descent."""
    if rng is None:
        rng = np.random.RandomState(0)
    h = h0.copy()
