import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def fft_c5(h):
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    H = fft(h, N)
    V = fft((1.0 - h)[::-1], N)
    corr = ifft(H * V).real[:2*n-1]
    return float(np.max(corr) * dx), corr

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def lbgfb_optimize(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    
    def f(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(hv) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-15, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def simulated_annealing_binary(h_bin, n, rng, T_init=0.008, T_final=0.0002, 
                                total_steps=500000, time_limit=600):
    """Simulated annealing on binary swaps using FFT."""
    start_t = time.time()
    h = h_bin.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    _, corr = fft_c5(h)
    current_c5 = np.max(corr) * (2.0/n)
    best_c5 = current_c5
    best_h = h.copy()
    
    n_steps = total_steps
    for step in range(n_steps):
        if time.time() - start_t > time_limit - 10:
            break
        
        # Temperature schedule
        frac = step / n_steps
        T = T_init * (T_final / T_init) ** frac
        
        # Pick random swap
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        p = ones[ii]  # was 1
        q = zeros[jj]  # was 0
        
        # Swap
        h[p] = 0.0
        h[q] = 1.0
        ones[ii] = q
        zeros[jj] = p
        
        # Compute new C5
        _, new_corr = fft_c5(h)
        new_c5 = np.max(new_corr) * (2.0/n)
        delta = new_c5 - current_c5
        
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            current_c5 = new_c5
            corr = new_corr
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Revert
            h[p] = 1.0
            h[q] = 0.0
            ones[ii] = p
            zeros[jj] = q
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)

def fft_c5(h):
    n = len(h)
    dx = 2.0 / n
    N = 2 * n
    H = fft(h, N)
    V = fft((1.0 - h)[::-1], N)
    corr = ifft(H * V).real[:2 * n - 1]
    return float(np.max(corr) * dx), corr

    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0

        if abs(diff) < 1e-15:
            break

        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
