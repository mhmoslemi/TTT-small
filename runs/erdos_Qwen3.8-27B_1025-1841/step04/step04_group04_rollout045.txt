    n = len(h)
    n_half = n - 1
    # lag is the actual lag k (can be negative)
    # c_k = sum_i h[i+k] * (1-h[i]) for valid i
    # d(c_k)/d(h[j]) = (1-h[j-k])*I(0<=j-k<n) - h[j+k]*I(0<=j+k<n)
    g = np.zeros(n)
    j = np.arange(n)
    # First term: (1-h[j-k]) where 0 <= j-k < n
    idx1 = j - lag
    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] += (1 - h[idx1[mask1]])
    # Second term: -h[j+k] where 0 <= j+k < n
    idx2 = j + lag
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
