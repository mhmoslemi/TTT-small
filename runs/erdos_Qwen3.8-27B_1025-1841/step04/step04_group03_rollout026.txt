def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _obj_lse(h, n, T, L):
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * (2.0 / n)
    k = min(L, len(c))
    idx = np.argpartition(c, -k)[-k:]
    top = c[idx]
    m = np.max(top)
    return float(m + np.log(np.sum(np.exp(T * (top - m)))) / T)

def _con_sum(x):
    return np.sum(x) - len(x) / 2.0

def _con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, L, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_sum_jac}]

    def obj(x):
        return _obj_lse(x, n, T, L)

    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    if res.success or res.x is not None:
        return res.x
    return h

            grad[i_start:i_end] -= h[i_start - m:i_end - m]

        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            candidates = [0.0, 1.0,
                          max(0.0, old - 0.05), max(0.0, old - 0.01),
                          max(0.0, old - 0.001), max(0.0, old - 0.0001),
                          min(1.0, old + 0.0001), min(1.0, old + 0.001),
                          min(1.0, old + 0.01), min(1.0, old + 0.05)]
            for cand in candidates:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def binary_sa_multi(h, n, start_time, budget_s, n_restarts=3):
    best_b = (h >= 0.5).astype(float)
    best_c5 = compute_c5(best_b)
    target = n // 2

    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 100:
            break
        if restart == 0:
            b = best_b.copy()
        else:
            b = np.zeros(n)
            idx = np.random.choice(n, size=target, replace=False)
            b[idx] = 1.0

        curr_c5 = compute_c5(b)
        T0 = 0.008
        T1 = 0.00005
        max_iters = 50000
        for it in range(max_iters):
            if time.time() - start_time > budget_s - 100:
                break
            frac = float(it) / max_iters
            T = T0 * (T1 / T0) ** frac
            # Try 1-swap or 2-swap
            if np.random.random() < 0.7:
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                b[i] = 0
                b[j] = 1
                new_c5 = compute_c5(b)
                if new_c5 <= curr_c5 or (T > 1e-13 and np.random.random() < np.exp(-(new_c5 - curr_c5) / T)):
                    curr_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_b = b.copy()
                else:
                    b[i] = 1
                    b[j] = 0
            else:
                # 2-swap
                ones = np.where(b == 1)[0]
                zeros = np.where(b == 0)[0]
                i1 = ones[np.random.randint(len(ones))]
                i2 = ones[np.random.randint(len(ones))]
                j1 = zeros[np.random.randint(len(zeros))]
                j2 = zeros[np.random.randint(len(zeros))]
                if i1 == i2 or j1 == j2:
                    continue
                b[i1] = 0
                b[i2] = 0
                b[j1] = 1
                b[j2] = 1
                new_c5 = compute_c5(b)
                if new_c5 <= curr_c5 or (T > 1e-13 and np.random.random() < np.exp(-(new_c5 - curr_c5) / T)):
                    curr_c5 = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_b = b.copy()
                else:
                    b[i1] = 1
                    b[i2] = 1
                    b[j1] = 0
                    b[j2] = 0
    return best_b, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP warmup with multiple temperatures
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        temps = [200.0, 1000.0, 5000.0, 20000.0]
        for T in temps:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
            try:
                x = slsqp_optimize(best_h, n0, T, 50, 100, 1e-14)
                c5 = compute_c5(x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = make_feasible(x, n0)
            except Exception:
                pass

    # Phase 2: Cutting-plane LP refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        tl = min(remaining - 80, 150)
        try:
            h_cp, c5_cp = cutting_plane(best_h, n0, start_time, start_time + tl + (time.time() - start_time))
            # Re-do with correct time tracking
            h_cp, c5_cp = cutting_plane(best_h, n0, start_time, start_time + (budget_s - 80) - (time.time() - start_time), max_iters=35)
            if c5_cp < best_c5:
                best_c5 = c5_cp
                best_h = h_cp.copy()
        except Exception:
            pass

    # Phase 3: Low-temperature SLSQP after cutting plane
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        for T in [50.0, 20.0, 5.0, 1.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 50:
                break
            try:
                x = slsqp_optimize(best_h, n0, T, 30, 80, 1e-15)
                c5 = compute_c5(x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = make_feasible(x, n0)
            except Exception:
                pass

    # Phase 4: Micro coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        best_h, best_c5 = micro_coord_descent(best_h.copy(), n0, start_time, start_time + min(remaining - 30, 90), max_rounds=20)

    # Phase 5: Binary SA with multi-swap for diversification
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        tl = min(remaining - 100, 150)
        b, c5_b = binary_sa_multi(best_h, n0, start_time, start_time + tl, n_restarts=3)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            # Refine the binary solution with SLSQP
            remaining = budget_s - (time.time() - start_time)
            if remaining > 80:
                for T in [100.0, 5.0]:
                    try:
                        x = slsqp_optimize(best_h, n0, T, 30, 60, 1e-15)
                        c5 = compute_c5(x)
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = make_feasible(x, n0)
                    except Exception:
                        pass

    # Phase 6: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 400:
        n1 = 1440
        old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n1, endpoint=False)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        # SLSQP at higher res
        for T in [500.0, 2000.0, 200.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 100:
                break
            try:
                x = slsqp_optimize(h_up, n1, T, 40, 80, 1e-14)
                c5 = compute_c5(x)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = make_feasible(x, n1)
            except Exception:
                pass
        # Cutting plane at higher res
        remaining = budget_s - (time.time() - start_time)
        if remaining > 150 and len(best_h) == n1:
            try:
                h_cp, c5_cp = cutting_plane(best_h, n1, start_time, start_time + (budget_s - 60) - (time.time() - start_time), max_iters=20)
                if c5_cp < best_c5:
                    best_c5 = c5_cp
                    best_h = h_cp.copy()
            except Exception:
                pass

    # Phase 7: Final cutting plane at best resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        n_pts = len(best_h)
        try:
            h_cp, c5_cp = cutting_plane(best_h, n_pts, start_time, start_time + min(remaining - 20, 80), max_iters=15)
            if c5_cp < best_c5:
                best_c5 = c5_cp
                best_h = h_cp.copy()
        except Exception:
            pass

    # Phase 8: Final micro coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        best_h, best_c5 = micro_coord_descent(best_h.copy(), n_pts, start_time, start_time + min(remaining - 5, 40), max_rounds=10)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
