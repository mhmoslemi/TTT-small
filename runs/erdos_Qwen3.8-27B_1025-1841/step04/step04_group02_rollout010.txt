import numpy as np
from scipy.optimize import minimize
import time

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def fast_corr_max(h):

    f2 = np.fft.rfft(1.0 - h, size)
    corr = np.fft.irfft(f1 * np.conj(f2), size)[:2 * n - 1]

def binary_sa_fft(h, n, n_iters, T0, T1, rng, time_limit_fn=None, verbose=False):
    """Binary simulated annealing with FFT-based evaluation."""

    zeros = np.where(b == 0)[0]
    stall = 0
    max_stall = 200000

    for it in range(n_iters):
        if time_limit_fn is not None and time_limit_fn() < 5:
            break
        frac = float(it) / n_iters
        T = T0 * (T1 / T0) ** (frac if T1 > T0 else (1.0 - frac) * 0.3 + frac * 0.7)
        T = max(T, 1e-10)

        j_idx = zeros[rng.randint(len(zeros))]
        b[i_idx] = 0.0
        b[j_idx] = 1.0

        # Update ones/zeros tracking
        ones = ones[np.where(ones != i_idx)]
        zeros = np.delete(zeros, np.where(zeros == j_idx)[0][0])
        ones = np.append(ones, j_idx)
        zeros = np.append(zeros, i_idx)

        new_c5 = fast_corr_max(b)
        delta = new_c5 - curr_c5

        if delta <= 0 or rng.random() < np.exp(-delta / (T * dx)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_b = b.copy()
                stall = 0
            else:
                stall += 1
        else:
            b[i_idx] = 1.0
            b[j_idx] = 0.0
            ones = ones[np.where(ones != j_idx)]
            zeros = np.delete(zeros, np.where(zeros == i_idx)[0][0])
            ones = np.append(ones, i_idx)
            zeros = np.append(zeros, j_idx)
            stall += 1

                zeros = np.append(zeros, oi)

def block_sa_fft(h, n, n_iters, T0, T1, rng, time_limit_fn=None):
    """Block-swap binary SA."""
    b = h.astype(float).copy()
    curr_c5 = fast_corr_max(b)
    best_c5 = curr_c5
    best_b = b.copy()
    dx = 2.0 / n

    for it in range(n_iters):
        if time_limit_fn is not None and time_limit_fn() < 5:
            break
        frac = float(it) / n_iters
        T = T0 * (T1 / T0) ** (1.0 - frac)
        T = max(T, 1e-10)

        blen = rng.randint(1, min(6, n // 10))
        s1 = rng.randint(0, n - blen)
        s2 = rng.randint(0, n - blen)

        seg1 = b[s1:s1+blen].copy()
        seg2 = b[s2:s2+blen].copy()

        if seg1.sum() == seg2.sum():
            # Swap blocks directly
            b[s1:s1+blen] = seg2
            b[s2:s2+blen] = seg1
        else:
            # Need to equalize counts: swap within segments
            ones1 = np.where(seg1 == 1)[0]
            zeros1 = np.where(seg1 == 0)[0]
            ones2 = np.where(seg2 == 1)[0]
            zeros2 = np.where(seg2 == 0)[0]
            cnt = min(len(ones1), len(zeros2))
            if cnt == 0:
                continue
            for k in range(cnt):
                b[s1 + ones1[k]] = 0.0
                b[s2 + zeros2[k]] = 1.0
            cnt2 = min(len(zeros1), len(ones2))
            for k in range(cnt2):
                b[s1 + zeros1[k]] = 1.0
                b[s2 + ones2[k]] = 0.0

        new_c5 = fast_corr_max(b)
        delta = new_c5 - curr_c5

        if delta <= 0 or rng.random() < np.exp(-delta / (T * dx)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[s1:s1+blen] = seg1
            b[s2:s2+blen] = seg2

    return best_b, best_c5 * dx

def coord_descent_fft(h, n, max_rounds, start_time, budget_s):
    """Coordinate descent with FFT evaluation."""
    best_c5 = fast_corr_max(h) * (2.0 / n)
    dx = 2.0 / n
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 2:
                return h, best_c5
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02),
                         max(0.0, old - 0.005), min(1.0, old + 0.005)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = fast_corr_max(trial) * dx
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def slsqp_objective(x, n, T):
    dx = 2.0 / n
    size = 1
    while size < 2 * n + 1:
        size *= 2
    f1 = np.fft.rfft(x, size)
    f2 = np.fft.rfft(1.0 - x, size)
    corr = np.fft.irfft(f1 * np.conj(f2), size)[:2 * n - 1]
    mc = np.max(corr)
    shifted = corr - mc
    soft_max = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return soft_max * dx

def slsqp_sum(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_sum, 'jac': slsqp_sum_jac}]
    try:
        res = minimize(slsqp_objective, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons, args=(n, T),
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def random_binary(n, rng):
    b = np.zeros(n)
    positions = rng.choice(n, size=n // 2, replace=False)

    # Verify with direct correlation (slower but exact)
    if len(best_h) <= 2000:
        c5_verify = compute_c5_verify(best_h)
        if c5_verify > best_c5 + 1e-10:
            best_c5 = c5_verify

    return best_h, best_c5, len(best_h)
