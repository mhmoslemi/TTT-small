Wait, one more consideration. The problem says the evaluation uses `np.correlate(h, 1-h, mode="full")`. This is a LINEAR correlation, not circular. So the effective "domain" for the overlap is [0, 2] with h extended by 0 outside. This is important!

If h is defined on [0,2] and we consider h(x+k) for x+k outside [0,2], then h(x+k) = 0 (zero extension). This means the correlation at large lags will be small (partial overlap).

This changes things: the maximum of the correlation might be near k=0 (where the overlap is largest) or at some intermediate k.

For the zero-lag: corr[n-1] = Σ h_i(1-h_i) ≤ n/2 (with equality when h_i ∈ {0,1} for all i, giving corr[n-1] = 0 for binary).

Wait for binary: h_i(1-h_i) = 0 for all i. So corr[n-1] = 0. Good.

For lag k (where the (1-h) is shifted by k positions relative to h):
corr[n-1+k] = Σ_i h[i] * (1-h[i+k]) for i where both indices are valid.

For positive k: i ranges from 0 to n-1-k, so there are n-k terms.
For negative k: i ranges from -k to n-1, so there are n+k terms.

The maximum will typically be at small |k| where there are many terms.

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Convert to binary with exactly n//2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Remove excess ones (those closest to 0.5)
        one_idx = np.where(h_bin == 1)[0]
        # Sort by original h value (lowest first)
        order = np.argsort(h[one_idx])
        for i in order[:n_ones - target]:
            h_bin[one_idx[i]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in order[:target - n_ones]:
            h_bin[zero_idx[i]] = 1.0
    return h_bin

    zeros = np.where(h == 0)[0].tolist()
    
    current = compute_c5(h)
    best = current
    best_h = h.copy()
    
    n_iters = max_iters
    for it in range(n_iters):
        T = T_start * (T_end / T_start) ** (it / n_iters)
        oi = ones[rng.randint(len(ones))]
        oz = zeros[rng.randint(len(zeros))]
        
        h[oi] = 0.0
        h[oz] = 1.0
        new_c = compute_c5(h)
        
        delta = new_c - current
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            current = new_c
            ones.remove(oi)
            zeros.remove(oz)
            ones.append(oz)
            zeros.append(oi)
            if new_c < best:
                best = new_c
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oz] = 0.0
    
    return best_h, best

def lbgfb_smooth(h, n, T, mu, max_iter=300):
    dx = 2.0 / n

        mc = np.max(corr)

                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

            if c5_b < best_c5:
                best_c5 = c5_b
                best_h = h_bin.copy()
            sa_t = min(40, max(10, tl() / 3.0))
            h_sa, c5_sa = sa_binary(best_h, n_target, rng, sa_t, 0.003, 0.00003)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()
        
        # Smooth optimization at new resolution
        if tl() > 40:
            for T in [500, 5000]:
                if tl() < 15:
                    break
                try:
                    h_opt = lbgfb_smooth(best_h, n_target, T, mu=300.0, max_iter=200)
                    c5_opt = compute_c5(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                except Exception:
                    pass
    
    # Phase 4: Final refinement - back to a good resolution for coordinate descent
    if tl() > 30:
        n_cd = min(len(best_h), 1280)
        h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        h_cd, c5_cd = coord_descent(h_cd, n_cd, time_limit=max(10, tl() - 15))
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Final SA squeeze at best resolution
    if tl() > 20:
        h_final_bin = to_binary(best_h, len(best_h))
        c5_fb = compute_c5(h_final_bin)
        if c5_fb < best_c5:
            best_c5 = c5_fb
            best_h = h_final_bin.copy()
        h_sa, c5_sa = sa_binary(best_h, len(best_h), rng, max(5, tl() - 10), 0.001, 0.000001)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
