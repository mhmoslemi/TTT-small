    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish at initial resolution
    if remaining() > 80:
        try:
            _G_N = n0
            _G_T = 300.0
            _G_TOP_IDX = None
            bounds = [(0.0, 1.0)] * n0
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            res = minimize(_obj_full, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n0)
        except Exception:
            pass

    if remaining() < 40:
        return best_h, best_c5, len(best_h)

    # Phase 2: Choose target resolution
    if remaining() > 700:
        n_target = 2880
    elif remaining() > 400:
        n_target = 1920
    else:
        n_target = n0 * 2

    h_up = upsample_h(best_h, n_target)
    c5_up = compute_c5(h_up)
    if c5_up <= best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()
    n = n_target

    # Phase 3: Binary local search (main workhorse)
    h_bin = snap_binary(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()

    search_result = binary_local_search(h_bin, n, remaining)
    if search_result[1] < best_c5:
        best_c5 = search_result[1]
        best_h = search_result[0]

    # Phase 4: Continuous SLSQP on the binary result
    if remaining() > 80:
        h_cont = make_feasible(best_h.copy(), n)
        _G_N = n
        _G_T = 500.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_full, h_cont, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-15})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 5: Shift-targeted SLSQP
    if remaining() > 50:
        corr_best = np.correlate(best_h, 1.0 - best_h, mode='full')
        mc_best = np.max(corr_best)
        threshold = mc_best - 0.008 * mc_best
        top_idx = np.where(corr_best >= threshold)[0]
        if len(top_idx) > 20:
            tv = corr_best[top_idx]
            so = np.argsort(-tv)
            top_idx = top_idx[so[:20]]
        elif len(top_idx) == 0:
            top_idx = np.array([int(np.argmax(corr_best))])
        _G_N = len(best_h)
        _G_T = 400.0
        _G_TOP_IDX = top_idx
        bounds = [(0.0, 1.0)] * len(best_h)
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_top, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 250, 'ftol': 1e-15})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, len(best_h))
        except Exception:
            pass

    # Phase 6: Binary snap check and micro-search
    if remaining() > 40:
        h_snap = snap_binary(best_h, len(best_h))
        c5_snap = compute_c5(h_snap)
        if c5_snap < best_c5:
            best_c5 = c5_snap
            best_h = h_snap.copy()
        # Quick binary micro-search
        h_micro = best_h.copy()
        corr_micro = np.correlate(h_micro, 1.0 - h_micro, mode='full')
        mc_micro = np.max(corr_micro)
        ones_m = np.where(h_micro == 1)[0]
        zeros_m = np.where(h_micro == 0)[0]
        if len(ones_m) > 0 and len(zeros_m) > 0:
            for _ in range(2000):
                if remaining() < 10:
                    break
                ii = ones_m[rng.randint(len(ones_m))]
                jj = zeros_m[rng.randint(len(zeros_m))]
                if ii == jj:
                    continue
                h_micro[ii], h_micro[jj] = 0.0, 1.0
                corr_m = np.correlate(h_micro, 1.0 - h_micro, mode='full')
                max_m = np.max(corr_m)
                if max_m < mc_micro - 1e-12:
                    mc_micro = max_m
                    corr_micro = corr_m
                    ones_m = np.where(h_micro == 1)[0]
                    zeros_m = np.where(h_micro == 0)[0]
                    c5_m = float(mc_micro * (2.0 / len(h_micro)))
                    if c5_m < best_c5:
                        best_c5 = c5_m
                        best_h = h_micro.copy()
                else:
                    h_micro[ii], h_micro[jj] = 1.0, 0.0

    # Phase 7: Fine coordinate descent
    if remaining() > 15:
        n_pts = len(best_h)
        for cd_pass in range(2):
            for j in range(n_pts):
                if remaining() < 5:
                    break
                old_val = best_h[j]
                best_val = old_val
                best_c = best_c5
                for step in [0.001, 0.003, 0.005, 0.01]:
                    for sign in [-1.0, 1.0]:
                        v = old_val + sign * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n_pts)
                            c = compute_c5(trial)
                            if c < best_c - 1e-15:
                                best_c = c
                                best_val = v
                if best_val != old_val:
                    best_h[j] = best_val
                    best_c5 = best_c

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
