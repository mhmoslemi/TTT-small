Hmm wait, I realize there might be an issue with the FFT approach. Let me verify:

For `np.correlate(a, v, mode='full')`:
The result c has length len(a) + len(v) - 1 = 2n-1.
c[k] = sum over m of a[m+k] * conj(v[m]), where the sum is over all valid m.

The FFT cross-correlation: ifft(fft(a, N) * conj(fft(v, N)))
This gives the circular cross-correlation, which for zero-padded arrays equals:
at index j: sum over m of a[(m+j) mod N] * conj(v[m])
For zero-padded (N ≥ 2n-1), this equals the linear cross-correlation.

The linear cross-correlation at lag k is: sum_m a[m+k] * conj(v[m]) for valid m.

So the FFT result at index j corresponds to lag j (or lag -j, depending on convention). The set of all values {c[0], c[1], ..., c[2n-2]} should be the same as the set of FFT outputs (just permuted).

Therefore, max(FFT_result) = max(direct_correlation). ✓

        return _N_cache[n]
    N = 1
    while N < 2 * (n - 1) + 1:

    n = len(h)
    N = get_N(n)
    H = fft(h, N)
    G = fft(1.0 - h, N)
    corr = ifft(H * np.conj(G)).real

    corr = np.correlate(h, 1.0 - h, mode='full')

    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):

        else:
            mask = h > 0.0

        if cnt == 0:

        h = np.clip(h, 0.0, 1.0)

                zeros_idx[zeros_idx == j] = i

import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

_N_cache = {}

def get_N(n):
    if n in _N_cache:
        return _N_cache[n]
    N = 1
    while N < 2 * (n - 1) + 1:
        N *= 2
    _N_cache[n] = N
    return N

def c5_fast(h):
    n = len(h)
    N = get_N(n)
    H = fft(h, N)
    G = fft(1.0 - h, N)
    corr = ifft(H * np.conj(G)).real
    return float(np.max(corr) * (2.0 / n))

def c5_exact(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while int(np.sum(b)) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:

        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

            idx_o = np.where(one_arr == i)[0]
            if len(idx_o) > 0:
                one_arr[idx_o[0]] = j
            idx_z = np.where(zero_arr == j)[0]
            if len(idx_z) > 0:
                zero_arr[idx_z[0]] = i
            if new_c < best:
                best = new_c
                best_b = b.copy()
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best

def sa_block_swap(h, n, n_iters, T0, T1, rng, deadline, max_block=5):
    b = to_binary(h, n)
    curr = c5_fast(b)
    best = curr
    best_b = b.copy()

def slsqp_smooth(h, n, T, maxiter):
    dx = 2.0 / n
    N = get_N(n)

    def obj(x):
        H = fft(x, N)
        G = fft(1.0 - x, N)
        corr = ifft(H * np.conj(G)).real
        mc = np.max(corr)
        sh = np.clip(T * (corr - mc), -50, 50)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * dx

    def con_fun(x):
        return np.sum(x) - n / 2.0

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun}]
    res = minimize(obj, h.copy(), method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
    return res.x

def coord_descent(h, n, start_time, deadline, max_rounds=15):
    best_c = c5_fast(h)
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline - 3:
                break
            old = h[j]
            best_val = old
            best_c_try = best_c
            for cand in [0.0, 1.0, max(0.0, old - 0.5), max(0.0, old - 0.1),
                         min(1.0, old + 0.1), min(1.0, old + 0.5)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = c5_fast(trial)
                if c < best_c_try - 1e-10:
                    best_c_try = c
                    best_val = trial
            if best_val is not old:
                h = best_val
                best_c = best_c_try
                improved = True
        if not improved:
            break
    return h, best_c

def random_binary(n, rng):
    b = np.zeros(n)
    half = n // 2
    idx = rng.permutation(n)[:half]
    b[idx] = 1.0
    return b

def periodic_init(n, period, duty, rng):
    b = np.zeros(n)
    for i in range(n):
        if (i % period) < int(period * duty):
            b[i] = 1.0
    diff = n // 2 - int(np.sum(b))
    if diff > 0:
        zeros = np.where(b == 0)[0]
        b[rng.choice(zeros, size=min(diff, len(zeros)), replace=False)] = 1
    elif diff < 0:
        ones = np.where(b == 1)[0]
        b[rng.choice(ones, size=min(-diff, len(ones)), replace=False)] = 0
    return b

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
