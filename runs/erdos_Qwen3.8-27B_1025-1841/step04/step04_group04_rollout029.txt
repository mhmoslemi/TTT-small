def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.arange(old_n) * (2.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _slsqp_solve(h_start, n, T, maxiter):
    global _g_n, _g_T
    _g_n = n
    _g_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    res = minimize(_obj_fn, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def run(seed=42, budget_s=1300, **kwargs):
    global _g_n, _g_T
    start = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start

    def remaining():
        return budget_s - elapsed()

    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Quick SLSQP at n=960 with T-annealing
    n1 = 960
    h_w = upsample_h(best_h, n1)
    for T in [25.0, 80.0, 200.0, 500.0]:
        if remaining() < 150:
            break
        try:
            res_x = _slsqp_solve(h_w, n1, T, 300)
            c = compute_c5(res_x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res_x, n1)
                h_w = best_h.copy()
        except Exception:
            pass

    # Phase 2: Upsample to n=1920, SLSQP with T-annealing
    n2 = 1920
    h_w2 = upsample_h(best_h, n2)
    for T in [40.0, 120.0, 400.0]:
        if remaining() < 150:
            break
        try:
            res_x = _slsqp_solve(h_w2, n2, T, 200)
            c = compute_c5(res_x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res_x, n2)
                h_w2 = best_h.copy()
        except Exception:
            pass

    # Phase 3: Try alternate starting point at n=960 (structured pattern)
    if remaining() > 300:
        n_alt = 960
        # Try a periodic pattern
        x = np.arange(n_alt) * (2.0 / n_alt)
        h_alt = 0.5 + 0.5 * np.cos(2.0 * np.pi * 3.0 * x)
        h_alt = make_feasible(h_alt, n_alt)
        c_alt = compute_c5(h_alt)
        # Also try block pattern
        h_block = np.zeros(n_alt)
        h_block[:n_alt // 2] = 1.0
        h_block = make_feasible(h_block, n_alt)
        c_block = compute_c5(h_block)
        # Pick the better alternate
        if c_alt < c_block:
            h_start_alt = h_alt
        else:
            h_start_alt = h_block
        for T in [50.0, 200.0, 600.0]:
            if remaining() < 120:
                break
            try:
                res_x = _slsqp_solve(h_start_alt, n_alt, T, 250)
                c = compute_c5(res_x)
                if c < best_c5:
                    best_c5 = c
                    best_h = make_feasible(res_x, n_alt)
                    h_start_alt = best_h.copy()
            except Exception:
                pass

    # Phase 4: If we found something better at lower res, upsample and re-optimize
    if len(best_h) < 1920 and remaining() > 200:
        h_up = upsample_h(best_h, 1920)
        for T in [80.0, 300.0]:
            if remaining() < 100:
                break
            try:
                res_x = _slsqp_solve(h_up, 1920, T, 200)
                c = compute_c5(res_x)
                if c < best_c5:
                    best_c5 = c
                    best_h = make_feasible(res_x, 1920)
                    h_up = best_h.copy()
            except Exception:
                pass

    # Phase 5: Binary local search (single-coordinate)
    n_f = len(best_h)
    for _pass in range(3):
        if remaining() < 30:
            break
        for j in range(n_f):
            if remaining() < 5:
                break
            for val in [0.0, 1.0]:
                if abs(best_h[j] - val) < 1e-6:
                    continue
                trial = best_h.copy()
                trial[j] = val
                trial = make_feasible(trial, n_f)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial

    # Phase 6: Random pairwise binary moves
    if remaining() > 60:
        for _ in range(800):
            if remaining() < 10:
                break
            j = rng.randint(n_f)
            k = rng.randint(n_f)
            if j == k:
                continue
            improved = False
            for vj in [0.0, 1.0]:
                if abs(best_h[j] - vj) < 1e-6:
                    continue
                for vk in [0.0, 1.0]:
                    if abs(best_h[k] - vk) < 1e-6:
                        continue
                    trial = best_h.copy()
                    trial[j] = vj
                    trial[k] = vk
                    trial = make_feasible(trial, n_f)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-14:
                        best_c5 = c
                        best_h = trial
                        improved = True
                        break
                if improved:
                    break

    # Phase 7: Final SLSQP polish at high T
    if remaining() > 80:
        n_p = len(best_h)
        T_p = 800.0
        try:
            res_x = _slsqp_solve(best_h, n_p, T_p, 150)
            c = compute_c5(res_x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res_x, n_p)
        except Exception:
            pass

    # Phase 8: Final coordinate descent (binary focus)
    n_cd = len(best_h)
    for _round in range(8):
        if remaining() < 5:
            break
        improved = False
        for j in range(n_cd):
            if remaining() < 3:
                break
            for cand in [0.0, 1.0]:
                if abs(best_h[j] - cand) < 1e-6:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_cd)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial
                    improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
