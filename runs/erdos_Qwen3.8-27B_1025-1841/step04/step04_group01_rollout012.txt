import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0
_G_PEN = 10000.0
_G_TOP_IDX = None

def _obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _G_TOP_IDX is not None:
        cu = corr[_G_TOP_IDX]
    else:
        cu = corr
    mc = np.max(cu)
    sh = cu - mc
    sm = mc + np.log(np.sum(np.exp(T * sh))) / T
    pen = _G_PEN * (np.sum(x) - n * 0.5) ** 2
    return sm * dx + pen

def _feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(500):
        d = tgt - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = float(np.sum(m))
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    d = tgt - np.sum(h)
    if abs(d) > 1e-14:
        idx = int(np.argmin(h)) if d > 0 else int(np.argmax(h))
        h[idx] += d
    np.clip(h, 0.0, 1.0, out=h)
    return h

def _c5(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * 2.0 / n)

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

                        t[j] = v

                h[j] = bv
                h = _feasible(h, n)
                c5 = bc
                imp = True
        if not imp:

        hr = _feasible(r.x, n)
        cr = _c5(hr)
