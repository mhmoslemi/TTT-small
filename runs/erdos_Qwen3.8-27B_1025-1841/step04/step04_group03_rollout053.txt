```python
import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x
```

```python
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

            valid2 = (idx2 >= 0) & (idx2 < n)
            coeff = np.zeros(nc)
            coeff[valid1] += (1.0 - h[idx1[valid1]])
            coeff[valid2] -= h[idx2[valid2]]
            
            # 1D search: grid + refine
            u_vals = np.concatenate([[0.0, 1.0],
                                     np.linspace(max(0, u-0.3), min(1, u+0.3), 15)])
            u_vals = np.unique(u_vals)
            best_u = u
            best_v = np.inf
            for up in u_vals:
                du = up - u
                if abs(du) < 1e-13:
                    continue
                nv = corr + du * coeff
                nv[m_mid] = corr[m_mid] + du * (1.0 - up - u)
                v = np.max(nv)
                if v < best_v:
                    best_v = v
                    best_u = up
            # Refine
            if 0.01 < best_u < 0.99:
                lo = max(0.0, best_u - 0.05)
                hi = min(1.0, best_u + 0.05)
                u_vals2 = np.linspace(lo, hi, 30)
                for up in u_vals2:
                    du = up - u
                    if abs(du) < 1e-13:
                        continue
                    nv = corr + du * coeff
                    nv[m_mid] = corr[m_mid] + du * (1.0 - up - u)
                    v = np.max(nv)
                    if v < best_v:
                        best_v = v
                        best_u = up
            # Even finer
            if 0.01 < best_u < 0.99:
                lo = max(0.0, best_u - 0.01)
                hi = min(1.0, best_u + 0.01)
                u_vals3 = np.linspace(lo, hi, 20)
                for up in u_vals3:
                    du = up - u
                    if abs(du) < 1e-13:
                        continue
                    nv = corr + du * coeff
                    nv[m_mid] = corr[m_mid] + du * (1.0 - up - u)
                    v = np.max(nv)
                    if v < best_v:
                        best_v = v
                        best_u = up
            
            if abs(best_u - u) > 1e-10:
                du = best_u - u
                h[j] = best_u
                corr + du * coeff
                corr = corr + du * coeff
                corr[m_mid] = corr[m_mid] - du * coeff[m_mid] + du * (1.0 - best_u - u)
                # Fix: recompute entry correctly
                # corr[m_mid] after linear update was corr_old[m_mid] + du*coeff[m_mid]
                # We want: corr_old[m_mid] + du*(1-best_u-u)
                # coeff[m_mid] = 1-2u (from formula), so:
                # corr[m_mid] = corr_old[m_mid] + du*(1-2u) [already applied]
                # Need: corr_old[m_mid] + du*(1-best_u-u)
                # Difference: du*((1-best_u-u) - (1-2u)) = du*(u-best_u)
                corr[m_mid] += du * (u - best_u)
                improved = True
        
        # Project to feasible
        h = make_feasible(h, n)
        actual_c5 = compute_c5(h)
        if actual_c5 < best_c5 - 1e-12:
            best_c5 = actual_c5
        elif actual_c5 < best_c5:
            best_c5 = actual_c5
        if not improved:
            break
    return h, best_c5

def fourier_perturb(h, n, rng, n_modes=4, amp=0.04):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.5, 1.5)
    return make_feasible(h_new, n)

def block_perturb(h, n, rng, n_segs=3):
    h_new = h.copy()
    for _ in range(n_segs):
        length = rng.randint(n // 20, n // 5)
        start = rng.randint(0, n - length)
        if rng.random() < 0.4:
            h_new[start:start+length] = 1.0 - h_new[start:start+length]
        else:
            h_new[start:start+length] += rng.uniform(-0.3, 0.3)
    return make_feasible(h_new, n)

def shift_perturb(h, n, rng):
    s = rng.randint(1, n // 4)
    h_shift = np.roll(h, s)
    alpha = rng.uniform(0.3, 0.7)
    h_new = alpha * h + (1 - alpha) * h_shift
    return make_feasible(h_new, n)

def multi_temp_slsqp(h, n, start_time, budget_s, temps=None):
    if temps is None:
        temps = [50.0, 200.0, 800.0, 3000.0, 12000.0, 50000.0]
    current_c5 = compute_c5(h)
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        x_res = slsqp_optimize(h, n, T, 200, 1e-14)
        x_res = np.clip(x_res, 0, 1)
        x_res = make_feasible(x_res, n)
        c5_res = compute_c5(x_res)
        if c5_res < current_c5:
            h = x_res
            current_c5 = c5_res
    return h, current_c5

def binary_sa_fast(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    dx = 2.0 / n
    nc = 2 * n - 1
    m_mid = n - 1
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b, dtype=int) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0: break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b, dtype=int) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0: break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    corr = np.correlate(b, 1.0 - b, mode='full').copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0; b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            corr = np.correlate(b, 1.0 - b, mode='full').copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1; b[j] = 0
    return best_b, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    
    # Phase 1: SLSQP multi-temperature
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s,
                                           temps=[50.0, 200.0, 1000.0, 5000.0, 20000.0])
    
    # Phase 2: Aggressive coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        best_h, best_c5 = fast_coord_descent(best_h, n0, 200, start_time, budget_s)
    
    # Phase 3: Perturbation + SLSQP + coord descent cycles
    perts = [fourier_perturb, block_perturb, shift_perturb]
    for pi, pf in enumerate(perts):
        for trial in range(3):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 150:
                break
            try:
                hp = pf(best_h, n0, rng)
                if pi == 0:
                    hr, cr = multi_temp_slsqp(hp, n0, start_time, budget_s,
                                              temps=[200.0, 2000.0, 10000.0])
                else:
                    hr, cr = multi_temp_slsqp(hp, n0, start_time, budget_s,
                                              temps=[500.0, 5000.0])
                # Coordinate descent refinement
                remaining = budget_s - (time.time() - start_time)
                if remaining > 50:
                    hr, cr = fast_coord_descent(hr, n0, 80, start_time, budget_s)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass
    
    # Phase 4: Binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        tl = min(remaining - 40, 120)
        b, c5_b = binary_sa_fast(best_h, n0, 80000, 0.005, 0.00001, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            best_h, best_c5 = fast_coord_descent(best_h, n0, 100, start_time, budget_s)
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s,
                                                   temps=[1000.0, 10000.0, 50000.0])
                best_h, best_c5 = fast_coord_descent(best_h, n0, 80, start_time, budget_s)
    
    # Phase 5: More perturbations with varied parameters
    for _ in range(4):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 120:
            break
        try:
            hp = fourier_perturb(best_h, n0, rng, n_modes=6, amp=0.06)
            hp = block_perturb(hp, n0, rng, n_segs=2)
            hr, cr = multi_temp_slsqp(hp, n0, start_time, budget_s,
                                      temps=[300.0, 3000.0, 15000.0])
            remaining = budget_s - (time.time() - start_time)
            if remaining > 40:
                hr, cr = fast_coord_descent(hr, n0, 60, start_time, budget_s)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()
        except Exception:
            pass
    
    # Phase 6: Higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        h_up, c5_up = multi_temp_slsqp(h_up, n1, start_time, budget_s,
                                       temps=[200.0, 2000.0, 10000.0])
        remaining = budget_s - (time.time() - start_time)
        if remaining > 60:
            h_up, c5_up = fast_coord_descent(h_up, n1, 60, start_time, budget_s)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            n0 = n1
    
    # Phase 7: Final SLSQP + coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        try:
            x_res = slsqp_optimize(best_h, n_pts, 100000.0, 300, 1e-15)
            x_res = np.clip(x_res, 0, 1)
            x_res = make_feasible(x_res, n_pts)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res
        except Exception:
            pass
    
    # Phase 8: Final coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        best_h, best_c5 = fast_coord_descent(best_h, n_pts, 300, start_time, budget_s)
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
