import numpy as np
from scipy.optimize import minimize
import time

_GLOBAL_T = 100.0
_GLOBAL_N = 960

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def slsqp_obj(x):

    T = _GLOBAL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_solve(h, T, maxiter):
    global _GLOBAL_T, _GLOBAL_N
    _GLOBAL_T = T
    _GLOBAL_N = len(h)
    bounds = [(0.0, 1.0)] * len(h)
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    res = minimize(slsqp_obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-15})
    return res.x

def targeted_local_search(b, n, max_rounds, deadline):

    best = compute_c5(b)

        max_val = np.max(corr)
        # Find shifts that are near the max
        threshold = max_val - 0.005
        hot_shifts = np.where(corr >= threshold)[0]
        
        improved = False
        for sh in hot_shifts:
            if time.time() > deadline:
                break

            # shift = sh - (n-1)

            if shift == 0:

                max_i = n - 1 - shift
            else:
                max_i = n - 1 + shift

                continue
