        if j1e > j1s:
            grad[j1s:j1e] += w_k * (1.0 - h)[j1s + s:j1e + s]
        j2s = max(0, s)
        j2e = min(n, n + s)
        if j2e > j2s:
            grad[j2s:j2e] -= w_k * h[j2s - s:j2e - s]
    grad *= dx
    return grad, compute_c5(h)

def coordinate_descent_exact(h, n, max_rounds=3, time_limit=None, start_time=None):
    """Exact coordinate descent: for each j, find optimal h[j] in [0,1] 
    that minimizes the max of correlations, while maintaining sum constraint."""
    best_c5 = compute_c5(h)
    for rd in range(max_rounds):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            old_val = h[j]
            # Try: 0, 1, and some intermediate values
            # The constraint sum(h)=n/2 means if we change h[j] by delta, 
            # we need to compensate. 
            # Strategy: try setting h[j] to v, then fix the sum by adjusting 
            # the single most flexible coordinate.
            best_v = old_val
            best_c = best_c5
            candidates = [0.0, 1.0]
            # Add fine grid around current value
            for dt in [0.01, 0.05, 0.1, 0.2, 0.3]:
                candidates.append(max(0.0, old_val - dt))
                candidates.append(min(1.0, old_val + dt))
            for v in candidates:
                if v == old_val:
                    continue
                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_v != old_val:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def get_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exp_s = np.exp(T * (corr - mc))
    weights = exp_s / np.sum(exp_s)
    grad = np.zeros(n)
    # For each lag s (from -(n-1) to n-1):
    # weights at position s+(n-1) in the full correlation
    for s in range(-(n-1), n):
        w = weights[s + n - 1]
        if w < 1e-14:
            continue
        # (1-h[m+s]) valid when 0 <= m+s < n, i.e., -s <= m <= n-1-s
        m1s = max(0, -s)
        m1e = min(n, n - s)  # exclusive
        if m1e > m1s:
            grad[m1s:m1e] += w * (1.0 - h)[m1s + s : m1e + s]
        # -h[m-s] valid when 0 <= m-s < n, i.e., s <= m <= n-1+s
        m2s = max(0, s)
        m2e = min(n, n + s)  # exclusive
        if m2e > m2s:
            grad[m2s:m2e] -= w * h[m2s - s : m2e - s]
    grad *= dx
    return grad

- autocorr_0 = n/2
- autocorr_k = n/2 - k for 0 < k < n/2 (the overlap of [0, n/2) and [k, n/2+k))
- autocorr_k = 0 for k >= n/2

So corr[k] = n/2 - (n/2 - k) = k for 0 < k < n/2.
The max of corr[k] for k ≠ 0 is at k = n/2 (approximately), giving corr ≈ n/2.
C5 ≈ (n/2) * (2/n) = 1.0. That's terrible!

So we need a balance: not too clumped (which gives high max correlation at large lags) but not too spread (which gives high correlation at small lags).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def get_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exp_s = np.exp(T * (corr - mc))
    sw = np.sum(exp_s)
    weights = exp_s / sw
    grad = np.zeros(n)
    for s in range(-(n - 1), n):
        w = weights[s + n - 1]
        if w < 1e-13:
            continue
        m1s = max(0, -s)
        m1e = min(n, n - s)
        if m1e > m1s:
            grad[m1s:m1e] += w * (1.0 - h)[m1s + s:m1e + s]
        m2s = max(0, s)
        m2e = min(n, n + s)
        if m2e > m2s:
            grad[m2s:m2e] -= w * h[m2s - s:m2e - s]
    grad *= dx
    return grad

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def adam_step(h, grad, m, v, t, lr, beta1=0.9, beta2=0.999, eps=1e-8):
    m = beta1 * m + (1 - beta1) * grad
    v = beta2 * v + (1 - beta2) * grad * grad
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    step = lr / (np.sqrt(v_hat) + eps)
    new_h = h - step * m_hat
    return new_h, m, v

def coordinate_descent_fast(h, n, max_rounds=5, time_lim=None, t0=None):
    best_c5 = compute_c5(h)
    for rd in range(max_rounds):
        if time_lim is not None and time.time() - t0 > time_lim:
            break
        improved = False
        for j in range(n):
            if time_lim is not None and time.time() - t0 > time_lim:
                break
            old_val = h[j]
            best_v = old_val
            best_c = best_c5
            cands = [0.0, 1.0]
            for dt in [0.02, 0.05, 0.1, 0.2, 0.4]:
                cands.append(max(0.0, old_val - dt))
                cands.append(min(1.0, old_val + dt))
            for v in cands:
                if abs(v - old_val) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_v != old_val:
                h = h.copy()
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def fine_coordinate_descent(h, n, time_lim=None, t0=None):
    best_c5 = compute_c5(h)
    improved = True
    while improved:
        if time_lim is not None and time.time() - t0 > time_lim:
            break
        improved = False
        for j in range(n):
            if time_lim is not None and time.time() - t0 > time_lim:
                break
            old_val = h[j]
            best_v = old_val
            best_c = best_c5
            for dt in [0.005, 0.01, 0.02, 0.03, 0.05]:
                for v in [max(0.0, old_val - dt), min(1.0, old_val + dt)]:
                    if abs(v - old_val) < 1e-10:

                    trial = h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:

                        best_v = v
            if best_v != old_val:
                h = h.copy()
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5
