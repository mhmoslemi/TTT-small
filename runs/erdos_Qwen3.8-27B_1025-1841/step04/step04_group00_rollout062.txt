    """Compute smooth approximation of C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Soft max
    cm = np.max(corr)
    exp_corr = np.exp((corr - cm) / T)
    log_sum = cm / T + np.log(np.sum(exp_corr))
    obj = T * log_sum * dx
    
    # Softmax weights
    p = exp_corr / np.sum(exp_corr)
    
    # Gradient: only top_m contribute significantly
    top_idx = np.argsort(p)[-top_m:]
    
    grad = np.zeros(n)
    for k in top_idx:
        if p[k] < 1e-10:
            continue
        # corr at index k corresponds to lag k - (n-1)
        lag = k - (n - 1)
        # corr_k = sum_{j valid} h[j] * (1 - h[j+lag])
        # ∂corr_k/∂h[j] = (1-h[j+lag]) if j valid as first + (-h[j-lag]) if j valid as second

    return obj, grad

Workaround: reparameterize. Let h[0:n-1] be free variables in [0,1], and h[n-1] = n/2 - sum(h[0:n-1]). But h[n-1] must be in [0,1], so sum(h[0:n-1]) must be in [n/2-1, n/2]. This is a linear constraint on n-1 variables, which L-BFGS-B can't handle.

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_smooth(h, n, T, maxiter, deadline):
    dx = 2.0 / n
    def f(x):
        h2 = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h2, 1.0 - h2, mode='full')
        cm = corr.max()
        return (cm + T * np.log(np.sum(np.exp((corr - cm) / T))) / T) * dx
    def g(x):
        h2 = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h2, 1.0 - h2, mode='full')
        cm = corr.max()
        exp_c = np.exp((corr - cm) / T)
        p = exp_c / exp_c.sum()
        top_m = 60
        top_idx = np.argsort(p)[-top_m:]

            if pk < 1e-14:

            lag = k - (n - 1)
            if lag >= 0:
                grad[:n - lag] += pk * (1.0 - h2[lag:])
                grad[lag:] += -pk * h2[:n - lag]
            else:
                l = -lag
                grad[l:] += pk * (1.0 - h2[:n - l])
                grad[:n - l] += -pk * h2[l:]
        return grad * dx
    def c1(x):
        return x.sum() - n / 2.0
    def c1g(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
