    """Get indices of top m correlation values."""
    return np.argsort(corr)[-m:]

def subgradient_top(corr, h, top_idx, n):
    """Compute subgradient from top shifts."""
    n_corr = len(corr)
    mid = n - 1  # offset for full correlation
    g = np.zeros(n)
    for s_idx in top_idx:
        s = s_idx - mid  # actual shift
        for j in range(n):
            # Contribution from h[j] as first factor: (1 - h[j+s]) if j+s in range
            if 0 <= j + s < n:

            if 0 <= j - s < n:
                g[j] -= h[j - s]
    g /= len(top_idx)
    return g

def subgradient_top_vec(corr, h, top_idx, n):
    n_corr = len(corr)
    mid = n - 1
    g = np.zeros(n)
    for s_idx in top_idx:
        s = s_idx - mid
        if s > 0:
            # j+s in [0,n) means j in [0, n-s)
            # (1-h[j+s]) for j in [0, n-s)
            g[:n-s] += (1.0 - h[s:])
            # j-s in [0,n) means j in [s, n)
            # -h[j-s] for j in [s, n)
            g[s:] -= h[:n-s]
        elif s < 0:
            t = -s
            # j+s = j-t in [0,n) means j in [t, n)
            g[t:] += (1.0 - h[:n-t])
            # j-s = j+t in [0,n) means j in [0, n-t)
            g[:n-t] -= h[t:]
        # s == 0: g[j] += (1-h[j]) - h[j] = 1 - 2h[j]
        # But for binary h, this is 0 or -1 or 1
        else:
            g += (1.0 - 2.0 * h)
    g /= len(top_idx)
    return g
