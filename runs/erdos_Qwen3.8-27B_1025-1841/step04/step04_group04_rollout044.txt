    return np.sum(x) - _OPT_N / 2.0

def _slsqp_jac(x):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_snap_pass(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        for val in [0.0, 1.0]:
            if abs(best[j] - val) < 1e-10:
                continue
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def coord_descent_pass(h, n, cands, elapsed_fn, time_budget):
    best_c = compute_c5(h)
    for j in range(n):
        if elapsed_fn() > time_budget:
            break
        old = h[j]
        best_lc = best_c
        best_lv = old
        for cand in cands:
            v = max(0.0, min(1.0, cand))
            if abs(v - old) < 1e-15:
                continue
            trial = h.copy()
            trial[j] = v
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_lc - 1e-14:
                best_lc = c
                best_lv = v
        if best_lv != old:
            h = h.copy()
            h[j] = best_lv
            h = make_feasible(h, n)
            best_c = best_lc
    return h, best_c

def fine_coord_descent(h, n, elapsed_fn, time_budget):
    best_c = compute_c5(h)
    improved_any = False
    for j in range(n):
        if elapsed_fn() > time_budget:
            break
        old = h[j]
        best_lc = best_c
        best_lv = old
        for d in [-0.02, -0.01, -0.005, -0.002, -0.001, 0.001, 0.002, 0.005, 0.01, 0.02]:
            v = old + d
            if v < -1e-12 or v > 1.0 + 1e-12:
                continue
            v = max(0.0, min(1.0, v))
            trial = h.copy()
            trial[j] = v
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_lc - 1e-15:
                best_lc = c
                best_lv = v
        if best_lv != old:
            h = h.copy()
            h[j] = best_lv
            h = make_feasible(h, n)
            best_c = best_lc
            improved_any = True
    return h, best_c, improved_any

def slsqp_polish(h, n, T, maxiter, elapsed_fn, time_budget):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_try = make_feasible(res.x, n)
        c_try = compute_c5(h_try)
        return h_try, c_try
    except Exception:
        return h, compute_c5(h)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(960, 0.5)
    n_pts = len(h0)
    h0 = make_feasible(h0, n_pts)
    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP polish from parent
    if remaining() > 150:
        hb, cb = slsqp_polish(best_h, n_pts, 60.0, 300, elapsed, budget_s - 120)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 2: SLSQP with different temperature from perturbed start
    if remaining() > 200:
        h_pert = best_h + rng.randn(n_pts) * 0.02
        h_pert = make_feasible(h_pert, n_pts)
        hb, cb = slsqp_polish(h_pert, n_pts, 120.0, 250, elapsed, budget_s - 150)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 3: Broad coordinate descent
    if remaining() > 300:
        cands_broad = [0.0, 1.0, 0.25, 0.5, 0.75, 0.1, 0.9, 0.15, 0.85, 0.4, 0.6]
        budget_cd = budget_s - 200
        for rnd in range(30):
            if elapsed() > budget_cd:
                break
            best_h, best_c5 = coord_descent_pass(best_h, n_pts, cands_broad, elapsed, budget_cd)

    # Phase 4: Binary snap + relaxation cycles
    for cycle in range(6):
        if remaining() < 60:
            break
        best_h, best_c5 = binary_snap_pass(best_h, n_pts)
        h_rel = best_h + rng.randn(n_pts) * 0.008
        h_rel = make_feasible(h_rel, n_pts)
        c_rel = compute_c5(h_rel)
        if c_rel < best_c5:
            best_h, best_c5 = h_rel, c_rel

    # Phase 5: Fine coordinate descent (multiple passes)
    if remaining() > 150:
        budget_fd = budget_s - 60
        for rnd in range(40):
            if elapsed() > budget_fd:
                break
            best_h, best_c5, improved = fine_coord_descent(best_h, n_pts, elapsed, budget_fd)
            if not improved:
                break

    # Phase 6: Another SLSQP polish
    if remaining() > 100:
        hb, cb = slsqp_polish(best_h, n_pts, 200.0, 200, elapsed, budget_s - 60)
        if cb < best_c5:
            best_h, best_c5 = hb, cb

    # Phase 7: Final binary snap
    if remaining() > 30:
        best_h, best_c5 = binary_snap_pass(best_h, n_pts)

    # Phase 8: Ultra-fine coordinate descent
    if remaining() > 30:
        for rnd in range(15):
            if remaining() < 10:
                break
            best_h, best_c5, improved = fine_coord_descent(best_h, n_pts, elapsed, budget_s - 5)
            if not improved:
                break

    # Phase 9: Final binary snap
    if remaining() > 10:
        best_h, best_c5 = binary_snap_pass(best_h, n_pts)

    best_h = make_feasible(best_h, n_pts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_pts
