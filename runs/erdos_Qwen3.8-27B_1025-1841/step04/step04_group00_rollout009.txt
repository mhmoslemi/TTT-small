        if mask.sum() > 0:

            proj = np.clip(proj, 0.0, 1.0)

def slsqp_smooth(h, n, T, top_m, maxiter, start_time, time_limit):
    dx = 2.0 / n
    
    def obj_and_grad(x):
        x = np.clip(x, 0.0, 1.0)
        corr = np.correlate(x, 1.0 - x, mode='full')
        _, weights = smooth_max_and_weights(corr, T)
        if top_m is not None and top_m < len(weights):
            idx = np.argsort(weights)[-top_m:]
            w2 = np.zeros_like(weights)
            w2[idx] = weights[idx]
            w2 /= w2.sum()
            weights = w2
        val = np.sum(weights * corr) * dx
        grad = grad_wrt_h(x, weights, n)
        return val, grad
    
    val0, _ = obj_and_grad(h)
    
    def obj(x):
        v, _ = obj_and_grad(np.clip(x, 0.0, 1.0))
        return v
    
    def jac(x):
        _, g = obj_and_grad(np.clip(x, 0.0, 1.0))
        # Penalty for sum constraint
        diff = np.sum(x) - n / 2.0
        g = g - diff * 2.0
        return g
    
    bounds = [(0.0, 1.0)] * n

    try:

def coord_descent_fine(h, n, start_time, time_limit, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    pass_count = 0
    while improved and time.time() - start_time < time_limit - 3:
        improved = False
        pass_count += 1
        for j in range(n):
            if time.time() - start_time > time_limit - 2:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in steps:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * s
                    if 0.0 <= v <= 1.0 and abs(v - old_val) > 1e-10:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = project_box_sum(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-14:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = project_box_sum(best_h, n)
                best_c5 = best_c
                improved = True
    return best_h, best_c5

def binary_projection(h, n):
    target = n // 2
    h_sorted = np.sort(h)[::-1]
    threshold = h_sorted[target - 1] if target < n else 0.0
    binary = (h >= threshold + 1e-10).astype(float)
    ones_count = int(binary.sum())
    if ones_count > target:
        idx = np.where(binary == 1.0)[0]
        vals = h[idx]
        excess = ones_count - target
        remove = idx[np.argsort(vals)[:excess]]
        binary[remove] = 0.0
    elif ones_count < target:
        idx = np.where(binary == 0.0)[0]
        vals = h[idx]
        deficit = target - ones_count
        add = idx[np.argsort(vals)[-deficit:]]
        binary[add] = 1.0
    return binary

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def remaining():
        return budget_s - (time.time() - start_time)
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    
    n = len(h_init)
    h_init = project_box_sum(h_init, n / 2.0)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Binary projection check
    h_bin = binary_projection(best_h, n)
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h, best_c5 = h_bin, c5_bin
    
    # Phase 2: Nesterov subgradient, moderate T
    if remaining() > 180:
        t_limit = min(200, remaining() - 20)
        h2, c5_2 = nesterov_subgrad(best_h, n, T_init=100.0, T_final=500.0,
                                     max_iter=3000, start_time=start_time,
                                     time_limit=time.time() + t_limit, top_frac=1.0)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        h_bin = binary_projection(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
    
    # Phase 3: SLSQP with smooth max
    if remaining() > 150:
        t_limit = min(120, remaining() - 20)
        h2, c5_2 = slsqp_smooth(best_h, n, T=500.0, top_m=20,
                                 maxiter=500, start_time=start_time,
                                 time_limit=time.time() + t_limit)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    # Phase 4: Nesterov with higher T (more exact max)
    if remaining() > 150:
        t_limit = min(150, remaining() - 20)
        h2, c5_2 = nesterov_subgrad(best_h, n, T_init=500.0, T_final=2000.0,
                                     max_iter=3000, start_time=start_time,
                                     time_limit=time.time() + t_limit, top_frac=0.8)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        h_bin = binary_projection(best_h, n)
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_h, best_c5 = h_bin, c5_bin
    
    # Phase 5: SLSQP with very high T, top only
    if remaining() > 100:
        t_limit = min(80, remaining() - 20)
        h2, c5_2 = slsqp_smooth(best_h, n, T=2000.0, top_m=8,
                                 maxiter=300, start_time=start_time,
                                 time_limit=time.time() + t_limit)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    # Phase 6: Coordinate descent
    if remaining() > 80:
        t_limit = min(80, remaining() - 10)
        h2, c5_2 = coord_descent_fine(best_h, n, start_time,
                                       time.time() + t_limit,
                                       steps=[0.005, 0.002, 0.001, 0.0005])
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    # Phase 7: Try upsampled resolution
    if remaining() > 200:
        n2 = n * 2
        old_x = np.linspace(0.0, 2.0, n)
        new_x = np.linspace(0.0, 2.0, n2)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = project_box_sum(h_up, n2 / 2.0)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5 + 0.003:
            if remaining() > 100:
                t_limit = min(100, remaining() - 60)
                h2, c5_2 = nesterov_subgrad(h_up, n2, T_init=200.0, T_final=800.0,
                                             max_iter=2000, start_time=start_time,
                                             time_limit=time.time() + t_limit, top_frac=0.7)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2
            if remaining() > 60:
                h_bin = binary_projection(best_h, len(best_h))
                c5_bin = compute_c5(h_bin)
                if c5_bin < best_c5:
                    best_h, best_c5 = h_bin, c5_bin
            if remaining() > 50:
                t_limit = min(50, remaining() - 10)
                h2, c5_2 = slsqp_smooth(best_h, len(best_h), T=800.0, top_m=15,
                                         maxiter=200, start_time=start_time,
                                         time_limit=time.time() + t_limit)
                if c5_2 < best_c5:
                    best_h, best_c5 = h2, c5_2
    
    # Phase 8: Random perturbations + local refine
    for restart in range(4):
        if remaining() < 50:
            break
        np_pts = len(best_h)
        scale = 0.008
        h_try = best_h + rng.randn(np_pts) * scale
        h_try = project_box_sum(h_try, np_pts / 2.0)
        t_limit = min(30, remaining() - 20)
        h2, c5_2 = nesterov_subgrad(h_try, np_pts, T_init=300.0, T_final=1000.0,
                                     max_iter=800, start_time=start_time,
                                     time_limit=time.time() + t_limit, top_frac=0.5)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    # Phase 9: Final Nesterov polish
    if remaining() > 30:
        t_limit = min(30, remaining() - 5)
        h2, c5_2 = nesterov_subgrad(best_h, len(best_h), T_init=2000.0, T_final=5000.0,
                                     max_iter=1000, start_time=start_time,
                                     time_limit=time.time() + t_limit, top_frac=0.3)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    # Phase 10: Final coordinate descent
    if remaining() > 15:
        t_limit = min(15, remaining() - 3)
        h2, c5_2 = coord_descent_fine(best_h, len(best_h), start_time,
                                       time.time() + t_limit,
                                       steps=[0.001, 0.0005, 0.0002])
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
    
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
