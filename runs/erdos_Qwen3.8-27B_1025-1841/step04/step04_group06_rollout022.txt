    #                    = (1-h[j-m+n-1]) - h[j+m-n+1]
    # But we need to be careful about which terms are valid
    
    n_corr = len(corr)  # 2n-1
    grad = np.zeros(n)
    
    for m in active:
        # For each j, the contribution to grad[j] from corr[m] is:
        # Term 1: h[j] appears as the first factor when i=j, second index is j-m+n-1
        #   Valid if 0 <= j-m+n-1 < n, i.e., m-n+1 <= j <= m
        #   Contribution: (1-h[j-m+n-1])
        # Term 2: h[j] appears as the second factor when i-m+n-1=j, i.e., i=j+m-n+1
        #   Valid if 0 <= j+m-n+1 < n, i.e., n-1-m <= j <= n-1
        #   Contribution: -h[j+m-n+1] (derivative of h[i]*(1-h[j]) w.r.t. h[j] is -h[i])
        
        # This is O(n) per active m, which could be expensive if many are active
        # Let's vectorize
        j_all = np.arange(n)
        
        # Term 1: valid when 0 <= j-m+n-1 < n
        idx1 = j_all - m + n - 1
        valid1 = (idx1 >= 0) & (idx1 < n)
        contrib1 = np.zeros(n)
        contrib1[valid1] = 1.0 - h[idx1[valid1]]
        
        # Term 2: valid when 0 <= j+m-n+1 < n
        idx2 = j_all + m - n + 1
        valid2 = (idx2 >= 0) & (idx2 < n)
        contrib2 = np.zeros(n)
        contrib2[valid2] = -h[idx2[valid2]]
        
        grad += contrib1 + contrib2
    
    # Average over active lags
    grad /= len(active)
    
    # Gradient step
    h_new = h - learning_rate * grad
    # Project onto [0,1]
    h_new = np.clip(h_new, 0.0, 1.0)
    # Project onto sum = n/2
    h_new = make_feasible(h_new, n)
    
    return h_new

def subgradient_method(h_start, n, max_iters, lr_init, lr_decay):
    h = h_start.copy()
    best_h = h_start.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    
    for it in range(max_iters):
        h_new = subgradient_step(h, n, lr)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Don't accept, but still decay learning rate
            pass
        lr *= lr_decay
    
    return best_h, best_c5

    """Efficient binary local search using direct recomputation."""
    h = h_bin.copy()
    current = compute_c5(h)
    corr = compute_corr(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(n_tries_per_round, len(ones) * len(zeros))
        for _ in range(n_tries):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            
            h[oi] = 0.0
            h[oj] = 1.0
            corr_new = compute_corr(h)
            c = float(np.max(corr_new) * (2.0 / n))
            
            if c < current - 1e-14:
                current = c
                corr = corr_new
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)
                zeros = np.append(zeros, oi)
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    
    return h, current

def coordinate_descent(h, n, max_rounds=10, deltas=None):
    """Coordinate descent with binary and continuous moves."""
    if deltas is None:
        deltas = [0.1, -0.1, 0.05, -0.05, 0.5, -0.5, 1.0, 0.0]
    h = h.copy()
    current = compute_c5(h)
    dx = 2.0 / n
    
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < -1e-9 or nv > 1 + 1e-9:
                    continue
                nv = np.clip(nv, 0.0, 1.0)
                if abs(nv - ov) < 1e-10:
                    continue

                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current - 1e-13:
                    current = c
                    h = tr
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    
    return h, current
