def _soft_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

    return np.sum(h) - n / 2.0

    return np.ones(n)

                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    # Get initial solution
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)
    n = len(h0)
    h0 = make_feasible(h0, n)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: Subgradient descent on true max (continuous)
    if time_left() > 200:
        h_sg, c5_sg = subgradient_descent(best_h, n, 3000, 0.08, rng)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Phase 2: Binary projection + local search
    if time_left() > 200:
        hb, c5_b = binary_local_search(best_h, n, 50000, rng)
        c5_b_cont = compute_c5(make_feasible(hb, n))
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = make_feasible(hb, n)
    
    # Phase 3: Simulated annealing on binary
    if time_left() > 300:
        hb_sa, c5_sa = binary_sa(best_h, n, 0.003, 1e-6, 5, rng)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = make_feasible(hb_sa, n)
    
    # Phase 4: SLSQP with log-sum-exp at various temperatures
    temp_schedule = [(50.0, 300), (200.0, 200), (500.0, 150), (1000.0, 100), (3000.0, 80)]
    for T, max_it in temp_schedule:
        if time_left() < 30:
            break
        try:
            h_opt = slsqp_optimize(best_h, n, T, max_it)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass
    
    # Phase 5: Binary local search again (from improved continuous)
    if time_left() > 100:
        hb2, c5_b2 = binary_local_search(best_h, n, 50000, rng)
        if c5_b2 < best_c5:
            best_c5 = c5_b2
            best_h = make_feasible(hb2, n)
    
    # Phase 6: Coordinate descent polish
    if time_left() > 100:
        best_h, best_c5 = coordinate_descent_polish(best_h, n, 5, rng)
    
    # Phase 7: Try at higher resolution
    if time_left() > 150:
        n2 = n * 2
        if n2 <= 5120:
            h2 = upsample_h(best_h, n2)
            c52 = compute_c5(h2)
            if c52 < best_c5:
                best_c5 = c52
                best_h = h2
            if time_left() > 60:
                try:
                    h2_opt = slsqp_optimize(h2, n2, 500.0, 150)
                    c52_opt = compute_c5(h2_opt)
                    if c52_opt < best_c5:
                        best_c5 = c52_opt
                        best_h = h2_opt
                except Exception:
                    pass
    
    # Phase 8: Another round of subgradient + binary
    if time_left() > 80:
        n_f = len(best_h)
        h_sg2, c5_sg2 = subgradient_descent(best_h, n_f, 2000, 0.05, rng)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()
        if time_left() > 40:
            hb3, c5_b3 = binary_local_search(best_h, n_f, 30000, rng)
            if c5_b3 < best_c5:
                best_c5 = c5_b3
                best_h = make_feasible(hb3, n_f)
    
    # Phase 9: Multi-start exploration
    if time_left() > 60:
        n_f = len(best_h)
        # Random perturbation
        h_rand = best_h + 0.1 * rng.randn(n_f)
        h_rand = make_feasible(h_rand, n_f)
        try:
            h_r_opt = slsqp_optimize(h_rand, n_f, 200.0, 150)
            c5_r = compute_c5(h_r_opt)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r_opt
        except Exception:
            pass
    
    # Final polish
    if time_left() > 20:
        n_f = len(best_h)
        best_h, best_c5 = coordinate_descent_polish(best_h, n_f, 3, rng)
    
    # Ensure feasibility
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, n_final
