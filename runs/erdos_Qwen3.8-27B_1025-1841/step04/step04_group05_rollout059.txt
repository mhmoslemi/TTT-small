```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff))+1:]
            for i in idx:
                h[i] = min(1.0, h[i] + diff / len(idx))
        else:
            idx = np.argsort(h)[:int(-diff)+1:]
            for i in idx:
                h[i] = max(0.0, h[i] + diff / len(idx))
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

# Global variables for objective function
_G_N = 0
_G_T = 1.0
_G_PEAK_LAGS = None

def smooth_max_obj(h):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Focused penalty on peak lags
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    base = (mc + log_sum / T) * dx
    if _G_PEAK_LAGS is not None and len(_G_PEAK_LAGS) > 0:
        # Extra penalty on specific lags
        peak_corr = corr[_G_PEAK_LAGS]
        peak_max = np.max(peak_corr)
        peak_shifted = peak_corr - peak_max
        peak_log_sum = np.log(np.sum(np.exp(100 * peak_shifted)))
        base += 0.1 * (peak_max + peak_log_sum / 100) * dx
    return base

def sum_constraint(h):
    return np.sum(h) - _G_N / 2.0

def sum_constraint_jac(h):
    return np.ones(_G_N)

def slsqp_optimize(h0, n, T, max_iter, peak_lags=None):
    global _G_N, _G_T, _G_PEAK_LAGS
    _G_N = n
    _G_T = T
    _G_PEAK_LAGS = peak_lags
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    res = minimize(smooth_max_obj, h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-13})
    return make_feasible(res.x, n)

def binary_local_search(h_binary, n, max_iters=5000, time_limit=None):
    """Local search on binary patterns: flip individual bits or pairs"""
    best_h = h_binary.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    
    for it in range(max_iters):
        if time_limit and (time.time() - start) > time_limit:
            break
        # Try flipping 1-3 consecutive bits
        n_flip = 1 + (it % 3)
        idx = np.random.randint(0, n - n_flip + 1)
        trial = best_h.copy()
        trial[idx:idx+n_flip] = 1.0 - trial[idx:idx+n_flip]
        trial = make_feasible(trial, n)
        c5 = compute_c5(trial)
        if c5 < best_c5:
            best_h = trial
            best_c5 = c5
    
    return best_h, best_c5

    """Simulated annealing on binary patterns"""
    best_h = h_binary.copy()
    best_c5 = compute_c5(best_h)
    cur_h = best_h.copy()
    cur_c5 = best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(200):
        diff = n / 2.0 - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return project_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return project_feasible(h_new, new_n)

_G_N = 0
_G_T = 1.0
_G_PEAK = None

def obj_fn(h):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    val = (mc + log_sum / T) * dx
    if _G_PEAK is not None:
        pk = corr[_G_PEAK]
        pm = np.max(pk)
        ps = pk - pm
        val += 0.05 * (pm + np.log(np.sum(np.exp(200 * ps))) / 200) * dx
    return val

def con_eq(h):
    return np.sum(h) - _G_N / 2.0

def con_jac(h):
    return np.ones(_G_N)

def slsqp(h0, n, T, iters, peak=None):
    global _G_N, _G_T, _G_PEAK
    _G_N = n
    _G_T = T
    _G_PEAK = peak
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_eq, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': iters, 'ftol': 1e-13})
        return project_feasible(res.x, n)
    except:
        return project_feasible(h0.copy(), n)

def subgrad_polish(h, n, steps=200, alpha=None):
    """Projected subgradient step on the max of correlation"""
    best = h.copy()
    best_c5 = compute_c5(best)
    if alpha is None:
        alpha = 0.01
    dx = 2.0 / n
    for s in range(steps):
        corr = np.correlate(best, 1.0 - best, mode='full')
        max_idx = np.argmax(corr)
        # Gradient of corr at the max lag
        lag = max_idx - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            for j in range(n - lag):
                grad[j + lag] += (1.0 - best[j])
                grad[j] -= best[j + lag]
        else:
            lk = -lag
            for j in range(lk, n):
                grad[j - lk] += (1.0 - best[j])
                grad[j] -= best[j - lk]
        # Take step reducing the max lag correlation
        best = project_feasible(best - alpha * grad, n)
        c5 = compute_c5(best)
        if c5 < best_c5:
            best_c5 = c5
            # Decrease alpha slightly
            alpha *= 0.999
        else:
            alpha *= 0.9
            # Revert
            # (keep best from previous iteration)
    return best, best_c5
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(corr) * dx)

def project_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

            grad[j] -= h[j + lag]
    else:
        lk = -lag
        for j in range(lk, n):
            grad[j - lk] += (1.0 - h[j])
            grad[j] -= h[j - lk]
    return grad
```
