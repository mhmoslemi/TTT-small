import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = h < 1.0 - 1e-12 if diff > 0 else h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def thue_morse(n):
    # Generate Thue-Morse sequence of length n
    t = np.zeros(n, dtype=float)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ 1 if (i & (i-1) != i) else t[i >> 1] + 1 - t[i >> 1]
    # Simpler: t[i] = popcount(i) % 2
    t = np.zeros(n, dtype=float)
    for i in range(n):
        t[i] = bin(i).count('1') % 2
    return t

            a = ones[np.argmin(s_o)]
            bb = zeros[np.argmin(s_z)]
        b[a] = 0.0
        b[bb] = 1.0
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-14 and rng.random() < np.exp(-max(0, new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-14:
                best_c5 = new_c5
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            b[a] = 1.0
            b[bb] = 0.0
            no_improve += 1
    return best_b, best_c5

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def multi_temp_slsqp(h, n, start_time, budget_s):
    temps = [100.0, 500.0, 2000.0, 10000.0, 50000.0]
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 30:
            break
        try:
            x_res = slsqp_optimize(h, n, T, 100, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < compute_c5(h):
                h = make_feasible(x_res, n)
        except Exception:
            pass
    return h, compute_c5(h)

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            for cand in [0.0, 1.0, max(0.0, old - 0.02), max(0.0, old - 0.005),
                         min(1.0, old + 0.005), min(1.0, old + 0.02)]:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def perturb_binary(b, n, rng, n_swaps=10):
    b_new = b.copy()
    ones = np.where(b_new == 1)[0]
    zeros = np.where(b_new == 0)[0]
    for _ in range(n_swaps):
        if len(ones) == 0 or len(zeros) == 0:
            break
        a = ones[rng.randint(len(ones))]
        z = zeros[rng.randint(len(zeros))]
        if a == z:
            continue
        b_new[a] = 0.0
        b_new[z] = 1.0
        ones = np.where(b_new == 1)[0]
        zeros = np.where(b_new == 0)[0]
    return b_new

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    deadline = start_time + budget_s

    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_parent = np.full(960, 0.5)

    best_c5 = 1.0
    best_h = None

    candidates = []

    # Candidate 1: parent (binary)
    n0 = len(h_parent)
    b0 = (h_parent >= 0.5).astype(float)
    target0 = n0 // 2
    while np.sum(b0) > target0:
        idx = np.where(b0 == 1)[0]
        b0[idx[-1]] = 0
    while np.sum(b0) < target0:
        idx = np.where(b0 == 0)[0]
        b0[idx[-1]] = 1.0
    candidates.append((b0, n0, "parent"))

    # Candidate 2: Thue-Morse at n=1024
    n_tm = 1024
    b_tm = thue_morse_gen(n_tm)
    candidates.append((b_tm, n_tm, "thue_morse"))

    # Candidate 3: Random
    n_r = 960
    b_r = random_balanced(n_r, rng)
    candidates.append((b_r, n_r, "random"))

    # Candidate 4: Golden ratio
    n_g = 960
    b_g = golden_ratio_pattern(n_g, rng)
    candidates.append((b_g, n_g, "golden"))

    # Evaluate all candidates
    for (b, n, name) in candidates:
        c = compute_c5(b)
        if best_h is None or c < best_c5:
            best_c5 = c
            best_h = b.copy()
        print(f"  {name} (n={n}): C5={c:.6f}")

    # Phase 1: Smart binary SA on best candidate
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        tl = min(remaining - 20, 250)
        bh, bc = smart_binary_sa(best_h.copy(), len(best_h), 200000, 0.004, 0.00001,
                                 rng, start_time, tl)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh.copy()
            print(f"  After SA1: C5={best_c5:.6f}")

    # Phase 2: ILS - perturb and re-optimize
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        for pert_level in [5, 15, 30]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 100:
                break
            bp = perturb_binary(best_h, len(best_h), rng, n_swaps=pert_level)
            tl = min(remaining - 30, 150)
            bh, bc = smart_binary_sa(bp, len(bp), 100000, 0.003, 0.000005,
                                     rng, start_time, tl)
            if bc < best_c5:
                best_c5 = bc
                best_h = bh.copy()
                print(f"  After ILS (pert={pert_level}): C5={best_c5:.6f}")

    # Phase 3: Try other candidates with SA
    remaining = budget_s - (time.time() - start_time)
    for (b, n, name) in candidates:
        if name == "parent":
            continue
        if remaining < 80:
            break
        tl = min(remaining - 20, 100)
        bh, bc = smart_binary_sa(b.copy(), n, 80000, 0.003, 0.00001,
                                 rng, start_time, tl)
        if bc < best_c5:
            best_c5 = bc
            best_h = bh.copy()
            print(f"  After SA on {name}: C5={best_c5:.6f}")

    # Phase 4: SLSQP refinement
    remaining = budget_s - (time.time() - start_time)
    if remaining > 80:
        n_pts = len(best_h)
        best_h, best_c5 = multi_temp_slsqp(best_h, n_pts, start_time, budget_s)
        print(f"  After SLSQP: C5={best_c5:.6f}")

    # Phase 5: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        for n_try in [len(best_h) + 128, 2048]:
            if n_try <= len(best_h):
                continue
            remaining = budget_s - (time.time() - start_time)
            if remaining < 150:
                break
            old_n = len(best_h)
            old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
            new_x = np.linspace(0.0, 2.0, n_try, endpoint=False)
            h_up = np.interp(new_x, old_x, best_h, period=2.0)
            h_up = np.clip(h_up, 0.0, 1.0)
            h_up = make_feasible(h_up, n_try)
            # Binary first
            b_up = (h_up >= 0.5).astype(float)
            tgt = n_try // 2
            while np.sum(b_up) > tgt:
                idx = np.where(b_up == 1)[0]
                b_up[idx[-1]] = 0
            while np.sum(b_up) < tgt:
                idx = np.where(b_up == 0)[0]
                b_up[idx[-1]] = 1.0
            tl = min(remaining - 60, 100)
            bh, bc = smart_binary_sa(b_up, n_try, 50000, 0.003, 0.00001,
                                     rng, start_time, tl)
            if bc < best_c5:
                best_c5 = bc
                best_h = bh.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                best_h, best_c5 = multi_temp_slsqp(best_h, len(best_h), start_time, budget_s)
                print(f"  After res n={len(best_h)}: C5={best_c5:.6f}")

    # Phase 6: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 15, start_time, budget_s)
        print(f"  After coord descent: C5={best_c5:.6f}")

    # Phase 7: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        try:
            x_res = slsqp_optimize(best_h, n_pts, 30000.0, 150, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 8: Final coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 8, start_time, budget_s)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    print(f"  FINAL: C5={best_c5:.6f}, n={len(best_h)}")
    return best_h, best_c5, len(best_h)
