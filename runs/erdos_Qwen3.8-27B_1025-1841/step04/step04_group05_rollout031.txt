def project(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = 0.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Fix any residual
    diff = target - np.sum(h_proj)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h_proj < 1.0 - 1e-12
        else:
            mask = h_proj > 1e-12
        if np.sum(mask) > 0:
            h_proj[mask] += diff / np.sum(mask)
            h_proj = np.clip(h_proj, 0.0, 1.0)
    return h_proj

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    n = len(h)
    dx = 2.0 / n
    R = periodic_corr(h)
    M = np.max(R)
    exp_T = np.exp(T * (R - M))
    p = exp_T / np.sum(exp_T)
    
    # Gradient
    Hp = np.fft.fft(h)
    Gp = np.fft.fft(1.0 - h)
    pp = np.fft.fft(p)
    term1 = np.fft.ifft(Gp * np.conj(pp)).real
    term2 = np.fft.ifft(Hp * pp).real
    grad = dx * (term1 - term2)
    
    return grad

def pgd_optimize(h_init, n, T, max_iter, lr, momentum=0.0):
    """PGD with optional momentum (Nesterov)."""
    h = h_init.copy()
    v = np.zeros(n)  # velocity for momentum
    best_obj = float('inf')
    best_h = h.copy()
    
    for it in range(max_iter):
        if momentum > 0:
            # Nesterov lookahead
            h_look = h - momentum * v
            grad = pgd_step(h_look, T, lr)
        else:
            grad = pgd_step(h, T, lr)
        
        v = momentum * v + lr * grad
        h_new = h - v
        h_new = project_simplex_box(h_new, n)
        
        # Evaluate
        R = periodic_corr(h_new)
        M = np.max(R)
        exp_T = np.exp(T * (R - M))
        obj = (M + np.log(np.sum(exp_T)) / T) * (2.0 / n)
        
        if obj < best_obj:
            best_obj = obj
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h

def slsqp_optimize(h_start, n, T, max_iter):

    dx = 2.0 / n

        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        log_sum = np.log(np.sum(np.exp(T * shifted)))
        return (mc + log_sum / T) * dx
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    res = minimize(obj, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

The gradient is: grad = dx * (term1 - term2)

term1 and term2 are each a convolution of p (which sums to 1) with a bounded function (≤ 1).
So |term1| ≤ 1, |term2| ≤ 1, and |grad| ≤ 2*dx = 4/n.

For n=2048: |grad| ≤ 4/2048 ≈ 0.002

The Lipschitz constant L of the gradient:
The gradient is a linear operator applied to h (plus a constant). The operator is:
A(h) = conv(p_rev, h) + conv(p, h)

The operator norm of A (as a map from ℓ∞ to ℓ∞) is at most ||p_rev||_1 + ||p||_1 = 2.
So ||A(h1) - A(h2)||_∞ ≤ 2 * ||h1-h2||_∞
And the gradient maps as: grad(h) = dx * (const - A(h))
So ||grad(h1) - grad(h2)||_∞ ≤ dx * 2 * ||h1-h2||_∞ = 2*dx * ||h1-h2||_∞

In the ℓ∞ norm: L = 2*dx = 4/n

For projected gradient, the step size should be ≤ 1/L = n/4.
For n=2048: step ≤ 512.
