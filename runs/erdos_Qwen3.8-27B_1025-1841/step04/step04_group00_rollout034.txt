def guided_swap_vectorized(cur, n, s):
    """Find best guided swap for shift s."""
    # For shift s, compute delta for all possible i (removing 1) 
    # and all possible j (adding 1)
    
    # Remove 1 at position i: delta = -(1-cur[i-s]) - cur[i+s] (where valid)
    # Add 1 at position j: delta = +(1-cur[j-s]) - cur[j+s] (where valid)
    
    # Vectorized computation
    remove_delta = np.zeros(n)
    add_delta = np.zeros(n)
    
    # For remove_delta[i]: -(1-cur[i-s]) if 0<=i-s<n, minus cur[i+s] if 0<=i+s<n
    valid_remove_left = (0 <= np.arange(n) - s) & (np.arange(n) - s < n)
    valid_remove_right = (0 <= np.arange(n) + s) & (np.arange(n) + s < n)
    remove_delta[valid_remove_left] -= (1 - cur[np.arange(n)[valid_remove_left] - s])
    remove_delta[valid_remove_right] -= cur[np.arange(n)[valid_remove_right] + s]
    
    # For add_delta[j]: +(1-cur[j-s]) if 0<=j-s<n, minus cur[j+s] if 0<=j+s<n
    valid_add_left = (0 <= np.arange(n) - s) & (np.arange(n) - s < n)
    valid_add_right = (0 <= np.arange(n) + s) & (np.arange(n) + s < n)
    add_delta[valid_add_left] += (1 - cur[np.arange(n)[valid_add_left] - s])
    add_delta[valid_add_right] -= cur[np.arange(n)[valid_add_right] + s]
    
    # Only consider positions where cur is 1 for remove, 0 for add
    remove_delta[cur != 1] = 0  # Can only remove 1s
    add_delta[cur != 0] = 0     # Can only add 1s (i.e., flip 0s)

    b = (h >= 0.5).astype(float)

    if count > target:
        one_idx = np.where(b == 1)[0]
        order = np.argsort(h[one_idx])
        for i in range(count - target):
            b[one_idx[order[i]]] = 0
    elif count < target:
        zero_idx = np.where(b == 0)[0]
        order = np.argsort(-h[zero_idx])
        for i in range(target - count):
            b[zero_idx[order[i]]] = 1
    return b

def guided_pair(cur, n, s):
    """Find best (i,j) swap to reduce correlation at shift s.
    Returns (i, j) where cur[i]=1, cur[j]=0, or (None, None) if no valid pair."""
    idx = np.arange(n)
    
    # Delta in correlation at shift s if we remove 1 at position p
    # (i.e., set cur[p] from 1 to 0):
    #   removes term cur[p]*(1-cur[p-s]) = 1*(1-cur[p-s]) if p-s valid
    #   removes term cur[p+s]*(1-cur[p]) = cur[p+s]*1 if p+s valid
    #   So delta_remove[p] = -(1-cur[p-s]) - cur[p+s] (sum of valid terms)
    
    # Delta in correlation at shift s if we add 1 at position q
    # (i.e., set cur[q] from 0 to 1):
    #   adds term cur[q]*(1-cur[q-s]) = 1*(1-cur[q-s]) if q-s valid
    #   removes term cur[q+s]*(1-cur[q]) = cur[q+s]*1 if q+s valid
    #   So delta_add[q] = (1-cur[q-s]) - cur[q+s] (sum of valid terms)
    
    vl = (idx - s >= 0) & (idx - s < n)
    vr = (idx + s >= 0) & (idx + s < n)
    
    d_remove = np.zeros(n)
    d_add = np.zeros(n)
    
    if np.any(vl):
        d_remove[vl] -= (1.0 - cur[idx[vl] - s])
        d_add[vl] += (1.0 - cur[idx[vl] - s])
    if np.any(vr):
        d_remove[vr] -= cur[idx[vr] + s]
        d_add[vr] -= cur[idx[vr] + s]
    
    # Only valid for positions with correct value
    d_remove[cur != 1] = 1e10
    d_add[cur != 0] = 1e10
    
    i = int(np.argmin(d_remove))
    j = int(np.argmin(d_add))
    
    if d_remove[i] >= 1e9 or d_add[j] >= 1e9:
        return None, None
    
    return i, j

    cur_c5 = float(np.max(cur_corr) * dx)

    best_c5 = cur_c5

                zeros = np.where(cur == 0)[0]
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            ones = np.where(cur == 1)[0]
            zeros = np.where(cur == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        
        if i == j:
            continue
        
        # Apply swap
        cur[i] = 0  # was 1
        cur[j] = 1  # was 0

        new_c5 = float(np.max(new_corr) * dx)
