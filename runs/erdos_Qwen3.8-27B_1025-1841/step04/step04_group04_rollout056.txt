```python
def bottleneck_grad_vec(h, shifts):
    n = len(h)
    grad = np.zeros(n)
    for p in shifts:
        # grad += (1-h[j+p]) where valid - h[j-p] where valid
        lo1 = max(0, -p)
        hi1 = min(n, n - p)
        if lo1 < hi1:
            grad[lo1:hi1] += (1.0 - h[lo1+p:hi1+p])
        lo2 = max(0, p)
        hi2 = min(n, n + p)
        if lo2 < hi2:
            grad[lo2:hi2] -= h[lo2-p:hi2-p]
    return grad
```

This is O(n * |shifts|) which is fast.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(np.float64), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # f = (1/T) * log(sum exp(T * corr)) * dx
    mc = np.max(corr)
    exp_T = np.exp(T * (corr - mc))
    S = np.sum(exp_T)
    w = exp_T / S  # softmax weights
    f = (1.0 / T) * np.log(S) * dx + mc * dx  # = (mc + log(S)/T) * dx
    
    # Gradient: dx * sum_m w[m] * d(corr[m])/dh
    # d(corr[m])/dh[j] = (1-h[j+p])[0<=j+p<n] - h[j-p][0<=j-p<n], p=m-(n-1)
    grad = np.zeros(n)
    for m in range(len(corr)):
        wm = w[m]
        if wm < 1e-12:
            continue
        p = m - (n - 1)
        lo = max(0, -p)
        hi = min(n, n - p)
        if lo < hi:
            grad[lo:hi] += wm * (1.0 - h[lo + p:hi + p])
        lo2 = max(0, p)
        hi2 = min(n, n + p)
        if lo2 < hi2:
            grad[lo2:hi2] -= wm * h[lo2 - p:hi2 - p]
    grad *= dx
    return f, grad
```

```python
def corr_update(h, delta_j, old_val, new_val, corr):
    """Update correlation when h[j] changes from old_val to new_val."""
    n = len(h)
    delta = new_val - old_val
    j = delta_j  # the index that changed
    # h[j] = new_val now
    # Update corr: for each shift m, corr[m] changes by
    # delta * (1-h[j+p]) [0<=j+p<n] - h[j-p] * delta [0<=j-p<n]
    # where p = m-(n-1)
    # Equivalent: the shift p contributes to corr at index m = p + (n-1)
    # Correlation at shift p: sum_i h[i]*(1-h[i+p])
    # Change: delta*(1-h[j+p])[0<=j+p<n] + sum_i h[i]*(-(delta))[i+p=j]
    #        = delta*(1-h[j+p])[0<=j+p<n] - delta*h[j-p][0<=j-p<n]
    
    # For all valid p:
    # We need to update corr[p + n - 1] for all p where either j+p or j-p is in [0,n)
    # p ranges from -(n-1) to n-1
    # j+p in [0,n) → p in [-j, n-1-j]
    # j-p in [0,n) → p in [j-(n-1), j]
    
    lo = max(-(n-1), -j)
    hi = min(n-1, n-1-j)
    # First term: delta * (1-h[j+p]) for this range
    # p = lo..hi, j+p = j+lo..j+hi (all in [0,n))
    m_lo = lo + n - 1
    m_hi = hi + n - 1
    corr[m_lo:m_hi+1] += delta * (1.0 - h[j+lo:j+hi+1])
    
    lo2 = max(-(n-1), j-(n-1))
    hi2 = min(n-1, j)
    # Second term: -delta * h[j-p] for this range
    # j-p ranges from j-hi2 to j-lo2 (decreasing as p increases)
    # When p=lo2: j-p = j-lo2; when p=hi2: j-p = j-hi2
    # j-p in [0, n) for all in this range
    m_lo2 = lo2 + n - 1
    m_hi2 = hi2 + n - 1
    # h[j-p] for p=lo2..hi2 is h[j-hi2..j-lo2] (reversed!)
    # Actually h[j-p] where p goes lo2 to hi2: values are h[j-lo2], h[j-lo2-1], ..., h[j-hi2]
    # = h[j-hi2 : j-lo2+1] in reverse
    corr[m_lo2:m_hi2+1] -= delta * h[j-hi2:j-lo2+1][::-1]
    
    return corr
```

```python
def smooth_obj_and_grad(h, T):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exp_T = np.exp(T * (corr - mc))
    S = np.sum(exp_T)
    w = exp_T / S

    grad = np.zeros(n)

        hi2 = min(n, n + p)

        corr_s = np.correlate(best_h, 1.0 - best_h, mode='full')
        dx_s = 2.0 / n_s
        bc5 = float(np.max(corr_s) * dx_s)
        for j in range(n_s):
            if remaining() < 5:
                break
            old_val = best_h[j]
            k = pick_partner(best_h, j, corr_s, n_s)
            old_k = best_h[k]
            for bv in (0.0, 1.0):
                delta = bv - old_val
                if abs(delta) < 1e-15:
                    continue
                trial_k = old_k - delta
                if trial_k < -1e-12 or trial_k > 1.0 + 1e-12:
                    continue
                trial_k = max(0.0, min(1.0, trial_k))
                trial_corr = corr_s.copy()
                pair_incr(trial_corr, best_h, j, bv, k, trial_k)
                tc = float(np.max(trial_corr) * dx_s)
                if tc < bc5 - 1e-14:
                    best_h[j] = bv
                    best_h[k] = trial_k
                    corr_s = trial_corr.copy()
                    bc5 = tc
        best_h = make_feasible(best_h, n_s)
        best_c5 = min(best_c5, compute_c5(best_h))

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
```
