import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_vec(h):
    """Return (c5, corr_vec, active_lags)"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = float(np.max(corr) * dx)
    active = np.where(corr > 0.9 * np.max(corr))[0]
    return c5, corr, active

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        # Need more ones
        order = np.argsort(-h)
        for i in range(diff):
            if b[order[i]] == 0:
                b[order[i]] = 1
    elif diff < 0:
        # Need fewer ones
        order = np.argsort(h)
        for i in range(-diff):
            if b[order[i]] == 1:

# SLSQP with log-sum-exp
_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    if T > 1e6:
        return mc * dx
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_pass(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def multi_pass_slsqp(h, n, T_schedule, iters_per_pass, ftol, start_time, budget_s):
    """Run SLSQP with progressively increasing T."""
    for T in T_schedule:

        if remaining < 30:

        x = make_feasible(x, n)
        c5_new = compute_c5(x)
        c5_old = compute_c5(h)
        if c5_new <= c5_old:
            h = x
    return h

        if len(ones) == 0 or len(zeros) == 0:
            break
        
        best_delta = 0
        best_pair = None
        # Sample a subset of pairs for efficiency
        n_checks = min(len(ones) * len(zeros), 20000)
        checked = set()
        attempts = 0
        while len(checked) < n_checks and attempts < n_checks * 3:
            attempts += 1
            i = ones[rng.randint(len(ones))] if hasattr(rng, 'randint') else ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]

                continue
            checked.add((i, j))
            b[i] = 0
            b[j] = 1
            c = compute_c5(b)
            b[i] = 1
            b[j] = 0
            delta = c - curr_c5
            if delta < best_delta - 1e-15:

                best_pair = (i, j)

                best_c5 = curr_c5
                best_b = b.copy()
            improved = True

Actually, one key insight: for the continuous relaxation, the problem is:
minimize max_k Σ_i h[i](1-h[i+k]) * dx

This can be rewritten as:
minimize max_k [Σ_i h[i] * (valid) - Σ_i h[i]h[i+k]] * dx

The first term is linear in h, the second is bilinear. For a fixed k, this is a quadratic in h.

What if I use a **block coordinate descent** approach? Fix all variables except a subset, optimize that subset, repeat. For example, fix all even-indexed variables, optimize odd-indexed ones (which is a quadratic problem), then vice versa.

This is still a quadratic in the odd variables, and the max of quadratics is non-convex. So this doesn't directly help.

One more idea: **use the connection to autocorrelation**. For binary h, define f = 2h - 1 (so f ∈ {-1, 1}). Then:
h[i](1-h[i+k]) = (f[i]+1)/2 * (1-f[i+k])/2 = (1 - f[i]f[i+k] - f[i] + f[i+k])/4

So the overlap at shift k is:
Σ_i h[i](1-h[i+k]) = (1/4) Σ_i (1 - f[i]f[i+k] - f[i] + f[i+k])
= (1/4)(N_k - Σ_i f[i]f[i+k] - Σ_i f[i] + Σ_i f[i+k])

where N_k is the number of valid pairs. The Σ_i f[i] and Σ_i f[i+k] terms are just sums of ±1, bounded by n. The autocorrelation term Σ_i f[i]f[i+k] is the key.

For a "balanced" sequence (equal +1 and -1), Σ_i f[i] = 0, so:
overlap_k = (1/4)(N_k - autocorr_k)

where autocorr_k = Σ_i f[i]f[i+k].

To minimize the max overlap, we want to maximize the min autocorr_k (over the range of valid k). This is a different way to look at it but doesn't immediately give an algorithm.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    diff = target - int(np.sum(b))
    if diff > 0:
        order = np.argsort(-h)
        cnt = 0
        for idx in order:
            if cnt >= diff:
                break
            if b[idx] == 0:
                b[idx] = 1
                cnt += 1
    elif diff < 0:
        order = np.argsort(h)
        cnt = 0
        for idx in order:
            if cnt >= -diff:
                break
            if b[idx] == 1:
                b[idx] = 0
                cnt += 1
    return b

def best_binary_rounding(h, n):
    best_b = None
    best_c = float('inf')
    for thresh in [0.3, 0.35, 0.4, 0.45, 0.5, 0.55, 0.6, 0.65, 0.7]:
        b = (h >= thresh).astype(float)
        target = n // 2
        diff = target - int(np.sum(b))
        if diff > 0:
            order = np.argsort(-h)
            for idx in order:
                if diff <= 0:
                    break
                if b[idx] == 0:
                    b[idx] = 1
                    diff -= 1
        elif diff < 0:
            order = np.argsort(h)
            for idx in order:
                if diff >= 0:
                    break
                if b[idx] == 1:
                    b[idx] = 0
                    diff += 1
        c = compute_c5(b)
        if c < best_c:
            best_c = c
            best_b = b.copy()
    return best_b, best_c

_SL_N = 0
_SL_T = 0.0

def obj_slsqp(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    if T > 1e6:
        return mc * dx
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_pass(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def multi_slsqp(h, n, T_list, iters, ftol, start_time, budget_s):
    for T in T_list:
        if time.time() - start_time > budget_s - 30:
            break
        x = slsqp_pass(h, n, T, iters, ftol)
        x = make_feasible(x, n)
        c5_new = compute_c5(x)
        c5_old = compute_c5(h)
        if c5_new <= c5_old + 1e-14:
            h = x
    return h

def targeted_sa(h, n, max_iters, T0, T1, rng, start_time, time_limit):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    worst_lag = 0
    update_every = 30

    for it in range(max_iters):
        if it % 1000 == 0 and time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac

        if it % update_every == 0:
            corr = np.correlate(b, 1.0 - b, mode='full')
            worst_lag = int(np.argmax(corr)) - (n - 1)

        if rng.random() < 0.35 and worst_lag != 0:
            lag = worst_lag
            valid = []
            if lag > 0:
                poss = np.where(b[:n - lag] == 1)[0]
                for i in poss:
                    if b[i + lag] == 0:
                        valid.append((i, i + lag))
            else:
                poss = np.where(b[-lag:] == 0)[0]
                for idx in poss:
                    i = idx - lag
                    if b[i] == 1:
                        valid.append((i, idx))
            if len(valid) > 0:
                i, j = valid[rng.randint(len(valid))]
            else:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]

        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
        else:
            b[i] = 1
            b[j] = 0

    return best_b, best_c5

def block_sa(h, n, max_iters, T0, T1, rng, start_time, time_limit):
    b = to_binary(h, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()

    for it in range(max_iters):
        if it % 1000 == 0 and time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        bs = rng.randint(2, 7)
        start = rng.randint(0, n - bs)
        seg = b[start:start + bs]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi), rng.randint(1, 4))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0

    return best_b, best_c5

def upsample_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return make_feasible(h.copy(), n_new)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return make_feasible(h_new, n_new)

def downsample_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return make_feasible(h.copy(), n_new)
    h_new = np.zeros(n_new)
    for i in range(n_new):
        lo = i * n_old // n_new
        hi = (i + 1) * n_old // n_new
        h_new[i] = np.mean(h[lo:hi])
    return make_feasible(h_new, n_new)
