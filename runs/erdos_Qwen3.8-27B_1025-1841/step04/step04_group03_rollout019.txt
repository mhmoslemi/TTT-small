def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            int_mask = (h == h.astype(int))
            mask = ~int_mask
            cnt = np.sum(mask)
            if cnt == 0:
                break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

        total_iters += 1
        # Cool temperature
        T = max(T_min, T_max * (T_min / T_max) ** (total_iters / 50000.0))

        mc = float(np.max(corr))

        top_idx = np.argsort(corr)[::-1][:top_k]

    if remaining > 180:
        tm = thue_morse_init(n0, rng)
        deadline_c = start_time + min(remaining - 90, 200)
        b, c5_b = focused_sa(tm, n0, deadline_c, rng, T0=0.012, T1=0.000003)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                h_sg, c5_sg = subgrad_opt(best_h.copy(), n0, start_time + min(remaining - 20, 80), top_k=8)
                if c5_sg < best_c5:
                    best_c5 = c5_sg
                    best_h = h_sg.copy()

    # Phase D: Random start + SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 160:
        rb = random_balanced_init(n0, rng)
        deadline_d = start_time + min(remaining - 80, 160)
        b, c5_b = focused_sa(rb, n0, deadline_d, rng, T0=0.015, T1=0.00001)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase E: Higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        n1 = n0 + 480
        h_up = interp_h(best_h, n0, n1)
        deadline_e1 = start_time + min(remaining - 150, 170)
        b, c5_b = focused_sa(h_up, n1, deadline_e1, rng, T0=0.006, T1=0.000004)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 80:
            h_sg, c5_sg = subgrad_opt(best_h.copy(), n1, start_time + min(remaining - 30, 130), top_k=12)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()

    # Phase F: SA on best (continued)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        n_pts = len(best_h)
        deadline_f = start_time + min(remaining - 40, 130)
        b, c5_b = focused_sa(best_h, n_pts, deadline_f, rng, T0=0.004, T1=0.000001)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase G: Final subgradient + local repair
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        h_sg, c5_sg = subgrad_opt(best_h.copy(), n_pts, start_time + min(remaining - 15, 50), top_k=12, T_max=20.0)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        h_lr, c5_lr = local_repair(best_h.copy(), n_pts, start_time + remaining - 5, n_trials=300)
        if c5_lr < best_c5:
            best_c5 = c5_lr
            best_h = h_lr.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
