import numpy as np
from scipy.optimize import minimize
import time

_SG_N = 0
_SG_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_slsqp(x):
    n = _SG_N
    T = _SG_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SG_N, _SG_T
    _SG_N = n
    _SG_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def multi_temp_slsqp(h, n, start_time, budget_s):
    temps = [100.0, 400.0, 1500.0, 6000.0, 25000.0, 100000.0]

def subgradient_optimize(h, n, start_time, time_limit, max_iters=200000):
    dx = 2.0 / n
    curr_c5 = compute_c5(h)
    best_c5 = curr_c5
    best_h = h.copy()
    alpha = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        # Subgradient of corr[k_star] wrt h[m]:
        # corr[k] = sum_j h[j+k]*(1-h[j]), valid 0<=j<n, 0<=j+k<n
        grad = np.zeros(n)
        # h[m] as h[j+k]: j=m-k, need 0<=m-k<n, 0<=m<n
        lo1 = max(0, k_star)
        hi1 = min(n, n + k_star)
        if lo1 < hi1:
            m_idx = np.arange(lo1, hi1)
            grad[m_idx] += (1.0 - h[m_idx - k_star])
        # h[m] as h[j]: j=m, need 0<=m<n, 0<=m+k_star<n
        lo2 = max(0, -k_star)
        hi2 = min(n, n - k_star)
        if lo2 < hi2:
            m_idx2 = np.arange(lo2, hi2)
            grad[m_idx2] -= h[m_idx2 + k_star]
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < curr_c5 - 1e-15:
            h = h_new
            curr_c5 = new_c5
            stall = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            stall += 1
            if stall > 50:
                alpha *= 0.9
                stall = 0
            if alpha < 1e-8:
                break
    return best_h, best_c5

        for j in range(n):

            if remaining < 3:

            old = h[j]
            best_c = best_c5
            best_val = old
            for s in steps:

                if cand <= 0.0:

                if abs(cand - old) < 1e-10:

            x_res = slsqp_optimize(best_h, n_pts, 200000.0, 300, 1e-15)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 9: Coordinate descent polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 25:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 15, start_time, budget_s)

    # Phase 10: Final micro SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15:
        n_pts = len(best_h)
        try:
            x_res = slsqp_optimize(best_h, n_pts, 500000.0, 100, 1e-16)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
