def weighted_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    weights = np.exp(T * (corr - np.max(corr)))
    weights /= np.sum(weights)

```python
def soft_max_grad(h, n, T):
    """Gradient of the log-sum-exp of correlations."""
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    # Softmax weights
    shifted = T * (corr - mc)
    # Clip to avoid underflow
    shifted = np.clip(shifted, -30, 30)
    weights = np.exp(shifted)
    weights /= np.sum(weights)
    
    # Gradient = weighted sum of individual lag gradients
    grad = np.zeros(n)
    # Only use lags with significant weight
    significant = weights > 1e-6
    for m in np.where(significant)[0]:
        shift = m - n + 1
        if shift == 0:
            grad += weights[m] * (1.0 - 2.0 * h)
        else:
            grad += weights[m] * grad_at_lag(h, n, shift)
    return grad
```

This is the gradient of log(Σ exp(T*corr_k)) / T * dx, which is a smooth approximation of max(corr_k) * dx.

As T → ∞, this approaches the subgradient of the max.

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def grad_at_lag(h, n, shift):
    grad = np.zeros(n)
    if shift >= 0:
        if shift < n:
            idx = np.arange(shift, n)
            grad[idx] += (1.0 - h[:n - shift])
            idx2 = np.arange(0, n - shift)
            grad[idx2] -= h[shift:]
    else:
        s = -shift
        if s < n:
            idx = np.arange(0, n - s)
            grad[idx] += (1.0 - h[s:])
            idx2 = np.arange(s, n)
            grad[idx2] -= h[:n - s]
    return grad

def softmax_grad(h, n, T):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -50, 50)
    weights = np.exp(shifted)
    wsum = np.sum(weights)
    weights /= wsum
    
    grad = np.zeros(n)
    sig = weights > 1e-7
    idxs = np.where(sig)[0]
    for m in idxs:
        shift = m - n + 1
        w = weights[m]
        if shift == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad += w * grad_at_lag(h, n, shift)
    return grad, mc * (2.0 / n)

def slsqp_refine(h, n, T, max_iter=200):
    _N = n
    _T = T
    dx = 2.0 / n
    
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        mc = np.max(corr)
        shifted = _T * (corr - mc)
        shifted = np.clip(shifted, -50, 50)
        return (mc + np.log(np.sum(np.exp(shifted))) / _T) * dx
    
    def cons_fun(v):
        return np.sum(v) - n / 2.0
    
    def cons_jac(v):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-12})
        return make_feasible(res.x, n)
    except Exception:
        return h
