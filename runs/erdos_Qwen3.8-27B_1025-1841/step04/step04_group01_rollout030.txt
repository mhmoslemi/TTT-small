- If lag ≥ 0: ∂corr_i/∂h[a] = (1-h[a+lag]) - h[a-lag] (with appropriate bounds)
- If lag < 0: similar

So for a fixed a, there are O(n) non-zero entries. Computing the full gradient is O(n²).

With the gradient, SLSQP would be O(n²) per iteration (evaluate gradient) + O(n³) for the Hessian update (BFGS). With n=960, that's about 920K per iteration for the gradient, which is fast.

```python
def smooth_max_obj(x, T, top_idx=None):
    """Compute smooth max of correlation values."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_idx is not None:
        vals = corr[top_idx]
    else:
        vals = corr
    mc = np.max(vals)
    shifted = T * (vals - mc)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (mc + log_sum / T) * dx

def smooth_max_grad(x, T, top_idx=None):
    """Compute gradient of smooth max objective."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_idx is not None:
        vals = corr[top_idx]
    else:
        vals = corr
        top_idx = np.arange(len(corr))
    
    mc = np.max(vals)
    shifted = T * (vals - mc)
    shifted = np.clip(shifted, -500, 500)
    exps = np.exp(shifted)
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for ii in range(len(top_idx)):
        idx = top_idx[ii]
        lag = idx - (n - 1)
        w = weights[ii]
        if lag >= 0:
            # First term: a in [0, n-1-lag]
            if n - 1 - lag >= 0:
                grad[:n-lag] += w * (1.0 - x[lag:n])
            # Second term: a in [lag, n-1]  
            if lag <= n - 1:
                grad[lag:] -= w * x[:n-lag]
        else:
            L = -lag
            # First term: a in [L, n-1]
            if L <= n - 1:
                grad[L:] += w * (1.0 - x[:n-L])
            # Second term: a in [0, n-1-L]
            if n - 1 - L >= 0:
                grad[:n-L] -= w * x[L:]
    grad *= dx
    return grad
```
