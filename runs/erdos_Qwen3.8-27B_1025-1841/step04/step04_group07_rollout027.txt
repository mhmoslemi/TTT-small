def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_smooth(h0, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h):
    n = len(h)
    h_b = np.zeros(n)
    target = n // 2
    order = np.argsort(-h)
    h_b[order[:target]] = 1.0
    return h_b

def soft_max_obj(corr, T):
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_smooth(h0, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        sm = soft_max_obj(corr, T)
        return sm * dx + mu * (x.sum() - n / 2.0) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-11})
    return make_feasible(res.x, n)

def binary_sa(h_bin, n, rng, time_limit, init_temp=0.03, cool=0.9985,
              tries_per_round=200, block_prob=0.15, max_block=5):
    t_start = time.time()
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    best_c5 = cur_c5
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    temp = init_temp
    stall = 0
    
    while time.time() - t_start < time_limit:
        improved = False
        for _ in range(tries_per_round):
            if not ones or not zeros:
                break
            if rng.random() < block_prob and n > 50:
                bs = rng.randint(1, min(max_block, n // 20)) + 1
                found_oi = -1
                for _ in range(8):
                    s = rng.randint(n - bs + 1)
                    if np.all(h[s:s+bs] == 1.0):
                        found_oi = s
                        break
                if found_oi < 0:
                    continue
                found_oj = -1
                for _ in range(8):
                    s = rng.randint(n - bs + 1)
                    e = s + bs
                    if (e <= found_oi or s >= found_oi + bs) and np.all(h[s:e] == 0.0):
                        found_oj = s
                        break
                if found_oj < 0:
                    continue
                h[found_oi:found_oi+bs] = 0.0

def coarse_to_fine(initial, target_n, rng, time_limit):

            if c_b < best_c5:
                best_c5 = c_b
                best_h = h_b.copy()
        
        # Hill climb
        if rem() > 40:
            h_hc, c_hc = binary_hill_climb(best_h, n2, rng, time_limit=min(rem()-20, 60))
            if c_hc < best_c5:
                best_c5 = c_hc
                best_h = h_hc.copy()
    
    # Phase 4: Multi-start perturbations
    if rem() > 120:
        n_ms = best_n
        for restart in range(6):
            if rem() < 80:
                break
            h_r = best_h.astype(float).copy()
            n_flips = max(10, n_ms // 8)
            flip_idx = rng.choice(n_ms, size=n_flips, replace=False)
            h_r[flip_idx] = 1.0 - h_r[flip_idx]
            h_r = make_feasible(h_r, n_ms)
            h_b = to_binary(h_r)
            t_sa = min(rem() - 50, 100)
            h_b, c_b = binary_sa(h_b, n_ms, rng, time_limit=t_sa,
                                 init_temp=0.06, cool=0.995, tries_per_round=250)
            if c_b < best_c5:
                best_c5 = c_b
                best_h = h_b.copy()
    
    # Phase 5: Intensive refinement on best
    if rem() > 30:
        h_b = to_binary(best_h)
        t_ref = max(rem() - 5, 15)
        h_b, c_b = binary_sa(h_b, best_n, rng, time_limit=t_ref,
                             init_temp=0.008, cool=0.9995, tries_per_round=400,
                             block_prob=0.1, max_block=4)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = h_b.copy()
    
    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
