    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def to_binary(h, n):

    target = n // 2

        ones = np.where(b == 1)[0]
        if len(ones) == 0:

        idx = ones[np.argmin(h[ones])]
        b[idx] = 0

        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
