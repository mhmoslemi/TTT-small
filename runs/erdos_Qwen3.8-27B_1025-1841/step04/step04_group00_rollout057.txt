def compute_lag_gradient(h, s):
    """Gradient of correlation at lag s with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    # First term: (1-h[i-s]) for valid i-s
    if s >= 0:
        # i-s in [0,n) => i in [s, s+n), intersect [0,n) => i in [s, n)
        grad[s:] += (1.0 - h[:n-s])
    else:
        # i-s in [0,n) => i in [s, s+n), s is negative, so i in [0, n+s)
        grad[:n+s] += (1.0 - h[-s:])

    if s >= 0:
        # i+s in [0,n) => i in [-s, n-s), intersect [0,n) => i in [0, n-s)
        grad[:n-s] -= h[s:]
    else:
        # i+s in [0,n) => i in [-s, n-s), -s > 0, so i in [-s, n)
        grad[-s:] -= h[:n+s]
    
    return grad

def lag_gradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        if n - s > 0:
            grad[s:] += (1.0 - h[:n-s])
            grad[:n-s] -= h[s:]
    else:
        ns = -s
        if n - ns > 0:
            grad[:n-ns] += (1.0 - h[ns:])
            grad[ns:] -= h[:n-ns]
    return grad

def smooth_max_obj_grad(h, n, T, top_m):
    dx = 2.0 / n
    corr = fast_corr(h)
    vals = corr * dx
    top_idx = np.argsort(vals)[-top_m:]

        s = int(idx - (n - 1))
        grad += w * lag_gradient(h, s)
    grad *= dx

        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    if s >= 0:
        ns = n - s
        if ns > 0:
            grad[s:] += (1.0 - h[:ns])
            grad[:ns] -= h[s:]
    else:
        ns = -s
        rem = n - ns
        if rem > 0:
            grad[:rem] += (1.0 - h[ns:])
            grad[ns:] -= h[:rem]
    return grad

def smooth_max_obj_grad(h, n, T, top_m):
    dx = 2.0 / n
    corr = fast_corr(h)
    vals = corr * dx
    top_idx = np.argsort(vals)[-top_m:]
    top_vals = vals[top_idx]
    vmax = np.max(top_vals)
    exps = np.exp(T * (top_vals - vmax))
    s_exp = np.sum(exps)
    weights = exps / s_exp
    obj = vmax + np.log(s_exp) / T
    grad = np.zeros(n)
    for w, idx in zip(weights, top_idx):
        s = int(idx - (n - 1))
        grad += w * lag_gradient(h, s)
    grad *= dx
    return obj, grad

def lbfgsb_smooth(h, n, T, top_m, maxiter, start_time, budget_s, penalty=5000.0):
    def obj(x):
        o, _ = smooth_max_obj_grad(x, n, T, top_m)
        return o + penalty * (np.sum(x) - n / 2.0) ** 2

    def grad(x):
        _, g = smooth_max_obj_grad(x, n, T, top_m)
        g = g + 2.0 * penalty * (np.sum(x) - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 20})
        res.x = project_feasible(res.x, n)
        c5 = compute_c5(res.x)
        if c5 < compute_c5(h):
            return res.x, c5
    except Exception:
        pass
    return h, compute_c5(h)

def slsqp_smooth(h, n, T, top_m, maxiter, start_time, budget_s):
    def obj(x):
        o, _ = smooth_max_obj_grad(x, n, T, top_m)
        return o

    def jac(x):
        _, g = smooth_max_obj_grad(x, n, T, top_m)
        return g

    def con_f(x):
        return np.sum(x) - n / 2.0

    def con_j(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    try:
        res = minimize(obj, h, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        res.x = project_feasible(res.x, n)
        c5 = compute_c5(res.x)
        if c5 < compute_c5(h):
            return res.x, c5
    except Exception:
        pass
    return h, compute_c5(h)

def pgd_fft(h, n, budget_s, start_time, top_m=8, T_sm=80.0):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    step_size = 0.005
    momentum = 0.9
    no_improve = 0
    it = 0

    while time.time() - start_time < budget_s - 5:
        it += 1
        corr = fast_corr(h) * dx
        c5_cur = float(np.max(corr))
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 80:
                step_size *= 0.5
                if step_size < 1e-9:
                    break
                no_improve = 0

        top_idx = np.argsort(corr)[-top_m:]
        tv = corr[top_idx]
        vmax = np.max(tv)
        exps = np.exp(T_sm * (tv - vmax))
        weights = exps / np.sum(exps)

        grad = np.zeros(n)
        for w, idx in zip(weights, top_idx):
            s = int(idx - (n - 1))
            grad += w * lag_gradient(h, s)
        grad *= dx

        y = h - momentum * velocity
        y = np.clip(y, 0, 1)
        tcorr = fast_corr(y) * dx
        ttop = np.argsort(tcorr)[-top_m:]
        tvg = tcorr[ttop]
        tvmax = np.max(tvg)
        texp = np.exp(T_sm * (tvg - tvmax))
        tw = texp / np.sum(texp)
        tgrad = np.zeros(n)
        for w, idx in zip(tw, ttop):
            s = int(idx - (n - 1))
            tgrad += w * lag_gradient(y, s)
        tgrad *= dx

        velocity = momentum * velocity - step_size * tgrad
        new_h = project_feasible(h + velocity, n)
        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur + 1e-16:
            h = new_h
        else:
            new_h2 = project_feasible(h - step_size * grad, n)
            c5_new2 = compute_c5(new_h2)
            if c5_new2 <= c5_cur:
                h = new_h2
                velocity *= 0.3
            else:
                step_size *= 0.7
                velocity *= 0.3
            if step_size < 1e-10:
                break
        if it % 100 == 0 and step_size < 0.003:
            step_size *= 1.5
        if time.time() - start_time > budget_s - 3:
            break

    h = project_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h, best_c5 = h, c5_h
    return best_h, best_c5

def subgrad_max(h, n, budget_s, start_time, top_m=5):
    """Subgradient method on the exact max."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step_size = 0.002
    no_improve = 0

    while time.time() - start_time < budget_s - 3:
        corr = fast_corr(h) * dx
        c5_cur = float(np.max(corr))
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 60:
                step_size *= 0.6
                if step_size < 1e-10:
                    break
                no_improve = 0

        top_val = np.max(corr)
        top_mask = corr >= top_val - 1e-12
        top_indices = np.where(top_mask)[0]
        if len(top_indices) > top_m:
            top_indices = np.argsort(corr)[-top_m:]

        grad = np.zeros(n)
        for idx in top_indices:
            s = int(idx - (n - 1))
            grad += lag_gradient(h, s)
        grad /= len(top_indices)
        grad *= dx

        new_h = project_feasible(h - step_size * grad, n)
        c5_new = compute_c5(new_h)
        if c5_new <= c5_cur:
            h = new_h
        else:
            step_size *= 0.7
        if time.time() - start_time > budget_s - 2:
            break

    h = project_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h, best_c5 = h, c5_h
    return best_h, best_c5

def coord_descent(h, n, start_time, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() - start_time < budget_s - 3:
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 3:
                break
            old_v = best_h[j]
            best_lc = best_c5
            best_lv = old_v
            base_c = best_c5
            for dv in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                v = old_v + dv
                if v < 0 or v > 1:
                    continue
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_lc - 1e-15:
                    best_lc = c
                    best_lv = v
            if best_lv != old_v:
                best_h[j] = best_lv
                best_h = make_feasible(best_h, n)
                best_c5 = best_lc
                improved = True
    return best_h, best_c5

def binary_anneal(h, n, start_time, budget_s, T_init=0.005, T_final=0.0001):
    h_b = (h > 0.5).astype(float)
    h_b = make_feasible(h_b, n)
    best_h = h_b.copy()
    best_c5 = compute_c5(h_b)
    curr_c5 = best_c5
    T = T_init
    rng2 = np.random.RandomState(12345)
    total_est = 80000
    step = 0

    while time.time() - start_time < budget_s - 5:
        step += 1
        frac = min(1.0, step / total_est)
        T = T_init * (T_final / T_init) ** frac

        j = rng2.randint(n)
        if h_b[j] > 0.5:
            zeros = np.flatnonzero(h_b == 0)
            if len(zeros) == 0:
                continue
            k2 = zeros[rng2.randint(len(zeros))]
            h_b[j] = 0.0
            h_b[k2] = 1.0
        else:
            ones = np.flatnonzero(h_b > 0.5)
            if len(ones) == 0:
                continue
            k2 = ones[rng2.randint(len(ones))]
            h_b[j] = 1.0
            h_b[k2] = 0.0

        new_c5 = compute_c5(h_b)
        delta = new_c5 - curr_c5
        if delta < 0 or (T > 0 and rng2.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_b.copy()
        else:
            h_b[k2] = 0.0 if h_b[k2] > 0.5 else 1.0
            h_b[j] = 1.0 if h_b[j] < 0.5 else 0.0

    return best_h, best_c5

def upsample(h, n_new):
    n_old = len(h)
    old_x = np.linspace(0.0, 2.0, n_old)
    new_x = np.linspace(0.0, 2.0, n_new)
    h_up = np.interp(new_x, old_x, h)
    return make_feasible(h_up, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: L-BFGS-B with smooth max, moderate T
    if rem() > 180:
        h1, c1 = lbfgsb_smooth(best_h, n, T=100, top_m=20, maxiter=200,
                                start_time=start_time, budget_s=budget_s)
        if c1 < best_c5:
            best_h, best_c5 = h1, c1

    # Phase 2: SLSQP with smoother objective
    if rem() > 150:
        h2, c2 = slsqp_smooth(best_h, n, T=200, top_m=15, maxiter=80,
                              start_time=start_time, budget_s=budget_s)
        if c2 < best_c5:
            best_h, best_c5 = h2, c2

    # Phase 3: PGD with FFT
    if rem() > 130:
        h3, c3 = pgd_fft(best_h, n, min(130, rem() - 30), start_time, top_m=10, T_sm=60)
        if c3 < best_c5:
            best_h, best_c5 = h3, c3

    # Phase 4: L-BFGS-B with sharper focus
    if rem() > 120:
        h4, c4 = lbfgsb_smooth(best_h, n, T=500, top_m=10, maxiter=150,
                                start_time=start_time, budget_s=budget_s)
        if c4 < best_c5:
            best_h, best_c5 = h4, c4

    # Phase 5: PGD tight focus
    if rem() > 100:
        h5, c5 = pgd_fft(best_h, n, min(100, rem() - 30), start_time, top_m=5, T_sm=100)
        if c5 < best_c5:
            best_h, best_c5 = h5, c5

    # Phase 6: Subgradient on exact max
    if rem() > 80:
        h6, c6 = subgrad_max(best_h, n, min(80, rem() - 20), start_time, top_m=4)
        if c6 < best_c5:
            best_h, best_c5 = h6, c6

    # Phase 7: L-BFGS-B very high T
    if rem() > 80:
        h7, c7 = lbfgsb_smooth(best_h, n, T=2000, top_m=5, maxiter=100,
                                start_time=start_time, budget_s=budget_s)
        if c7 < best_c5:
            best_h, best_c5 = h7, c7

    # Phase 8: Coordinate descent
    if rem() > 80:
        h8, c8 = coord_descent(best_h, n, start_time, min(80, rem() - 20))
        if c8 < best_c5:
            best_h, best_c5 = h8, c5

    # Phase 9: Binary annealing (explore different basin)
    if rem() > 100:
        hb, cb = binary_anneal(best_h, n, start_time, min(80, rem() - 40),
                                T_init=0.003, T_final=0.00005)
        if cb < best_c5 + 0.001:
            # Refine the binary solution in continuous
            h9, c9 = lbfgsb_smooth(hb, n, T=200, top_m=15, maxiter=100,
                                    start_time=start_time, budget_s=budget_s)
            if c9 < best_c5:
                best_h, best_c5 = h9, c9
            h10, c10 = pgd_fft(best_h if c9 < best_c5 else hb, n,
                                min(40, rem() - 30), start_time, top_m=8)
            if c10 < best_c5:
                best_h, best_c5 = h10, c10

    # Phase 10: Random restarts
    for Restart in range(3):
        if rem() < 50:
            break
        scale = 0.015
        h_try = make_feasible(best_h + rng.randn(len(best_h)) * scale, len(best_h))
        hr, cr = pgd_fft(h_try, len(best_h), min(30, rem() - 30), start_time, top_m=8)
        if cr < best_c5:
            best_h, best_c5 = hr, cr
            if rem() > 40:
                h2r, c2r = lbfgsb_smooth(best_h, len(best_h), T=500, top_m=8,
                                          maxiter=60, start_time=start_time, budget_s=budget_s)
                if c2r < best_c5:
                    best_h, best_c5 = h2r, c2r

    # Phase 11: Final SLSQP polish
    if rem() > 40:
        h11, c11 = slsqp_smooth(best_h, len(best_h), T=1000, top_m=4, maxiter=50,
                                 start_time=start_time, budget_s=budget_s)
        if c11 < best_c5:
            best_h, best_c5 = h11, c11

    # Phase 12: Final subgradient + coord
    if rem() > 25:
        h12, c12 = subgrad_max(best_h, len(best_h), min(20, rem() - 10), start_time, top_m=3)
        if c12 < best_c5:
            best_h, best_c5 = h12, c12
    if rem() > 15:
        h13, c13 = coord_descent(best_h, len(best_h), start_time, min(12, rem() - 3))
        if c13 < best_c5:
            best_h, best_c5 = h13, c13

    # Phase 13: Upsample if time remains
    if rem() > 120 and len(best_h) <= 960:
        n2 = len(best_h) * 2
        if n2 <= 1920:
            h_up = upsample(best_h, n2)
            c_up = compute_c5(h_up)
            if c_up < best_c5 + 0.002:
                hu, cu = lbfgsb_smooth(h_up, n2, T=200, top_m=20, maxiter=80,
                                        start_time=start_time, budget_s=budget_s)
                if cu < best_c5:
                    best_h, best_c5 = hu, cu
                if rem() > 60:
                    hu2, cu2 = pgd_fft(best_h, n2, min(50, rem() - 40), start_time, top_m=8)
                    if cu2 < best_c5:
                        best_h, best_c5 = hu2, cu2
                if rem() > 40:
                    hu3, cu3 = coord_descent(best_h, n2, start_time, min(30, rem() - 20))
                    if cu3 < best_c5:
                        best_h, best_c5 = hu3, cu3

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
