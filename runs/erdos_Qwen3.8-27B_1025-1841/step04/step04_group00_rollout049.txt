import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_smooth_max(corr, T):
    """Compute log-sum-exp approximation of max(corr)."""
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T

def compute_smooth_max_grad(corr, T):
    """Compute softmax weights for smooth max."""
    mc = np.max(corr)
    w = np.exp(T * (corr - mc))
    w /= np.sum(w)
    return w, mc + np.log(np.sum(w)) / T

def smooth_max_gradient(h, T, top_m=None):
    """Compute gradient of smooth max of correlation w.r.t. h."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m is not None and top_m < len(corr):
        # Zero out non-top to save computation
        idx = np.argsort(corr)[-top_m:]
        mask = np.zeros(len(corr), dtype=bool)
        mask[idx] = True
        corr_work = np.where(mask, corr, -np.inf)
    else:
        corr_work = corr
    w, smax = compute_smooth_max_grad(corr_work, T)

    # where k_c = c - (n-1)

            v *= 0.5

def slsqp_smooth(h, n, T=1000.0, maxiter=300):
    """SLSQP with smooth max."""
    dx = 2.0 / n
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        val = mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T
        return val * dx
    
    def obj_grad(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        w, smax = compute_smooth_max_grad(corr, T)
        grad = np.zeros(n)
        for c in range(len(corr)):
            if w[c] < 1e-12:
                continue
            k = c - (n - 1)
            j_s = max(0, -k)
            j_e = min(n, n - k)
            grad[j_s:j_e] += w[c] * (1.0 - x[j_s + k:j_e + k])
            j_s2 = max(0, k)
            j_e2 = min(n, n + k)
            grad[j_s2:j_e2] -= w[c] * x[j_s2 - k:j_e2 - k]
        return grad * dx
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    res = minimize(obj, h, jac=obj_grad, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    c5_res = compute_c5(res.x)
    h0_c5 = compute_c5(h)
    if c5_res < h0_c5:
        return make_feasible(res.x, n), c5_res
    return h, h0_c5

def coord_descent_fine(h, n, time_budget, start_time):

    best_h = h.copy()

    improved = True

            for v in [0.0, 1.0]:

                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        best_c5 = c
                        best_h = trial.copy()
                        improved = True
    return best_h, best_c5

def upsample(h, n_new):

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.count_nonzero(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, T):
    mc = np.max(corr)
    return mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T

def smooth_max_weights(corr, T):
    mc = np.max(corr)
    w = np.exp(T * (corr - mc))
    s = np.sum(w)
    w /= s
    return w

def compute_gradient_smooth(h, corr, weights):
    n = len(h)
    grad = np.zeros(n)
    active = weights > 1e-14
    if not np.any(active):
        return grad
    for c in np.where(active)[0]:
        w = weights[c]
        k = c - (n - 1)
        j_s = max(0, -k)
        j_e = min(n, n - k)
        if j_e > j_s:
            grad[j_s:j_e] += w * (1.0 - h[j_s + k:j_e + k])
        j_s2 = max(0, k)
        j_e2 = min(n, n + k)
        if j_e2 > j_s2:
            grad[j_s2:j_e2] -= w * h[j_s2 - k:j_e2 - k]
    return grad

        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: Try upsample to 2880
    n3 = min(n * 3, 2880)
    if n3 > len(best_h) and rem() > 300:
        h3 = upsample(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5 + 0.004:
            if rem() > 180:
                h3, c5_3 = pgd_optimize(h3, n3, min(150, rem() - 20), start_time, T=800, top_m=15, lr=0.005, momentum=0.82)
            if rem() > 100:
                h3, c5_3 = slsqp_optimize(h3, n3, T=1000, maxiter=200)
            if rem() > 50:
                h3, c5_3 = coord_descent(h3, n3, min(35, rem() - 10), start_time)
            if c5_3 < best_c5:
                best_h, best_c5 = h3, c5_3

    # Phase 7: Multi-restart with different scales
    n_cur = len(best_h)
    for r in range(5):
        if rem() < 50:
            break
        scale = 0.01 + 0.008 * r
        h_try = make_feasible(best_h + rng.randn(n_cur) * scale, n_cur)
        h_try, c5_try = pgd_optimize(h_try, n_cur, min(35, rem() - 20), start_time, T=800, top_m=10, lr=0.006, momentum=0.8)
        if c5_try < best_c5:
            best_h, best_c5 = h_try, c5_try

    # Phase 8: Final SLSQP polish
    if rem() > 30:
        h, c5 = slsqp_optimize(best_h, len(best_h), T=3000, maxiter=150)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 9: Ultra-fine coord descent
    if rem() > 20:
        h, c5 = coord_descent(best_h, len(best_h), min(15, rem() - 3), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
