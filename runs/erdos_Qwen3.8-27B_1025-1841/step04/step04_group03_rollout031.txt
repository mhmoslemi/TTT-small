```python
    n = len(h)

    log_weights -= log_weights.max()

    top_idx = np.argsort(weights)[::-1][:topk]
```

```python
                    delta = sign * d * scale
                    nv = old + delta
                    if nv < 0.0 or nv > 1.0:
                        continue
                    # Fast comparison using delta_corr
                    dc = delta_corr(h, j, delta, n)
                    new_max = float((corr + dc).max())
                    if new_max < best_max - 1e-14:
                        best_max = new_max
                        best_new = nv
            
            if abs(best_new - old) > 1e-12:
                # Apply change
                dc = delta_corr(h, j, best_new - old, n)
                h[j] = best_new
                corr = corr + dc  # Approximate (linear)
                # Occasionally recompute exactly to avoid drift
                if round_num % 5 == 0 or j == n - 1:
                    corr = compute_corr(h)
                improved = True
                if float(corr.max()) * dx < best_c5:
                    best_c5 = float(corr.max()) * dx
                    best_h = h.copy()
        
        if not improved:
            break
    
    # Final exact evaluation
    h = project_feasible(h, n)
    c5 = compute_c5(h)
    if c5 <= best_c5 + 1e-10:
        return h, c5
    best_h = project_feasible(best_h, n)
    return best_h, compute_c5(best_h)
```

The `delta_corr` function computes the change in O(n) time (a few array operations). So each trial is O(n) ≈ 1000 operations. With 960*14 = 13440 trials per pass, that's about 13M operations, taking maybe 0.1-0.2 seconds per pass. In 100 seconds, I can do 500+ passes! That's plenty for convergence.

The approximation error from using the linear delta_corr accumulates over many steps, so I periodically recompute the exact correlation. This should be accurate enough.

```python
def binary_local_search(h, n, start_time, time_limit, rng):
    """Local search on binary sequences using fast delta evaluation."""
    b = (h >= 0.5).astype(float)
    target = n // 2
    # Adjust
    while b.sum() > target:
        ones = np.where(b == 1)[0]
        b[ones[rng.randint(len(ones))]] = 0
    while b.sum() < target:
        zeros = np.where(b == 0)[0]
        b[zeros[rng.randint(len(zeros))]] = 1
    
    corr = compute_corr(b)
    dx = 2.0 / n
    curr_c5 = float(corr.max()) * dx
    best_c5 = curr_c5
    best_b = b.copy()
    
    T = 0.01  # Temperature for SA
    it = 0
    while time.time() - start_time < time_limit:
        it += 1
        T = 0.01 * (0.999 ** it)
        
        # Random swap
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Compute delta for swapping i (1→0) and j (0→1)
        dc = delta_corr(b, i, -1.0, n) + delta_corr(b, j, 1.0, n)
        new_corr = corr + dc
        new_c5 = float(new_corr.max()) * dx
        
        if new_c5 <= curr_c5 or (T > 1e-15 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            b[i] = 0
            b[j] = 1
            corr = new_corr
            curr_c5 = new_c5
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_b = b.copy()
        else:
            pass  # Reject, no change needed
        
        # Periodic exact recompute
        if it % 1000 == 0:
            corr = compute_corr(b)
            curr_c5 = float(corr.max()) * dx
    
    return best_b, best_c5
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def project_feasible(h, n):
    target = float(n) / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        x = np.clip(h - mid, 0.0, 1.0)
        if x.sum() > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - mid, 0.0, 1.0)

def delta_corr(h, j, delta, n):
    dc = np.zeros(2 * n - 1)
    s_lo = max(-(n-1), -j)
    s_hi = min(n-1, n-1-j)
    if s_lo <= s_hi:
        s = np.arange(s_lo, s_hi + 1)
        dc[n-1+s] += delta * (1.0 - h[j+s])
    s_lo2 = max(-(n-1), j-(n-1))
    s_hi2 = min(n-1, j)
    if s_lo2 <= s_hi2:
        s2 = np.arange(s_lo2, s_hi2 + 1)
        dc[n-1+s2] -= delta * h[j-s2]
    return dc
