from scipy.optimize import linprog, minimize
import time

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # shape (2n-1,), index k corresponds to shift k-(n-1)

def compute_gradients_active(h, active_shifts):
    """Compute gradients of overlap for active shifts.
    Returns (n, len(active_shifts)) array where column i is grad of overlap for shift i."""
    n = len(h)
    dx = 2.0 / n
    # For shift s (0-indexed from 0 to n-1 for positive, or negative):
    # overlap(s) = sum_{j} h[j] * (1-h[j+s]) * dx (for j+s in [0,n-1])
    # d/dh[j] = dx * [(1-h[j+s]) if j+s in range - h[j-s] if j-s in range]
    grads = np.zeros((n, len(active_shifts)))
    for i, s in enumerate(active_shifts):
        # s is the shift in correlation index: actual shift = s - (n-1)
        actual_s = s - (n - 1)
        if actual_s >= 0:
            # overlap = sum_{j=0}^{n-1-s} h[j]*(1-h[j+s]) * dx
            # d/dh[j] for j in [0, n-1-s]: (1-h[j+s]) * dx
            # d/dh[j] for j in [s, n-1]: -h[j-s] * dx
            for j in range(n - actual_s):
                grads[j, i] += (1.0 - h[j + actual_s]) * dx
            for j in range(actual_s, n):
                grads[j, i] -= h[j - actual_s] * dx
        else:

            s_prime = -actual_s
            # overlap = sum_{j=s'}^{n-1} h[j]*(1-h[j-s']) * dx
            # d/dh[j] for j in [0, n-1-s']: -h[j+s'] * dx (from the (1-h[j-s']) term when j - actual_s = j + s' is the first arg)
            # Wait, let me redo this.

            # overlap = sum over valid j: h[j] * (1 - h[j + actual_s]) = h[j] * (1 - h[j - s_prime])
            # valid j: j - s_prime >= 0, i.e., j >= s_prime, and j < n
            # So overlap = sum_{j=s_prime}^{n-1} h[j] * (1 - h[j - s_prime]) * dx
            # d/dh[j]:

        active_idx = np.where(active_mask)[0]

        A_ub = np.zeros((m_active, n + 1))
        b_ub = np.zeros(m_active)
        for i, ai in enumerate(active_idx):

            A_ub[i, n] = -1.0
            b_ub[i] = -overlaps[ai] + grads[:, i] @ h

        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0

        bounds = [(0, 1)] * n + [(0, None)]

                    best_c5 = c5_new
                    h = h_new
                else:

            else:
                break

import numpy as np
from scipy.optimize import linprog, minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_gradient_shift(h, s):
    """Gradient of overlap for shift s (actual shift, not correlation index)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n-s] += (1.0 - h[s:]) * dx
            grad[s:] -= h[:n-s] * dx
    else:
        p = -s
        if p < n:
            grad[p:] += (1.0 - h[:n-p]) * dx
            grad[:n-p] -= h[p:] * dx
    return grad

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.argmin(h)
        else:
            idx = np.argmax(h)
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_round(h, n, top_m, epsilon, time_limit, start_time):
    """Iterated linearization approach."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    while time.time() - start_time < time_limit:
        # Compute overlaps
        overlaps = compute_all_overlaps(best_h)
        current_max = np.max(overlaps)
        
        # Select active shifts
        threshold = current_max - epsilon
        active_mask = overlaps >= threshold
        active_idx = np.where(active_mask)[0]
        top_idx = np.argsort(overlaps)[-top_m:]
        active_idx = np.unique(np.concatenate([active_idx, top_idx]))
        
        # Compute gradients
        n_active = len(active_idx)
        grads = np.zeros((n, n_active))
        for i, ci in enumerate(active_idx):
            s = ci - (n - 1)  # actual shift
            grads[:, i] = compute_gradient_shift(best_h, s)
        
        # LP setup
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0  # minimize t
        
        A_ub = np.zeros((n_active, n + 1))
        b_ub = np.zeros(n_active)
        for i in range(n_active):
            A_ub[i, :n] = grads[:, i]
            A_ub[i, n] = -1.0
            b_ub[i] = -overlaps[active_idx[i]] + grads[:, i] @ best_h
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.success:
                h_new = res.x[:n]
                h_new = make_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5 - 1e-14:
                    best_h = h_new
                    best_c5 = c5_new
                else:
                    # Check if we can still make progress
                    if abs(c5_new - best_c5) < 1e-10:
                        break
                    # Try a step toward the LP solution
                    alpha = 0.5
                    h_trial = make_feasible(alpha * h_new + (1-alpha) * best_h, n)
                    c5_trial = compute_c5(h_trial)
                    if c5_trial < best_c5 - 1e-14:
                        best_h = h_trial
                        best_c5 = c5_trial
                    else:
                        break
            else:
                break
        except Exception:
            break
    
    return best_h, best_c5

def pgd_round(h, n, time_limit, start_time, top_m=5):
    """Projected gradient descent with momentum and line search."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    velocity = np.zeros(n)
    momentum = 0.85
    step_size = 0.05
    no_improve = 0
    
    while time.time() - start_time < time_limit:
        # Compute overlaps
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c5_cur = np.max(corr) * dx
        
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 50:
                step_size *= 0.5
                if step_size < 1e-7:
                    break
                no_improve = 0
        
        # Top shifts
        top_idx = np.argsort(corr)[-top_m:]
        
        # Compute gradient
        grad = np.zeros(n)
        for ci in top_idx:
            s = ci - (n - 1)
            grad += compute_gradient_shift(best_h, s)
        grad /= len(top_idx)
        
        # Nesterov
        y = best_h - momentum * velocity
        y = np.clip(y, 0, 1)
        
        # Gradient at y (approximate)
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        top_idx_y = np.argsort(corr_y)[-top_m:]
        grad_y = np.zeros(n)
        for ci in top_idx_y:
            s = ci - (n - 1)
            grad_y += compute_gradient_shift(y, s)
        grad_y /= len(top_idx_y)
        
        velocity = momentum * velocity - step_size * grad_y
        h_new = best_h + velocity
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new <= c5_cur:
            best_h = h_new
        else:
            # Backtrack
            h_new2 = make_feasible(best_h - step_size * grad, n)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 <= c5_cur:
                best_h = h_new2
                velocity *= 0.5
            else:
                step_size *= 0.7
                velocity *= 0.3
            if step_size < 1e-8:
                break
        
        if no_improve % 100 == 0 and step_size < 0.1:
            step_size *= 1.2
    
    best_h = make_feasible(best_h, n)
    c5_final = compute_c5(best_h)
    if c5_final < best_c5:
        best_h, best_c5 = best_h, c5_final
    return best_h, best_c5

def coordinate_descent(h, n, time_limit, start_time, steps=None):

import numpy as np
from scipy.optimize import linprog
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def compute_gradient_shift(h, s):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n - s] += (1.0 - h[s:]) * dx
            grad[s:] -= h[:n - s] * dx
    else:
        p = -s
        if p < n:
            grad[p:] += (1.0 - h[:n - p]) * dx
            grad[:n - p] -= h[p:] * dx
    return grad

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def lp_round(h, n, top_m, epsilon, time_limit, start_time):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() - start_time < time_limit:
        overlaps = compute_all_overlaps(best_h)
        current_max = np.max(overlaps)
        threshold = current_max - epsilon
        active_mask = overlaps >= threshold
        active_idx = np.where(active_mask)[0]
        top_idx = np.argsort(overlaps)[-top_m:]
        active_idx = np.unique(np.concatenate([active_idx, top_idx]))
        # Cap active set size
        if len(active_idx) > 200:
            top200 = np.argsort(overlaps)[::-1][:200]
            active_idx = top200
        n_active = len(active_idx)
        grads = np.zeros((n, n_active))
        for i, ci in enumerate(active_idx):
            s = (n - 1) - ci
            grads[:, i] = compute_gradient_shift(best_h, s)
        c_vec = np.zeros(n + 1)
        c_vec[-1] = 1.0
        A_ub = np.zeros((n_active, n + 1))
        b_ub = np.zeros(n_active)
        for i in range(n_active):
            A_ub[i, :n] = grads[:, i]
            A_ub[i, n] = -1.0
            b_ub[i] = -overlaps[active_idx[i]] + grads[:, i] @ best_h
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds = [(0, 1)] * n + [(0, None)]
        try:
            res = linprog(c_vec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                break
            h_new = res.x[:n]
            dist = np.max(np.abs(h_new - best_h))
            if dist > 0.3:
                h_new = best_h + 0.3 * (h_new - best_h)
            h_new = make_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-14:
                best_h = h_new
                best_c5 = c5_new
            else:
                alpha = 0.3
                h_trial = make_feasible(alpha * res.x[:n] + (1 - alpha) * best_h, n)
                c5_trial = compute_c5(h_trial)
                if c5_trial < best_c5 - 1e-14:
                    best_h = h_trial
                    best_c5 = c5_trial
                else:
                    break
        except Exception:
            break
    return best_h, best_c5

def pgd_round(h, n, time_limit, start_time, top_m=5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    velocity = np.zeros(n)
    momentum = 0.8
    step_size = 0.03
    no_improve = 0
    while time.time() - start_time < time_limit:
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c5_cur = np.max(corr) * (2.0 / n)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                step_size *= 0.6
                if step_size < 1e-7:
                    break
                no_improve = 0
        top_idx = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        for ci in top_idx:
            s = (n - 1) - ci
            grad += compute_gradient_shift(best_h, s)
        grad /= len(top_idx)
        y = best_h - momentum * velocity
        yc = np.clip(y, 0, 1)
        corr_y = np.correlate(yc, 1.0 - yc, mode='full')
        top_y = np.argsort(corr_y)[-top_m:]
        grad_y = np.zeros(n)
        for ci in top_y:
            s = (n - 1) - ci
            grad_y += compute_gradient_shift(yc, s)
        grad_y /= len(top_y)
        velocity = momentum * velocity - step_size * grad_y
        h_new = make_feasible(best_h + velocity, n)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_cur:
            best_h = h_new
        else:
            h_new2 = make_feasible(best_h - step_size * grad, n)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 <= c5_cur:
                best_h = h_new2
                velocity *= 0.5
            else:
                step_size *= 0.7
                velocity *= 0.3
            if step_size < 1e-8:
                break
    best_h = make_feasible(best_h, n)
    c5_f = compute_c5(best_h)
    if c5_f < best_c5:
        best_c5 = c5_f
    return best_h, best_c5

def bit_flip_search(h, n, time_limit, start_time, rng):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() - start_time < time_limit:
        improved = False
        order = rng.permutation(n)
        for j in order:
            if time.time() - start_time > time_limit - 2:
                break
            for v in [0.0, 1.0]:
                if abs(best_h[j] - v) < 1e-10:
                    continue
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_h = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def slsqp_polish(h, n, time_limit, start_time, T=500.0):
    import scipy.optimize as so
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        safe = np.minimum(T * shifted, 500.0)
        return (mc + np.log(np.sum(np.exp(safe))) / T) * dx
    def con_fun(x):
        return np.sum(x) - n / 2.0
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    try:
        res = so.minimize(obj, h, method='SLSQP', bounds=bounds,
                         constraints=cons, options={'maxiter': 200, 'ftol': 1e-14})
        if res.success:
            h_new = make_feasible(res.x, n)
            c5_new = compute_c5(h_new)
            if c5_new < compute_c5(h):
                return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)
