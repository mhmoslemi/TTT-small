    if rem() > 15:
        for rd in range(3):
            improved = False
            for j in range(n):
                if rem() < 3:
                    break
                hj = best_h[j]
                for step in [0.0005, 0.001, 0.002, 0.005, 0.01]:
                    for s in [-1.0, 1.0]:
                        v = hj + s * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n)
                            c = compute_c5(trial)
                            if c < best_c5 - 1e-15:
                                best_c5 = c
                                best_h = trial
                                improved = True
                                break
                    if improved:
                        break
            if not improved:
                break

    # Phase 8: Random perturbations + L-BFGS-B
    for restart in range(3):
        if rem() < 60:
            break
        scale = 0.01 + 0.01 * restart
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        _G_T = 600.0
        _G_LAM = 5000.0
        try:
            r = minimize(_obj_lbfgs, h_try, method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 200, 'ftol': 1e-17, 'gtol': 1e-14})
            hr = make_feasible(r.x, n)
            c = compute_c5(hr)
            if c < best_c5:
                best_c5, best_h = c, hr
        except Exception:
            pass

    # Phase 9: Try complement symmetry
    comp = make_feasible(1.0 - best_h, n)
    c_comp = compute_c5(comp)
    if c_comp < best_c5:
        best_c5, best_h = c_comp, comp

    # Phase 10: Final SLSQP polish
    if rem() > 30:
        _G_T = 1500.0
        cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
        try:
            r = minimize(_obj_slsqp, best_h, method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 200, 'ftol': 1e-17})
            hr = make_feasible(r.x, n)
            c = compute_c5(hr)
            if c < best_c5:
                best_c5, best_h = c, hr
        except Exception:
            pass

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
