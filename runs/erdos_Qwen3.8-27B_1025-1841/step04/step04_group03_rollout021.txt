If A = [0,1], then |A ∩ (A-k)| = |(-k, 1-k) ∩ [0,1]| = max(0, 1-|k|) for k ∈ [-1,1], and 0 otherwise.
So min_k |A ∩ (A-k)| = 0 (at k=1 or k=-1). C₅ = 1. Bad.

If A is "random" (each point included with prob 0.5), then E[|A ∩ (A-k)|] ≈ 0.5 * 2 * (1-|k|/2) for |k|≤2... actually this gets complicated.

For a periodic-like structure: if A consists of K intervals of length 1/K spread evenly over [0,2], then the autocorrelation will have peaks at multiples of 2/K.

Actually, I recall that for the continuous version, the optimal is related to " discrepancy theory" or "equidistribution." A good heuristic is that h should be "as close to 0.5 as possible" in a structured way that avoids having large blocks of 1's or 0's.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return make_feasible(h_new, n_new)

_G_N = 0
_G_T = 0.0

def obj_slsqp(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + T * np.log(np.sum(np.exp(shifted / T)))) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def compute_shift_gradient(h, s):
    n = len(h)
    grad = np.zeros(n)
    ss = abs(int(s))
    if ss < n:
        grad[0:n-ss] += (1.0 - h[ss:])
        grad[ss:n] += -h[0:n-ss]
    return grad

def bottleneck_subgrad(h, n, n_steps, lr0, start_time, budget_s):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for step in range(n_steps):
        if time.time() - start_time > budget_s - 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m = int(np.argmax(corr))
        s = m - (n - 1)
        grad = compute_shift_gradient(h, s)
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        lr *= 0.995
    return best_h, best_c5

def multi_shift_subgrad(h, n, n_steps, lr0, start_time, budget_s, n_top=3):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr0
    for step in range(n_steps):
        if time.time() - start_time > budget_s - 15:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[::-1][:n_top]
        total_grad = np.zeros(n)
        for m in top_idx:
            s = m - (n - 1)
            total_grad += compute_shift_gradient(h, s)
        total_grad /= n_top
        h_new = h - lr * total_grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        lr *= 0.995
    return best_h, best_c5

def coord_descent_fine(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0,
                     max(0.0, old - 0.05), max(0.0, old - 0.02), max(0.0, old - 0.005),
                     min(1.0, old + 0.005), min(1.0, old + 0.02), min(1.0, old + 0.05)]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def fourier_perturb(h, n, rng, n_modes=5, amp=0.05):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.3, 1.0)
    return make_feasible(h_new, n)

def local_block_perturb(h, n, rng):
    h_new = h.copy()
    n_blocks = rng.randint(2, 6)
    for _ in range(n_blocks):
        bsz = rng.randint(n // 20, n // 6)
        start = rng.randint(0, n - bsz)
        if rng.random() < 0.4:
            h_new[start:start+bsz] = 1.0 - h_new[start:start+bsz]
        else:
            h_new[start:start+bsz] += rng.uniform(-0.3, 0.3)
    return make_feasible(h_new, n)

def shift_blend_perturb(h, n, rng):
    s = rng.choice([1, 2, 3, 4, 5, 8, 12])
    h_shift = np.roll(h, s)
    alpha = rng.uniform(0.3, 0.7)
    h_new = alpha * h + (1 - alpha) * h_shift
    return make_feasible(h_new, n)
