    # j1 is valid if 0 <= j1 < n, i.e., 0 <= i+m-(n-1) < n, i.e., n-1-i <= m <= 2n-2-i
    # In padded: i+m is in [n-1, 2n-2] which is always in [0, 2n-2] for valid i,m.
    # But we need the mask: n-1-i <= m <= 2n-2-i
    
    # Actually, h_padded[i+m] is 0 when i+m < n-1 or i+m > 2n-2.
    # i+m < n-1 means m < n-1-i. Since m >= 0, this happens when m < n-1-i (i.e., i > n-1-m).
    # i+m > 2n-2 means m > 2n-2-i. Since m <= 2n-2, this happens when i+m > 2n-2 (i.e., m > 2n-2-i).
    
    # So h_padded[i+m] already handles the validity! It's 0 when out of range.
    
    # A[i] = sum_m w[m] * (1 - h_padded[i+m]) * [h_padded is "active" at i+m]
    # But we need (1-h[j]) only when j is valid (0<=j<n). When j is invalid, the term shouldn't contribute.
    # h_padded[i+m] = 0 when invalid, so (1-h_padded[i+m]) = 1 when invalid. That's wrong!
    
    # I need to mask it. Let me create a validity mask.
    # valid[i,m] = (0 <= i+m-(n-1) < n) = (n-1 <= i+m <= 2n-2)
    
    # For the first term: A[i] = sum_{m: n-1<=i+m<=2n-2} w[m] * (1 - h[i+m-(n-1)])
    # = sum_{m} w[m] * (1 - h[padded_idx]) * valid_mask
    
    # Let me just use the validity mask approach.
    
    # j1[i,m] = i + m - (n-1), valid if 0 <= j1 < n
    j1 = i_idx[:, None] + m_idx[None, :] - (n-1)  # shape (n, 2n-1)
    mask1 = (j1 >= 0) & (j1 < n)
    
    # For valid entries, h[j1[i,m]]; for invalid, 0
    # We can use np.where or advance indexing with clipping
    j1_clipped = np.clip(j1, 0, n-1)
    h_j1 = h[j1_clipped]  # This is h[j1_clipped[i,m]] for each (i,m)
    
    A = np.sum(w[None, :] * np.where(mask1, 1.0 - h_j1, 0.0), axis=1)
    
    # j2[i,m] = i - m + (n-1), valid if 0 <= j2 < n
    j2 = i_idx[:, None] - m_idx[None, :] + (n-1)
    mask2 = (j2 >= 0) & (j2 < n)
    j2_clipped = np.clip(j2, 0, n-1)
    h_j2 = h[j2_clipped]
    
    B = np.sum(w[None, :] * np.where(mask2, h_j2, 0.0), axis=1)
    
    return dx * (A - B)

def project_feasible(h, n):
    target = n / 2.0
    # Water-filling: find λ such that sum(clip(h - λ, 0, 1)) = target
    # h - λ clipped to [0,1] means:
    # If h[i] - λ <= 0: result is 0
    # If h[i] - λ >= 1: result is 1
    # Otherwise: result is h[i] - λ
    # Sum = count(h[i] >= 1+λ) * 1 + sum_{0 < h[i]-λ < 1} (h[i]-λ)
    # Let's binary search on λ.

    for _ in range(80):

        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

        else:
