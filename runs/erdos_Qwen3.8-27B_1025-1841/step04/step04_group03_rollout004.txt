    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_smooth, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h

def multi_temp_refine(h, n, start_time, budget_s, temps, iters_base=250):
    curr_c5 = compute_c5(h)
    for T in temps:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 25:
            break
        iters = min(int(remaining * 0.15), iters_base)
        iters = max(iters, 30)
        try:
            x_res = slsqp_opt(h, n, T, iters, 1e-14)
            x_res = make_feasible(x_res, n)
            c5_res = compute_c5(x_res)
            if c5_res < curr_c5 - 1e-15:
                h = x_res
                curr_c5 = c5_res
        except Exception:
            pass
    return h, curr_c5

def fourier_perturb(h, n, rng, n_modes=5, amp=0.05):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 4)
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.4, 1.4)
    return make_feasible(h_new, n)

def peak_target_perturb(h, n, rng, amp=0.04):
    corr = compute_corr_vec(h)
    peak_val = np.max(corr)
    peak_lags = np.where(corr > peak_val * 0.9)[0]
    h_new = h.copy()
    x = np.arange(n)
    for _ in range(rng.randint(3, 7)):
        lag = peak_lags[rng.randint(len(peak_lags))]
        f = rng.uniform(1.0, 6.0) / n
        phase = rng.uniform(0, 2 * np.pi)
        sig = np.sin(2 * np.pi * f * x + phase)
        sig /= (np.max(np.abs(sig)) + 1e-10)
        h_new += amp * sig * rng.uniform(0.3, 1.0)
    return make_feasible(h_new, n)

def segment_perturb(h, n, rng):
    h_new = h.copy()
    n_segs = rng.randint(1, 4)
    for _ in range(n_segs):
        length = rng.randint(n // 16, n // 5)
        start = rng.randint(0, max(1, n - length))
        if rng.random() < 0.5:
            h_new[start:start + length] = 1.0 - h_new[start:start + length]
        else:
            h_new[start:start + length] = rng.uniform(0, 1)
    return make_feasible(h_new, n)

def swap_sa(h, n, max_iters, T0, T1, rng, time_limit, start_time):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    stall = 0
    for it in range(max_iters):
        if time.time() - start_time > start_time + time_limit:
            break
        frac = min(1.0, float(it) / max_iters)
        T = T0 * (T1 / T0) ** frac
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        delta = new_c5 - curr_c5
        if delta <= 0 or (T > 1e-14 and rng.random() < np.exp(-delta / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_b = b.copy()
                stall = 0
            else:
                stall += 1
        else:
            b[i] = 1
            b[j] = 0
            stall += 1
        if stall > 8000:
            for _ in range(80):
                i2 = ones[rng.randint(len(ones))]
                j2 = zeros[rng.randint(len(zeros))]
                b[i2] = 1 - b[i2]
                b[j2] = 1 - b[j2]
            curr_c5 = compute_c5(b)
            stall = 0
    return best_b, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    rdc = np.random.RandomState(12345)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        order = rdc.permutation(n)
        for j in order:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.1), max(0.0, old - 0.03),
                     max(0.0, old - 0.008), max(0.0, old - 0.002),
                     min(1.0, old + 0.002), min(1.0, old + 0.008),
                     min(1.0, old + 0.03), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Thorough multi-temp SLSQP from initial point
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        hot_temps = [100000.0, 30000.0, 10000.0, 5000.0, 2000.0, 1000.0, 500.0, 250.0, 100.0, 50.0, 20.0]
        best_h, best_c5 = multi_temp_refine(best_h, n0, start_time, budget_s, hot_temps, 300)

    # Phase 2: Diverse population search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 350:
        pop = [best_h.copy()]
        pop.append(fourier_perturb(best_h, n0, rng, n_modes=6, amp=0.06))
        pop.append(peak_target_perturb(best_h, n0, rng, amp=0.05))
        pop.append(segment_perturb(best_h, n0, rng))
        pop.append(fourier_perturb(best_h, n0, rng, n_modes=3, amp=0.03))

        med_temps = [5000.0, 2000.0, 800.0, 400.0, 200.0, 80.0, 30.0]
        for gen in range(4):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 100:
                break
            for i in range(len(pop)):
                remaining = budget_s - (time.time() - start_time)
                if remaining < 50:
                    break
                hp, cp = multi_temp_refine(pop[i], n0, start_time, budget_s, med_temps, 200)
                pop[i] = hp
                if cp < best_c5:
                    best_c5 = cp
                    best_h = hp.copy()
            # Evolve population
            remaining = budget_s - (time.time() - start_time)
            if remaining > 60:
                i1 = rng.randint(len(pop))
                i2 = rng.randint(len(pop))
                if i1 != i2:
                    a = rng.uniform(0.3, 0.7)
                    hc = make_feasible(a * pop[i1] + (1 - a) * pop[i2], n0)
                    worst = max(range(len(pop)), key=lambda x: compute_c5(pop[x]))
                    pop[worst] = hc
                hp_new = peak_target_perturb(best_h, n0, rng, amp=0.04)
                worst2 = max(range(len(pop)), key=lambda x: compute_c5(pop[x]))
                pop[worst2] = hp_new

    # Phase 3: Binary SA for structural improvements
    remaining = budget_s - (time.time() - start_time)
    if remaining > 120:
        tl = min(remaining - 20, 130)
        b, c5_b = swap_sa(best_h, n0, 300000, 0.003, 0.000003, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 80:
            final_temps = [3000.0, 1000.0, 500.0, 200.0, 80.0, 30.0, 10.0]
            best_h, best_c5 = multi_temp_refine(best_h, n0, start_time, budget_s, final_temps, 250)

    # Phase 4: More targeted local search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        for _ in range(4):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
            hp = peak_target_perturb(best_h, n0, rng, amp=0.03)
            lt_temps = [2000.0, 800.0, 300.0, 100.0, 50.0, 20.0, 10.0]
            hp, cp = multi_temp_refine(hp, n0, start_time, budget_s, lt_temps, 200)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp

    # Phase 5: Try higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 350:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n1, endpoint=False)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        up_temps = [30000.0, 10000.0, 5000.0, 2000.0, 800.0, 300.0, 100.0, 50.0, 20.0]
        h_up, c5_up = multi_temp_refine(h_up, n1, start_time, budget_s, up_temps, 200)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100 and c5_up <= best_c5:
            hp = fourier_perturb(best_h, len(best_h), rng, n_modes=4, amp=0.03)
            hp, cp = multi_temp_refine(hp, len(best_h), start_time, budget_s,
                                       [5000.0, 2000.0, 800.0, 300.0, 100.0, 50.0], 200)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp

    # Phase 6: Extended coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 40, start_time, budget_s)

    # Phase 7: Final SLSQP polish with very low temperatures
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        polish_temps = [5000.0, 2000.0, 800.0, 300.0, 100.0, 50.0, 20.0, 10.0, 5.0]
        best_h, best_c5 = multi_temp_refine(best_h, n_pts, start_time, budget_s, polish_temps, 300)

    # Phase 8: Final coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 15:
        best_h, best_c5 = coord_descent(best_h.copy(), len(best_h), 15, start_time, budget_s)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
