    cons_now = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_eq_jac}]
    for r in range(4):
        if rem() < 45:
            break
        scale = 0.008 + 0.012 * r
        h_try = make_feasible(best_h + rng.randn(n_now) * scale, n_now)
        _G_N = n_now
        _G_T = 600.0
        _G_TOP_IDX = None
        try:
            res = minimize(_obj_soft, h_try, method='SLSQP',
                          bounds=bnds_now, constraints=cons_now,
                          options={'maxiter': 400, 'ftol': 1e-17})
            ch = make_feasible(res.x, n_now)
            c = compute_c5(ch)
            if c < best_c5:
                best_c5 = c
                best_h = ch
        except Exception:
            pass

    # Phase 8: Final ultra-fine coordinate descent
    for rd in range(3):
        if rem() < 8:
            break
        steps = [0.0005, 0.001, 0.002, 0.004, 0.008] if rd == 0 else [0.0003, 0.0008, 0.002]
        best_h, best_c5, _ = cd_pass(best_h, len(best_h), best_c5, steps, len(best_h))

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
