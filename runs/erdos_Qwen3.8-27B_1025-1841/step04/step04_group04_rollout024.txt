        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h


def to_binary(h, n):
    sorted_idx = np.argsort(h)[::-1]
    binary = np.zeros(n)
    binary[sorted_idx[:n // 2]] = 1.0
    return binary


def _obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_G_T * shifted))) / _G_T) * dx


def _con_fun(x):
    return np.sum(x) - _G_N / 2.0


def _con_jac(x):
    return np.ones(_G_N)


def _time_remaining():
    return _G_BUDGET - (time.time() - _G_START)


def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T, _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    _G_N = n
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # === Phase 1: Binary Simulated Annealing ===
    h_bin = to_binary(h_init, n)
    curr_c5 = compute_c5(h_bin)
    if curr_c5 < best_c5:
        best_c5 = curr_c5
        best_h = h_bin.copy()

    n_ones = n // 2
    n_zeros = n - n_ones
    ones_arr = np.where(h_bin == 1.0)[0].copy()
    zeros_arr = np.where(h_bin == 0.0)[0].copy()

    T_start = 0.003
    T_end = 0.000005
    sa_time_fрак = 0.55
    sa_limit = sa_time_fрак * budget_s

    best_sa_c5 = curr_c5
    best_sa_h = h_bin.copy()
    no_improve = 0
    max_no_improve = 40000
    reheat_count = 0
    max_reheats = 15

    while (time.time() - _G_START) < _G_BUDGET - 350:
        sa_frac = min(1.0, (time.time() - _G_START) / sa_limit)
        T = T_start * (T_end / T_start) ** sa_frac

        batch_size = 4
        best_delta = float('inf')
        best_i = best_j = -1
        best_i_idx = best_j_idx = -1

        for _ in range(batch_size):
            i_idx = rng.randint(n_ones)
            j_idx = rng.randint(n_zeros)
            i = ones_arr[i_idx]
            j = zeros_arr[j_idx]

            h_bin[i] = 0.0
            h_bin[j] = 1.0
            c = compute_c5(h_bin)
            delta = c - curr_c5
            h_bin[i] = 1.0
            h_bin[j] = 0.0

            if delta < best_delta:
                best_delta = delta
                best_i, best_j = i, j
                best_i_idx, best_j_idx = i_idx, j_idx

        if best_delta < 0 or (T > 1e-9 and rng.random() < np.exp(-best_delta / T)):
            h_bin[best_i] = 0.0
            h_bin[best_j] = 1.0
            curr_c5 += best_delta
            ones_arr[best_i_idx] = best_j
            zeros_arr[best_j_idx] = best_i

            if curr_c5 < best_sa_c5 - 1e-15:
                best_sa_c5 = curr_c5
                best_sa_h = h_bin.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1

        if no_improve > max_no_improve:
            if reheat_count < max_reheats:
                # Reheat: random perturbation
                for _ in range(15 + reheat_count * 5):
                    i_idx = rng.randint(n_ones)
                    j_idx = rng.randint(n_zeros)
                    i = ones_arr[i_idx]
                    j = zeros_arr[j_idx]
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    ones_arr[i_idx] = j
                    zeros_arr[j_idx] = i
                curr_c5 = compute_c5(h_bin)
                reheat_count += 1
            else:
                # Aggressive restart: re-derive binary from best
                h_bin[:] = 0.0
                h_bin[ones_arr] = 1.0
                curr_c5 = compute_c5(h_bin)
                # Try a different initial binary from a perturbed version
                for _ in range(30):
                    i_idx = rng.randint(n_ones)
                    j_idx = rng.randint(n_zeros)
                    i = ones_arr[i_idx]
                    j = zeros_arr[j_idx]
                    h_bin[i] = 0.0
                    h_bin[j] = 1.0
                    ones_arr[i_idx] = j
                    zeros_arr[j_idx] = i
                curr_c5 = compute_c5(h_bin)
            no_improve = 0

    if best_sa_c5 < best_c5:
        best_c5 = best_sa_c5
        best_h = best_sa_h.copy()

    # === Phase 2: SLSQP continuous refinement ===
    if _time_remaining() > 150:
        _G_T = 100.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # === Phase 3: Targeted coordinate descent ===
    if _time_remaining() > 60:
        for _round in range(60):
            if _time_remaining() < 10:
                break
            improved = False
            # Focus on positions with intermediate values first
            order = np.argsort(np.abs(best_h - 0.5))[::-1]
            for j in order:
                if _time_remaining() < 5:
                    break
                old = best_h[j]
                best_trial = old
                best_c = best_c5
                cands = [0.0, 1.0, 0.5]
                for d in [-0.2, -0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1, 0.2]:
                    c = max(0.0, min(1.0, old + d))
                    cands.append(c)
                for cand in cands:
                    if abs(cand - old) < 1e-10:
                        continue
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-14:
                        best_c = c
                        best_trial = cand
                if best_trial != old:
                    best_h[j] = best_trial
                    best_h = make_feasible(best_h, n)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break

    # === Phase 4: Final SLSQP polish at high temperature ===
    if _time_remaining() > 40:
        _G_T = 400.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-15})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # === Phase 5: Final fine-tune coordinate descent ===
    if _time_remaining() > 20:
        for _round in range(30):
            if _time_remaining() < 5:
                break
            improved = False
            for j in range(n):
                if _time_remaining() < 3:
                    break
                old = best_h[j]
                best_c = best_c5
                best_trial = old
                for d in [-0.005, -0.002, -0.001, 0.001, 0.002, 0.005]:
                    c = max(0.0, min(1.0, old + d))
                    trial = best_h.copy()
                    trial[j] = c
                    trial = make_feasible(trial, n)
                    val = compute_c5(trial)
                    if val < best_c - 1e-15:
                        best_c = val
                        best_trial = c
                if best_trial != old:
                    best_h[j] = best_trial
                    best_h = make_feasible(best_h, n)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
