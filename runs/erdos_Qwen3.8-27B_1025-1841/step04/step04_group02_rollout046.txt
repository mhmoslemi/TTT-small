import numpy as np
from scipy.optimize import minimize
from scipy.signal import fftconvolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_corr(h):
    """Returns both C5 value and the correlation vector."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return mc * dx, corr

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    # Adjust to have exactly target ones
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        # Flip the one with lowest h value
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h_init, n, max_iters, T0, T1, rng, use_blocks=True):
    """Simulated annealing with point and block swaps."""
    b = to_binary(h_init, n)
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    ones = np.where(b == 1)[0]
    zeros = np.where(b == 0)[0]
    
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        
        if use_blocks and rng.random() < 0.3:
            # Block move: pick a contiguous block, swap its content with another
            block_size = rng.randint(2, 8)
            if n < 2 * block_size + 2:
                continue
            start1 = rng.randint(0, n - block_size)
            seg1 = b[start1:start1+block_size]
            ones_in_seg = np.where(seg1 == 1)[0]
            zeros_in_seg = np.where(seg1 == 0)[0]
            if len(ones_in_seg) == 0 or len(zeros_in_seg) == 0:
                continue
            # Swap within the block
            n_swap = min(len(ones_in_seg), len(zeros_in_seg))
            chosen_ones = rng.choice(ones_in_seg, size=n_swap, replace=False)
            chosen_zeros = rng.choice(zeros_in_seg, size=n_swap, replace=False)
            b[start1 + chosen_ones] = 0
            b[start1 + chosen_zeros] = 1
        else:
            # Point swap
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
        
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            if use_blocks and rng.random() < 0.3:
                # Revert block
                b[start1 + chosen_ones] = 1
                b[start1 + chosen_zeros] = 0
            else:
                b[i] = 1
                b[j] = 0
    
    return best_b, best_c5

def greedy_binary_local_search(b, n, max_sweeps=50, time_limit=None, start_time=None):

    curr_c5 = compute_c5(b)

        improved = False
        ones = np.where(b == 1)[0]

        # For each 1-position, find the best 0-position to swap with
        for i in ones:
            # Try swapping i with each 0-position
            b[i] = 0
            # Compute new correlation once (expensive but only once per i)
            # Actually we need to try each j...
            # Better: compute the change for each j at once
            # Δcorr[k] depends on h[i+k] and h[j+k]

            b[i] = 1
            
            best_j = -1
            best_new_c5 = curr_c5
            
            for j in zeros:
                b[i] = 0

                new_c5 = compute_c5(b)
                b[j] = 0
                b[i] = 1
                if new_c5 < best_new_c5 - 1e-15:

                    best_b = b.copy()
                # Update zeros (j is no longer zero, i is now zero)
                zeros = np.where(b == 0)[0]

def targeted_local_search(b, n, max_sweeps=100, start_time=None, time_limit=None):
    """Targeted local search focusing on the critical lag."""
    curr_c5, corr = compute_c5_corr(b)
    best_c5 = curr_c5
    best_b = b.copy()
    
    for sweep in range(max_sweeps):
        if start_time and time.time() - start_time > time_limit:
            break
        
        improved = False
        cu, corr = compute_c5_corr(b)
        
        # Identify critical lags (within 5% of max)
        max_val = np.max(corr)
        critical = np.where(corr >= 0.99 * max_val)[0]
        
        # Try to reduce overlap at critical lags
        for k in critical:
            # k is the index in the (2n-1) length correlation
            # Lag = k - (n-1)
            lag = k - (n - 1)
            if lag >= 0:
                # corr[k] = sum_{i=0}^{n-1-lag} b[i] * (1-b[i+lag])
                # We want to reduce this
                # Best move: find i where b[i]=1 and b[i+lag]=0, swap with j where b[j]=0
                for i in range(n - lag):
                    if b[i] == 1 and b[i+lag] == 0:
                        # Try setting b[i]=0, b[i+lag]=1
                        b[i] = 0
                        b[i+lag] = 1
                        new_c5 = compute_c5(b)
                        if new_c5 < cu - 1e-15:
                            cu = new_c5
                            improved = True
                            if cu < best_c5:
                                best_c5 = cu
                                best_b = b.copy()
                            # Recompute correlation
                            cu, corr = compute_c5_corr(b)
                            break
                        else:
                            b[i] = 1
                            b[i+lag] = 0
            elif lag < 0:
                nl = -lag
                for i in range(n - nl):
                    if b[i+nl] == 1 and b[i] == 0:
                        b[i] = 1
                        b[i+nl] = 0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_c5_and_corr(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx, corr

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        idx = ones[np.argmin(h[ones])]
        b[idx] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        idx = zeros[np.argmax(h[zeros])]
        b[idx] = 1
    return b

def binary_sa(h_init, n, max_iters, T0, T1, rng):

    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        r = rng.random()
        if r < 0.25:
            bs = 1
            ones = np.where(b == 1)[0]
            zeros = np.where(b == 0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i] = 0
            b[j] = 1
            si, sj = i, j
        elif r < 0.55:
            bs = rng.randint(2, 7)
            if n < bs + 2:
                continue
            s = rng.randint(0, n - bs)
            seg = b[s:s+bs]
            oi = np.where(seg == 1)[0]
            zi = np.where(seg == 0)[0]
            if len(oi) == 0 or len(zi) == 0:
                continue
            nc = min(len(oi), len(zi))
            co = rng.choice(oi, size=nc, replace=False)
            cz = rng.choice(zi, size=nc, replace=False)
            b[s+co] = 0
            b[s+cz] = 1
            si, sj = s, bs

def obj_slsqp(x):
    n = _sln
    T = _slT
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    logsum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + logsum / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_run(h, n, T, maxiter, ftol):
    global _sln, _slT
    _sln = n
    _slT = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_slsqp, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

        b3, c5_3 = binary_sa(best_h, n0, 35000, 0.006, 3e-5, rng)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, b3

    # Phase 4: Greedy local search
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        b4, c5_4 = greedy_local_search(best_h.copy(), n0, 200, start_time, budget_s)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, b4

    # Phase 5: SA from random start
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        rand_b = np.zeros(n0)
        rand_idx = rng.choice(n0, size=n0 // 2, replace=False)
        rand_b[rand_idx] = 1.0
        b5, c5_5 = binary_sa(rand_b, n0, 50000, 0.08, 5e-5, rng)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, b5

    # Phase 6: SLSQP continuous polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        try:
            x_res = slsqp_run(best_h, n0, 80.0, 400, 1e-13)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5, best_h = c5_res, make_feasible(x_res, n0)
        except Exception:
            pass

    # Phase 7: Targeted search after SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        b7 = to_binary(best_h, n0)
        b7, c5_7 = targeted_search(b7, n0, 3000, start_time, budget_s)
        if c5_7 < best_c5:
            best_c5, best_h = c5_7, b7

    # Phase 8: Another SA round
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        b8, c5_8 = binary_sa(best_h, n0, 30000, 0.004, 2e-5, rng)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, b8
        # Follow with targeted
        remaining = budget_s - (time.time() - start_time)
        if remaining > 100:
            b8b, c5_8b = targeted_search(best_h.copy(), n0, 3000, start_time, budget_s)
            if c5_8b < best_c5:
                best_c5, best_h = c5_8b, b8b

    # Phase 9: Higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 500:
        n1 = 1920
        old_x = np.linspace(0.0, 2.0, n0)
        new_x = np.linspace(0.0, 2.0, n1)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        b_up, c5_up = binary_sa(h_up, n1, 40000, 0.006, 3e-5, rng)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, b_up
            remaining = budget_s - (time.time() - start_time)
            if remaining > 200:
                b_up2, c5_up2 = targeted_search(b_up, n1, 2000, start_time, budget_s)
                if c5_up2 < best_c5:
                    best_c5, best_h = c5_up2, b_up2
            remaining = budget_s - (time.time() - start_time)
            if remaining > 150:
                try:
                    x_res = slsqp_run(best_h, n1, 100.0, 300, 1e-13)
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5, best_h = c5_res, make_feasible(x_res, n1)
                except Exception:
                    pass

    # Phase 10: Coordinate descent
    n_pts = len(best_h)
    best_h, best_c5 = coord_descend(best_h.copy(), n_pts, 15, start_time, budget_s, 0.01)
    best_h, best_c5 = coord_descend(best_h.copy(), n_pts, 10, start_time, budget_s, 0.003)

    # Phase 11: Final SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        n_pts = len(best_h)
        try:
            x_res = slsqp_run(best_h, n_pts, 500.0, 250, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5, best_h = c5_res, make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 12: Final binary polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 60:
        n_pts = len(best_h)
        b_fin = to_binary(best_h, n_pts)
        b_fin, c5_fin = targeted_search(b_fin, n_pts, 2000, start_time, budget_s)
        if c5_fin < best_c5:
            best_c5, best_h = c5_fin, b_fin

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
