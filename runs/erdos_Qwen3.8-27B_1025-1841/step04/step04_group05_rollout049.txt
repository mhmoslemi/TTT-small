2. For each position, try flipping it (and adjusting one other to maintain sum)
3. Accept if C5 decreases
4. Also try multi-flip moves (swap two 1s, or flip one 1 to 0 and another 0 to 1)

    return float(np.max(corr) * dx)

    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:

        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)

    grad = np.zeros(n)
    grad[k_eff:] += (1.0 - h[:n - k_eff])
    grad[:n - k_eff] -= h[k_eff:]
    return grad

    """Compute softmax(T * corr) but numerically stable."""
    m = np.max(corr)
    exps = np.exp(T * (corr - m))

    return exps / s

        else:

        if cnt > 0:

            h_new[mask] = np.clip(h_new[mask] + step, 0.0, 1.0)
    return h_new

def make_feasible(h, n):

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def thue_morse(n):
    """Generate Thue-Morse binary sequence of length n."""
    tm = np.zeros(n, dtype=int)

        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm.astype(float)
