        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    T = _G_T
    shifted = T * (corr - mc)
    m = np.max(shifted)
    lse = m + np.log(np.sum(np.exp(shifted - m)))
    return (mc + lse / T) * dx

def smooth_grad(x):
    n = _G_N
    dx = 2.0 / n
    T = _G_T
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    m = np.max(shifted)
    log_w = shifted - m
    w = np.exp(log_w)
    w_sum = np.sum(w)
    weights = w / w_sum
    grad = np.zeros(n)
    active = np.where(weights > 1e-10)[0]
    for idx in active:
        wi = weights[idx]
        k = idx - (n - 1)
        if k >= 0:
            cnt = n - k
            if cnt > 0:
                grad[:cnt] += wi * (1.0 - x[k:k + cnt])
                grad[k:] -= wi * x[:cnt]
        else:
            kk = -k
            cnt = n - kk
            if cnt > 0:
                grad[kk:kk + cnt] += wi * (1.0 - x[:cnt])
                grad[:cnt] -= wi * x[kk:kk + cnt]
    return grad * dx

def eq_con_fun(x):
    return np.sum(x) - _G_N / 2.0

def eq_con_jac(x):
    return np.ones(_G_N)

def slsqp_run(h_start, n_pts, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n_pts
    _G_T = T
    bounds = [(0.0, 1.0)] * n_pts
    cons = [{'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac}]
    try:
        res = minimize(smooth_obj, h_start, jac=smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_start

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def grad_subgrad_step(h, n, lr, top_k):
    dx_val = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_k:]
    grad = np.zeros(n)
    for idx in top_idx:
        k = idx - (n - 1)
        if k >= 0:
            cnt = n - k
            if cnt > 0:
                grad[:cnt] += (1.0 - h[k:k + cnt])
                grad[k:] -= h[:cnt]
        else:
            kk = -k
            cnt = n - kk
            if cnt > 0:
                grad[kk:kk + cnt] += (1.0 - h[:cnt])
                grad[:cnt] -= h[kk:kk + cnt]
    grad *= dx_val / top_k
    h_new = h - lr * grad
    h_new = make_feasible(h_new, n)
    return h_new

def grad_descent_loop(h, n, n_steps, lr_init, decay, top_k):
    best = h.copy()
    best_c = compute_c5(h)
    cur = h.copy()
    for step in range(n_steps):
        lr = lr_init * (decay ** step)
        if lr < 1e-8:
            break
        cur = grad_subgrad_step(cur, n, lr, top_k)
        c = compute_c5(cur)
        if c < best_c - 1e-15:
            best = cur.copy()
            best_c = c
    return best, best_c

def coord_descent_pass(h, n, n_rounds, time_limit):
    start = time.time()
    best = h.copy()
    best_c = compute_c5(h)
    for r in range(n_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            v = best[j]
            cands = [0.0, 1.0, 0.5,
                     max(0.0, v - 0.1), max(0.0, v - 0.05), max(0.0, v - 0.02),
                     max(0.0, v - 0.01), min(1.0, v + 0.01),
                     min(1.0, v + 0.02), min(1.0, v + 0.05), min(1.0, v + 0.1)]
            best_trial = v
            best_cj = best_c
            for cand in cands:
                if abs(cand - v) < 1e-12:
                    continue
                trial = best.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_cj - 1e-14:
                    best_cj = c
                    best_trial = cand
            if best_trial != v:
                best = best.copy()
                best[j] = best_trial
                best = make_feasible(best, n)
                best_c = best_cj
                improved = True
        if not improved:
            break
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(480, 0.5)

    n_init = len(h)
    h = make_feasible(h, n_init)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: SLSQP polish at initial resolution (multiple T)
    for T in [80.0, 200.0, 500.0]:
        if budget_s - (time.time() - t0) < 100:
            break
        h_res = slsqp_run(best_h, n_init, T, 400, 1e-14)
        h_res = make_feasible(h_res, n_init)
        c = compute_c5(h_res)
        if c < best_c5:
            best_c5 = c
            best_h = h_res

    # Phase 2: Subgradient descent refinement
    if budget_s - (time.time() - t0) > 80:
        h_gd, c_gd = grad_descent_loop(best_h, n_init, 3000, 0.004, 0.9995, 5)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd

    # Phase 3: SLSQP after gradient refinement
    if budget_s - (time.time() - t0) > 100:
        h_res = slsqp_run(best_h, n_init, 400.0, 300, 1e-14)
        h_res = make_feasible(h_res, n_init)
        c = compute_c5(h_res)
        if c < best_c5:
            best_c5 = c
            best_h = h_res

    # Phase 4: Upsample to higher resolution and optimize
    n_high = 1920
    if budget_s - (time.time() - t0) > 200:
        h_high = upsample_h(best_h, n_high)
        for T in [150.0, 400.0]:
            if budget_s - (time.time() - t0) < 80:
                break
            h_res = slsqp_run(h_high, n_high, T, 250, 1e-14)
            h_res = make_feasible(h_res, n_high)
            c = compute_c5(h_res)
            if c < best_c5:
                best_c5 = c
                best_h = h_res
                h_high = h_res.copy()

    # Phase 5: Gradient descent at best resolution
    n_cur = len(best_h)
    if budget_s - (time.time() - t0) > 60:
        h_gd, c_gd = grad_descent_loop(best_h, n_cur, 3000, 0.003, 0.9995, 5)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd

    # Phase 6: Second SLSQP polish
    if budget_s - (time.time() - t0) > 100:
        h_res = slsqp_run(best_h, n_cur, 600.0, 250, 1e-14)
        h_res = make_feasible(h_res, n_cur)
        c = compute_c5(h_res)
        if c < best_c5:
            best_c5 = c
            best_h = h_res

    # Phase 7: Coordinate descent
    n_cur = len(best_h)
    if budget_s - (time.time() - t0) > 50:
        remaining_cd = max(20, min(120, int(budget_s - (time.time() - t0) - 60)))
        best_h, best_c5 = coord_descent_pass(best_h, n_cur, 5, remaining_cd)

    # Phase 8: Final SLSQP
    if budget_s - (time.time() - t0) > 60:
        n_cur = len(best_h)
        h_res = slsqp_run(best_h, n_cur, 500.0, 200, 1e-14)
        h_res = make_feasible(h_res, n_cur)
        c = compute_c5(h_res)
        if c < best_c5:
            best_c5 = c
            best_h = h_res

    # Phase 9: Final tight coordinate descent
    n_cur = len(best_h)
    if budget_s - (time.time() - t0) > 20:
        best = best_h.copy()
        best_c = best_c5
        for r in range(10):
            if budget_s - (time.time() - t0) < 5:
                break
            improved = False
            for j in range(n_cur):
                v = best[j]
                cands = [0.0, 1.0, max(0.0, v - 0.02), max(0.0, v - 0.01),
                         max(0.0, v - 0.005), min(1.0, v + 0.005),
                         min(1.0, v + 0.01), min(1.0, v + 0.02)]
                best_trial = v
                best_cj = best_c
                for cand in cands:
                    if abs(cand - v) < 1e-12:
                        continue
                    trial = best.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n_cur)
                    c = compute_c5(trial)
                    if c < best_cj - 1e-14:
                        best_cj = c
                        best_trial = cand
                if best_trial != v:
                    best = best.copy()
                    best[j] = best_trial
                    best = make_feasible(best, n_cur)
                    best_c = best_cj
                    improved = True
            if not improved:
                break
        best_c5 = best_c
        best_h = best

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
