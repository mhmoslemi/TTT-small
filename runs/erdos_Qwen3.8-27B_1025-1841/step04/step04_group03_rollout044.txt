def binary_sa(b, n, max_iters, T0, T1, rng, time_limit, start_time):
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    ones_idx = np.where(b == 1)[0]
    zeros_idx = np.where(b == 0)[0]
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        b[i] = 0
        b[j] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
            ones_idx = np.where(b == 1)[0]
            zeros_idx = np.where(b == 0)[0]
        else:
            b[i] = 1
            b[j] = 0
    return best_b, best_c5

def block_sa(b, n, max_iters, T0, T1, rng, time_limit, start_time):
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = float(it) / max_iters
        T = T0 * (T1 / T0) ** frac
        bsize = rng.randint(2, 6)
        start = rng.randint(0, max(1, n - bsize))
        end = start + bsize
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) == 0 or len(zi) == 0:
            continue
        sc = min(len(oi), len(zi))
        co = rng.choice(oi, size=sc, replace=False)
        cz = rng.choice(zi, size=sc, replace=False)
        b[start + co] = 0
        b[start + cz] = 1
        new_c5 = compute_c5(b)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[start + co] = 1
            b[start + cz] = 0
    return best_b, best_c5

def coord_descent(h, n, max_rounds, start_time, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        improved = False
        for j in range(n):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.02), max(0.0, old - 0.005),
                     min(1.0, old + 0.005), min(1.0, old + 0.02)]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Multi-temp SLSQP (non-periodic)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        best_h, best_c5 = multi_temp_slsqp(best_h, n0, start_time, budget_s, periodic=False)

    # Phase 2: Periodic SLSQP
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        h_per = make_periodic(best_h, n0)
        h_per, c5_per = multi_temp_slsqp(h_per, n0, start_time, budget_s, periodic=True)
        if c5_per < best_c5:
            best_c5 = c5_per
            best_h = h_per.copy()

    # Phase 3: Perturb + SLSQP
    perts = [fourier_perturb, segment_perturb]
    for pf in perts:
        remaining = budget_s - (time.time() - start_time)
        if remaining < 100:
            break
        for _ in range(2):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 50:
                break
            try:
                hp = pf(best_h, n0, rng)
                hr, cr = multi_temp_slsqp(hp, n0, start_time, budget_s, periodic=False)
                if cr < best_c5:
                    best_c5 = cr
                    best_h = hr.copy()
            except Exception:
                pass

    # Phase 4: Binary SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        tl = min(remaining - 20, 150)
        b = (best_h >= 0.5).astype(float)
        b, c5_b = binary_sa(b, n0, 200000, 0.005, 0.000005, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase 5: Block SA
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        tl = min(remaining - 20, 150)
        b = (best_h >= 0.5).astype(float)
        b, c5_b = block_sa(b, n0, 100000, 0.004, 0.00002, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase 6: Upsample to higher resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        n1 = n0 + 480
        old_x = np.linspace(0.0, 2.0, n0, endpoint=False)
        new_x = np.linspace(0.0, 2.0, n1, endpoint=False)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n1)
        h_up, c5_up = multi_temp_slsqp(h_up, n1, start_time, budget_s, periodic=False)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            n0 = n1

    # Phase 7: SA at current resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        n_pts = len(best_h)
        tl = min(remaining - 20, 120)
        b = (best_h >= 0.5).astype(float)
        b, c5_b = binary_sa(b, n_pts, 150000, 0.004, 0.000003, rng, tl, start_time)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = b.copy()

    # Phase 8: Periodic at higher res
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        n_pts = len(best_h)
        h_per = make_periodic(best_h, n_pts)
        h_per, c5_per = multi_temp_slsqp(h_per, n_pts, start_time, budget_s, periodic=True)
        if c5_per < best_c5:
            best_c5 = c5_per
            best_h = h_per.copy()

    # Phase 9: Coordinate descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 30, start_time, budget_s)

    # Phase 10: Final SLSQP polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        n_pts = len(best_h)
        try:
            x_res = slsqp_optimize(best_h, n_pts, 50000.0, 150, 1e-15, periodic=False)
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(x_res, n_pts)
        except Exception:
            pass

    # Phase 11: Final coord descent
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 10, start_time, budget_s)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
