    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0

def coordinate_descent(h, n, n_passes=5, time_check=None, start_time=0, budget=1300):

    deltas = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.05, -0.05]

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(corr - mc, -50.0/T, 0.0)
    exp_s = np.exp(T * shifted)
    weights = exp_s / exp_s.sum()
    
    grad = np.zeros(n)
    for m in range(2*n - 1):
        wm = weights[m]
        if wm < weight_thresh:
            continue
        k = m - (n - 1)
        # First term: (1-h[j-k]) for j where 0≤j-k<n
        js1 = max(0, k)
        je1 = min(n, k + n)
        if js1 < je1:
            idx = np.arange(js1, je1)
            grad[idx] += wm * (1.0 - h[idx - k])
        # Second term: -h[j+k] for j where 0≤j+k<n
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if js2 < je2:
            idx2 = np.arange(js2, je2)
            grad[idx2] -= wm * h[idx2 + k]
    
    grad *= dx
    sv = h.sum() - n / 2.0
    grad += 2.0 * penalty * sv
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _sm_obj(h, T, pen):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = np.clip(corr - mc, -50.0 / T, 0.0)
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx + pen * (h.sum() - n / 2.0) ** 2

    dx = 2.0 / n
