```python
def reweighted_optimize(h0, n, T_start=1000, T_end=5, n_outer=30, T_decay=0.9, max_inner=200):
    dx = 2.0 / n
    h = h0.copy()
    T = T_start
    
    for outer in range(n_outer):
        # Compute current correlation
        corr = compute_corr(h)
        # Compute weights (softmax)
        shifted = np.clip(corr / T, -50, 50)
        w = np.exp(shifted)
        w /= w.sum()
        
        # Inner optimization: minimize sum_k w_k * corr_k + penalty
        def obj(x):
            c = np.correlate(x, 1.0 - x, mode='full')
            return np.dot(w, c) * dx + 100.0 * (x.sum() - n/2.0)**2
        
        res = minimize(obj, h, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': max_inner, 'ftol': 1e-14, 'gtol': 1e-9})
        h = make_feasible(res.x, n)
        T *= T_decay
    
    return h
```

```python
        if c1 < best_c5:
            best_c5, best_h, best_n = c1, h1.copy(), n1

    # Phase 2: Active-set refinement at n=1280
    if rem() > 120:
        h1 = active_set_optimize(best_h, n1, n_rounds=25, topk_frac=0.08)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h, best_n = c1, h1.copy(), n1
        # Second pass with tighter focus
        h1 = active_set_optimize(best_h, n1, n_rounds=20, topk_frac=0.05)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5, best_h, best_n = c1, h1.copy(), n1

    # Phase 3: Binary local search at n=1024
    if rem() > 150:
        n_bin = 1024
        h_b = upsample_h(best_h, n_bin)
        h_bin = make_binary(h_b, n_bin, rng)
        h_bs, c_bs = binary_local_search(h_bin, n_bin, rng, max_rounds=40, n_tries=6000)
        if c_bs < best_c5:
            best_c5, best_h, best_n = c_bs, h_bs.copy(), n_bin
        # Try again from a different random binary init
        if rem() > 80:
            h_bin2 = make_binary(h_b + 0.15 * rng.randn(n_bin), n_bin, rng)
            h_bs2, c_bs2 = binary_local_search(h_bin2, n_bin, rng, max_rounds=30, n_tries=5000)
            if c_bs2 < best_c5:
                best_c5, best_h, best_n = c_bs2, h_bs2.copy(), n_bin

    # Phase 4: Upsample to n=2560 and refine
    n2 = 2560
    if rem() > 150:
        h2 = upsample_h(best_h, n2)
        for T in [500, 250, 100, 50, 25, 15, 10, 7, 5, 3, 2]:
            if rem() < 50:
                break
            h2 = lbgfb_solve(h2, n2, T, mu=300.0, max_inner=150)
            c2 = compute_c5(h2)
            if c2 < best_c5:
                best_c5, best_h, best_n = c2, h2.copy(), n2

    # Phase 5: Active-set at n=2560
    if rem() > 80:
        h2 = active_set_optimize(best_h, n2, n_rounds=15, topk_frac=0.06)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5, best_h, best_n = c2, h2.copy(), n2

    # Phase 6: Multi-start at n=1280
    n_ms = 1280
    if rem() > 80:
        for restart in range(4):
            if rem() < 30:
                break
            h_r = upsample_h(best_h, n_ms) + 0.08 * rng.randn(n_ms)
            h_r = make_feasible(h_r, n_ms)
            for T in [200, 80, 30, 10, 5]:
                if rem() < 20:
                    break
                h_r = lbgfb_solve(h_r, n_ms, T, mu=600.0, max_inner=200)
            c_r = compute_c5(h_r)
            if c_r < best_c5:
                best_c5, best_h, best_n = c_r, h_r.copy(), n_ms

    # Phase 7: Binary search at n=2048
    if rem() > 100:
        n_b3 = 2048
        h_b3 = upsample_h(best_h, n_b3)
        h_bin3 = make_binary(h_b3, n_b3, rng)
        h_bs3, c_bs3 = binary_local_search(h_bin3, n_b3, rng, max_rounds=20, n_tries=4000)
        if c_bs3 < best_c5:
            best_c5, best_h, best_n = c_bs3, h_bs3.copy(), n_b3

    # Phase 8: Final coordinate descent polish
    if rem() > 30:
        n_cd = len(best_h)
        h_cd, c_cd = coord_polish(best_h, n_cd, max_rounds=4,
                                  step_list=[0.015, -0.015, 0.007, -0.007, 0.003, -0.003])
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Phase 9: Very fine T polishing at best resolution
    if rem() > 25:
        n_fin = len(best_h)
        for T in [0.5, 0.3, 0.2, 0.1]:
            if rem() < 10:
                break
            h_f = lbgfb_solve(best_h, n_fin, T, mu=1000.0, max_inner=200)
            c_f = compute_c5(h_f)
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f.copy()

    # Final feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
