import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-9
        else:
            mask = h > 1e-9
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max(corr, T):
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted = np.clip(shifted, -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_optimize(h_start, n, T, mu, max_iter):

        corr = np.correlate(h, 1.0 - h, mode='full')
        sm = smooth_max(corr, T) * dx
        viol = np.sum(h) - n / 2.0
        return sm + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

    best_c = current_c
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / n_iter)
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        new_c, _ = compute_c5(h)
        delta = new_c - current_c
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c = new_c
            ones = np.delete(ones, ii)
            zeros = np.delete(zeros, jj)
            ones = np.append(ones, oj)
            zeros = np.append(zeros, oi)
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best_c

    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -80, 80)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def lbgfb_optimize(h_start, n, T, mu, max_iter):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        sm = smooth_max(corr, T) * dx
        viol = np.sum(h) - n / 2.0
        return sm + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-10})
    return make_feasible(res.x, n)

def fix_binary(h, n, rng):
    h = (h > 0.5).astype(float)
    n_ones = int(np.sum(h))
    n_target = n // 2
    if n_ones > n_target:
        oi = np.where(h == 1)[0]
        perm = rng.permutation(len(oi))
        for k in range(n_ones - n_target):
            h[oi[perm[k]]] = 0.0
    elif n_ones < n_target:
        zi = np.where(h == 0)[0]
        perm = rng.permutation(len(zi))
        for k in range(n_target - n_ones):
            h[zi[perm[k]]] = 1.0
    return h

def binary_sa(h_bin, n, rng, T_start, T_end, n_iter):
    h = h_bin.copy()
    current_c = compute_c5(h)
    best_c = current_c
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = int(ones[ii]), int(zeros[jj])
        h[oi] = 0.0
        h[oj] = 1.0
        new_c = compute_c5(h)
        delta = new_c - current_c
        if delta < 0 or rng.rand() < np.exp(-min(delta / max(T, 1e-12), 50)):
            current_c = new_c
            ones[ii] = oj
            zeros[jj] = oi
            if current_c < best_c:
                best_c = current_c
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best_c

        best_c5 = final_c5
