def project_feasible(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Find lambda such that sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0

def binary_sa_optimize(h, n, n_iters, T0, T1, rng, start_time, time_limit):

    # Ensure exactly n/2 ones
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        b[zeros[rng.randint(len(zeros))]] = 1
    
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5

        T = T0 * (T1 / T0) ** frac

        start = rng.randint(0, n - bsize)
        end = start + bsize
        seg = b[start:end]

        if len(ones_idx) == 0 or len(zeros_idx) == 0:

            if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):

        else:

            chosen_o = rng.choice(ones_idx, size=nswap, replace=False)
            chosen_z = rng.choice(zeros_idx, size=nswap, replace=False)
            b[start + chosen_o] = 0

def coord_descent(h, n, max_rounds, start_time, time_limit):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        if time.time() - start_time > time_limit - 10:
            break
        for j in range(n):
            if time.time() - start_time > time_limit - 5:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01),
                     max(0.0, old - 0.001), min(1.0, old + 0.001)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    curr = np.sum(h)
    if abs(curr - target) < 1e-14:
        return np.clip(h, 0.0, 1.0)
    lo, hi = -5.0, 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n

    m = np.max(corr)

    shifted -= np.max(shifted)

    g = 1.0 - h

                    best_b = b.copy()
            else:
                b[start + co] = 1
                b[start + cz] = 0
    return best_b, best_c5

def continuous_sa(h, n, n_iters, T0, T1, start_time, time_limit, rng):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    curr_c5 = best_c5
    for it in range(n_iters):
        if time.time() - start_time > time_limit:
            break
        frac = it / max(n_iters - 1, 1)
        T = T0 * (T1 / T0) ** frac
        h_new = h.copy()
        n_change = rng.randint(1, 5)
        idx = rng.choice(n, size=n_change, replace=False)
        h_new[idx] = rng.random(size=n_change)
        h_new = project_feasible(h_new, n)
        new_c5 = compute_c5(h_new)
        if new_c5 <= curr_c5 or (T > 1e-12 and rng.random() < np.exp(-(new_c5 - curr_c5) / T)):
            h = h_new
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
    return best_h, best_c5

def coord_descent(h, n, max_rounds, start_time, time_limit):
    best_c5 = compute_c5(h)
    improved = True
    rnd = 0
    while improved and rnd < max_rounds:
        rnd += 1
        improved = False
        if time.time() - start_time > time_limit - 5:
            break
        for j in range(n):
            if time.time() - start_time > time_limit - 3:
                break
            old = h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01),
                     max(0.0, old - 0.001), min(1.0, old + 0.001)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
    return h, best_c5

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return make_feasible(h_old.copy(), n_new)
    old_x = np.linspace(0.0, 2.0, n_old)
    new_x = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(new_x, old_x, h_old)
    return make_feasible(h_new, n_new)

def run(seed=42, budget_s=1300, **kwargs):
    global rng_global
    start_time = time.time()
    time_limit = start_time + budget_s - 25
    rng = np.random.RandomState(seed)
    rng_global = np.random.RandomState(seed + 1)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(960, 0.5)
    n0 = len(h0)
    h0 = make_feasible(h0, n0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Binary SA on parent
    b1, c5_1 = binary_sa_opt(h0, n0, 35000, 0.01, 0.00001, rng, start_time, time_limit - 300)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = b1.copy()

    # Phase 2: Projected gradient refinement
    rem = budget_s - (time.time() - start_time)
    if rem > 120:
        h2, c5_2 = projected_gradient_opt(best_h.copy(), n0, 20.0, 400.0, 2000, start_time, time_limit - 100)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Continuous SA for exploration
    rem = budget_s - (time.time() - start_time)
    if rem > 100:
        h3, c5_3 = continuous_sa(best_h.copy(), n0, 20000, 0.02, 0.001, start_time, time_limit - 80, rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        rem = budget_s - (time.time() - start_time)
        if rem > 80:
            h3b, c5_3b = projected_gradient_opt(best_h.copy(), n0, 50.0, 800.0, 1000, start_time, time_limit - 40)
            if c5_3b < best_c5:
                best_c5 = c5_3b
                best_h = h3b.copy()

    # Phase 4: Multi-resolution search
    for n_r in [480, 720, 1200, 1440]:
        rem = budget_s - (time.time() - start_time)
        if rem < 120:
            break
        h_r = upsample(best_h, n_r)
        b_r, c5_br = binary_sa_opt(h_r, n_r, 15000, 0.006, 0.00001, rng, start_time, time_limit - 60)
        if c5_br < best_c5:
            best_c5 = c5_br
            best_h = b_r.copy()
        rem = budget_s - (time.time() - start_time)
        if rem > 80:
            h_cg, c5_cg = projected_gradient_opt(best_h.copy(), n_r, 30.0, 600.0, 1000, start_time, time_limit - 30)
            if c5_cg < best_c5:
                best_c5 = c5_cg
                best_h = h_cg.copy()

    # Phase 5: Second binary SA round on best
    rem = budget_s - (time.time() - start_time)
    if rem > 80:
        n_pts = len(best_h)
        b5, c5_5 = binary_sa_opt(best_h, n_pts, 25000, 0.004, 0.000005, rng, start_time, time_limit - 50)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = b5.copy()

    # Phase 6: High-temperature projected gradient
    rem = budget_s - (time.time() - start_time)
    if rem > 50:
        n_pts = len(best_h)
        h6, c5_6 = projected_gradient_opt(best_h.copy(), n_pts, 100.0, 2000.0, 1000, start_time, time_limit - 20)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Coordinate descent
    rem = budget_s - (time.time() - start_time)
    if rem > 30:
        n_pts = len(best_h)
        best_h, best_c5 = coord_descent(best_h.copy(), n_pts, 5, start_time, time_limit - 5)

    # Phase 8: Final projected gradient with very high T
    rem = budget_s - (time.time() - start_time)
    if rem > 40:
        n_pts = len(best_h)
        h8, c5_8 = projected_gradient_opt(best_h.copy(), n_pts, 500.0, 5000.0, 500, start_time, time_limit - 5)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
