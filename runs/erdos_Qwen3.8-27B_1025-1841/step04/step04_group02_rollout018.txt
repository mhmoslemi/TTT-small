So ∑_m c[m] = ∑ᵢ h[i] · n/2 = (n/2)·(n/2) = n²/4

There are 2n-1 values of m, so the average is n²/4 / (2n-1) ≈ n/8.
This gives a lower bound: max c[m] ≥ n/8, so C₅ ≥ (n/8)·(2/n) = 1/4 = 0.25.

We're at 0.38, so there's room for improvement down to 0.25 in theory.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h.astype(float), 1.0 - h.astype(float), mode='full')
    return float(np.max(corr)) * dx

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = (h >= 0.5).astype(np.int8)
    target = n // 2
    current = int(np.sum(b))
    if current > target:
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])[:current - target]
        b[ones[order]] = 0
    elif current < target:
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])[:target - current]
        b[zeros[order]] = 1
    return b

def fast_sa_step(c, b, a, bb, n, rng):
    """Compute delta correlation for swapping b[a]:1->0, b[bb]:0->1.
    Returns (delta, new_max) or None if invalid."""
    # Δc[m + n - 1] for m in range [-(n-1), n-1]
    # = -(1-b[a-m]) for a-m in [0,n-1]
    #   + b[a+m] for a+m in [0,n-1]
    #   + (1-b[b-m]) for b-m in [0,n-1]
    #   - b[b+m] for b+m in [0,n-1]
    
    # Range for a-m valid: m in [a-(n-1), a]
    m1_start = max(a - n + 1, -(n-1))
    m1_end = min(a, n-1)
    
    # Range for a+m valid: m in [-a, n-1-a]
    m2_start = max(-a, -(n-1))
    m2_end = min(n-1-a, n-1)
    
    # Range for bb-m valid: m in [bb-(n-1), bb]
    m3_start = max(bb - n + 1, -(n-1))
    m3_end = min(bb, n-1)
    
    # Range for bb+m valid: m in [-bb, n-1-bb]
    m4_start = max(-bb, -(n-1))
    m4_end = min(n-1-bb, n-1)
    
    # Build delta array
    # Instead of full array, compute new_c = c + delta
    # But we need max(new_c), so we need the full array
    
    delta = np.zeros(2*n-1)
    
    if m1_end >= m1_start:
        ms = np.arange(m1_start, m1_end + 1)
        delta[ms + (n-1)] -= (1.0 - b[a - ms].astype(np.float64))
    
    if m2_end >= m2_start:
        ms = np.arange(m2_start, m2_end + 1)
        delta[ms + (n-1)] += b[a + ms].astype(np.float64)
    
    if m3_end >= m3_start:
        ms = np.arange(m3_start, m3_end + 1)
        delta[ms + (n-1)] += (1.0 - b[bb - ms].astype(np.float64))
    
    if m4_end >= m4_start:
        ms = np.arange(m4_start, m4_end + 1)
        delta[ms + (n-1)] -= b[bb + ms].astype(np.float64)
    
    new_c = c + delta
    return new_c

def fast_binary_sa(h_init, n, max_iters, T0, T1, rng, start_time, deadline):
    """Fast SA with O(n) per step."""
    b = to_binary(h_init, n).copy()
    c = np.correlate(b.astype(np.float64), 1.0 - b.astype(np.float64), mode='full')
    curr_max = float(np.max(c))
    best_max = curr_max
    best_b = b.copy()
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        frac = float(min(it, max_iters - 1)) / max(max_iters - 1, 1)
        T = T0 * (T1 / T0) ** frac
        
        # Pick random a (currently 1) and bb (currently 0)
        # Use reservoir or precomputed lists
        ones = np.where(b == 1)[0]
        zeros = np.where(b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        a = int(ones[rng.randint(len(ones))])
        bb = int(zeros[rng.randint(len(zeros))])
        if a == bb:
            continue
        
        new_c = fast_sa_step(c, b, a, bb, n, rng)
        new_max = float(np.max(new_c))
        
        if new_max <= curr_max or (T > 1e-12 and rng.random() < np.exp(-(new_max - curr_max) / T)):
            c = new_c
            curr_max = new_max
            b[a] = 0
            b[bb] = 1
            if new_max < best_max:
                best_max = new_max
                best_b = b.copy()
        
        if it % 50000 == 0 and it > 0:
            # Verify periodically
            c_check = np.correlate(b.astype(np.float64), 1.0 - b.astype(np.float64), mode='full')
            if abs(float(np.max(c_check)) - curr_max) > 1e-6:
                c = c_check
                curr_max = float(np.max(c))
    
    return best_b, best_max * 2.0 / n

def slsqp_obj(x, n, tmp):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(tmp * shifted))) / tmp) * dx

def slsqp_con(x):
    return np.sum(x) - len(x) / 2.0

def slsqp_con_jac(x):
    return np.ones(len(x))

def slsqp_refine(h, T, maxiter, ftol):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': slsqp_con, 'jac': slsqp_con_jac}]
    try:
        res = minimize(slsqp_obj, h.copy(), args=(n, T), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(res.x, n)
    except Exception:
        return h
