import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_feasible(v, n):
    target = n / 2.0
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h_proj = np.clip(v + mid, 0.0, 1.0)
        s = np.sum(h_proj)
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(v + (lo + hi) / 2.0, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_obj_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    obj = (mc + np.log(sum_exp) / T) * dx
    g = np.zeros(n)
    significant = np.where(weights > 1e-10)[0]
    for k in significant:
        w = weights[k]
        m = k - (n - 1)
        if m >= 0:
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] -= w * h[:n - m]
        else:
            mp = -m
            g[mp:] += w * (1.0 - h[:n - mp])
            g[:n - mp] -= w * h[mp:]
    return obj, g * dx

def proj_grad_descent(h, n, T_start, T_end, n_iter, lr0, momentum, start_time, budget):
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - start_time > budget - 30:
            break
        T_now = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        lr = lr0 * (0.9999 ** it)
        obj, grad = soft_max_obj_grad(h, n, T_now)
        v = momentum * v - lr * grad
        h = project_onto_feasible(h + v, n)
        if (it + 1) % 500 == 0:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5

def binary_local_search(h, n, max_swaps, start_time, budget):
    h_bin = np.where(h >= 0.5, 1.0, 0.0)
    h_bin = make_feasible(h_bin, n)
    best_c = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(123)
    ones = np.where(best_h == 1.0)[0]
    zeros = np.where(best_h == 0.0)[0]
    count = 0
    while count < max_swaps:
        if time.time() - start_time > budget - 15:
            break
        i1 = ones[rng.randint(len(ones))]
        i2 = zeros[rng.randint(len(zeros))]
        if i1 == i2:
            count += 1
            continue
        best_h[i1] = 0.0
        best_h[i2] = 1.0
        c = compute_c5(best_h)
        if c < best_c - 1e-14:
            best_c = c
            ones = np.where(best_h == 1.0)[0]
            zeros = np.where(best_h == 0.0)[0]
        else:
            best_h[i1] = 1.0
            best_h[i2] = 0.0
        count += 1
    return best_h, best_c

def coord_descent(h, n, n_rounds, start_time, budget):
    best_c = compute_c5(h)
    for rnd in range(n_rounds):
        if time.time() - start_time > budget - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget - 10:
                return h, best_c
            old = h[j]
            best_trial = old
            bc = best_c
            for cand in (0.0, 1.0):
                if abs(cand - old) < 1e-12:
                    continue
                h[j] = cand
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < bc - 1e-14:
                    bc = c
                    best_trial = cand
            if best_trial != old:
                h[j] = best_trial
                h = make_feasible(h, n)
                best_c = bc
                improved = True
            else:
                h[j] = old
                h = make_feasible(h, n)
        if not improved:
            break
    return h, best_c

def slsqp_polish(h, n, maxiter, start_time, budget):
    if time.time() - start_time > budget - 30:
        return h, compute_c5(h)
    T = 200.0
    dx = 2.0 / n

    def obj_fn(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj_fn, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_new = make_feasible(res.x, n)
        c_new = compute_c5(h_new)
        c_old = compute_c5(h)
        if c_new < c_old:
            return h_new, c_new
    except Exception:
        pass
    return h, compute_c5(h)
