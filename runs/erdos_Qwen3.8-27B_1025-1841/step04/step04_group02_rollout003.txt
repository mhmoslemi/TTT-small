_sl_n = 0
_sl_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    b = np.zeros(n)
    order = np.argsort(h)[::-1]
    b[order[:n // 2]] = 1
    return b

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _obj_func(x):
    n = _sl_n
    T = _sl_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

    n = len(h)

    _sl_T = T

                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass

    # Phase 5: SA from best so far
    if time.time() < deadline - 120:
        n_cur = len(best_h)
        b_start3 = to_binary(best_h, n_cur)
        b_out3, c5_out3 = sa_binary(b_start3, n_cur, 150000, 0.008, 0.00001, rng, deadline - 80)
        if c5_out3 < best_c5:
            best_c5 = c5_out3
            best_h = b_out3.copy()

    # Phase 6: Coordinate descent
    n_final = len(best_h)
    if time.time() < deadline - 15:
        best_h, best_c5 = coord_descent(best_h.copy(), n_final, 15, deadline - 3)

    # Phase 7: Final SLSQP
    if time.time() < deadline - 20:
        try:
            x_res = slsqp_run(best_h, 3.0, 200, 1e-15)
            x_res = make_feasible(x_res, len(x_res))
            c5_res = compute_c5(x_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = x_res.copy()
        except Exception:
            pass

    # Phase 8: Try upsampled resolution if time remains
    if time.time() < deadline - 200:
        n_up = min(len(best_h) + 480, 1920)
        h_up = upsample_h(best_h, n_up)
        b_up = to_binary(h_up, n_up)
        b_up_o, c5_up = sa_binary(b_up, n_up, 60000, 0.008, 0.00002, rng, deadline - 100)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = b_up_o.copy()
            if time.time() < deadline - 40:
                try:
                    x_res = slsqp_run(best_h, 10.0, 150, 1e-14)
                    x_res = make_feasible(x_res, len(x_res))
                    c5_res = compute_c5(x_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = x_res.copy()
                except Exception:
                    pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
