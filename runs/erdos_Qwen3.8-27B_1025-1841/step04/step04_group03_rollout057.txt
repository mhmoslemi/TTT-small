import numpy as np
from scipy.optimize import minimize
import time

_GLOB_N = 0
_GLOB_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.any(mask):
            idx = np.where(mask)[0][np.argmax(np.abs(h[mask] - 0.5))]
            h[idx] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def obj_smooth(x):
    n = _GLOB_N
    T = _GLOB_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_step(h, n, T, maxiter, ftol):
    global _GLOB_N, _GLOB_T
    _GLOB_N = n
    _GLOB_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    res = minimize(obj_smooth, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def project_slice(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2.0
    proj = np.clip(h - tau, 0.0, 1.0)
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-13:
        if diff > 0:
            mask = proj < 1.0
        else:
            mask = proj > 0.0
        if np.any(mask):
            w = np.where(mask)[0]
            idx = w[np.argmax(np.abs(proj[w] - 0.5))]
            proj[idx] += diff
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def logsumexp_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    log_vals = T * (corr - mc)
    log_vals -= np.max(log_vals)
    exp_vals = np.exp(log_vals)
    p = exp_vals / np.sum(exp_vals)
    p = p * dx
    grad = np.zeros(n)
    k_range = np.arange(-(n-1), n)
    for m in range(2*n-1):
        if p[m] < 1e-15:
            continue
        k = m - (n-1)
        term1 = np.zeros(n)
        term2 = np.zeros(n)
        if k >= 0:
            term1[:n-k] = 1.0 - h[k:]
        else:
            term1[-k:] = 1.0 - h[:n+k]
        if k <= 0:
            term2[max(0,-k):] = h[:max(0,n+k)] if k<=0 else h[-k:]
        else:
            term2[k:] = h[:n-k]
        grad += p[m] * (term1 - term2)
    return grad, mc + np.log(np.sum(np.exp(log_vals)) + 1e-300) / T

def proj_grad_descent(h, n, T, lr, n_iter, start_time, budget):
    target = n / 2.0
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        if time.time() - start_time > budget:
            break
        grad, _ = logsumexp_grad(h, T)
        h_new = h - lr * grad
        h_proj = project_slice(h_new, target)
        c5_new = compute_c5(h_proj)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_proj.copy()
        h = h_proj
    return best_h, best_c5

def binary_sa_fast(h, n, max_iters, T0, T1, rng, time_limit, start_time):

def segment_perturb(h, n, rng, n_seg_min=3, n_seg_max=10):
    h_new = h.copy()
    n_segs = rng.randint(n_seg_min, n_seg_max)
    cut = np.sort(rng.randint(1, n, size=n_segs-1))
    cut = np.concatenate([[0], cut, [n]])
    for i in range(n_segs):
        s, e = cut[i], cut[i+1]
        val = rng.uniform(0, 1)
        h_new[s:e] = val
    return make_feasible(h_new, n)

def fourier_perturb(h, n, rng, n_modes=5, amp=0.05):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, n // 5)
        phase = rng.uniform(0, 2 * np.pi)
        a = amp * rng.uniform(0.3, 2.0)
        h_new += a * np.sin(2.0 * np.pi * freq * x / n + phase)
    return make_feasible(h_new, n)

def blend_perturb(h, n, rng):
    s = rng.randint(1, n // 6)
    h_shift = np.roll(h, s)
    alpha = rng.uniform(0.3, 0.7)
    h_new = alpha * h + (1 - alpha) * h_shift
    return make_feasible(h_new, n)

def coord_descent_fine(h, n, max_rounds, start_time, budget):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):

            old = h[j]
            best_c = best_c5
            best_val = old

                          old - 0.05, old + 0.05]
            for cand in candidates:
                cand = max(0.0, min(1.0, cand))

                    continue
                trial = h.copy()
                trial[j] = cand

                    best_val = cand
            if abs(best_val - old) > 1e-12:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def multi_temp_refine(h, n, start_time, budget):
    temps = [50.0, 200.0, 1000.0, 5000.0, 20000.0, 100000.0]
    for T in temps:
        remaining = budget - (time.time() - start_time)
        if remaining < 30:
            break
        try:
            x_res = slsqp_step(h, n, T, 120, 1e-14)
            c5_res = compute_c5(x_res)
            if c5_res < compute_c5(h):
                h = make_feasible(x_res, n)
        except Exception:
            pass
    return h, compute_c5(h)

def upsample(h, n_old, n_new):
    old_x = np.linspace(0.0, 2.0, n_old)
    new_x = np.linspace(0.0, 2.0, n_new)
    h_up = np.interp(new_x, old_x, h)
    return make_feasible(h_up, n_new)
