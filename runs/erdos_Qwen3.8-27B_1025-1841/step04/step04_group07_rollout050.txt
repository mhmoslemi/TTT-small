def gradient_of_corr(h, tau, n):
    """Gradient of correlation at lag tau w.r.t. h."""
    grad = np.zeros(n)
    if tau >= 0:
        # j in [tau, n-1]: contribution (1-h[j-tau])
        j_range = np.arange(tau, n)
        grad[j_range] += (1.0 - h[j_range - tau])
        # j in [0, n-1-tau]: contribution -h[j+tau]
        j_range2 = np.arange(0, n - tau)
        grad[j_range2] -= h[j_range2 + tau]
    else:
        m = -tau
        # j in [0, n-1-m]: contribution (1-h[j+m])
        j_range = np.arange(0, n - m)
        grad[j_range] += (1.0 - h[j_range + m])
        # j in [m, n-1]: contribution -h[j-m]
        j_range2 = np.arange(m, n)
        grad[j_range2] -= h[j_range2 - m]
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-13
        if not mask.any():
            break
        step = diff / max(mask.sum(), 1)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def grad_corr(h, tau, n):
    grad = np.zeros(n)
    if tau >= 0:
        if tau < n:
            j1 = np.arange(tau, n)
            grad[j1] += (1.0 - h[j1 - tau])
            j2 = np.arange(0, n - tau)
            grad[j2] -= h[j2 + tau]
    else:
        m = -tau
        if m < n:
            j1 = np.arange(0, n - m)
            grad[j1] += (1.0 - h[j1 + m])
            j2 = np.arange(m, n)
            grad[j2] -= h[j2 - m]
    return grad

def subgradient_c5(h, n, topk=5):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    k = min(topk, len(corr))
    idx = np.argpartition(corr, -k)[-k:]
    weights = corr[idx] - mc + 1.0
    weights = weights ** 2
    weights /= weights.sum()
    grad = np.zeros(n)
    for wi, ki in zip(weights, idx):
        tau = int(ki) - (n - 1)
        grad += wi * grad_corr(h, tau, n)
    return grad * dx, mc * dx

def subgradient_minmax(h0, n, max_iters=300, topk=5):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    patience = 0
    for it in range(max_iters):
        grad, c5_val = subgradient_c5(h, n, topk)
        if c5_val < best_c5 - 1e-12:
            best_c5 = c5_val
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        gn = np.linalg.norm(grad) + 1e-12
        lr = 2.0 * best_c5 / ((it + 1) ** 0.7) / gn
        lr = min(lr, 0.05)
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        if c_new <= best_c5 + 1e-10:
            h = h_new
        else:
            lr *= 0.5
            h_new2 = h - lr * grad
            h_new2 = make_feasible(h_new2, n)
            if compute_c5(h_new2) <= best_c5 + 1e-10:
                h = h_new2
        if patience > 80:
            break
    return best_h, best_c5

def lbgfb_power(h0, n, p, mu, max_iter=250):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        c = np.maximum(corr, 0.0)
        return (np.sum(c ** p)) ** (1.0 / p) * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def lbgfb_smax(h0, n, T, mu, max_iter=250):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -80, 80)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx + mu * (x.sum() - target) ** 2
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

            h_new = make_feasible(res.x, n)
            c_new = compute_c5(h_new)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            h = h_new
        except Exception:
            pass
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def swap_local_search(h0, n, rng, max_rounds=40, tries=8000):
    h_bin = (h0 > 0.5).astype(float)
    target_ones = n // 2
    n_ones = int(h_bin.sum())
    if n_ones > target_ones:
        idx = np.where(h_bin == 1)[0]
        rng.shuffle(idx)
        h_bin[idx[:n_ones - target_ones]] = 0.0
    elif n_ones < target_ones:
        idx = np.where(h_bin == 0)[0]
        rng.shuffle(idx)
        h_bin[idx[:target_ones - n_ones]] = 1.0
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    dx = 2.0 / n
    for rnd in range(max_rounds):
        improved = False
        for _ in range(min(tries, len(ones) * len(zeros))):
            if len(ones) == 0 or len(zeros) == 0:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def coord_descent(h0, n, max_rounds=5, steps=None):
    if steps is None:
        steps = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur
