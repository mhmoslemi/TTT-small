# Top-level parameters (will be set in run)
_T_CUR = 100.0
_N_CUR = 960

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj(x):
    n = _N_CUR
    T = _T_CUR
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.clip(shifted, -min(50.0, 40.0/T), 0.0)
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def block_obj(params):
    n = _N_CUR
    widths = np.maximum(params, 1.0)
    h = blocks_to_h(widths, n)
    return compute_c5(h)

def blocks_to_h(widths, n):
    h = np.zeros(n)
    pos = 0
    for i, w in enumerate(widths):
        w_int = max(1, int(round(w)))
        if pos + w_int > n:
            w_int = n - pos
        if w_int <= 0:
            break
        if i % 2 == 0:
            h[pos:pos+w_int] = 1.0
        pos += w_int
    if pos < n:
        h[pos:] = 0.0
    return make_feasible(h, n)

def periodic_h(pattern, n):
    p = len(pattern)
    h = np.tile(pattern, n // p + 1)[:n]
    return make_feasible(h, n)
