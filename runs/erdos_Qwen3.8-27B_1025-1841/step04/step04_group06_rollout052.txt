def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()

    for _ in range(200):

            break

        else:

        if cnt == 0:
            break

        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h


def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)


def lbe_opt(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = np.clip(sh, -100, 100)
        soft = (mc + np.log(np.sum(np.exp(sh))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13})
    return make_feasible(res.x, n)


def pnorm_opt(h_start, n, p, mu, max_iter=200):
    dx = 2.0 / n
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        pos = np.maximum(corr, 0.0)
        val = (np.sum(pos ** p) / len(pos)) ** (1.0 / p)
        return val * dx + mu * (np.sum(h) - n / 2.0) ** 2
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)


def slsqp_opt(h_start, n, T, max_iter=100):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        sh = T * (corr - mc)
        sh = np.clip(sh, -80, 80)
        return (mc + np.log(np.sum(np.exp(sh))) / T) * dx
    def con(h):
        return np.sum(h) - n / 2.0
    def jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

            h_bin = (h_c > 0.5).astype(float)
            n_ones = int(np.sum(h_bin))
            n_target = n3 // 2
            if n_ones > n_target:
                idx = np.where(h_bin == 1)[0]
                for k in range(n_ones - n_target):
                    h_bin[idx[k]] = 0.0
            elif n_ones < n_target:
                idx = np.where(h_bin == 0)[0]
                for k in range(n_target - n_ones):
                    h_bin[idx[k]] = 1.0
            h_sa, c5_sa = binary_sa(h_bin, n3, rng, max_rounds=50, T0=0.2, T1=0.002, n_tries=1000)
            if c5_sa < best_c5:
                best_c5 = c5_sa
                best_h = h_sa.copy()

    # Phase 6: Final SLSQP polish at manageable resolution
    if tl() > 20:
        n_f = len(best_h)
        if n_f > 1200:
            h_s = upsample_h(best_h, 1024)
        else:
            h_s = best_h.copy()
            n_f = len(h_s)
        try:
            h_opt = slsqp_opt(h_s, n_f, 3000, max_iter=80)
            c5o = compute_c5(h_opt)
            if c5o < best_c5:
                if n_f < len(best_h):
                    h_opt = upsample_h(h_opt, len(best_h))
                    c5o = compute_c5(h_opt)
                if c5o < best_c5:
                    best_c5 = c5o
                    best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 7: Coordinate descent on best solution
    if tl() > 15:
        n_cd = len(best_h)
        h_cd = best_h.copy()
        c5_cd = best_c5
        deltas = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01, 0.005, -0.005]
        for rnd in range(4):
            if tl() < 8:
                break
            improved = False
            step_j = max(1, n_cd // 512)
            for j in range(0, n_cd, step_j):
                if tl() < 5:
                    break
                for d in deltas:
                    nv = h_cd[j] + d
                    if nv < 0 or nv > 1:
                        continue
                    old = h_cd[j]
                    h_cd[j] = nv
                    h_cd = make_feasible(h_cd, n_cd)
                    c = compute_c5(h_cd)
                    if c < c5_cd - 1e-14:
                        c5_cd = c
                        improved = True
                        break
                    else:
                        h_cd[j] = old
                        h_cd = make_feasible(h_cd, n_cd)
            if not improved:
                break
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
