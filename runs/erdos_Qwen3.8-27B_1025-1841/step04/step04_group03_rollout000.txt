def slsqp_optimize(h, n, T, maxiter, ftol):

        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

        return np.sum(x) - len(x) / 2.0

    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _grad_single_shift(h, k_idx):
    n = len(h)
    dx = 2.0 / n
    d = n - 1 - k_idx
    grad = np.zeros(n)
    if d >= 0:
        grad[:n-d] += dx * (1.0 - h[d:n])
        grad[d:] -= dx * h[:n-d]
    else:
        e = -d
        grad[e:] += dx * (1.0 - h[:n-e])
        grad[:n-e] -= dx * h[e:]
    return grad

def proj_subgrad(h, n, alpha, steps, start_time, budget_s):
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    for s in range(steps):
        if time.time() - start_time > budget_s:
            break
        dx = 2.0 / n
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        top_k = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for k_idx in top_k:
            grad += _grad_single_shift(h_curr, k_idx)
        h_new = h_curr - alpha * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            h_curr = h_new
        else:
            alpha *= 0.93
    return h_curr, best_c5

def _con_sum(x):
    return np.sum(x) - len(x) / 2.0

def _con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_sum, 'jac': _con_sum_jac}]
    res = minimize(obj, h.copy(), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def fast_local(h, n, start_time, budget_s):
    best_c5 = compute_c5(h)
    alpha = 0.02
    for phase in range(4):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        h, c5 = proj_subgrad(h, n, alpha, 300, start_time, budget_s)
        if c5 < best_c5:
            best_c5 = c5
        alpha *= 0.6
    remaining = budget_s - (time.time() - start_time)

        for T in [80.0, 400.0, 2000.0]:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 15:

            try:
                x_res = slsqp_optimize(h, n, T, 80, 1e-14)
                c5_res = compute_c5(x_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    h = make_feasible(x_res, n)
            except Exception:
                pass
    return h, best_c5

def diverse_perturb(h, n, rng, intensity=0.4):
    h_p = h.copy()
    n_flips = max(1, int(n * intensity * 0.05))
    for _ in range(n_flips):
        blen = rng.randint(n//20, n//6)
        bstart = rng.randint(0, n - blen)
        h_p[bstart:bstart+blen] = 1.0 - h_p[bstart:bstart+blen]
    n_rand = max(1, int(n * intensity * 0.02))
    for _ in range(n_rand):
        idx = rng.randint(n)
        h_p[idx] = rng.uniform(0, 1)
    return make_feasible(h_p, n)
