from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = min(max(h[idx] + diff, 0.0), 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def thue_morse(n):
    """Generate Thue-Morse sequence of length n (padded/truncated to n)."""
    # Find the next power of 2 >= n
    m = 1
    while m < n:
        m *= 2
    tm = np.zeros(m)
    for i in range(1, m):
        # t(i) = t(i >> 1) ^ (i & 1)
        pass
    # More efficient:
    tm = np.zeros(m, dtype=np.float64)
    for i in range(1, m):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return tm[:n]

def lbgfb_smooth(h_start, n, T, mu, max_iter=200):
    dx = 2.0 / n
    h0 = h_start.copy()
    
    def f(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -100, 100)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(h) - n / 2.0
        return soft + mu * viol * viol
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

    h = h_bin.copy()
    current = compute_c5(h)
    best = current
    best_h = h.copy()
    
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    
    for step in range(n_steps):
        T = T_start * (T_end / T_start) ** (step / max(n_steps - 1, 1))
        
        # Choose move
        r = rng.random()
        if r < 0.2 and len(ones) > 5 and len(zeros) > 5:
            # Block swap
            bs = rng.randint(2, 6)
            if len(ones) < bs or len(zeros) < bs:
                bs = min(bs, len(ones), len(zeros))
            if bs < 1:
                continue
            idx_o = rng.randint(len(ones) - bs)
            idx_z = rng.randint(len(zeros) - bs)
            o_block = ones[idx_o:idx_o+bs]
            z_block = zeros[idx_z:idx_z+bs]
            
            h[o_block] = 0.0
            h[z_block] = 1.0
            new_c = compute_c5(h)
            
            delta = new_c - current
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
                current = new_c
                # Update ones and zeros
                new_ones = np.setdiff1d(ones, o_block)
                new_zeros = np.setdiff1d(zeros, z_block)
                ones = np.concatenate([new_ones, z_block])
                zeros = np.concatenate([new_zeros, o_block])
                if current < best:
                    best = current
                    best_h = h.copy()
            else:
                h[o_block] = 1.0
                h[z_block] = 0.0
        else:
            # Single swap

                continue
            io = rng.randint(len(ones))
            iz = rng.randint(len(zeros))
