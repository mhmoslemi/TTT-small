def _slsqp_obj_full(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Use a stable log-sum-exp
    log_sum = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return log_sum * dx
