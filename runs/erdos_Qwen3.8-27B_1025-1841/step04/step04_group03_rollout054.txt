import numpy as np
from scipy.optimize import minimize
import time

# Global state for SLSQP objective
_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_soft(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    if T < 1e-10:
        return mc * dx
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol, time_limit=None, start_time=None):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(obj_soft, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
    except Exception:
        return h.copy()
    return res.x

def thue_morse(n):
    """Generate Thue-Morse-like binary sequence of length n with exactly n/2 ones."""
    n2 = n // 2
    # Generate Thue-Morse of length > n
    tm = np.zeros(1)
    while len(tm) < n:
        tm = np.concatenate([tm, 1 - tm])
    tm = tm[:n]
    # Adjust to have exactly n/2 ones
    ones = np.sum(tm)
    if ones > n2:
        idx = np.where(tm == 1)[0]
        rng = np.random.RandomState(0)
        to_flip = idx[rng.choice(len(idx), size=ones - n2, replace=False)]
        tm[to_flip] = 0
    elif ones < n2:
        idx = np.where(tm == 0)[0]
        rng = np.random.RandomState(0)
        to_flip = idx[rng.choice(len(idx), size=n2 - ones, replace=False)]
        tm[to_flip] = 1
    return tm.astype(float)

import numpy as np
from scipy.optimize import minimize
import time

_SL_N = 0
_SL_T = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.count_nonzero(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_soft(x):
    n = _SL_N
    T = _SL_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    if T < 1e-10:
        return mc * dx
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_sum(x):
    return np.sum(x) - len(x) / 2.0

def con_sum_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, ftol):
    global _SL_N, _SL_T
    _SL_N = n
    _SL_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_sum, 'jac': con_sum_jac}]
    try:
        res = minimize(obj_soft, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h.copy()

def thue_morse_binary(n, seed=0):
    n2 = n // 2
    tm = np.zeros(1)
    while len(tm) < n:
        tm = np.concatenate([tm, 1.0 - tm])
    tm = tm[:n].copy()
    rng = np.random.RandomState(seed)
    ones = int(np.sum(tm))
    if ones > n2:
        idx = np.where(tm == 1)[0]
        to_flip = idx[rng.choice(len(idx), size=ones - n2, replace=False)]
        tm[to_flip] = 0.0
    elif ones < n2:
        idx = np.where(tm == 0)[0]
        to_flip = idx[rng.choice(len(idx), size=n2 - ones, replace=False)]
        tm[to_flip] = 1.0
    return tm

def sa_binary(h, n, max_iters, T0, T1, rng, time_limit, start_time, block_sizes):
    b = (h >= 0.5).astype(float)
    target = n // 2
    while np.sum(b) > target:
        ones = np.where(b == 1)[0]
        if len(ones) == 0:
            break
        b[ones[rng.randint(len(ones))]] = 0
    while np.sum(b) < target:
        zeros = np.where(b == 0)[0]
        if len(zeros) == 0:
            break
        b[zeros[rng.randint(len(zeros))]] = 1
    curr_c5 = compute_c5(b)
    best_c5 = curr_c5
    best_b = b.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        frac = min(1.0, float(it) / max_iters)
        T = T0 * (T1 / T0) ** frac
        bs = block_sizes[rng.randint(len(block_sizes))]
        start = rng.randint(0, max(1, n - bs + 1))
        end = min(start + bs, n)
        seg = b[start:end]
        oi = np.where(seg == 1)[0]
        zi = np.where(seg == 0)[0]
        if len(oi) > 0 and len(zi) > 0:
            sc = min(len(oi), len(zi))
            co = rng.choice(oi, size=sc, replace=False)
            cz = rng.choice(zi, size=sc, replace=False)
            b[start + co] = 0
            b[start + cz] = 1
        elif len(oi) > 0:
            j = start + oi[0]
            b[j] = 0
        else:
            j = start + zi[0]
            b[j] = 1
        new_c5 = compute_c5(b)
        accept = new_c5 <= curr_c5
        if not accept and T > 1e-12:
            delta = (new_c5 - curr_c5) / T
            if delta < 500.0:
                accept = rng.random() < np.exp(-delta)
        if accept:
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            if len(oi) > 0 and len(zi) > 0:
                b[start + co] = 1
                b[start + cz] = 0
            elif len(oi) > 0:
                b[j] = 1
            else:
                b[j] = 0
    return best_b, best_c5

def worst_shift_gradient_step(h, n, lr, rng, nsamples=200):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    if k_star < n:
        k = k_star
    else:
        k = 2 * (n - 1) - k_star
    if k == 0:
        return h.copy()
    grad = np.zeros(n)
    for j in range(k, n):
        grad[j] += (1.0 - h[j - k])
    for j in range(0, n - k):
        grad[j] -= h[j + k]
    h_new = h - lr * grad * dx
    h_new = make_feasible(h_new, n)
    return h_new

def slsqp_multi(h, n, temps, maxiter, ftol, start_time, budget_min_left):
    for T in temps:
        if time.time() - start_time > budget_min_left:
            break
        x_res = slsqp_optimize(h, n, T, maxiter, ftol)
        x_res = make_feasible(x_res, n)
        c5_res = compute_c5(x_res)
        if c5_res < compute_c5(h):
            h = x_res
    return h, compute_c5(h)

def fourier_perturb(h, n, rng, n_modes, amp):
    x = np.arange(n)
    h_new = h.copy()
    for _ in range(n_modes):
        freq = rng.randint(1, max(2, n // 8))
        phase = rng.uniform(0, 2 * np.pi)
        h_new += np.sin(2 * np.pi * freq * x / n + phase) * amp * rng.uniform(0.3, 1.5)
    return make_feasible(h_new, n)

def segment_flip(h, n, rng, n_segs):
    h_new = h.copy()
    seg = n // max(1, n_segs)
    for _ in range(n_segs):
        s = rng.randint(0, max(1, n - seg))

            h_new[s:s+seg] = 1.0 - h_new[s:s+seg]
        else:
            h_new[s:s+seg] *= rng.uniform(0.3, 1.2)
    return make_feasible(h_new, n)

            if c5_sa < best_c5 + 0.003:
                if time.time() < budget_end - 80:
                    h_sa2, c5_sa2 = slsqp_multi(h_sa, n0, [20.0, 100.0, 500.0], 100, 1e-14,
                                                start_time, budget_end - 80)
                    if c5_sa2 < best_c5:
                        best_c5 = c5_sa2
                        best_h = h_sa2.copy()

    # Phase 4: Worst-shift gradient steps + SLSQP
    if time.time() < budget_end - 100:
        h_ws = best_h.copy()
        for step in range(30):
            if time.time() > budget_end - 80:
                break
            lr = 0.02 * (0.9 ** step)
            h_ws = worst_shift_gradient_step(h_ws, n0, lr, rng)
            c5_ws = compute_c5(h_ws)
            if c5_ws < best_c5:
                best_c5 = c5_ws
                best_h = h_ws.copy()
        if time.time() < budget_end - 60:
            best_h, best_c5 = slsqp_multi(best_h, n0, [20.0, 100.0, 500.0, 3000.0], 120, 1e-14,
                                          start_time, budget_end - 60)

    # Phase 5: Structured perturbations + SLSQP
    if time.time() < budget_end - 150:
        for _ in range(3):
            if time.time() > budget_end - 80:
                break
            ns = rng.randint(2, 5)
            h_p = segment_flip(best_h, n0, rng, ns)
            h_p2, c5_p2 = slsqp_multi(h_p, n0, [50.0, 300.0], 80, 1e-14,
                                      start_time, budget_end - 50)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2.copy()
        if time.time() < budget_end - 80:
            h_f = fourier_perturb(best_h, n0, rng, 4, 0.04)
            h_f2, c5_f2 = slsqp_multi(h_f, n0, [50.0, 300.0, 2000.0], 100, 1e-14,
                                      start_time, budget_end - 50)
            if c5_f2 < best_c5:
                best_c5 = c5_f2
                best_h = h_f2.copy()

    # Phase 6: Coordinate descent with golden section
    if time.time() < budget_end - 40:
        best_h, best_c5 = coord_descent_golden(best_h.copy(), n0, 4, start_time, budget_end)

    # Phase 7: Final SLSQP at very low temp
    if time.time() < budget_end - 30:
        best_h, best_c5 = slsqp_multi(best_h, n0, [5.0, 20.0, 100.0], 150, 1e-15,
                                      start_time, budget_end - 20)

    # Phase 8: Final coord descent
    if time.time() < budget_end - 10:
        best_h, best_c5 = coord_descent_golden(best_h.copy(), n0, 2, start_time, budget_end)

    best_h = make_feasible(best_h, n0)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n0
