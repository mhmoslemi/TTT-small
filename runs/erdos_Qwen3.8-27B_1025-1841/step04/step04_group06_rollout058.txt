            hm[j] = max(0.0, h[j] - eps)
            cp = np.correlate(hp, 1.0 - hp, mode='full')
            cm = np.correlate(hm, 1.0 - hm, mode='full')
            # Use same lag selection
            gp = np.max(cp[idx])
            gm = np.max(cm[idx])
            grad[j] = (gp - gm) / (2 * eps)
        # Extrapolate to all indices (approximation)
        # Actually just use L-BFGS-B instead
        break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Work at n=1024 (good balance of resolution and speed)
    n1 = 1024
    hw = upsample_h(best_h, n1)
    c5w = compute_c5(hw)
    if c5w < best_c5:
        best_c5, best_h = c5w, hw.copy()

    # Multi-temperature, multi-topk L-BFGS-B
    configs = [
        (20, 5, 800), (20, 10, 800), (50, 5, 600), (50, 10, 600),
        (150, 5, 500), (150, 10, 500), (500, 10, 400), (500, 20, 400),
        (2000, 20, 300), (5000, 30, 250),
    ]
    for T, topm, mxit in configs:
        if rem() < 30:
            break
        try:
            ho = lbgfb_topk(hw, n1, T, topm, 500.0, mxit, tlimit=rem() - 10, t0=t0)
            c5o = compute_c5(ho)
            if c5o < best_c5:
                best_c5, best_h = c5o, ho.copy()
            hw = ho.copy()
        except Exception:
            pass

    # Restart with noise
    if rem() > 50:
        hrr = make_feasible(hw + 0.2 * rng.randn(n1), n1)
        for T, topm, mxit in [(100, 10, 300), (500, 10, 250)]:
            if rem() < 20:
                break
            try:
                ho = lbgfb_topk(hrr, n1, T, topm, 500.0, mxit, tlimit=rem() - 10, t0=t0)
                c5o = compute_c5(ho)
                if c5o < best_c5:
                    best_c5, best_h = c5o, ho.copy()
                    hrr = ho.copy()
            except Exception:
                pass

    # Phase 2: Binary local search at n1
    if rem() > 80:
        hb = (hw > 0.5).astype(float)
        # Fix count
        nones = int(np.sum(hb))
        nt = n1 // 2
        if nones > nt:
            oi = np.where(hb > 0.5)[0]
            for _ in range(nones - nt):
                hb[int(oi[rng.randint(len(oi))])] = 0.0
                oi = np.where(hb > 0.5)[0]
        elif nones < nt:
            zi = np.where(hb <= 0.5)[0]
            for _ in range(nt - nones):
                hb[int(zi[rng.randint(len(zi))])] = 1.0
                zi = np.where(hb <= 0.5)[0]
        c5b = compute_c5(hb)
        if c5b < best_c5:
            best_c5, best_h = c5b, hb.copy()

        hb2, c5b2 = bin_local_search(hb, n1, rng, max_rounds=60, tries_per_round=1200, tlimit=rem() - 40, t0=t0)
        if c5b2 < best_c5:
            best_c5, best_h = c5b2, hb2.copy()
            hb = hb2.copy()

        # Block swap search
        hb3, c5b3 = block_swap_search(hb, n1, rng, max_rounds=20, tlimit=rem() - 30, t0=t0)
        if c5b3 < best_c5:
            best_c5, best_h = c5b3, hb3.copy()

    # Phase 3: Higher resolution n=2048
    if rem() > 120:
        n2 = 2048
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

        for T, topm, mxit in [(30, 5, 400), (100, 10, 300), (400, 10, 250), (1500, 20, 200)]:
            if rem() < 40:
                break
            try:
                ho = lbgfb_topk(h2, n2, T, topm, 400.0, mxit, tlimit=rem() - 15, t0=t0)
                c5o = compute_c5(ho)
                if c5o < best_c5:
                    best_c5, best_h = c5o, ho.copy()
                    h2 = ho.copy()
            except Exception:
                pass

        # Binary search at n2
        if rem() > 80:
            h2b = (h2 > 0.5).astype(float)
            n2t = n2 // 2
            n2o = int(np.sum(h2b))
            if n2o > n2t:
                oi2 = np.where(h2b > 0.5)[0]
                for _ in range(n2o - n2t):
                    h2b[int(oi2[rng.randint(len(oi2))])] = 0.0
                    oi2 = np.where(h2b > 0.5)[0]
            elif n2o < n2t:
                zi2 = np.where(h2b <= 0.5)[0]
                for _ in range(n2t - n2o):
                    h2b[int(zi2[rng.randint(len(zi2))])] = 1.0
                    zi2 = np.where(h2b <= 0.5)[0]
            c5_2b = compute_c5(h2b)
            if c5_2b < best_c5:
                best_c5, best_h = c5_2b, h2b.copy()

            h2b2, c5_2b2 = bin_local_search(h2b, n2, rng, max_rounds=40, tries_per_round=800, tlimit=rem() - 30, t0=t0)
            if c5_2b2 < best_c5:
                best_c5, best_h = c5_2b2, h2b2.copy()

    # Phase 4: Even higher resolution n=3072 if time permits
    if rem() > 150:
        n3 = 3072
        h3 = upsample_h(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

        for T, topm, mxit in [(50, 5, 300), (200, 10, 200), (800, 10, 150)]:
            if rem() < 50:
                break
            try:
                ho = lbgfb_topk(h3, n3, T, topm, 300.0, mxit, tlimit=rem() - 20, t0=t0)
                c5o = compute_c5(ho)
                if c5o < best_c5:
                    best_c5, best_h = c5o, ho.copy()
                    h3 = ho.copy()
            except Exception:
                pass

        if rem() > 60:
            h3b = (h3 > 0.5).astype(float)
            n3t = n3 // 2
            n3o = int(np.sum(h3b))
            if n3o > n3t:
                oi3 = np.where(h3b > 0.5)[0]
                for _ in range(n3o - n3t):
                    h3b[int(oi3[rng.randint(len(oi3))])] = 0.0
                    oi3 = np.where(h3b > 0.5)[0]
            elif n3o < n3t:
                zi3 = np.where(h3b <= 0.5)[0]
                for _ in range(n3t - n3o):
                    h3b[int(zi3[rng.randint(len(zi3))])] = 1.0
                    zi3 = np.where(h3b <= 0.5)[0]
            c5_3b = compute_c5(h3b)
            if c5_3b < best_c5:
                best_c5, best_h = c5_3b, h3b.copy()
            h3b2, c5_3b2 = bin_local_search(h3b, n3, rng, max_rounds=30, tries_per_round=600, tlimit=rem() - 20, t0=t0)
            if c5_3b2 < best_c5:
                best_c5, best_h = c5_3b2, h3b2.copy()

    # Phase 5: Final SLSQP polish at moderate n
    if rem() > 40:
        ns = min(len(best_h), 1280)
        hs = upsample_h(best_h, ns) if ns < len(best_h) else best_h.copy()
        dx = 2.0 / ns
        target = ns / 2.0

        def slsqp_obj(hv):
            corr = np.correlate(hv, 1.0 - hv, mode='full')
            topm = 10
            idx = np.argpartition(corr, -topm)[-topm:]
            tv = corr[idx]
            mx = np.max(tv)
            sh = np.clip(300 * (tv - mx), -50, 50)
            return (mx + np.log(np.sum(np.exp(sh))) / 300) * dx

        def slsqp_con(hv):
            return np.sum(hv) - target

        try:
            res = minimize(slsqp_obj, hs, method='SLSQP',
                           bounds=[(0.0, 1.0)] * ns,
                           constraints=[{'type': 'eq', 'fun': slsqp_con, 'jac': lambda x: np.ones(ns)}],
                           options={'maxiter': 100, 'ftol': 1e-16})
            if res.success or res.fun < slsqp_obj(hs):
                hsp = make_feasible(res.x, ns)
                c5sp = compute_c5(hsp)
                if c5sp < best_c5:
                    best_c5, best_h = c5sp, hsp.copy()
        except Exception:
            pass

    # Phase 6: Final coordinate descent / binary polish
    if rem() > 20:
        nf = len(best_h)
        hcd = best_h.copy()
        cd_improved = True
        max_cd_rounds = 3
        for rnd in range(max_cd_rounds):
            if rem() < 10:
                break
            cd_improved = False
            # Try binary flips at random positions
            for _ in range(min(3000, nf * 5)):
                j = int(rng.randint(nf))
                old_v = hcd[j]
                new_v = 1.0 - old_v if old_v < 0.99 or old_v > 0.01 else (0.0 if old_v > 0.5 else 1.0)
                if abs(new_v - old_v) < 1e-9:
                    continue
                hcd[j] = new_v
                hcd = make_feasible(hcd, nf)
                c = compute_c5(hcd)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = hcd.copy()
                    cd_improved = True
                else:
                    hcd[j] = old_v
                    hcd = make_feasible(hcd, nf)
            if not cd_improved:
                break
        c5f = compute_c5(hcd)
        if c5f < best_c5:
            best_c5, best_h = c5f, hcd.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    # Ensure best_h matches best_c5
    if compute_c5(best_h) != best_c5:
        # Re-verify
        best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
