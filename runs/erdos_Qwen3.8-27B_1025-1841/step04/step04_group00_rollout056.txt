from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_obj(x, n, T, top_m=None):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    if top_m is not None and top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        vals = corr[top_idx]
        mc = vals.max()
        s = vals - mc
        return (mc + np.log(np.sum(np.exp(T * s))) / T) * dx
    mc = corr.max()
    s = corr - mc
    return (mc + np.log(np.sum(np.exp(T * s))) / T) * dx

def slsqp_optimize(h, n, T, maxiter, top_m=None):
    def obj(x):
        return smooth_max_obj(x, n, T, top_m)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}]
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = make_feasible(res.x, n)
        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_old:
            return h_new, c5_new
    except:
        pass
    return h, compute_c5(h)

def pgd_optimize(h, n, budget_s, start_time, top_m=8, lr=0.05):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    mu = 0.9

        corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_max_grad(x, n, T, top_m=None):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    if top_m is not None and top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        vals = corr[top_idx]
    else:
        top_idx = np.arange(len(corr))
        vals = corr
    
    mc = vals.max()
    shifted = vals - mc
    exps = np.exp(T * shifted)
    w = exps / np.sum(exps)
    
    grad = np.zeros(n)

Wait, the projection onto {0≤h≤1, sum h = n/2} needs to be done correctly. The simple "clip then adjust" approach might not be the true projection. Let me use a proper one:

The projection of z onto {h: 0≤h≤1, sum h = n/2} solves:
min ||h - z||² subject to 0≤h≤1, sum h = n/2

The solution is h_i = clip(z_i - λ, 0, 1) for some λ.

To find λ: bisect on λ such that sum_i clip(z_i - λ, 0, 1) = n/2.

```python
def project_onto_constraint(z, n):
    target = n / 2.0
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(50):
        lam = (lo + hi) / 2.0
        h = np.clip(z - lam, 0.0, 1.0)
        s = h.sum()
        if s < target:
            hi = lam
        else:
            lo = lam
    h = np.clip(z - lam, 0.0, 1.0)
    # Final adjustment
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_constraint(z, n):
    target = n / 2.0
    lo_val = z.min() - 1.0
    hi_val = z.max() + 1.0
    lam = (lo_val + hi_val) / 2.0
    for _ in range(60):
        lam = (lo_val + hi_val) / 2.0
        h = np.clip(z - lam, 0.0, 1.0)
        s = h.sum()
        if s < target:
            hi_val = lam
        else:
            lo_val = lam
    h = np.clip(z - lam, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_grad(x, n, T, top_m=None):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    full_len = len(corr)
    if top_m is not None and top_m < full_len:
        top_idx = np.argsort(corr)[-top_m:]
        vals = corr[top_idx]
    else:
        top_idx = np.arange(full_len)
        vals = corr
    mc = vals.max()
    shifted = vals - mc
    exps = np.exp(T * shifted)
    w = exps / np.sum(exps)
    grad = np.zeros(n)
    ren = range(int(-n + 1), int(n))
    for idx, wi in zip(top_idx, w):
        if wi < 1e-14:
            continue
        s = int(idx) - (n - 1)
        if s == 0:
            grad += wi * (1.0 - 2.0 * x)
        elif s > 0:
            grad[:n - s] += wi * (1.0 - x[s:])
            grad[s:] -= wi * x[:n - s]
        else:
            ss = -s
            grad[ss:] += wi * (1.0 - x[:n - ss])
            grad[:n - ss] -= wi * x[ss:]
    grad *= dx
    obj_val = float((mc + np.log(np.sum(exps)) / T) * dx)
    return obj_val, grad

def fista_optimize(h, n, budget_s, start_time, T, top_m=None):
    h = h.copy()
    y = h.copy()
    t = 1.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    L = 50.0
    stall = 0
    while time.time() - start_time < budget_s:
        _, grad = smooth_max_grad(y, n, T, top_m)
        z = y - grad / L
        h_new = project_onto_constraint(z, n)
        if np.max(np.abs(h_new - h)) < 1e-13:
            stall += 1
            if stall > 50:
                break
        else:
            stall = 0
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_new.copy()
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_new + ((t - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)
        h = h_new
        t = t_new
        if int(t) % 100 == 0:
            L *= 1.05
    h = make_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h, best_c5 = h, c5_h
    return best_h, best_c5

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, n, T, maxiter, top_m=None):
    def obj_and_grad(x):
        return smooth_max_grad(x, n, T, top_m)

    def sfun(x):
        val, g = smooth_max_grad(x, n, T, top_m)
        return val, g

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(sfun, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-18})
        h_new = make_feasible(res.x, n)
        c5_old = compute_c5(h)
        c5_new = compute_c5(h_new)
        if c5_new <= c5_old + 1e-15:
            return h_new, c5_new
    except Exception:
        pass
    return h, compute_c5(h)

def pgd_optimize(h, n, budget_s, start_time, top_m=8, lr=0.03):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    mu = 0.85
    stall = 0
    while time.time() - start_time < budget_s:
        _, grad = smooth_max_grad(h, n, 50.0, top_m)
        y = h - mu * vel
        y = np.clip(y, 0.0, 1.0)
        _, grad_y = smooth_max_grad(y, n, 50.0, top_m)
        vel = mu * vel - lr * grad_y
        h_new = h + vel
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        c5_cur = compute_c5(h)
        if c5_new <= c5_cur + 1e-15:
            h = h_new
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            lr *= 0.7
            vel *= 0.5
            stall += 1
        if stall > 80:
            break
        if lr < 1e-9:
            break
    h = make_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h, best_c5 = h, c5_h
    return best_h, best_c5

def coord_descent(h, n, budget_s, start_time, steps=None):
    if steps is None:
        steps = [0.003, 0.001, 0.0005, 0.0002]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    any_improved = True
    while any_improved and time.time() - start_time < budget_s:
        any_improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 2:
                return best_h, best_c5
            old_val = best_h[j]
            best_c = best_c5
            best_val = old_val
            for s in steps:
                for sign_val in (-1.0, 1.0):
                    v = old_val + sign_val * s
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            for v in (0.0, 1.0):
                if abs(v - old_val) > 0.02:
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                any_improved = True
    return best_h, best_c5

def binary_search_and_recover(h, n, rng, budget_s, start_time):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    while time.time() - start_time < budget_s:
        h_bin = (h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, n)
        improved = False
        ones_list = np.flatnonzero(h_bin == 1.0)
        zeros_list = np.flatnonzero(h_bin == 0.0)
        for _ in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
            if len(ones_list) == 0 or len(zeros_list) == 0:
                break
            i_old = int(rng.randint(len(ones_list)))
            j_new = int(rng.randint(len(zeros_list)))
            a = ones_list[i_old]
            b = zeros_list[j_new]
            h_bin[a] = 0.0
            h_bin[b] = 1.0
            c5_trial = compute_c5(h_bin)
            if c5_trial < best_c5 - 1e-15:
                best_c5 = c5_trial
                best_h = h_bin.copy()
                improved = True
                ones_list = np.flatnonzero(h_bin == 1.0)
                zeros_list = np.flatnonzero(h_bin == 0.0)
            else:
                h_bin[a] = 1.0
                h_bin[b] = 0.0
        if not improved:
            break
    return best_h, best_c5
