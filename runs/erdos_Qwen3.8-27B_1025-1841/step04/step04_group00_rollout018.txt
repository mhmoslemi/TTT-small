import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0 - 1e-15) if diff > 0 else (h > 1e-15)
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def to_binary(h, n):
    """Project to binary with exactly n/2 ones."""
    target = n // 2
    h_sorted = h.copy()
    if np.sum(h > 0.5) >= target:
        # Too many ones: flip the smallest >0.5 values to 0
        above = np.where(h > 0.5)[0]
        n_extra = len(above) - target
        if n_extra > 0:
            above_sorted = above[np.argsort(h[above])]
            h[above_sorted[:n_extra]] = 0
    else:
        # Too few ones: flip the largest <0.5 values to 1
        below = np.where(h < 0.5)[0]
        n_missing = target - len(above)
        if n_missing > 0:
            below_sorted = below[np.argsort(-h[below])]
            h[below_sorted[:n_missing]] = 1
    h[h > 0.5] = 1
    h[h <= 0.5] = 0
    # Verify
    assert np.sum(h) == target
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

        else:
            mask = h > 1e-12
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def project_box_sum(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    mask = (h < 1.0 - 1e-12) if diff > 0 else (h > 1e-12)
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0, 1)
    return h

def vec_grad_shift(h, n, s):
    grad = np.zeros(n)
    if s >= 0:
        if s < n:
            grad[:n-s] += (1.0 - h[s:])
            if s > 0:
                grad[s:] += -h[:n-s]
    else:
        m = -s
        if m < n:
            grad[m:] += (1.0 - h[:n-m])
            if m > 0:
                grad[:n-m] += -h[m:]
    return grad

def top_grad(h, n, top_m):
    corr = compute_corr_full(h)
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for idx in top_idx:
        s = int(idx) - (n - 1)
        grad += vec_grad_shift(h, n, s)
    return grad / len(top_idx), corr

def pgd_opt(h, n, t_budget, t_start, top_m=6, init_step=0.04):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    step = init_step
    mom = 0.8
    no_imp = 0
    while time.time() - t_start < t_budget:
        grad, corr = top_grad(h, n, top_m)
        c5_cur = float(np.max(corr) * dx)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            no_imp = 0
        else:
            no_imp += 1
            if no_imp > 60:
                step *= 0.5
                if step < 1e-8:
                    break
                no_imp = 0
        y = np.clip(h + mom * vel, 0, 1)
        gy, _ = top_grad(y, n, top_m)
        vel = mom * vel - step * gy
        nh = project_box_sum(h + vel, n)
        c5_new = compute_c5(nh)
        if c5_new <= c5_cur:
            h = nh
        else:
            nh2 = project_box_sum(h - step * grad, n)
            c5_2 = compute_c5(nh2)
            if c5_2 <= c5_cur:
                h = nh2
            else:
                step *= 0.65
                vel *= 0.3
            if step < 1e-9:
                break
        if no_imp % 150 == 0 and step < 0.04:
            step = min(0.05, step * 1.4)
            mom = min(0.93, mom + 0.01)
    h = project_box_sum(h, n)
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_h, best_c5 = h, c5_final
    return best_h, best_c5

def compute_E(h, n):
    E = np.zeros((n, 2*n-1))
    for s in range(-(n-1), n):
        idx = s + (n-1)
        if s >= 0:
            if s < n:
                E[0:n-s, idx] += h[s:n]
                E[s:n, idx] += h[0:n-s]
        else:
            m = -s
            if m < n:
                E[m:n, idx] += h[0:n-m]
                E[0:n-m, idx] += h[m:n]
    return E

def binary_greedy(h, n, t_budget, t_start, rng):
    h_b = h.copy()
    target = n // 2
    si = np.argsort(-h_b)
    h_b = np.zeros(n)
    h_b[si[:target]] = 1.0
    corr = compute_corr_full(h_b)
    best_c5 = float(np.max(corr) * 2.0 / n)
    best_h = h_b.copy()
    ones = np.where(h_b == 1)[0]
    zeros = np.where(h_b == 0)[0]
    E = compute_E(h_b, n)
    cur_max = float(np.max(corr))
    stalled = 0
    while time.time() - t_start < t_budget:
        E_zeros = E[zeros, :]
        best_j_local = -1
        best_improve = 0.0
        for ii in range(len(ones)):
            i = ones[ii]
            delta = E_zeros - E[i]
            new_max = np.max(corr[np.newaxis, :] + delta, axis=1)
            bi = np.argmin(new_max)
            imp = cur_max - new_max[bi]
            if imp > best_improve:
                best_improve = imp
                best_j_local = bi
                best_i_local = i
        if best_j_local < 0:
            # Try random swaps
            found = False
            for _ in range(500):
                i = int(ones[rng.randint(len(ones))])
                j = int(zeros[rng.randint(len(zeros))])
                nc = corr + E[i] - E[j]
                nm = float(np.max(nc))
                if nm < cur_max - 1e-12:
                    h_b[i] = 0
                    h_b[j] = 1
                    corr = nc
                    cur_max = nm
                    # Update E
                    E[i, :] -= 1.0  # placeholder, handled below
                    # Actually recompute affected rows
                    for s in range(-(n-1), n):
                        si_ = s + (n-1)
                        # h[i] went 1->0: affects E[p,si_] where p+s=i or p-s=i
                        # p = i-s: if 0<=i-s<n, E[i-s, si_] -= 1 (from h[i] term in h[p+s] or h[p-s])
                        # This is complex; just recompute E fully if needed
                    E = compute_E(h_b, n)
                    ones = np.where(h_b == 1)[0]
                    zeros = np.where(h_b == 0)[0]
                    if nm < best_c5 * n / 2.0 - 1e-12:
                        best_c5 = nm * 2.0 / n
                        best_h = h_b.copy()
                    stalled = 0
                    found = True
                    break
            if not found:
                break
        else:
            i = best_i_local
            j = zeros[best_j_local]
            h_b[i] = 0
            h_b[j] = 1
            E = compute_E(h_b, n)
            corr = corr + E[zeros[best_j_local], :] - E[ones[int(np.where(ones==i)[0][0])], :] if False else None
            # Recompute corr properly
            corr = compute_corr_full(h_b)
            cur_max = float(np.max(corr))
            ones = np.where(h_b == 1)[0]
            zeros = np.where(h_b == 0)[0]
            stalled = 0
            if cur_max < best_c5 * n / 2.0 - 1e-12:
                best_c5 = cur_max * 2.0 / n
                best_h = h_b.copy()
        stalled += 1
        if stalled > 50:
            break
    return best_h, best_c5

def slsqp_solve(h, n, T=2000.0, top_m=None, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m and top_m < len(corr):
            ti = np.argsort(corr)[-top_m:]
            mc = np.max(corr[ti])
            sh = corr[ti] - mc
            return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx
        mc = np.max(corr)
        sh = corr - mc
        return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    def cons_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-16})
        ch = make_feasible(res.x, n)
        c5 = compute_c5(ch)
        if c5 < compute_c5(h) - 1e-15:
            return ch, c5
    except Exception:
        pass
    return h, compute_c5(h)

def coord_desc(h, n, t_budget, t_start, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(5):
        if time.time() - t_start > t_budget:
            break
        improved = False
        n_pts = len(best_h)
        for j in range(n_pts):
            if time.time() - t_start > t_budget - 2:
                break
            old = best_h[j]
            local_best = old
            local_c = best_c5
            for s in steps:
                for sig in [-1.0, 1.0]:
                    v = old + sig * s
                    if 0.0 < v < 1.0 and abs(v - old) > 1e-10:
                        tr = best_h.copy()
                        tr[j] = v
                        tr = make_feasible(tr, n_pts)
                        c = compute_c5(tr)
                        if c < local_c - 1e-14:
                            local_c = c
                            local_best = v
            if local_best != old:
                best_h[j] = local_best
                best_h = make_feasible(best_h, n_pts)
                best_c5 = local_c
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - t0

    def rem():
        return budget_s - elapsed()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: PGD
    if rem() > 100:
        h, c5 = pgd_opt(best_h, n, min(80, rem() - 30), t0, top_m=6, init_step=0.04)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 2: SLSQP
    if rem() > 80:
        h, c5 = slsqp_solve(best_h, n, T=1500.0, top_m=6, maxiter=300)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: PGD again (tighter)
    if rem() > 80:
        h, c5 = pgd_opt(best_h, n, min(60, rem() - 30), t0, top_m=8, init_step=0.02)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: Binary greedy local search
    if rem() > 200:
        h_b, c5_b = binary_greedy(best_h, n, min(150, rem() - 30), t0, rng)
        if c5_b < best_c5:
            best_h, best_c5 = h_b, c5_b
            # Continue binary search if still improving
            if rem() > 100:
                h_b2, c5_b2 = binary_greedy(best_h, n, min(80, rem() - 30), t0, rng)
                if c5_b2 < best_c5:
                    best_h, best_c5 = h_b2, c5_b2

    # Phase 5: Coordinate descent (fine)
    if rem() > 60:
        h, c5 = coord_desc(best_h, n, min(50, rem() - 15), t0, steps=[0.005, 0.002, 0.001, 0.0005])
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: SLSQP with high T
    if rem() > 60:
        h, c5 = slsqp_solve(best_h, n, T=3000.0, top_m=4, maxiter=400)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: PGD with top-4
    if rem() > 60:
        h, c5 = pgd_opt(best_h, n, min(50, rem() - 15), t0, top_m=4, init_step=0.015)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 8: Coord descent ultra-fine
    if rem() > 40:
        h, c5 = coord_desc(best_h, n, min(30, rem() - 10), t0, steps=[0.002, 0.001, 0.0005, 0.0002])
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 9: Final SLSQP
    if rem() > 40:
        h, c5 = slsqp_solve(best_h, n, T=5000.0, top_m=3, maxiter=500)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 10: Random perturbations + local refine
    for restart in range(4):
        if rem() < 30:
            break
        nz = len(best_h)
        scale = 0.005 + restart * 0.005
        h_try = make_feasible(best_h + rng.randn(nz) * scale, nz)
        h2, c5_2 = pgd_opt(h_try, nz, min(25, rem() - 10), t0, top_m=5, init_step=0.02)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2
        if rem() < 20:
            break

    # Phase 11: Final polish
    if rem() > 20:
        h, c5 = slsqp_solve(best_h, n, T=8000.0, top_m=3, maxiter=600)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
