    G = np.fft.rfft(1.0 - h, n)

    f = np.empty(n)
    f[0] = r[0]
    f[1:] = r[n-1:0:-1]

           = ∑_k w_k * [(1-h[(j+k)%n])] - ∑_k w_k * h[(j-k)%n]
           = (1 - ∑_k w_k * h[(j+k)%n]) - ∑_k w_k * h[(j-k)%n]
           = 1 - (w*h_circ)[j] - (w*h_circ_rev)[j]

where (w*h_circ)[j] = ∑_k w[k] * h[(j+k)%n] (circular convolution of w with h)
and (w*h_circ_rev)[j] = ∑_k w[k] * h[(j-k)%n] (circular correlation)

These can be computed via FFT in O(n log n)!

(w * h_circ)[j] = ∑_k w[k] * h[(j+k)%n] = circular_conv(w, h)[j]
= ifft(fft(w) * fft(h))[j]

(w * h_corr)[j] = ∑_k w[k] * h[(j-k)%n] = circular_corr(w, h)[j]  
= ifft(fft(w) * conj(fft(h)))[j]

So the full gradient is:
grad[j] = 1 - ifft(fft(w)*fft(h))[j] - ifft(fft(w)*conj(fft(h)))[j]

All in O(n log n)! Beautiful.

The integral projection:

- If sum > n/2: need to reduce sum by δ. Find threshold τ such that sum(clip(h_new - τ, 0, 1)) = n/2.
  This is a 1D root-finding problem (bisection on τ).
- Similarly for sum < n/2.

Actually, a simpler approach:

Actually, I realize there might be a simpler and more effective approach:

**Key observation**: f_k = n/2 - ∑_i h[i]·h[(i+k)%n]

So C₅ = max_k f_k · dx = (n/2 - min_k ∑_i h[i]·h[(i+k)%n]) · dx

Minimizing C₅ = maximizing min_k ∑_i h[i]·h[(i+k)%n]

This is a min-max problem: max_h min_k (h^T A_k h) subject to 0≤h≤1, sum h = n/2.

The inner maximization (over h for fixed k) is: max h^T A_k h. The matrix A_k for lag k is a circulant matrix. Its eigenvalues are the FFT of one row.

For a circulant matrix A_k where (A_k)_{i,j} = 1 if j-i ≡ k (mod n), the eigenvalues are:
λ_m = ∑_j e^{-2πi(mj)/n} · (A_k)_{0,j} = e^{-2πi(mk)/n} (only one non-zero entry per row at j=k)

Wait, A_k has (A_k)_{i,j} = 1 if j = (i+k) mod n. So each row has exactly one 1.
The eigenvalues: λ_m = e^{2πi·m·k/n} (the DFT of the first row which is [0,...,1,0,...,0] with 1 at position k).

So the quadratic h^T A_k h = ∑_i h[i]·h[(i+k)%n] is a sum of products.

The maximum of h^T A_k h over {0≤h≤1, sum h = n/2} depends on the eigenvalues.

For k=1 (adjacent): A_1 is a cyclic shift. The max of h^T A_1 h with sum h = n/2 is achieved by making h as "blocky" as possible (all 1s in one contiguous block of length n/2).

For k=n/2 (opposite): A_{n/2} pairs h[i] with h[(i+n/2)%n]. The max is achieved by making h[i] = h[(i+n/2)%n] for all i (both 0 or both 1), giving ∑h[i]h[(i+n/2)%n] = ∑h[i]² ≤ n/2.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            room = 1.0 - h
        else:
            room = h
        mask = room > 1e-12
        if not mask.any():
            break
        step = diff / mask.sum()
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def fft_circular_corr(h, g):
    """Circular cross-correlation: r[k] = sum_i h[i]*g[(i+k)%n]"""
    n = len(h)
    H = np.fft.rfft(h, n)
    G = np.fft.rfft(g, n)
    return np.fft.irfft(H * np.conj(G), n)

def circular_f(h):
    """f[k] = sum_i h[i]*(1-h[(i+k)%n]) for all k."""
    n = len(h)
    r = fft_circular_corr(h, 1.0 - h)
    # f[k] = r[(n-k)%n]
    f = np.empty(n)
    f[0] = r[0]
    f[1:] = r[n-1:0:-1]
    return f

def softmax_cg(h, T):
    """Compute soft-max gradient using FFT."""
    n = len(h)
    f = circular_f(h)
    f_shifted = f - f.max()
    w = np.exp(T * f_shifted)
    w /= w.sum()
    
    W = np.fft.rfft(w, n)
    H = np.fft.rfft(h, n)
    
    # grad[j] = 1 - sum_k w[k]*h[(j+k)%n] - sum_k w[k]*h[(j-k)%n]
    # = 1 - circ_conv(w,h)[j] - circ_corr(w,h)[j]
    conv = np.fft.irfft(W * H, n)  # circ_conv(w,h)[j] = sum_k w[k]*h[(j+k)%n]
    corr = np.fft.irfft(W * np.conj(H), n)  # circ_corr(w,h)[j] = sum_k w[k]*h[(j-k)%n]
    
    grad = 1.0 - conv - corr
    return f, w, grad

def projected_gradient_step(h, grad, lr):
    """Take a gradient step and project onto {0<=h<=1, sum(h)=n/2}."""
    n = len(h)
    h_new = h - lr * grad
    h_new = np.clip(h_new, 0.0, 1.0)
    return make_feasible(h_new, n)

def bb_step_size(s, y, prev_lr):
    """Barzilai-Borwein step size."""
    sy = np.dot(s, y)
    yy = np.dot(y, y)
    if yy < 1e-30:
        return prev_lr
    if sy <= 1e-30:
        return prev_lr
    lr = sy / yy
    return np.clip(lr, 1e-4, 1.0)

def fft_projected_gradient(h0, n, T=50.0, max_iter=500, verbose=False):
    """Main FFT-based projected gradient optimizer."""
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = 0.01
    prev_h = h.copy()
    prev_grad = None
    
    for it in range(max_iter):
        f, w, grad = softmax_cg(h, T)
        c5_est = f.max() * dx
        
        # Projected gradient step
        h_new = projected_gradient_step(h, grad, lr)
        s = h_new - h
        s_norm = np.linalg.norm(s)
        
        if s_norm < 1e-14:
            lr *= 0.5
            continue
        
        # BB step size
        if prev_grad is not None:
            y = grad - prev_grad
            lr = bb_step_size(s, y, lr)
        
        h = h_new
        prev_h = h.copy()
        prev_grad = grad
        
        # Evaluate periodically

            if c5 < best_c5:

                best_h = h.copy()

    return best_h, best_c5

def topk_gradient_step(h, k_active, lr):
    """Gradient step using top-k lags uniformly weighted."""
    n = len(h)
    f = circular_f(h)
    # Find top k_active lags
    idx = np.argpartition(f, -k_active)[-k_active:]
    w = np.zeros(n)
    w[idx] = 1.0 / k_active
    
    W = np.fft.rfft(w, n)
    H = np.fft.rfft(h, n)
    conv = np.fft.irfft(W * H, n)
    corr = np.fft.irfft(W * np.conj(H), n)
    grad = 1.0 - conv - corr
    
    return h - lr * grad

    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

def binary_local_search(h_bin, n, rng, max_rounds=40, tries=3000):
    """Local search on binary h via 1-flip and 2-flip moves."""
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()

        n_tries = min(tries, len(ones) * len(zeros))

                zeros[jj] = oi
                improved = True
            else:

                h[oj] = 0.0

        return h
    if diff > 0:

        room = 1.0 - h
        mask = room > 1e-14
        if room[mask].sum() < diff:
            # Can't reach target with just additions
            h[mask] = 1.0
            return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() == 0:
            break
        h[mask] += diff / mask.sum()
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        idx = np.argmax(1.0 - h) if diff > 0 else np.argmin(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def circular_f(h):
    n = len(h)
    H = np.fft.rfft(h, n)
    G = np.fft.rfft(1.0 - h, n)
    r = np.fft.irfft(H * np.conj(G), n)
    f = np.empty(n)
    f[0] = r[0]
    f[1:] = r[n-1:0:-1]
    return f

def softmax_gradient(h, T):
    n = len(h)
    f = circular_f(h)
    fs = f - f.max()
    w = np.exp(T * fs)
    w /= w.sum()
    W = np.fft.rfft(w, n)
    H = np.fft.rfft(h, n)
    conv = np.fft.irfft(W * H, n)
    corr = np.fft.irfft(W * np.conj(H), n)
    grad = 1.0 - conv - corr
    return f, w, grad

def fft_projected_gradient(h0, n, T=50.0, max_iter=400, beta=0.9):
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.05
    v = np.zeros(n)
    prev_h = h.copy()
    prev_grad = None
    for it in range(max_iter):
        f, w, grad = softmax_gradient(h, T)
        v = beta * v + (1.0 - beta) * grad
        h_new = h - lr * v
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        s = h_new - h
        s_norm = np.linalg.norm(s)
        if s_norm < 1e-15:
            lr *= 0.5
            h = h_new
            continue
        if prev_grad is not None:
            y = grad - prev_grad
            sy = np.dot(s, y)
            yy = np.dot(y, y)
            if yy > 1e-20 and sy > 1e-20:
                lr = np.clip(sy / yy, 0.005, 0.5)
        h = h_new
        prev_h = h.copy()
        prev_grad = grad
        if it % 25 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        if it % 80 == 79 and T < 2000:
            T *= 1.4
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def alt_lag_descent(h0, n, n_iters=200, topk=8, lr=0.003):
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        f = circular_f(h)
        idx = np.argpartition(f, -topk)[-topk:]
        for k in idx:
            kk = k % n
            if kk == 0:
                grad_k = 1.0 - 2.0 * h
            else:
                grad_k = (1.0 - np.roll(h, -kk)) - np.roll(h, kk)
            h_new = h - lr * grad_k
            h_new = np.clip(h_new, 0.0, 1.0)
            h = make_feasible(h_new, n)
        if it % 20 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def binary_swap_search(h_bin, n, rng, max_rounds=60, tries=5000):
    h = h_bin.copy()
    cur = compute_c5(h)
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for rnd in range(max_rounds):
        improved = False
        n_tries = min(tries, max(1, len(ones) * len(zeros)))
        for _ in range(n_tries):
            if not ones or not zeros:
                break
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur - 1e-15:
                cur = c
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur

def simulated_anneal_binary(h_bin, n, rng, n_iters=3000, T_start=0.002, T_end=0.0001):
    h = h_bin.copy()
    cur = compute_c5(h)
    best_c5 = cur
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    for it in range(n_iters):
        T = T_start * (T_end / T_start) ** (it / max(1, n_iters - 1))
        if not ones or not zeros:
            break
        ii = rng.randint(len(ones))
        jj = rng.randint(len(zeros))
        oi, oj = ones[ii], zeros[jj]
        h[oi] = 0.0
        h[oj] = 1.0
        c = compute_c5(h)
        delta = c - cur
        if delta < 0 or rng.rand() < np.exp(-delta / T):
            cur = c
            ones[ii] = oj
            zeros[jj] = oi
            if cur < best_c5:
                best_c5 = cur
                best_h = h.copy()
        else:
            h[oi] = 1.0
            h[oj] = 0.0
    return best_h, best_c5

def coord_polish(h0, n, max_rounds=6, steps=None):
    if steps is None:
        steps = [0.05, -0.05, 0.025, -0.025, 0.01, -0.01, 0.005, -0.005]
    h = h0.copy()
    cur = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            if h[j] <= 1e-10 or h[j] >= 1.0 - 1e-10:
                continue
            ov = h[j]
            for d in steps:
                nv = ov + d
                if nv < -1e-12 or nv > 1.0 + 1e-12:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cur - 1e-14:
                    cur = c
                    h = trial
                    ov = nv
                    improved = True
                    break
        if not improved:
            break
    return h, cur
