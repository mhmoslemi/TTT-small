import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 960
_OPT_T = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def subgrad_active(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    active = np.where(np.abs(corr - max_val) < 1e-12)[0]
    grad = np.zeros(n)
    for idx in active:
        l = idx - n + 1
        if l >= 0:
            grad[l:] += (1.0 - h[:n - l])
            grad[:n - l] -= h[l:]
        else:
            m = -l
            grad[:n - m] += (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    grad /= len(active)
    return grad, max_val

def slsqp_optimize(h, T, maxiter, ftol):
    global _OPT_N, _OPT_T
    n = len(h)
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        res_h = make_feasible(res.x, n)
        res_c = compute_c5(res_h)
        return res_h, res_c
    except Exception:
        return h, compute_c5(h)

def coord_descent(h, n, deltas, max_rounds, tol):
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_val = old
            best_cj = best_c
            for d in deltas:
                cand = max(0.0, min(1.0, old + d))
                if abs(cand - old) < 1e-15:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_cj - tol:
                    best_cj = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_cj
                improved = True
        if not improved:
            break
    return h, best_c
