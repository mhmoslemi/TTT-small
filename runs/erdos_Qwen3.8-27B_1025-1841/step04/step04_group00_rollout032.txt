def make_feasible(h, n):

def soft_max_objective(h, T=500.0, top_m=None):
    """Smooth approximation of max correlation"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_correlations(h)
    if top_m is not None and top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        c = corr[top_idx]
    else:
        c = corr
    mc = np.max(c)
    # Log-sum-exp for numerical stability
    shifted = c - mc
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    val = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return val * dx

    def obj(x):
        return soft_max_objective(x, T=T, top_m=top_m)
    
    def obj_grad(x):
        n = len(x)
        dx = 2.0 / n
        corr = compute_correlations(x)
        if top_m is not None and top_m < len(corr):

            c = corr[top_idx]
        else:
            top_idx = np.arange(len(corr))
            c = corr
        mc = np.max(c)
        shifted = c - mc
        weights = np.exp(T * np.clip(shifted, -50, 0))
        weights /= np.sum(weights)

                    grad[j] += w * (1.0 - x[j])
                if 0 <= j - s < n:
                    grad[j] -= w * x[j - s]
        return grad * dx
    
    bounds = [(0.0, 1.0)] * n

import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def project(h, n):
    h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = np.sum(mask)
    if cnt > 0:
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = n / 2.0 - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def subgradient(h, top_m):
    """Compute subgradient of max of top correlations"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in top_k:
        s = k - (n - 1)
        # d/dh[j] corr[k] = (1-h[j-s]) if 0<=j-s<n, minus h[j+s] if 0<=j+s<n
        js_minus = np.arange(n) - s
        valid_plus = (js_minus >= 0) & (js_minus < n)
        grad[valid_plus] += (1.0 - h[js_minus[valid_plus]])
        js_plus = np.arange(n) + s
        valid_minus = (js_plus >= 0) & (js_plus < n)
        grad[valid_minus] -= h[js_plus[valid_minus]]
    grad /= len(top_k)
    return grad, corr

def soft_corr_value(h, T, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < len(corr):
        top = corr[np.argsort(corr)[-top_m:]]
    else:
        top = corr
    mc = np.max(top)
    shifted = np.clip(top - mc, -40, 0)
    val = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return val * dx, corr

def soft_corr_grad(h, T, top_m):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
    else:
        top_idx = np.arange(len(corr))
    top_corr = corr[top_idx]
    mc = np.max(top_corr)
    shifted = np.clip(top_corr - mc, -40, 0)
    weights = np.exp(T * shifted)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    js_arr = np.arange(n)
    for w, k in zip(weights, top_idx):
        s = k - (n - 1)
        js_minus = js_arr - s
        valid_p = (js_minus >= 0) & (js_minus < n)
        grad[valid_p] += w * (1.0 - h[js_minus[valid_p]])
        js_plus = js_arr + s
        valid_m = (js_plus >= 0) & (js_plus < n)
        grad[valid_m] -= w * h[js_plus[valid_m]]
    return grad * T * dx

def nesterov_descent(h, n, start_time, time_limit, top_m=10, step0=0.05):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    step = step0
    mu = 0.3
    no_imp = 0
    
    while time.time() - start_time < time_limit:
        y = np.clip(h - mu * vel, 0, 1)
        grad_y, _ = subgradient(y, top_m)
        vel = mu * vel - step * grad_y
        mu = 0.5 * (1 + mu)
        new_h = project(h + vel, n)
        c5_new = compute_c5(new_h)
        
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = new_h.copy()
            h = new_h
            no_imp = 0
        else:
            # Try plain gradient step
            grad_h, _ = subgradient(h, top_m)
            new_h2 = project(h - step * grad_h, n)
            c5_2 = compute_c5(new_h2)
            if c5_2 < best_c5 - 1e-15:
                best_c5 = c5_2
                best_h = new_h2.copy()
                h = new_h2
                vel *= 0.5
                no_imp = 0
            else:
                no_imp += 1
                step *= 0.6
                if no_imp > 60 or step < 1e-9:
                    break
        if (time.time() - start_time) % 2 < 1 and no_imp == 0:
            step = min(step * 1.3, 0.3)
    
    return best_h, best_c5

def slsqp_phase(h, n, start_time, time_limit, T=400, top_m=8, maxiter=200):
    def obj(x):
        v, _ = soft_corr_value(x, T, top_m)
        return v
    def grad(x):
        return soft_corr_grad(x, T, top_m)
    def con(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, jac=grad,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        c5 = compute_c5(res.x)
        if c5 < compute_c5(h):
            return make_feasible(res.x, n), c5
    except Exception:
        pass
    return h, compute_c5(h)

def coord_fine(h, n, start_time, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    dx = 2.0 / n
    steps = [0.001, 0.0005, 0.0002, 0.0001]
    
    while time.time() - start_time < time_limit:

        top_k = np.argsort(corr)[-8:]

        for k in top_k:

            jm = js_arr - s
            important[(jm >= 0) & (jm < n)] = True
            jp = js_arr + s

            if time.time() - start_time > time_limit - 2:

- ∫h = 1 means the DC component (k=0) is 1/2
- The overlap integral at shift s: ∫ h(x)(1-h(x+s)) dx

Specifically, if h(x) = 1/2 + sum_{k=1}^{K} a_k cos(pi*k*x) + b_k sin(pi*k*x), then:

def fourier_h(coeffs, K, x):
    """Evaluate h(x) from Fourier coefficients"""
    # coeffs = [a1, b1, a2, b2, ..., aK, bK]
    h = np.full(len(x), 0.5)
    for k in range(1, K+1):
        a = coeffs[2*(k-1)]
        b = coeffs[2*(k-1)+1]
        h += a * np.cos(np.pi * k * x) + b * np.sin(np.pi * k * x)
    return h

def fourier_objective(coeffs, K, T=200):

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def subgradient(h, top_m):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_k = np.argsort(corr)[-top_m:]
    js = np.arange(n)
    grad = np.zeros(n)
    for k in top_k:
        s = k - (n - 1)
        jm = js - s
        vp = (jm >= 0) & (jm < n)
        grad[vp] += (1.0 - h[jm[vp]])
        jp = js + s
        vm = (jp >= 0) & (jp < n)
        grad[vm] -= h[jp[vm]]
    grad /= len(top_k)
    return grad, corr

def nesterov_descent(h, n, t0, time_limit, top_m=10, step0=0.04):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    vel = np.zeros(n)
    step = step0
    mu = 0.3
    no_imp = 0
    it = 0
    while time.time() - t0 < time_limit:
        it += 1
        y = np.clip(h - mu * vel, 0.0, 1.0)
        grad_y, _ = subgradient(y, top_m)
        vel = mu * vel - step * grad_y
        mu = 0.5 * (1.0 + mu)
        new_h = project(h + vel, n)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = new_h.copy()
            h = new_h
            no_imp = 0
        else:
            grad_h, _ = subgradient(h, top_m)
            new_h2 = project(h - step * 0.5 * grad_h, n)
            c5_2 = compute_c5(new_h2)
            if c5_2 < best_c5 - 1e-15:
                best_c5 = c5_2
                best_h = new_h2.copy()
                h = new_h2
                vel *= 0.5
                no_imp = 0
            else:

                step *= 0.65
                if no_imp > 80 or step < 1e-10:
                    break
        if it % 100 == 0 and no_imp == 0:
            step = min(step * 1.4, 0.4)
    return best_h, best_c5

def soft_obj(x, T, top_m, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_m < len(corr):
        idx = np.argsort(corr)[-top_m:]
        c = corr[idx]
    else:
        c = corr
    mc = np.max(c)
    sh = np.clip(c - mc, -40.0, 0.0)
    return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx

def soft_grad(x, T, top_m, n):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
    else:
        top_idx = np.arange(len(corr))
    tc = corr[top_idx]
    mc = np.max(tc)
    sh = np.clip(tc - mc, -40.0, 0.0)
    w = np.exp(T * sh)
    w /= np.sum(w)
    js = np.arange(n)
    grad = np.zeros(n)
    for wi, k in zip(w, top_idx):
        s = k - (n - 1)
        jm = js - s
        vp = (jm >= 0) & (jm < n)
        grad[vp] += wi * (1.0 - x[jm[vp]])
        jp = js + s
        vm = (jp >= 0) & (jp < n)
        grad[vm] -= wi * x[jp[vm]]
    return grad * T * dx

        return soft_grad(x, T, top_m, n)

        return np.sum(x) - n / 2.0
    def conj(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n

    try:

                       constraints=cons, jac=objg,

        pass

        top_k = np.argsort(corr)[-8:]

            jm = js - s
            important[(jm >= 0) & (jm < n)] = True
            jp = js + s
            important[(jp >= 0) & (jp < n)] = True
