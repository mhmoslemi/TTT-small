    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj_func, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return np.asarray(res.x, dtype=float)
    except Exception:
        return h.copy()

def do_coord_descent(h, n, mode):
    best_c = compute_c5(h)
    for j in range(n):
        if rem_time() < 3:
            break
        old_val = h[j]
        best_val = old_val
        best_c_local = best_c
        if mode == 'binary':
            cands = [0.0, 1.0]
        elif mode == 'fine':
            cands = [0.0, 1.0, old_val-0.001, old_val+0.001,
                     old_val-0.003, old_val+0.003, old_val-0.005, old_val+0.005]
        else:
            cands = [0.0, 1.0, old_val-0.01, old_val+0.01,
                     old_val-0.02, old_val+0.02, old_val-0.04, old_val+0.04]
        for cand in cands:
            v = float(np.clip(cand, 0.0, 1.0))
            if abs(v - old_val) < 1e-10:
                continue
            trial = h.copy()
            trial[j] = v
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c_local - 1e-14:
                best_c_local = c
                best_val = v
        if best_val != old_val:
            h[j] = best_val
            h = make_feasible(h, n)
            best_c = best_c_local
    return h, best_c

def do_block_moves(h, n):
    best_c = compute_c5(h)
    for blen in [2, 3, 4, 5, 7, 10, 14, 20]:
        if rem_time() < 3:
            break
        step = max(1, blen // 3)
        for start in range(0, n - blen, step):
            if rem_time() < 2:
                break
            block = h[start:start+blen]
            flipped = 1.0 - block
            if np.allclose(block, flipped, atol=1e-10):
                continue
            trial = h.copy()
            trial[start:start+blen] = flipped
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-13:
                best_c = c
                h = trial
    return h, best_c

def do_neighbor_swaps(h, n, max_dist=5):
    best_c = compute_c5(h)
    for d in range(1, max_dist+1):
        if rem_time() < 3:
            break
        for j in range(0, n - d, max(1, d // 2)):
            if rem_time() < 2:
                break
            if abs(h[j] - h[j+d]) < 1e-10:
                continue
            trial = h.copy()
            trial[j], trial[j+d] = h[j+d], h[j]
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-13:
                best_c = c
                h = trial
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _G_START, _G_BUDGET
    _G_START = time.time()
    _G_BUDGET = budget_s
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)
    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Progressive SLSQP with increasing sharpness
    schedule = [
        (40, 80.0, 400),
        (25, 200.0, 350),
        (15, 400.0, 350),
        (10, 800.0, 300),
        (8, 1500.0, 300),
        (5, 2500.0, 250),
        (5, 4000.0, 250),
        (3, 6000.0, 200),
        (3, 8000.0, 200),
        (2, 10000.0, 150),
        (2, 15000.0, 150),
    ]
    for (k_top, T_val, maxit) in schedule:
        if rem_time() < 45:
            break
        n_pts = len(best_h)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(-corr)[:k_top]
        h_new = slsqp_run(best_h, n_pts, top_idx, T_val, maxit)
        h_new = make_feasible(h_new, n_pts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: Binary coordinate descent (multiple passes)
    for _ in range(4):
        if rem_time() < 20:
            break
        n_pts = len(best_h)
        best_h, best_c5 = do_coord_descent(best_h, n_pts, 'binary')

    # Phase 3: Block moves + re-optimize
    if rem_time() > 60:
        n_pts = len(best_h)
        best_h, best_c5 = do_block_moves(best_h, n_pts)
        if rem_time() > 40:
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            top_idx = np.argsort(-corr)[:5]
            h_new = slsqp_run(best_h, n_pts, top_idx, 3000.0, 200)
            h_new = make_feasible(h_new, n_pts)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
            best_h, best_c5 = do_coord_descent(best_h, n_pts, 'binary')

    # Phase 4: Neighbor swaps
    if rem_time() > 40:
        n_pts = len(best_h)
        best_h, best_c5 = do_neighbor_swaps(best_h, n_pts, max_dist=4)
        if best_c5 < best_c5 + 1:  # always true, just to keep flow
            pass
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        c5_chk = float(np.max(corr) * (2.0 / n_pts))
        if c5_chk < best_c5:
            best_c5 = c5_chk

    # Phase 5: Random restarts
    for restart in range(6):
        if rem_time() < 55:
            break
        n_pts = len(best_h)
        scale = 0.012 + 0.008 * restart
        h_try = make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
        corr_t = np.correlate(h_try, 1.0 - h_try, mode='full')
        top_t = np.argsort(-corr_t)[:12]
        h_try = slsqp_run(h_try, n_pts, top_t, 400.0, 200)
        h_try = make_feasible(h_try, n_pts)
        h_try, c5_try = do_coord_descent(h_try, n_pts, 'binary')
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try
            # Additional polish from new best
            if rem_time() > 30:
                corr_n = np.correlate(best_h, 1.0 - best_h, mode='full')
                top_n = np.argsort(-corr_n)[:5]
                h_n = slsqp_run(best_h, n_pts, top_n, 3000.0, 150)
                h_n = make_feasible(h_n, n_pts)
                c5_n = compute_c5(h_n)
                if c5_n < best_c5:
                    best_c5 = c5_n
                    best_h = h_n

    # Phase 6: Try a different structural starting point (corridor-like)
    if rem_time() > 120:
        n_pts = n_init
        # Try a "two-band" pattern
        h_band = np.zeros(n_pts)
        # Place 1s in two bands to minimize auto-correlation
        band_w = n_pts // 12
        for i in range(n_pts):
            pos_frac = i / n_pts
            if (0.0 <= pos_frac < 0.08) or (0.25 <= pos_frac < 0.33) or \
               (0.5 <= pos_frac < 0.58) or (0.75 <= pos_frac < 0.83):
                h_band[i] = 1.0
        h_band = make_feasible(h_band, n_pts)
        c5_band = compute_c5(h_band)
        if c5_band < 0.42:
            for k_top in [20, 8]:
                if rem_time() < 40:
                    break
                corr_b = np.correlate(h_band, 1.0 - h_band, mode='full')
                top_b = np.argsort(-corr_b)[:k_top]
                T_b = 300.0 if k_top > 10 else 1000.0
                h_b = slsqp_run(h_band, n_pts, top_b, T_b, 250)
                h_b = make_feasible(h_b, n_pts)
                h_b, c5_b = do_coord_descent(h_b, n_pts, 'binary')
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b

    # Phase 7: Fine coordinate descent
    if rem_time() > 15:
        n_pts = len(best_h)
        best_h, best_c5 = do_coord_descent(best_h, n_pts, 'fine')

    # Phase 8: Final SLSQP polish on worst 2 shifts
    if rem_time() > 30:
        n_pts = len(best_h)
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        top_idx = np.argsort(-corr)[:2]
        h_new = slsqp_run(best_h, n_pts, top_idx, 8000.0, 150)
        h_new = make_feasible(h_new, n_pts)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
        best_h, best_c5 = do_coord_descent(best_h, n_pts, 'fine')

    # Phase 9: Ultra-fine final pass
    if rem_time() > 10:
        n_pts = len(best_h)
        for j in range(n_pts):
            if rem_time() < 3:
                break
            old_val = best_h[j]
            for step in [0.0005, 0.001, 0.002]:
                for sign in [-1, 1]:
                    v = old_val + sign * step
                    v = float(np.clip(v, 0.0, 1.0))
                    if abs(v - old_val) < 1e-12:
                        continue
                    trial = best_h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n_pts)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        best_c5 = c
                        best_h = trial
                        old_val = v

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
