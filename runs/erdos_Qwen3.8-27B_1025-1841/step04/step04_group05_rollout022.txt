def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff))+1:]
        else:
            idx = np.argsort(h)[:int(abs(diff))+1:]
        for i in idx:
            adj = min(diff if diff > 0 else 0, 1.0 - h[i]) if diff > 0 else max(diff if diff < 0 else 0, -h[i])
            h[i] += adj
            diff -= (adj if diff > 0 else adj)
            if abs(diff) < 1e-14:
                break
    return h

            sub = min(-diff, h[idx])
            h[idx] -= sub
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)
