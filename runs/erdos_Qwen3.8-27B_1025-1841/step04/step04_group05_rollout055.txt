    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(T * shifted)
    weights /= np.sum(weights)
    
    # Gradient of sum_l weights[l] * corr[l] with respect to h
    # corr[j] where j = 0..2n-2, lag l = j - (n-1)
    # corr[j] = sum_{valid i} h[i + l] * (1-h[i])
    # where l = j - (n-1)

    # where l = j - (n-1)
    
    grad = np.zeros(n)
    for j in range(2*n - 1):
        l = j - (n - 1)
        wj = weights[j]
        if wj < 1e-15:
            continue
        # First term: (1-h[m-l]) for valid m
        # m-l in [0, n-1] and m in [0, n-1]

        if m_end > m_start:

    grad *= dx
    return grad

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argsort(h)[-1]
        else:
            idx = np.argsort(h)[0]
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)
