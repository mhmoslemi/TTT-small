    # Softmax weights
    if top_m is not None and top_m < len(corr):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def golden_ratio_init(n):
    """Initialize with golden ratio Beatty sequence."""
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    # Adjust count
    ones = np.sum(h)
    target = n / 2
    if ones > target:
        # Remove excess ones
        # Remove ones that are "least supported" (neighbors are also 1)
        scores = h.copy()
        for d in range(1, min(5, n)):
            scores += np.roll(h, d) + np.roll(h, -d)
        excess = int(ones - target)
        one_idx = np.where(h == 1.0)[0]
        # Remove the ones with highest scores (most clustered)
        remove_order = one_idx[np.argsort(-scores[one_idx])]
        h[remove_order[:excess]] = 0.0
    elif ones < target:
        deficit = int(target - ones)
        zero_idx = np.where(h == 0.0)[0]
        scores = -h.copy()
        for d in range(1, min(5, n)):
            scores -= np.roll(h, d) + np.roll(h, -d)
        add_order = zero_idx[np.argsort(-scores[zero_idx])]
        h[add_order[:deficit]] = 1.0
    return h

def smooth_grad(h, n, T, top_m=10):
    """Vectorized gradient of smooth max of correlations."""
    corr = compute_corr(h)
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        c = corr[top_idx]
        w = np.exp(T * (c - c.max()))
        w /= w.sum()
    else:
        top_idx = np.arange(len(corr))
        w = np.exp(T * (corr - corr.max()))
        w /= w.sum()
    
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        wt = w[idx]
        if wt < 1e-15:
            continue
        m = top_idx[idx]
        s = m - (n - 1)
        # ∂corr_m/∂h[i] = (1-h[i-s])*[0≤i-s<n] - h[i+s]*[0≤i+s<n]
        lo1, hi1 = max(0, s), min(n, s + n)
        lo2, hi2 = max(0, -s), min(n, n - s)
        if hi1 > lo1:
            grad[lo1:hi1] += wt * (1.0 - h[lo1 - s:hi1 - s])
        if s != 0:
            if hi2 > lo2:
                grad[lo2:hi2] -= wt * h[lo2 + s:hi2 + s]
        else:
            if hi2 > lo2:
                grad[lo2:hi2] -= wt * h[lo2:hi2]
    
    return grad, corr

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def grad_desc(h, n, budget_s, start_time, T_start=20.0, T_end=500.0, top_m=10, lr=None):
    """Nesterov gradient descent on smooth max."""
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = 0.85
    
    if lr is None:
        lr = 1.0 / (2.0 * T_end)  # rough Lipschitz bound
    
    it = 0
    n_it = 0
    while time.time() - start_time < budget_s - 3:
        it += 1
        n_it += 1
        T = T_start + (T_end - T_start) * min(1.0, it / max(50, n_it // 3))
        
        # Nesterov
        y = np.clip(h - momentum * v, 0.0, 1.0)
        grad_y, _ = smooth_grad(y, n, T, top_m)
        
        # Scale gradient
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-15:
            break
        
        # Adaptive step: try to reduce the objective
        step = lr * (1.0 + 0.5 * np.sin(it * 0.01))  # oscillate to escape
        v = momentum * v - step * grad_y
        new_h = h + v
        new_h = project_feasible(new_h, n)
        
        c5_new = compute_c5(new_h)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
        
        # If worsened, reduce step
        c5_cur = compute_c5(h)
        if c5_new > c5_cur * 1.01:
            lr *= 0.8
            v *= 0.5
        elif it % 100 == 0:
            lr = min(1.0, lr * 1.05)
        
        h = new_h
        
        if lr < 1e-8:
            break
    
    return best_h, best_c5

1. For each j with h[j]=1 and delta_j[s*] = -1 (reduces the max):
2.   Find m with h[m]=0 that minimizes max(corr + delta_j + delta_m)

For step 2, we can approximate by looking at the top few shifts and checking at those.

OK let me just write a practical implementation that's fast enough:

def binary_flip_search(h, n, budget_s, start_time):
    h = h.copy()
    idx = np.argsort(h)[::-1]
    h[idx[:n//2]] = 1.0
    h[idx[n//2:]] = 0.0
    
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    dx = 2.0/n
    
    ones = np.where(h==1.0)[0]
    zeros = np.where(h==0.0)[0]
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 2:
            break
        
        max_idx = int(np.argmax(corr))
        s = max_idx - (n-1)
        
        # Find j: h[j]=1, h[j+s]=0, h[j-s]=0 (if valid)
        # This gives delta_j[max_idx] = -1
        candidate_js = []
        for j in ones:
            cond = True
            if 0 <= j+s < n:
                if h[j+s] != 0:
                    cond = False
            if 0 <= j-s < n and cond:
                if h[j-s] != 0:
                    cond = False
            if cond:
                candidate_js.append(j)
        
        if not candidate_js:
            # Relax: just find j where h[j]=1 and h[j+s]=0
            for j in ones:
                if 0 <= j+s < n and h[j+s] == 0:
                    candidate_js.append(j)
        
        if not candidate_js:
            break
        
        best_improvement = 0
        best_pair = None
        
        for j in candidate_js[:30]:
            # Compute delta_j
            delta_j = np.zeros(2*n-1)
            for s2 in range(-n+1, n):
                d = 0
                if 0 <= j+s2 < n:
                    d += int(h[j+s2])
                if 0 <= j-s2 < n:
                    d -= int(1-h[j-s2])
                delta_j[n-1+s2] = d
            # Wait, this is O(n) per j, and we have 30 js, so 30*n = 28800. Fine.
            
            new_corr_j = corr + delta_j
            new_max_j = new_corr_j.max()
            
            if new_max_j >= corr[max_idx]:
                continue  # Doesn't help at the max shift
            
            # Find best m
            for m in zeros[:50]:
                if m == j:
                    continue
                # delta_m = -delta_j pattern but for m
                # Actually delta_m[s2] = (1-h[m-s2]) - h[m+s2] (for 0→1 flip)
                # We need to compute this quickly
                # Approximate: just check the effect at the top few shifts
                new_max = new_max_j
                for s2 in [s, int(np.argsort(corr + delta_j)[-2] - (n-1))]:
                    if 0 <= m+s2 < n:
                        new_val = new_corr_j[n-1+s2] + (1-h[m-s2] if 0<=m-s2<n else 0) - h[m+s2]
                    elif 0 <= m-s2 < n:
                        new_val = new_corr_j[n-1+s2] + (1-h[m-s2]) - 0
                    else:
                        new_val = new_corr_j[n-1+s2]
                    new_max = max(new_max, new_val)
                
                if new_max < best_c5/dx - 0.5:  # at least 0.5 improvement in count
                    best_improvement = best_c5 - new_max*dx
                    best_pair = (j, m, delta_j)
                    break
        
        if best_pair is None:
            break
        
        j, m, delta_j = best_pair
        h[j] = 0.0
        h[m] = 1.0
        corr = compute_corr(h)
        best_c5 = compute_c5(h)
        ones = np.where(h==1.0)[0]
        zeros = np.where(h==0.0)[0]
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def golden_ratio_init(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    fr = (np.arange(n) * phi) % 1.0
    h[fr < 0.5] = 1.0
    ones = np.sum(h)
    target = n // 2
    if ones > target:
        idx1 = np.where(h == 1.0)[0]
        scores = np.zeros(n)
        for d in [1, 2, 3, 5, 8]:
            scores += np.roll(h, d) + np.roll(h, -d)
        order = idx1[np.argsort(-scores[idx1])]
        h[order[:int(ones - target)]] = 0.0
    elif ones < target:
        idx0 = np.where(h == 0.0)[0]
        scores = np.zeros(n)
        for d in [1, 2, 3, 5, 8]:
            scores += np.roll(h, d) + np.roll(h, -d)
        order = idx0[np.argsort(scores[idx0])]
        h[order[:int(target - ones)]] = 1.0
    return h

def smooth_grad(h, n, T, top_m=10):
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_m < len(corr):
        top_idx = np.argsort(corr)[-top_m:]
        c = corr[top_idx]
        w_raw = np.exp(T * (c - c.max()))
        w_total = w_raw.sum()
        if w_total < 1e-300:
            w_raw = np.ones(len(c))
        else:
            w_raw = w_raw / w_total
    else:
        top_idx = np.arange(len(corr))
        w_raw = np.exp(T * (corr - corr.max()))
        w_total = w_raw.sum()
        w_raw = w_raw / w_total
    grad = np.zeros(n)
    for idx in range(len(top_idx)):
        wt = w_raw[idx]
        if wt < 1e-14:
            continue
        m = int(top_idx[idx])
        s = m - (n - 1)
        lo1 = max(0, s)
        hi1 = min(n, s + n)
        if hi1 > lo1:
            grad[lo1:hi1] += wt * (1.0 - h[lo1 - s:hi1 - s])
        lo2 = max(0, -s)
        hi2 = min(n, n - s)
        if hi2 > lo2 and s != 0:
            grad[lo2:hi2] -= wt * h[lo2 + s:hi2 + s]
        elif s == 0:
            grad[lo2:hi2] -= wt * h[lo2:hi2]
    return grad, corr

def nesterov_opt(h, n, budget_s, start_time, T_start=5.0, T_end=1000.0, top_m=10, max_iters=5000):
    h = h.copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    momentum = 0.85
    lr = 0.5
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 2:
            break
        frac = min(1.0, it / (max_iters * 0.7))
        T = T_start * (T_end / T_start) ** frac
        
        y = h - momentum * v
        y = np.clip(y, 0.0, 1.0)
        grad_y, _ = smooth_grad(y, n, T, top_m)
        
        gnorm = np.linalg.norm(grad_y)
        if gnorm < 1e-14:
            break
        
        v = momentum * v + lr * grad_y
        new_h = h - v
        new_h = project_feasible(new_h, n)
        
        c5_new = compute_c5(new_h)
        if c5_new < best_c5 - 1e-15:
            best_h = new_h.copy()
            best_c5 = c5_new
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                lr *= 0.7
                v *= 0.3
                no_improve = 0
                if lr < 1e-9:
                    break
        
        h = new_h
        if it % 200 == 0 and lr < 2.0:
            lr = min(2.0, lr * 1.1)
    
    h_final = project_feasible(h, n)
    c5_f = compute_c5(h_final)
    if c5_f < best_c5:
        best_h, best_c5 = h_final, c5_f
    return best_h, best_c5

def binary_project(h, n):
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    half = n // 2
    b[idx[:half]] = 1.0
    return b

def binary_greedy(h, n, budget_s, start_time):
    h = binary_project(h, n)
    best_c5 = compute_c5(h)
    corr = compute_corr(h)
    dx = 2.0 / n
    threshold = best_c5 / dx
    
    for it in range(5000):
        if time.time() - start_time > budget_s - 2:
            break
        
        max_idx = int(np.argmax(corr))
        s = max_idx - (n - 1)
        ones = np.where(h > 0.5)[0]
        zeros = np.where(h <= 0.5)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        found = False
        np.random.shuffle(ones)
        for j in ones:
            if time.time() - start_time > start_time + budget_s - 2:
                break
            # Check if flipping j (1→0) helps at shift s
            # delta at s: h[j+s] - (1-h[j-s])
            dj_s = 0.0
            if 0 <= j + s < n:
                dj_s += h[j + s]
            if 0 <= j - s < n:
                dj_s -= (1.0 - h[j - s])
            if dj_s >= 0:
                continue
            
            # Try to find m (0→1) that doesn't make things worse
            np.random.shuffle(zeros)
            for m in zeros:
                if m == j:
                    continue
                h[j] = 0.0
                h[m] = 1.0
                c_new = compute_c5(h)
                h[j] = 1.0
                h[m] = 0.0
                if c_new < best_c5 - 1e-15:
                    h[j] = 0.0
                    h[m] = 1.0
                    best_c5 = c_new
                    corr = compute_corr(h)
                    threshold = best_c5 / dx
                    found = True
                    break
            if found:
                break
        if not found:
            break
    
    return h, best_c5

def slsqp_refine(h, n, T=500.0, top_m=8, maxiter=300):
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        if top_m < len(c):
            top = np.sort(c)[-top_m:]
        else:
            top = c
        mx = top.max()
        vals = np.exp(T * (top - mx))
        return (mx + np.log(vals.sum() / len(top)) / T) * dx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        h_r = project_feasible(res.x, n)
        c5_r = compute_c5(h_r)
        if c5_r < compute_c5(h):
            return h_r, c5_r
    except Exception:
        pass
    return h, compute_c5(h)
