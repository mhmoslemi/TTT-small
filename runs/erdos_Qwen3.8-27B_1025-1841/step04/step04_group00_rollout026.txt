        if s == 0:
            # C[0] = sum h[i]^2 ... wait no
            # C[0] = sum_i h[i] * (1-h[i]) = sum(h - h^2)
            # dC[0]/dh[j] = 1 - 2*h[j]
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            # First term: (1-h[j-s]) for j in [s, n-1]
            grad[s:] += (1.0 - h[:n-s])
            # Second term: -h[j+s] for j in [0, n-1-s]
            grad[:n-s] -= h[s:]
        else:  # s < 0
            # First term: (1-h[j-s]) for j in [0, n-1+s]
            grad[:n+s] += (1.0 - h[-s:])
            # Second term: -h[j+s] for j in [-s, n-1]
            grad[-s:] -= h[:n+s]
    return grad / top_m, corr

def bb_step(h_prev, g_prev, h_curr, g_curr, step_prev):
    """Barzilai-Borwein step size."""
    s = h_curr - h_prev
    y = g_curr - g_prev
    ss = np.dot(s, s)
    yy = np.dot(y, y)
    sy = np.dot(s, y)
    if yy < 1e-16 or ss < 1e-16:
        return step_prev
    bb_plus = ss / sy if sy > 1e-16 else step_prev
    bb_minus = ss / yy
    # Clip to reasonable range
    bb_plus = np.clip(bb_plus, 1e-8, 10.0)
    bb_minus = np.clip(bb_minus, 1e-8, 10.0)
    return 0.5 * (bb_plus + bb_minus)

def subgradient_optimize(h, n, budget_s, start_time, top_m=10, max_iter=5000):
    """Subgradient method with BB step size and Nesterov momentum."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.05
    momentum = 0.7
    velocity = np.zeros(n)
    h_prev = h.copy()
    g_prev = None
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        grad, corr = subgradient_top(h, n, top_m)
        c5_cur = np.max(corr) * (2.0 / n)
        
        if c5_cur < best_c5 - 1e-12:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Nesterov lookahead
        y = h - momentum * velocity
        y = np.clip(y, 0.0, 1.0)
        grad_y, _ = subgradient_top(y, n, top_m)
        
        # BB step size
        if g_prev is not None:
            step = bb_step(h_prev, g_prev, h, grad, step)
        
        # Update
        velocity = momentum * velocity - step * grad_y
        h_new = h + velocity
        h_new = make_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur + 1e-10:
            h_prev = h.copy()
            g_prev = grad.copy()
            h = h_new
        else:
            # Backtrack
            step *= 0.5
            velocity *= 0.8
            h_new2 = make_feasible(h - step * grad, n)
            c5_new2 = compute_c5(h_new2)
            if c5_new2 < c5_cur + 1e-10:
                h_prev = h.copy()
                g_prev = grad.copy()
                h = h_new2
            else:
                step *= 0.5
                if step < 1e-8:
                    break
        
        # Decay momentum
        momentum = max(0.3, momentum - 0.001)
    
    h = make_feasible(h, n)
    c5_h = compute_c5(h)
    if c5_h < best_c5:
        best_h, best_c5 = h, c5_h
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_top(h, n, top_m):
    corr = np.correlate(h, 1.0 - h, mode='full')
    top_idx = np.argsort(corr)[-top_m:]
    grad = np.zeros(n)
    for k in top_idx:
        s = k - (n - 1)
        if s == 0:
            grad += (1.0 - 2.0 * h)
        elif s > 0:
            grad[s:] += (1.0 - h[:n - s])
            grad[:n - s] -= h[s:]
        else:
            grad[:n + s] += (1.0 - h[-s:])
            grad[-s:] -= h[:n + s]
    return grad / top_m, corr

def bb_step_size(h_prev, g_prev, h_curr, g_curr, step_prev):
    s_vec = h_curr - h_prev
    y_vec = g_curr - g_prev
    ss = np.dot(s_vec, s_vec)
    yy = np.dot(y_vec, y_vec)
    sy = np.dot(s_vec, y_vec)
    if yy < 1e-18 or ss < 1e-18:
        return step_prev
    if sy > 1e-12:
        bb_plus = ss / sy
    else:
        bb_plus = step_prev
    bb_minus = ss / yy
    bb_plus = np.clip(bb_plus, 1e-7, 5.0)
    bb_minus = np.clip(bb_minus, 1e-7, 5.0)
    return 0.5 * (bb_plus + bb_minus)

def subgradient_optimize(h, n, budget_s, start_time, top_m=10, max_iter=8000):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.03
    momentum = 0.5
    velocity = np.zeros(n)
    h_prev = h.copy()
    g_prev = None
    no_improve = 0
    max_no_improve = 200

    for it in range(max_iter):
        if time.time() - start_time > budget_s - 3:
            break
        grad, corr = subgradient_top(h, n, top_m)
        c5_cur = np.max(corr) * (2.0 / n)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > max_no_improve:
                step = max(step * 0.7, 0.005)
                momentum = max(0.2, momentum - 0.1)
                no_improve = 0
                if step < 0.002:
                    break

        y = h - momentum * velocity
        y_clipped = np.clip(y, 0.0, 1.0)
        grad_y, _ = subgradient_top(y_clipped, n, top_m)

        if g_prev is not None:
            step_new = bb_step_size(h_prev, g_prev, h, grad, step)
            step = 0.7 * step + 0.3 * step_new

        h_new = make_feasible(h_new, n)

def smooth_max_obj(x, n, T, top_m=None):
    corr = np.correlate(x, 1.0 - x, mode='full')
    dx = 2.0 / n
    if top_m is not None and top_m < len(corr):
        top = np.argsort(corr)[-top_m:]
        mc = np.max(corr[top])
        shifted = corr[top] - mc
        obj = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    else:
        mc = np.max(corr)
        shifted = corr - mc
        obj = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    return obj

def slsqp_polish(h, n, budget_s, start_time, T=2000.0, top_m=8):
    def obj(x):
        return smooth_max_obj(x, n, T, top_m)

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 200, 'ftol': 1e-18})
        h_res = make_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        if c5_res < compute_c5(h) - 1e-12:
            return h_res, c5_res
    except Exception:
        pass
    return h, compute_c5(h)

    ones_idx = np.where(best_h > 0.5)[0]
    zeros_idx = np.where(best_h <= 0.5)[0]

    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 5:
            break
        improved = False
        for _ in range(len(ones_idx)):
            if time.time() - start_time > budget_s - 5:
                break
            j = int(ones_idx[np.random.randint(len(ones_idx))])
            k = int(zeros_idx[np.random.randint(len(zeros_idx))])
            trial = best_h.copy()
            trial[j] = 0.0
            trial[k] = 1.0
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-13:

                best_c5 = c
                improved = True
                ones_idx = np.where(best_h > 0.5)[0]
                zeros_idx = np.where(best_h <= 0.5)[0]
                break
        if not improved:
            break
    return best_h, best_c5

def coord_descent_fast(h, n, budget_s, start_time, steps=None):
    if steps is None:
        steps = [0.01, 0.005, 0.002, 0.001]

    # Phase 5: Coordinate descent
    if remaining() > 50:
        h, c5 = coord_descent_fast(best_h, n, min(60, remaining() - 10), start_time)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 6: SLSQP with higher T
    if remaining() > 40:
        h, c5 = slsqp_polish(best_h, n, remaining() - 10, start_time, T=3000.0, top_m=5)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 7: Try upsampling to 2n
    if remaining() > 250:
        n2 = n * 2
        if n2 <= 2880:
            old_x = np.linspace(0.0, 2.0, n)
            new_x = np.linspace(0.0, 2.0, n2)
            h_up = np.interp(new_x, old_x, best_h)
            h_up = make_feasible(h_up, n2)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5 + 0.005:
                if remaining() > 120:
                    h2, c5_2 = subgradient_optimize(h_up, n2, min(120, remaining() - 60), start_time, top_m=10)
                    if c5_2 < best_c5:
                        best_h, best_c5 = h2, c5_2
                if remaining() > 60:
                    h2, c5_2 = slsqp_polish(best_h, len(best_h), remaining() - 20, start_time, T=1000.0, top_m=10)
                    if c5_2 < best_c5:
                        best_h, best_c5 = h2, c5_2
                if remaining() > 40:
                    h2, c5_2 = coord_descent_fast(best_h, len(best_h), min(40, remaining() - 10), start_time,
                                                   steps=[0.005, 0.002, 0.001])
                    if c5_2 < best_c5:
                        best_h, best_c5 = h2, c5_2

    # Phase 8: Random restarts with subgradient
    for restart in range(6):
        if remaining() < 50:
            break
        n_pts = len(best_h)
        scales = [0.01, 0.015, 0.02, 0.03, 0.01, 0.005]
        scale = scales[min(restart, len(scales) - 1)]
        h_try = make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
        h2, c5_2 = subgradient_optimize(h_try, n_pts, min(40, remaining() - 20), start_time, top_m=8)
        if c5_2 < best_c5:
            best_h, best_c5 = h2, c5_2

    # Phase 9: Final subgradient with very tight top_m
    if remaining() > 40:
        h, c5 = subgradient_optimize(best_h, len(best_h), min(40, remaining() - 10), start_time, top_m=3)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 10: Final SLSQP
    if remaining() > 30:
        h, c5 = slsqp_polish(best_h, len(best_h), remaining() - 5, start_time, T=5000.0, top_m=3)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 11: Ultra-fine coordinate descent
    if remaining() > 15:
        h, c5 = coord_descent_fast(best_h, len(best_h), min(15, remaining() - 3), start_time,
                                    steps=[0.001, 0.0005, 0.0002])
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
