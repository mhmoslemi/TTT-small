import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 0
_OPT_P = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def pnorm_obj(h):
    dx = 2.0 / _OPT_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr = np.maximum(corr, 1e-30)
    return (np.sum(corr ** _OPT_P)) ** (1.0 / _OPT_P) * dx

def eq_con(h):
    return np.sum(h) - _OPT_N / 2.0

def eq_con_jac(h):
    return np.ones(_OPT_N)

# Harmonic helpers
_HARM_N = 0
_HARM_M = 0
_HARM_P = 1.0

def harmonic_params_to_h(params, N, M):
    a = params[:M]
    b = params[M:2*M]
    t = np.arange(N)
    h = np.full(N, 0.5)
    for m in range(1, M + 1):
        h += a[m-1] * np.cos(2.0 * np.pi * m * t / N)
        h += b[m-1] * np.sin(2.0 * np.pi * m * t / N)
    return np.clip(h, 0.0, 1.0)

def harmonic_obj(params):
    N = _HARM_N
    M = _HARM_M
    h = harmonic_params_to_h(params, N, M)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    corr = np.maximum(corr, 1e-30)
    return (np.sum(corr ** _HARM_P)) ** (1.0 / _HARM_P) * dx

def slsqp_optimize(h_start, n, p, max_iter):
    global _OPT_N, _OPT_P
    _OPT_N = n
    _OPT_P = p
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': eq_con, 'jac': eq_con_jac}]
    res = minimize(pnorm_obj, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def binary_local_search(h_init, n, max_iters):

    h = (h_init > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng_local = np.random.RandomState(0)
        to_flip = rng_local.choice(idx, n_ones - target, replace=False)
        h[to_flip] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng_local = np.random.RandomState(0)
        to_flip = rng_local.choice(idx, target - n_ones, replace=False)
        h[to_flip] = 1
    
    c5_cur = compute_c5(h)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))  # lag index in [0, 2n-2]
        k_val = k_star - n + 1  # actual lag value
        
        # Try to find a beneficial pair flip
        # For flipping h[j]: 0→1 and h[i]: 1→0
        # Δc[k] = (1-h[j-k]) - h[j+k] - (1-h[i-k]) + h[i+k] (for valid indices)
        
        best_delta = 0
        best_j = -1
        best_i = -1
        
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        # For efficiency, only try a subset of candidates
        # The change in c at lag k_val for flipping j (0→1):
        # Δ_j[k] = (1-h[j-k_val] if valid) - (h[j+k_val] if valid)
        
        best_total_delta = 0
        for j in zeros:
            d_j = 0.0
            j_minus = j - k_val
            j_plus = j + k_val
            if 0 <= j_minus < n:
                d_j += (1.0 - h[j_minus])
            if 0 <= j_plus < n:
                d_j -= h[j_plus]
            # Now find i in ones that minimizes:
            # -[(1-h[i-k_val]) - h[i+k_val]]
            # = -(1-h[i-k_val] if valid) + (h[i+k_val] if valid)
            # We want this to be as small as possible (making total delta most negative)
            # Total delta = d_j - d_i where d_i = (1-h[i-k_val]) - h[i+k_val]
            # We want d_j - d_i < 0, i.e., d_i > d_j
            
            # For simplicity, just check a few candidates

                d_i = 0.0
                i_minus = i - k_val
                i_plus = i + k_val
                if 0 <= i_minus < n:
                    d_i += (1.0 - h[i_minus])
                if 0 <= i_plus < n:
                    d_i -= h[i_plus]
                total_d = d_j - d_i
                if total_d < best_delta - 1e-12:
                    best_delta = total_d
                    best_j = j
                    best_i = i

                c5_cur = new_c5
            else:
                h[best_j] = 0.0

            new_c5 = compute_c5(h)
            if new_c5 > c5_cur + 0.001:
