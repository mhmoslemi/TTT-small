    grad += 2.0 * mu * (np.sum(h) - n/2.0)
    return grad

def proj_subgradient(h0, n, T, mu, n_iters=200, lr=None):
    """Projected subgradient / mirror descent method."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        g = fourier_grad(h, n, T, mu)
        if lr is None:
            lr = 0.1 / (it + 10)
        h_new = h - lr * g
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def fmin_lbfgs(T, mu, h0, n, maxiter=500):
    def obj(x):
        return fourier_obj(x, n, T, mu)
    def grad(x):
        return fourier_grad(x, n, T, mu)
    res = fmin_l_bfgs_b(obj, h0, fprime=grad, m=100, pgtol=1e-8, 
                          maxiter=maxiter, iprint=0, factr=1e6)
    h_new = make_feasible(res[0], n)
    return h_new, res[0]

def binary_swap_search(h_bin, n, rng, max_rounds=100, tries_per_round=3000):
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    
    for rnd in range(max_rounds):
        improved = False
        n_ones = len(ones)
        n_zeros = len(zeros)
        if n_ones == 0 or n_zeros == 0:
            break
        n_tries = min(tries_per_round, n_ones * n_zeros)
        for _ in range(n_tries):
            ii = rng.randint(n_ones)
            jj = rng.randint(n_zeros)
            oi = ones[ii]
            oj = zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < cur_c5 - 1e-15:
                cur_c5 = c
                # Update lists
                ones[ii] = oj
                zeros[jj] = oi
                improved = True
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:
            break
    return h, cur_c5

def binary_block_search(h_bin, n, rng, block_sizes=[2,3,4,5,6], max_rounds=50):
    """Try swapping blocks of consecutive 1s with consecutive 0s."""
    h = h_bin.copy()
    cur_c5 = compute_c5(h)
    
    for rnd in range(max_rounds):
        improved = False
        for bs in block_sizes:
            for start_1 in range(n - bs + 1):
                block_1 = h[start_1:start_1+bs]
                if np.any(block_1 == 0):
                    continue
                for start_0 in range(n - bs + 1):
                    block_0 = h[start_0:start_0+bs]
                    if np.any(block_0 == 1):
                        continue
                    # Try the swap
                    h[start_1:start_1+bs] = 0.0
                    h[start_0:start_0+bs] = 1.0
                    c = compute_c5(h)
                    if c < cur_c5 - 1e-15:
                        cur_c5 = c
                        improved = True
                    else:
                        h[start_0:start_0+bs] = 0.0
                        h[start_1:start_1+bs] = 1.0
                    if improved:
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, cur_c5

def spectral_flattening(h0, n, n_iters=100, alpha=0.1):
    """Iteratively adjust h to flatten the power spectrum."""
    h = h0.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        H = np.fft.fft(h)
        r = np.abs(H)**2
        # Target: flat spectrum for j≠0
        r_target = np.sum(r[1:]) / (n-1)
        r_target = np.full(n, r_target)
        r_target[0] = r[0]  # Keep DC
        
        # Adjust: increase power at weak frequencies, decrease at strong
        delta_r = alpha * (r_target - r)
        delta_r[0] = 0
        
        # Convert spectrum change to time-domain change
        # This is approximate: δh ≈ IDFT(δ√H * phase(H))
        # Simpler: just do a small step in the time domain
        # Autocorrelation adj = IDFT(δr)
        adj = np.fft.ifft(delta_r).real
        h_new = h + 0.01 * adj
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

For binary sequences, this is related to "aperiodic autocorrelation" or "Barker sequences" (minimizing aperiodic autocorrelation).

A Barker sequence of length n has the property that its aperiodic autocorrelation is 0 or -1 for all non-zero shifts. This is exactly what we want!

Known Barker sequences exist for n = 2, 3, 4, 5, 7, 11, 13. For larger n, we can use "low-PSD" or "low-autocorrelation" sequences.
