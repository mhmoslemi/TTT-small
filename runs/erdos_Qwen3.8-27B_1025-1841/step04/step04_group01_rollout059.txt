import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0
_G_TOP_IDX = None

def _softmax_obj(x):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    idx = _G_TOP_IDX
    if idx is not None:
        corr = corr[idx]
    mc = np.max(corr)
    sh = corr - mc
    return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def run_slsqp(h, n, T, top_idx=None, maxiter=300):
    global _G_N, _G_T, _G_TOP_IDX
    _G_N = n
    _G_T = T
    _G_TOP_IDX = top_idx
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_softmax_obj, h, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-15})
        return make_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_desc(h, n, passes, steps, t0, budget_s):
    best_c = compute_c5(h)
    for p in range(passes):
        if (time.time() - t0) > budget_s - 15:
            break
        improved = False
        for j in range(n):
            if (time.time() - t0) > budget_s - 8:
                return h, best_c
            old = h[j]
            best_v = old
            best_cj = best_c
            for s in steps:
                for sg in (-1.0, 1.0):
                    v = old + sg * s
                    if v < -1e-10 or v > 1.0 + 1e-10:
                        continue
                    v = max(0.0, min(1.0, v))
                    if abs(v - old) < 1e-12:
                        continue
                    trial = h.copy()
                    trial[j] = v
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_cj - 1e-15:
                        best_cj = c
                        best_v = v
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c = best_cj
                improved = True
        if not improved:
            break
    return h, best_c

def get_top_idx(h, frac=0.003, max_count=20):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    thr = mc - frac * mc
    idx = np.where(corr >= thr)[0]
    if len(idx) > max_count:
        sv = corr[idx]
        idx = idx[np.argsort(-sv)[:max_count]]
    if len(idx) == 0:
        idx = np.array([int(np.argmax(corr))])
    return idx

def sa_binary(h_start, n, budget_s, rng):
    h = (h_start > 0.5).astype(float)
    h = make_feasible(h, n)
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    T = 0.003
    T_min = 0.00001
    t_start = time.time()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]

        T = max(T * 0.99998, T_min)

        trial = h.copy()
        trial[j] = 0.0
        trial[k] = 1.0
        c_new = compute_c5(trial)
        delta = c_new - curr_c
        if delta < 0 or (T > T_min * 2 and rng.random() < np.exp(-delta / T)):
            h = trial
            curr_c = c_new
            if c_new < best_c:

                best_h = h.copy()
            ones = np.where(h > 0.5)[0]

    except Exception:
        return h.copy(), compute_c5(h)
