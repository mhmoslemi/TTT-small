- But we need to sum over all k, which is O(N) per j.
- Total gradient computation: O(N²).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(500):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute among all non-boundary elements
        mask = (h > 1e-6) & (h < 1.0 - 1e-6)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def power_mean_c5(h, p):
    """Compute power-mean smooth approximation of C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.maximum(corr, 0.0)
    M = len(c)
    cmax = np.max(c)
    if cmax < 1e-30:
        return 0.0, c
    cn = c / cmax
    val = cmax * (np.sum(cn**p) / M)**(1.0/p) * dx
    return val, c

def power_mean_grad(h, p):
    """Compute gradient of power-mean C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = np.maximum(corr, 0.0)
    M = len(c)
    cmax = np.max(c)
    if cmax < 1e-30:
        return np.zeros(n)
    cn = c / cmax
    sp = np.sum(cn**p) / M
    if sp < 1e-30:
        return np.zeros(n)
    # dF/dc[m] = (1/M) * sp^(1/p - 1) * cn[m]^(p-1) * (1/cmax) ... wait
    # F = cmax * sp^(1/p) * dx
    # dF/dc[m] = sp^(1/p) * dx * d(cmax)/dc[m] * ... this is messy because cmax depends on c.
    
    # Let me just use: F = (sum c^p / M)^(1/p) * dx (without normalization by cmax)
    # This is numerically less stable but simpler.

    # Use log-sum-exp trick:
    # log(sum c^p / M) = p*log(cmax) + log(sum (c/cmax)^p / M)
    # F = exp((1/p) * log(sum c^p / M)) * dx = cmax * (sum (c/cmax)^p / M)^(1/p) * dx

    #           = dx * (1/M) * (sum c^p / M)^(1/p - 1) * c[m]^(p-1)

    prefac = dx * sp**(1.0/p - 1.0) / M
    w = prefac * cn**(p-1)  # length 2n-1

    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.05
    n_steps_done = 0
    
    for step in range(max_steps):
        remaining = budget_s - (time.time() - start_time)
        if remaining < 5:
            break
        # Adaptive p
        frac = float(step) / max_steps
        p = p_start + (p_end - p_start) * frac**0.7
        p = max(p, 2.0)
        
        # Compute gradient
        grad = power_mean_grad(h, p)
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-15:
            break
        
        # Gradient descent step with backtracking
        for _ in range(30):
            h_trial = h - lr * grad / (grad_norm + 1e-15)  # normalize grad
            # Actually, better to not normalize: just step by lr * grad
            h_trial = h - lr * grad
            h_trial = make_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial <= best_c5:  # accept if improving or non-worsening vs best
                break
            lr *= 0.5
        else:
            lr = max(lr, 1e-8)
            h_trial = make_feasible(h - lr * grad, n)
        
        h = h_trial
        n_steps_done += 1
        
        # Track best
        c5_curr = compute_c5(h)
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
        
        # Adapt learning rate
        if step % 50 == 0 and step > 0:
            # Increase lr slowly
            lr = min(lr * 1.05, 0.5)
    
    return best_h, best_c5, n_steps_done
