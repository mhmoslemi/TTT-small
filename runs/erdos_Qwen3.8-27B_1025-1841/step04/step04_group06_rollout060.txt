def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(h)[-int(abs(diff))+1:] if abs(diff) > 1 else [int(np.argmin(h))]
            for i in idx:
                h[i] = min(1.0, h[i] + diff / len(idx))
        else:
            idx = np.argsort(h)[:int(abs(diff))+1] if abs(diff) > 1 else [int(np.argmax(h))]
            for i in idx:
                h[i] = max(0.0, h[i] + diff / len(idx))
    return h

def upsample_h(h, new_n):

        return make_feasible(h.copy(), new_n)
    old_x = np.arange(old_n) * 2.0 / old_n

    x_tiled = np.concatenate([np.arange(-old_n//10, 0) * 2.0/old_n, old_x, 
                               np.arange(old_n, old_n + old_n//10) * 2.0/old_n])

        mc = np.max(corr)

    h = h.copy()

    for it in range(iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        k = k_star - (n - 1)  # actual lag
        # Gradient of corr[k] w.r.t. h[i]
        # corr[k] = sum_n h[n+k]*(1-h[n]) for valid n
        # d/dh[i] = (1-h[i-k]) - h[i+k] (with bounds)

            #           - h[i+k] for i from 0 to n-1-k (from second term)

            else:

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    pad = old_n // 8
    x_old = np.arange(-pad, old_n + pad) * 2.0 / old_n
    h_pad = np.concatenate([h[-pad:], h, h[:pad]])
    x_new = np.arange(new_n) * 2.0 / new_n
    h_new = np.interp(x_new, x_old, h_pad)
    return make_feasible(h_new, new_n)

def smooth_obj(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(T * (corr - mc), -100, 100)
    soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    viol = np.sum(h) - n / 2.0
    return soft + mu * viol * viol

def smooth_optimize(h_start, n, T, mu, max_iter=300):
    dx = 2.0 / n
    def f(h):
        return smooth_obj(h, n, T, mu)
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-18, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def subgradient_optimize(h_start, n, lr_init, n_iters=200, decay=0.999):
    dx = 2.0 / n
    h = h_start.copy()
    best = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        k = k_star - (n - 1)
        grad = np.zeros(n)
        if k >= 0:
            hi = n - k
            grad[:hi] += -(h[k:])
            grad[k:] += (1.0 - h[:hi])
        else:
            kk = -k
            hi = n - kk
            grad[:hi] += (1.0 - h[kk:])
            grad[kk:] += -(h[:hi])
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < best:
            best = c
            best_h = h_new.copy()
        h = h_new
        lr *= decay
    return best_h, best

def binary_local_search(h_bin, n, rng, max_rounds=200, n_tries=3000):
    h = h_bin.copy()
    current = compute_c5(h)
    best = current
    best_h = h.copy()
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    for rnd in range(max_rounds):
        improved = False
        nt = min(n_tries, max(1, len(ones) * len(zeros)))
        for _ in range(nt):
            ii = rng.randint(len(ones))
            jj = rng.randint(len(zeros))
            oi, oj = ones[ii], zeros[jj]
            h[oi] = 0.0
            h[oj] = 1.0
            c = compute_c5(h)
            if c < current:
                current = c
                ones = np.delete(ones, ii)
                zeros = np.delete(zeros, jj)
                ones = np.append(ones, oj)

                improved = True
                if c < best:
                    best = c
                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved:

    return best_h, best

                    best_h = h.copy()
            else:
                h[oi] = 1.0
                h[oj] = 0.0
        if not improved and T < 0.005:
            break
    return best_h, best

def multi_lag_target(h, n, n_rounds=30):
    """Target top few lags simultaneously using smooth optimization."""
    h = h.copy()
    best = compute_c5(h)
    for _ in range(n_rounds):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = np.argsort(corr)[-5:]
        # Build a smooth objective that targets top lags
        T = 500.0
        mu = 100.0
        def f_target(hh):
            cc = np.correlate(hh, 1.0 - hh, mode='full')
            mc = np.max(cc)
            shifted = np.clip(T * (cc - mc), -80, 80)
            soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * (2.0 / n)
            viol = np.sum(hh) - n / 2.0
            return soft + mu * viol * viol
        bounds = [(0.0, 1.0)] * n
        res = minimize(f_target, h, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': 80, 'ftol': 1e-16})
        h_new = make_feasible(res.x, n)
        c = compute_c5(h_new)
        if c < best:
            best = c
            h = h_new
        else:
            break
    return h, best

def coord_descent(h, n, n_rounds=4, deltas=None):
    if deltas is None:
        deltas = [0.04, -0.04, 0.02, -0.02, 0.01, -0.01]
    h = h.copy()
    best = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                h[j] = nv
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < best - 1e-14:
                    best = c
                    ov = nv
                    improved = True
                    break
                else:
                    h[j] = ov
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-8:
                    continue
                h[j] = bv
                h = make_feasible(h, n)
                c = compute_c5(h)
                if c < best - 1e-14:
                    best = c
                    improved = True
            if not improved and rnd > 0:
                pass
    return h, best

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Upsample to n=2048, smooth optimize
    n1 = 2048
    h_w = upsample_h(best_h, n1)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()

    temps = [20, 50, 120, 300, 800, 2000]
    for T in temps:
        if tl() < 25:
            break
        try:
            h_opt = smooth_optimize(h_w, n1, T, mu=500.0, max_iter=250)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h_w = h_opt.copy()
        except Exception:
            pass

    # Random restarts at n1
    for rs in range(3):
        if tl() < 30:
            break
        h_r = h_w + 0.15 * rng.randn(n1)
        h_r = make_feasible(h_r, n1)
        for T in [100, 500]:
            if tl() < 15:
                break
            try:
                h_opt = smooth_optimize(h_r, n1, T, mu=500.0, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_r = h_opt.copy()
            except Exception:
                pass

    # Phase 2: Subgradient refinement
    if tl() > 40:
        for lr in [0.5, 0.2, 0.1, 0.05]:
            if tl() < 15:
                break
            try:
                h_sg, c5_sg = subgradient_optimize(h_w, n1, lr, n_iters=150)
                if c5_sg < best_c5:
                    best_c5, best_h = c5_sg, h_sg.copy()
                    h_w = h_sg.copy()
            except Exception:
                pass

    # Phase 3: Binary local search at n1
    if tl() > 40:
        h_bin = (h_w > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_target = n1 // 2
        if n_ones > n_target:
            oi = np.where(h_bin == 1)[0]
            rng.shuffle(oi)
            for k in range(n_ones - n_target):
                h_bin[oi[k]] = 0.0
        elif n_ones < n_target:
            zi = np.where(h_bin == 0)[0]
            rng.shuffle(zi)
            for k in range(n_target - n_ones):
                h_bin[zi[k]] = 1.0
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

        h_bs, c5_bs = binary_local_search(h_bin, n1, rng, max_rounds=200, n_tries=4000)
        if c5_bs < best_c5:
            best_c5, best_h = c5_bs, h_bs.copy()

        # Annealed search from perturbed binary
        if tl() > 40:
            h_pert = h_bs.copy()
            ones_p = np.where(h_pert > 0.5)[0]
            zeros_p = np.where(h_pert <= 0.5)[0]
            n_flips = 80
            for _ in range(n_flips):
                oi = rng.randint(len(ones_p))
                oz = rng.randint(len(zeros_p))
                h_pert[ones_p[oi]], h_pert[zeros_p[oz]] = 0.0, 1.0
                ones_p, zeros_p = np.where(h_pert > 0.5)[0], np.where(h_pert <= 0.5)[0]
            h_ana, c5_ana = binary_anneal(h_pert, n1, rng, max_rounds=150, T0=0.03, T1=0.003)
            if c5_ana < best_c5:
                best_c5, best_h = c5_ana, h_ana.copy()

    # Phase 4: Multi-lag targeting
    if tl() > 30:
        try:
            h_mt, c5_mt = multi_lag_target(h_w, n1, n_rounds=20)
            if c5_mt < best_c5:
                best_c5, best_h = c5_mt, h_mt.copy()
        except Exception:
            pass

    # Phase 5: Higher resolution n=4096
    if tl() > 100:
        n2 = 4096
        h_hi = upsample_h(best_h, n2)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5, best_h = c5_hi, h_hi.copy()

        for T in [50, 200, 800]:
            if tl() < 25:
                break
            try:
                h_opt = smooth_optimize(h_hi, n2, T, mu=300.0, max_iter=120)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_hi = h_opt.copy()
            except Exception:
                pass

        # Binary at high res
        if tl() > 50:
            h_bin_hi = (h_hi > 0.5).astype(float)
            n_ones_hi = int(np.sum(h_bin_hi))
            n_target_hi = n2 // 2
            if n_ones_hi > n_target_hi:
                oi = np.where(h_bin_hi == 1)[0]
                rng.shuffle(oi)
                for k in range(n_ones_hi - n_target_hi):
                    h_bin_hi[oi[k]] = 0.0
            elif n_ones_hi < n_target_hi:
                zi = np.where(h_bin_hi == 0)[0]
                rng.shuffle(zi)
                for k in range(n_target_hi - n_ones_hi):
                    h_bin_hi[zi[k]] = 1.0
            h_bshi, c5_bshi = binary_local_search(h_bin_hi, n2, rng, max_rounds=100, n_tries=3000)
            if c5_bshi < best_c5:
                best_c5, best_h = c5_bshi, h_bshi.copy()

    # Phase 6: Even higher res n=8192 if time permits
    if tl() > 120:
        n3 = 8192
        h_xx = upsample_h(best_h, n3)
        c5_xx = compute_c5(h_xx)
        if c5_xx < best_c5:
            best_c5, best_h = c5_xx, h_xx.copy()
        if tl() > 40:
            try:
                h_opt = smooth_optimize(h_xx, n3, 200, mu=200.0, max_iter=80)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
            except Exception:
                pass

    # Phase 7: Final coordinate descent polish
    if tl() > 20:
        n_cd = min(len(best_h), 2048)
        h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
        h_cd, c5_cd = coord_descent(h_cd, n_cd, n_rounds=3,
                                     deltas=[0.03, -0.03, 0.015, -0.015, 0.008, -0.008])
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Final: one more smooth+subgradient squeeze
    if tl() > 15:
        n_f = min(len(best_h), 2048)
        h_f = upsample_h(best_h, n_f) if n_f < len(best_h) else best_h.copy()
        try:
            h_opt = smooth_optimize(h_f, n_f, 500, mu=1000.0, max_iter=100)
            c5_o = compute_c5(h_opt)
            if c5_o < best_c5:
                best_c5, best_h = c5_o, h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
