_OPT_N = 960
_OPT_T = 300.0

def _obj(x):
    n = _OPT_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    h = h.copy()
    binary = (h >= 0.5).astype(float)
    n_ones = int(np.sum(binary))
    target = n // 2
    if n_ones > target:
        ones = np.where(binary == 1.0)[0]
        sorted_idx = ones[np.argsort(h[ones])]
        for i in range(n_ones - target):
            binary[sorted_idx[i]] = 0.0
    elif n_ones < target:
        zeros = np.where(binary == 0.0)[0]
        sorted_idx = zeros[np.argsort(-h[zeros])]
        for i in range(target - n_ones):
            binary[sorted_idx[i]] = 1.0
    return binary

def greedy_binary_snap(h, n, max_passes=5):
    best = h.copy()
    best_c = compute_c5(best)
    for _pass in range(max_passes):
        improved = False
        for j in range(n):
            for val in [0.0, 1.0]:
                trial = best.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best = trial
                    best_c = c
                    improved = True
        if not improved:
            break
    return best, best_c

def swap_sa(h_bin, n, start_time, budget_s, margin, seed, iters_per_sec, T0=0.004, Tend=1e-9):

    current = h_bin.copy()

    best = current.copy()

    zeros_arr = np.where(current == 0.0)[0].astype(np.int32).copy()
    n_ones = len(ones_arr)
    n_zeros = len(zeros_arr)
    
    elapsed = time.time() - start_time
    total_avail = max(budget_s - margin - elapsed, 10)
    total_iters = int(total_avail * iters_per_sec)
    total_iters = max(total_iters, 10000)
    
    for it in range(total_iters):
        if time.time() - start_time > budget_s - margin:
            break
        
        frac = (it + 1) / total_iters
        T = T0 * (Tend / T0) ** frac
        
        ii = rng.randint(n_ones)
        jj = rng.randint(n_zeros)
        i = ones_arr[ii]
        j = zeros_arr[jj]
        
        current[i] = 0.0
        current[j] = 1.0
        new_c = compute_c5(current)
        
        delta = new_c - current_c
        if delta <= 0.0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            current_c = new_c
            if new_c < best_c - 1e-14:
                best = current.copy()
                best_c = new_c
            ones_arr[ii] = j
            zeros_arr[jj] = i
        else:
            current[i] = 1.0
            current[j] = 0.0
    
    return best, best_c

def fine_cd(h, n, start_time, budget_s, margin):
    best = h.copy()
    best_c = compute_c5(best)
    delta_vals = [0.5, 0.3, 0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01, 0.005, 0.002]
    
    while time.time() - start_time < budget_s - margin:
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - margin:
                return best, best_c
            old = best[j]
            best_val = old
            best_c_local = best_c
            for d in delta_vals:
                for sd, cand in [(-1, old - d), (1, old + d)]:
                    cand = max(0.0, min(1.0, cand))
                    if abs(cand - old) < 1e-9:
                        continue
                    trial = best.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c_local - 1e-14:
                        best_c_local = c
                        best_val = cand
            if best_val != old:
                best[j] = best_val
                best = make_feasible(best, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return best, best_c
