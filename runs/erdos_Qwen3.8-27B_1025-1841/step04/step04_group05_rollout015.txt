def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

        else:

def _make_feasible(h, n):

    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:

        cnt = np.sum(mask)

import numpy as np
from scipy.optimize import minimize
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        d = target - h.sum()
        if abs(d) < 1e-15:
            break
        mask = h < 1.0 - 1e-12 if d > 0 else h > 1e-12
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + d / cnt, 0.0, 1.0)
    d = target - h.sum()
    if abs(d) > 1e-12:
        i = np.argmin(h) if d > 0 else np.argmax(h)
        h[i] = np.clip(h[i] + d, 0.0, 1.0)
    return h

def _up(h, n2):
    n1 = len(h)
    if n2 == n1:
        return _feasible(h.copy(), n2)
    x1 = np.linspace(0, 2, n1, endpoint=False)
    x2 = np.linspace(0, 2, n2, endpoint=False)
    return _feasible(np.interp(x2, x1, h, period=2.0), n2)

def _lag_g(h, m, n):
    """d/dh of raw correlation at lag m (no dx factor)."""
    g = np.zeros(n)
    if m == 0:
        g[:] = 1.0 - 2.0 * h
    elif 0 < m < n:
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    elif -n < m < 0:
        p = -m
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g

def _smog(h, n, T, mu):
    """Smoothed objective + penalty gradient."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cm = corr.max()
    e = np.exp(T * (corr - cm))
    S = e.sum()
    obj = (cm + np.log(S) / T) * dx
    s = h.sum()
    obj += mu * (s - n / 2.0) ** 2
    w = e / S
    g = np.zeros(n)
    act = np.where(w > 1e-10)[0]
    for k in act:
        g += w[k] * _lag_g(h, int(k - (n - 1)), n)
    g *= dx
    g += 2.0 * mu * (s - n / 2.0)
    return obj, g

def _sls(h, n, niters, a0, ktop):
    """Subgradient polish."""
    h = h.copy()
    best = _c5(h)
    a = a0
    for _ in range(niters):
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full')
        topk = np.argsort(corr)[::-1][:ktop]
        g = np.zeros(n)
        for ki in topk:
            g += _lag_g(h, int(ki - (n - 1)), n)
        g = g / ktop * dx
        h2 = _feasible(h - a * g, n)
        c2 = _c5(h2)
        if c2 < best:
            h, best = h2, c2
        else:
            a *= 0.95
        if a < 1e-10:
            break
    return h, best

    t0 = time.time()
    rng = np.random.RandomState(seed)

    # Use a global to pass n, T to the objective
    global _GC_N, _GC_T, _GC_MU
    _GC_N, _GC_T, _GC_MU = n, 50.0, 100.0
    
    def _obj_u(x):
        o, g = _smog(x, _GC_N, _GC_T, _GC_MU)
        return o, g

                   options={'maxiter': 300, 'ftol': 1e-15})
