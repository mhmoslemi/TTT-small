def subgradient_step(h, n, step_size, alpha):
    """One subgradient step for the max correlation problem."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    max_idx = np.where(corr == max_corr)[0]
    
    # Compute subgradient (average of gradients at active lags)
    grad = np.zeros(n)
    for lag_idx in max_idx:
        k = lag_idx - (n - 1)  # actual lag
        # For this lag, g_k = sum_i h[i] * (1-h[i+k])
        # Gradient w.r.t. h[m]:
        # For each valid pair (i, i+k):
        #   d/dh[i] = 1 - h[i+k]
        #   d/dh[i+k] = -h[i]
        if k >= 0:

        else:
            valid_i = range(-k, n)

                grad[j] += (-h[i])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-16
        else:
            mask = h > 1e-16
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def smooth_obj(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

        return smooth_obj(h)

        return np.sum(h) - n / 2.0

        return np.ones(n)

                   bounds=bounds,

                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

    best_c5 = compute_c5(h)
    dx = 2.0 / n

    for it in range(n_iter):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        max_corr = np.max(corr)

        # Compute subgradient
        grad = np.zeros(n)
        for lag_idx in top_indices:

            if k >= 0:

                else:
                    grad[:n-k] += (1.0 - h_curr[k:])
                    grad[k:] += (-h_curr[:n-k])

    best_c5 = compute_c5(h)

    v = np.zeros(n)  # momentum velocity

        t = it + 1

OK let me also think about whether there's a smarter optimization approach.

**Key insight**: The problem is non-convex (bilinear), but it has a special structure. The function g_k(h) = ∑_i h[i]*(1-h[i+k]) is supermodular (or submodular) in h, which means the maximum might have nice properties.

Actually, g_k(h) = ∑_i h[i] - ∑_i h[i]*h[i+k]. The first term is linear, the second is a "quadratic" term. Since h[i]*h[i+k] ≥ 0, maximizing ∑_i h[i]*(1-h[i+k]) is equivalent to minimizing ∑_i h[i]*h[i+k] (the autocorrelation at lag k).

So the problem is: minimize max_k (1 - R_k(h) * dx) where R_k(h) = ∑_i h[i]*h[i+k] * dx is the autocorrelation.

Equivalently: maximize min_k R_k(h).

This is a "max-min" problem. The optimal solution equalizes the autocorrelations at the worst lags.

This gives me an idea: **water-filling approach**.
1. Start with a feasible h
2. Compute all autocorrelations R_k
3. Find the minimum R_k (worst lag)
4. Adjust h to increase the minimum R_k while not decreasing others
5. Repeat

The adjustment in step 4: to increase R_k = ∑_i h[i]*h[i+k], we want to increase h[i] and h[i+k] simultaneously (where they're not already at 1). This is like a "positive feedback" for pairs that should be high.

But this is essentially what the subgradient method does (with the sign flipped). So I think the subgradient method is the right approach.

import numpy as np
from scipy.optimize import minimize
import time

_global_n = 0
_global_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _smooth_obj(h):
    n = _global_n
    T = _global_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_eq(h):
    return np.sum(h) - _global_n / 2.0

def _cons_jac(h):
    return np.ones(_global_n)

def slsqp_optimize(h0, n, T, max_iter):
    global _global_n, _global_T
    _global_n = n
    _global_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    res = minimize(_smooth_obj, h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def subgrad_optimize(h0, n, n_iter, top_k=3, use_momentum=True):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    dx = 2.0 / n
    h_curr = h0.copy()
    v = np.zeros(n) if use_momentum else None
    
    for it in range(n_iter):
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        max_corr = np.max(corr)
        
        top_indices = np.argsort(corr)[-top_k:]
        
        grad = np.zeros(n)
        for lag_idx in top_indices:
            k = int(lag_idx) - (n - 1)
            if k >= 0:
                if k == 0:
                    grad += 1.0 - 2.0 * h_curr
                else:
                    grad[:n-k] += (1.0 - h_curr[k:])
                    grad[k:] += (-h_curr[:n-k])
            else:
                kp = -k
                grad[kp:] += (1.0 - h_curr[:n-kp])
                grad[:n-kp] += (-h_curr[kp:])
        
        grad /= top_k
        grad *= dx
        
        t = float(it + 1)
        lr = 0.08 / (t ** 0.5) if use_momentum else 0.01 / (1 + 0.0005 * it)
        
        if use_momentum:
            beta = (t - 1.0) / (t + 2.0) if it > 0 else 0.0
            v = beta * v - lr * grad
            h_new = h_curr + v
        else:
            h_new = h_curr - lr * grad
        
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h_curr = h_new
    
    return best_h, best_c5

def local_search(h0, n, n_iters, rng, max_pert=4):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    step_size = 0.06
    
    for it in range(n_iters):
        n_pert = rng.randint(1, max_pert + 1)
        idx = rng.choice(n, size=n_pert, replace=False)
        trial = best_h.copy()
        for j in idx:
            delta = rng.uniform(-step_size, step_size)
            trial[j] = np.clip(trial[j] + delta, 0.0, 1.0)
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c5:
            best_c5 = c
            best_h = trial.copy()
            if it % 200 == 0:
                step_size = min(0.15, step_size * 1.05)
        elif it % 1000 == 0:
            step_size = max(0.01, step_size * 0.85)
    
    return best_h, best_c5

def coord_descent(h0, n, n_rounds=3):
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            old_val = best_h[j]
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                candidate = old_val + delta
                if candidate < 0.0 or candidate > 1.0:
                    continue
                trial = best_h.copy()
                trial[j] = candidate
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial
                    old_val = candidate
                    improved = True
            for bv in (0.0, 1.0):
                if abs(bv - best_h[j]) < 1e-9:
                    continue
                trial = best_h.copy()
                trial[j] = bv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial
                    improved = True
        if not improved:
            break
    
    return best_h, best_c5
