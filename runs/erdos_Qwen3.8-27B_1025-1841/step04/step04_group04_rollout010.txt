def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h[0] += diff  # crude fix
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

        h[to_flip] = 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        free = np.where((h > 1e-12) & (h < 1.0 - 1e-12))[0]
        if len(free) > 0:
            h[free[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def objective_and_grad(x, n, T, penalty):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = np.max(corr)
    shifted = corr - M
    max_sh = np.max(shifted)
    exps = np.exp(T * (shifted - max_sh))
    sum_exps = np.sum(exps)
    w = exps / sum_exps
    LSE = M + max_sh + np.log(sum_exps) / T
    F = LSE * dx
    sv = np.sum(x) - n / 2.0
    F += penalty * sv * sv
    W_rev = w[::-1]
    conv_A = np.convolve(x, W_rev, mode='full')
    conv_B = np.convolve(x, w, mode='full')
    A = conv_A[n - 1:2 * n - 1]
    B = conv_B[n - 1:2 * n - 1]
    grad = dx * (1.0 - A - B) + 2.0 * penalty * sv
    return F, grad

def lbfgsb_opt(h, n, T, penalty, maxiter, ftol):
    def fun(x):
        return objective_and_grad(x, n, T, penalty)[0]
    def jac(x):
        return objective_and_grad(x, n, T, penalty)[1]
    def funjac(x):
        return objective_and_grad(x, n, T, penalty)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(funjac, h, method='L-BFGS-B', bounds=bounds,
                       jac=True, options={'maxiter': maxiter, 'ftol': ftol,
                                          'maxfun': maxiter * 10})
        return res.x
    except Exception:
        return h

def binary_sa(h, n, rng, n_iter, T_start, T_end):
    h = h.copy().astype(float)
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n_ones = int(np.sum(h))
    target_ones = n // 2
    if n_ones > target_ones:
        ones = np.where(h == 1.0)[0]
        to_flip = rng.choice(ones, n_ones - target_ones, replace=False)
        h[to_flip] = 0.0
    elif n_ones < target_ones:
        zeros = np.where(h == 0.0)[0]
        to_flip = rng.choice(zeros, target_ones - n_ones, replace=False)
        h[to_flip] = 1.0
    best_h = h.copy()
    best_c = compute_c5(h)
    cur_c = best_c
    ratio = T_end / T_start
    for it in range(n_iter):
        T_sa = T_start * (ratio ** (it / max(n_iter - 1, 1)))
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - cur_c
        if delta < 0 or (T_sa > 1e-15 and rng.random() < np.exp(-delta / T_sa)):
            cur_c = new_c
            if cur_c < best_c:
                best_c = cur_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    return best_h, best_c

def coord_descent(h, n, n_rounds, remaining_fn):
    for _ in range(n_rounds):
        if remaining_fn() < 5:
            break
        improved = False
        for j in range(n):
            if remaining_fn() < 3:
                break
            old = h[j]
            best_c = compute_c5(h)
            best_val = old
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old - 0.1), max(0.0, old - 0.05), max(0.0, old - 0.01),
                     min(1.0, old + 0.01), min(1.0, old + 0.05), min(1.0, old + 0.1)]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                improved = True
        if not improved:
            break
    return h, compute_c5(h)
