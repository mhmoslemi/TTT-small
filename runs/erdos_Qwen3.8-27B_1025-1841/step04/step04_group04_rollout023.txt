    corr = np.correlate(h, 1.0-h, mode='full')
    R = corr * dx  # length 2n-1

    # g_j(m) = (1-h[j+m]) if j+m < n else 0, minus h[j-m] if j-m >= 0 else 0
    # Note: m ranges from -(n-1) to n-1, indexed as m + (n-1) in the correlate output

    for m_idx in range(2*n-1):
        m = m_idx - (n-1)
        gj = 0.0
        if j + m >= 0 and j + m < n:
            gj += (1.0 - h[j+m])
        if j - m >= 0 and j - m < n:
            gj -= h[j-m]
        gjp = 0.0
        if jp + m >= 0 and jp + m < n:
            gjp += (1.0 - h[jp+m])
        if jp - m >= 0 and jp - m < n:
            gjp -= h[jp-m]
        d[m_idx] = gj - gjp
    
    # Binary search on c
    # δ range
    delta_lo = max(-h[j], h[jp]-1.0)
    delta_hi = min(1.0-h[j], h[jp])
    
    if delta_lo >= delta_hi:
        return 0.0, R.max()
    
    # The minimum possible max is at least R.max() (current)
    # and at most R.max() + max(|d|) * (delta_hi - delta_lo)
    
    lo_c = R.max() - abs(d).max() * (delta_hi - delta_lo)
    hi_c = R.max() + abs(d).max() * (delta_hi - delta_lo)
    
    for _ in range(50):
        mid_c = (lo_c + hi_c) / 2
        # Check feasibility: exists δ ∈ [delta_lo, delta_hi] s.t. R + δ*d ≤ mid_c for all components
        # For each m: if d[m] > 0: δ ≤ (mid_c - R[m]) / d[m]
        #              if d[m] < 0: δ ≥ (mid_c - R[m]) / d[m]
        #              if d[m] = 0: need R[m] ≤ mid_c
        feasible_lo = delta_lo
        feasible_hi = delta_hi
        for m_idx in range(2*n-1):
            if d[m_idx] > 1e-15:
                feasible_hi = min(feasible_hi, (mid_c - R[m_idx]) / d[m_idx])
            elif d[m_idx] < -1e-15:
                feasible_lo = max(feasible_lo, (mid_c - R[m_idx]) / d[m_idx])
            else:
                if R[m_idx] > mid_c:
                    break
        if feasible_lo <= feasible_hi:
            hi_c = mid_c
            # Found a feasible δ
            best_delta = (feasible_lo + feasible_hi) / 2
        else:
            lo_c = mid_c
            best_delta = 0.0
    
    return best_delta, hi_c

        fe_hi = delta_hi

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_d_vec(h, j, jp, n, m_vals):
    """Compute d(m) = g_j(m) - g_jp(m) where g_j(m) is the sensitivity of R(m) to h[j]."""
    idx1 = j + m_vals
    idx2 = j - m_vals
    gj = np.zeros(2 * n - 1)
    m1 = (idx1 >= 0) & (idx1 < n)
    m2 = (idx2 >= 0) & (idx2 < n)
    if m1.any():
        gj[m1] += (1.0 - h[idx1[m1]])
    if m2.any():
        gj[m2] -= h[idx2[m2]]
    
    idx3 = jp + m_vals
    idx4 = jp - m_vals
    gjp = np.zeros(2 * n - 1)
    m3 = (idx3 >= 0) & (idx3 < n)
    m4 = (idx4 >= 0) & (idx4 < n)
    if m3.any():
        gjp[m3] += (1.0 - h[idx3[m3]])
    if m4.any():
        gjp[m4] -= h[idx4[m4]]
    
    return gj - gjp

def solve_pair(h, j, jp, n, R, dx):
    """Find optimal delta for paired swap (j, jp) given current R."""
    m_vals_local = np.arange(-(n-1), n)
    d = compute_d_vec(h, j, jp, n, m_vals_local)
    
    dmax = np.abs(d).max()
    if dmax < 1e-14:
        return 0.0
    
    d_lo = max(-h[j], h[jp] - 1.0)
    d_hi = min(1.0 - h[j], h[jp])
    if d_lo >= d_hi - 1e-14:
        return 0.0
    
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    zero_mask = ~(pos_mask | neg_mask)

        if neg_mask.any():

        else:
