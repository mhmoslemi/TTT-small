def compute_corr_fft(h):

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = int(np.argmin(h)) if diff > 0 else int(np.argmax(h))
        h[idx] += diff
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr_fast(h):
    """FFT-based cross-correlation of h with (1-h)."""
    n = len(h)
    g = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    Gr = np.fft.rfft(g[::-1], N)
    corr = np.fft.irfft(H * Gr, N)[:2 * n - 1]
    return corr

def subgradient_optimize(h, n, budget_s, start_time, top_m=8):

    dx = 2.0 / n

    momentum = 0.85
    velocity = np.zeros(n)
    no_improve = 0

        it += 1
        corr = compute_corr_fast(h)
        c5_cur = float(np.max(corr)) * dx

        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                step *= 0.65
                if step < 1e-8:
                    break

                if s > 0:

def slsqp_optimize(h, n, budget_s, start_time, T=2000.0, top_m=5):
    """SLSQP with softmax-smoothed max."""
    dx = 2.0 / n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        if top_m is not None and top_m < len(corr):
            top = np.sort(corr)[-top_m:]
            mc = top[-1]
            shifted = top - mc
            return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': 400, 'ftol': 1e-17})
        h_res = make_feasible(res.x, n)
        c5_res = compute_c5(h_res)
        if c5_res < compute_c5(h) - 1e-15:
            return h_res, c5_res
    except Exception:
        pass
    return h, compute_c5(h)

def coord_refine(h, n, budget_s, start_time, steps=None):
    """Coordinate descent with feasibility repair."""
    if steps is None:
        steps = [0.005, 0.002, 0.001]
    best_h = h.copy()
