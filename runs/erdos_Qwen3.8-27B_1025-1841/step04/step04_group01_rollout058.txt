```python
    # The first-index term becomes t*(1-h[j+m-n+1])
    # So corr[m](t) = corr0[m] + t*(1-h[j+m-n+1])*I1 - t*h[j-m+n-1]*I2
    # = corr0[m] + t * [(1-h[j+m-n+1])*I1 - h[j-m+n-1]*I2]
    
    # A = corr0 (computed above)
    # B[m] = (1-h[j+m-n+1])*I(0<=j+m-n+1<n) - h[j-m+n-1]*I(0<=j-m+n-1<n)
    
    m_arr = np.arange(n_full)
    idx1 = j + m_arr - n + 1  # second index when j is first
    idx2 = j - m_arr + n - 1  # first index when j is second
    
    I1 = (idx1 >= 0) & (idx1 < n)
    I2 = (idx2 >= 0) & (idx2 < n)
    
    B = np.zeros(n_full)
    B[I1] = (1.0 - h[idx1[I1]])
    B[I2] = B[I2] - h[idx2[I2]]
    
    h[j] = h_save
    
    # Now minimize max_m (A[m] + B[m]*t) over t in [0,1]
    # This is convex PL. Use the "walk" algorithm.
    # Start at t=0, find active m, walk to next breakpoint.
    
    # For efficiency, only consider top shifts (those with A[m] or A[m]+B[m] near max)
    # But for correctness, we need all. Let's use all 2n-1 shifts.
    
    t = 0.0
    for _ in range(50):  # max iterations
        val = A + B * t
        m_star = np.argmax(val)
        B_star = B[m_star]
        if B_star >= 0:
            break  # minimum at current t (slope >= 0)
        # Find next breakpoint: min over m where B[m] > B_star of (A[m_star]-A[m])/(B[m]-B_star)
        candidates = B > B_star
        if not np.any(candidates):
            break  # no crossing, go to t=1
        diffs = (A[m_star] - A[candidates]) / (B[candidates] - B_star)
        diffs = diffs[diffs > 1e-12]
        if len(diffs) == 0:
            break
        t_next = np.min(diffs)
        if t_next >= 1.0 - 1e-12:
            t = 1.0
            break
        t = max(t, t_next) + 1e-10  # advance slightly past breakpoint
        if t > 1.0:
            t = 1.0
            break
    
    # Evaluate at t, 0, 1
    best_t = 0.0
    best_val = np.max(A + B * 0.0)
    for t_cand in [t, 0.0, 1.0]:
        t_cand = max(0.0, min(1.0, t_cand))
        v = np.max(A + B * t_cand)
        if v < best_val:
            best_val = v
            best_t = t_cand
    
    return best_t, best_val * dx
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = int(np.argmin(h))
        else:
            idx = int(np.argmax(h))
        h[idx] += diff
    return h

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _soft_max_obj(x, T, top_idx, dx, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    if top_idx is not None:
        c = corr[top_idx]
    else:
        c = corr
    mc = np.max(c)
    shifted = c - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _coord_1d_sweep(j, h, n, best_c5):

    h[j] = 0.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    h[j] = old_val
    m_arr = np.arange(2 * n - 1)
    idx1 = j + m_arr - n + 1
    idx2 = j - m_arr + n + 1 - 2  # j - m + n - 1
    idx2 = j - m_arr + n - 1
    I1 = (idx1 >= 0) & (idx1 < n)
    I2 = (idx2 >= 0) & (idx2 < n)
    B = np.zeros(2 * n - 1)
    B[I1] = (1.0 - h[idx1[I1]])
    B[I2] = B[I2] - h[idx2[I2]]
    A = corr0
    # Find optimal t via convex PL minimization
    candidates_t = [0.0, 1.0]
    # Walk algorithm
    t = 0.0
    for _ in range(30):
        val = A + B * t
        m_star = int(np.argmax(val))
        B_star = B[m_star]
        if B_star >= -1e-12:

            break
        cands = B > B_star + 1e-12
        if not np.any(cands):

        diffs = (A[m_star] - A[cands]) / (B[cands] - B_star)
        diffs = diffs[diffs > 1e-12]
        if len(diffs) == 0:
            candidates_t.append(1.0)
            break
        t_next = float(np.min(diffs))
        if t_next >= 1.0 - 1e-10:
            candidates_t.append(1.0)
            break
        t = min(1.0, t_next + 1e-8)
        candidates_t.append(t)
        if t >= 1.0 - 1e-10:
            break
    # Evaluate candidates

        t_c = max(0.0, min(1.0, t_c))
        if abs(t_c - old_val) < 1e-10:
            continue
        trial = h.copy()
        trial[j] = t_c
        trial = _make_feasible(trial, n)
        c = _compute_c5(trial)
        if c < best_c - 1e-14:
```
