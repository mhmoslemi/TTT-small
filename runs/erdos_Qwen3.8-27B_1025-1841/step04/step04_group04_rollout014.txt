import time
from scipy.optimize import minimize

_G_N = 960
_G_T = 50.0

def _obj(x):
    dx = 2.0 / _G_N
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_G_T * shifted))) / _G_T) * dx

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_snap(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    changed = True
    while changed:
        changed = False
        for j in range(n):
            for val in (0.0, 1.0):
                if abs(best[j] - val) < 1e-12:
                    continue
                trial = best.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best = trial
                    best_c = c
                    changed = True
                    break
    return best, best_c

def swap_search(h, n, max_iters, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c = compute_c5(h)
    for it in range(max_iters):
        order = np.argsort(h)
        i_hi = order[rng.randint(n // 3, n)]
        i_lo = order[rng.randint(0, n // 3)]
        if abs(h[i_hi] - h[i_lo]) < 1e-10:
            continue
        old_hi = h[i_hi]
        old_lo = h[i_lo]
        h[i_hi] = old_lo
        h[i_lo] = old_hi
        h = make_feasible(h, n)
        c = compute_c5(h)
        if c < best_c - 1e-15:
            best_c = c
        else:
            h[i_hi] = old_hi
            h[i_lo] = old_lo
            h = make_feasible(h, n)
    return h, best_c

def coord_descent(h, n, step, rounds, tol):
    best_c = compute_c5(h)
    for r in range(rounds):
        improved = False
        for j in range(n):
            old = h[j]
            best_v = old
            best_cj = best_c
            cands = [0.0, 1.0, max(0.0, old - step), max(0.0, old - step * 0.5),
                     max(0.0, old - step * 2.0), min(1.0, old + step * 0.5),
                     min(1.0, old + step), min(1.0, old + step * 2.0)]
            for d in cands:
                if abs(d - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = d
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_cj - tol:
                    best_cj = c
                    best_v = d
            if best_v != old:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c = best_cj
                improved = True
        if not improved:
            break
    return h, best_c
