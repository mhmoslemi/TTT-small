        if len(zeros) == 0:
            break
        idx = zeros[rng.randint(len(zeros))]
        b[idx] = 1
        s += 1

    top_idx = np.argsort(corr)[-top_k:]
    dx = 2.0 / n
    grad = np.zeros(n)

            alpha *= 0.6
            stalled += 1
            if stalled > 30:
                top_k = min(top_k + 3, 20)
                alpha = min(alpha * 2.0, 0.1)
                stalled = 0
            if stalled > 200:
                break
    return best_h, best_c5

def binary_sa(b, n, rng, start_time, time_limit, T0=0.005, T1=1e-6):
    curr = compute_c5(b)
    best = curr
    best_b = b.copy()
    it = 0
    evals = 0
    while time.time() - start_time < time_limit:
        it += 1
        frac = min(1.0, it / 100000.0)
        T = T0 * (T1 / T0) ** frac
        r = rng.random()
        if r < 0.35:
            ones = np.flatnonzero(b)
            zeros = np.flatnonzero(1.0 - b)
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            b[i], b[j] = 0, 1
            nc = compute_c5(b)
            evals += 1
            if nc > curr and (T < 1e-14 or rng.random() >= np.exp(-(nc - curr) / T)):
                b[i], b[j] = 1, 0
            else:
                curr = nc
                if nc < best:
                    best = nc
                    best_b = b.copy()
        elif r < 0.6:
            ones = np.flatnonzero(b)
            zeros = np.flatnonzero(1.0 - b)
            if len(ones) < 2 or len(zeros) < 2:
                continue
            idx_o = rng.choice(len(ones), 2, replace=False)
            idx_z = rng.choice(len(zeros), 2, replace=False)
            p = [ones[idx_o[0]], ones[idx_o[1]], zeros[idx_z[0]], zeros[idx_z[1]]]
            b[p[0]], b[p[1]] = 0, 0
            b[p[2]], b[p[3]] = 1, 1
            nc = compute_c5(b)
            evals += 1
            if nc > curr and (T < 1e-14 or rng.random() >= np.exp(-(nc - curr) / T)):
                b[p[0]], b[p[1]] = 1, 1
                b[p[2]], b[p[3]] = 0, 0
            else:
                curr = nc
                if nc < best:
                    best = nc
                    best_b = b.copy()
        elif r < 0.85:
            L = rng.randint(4, max(5, n // 12))
            s = rng.randint(0, n - L)
            b[s:s+L] = b[s:s+L][::-1]
            nc = compute_c5(b)
            evals += 1
            if nc > curr and (T < 1e-14 or rng.random() >= np.exp(-(nc - curr) / T)):
                b[s:s+L] = b[s:s+L][::-1]
            else:
                curr = nc
                if nc < best:
                    best = nc
                    best_b = b.copy()
        else:
            L = rng.randint(3, max(4, n // 15))
            s = rng.randint(0, n - L)
            amp = rng.randint(1, L)
            seg = b[s:s+L].copy()
            b[s:s+L] = np.concatenate([seg[amp:], seg[:amp]])
            nc = compute_c5(b)
            evals += 1
            if nc > curr and (T < 1e-14 or rng.random() >= np.exp(-(nc - curr) / T)):
                b[s:s+L] = seg
            else:
                curr = nc
                if nc < best:
                    best = nc
                    best_b = b.copy()
    return best_b, best

def hill_climb_binary(b, n, rng, start_time, time_limit):
    curr = compute_c5(b)
    while time.time() - start_time < time_limit:
        ones = np.flatnonzero(b)
        zeros = np.flatnonzero(1.0 - b)
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        b[i], b[j] = 0, 1
        nc = compute_c5(b)
        if nc < curr:
            curr = nc
        else:
            b[i], b[j] = 1, 0
    return b, curr

def two_level_search(b, n, rng, start_time, time_limit):
    """Alternating hill climb and SA with restarts."""
    best_b = b.copy()
    best_c5 = compute_c5(b)
    remaining = time_limit - (time.time() - start_time)
    while remaining > 10:
        hc_time = min(20, remaining * 0.3)
        b, c = hill_climb_binary(b.copy(), n, rng, start_time, start_time + hc_time)
        if c < best_c5:
            best_c5 = c
            best_b = b.copy()
        remaining = time_limit - (time.time() - start_time)
        if remaining < 30:
            break
        sa_time = min(remaining * 0.5, 120)
        b, c = binary_sa(b, n, rng, start_time, start_time + sa_time)
        if c < best_c5:
            best_c5 = c
            best_b = b.copy()
    return best_b, best_c5

def init_random_binary(n, rng):
    h = np.zeros(n)
    h[rng.choice(n, n // 2, replace=False)] = 1.0
    return h

def init_1periodic(n, rng):
    half = n // 2
    base = np.zeros(half)
    idx = rng.choice(half, half // 2, replace=False)
    base[idx] = 1.0
    return np.tile(base, 2).astype(float)

def init_stripe(n, rng, n_stripes=4):
    h = np.zeros(n)
    stripe_len = n // n_stripes
    for k in range(0, n_stripes, 2):
        h[k * stripe_len:(k + 1) * stripe_len] = 1.0
    return h

def validate_and_fix(h, n):
    h = np.clip(h, 0.0, 1.0)
    h = make_feasible(h, n)
    assert len(h) == n
    assert np.all(h >= -1e-10) and np.all(h <= 1.0 + 1e-10)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Quick continuous refinement of parent (200s)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        tl = min(180, remaining - 60)
        h_r, c_r = continuous_refine(best_h.copy(), n, rng, start_time, start_time + tl)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Phase 2: Binary SA from parent (thresholded) - main phase
    remaining = budget_s - (time.time() - start_time)
    if remaining > 300:
        b_parent = (best_h >= 0.5).astype(float)
        target = n // 2
        s = np.sum(b_parent)
        while s > target:
            ones = np.flatnonzero(b_parent)
            b_parent[ones[rng.randint(len(ones))]] = 0
            s -= 1
        while s < target:
            zeros = np.flatnonzero(1.0 - b_parent)
            b_parent[zeros[rng.randint(len(zeros))]] = 1
            s += 1
        tl = min(280, remaining - 100)
        b, c_b = binary_sa(b_parent, n, rng, start_time, start_time + tl)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = b.copy()

    # Phase 3: Binary SA from random start
    remaining = budget_s - (time.time() - start_time)
    if remaining > 250:
        b_rand = init_random_binary(n, rng)
        tl = min(230, remaining - 80)
        b, c_b = binary_sa(b_rand, n, rng, start_time, start_time + tl, T0=0.01, T1=1e-5)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = b.copy()

    # Phase 4: Binary SA from 1-periodic pattern
    remaining = budget_s - (time.time() - start_time)
    if remaining > 200:
        b_periodic = init_1periodic(n, rng)
        tl = min(180, remaining - 60)
        b, c_b = binary_sa(b_periodic, n, rng, start_time, start_time + tl, T0=0.008, T1=1e-5)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = b.copy()

    # Phase 5: Continuous refinement of best
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        tl = min(100, remaining - 30)
        h_r, c_r = continuous_refine(best_h.copy(), n, rng, start_time, start_time + tl)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Phase 6: Another round of binary SA from current best (thresholded)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 150:
        b_cur = (best_h >= 0.5).astype(float)
        target = n // 2
        s = np.sum(b_cur)
        while s > target:
            ones = np.flatnonzero(b_cur)
            b_cur[ones[rng.randint(len(ones))]] = 0
            s -= 1
        while s < target:
            zeros = np.flatnonzero(1.0 - b_cur)
            b_cur[zeros[rng.randint(len(zeros))]] = 1
            s += 1
        tl = min(140, remaining - 40)
        b, c_b = two_level_search(b_cur, n, rng, start_time, start_time + tl)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = b.copy()

    # Phase 7: Final continuous polish
    remaining = budget_s - (time.time() - start_time)
    if remaining > 40:
        tl = min(40, remaining - 5)
        h_r, c_r = continuous_refine(best_h.copy(), n, rng, start_time, start_time + tl)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()

    # Final validation
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
