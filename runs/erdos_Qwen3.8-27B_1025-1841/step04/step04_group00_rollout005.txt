        g[:n-t] += (1.0 - h[t:])

        top_idx = np.arange(len(corr))
        c_top = corr

    exp_c = np.exp(T * (c_top - mc))
    sum_exp = np.sum(exp_c)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(objf, h_in, jac=gradf, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = make_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h_in, compute_c5(h_in)

def subgrad_pgd(h_in, n, n_iter, start_time, budget_s):
    h = h_in.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 0.08
    for it in range(n_iter):
        if time.time() - start_time > budget_s - 3:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-5:]
        grad = np.zeros(n)
        for k in top_idx:
            s = int(k) - (n - 1)
            grad += shift_grad_vec(h, s, n)
        grad /= 5.0
        h = h - lr * grad
        h = make_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        if it % 40 == 0:
            lr *= 0.88
        if lr < 1e-6:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    def rem():
        return budget_s - (time.time() - start_time)

    # Phase 1: SLSQP with moderate T, varying top_m
    for T_val in [50, 100, 200]:
        if rem() < 80:
            break
        for top_m in [15, 25]:
            if rem() < 50:
                break
            h, c5 = slsqp_opt(best_h, n, T_val, top_m, 250)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Phase 2: Subgradient PGD for quick improvement
    if rem() > 50:
        h, c5 = subgrad_pgd(best_h, n, 3000, start_time, min(50, rem()))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 3: SLSQP higher T
    for T_val in [400, 800, 1500]:
        if rem() < 50:
            break
        h, c5 = slsqp_opt(best_h, n, T_val, 10, 200)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 4: More PGD
    if rem() > 40:
        h, c5 = subgrad_pgd(best_h, n, 2000, start_time, min(40, rem()))
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 5: Random restarts with SLSQP
    for r in range(12):
        if rem() < 70:
            break
        scale = 0.015 + 0.01 * (r % 4)
        h_try = make_feasible(best_h + rng.randn(n) * scale, n)
        h, c5 = slsqp_opt(h_try, n, 150, 15, 150)
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        if rem() > 50:
            h2, c52 = slsqp_opt(best_h, n, 1000, 5, 120)
            if c52 < best_c5:
                best_h, best_c5 = h2, c52

    # Phase 6: PGD + SLSQP cycle
    if rem() > 80:
        h, c5 = subgrad_pgd(best_h, n, 2000, start_time, min(40, rem()))
        if c5 < best_c5:
            best_h, best_c5 = h, c5
        if rem() > 40:
            h, c5 = slsqp_opt(best_h, n, 500, 12, 200)
            if c5 < best_c5:
                best_h, best_c5 = h, c5

    # Phase 7: Very high T final polish
    if rem() > 30:
        h, c5 = slsqp_opt(best_h, n, 5000, 3, 200)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    # Phase 8: Try 2x resolution
    if rem() > 250:
        n2 = n * 2
        old_x = np.linspace(0, 2, n)
        new_x = np.linspace(0, 2, n2)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, n2)
        c5_up = compute_c5(h_up)
        if rem() > 150:
            h3, c53 = slsqp_opt(h_up, n2, 150, 20, 150)
            if c53 < best_c5:
                best_h, best_c5 = h3, c53
            if rem() > 80:
                h4, c54 = slsqp_opt(best_h, len(best_h), 800, 8, 150)
                if c54 < best_c5:
                    best_h, best_c5 = h4, c54

    # Phase 9: Final ultra-polish
    if rem() > 25:
        h, c5 = slsqp_opt(best_h, len(best_h), 3000, 4, 200)
        if c5 < best_c5:
            best_h, best_c5 = h, c5

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
