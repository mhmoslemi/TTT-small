        return soft + mu * viol * viol

    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-12})
    return make_feasible(res.x, n)

def simulated_annealing(h_bin, n, rng, time_limit, T0=8.0, alpha=0.99995, tabu_size=300):
    dx = 2.0 / n
    start_ts = time.time()
    h = h_bin.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    current_cost = fast_corr_max_binary(h, n)
    best_cost = current_cost
    best_h = h.copy()
    T = T0
    tabu = set()
    tabu_queue = []
    iter_count = 0
    no_improve = 0
    last_best_cost = best_cost

    while time.time() - start_ts < time_limit:
        if no_improve > 100000:
            T = max(T, T0 * 0.3)
            no_improve = 0

        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        oi = int(ones[i_idx])
        oj = int(zeros[j_idx])

        if (oi, oj) in tabu:
            iter_count += 1
            continue

        h[oi] = 0.0
        h[oj] = 1.0
        new_cost = fast_corr_max_binary(h, n)

        if new_cost < current_cost:
            current_cost = new_cost
            ones[i_idx] = oj
            zeros[j_idx] = oi
            if new_cost < best_cost:
                best_cost = new_cost
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            delta = new_cost - current_cost
            if delta * dx < T * dx and rng.random() < np.exp(-delta / T):
                ones[i_idx] = oj
                zeros[j_idx] = oi
                current_cost = new_cost
                no_improve += 1
            else:
                h[oi] = 1.0
                h[oj] = 0.0
                no_improve += 1

        tabu_queue.append((oi, oj))
        tabu.add((oi, oj))
        if len(tabu_queue) > tabu_size:
            old = tabu_queue.pop(0)
            tabu.discard(old)

        T = max(T * alpha, 0.05)
        iter_count += 1

    return best_h, best_cost * dx, current_cost * dx

def coordinate_descent(h, n, deltas, max_rounds, time_limit):
    start_ts = time.time()
    current_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_ts > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_ts > time_limit:
                break
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < current_c5 - 1e-14:
                    current_c5 = c
                    h = tr
                    ov = nv
                    improved = True
                    break
            if time.time() - start_ts > time_limit:
                break
        if not improved:
            break
    return h, current_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)[:640]

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    n1 = 4096
    h_work = upsample_h(best_h, n1)
    c5w = compute_c5(h_work)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_work.copy()

    h_bin = binarize_fixed(h_work, n1, rng)
    c5b = compute_c5(h_bin)
    if c5b < best_c5:
        best_c5, best_h = c5b, h_bin.copy()

    # Structured patterns
    for period in [63, 127, 191, 255, 315, 511, 767, 1023]:
        if tl() < 10:
            break
        for frac in [0.45, 0.5, 0.55]:
            for offset in range(min(5, period)):
                pat = structured_pattern(n1, period, frac, offset)
                pb = binarize_fixed(pat, n1, rng)
                cpb = fast_corr_max_binary(pb, n1) * (2.0 / n1)
                if cpb < best_c5:
                    best_c5 = cpb
                    best_h = pb.copy()
                    h_bin = pb.copy()

    # SA Phase 1 at n=4096
    if tl() > 300:
        sa_time = min(tl() - 300, 600)
        h_sa, c5_sa, _ = simulated_annealing(h_bin, n1, rng, sa_time, T0=8.0, alpha=0.99999, tabu_size=200)
        if c5_sa < best_c5:
            best_c5, best_h = c5_sa, h_sa.copy()
            h_bin = h_sa.copy()

    # L-BFGS-B continuous at n=512 (quick)
    if tl() > 80:
        n_c = 512
        h_c = upsample_h(best_h, n_c)
        for T in [100, 500, 2000]:
            if tl() < 30:
                break
            try:
                h_opt = lbgfb_smooth(h_c, n_c, T, mu=500.0, max_iter=150, time_limit=20)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                h_c = h_opt.copy()
            except Exception:
                pass

    # SA Phase 2 at n=4096 (second round, lower temp)
    if tl() > 200:
        h_b2 = binarize_fixed(upsample_h(best_h, n1), n1, rng)
        c5b2 = compute_c5(h_b2)
        if c5b2 < best_c5:
            best_c5, best_h = c5b2, h_b2.copy()
        sa_time2 = min(tl() - 200, 350)
        h_sa2, c5_sa2, _ = simulated_annealing(h_b2, n1, rng, sa_time2, T0=3.0, alpha=0.99998, tabu_size=300)
        if c5_sa2 < best_c5:
            best_c5, best_h = c5_sa2, h_sa2.copy()
            h_bin = h_sa2.copy()

    # Try n=8192
    n2 = 8192
    if tl() > 250:
        h_high = upsample_h(best_h, n2)
        h_bh = binarize_fixed(h_high, n2, rng)
        c5bh = compute_c5(h_bh)
        if c5bh < best_c5:
            best_c5, best_h = c5bh, h_bh.copy()
        sa_time3 = min(tl() - 250, 280)
        h_sa3, c5_sa3, _ = simulated_annealing(h_bh, n2, rng, sa_time3, T0=5.0, alpha=0.99999, tabu_size=200)
        if c5_sa3 < best_c5:
            best_c5, best_h = c5_sa3, h_sa3.copy()

    # Final coordinate descent at moderate resolution
    n_cd = min(len(best_h), 2048)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    if tl() > 30:
        deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
        h_cd, c5_cd = coordinate_descent(h_cd, n_cd, deltas, 4, max(0, tl() - 20))
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Final exact verification and feasibility
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
