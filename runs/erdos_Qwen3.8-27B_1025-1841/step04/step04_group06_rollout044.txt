import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def soft_max_obj(h, n, T, mu, top_k=None):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if top_k is not None and top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        c = corr[idx]
    else:
        c = corr
    mc = np.max(c)
    shifted = T * (c - mc)
    shifted = np.clip(shifted, -100, 100)
    soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    viol = np.sum(h) - n / 2.0
    return soft + mu * viol * viol

        return soft_max_obj(h, n, T, mu, top_k)
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14,
                            'maxls': 50})
    return make_feasible(res.x, n)

    zeros = list(np.where(h <= 0.5)[0])

                    ones.pop(ii)

def slsqp_polish(h_start, n, T, max_iter):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
    def con(h):
        return np.sum(h) - n / 2.0
    def jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': jac}]
    res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

        order = rng.permutation(n)
        for j in order:
            if time_limit_fn is not None and time_limit_fn() < 10:
                break
            ov = h[j]
            best_val = ov
            best_c = current
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n)
                c = compute_c5(tr)
                if c < best_c:
                    best_c = c
                    best_val = nv
            # Try binary
            for bv in (0.0, 1.0):
                if abs(bv - ov) < 1e-9:

                tr[j] = bv
                tr = make_feasible(tr, n)

                    best_val = bv
            if best_val != ov:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
