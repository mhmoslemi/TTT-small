```python
    cur_c5 = compute_c5(h_cur)
    
    for epoch in range(20):
        if time_left() < 100: break
        no_improve = 0
        for _ in range(5000):

            # Random pair swap

            zeros = np.where(h_cur == 0)[0]
            a = ones[rng.randint(len(ones))]
            b = zeros[rng.randint(len(zeros))]
            h_cur[a], h_cur[b] = 1-h_cur[a], 1-h_cur[b]
            new_c5 = compute_c5(h_cur)
            if new_c5 < cur_c5 - 1e-14:
                cur_c5 = new_c5
                no_improve = 0
                if new_c5 < best_c5:

            else:
                h_cur[a], h_cur[b] = 1-h_cur[a], 1-h_cur[b]
                no_improve += 1
            if no_improve > 2000:
                break
        if no_improve >= 5000:
            break
    
    # Phase 3: Continuous optimization
    h_cont = best_h.copy() if len(best_h) == n else upsample_to(best_h, n)
    for T in [50, 200, 1000, 5000]:
        if time_left() < 60: break
        h_opt = lbgfb_optimize(h_cont, n, T)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5, best_h = c, h_opt.copy()
            h_cont = h_opt.copy()

    if time_left() > 50:
        h_bin2 = (best_h > 0.5).astype(float)
        fix_count(h_bin2, n)
        # Quick local search
        ...
    
    return best_h, best_c5, len(best_h)
```

Actually, let me think about a completely different approach: **direct optimization using the structure of the problem**.

The key observation is that C₅ = max_k I(k), and I(k) = ∑ᵢ h[i+k](1-h[i]) * (2/n).

If we define f_k(i) = h[i+k](1-h[i]), then I(k) = (2/n) * ∑ᵢ f_k(i).

For a binary h, f_k(i) ∈ {0, 1}, and I(k) = (2/n) * #{i: h[i]=0, h[i+k]=1}.

The problem is equivalent to: arrange n/2 ones and n/2 zeros in a sequence to minimize the maximum over all lags k of the count of "0 at position i, 1 at position i+k" pairs.

This is a **combinatorial design problem**. It's related to:
- Difference sets
- Golomb rulers
- Optimal beacon patterns

One known good construction: **Barker sequences** (but they only exist for specific lengths).

Another approach: **greedy with backtracking**. Place elements one by one, backtracking when stuck.

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def upsample_to(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return h_new

def fix_binary_count(h, n):
    h = (h > 0.5).astype(float).copy()
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        one_idx = np.where(h == 1)[0]
        for k in range(n_ones - target):
            h[one_idx[k]] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h == 0)[0]
        for k in range(target - n_ones):
            h[zero_idx[k]] = 1.0
    return h

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def lbgfb_smooth(h_start, n, T, mu=500.0, max_iter=300):
    dx = 2.0 / n
    def f(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        shifted = np.clip(shifted, -80, 80)
        soft = (mc + np.log(np.sum(np.exp(shifted))) / T) * dx
        viol = np.sum(hv) - n / 2.0
        return soft + mu * viol * viol
    bounds = [(0.0, 1.0)] * n
    res = minimize(f, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-13})
    return make_feasible(res.x, n)

def binary_local_search(h_bin, n, rng, time_fn, min_time=60):
    h = h_bin.copy()
    current = compute_c5(h)
    ones = np.where(h > 0.5)[0].copy()
    zeros = np.where(h <= 0.5)[0].copy()
    best_c5 = current
    best_h = h.copy()
    no_improve = 0
    total_tries = 0
    while time_fn() > min_time and no_improve < 15000:
        a = ones[rng.randint(len(ones))]
        b = zeros[rng.randint(len(zeros))]
        h[a] = 0.0
        h[b] = 1.0
        c = compute_c5(h)
        total_tries += 1
        if c < current - 1e-15:
            current = c
            idx_a = np.where(ones == a)[0][0]
            idx_b = np.where(zeros == b)[0][0]
            ones = np.delete(ones, idx_a)
            zeros = np.delete(zeros, idx_b)
            ones = np.append(ones, b)
            zeros = np.append(zeros, a)
            no_improve = 0
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
            no_improve += 1
    return best_h, best_c5

def block_swap_search(h_bin, n, rng, time_fn, min_time=30):
    h = h_bin.copy()
    current = compute_c5(h)
    best_c5 = current
    best_h = h.copy()
    no_improve = 0
    max_block = max(2, n // 64)
    while time_fn() > min_time and no_improve < 5000:
        block_len = rng.randint(1, max_block + 1)
        start_a = rng.randint(0, n - block_len)
        start_b = rng.randint(0, n - block_len)
        if abs(start_a - start_b) < block_len:
            no_improve += 1
            continue
        seg_a = h[start_a:start_a + block_len].copy()
        seg_b = h[start_b:start_b + block_len].copy()
        # Swap if one is all 1s and other is all 0s, or vice versa
        sum_a = np.sum(seg_a)
        sum_b = np.sum(seg_b)
        if abs(sum_a - sum_b) > 0.5:
            h[start_a:start_a + block_len] = seg_b
            h[start_b:start_b + block_len] = seg_a
            c = compute_c5(h)
            if c < current - 1e-15:
                current = c
                no_improve = 0
                if c < best_c5:
                    best_c5 = c
                    best_h = h.copy()
            else:
                h[start_a:start_a + block_len] = seg_a
                h[start_b:start_b + block_len] = seg_b
                no_improve += 1
        else:
            no_improve += 1
    return best_h, best_c5

def coordinate_descent(h_in, n, time_fn, min_time=15):
    h = h_in.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    deltas = [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]
    for rnd in range(20):
        if time_fn() < min_time:
            break
        improved = False
        n_cd = len(h)
        for j in range(0, n_cd, 1):
            if time_fn() < min_time:
                break
            ov = h[j]
            for d in deltas:
                nv = ov + d
                if nv < 0 or nv > 1:
                    continue
                tr = h.copy()
                tr[j] = nv
                tr = make_feasible(tr, n_cd)
                c = compute_c5(tr)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = tr
                    ov = nv
                    improved = True
                    break
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                tr = h.copy()
                tr[j] = bv
                tr = make_feasible(tr, n_cd)
                c = compute_c5(tr)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = tr
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    try:

    n = 1280
    h_w = upsample_to(best_h, n)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()

    # Phase 2: Binary local search (primary)
    h_bin = fix_binary_count(h_w, n)
    c5b = compute_c5(h_bin)
    if c5b < best_c5:
        best_c5, best_h = c5b, h_bin.copy()

    h_bs, c5_bs = binary_local_search(h_bin, n, rng, tl, min_time=120)
    if c5_bs < best_c5:

    # Phase 3: Block swap search

        h_sw, c5_sw = block_swap_search(best_h, n, rng, tl, min_time=50)
        if c5_sw < best_c5:

    # Phase 4: More binary local search from improved point
    if tl() > 100:
        h_bin2 = fix_binary_count(best_h, n)
        h_bs2, c5_bs2 = binary_local_search(h_bin2, n, rng, tl, min_time=60)
        if c5_bs2 < best_c5:
            best_c5, best_h = c5_bs2, h_bs2.copy()

    # Phase 5: Continuous L-BFGS optimization
    if tl() > 80:
        h_cont = best_h.copy()
        for T in [30, 100, 400, 1200]:

                    best_c5, best_h = c5_opt, h_opt.copy()

            except Exception:

        except Exception:
            pass

    # Phase 9: One more binary pass

        h_bin4 = fix_binary_count(best_h, n)
```
