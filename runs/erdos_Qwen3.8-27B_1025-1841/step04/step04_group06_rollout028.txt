from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(300):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def lse_objective(h, n, T, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

    h_w = upsample_h(best_h, n1)
    c5w = compute_c5(h_w)
    if c5w < best_c5:
        best_c5, best_h = c5w, h_w.copy()

    for T in [200, 80, 30, 15, 8, 4, 2]:
        if tl() < 25:
            break
        try:
            h_opt = lbgfb_smooth(h_w, n1, T, 200.0, max_iter=180)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
            h_w = h_opt.copy()
        except Exception:
            pass

    # Random restarts
    for restart in range(4):
        if tl() < 20:
            break
        h_init = h_w + 0.35 * rng.randn(n1)
        h_init = make_feasible(h_init, n1)
        for T in [60, 10]:
            if tl() < 12:
                break
            try:
                h_opt = lbgfb_smooth(h_init, n1, T, 200.0, max_iter=120)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_w = h_opt.copy()
            except Exception:
                pass

    # Phase 2: Binary optimization at n=512
    if tl() > 80:
        h_bin = (h_w > 0.5).astype(float)
        n_ones = int(np.sum(h_bin))
        n_target = n1 // 2
        if n_ones > n_target:
            one_idx = np.where(h_bin == 1)[0]
            for k in range(n_ones - n_target):
                h_bin[one_idx[k]] = 0.0
        elif n_ones < n_target:
            zero_idx = np.where(h_bin == 0)[0]
            for k in range(n_target - n_ones):
                h_bin[zero_idx[k]] = 1.0
        c5b = compute_c5(h_bin)
        if c5b < best_c5:
            best_c5, best_h = c5b, h_bin.copy()

        h_bs, c5_bs = greedy_binary(h_bin, n1, rng, n_rounds=40, n_tries=3000)
        if c5_bs < best_c5:
            best_c5, best_h = c5_bs, h_bs.copy()

        h_sw, c5_sw = two_swap_search(best_h, n1, rng, n_rounds=10, n_tries=1500)
        if c5_sw < best_c5:
            best_c5, best_h = c5_sw, h_sw.copy()

        if tl() > 150:
            h_sa, c5_sa = binary_sa(best_h, n1, rng, n_iter=40000, T_start=0.15, T_end=0.01)
            if c5_sa < best_c5:
                best_c5, best_h = c5_sa, h_sa.copy()

    # Phase 3: Higher resolution optimization
    for n_pts in [1024, 2048]:
        if tl() < 50:
            break
        h_c = upsample_h(best_h, n_pts)
        c5c = compute_c5(h_c)
        if c5c < best_c5:
            best_c5, best_h = c5c, h_c.copy()

        for T in [40, 15, 5, 2]:
            if tl() < 20:
                break
            try:
                h_opt = lbgfb_smooth(h_c, n_pts, T, 300.0, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass

        # Random restart at this resolution
        if tl() > 35:
            h_r = h_c + 0.2 * rng.randn(n_pts)
            h_r = make_feasible(h_r, n_pts)
            try:
                h_opt = lbgfb_smooth(h_r, n_pts, 20, 300.0, max_iter=120)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5, best_h = c5_opt, h_opt.copy()
                    h_c = h_opt.copy()
            except Exception:
                pass

        # Binary SA at this resolution
        if tl() > 80:
            h_b2 = (h_c > 0.5).astype(float)
            n_ones2 = int(np.sum(h_b2))
            n_t2 = n_pts // 2
            if n_ones2 > n_t2:
                oi2 = np.where(h_b2 == 1)[0]
                for k in range(n_ones2 - n_t2):
                    h_b2[oi2[k]] = 0.0
            elif n_ones2 < n_t2:
                zi2 = np.where(h_b2 == 0)[0]
                for k in range(n_t2 - n_ones2):
                    h_b2[zi2[k]] = 1.0
            c5_b2 = compute_c5(h_b2)
            if c5_b2 < best_c5:
                best_c5, best_h = c5_b2, h_b2.copy()
            if tl() > 60:
                h_sa2, c5_sa2 = binary_sa(h_b2, n_pts, rng, n_iter=20000, T_start=0.12, T_end=0.01)
                if c5_sa2 < best_c5:
                    best_c5, best_h = c5_sa2, h_sa2.copy()

    # Phase 4: Coordinate descent polish
    n_cd = min(len(best_h), 1024)
    h_cd = upsample_h(best_h, n_cd) if n_cd < len(best_h) else best_h.copy()
    if tl() > 25:
        h_cd, c5_cd = coord_descent(h_cd, n_cd, max_rounds=4)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 5: Final L-BFGS-B squeeze
    if tl() > 15:
        n_f = len(best_h)
        try:
            h_opt = lbgfb_smooth(best_h, n_f, 3, 500.0, max_iter=80)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5, best_h = c5_opt, h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
