import numpy as np
from scipy.optimize import minimize
import time

_OPT = {}

def _obj_fn(x):
    n = _OPT['n']
    T = _OPT['T']
    top_k = _OPT['top_k']
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    if 0 < top_k < len(corr):
        idx = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[idx]
    else:
        vals = corr
    mc = np.max(vals)
    shifted = T * (vals - mc)
    return mc + np.log(np.sum(np.exp(shifted))) / T

def _con_fun(x):
    return np.sum(x) - _OPT['n'] / 2.0

def _con_jac(x):
    return np.ones(_OPT['n'])

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

    _OPT['T'] = T
    _OPT['top_k'] = top_k
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:

                       bounds=bounds, constraints=cons,

        h3 = upsample_h(best_h, n2)

        h3 = make_feasible(h3, n2)
        h3 = slsqp_optimize(h3, n2, 300, 0, 250, 1e-13)
        h3 = slsqp_optimize(h3, n2, 700, 5, 200, 1e-13)
        c = compute_c5(h3)
        if c < best_c5:
            best_c5 = c
            best_h = h3.copy()
            best_n = n2

    # Phase 5: Binary refinement
    if remaining() > 40:
        n_cur = len(best_h)
        best_h, best_c5 = binary_refine(best_h, n_cur, remaining)

    # Phase 6: Swap refinement
    if remaining() > 30:
        n_cur = len(best_h)
        best_h, best_c5 = swap_refine(best_h, n_cur, remaining, max_pairs=3000)

    # Phase 7: Focused coordinate descent
    if remaining() > 30:
        n_cur = len(best_h)
        best_h, best_c5 = coord_descent_focused(best_h, n_cur, remaining, rounds=25)

    # Phase 8: Final SLSQP polish
    if remaining() > 50:
        n_cur = len(best_h)
        best_h = slsqp_optimize(best_h, n_cur, 600, 3, 150, 1e-14)
        c = compute_c5(best_h)
        if c < best_c5:
            best_c5 = c

    # Phase 9: Final binary snap
    if remaining() > 20:
        n_cur = len(best_h)
        best_h, best_c5 = binary_refine(best_h, n_cur, remaining)

    # Phase 10: One more coord descent
    if remaining() > 10:
        n_cur = len(best_h)
        best_h, best_c5 = coord_descent_focused(best_h, n_cur, remaining, rounds=10)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
