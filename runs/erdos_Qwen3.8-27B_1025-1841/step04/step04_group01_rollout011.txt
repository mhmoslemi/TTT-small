    """Find optimal h[j] by considering top_k shifts."""
    n = len(h)
    dx = 2.0/n
    # Compute current correlations
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top_k shifts
    top_idx = np.argsort(-corr)[:top_k]
    
    # For each top shift, compute f_s as function of t = h[j]
    # f_s(t) = base_s + slope_s * t (for s != 0)
    # f_0(t) = base_0 + slope_0 * t - t^2 (for s = 0, i.e., shift index n-1)
    
    slopes = np.zeros(len(top_idx))
    bases = np.zeros(len(top_idx))
    zero_idx = -1  # index in top_idx where shift = 0 (i.e., top_idx[k] == n-1)

        if s == 0:

            # slope is (1-2t), base is 0 for the t-dependent part

            # = sum_{i≠j} h[i] - h[i]² + t - t²
