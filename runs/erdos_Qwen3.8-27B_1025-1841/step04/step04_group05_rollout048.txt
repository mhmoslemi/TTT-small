def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def _time_left():
        return budget_s - (time.time() - start_time)
    
    best_c5 = float('inf')
    best_h = None
    
    # Get initial values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.5], 320)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: Upsample to n=2048 and optimize with SLSQP
    n1 = 2048
    h_curr = upsample_h(best_h, n1)
    c5_curr = compute_c5(h_curr)
    if c5_curr < best_c5:
        best_c5 = c5_curr
        best_h = h_curr.copy()
    
    # SLSQP optimization at n=2048
    def _slsqp(h_start, n, T, max_iter):
        dx = 2.0 / n
        def obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full') * dx
            mc = np.max(corr)
            shifted = corr - mc
            return mc + np.log(np.sum(np.exp(T * shifted))) / T
        def cons_fun(h):
            return np.sum(h) - n / 2.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': cons_fun, 'jac': lambda h: np.ones(n)}]
        res = minimize(obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        return make_feasible(res.x, n)
    
    # Temperature schedule for SLSQP
    for T in [20.0, 80.0, 300.0, 1000.0]:
        if _time_left() < 30:
            break
        n_iter = min(200, int(60 / _time_left() * 1300))  # scale iterations to time
        n_iter = max(50, min(n_iter, 300))
        try:
            h_opt = _slsqp(h_curr, n1, T, n_iter)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_curr = h_opt.copy()
        except Exception as e:
            pass
    
    # Phase 2: Binary local search (simulated annealing style)
    if _time_left() > 120:
        h_bin = (h_curr > 0.5).astype(float)
        # Ensure exactly n/2 ones
        n_ones = int(np.sum(h_bin))
        target_ones = n1 // 2
        if n_ones > target_ones:
            # Set some 1s to 0s (the ones closest to 0.5)
            one_indices = np.where(h_bin == 1)[0]
            scores = h_curr[one_indices]  # lower score = closer to 0
            order = np.argsort(scores)[:n_ones - target_ones]
            h_bin[one_indices[order]] = 0
        elif n_ones < target_ones:
            zero_indices = np.where(h_bin == 0)[0]
            scores = 1 - h_curr[zero_indices]  # lower score = closer to 0
            order = np.argsort(h_curr[zero_indices], kind='stable')[-(target_ones - n_ones):]
            h_bin[zero_indices[order]] = 1
        
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        
        # Simulated annealing on binary
        sa_start = time.time()
        T_sa = 0.01  # initial temperature for acceptance
        T_sa_end = 0.001
        n_sa_iters = min(50000, int(80 / (2.0/n1 + 0.001) * 100))  # rough time-based
        n_sa_iters = max(10000, min(n_sa_iters, 100000))
        
        h_sa = h_bin.copy()
        c5_sa = c5_bin
        one_pos = np.where(h_sa == 1)[0]
        zero_pos = np.where(h_sa == 0)[0]
        
        for it in range(n_sa_iters):
            if time.time() - sa_start > 80:
                break
            # Pick random 1 and random 0, swap them
            i1 = one_pos[rng.randint(len(one_pos))]
            i0 = zero_pos[rng.randint(len(zero_pos))]
            h_sa[i1] = 0
            h_sa[i0] = 1
            c5_new = compute_c5(h_sa)
            delta = c5_new - c5_sa
            if delta < 0 or rng.rand() < np.exp(-delta / T_sa):
                c5_sa = c5_new
                # Update position arrays
                one_pos = np.where(h_sa == 1)[0]
                zero_pos = np.where(h_sa == 0)[0]
                if c5_sa < best_c5:
                    best_c5 = c5_sa
                    best_h = h_sa.copy()
            else:
                h_sa[i1] = 1
                h_sa[i0] = 0
            # Cool down
            T_sa += (T_sa_end - T_sa) * 0.001
        
    # Phase 3: Polish with SLSQP at higher T from best
    if _time_left() > 40:
        h_curr = upsample_h(best_h, n1)
        for T in [500.0, 2000.0]:
            if _time_left() < 25:
                break
            try:

                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
