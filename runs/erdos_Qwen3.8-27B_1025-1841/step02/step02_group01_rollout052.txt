def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)

    old_x = (np.arange(old_n) + 0.5) * 2.0 / old_n
    new_x = (np.arange(new_n) + 0.5) * 2.0 / new_n
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_obj(h, t, mu):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Soft max
    sm = mc + np.log(np.sum(np.exp(t * shifted))) / t
    # Penalty for sum constraint
    penalty = mu * (np.sum(h) - n / 2.0) ** 2
    return sm * dx + penalty

    bounds = [(0.0, 1.0)] * n
    res = minimize(lambda x: soft_max_obj(x, t, mu), h, method='L-BFGS-B',
                   bounds=bounds, options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-6})
    return make_feasible(res.x, n)

def slsqp_optimize(h, t=50.0, maxiter=500, ftol=1e-10):
    n = len(h)
    dx = 2.0 / n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        sm = mc + np.log(np.sum(np.exp(t * shifted))) / t
        return sm * dx
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return make_feasible(res.x, n)

                h1 = opt_lbgfsb(h1024, t=temp, mu=1000, maxiter=600)
                c1 = compute_c5(h1)
                if c1 < best_c5:
                    best_h, best_c5 = h1, c1
                    h1024 = h1
            except Exception:
                pass

    # Phase 8: SLSQP at 1024
    if time_left() > 150:
        try:
            h1 = opt_slsqp(h1024, t=60, maxiter=400, ftol=1e-12)
            c1 = compute_c5(h1)
            if c1 < best_c5:
                best_h, best_c5 = h1, c1
                h1024 = h1
        except Exception:
            pass

    # Phase 9: Coordinate descent at 1024
    if time_left() > 50:
        tl = time_left() - 20
        best_h, best_c5 = coord_descent(best_h, max_rounds=4, n_grid=35, time_limit=tl)

    # Phase 10: Final polish - L-BFGS-B then coord descent
    if time_left() > 80:
        try:
            h1 = opt_lbgfsb(best_h, t=80, mu=1000, maxiter=300)
            c1 = compute_c5(h1)
            if c1 < best_c5:
                best_h, best_c5 = h1, c1
        except Exception:
            pass
    if time_left() > 30:
        tl = time_left() - 10
        best_h, best_c5 = coord_descent(best_h, max_rounds=3, n_grid=50, time_limit=tl)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
