import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    hb = (h >= 0.5).astype(float)
    n_ones = int(np.sum(hb))
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        ones = np.where(hb == 1)[0]
        # Flip the ones with lowest h values
        order = np.argsort(h[ones])[:n_ones - target]
        hb[ones[order]] = 0
    elif n_ones < target:
        zeros = np.where(hb == 0)[0]
        order = np.argsort(-h[zeros])[:target - n_ones]
        hb[zeros[order]] = 1
    return hb

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h_bin, n_max_iters=500000, temp_start=0.01, temp_end=0.001, rng=None):
    """Local search on binary sequences using swap moves."""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h_bin)
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = np.max(corr)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_max_iters):
        temp = temp_start * (temp_end / temp_start) ** (it / n_max_iters)
        
        # Pick random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Try the swap
        h[i] = 0
        h[j] = 1
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(corr_new)
        
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / (temp * 2.0 / n)):
            # Accept
            current_c5 = new_c5
            corr = corr_new
            # Update ones/zeros
            ones = ones[ones != i]
            zeros = zeros[zeros != j]
            ones = np.append(ones, j)
            zeros = np.append(zeros, i)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Reject
            h[i] = 1
            h[j] = 0
    
    return best_h, best_c5 * 2.0 / n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    logsum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + logsum / T) * dx

def soft_obj_grad(h, T, top_k=50):
    """Gradient of the soft-max objective using top-k approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)

    weights = exp_s / total
    # Use top-k for efficiency
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for ki in idx:
        w = weights[ki]
        if w < 1e-10:
            break
        shift = ki - (n - 1)
        grad += w * subgrad(h, shift)

def subgrad(h, shift):
    """Subgradient of corr_k(h) = sum_n h[n+k]*(1-h[n]) w.r.t. h."""
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        # corr = sum_{n=0}^{n-1-s} h[n+s]*(1-h[n])
        # d/dh[i]: appears in h[n+s] term (when i = n+s, n = i-s) and (1-h[n]) term (when i = n, n = i)
        # d/dh[i] from h[i] term: (1-h[i-s]) for i >= s
        # d/dh[i] from (1-h[i]) term: -h[i+s] for i < n-s
        grad[s:] += (1.0 - h[:n-s])
        grad[:n-s] += (-h[s:])
    else:
        s = -shift
        # corr = sum_{n=s}^{n-1} h[n-s]*(1-h[n])
        # d/dh[i] from h[i] term (i = n-s, n = i+s): (1-h[i+s]) for i < n-s
        # d/dh[i] from (1-h[i]) term (i = n): -h[i-s] for i >= s
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] += (-h[:n-s])
    return grad

def gd_optimize(h, n_iter, T_start, T_end, lr, decay=0.999, check_every=30):

    n = len(h)

    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr_t = lr

        T = T_start * (T_end / T_start) ** frac

    c5 = compute_c5(h)

        best_h = h.copy()
    return best_h, best_c5

def sa_optimize(h, n_iter, T_start, T_end, step_size, rng):

    n = len(h)

    best_c5 = compute_c5(h)
    best_h = h.copy()

        delta_c = c5_new - current_c5

            h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    return best_h, best_c5
