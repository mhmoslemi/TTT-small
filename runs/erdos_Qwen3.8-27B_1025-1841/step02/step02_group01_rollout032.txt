import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def grad_lag_vec(h, k_full, n):
    """Vectorized gradient of f_k for lag index k_full in full correlation."""
    dx = 2.0 / n
    lag = k_full - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += dx * (1.0 - h[lag:])
        grad[lag:] -= dx * h[:n - lag]
    else:
        m = -lag
        grad[m:] += dx * (1.0 - h[:n - m])
        grad[:n - m] -= dx * h[m:]
    return grad

def pgd_step(h, K=5, temp=5.0, step_size=None):
    """One PGD step with top-K gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    
    top_idx = np.argsort(corr)[-K:]
    shifts = corr[top_idx] - corr[top_idx[-1]]
    weights = np.exp(temp * shifts)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        if weights[i] > 1e-8:
            grad += weights[i] * grad_lag_vec(h, int(k), n)
    
    # Scale gradient
    # The gradient of f_k has norm roughly O(1) since f_k is O(1)
    # We want step size to be small enough for convergence
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-10:
        return h.copy(), corr
    grad = grad / gnorm  # normalize
    
    # Line search
    best_h = h.copy()
    best_f = corr.max()
    for s in [step_size or 0.01, 0.005, 0.002, 0.001, 0.0005]:
        h_new = h - s * grad
        h_new = make_feasible(h_new, n)
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
        f_new = corr_new.max()
        if f_new < best_f:
            best_h = h_new
            best_f = f_new
            break
        best_h = h_new  # still move, just less
        best_f = f_new
    return best_h, corr_new

def pgd_optimize(h, n, max_iter=5000, time_limit=None, start_time=None):
    """Run PGD optimization."""
    h = h.copy()
    best_h = h.copy()

    weights = np.exp(M * shifted)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    # Only process lags with significant weight
    thresh = 1e-8
    for k in range(len(corr)):
        if weights[k] > thresh:
            grad += weights[k] * grad_at_lag(h, k, n)
    return corr, grad

def pgd_optimize(h, n, M_start=2.0, M_end=30.0, n_iter=3000):
    """PGD with annealed log-sum-exp."""
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    base_step = 0.02
    
    for it in range(n_iter):
        progress = it / max(1, n_iter)
        M = M_start + (M_end - M_start) * progress
        
        corr, grad = lse_gradient(h, n, M)
        c5 = corr.max()
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        # Line search
        step = base_step
        improved = False
        for s in [step, step*0.5, step*0.25, step*0.125, step*0.0625]:

                h = h_new
                improved = True
                break
        if not improved:
            h = make_feasible(h - step*0.0625 * grad, n)
    
    return best_h, best_c5

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def grad_at_lag(h, k_full, n):
    dx = 2.0 / n
    lag = k_full - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        grad[:n - lag] += dx * (1.0 - h[lag:])
        grad[lag:] -= dx * h[:n - lag]
    else:
        m = -lag
        grad[m:] += dx * (1.0 - h[:n - m])
        grad[:n - m] -= dx * h[m:]
    return grad

def lse_grad(corr_values, h, n, M):
    shifted = corr_values - corr_values.max()
    weights = np.exp(M * shifted)
    weights /= weights.sum()
    grad = np.zeros(n)
    thresh = max(1e-8, 1.0 / (n * 50))
    active = np.where(weights > thresh)[0]
    for k in active:
        grad += weights[k] * grad_at_lag(h, int(k), n)
    return grad

def pgd_optimize(h, n, M_start=2.0, M_end=40.0, n_iter=3000):
    h = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    stalled = 0
    base_step = 0.03

    for it in range(n_iter):
        progress = it / max(1, n_iter)
        M = M_start + (M_end - M_start) * (progress ** 0.7)

        corr_raw = np.correlate(h, 1.0 - h, mode='full')
        corr = corr_raw * dx
        c5 = corr.max()
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1

        if stalled > 800:
            h = make_feasible(h + 0.02 * np.random.RandomState(it).randn(n), n)
            stalled = 0

        grad = lse_grad(corr, h, n, M)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break

        grad_dir = grad / gnorm
        improved = False
        for s in [base_step, base_step * 0.5, base_step * 0.25, base_step * 0.1]:
            h_new = make_feasible(h - s * grad_dir, n)
            c5_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full')) * dx
            if c5_new < c5 - 1e-14:
                h = h_new
                improved = True
                break
        if not improved:
            h = make_feasible(h - base_step * 0.1 * grad_dir, n)

    return best_h, best_c5

def random_binary(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n_ones:
        to_flip = rng.choice(ones, len(ones) - n_ones, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < n_ones:
        to_flip = rng.choice(zeros, n_ones - len(ones), replace=False)
        h[to_flip] = 1.0
    return h

def coord_descent(h, n, max_rounds=8, max_time=60.0, t_start=None):
    if t_start is None:
        t_start = time.time()
    h = h.copy()
    dx = 2.0 / n
    c5 = compute_c5(h)

    for rnd in range(max_rounds):
        if time.time() - t_start > max_time:
            break
        improved = False
        step_vals = [0.0, 1.0, 0.5]
        for j in range(n):
            if time.time() - t_start > max_time:
                break
            old = h[j]
            local_best = old
            local_c5 = c5
            candidates = step_vals + [old + 0.02, old - 0.02, old + 0.05, old - 0.05, old + 0.1, old - 0.1]
            for cand in candidates:
                if cand < -1e-9 or cand > 1 + 1e-9:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = np.max(np.correlate(trial, 1.0 - trial, mode='full')) * dx
                if c < local_c5 - 1e-12:
                    local_c5 = c
                    local_best = cand
            if local_best != old:
                h[j] = local_best
                h = make_feasible(h, n)
                c5 = local_c5
                improved = True
        if not improved:
            break
    return h, c5

        h_opt, c5_opt = pgd_optimize(h, n, M_start=2.0, M_end=40.0, n_iter=n_iter)
        if c5_opt < best_c5:

        if c5_opt2 < best_c5:

            best_h = h_opt3.copy()

    # Phase 2: Focused refinement at best resolution
    n_best = len(best_h)

        h2 = best_h + rng.randn(n_best) * 0.01
        h2 = make_feasible(h2, n_best)
        h_ref, c5_ref = pgd_optimize(h2, n_best, M_start=10.0, M_end=60.0, n_iter=3000)
