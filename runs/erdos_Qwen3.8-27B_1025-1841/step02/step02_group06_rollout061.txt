        s = np.sum(np.clip(g - mid, 0.0, 1.0))

        else:

```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    xo = np.arange(n) * (2.0 / n)
    xn = np.arange(target_n) * (2.0 / target_n)
    return np.interp(xn, xo, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def soft_grad_from_corr(corr, h, n, dx, tau, topM=0):
    g = corr * dx
    gs = g - g.max()
    eg = np.exp(np.clip(tau * gs, -500.0, 0.0))
    s = eg.sum()
    p = eg / s if s > 1e-300 else np.ones(len(g)) / len(g)
    if topM > 0 and topM < len(p):
        ti = np.argpartition(p, -topM)[-topM:]
        ps = p[ti] / p[ti].sum()
        idxs = [(ti[i], ps[i]) for i in range(len(ti)) if ps[i] > 1e-15]
    else:
        idxs = [(k, p[k]) for k in range(len(p)) if p[k] > 1e-15]
    grad = np.zeros(n)
    for k, pk in idxs:
        d = k - n + 1
        j0, j1 = max(0, d), min(n, n + d)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - d:j1 - d])
        j0b, j1b = max(0, -d), min(n, n - d)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + d:j1b + d]
    return grad * dx

def adam_opt(h, n, tau_list, lr, time_limit, topM=0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr(h)
    bh, bc = h.copy(), corr.max() * dx
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    s0 = time.time()
    b1, b2, eps = 0.85, 0.999, 1e-8
    for tau in tau_list:
        while time.time() - s0 < time_limit:
            t += 1
            corr = compute_corr(h)
            c5 = corr.max() * dx
            if c5 < bc - 1e-15:
                bc, bh = c5, h.copy()
            grad = soft_grad_from_corr(corr, h, n, dx, tau, topM)
            lr_t = lr * np.sqrt(1.0 - b2 ** t) / (1.0 - b1 ** t)
            m = b1 * m + (1.0 - b1) * grad
            v = b2 * v + (1.0 - b2) * grad * grad
            mh = m / (1.0 - b1 ** t)
            vh = v / (1.0 - b2 ** t)
            h = project_to_feasible(h - lr_t * mh / (np.sqrt(vh) + eps), n)
    return bh, bc

def local_perturb(h, n, tlim, noise, ntr, npert=200):
    h = project_to_feasible(h, n).copy()
    bh, bc = h.copy(), compute_c5(h)
    s0 = time.time()
    for _ in range(ntr):
        if time.time() - s0 > tlim:
            break
        idx = np.random.randint(0, n, size=min(npert, n))
        ht = h.copy()
        ht[idx] += noise * np.random.randn(len(idx))
        ht = project_to_feasible(ht, n)
        ct = compute_c5(ht)
        if ct < bc - 1e-15:
            bc, bh, h = ct, ht.copy(), ht.copy()
    return bh, bc

def local_swap(h, n, tlim, ntr):
    h = project_to_feasible(h, n).copy()
    bh, bc = h.copy(), compute_c5(h)
    s0 = time.time()
    for _ in range(ntr):
        if time.time() - s0 > tlim:
            break
        i, j = np.random.randint(0, n, 2)
        if i == j:
            continue
        alpha = np.random.rand()
        ht = h.copy()
        di = h[j] - h[i]
        ht[i] = h[i] + alpha * di
        ht[j] = h[j] - alpha * di
        if ht[i] < 0 or ht[i] > 1 or ht[j] < 0 or ht[j] > 1:
            continue
        ct = compute_c5(ht)
        if ct < bc - 1e-15:
            bc, bh, h = ct, ht.copy(), ht.copy()
    return bh, bc

def subpolish(h, n, tlim, lr0, topK):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr(h)
    bh, bc = h.copy(), corr.max() * dx
    s0 = time.time()
    v = np.zeros(n)
    beta = 0.85
    it = 0
    while time.time() - s0 < tlim:
        it += 1
        corr = compute_corr(h)
        c5 = corr.max() * dx
        if c5 < bc - 1e-15:
            bc, bh = c5, h.copy()
        top = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top:
            d = ki - n + 1
            j0, j1 = max(0, d), min(n, n + d)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[j0 - d:j1 - d])
            j0b, j1b = max(0, -d), min(n, n - d)
            if j0b < j1b:
                grad[j0b:j1b] -= h[j0b + d:j1b + d]
        lr = lr0 / (1.0 + it * 0.005)
        v = beta * v - lr * grad * dx / topK
        h = project_to_feasible(h + v, n)
    return bh, bc
