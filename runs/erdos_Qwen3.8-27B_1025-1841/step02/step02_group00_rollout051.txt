_O_N = 0
_O_T = 1.0

def _smo(h):
    n = _O_N
    T = _O_T
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * (2.0/n)

def _ce(h):
    return np.sum(h) - _O_N / 2.0

def _cej(h):
    return np.ones(_O_N)

def _slsqp(h0, n, T, iters):
    global _O_N, _O_T
    _O_N = n
    _O_T = T
    r = minimize(_smo, h0, method='SLSQP',
                 bounds=[(0.,1.)]*n,
                 constraints=[{'type':'eq','fun':_ce,'jac':_cej}],
                 options={'maxiter':iters, 'ftol':1e-14})
    return np.clip(r.x, 0, 1)

        if c==0: break

    k = ki - (n-1)  # lag

        for j in range(kp, n):
            g[j] -= h[j-kp]
