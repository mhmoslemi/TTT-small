import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -600, 600)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    active = np.where(p > 1e-11)[0]
    for k in active:
        pk = p[k]
        sh = k - n + 1
        if sh >= 0:
            L = n - sh
            if L > 0:
                grad[:L] += pk * (1.0 - h[sh:n])
                grad[sh:] -= pk * h[:L]
        else:
            ssh = -sh
            L = n - ssh
            if L > 0:
                grad[ssh:] += pk * (1.0 - h[:L])
                grad[:L] -= pk * h[ssh:n]
    grad *= dx
    c5_now = max_c * dx
    return c5_now, grad


def optimize_smooth(h, n, tau, time_limit, lr, momentum=0.9, reheat_interval=3000, reheat_mag=0.003):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = momentum
    stall = 0
    it = 0

    while time.time() - start < time_limit:
        c5_now, grad = smooth_obj_grad(h, tau)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        it += 1

        if stall > reheat_interval:
            h += reheat_mag * np.random.randn(n)
            h = project_to_feasible(h, n)
            lr *= 0.5
            v *= 0.1
            stall = 0
            if lr < 1e-14:
                break

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5


def optimize_multishift(h, n, time_limit, lr=0.003, top_k=10):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.82
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            lr *= 0.6
            v *= 0.2
            stall = 0
            if lr < 1e-15:
                break

        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        top_vals = corr[top_idx]
        margin = top_vals[0] - top_vals
        w = np.exp(-margin * 5.0)
        w /= w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            wi = w[i]
            sh = k - n + 1
            if sh >= 0:

                    grad[sh:] -= wi * h[:L]
            else:
                ssh = -sh
                L = n - ssh

        grad *= dx

        h = project_to_feasible(h + v, n)
