    best_h = h.copy()

        frac = it / max(n_iter - 1, 1)

        mc = np.max(corr)
        shifted = np.clip(corr - mc, -100.0/T, 0)
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

        val, g = softmax_value_grad(x, T, top_k=n)
        return g

            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def get_corr(h):

def softmax_val_grad(h, T, n, top_k=None):

    corr = get_corr(h)
    mc = np.max(corr)
    shifted = corr - mc

                       bounds=bounds, constraints=cons,

        pass
    return h, compute_c5(h)

def coord_descent(h, n, passes=4, eps_init=0.004):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for p in range(passes):
        eps = eps_init * (0.6 ** p)
        improved = False
        for i in range(n):
            for delta in [eps, -eps]:
                trial = best_h.copy()
                trial[i] += delta
                trial = make_feasible(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c5 - 1e-12:
                    best_h = trial
                    best_c5 = c5
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n_init = 320
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        n_ones = n_init // 2
        ones = np.where(h0 == 1)[0]
        zeros = np.where(h0 == 0)[0]
        if len(ones) > n_ones:
            h0[rng.choice(ones, len(ones) - n_ones, replace=False)] = 0.0
        elif len(ones) < n_ones:
            h0[rng.choice(zeros, n_ones - len(ones), replace=False)] = 1.0

    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    print(f"Initial: n={n0}, C5={best_c5:.8f}", flush=True)

    def remaining():
        return budget_s - (time.time() - start_time)

    # Phase 1: GD at initial resolution, moderate T
    if remaining() > 250:
        h1, c1 = gd_run(best_h, 2500, 40.0, 1.0, 0.10, 0.008, check_every=30, top_k=150)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        print(f"P1 GD: C5={best_c5:.8f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: SLSQP at initial resolution
    if n0 <= 420 and remaining() > 150:
        for T_val in [8.0, 2.0]:
            if remaining() < 30:
                break
            hx, cx = slsqp_run(best_h, T_val, maxiter=700)
            if cx < best_c5:
                best_c5, best_h = cx, hx.copy()
        print(f"P2 SLSQP: C5={best_c5:.8f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: Low-T GD for precise max optimization
    if remaining() > 120:
        h3, c3 = gd_run(best_h, 4000, 1.5, 0.03, 0.008, 0.0005, check_every=20, top_k=60)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
        print(f"P3 LowT: C5={best_c5:.8f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 4: Coordinate descent
    if remaining() > 80:
        h4, c4 = coord_descent(best_h, len(best_h), passes=3, eps_init=0.003)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
        print(f"P4 CoordDescent: C5={best_c5:.8f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 5: Try alternate resolutions
    n_cur = len(best_h)
    alt_ns = []
    for cand in [256, 384, 448, 512, 640]:
        if cand != n_cur:
            alt_ns.append(cand)
    for new_n in alt_ns[:3]:
        if remaining() < 80:
            break
        h_up = upsample_h(best_h, new_n)
        c_up = compute_c5(h_up)
        if remaining() > 60:
            h_g, c_g = gd_run(h_up, 2000, 20.0, 0.2, 0.08, 0.004, check_every=40, top_k=100)
            if c_g < c_up:
                h_up, c_up = h_g, c_g
        if new_n <= 420 and remaining() > 40:
            hx, cx = slsqp_run(h_up, 2.0, maxiter=400)
            if cx < c_up:
                h_up, c_up = hx, cx
        if remaining() > 30:
            h_cd, c_cd = coord_descent(h_up, new_n, passes=2, eps_init=0.003)
            if c_cd < c_up:
                h_up, c_up = h_cd, c_cd
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up.copy()
            print(f"  n={new_n}: C5={best_c5:.8f}", flush=True)
    print(f"P5 Resolutions: C5={best_c5:.8f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 6: Multi-start perturbation
    n_pts = len(best_h)
    if remaining() > 120:
        for scale in [0.015, 0.04, 0.08]:
            if remaining() < 30:
                break
            h_p = make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
            h_r, c_r = gd_run(h_p, 2000, 15.0, 0.1, 0.06, 0.003, check_every=50, top_k=100)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  Perturb {scale}: C5={best_c5:.8f}", flush=True)

    # Phase 7: Final ultra-low-T GD
    if remaining() > 40:
        n_pts = len(best_h)
        h_f, c_f = gd_run(best_h, 5000, 0.8, 0.005, 0.003, 0.0002, check_every=15, top_k=30)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
        print(f"P7 FinalGD: C5={best_c5:.8f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 8: Final SLSQP polish
    n_pts = len(best_h)
    if n_pts <= 420 and remaining() > 25:
        hx, cx = slsqp_run(best_h, 0.8, maxiter=800)
        if cx < best_c5:
            best_c5, best_h = cx, hx.copy()

    # Phase 9: Final coordinate descent
    if remaining() > 15:
        h_cd, c_cd = coord_descent(best_h, len(best_h), passes=2, eps_init=0.002)
        if c_cd < best_c5:
            best_c5, best_h = c_cd, h_cd.copy()

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.8f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
