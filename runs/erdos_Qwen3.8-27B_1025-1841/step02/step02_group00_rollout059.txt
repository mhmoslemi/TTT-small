def weighted_gradient(h, tau, top_m=50):
    n = len(h)
    dx = 2.0 / n
    overlap = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(overlap)

    top_w = weights[top_idx]
    top_w /= np.sum(top_w)

        # -h[k-lag] for k where 0<=k-lag<n
        if lag >= 0:

            # h[k-lag] = h[k+|lag|]: k in [0, n-1-|lag|]
            grad[:n-abs(lag)] -= w * h[abs(lag):]
    grad *= dx
    return m, grad

            # Or: keep going (momentum-like)

            tau = max(tau, 0.0001)
            alpha = 0.15 * (1.0 - frac * 0.8)
            top_m = max(3, int(15 * (1 - frac)) + 3)

            c5_now, grad = compute_grad_and_max(h_curr, tau, top_m)
            h_new = make_feasible(h_curr - alpha * grad, n_pts)
            c5_new = compute_c5(h_new)
            if c5_new < c5_now - 1e-15:
                h_curr = h_new
                c5_curr = c5_new
                stalled = 0
                if c5_curr < best_c5:
                    best_c5 = c5_curr
                    best_h = h_curr.copy()
            else:
                stalled += 1
                if stalled > 200:
                    break
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        # Phase C: local moves (only for manageable n)
        if time_left() > 30 and n_pts <= 1024:
            h_local = h_curr.copy()
            for sweep in range(2):
                if time_left() < 10:
                    break
                improved = False
                for j in range(n_pts):
                    if time_left() < 5:
                        break
                    for bv in (0.0, 1.0):
                        if abs(h_local[j] - bv) < 1e-9:
                            continue
                        trial = h_local.copy()
                        trial[j] = bv
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-14:
                            h_local = trial
                            best_c5 = c
                            best_h = h_local.copy()
                            improved = True
                            break
                    if h_local[j] != 0.0 and h_local[j] != 1.0:
                        trial = h_local.copy()
                        trial[j] = 1.0 - h_local[j]
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-14:
                            h_local = trial
                            best_c5 = c
                            best_h = h_local.copy()
                            improved = True
                if not improved:
                    break
            h_curr = h_local
            c5_curr = compute_c5(h_curr)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h_curr.copy()

    # 5. Random restart perturbations at best resolution
    if time_left() > 60:
        n_final = len(best_h)
        for restart in range(5):
            if time_left() < 30:
                break
            pert = rng.normal(0, 0.1, size=n_final)
            h_pert = np.clip(best_h + pert, 0.0, 1.0)
            h_pert = make_feasible(h_pert, n_final)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h_start = h_pert.copy()
            else:
                h_start = best_h.copy()
            # Short gradient descent from perturbation
            n_short = 300
            for it in range(n_short):
                if time_left() < 15:
                    break
                frac = it / n_short
                tau = 0.005 * (1 - frac) + 0.0005
                alpha = 0.2 * (1 - frac * 0.7)
                c5_now, grad = compute_grad_and_max(h_start, tau, 15)
                h_new = make_feasible(h_start - alpha * grad, n_final)
                c5_new = compute_c5(h_new)
                if c5_new < c5_now - 1e-15:
                    h_start = h_new
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_start.copy()

    # 6. Final polish
    if time_left() > 10:
        n_final = len(best_h)
        for it in range(300):
            if time_left() < 5:
                break
            alpha = 0.05 * (1.0 - it / 300.0) + 0.005
            c5_now, grad = compute_grad_and_max(best_h, 0.0005, 8)
            h_new = make_feasible(best_h - alpha * grad, n_final)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-15:
                best_h = h_new
                best_c5 = c5_new

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
