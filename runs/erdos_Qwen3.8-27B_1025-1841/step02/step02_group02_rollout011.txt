```python
def compute_corr_fft(h, n):
    N = 1
    while N < 2*n-1:
        N *= 2
    H = np.fft.rfft(h, N)
    Oh = np.fft.rfft(1-h, N)
    corr_circ = np.fft.irfft(H * np.conj(Oh), N)
    # This gives circular correlation. Need to convert to linear.
    # Circular corr at lag s = linear corr at lag s + wrap-around
    # For |s| < n, the circular corr equals linear corr if no wrap-around.
    # With zero-padding to size N >= 2n-1, the circular corr IS the linear corr.
    # Actually, with zero-padding:
    # h is padded to [h, 0, ..., 0] (total length N)
    # 1-h is padded to [1-h, 0, ..., 0] (total length N)
    # Circular correlation at lag s: sum_i h_pad[i]*(1-h)_pad[i+s]
    # For s >= 0: sum_{i=0}^{N-1-s} ... but with the padding, this equals
    #   sum_{i=0}^{n-1-s} h[i]*(1-h)[i+s] (for s < n) + 0 (for s >= n)
    # For s < 0 (i.e., N-s in circular): ...
    
    # Actually, the standard trick:
    # corr_full = ifft(fft(h, N) * conj(fft(1-h, N)))
    # This gives the circular cross-correlation.
    # With zero-padding to N >= 2n-1, the first n values (lags 0 to n-1) 
    # give the "forward" part and the last n-1 values give the "backward" part.
    # Specifically: corr_full[s] for s=0..n-1 gives corr at shift +s
    #             corr_full[N-s] for s=1..n-1 gives corr at shift -s
    
    # np.correlate(h, 1-h, 'full') gives:
    # index 0: shift -(n-1), index n-1: shift 0, index 2n-2: shift n-1
    # So result[k] corresponds to shift k-(n-1)
    
    # From circular correlation with zero-padding:
    # corr_circ[s] for s=0: shift 0
    # corr_circ[s] for s=1..n-1: shift +s
    # corr_circ[N-s] for s=1..n-1: shift -s (= shift -(n-1) to -(1))
    
    # So: full_corr[k] = 
    #   for k=0..n-1 (shift = -(n-1)..0): 
    #     shift = -(n-1)+k, i.e., shift from -(n-1) to 0
    #     if shift = 0: corr_circ[0]
    #     if shift = -s (s>0): corr_circ[N-s]
    #   for k=n..2n-2 (shift = 1..n-1):
    #     shift = k-(n-1)
    #     corr_circ[shift]
    
    # Let me just assemble:
    result = np.zeros(2*n-1)
    result[n-1] = corr_circ[0]  # shift 0
    for s in range(1, n):
        result[n-1+s] = corr_circ[s]  # shift +s
        result[n-1-s] = corr_circ[N-s]  # shift -s
    return result
```
