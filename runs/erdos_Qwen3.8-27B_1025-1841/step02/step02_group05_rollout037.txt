def coordinate_descent(h, n, time_limit, max_passes=100):
    dx = 2.0 / n
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    start = time.time()
    
    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            # Try h[j] = 0
            h[j] = 0.0
            c5_0 = compute_c5(h)
            # Try h[j] = 1
            h[j] = 1.0
            c5_1 = compute_c5(h)
            # Keep original if both are worse
            if c5_0 <= c5_1 and c5_0 < best_c5:
                h[j] = 0.0
                best_c5 = c5_0
                best_h = h.copy()
                improved = True
            elif c5_1 < c5_0 and c5_1 < best_c5:
                h[j] = 1.0
                best_c5 = c5_1
                best_h = h.copy()
                improved = True
            else:
                # Try keeping the value that gives lower c5
                if c5_0 <= c5_1:
                    h[j] = 0.0
                else:
                    h[j] = 1.0
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def coordinate_descent(h, n, time_limit, max_passes=500, start_time=None):
    if start_time is None:
        start_time = time.time()
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = corr.max()
    best_c5 = current_max * dx
    best_h = h.copy()
    
    for p in range(max_passes):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            old_val = h[j]
            current_max = corr.max()
            best_val = old_val
            best_corr = None
            best_max = current_max
            
            for new_val in (0.0, 1.0):
                if abs(new_val - old_val) < 1e-10:
                    continue
                delta = new_val - old_val
                # Apply
                k_start = n - 1 - j
                k_end_excl = min(n - 1 + j, 2 * n - 2 - j) + 1
                if k_start < k_end_excl:
                    h_idx = np.arange(0, k_end_excl - k_start)
                    corr[k_start:k_end_excl] += delta * (1.0 - h[h_idx])
                k_start2 = j
                k_end2 = j + n
                h_idx2 = np.arange(n - 1, -1, -1)
                corr[k_start2:k_end2] -= delta * h[h_idx2]
                
                new_max = corr.max()
                if new_max < best_max - 1e-14:
                    best_max = new_max
                    best_val = new_val
                    best_corr = corr.copy()
                
                # Undo
                if k_start < k_end_excl:
                    h_idx = np.arange(0, k_end_excl - k_start)
                    corr[k_start:k_end_excl] -= delta * (1.0 - h[h_idx])
                corr[k_start2:k_end2] += delta * h[h_idx2]
            
            if best_corr is not None:
                h[j] = best_val
                corr = best_corr
                if best_max * dx < best_c5 - 1e-14:
                    best_c5 = best_max * dx
                    best_h = h.copy()
                improved = True
        
        if not improved:
            break
    
    return best_h, best_c5

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

R[ℓ] = (number of positions where both x[i]=1 and x[i+ℓ]=1)

And the "1→0" count at lag ℓ is:
C[ℓ] = (number of 1s in overlap) - R[ℓ]

The number of 1s in the overlap at lag ℓ is at most min(N-|ℓ|, N/2) ≈ N/2 for small ℓ.

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
