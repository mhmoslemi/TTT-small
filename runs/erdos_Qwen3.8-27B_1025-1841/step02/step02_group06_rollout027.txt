import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr_fft(h):
    n = len(h)
    m = 1.0 - h
    total = 2 * n - 1
    fsize = 1
    while fsize < total:
        fsize *= 2
    H = np.fft.rfft(h, fsize)
    M = np.fft.rfft(m[::-1], fsize)
    result = np.fft.irfft(H * M, fsize)[:total]
    return result[::-1]


def compute_c5(h):
    n = len(h)
    return compute_corr_fft(h).max() * (2.0 / n)


def gradient_top_k(h, corr_vals, n, dx, topK):
    """Compute gradient focusing on top-K worst shifts."""
    top_idx = np.argsort(corr_vals)[-topK:]
    grad = np.zeros(n)
    for k in top_idx:
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= h[idx + d]
    return grad * dx / topK


def softmax_gradient(h, corr_vals, n, dx, tau, topM=0):
    """Compute softmax-weighted gradient."""
    if topM > 0 and topM < (2 * n - 1):
        top_idx = np.argpartition(corr_vals, -topM)[-topM:]
        gv = corr_vals[top_idx]
    else:
        top_idx = np.arange(2 * n - 1)
        gv = corr_vals
    max_gv = gv.max()
    shifted = np.clip(tau * (gv - max_gv), -500, 500)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p[i]
        if pk < 1e-10:
            continue
        k = top_idx[i]
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx


def optimize_softmax(h, n, tau, max_iters, time_limit, lr_init=0.15, topM=0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr_fft(h)
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(50, max_iters // 20)
    current_tau = tau

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr_fft(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        # Adaptively increase tau to focus on worst shifts
        current_tau = tau * (1.0 + it / max(1, max_iters // 3))
        grad = softmax_gradient(h, g, n, dx, current_tau, topM)

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        h = h_new

    return best_h, best_obj


def optimize_subgrad(h, n, max_iters, time_limit, lr_init=0.004, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    stall = 0
    patience = 300

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr_fft(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        grad = gradient_top_k(h, corr, n, dx, topK)
        cur_lr = lr / (1.0 + it * 0.003)
        h_new = project_to_feasible(h - cur_lr * grad, n)

        if stall > patience:
            lr *= 0.65
            stall = 0
            if lr < 1e-14:
                break
        h = h_new

    return best_h, best_c5


def optimize_averaged(h, n, max_iters, time_limit, lr_init=0.008, topK=8):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    h_sum = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    count = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr_fft(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()

        grad = gradient_top_k(h, corr, n, dx, topK)
        cur_lr = lr_init / (1.0 + it * 0.01)
        h_new = project_to_feasible(h - cur_lr * grad, n)
        h_sum += h_new
        count += 1

        if count % 500 == 0:
            h_avg = project_to_feasible(h_sum / count, n)
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg.copy()
        h = h_new

    if count > 100:
        h_avg = project_to_feasible(h_sum / count, n)
        c5_avg = compute_c5(h_avg)
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_avg.copy()

    return best_h, best_c5


def structured_perturb(h, n, rng):
    """Create structured perturbation (block-like changes)."""
    h_p = h.copy()
    n_blocks = rng.randint(3, 8)
    block_size = max(1, n // (n_blocks * 4))
    for _ in range(n_blocks):
        pos = rng.randint(0, n - block_size)
        delta = rng.uniform(-0.15, 0.15)
        h_p[pos:pos + block_size] += delta
    return project_to_feasible(h_p, n)
