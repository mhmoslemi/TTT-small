                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1

            if stall < 3000:
                K = 5
            elif stall < 8000:
                K = 3
            elif stall < 15000:
                K = 1
            else:
                K = 4

            top_k = np.argsort(corr)[-K:]
            grad = np.zeros(n)
            for ki in top_k:
                grad += subgrad_at(h, int(ki), n)
            grad /= K

            v = beta * v - lr * grad
            h = project_to_feasible(h + v, n)

            if stall > 20000:
                h = h + 0.005 * np.random.randn(n)
                h = project_to_feasible(h, n)
                lr *= 0.6
                v *= 0.3
                stall = 0

            if stall % 2000 == 0 and stall > 0:
                lr *= 0.85

            it += 1

        c5_chk, _ = compute_c5(h)
        if c5_chk < best_c5:
            best_c5 = c5_chk
            best_h = h.copy()

    # Phase 3: Binary projection + local search
    if rem() > 50:
        h_bin = binary_projection(best_h, n)
        c5_bin, _ = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        h_ls, c5_ls = local_search_binary(best_h, n, 20000, rem() - 30, t0)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 4: Higher resolution refinement
    if rem() > 80:
        n2 = 8000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        dx2 = 2.0 / n2
        best_c5_2 = best_c5
        best_h2 = h2.copy()
        lr2 = 0.5
        v2 = np.zeros(n2)
        beta2 = 0.6
        for it2 in range(60000):
            if rem() < 40:
                break
            corr2 = np.correlate(h2, 1.0 - h2, mode='full')
            c5_2 = corr2.max() * dx2
            if c5_2 < best_c5_2 - 1e-14:
                best_c5_2 = c5_2
                best_h2 = h2.copy()

            k_star = np.argmax(corr2)
            grad2 = subgrad_at(h2, k_star, n2)
            # Mix with second worst
            top2 = np.argsort(corr2)[-2:]
            grad2 += subgrad_at(h2, int(top2[0]), n2)
            grad2 += subgrad_at(h2, int(top2[1]), n2)
            grad2 /= 2.0

            v2 = beta2 * v2 - lr2 * grad2
            h2 = project_to_feasible(h2 + v2, n2)

        if best_c5_2 < best_c5:
            best_c5 = best_c5_2
            best_h = best_h2.copy()

    # Phase 5: Final subgradient polish on best_h
    if rem() > 10:
        nf = len(best_h)
        dx_f = 2.0 / nf
        h_f = best_h.copy()
        lr_f = 0.3
        v_f = np.zeros(nf)
        for itf in range(100000):
            if rem() < 3:
                break
            corr_f = np.correlate(h_f, 1.0 - h_f, mode='full')
            c5_f = corr_f.max() * dx_f
            if c5_f < best_c5 - 1e-14:
                best_c5 = c5_f
                best_h = h_f.copy()
            k_star = np.argmax(corr_f)
            grad_f = subgrad_at(h_f, k_star, nf)
            v_f = 0.5 * v_f - lr_f * grad_f
            h_f = project_to_feasible(h_f + v_f, nf)

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5, _ = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
