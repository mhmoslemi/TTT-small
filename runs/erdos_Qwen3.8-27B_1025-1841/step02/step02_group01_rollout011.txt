def soft_max_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = corr - mx
    soft = mx + np.log(np.sum(np.exp(T * shifted))) / T
    return soft * dx

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = corr - mx
    weights = T * np.exp(T * shifted) / np.sum(np.exp(T * shifted))
    # gradient of soft-max wrt corr
    # gradient of corr wrt h
    grad = np.zeros(2 * n - 1)
    grad += weights
    # Now multiply by Jacobian of corr wrt h
    # corr[m] = sum_i h[i] * (1 - h[i + shift(m)])
    # where shift(m) = m - (n-1)
    # dc_m/dh[j] = (1-h[j+s]) - h[j-s] (with bounds)
    # where s = m - (n-1)
    jac = np.zeros((2*n-1, n))
    for m in range(2*n-1):
        s = m - (n - 1)
        if s >= 0:
            # corr[m] = sum_{i=0}^{n-1-s} h[i+s] * (1-h[i])
            for i in range(n - s):
                j = i + s  # h[j] appears with coefficient (1-h[i])
                jac[m, j] += (1 - h[i])
                j2 = i  # h[j2] appears with coefficient -h[i+s]
                jac[m, j2] -= h[j]
        else:
            s2 = -s
            # corr[m] = sum_{i=s2}^{n-1} h[i] * (1-h[i-s2])
            for i in range(s2, n):
                j = i  # h[j] appears with coefficient (1-h[i-s2])
                jac[m, j] += (1 - h[i - s2])
                j2 = i - s2  # h[j2] appears with coefficient -h[i]
                jac[m, j2] -= h[j]
    result = jac.T.dot(grad)
    return result * dx

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _soft_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    sh = corr - mx
    exp_sh = np.exp(T * sh)
    w = T * exp_sh / np.sum(exp_sh)
    
    W = np.zeros(n)
    W[0] = w[n-1]
    if n > 1:
        W[1:] = w[n-2::-1] + w[n:]
    
    one_minus_h = 1.0 - h
    first_term = np.convolve(one_minus_h, W, mode='full')[:n]
    h_rev = h[::-1]
    second_term = np.cumsum(W * h_rev)[::-1]
    
    return (first_term - second_term) * dx

def _soft_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    sh = corr - mx
    return (mx + np.log(np.sum(np.exp(T * sh))) / T) * dx

def _constraint(h, n):
    return np.sum(h) - n / 2.0

def coord_descent(h, n, max_rounds=10, tol=1e-12):
    """Coordinate descent with re-projection."""
    best = h.copy()
    best_c5 = compute_c5(best)
    for round_idx in range(max_rounds):
        improved = False
        for j in range(n):
            if time.time() > _TIME_LIMIT:
                return best, best_c5
            old = best[j]
            # Try a set of candidate values
            candidates = [0.0, 1.0]
            for delta in [0.05, 0.02, 0.01, 0.005, 0.001]:
                candidates.append(old + delta)
                candidates.append(old - delta)
            candidates = [c for c in candidates if 0.0 <= c <= 1.0]
            best_candidate = old
            best_c = best_c5
            for c in candidates:
                trial = best.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c - tol:
                    best_c = c5
                    best_candidate = c
            if best_candidate != old:
                best[j] = best_candidate
                best = make_feasible(best, n)
                best_c5 = compute_c5(best)
                improved = True
        if not improved:
            break
    return best, best_c5

_TIME_LIMIT = 0

    _TIME_LIMIT = start_time + budget_s - 5
    
    rng = np.random.RandomState(seed)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: Optimize at original resolution
    n1 = len(h0)
    _TIME_LIMIT = start_time + budget_s * 0.15
    T1 = 80.0
    bounds = [(0.0, 1.0)] * n1
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n1/2.0, 'jac': lambda x: np.ones(n1)}]
    
    try:
        res = minimize(lambda x: _soft_obj(x, T1), h0, jac=lambda x: _soft_obj_grad(x, T1),
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-12})
        if compute_c5(res.x) < best_c5:
            best_c5 = compute_c5(res.x)
            best_h = res.x.copy()
    except Exception:

    # Phase 2: Upsample to higher resolution and optimize
    n2 = 768

    best_c5_2 = compute_c5(h2)
    if best_c5_2 < best_c5:

        best_h = h2.copy()
    
    T2 = 60.0

Actually, you know what, let me try a completely different approach for the local optimization: **direct gradient descent on the max function using subgradient**.

The idea:
1. Compute corr = np.correlate(h, 1-h, 'full')
2. Find all k where corr[k] is within ε of the max (active set)
3. Compute the gradient of the average of active corr values
4. Take a step in the negative gradient direction
5. Project to feasible set (clip to [0,1], adjust sum)
6. Repeat with decreasing step size

This avoids the soft-max approximation and directly targets the max.

The gradient of corr[k] wrt h[j] (using my formula):
d(corr[k])/d(h[j]) = (1-h[j-s])*I[j>=s] - h[j+s]*I[j<=n-1-s] where s = |k-(n-1)|

For the average of active set A:
grad_A[j] = (1/|A|) * sum_{k in A} [(1-h[j-s_k])*I[j>=s_k] - h[j+s_k]*I[j<=n-1-s_k]]

This can be computed efficiently using the W vector approach but with a different W (only active shifts).

Let me implement this:

```python
def subgradient_step(h, n, step_size, active_frac=0.05):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    # Active set: corr values within some fraction of max
    threshold = mx - active_frac * (mx - np.min(corr))
    active = corr >= threshold
    
    # Build W from active set
    w = np.zeros(2*n-1)
    w[active] = 1.0 / np.sum(active)
    
    W = np.zeros(n)
    W[0] = w[n-1]
    if n > 1:
        W[1:] = w[n-2::-1] + w[n:]
    
    one_minus_h = 1.0 - h
    first_term = np.convolve(one_minus_h, W, mode='full')[:n]
    h_rev = h[::-1]
    second_term = np.cumsum(W * h_rev)[::-1]
    
    grad = (first_term - second_term) * dx
    
    # Gradient descent step
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _grad_softmax(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    sh = corr - mx
    exp_sh = np.exp(T * sh)
    w = T * exp_sh / np.sum(exp_sh)
    
    W = np.zeros(n)
    W[0] = w[n - 1]
    if n > 1:
        W[1:] = w[n-2::-1] + w[n:]
    
    one_minus_h = 1.0 - h
    first_term = np.convolve(one_minus_h, W, mode='full')[:n]
    h_rev = h[::-1]
    second_term = np.cumsum(W * h_rev)[::-1]
    
    return (first_term - second_term) * dx

def _softmax_obj(h, T):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    sh = corr - mx
    return (mx + np.log(np.sum(np.exp(T * sh))) / T) * dx

def _constraint_fun(h):
    return np.sum(h) - _CUR_N / 2.0

def _constraint_jac(h):
    return np.ones(_CUR_N)

def _subgrad_step(h, n, step_size):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    mn = np.min(corr)
    if mx - mn < 1e-15:
        return h
    threshold = mx - 0.05 * (mx - mn)
    active = (corr >= threshold).astype(float)
    active /= np.sum(active)
    
    w = active
    W = np.zeros(n)
    W[0] = w[n - 1]
    if n > 1:
        W[1:] = w[n-2::-1] + w[n:]
    
    one_minus_h = 1.0 - h
    first_term = np.convolve(one_minus_h, W, mode='full')[:n]
    h_rev = h[::-1]
    second_term = np.cumsum(W * h_rev)[::-1]
    
    grad = (first_term - second_term) * dx
    h_new = h - step_size * grad
    return make_feasible(h_new, n)

def _coord_descent(h, n, deadline, max_rounds=10):
    best = h.copy()
    best_c5 = compute_c5(best)
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            if time.time() > deadline:
                return best, best_c5
            old = best[j]
            # Generate candidates
            cands = [0.0, 1.0, 0.5]
            for d in [0.05, 0.02, 0.01, 0.005]:
                cands.append(old + d)
                cands.append(old - d)
            cands = [c for c in cands if 0.0 <= c <= 1.0 and c != old]
            
            best_c = best_c5
            best_j = old
            for c in cands:
                trial = best.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                c5 = compute_c5(trial)
                if c5 < best_c - 1e-14:
                    best_c = c5
                    best_j = c
            if best_j != old:
                best[j] = best_j
                best = make_feasible(best, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return best, best_c5

_CUR_N = 0

def run(seed=42, budget_s=1300, **kwargs):
    global _CUR_N
    start_time = time.time()
    deadline = start_time + budget_s - 5
    rng = np.random.RandomState(seed)
    
    # Get initial values
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):

    if h0 is None:
        n_init = 320
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        h0 = make_feasible(h0, n_init)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: SLSQP at initial resolution
    _CUR_N = len(h0)
    T = 80.0
    bounds = [(0.0, 1.0)] * _CUR_N
    cons = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
    dl1 = start_time + budget_s * 0.12
    
    try:
        res = minimize(_softmax_obj, h0, args=(T,), jac=_grad_softmax,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-13})
        c5 = compute_c5(res.x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = res.x.copy()
    except Exception:
        pass
    
    # Phase 2: Upsample to n=768 and optimize
    N = 768
    _CUR_N = N
    h2 = upsample_h(best_h, N)
    c5_2 = compute_c5(h2)
    if c5_2 < best_c5:
        best_c5 = c5_2
        best_h = h2.copy()
    
    T2 = 50.0
    bounds2 = [(0.0, 1.0)] * N
    cons2 = [{'type': 'eq', 'fun': _constraint_fun, 'jac': _constraint_jac}]
    dl2 = start_time + budget_s * 0.40
    
    try:
        res2 = minimize(_softmax_obj, h2, args=(T2,), jac=_grad_softmax,
                        method='SLSQP', bounds=bounds2, constraints=cons2,
                        options={'maxiter': 250, 'ftol': 1e-13})
        c5 = compute_c5(res2.x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = res2.x.copy()
    except Exception:
        pass
    
    # Phase 3: Subgradient descent
    dl3 = start_time + budget_s * 0.55
    step = 0.02
    for it in range(200):
        if time.time() > dl3:
            break
        h_sg = _subgrad_step(best_h, N, step)
        c5 = compute_c5(h_sg)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sg
        else:
            step *= 0.95
    # Decay step
    step = 0.005
    for it in range(200):
        if time.time() > dl3:
            break
        h_sg = _subgrad_step(best_h, N, step)
        c5 = compute_c5(h_sg)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sg
        else:
            step *= 0.97
    
    # Phase 4: Coordinate descent
    dl4 = start_time + budget_s * 0.70
    best_h, best_c5 = _coord_descent(best_h, N, dl4, max_rounds=5)
    
    # Phase 5: Perturb + SLSQP + coord descent (multiple times)
    dl5 = start_time + budget_s * 0.88
    for attempt in range(8):
        if time.time() > dl5:
            break
        noise_scale = 0.015 * (1.0 - attempt / 10.0)
        h_pert = best_h + rng.randn(N) * noise_scale

                           method='SLSQP', bounds=bounds2, constraints=cons2,
                           options={'maxiter': 150, 'ftol': 1e-13})
            c5 = compute_c5(res3.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res3.x.copy()
                # Quick coord descent
                dl_cd = time.time() + 15
                best_h, best_c5 = _coord_descent(best_h, N, dl_cd, max_rounds=2)
        except Exception:
            pass
    
    # Phase 6: Final coordinate descent
    dl6 = deadline
    best_h, best_c5 = _coord_descent(best_h, N, dl6, max_rounds=5)
    
    # Final validation
    best_h = make_feasible(best_h, N)
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, N
