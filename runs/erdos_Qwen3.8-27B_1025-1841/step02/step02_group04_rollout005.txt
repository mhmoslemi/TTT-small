def binary_project(h, n):
    """Push h toward binary {0,1} while maintaining sum constraint."""
    target = n // 2
    h = h.copy()
    # Sort by value, set top n/2 to 1, rest to 0
    indices = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[indices[:target]] = 1.0
    return h_bin

# SLSQP objective
_OPT = {}

def _slsqp_obj(h):
    T = _OPT['T']
    n = _OPT['n']
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_cons(h):
    return np.sum(h) - _OPT['n'] / 2.0

def _slsqp_jac(h):
    return np.ones(_OPT['n'])

def slsqp_optimize(h, T=5.0, maxiter=500):
    n = len(h)
    _OPT['T'] = T
    _OPT['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            return hx
    except Exception:
        pass
    return h

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    tgt = n * 0.5
    for _ in range(50):
        d = tgt - h.sum()
        if abs(d) < 1e-15: break
        if d > 0: m = h < 1.0
        else: m = h > 0.0
        c = m.sum()
        if c == 0: break
        h[m] += d / c
        np.clip(h, 0, 1, out=h)
    return h

def _interp(h, nn):
    on = len(h)
    if nn == on: return h.copy()
    xo = np.linspace(0, 2, on)
    xn = np.linspace(0, 2, nn)
    return _feasible(np.interp(xn, xo, h), nn)

def _grad_topk(h, K=25, T=5.0):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    M = len(c)
    mx = c.max()
    w = np.exp(T * (c - mx))
    s = w.sum()
    if s < 1e-300:
        w = np.eye(1, M, c.argmax())[0]
    else:
        w /= s
    # top K
    idx = np.argsort(c)[-K:]
    wk = w[idx]
    wk /= wk.sum()
    g = np.zeros(n)
    for i, wi in zip(idx, wk):
        if wi < 1e-14: continue
        sh = i - (n - 1)
        if sh >= 0:
            g[:n-sh] += wi * (1.0 - h[sh:])
            g[sh:]   -= wi * h[:n-sh]
        else:
            sp = -sh
            g[sp:]     += wi * (1.0 - h[:n-sp])
            g[:n-sp]   -= wi * h[sp:]
    return g * dx

def _gd(h, niter, K, T0, T1, lr0, lr1, dt=50, tlimit=None, t0=None):
    n = len(h)
    h = h.astype(float).copy()
    bc, bh = _c5(h), h.copy()
    for it in range(niter):
        if tlimit is not None and time.time() - t0 > tlimit + (t0 - t0): 
            pass
        if tlimit is not None and time.time() - t0 > t0 + tlimit:
            break
        f = it / max(niter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        g = _grad_topk(h, K, T)
        h = np.clip(h - lr * g, 0, 1)
        h = _feasible(h, n)
        if (it + 1) % dt == 0:
            c = _c5(h)
            if c < bc: bc, bh = c, h.copy()
    c = _c5(h)
    if c < bc: bc, bh = c, h.copy()
    return bh, bc

import numpy as np
from scipy.optimize import minimize
import time

def _corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def _c5(h):
    n = len(h)
    return float(np.max(_corr(h)) * 2.0 / n)

def _feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    tgt = n * 0.5
    for _ in range(60):
        d = tgt - h.sum()
        if abs(d) < 1e-15:
            break
        if d > 0:
            m = h < 1.0
        else:
            m = h > 0.0
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp_up(h, nn):
    on = len(h)
    if nn == on:
        return h.copy()
    xo = np.linspace(0.0, 2.0, on)
    xn = np.linspace(0.0, 2.0, nn)
    return _feasible(np.interp(xn, xo, h), nn)

def _grad_topk(h, K, T):
    n = len(h)
    dx = 2.0 / n
    c = _corr(h)
    mx = c.max()
    ex = np.exp(T * (c - mx))
    tot = ex.sum()
    if tot < 1e-300:
        w = np.zeros(len(c))
        w[np.argmax(c)] = 1.0
    else:
        w = ex / tot
    idx = np.argsort(c)[-K:]
    wk = w[idx].copy()
    ws = wk.sum()
    if ws < 1e-15:
        return np.zeros(n)
    wk /= ws
    g = np.zeros(n)
    for ii, wi in zip(idx, wk):
        if wi < 1e-14:
            continue
        sh = ii - (n - 1)
        if sh >= 0:
            g[:n - sh] += wi * (1.0 - h[sh:])
            g[sh:] -= wi * h[:n - sh]
        else:
            sp = -sh
            g[sp:] += wi * (1.0 - h[:n - sp])
            g[:n - sp] -= wi * h[sp:]
    return g * dx

def _gd(h, niter, K, T0, T1, lr0, lr1, ck, tlimit, t0):
    n = len(h)
    h = h.astype(float).copy()
    bc, bh = _c5(h), h.copy()
    for it in range(niter):
        if time.time() - t0 > t0 + tlimit:
            break
        f = it / max(niter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lr = lr0 * (lr1 / lr0) ** f
        g = _grad_topk(h, K, T)
        h = np.clip(h - lr * g, 0.0, 1.0)
        h = _feasible(h, n)
        if (it + 1) % ck == 0:
            cv = _c5(h)
            if cv < bc:
                bc, bh = cv, h.copy()
    cv = _c5(h)
    if cv < bc:
        bc, bh = cv, h.copy()
    return bh, bc

_opt_p = {'T': 5.0, 'n': 320}

def _soft_obj(h):
    n = _opt_p['n']
    dx = 2.0 / n
    c = _corr(h)
    mc = c.max()
    sh = c - mc
    return (mc + np.log(np.sum(np.exp(_opt_p['T'] * sh))) / _opt_p['T']) * dx

def _eq_fun(h):
    return np.sum(h) - _opt_p['n'] * 0.5

def _eq_jac(h):
    return np.ones(_opt_p['n'])

def _slsqp(h, T, maxit):
    n = len(h)
    _opt_p['T'] = T
    _opt_p['n'] = n
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    try:
        r = minimize(_soft_obj, h, method='SLSQP', bounds=bnds,
                     constraints=cons, options={'maxiter': maxit, 'ftol': 1e-15})
        if r.x is not None:
            return _feasible(r.x, n)
    except Exception:
        pass
    return h

def _binary_local(h, n, ntrials, seed):
    rng = np.random.RandomState(seed)
    bc, bh = _c5(h), h.copy()
    tgt = n // 2
    for _ in range(ntrials):
        i = rng.randint(n)
        h2 = h.copy()
        h2[i] = 1.0 - h2[i]
        h2 = _feasible(h2, n)
        cv = _c5(h2)
        if cv < bc:
            bc, bh = cv, h2.copy()
    return bh, bc

def _swap_local(h, n, ntrials, seed):
    rng = np.random.RandomState(seed)
    bc, bh = _c5(h), h.copy()
    for _ in range(ntrials):
        i, j = rng.choice(n, 2, replace=False)
        h2 = h.copy()
        h2[i], h2[j] = h2[j], h2[i]
        cv = _c5(h2)
        if cv < bc:
            bc, bh = cv, h2.copy()
    return bh, bc

    def tl():
        return budget_s - (time.time() - t0)
    rng = np.random.RandomState(seed)

    except Exception:

        elif len(o) < 160:
            h0[rng.choice(z, 160 - len(o), replace=False)] = 1.0

    n0 = len(h0)
    best_c5 = _c5(h0)
    best_h = h0.copy()
    print(f"Init: n={n0} C5={best_c5:.6f}", flush=True)

    # Phase 1: GD at n0
    if tl() > 100:

        if c < best_c5:
            best_c5, best_h = c, h
        print(f"GD1 n={n0} C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 2: SLSQP chain at n0
    if n0 <= 512 and tl() > 80:
        for T in [10.0, 5.0, 2.0, 1.0, 0.5]:
            if tl() < 25:

            h = _slsqp(best_h, T, 500)
            c = _c5(h)
            if c < best_c5:
                best_c5, best_h = c, h
        print(f"SLSQP1 n={n0} C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 3: Binary local search at n0

        print(f"n640 C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 6: Multi-start perturbations at best resolution
    np_pts = len(best_h)

    # Phase 8: Final fine GD

    return best_h, best_c5, len(best_h)
