import time

        return 1.0 - 2.0 * h
    a = abs(s)
    if a >= n:
        return np.zeros(n)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr0 * (1.0 - 0.5 * frac)
        
        # Nesterov: evaluate gradient at h - v
        h_eval = make_feasible(h - v, n)
        grad = full_smooth_grad(h_eval, T)
        
        v = momentum * v - lr * grad
        h_new = make_feasible(h + v, n)
        
        # Check improvement
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        
        h = h_new
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def corrected_subgrad(h, s):
    n = len(h)
    if s == 0:
        return 1.0 - 2.0 * h
    a = abs(s)
    if a >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if s > 0:
        grad[0:n-a] = 1.0 - h[a:n]
        grad[a:n] += -h[0:n-a]
    else:
        grad[a:n] = 1.0 - h[0:n-a]
        grad[0:n-a] += -h[a:n]
    return grad

def full_smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    shifted = corr - c_max
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        idx = int(np.argmax(corr))
        s = idx - (n - 1)
        return corrected_subgrad(h, s) * dx
    w = exp_s / total
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        wk = w[k]
        if wk > 1e-14:
            s = k - (n - 1)
            grad += wk * corrected_subgrad(h, s)
    return grad * dx

def nesterov_gd(h, n_iter, T_start, T_end, lr0, momentum=0.85, check_every=30):
    n = len(h)
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr0 * (1.0 - 0.4 * frac)
        h_eval = make_feasible(h - v, n)
        grad = full_smooth_grad(h_eval, T)
        v = momentum * v - lr * grad
        h = make_feasible(h + v, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def coordinate_refine(h, n_sweeps=3):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for sweep in range(n_sweeps):
        order = np.random.permutation(n)
        for i in order:
            corr = np.correlate(h, 1.0 - h, mode='full')
            s_max = np.argmax(corr)
            best_val = h[i]
            best_obj = float(np.max(corr))
            for test_v in [0.0, 1.0, h[i] - 0.05, h[i] + 0.05, 0.25, 0.5, 0.75]:
                if test_v < 0 or test_v > 1:
                    continue
                h[i] = test_v
                diff = n / 2.0 - np.sum(h)
                if abs(diff) > 1e-12:
                    others = np.ones(n, dtype=bool)
                    others[i] = False
                    h[others] += diff / (n - 1)
                    h = np.clip(h, 0.0, 1.0)
                    h[i] = test_v
                    diff2 = n / 2.0 - np.sum(h)
                    if diff2 > 0:
                        m = h < 1.0
                    else:
                        m = h > 0.0
                    if np.sum(m) > 0:
                        h[m] += diff2 / np.sum(m)
                        h = np.clip(h, 0.0, 1.0)
                c_new = np.max(np.correlate(h, 1.0 - h, mode='full'))
                if c_new < best_obj:
                    best_obj = c_new
                    best_val = test_v
                    h[i] = test_v
                    h = make_feasible(h, n)
            h[i] = best_val
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
    h = make_feasible(h, n)
    return h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _slsqp_obj(h):
    T = _OPT_PARAMS['T']
    n = _OPT_PARAMS['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_cons(h):
    return np.sum(h) - _OPT_PARAMS['n'] / 2.0

def _slsqp_jac(h):
    return np.ones(_OPT_PARAMS['n'])

_OPT_PARAMS = {}

        zeros = np.where(h0 == 0)[0]
        if len(ones) > n_ones:
            h0[rng.choice(ones, len(ones) - n_ones, replace=False)] = 0.0
        elif len(ones) < n_ones:
            h0[rng.choice(zeros, n_ones - len(ones), replace=False)] = 1.0

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    n0 = len(best_h)
    print(f"Initial: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Nesterov GD with corrected gradient
    if elapsed() < budget_s - 120:
        h1, c1 = nesterov_gd(best_h, 6000, T_start=60.0, T_end=1.0, lr0=2.0, momentum=0.85, check_every=50)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        print(f"Nesterov1: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 2: Second Nesterov pass with different params
    if elapsed() < budget_s - 100:
        h2, c2 = nesterov_gd(best_h, 4000, T_start=20.0, T_end=0.5, lr0=1.0, momentum=0.9, check_every=50)
        if c2 < best_c5:

        print(f"Nesterov2: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 3: Coordinate refinement
    if elapsed() < budget_s - 80:

            if c_r < best_c5:
