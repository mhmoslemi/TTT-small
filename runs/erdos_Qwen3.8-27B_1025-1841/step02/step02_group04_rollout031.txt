import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(np.abs(h - 0.5))
        h[idx] += diff
    return np.clip(h, 0.0, 1.0)

def softmax_gradient(h, T, top_k=50):
    """Compute gradient of soft-max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return _subgrad_single(h, k_max - (n - 1)) * dx
    weights = exp_vals / total
    # Only use top contributions
    sorted_idx = np.argsort(weights)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in sorted_idx:
        w = weights[idx]
        if w < 1e-10:
            break
        shift = idx - (n - 1)
        grad += w * _subgrad_single(h, shift)
    return grad * dx

def _subgrad_single(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def nesterov_gd(h, n_iter=5000, T_start=30.0, T_end=0.5, lr=0.1, momentum=0.9, check_every=30):
    """Nesterov accelerated gradient descent."""
    n = len(h)
    h_curr = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h_curr)
    best_h = h_curr.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        
        # Nesterov lookahead
        h_look = h_curr + momentum * v
        h_look = np.clip(h_look, 0.0, 1.0)
        
        grad = softmax_gradient(h_look, T)
        
        # Update velocity
        v = momentum * v - lr_t * grad
        h_curr = h_curr + v
        h_curr = make_feasible(h_curr, n)
        
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h_curr)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_curr.copy()
    
    c5 = compute_c5(h_curr)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_curr.copy()
    
    return best_h, best_c5

def thue_morse(n):
    """Generate Thue-Morse sequence of length n."""
    t = [0]
    for _ in range(int(np.ceil(np.log2(n)))):
        t = t + [1 - x for x in t]
    return np.array(t[:n], dtype=float)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_single(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def softmax_grad(h, T, top_k=60):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(T * shifted)
    total = np.sum(exp_vals)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return _subgrad_single(h, k_max - (n - 1)) * dx
    weights = exp_vals / total
    sorted_idx = np.argsort(weights)[::-1][:top_k]
    grad = np.zeros(n)
    for idx in sorted_idx:
        w = weights[idx]
        if w < 1e-12:
            break
        shift = idx - (n - 1)
        grad += w * _subgrad_single(h, shift)
    return grad * dx

def fista(h, n_iter=8000, T_start=40.0, T_end=0.1, lr=0.15, check_every=50):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    t_k = 1.0
    best_c5 = compute_c5(x)
    best_h = x.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.4 * frac)
        grad = softmax_grad(z, T)
        x_new = project_c(z - lr_t * grad, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = x_new + (t_k - 1.0) / t_new * (x_new - x)
        z = np.clip(z, -1.0, 2.0)
        x = x_new
        t_k = t_new
        if (it + 1) % check_every == 0:
            c5 = compute_c5(x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.copy()
    c5 = compute_c5(x)
    if c5 < best_c5:
        best_c5 = c5
        best_h = x.copy()
    return best_h, best_c5

def thue_morse(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_c(h_new, new_n)

_OB = {}

def _slsqp_obj(h):
    T = _OB['T']
    n = _OB['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_con(h):
    return np.sum(h) - _OB['n'] / 2.0

def _slsqp_jac(h):
    return np.ones(_OB['n'])

def slsqp_polish(h, T=5.0, maxiter=800):
    n = len(h)
    _OB['T'] = T
    _OB['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = project_c(res.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h

from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=60):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter=5000, T0=40.0, T1=0.1, lr=0.15, check=50):

        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_c(h, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _subgrad_one(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += 1.0 - h[:n - s]
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += 1.0 - h[s:]
        g[s:] -= h[:n - s]
    return g

def sm_grad(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exps = np.exp(T * (corr - mc))
    tot = np.sum(exps)
    if tot < 1e-300:
        return _subgrad_one(h, int(np.argmax(corr)) - (n - 1)) * dx
    w = exps / tot
    idx = np.argsort(w)[::-1][:top_k]
    g = np.zeros(n)
    for i in idx:
        wi = w[i]
        if wi < 1e-12:
            break
        g += wi * _subgrad_one(h, i - (n - 1))
    return g * dx

def fista(h, n_iter=5000, T0=40.0, T1=0.1, lr=0.15, check=40):
    n = len(h)
    x = h.astype(float).copy()
    z = x.copy()
    tk = 1.0
    bc = compute_c5(x)
    bh = x.copy()
    for it in range(n_iter):
        f = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** f
        lrt = lr * (1.0 - 0.4 * f)
        g = sm_grad(z, T)
        xn = project_c(z - lrt * g, n)
        tn = (1.0 + np.sqrt(1.0 + 4.0 * tk * tk)) / 2.0
        z = xn + (tk - 1.0) / tn * (xn - x)
        np.clip(z, -1.0, 2.0, out=z)
        x = xn
        tk = tn
        if (it + 1) % check == 0:
            c = compute_c5(x)
            if c < bc:
                bc, bh = c, x.copy()
    c = compute_c5(x)
    if c < bc:
        bc, bh = c, x.copy()
    return bh, bc

def tm_seq(n):
    t = np.array([0])
    while len(t) < n:
        t = np.concatenate([t, 1 - t])
    return t[:n].astype(float)

def upsample(h, nn):
    if nn == len(h):
        return h.copy()
    ox = np.linspace(0, 2, len(h))
    nx = np.linspace(0, 2, nn)
    return project_c(np.interp(nx, ox, h), nn)

_OB = {}

def _sobj(h):
    T = _OB['T']
    n = _OB['n']
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(c)
    return (mc + np.log(np.sum(np.exp(T * (c - mc)))) / T) * dx

def _scon(h):
    return np.sum(h) - _OB['n'] / 2.0

def _sjac(h):
    return np.ones(_OB['n'])

def slsqp(h, T=5.0, mi=600):
    n = len(h)
    _OB['T'], _OB['n'] = T, n
    try:
        r = minimize(_sobj, h, method='SLSQP',
                     bounds=[(0.0, 1.0)] * n,
                     constraints=[{'type': 'eq', 'fun': _scon, 'jac': _sjac}],
                     options={'maxiter': mi, 'ftol': 1e-14})
        if r.x is not None:
            hx = project_c(r.x, n)
            if compute_c5(hx) < compute_c5(h):
                return hx
    except Exception:
        pass
    return h
