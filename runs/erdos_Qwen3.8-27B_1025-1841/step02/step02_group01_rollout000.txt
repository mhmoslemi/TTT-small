where A_k and B_k depend on the other h values. So corr[k] is linear in v!

Wait: corr[k] = sum_i h[i] * (1-h[i+k])
If we change h[j] to v:

- Terms where i+k-(n-1)=j (i.e., the second factor): h[i]*(1-v)

This is piecewise linear in v (actually just linear since it's a sum of products where only one factor is v).

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_obj(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(t * shifted))) / t
    return soft * dx

def softmax_obj_grad(h, n, t):
    dx = 2.0 / n
    ones = 1.0 - h
    corr = np.correlate(h, ones, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_vals = np.exp(t * shifted)
    sum_exp = np.sum(exp_vals)
    weights = exp_vals / sum_exp
    obj = (mc + np.log(sum_exp) / t) * dx
    # Gradient via correlation structure
    # corr[k] = sum_n h[n+k]*(1-h[n]) for valid indices
    # d(corr[k])/dh[j] = (1-h[j-k])*I[0<=j-k<n] - h[j+k]*I[0<=j+k<n]
    # grad[j] = dx * sum_k weights[k] * [(1-h[j-k])*I - h[j+k]*I]
    # Use convolution for efficiency:
    w = weights
    # term1[j] = sum_k w[k]*(1-h[j-k]) for valid j-k
    # term2[j] = sum_k w[k]*h[j+k] for valid j+k
    # term1 = correlate(1-h, w_rev, ...) or use full and extract
    # Simpler: use np.correlate
    # For term1: sum_k w[k]*(1-h[j-k]) = sum_m (1-h[m])*w[j-m] (where j-m in [0,2n-2])
    # This is like convolve(1-h, w) at position j offset
    full1 = np.correlate(1.0 - h, w, mode='full')  # length n + (2n-1) - 1 = 3n-2
    # full1[k] = sum_m (1-h[m+k]) * w[m]
    # We want term1[j] = sum_m (1-h[m]) * w[j-m] for valid m, j-m
    # Let l = j-m: term1[j] = sum_l (1-h[j-l]) * w[l]
    # full1[j] = sum_m (1-h[m+j]) * w[m] -- not quite
    # Let me just do it directly for correctness (n <= 700, so O(n*2n) is fine)
    grad = np.zeros(n)
    for j in range(n):
        s = 0.0
        # (1-h[j-k]) valid when 0 <= j-k < n => j-n+1 <= k <= j
        k_lo = max(0, j - n + 1)
        k_hi = min(2 * n - 2, j)
        if k_lo <= k_hi:
            idx = j - np.arange(k_hi - k_lo + 1)
            s += np.dot(w[k_lo:k_hi + 1], 1.0 - h[idx])
        # -h[j+k] valid when 0 <= j+k < n => -j <= k <= n-1-j
        k_lo2 = max(0, -j)
        k_hi2 = min(2 * n - 2, n - 1 - j)
        if k_lo2 <= k_hi2:
            idx2 = j + np.arange(k_hi2 - k_lo2 + 1)
            s -= np.dot(w[k_lo2:k_hi2 + 1], h[idx2])
        grad[j] = dx * s
    return obj, grad

def slsqp_optimize(h_start, n, t, maxiter, budget_check):
    def obj(x):
        return softmax_obj(x, n, t)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12,
                                                   'disp': False})
        if res.x is not None and np.all(res.x >= -1e-9) and np.all(res.x <= 1.0 + 1e-9):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h_start

def coord_descent_1d(h, n, max_rounds, time_limit):
