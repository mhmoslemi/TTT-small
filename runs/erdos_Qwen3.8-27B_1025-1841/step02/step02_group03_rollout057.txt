import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = h.astype(float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    ox = np.linspace(0.0, 1.0, old_n)
    nx = np.linspace(0.0, 1.0, new_n)
    return make_feasible(np.interp(nx, ox, h), new_n)

def binarize(h, n):
    h = np.array(h, dtype=float)
    target_ones = n // 2
    threshold = np.sort(h)[n - target_ones - 1]
    b = (h >= threshold).astype(float)
    n_ones = int(np.sum(b))
    if n_ones > target_ones:
        one_idx = np.where(b == 1)[0]
        excess = n_ones - target_ones
        order = np.argsort(h[one_idx])[:excess]
        b[one_idx[order]] = 0.0
    elif n_ones < target_ones:
        zero_idx = np.where(b == 0)[0]
        deficit = target_ones - n_ones
        order = np.argsort(-h[zero_idx])[:deficit]
        b[zero_idx[order]] = 1.0
    return b

    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
