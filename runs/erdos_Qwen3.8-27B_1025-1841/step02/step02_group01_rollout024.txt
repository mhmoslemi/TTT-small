            fi = 0.0
            if 0 <= i + k_star < n:
                fi += h[i + k_star]
            if 0 <= i - k_star < n:
                fi -= (1.0 - h[i - k_star])
            f_vals[idx] = fi
        
        g_vals = np.zeros(len(zeros))
        for idx, j in enumerate(zeros):
            gj = 0.0
            if 0 <= j + k_star < n:
                gj += h[j + k_star]
            if 0 <= j - k_star < n:
                gj -= (1.0 - h[j - k_star])
            g_vals[idx] = gj
        
        best_i_idx = int(np.argmin(f_vals))
        best_j_idx = int(np.argmax(g_vals))
        best_delta = f_vals[best_i_idx] - g_vals[best_j_idx]
        
        if best_delta < -1e-10:
            i = ones[best_i_idx]
            j = zeros[best_j_idx]
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            if new_c5 < best_c5 - 1e-12:
                best_c5 = new_c5
                stagnant = 0
            else:
                h[i] = 1.0
                h[j] = 0.0
                stagnant += 1
        else:
            stagnant += 1
            if rng.random() < 0.15:
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    stagnant = 0
                else:
                    h[i] = 1.0
                    h[j] = 0.0
    
    return h, best_c5

def multi_lag_binary_search(h, n, rng, max_sweeps=100, time_budget=60):
    """Try improving multiple lags, not just the max."""
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_budget:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_lags = np.argsort(corr)[::-1][:3]
        improved = False
        
        for k_star in top_lags:
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if len(ones) == 0 or len(zeros) == 0:
                break
            
            f_vals = np.zeros(len(ones))
            for idx, i in enumerate(ones):
                fi = 0.0
                if 0 <= i + k_star < n:
                    fi += h[i + k_star]
                if 0 <= i - k_star < n:
                    fi -= (1.0 - h[i - k_star])
                f_vals[idx] = fi
            
            g_vals = np.zeros(len(zeros))
            for idx, j in enumerate(zeros):
                gj = 0.0
                if 0 <= j + k_star < n:
                    gj += h[j + k_star]
                if 0 <= j - k_star < n:
                    gj -= (1.0 - h[j - k_star])
                g_vals[idx] = gj
            
            best_i_idx = int(np.argmin(f_vals))
            best_j_idx = int(np.argmax(g_vals))
            best_delta = f_vals[best_i_idx] - g_vals[best_j_idx]
            
            if best_delta < -1e-10:
                i = ones[best_i_idx]
                j = zeros[best_j_idx]
                h[i] = 0.0
                h[j] = 1.0
                new_c5 = compute_c5(h)
                if new_c5 < best_c5 - 1e-12:
                    best_c5 = new_c5
                    improved = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        
        if not improved:
            break
    
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_c5 = float('inf')
    best_h = None
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = random_binary_h(320, rng)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    n_pts = len(h0)
    
    # Phase 1: SLSQP with soft-max at parent resolution
    for T in [40, 70, 100]:
        if time.time() - start_time > budget_s - 300:
            break
        for restart in range(2):
            if time.time() - start_time > budget_s - 300:
                break
            if restart == 0:
                h_try = best_h.copy()
            else:
                h_try = best_h + rng.randn(n_pts) * 0.04
                h_try = make_feasible(h_try, n_pts)
            
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
            try:
                res = minimize(soft_objective, h_try, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-10},
                              args=(T,))
                true_c5 = compute_c5(res.x)
                if true_c5 < best_c5:
                    best_c5 = true_c5
                    best_h = res.x.copy()
            except Exception:
                pass
    
    # Phase 2: Binary local search (multiple strategies)
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = n_pts // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        order = np.argsort(best_h[ones_idx])
        for idx in order[:n_ones - target_ones]:
            h_bin[ones_idx[idx]] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(-best_h[zeros_idx])
        for idx in order[:target_ones - n_ones]:
            h_bin[zeros_idx[idx]] = 1
    
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = h_bin.copy()
    
    # Run binary local search with time budget
    time_for_bs = min(200, int((budget_s - (time.time() - start_time)) * 0.3))
    h_ls, c5_ls = binary_local_search(best_h, n_pts, rng, max_sweeps=500,
                                       time_budget=time_for_bs)
    if c5_ls < best_c5:
        best_c5 = c5_ls
        best_h = h_ls.copy()
    
    # Multi-lag search
    time_for_ml = min(80, int((budget_s - (time.time() - start_time)) * 0.2))
    if time_for_ml > 10:
        h_ml, c5_ml = multi_lag_binary_search(best_h, n_pts, rng, max_sweeps=100,
                                               time_budget=time_for_ml)
        if c5_ml < best_c5:
            best_c5 = c5_ml
            best_h = h_ml.copy()
    
    # Try random binary restarts
    time_remaining = budget_s - (time.time() - start_time)
    n_restarts = max(2, int(time_remaining / 80))
    for r in range(n_restarts):
        if time.time() - start_time > budget_s - 100:
            break
        h_rand = random_binary_h(n_pts, rng)
        time_for_r = min(40, int((budget_s - (time.time() - start_time)) * 0.3))
        if time_for_r < 10:
            break
        h_r, c5_r = binary_local_search(h_rand, n_pts, rng, max_sweeps=200,
                                         time_budget=time_for_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()
    
    # Phase 3: SLSQP refinement on best (possibly binary) solution
    if time.time() - start_time < budget_s - 60:
        for T in [60, 100, 150]:
            if time.time() - start_time > budget_s - 30:
                break
            h_try = best_h.copy()
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
            try:
                res = minimize(soft_objective, h_try, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 300, 'ftol': 1e-11},
                              args=(T,))
                true_c5 = compute_c5(res.x)
                if true_c5 < best_c5:
                    best_c5 = true_c5
                    best_h = res.x.copy()
            except Exception:
                pass
    
    # Phase 4: Coordinate descent polish
    n = len(best_h)
    improved = True
    cd_iter = 0
    while improved and cd_iter < 30:
        improved = False
        cd_iter += 1
        if time.time() - start_time > budget_s - 10:
            break
        for j in range(n):
            if time.time() - start_time > budget_s - 10:
                break
            old_val = best_h[j]
            for candidate in [0.0, 1.0, 0.5, old_val + 0.02, old_val - 0.02]:
                if candidate < 0 or candidate > 1:
                    continue
                trial = best_h.copy()
                trial[j] = candidate
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-10:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    break
    
    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
