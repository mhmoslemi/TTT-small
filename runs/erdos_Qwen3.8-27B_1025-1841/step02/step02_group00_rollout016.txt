    s = np.clip(s, -500, 0)
    W = np.exp(s)
    W /= np.sum(W)
    j = np.arange(n)
    s_vals = np.arange(-(n - 1), n)
    chunk = max(1, 1000000 // (2 * n - 1))
    grad = np.zeros(n)
    for st in range(0, n, chunk):
        en = min(st + chunk, n)
        jj = j[st:en, None]
        idx1 = jj + s_vals[None, :]
        idx2 = jj - s_vals[None, :]
        v1 = (idx1 >= 0) & (idx1 < n)
        v2 = (idx2 >= 0) & (idx2 < n)
        t1 = np.zeros((en - st, len(s_vals)))
        t2 = np.zeros((en - st, len(s_vals)))
        t1[v1] = W[None, :][v1] * (1.0 - h[idx1[v1]])
        t2[v2] = W[None, :][v2] * h[idx2[v2]]
        grad[st:en] = dx * (t1.sum(axis=1) - t2.sum(axis=1))
    return grad

def slsqp_opt(h_start, n, T, max_iter):
    global _G_N, _G_T
    _G_N = n
    _G_T = T

    def obj_with_grad(x):
        f = _obj_fn(x)
        g = _grad_logsumexp(x, n, T)
        return f, np.concatenate([g, np.full((1,), 0.0)])

    def cons_with_jac(x):
        c_val = _cons_fun(x)
        c_jac = _cons_jac(x)
        return np.array([c_val]), np.array([c_jac])

    def augmented_obj(x):
        f, g = obj_with_grad(x)
        return f

    def augmented_obj_jac(x):
        f, g = obj_with_grad(x)
        return g[:n]

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_fn, h_start, method='SLSQP', jac=augmented_obj_jac,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-16})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def binary_local_search(h, n, max_rounds=5, max_check=400):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for r in range(max_rounds):
        improved = False
        unc = np.abs(best_h - 0.5)
        order = np.argsort(unc)
        checked = 0
        for idx in order:
            checked += 1
            if checked > max_check:
                break
            old = best_h[idx]
            candidates = [0.0, 1.0]
            if 0.1 < old < 0.9:
                candidates += [0.25, 0.5, 0.75]
            for nv in candidates:
                if abs(nv - old) < 1e-10:
                    continue
                trial = best_h.copy()
                trial[idx] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    best_c5 = c
                    best_h = trial
                    improved = True
                    break
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = (rng.rand(640) < 0.5).astype(float)
        h0 = make_feasible(h0, 640)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Optimize at n=640
    for T in [30, 100, 500]:
        if time_left() < 30:
            break
        h_opt = slsqp_opt(best_h, 640, T, 200)
        c5 = compute_c5(h_opt)
        if c5 <= best_c5:
            best_h = h_opt.copy()
            best_c5 = c5
        else:
            best_h = h_opt.copy()

    # Phase 2: Upsample to 2048 and optimize
    if time_left() > 90:
        h2048 = upsample_h(best_h, 2048)
        c5 = compute_c5(h2048)
        best_c5 = min(best_c5, c5)
        best_h = h2048.copy()

        for T in [50, 200, 800, 3000]:
            if time_left() < 50:
                break
            h_opt = slsqp_opt(best_h, 2048, T, 120)
            c5 = compute_c5(h_opt)
            best_h = h_opt.copy()
            if c5 < best_c5:
                best_c5 = c5

    # Phase 3: Try n=4096 if time allows
    if time_left() > 120:
        h4096 = upsample_h(best_h, 4096)
        c5 = compute_c5(h4096)
        best_c5 = min(best_c5, c5)
        best_h = h4096.copy()

        for T in [200, 1000, 5000]:
            if time_left() < 60:
                break
            h_opt = slsqp_opt(best_h, 4096, T, 80)
            c5 = compute_c5(h_opt)
            best_h = h_opt.copy()
            if c5 < best_c5:
                best_c5 = c5

    # Phase 4: Work back at n=2048 for more iterations if 4096 didn't help
    n_sol = len(best_h)
    if n_sol > 2048:
        h2048 = upsample_h(best_h, 2048)
        c5 = compute_c5(h2048)
        if c5 <= best_c5:
            best_h = h2048.copy()
            best_c5 = c5
            n_sol = 2048

    # Phase 5: Multiple restarts at current resolution
    n_work = len(best_h)
    n_restart = min(n_work, 2048)
    if n_restart < n_work:
        h_work = upsample_h(best_h, n_restart)
    else:
        h_work = best_h.copy()

    for restart in range(4):
        if time_left() < 60:
            break
        if restart == 0:
            h_start = h_work.copy()
        elif restart == 1:
            h_start = 0.8 * h_work + 0.2 * rng.rand(n_restart)
            h_start = make_feasible(h_start, n_restart)
        elif restart == 2:
            h_bin = (rng.rand(n_restart) < 0.5).astype(float)
            h_start = 0.6 * h_work + 0.4 * h_bin
            h_start = make_feasible(h_start, n_restart)
        else:
            h_start = h_work.copy()
            h_start[rng.choice(n_restart, size=n_restart//4, replace=False)] = 1.0 - h_start[rng.choice(n_restart, size=n_restart//4, replace=False)]
            h_start = make_feasible(h_start, n_restart)

        for T in [100, 500, 2000]:
            if time_left() < 30:
                break
            h_opt = slsqp_opt(h_start, n_restart, T, 100)
            c5 = compute_c5(h_opt)
            h_start = h_opt.copy()
            if c5 < best_c5:
                best_c5 = c5
                if n_restart == len(best_h):
                    best_h = h_opt.copy()
                else:
                    best_h = upsample_h(h_opt, len(best_h))
                    best_c5 = min(best_c5, compute_c5(best_h))

    # Re-assert resolution
    n_final = len(best_h)

    # Phase 6: Binary local search
    if time_left() > 40:
        best_h, best_c5 = binary_local_search(best_h, n_final, max_rounds=4, max_check=300)

    # Phase 7: Final SLSQP polish
    if time_left() > 30:
        h_opt = slsqp_opt(best_h, n_final, 5000, 80)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Return at a reasonable resolution
    if len(best_h) > 2048 and time_left() > 10:
        h_ret = upsample_h(best_h, 2048)
        c5_ret = compute_c5(h_ret)
        if c5_ret <= best_c5 + 1e-6:
            return h_ret, c5_ret, 2048

    return best_h, best_c5, len(best_h)
