import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_one(h, s):
    n = len(h)
    g = np.zeros(n)
    if s > 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += -h[:n-s]
    elif s < 0:
        s2 = -s
        g[s2:] += (1.0 - h[:n-s2])
        g[:n-s2] += -h[s2:]
    else:
        g[:] = 1.0 - 2.0 * h
    return g

def smooth_grad(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        return mc * dx, subgrad_one(h, s) * dx
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    nnm1 = n - 1
    for ki in idx:
        w = weights[ki]
        if w < 1e-14:
            break
        s = ki - nnm1
        grad += w * subgrad_one(h, s)
    return mc * dx, grad * dx

    n = params['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

    n = params['n']

    mc = np.max(corr)

        grad += w * subgrad_one(h, s)

    return fval, grad * dx

    c5_avg = compute_c5(h_avg)
    if c5_avg < best_c5:
        best_c5, best_h = c5_avg, h_avg.copy()
    
    return best_h, best_c5

def slsqp_optimize(h, n, T, maxiter=800, ftol=1e-14):

            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception as e:
        pass
    return None, float('inf')
