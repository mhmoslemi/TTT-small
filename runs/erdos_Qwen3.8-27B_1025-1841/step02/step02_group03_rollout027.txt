import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_max(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr))

def make_binary_feasible(h, n):
    n_half = n // 2
    h = np.clip(h, 0.0, 1.0)
    b = (h >= 0.5).astype(float)
    n_ones = int(np.sum(b))
    if n_ones == n_half:
        return b
    ones_idx = np.where(b == 1)[0]
    zeros_idx = np.where(b == 0)[0]
    if n_ones > n_half:
        to_flip = n_ones - n_half
        sorted_ones = ones_idx[np.argsort(h[ones_idx])]
        for i in range(to_flip):
            b[sorted_ones[i]] = 0
    else:
        to_flip = n_half - n_ones
        sorted_zeros = zeros_idx[np.argsort(-h[zeros_idx])]
        for i in range(to_flip):
            b[sorted_zeros[i]] = 1
    return b

def thue_morse_binary(n):
    tm = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=float)
    n_ones = int(np.sum(tm))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(tm == 1)[0]
        extra = n_ones - target
        # Flip the ones with highest index
        for i in range(extra):
            tm[ones_idx[-(i+1)]] = 0
    elif n_ones < target:
        zeros_idx = np.where(tm == 0)[0]
        missing = target - n_ones
        for i in range(missing):
            tm[zeros_idx[i]] = 1
    return tm

def sa_search(h, n, max_iter, rng, T_start, cooldown):
    best_h = h.copy()
    best_c = compute_corr_max(h)
    cur_h = h.copy()
    cur_c = best_c
    T = T_start
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    comp = 1.0 - cur_h
    for it in range(max_iter):
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        cur_h[i] = 0
        cur_h[j] = 1
        comp[i] = 1
        comp[j] = 0
        new_corr = np.correlate(cur_h, comp, mode='full')
        new_c = float(np.max(new_corr))
        delta = new_c - cur_c
        if delta < 0 or (T > 1e-9 and rng.random() < np.exp(-delta / T)):
            cur_c = new_c
            ones = np.delete(ones, np.where(ones == i)[0][0])
            ones = np.append(ones, j)
            zeros = np.delete(zeros, np.where(zeros == j)[0][0])
            zeros = np.append(zeros, i)
            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
            comp[i] = 0
            comp[j] = 1
        T *= cooldown
    return best_h, best_c * (2.0 / n)

def greedy_search(h, n, max_iter, rng):
    best_h = h.copy()
    best_c = compute_corr_max(h)
    cur_h = h.copy()
    cur_c = best_c
    comp = 1.0 - cur_h
    ones = np.where(cur_h == 1)[0]
    zeros = np.where(cur_h == 0)[0]
    no_improve = 0
    max_no_improve = 5000

        i = ones[rng.randint(len(ones))]

        cur_h[i] = 0

        new_c = float(np.max(np.correlate(cur_h, comp, mode='full')))

            oi = np.where(ones == i)[0][0]

            ones = np.append(ones, j)
            zeros = np.delete(zeros, oz)

            if new_c < best_c:
                best_c = new_c
                best_h = cur_h.copy()
        else:
            cur_h[i] = 1
            cur_h[j] = 0
            comp[i] = 0
            comp[j] = 1
            no_improve += 1
    return best_h, best_c * (2.0 / n)

def upsample_binary(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_binary_feasible(h_new, new_n)
