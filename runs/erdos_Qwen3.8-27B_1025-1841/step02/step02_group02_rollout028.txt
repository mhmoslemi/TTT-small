import numpy as np
import time
from scipy.optimize import minimize


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_calc(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def lfs_obj(h, tau, pen_lambda):
    o, _ = smooth_calc(h, tau)
    n = len(h)
    sum_dev = np.sum(h) - n / 2.0
    return o + pen_lambda * sum_dev * sum_dev


def lfs_grad(h, tau, pen_lambda):
    _, g = smooth_calc(h, tau)
    n = len(h)
    sum_dev = np.sum(h) - n / 2.0
    return g + 2.0 * pen_lambda * sum_dev


def run_lbfgs(h, tau, pen_lambda, max_iter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    result = minimize(
        lfs_obj, h, method='L-BFGS-B', jac=lfs_grad,
        args=(tau, pen_lambda), bounds=bounds,
        options={'maxiter': max_iter, 'maxfun': max_iter * 5,
                 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50}
    )
    return result.x


def multi_subgrad(h, n, max_iters, time_limit, num_worst, initial_lr=None):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.004 if initial_lr is None else initial_lr
    v = np.zeros(n)
    beta = 0.8
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_k = np.argsort(corr)[-num_worst:]
        grad = np.zeros(n)
        for k_star in top_k:
            sh = k_star - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[j0:j1] += (1.0 - h[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[j0b:j1b] -= h[idx + sh]
        grad *= dx / num_worst

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 300:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        h = h_new

    return best_h, best_c5
