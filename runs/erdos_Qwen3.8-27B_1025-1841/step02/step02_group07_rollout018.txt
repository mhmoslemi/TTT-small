from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_full(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, T):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def smooth_max_grad(h, n, T):
    """Compute gradient of smooth max of corr*dx"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    p = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    n_lags = len(p)
    for m_idx in range(n_lags):
        if p[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)  # lag offset
        # grad contribution: p[m_idx] * ((1-h[j+k]) - h[j-k]) for valid j
        if k >= 0:
            # (1-h[j+k]) valid for j in [0, n-1-k]
            t1 = np.zeros(n)
            t1[:n-k] = 1.0 - h[k:]
            # h[j-k] valid for j in [k, n-1]
            t2 = np.zeros(n)
            t2[k:] = h[:n-k]
        else:
            kk = -k
            # (1-h[j-kk]) valid for j in [kk, n-1]
            t1 = np.zeros(n)
            t1[kk:] = 1.0 - h[:n-kk]
            # h[j+kk] valid for j in [0, n-1-kk]
            t2 = np.zeros(n)
            t2[:n-kk] = h[kk:]
        grad += p[m_idx] * (t1 - t2)
    return grad

def gradient_descent(h0, n, T, deadline, n_steps=500, lr0=0.01):

    for step in range(n_steps):

        g = smooth_max_grad(h, n, T)
        # Adaptive step size
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-15:
            h -= lr * g / g_norm * 0.1  # normalized step
        else:
            break
        h = project_full(h, n)
        lr *= 0.995
    return h

def slob_opt(h0, n, T, deadline, maxiter=200):
    """SL0B optimization with analytic gradient"""
    def f(x):
        h = project_full(x, n)
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0/n)
        return smooth_max_val(corr, T)
    
    def fprime(x):
        h = project_full(x, n)
        return smooth_max_grad(h, n, T)
    
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=fprime, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_full(res.x, n)
    except Exception:
        pass
    return h0

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_full(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def smooth_max_val(corr, T):
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def smooth_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    s = np.sum(exp_vals)
    p = exp_vals / s
    grad = np.zeros(n)
    for m_idx in range(len(p)):
        if p[m_idx] < 1e-12:
            continue
        k = m_idx - (n - 1)
        if k >= 0:
            t1 = np.zeros(n)
            t1[:n - k] = 1.0 - h[k:]
            t2 = np.zeros(n)
            t2[k:] = h[:n - k]
        else:
            kk = -k
            t1 = np.zeros(n)
            t1[kk:] = 1.0 - h[:n - kk]
            t2 = np.zeros(n)
            t2[:n - kk] = h[kk:]
        grad += p[m_idx] * (t1 - t2)
    return grad * dx

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def lbgfb_opt(h0, n, T, deadline, maxiter=400, lam=1e5):
    def f(x):
        h = np.clip(x, 0.0, 1.0)
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return smooth_max_val(corr, T) + lam * (np.sum(h) - n / 2.0) ** 2

    def fprime(x):
        h = np.clip(x, 0.0, 1.0)
        g = smooth_max_grad(h, n, T)
        g += 2.0 * lam * (np.sum(h) - n / 2.0)
        return g

    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(f, h0, jac=fprime, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxls': 50})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_full(res.x, n)
    except Exception:
        pass
    return h0

def grad_desc(h0, n, T, deadline, n_steps=500, lr0=0.08):
    h = h0.copy()
    lr = lr0
    prev_obj = None
    for step in range(n_steps):
        if time.time() > deadline:
            break
        g = smooth_max_grad(h, n, T)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        h_new = h - lr * g / gn * 0.05
        h_new = project_full(h_new, n)
        c5_new = evaluate_c5(h_new)
        if prev_obj is not None and c5_new > prev_obj + 1e-8:
            lr *= 0.7
        else:
            lr = min(lr * 1.02, 0.15)
        h = h_new
        prev_obj = c5_new
    return h

def slsqp_opt(h0, n, T, deadline, maxiter=300):
    def f(x):
        h = project_full(x, n)
        corr = np.correlate(h, 1.0 - h, mode='full') * (2.0 / n)
        return smooth_max_val(corr, T)

    def c_eq(x):
        return np.sum(x) - n / 2.0

    try:
        res = minimize(f, h0, method='SLSQP', bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_full(res.x, n)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(best_h)

    for T in [30.0, 80.0, 150.0, 400.0]:
        if time.time() > deadline - 40:
            break
        h_new = lbgfb_opt(best_h, n, T, deadline - 40, maxiter=300)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 2: SLSQP for equality constraint
    for T in [100.0, 300.0]:
        if time.time() > deadline - 30:
            break
        h_new = slsqp_opt(best_h, n, T, deadline - 30, maxiter=200)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 3: Gradient descent fine-tuning
    if time.time() < deadline - 25:
        h_new = grad_desc(best_h, n, 500.0, deadline - 25, n_steps=400, lr0=0.05)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 4: Higher resolution (2x)
    n2 = 2 * n
    if time.time() < deadline - 80:
        h_high = interp_to(best_h, n, n2)
        h_high = project_full(h_high, n2)
        for T in [80.0, 200.0, 500.0]:
            if time.time() > deadline - 40:
                break
            h_high = lbgfb_opt(h_high, n2, T, deadline - 40, maxiter=200)
            c5 = evaluate_c5(h_high)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_high.copy()
        if time.time() < deadline - 20:
            h_high = grad_desc(h_high, n2, 800.0, deadline - 20, n_steps=200, lr0=0.03)
            c5 = evaluate_c5(h_high)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_high.copy()

    # Phase 5: Random perturbations + local optimization
    n_cur = len(best_h)
    for attempt in range(40):
        if time.time() > deadline - 15:
            break
        sigma = 0.02 + 0.015 * (attempt % 5)
        h_pert = best_h + np.random.randn(n_cur) * sigma
        h_pert = project_full(h_pert, n_cur)
        T_use = 150.0 if attempt % 2 == 0 else 400.0
        h_new = lbgfb_opt(h_pert, n_cur, T_use, deadline - 10, maxiter=80, lam=1e6)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    # Phase 6: Final fine-tuning
    if time.time() < deadline - 5:
        h_new = grad_desc(best_h, len(best_h), 1000.0, deadline - 5, n_steps=150, lr0=0.02)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
