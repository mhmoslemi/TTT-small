            C_hi = max(C_hi, (corr - old_v * g).max(), (corr + (1.0 - old_v) * g).max())
            C_lo = 0.0
            
            best_v = old_v
            for _ in range(35):
                C_mid = (C_lo + C_hi) / 2.0
                upper_v = 1.0
                lower_v = 0.0
                if pos.any():

                    best_v = (lower_v + upper_v) / 2.0

            new_v = max(0.0, min(1.0, best_v))
            if abs(new_v - old_v) > 1e-12:
                dv = new_v - old_v
                corr += dv * g
                h[j] = new_v
                one_minus_h[j] = 1.0 - new_v
                neg_h_rev[n-1-j] = -new_v
        
        sweep += 1
        # Project
        h = project_to_feasible(h, n)
        # Recompute derived arrays
        one_minus_h = 1.0 - h
        neg_h_rev = -h[::-1].copy()
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
    
    return best_h, best_c5

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def coord_descent_h(h_init, n, time_limit, seed=42):
    """Coordinate descent: exact 1D minimization per coordinate."""
    h = project_to_feasible(h_init, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    one_minus_h = 1.0 - h
    neg_h_rev = -h[::-1].copy()
    ncorr = 2 * n - 1
    best_c5 = corr.max() * (2.0 / n)
    best_h = h.copy()
    dx = 2.0 / n
    start = time.time()
    rng = np.random.default_rng(seed)
    stall = 0
    sweep = 0

    while True:
        rem = time_limit - (time.time() - start)
        if rem < 1.0:
            break

        order = rng.permutation(n)
        for j in order:
            old_v = h[j]

            # Build gradient g: dcorr[m]/dh[j]
            g = np.zeros(ncorr)
            g[n - 1 - j:2 * n - 1 - j] = one_minus_h
            g[j:j + n] += neg_h_rev

            # Precompute masks
            pos_idx = np.where(g > 1e-14)[0]
            neg_idx = np.where(g < -1e-14)[0]

            if len(pos_idx) == 0 and len(neg_idx) == 0:
                continue

            g_pos = g[pos_idx]
            g_neg = g[neg_idx]
            corr_pos = corr[pos_idx]
            corr_neg = corr[neg_idx]

            # Upper bound for C
            C_hi = corr.max()
            if len(pos_idx) > 0 or len(neg_idx) > 0:
                C_hi = max(C_hi,
                           (corr - old_v * g).max(),
                           (corr + (1.0 - old_v) * g).max())
            C_lo = 0.0

            best_v = old_v
            for _ in range(40):
                C_mid = (C_lo + C_hi) / 2.0
                upper_v = 1.0
                lower_v = 0.0
                if len(pos_idx) > 0:
                    ub = ((C_mid - corr_pos) / g_pos + old_v).min()
                    if ub < upper_v:
                        upper_v = ub
                if len(neg_idx) > 0:
                    lb = ((C_mid - corr_neg) / g_neg + old_v).max()
                    if lb > lower_v:
                        lower_v = lb
                if lower_v <= upper_v + 1e-15:
                    C_hi = C_mid
                    best_v = max(0.0, min(1.0, (lower_v + upper_v) / 2.0))
                else:
                    C_lo = C_mid

            new_v = max(0.0, min(1.0, best_v))
            if abs(new_v - old_v) > 1e-13:
                dv = new_v - old_v
                corr += dv * g
                h[j] = new_v
                one_minus_h[j] = 1.0 - new_v
                neg_h_rev[n - 1 - j] = -new_v

        sweep += 1
        # Project to feasible
        h = project_to_feasible(h, n)
        one_minus_h = 1.0 - h
        neg_h_rev = -h[::-1].copy()
        corr = np.correlate(h, 1.0 - h, mode='full')

        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 40:
            # Perturb to escape local optimum
            if rem > 5.0:
                h = best_h + 0.05 * rng.standard_normal(n)
                h = project_to_feasible(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                one_minus_h = 1.0 - h
                neg_h_rev = -h[::-1].copy()
                stall = 0
            else:
                break

    return best_h, best_c5


def smooth_max_optimize(h_init, n, time_limit, tau=20.0, lr=0.05, momentum=0.9):
    """Lightweight smooth-max gradient descent as fallback."""
    h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    ncorr = 2 * n - 1
    v = np.zeros(n)
    stall = 0

    for it in range(100000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -500, 500)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()

        grad = np.zeros(n)
        for k in range(ncorr):
            pk = p[k]
            if pk < 1e-12:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
        grad *= dx

        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5

    t0 = time.time()

    def elapsed():
        return time.time() - t0

    def remaining():
        return max(0.0, budget_s - elapsed())

    np.random.seed(seed)

    # Get initial solution
    h_init = initial_h_values.copy()
    n_init = len(h_init)

    # Phase 1: Coordinate descent at n=2000 (~450s)

            best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Another coord descent with different seed for diversity
    if remaining() > 200:
        h_pert = best_h + 0.03 * np.random.randn(n_init)
        h_pert = project_to_feasible(h_pert, n_init)
        tl2 = min(250.0, remaining() * 0.20)
        h2, c5_2 = coord_descent_h(h_pert, n_init, tl2, seed=seed + 1)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Upsample to n=4000 and coord descent
    if remaining() > 150:
        n_up = 4000
        h_up = interp_periodic(best_h, n_up)
        h_up = project_to_feasible(h_up, n_up)
        tl3 = min(500.0, remaining() * 0.55)
        h3, c5_3 = coord_descent_h(h_up, n_up, tl3, seed=seed + 2)
        if c5_3 < best_c5:

    # Phase 4: Smooth max polish on best at its resolution
    if remaining() > 60:
        n_best = len(best_h)
        tl4 = min(80.0, remaining() * 0.4)
        h4, c5_4 = smooth_max_optimize(best_h, n_best, tl4, tau=15.0, lr=0.03)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Final coord descent polish

        n_best = len(best_h)
        tl5 = min(60.0, remaining() * 0.6)
        h5, c5_5 = coord_descent_h(best_h, n_best, tl5, seed=seed + 99)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
