def smooth_max_grad_fft(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = fft_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -300, 300)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    P_arr = p
    # S1
    prefix = np.zeros(2*n)
    prefix[1:] = np.cumsum(P_arr)
    idx = np.arange(n)
    S1 = prefix[2*n-1-idx] - prefix[n-1-idx]
    
    # S2 via FFT
    Q = P_arr[::-1]
    conv_size = n + len(Q) - 1
    H_f = np.fft.rfft(h, conv_size)
    Q_f = np.fft.rfft(Q, conv_size)
    S2_full = np.fft.irfft(H_f * Q_f, conv_size)
    S2 = S2_full[n-1:n-1+n]
    
    # S3 via FFT
    P_f = np.fft.rfft(P_arr, conv_size)
    H_f2 = np.fft.rfft(h, conv_size)
    S3_full = np.fft.irfft(P_f * H_f2, conv_size)
    S3 = S3_full[n-1:n-1+n]
    
    grad = (S1 - S2 - S3) * dx
    return obj, grad, corr

def nesterov_smooth(h, n, tau, time_limit, lr=None, max_iter=50000):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.5 / tau
    
    t = 1.0
    h_prev = h.copy()
    y = h.copy()
    stall = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        obj, grad, corr = smooth_max_grad_fft(y, tau)
        c5_now = corr.max() * (2.0 / n)
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = y.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            lr *= 0.5
            t = 1.0
            h = y.copy()
            h_prev = h.copy()
            stall = 0
            if lr < 1e-10:
                break
        
        h_new = project_to_feasible(y - lr * grad, n)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y = h_new + beta * (h_new - h)
        y = np.clip(y, 0.0, 1.0)  # keep in box
        h = h_new
        t = t_new
    
    return best_h, best_c5

def subgrad_fft(h, n, time_limit, lr=None, top_k=1, max_iter=200000):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.002 * dx * n
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        
        corr = fft_corr(h)
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 4000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break
        
        if top_k == 1:
            k_star = np.argmax(corr)
            sh = k_star - n + 1
            grad = np.zeros(n)
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[j0-sh:j1-sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= h[j0b+sh:j1b+sh]
        else:
            idx_sorted = np.argsort(corr)[::-1]
            top_idx = idx_sorted[:top_k]
            top_vals = corr[top_idx]
            w = np.exp(top_vals - top_vals[0])
            w /= w.sum()
            grad = np.zeros(n)
            for i, k in enumerate(top_idx):
                sh = k - n + 1
                j0 = max(0, sh)
                j1 = min(n, sh + n)
                if j0 < j1:
                    grad[j0:j1] += w[i] * (1.0 - h[j0-sh:j1-sh])
                j0b = max(0, -sh)
                j1b = min(n, n - sh)
                if j0b < j1b:
                    grad[j0b:j1b] -= w[i] * h[j0b+sh:j1b+sh]
        
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5
