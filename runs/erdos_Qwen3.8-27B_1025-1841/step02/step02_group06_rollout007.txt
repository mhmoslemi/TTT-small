from scipy.fft import fft, ifft, next_fast_len

def project_to_feasible(g, n):
    """Project g to [0,1]^n with sum = n/2."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def project_grad_optimize_v2(h, n, tau, max_iters, time_limit, lr_init=0.2, topM=0, use_fft=False):

    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(50, max_iters // 20)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr_fft(h) if use_fft else compute_corr(h)
        g = corr * dx
        cur_max = g.max()

            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p_top = exp_c / s

def level_set_optimize(h, n, time_limit, lr_init=0.05, max_iters=20000):
    """Optimize by focusing on correlations above a moving threshold."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()

    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()

        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()

        threshold = g.max() - (g.max() - g.min()) * 0.3
        mask = g > threshold

        for k in np.where(mask)[0]:

        h = h_new
        lr *= 0.999  # slow decay
        
        if lr < 1e-10:
            break

    return best_h, best_obj

def topk_subgrad(h, n, time_limit, topK=5, lr_init=0.003):

    dx = 2.0 / n

    best_h = h.copy()

    lr = lr_init

        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()

            grad += gradient_for_lag(h, ki, n)

        n_iter += 1
        
        if n_iter % 5000 == 0 and cur_lr < 1e-6:
            break
    
    return best_h, best_c5

import numpy as np
import time
from scipy.fft import fft, ifft, next_fast_len


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr_direct(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_corr_fft(h):
    n = len(h)
    N = next_fast_len(2 * n - 1)
    H = fft(h, N)
    G = fft(1.0 - h, N)
    C = ifft(H * np.conj(G)).real
    indices = np.mod(n - 1 - np.arange(2 * n - 1), N)
    return C[indices]


def compute_corr(h):
    n = len(h)
    if n >= 3000:
        return compute_corr_fft(h)
    return compute_corr_direct(h)


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def gradient_single_lag(h, k, n):
    d = k - n + 1
    grad = np.zeros(n)
    j0 = max(0, d)
    j1 = min(n, n + d)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[idx] += (1.0 - h[idx - d])
    j0b = max(0, -d)
    j1b = min(n, n - d)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        grad[idx] -= h[idx + d]
    return grad


def softmax_gradient(h, g, n, dx, tau, topM):
    corr_len = 2 * n - 1
    if topM > 0 and topM < corr_len:
        top_idx = np.argpartition(g, -topM)[-topM:]
        g_top = g[top_idx]
        max_top = g_top.max()
        shifted = np.clip(tau * (g_top - max_top), -200, 200)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        if s < 1e-30:
            p_top = np.ones(topM) / topM
        else:
            p_top = exp_c / s
        grad = np.zeros(n)
        for i in range(topM):
            pk = p_top[i]
            if pk < 1e-12:
                continue
            grad += pk * gradient_single_lag(h, top_idx[i], n)
    else:
        max_g = g.max()
        shifted = np.clip(tau * (g - max_g), -200, 200)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        if s < 1e-30:
            p = np.ones(corr_len) / corr_len
        else:
            p = exp_c / s
        grad = np.zeros(n)
        for k in range(corr_len):
            pk = p[k]
            if pk < 1e-12:
                continue
            grad += pk * gradient_single_lag(h, k, n)
    return grad * dx


def optimize_softgrad(h, n, tau, max_iters, time_limit, lr_init, topM, momentum=0.85):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr(h)
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    patience = max(80, max_iters // 20)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        grad = softmax_gradient(h, g, n, dx, tau, topM)
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.8
            v *= 0.4
            stall = 0
            if lr < 1e-13:
                break

    return best_h, best_obj


def subgrad_topk(h, n, time_limit, topK, lr_init):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    it = 0

    while time.time() - start < time_limit:
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            grad += gradient_single_lag(h, ki, n)
        grad *= dx / topK

        cur_lr = lr_init / (1.0 + 0.008 * it)
        h = project_to_feasible(h - cur_lr * grad, n)
        it += 1
        if cur_lr < 1e-8:
            break

    return best_h, best_c5


def multifreq_subgrad(h, n, time_limit, lr_init=0.004):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    it = 0
    topK_cycle = [5, 10, 20, 3]
