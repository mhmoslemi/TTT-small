```python
def compute_gradient(h, T, dx):
    n = len(h)
    one_minus_h = 1.0 - h
    r = np.correlate(h, one_minus_h, mode='full') * dx
    # r has length 2n-1, r[p] corresponds to lag k = p - (n-1)
    
    # Compute softmax weights
    r_scaled = T * r
    r_max = np.max(r_scaled)
    exp_r = np.exp(r_scaled - r_max)
    W = exp_r / np.sum(exp_r)
    
    # lags
    lags = np.arange(-(n-1), n)
    
    # g[j] = dx * sum_p W[p] * ((1-h[j+lag_p]) * 1_{valid} - h[j-lag_p] * 1_{valid})
    g = np.zeros(n)
    for j in range(n):
        # term1: sum over p where 0 <= j + lags[p] < n
        valid1 = (j + lags >= 0) & (j + lags < n)
        term1 = np.sum(W[valid1] * (1.0 - h[j + lags[valid1]]))
        
        # term2: sum over p where 0 <= j - lags[p] < n
        valid2 = (j - lags >= 0) & (j - lags < n)
        term2 = np.sum(W[valid2] * h[j - lags[valid2]])
        
        g[j] = dx * (term1 - term2)
    
    return g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def project_onto_simplex(h, n):
    """Project h onto {h : 0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def objective_and_gradient(h, n, T, mu):
    """Compute soft-max objective and its gradient, plus penalty for sum constraint."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    r = np.correlate(h, one_minus_h, mode='full') * dx  # shape (2n-1,)
    
    # Soft-max value
    scaled = T * r
    r_max = np.max(scaled)
    exp_r = np.exp(scaled - r_max)
    Z = np.sum(exp_r)
    obj_sm = (r_max + np.log(Z)) / T
    
    # Softmax weights
    W = exp_r / Z  # shape (2n-1,)
    
    # Gradient of soft-max: g_sm[j] = dx * sum_p W[p] * ((1-h[j+lag_p]) - h[j-lag_p])
    # where lag_p = p - (n-1)
    
    # P[j] = sum of W[p] for valid p (where 0 <= j + lag_p < n)
    # valid p: 0 <= j + p - (n-1) < n => n-1-j <= p <= n-1+j
    cumW = np.concatenate([[0.0], np.cumsum(W)])
    P = np.zeros(n)
    for j in range(n):
        p_lo = max(0, n - 1 - j)
        p_hi = min(2 * n - 2, n - 1 + j)
        P[j] = cumW[p_hi + 1] - cumW[p_lo]
    
    # Q[j] = sum_p W[p] * h[j + p - (n-1)] for valid p
    # = correlate(W, h, 'full')[n-1-j]
    Q_full = np.correlate(W, h, mode='full')
    Q = Q_full[:n][::-1]  # Q[j] = Q_full[n-1-j]
    
    # R[j] = sum_p W[p] * h[j - p + (n-1)] for valid p  
    # = convolve(h, W, 'full')[n-1+j]
    R_full = np.convolve(h, W, mode='full')
    R = R_full[n-1:2*n-1]  # R[j] = R_full[n-1+j]
    
    g_sm = dx * (P - Q - R)
    
    # Penalty gradient
    sum_h = np.sum(h)
    penalty_obj = (mu / 2.0) * (sum_h - n/2.0)**2
    g_penalty = mu * (sum_h - n/2.0) * np.ones(n)
    
    total_obj = obj_sm + penalty_obj
    total_g = g_sm + g_penalty
    
    return total_obj, total_g

def lbfgsb_optimize(h0, n, T, mu, deadline, maxiter=500, ftol=1e-15):
    """Optimize using L-BFGS-B with penalty method."""
    def fun_and_grad(x):
        return objective_and_gradient(x, n, T, mu)
    
    try:
        res = minimize(fun_and_grad, h0, method='L-BFGS-B',
                       jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': ftol, 'gtol': 1e-14,
                                'maxls': 50})
        if time.time() < deadline:
            if res.x is not None and np.all(np.isfinite(res.x)):
                return project_onto_simplex(res.x, n)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(42, 0.5)
    
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: Optimize at current resolution with increasing T
    n = len(best_h)
    for T in [50.0, 100.0, 200.0, 400.0]:
        if time.time() > deadline - 30:
            break
        for mu in [1.0, 5.0, 20.0]:
            if time.time() > deadline - 10:
                break
            h_new = lbfgsb_optimize(best_h, n, T, mu, deadline, maxiter=300)
            h_new = project_onto_simplex(h_new, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Phase 2: Try higher resolutions
    for n_target in [504, 768, 1024, 1536, 2048]:
        if time.time() > deadline - 60:
            break
        
        h = interp_to(best_h, len(best_h), n_target)
        h = project_onto_simplex(h, n_target)
        
        for T in [100.0, 200.0, 400.0]:
            if time.time() > deadline - 20:
                break
            h_new = lbfgsb_optimize(h, n_target, T, 5.0, deadline, maxiter=400)
            h_new = project_onto_simplex(h_new, n_target)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h = h_new
    
    # Phase 3: Random restarts
    n_best = len(best_h)
    n_restarts = int(max(0, (deadline - time.time()) / 30))
    for i in range(min(n_restarts, 20)):
        if time.time() > deadline - 10:
            break
        h_pert = best_h + np.random.randn(n_best) * 0.05
        h_pert = project_onto_simplex(h_pert, n_best)
        h_new = lbfgsb_optimize(h_pert, n_best, 200.0, 10.0, deadline, maxiter=200)
        h_new = project_onto_simplex(h_new, n_best)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Final: ensure feasibility
    best_h = project_onto_simplex(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
```
