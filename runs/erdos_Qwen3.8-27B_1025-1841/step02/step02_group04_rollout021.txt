            grad[sa:] += w * (1.0 - h[:n-sa])
            if sa > 0:
                grad[:sa] += w * (-h[sa:])
    
    return grad * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    m_arr = np.zeros(n)
    v_arr = np.zeros(n)
    
    for it in range(n_iter):
        t = it + 1
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 + 0.5 * (1.0 - frac))

        v_arr = b2 * v_arr + (1 - b2) * (grad * grad)
        m_hat = m_arr / (1.0 - b1 ** t)
        v_hat = v_arr / (1.0 - b2 ** t)

        h = make_feasible(h, n)

            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()

import numpy as np

def _fft_corr(h):
    n = len(h)
    size = 1
    while size < 2 * n - 1:
        size *= 2
    H = np.fft.rfft(h, size)
    Gh = np.fft.rfft((1.0 - h)[::-1], size)
    return np.fft.irfft(H * Gh, size)[:2 * n - 1]

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = _fft_corr(h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _grad_single(h, s, n):
    grad = np.zeros(n)
    if s >= 0:
        grad[:n-s] += (1.0 - h[s:])
        if s > 0:
            grad[n-s:] += (-h[:s])
    else:
        sa = -s
        grad[sa:] += (1.0 - h[:n-sa])
        if sa > 0:
            grad[:sa] += (-h[sa:])
    return grad

def get_grad(h, T, top_k=40):
    n = len(h)
    dx = 2.0 / n
    corr = _fft_corr(h)
    c_max = np.max(corr)
    shifted = np.clip(corr - c_max, -100.0, 0.0)
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)

        return _grad_single(h, m_idx - (n - 1), n) * dx
    weights = exp_s / total
    k_use = min(top_k, len(weights))
    idx = np.argsort(weights)[-k_use:]
    grad = np.zeros(n)
    for m in idx:
        w = weights[m]
        if w < 1e-10:
            continue
        grad += w * _grad_single(h, m - (n - 1), n)
    return grad * dx
