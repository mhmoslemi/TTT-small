    obj = dx * (max_c + np.log(s) / tau)

            u *= 0.5

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.3
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    t = 0
    stall = 0
    stall_threshold = 150

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h - step, n)

        if stall > stall_threshold:
            lr *= 0.6
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break

        h = h_new

    return best_h, best_obj


def multi_shift_grad(h, n, top_k):
    dx = 2.0 / n
    corr = compute_corr(h)
    overlap = corr * dx
    k_max = overlap.max()
    if top_k >= len(overlap):
        top_k = len(overlap)
    idx = np.argpartition(overlap, -top_k)[-top_k:]
    w = overlap[idx]
    w = np.clip(w - 0.9 * k_max, 0.0, None)
    wsum = w.sum()
    if wsum < 1e-16:
        idx = np.argmax(overlap)
        idx = np.array([idx])
        w = np.array([1.0])
    else:
        w = w / wsum
    grad = np.zeros(n)
    for wi, ki in zip(w, idx):
        sh = ki - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += wi * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= wi * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    return k_max, grad
