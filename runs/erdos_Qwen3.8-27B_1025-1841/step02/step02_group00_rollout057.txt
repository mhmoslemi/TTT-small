from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_detail(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), int(np.argmax(corr)), corr

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Convert to binary pattern with exactly n/2 ones"""
    h = np.asarray(h, dtype=float)
    ones = np.sum(h > 0.5)
    target = n // 2
    if ones == target:
        return (h > 0.5).astype(float)
    elif ones > target:
        # Need to flip some 1s to 0s
        one_idx = np.where(h > 0.5)[0]
        # Sort by value descending, flip the smallest ones
        order = np.argsort(h[one_idx])[::-1]
        flip_count = ones - target
        for idx in order[:flip_count]:
            h[one_idx[idx]] = 0.0
    else:
        # Need to flip some 0s to 1s
        zero_idx = np.where(h <= 0.5)[0]
        order = np.argsort(h[zero_idx])[::-1]
        flip_count = target - ones
        for idx in order[:flip_count]:
            h[zero_idx[idx]] = 1.0
    return (h > 0.5).astype(float)

def binary_local_search(h, n, max_rounds=200, time_limit=None, start_time=None):
    """Local search on binary pattern by flipping pairs"""
    h = to_binary(h, n)
    best_c5 = compute_c5(h)
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]

            if time.time() - start_time > time_limit:
                break
        improved = False

        rng = np.random.RandomState(round)
        
        # Greedy: try all single-pair flips but sample if too many
        if n_trials <= 5000:
            pairs = [(i, j) for i in ones_idx for j in zeros_idx]
            rng.shuffle(pairs)
        else:
            pairs = [(ones_idx[rng.randint(len(ones_idx))], zeros_idx[rng.randint(len(zeros_idx))]) for _ in range(5000)]
            rng.shuffle(pairs)
        
        for i, j in pairs:
            h[i], h[j] = 1.0, 0.0  # flip
            c = compute_c5(h)
            if c < best_c5 - 1e-14:
                best_c5 = c
                ones_idx = np.where(h == 1.0)[0]
                zeros_idx = np.where(h == 0.0)[0]
                improved = True
                n_trials = min(len(ones_idx) * len(zeros_idx), 5000)
            else:
                h[i], h[j] = 0.0, 1.0  # unflip
        if not improved:
            break
    
    return h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if old_n == new_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n // 2
    mask_one = h >= 0.5
    ones = int(np.sum(mask_one))
    if ones > target:
        one_idx = np.where(mask_one)[0]
        order = np.argsort(h[one_idx])
        h[one_idx[order[:ones - target]]] = 0.0
        mask_one = h >= 0.5
    elif ones < target:
        zero_idx = np.where(~mask_one)[0]
        order = np.argsort(h[zero_idx])[::-1]
        h[zero_idx[order[:target - ones]]] = 1.0
    h[mask_one] = 1.0
    h[~mask_one] = 0.0
    return h

def grad_guided_binary_ls(h, n, max_iters=200000, time_limit=400, start_time=0, seed=42):
    """Gradient-guided binary local search with simulated annealing"""
    h = to_binary(h, n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    T_sa = 0.01  # SA temperature (in units of c5)
    T_decay = 0.999
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Compute active lag and gradient
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s_star = k_star - n + 1  # the shift
        
        # Compute subgradient
        # dc/dh[i] = dx * ((1-h[i+s]) - h[i-s]) for valid indices
        i_plus = np.arange(n) + s_star
        i_minus = np.arange(n) - s_star
        grad = np.zeros(n)
        valid_p = (i_plus >= 0) & (i_plus < n)
        valid_m = (i_minus >= 0) & (i_minus < n)
        grad[valid_p] += (1.0 - h[i_plus[valid_p]])
        grad[valid_m] -= h[i_minus[valid_m]]
        grad *= dx
        
        # Find best flip pair
        ones_idx = np.where(h == 1.0)[0]
        zeros_idx = np.where(h == 0.0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        
        # Greedy: find i (1->0) with largest positive grad, j (0->1) with most negative grad
        # But we need to maintain sum, so we flip one 1->0 and one 0->1
        # Score for flipping i from 1 to 0: -grad[i] (we want to decrease c, so if grad[i]>0, decreasing h[i] helps)
        # Score for flipping j from 0 to 1: -grad[j] (if grad[j]<0, increasing h[j] helps)
        
        # Actually, flipping i:1->0 changes c by approximately -grad[i]*step where step=1
        # Flipping j:0->1 changes c by approximately grad[j]*step where step=1
        # Total change: -grad[i] + grad[j]
        # We want this to be negative (decrease c)
        # So we want grad[i] - grad[j] > 0, i.e., grad[i] > grad[j]
        
        # Best greedy: maximize grad[i] - grad[j] over i in ones, j in zeros
        # This is: max over i in ones of grad[i] - min over j in zeros of grad[j]
        best_i_score = np.max(grad[ones_idx]) if len(ones_idx) > 0 else 0
        best_j_score = np.min(grad[zeros_idx]) if len(zeros_idx) > 0 else 0
        
        if best_i_score - best_j_score < 1e-12:
            # No improving move found greedily; try random
            i = ones_idx[rng.randint(len(ones_idx))]
            j = zeros_idx[rng.randint(len(zeros_idx))]
        else:
            i = ones_idx[np.argmax(grad[ones_idx])]
            j = zeros_idx[np.argmin(grad[zeros_idx])]
        
        # Try the flip
        h[i] = 0.0
        h[j] = 1.0
        c_new = compute_c5(h)
        
        # Accept if better, or with SA probability
        if c_new < best_c5 - 1e-14:
            best_c5 = c_new
            T_sa *= T_decay
        elif rng.random() < np.exp(-(c_new - best_c5) / (T_sa + 1e-10)):
