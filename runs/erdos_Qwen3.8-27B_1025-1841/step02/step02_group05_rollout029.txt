def fft_grad(h, p_weights, n):
    dx = 2.0 / n
    # Need FFT size >= n + 2n-1 - 1 = 3n-2
    fft_size = 1
    while fft_size < 3*n - 1:
        fft_size *= 2
    H = np.fft.rfft(h, fft_size)
    P = np.fft.rfft(p_weights, fft_size)
    P_rev = np.fft.rfft(p_weights[::-1], fft_size)
    
    conv1 = np.fft.irfft(H * P, fft_size)
    conv2 = np.fft.irfft(H * P_rev, fft_size)
    
    part1 = conv2[n-1:n-1+n]  # correlate(h, p_weights) = convolve(h, p_weights_rev)
    part2 = conv1[n-1:n-1+n]  # convolve(h, p_weights)
    
    return dx * (1.0 - part1 - part2)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def fft_smooth_grad(h, tau, n):
    """Compute objective and gradient for log-sum-exp smoothed objective using FFT."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
