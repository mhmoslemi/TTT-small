    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def compute_gradient_topM(h, p_top, top_idx, n, dx):

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def compute_gradient_topM(h, p_top, top_idx, n, dx):
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_top[i]
        if pk < 1e-15:
            continue
        k = top_idx[i]
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx


def grad_optimize(h, n, tau, max_iters, time_limit, lr_init=0.15, topM=0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(80, max_iters // 15)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
        else:
            top_idx = np.argpartition(g, -min(300, 2 * n - 1))[-min(300, 2 * n - 1):]
            g_top = g[top_idx]

        shifted = np.clip(tau * (g_top - g_top.max()), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        if s > 0:
            p_top = exp_c / s
        else:
            p_top = np.ones(len(top_idx)) / len(top_idx)
        grad = compute_gradient_topM(h, p_top, top_idx, n, dx)

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new

    return best_h, best_obj


def swap_delta_corr(h, i, j, n):
    delta = np.zeros(2 * n - 1)
    ds = max(-i, -(n - 1))
    de = min(n - 1 - i, n - 1)
    if ds <= de:
        delta[ds + n - 1:de + n] += h[i + ds:i + de + 1]
    ds = max(-j, -(n - 1))
    de = min(n - 1 - j, n - 1)
    if ds <= de:
        delta[ds + n - 1:de + n] -= h[j + ds:j + de + 1]
    ds = max(i - n + 1, -(n - 1))
    de = min(i, n - 1)
    if ds <= de:
        idx = np.arange(i - ds, i - de - 1, -1)
        delta[ds + n - 1:de + n] += h[idx]
    ds = max(j - n + 1, -(n - 1))
    de = min(j, n - 1)
    if ds <= de:
        idx = np.arange(j - ds, j - de - 1, -1)
        delta[ds + n - 1:de + n] -= h[idx]
    return delta


def binary_local_search(h_init_bin, n, time_limit, max_sweeps=500, tries_per_sweep=5000):
    dx = 2.0 / n
    h_b = h_init_bin.copy()
    corr = compute_corr(h_b)
    best_c5 = corr.max() * dx
    best_h = h_b.copy()
    start = time.time()
    no_improve = 0

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        improved = False
        n_tries = min(tries_per_sweep, len(ones) * len(zeros))
        for t in range(n_tries):
            if time.time() - start > time_limit:
                break
            i = ones[t % len(ones)]
            j = zeros[t % len(zeros)]
            if h_b[i] != 1 or h_b[j] != 0:
                continue
            delta = swap_delta_corr(h_b, i, j, n)
            new_max = (corr + delta).max() * dx
            if new_max < best_c5 - 1e-15:
                h_b[i] = 0
                h_b[j] = 1
                corr = corr + delta
                best_c5 = new_max
                best_h = h_b.copy()
                improved = True
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 40:
                break
    return best_h, best_c5


def binary_sa(h_init_bin, n, time_limit, T_init=0.008, T_floor=0.0005, tries_per_sweep=3000):
    dx = 2.0 / n
    h_b = h_init_bin.copy()
    corr = compute_corr(h_b)
    cur_c5 = corr.max() * dx
    best_c5 = cur_c5
    best_h = h_b.copy()
    start = time.time()

    while time.time() - start < time_limit:
        prog = (time.time() - start) / max(time_limit, 1e-9)
        T = T_init * (T_floor / T_init) ** prog
        ones = np.where(h_b == 1)[0]
        zeros = np.where(h_b == 0)[0]
        np.random.shuffle(ones)
        np.random.shuffle(zeros)
        n_tries = min(tries_per_sweep, len(ones) * len(zeros))
        for t in range(n_tries):
            if time.time() - start > time_limit:
                break
            i = ones[t % len(ones)]
            j = zeros[t % len(zeros)]
            if h_b[i] != 1 or h_b[j] != 0:
                continue
            delta = swap_delta_corr(h_b, i, j, n)
            new_max = (corr + delta).max() * dx
            dc = new_max - cur_c5
            if dc < 0 or np.random.random() < np.exp(-dc / max(T, 1e-8)):
                h_b[i] = 0
                h_b[j] = 1
                corr = corr + delta
                cur_c5 = new_max
                if cur_c5 < best_c5 - 1e-15:
                    best_c5 = cur_c5
                    best_h = h_b.copy()
    return best_h, best_c5


def to_binary(h, n):
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:n // 2]] = 1.0
    return h_bin


def subgrad_polish(h, n, max_iters, time_limit, lr_init=0.004, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    patience = 3000

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK

        cur_lr = lr_init / (1.0 + it * 0.003)
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
            lr_init *= 0.7
            v *= 0.5
            stall = 0
            if lr_init < 1e-12:
                break
        h = h_new

    return best_h, best_c5
