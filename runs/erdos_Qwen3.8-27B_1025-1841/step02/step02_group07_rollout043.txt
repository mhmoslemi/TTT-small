```python
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}"""
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmin_objective_and_grad(h, T):
    """Compute softmin of autocorrelation and its gradient."""
    n = len(h)
    # Compute autocorrelation A[0..n-1]
    h_pad = np.zeros(2*n - 1)
    h_pad[:n] = h
    H = np.fft.fft(h_pad)
    A = np.fft.ifft(H * np.conj(H)).real[:n]
    A = np.maximum(A, 0)  # numerical safety
    
    # Softmax weights for softmin
    neg_T_A = -T * A
    neg_T_A -= np.max(neg_T_A)
    p = np.exp(neg_T_A)
    p_sum = np.sum(p)
    p = p / p_sum
    
    # Objective: softmin(A) = -(1/T) * log(sum(exp(-T*A)))
    obj = -(1.0/T) * (np.max(neg_T_A) + np.log(p_sum))
    
    # Gradient: d(softmin)/d(h[j]) = -(2/T) * sum_s p_s * h[j+s]
    # = -(2/T) * (p * h)[j] (linear convolution)
    p_pad = np.zeros(2*n - 1)
    p_pad[:n] = p
    P = np.fft.fft(p_pad)
    conv = np.fft.ifft(P * H).real[:n]
    
    grad = -(2.0/T) * conv
    
    return obj, grad, A

def gradient_descent(h0, T_schedule, lr_init, deadline, n_total_steps=None):
    """Run gradient descent with annealed temperature."""
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    lr = lr_init
    step_count = 0
    
    for T in T_schedule:
        if time.time() > deadline:
            break
        # Run several steps at this T
        while time.time() < deadline:
            if n_total_steps and step_count >= n_total_steps:
                break
            obj, grad, A = softmin_objective_and_grad(h, T)
            
            # Gradient descent step
            h_new = h - lr * grad
            h_new = project_h(h_new)
            
            # Evaluate

            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            
            # Simple adaptive lr
            if c5_new > best_c5 + 0.001:
                lr *= 0.5
            elif c5_new < best_c5 - 1e-6 and lr < 0.5:
                lr *= 1.1
            else:
                lr *= 0.95
            
            h = h_new
            step_count += 1
            
            if step_count % 100 == 0 and time.time() > deadline:
                break
    
    return best_h, best_c5
```

```python
def softmin_obj_grad(h, T):
    n = len(h)
    size = 2 * n - 1
    # Pad to next power of 2 for FFT efficiency
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    
    h_pad = np.zeros(fft_size)
    h_pad[:n] = h
    
    # Autocorrelation
    H = np.fft.fft(h_pad)
    A = np.fft.ifft(H * np.conj(H)).real[:n]
    A = np.maximum(A, 0.0)
    
    # Softmax weights
    x = -T * A
    x_max = np.max(x)
    p = np.exp(x - x_max)
    p_sum = np.sum(p)
    obj = -(1.0/T) * (x_max + np.log(p_sum))
    p = p / p_sum
    
    # Compute (p * h) and (p_rev * h)
    p_pad = np.zeros(fft_size)
    p_pad[:n] = p
    P = np.fft.fft(p_pad)
    
    p_rev_pad = np.zeros(fft_size)
    p_rev_pad[:n] = p[::-1]
    P_rev = np.fft.fft(p_rev_pad)
    
    conv1 = np.fft.ifft(P * H).real[:n]       # (p*h)[0..n-1]
    conv2 = np.fft.ifft(P_rev * H).real[n-1:2*n-1]  # (p_rev*h)[n-1..2n-2]
    
    # Wait, I need (p_rev*h)[j+n-1] for j=0..n-1, which is elements [n-1, n, ..., 2n-2]
    conv2 = np.fft.ifft(P_rev * H).real
    conv2 = conv2[n-1:2*n-1]  # elements at indices n-1, n, ..., 2n-2
    
    grad = -(1.0/T) * (conv1 + conv2)
    
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h):
    n = len(h)
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) * 0.5
    return np.clip(h - tau, 0.0, 1.0)

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False) + 1.0 / n_from
    xt = np.linspace(0, 2, n_to, endpoint=False) + 1.0 / n_to
    return np.interp(xt, xf, h)

def softmin_obj_grad(h, T):
    n = len(h)
    fft_size = 1
    while fft_size < 2 * n - 1:
        fft_size *= 2
    h_pad = np.zeros(fft_size)
    h_pad[:n] = h
    H = np.fft.fft(h_pad)
    A = np.fft.ifft(H * np.conj(H)).real[:n]
    A = np.maximum(A, 0.0)
    x = -T * A
    x_max = np.max(x)
    p = np.exp(x - x_max)
    p_sum = np.sum(p)
    obj = -(1.0 / T) * (x_max + np.log(p_sum))
    p = p / p_sum
    p_pad = np.zeros(fft_size)
    p_pad[:n] = p
    P = np.fft.fft(p_pad)
    p_rev_pad = np.zeros(fft_size)
    p_rev_pad[:n] = p[::-1]
    P_rev = np.fft.fft(p_rev_pad)
    conv1 = np.fft.ifft(P * H).real[:n]
    conv2_full = np.fft.ifft(P_rev * H).real
    conv2 = conv2_full[n - 1:2 * n - 1]
    grad = -(1.0 / T) * (conv1 + conv2)
    return obj, grad

def anneal_optimize(h0, deadline, T_list=None, lr_init=0.8):
    h = h0.copy()
    n = len(h)
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    if T_list is None:
        T_list = [5, 10, 20, 40, 80, 160, 320, 640]
    lr = lr_init
    for T in T_list:
        if time.time() > deadline - 10:
            break
        steps_at_T = 0
        max_steps = 3000
        while steps_at_T < max_steps:
            if time.time() > deadline - 10:
                break
            obj, grad = softmin_obj_grad(h, T)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                break
            step = lr * grad / gnorm
            h_new = h - step
            h_new = project_h(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new
                lr = min(lr * 1.02, 3.0)
            else:
                lr *= 0.7
                h = h_new if c5_new < best_c5 + 0.005 else h
            steps_at_T += 1
            if steps_at_T % 500 == 0 and lr < 0.01:
                break
    return best_h, best_c5

def exact_subgradient_step(h, lr):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    shift = k_star - (n - 1)
    grad = np.ones(n)
    idx_plus = np.arange(n) + shift
    idx_minus = np.arange(n) - shift
    valid_p = (idx_plus >= 0) & (idx_plus < n)
    valid_m = (idx_minus >= 0) & (idx_minus < n)
    grad -= np.where(valid_p, h[idx_plus * valid_p if valid_p.any() else 0], 0.0) if valid_p.any() else 0.0
    if valid_p.any():
        grad[np.where(valid_p)[0]] -= h[idx_plus[valid_p]]
    if valid_m.any():
        grad[np.where(valid_m)[0]] -= h[idx_minus[valid_m]]
    else:
        grad -= np.where(valid_m, h[idx_minus[valid_m]], 0.0)
    h_new = h + lr * grad
    h_new = project_h(h_new)
    return h_new

def exact_polish(h0, deadline, n_steps=2000):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = 0.05
    for i in range(n_steps):
        if time.time() > deadline:
            break
        n = len(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        shift = k_star - (n - 1)
        grad = np.ones(n)
        j = np.arange(n)
        jp = j + shift
        jm = j - shift
        vp = (jp >= 0) & (jp < n)
        vm = (jm >= 0) & (jm < n)
        if vp.any():
            grad[vp] -= h[jp[vp]]
        if vm.any():
            grad[vm] -= h[jm[vm]]
        h_new = h + lr * grad
        h_new = project_h(h_new)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
        else:
            lr *= 0.85
            if lr < 0.001:
                break
    return best_h, best_c5

def random_feasible(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_h(h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 30

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    else:
        h_parent = np.full(100, 0.5)

    best_h = h_parent.copy()
    best_c5 = evaluate_c5(best_h)

    resolutions = sorted(set([len(h_parent), 448, 672, 896]))

    for n in resolutions:
        if time.time() > deadline - 120:
            break
        if n > len(best_h):
            h0 = interp_to(best_h, len(best_h), n)
            h0 = project_h(h0)
        elif n == len(best_h):
            h0 = best_h.copy()
        else:
            continue

        T_list = [5, 10, 20, 50, 100, 200, 500]
        result = anneal_optimize(h0, deadline, T_list=T_list, lr_init=0.8)
        if result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0].copy()

        if time.time() < deadline - 60:
            polished = exact_polish(best_h, deadline, n_steps=1500)
            if polished[1] < best_c5:
                best_c5 = polished[1]
                best_h = polished[0].copy()

    # Random restarts
    n_best = len(best_h)
    for attempt in range(10):
        if time.time() > deadline - 30:
            break
        h_pert = best_h + np.random.randn(n_best) * 0.05
        h_pert = project_h(h_pert)
        T_list = [20, 50, 100]
        result = anneal_optimize(h_pert, deadline, T_list=T_list, lr_init=0.5)
        if result[1] < best_c5:
            best_c5 = result[1]
            best_h = result[0].copy()
            polished = exact_polish(best_h, deadline, n_steps=500)
            if polished[1] < best_c5:
                best_c5 = polished[1]
                best_h = polished[0].copy()

    # Try from pure random

                best_c5 = result[1]

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
