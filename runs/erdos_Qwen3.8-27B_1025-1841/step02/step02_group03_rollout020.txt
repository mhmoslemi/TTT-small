                trial[j + 1] = max(0.0, min(1.0, trial[j + 1] + dk[1]))
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

def binary_snap(h, n, time_limit, start_time):
    best = h.copy()
    best_c = compute_c5(h)
    improved = True
    passes = 0
    while improved and passes < 5 and time.time() - start_time < time_limit:
        improved = False
        passes += 1
        for j in range(n):
            if time.time() - start_time > time_limit:
                break
            for val in [0.0, 1.0, 0.05, 0.1, 0.9, 0.95]:
                trial = best.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best = trial
                    best_c = c
                    improved = True
                    break
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(960, 0.5)

    best_h = make_feasible(h0, len(h0))
    best_c5 = compute_c5(best_h)
    n = len(best_h)

    # Phase 1: SLSQP at initial resolution with progressive T
    for T in [50.0, 150.0, 400.0, 1000.0, 2500.0]:
        if remaining() < 80:
            break
        h_start = best_h.copy()
        h_res = slsqp_optimize(h_start, n, T, 600)
        c5 = compute_c5(h_res)
        if c5 < best_c5:
            best_c5 = c5
            best_h = make_feasible(h_res, n)
            n = len(best_h)

    # Phase 2: Upsample and optimize at higher resolution
    for n_up in [1280, 1920]:
        if remaining() < 150:
            break
        h_up = upsample_h(best_h, n_up)
        for T in [100.0, 500.0, 1500.0]:
            if remaining() < 60:
                break
            h_res = slsqp_optimize(h_up, n_up, T, 400)
            c5 = compute_c5(h_res)
            if c5 < best_c5:
                best_c5 = c5
                best_h = make_feasible(h_res, n_up)
                n = n_up
                h_up = best_h.copy()

    # Phase 3: Coarse coordinate descent
    if remaining() > 100:
        steps_coarse = [0.3, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005]
        best_h, best_c5 = coord_descent(best_h, n, steps_coarse, 20,
                                        90, start_time)

    # Phase 4: Binary snap
    if remaining() > 60:
        best_h, best_c5 = binary_snap(best_h, n, 50, start_time)

    # Phase 5: Fine coordinate descent (multiple passes)
    if remaining() > 120:
        steps_fine = [0.01, 0.005, 0.002, 0.001]
        best_h, best_c5 = coord_descent(best_h, n, steps_fine, 30,
                                        110, start_time)

    # Phase 6: Pair coordinate descent
    if remaining() > 80:
        best_h, best_c5 = pair_cds(best_h, n, 70, start_time)

    # Phase 7: SLSQP polish with very high T
    if remaining() > 60:
        for T in [5000.0, 10000.0]:
            if remaining() < 40:
                break
            h_res = slsqp_optimize(best_h, n, T, 400)
            h_res = make_feasible(h_res, n)
            c5 = compute_c5(h_res)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_res
                n = len(best_h)

    # Phase 8: Final fine coordinate descent
    if remaining() > 40:
        steps_final = [0.005, 0.002, 0.001, 0.0005]
        best_h, best_c5 = coord_descent(best_h, n, steps_final, 20,
                                        35, start_time)

    # Phase 9: Final pair passes
    if remaining() > 20:
        best_h, best_c5 = pair_cds(best_h, n, 15, start_time)

    # Final feasibility check and evaluation
    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n
