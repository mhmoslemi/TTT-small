def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    """Compute gradient focusing on top M lags."""
    grad = np.zeros(n)
    top_idx = np.argpartition(g, -topM)[-topM:]
    for k in top_idx:
        pk = g[k]  # weight proportional to g value
        if pk < 1e-15:

        d = k - n + 1
        # For this lag d, ∂g_d/∂h[j] = dx * [(1-h[j+d])*1_{valid} - h[j-d]*1_{valid}]
        j0 = max(0, -d)

            grad[j0:j1] += (1.0 - h[j0+d:j1+d])

        j1b = min(n, n + d)
        if j0b < j1b:

    return grad * dx

def project_to_feasible(g, n):

    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)

    n = len(h)
    if n == target_n:

    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_grad_weighted(h, p, n, dx, topM=0):
    grad = np.zeros(n)
    if topM > 0 and topM < len(p):
        top_idx = np.argpartition(p, -topM)[-topM:]
        top_p = p[top_idx]
    else:
        top_idx = np.arange(len(p))
        top_p = p
    for i in range(len(top_idx)):
        pk = top_p[i]

            continue

        d = k - n + 1
        j0 = max(0, -d)
        j1 = min(n, n - d)

            grad[j0:j1] += pk * (1.0 - h[j0 + d:j1 + d])
        j0b = max(0, d)
        j1b = min(n, n + d)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - d:j1b - d]
    return grad * dx


def adam_opt(h0, n, tau, max_iters, time_limit, lr=0.03, topM=300,
             beta1=0.9, beta2=0.999, tau_anneal=0.9998):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()

    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    t = 0
    stall = 0
    patience = max(300, max_iters // 12)
    cur_tau = tau

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1

        g = compute_corr(h) * dx
        cur_max = g.max()
        if cur_max < best_c5 - 1e-14:
            best_c5 = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            cur_tau = max(cur_tau * 0.8, 2.0)
            lr *= 0.7
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break

        max_g = g.max()
        shifted = np.clip(cur_tau * (g - max_g), -500, 500)
        exp_c = np.exp(shifted)

        cur_tau = max(cur_tau * tau_anneal, 3.0)

def subgrad_polish(h0, n, max_iters, time_limit, lr_init=0.002, topK=3):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    stall = 0
    patience = 500

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            lr_init *= 0.6
            stall = 0
            if lr_init < 1e-12:
                break

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, -d)
            j1 = min(n, n - d)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[j0 + d:j1 + d])
            j0b = max(0, d)
            j1b = min(n, n + d)
            if j0b < j1b:
                grad[j0b:j1b] -= h[j0b - d:j1b - d]
        grad *= dx / topK

        cur_lr = lr_init / (1.0 + it * 0.002)
        h = project_to_feasible(h - cur_lr * grad, n)

    return best_h, best_c5


def binarize_refine(h, n):
    """Try to improve by pushing toward binary values."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for p in [2, 4, 8]:
        mask = h > 0.5
        h_bin = mask.astype(float)
        s = h_bin.sum()
        if s == 0 or s == n:
            continue
        if abs(s - n / 2) > n * 0.1:
            h_soft = h ** p / (h ** p + (1 - h) ** p + 1e-15)
            h_cand = project_to_feasible(h_soft, n)
        else:
            h_cand = project_to_feasible(h_bin + 0.1 * (h - 0.5), n)
        c5 = compute_c5(h_cand)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_cand.copy()
    return best_h, best_c5
