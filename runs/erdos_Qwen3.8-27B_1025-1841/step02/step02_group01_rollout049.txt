def smooth_obj_and_grad(h, n, temp):
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_shifted = np.exp(temp * shifted)
    sum_exp = np.sum(exp_shifted)
    w = exp_shifted / sum_exp  # weights, length 2n-1
    
    obj = (mc + np.log(sum_exp) / temp) * dx
    
    # Gradient: grad[i] = dx * sum_m w[m] * ((1-h[p]) - h[q])
    # where p = i + m - n + 1, q = i - m + n - 1
    # This is: dx * (correlate(w, h) related terms)
    
    # Let's compute it directly
    grad = np.zeros(n)
    # For each i, sum over valid m:
    # dcorr[m]/dh[i] = (1-h[p]) - h[q] where p=i+m-n+1, q=i-m+n+1
    # m ranges from 0 to 2n-2
    # p valid: 0 <= i+m-n+1 <= n-1, i.e., n-1-i <= m <= 2n-2-i
    # q valid: 0 <= i-m+n-1 <= n-1, i.e., i-n+1 <= m <= i+n-1
    
    for i in range(n):
        # p = i + m - n + 1, valid for m in [n-1-i, 2n-2-i] ∩ [0, 2n-2]
        m_lo_p = max(0, n-1-i)
        m_hi_p = min(2*n-2, 2*n-2-i)
        # q = i - m + n - 1, valid for m in [i-n+1, i+n-1] ∩ [0, 2n-2]
        m_lo_q = max(0, i-n+1)
        m_hi_q = min(2*n-2, i+n-1)
        
        # Sum w[m] * (1-h[p]) for valid p
        if m_lo_p <= m_hi_p:
            p_vals = i + np.arange(m_lo_p, m_hi_p+1) - n + 1
            grad[i] += np.dot(w[m_lo_p:m_hi_p+1], 1.0 - h[p_vals])
        
        # Sum w[m] * (-h[q]) for valid q
        if m_lo_q <= m_hi_q:
            q_vals = i - np.arange(m_lo_q, m_hi_q+1) + n - 1
            grad[i] -= np.dot(w[m_lo_q:m_hi_q+1], h[q_vals])
    
    grad *= dx
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

_C_N = 80
_C_T = 50.0

def _smooth_obj(h):
    n = _C_N
    t = _C_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(t * shifted))) / t
    return soft * dx

def _smooth_grad(h):
    n = _C_N
    t = _C_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(t * shifted)
    sum_exp = np.sum(exp_s)
    w = exp_s / sum_exp
    grad = np.zeros(n)
    for i in range(n):
        s1 = 0.0
        s2 = 0.0
        for m in range(2 * n - 1):
            p = i + m - n + 1
            if 0 <= p < n:
                s1 += w[m] * (1.0 - h[p])
            q = i - m + n - 1
            if 0 <= q < n:
                s2 += w[m] * h[q]
        grad[i] = (s1 - s2) * dx
    return grad

def _constraint_fun(x):
    return np.sum(x) - _C_N / 2.0

def _constraint_jac(x):
    return np.ones(_C_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-13:
        return h
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = np.zeros(n)
    ones = rng.choice(n, n // 2, replace=False)
    h[ones] = 1.0
    return h

    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
