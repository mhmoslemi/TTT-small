The problem is: min_h max_k corr(h, k)

This is equivalent to: min_h, t t subject to corr(h,k) ≤ t for all k, 0≤h≤1, sum(h)=n/2

This is a SDO (second-order cone) problem! We can write:
- corr(h,k) = h^T A_k (1-h) where A_k is a shift matrix

import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    if abs(target - np.sum(h)) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += (target - np.sum(h)) / np.sum(mask)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def downsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n)
    new_x = np.linspace(0, 2, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def subgrad_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    else:
        grad[:n - s] += (1.0 - h[-s:])
        grad[-s:] += (-h[:n - s])
    return grad

def smooth_max_grad(h, T, top_frac=0.15):
    corr = np.correlate(h, 1.0 - h, mode='full')
    n = len(h)
    mc = np.max(corr)
    shifted = corr - mc
    if T < 0.3:
        k_top = max(3, int(0.05 * len(corr)))
        top_idx = np.argsort(corr)[-k_top:]
        weights = np.zeros_like(corr)
        weights[top_idx] = 1.0 / len(top_idx)
    else:
        exp_s = np.exp(T * shifted)
        total = np.sum(exp_s)
        if total < 1e-300:
            idx = int(np.argmax(corr))
            weights = np.zeros(len(corr))
            weights[idx] = 1.0
        else:
            weights = exp_s / total
        n_top = max(3, int(top_frac * len(corr)))
        top_idx = np.argsort(weights)[-n_top:]
        mask = np.zeros_like(weights)
        mask[top_idx] = 1.0
        weights = weights * mask
        s = np.sum(weights)
        if s > 0:
            weights /= s
    grad = np.zeros(n)
    sig = np.where(weights > 1e-10)[0]
    for k_idx in sig:
        shift = int(k_idx) - (n - 1)
        grad += weights[k_idx] * subgrad_shift(h, shift)
    return grad

def _obj(h):
    T = _P['T']
    n = _P['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _eq(h):
    return np.sum(h) - _P['n'] / 2.0

def _eqj(h):
    return np.ones(_P['n'])

def slsqp(h, T, maxiter=400):
    n = len(h)
    _P['T'] = T
    _P['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq, 'jac': _eqj}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h
