        best_x = x.copy()
    return best_x, best_c5

def _slsqp_obj(h):
    T = _SP['T']
    n = _SP['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_cfun(h):
    return np.sum(h) - _SP['n'] * 0.5

def _slsqp_cjac(h):
    return np.ones(_SP['n'])

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n * 0.5)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n_init = 320
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        n_ones = n_init // 2
        idx = np.arange(n_init)
        rng.shuffle(idx)
        h0 = np.zeros(n_init)
        h0[idx[:n_ones]] = 1.0

    n0 = len(h0)
    h0 = project_feasible(h0, n0 * 0.5)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    print(f"Init: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: FISTA at n0 with aggressive cooling
    if time.time() - start_time < budget_s - 150:
        h1, c1 = fista(best_h, 7000, 120.0, 1.0, top_k=60, check_every=30, step_scale=0.9)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        print(f"FISTA1 n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: SLSQP at n0 (multiple T)
    if n0 <= 400 and time.time() - start_time < budget_s - 100:
        for Tv in [12.0, 4.0, 1.5]:
            if time.time() - start_time > budget_s - 40:
                break
            _SP['T'] = Tv
            _SP['n'] = n0
            bounds = [(0.0, 1.0)] * n0
            cons = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
            try:
                res = minimize(_slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 1200, 'ftol': 1e-15})
                if res.x is not None:
                    hx = project_feasible(res.x, n0 * 0.5)
                    cx = compute_c5(hx)
                    if cx < best_c5:
                        best_c5, best_h = cx, hx
            except Exception:
                pass
        print(f"SLSQP: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: Upsample to higher resolution
    for new_n in [n0 * 2]:
        if time.time() - start_time > budget_s - 250:
            break
        h_up = upsample_h(best_h, new_n)
        c_up_init = compute_c5(h_up)
        h_f, c_f = fista(h_up, 6000, 80.0, 0.5, top_k=60, check_every=40, step_scale=0.85)
        if c_f < c_up_init:
            h_up = h_f
        # SLSQP at new res if feasible
        if new_n <= 512 and time.time() - start_time < budget_s - 100:
            _SP['T'] = 6.0
            _SP['n'] = new_n
            bounds2 = [(0.0, 1.0)] * new_n
            cons2 = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
            try:
                res = minimize(_slsqp_obj, h_up, method='SLSQP', bounds=bounds2,
                              constraints=cons2, options={'maxiter': 800, 'ftol': 1e-14})
                if res.x is not None:
                    hx = project_feasible(res.x, new_n * 0.5)
                    cx = compute_c5(hx)
                    if cx < compute_c5(h_up):
                        h_up = hx
            except Exception:
                pass
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up
        print(f"Up n={new_n}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 4: Multi-start perturbations
    if time.time() - start_time < budget_s - 150:
        np_best = len(best_h)
        for scale in [0.02, 0.05, 0.1, 0.2, 0.35]:
            if time.time() - start_time > budget_s - 60:
                break
            h_p = best_h + rng.randn(np_best) * scale
            h_f, c_f = fista(h_p, 3500, 50.0, 1.0, top_k=50, check_every=40, step_scale=0.8)
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f
                print(f"  Pert {scale}: C5={best_c5:.6f}", flush=True)
        print(f"Pert: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 5: Targeted FISTA at low T (fine structure)
    if time.time() - start_time < budget_s - 40:
        h_f, c_f = fista(best_h, 5000, 4.0, 0.05, top_k=20, check_every=20, step_scale=0.7)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f

    # Phase 6: Final SLSQP squeeze (only if small enough)
    n_pts = len(best_h)
    if n_pts <= 384 and time.time() - start_time < budget_s - 25:
        _SP['T'] = 2.0
        _SP['n'] = n_pts
        bounds3 = [(0.0, 1.0)] * n_pts
        cons3 = [{'type': 'eq', 'fun': _slsqp_cfun, 'jac': _slsqp_cjac}]
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP', bounds=bounds3,
                          constraints=cons3, options={'maxiter': 1500, 'ftol': 1e-16})
            if res.x is not None:
                hx = project_feasible(res.x, n_pts * 0.5)
                cx = compute_c5(hx)
                if cx < best_c5:
                    best_c5, best_h = cx, hx
        except Exception:
            pass

    # Final validation
    best_h = project_feasible(best_h, len(best_h) * 0.5)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
