    A = xc_fP[idxs]
    
    grad = (A - B) * dx
    return obj, grad

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def project_to_feasible(g, n):
    target = n / 2.0
    lo, hi = g.min() - 2.0, g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def fft_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -50, 50)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    N = 1
    while N < 3 * n:
        N <<= 1
    
    f = 1.0 - h
    f_pad = np.zeros(N)
    f_pad[:n] = f
    g_pad = np.zeros(N)
    g_pad[:n] = h
    P_pad = np.zeros(N)
    P_pad[:2*n-1] = p
    
    f_hat = np.fft.rfft(f_pad)
    g_hat = np.fft.rfft(g_pad)
    P_hat = np.fft.rfft(P_pad)
    
    xc_fP = np.fft.irfft(f_hat * np.conj(P_hat), N)
    conv_Pg = np.fft.irfft(P_hat * g_hat, N)
    
    idxs = (np.arange(n) - (n - 1)) % N
    A = xc_fP[idxs]
    B = conv_Pg[n-1:n-1+n]
    
    grad = (A - B) * dx
    return obj, grad

def adam_smooth(h, n, tau, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    tau_cur = tau
    
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        
        obj, grad = fft_obj_grad(h, tau_cur)
        c5_now = compute_c5(h)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            # Decrease tau and reset some state
            tau_cur *= 0.7
            if tau_cur < 3:
                tau_cur = 3
            m *= 0.5
            v *= 0.5
            stall = 0
            if tau_cur <= 3.01 and lr < 1e-6:
                break
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)
    
    return best_h, best_c5

def subgrad_opt(h, n, time_limit, top_k=3, lr=0.001):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        
        idx_sorted = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[idx_sorted]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        
        grad = np.zeros(n)
        for i, k in enumerate(idx_sorted):
            sh = k - n + 1
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 < i1:
                grad[i0:i1] += w[i] * (1.0 - h[i0+sh:i1+sh])
            i0b = max(0, sh)
            i1b = min(n, sh + n)
            if i0b < i1b:
                grad[i0b:i1b] -= w[i] * h[i0b-sh:i1b-sh]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5
