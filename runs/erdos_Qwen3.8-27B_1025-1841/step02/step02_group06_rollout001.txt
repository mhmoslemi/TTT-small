    while time.time() - start < time_limit:
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur = g.max()
        if cur < best_c5 - 1e-12:
            best_c5 = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        grad, _ = softmax_grad(h, n, corr, dx, tau, topM)

        m_s = b1 * m_s + (1 - b1) * grad
        v_s = b2 * v_s + (1 - b2) * grad * grad
        mh = m_s / (1.0 - b1 ** t)
        vh = v_s / (1.0 - b2 ** t)
        h_new = h - lr * mh / (np.sqrt(vh) + eps)
        h = project_to_feasible(h_new, n)

        if stall > patience:
            lr *= 0.7
            tau *= 1.2
            m_s *= 0.5
            v_s *= 0.5
            stall = 0
            if lr < 1e-8:
                break

    return best_h, best_c5

def subgrad_optimize(h, n, time_limit, lr0, topK):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = get_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur = g.max()
        if cur < best_c5 - 1e-12:
            best_c5 = cur
            best_h = h.copy()

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            if d >= 0:
                m = n - d
                grad[:m] += (1.0 - h[d:])
                grad[d:] -= h[:m]
            else:
                dd = -d
                m = n - dd
                grad[dd:] += (1.0 - h[:m])
                grad[:m] -= h[dd:]
        grad *= dx / topK

        cur_lr = lr / (1.0 + 0.002 * (time.time() - start))
        h = project_to_feasible(h - cur_lr * grad, n)

    return best_h, best_c5

def sharpen(h, n, p):
    """Push h toward {0,1}."""
    h_safe = np.clip(h, 1e-10, 1.0 - 1e-10)
    h_sharp = h_safe ** p
    s = h_sharp.sum()
    if s < 1e-15:
        return h.copy()
    h_sharp = h_sharp * (n / 2.0) / s
    return project_to_feasible(h_sharp, n)
