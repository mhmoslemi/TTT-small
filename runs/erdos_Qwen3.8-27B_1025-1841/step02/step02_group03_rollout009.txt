        idx = np.where(h_r == 1)[0]
        order = np.argsort(h[idx])
        for k in range(s - target):
            h_r[idx[order[k]]] = 0.0
    elif s < target:
        idx = np.where(h_r == 0)[0]
        order = np.argsort(h[idx])[::-1]
        for k in range(target - s):
            h_r[idx[order[k]]] = 1.0
    return h_r

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_search(h, n, rng, time_limit, sa_temp=0.005):
    h = h.copy()
    best_c = compute_c5(h)
    stall = 0
    while time.time() < time_limit and stall < 12:
        improved = False
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        rng.shuffle(ones)
        rng.shuffle(zeros)
        for i in ones:
            if time.time() > time_limit:
                break
            h[i] = 0.0
            found = False
            for j in zeros:
                if time.time() > time_limit:
                    break
                h[j] = 1.0
                c = compute_c5(h)
                if c < best_c - 1e-15:
                    best_c = c
                    improved = True
                    found = True
                    break
                h[j] = 0.0
            if not found:
                h[i] = 1.0
            else:
                zeros = np.where(h == 0)[0]
        if not improved:
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            broke_free = False
            for _ in range(20):
                if time.time() > time_limit:
                    break
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0.0
                h[j] = 1.0
                c = compute_c5(h)
                delta = c - best_c
                if c < best_c or (sa_temp > 0 and rng.random() < np.exp(-delta / sa_temp)):
                    if c < best_c:
                        best_c = c
                    improved = True
                    broke_free = True
                    break
                h[i] = 1.0
                h[j] = 0.0
            if not broke_free:
                stall += 1
        else:
            stall = 0
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    global _G_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def rem():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n0 = len(h0)
    except Exception:
        n0 = 960
        h0 = np.full(n0, 0.5)

    h0 = make_feasible(h0, n0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: SLSQP at moderate temperatures
    for T in [60, 150, 400]:
        if rem() < 130:
            break
        _G_T = T
        bounds = [(0.0, 1.0)] * n0
        cons = [{'type': 'eq', 'fun': g_con_fun, 'jac': g_con_jac}]
        h_start = best_h + rng.randn(n0) * 0.012
        h_start = make_feasible(h_start, n0)
        try:
            res = minimize(g_obj, h_start, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 350, 'ftol': 1e-13})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, n0)
        except Exception:
            pass

    # Phase 2: Binary local search (main workhorse)
    if rem() > 350:
        h_bin = snap_binary(best_h, n0)
        search_limit = time.time() + (budget_s - 400) * 0.45
        h_bin, c_bin = binary_search(h_bin, n0, rng, search_limit, sa_temp=0.008)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()

    # Phase 3: SLSQP high-T polish
    if rem() > 120:
        for T in [800, 1500]:
            if rem() < 60:
                break
            _G_T = T
            bounds = [(0.0, 1.0)] * n0
            cons = [{'type': 'eq', 'fun': g_con_fun, 'jac': g_con_jac}]
            h_s = best_h + rng.randn(n0) * 0.005
            h_s = make_feasible(h_s, n0)
            try:
                res = minimize(g_obj, h_s, method='SLSQP', bounds=bounds,
                               constraints=cons, options={'maxiter': 250, 'ftol': 1e-14})
                c = compute_c5(res.x)
                if c < best_c5:
                    best_c5 = c
                    best_h = make_feasible(res.x, n0)
            except Exception:
                pass

    # Phase 4: Second binary search round
    if rem() > 250:
        h_bin = snap_binary(best_h, n0)
        search_limit = time.time() + (budget_s - 200) * 0.35
        h_bin, c_bin = binary_search(h_bin, n0, rng, search_limit, sa_temp=0.003)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()

    # Phase 5: Try alternate resolutions
    if rem() > 150:
        for n_alt in [n0 - 80, n0 + 80]:
            if rem() < 80:
                break
            if n_alt < 100:
                continue
            h_alt = upsample_h(best_h, n_alt)
            h_b = snap_binary(h_alt, n_alt)
            sl = time.time() + (budget_s - 100) * 0.18
            h_b, c_b = binary_search(h_b, n_alt, rng, sl, sa_temp=0.004)
            if c_b < best_c5:
                best_c5 = c_b
                best_h = h_b.copy()

    # Phase 6: Final SLSQP polish
    if rem() > 40:
        _G_T = 3000
        bounds = [(0.0, 1.0)] * len(best_h)
        cons = [{'type': 'eq', 'fun': g_con_fun, 'jac': g_con_jac}]
        try:
            res = minimize(g_obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 150, 'ftol': 1e-14})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = make_feasible(res.x, len(best_h))
        except Exception:
            pass

    # Phase 7: Final binary nudge
    if rem() > 20:
        n_f = len(best_h)
        h_f = snap_binary(best_h, n_f)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_h = h_f
            best_c5 = c_f
        sl = time.time() + (budget_s - 5) * 0.05
        best_h, best_c5 = binary_search(best_h, n_f, rng, sl, sa_temp=0.0)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
