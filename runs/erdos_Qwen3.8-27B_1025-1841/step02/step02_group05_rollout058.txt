import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return obj, grad

Wait, I need to double-check the FISTA step size. The issue is that I'm not sure what the right step size is. Let me think about it more carefully.

The objective is f(h) = dx * [max_c + (1/tau) * log(∑ exp(tau*(corr_k - max_c)))]

The gradient is ∇f = dx * ∑_k p_k * ∇corr_k

where p_k = exp(tau*(corr_k - max_c)) / ∑_m exp(tau*(corr_m - max_c))

And ∇corr_k has entries in [-1, 1].

The Hessian is ∇²f = dx * [∑_k (tau * p_k * (1-p_k) * ∇corr_k)(∇corr_k)^T + ∑_k p_k * ∇²corr_k]

The second term is the Hessian of the max (for large tau), which has spectral norm ≈ 2.
The first term is positive semidefinite (it's the covariance of the random vector ∑ p_k ∇corr_k weighted by tau).

So ||∇²f|| ≤ dx * [2 + tau * ||cov||]

For large tau, the first term dominates and ||∇²f|| ≈ dx * tau * (something). The "something" is bounded by the variance of ∇corr_k under the distribution p, which is at most 2 (since each entry is in [-1,1]).

So ||∇²f|| ≈ 2 * dx * tau = 2 * (2/n) * tau = 4*tau/n.

Thus L = 4*tau/n, and 1/L = n/(4*tau).

For n=2000, tau=50: 1/L = 2000/200 = 10.
For n=2000, tau=25: 1/L = 2000/100 = 20.
For n=3000, tau=40: 1/L = 3000/160 ≈ 19.
