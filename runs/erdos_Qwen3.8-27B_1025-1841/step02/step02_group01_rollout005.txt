import time

    return float(np.max(corr) * dx)

    best_c5 = compute_c5(h)
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                break
            old_val = h[j]
            # Try several candidate values
            candidates = [0.0, 1.0, 0.5]
            for frac in [0.25, 0.75, 0.1, 0.9, 0.15, 0.85, 0.4, 0.6]:
                candidates.append(frac)
            # Also perturb around current value
            for delta in [0.05, -0.05, 0.02, -0.02, 0.01, -0.01]:
                v = old_val + delta
                if 0.0 <= v <= 1.0:
                    candidates.append(v)

            best_c = best_c5

                if cv < -1e-9 or cv > 1.0 + 1e-9:

                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break

        # Work at a lower resolution for speed

        else:

        h_cd, c_cd = _coord_descent(h_cd, len(h_cd), 20, remaining - 20, start_time)
        if c_cd < best_c5:

                pass
        no_improve_count += 1
    return h, best_c5

def coord_descent_fast(h, n, time_limit, start_time):
    best_c5 = compute_c5(h)
    while time.time() - start_time < time_limit:
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                return h, best_c5
            old_val = h[j]
            best_c = best_c5
            best_v = old_val
            for cv in [0.0, 1.0]:
                if abs(cv - old_val) < 0.01:
                    continue
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-11:
                    best_c = c
                    best_v = cv
            for d in [0.05, -0.05, 0.02, -0.02]:
                v = old_val + d
                if v < 0 or v > 1:
                    continue
                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-11:
                    best_c = c
                    best_v = v
            if best_v != old_val:
                h[j] = best_v
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _CUR_N, _CUR_TEMP
    start_time = time.time()
    rng = np.random.RandomState(seed)

    h0 = get_initial()
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: SLSQP at parent resolution (quick)
    n_parent = len(h0)
    if n_parent < 320:
        h_work = upsample_h(h0, 320)
    else:
        h_work = h0.copy()
    n_p = len(h_work)

    for temp in [40, 80]:
        if time.time() - start_time > budget_s - 100:
            break
        h_opt = slsqp_optimize(h_work, n_p, temp, 300, 1e-10)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Phase 2: Main optimization at higher resolutions
    n_main = 512
    h_work = upsample_h(best_h, n_main)

    # Multiple SLSQP passes with different temperatures
    temps_main = [30, 60, 120, 250]
    for temp in temps_main:
        if time.time() - start_time > budget_s - 150:
            break
        if temp == 250:
            maxiter = 200
        else:
            maxiter = 400
        h_opt = slsqp_optimize(h_work, n_main, temp, maxiter, 1e-11)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()
            h_work = h_opt.copy()

    # Try a slightly perturbed restart
    if time.time() - start_time < budget_s - 150:
        h_pert = h_work + rng.randn(n_main) * 0.02
        h_pert = make_feasible(h_pert, n_main)
        h_opt = slsqp_optimize(h_pert, n_main, 100, 300, 1e-11)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Phase 3: Higher resolution
    n_hi = 800
    if time.time() - start_time < budget_s - 200:
        h_hi = upsample_h(best_h, n_hi)
        for temp in [50, 100]:
            if time.time() - start_time > budget_s - 100:
                break
            h_opt = slsqp_optimize(h_hi, n_hi, temp, 300, 1e-11)
            c_opt = compute_c5(h_opt)
            if c_opt < best_c5:
                best_c5 = c_opt
                best_h = h_opt.copy()

    # Phase 4: Coordinate descent polish at moderate resolution
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 60:
        n_cd = min(len(best_h), 512)
        if n_cd < len(best_h):
            h_cd = upsample_h(best_h, n_cd)
        else:
            h_cd = best_h.copy()
        cd_limit = min(remaining - 20, 200)
        h_cd, c_cd = coord_descent_fast(h_cd, n_cd, start_time + cd_limit, start_time)
        if c_cd < best_c5:
            best_c5 = c_cd
            if n_cd < 512:
                best_h = upsample_h(h_cd, 512)
                best_c5 = compute_c5(best_h)
            else:
                best_h = h_cd.copy()

    # Phase 5: Final SLSQP polish
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 40:
        n_f = len(best_h)
        h_opt = slsqp_optimize(best_h, n_f, 300, 200, 1e-12)
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5 = c_opt
            best_h = h_opt.copy()

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
