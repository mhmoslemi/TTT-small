    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

        else:

        return h.copy()

        delta_c[ks] += d * h[j + ks]
    return delta_c

def sa_phase(h0, n, deadline, seed):
    rng = np.random.RandomState(seed)
    h = h0.copy()
    best_h = h.copy()
    dx = 2.0 / n
    best_c5 = evaluate_c5(h)
    curr_corr = np.correlate(h, 1.0 - h, mode='full') * dx
    curr_max = float(np.max(curr_corr))
    T_init = 0.008
    T_final = 1e-7
    step = 0
    total_est = int((deadline - time.time()) * 40000)
    no_improve = 0
    restart_threshold = 200000
    while time.time() < deadline:
        frac = min(1.0, step / max(total_est, 1))
        T = max(T_init * (T_final / T_init) ** frac, T_final)
        scale_factor = 0.1 + 0.9 * (T / T_init)
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            step += 1
            continue
        d_lo = max(-h[i], h[j] - 1.0)
        d_hi = min(1.0 - h[i], h[j])
        if d_hi - d_lo < 1e-12:
            step += 1
            continue
        d = d_lo + rng.rand() * (d_hi - d_lo) * scale_factor
        if abs(d) < 1e-14:
            step += 1
            continue
        delta_c = compute_delta_c(h, n, i, j, d) * dx
        new_max = float(np.max(curr_corr + delta_c))
        delta_obj = new_max - curr_max
        if delta_obj < 0 or (T > T_final * 10 and rng.rand() < np.exp(-delta_obj / T)):
            h[i] += d
            h[j] -= d
            curr_corr += delta_c
            curr_max = new_max
            if curr_max < best_c5 - 1e-15:
                best_c5 = curr_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        step += 1
        if no_improve > restart_threshold and T > 0.001:
            h = best_h.copy()
            curr_corr = np.correlate(h, 1.0 - h, mode='full') * dx
            curr_max = float(np.max(curr_corr))
            no_improve = 0
        if step % 500000 == 0:
            check = float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))
            if abs(check - curr_max) > 1e-8:
                curr_corr = np.correlate(h, 1.0 - h, mode='full') * dx
                curr_max = float(np.max(curr_corr))
    return best_h, best_c5

def binary_local_search(h0, n, deadline):
    h = (h0 > 0.5).astype(float)
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1.0)[0]
        order = np.argsort(h0[ones_idx])
        for idx in order[:n_ones - target]:
            h[ones_idx[idx]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0.0)[0]
        order = np.argsort(-h0[zeros_idx])
        for idx in order[:target - n_ones]:
            h[zeros_idx[idx]] = 1.0
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    dx = 2.0 / n
    improved = True
    while improved and time.time() < deadline:
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        max_k = int(np.argmax(corr))
        for i in list(zeros_set):
            if time.time() > deadline:
                break
            for j in list(ones_set):
                if abs(i - j) > n // 2:
                    continue
                h[i] = 1.0
                h[j] = 0.0
                new_c5 = evaluate_c5(h)
                if new_c5 < best_c5 - 1e-14:
                    best_c5 = new_c5
                    best_h = h.copy()
                    ones_set.add(i)
                    ones_set.discard(j)
                    zeros_set.discard(i)
                    zeros_set.add(j)
                    improved = True
                    break
                else:
                    h[i] = 0.0
                    h[j] = 1.0
                    break
    return best_h, best_c5

def qr_start(n, p):
    h = np.zeros(n)
    for x in range(1, p):
        r = (x * x) % p
        idx = (r - 1) % n
        h[idx] = 1.0
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h == 1.0)[0]
        idx = ones_idx[0]
        h[idx] = 0.0
    h = project(h, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    n = len(h_start)
    h_start = project(h_start, n)
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: SLSQP refinement
    t1 = min(100, budget_s * 0.08)
    t1_end = t0 + t1
    for T in [150.0, 400.0, 800.0]:
        if time.time() > t1_end:
            break
        h_new = slsqp_opt(best_h, n, T, maxiter=150)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: SA (main optimization)
    t2_end = deadline - 150
    h_sa, c5_sa = sa_phase(best_h, n, t2_end, seed)
    if c5_sa < best_c5:
        best_c5 = c5_sa
        best_h = h_sa.copy()

    # Phase 3: Binary local search
    if time.time() < deadline - 60:
        h_bin_ls, c5_bin = binary_local_search(best_h, n, deadline - 60)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_ls.copy()

    # Phase 4: Try QR-structured start
    if time.time() < deadline - 200 and n == 336:
        h_qr = qr_start(336, 337)
        h_qr_opt = slsqp_opt(h_qr, 336, 300.0, maxiter=100)
        c5_qr = evaluate_c5(h_qr_opt)
        if c5_qr < best_c5:
            best_c5 = c5_qr
            best_h = h_qr_opt.copy()

    # Phase 5: Final SLSQP polish
    if time.time() < deadline - 5:
        for T in [300.0, 600.0]:
            if time.time() > deadline - 3:
                break
            h_new = slsqp_opt(best_h, n, T, maxiter=100)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n)
