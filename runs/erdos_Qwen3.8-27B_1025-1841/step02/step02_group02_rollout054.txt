```python
import numpy as np
import time
from scipy.optimize import minimize

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_and_grad(h, tau, lambda_sum=1000.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip((corr - max_c) / tau, -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # Smooth max value
    obj_sm = dx * (max_c + tau * np.log(s))
    
    # Sum constraint penalty
    sum_h = h.sum()
    target = n / 2.0
    obj_pen = lambda_sum * (sum_h - target) ** 2
    obj = obj_sm + obj_pen
    
    # Gradient of smooth max
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(sh, 0)
        j1 = min(n, n + sh)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(-sh, 0)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    
    # Gradient of penalty
    grad += 2.0 * lambda_sum * (sum_h - target)
    
    return obj, grad

def optimize_lbfgsb(h, tau, time_limit, lambda_sum=1000.0):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    bounds = [(0.0, 1.0)] * n
    
    def fun(h_flat):
        return smooth_obj_and_grad(h_flat, tau, lambda_sum)[0]
    
    def fun_and_grad(h_flat):
        return smooth_obj_and_grad(h_flat, tau, lambda_sum)
    
    start = time.time()
    res = minimize(fun_and_grad, h, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8})
    
    best_h = project_to_feasible(h, n)
    best_c5 = compute_c5(best_h)
    if res.success or res.fun < best_c5 / (2.0/n):
        best_h = project_to_feasible(res.x, n)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5
```

```python
    x_new = np.arange(target_n) * (2.0 / target_n)

    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)
```
