def project_to_feasible(g, n):

    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr_fft(h):
    n = len(h)
    b = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    hb = np.fft.rfft(h, N)
    bf = np.fft.rfft(b[::-1], N)
    conv = np.fft.irfft(hb * bf, N)
    return conv[:2 * n - 1][::-1]


def compute_c5(h):
    n = len(h)
    return compute_corr_fft(h).max() * (2.0 / n)


def compute_weighted_grad(h, top_idx, p_top, n, dx):
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_top[i]
        if pk < 1e-12:
            continue
        k = int(top_idx[i])
        d = k - (n - 1)
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - d:j1 - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + d:j1b + d]
    return grad * dx


def softmax_optimize(h, n, time_limit, tau_list, topM, lr_init):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr_fft(h)
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = 400
    tau_ptr = 0
    cur_tau = tau_list[0]
    tau_period = max(200, int(time_limit * 100))

    it = 0
    while True:
        elapsed = time.time() - start
        if elapsed > time_limit:
            break

        if tau_ptr < len(tau_list) - 1 and it >= (tau_ptr + 1) * tau_period:
            tau_ptr += 1
            cur_tau = tau_list[tau_ptr]

        corr = compute_corr_fft(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_sub = g[top_idx]
        else:
            top_idx = np.arange(2 * n - 1)
            g_sub = g

        max_sub = g_sub.max()
        shifted = np.clip(cur_tau * (g_sub - max_sub), -300.0, 300.0)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p_top = exp_c / s

        grad = compute_weighted_grad(h, top_idx, p_top, n, dx)
        lr_eff = lr / (1.0 + 0.0001 * it)
        v = beta * v - lr_eff * grad
        h = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.7
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        it += 1

    return best_h, best_obj


def subgrad_optimize(h, n, time_limit, lr_init, topK):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    patience = 200

    it = 0
    while True:
        if time.time() - start > time_limit:
            break
        corr = compute_corr_fft(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = int(ki) - (n - 1)
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[j0 - d:j1 - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                grad[j0b:j1b] -= h[j0b + d:j1b + d]
        grad *= dx / topK

        cur_lr = lr / (1.0 + it * 0.002)
        v = beta * v - cur_lr * grad
        h = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.7
            v *= 0.4
            stall = 0
            if lr < 1e-13:
                break
        it += 1

    return best_h, best_c5


def averaged_subgrad(h, n, time_limit, lr_init, topK):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    h_sum = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    count = 0

    it = 0
    while True:
        if time.time() - start > time_limit:
            break
        corr = compute_corr_fft(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = int(ki) - (n - 1)
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[j0 - d:j1 - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                grad[j0b:j1b] -= h[j0b + d:j1b + d]
        grad *= dx / topK

        cur_lr = lr_init / (1.0 + it * 0.01)
        h_new = project_to_feasible(h - cur_lr * grad, n)
        h_sum += h_new
        count += 1

        if count % 500 == 0:
            h_avg = project_to_feasible(h_sum / count, n)
            c5_avg = compute_c5(h_avg)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = h_avg.copy()
        h = h_new
        it += 1

    if count > 0:
        h_avg = project_to_feasible(h_sum / count, n)
        c5_avg = compute_c5(h_avg)
        if c5_avg < best_c5:
            best_c5 = c5_avg
            best_h = h_avg.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def rem():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: Optimize at initial resolution with FFT-accelerated softmax
    n1 = n_init
    if rem() > 80:
        h1, _ = softmax_optimize(best_h, n1, rem() * 0.15,
                                  tau_list=[5.0, 15.0, 40.0, 100.0],
                                  topM=200, lr_init=0.2)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Upsample to 2x and optimize
    n2 = n_init * 2
    if rem() > 120:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        h2, _ = softmax_optimize(h2, n2, rem() * 0.2,
                                  tau_list=[8.0, 25.0, 60.0, 150.0],
                                  topM=200, lr_init=0.15)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Upsample to 4x (n=14400) and optimize
    n3 = n_init * 4
    if rem() > 150:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        h3, _ = softmax_optimize(h3, n3, rem() * 0.22,
                                  tau_list=[10.0, 30.0, 80.0, 200.0],
                                  topM=150, lr_init=0.12)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Multi-start at current best resolution
    n_cur = len(best_h)
    if rem() > 100:
        attempts = 6
        t_avail = rem() * 0.2
        t_each = t_avail / attempts
        for att in range(attempts):
            if rem() < 30:
                break
            noise_scale = 0.03 * (att + 1)
            h_p = best_h + noise_scale * np.random.randn(n_cur)
            h_p = project_to_feasible(h_p, n_cur)
            h_p, _ = softmax_optimize(h_p, n_cur, min(t_each, rem() * 0.15),
                                       tau_list=[20.0, 60.0, 150.0],
                                       topM=100, lr_init=0.1)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 5: Try different initial structures at high res
    if rem() > 80:
        n_try = n_cur
        for n_half_src in [2, 3, 4]:
            if rem() < 40:
                break
            t_sub = min(rem() * 0.08, 80.0)
            # Create a multi-pulse initial guess
            h_struct = np.zeros(n_try)
            pulse_w = n_try // (2 * n_half_src)
            for i in range(n_half_src):
                center = i * (2 * pulse_w) + pulse_w // 2
                s_idx = center - pulse_w // 2
                e_idx = center + pulse_w // 2
                h_struct[s_idx:e_idx] = 1.0
            h_struct = project_to_feasible(h_struct, n_try)
            h_struct, _ = softmax_optimize(h_struct, n_try, t_sub,
                                            tau_list=[10.0, 40.0, 100.0],
                                            topM=150, lr_init=0.15)
            c5_struct = compute_c5(h_struct)
            if c5_struct < best_c5:
                best_c5 = c5_struct
                best_h = h_struct.copy()

    # Phase 6: Subgradient polish
    if rem() > 60:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = subgrad_optimize(h_f, n_f, rem() * 0.15,
                                      lr_init=0.004, topK=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 7: Averaged subgradient
    if rem() > 50:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = averaged_subgrad(h_f, n_f, rem() * 0.12,
                                      lr_init=0.008, topK=5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Final fine-tune with very small subgradient steps
    if rem() > 30:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = subgrad_optimize(h_f, n_f, rem() * 0.25,
                                      lr_init=0.001, topK=2)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
