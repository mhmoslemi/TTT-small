    n = len(h)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):

    if start_time is None:
        start_time = time.time()

    best_c5 = compute_c5(h)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def topk_obj_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if K >= len(corr):
        k_indices = np.arange(len(corr))
    else:
        k_indices = np.argpartition(corr, -K)[-K:]
    obj = corr[k_indices].mean() * dx
    grad = np.zeros(n)
    for k in k_indices:
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        if lo > hi:
            continue
        grad[lo:hi + 1] -= h[lo + k:hi + 1 + k]
        j0 = lo + k
        j1 = hi + k
        if j0 >= 0 and j1 < n:
            grad[j0:j1 + 1] += (1.0 - h[j0 - k:j1 - k + 1])
    grad *= dx / K
    return obj, grad, corr


def smooth_max_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    for k in range(len(corr)):
        pk = p[k]
        if pk < 1e-18:
            continue
        lo = max(0, -k)
        hi = min(n - 1, n - 1 - k)
        if lo > hi:
            continue
        grad[lo:hi + 1] -= h[lo + k:hi + 1 + k] * pk
        j0 = lo + k
        j1 = hi + k
        if j0 >= 0 and j1 < n:
            grad[j0:j1 + 1] += (1.0 - h[j0 - k:j1 - k + 1]) * pk
    grad *= dx
    return obj, grad, corr


def adam_optimize(h, n, lr, max_iters, time_limit, start_time, mode, param, track_best=True):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    it = 0
    stall = 0
    current_lr = lr

    while it < max_iters:
        if time.time() - start_time > time_limit:
            break
        if mode == 'topk':
            obj, grad, corr = topk_obj_grad(h, param)
        else:
            obj, grad, corr = smooth_max_obj_grad(h, param)

        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        it += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** it)
        v_hat = v / (1 - beta2 ** it)
        h_new = h - current_lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, n)

        if stall > 150:
            current_lr *= 0.5
            m *= 0.5
            v *= 0.5
            stall = 0
            if current_lr < 1e-12:
                break
        elif stall > 80:
            current_lr *= 0.7
            stall = 0

        h = h_new

    return best_h, best_c5


def subgrad_step(h, n, lr, start_time, time_limit, max_iters):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    current_lr = lr

    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        grad = np.zeros(n)
        lo = max(0, -k_star)
        hi = min(n - 1, n - 1 - k_star)
        if lo <= hi:
            grad[lo:hi + 1] -= h[lo + k_star:hi + 1 + k_star]
            j0 = lo + k_star
            j1 = hi + k_star
            if j0 >= 0 and j1 < n:
                grad[j0:j1 + 1] += (1.0 - h[j0 - k_star:j1 - k_star + 1])
        grad *= (2.0 / n)

        v = beta * v - current_lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 200:
            current_lr *= 0.5
            v *= 0.5
            stall = 0
            if current_lr < 1e-14:
                break

        h = h_new

    return best_h, best_c5
