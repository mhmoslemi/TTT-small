```python
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):

    return compute_corr(h).max() * (2.0 / n)

def project_to_feasible(g, n):

    lo = g.min() - 2.0
    hi = g.max() + 2.0

        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1) - sh
            grad[j0:j1] += pk * (1.0 - h[idx])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:

            grad[j0b:j1b] -= pk * h[idx2]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def adam_optimize(h, n, tau, max_iters, time_limit, lr=0.1):
    h = h.copy()

    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    start = time.time()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        mhat = m / (1 - beta1**t)
        vhat = v / (1 - beta2**t)
        
        h_new = h - lr * mhat / (np.sqrt(vhat) + eps)
        h = project_to_feasible(h_new, n)
        
        if stall > 200:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
    
    return best_h, best_obj

    stall = 0

        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:

            best_h = h.copy()

        j1 = min(n, sh + n)
        if j0 < j1:

            grad[j0b:j1b] -= h[idx2]

        h = project_to_feasible(h + v, n)
        
        if stall > 300:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
    
    return best_h, best_c5
```

```python
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx2 = np.arange(j0b, j1b) + sh
            grad[j0b:j1b] -= pk * h[idx2]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad
```

This skips small p[k] values, which is a big speedup for large tau.

For tau=500, typically only the top 5-20 corr values will have p[k] > 1e-10. So the gradient loop goes from 1799 iterations to ~10-20 iterations. That's a 100x speedup!

```python
import numpy as np
import time


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -600, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    thresh = 1e-12
    active = np.where(p > thresh)[0]
    grad = np.zeros(n)
    for k in active:
        pk = p[k]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1) - sh
            grad[j0:j1] += pk * (1.0 - h[idx])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx2 = np.arange(j0b, j1b) + sh
            grad[j0b:j1b] -= pk * h[idx2]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def adam_stage(h, n, tau, max_iters, time_limit, lr):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_grad(h, tau)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    stall = 0
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_grad(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        mhat = m / (1.0 - beta1 ** t)
        vhat = v / (1.0 - beta2 ** t)
        h_new = h - lr * mhat / (np.sqrt(vhat) + eps)
        h = project_to_feasible(h_new, n)
        if stall > 300:
            lr *= 0.5
            stall = 0
            if lr < 1e-12:
                break
    return best_h, best_obj


def subgrad_stage(h, n, max_iters, time_limit, lr):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1) - sh
            grad[j0:j1] += (1.0 - h[idx])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx2 = np.arange(j0b, j1b) + sh
            grad[j0b:j1b] -= h[idx2]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 500:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    h_ext = np.concatenate([h, h[:1]])
    x_ext = np.concatenate([x_old, [2.0]])
    return np.interp(x_new, x_ext, h_ext)
```
