def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(5):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)

def lbfgs_pass(h, T, pen=1e5, maxiter=400):
    N = len(h)
    dx = 2.0 / N

    def obj(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        mc = np.max(corr)
        sh = corr - mc
        ev = np.exp(T * sh)
        base = (mc + np.log(np.sum(ev)) / T) * dx
        diff = np.sum(hc) - N / 2.0
        return base + pen * diff * diff

    def grad(x):
        hc = np.clip(x, 0.0, 1.0)
        corr = np.correlate(hc, 1.0 - hc, mode='full')
        mc = np.max(corr)
        sh = corr - mc
        ev = np.exp(T * sh)
        total = np.sum(ev)
        weights = ev / total
        g = np.zeros(N)
        for s in range(1, N):
            wp = weights[N - 1 + s]
            wn = weights[N - 1 - s]
            if wp > 1e-10:
                g[s:] += wp * (1.0 - hc[:N - s])
                g[:N - s] += wp * (-hc[s:])
            if wn > 1e-10:
                g[:N - s] += wn * (1.0 - hc[s:])
                g[s:] += wn * (-hc[:N - s])
        diff = np.sum(hc) - N / 2.0
        g = g * dx + 2.0 * pen * diff
        g[(hc <= 0) | (hc >= 1)] = 0.0
        return g

    res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * N,
                   options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
    return make_feasible(res.x, N)

    best_h = h.copy()
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    s_star = k_star - (N - 1)
    for it in range(n_iter):
        lr_t = lr_start * (lr_end / lr_start) ** (it / max(n_iter - 1, 1))
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s_star = k_star - (N - 1)
        grad = np.zeros(N)
        if s_star >= 0:
            grad[s_star:] += (1.0 - h[:N - s_star])
            grad[:N - s_star] += (-h[s_star:])
        else:
            ss = -s_star
            grad[:N - ss] += (1.0 - h[ss:])
            grad[ss:] += (-h[:N - ss])
        h = make_feasible(h - lr_t * grad * dx, N)
        if (it + 1) % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    elapsed = lambda: time.time() - t0

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n_init = 320
        h0 = np.clip(0.5 + 0.3 * rng.randn(n_init), 0, 1)
        h0 = make_feasible(h0, n_init)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    print(f"Init: n={len(best_h)}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Upsample to 640
    n1 = 640
    h1 = upsample_h(best_h, n1)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5, best_h = c1, h1
    else:
        h1 = best_h.copy() if len(best_h) == n1 else upsample_h(best_h, n1)
    print(f"Up to {n1}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 2: L-BFGS-B at n=640 with decreasing T
    for T in [80.0, 30.0, 10.0, 4.0]:
        if elapsed() > budget_s - 200:
            break
        hx = lbfgs_pass(h1, T, pen=1e5, maxiter=300)
        cx = compute_c5(hx)
        if cx < best_c5:
            best_c5, best_h = cx, hx
            h1 = hx
        else:
            h1 = hx if cx < compute_c5(h1) else h1
    print(f"L-BFGS640: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 3: Nesterov GD at n=640
    if elapsed() < budget_s - 150:
        h1, c1 = nesterov_gd(h1, n_iter=3000, T_start=20.0, T_end=0.5, lr=0.05)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        else:
            h1 = best_h if len(best_h) == 640 else upsample_h(best_h, 640)
    print(f"Nest640: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 4: Subgradient polish at n=640
    if elapsed() < budget_s - 100:
        h1, c1 = subgrad_polish(h1, n_iter=4000, lr_start=0.02, lr_end=0.0005)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        else:
            h1 = best_h if len(best_h) == 640 else upsample_h(best_h, 640)
    print(f"SG640: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 5: Upsample to 1280 and optimize
    n2 = 1280
    if elapsed() < budget_s - 300:
        h2 = upsample_h(h1, n2)
        for T in [40.0, 10.0, 3.0]:
            if elapsed() > budget_s - 200:
                break
            hx = lbfgs_pass(h2, T, pen=1e5, maxiter=250)
            cx = compute_c5(hx)
            if cx < best_c5:
                best_c5, best_h = cx, hx
            h2 = hx
        if elapsed() < budget_s - 150:
            h2, c2 = nesterov_gd(h2, n_iter=3000, T_start=10.0, T_end=0.3, lr=0.04)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
            else:
                h2 = best_h if len(best_h) == n2 else upsample_h(best_h, n2)
        if elapsed() < budget_s - 100:
            h2, c2 = subgrad_polish(h2, n_iter=5000, lr_start=0.015, lr_end=0.0003)
            if c2 < best_c5:
                best_c5, best_h = c2, h2
        print(f"Res1280: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 6: Multi-start from perturbations
    n_pts = len(best_h)
    for pert in [0.02, 0.06, 0.12]:
        if elapsed() > budget_s - 80:
            break
        hp = best_h + rng.randn(n_pts) * pert
        hp = make_feasible(hp, n_pts)
        hr, cr = nesterov_gd(hp, n_iter=2000, T_start=15.0, T_end=0.3, lr=0.04)
        if cr < best_c5:
            best_c5, best_h = cr, hr
    print(f"MultiStart: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 7: Final subgradient polish at best resolution
    if elapsed() < budget_s - 30:
        n_pts = len(best_h)
        hf, cf = subgrad_polish(best_h, n_iter=5000, lr_start=0.01, lr_end=0.0002)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    print(f"Polish: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 8: If time remains, try one more L-BFGS pass at low T
    if elapsed() < budget_s - 40:
        n_pts = len(best_h)
        hf = lbfgs_pass(best_h, T=1.5, pen=1e6, maxiter=300)
        cf = compute_c5(hf)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    print(f"FinalLB: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
