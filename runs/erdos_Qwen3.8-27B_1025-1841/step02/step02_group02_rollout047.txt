def project_to_feasible(g, n):
    # Bisection to find threshold for sum constraint
    target = n / 2.0
    lo, hi = g.min() - 1.0, g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(g - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_and_grad(h, tau):

        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_to_feasible(h_new, n)
        
        if stall_count > 100:
            # Try a smaller step
            h_new = h - 0.1 * lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h_new = project_to_feasible(h_new, n)
            stall_count = 0
        
        h = h_new
    
    return best_h, best_obj

def local_search_binary(h, n, max_rounds, time_limit):
    # Try to improve by adjusting near-binary values
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for round_ in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        c5_val = corr.max() * dx
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find active shifts
        top_vals = np.sort(corr)[::-1]
        threshold = top_vals[min(20, len(top_vals)-1)]
        active = np.where(corr >= threshold)[0]
        
        improved = False
        # Try setting near-0 values to 0 and near-1 values to 1
        for i in range(n):
            if time.time() - start > time_limit:
                break
            if h[i] < 0.05:
                h[i] = 0.0
                h = project_to_feasible(h, n)
            elif h[i] > 0.95:
                h[i] = 1.0
                h = project_to_feasible(h, n)
        
        c5_val = compute_c5(h)
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            improved = True
    
    return best_h, best_c5
