def softgrad(h, T, top_k=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = corr - mc
    exp_s = np.exp(T * shifted)

        shift = idx - (n - 1)
        grad = _subgrad(h, shift)
        return grad * dx
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]

        w = weights[ki]
        if w < 1e-10:

        shift = ki - (n - 1)

def _subgrad(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:

        grad[:n-s] += (-h[s:])
    else:
        grad[:n-s] += (1.0 - h[s:])
