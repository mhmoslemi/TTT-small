def slsqp(h, t, maxiter, time_limit, start_time):
    global _OPT_N, _OPT_T
    n = len(h)
    _OPT_N = n
    _OPT_T = t
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if time.time() - start_time > time_limit:
            return h
        return make_feasible(res.x, n)
    except:
        return h

def coord_descent(h, max_sweeps=10, time_limit=100, start_time=0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    lam = 0.0
    c5 = np.max(corr) * dx
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        for j in range(n):
            if time.time() - start_time > time_limit:
                break
            old = h[j]
            # Compute slopes s[m] for all m
            m_idx = np.arange(2 * n - 1)
            idx1 = j + m_idx - (n - 1)  # j + m - (n-1)
            idx2 = j - m_idx + (n - 1)  # j - m + (n-1)
            valid1 = (idx1 >= 0) & (idx1 < n)
            valid2 = (idx2 >= 0) & (idx2 < n)
            
            h1 = np.zeros(2 * n - 1)
            h2 = np.zeros(2 * n - 1)
            h1[valid1] = h[idx1[valid1]]
            h2[valid2] = h[idx2[valid2]]
            
            s = 1.0 - h1 - h2  # slope for each lag m
            
            # Find optimal v in [0,1] minimizing max_m (corr[m] - old*s[m] + v*s[m]) * dx - lam*v
            # = max_m (corr[m] - old*s[m] + v*s[m]) * dx - lam*v
            # Let base[m] = corr[m] - old*s[m] (correlation when h[j]=0)
            base = corr - old * s
            
            # Piecewise linear convex: min_v max_m (base[m] + v*s[m]) * dx - lam*v
            # Try a grid
            grid = np.linspace(0, 1, 51)
            # Vectorized: for each v in grid, compute max_m(base + v*s)
            # This is a (51, 2n-1) matrix operation
            vals = np.max(base[None, :] + grid[:, None] * s[None, :], axis=1) * dx - lam * grid
            best_idx = np.argmin(vals)
            v_new = grid[best_idx]
            
            if v_new != old:
                # Update corr
                corr = base + v_new * s
                h[j] = v_new
        
        # Adjust lambda
        current_sum = np.sum(h)
        target_sum = n / 2.0
        lam += 0.05 * (current_sum - target_sum)
        h = make_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < c5:
            c5 = new_c5
        else:
            # Reset to best
            pass  # h is already modified, can't easily revert
    return h, c5

def stochastic_ls(h, num_steps, time_limit, start_time, rng):
    n = len(h)
    c5 = compute_c5(h)
    for step in range(num_steps):
        if time.time() - start_time > time_limit:
            break
        # Pick 2-5 random coordinates to perturb
        n_perturb = rng.randint(2, 6)
        idx = rng.choice(n, n_perturb, replace=False)
        h_trial = h.copy()
        step_size = 0.01 + 0.02 * (step / num_steps)  # decreasing step size
        h_trial[idx] += rng.randn(n_perturb) * step_size
        h_trial = make_feasible(h_trial, n)
        c5_trial = compute_c5(h_trial)
        if c5_trial <= c5 + 1e-12:  # accept equal or better (slight tolerance)
            h = h_trial
            c5 = min(c5, c5_trial)
    return h, c5

_OPT_N = 0
_OPT_T = 100.0

def _obj(h):
    n = _OPT_N
    t = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(t * shifted))) / t
    return soft * dx

def _con_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _con_jac(h):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n

            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def compute_slope_vec(h, j):
    n = len(h)
    m_idx = np.arange(2 * n - 1)
    idx1 = j + m_idx - (n - 1)
    idx2 = j - m_idx + (n - 1)
    valid1 = (idx1 >= 0) & (idx1 < n)
    valid2 = (idx2 >= 0) & (idx2 < n)
    slope = np.zeros(2 * n - 1)
    slope[valid1] += (1.0 - h[idx1[valid1]])
    slope[valid2] -= h[idx2[valid2]]
    return slope

def coord_descent(h, max_sweeps, deadline):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    top_k = 8

            break

            if time.time() > deadline:
                break
            old = h[j]
            s = compute_slope_vec(h, j)
            base = corr - old * s
            # Find candidates: top lags at v=0 and v=1, pairwise crossings
            idx0 = np.argsort(-base)[:top_k]
            idx1 = np.argsort(-(base + s))[:top_k]
            all_idx = np.unique(np.concatenate([idx0, idx1]))
            candidates = [0.0, 1.0, old]

                    ma, mb = all_idx[a], all_idx[b]

            candidates = np.array(candidates)
