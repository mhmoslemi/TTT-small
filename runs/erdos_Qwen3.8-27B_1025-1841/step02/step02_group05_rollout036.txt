import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_grad(h, p, dx):
    """Compute gradient of smooth objective given precomputed p (softmax weights)."""
    n = len(h)
    # A[j] = sum of p[k] for k in [n-1-j, 2n-2-j]
    prefix = np.zeros(2*n)
    prefix[1:] = np.cumsum(p)
    A = prefix[2*n-1] - prefix[n-1]  # this is sum of all p = 1
    # Actually A[j] = sum_{k=n-1-j}^{2n-2-j} p[k]
    # = prefix[2n-1-j] - prefix[n-1-j]
    # For j=0: prefix[2n-1] - prefix[n-1]
    # For j=n-1: prefix[n] - prefix[0]
    
    # Let me compute A for all j at once
    # A[j] = prefix[2n-1-j] - prefix[n-1-j] for j=0,...,n-1
    # indices for prefix: 2n-1, 2n-2, ..., n (for the first term)
    #                    n-1, n-2, ..., 0 (for the second term)
    idx1 = np.arange(2*n-1, n-1, -1)  # 2n-1, 2n-2, ..., n
    idx2 = np.arange(n-1, -1, -1)      # n-1, n-2, ..., 0
    A = prefix[idx1] - prefix[idx2]
    
    # B[j] = np.correlate(p, h, 'full')[n-1-j]
    corr_ph = np.correlate(p, h, mode='full')
    B = corr_ph[n-1::-1][:n]  # B[j] = corr_ph[n-1-j]
    
    # C[j] = np.convolve(p, h, 'full')[n-1+j]
    conv_ph = np.convolve(p, h, mode='full')
    C = conv_ph[n-1:2*n-1]  # C[j] = conv_ph[n-1+j]
    
    grad = dx * (A - B - C)
    return grad

def smooth_obj_grad_fast(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = smooth_grad(h, p, dx)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr

def optimize_fista(h0, n, tau_schedule, time_limit, lr=None, momentum_init=0.9):
    """FISTA-like optimization with projected gradient."""
    h = project_to_feasible(h0, n).copy()
    y = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    t_k = 1.0
    stall = 0
    
    # Generate tau schedule
    if tau_schedule is None:
        tau_schedule = [100, 50, 20, 8, 4, 2]
    
    total_steps = len(tau_schedule)
    steps_per_phase = 200  # adjust based on time
    
    for phase, tau in enumerate(tau_schedule):
        if lr is not None:
            lr_phase = lr
        else:
            lr_phase = 0.5 / tau
        
        for step in range(10000):
            if time.time() - start > time_limit:
                break
            
            obj, grad, corr = smooth_obj_grad_fast(y, tau)
            c5_now = corr.max() * (2.0/n)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > 500:
                lr_phase *= 0.7
                stall = 0
                if lr_phase < 1e-14:
                    break
            
            # FISTA update
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
            momentum = (t_k - 1.0) / t_new
            
            y_new = project_to_feasible(y - lr_phase * grad, n)
            h_new = y_new + momentum * (y_new - h)
            h_new = project_to_feasible(h_new, n)
            
            h = h_new
            y = y_new
            t_k = t_new
        
        # Reset FISTA at phase boundaries
        h = project_to_feasible(h, n)
        y = h.copy()
        t_k = 1.0
    
    return best_h, best_c5

Actually, one more idea: **alternating direction method of multipliers (ADMM)** or **proximal linearization**.

The problem is: min_h max_k f_k(h) s.t. 0≤h≤1, sum(h)=n/2

We can introduce a variable t and write: min_{h,t} t s.t. f_k(h) ≤ t for all k, 0≤h≤1, sum(h)=n/2

This is a non-convex QCQP (quadratically constrained QP). Not directly solvable with standard convex solvers.

But we can use an iterative approach:
1. Fix t, find h that minimizes the number of violated constraints
2. Fix h, update t = max_k f_k(h)
This is basically what the smooth method does.

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad_fast(h, tau):
    """Compute smooth objective and gradient efficiently."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # Compute gradient using correlate and convolve
    # A[j] = sum_{k=n-1-j}^{2n-2-j} p[k]
    prefix = np.zeros(2*n)
    prefix[1:] = np.cumsum(p)
    idx1 = np.arange(2*n-1, n-1, -1)  # 2n-1, 2n-2, ..., n (length n)
    idx2 = np.arange(n-1, -1, -1)      # n-1, n-2, ..., 0 (length n)
    A = prefix[idx1] - prefix[idx2]
    
    # B[j] = correlate(p, h, 'full')[n-1-j]
    corr_ph = np.correlate(p, h, mode='full')
    B = corr_ph[n-1::-1][:n]
    
    # C[j] = convolve(p, h, 'full')[n-1+j]
    conv_ph = np.convolve(p, h, mode='full')
    C = conv_ph[n-1:2*n-1]
    
    grad = dx * (A - B - C)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr

def fista_optimize(h0, n, tau_list, time_limit, base_lr=None):
    """FISTA optimization with tau annealing."""
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for tau in tau_list:
        if time.time() - start > time_limit:
            break
        lr = base_lr / tau if base_lr is not None else 0.5 / tau
        y = h.copy()
        t_k = 1.0
        stall = 0
        stall_patience = 300
        
        while time.time() - start < time_limit:
            obj, grad, corr = smooth_obj_grad_fast(y, tau)
            c5_now = corr.max() * (2.0 / n)
            if c5_now < best_c5 - 1e-15:
                best_c5 = c5_now
                best_h = y.copy()
                stall = 0
            else:
                stall += 1
            
            if stall > stall_patience:
                lr *= 0.7
                stall = 0
                if lr < 1e-14:
                    break
            
            # FISTA step
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k))
            momentum = (t_k - 1.0) / t_new
            
            y_new = project_to_feasible(y - lr * grad, n)
            h_new = y_new + momentum * (y_new - y)
            h_new = project_to_feasible(h_new, n)
            
            h = y_new
            y = h_new
            t_k = t_new
    
    return best_h, best_c5

def subgrad_multishift(h0, n, time_limit, top_k=3, lr_init=0.005):
    """Subgradient method using top-k shifts."""
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    lr = lr_init
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            lr *= 0.7
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break
        
        # Top-k subgradient
        idx_sorted = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[idx_sorted]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        
        grad = np.zeros(n)
        for i, k in enumerate(idx_sorted):
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0-sh:j1-sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b+sh:j1b+sh]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5
