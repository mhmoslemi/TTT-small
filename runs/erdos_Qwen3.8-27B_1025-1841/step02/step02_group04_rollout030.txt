def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary(h, n):
    h = np.asarray(h, dtype=float)
    h_bin = (h >= 0.5).astype(float)
    n_ones = int(np.sum(h_bin))

    best_h = h.copy()
    
    no_improve = 0
    patience = 20000
    
    for it in range(n_iter):
        if time_limit is not None and (time.time() - start_time) > time_limit:
            break
        
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        
        ia = rng.randint(len(ones))
        ib = rng.randint(len(zeros))
        a = ones[ia]
        b = zeros[ib]
        
        # Swap
        h[a] = 0.0
        h[b] = 1.0
        hcomp[a] = 1.0
        hcomp[b] = 0.0
        ones[ia] = b
        zeros[ib] = a
        
        corr = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(hcomp, N).conj(), N)[:2*n-1]
        new_max = float(np.max(corr))
        
        delta = current_max - new_max
        if delta >= 0 or rng.random() < np.exp(delta / T):
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[a] = 1.0
            h[b] = 0.0
            hcomp[a] = 0.0
            hcomp[b] = 1.0
            ones[ia] = a
            zeros[ib] = b
            no_improve += 1
        
        if no_improve > patience:
            # Perturb
            for _ in range(max(3, n // 30)):
                ia2 = rng.randint(len(ones))
                ib2 = rng.randint(len(zeros))
                h[ones[ia2]], h[zeros[ib2]] = h[zeros[ib2]], h[ones[ia2]]
                hcomp[ones[ia2]], hcomp[zeros[ib2]] = hcomp[zeros[ib2]], hcomp[ones[ia2]]
                tmp = ones[ia2]
                ones[ia2] = zeros[ib2]
                zeros[ib2] = tmp
            corr = np.fft.irfft(np.fft.rfft(h, N) * np.fft.rfft(hcomp, N).conj(), N)[:2*n-1]
            current_max = float(np.max(corr))
            no_improve = 0
    
    return best_h, best_max
