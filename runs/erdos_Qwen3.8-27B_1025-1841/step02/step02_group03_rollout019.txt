Let me think of a better approach:

```python
                best_h = _fix(res.x, n)
        except Exception:
            pass

    # Phase 7: Final coordinate descent on fractional values
    if rem() > 10:
        h = best_h.copy()
        for _ in range(50000):
            if rem() < 10:
                break
            improved = False
            for j in range(0, n, max(1, n // 200)):
                if rem() < 5:
                    break
                old = h[j]
                for cand in [0.0, 1.0, max(0.0, old - 0.01), min(1.0, old + 0.01)]:
                    if cand == old:
                        continue
                    h[j] = cand
                    h = _fix(h, n)
                    c2 = _c5(h)
                    if c2 < best_c - 1e-15:
                        best_c = c2
                        best_h = h.copy()
                        improved = True
                        break
                if not improved:
                    h[j] = old
                    h = _fix(h, n)
            if not improved:
                break

    best_h = _fix(best_h, n)
    best_c = _c5(best_h)
    return best_h, best_c, n
```
