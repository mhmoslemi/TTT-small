New output[t] = ∑_m h_new[m] * v_new[m+t-n+1]

Difference:
Δoutput[t] = h_new[i]*v_new[i+t-n+1] - h[i]*v[i+t-n+1] 
           + h_new[j]*v_new[j+t-n+1] - h[j]*v[j+t-n+1]

For m≠i,j: h_new[m]=h[m], and v_new[m+t-n+1]=v[m+t-n+1] UNLESS m+t-n+1 equals i or j.

           + [0*v_new[j+t-n+1] - 1*v[j+t-n+1]]
           + ∑_{m≠i,j: m+t-n+1=i} (h[m]*v_new[i] - h[m]*v[i])
           + ∑_{m≠i,j: m+t-n+1=j} (h[m]*v_new[j] - h[m]*v[j])

The sums are over m such that m+t-n+1=i (i.e., m=i-t+n-1) and m≠i,j,
and m such that m+t-n+1=j (i.e., m=j-t+n-1) and m≠i,j.

For the first sum (m=i-t+n-1, m≠i,j, and 0≤m≤n-1):
If m is valid and m≠i,j: h[m]*(v_new[i]-v[i]) = h[m]*(-1) = -h[m]
(only one term if m is in range and m≠i,j)

For the second sum (m=j-t+n-1, m≠i,j, and 0≤m≤n-1):
If m is valid and m≠i,j: h[m]*(v_new[j]-v[j]) = h[m]*(+1) = h[m]
(only one term if m is in range and m≠i,j)

So:
Δoutput[t] = v_new[i+t-n+1] - v[j+t-n+1] - [h[i-t+n-1] if valid and ≠i,j] + [h[j-t+n-1] if valid and ≠i,j]

Where v_new[i+t-n+1] = v[i+t-n+1] - 1 if i+t-n+1=j (i.e., t=j-i), else v[i+t-n+1]
And v[j+t-n+1] = v[j+t-n+1] + 1 if j+t-n+1=i (i.e., t=i-j), else v[j+t-n+1]

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_at_max(h, k_max):
    """Gradient of correlation at shift k_max w.r.t. h."""
    n = len(h)
    v = 1.0 - h
    shift = k_max - (n - 1)  # actual shift
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        # corr = sum_{i=0}^{n-1-s} h[i] * v[i+s]
        grad[:n-s] += v[s:]      # d/dh[i] of h[i]*v[i+s]
        grad[s:] += -h[:n-s]     # d/dh[i] of h[i-s]*v[i] (via v[i] = 1-h[i])
    else:
        # shift < 0, s = -shift
        # corr = sum_{i=0}^{n-1-s} h[i+s] * v[i]
        grad[s:] += v[:n-s]      # d/dh[i] of h[i]*v[i-s]
        grad[:n-s] += -h[s:]     # d/dh[i] of h[i+s]*v[i] (via v[i])
    return grad

def grad_descent_momentum(h, n_iter, lr=0.05, momentum=0.9, T_starts=None):

    n = len(h)

    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        T = T0 * (T_end / T0) ** (it / max(n_iter - 1, 1))
        current_c5 = compute_c5(h)  # This is expensive; track it instead

        zeros_idx = np.where(h < 0.5)[0]

        best_delta = 0  # best improvement (negative = improvement)
        best_swap = None
        for _ in range(min(K, len(ones_idx))):

            j = ones_idx[np.random.randint(len(ones_idx))]

            new_c5 = compute_c5(h)
            delta = new_c5 - current_c5
            if delta < best_delta:
                best_delta = delta
                best_swap = (i, j, h[i], h[j])
            h[i], h[j] = 1.0 - h[j], 1.0 - h[i]  # undo
        
        if best_swap is not None:
            i, j, _, _ = best_swap
            h[i], h[j] = 1.0 - h[j], 1.0 - h[i]
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # No improvement found; try a random move with annealing
            i = zeros_idx[np.random.randint(len(zeros_idx))]
            j = ones_idx[np.random.randint(len(ones_idx))]
            h[i], h[j] = 1.0 - h[j], 1.0 - h[i]
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            else:
                # Reject (or accept with prob)
                delta = current_c5 - best_c5
                if np.random.rand() > np.exp(-delta / max(T, 1e-8)):
                    h[i], h[j] = h[j], h[i]  # undo
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad(h, k_max, n):
    """Subgradient of correlation at lag k_max."""
    v = 1.0 - h
    shift = k_max - (n - 1)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[:n-s] += v[s:]
        g[s:] += -h[:n-s]
    else:
        g[s:] += v[:n-s]
        g[:n-s] += -h[s:]
    return g

def soft_subgrad(h, T, top_k=10):
    """Weighted subgradient using top-k max lags."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k = int(np.argmax(corr))
        return subgrad(h, k, n)
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]
    g = np.zeros(n)
    for ki in idx:
        w = weights[ki]
        if w < 1e-10:
            break
        k = ki - (n - 1)
        g += w * subgrad(h, k, n)
    return g

def gd_momentum(h, n_iter, lr=0.05, mom=0.85, T0=20, T1=2, check=50):
    """Gradient descent with Nesterov momentum."""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        g = soft_subgrad(h, T) * dx
        v = mom * v - lr_t * g
        h = make_feasible(h + v, n)
        if (it + 1) % check == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5, best_h = c5, h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5, best_h = c5, h.copy()
    return best_h, best_c5

def binary_sa(h, n_steps, T0=0.005, Tend=0.0001, rng_seed=42):
    """Simulated annealing on binary sequence with single-swap moves."""
    rng = np.random.RandomState(rng_seed)
    n = len(h)
    h = (h > 0.5).astype(float).copy()
    # Ensure correct count
    n_ones = int(np.sum(h))
    target = n // 2
    if n_ones != target:
        if n_ones > target:
            idx = rng.choice(np.where(h == 1)[0], n_ones - target, replace=False)
            h[idx] = 0
        else:
            idx = rng.choice(np.where(h == 0)[0], target - n_ones, replace=False)
            h[idx] = 1
    
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(n_steps):
        T = T0 * (Tend / T0) ** (it / max(n_steps - 1, 1))
        # Random swap
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]
        h[i], h[j] = 1.0, 0.0
        new_c5 = compute_c5(h)
        delta = new_c5 - current_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5, best_h = current_c5, h.copy()
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
        else:
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _subgrad_one(h, k_max, n):
    v = 1.0 - h
    shift = k_max - (n - 1)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[:n - s] += v[s:]
        g[s:] += -h[:n - s]
    else:
        g[s:] += v[:n - s]
        g[:n - s] += -h[s:]
    return g

def _soft_subgrad(h, T, top_k=12):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k = int(np.argmax(corr))
        return _subgrad_one(h, k, n)
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]
    g = np.zeros(n)
    for ki in idx:
        w = weights[ki]
        if w < 1e-12:
            break
        k = ki - (n - 1)
        g += w * _subgrad_one(h, k, n)
    return g

def coordinate_descent(h, n_sweeps=5, n=0):

            a_k = np.zeros(2 * n - 1)

            hg, cg = gd_momentum(hs, 800, lr=0.03, mom=0.8, T0=15.0, T1=2.0, check=50)
            if cg < best_c5:
                best_c5, best_h = cg, hg
                print(f"  Structured: C5={best_c5:.6f}", flush=True)
        print(f"Phase2: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 3: Binary SA (main improvement)
    if time_left(60):
        h_bin = (best_h > 0.5).astype(float)
        h_bin = _fix_binary_count(h_bin, n0, rng)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5:
            best_c5, best_h = c_bin, h_bin.copy()

        sa_budget = min(250, int(time_left(60)))
        n_steps = int(sa_budget / 0.0008)
        n_steps = min(n_steps, 200000)
        print(f"SA: {n_steps} steps, budget={sa_budget}s", flush=True)
        h_sa, c_sa = binary_sa(best_h, n_steps, T0=0.004, Tend=0.00003, rng_seed=seed + 1)
        if c_sa < best_c5:
            best_c5, best_h = c_sa, h_sa
        print(f"SABinary: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

        # Second SA round from perturbed best
        if time_left(40):
            h_pert = best_h.copy()
            n_flip = max(5, n0 // 20)
            flip_idx = rng.choice(n0, n_flip, replace=False)
            h_pert[flip_idx] = 1.0 - h_pert[flip_idx]
            h_pert = _fix_binary_count(h_pert, n0, rng)
            n_steps2 = min(50000, int(min(80, time_left(20)) / 0.0008))
            h_sa2, c_sa2 = binary_sa(h_pert, n_steps2, T0=0.003, Tend=0.00005, rng_seed=seed + 2)
            if c_sa2 < best_c5:
                best_c5, best_h = c_sa2, h_sa2
                print(f"SA2: C5={best_c5:.6f}", flush=True)

    # Phase 4: GD fine-tuning
    if time_left(30):
        h4, c4 = gd_momentum(best_h, 3000, lr=0.015, mom=0.8, T0=5.0, T1=0.3, check=50)
        if c4 < best_c5:
            best_c5, best_h = c4, h4
        print(f"GD2: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 5: SLSQP
    if time_left(50):
        n_pts = len(best_h)
        if n_pts <= 400:
            hx, cx = slsqp_solve(best_h, n_pts, T_val=5.0, maxiter=300)
            if cx < best_c5:
                best_c5, best_h = cx, hx
            if time_left(20):
                hx, cx = slsqp_solve(best_h, n_pts, T_val=2.0, maxiter=300)
                if cx < best_c5:
                    best_c5, best_h = cx, hx
        print(f"SLSQP: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 6: Coordinate descent
    if time_left(40):
        n_pts = len(best_h)
        if n_pts <= 320:
            hc, cc = coordinate_descent(best_h, n_sweeps=3)
            if cc < best_c5:
                best_c5, best_h = cc, hc
                print(f"CoordDesc: C5={best_c5:.6f}", flush=True)

    # Phase 7: Upsample to higher resolution
    if time_left(80):
        new_n = min(n0 * 2, 640)
        h_up = upsample_h(best_h, new_n)
        h_up2, c_up = gd_momentum(h_up, 3000, lr=0.015, mom=0.85, T0=15.0, T1=0.5, check=40)
        c_base = compute_c5(h_up)
        if c_up < c_base:
            h_up = h_up2
        c_final = compute_c5(h_up)
        if c_final < best_c5:
            best_c5, best_h = c_final, h_up
        print(f"Upsample n={new_n}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

        # Binary SA at higher res
        if time_left(30):
            h_b2 = (best_h > 0.5).astype(float)
            h_b2 = _fix_binary_count(h_b2, new_n, rng)
            c_b2 = compute_c5(h_b2)
            if c_b2 < best_c5:
                best_c5, best_h = c_b2, h_b2.copy()
            n_steps3 = min(30000, int(min(30, time_left(10)) / 0.0015))
            if n_steps3 > 100:
                h_sa3, c_sa3 = binary_sa(best_h, n_steps3, T0=0.002, Tend=0.00005, rng_seed=seed + 3)
                if c_sa3 < best_c5:
                    best_c5, best_h = c_sa3, h_sa3
                print(f"SA_high: C5={best_c5:.6f}", flush=True)

    # Phase 8: Final GD polish
    if time_left(15):
        n_pts = len(best_h)
        h_f, c_f = gd_momentum(best_h, 2000, lr=0.008, mom=0.7, T0=3.0, T1=0.1, check=20)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
```
