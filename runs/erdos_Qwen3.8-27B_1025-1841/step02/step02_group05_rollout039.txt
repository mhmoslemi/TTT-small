    n = len(h)
    pos = np.zeros(n)  # B[j] = sum_k p[k] * h[j+lag_k] (valid)
    neg = np.zeros(n)  # C[j] = sum_k p[k] * h[j-lag_k] (valid)
    
    for k in range(2*n - 1):

        if wk < 1e-14:
            continue
        lag = k - n + 1
        if lag >= 0:
            pos[:n-lag] += wk * h[lag:]
            neg[lag:] += wk * h[:n-lag]
        else:
            pos[-lag:] += wk * h[:n+lag]
            neg[:-lag] += wk * h[-lag:]

        hi = min(2*n-2, n-1+j) + 1

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))

def smooth_obj_grad(h, tau):
    """Compute smooth objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(corr.max())
    
    # Softmax weights
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # Compute gradient components
    pos = np.zeros(n)
    neg = np.zeros(n)
    
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-14:
            continue
        lag = k - n + 1
        if lag >= 0:
            pos[:n - lag] += wk * h[lag:]
            neg[lag:] += wk * h[:n - lag]
        else:
            pos[-lag:] += wk * h[:n + lag]
            neg[:-lag] += wk * h[-lag:]
    
    # A[j] via cumsum
    p_cs = np.concatenate([[0.0], np.cumsum(p)])
    A = np.zeros(n)
    for j in range(n):
        lo_idx = max(0, n - 1 - j)
        hi_idx = min(2 * n - 2, n - 1 + j) + 1
        A[j] = p_cs[hi_idx] - p_cs[lo_idx]
    
    grad = dx * (A - pos - neg)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def adam_smooth_optimize(h, n, time_limit, tau0, tau_min, lr=0.1):
    """
    Smooth optimization with Adam and adaptive tau cooling.
    """
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    
    tau = tau0
    cool_factor = 0.9995  # cool tau slowly
    stall = 0
    
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        if tau < tau_min:
            tau = tau_min
        
        obj, grad = smooth_obj_grad(h, tau)
        
        # Check C5
        if it % 5 == 0:  # don't compute every iteration for speed
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-12:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        step = -lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h + step, n)
        
        # Cool tau
        tau *= cool_factor
        
        # If stalled, restart with higher tau
        if stall > 50 and tau > tau_min * 2:
            tau = min(tau0 * 0.5, tau * 2)
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
            stall = 0
    
    return best_h, best_c5

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = float(g.min()) - 2.0
    hi = float(g.max()) + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * (2.0 / n))


def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(corr.max())
    shifted = np.clip(tau * (corr - max_c), -400.0, 400.0)
    exp_c = np.exp(shifted)
    s = float(exp_c.sum())
    p = exp_c / s
    pos = np.zeros(n)
    neg = np.zeros(n)
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-13:
            continue
        lag = k - n + 1
        if lag >= 0:
            pos[:n - lag] += wk * h[lag:]
            neg[lag:] += wk * h[:n - lag]
        else:
            pos[-lag:] += wk * h[:n + lag]
            neg[:-lag] += wk * h[-lag:]
    p_cs = np.concatenate([[0.0], np.cumsum(p)])
    js = np.arange(n)
    lo_idx = np.maximum(0, n - 1 - js)
    hi_idx = np.minimum(2 * n - 1, n + js)
    A = p_cs[hi_idx] - p_cs[lo_idx]
    grad = dx * (A - pos - neg)
    obj = dx * (max_c + np.log(s) / tau)
    c5 = max_c * dx
    return obj, grad, c5


def adam_smooth(h, n, time_limit, tau, lr, momentum=0.9, max_iters=500000):
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    d1 = np.zeros(n)
    d2 = np.zeros(n)
    beta = momentum
    stall = 0
    prev_obj = None
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad, c5_now = smooth_obj_and_grad(h, tau)
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            lr *= 0.5
            d1 *= 0.3
            d2 *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        d1 = beta * d1 + (1.0 - beta) * grad
        d2 = beta * d2 + (1.0 - beta) * (grad - d1) ** 2
        step = -lr * d1 / (np.sqrt(d2) + 1e-8)
        h = project_to_feasible(h + step, n)
    return best_h, best_c5


def subgrad_optimize(h, n, time_limit, top_k=3, lr=None):
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.003
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(corr.max()) * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.65
            v *= 0.4
            stall = 0
            if lr < 1e-15:
                break
        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for i, ki in enumerate(top_idx):
            wk = w[i]
            if wk < 1e-12:
                continue
            lag = ki - n + 1
            if lag >= 0:
                grad[:n - lag] += wk * (1.0 - h[lag:])
                grad[lag:] -= wk * h[:n - lag]
            else:
                grad[-lag:] += wk * (1.0 - h[:n + lag])
                grad[:-lag] -= wk * h[-lag:]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5
