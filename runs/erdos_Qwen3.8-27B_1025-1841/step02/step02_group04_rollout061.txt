import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max_grad(corr, T, top_k=20):
    """Compute soft-max weights and top-k indices."""
    mc = np.max(corr)
    shifted = corr - mc
    # For numerical stability
    if T > 100:
        exp_s = np.exp(T * shifted)
    else:
        exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        idx = np.argmax(corr)
        return np.array([1.0]), np.array([idx])
    weights = exp_s / total
    # Get top-k by weight
    k = min(top_k, len(weights))
    top_idx = np.argsort(weights)[-k:][::-1]
    top_w = weights[top_idx]
    # Re-normalize
    top_w = top_w / np.sum(top_w)
    return top_w, top_idx

def compute_grad(h, T, top_k=20):
    """Compute gradient of soft-max overlap objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    w, idx = smooth_max_grad(corr, T, top_k)
    
    grad = np.zeros(n)
    n_half = n - 1
    for wi, mi in zip(w, idx):
        if wi < 1e-12:
            continue
        lag = mi - n_half
        # d(corr[mi])/dh[j] = (1-h[j+lag]) for valid j+lag, -h[j-lag] for valid j-lag
        if lag >= 0:
            # j+lag valid: j < n-lag → j in [0, n-1-lag]
            # j-lag valid: j >= lag → j in [lag, n-1]
            valid1 = np.arange(0, n - lag)
            grad[valid1] += wi * (1.0 - h[valid1 + lag])
            valid2 = np.arange(lag, n)
            grad[valid2] += wi * (-h[valid2 - lag])
        else:
            lag2 = -lag
            # j+lag valid: j >= lag2 → j in [lag2, n-1]
            # j-lag valid: j < n-lag2 → j in [0, n-1-lag2]
            valid1 = np.arange(lag2, n)
            grad[valid1] += wi * (1.0 - h[valid1 + lag])  # valid1 + lag = valid1 - lag2
            valid2 = np.arange(0, n - lag2)
            grad[valid2] += wi * (-h[valid2 - lag])  # valid2 - lag = valid2 + lag2
    return grad * dx

def make_feasible(h, n):
    """Project onto {0<=h<=1, sum=h=n/2} using water-filling."""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    # Binary search on the offset
    lo, hi = -1.0, 1.0  # or tighter bounds
    for _ in range(60):
        mid = (lo + hi) / 2
        h_c = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(h_c)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Final exact adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    top_w = top_w / top_w.sum()

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    lo, hi = -1.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hc = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(hc)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=float)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def _grad_at_lag(h, lag):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        if lag < n:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
    else:
        l2 = -lag
        if l2 < n:
            grad[l2:] += (1.0 - h[:-l2])
            grad[:n - l2] -= h[lag + n - l2 - (n - l2)]  # fix below
            grad[:n - l2] -= h[l2:]
    return grad

def subgrad_exact(h, top_k=1):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    thr = mc - 1e-10
    active = np.where(corr >= thr)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    active = active[-top_k:]
    grad = np.zeros(n)
    for mi in active:
        lag = mi - (n - 1)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            l2 = -lag
            grad[l2:] += (1.0 - h[:-l2])
            grad[:n - l2] -= h[l2:]
    return grad / len(active) * dx

def smooth_grad(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        mi = int(np.argmax(corr))
        lag = mi - (n - 1)
        grad = np.zeros(n)
        if lag >= 0:
            grad[:n - lag] += (1.0 - h[lag:])
            grad[lag:] -= h[:n - lag]
        else:
            l2 = -lag
            grad[l2:] += (1.0 - h[:-l2])
            grad[:n - l2] -= h[l2:]
        return grad * dx
    weights = exp_s / total
    k = min(top_k, len(weights))
    top_idx = np.argsort(weights)[-k:][::-1]
    top_w = weights[top_idx].copy()
    top_w /= top_w.sum()
    grad = np.zeros(n)
    for wi, mi in zip(top_w, top_idx):
        if wi < 1e-12:
            continue
        lag = mi - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wi * (1.0 - h[lag:])
            grad[lag:] -= wi * h[:n - lag]
        else:
            l2 = -lag
            grad[l2:] += wi * (1.0 - h[:-l2])
            grad[:n - l2] -= wi * h[l2:]
    return grad * dx

def adam_opt(h, n_iter, T_start, T_end, lr, top_k=30, use_subgrad_tail=200):
    n = len(h)
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        if T < 0.3 and use_subgrad_tail > 0 it <= n_iter - use_subgrad_tail:
            pass
        g = smooth_grad(h, T, top_k)
        m = b1 * m + (1.0 - b1) * g
        v = b2 * v + (1.0 - b2) * g * g
        m_hat = m / (1.0 - b1 ** (it + 1))
        v_hat = v / (1.0 - b2 ** (it + 1))
        h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, n)
        if (it + 1) % 150 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def subgrad_opt(h, n_iter, lr_start, lr_end, top_k=3):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        lr_t = lr_start * (lr_end / lr_start) ** frac
        g = subgrad_exact(h, top_k)
        h = h - lr_t * g
        h = make_feasible(h, n)
        if (it + 1) % 100 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
