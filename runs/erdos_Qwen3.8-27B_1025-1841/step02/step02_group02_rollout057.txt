import time

def project_to_feasible(g, n):
    target = n / 2.0
    s = np.sort(g)
    # Binary search for the threshold
    lo = s[0] - 1.0
    hi = s[-1] + 1.0
    for _ in range(60):

        total = np.sum(np.clip(g - mid, 0.0, 1.0))
        if total > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -350, 350)
    exp_c = np.exp(shifted)
    s = exp_c.sum()

    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    obj = max_c * dx
    
    shifted = np.clip(tau * (corr - max_c), -350, 350)
    exp_c = np.exp(shifted)
    s_exp = exp_c.sum()
    p = exp_c / s_exp
    
    if top_k < 2 * n - 1:
        idx = np.argsort(p)[::-1][:top_k]
    else:
        idx = np.arange(2 * n - 1)
    
    p_sel = p[idx]
    p_sum = p_sel.sum()
    if p_sum < 1e-30:
        return obj, np.zeros(n)
    p_sel = p_sel / p_sum
    
    grad = np.zeros(n)
    for i in range(len(idx)):
        k = int(idx[i])
        pk = p_sel[i]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        lo1 = max(0, -sh)
        hi1 = min(n, n - sh)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1+sh:hi1+sh])
        lo2 = max(0, sh)
        hi2 = min(n, n + sh)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2-sh:hi2-sh]
    grad *= dx
    return obj, grad

def momentum_optimize(h, n, tau, top_k, max_iters, time_limit, lr_init):

    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    stall = 0
    start = time.time()

        if time.time() - start > time_limit:
            break
        obj, grad = smooth_grad(h, tau, top_k)
        if obj < best_c5 - 1e-13:
            best_c5 = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        h_new = project_to_feasible(h + v, n)
        h = h_new
        
        if stall > 80:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        if stall > 3000:
            break
    
    return best_h, best_c5

    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()

    lr = lr_init
    stall = 0

            break
        max_c = corr.max()
        # Use top n_active lags
        top_idx = np.argsort(corr)[::-1][:n_active]
        # Normalize by count
        grad = np.zeros(n)
        for k in top_idx:
            sh = int(k) - n + 1
            lo1 = max(0, -sh)
            hi1 = min(n, n - sh)
            if lo1 < hi1:
                grad[lo1:hi1] += (1.0 - h[lo1+sh:hi1+sh])
            lo2 = max(0, sh)
            hi2 = min(n, n + sh)
            if lo2 < hi2:
                grad[lo2:hi2] -= h[lo2-sh:hi2-sh]
        grad *= dx / n_active
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        corr = compute_corr(h_new)
        c5_new = corr.max() * dx

            v *= 0.5
            stall = 0
            if lr < 1e-12:

def interp_periodic(h, target_n):

    x_new = np.arange(target_n) * (2.0 / target_n)

        h_try = h_alt if compute_c5(h_alt) < compute_c5(h_alt_comp) else h_alt_comp
        top_kAlt = max(30, nf // 15)
        tl = min(25, get_time_left(t0, budget_s) * 0.3)
        h_try, c5_try = momentum_optimize(h_try, nf, 10, top_kAlt,
                                           max_iters=8000, time_limit=tl,
                                           lr_init=0.2)
        if get_time_left(t0, budget_s) > 15:
            h_try, c5_try2 = subgrad_polish(h_try, nf, max_iters=20000,
                                             time_limit=min(12, get_time_left(t0, budget_s) * 0.2),
                                             lr_init=0.003, n_active=3)
            if c5_try2 < c5_try:
                c5_try = c5_try2
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()

    # Final feasibility and evaluation
    nf = len(best_h)
    best_h = project_to_feasible(best_h, nf)
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, best_c5, nf
