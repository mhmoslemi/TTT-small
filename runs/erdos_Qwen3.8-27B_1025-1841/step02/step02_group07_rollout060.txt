import time

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        d = target - s
        if abs(d) < 1e-14:

        if d > 0:

        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def symmetrize(h, n):
    """Enforce h[i] = h[n-1-i]"""
    return (h + h[::-1]) / 2.0

def to_full_sym(h_half, n_half):
    """Expand symmetric half to full"""
    n_full = 2 * n_half
    full = np.empty(n_full)
    full[:n_half] = h_half
    full[n_half:] = h_half[::-1]
    return full

def subgrad_max_corr(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        m = -lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g, c[k_star]

    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    mx = np.max(scaled)

        if cnt == 0:

        h_half[mask] += d / cnt

def evaluate_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        step = d / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_corr(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)

    else:
        m = -lag
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    return g, c[k_star]

    if denom < 1e-30:
        return h, 0.0
    lr = alpha * (f_now - f_best) / denom
    h_new = h - lr * fg
    return h_new, lr

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    mx = np.max(scaled)
    return float((mx + np.log(np.sum(np.exp(scaled - mx)))) / T * dx)

The 1D problem: min_t max_k (A_k + B_k * t)
This is minimized at the point where the maximum is balanced. We can solve it by:
- Binary search on the value V: check if there exists t ∈ [0,1] such that A_k + B_k * t ≤ V for all k.
  This means: for all k, A_k + B_k * t ≤ V
  → for B_k > 0: t ≤ (V - A_k) / B_k
  → for B_k < 0: t ≥ (V - A_k) / B_k
  → for B_k = 0: need A_k ≤ V
  
  So we need: max_{k: B_k<0} (V-A_k)/B_k ≤ t ≤ min_{k: B_k>0} (V-A_k)/B_k
  And all A_k ≤ V for B_k=0.
  And t ∈ [0,1].

This is a feasibility check in 1D. We can binary search on V.

But doing this for all 2n-1 k values for each coordinate is O(n * n) = O(n²) per coordinate, and O(n³) total per sweep. For n=336, that's about 38 million operations per sweep. With maybe 100 sweeps, that's 3.8 billion - too slow.

Optimization: for each coordinate i, B_k is non-zero for only O(n) values of k (specifically, k = i-j and k = i-j+n-1 for various j, but really only the lags where h[i] participates). Actually, h[i] participates in c[k] for all k where i appears as either the "a" or "b" index. That's all k from -(n-1) to n-1. So it's O(n) non-zero B_k's.

Hmm, that's still O(n²) per sweep. For n=336, that's 113K per coordinate, 38M per sweep. In Python with numpy, maybe 1-2 seconds per sweep. 100 sweeps = 100-200 seconds. That could work!

  Wait no. c[k] = sum_j h[j+k]*(1-h[j]). h[i] appears when j+k=i (as h[j+k]) or j=i (in (1-h[j])).
  
  As h[j+k]=h[i]: j = i-k. Valid if 0≤i-k<n and 0≤j<n. Contribution to B: (1-h[j]) = (1-h[i-k]).
  As (1-h[j])=(1-h[i]): j=i. Valid if 0≤i<n and 0≤i+k<n. Contribution to B: -h[i+k].

So B[k] = (1-h[i-k])*valid1 - h[i+k]*valid2 where:
valid1 = 1 if 0≤i-k<n (i.e., i-n<k≤i)
valid2 = 1 if 0≤i+k<n (i.e., -i<k<n-i)

This can be computed vectorized for all k at once.

Then the 1D problem: min_t max_k (c[k] - B[k]*h[i] + B[k]*t) = min_t max_k (c_new[k])
where c_new[k] = c[k] + B[k]*(t - h[i])

This is min_t max_k (c[k] + B[k]*t - B[k]*h[i])

Let me define D[k] = c[k] - B[k]*h[i] (the "base" without h[i]'s contribution)
Then we want min_t max_k (D[k] + B[k]*t) over t ∈ [0,1].

This is a 1D convex minimization. The optimal t* can be found by:
- If B[k] is all ≤ 0: t* = 0 (decreasing or flat)
- If B[k] is all ≥ 0: t* = 1 (increasing or flat)
- Otherwise: find the t where the maximum is balanced.

A simple approach: ternary search or binary search on the subgradient.
The derivative of max_k (D[k]+B[k]*t) w.r.t. t is B[k*] where k* is argmax. So:
- If B[argmax] > 0: increase t
- If B[argmax] < 0: decrease t
- If B[argmax] = 0: at optimum (or flat)

But this converges linearly. Better: binary search on the value.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_box_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_corr(h):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(c))
    lag = k_star - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        m = -lag
        g[:n - m] += (1.0 - h[m:])
        g[m:] -= h[:n - m]
    return g, c[k_star]

def smooth_subgrad(h, T_weight):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    weights = np.exp((c - c_max) * T_weight)
    w_sum = np.sum(weights)
    if w_sum < 1e-30:
        g, _ = subgrad_corr(h)
        return g, c_max
    weights /= w_sum
    nz = np.where(weights > 1e-8)[0]
    g = np.zeros(n)
    for k_idx in nz:
        lag = k_idx - (n - 1)
        gk = np.zeros(n)
        if lag >= 0:
            gk[lag:] += (1.0 - h[:n - lag])
            gk[:n - lag] -= h[lag:]
        else:
            m = -lag
            gk[:n - m] += (1.0 - h[m:])
            gk[m:] -= h[:n - m]
        g += weights[k_idx] * gk
    return g, c_max

def projected_subgrad(h0, n, n_iter, deadline, use_sym=False, T_weight=5.0):
    h = h0.copy()
    target = n / 2.0
    corr0 = np.correlate(h, 1.0 - h, mode='full')
    f_best = float(np.max(corr0))
    h_best = h.copy()
    f_lower = 0.0
    lr_base = 0.5
    for it in range(n_iter):
        if time.time() > deadline:
            break
        g, f_now = smooth_subgrad(h, T_weight)
        norm_g = np.linalg.norm(g)
        if norm_g < 1e-15:
            break
        gap = f_now - f_lower
        if gap < 1e-12:
            lr = lr_base / (1.0 + 0.01 * it)
        else:
            lr = np.sqrt(gap) / norm_g
        lr = min(lr, 0.5)
        h_new = h - lr * g
        if use_sym:
            h_new = (h_new + h_new[::-1]) / 2.0
        h_new = project_box_sum(h_new, target)
        c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))
        if c_new < f_best:
            f_best = c_new
            h_best = h_new.copy()
        h = h_new
    return h_best, f_best

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    mx = np.max(scaled)
    return float((mx + np.log(np.sum(np.exp(scaled - mx)))) / T * dx)

def slsqp_opt(h0, n, T, maxiter, deadline):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_box_sum(res.x, target)
    except Exception:
        pass
    return h0

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def coord_descent_step(h, n, target, deadline):
    """Single coordinate descent sweep minimizing max_k c[k] (ignoring sum constraint, project at end)"""
    c = np.correlate(h, 1.0 - h, mode='full')
    for i in range(n):
        if time.time() > deadline:
            break
        h_old = h[i]
        B = np.zeros(len(c))
        B_as_a = 1.0
        for k_idx in range(len(c)):
            lag = k_idx - (n - 1)
            if lag >= 0:
                if lag <= i < n:
                    B[k_idx] += (1.0 - h[i - lag])
                if 0 <= i < n - lag:
                    B[k_idx] -= h[i + lag]
            else:
                m = -lag
                if 0 <= i < n - m:
                    B[k_idx] += (1.0 - h[i + m])
                if m <= i < n:
                    B[k_idx] -= h[i - m]
        D = c - B * h_old
        lo, hi = 0.0, 1.0
        for _ in range(30):
            mid = (lo + hi) / 2.0
            vals = D + B * mid
            s = B[int(np.argmax(vals))]
            if s > 1e-12:
                hi = mid
            elif s < -1e-12:
                lo = mid
            else:
                break
        h[i] = (lo + hi) / 2.0
    h = project_box_sum(h, target)
    return h
