import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_corr(h):
    """Return the correlation array (not yet multiplied by dx)."""
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
            if np.sum(mask) == 0:
                break
            h[mask] += diff / np.sum(mask)
        else:
            mask = h > 0.0
            if np.sum(mask) == 0:
                break
            h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    # Final exact fix
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        # Adjust fractional entries
        frac = h[(h > 0) & (h < 1)]
        if len(frac) > 0:
            idx_frac = np.where((h > 0) & (h < 1))[0]
            h[idx_frac[0]] += diff
            h = np.clip(h, 0.0, 1.0)
    return h

def _obj_slsqp(x, T):
    """Smooth surrogate for max(corr)."""
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, T=100.0, maxiter=300):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(lambda x: _obj_slsqp(x, T), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, n)
    except Exception:
        return h.copy()

def threshold_binary(h, n):
    """Convert continuous h to binary with exactly n/2 ones."""
    sorted_idx = np.argsort(-h)  # descending
    binary = np.zeros(n)
    binary[sorted_idx[:n//2]] = 1.0
    return binary

def binary_swap_sa(h_binary, n, max_time, rng, verbose=False):
    """Simulated annealing with swap moves on binary h."""
    dx = 2.0 / n
    h = h_binary.copy()
    best_h = h.copy()
    best_c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    T_start = 0.001  # initial temperature (in units of C5)
    T_end = 1e-6
    n_halving = 30  # number of halvings
    
    current_c = best_c
    start_t = time.time()
    
    for iteration in range(1000000):
        if time.time() - start_t > max_time:
            break
        
        T = T_start * (T_end / T_start) ** (iteration / (n_halving * 1000))
        # Actually let's use a simpler schedule
        progress = min(1.0, (time.time() - start_t) / max_time)
        T = T_start * (T_end / T_start) ** progress
        
        # Pick a random swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        # Perform swap
        h[i] = 0.0
        h[j] = 1.0
        
        # Update ones and zeros
        ones = ones[ones != i]
        zeros = zeros[zeros != j]
        zeros = np.append(zeros, i)
        ones = np.append(ones, j)
        
        # Evaluate
        c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
        delta = c_new - current_c
        
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            current_c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            # Undo
            h[i] = 1.0
            h[j] = 0.0
            ones = ones[ones != j]
            zeros = zeros[zeros != i]
            ones = np.append(ones, i)
            zeros = np.append(zeros, j)
    
    return best_h, best_c

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _slsqp_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x):
    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

def slsqp_optimize(h, T=100.0, maxiter=200):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(lambda x: _slsqp_obj(x, T), h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.success or res.fun < _slsqp_obj(h, T):
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h.copy()

def threshold_binary(h, n):
    sorted_idx = np.argsort(-h)
    binary = np.zeros(n)
    binary[sorted_idx[:n // 2]] = 1.0
    return binary

def greedy_binary_local_search(h, n, max_time=30.0):
    """Greedy swap-based local search on binary h."""
    dx = 2.0 / n
    h = h.copy()
    c = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    start_t = time.time()
    
    improving = True
    while improving and time.time() - start_t < max_time:
        improving = False
        # Try all "nearby" swaps: for each 1, try swapping with a few random zeros
        rng_local = np.random.RandomState(42)
        for i in ones:
            for _ in range(5):
                j = zeros[rng_local.randint(len(zeros))]
                h[i] = 0.0
                h[j] = 1.0
                c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)
                if c_new < c - 1e-14:
                    c = c_new
                    improving = True
                    zeros = zeros[zeros != j]
                    ones = ones[ones != i]
                    zeros = np.append(zeros, i)
                    ones = np.append(ones, j)
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if time.time() - start_t > max_time:
            break
    return h, c

def binary_sa(h, n, max_time, rng, T_start=0.002, T_end=1e-5):
    """Simulated annealing with swap moves on binary h."""
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_c = float(np.max(corr)) * dx
    current_c = best_c
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    start_t = time.time()
    no_improve = 0
    improve_thresh = 50000
    
    while time.time() - start_t < max_time:
        elapsed = time.time() - start_t
        progress = min(1.0, elapsed / max_time)
        T = T_start * (T_end / T_start) ** progress
        
        i_idx = rng.randint(len(ones))
        j_idx = rng.randint(len(zeros))
        i = ones[i_idx]
        j = zeros[j_idx]
        
        h[i] = 0.0
        h[j] = 1.0
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c_new = float(np.max(corr)) * dx
        delta = c_new - current_c
        
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            current_c = c_new
            ones = list(ones)
            zeros = list(zeros)
            ones.pop(i_idx)
            zeros.pop(j_idx)
            ones.append(j)
            zeros.append(i)
            ones = np.array(ones)
            zeros = np.array(zeros)
            
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            h[i] = 1.0
            h[j] = 0.0
        
        # Kick: if stuck, restart from best with higher T
        if no_improve > improve_thresh and T > T_end * 10:
            h = best_h.copy()
            current_c = best_c
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0
            improve_thresh = max(20000, improve_thresh // 2)
    
    return best_h, best_c
