def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    # Iterative correction
    for _ in range(5):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final exact correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute to elements with most room
        if diff > 0:
            idx = np.argsort(-h)[:int(abs(diff)) + 1]
        else:
            idx = np.argsort(h)[:int(abs(-diff)) + 1]

            h[i] += np.sign(diff) * min(abs(step), abs(diff))
            diff -= np.sign(diff) * min(abs(step), abs(diff))
    return h

    dx = 2.0 / n

    # corr_k = base_k + h[j] * slope_k (when only h[j] varies)
    # Actually: corr_k(new) = corr_k(old) + (new - old) * (1 - h[j+k] - h[j-k])

    #     b_k = corr_k(old) - h[j]_old * a_k

∂g(x) = {a_k : k ∈ argmax}

If min(∂g) > 0: function is increasing → min is to the left (decrease x)
If max(∂g) < 0: function is decreasing → min is to the right (increase x)
If 0 ∈ [min(∂g), max(∂g)]: at minimum

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(10):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    # exact fix
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            d = min(abs(diff), max(0.0, (1.0 - h[idx]) if diff > 0 else h[idx]))
            h[idx] += np.sign(diff) * d
            diff -= np.sign(diff) * d
            if abs(diff) < 1e-14:
                break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def optimize_single_coord(h, n, j):
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = len(corr)
    slopes = np.zeros(m)
    for k in range(m):
        s = k - (n - 1)
        v = 0.0
        if 0 <= j + s < n:
            v += (1.0 - h[j + s])
        if 0 <= j - s < n:
            v -= h[j - s]
        slopes[k] = v
    intercepts = corr - h[j] * slopes
    # g(x) = max_k(slopes[k]*x + intercepts[k]), convex in x
    # subgradient bisection on [0,1]
    def g(x):
        return np.max(slopes * x + intercepts)
    g0 = g(0.0)
    g1 = g(1.0)
    # check endpoints via subgradient
    vals0 = intercepts.copy()
    mx0 = np.max(vals0)
    act0 = np.where(vals0 >= mx0 - 1e-12)[0]
    if slopes[act0].min() >= -1e-12:
        return 0.0
    vals1 = slopes + intercepts
    mx1 = np.max(vals1)
    act1 = np.where(vals1 >= mx1 - 1e-12)[0]
    if slopes[act1].max() <= 1e-12:
        return 1.0
    lo, hi = 0.0, 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        vals = slopes * mid + intercepts
        mx = np.max(vals)
        act = np.where(vals >= mx - 1e-12)[0]
        if act.size == 0:
            act = np.array([np.argmax(vals)])
        smin = slopes[act].min()
        smax = slopes[act].max()
        if smin >= -1e-12:
            hi = mid
        elif smax <= 1e-12:
            lo = mid
        else:
            break
    opt = 0.5 * (lo + hi)
    # verify against endpoints
    best_v = g(opt)
    if g0 < best_v:
        return 0.0
    if g1 < best_v:
        return 1.0
    return opt

def coord_pass(h, n, time_limit=None, start_time=None):
    best_c5 = compute_c5(h)
    improved = False
    order = np.arange(n)
    for j in order:
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 10:
                break
        old = h[j]
        opt = optimize_single_coord(h, n, j)
        if abs(opt - old) < 1e-13:
            continue
        h[j] = opt
        h = make_feasible(h, n)
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            improved = True
        else:
            # revert
            h[j] = old
            h = make_feasible(h, n)
    return h, best_c5, improved

def soft_obj(h, n, t=50.0, lam=1e4):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    soft = mc + np.log(np.sum(np.exp(t * (corr - mc)))) / t
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return soft * dx + penalty
