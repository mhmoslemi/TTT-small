import numpy as np
from scipy.optimize import minimize
import time

_G_N = 960
_G_T = 100.0
_G_TOPK = 40

def _obj_smooth(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    top_k = min(_G_TOPK, len(corr))
    top_idx = np.argsort(-corr)[:top_k]
    top_corr = corr[top_idx]
    top_mc = np.max(top_corr)
    top_shifted = top_corr - top_mc
    soft_max = top_mc + np.log(np.sum(np.exp(_G_T * top_shifted))) / _G_T
    return soft_max * dx

def _eq_fun(x):
    return np.sum(x) - _G_N / 2.0

def _eq_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _slsqp_solve(h_start, n, T, maxiter, ftol):
    global _G_N, _G_T, _G_TOPK
    _G_N = n
    _G_T = T
    _G_TOPK = 40
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    res = minimize(_obj_smooth, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': ftol})
    return res.x

def _binary_greedy(h_bin, n, time_limit):
    t0 = time.time()
    c5_cur = compute_c5(h_bin)
    improved = True
    while improved and (time.time() - t0) < time_limit:
        improved = False
        for j in range(n):
            if (time.time() - t0) > time_limit:
                break
            for val in [0.0, 1.0]:
                if abs(h_bin[j] - val) < 1e-10:
                    continue
                trial = h_bin.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < c5_cur - 1e-14:
                    h_bin = trial
                    c5_cur = c
                    improved = True
                    break
    return h_bin, c5_cur

def _swap_search(h, n, rng, n_swaps, time_limit):
    t0 = time.time()
    c5_cur = compute_c5(h)
    for _ in range(n_swaps):
        if (time.time() - t0) > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        trial = h.copy()
        trial[i], trial[j] = h[j], h[i]
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < c5_cur - 1e-14:
            h = trial
            c5_cur = c
    return h, c5_cur

def _coord_descent(h, n, rng, n_rounds, time_limit):
    t0 = time.time()
    c5_cur = compute_c5(h)
    for rnd in range(n_rounds):
        if (time.time() - t0) > time_limit:
            break
        improved = False
        for j in range(n):
            if (time.time() - t0) > time_limit:
                break
            old = h[j]
            best_c = c5_cur
            best_cand = old
            cands = [0.0, 1.0, max(0.0, old - 0.5), min(1.0, old + 0.5),
                     max(0.0, old - 0.25), min(1.0, old + 0.25),
                     max(0.0, old - 0.05), min(1.0, old + 0.05),
                     max(0.0, old - 0.01), min(1.0, old + 0.01)]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_cand = cand
            if abs(best_cand - old) > 1e-12:
                h = h.copy()
                h[j] = best_cand
                h = make_feasible(h, n)
                c5_cur = best_c
                improved = True
        if not improved:
            break
    return h, c5_cur

def _block_search(h, n, rng, time_limit):
    t0 = time.time()
    c5_cur = compute_c5(h)
    for _ in range(500):
        if (time.time() - t0) > time_limit:
            break
        block_len = rng.randint(2, 8)
        start = rng.randint(0, n - block_len)
        end = start + block_len
        trial = h.copy()
        # Try flipping a block
        flip = rng.random() < 0.5
        if flip:
            trial[start:end] = 1.0 - h[start:end]
        else:
            trial[start:end] = h[start:end] + rng.randn(block_len) * 0.1
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < c5_cur - 1e-14:
            h = trial
            c5_cur = c
    return h, c5_cur

    T_schedule = [30, 80, 200, 500, 1000]

                best_h = make_feasible(x_new, n_pts)
        except Exception:
            pass

    # Phase 2: Multi-restart with perturbations
    for r in range(4):
        if rem() < 100:

        scale = 0.02 * (1 + r)
        h_pert = best_h + rng.randn(n_pts) * scale
        h_pert = make_feasible(h_pert, n_pts)

                    best_c5 = c5_val
                    best_h = make_feasible(x_new, n_pts)
                    h_pert = best_h.copy()
            except Exception:
                pass

    # Phase 3: Binary snap and greedy
    if rem() > 120:

        h_bin = make_feasible(h_bin, n_pts)

    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
