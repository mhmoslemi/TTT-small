def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):

    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_c5_grad(h, tau):

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:

    obj = dx * (max_c + np.log(s) / tau)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 3.0
    hi = g.max() + 3.0

        s = np.sum(np.clip(g - mid, 0.0, 1.0))

        else:

        h_p, c5_p = perturbation_search(best_h, n1, time_limit=tl, n_perturb=10)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 5: Higher resolution polish
    if _remaining() > 180:
        n2 = 3000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tau_sched2 = [100.0, 40.0, 12.0, 5.0]
        tl = min(100, _remaining() * 0.18)
        h2, c5_2 = adam_optimize(h2, n2, tau_sched2, time_limit=tl, lr=0.05)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()
        if _remaining() > 40:
            tl2 = min(30, _remaining() * 0.1)
            h2b, c5_2b = subgrad_optimize(best_h, len(best_h), time_limit=tl2, lr=0.001, top_k=8)
            if c5_2b < best_c5:
                best_c5, best_h = c5_2b, h2b.copy()

    # Phase 6: Final low-tau refinement
    if _remaining() > 50:
        n_f = len(best_h)
        tau_sched_f = [4.0, 2.0]
        tl = min(40, _remaining() * 0.25)
        h_f, c5_f = adam_optimize(best_h, n_f, tau_sched_f, time_limit=tl, lr=0.03)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Phase 7: Final subgradient + perturbation
    if _remaining() > 30:
        n_f = len(best_h)
        tl = min(20, _remaining() * 0.4)
        h_f, c5_f = subgrad_optimize(best_h, n_f, time_limit=tl, lr=0.0008, top_k=12)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    if _remaining() > 15:
        n_f = len(best_h)
        tl = min(15, _remaining() * 0.5)
        h_p, c5_p = perturbation_search(best_h, n_f, time_limit=tl, n_perturb=4)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
