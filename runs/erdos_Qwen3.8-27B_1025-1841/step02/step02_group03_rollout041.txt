def eval_flip_pair(h, corr, i, j, n):
    """Compute new max correlation after flipping h[i] 1->0 and h[j] 0->1"""
    one_minus_h = 1 - h
    # Effect of flipping h[i] from 1 to 0:
    # For each shift k, the term h[i]*(1-h[i+k-(n-1)]) becomes 0
    # So corr[k] decreases by (1-h[i+k-(n-1)]) where valid
    # Effect of flipping h[j] from 0 to 1:
    # For each shift k, the term h[j]*(1-h[j+k-(n-1)]) becomes (1-h[j+k-(n-1)])
    # So corr[k] increases by (1-h[j+k-(n-1)]) where valid

    # So k ranges from (n-1-i) to (n-1-i+n-1) = (2n-2-i)
    # And k must be in [0, 2n-2]
    
    lo_i = max(0, n-1-i)
    hi_i = min(2*n-2, 2*n-2-i)

    #   if n-1-i < 0: 0 + i - n + 1 = i - n + 1
