def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def corr_fft(h):
    n = len(h)
    N = 2 * n
    H = np.fft.rfft(h, N)
    F = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(F), N)[:2*n-1]
    return corr

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def corr_fft(h):
    n = len(h)
    N = 2 * n - 1
    N_fft = 1
    while N_fft < N:
        N_fft *= 2
    A = np.fft.fft(h, N_fft)
    B = np.fft.fft((1.0 - h)[::-1], N_fft)
    result = np.fft.ifft(A * B).real[:N]
    return result


def compute_c5(h):
    n = len(h)
    corr = corr_fft(h)
    return corr.max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_gradient_topM(h, p_top, top_idx, n, dx):
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_top[i]
        if pk < 1e-15:
            continue
        k = top_idx[i]
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx

        if eff_topM < (2 * n - 1):
            top_idx = np.argpartition(g, -eff_topM)[-eff_topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p_top = exp_c / s
            grad = compute_gradient_topM(h, p_top, top_idx, n, dx)
        else:
            max_g = g.max()
            shifted = np.clip(tau * (g - max_g), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p = exp_c / s
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if p[k] < 1e-15:
                    continue
                d = k - n + 1
                j0 = max(0, d)
                j1 = min(n, n + d)
                if j0 < j1:
                    idx = np.arange(j0, j1)
                    grad[idx] += p[k] * (1.0 - h[idx - d])
                j0b = max(0, -d)
                j1b = min(n, n - d)
                if j0b < j1b:
                    idx = np.arange(j0b, j1b)
                    grad[idx] -= p[k] * h[idx + d]
            grad *= dx

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.6
            v *= 0.2
            stall = 0
            tau *= 0.5
            if lr < 1e-13:
                break
        elif stall > patience // 2:
            lr *= 0.8
            v *= 0.5
            stall = 0

        h = h_new

    return best_h, best_obj


def optimize_subgrad(h, n, time_limit, lr_init=0.002, topK=3):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    patience = 600

    for it in range(300000):
        if time.time() - start > time_limit:
            break
        corr = corr_fft(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK

        cur_lr = lr / (1.0 + it * 0.002)
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break
        h = h_new

    return best_h, best_c5


def anneal_phase(h, n, time_limit, n_iter=50000):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    T = 0.05

    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        if it % 5000 == 4000:
            T *= 0.7

        n_block = max(2, n // 8)
        idx0 = np.random.randint(0, n - n_block)
        idx1 = idx0 + np.random.randint(n_block // 2, n_block)

        h_trial = h.copy()
        h_trial[idx0:idx1] = np.random.uniform(0, 1, idx1 - idx0)
        h_trial = project_to_feasible(h_trial, n)

        c5_trial = compute_c5(h_trial)
        delta = c5_trial - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / max(T, 1e-10)):
            h = h_trial
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Early optimization at initial resolution with low tau (smooth)
    if remaining() > 150:
        h1 = best_h.copy()
        tl = min(remaining() * 0.12, 100.0)
        h1, _ = optimize_softmax(h1, n_init, tl, lr_init=0.2, tau_init=3.0,
                                  topM=400, tau_rate=1.02)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Medium tau at initial resolution
    if remaining() > 120:
        h1 = best_h.copy()
        tl = min(remaining() * 0.12, 100.0)
        h1, _ = optimize_softmax(h1, n_init, tl, lr_init=0.12, tau_init=15.0,
                                  topM=300, tau_rate=1.04)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 3: High tau at initial resolution
    if remaining() > 100:
        h1 = best_h.copy()
        tl = min(remaining() * 0.12, 90.0)
        h1, _ = optimize_softmax(h1, n_init, tl, lr_init=0.08, tau_init=50.0,
                                  topM=200, tau_rate=1.06)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 4: Resolution 2x
    n2 = n_init * 2
    if remaining() > 120:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(remaining() * 0.15, 150.0)
        h2, _ = optimize_softmax(h2, n2, tl, lr_init=0.08, tau_init=8.0,
                                  topM=300, tau_rate=1.04)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        if remaining() > 60:
            h2, _ = optimize_softmax(h2, n2, min(remaining() * 0.08, 60.0),
                                       lr_init=0.05, tau_init=40.0,
                                       topM=200, tau_rate=1.06)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 5: Resolution 3x
    n3 = n_init * 3
    if remaining() > 100:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        tl = min(remaining() * 0.15, 130.0)
        h3, _ = optimize_softmax(h3, n3, tl, lr_init=0.06, tau_init=10.0,
                                  topM=400, tau_rate=1.04)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        if remaining() > 50:
            h3, _ = optimize_softmax(h3, n3, min(remaining() * 0.08, 50.0),
                                       lr_init=0.04, tau_init=50.0,
                                       topM=250, tau_rate=1.06)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

    # Phase 6: Resolution 4x
    n4 = n_init * 4
    if remaining() > 80:
        h4 = interp_periodic(best_h, n4)
        h4 = project_to_feasible(h4, n4)
        tl = min(remaining() * 0.15, 120.0)
        h4, _ = optimize_softmax(h4, n4, tl, lr_init=0.05, tau_init=12.0,
                                  topM=500, tau_rate=1.04)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 7: Subgradient polish at best resolution
    if remaining() > 60:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = optimize_subgrad(h_f, n_f, time_limit=remaining() * 0.15,
                                      lr_init=0.002, topK=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 8: Random restarts with optimization
    if remaining() > 100:
        for attempt in range(6):
            if remaining() < 40:
                break
            n_p = len(best_h)
            noise_scale = 0.015 * (attempt + 1)
            h_p = best_h + noise_scale * np.random.randn(n_p)
            h_p = project_to_feasible(h_p, n_p)
            tl = min(remaining() * 0.12, 60.0)
            h_p, _ = optimize_softmax(h_p, n_p, tl, lr_init=0.1, tau_init=15.0,
                                       topM=min(200, 2 * n_p - 1), tau_rate=1.05)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 9: Annealing phase
    if remaining() > 50:
        n_a = len(best_h)
        h_a = best_h.copy()
        h_a, c5_a = anneal_phase(h_a, n_a, time_limit=remaining() * 0.2,
                                  n_iter=int(remaining() * 0.2 * 100))
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Phase 10: Final subgradient polish
    if remaining() > 30:
        n_final = len(best_h)
        h_final = best_h.copy()
        h_final, c5_final = optimize_subgrad(h_final, n_final,
                                              time_limit=remaining() * 0.35,
                                              lr_init=0.001, topK=3)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
