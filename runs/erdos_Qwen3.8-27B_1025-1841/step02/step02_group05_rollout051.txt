import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def fw_step(grad, n):
    """Frank-Wolfe linear minimization: find vertex of {0<=h<=1, sum(h)=n/2} minimizing grad·h"""
    # Sort by gradient, pick n/2 smallest to set to 1
    order = np.argsort(grad)
    v = np.zeros(n)
    v[order[:n//2]] = 1.0
    return v

def nesterow_fw(h, n, tau, time_limit, lr_scale=1.0):
    """Nesterov-accelerated Frank-Wolfe for smoothed objective"""
    start = time.time()
    h = project_to_feasible(h.copy(), n)
    y = h.copy()  # Nesterov momentum variable
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    t = 1.0
    prev_c5 = best_c5
    
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        
        # Nesterov step
        beta_t = (1 + np.sqrt(1 + 4*t*t)) / 2
        t_new = t + 1
        beta_new = (1 + np.sqrt(1 + 4*t_new*t_new)) / 2
        alpha = 1 - beta_t / (t * beta_new)
        
        y = (1 - alpha) * h + alpha * y
        
        # Gradient at y (project to feasible for gradient computation)
        y_feas = project_to_feasible(y, n)
        obj, grad = smooth_obj_grad(y_feas, tau)
        
        # FW vertex
        v = fw_step(grad, n)
        
        # Step size: minimize over gamma in [0,1]
        # f((1-gamma)*h + gamma*v)
        # Use given step: gamma = 2/(t+2)
        gamma = 2.0 / (t + 2.0)
        
        h = (1 - gamma) * h + gamma * v
        t = t_new

    return best_h, best_c5

    target = n / 2.0

        else:

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s_exp = exp_c.sum()
    p = exp_c / s_exp
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0-sh:j1-sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b+sh:j1b+sh]
    grad *= dx
    obj = dx * (max_c + np.log(s_exp) / tau)
    return obj, grad
