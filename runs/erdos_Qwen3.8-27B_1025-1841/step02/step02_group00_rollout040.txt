    """Compute log-sum-exp smoothed objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1

    #       for the second term, we need 0 <= j+ell < n and j in valid range

    for k in range(2 * n - 1):
        if p[k] < 1e-10:

        i_start = max(0, -ell)

        # i_start+ell = max(0,-ell)+ell = max(ell, 0)
        # i_end-1+ell = min(n, n-ell)-1+ell

        j2_end = i_end
        # Additionally need j+ell in [0, n): j+ell >= 0 => j >= -ell, j+ell < n => j < n-ell

        # Wait: we need 0 <= j+ell < n. j ranges [j2_start, j2_end).
        # j+ell ranges [j2_start+ell, j2_end+ell).
        # j2_start+ell = max(0,-ell)+ell = max(ell, 0) >= 0 ✓
        # j2_end+ell = min(n, n-ell)+ell = min(n+ell, n) <= n (if ell <= 0) or n (if ell > 0, then min(n, n-ell) = n-ell, so n-ell+ell = n) ✓

            h_vals = h[indices + ell]  # h[j+ell]
            grad[j2_start:j2_end] -= p[k] * h_vals
    
    grad *= dx
