A simpler but effective approach:

2. For all possible swaps (i, j), compute Δc[L*] (the change in the critical correlation)
3. Among swaps with Δc[L*] < 0, pick the one that most reduces the overall max (check a few top correlations)
4. Apply the best swap
5. Repeat

The key optimization: for a fixed L*, we can precompute for all positions p the value "score(p) = h[p+L*] + h[p-L*]" (with boundary handling). Then for a swap (i, j):
Δc[L*] = score(j) - score(i)

So we want i (a 0) with high score and j (a 1) with low score.

This is O(n) to compute scores, and O(n) to find the best (i, j) pair. Then we verify by recomputing the full correlation.

Total per iteration: O(n) for finding the candidate swap + O(n²) for verification. With n=960, that's fast.

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() - start > time_limit:
            return h, _compute_c5(h)
        c = _compute_c5(res.x)
        if c < _compute_c5(h):
            return _make_feasible(res.x, n), c
    except Exception:
        pass
    return h, _compute_c5(h)

def _upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return _make_feasible(np.interp(new_x, old_x, h), new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n = len(h_init)
    h_init = _make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = _compute_c5(h_init)

    # Phase 1: Binary snap + swap search at native resolution
    if remaining() > 200:
        h_bin = _to_binary(h_init, n)
        budget1 = min(remaining() - 200, 400)
        h_bin, c_bin = _swap_search(h_bin, n, budget1, rng)
        c_bin = _compute_c5(h_bin)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()

    # Phase 2: Coordinate descent on the best solution
    if remaining() > 100:
        budget2 = min(remaining() - 100, 200)
        best_h, best_c5 = _coord_descent(best_h, n, budget2, n_passes=4)

    # Phase 3: SLSQP polish with multiple temperatures
    if remaining() > 80:
        for T in [50.0, 200.0, 500.0]:
            if remaining() < 40:
                break
            best_h, c_new = _slsqp_polish(best_h, n, T, 150, min(remaining() - 30, 120))
            if c_new < best_c5:
                best_c5 = c_new

    # Phase 4: Another round of coord descent after SLSQP
    if remaining() > 60:
        best_h, best_c5 = _coord_descent(best_h, n, min(remaining() - 40, 100), n_passes=3)

    # Phase 5: Try higher resolution
    if remaining() > 120:
        n2 = n * 2
        h2 = _upsample(best_h, n2)
        h2_bin = _to_binary(h2, n2)
        budget5 = min(remaining() - 80, 200)
        h2_bin, c2_bin = _swap_search(h2_bin, n2, budget5, rng)
        c2_bin = _compute_c5(h2_bin)
        if c2_bin < best_c5:
            best_c5 = c2_bin
            best_h = h2_bin.copy()
            n = n2
        if remaining() > 60:
            best_h, best_c5 = _coord_descent(best_h, len(best_h), min(remaining() - 40, 100), n_passes=3)
        if remaining() > 50:
            best_h, c_new = _slsqp_polish(best_h, len(best_h), 300.0, 150, min(remaining() - 20, 80))
            if c_new < best_c5:
                best_c5 = c_new

    # Phase 6: Final binary swap refinement
    if remaining() > 40:
        n_f = len(best_h)
        h_f = _to_binary(best_h, n_f)
        budget6 = min(remaining() - 10, 60)
        h_f, c_f = _swap_search(h_f, n_f, budget6, rng)
        c_f = _compute_c5(h_f)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()

    # Final feasibility check
    n_final = len(best_h)
    best_h = _make_feasible(best_h, n_final)
    best_c5 = _compute_c5(best_h)

    return best_h, best_c5, n_final
