    corr = np.correlate(h, 1.0 - h, mode='full')
    g_vals = corr * dx
    best_obj = g_vals.max()
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(200, max_iters // 10)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        g_vals = corr * dx
        cur_max = g_vals.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argpartition(g_vals, -topM)[-topM:]
        g_top = g_vals[top_idx]
        max_top = g_top.max()
        shifted = np.clip(tau * (g_top - max_top), -500, 500)
        exp_c = np.exp(shifted)
        s_exp = exp_c.sum()
        p_top = exp_c / s_exp

        M = len(top_idx)
        grad_weighted = np.zeros(n)
        for i in range(M):
            w = p_top[i]
            if w < 1e-12:
                continue
            ki = top_idx[i]
            d = ki - n + 1
            if d >= 0:
                p1_end = min(n - d, n)
                if p1_end > 0:
                    idx1 = np.arange(0, p1_end)
                    grad_weighted[idx1] += w * (1.0 - h[idx1 + d])
                if d < n:
                    idx2 = np.arange(d, n)
                    grad_weighted[idx2] -= w * h[idx2 - d]
            else:
                e = -d
                if e < n:
                    idx1 = np.arange(e, n)
                    grad_weighted[idx1] += w * (1.0 - h[idx1 - e])
                p2_end = min(n - e, n)
                if p2_end > 0:
                    idx2 = np.arange(0, p2_end)
                    grad_weighted[idx2] -= w * h[idx2 + e]
        grad = grad_weighted * dx

        lr = lr_init / (1.0 + it * 0.003)
        y = h + (beta / max(0.9, 0.9)) * v if it > 0 else h
        v_new = beta * v - lr * grad
        h_new = project_to_feasible(h + v_new, n)
        v = v_new
        h = h_new

        if stall > patience:
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break

    return best_h, best_obj


def subgrad_polish(h, n, max_iters, time_limit, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr_init = 0.004
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    patience = 300

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            if d >= 0:
                p1_end = min(n - d, n)
                if p1_end > 0:
                    idx1 = np.arange(0, p1_end)
                    grad[idx1] += 1.0 - h[idx1 + d]
                if d < n:
                    idx2 = np.arange(d, n)
                    grad[idx2] -= h[idx2 - d]
            else:
                e = -d
                if e < n:
                    idx1 = np.arange(e, n)
                    grad[idx1] += 1.0 - h[idx1 - e]
                p2_end = min(n - e, n)
                if p2_end > 0:
                    idx2 = np.arange(0, p2_end)
                    grad[idx2] -= h[idx2 + e]
        grad *= dx / topK

        cur_lr = lr_init / (1.0 + it * 0.008)
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)
        h = h_new

        if stall > patience:
            lr_init *= 0.6
            v *= 0.5
            stall = 0
            if lr_init < 1e-14:
                break

    return best_h, best_c5


def coordinate_descent(h, n, max_passes, time_limit, tol=1e-6):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    corr = np.correlate(h, 1.0 - h, mode='full') * dx

    for p in range(max_passes):
        if time.time() - start > time_limit:
            break
        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            if old_val <= 1e-10 or old_val >= 1.0 - 1e-10:
                continue
            # Try moving toward 0 or 1
            for new_val in [0.0, 1.0]:
                h[i] = new_val
                # Quick recompute of affected correlations
                # Only lags where h[i] appears change
                # For lag d: h[i] appears if 0<=i-d<n or 0<=i<n-d (d>=0) etc.
                # Just recompute all is O(n^2) which is too slow for per-coordinate
                # Skip CD, use gradient step instead
                break
            h[i] = old_val
        # Fallback: small gradient step
        top_idx = np.argsort(corr)[-10:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            if d >= 0:
                p1_end = min(n - d, n)
                if p1_end > 0:
                    idx1 = np.arange(0, p1_end)
                    grad[idx1] += 1.0 - h[idx1 + d]
                if d < n:
                    idx2 = np.arange(d, n)
                    grad[idx2] -= h[idx2 - d]
            else:
                e = -d
                if e < n:
                    idx1 = np.arange(e, n)
                    grad[idx1] += 1.0 - h[idx1 - e]
                p2_end = min(n - e, n)
                if p2_end > 0:
                    idx2 = np.arange(0, p2_end)
                    grad[idx2] -= h[idx2 + e]
        grad *= dx / 10.0
        lr = 0.002 / (1.0 + p * 0.02)
        h_new = project_to_feasible(h - lr * grad, n)
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new
            improved = True
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        if not improved and p > 50:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n0 = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: Optimize at initial resolution with decreasing tau
    if remaining() > 120:
        n1 = n0
        h_curr = best_h.copy()
        taus = [60.0, 30.0, 15.0, 8.0, 4.0, 2.0, 1.0]
        for tau in taus:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.14, 130.0)
            h_curr, _ = nesterov_optimize(h_curr, n1, tau=tau, max_iters=30000,
                                           time_limit=tl, lr_init=0.12, topM=300)
            c5_1 = compute_c5(h_curr)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h_curr.copy()

    # Phase 2: Random restarts at same resolution
    if remaining() > 150:
        n1 = n0
        for attempt in range(4):
            if remaining() < 50:
                break
            noise_scale = 0.03 + 0.04 * attempt
            h_try = best_h + noise_scale * np.random.randn(n1)
            h_try = project_to_feasible(h_try, n1)
            tl = min(remaining() * 0.15, 100.0)
            h_try, _ = nesterov_optimize(h_try, n1, tau=8.0, max_iters=15000,
                                          time_limit=tl, lr_init=0.1, topM=250)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Phase 3: Double resolution
    n2 = n1 * 2
    if remaining() > 120:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        for tau in [12.0, 6.0, 3.0]:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.2, 160.0)
            h2, _ = nesterov_optimize(h2, n2, tau=tau, max_iters=10000,
                                       time_limit=tl, lr_init=0.1, topM=200)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 4: Quadruple resolution
    n3 = n1 * 4
    if remaining() > 100:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        for tau in [6.0, 3.0, 1.5]:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.22, 150.0)
            h3, _ = nesterov_optimize(h3, n3, tau=tau, max_iters=5000,
                                       time_limit=tl, lr_init=0.08, topM=150)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

    # Phase 5: Subgradient polish at best resolution
    if remaining() > 50:
        n_f = len(best_h)
        h_f = best_h.copy()
        tl = min(remaining() * 0.35, 200.0)
        h_f, c5_f = subgrad_polish(h_f, n_f, max_iters=100000,
                                    time_limit=tl, topK=5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: Fine subgradient with increasing topK
    if remaining() > 40:
        n_f = len(best_h)
        h_f = best_h.copy()
        tl = min(remaining() * 0.3, 120.0)
        h_f, c5_f = subgrad_polish(h_f, n_f, max_iters=80000,
                                    time_limit=tl, topK=8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 7: Another nesterov pass with very low tau
    if remaining() > 40:
        n_f = len(best_h)
        h_f = best_h.copy()
        tl = min(remaining() * 0.3, 100.0)
        h_f, _ = nesterov_optimize(h_f, n_f, tau=1.0, max_iters=20000,
                                    time_limit=tl, lr_init=0.05, topM=500)
        c5_f = compute_c5(h_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
