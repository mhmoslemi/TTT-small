import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - target) > 1e-12:
            if np.sum(h) < target:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += (target - np.sum(h)) / cnt
    return np.clip(h, 0.0, 1.0)

def soft_max_gd(h, n_iter, T_start, T_end, lr_base, top_k):
    h = h.astype(float).copy()
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_base * (1.0 - 0.7 * frac) + lr_base * 0.1
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        exp_s = np.exp(T * shifted)
        total = np.sum(exp_s)
        if total < 1e-300:
            k = np.argmax(corr) - (n - 1)
            grad = _grad_single(h, k)
        else:
            weights = exp_s / total
            top_idx = np.argsort(weights)[-top_k:][::-1]
            grad = np.zeros(n)
            for idx in top_idx:
                w = weights[idx]
                if w < 1e-12:
                    break
                k = idx - (n - 1)
                grad += w * _grad_single(h, k)
        
        h = h - lr * grad
        h = make_feasible(h, n)
        
        if (it + 1) % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _grad_single(h, k):
    """Gradient of correlation at lag k w.r.t. h, times dx"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        else:
            grad[:n-k] += (1.0 - h[k:]) * dx
            grad[k:] += (-h[:n-k]) * dx
    else:
        s = -k
        grad[s:] += (1.0 - h[:n-s]) * dx
        grad[:n-s] += (-h[s:]) * dx
    return grad

def binary_sa(h_bin, n_iter, T_start, T_end, rng):
    """Simulated annealing on binary sequence"""
    n = len(h_bin)
    h = h_bin.copy()
    dx = 2.0 / n

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        if abs(np.sum(h) - target) > 1e-12:
            if np.sum(h) < target:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += (target - np.sum(h)) / cnt
    return np.clip(h, 0.0, 1.0)

def _grad_single(h, k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        if k == 0:
            grad[:] = (1.0 - 2.0 * h) * dx
        else:
            grad[:n-k] += (1.0 - h[k:]) * dx
            grad[k:] += (-h[:n-k]) * dx
    else:
        s = -k
        grad[s:] += (1.0 - h[:n-s]) * dx
        grad[:n-s] += (-h[s:]) * dx
    return grad

def make_binary(h, n, rng):
    h_b = (h > 0.5).astype(float)
    n_target = n // 2
    n_ones = int(np.sum(h_b))
    if n_ones != n_target:
        if n_ones > n_target:
            ones_idx = np.where(h_b == 1)[0]
            remove = rng.choice(ones_idx, n_ones - n_target, replace=False)
            h_b[remove] = 0
        else:
            zeros_idx = np.where(h_b == 0)[0]
            add = rng.choice(zeros_idx, n_target - n_ones, replace=False)
            h_b[add] = 1
    return h_b

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n = 320
        h0 = np.zeros(n)
        h0[rng.choice(n, n // 2, replace=False)] = 1.0
        h0 = make_feasible(h0, n)

    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    print(f"Initial: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Quick GD to get a good starting point
    if time.time() - start_time < budget_s - 200:
        h1, c1 = soft_max_gd(best_h, 1500, 50.0, 1.0, 0.05, 10)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        print(f"GD: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: Binary SA at initial resolution
    if time.time() - start_time < budget_s - 150:
        h_bin = make_binary(best_h, n0, rng)
        time_left = budget_s - 120 - (time.time() - start_time)
        n_iter_sa = min(200000, max(20000, int(time_left * 50000)))
        h_b, c_b = binary_sa(h_bin, n_iter_sa, 3.0, 0.01, rng,
                             time_limit=budget_s - 100 - (start_time + 0.0),
                             start_t=start_time)
        if c_b < best_c5:
            best_c5, best_h = c_b, h_b.copy()
            print(f"  SA n={n0}: C5={best_c5:.6f}", flush=True)

        # Greedy polish
        if time.time() - start_time < budget_s - 80:
            h_g, c_g = greedy_polish(best_h, n_passes=3, rng=rng)
            if c_g < best_c5:
                best_c5, best_h = c_g, h_g.copy()
                print(f"  Polish n={n0}: C5={best_c5:.6f}", flush=True)

    print(f"After SA n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: Multi-resolution SA
    for target_n in [480, 640, 960]:
        if time.time() - start_time > budget_s - 100:
            break
        h_up = upsample(best_h, target_n)
        h_bin_up = make_binary(h_up, target_n, rng)
        c_start = compute_c5(h_bin_up)
        
        time_left = budget_s - 60 - (time.time() - start_time)
        if time_left < 20:
            break
        n_iter = min(150000, max(10000, int(time_left * 40000)))
        
        h_bu, c_bu = binary_sa(h_bin_up, n_iter, 2.5, 0.005, rng,
                               time_limit=time_left - 10,
                               start_t=time.time())
        # Polish
        if time.time() - start_time < budget_s - 40:
            h_pol, c_pol = greedy_polish(h_bu, n_passes=2, rng=rng)
            if c_pol < c_bu:
                h_bu, c_bu = h_pol, c_pol
        
        if c_bu < best_c5:
            best_c5, best_h = c_bu, h_bu.copy()
            print(f"  SA n={target_n}: C5={best_c5:.6f}", flush=True)
        else:
            print(f"  SA n={target_n}: C5={c_bu:.6f} (no improvement)", flush=True)

    # Phase 4: Continuous refinement at best resolution
    if time.time() - start_time < budget_s - 50:
        n_pts = len(best_h)
        # Fine continuous optimization
        h_f, c_f = soft_max_gd(best_h, 3000, 20.0, 0.3, 0.03, 8)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
            print(f"  ContRefine: C5={best_c5:.6f}", flush=True)

        # SLSQP for small n
        if n_pts <= 512 and time.time() - start_time < budget_s - 20:
            n = n_pts
            dx = 2.0 / n
            T_s = 6.0
            def obj_fn(hv):
                corr = np.correlate(hv, 1.0 - hv, mode='full')
                mc = np.max(corr)
                return (mc + np.log(np.sum(np.exp(T_s * (corr - mc)))) / T_s) * dx
            def cons_fn(hv):
                return np.sum(hv) - n / 2.0
            def cons_jac(hv):
                return np.ones(n)
            try:
                res = minimize(obj_fn, best_h, method='SLSQP',
                               bounds=[(0.0, 1.0)] * n,
                               constraints=[{'type': 'eq', 'fun': cons_fn, 'jac': cons_jac}],
                               options={'maxiter': 400, 'ftol': 1e-14})
                if res.x is not None:
                    hx = make_feasible(res.x, n)
                    cx = compute_c5(hx)
                    if cx < best_c5:
                        best_c5, best_h = cx, hx.copy()
                        print(f"  SLSQP: C5={best_c5:.6f}", flush=True)
            except Exception:
                pass

    # Phase 5: Final SA round on best if time remains
    if time.time() - start_time < budget_s - 15:
        n_pts = len(best_h)
        h_bin_final = make_binary(best_h, n_pts, rng)
        c_final_start = compute_c5(h_bin_final)
        if c_final_start <= best_c5 + 1e-9:
            time_left = budget_s - 10 - (time.time() - start_time)
            if time_left > 5:
                h_fb, c_fb = binary_sa(h_bin_final, min(80000, int(time_left * 30000)),
                                       1.5, 0.003, rng, time_limit=time_left - 2,
                                       start_t=time.time())
                if c_fb < best_c5:
                    best_c5, best_h = c_fb, h_fb.copy()
                    print(f"  FinalSA: C5={best_c5:.6f}", flush=True)

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
