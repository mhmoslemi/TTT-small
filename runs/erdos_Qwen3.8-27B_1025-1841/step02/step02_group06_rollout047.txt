Let me think about the coordinate descent more carefully. For n=3600, a full sweep is 3600 coordinate updates. Each update requires:
- Computing S (vector of length 7199): this involves indexing into h at shifted positions
- 10-15 bisection iterations, each computing a weighted sum of S

The computation of S for coordinate j:
```python
d = np.arange(2*n-1) - (n-1)
jd = j + d
term1 = np.where((jd >= 0) & (jd < n), 1.0 - h[np.clip(jd, 0, n-1)], 0.0)
jm = j - d
term2 = np.where((jm >= 0) & (jm < n), h[np.clip(jm, 0, n-1)], 0.0)
S = term1 - term2
```

This is O(n) with good constants. For 3600 coordinates, that's 3600 * 7199 ≈ 26M operations per sweep. At ~100M ops/s in numpy, that's ~0.26s per sweep. With 10 bisection iterations (each O(n)), total per sweep ≈ 0.26s * 11 ≈ 2.9s.

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    g = np.clip(g, -10.0, 10.0)
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr


def softmax_gradient_step(h, corr, n, dx, tau, topM, lr, v, beta):
    glen = 2 * n - 1
    if topM > 0 and topM < glen:
        top_idx = np.argpartition(corr, -topM)[-topM:]
        gsub = corr[top_idx]
        mx = gsub.max()
        shifted = np.clip(tau * (gsub - mx), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        if s < 1e-300:
            p_top = np.ones(topM) / topM
        else:
            p_top = exp_c / s
        grad = np.zeros(n)
        for i in range(topM):
            pk = p_top[i]
            if pk < 1e-10:
                continue
            d = top_idx[i] - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += pk * (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= pk * h[idx + d]
    else:
        mx = corr.max()
        shifted = np.clip(tau * (corr - mx), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        if s < 1e-300:
            p = np.ones(glen) / glen
        else:
            p = exp_c / s
        grad = np.zeros(n)
        for k in range(glen):
            pk = p[k]
            if pk < 1e-10:
                continue
            d = k - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += pk * (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= pk * h[idx + d]
    grad *= dx
    v = beta * v - lr * grad
    h_new = project_to_feasible(h + v, n)
    return h_new, v


def softmax_optimize(h, n, tau_list, max_iters_each, time_limit, topM=0, lr_init=0.15):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj, _ = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9

    for tau in tau_list:
        if time.time() - start > time_limit:
            break
        tl = time_limit - (time.time() - start)
        if tl < 10:
            break
        lr = lr_init
        stall = 0
        patience = max(50, max_iters_each // 10)
        corr, _ = compute_c5(h)

        for it in range(max_iters_each):
            if it < max_iters_each and (time.time() - start) > time_limit:
                break
            cur_max = corr.max() * dx
            if cur_max < best_obj - 1e-14:
                best_obj = cur_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1

            h, v = softmax_gradient_step(h, corr, n, dx, tau, topM, lr, v, beta)
            corr = np.correlate(h, 1.0 - h, mode='full')

            if stall > patience:
                lr *= 0.6
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break

    return best_h, best_obj


def coord_descent(h, n, max_sweeps, time_limit, tau=100.0, bisection_iters=12):
    dx = 2.0 / n
    h = h.copy()
    best_obj, corr = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    glen = 2 * n - 1
    d_all = np.arange(glen) - (n - 1)
    order = np.arange(n)

    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        np.random.shuffle(order)
        for j in order:
            if time.time() - start > time_limit:
                break
            jd = j + d_all
            jm = j - d_all
            jd_clipped = np.clip(jd, 0, n - 1)
            jm_clipped = np.clip(jm, 0, n - 1)
            term1 = np.where((jd >= 0) & (jd < n), 1.0 - h[jd_clipped], 0.0)
            term2 = np.where((jm >= 0) & (jm < n), h[jm_clipped], 0.0)
            S = term1 - term2

            hj = h[j]
            lo_d = -hj
            hi_d = 1.0 - hj
            if hi_d - lo_d < 1e-12:
                continue

            for _ in range(bisection_iters):
                mid_d = (lo_d + hi_d) / 2.0
                basis = corr + mid_d * S
                mx = basis.max()
                shifted = np.clip(tau * (basis - mx), -500, 500)
                p = np.exp(shifted)
                snorm = p.sum()
                if snorm < 1e-300:
                    fprime = 0.0
                else:
                    fprime = np.dot(S, p) / snorm
                if fprime > 0:
                    hi_d = mid_d
                else:
                    lo_d = mid_d

            delta = (lo_d + hi_d) / 2.0
            if abs(delta) < 1e-10:
                continue
            h[j] += delta
            corr += delta * S

        if sweep % 3 == 2:
            h = project_to_feasible(h, n)
            cur_c5, corr = compute_c5(h)
            if cur_c5 < best_obj - 1e-14:
                best_obj = cur_c5
                best_h = h.copy()

    h = project_to_feasible(h, n)
    cur_c5, _ = compute_c5(h)
    if cur_c5 < best_obj:
        best_obj = cur_c5
        best_h = h.copy()
    return best_h, best_obj


def subgrad_polish(h, n, max_iters, time_limit, lr_init=0.003, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        _, corr = compute_c5(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK
        cur_lr = lr / (1.0 + it * 0.01)
        h = project_to_feasible(h - cur_lr * grad, n)
        if it % 500 == 499 and lr > 1e-8:
            lr *= 0.8

    return best_h, best_c5
