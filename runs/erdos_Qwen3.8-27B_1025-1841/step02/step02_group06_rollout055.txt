import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x = np.arange(target_n, dtype=np.float64) * (float(n) / float(target_n))
    i0 = np.floor(x).astype(np.int64) % n
    i1 = (i0 + 1) % n
    f = x - np.floor(x)
    return h[i0] * (1.0 - f) + h[i1] * f


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def get_grad(h, lags, weights, n, dx):
    n_lags = len(lags)
    grad = np.zeros(n)
    for i in range(n_lags):
        w = weights[i]
        if w < 1e-14:
            continue
        d = int(lags[i]) - n + 1
        a = max(0, d)
        b = min(n, n + d)
        if a < b:
            idx = np.arange(a, b)
            grad[idx] += w * (1.0 - h[idx - d])
        a = max(0, -d)
        b = min(n, n - d)
        if a < b:
            idx = np.arange(a, b)
            grad[idx] -= w * h[idx + d]
    return grad * dx


def optimize_phase(h0, n, t_limit, phases):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    v = np.zeros(n)

    for (tau, topM, lr, n_iters) in phases:
        if time.time() - t0 >= t_limit:
            break
        mom = 0.9
        for it in range(n_iters):
            if time.time() - t0 >= t_limit:
                break
            corr = np.correlate(h, 1.0 - h, mode='full')
            g = corr * dx
            cur = g.max()
            if cur < best_c5 - 1e-15:
                best_c5 = cur
                best_h = h.copy()

            M = min(topM, 2 * n - 1)
            if M < 2 * n - 1:
                top = np.argpartition(g, -M)[-M:]
            else:
                top = np.arange(2 * n - 1)

            gt = g[top]
            sh = np.clip(tau * (gt - gt.max()), -500.0, 500.0)
            e = np.exp(sh)
            p = e / e.sum()

            grad = get_grad(h, top, p, n, dx)
            v = mom * v - lr * grad
            h = project_to_feasible(h + v, n)

            if it % 5000 == 4999:
                mom = max(0.5, mom - 0.02)

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def subgrad_final(h0, n, t_limit, lr0, topK):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

            best_h = h.copy()

    h = project_to_feasible(h0, n).copy()
    h_sum = np.zeros(n)
