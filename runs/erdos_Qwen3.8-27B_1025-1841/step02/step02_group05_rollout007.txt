        v = b2 * v + (1.0 - b2) * grad * grad
        m_h = m / (1.0 - b1 ** t)
        v_h = v / (1.0 - b2 ** t)
        h = project_to_feasible(h - lr * m_h / (np.sqrt(v_h) + eps), n)
    return best_h, best_c5


def optimize_anneal(h, n, tau_start, tau_end, time_limit, lr=0.01):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    stall = 0
    for it in range(500000):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break
        frac = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** frac
        t += 1
        obj, grad = smooth_grad(h, tau)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            lr *= 0.5
            m *= 0.1
            v *= 0.1
            stall = 0
            if lr < 1e-10:
                break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_h = m / (1.0 - b1 ** t)
        v_h = v / (1.0 - b2 ** t)
        h = project_to_feasible(h - lr * m_h / (np.sqrt(v_h) + eps), n)
    return best_h, best_c5


def subgrad_polish(h, n, time_limit, lr=0.001, top_k=3):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for i in range(top_k):
            k = top_idx[i]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b + sh:j1b + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    h0 = initial_h_values.copy()
    n = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    remaining = max(0.0, budget_s - (time.time() - t0))
    if remaining > 600:
        tl = min(700, remaining * 0.55)
        h_a, c5_a = optimize_anneal(best_h, n, 100.0, 3.0, tl, lr=0.015)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a

    remaining = max(0.0, budget_s - (time.time() - t0))
    if remaining > 350:
        tl = min(380, remaining * 0.32)
        h_a, c5_a = optimize_anneal(best_h, n, 8.0, 1.5, tl, lr=0.007)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a

    remaining = max(0.0, budget_s - (time.time() - t0))
    if remaining > 120:
        tl = min(160, remaining * 0.15)
        h_s, c5_s = subgrad_polish(best_h, n, tl, lr=0.0005, top_k=3)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s

    remaining = max(0.0, budget_s - (time.time() - t0))
    if remaining > 300:
        n2 = 4000
        h2 = project_to_feasible(interp_periodic(best_h, n2), n2)
        c5_2 = compute_c5(h2)
        remaining = max(0.0, budget_s - (time.time() - t0))
        if remaining > 180:
            tl = min(160, remaining * 0.35)
            h2, c5_2 = optimize_anneal(h2, n2, 60.0, 3.0, tl, lr=0.01)
        remaining = max(0.0, budget_s - (time.time() - t0))
        if remaining > 80:
            tl = min(70, remaining * 0.3)
            h2, c5_2 = subgrad_polish(h2, n2, tl, lr=0.0003, top_k=3)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2

    remaining = max(0.0, budget_s - (time.time() - t0))
    if remaining > 40:
        nf = len(best_h)
        tl = min(50, remaining * 0.6)
        h_s, c5_s = subgrad_polish(best_h, nf, tl, lr=0.0002, top_k=2)
        if c5_s < best_c5:
            best_c5, best_h = c5_s, h_s

    remaining = max(0.0, budget_s - (time.time() - t0))
    if remaining > 30:
        nf = len(best_h)
        tl = min(35, remaining * 0.8)
        h_a, c5_a = optimize_anneal(best_h, nf, 5.0, 1.0, tl, lr=0.004)
        if c5_a < best_c5:
            best_c5, best_h = c5_a, h_a

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
