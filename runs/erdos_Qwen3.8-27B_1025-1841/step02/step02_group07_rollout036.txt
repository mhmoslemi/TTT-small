def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def quantize_to_binary(h, n):
    """Quantize h to binary, maintaining sum = n/2."""
    target = n / 2.0
    # Sort by value, set top n/2 to 1
    sorted_idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[sorted_idx[:int(target)]] = 1.0
    return h_bin

def compute_corr(h):
    """Compute the full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def binary_local_search(h_bin, n, max_passes=100, deadline=None, verbose=False):
    """Local search on binary solutions by swapping 1s and 0s."""
    target = n // 2
    h = h_bin.copy()
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    corr = compute_corr(h)
    best_c5 = np.max(corr)
    dx = 2.0 / n
    
    for pass_num in range(max_passes):
        if deadline is not None and time.time() > deadline:
            break
        
        improved = False
        # For each 1, try swapping with each 0
        for i in ones:
            if deadline is not None and time.time() > deadline:
                break
            for j in zeros:

                    continue

def compute_swap_change(h, i, j, n):
    """Compute the change in correlation vector when h[i]:1->0 and h[j]:0->1."""
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    
    delta = np.zeros(2*n - 1)
    
    # h[i]: 1->0
    # Subtract (1-h) from positions [n-1-i, 2n-2-i]
    start1 = n - 1 - i
    end1 = 2*n - 1 - i  # exclusive
    delta[start1:end1] -= one_minus_h
    
    # Add h[::-1] to positions [i, i+n-1]
    start2 = i
    end2 = i + n  # exclusive
    delta[start2:end2] += h_rev
    
    # h[j]: 0->1
    # Add (1-h) to positions [n-1-j, 2n-2-j]
    start3 = n - 1 - j
    end3 = 2*n - 1 - j
    delta[start3:end3] += one_minus_h
    
    # Subtract h[::-1] from positions [j, j+n-1]
    start4 = j
    end4 = j + n
    delta[start4:end4] -= h_rev
    
    return delta

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)

    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def binary_local_search(h_bin, n, deadline, max_scans=50):
    """Systematic local search on binary solutions."""
    h = h_bin.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    best_max = np.max(corr)
    dx = 2.0 / n
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    
    for scan in range(max_scans):
        if time.time() > deadline:
            break
        
        improved = False
        # Try all pairs (i in ones, j in zeros)
        for i in ones:
            if time.time() > deadline:
                break
            # Precompute the delta contribution from i
            delta_i = np.zeros(2*n - 1)
            delta_i[n-1-i : 2*n-1-i] -= one_minus_h
            delta_i[i : i+n] += h_rev
            
            for j in zeros:

                    continue

                delta_j[n-1-j : 2*n-1-j] += one_minus_h
                delta_j[j : j+n] -= h_rev
                
                total_delta = delta_i + delta_j
                new_corr = corr + total_delta
                new_max = np.max(new_corr)
                
                if new_max < best_max - 1e-15:
                    # Accept
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    best_max = new_max
                    one_minus_h = 1.0 - h
                    h_rev = h[::-1]
                    ones = np.where(h == 1.0)[0]
                    zeros = np.where(h == 0.0)[0]

                    break
            
            if improved:
                break
        
        if not improved:

            if len(ones) > 0 and len(zeros) > 0:
                i = ones[np.random.randint(len(ones))]
                j = zeros[np.random.randint(len(zeros))]
                if i != j:

                    ones = np.where(h == 1.0)[0]

            else:
                break
        
        if time.time() > deadline:
            break
    
    return h, best_max * dx
