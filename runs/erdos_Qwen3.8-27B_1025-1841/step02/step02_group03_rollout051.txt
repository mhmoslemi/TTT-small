            for cand in [0.0, 1.0, 0.25, 0.75]:
                if abs(cand - old_val) < 1e-10:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-15:
                    best_c = c
                    best_val = cand
            for scale in [0.005, 0.01, 0.02, 0.05]:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * scale
                    if 0.0 < v < 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved_any = True
        if not improved_any:
            break

    # Phase 4: SLSQP polish with high T
    if remaining() > 40:
        n_pts = len(best_h)
        _G_N = n_pts
        _G_T = 800.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_full, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 300, 'ftol': 1e-15})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 5: Interleaved coordinate descent + shift-targeted SLSQP
    for interleaved in range(4):
        if remaining() < 60:
            break
        # Coord descent pass
        n_pts = len(best_h)
        for j in range(n_pts):
            if remaining() < 15:
                break
            old_val = best_h[j]
            best_val = old_val
            best_c = best_c5
            for step in [0.002, 0.005, 0.01, 0.02, 0.05]:
                for sign in [-1.0, 1.0]:
                    v = old_val + sign * step
                    if 0.0 <= v <= 1.0:
                        trial = best_h.copy()
                        trial[j] = v
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c - 1e-15:
                            best_c = c
                            best_val = v
            if best_val != old_val:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c

        # Shift-targeted SLSQP
        if remaining() > 30:
            corr = get_corr(best_h)
            mc = np.max(corr)
            threshold = mc - 0.005 * mc
            top_idx = np.where(corr >= threshold)[0]
            if len(top_idx) > 15:
                top_vals = corr[top_idx]
                sort_order = np.argsort(-top_vals)
                top_idx = top_idx[sort_order[:15]]
            elif len(top_idx) == 0:
                top_idx = np.array([int(np.argmax(corr))])
            _G_N = n_pts
            _G_T = 400.0
            _G_TOP_IDX = top_idx
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
            try:
                res = minimize(_obj_top_shifts, best_h, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 200, 'ftol': 1e-15})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_pts)
            except Exception:
                pass

    # Phase 6: Try upsample to 1920 if time allows
    if remaining() > 250:
        n_up = len(best_h) * 2
        if n_up <= 1920:
            h_up = upsample_h(best_h, n_up)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5 + 0.001:
                _G_N = n_up
                _G_T = 200.0
                _G_TOP_IDX = None
                bounds = [(0.0, 1.0)] * n_up
                cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
                try:
                    res = minimize(_obj_full, h_up, method='SLSQP',
                                  bounds=bounds, constraints=cons,
                                  options={'maxiter': 250, 'ftol': 1e-15})
                    c5_res = compute_c5(res.x)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = make_feasible(res.x, n_up)
                except Exception:
                    pass
                # Shift-targeted at higher res
                if remaining() > 100:
                    n_pts2 = len(best_h)
                    corr2 = get_corr(best_h)
                    mc2 = np.max(corr2)
                    thr2 = mc2 - 0.01 * mc2
                    ti2 = np.where(corr2 >= thr2)[0]
                    if len(ti2) > 20:
                        tv2 = corr2[ti2]
                        so2 = np.argsort(-tv2)
                        ti2 = ti2[so2[:20]]
                    elif len(ti2) == 0:
                        ti2 = np.array([int(np.argmax(corr2))])
                    _G_N = n_pts2
                    _G_T = 300.0
                    _G_TOP_IDX = ti2
                    bounds2 = [(0.0, 1.0)] * n_pts2
                    cons2 = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
                    try:
                        res2 = minimize(_obj_top_shifts, best_h, method='SLSQP',
                                       bounds=bounds2, constraints=cons2,
                                       options={'maxiter': 200, 'ftol': 1e-15})
                        c5_res2 = compute_c5(res2.x)
                        if c5_res2 < best_c5:
                            best_c5 = c5_res2
                            best_h = make_feasible(res2.x, n_pts2)
                    except Exception:
                        pass

    # Phase 7: Random restarts at best resolution
    for restart in range(4):
        if remaining() < 50:
            break
        n_pts = len(best_h)
        scale = 0.015 if restart < 2 else 0.03
        h_try = make_feasible(best_h + rng.randn(n_pts) * scale, n_pts)
        _G_N = n_pts
        _G_T = 300.0
        _G_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_full, h_try, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 250, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 8: Binary snap
    if remaining() > 20:
        n_pts = len(best_h)
        for j in range(n_pts):
            if remaining() < 5:
                break
            for val in [0.0, 1.0]:
                if abs(best_h[j] - val) < 1e-6:
                    continue
                trial = best_h.copy()
                trial[j] = val
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial

    # Phase 9: Final ultra-fine coordinate descent
    if remaining() > 15:
        n_pts = len(best_h)
        for rd in range(2):
            for j in range(n_pts):
                if remaining() < 3:
                    break
                old_val = best_h[j]
                best_val = old_val
                best_c = best_c5
                for step in [0.0005, 0.001, 0.002, 0.003, 0.005]:
                    for sign in [-1.0, 1.0]:
                        v = old_val + sign * step
                        if 0.0 <= v <= 1.0:
                            trial = best_h.copy()
                            trial[j] = v
                            trial = make_feasible(trial, n_pts)
                            c = compute_c5(trial)
                            if c < best_c - 1e-16:
                                best_c = c
                                best_val = v
                if best_val != old_val:
                    best_h[j] = best_val
                    best_h = make_feasible(best_h, n_pts)
                    best_c5 = best_c

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
