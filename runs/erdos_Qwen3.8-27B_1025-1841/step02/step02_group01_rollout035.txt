import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def resample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def slsqp_optimize(h, n_t, maxiter=500):
    n = len(h)
    dx = 2.0 / n
    
    def objective(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        soft = mc + np.log(np.sum(np.exp(n_t * shifted))) / n_t
        return soft * dx
    
    def eq_fun(h):
        return np.sum(h) - n / 2.0
    
    def eq_jac(h):
        return np.ones(n)
    
    res = minimize(objective, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

def coordinate_descent(h, max_rounds=20, time_limit=None, start_time=None):

    best_c5 = compute_c5(h)

                    return h, best_c5
            old_val = h[j]
            best_local = old_val
            best_local_c5 = best_c5
            
            candidates = [0.0, 1.0]
            # Adaptive candidates around current value
            for frac in [0.1, 0.2, 0.3, 0.5]:
                candidates.append(old_val + frac * (1.0 - old_val))
                candidates.append(old_val - frac * old_val)
            # Fine steps
            for delta in [0.005, 0.01, 0.02, 0.05]:
                candidates.append(old_val + delta)
                candidates.append(old_val - delta)
            
            for cand in candidates:
                if cand < 0.0 or cand > 1.0:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_local_c5 - 1e-12:
                    best_local_c5 = c
                    best_local = cand
            
            if best_local != old_val:
                h[j] = best_local
                h = make_feasible(h, n)
                best_c5 = best_local_c5
                improved = True
        
        if not improved:

    return h, best_c5

def subgradient_step(h, n_active=3):
    """Compute multi-lag subgradient direction."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    k = min(n_active, len(corr))
    top_lags = np.argpartition(corr, -k)[-k:]
    
    valid = np.arange(n)
    grad = np.zeros(n)
    for k_idx in top_lags:
        shift = k_idx - (n - 1)
        m1 = (valid - shift >= 0) & (valid - shift < n)
        grad[m1] += (1.0 - h[valid[m1] - shift])
        m2 = (valid + shift >= 0) & (valid + shift < n)
        grad[m2] -= h[valid[m2] + shift]
    
    grad *= dx / k
    return grad

def subgradient_method(h, n_iter=2000, lr_init=0.03, decay=0.9995, n_active=3):

import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 320
_OPT_T = 50.0
_OPT_DX = 2.0 / 320
_OPT_LAG = -1

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.array(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def resample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _obj_lse(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T
    return soft * _OPT_DX

def _obj_single_lag(h):
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr[_OPT_LAG] * _OPT_DX

def _eq_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _eq_jac(h):
    return np.ones(_OPT_N)

def slsqp_lse(h, t_val, maxiter=800):
    global _OPT_N, _OPT_T, _OPT_DX
    n = len(h)
    _OPT_N = n
    _OPT_T = t_val
    _OPT_DX = 2.0 / n
    res = minimize(_obj_lse, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def slsqp_lag(h, lag_idx, maxiter=300):
    global _OPT_N, _OPT_DX, _OPT_LAG
    n = len(h)
    _OPT_N = n
    _OPT_DX = 2.0 / n
    _OPT_LAG = lag_idx
    res = minimize(_obj_single_lag, h, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def multi_lag_subgradient(h, n_iter, lr_init, decay, n_active):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = lr_init
    valid = np.arange(n)
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = min(n_active, len(corr))
        top_lags = np.argpartition(corr, -k)[-k:]
        grad = np.zeros(n)
        for ki in top_lags:
            shift = int(ki) - (n - 1)
            m1 = (valid - shift >= 0) & (valid - shift < n)
            grad[m1] += (1.0 - h[valid[m1] - shift])
            m2 = (valid + shift >= 0) & (valid + shift < n)
            grad[m2] -= h[valid[m2] + shift]
        grad *= dx / k
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        lr *= decay
    return best_h, best_c5

def coord_descent_zoom(h, max_rounds, time_limit, start_time):
    n = len(h)
    best_c5 = compute_c5(h)
    for rd in range(max_rounds):
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit:
                    return h, best_c5
            ov = h[j]
            best_val = ov
            best_c = best_c5
            coarse = [0.0, 1.0, 0.5]
            for f in [0.1, 0.2, 0.3, 0.5, 0.8]:
                coarse.append(ov + f * (1.0 - ov))
                coarse.append(ov - f * ov)
            coarse = [c for c in coarse if 0.0 <= c <= 1.0]
            for cand in coarse:
                if abs(cand - ov) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_val = cand
            if best_val != ov:
                lo = max(0.0, best_val - 0.05)
                hi = min(1.0, best_val + 0.05)
                for _ in range(12):
                    mid_lo = lo + (hi - lo) * 0.33
                    mid_hi = lo + (hi - lo) * 0.67
                    for cand in [lo, mid_lo, (lo+hi)/2, mid_hi, hi]:
                        trial = h.copy()
                        trial[j] = cand
                        trial = make_feasible(trial, n)
                        c = compute_c5(trial)
                        if c < best_c - 1e-13:
                            best_c = c
                            best_val = cand
                    lo = mid_lo
                    hi = mid_hi
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return h, best_c5

def exact_lag_iterate(h, n_rounds, time_limit, start_time):
    n = len(h)
    best_c5 = compute_c5(h)
    for rd in range(n_rounds):
        if start_time is not None and time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_k = min(5, len(corr))
        top_lags = np.argpartition(corr, -top_k)[-top_k:]
        for lag in top_lags:
            if start_time is not None and time_limit is not None:
                if time.time() - start_time > time_limit:
                    break
            try:
                h_new = slsqp_lag(h, int(lag), maxiter=200)
                h_new = make_feasible(h_new, n)
                c = compute_c5(h_new)
                if c < best_c5:
                    h = h_new
                    best_c5 = c
            except Exception:
                pass
    return h, best_c5
