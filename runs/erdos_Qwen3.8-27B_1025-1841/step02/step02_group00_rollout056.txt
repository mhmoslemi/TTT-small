    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0

        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break

        else:
            mask = h > 1e-14

        if time.time() > deadline:
            return 1e10
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = corr.max()
        sh = corr - mc
        return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx

        return x.sum() - n / 2.0

        return np.ones(n)

        return make_feasible(res.x, n)
    except:
        return make_feasible(h_start, n)

                      options={'maxiter': max_iter, 'ftol': 1e-13})
        return make_feasible(res.x, n)
    except:
        return make_feasible(h_start, n)

def sa_search(h_start, n, deadline, rng, n_iters):
    h = to_binary(h_start, n)
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    
    T = 0.008
    T_min = 0.0002
    cool = (T_min / T) ** (1.0 / n_iters)
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        T = max(T * cool, T_min)
        
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        
        h[i], h[j] = h[j], h[i]
        c5_new = compute_c5(h)
        delta = c5_new - c5_curr
        
        if delta <= 0 or rng.rand() < np.exp(-delta / T):
            c5_curr = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

def reheat_sa(h_start, n, deadline, rng, n_phases=5, iters_per_phase=40000):
    """SA with reheating"""
    h = to_binary(h_start, n)
    c5_curr = compute_c5(h)
    best_c5 = c5_curr
    best_h = h.copy()
    
    for phase in range(n_phases):
        if time.time() > deadline:
            break
        T = 0.005 / (phase + 1)
        T_min = T * 0.1
        cool = (T_min / T) ** (1.0 / iters_per_phase)
        
        for it in range(iters_per_phase):
            if time.time() > deadline:
                break
            T = max(T * cool, T_min)
            
            i = rng.integers(n)
            j = rng.integers(n)
            if i == j:
                continue
            
            h[i], h[j] = h[j], h[i]
            c5_new = compute_c5(h)
            delta = c5_new - c5_curr
            
            if delta <= 0 or rng.rand() < np.exp(-delta / T):
                c5_curr = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
            else:
                h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5
