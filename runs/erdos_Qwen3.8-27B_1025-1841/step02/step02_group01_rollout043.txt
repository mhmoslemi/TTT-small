```python
def _cd_step(h, n, j, new_val):
    """Compute change in all correlation values when h[j] changes to new_val."""
    old_val = h[j]
    delta = new_val - old_val
    if abs(delta) < 1e-15:
        return np.zeros(2*n-1)
    dx = 2.0 / n
    idx = np.arange(n)
    dc = np.zeros(2*n-1)
    for k in range(2*n-1):
        shift = k - (n-1)
        coeff = 1.0
        ip = j + shift
        im = j - shift
        if 0 <= ip < n:
            coeff -= h[ip]
        if 0 <= im < n:
            coeff -= h[im]
        dc[k] = delta * coeff
```

For efficiency, I can use a simpler approach: just try candidate = 0 and candidate = 1, and also the intersection point of the two top lines.

Actually, for a proper implementation:
1. For the current h, compute corr (length 2n-1)
2. Identify the top M shifts (M=10 or so)
3. For each j:
   a. Compute coeff_k for each of the top M shifts (O(M))
   b. The new value at shift k is: val_k = corr[k] + (candidate - h[j]) * coeff_k
   c. We want to min over candidate ∈ [0,1] of max_{k in top M} val_k
   d. This is a 1D piecewise linear minimax. Find the optimal candidate.
   e. Update h[j] and recompute corr.

Step 2d: The function to minimize is f(x) = max_i (a_i * x + b_i) where a_i = coeff_k, b_i = corr[k] - h[j]*coeff_k.
The minimum is at a point where two of the lines cross, or at x=0 or x=1.

To find the intersection points: for each pair (i,j), x = (b_j - b_i) / (a_i - a_j) if a_i ≠ a_j.
Then evaluate f at all intersection points (that are in [0,1]) and at 0 and 1.
Take the best.

With M=10, there are C(10,2)=45 pairs, so 45+2 = 47 candidates to evaluate. Each evaluation is O(M). Total: O(M^2) per coordinate. For n coordinates: O(n*M^2). For n=320, M=10: 32000 ops per sweep. Very fast!

```python
def _minimax_1d(a, b, lo=0.0, hi=1.0):

    Returns the optimal x."""
    n = len(a)
    # Collect candidate x values
    candidates = [lo, hi]
    for i in range(n):

    for x in candidates:
        val = np.max(a * x + b)
        if val < best_val:
            best_val = val
            best_x = x
    return best_x
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_GN = 0

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _project(v, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def _softobj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (mc + log_sum / temp) * dx

def _con_eq(h):
    return np.sum(h) - _GN / 2.0

def _con_jac(h):
    return np.ones(_GN)

def _slsqp(h0, n, temp, maxiter):
    global _GN
    _GN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    try:
        res = minimize(_softobj, h0, args=(n, temp), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})
        return res.x
    except Exception:
        return h0.copy()

def _upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return _make_feasible(h_new, new_n)

def _subgradient(h0, n, max_iters, step0, decay):
    h = h0.copy()
    best_c5 = _compute_c5(h)
    best_h = h.copy()
    step = step0
    dx = 2.0 / n
    idx = np.arange(n)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        thresh = mc - max(0.005 * mc, 1.0)
        near = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        for k in near:
            shift = int(k) - (n - 1)
            g = np.ones(n)
            vp = (idx + shift >= 0) & (idx + shift < n)
            g[vp] -= h[idx[vp] + shift]
            vm = (idx - shift >= 0) & (idx - shift < n)
            g[vm] -= h[idx[vm] - shift]
            grad += g
        grad /= (len(near) * dx)
        h = _project(h - step * grad, n)
        c5 = _compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= decay
    return best_h, best_c5

def _minimax_1d(a, b):
    n = len(a)
    candidates = [0.0, 1.0]
    for i in range(n):
        for j in range(i + 1, n):
            denom = a[i] - a[j]
            if abs(denom) > 1e-14:
                x = (b[j] - b[i]) / denom
                if 0.0 <= x <= 1.0:
                    candidates.append(x)
    best_x = 0.0
    best_val = float('inf')
    for x in candidates:
        val = float(np.max(a * x + b))
        if val < best_val:
            best_val = val
            best_x = x
    return best_x

def _coord_desc_minimax(h0, n, max_sweeps, active_count):
    h = h0.copy()
    best_c5 = _compute_c5(h)
    dx = 2.0 / n
    for sweep in range(max_sweeps):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            # Get active shifts
            top_idx = np.argsort(corr)[-active_count:]
            shifts = top_idx - (n - 1)
            # Compute coefficients for each active shift
            a = np.ones(active_count)
            for i, sh in enumerate(shifts):
                ip = j + int(sh)
                im = j - int(sh)
                if 0 <= ip < n:
                    a[i] -= h[ip]
                if 0 <= im < n:
                    a[i] -= h[im]
            # b_k = corr[k] - h[j] * a_k
            b = corr[top_idx].astype(float) - h[j] * a
            # Find optimal h[j]
            best_j = _minimax_1d(a, b)
            if abs(best_j - h[j]) > 1e-10:
                old_max = np.max(corr)
                h[j] = best_j
                h = _make_feasible(h, n)
                # Recompute corr
                corr = np.correlate(h, 1.0 - h, mode='full')
                if np.max(corr) * dx < best_c5 - 1e-12:
                    best_c5 = np.max(corr) * dx
                    improved = True
                else:
                    # Revert if sum constraint rebalancing hurt
                    h[j] = best_j  # keep the value, the rebalancing happened
                    # Actually let's just keep it if the max didn't increase
                    if np.max(corr) * dx <= best_c5 + 1e-10:
                        best_c5 = np.max(corr) * dx
                        improved = True
        if not improved:
            break
    return h, best_c5

def _binary_round(h, n):
    """Round h to binary, maintaining sum = n/2."""
    h_bin = (h >= 0.5).astype(float)
    ones = int(np.sum(h_bin))
    target = n // 2
    if ones > target:
        # Flip some 1s to 0s (those with lowest h values)
        ones_idx = np.where(h_bin == 1)[0]
        sorted_by_h = ones_idx[np.argsort(h[ones_idx])]
        h_bin[sorted_by_h[:ones - target]] = 0.0
    elif ones < target:
        # Flip some 0s to 1s (those with highest h values)
        zeros_idx = np.where(h_bin == 0)[0]
        sorted_by_h = zeros_idx[np.argsort(-h[zeros_idx])]
        h_bin[sorted_by_h[:target - ones]] = 1.0
    return h_bin

def _binary_local_search(h_bin, n, max_sweeps=30):
    """Local search on binary: try all swaps, accept improving ones."""
    h = h_bin.copy().astype(float)
    best_c5 = _compute_c5(h)
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    idx = np.arange(n)
    
    for sweep in range(max_sweeps):
        improved = False
        corr = np.correlate(h, 1.0 - h, mode='full')
        dx = 2.0 / n
        for i in ones:
            for j in zeros:
                if i == j:
                    continue
                # Compute Δc_k for swap i:1->0, j:0->1
                # For each k, shift = k - (n-1)
                # Δc_k = -(1-h[i+sh])*1_{valid} + h[i-sh]*1_{valid} + (1-h[j+sh])*1_{valid} - h[j-sh]*1_{valid}
                shifts_full = np.arange(2*n-1) - (n-1)
                dc = np.zeros(2*n-1)
                
                ip = i + shifts_full
                vmip = (ip >= 0) & (ip < n)
                # -(1-h[ip]) for valid ip = -1 + h[ip] for valid ip
                dc[vmip] += (-1.0 + h[ip[vmip]])
                
                im = i - shifts_full
                vmim = (im >= 0) & (im < n)
                dc[vmim] += h[im[vmim]]
                
                jp = j + shifts_full
                vmjp = (jp >= 0) & (jp < n)
                dc[vmjp] += (1.0 - h[jp[vmjp]])
                
                jm = j - shifts_full
                vmjm = (jm >= 0) & (jm < n)
                dc[vmjm] += (-h[jm[vmjm]])
                
                new_corr = corr + dc
                new_max = np.max(new_corr) * dx
                if new_max < best_c5 - 1e-12:
                    # Accept swap
                    h[i] = 0.0
                    h[j] = 1.0
                    corr = new_corr
                    best_c5 = new_max
                    best_h = h.copy()
                    improved = True
                    # Update ones and zeros
                    ones = ones[ones != i]
                    zeros = zeros[(zeros != j)]
                    ones = np.append(ones, j)
                    zeros = np.append(zeros, i)
        if not improved:
            break
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_GN = 0

def _compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(1000):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def _project(v, n):
    target = n / 2.0
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def _softobj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(shifted)))
    return (mc + log_sum / temp) * dx

def _con_eq(h):
    return np.sum(h) - _GN / 2.0

def _con_jac(h):
    return np.ones(_GN)

def _slsqp(h0, n, temp, maxiter):
    global _GN
    _GN = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_eq, 'jac': _con_jac}]
    try:
        res = minimize(_softobj, h0, args=(n, temp), method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-12})

    old_n = len(h)

    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return _make_feasible(h_new, new_n)

def _subgradient(h0, n, max_iters, step0, decay):
    h = h0.copy()
    best_c5 = _compute_c5(h)
    best_h = h.copy()
    step = step0
    dx = 2.0 / n
    idx = np.arange(n)
    for it in range(max_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        thresh = mc - max(0.005 * mc, 1.0)
        near = np.where(corr >= thresh)[0]
        grad = np.zeros(n)
        for k in near:
            shift = int(k) - (n - 1)
            g = np.ones(n)
            vp = (idx + shift >= 0) & (idx + shift < n)
            g[vp] -= h[idx[vp] + shift]
            vm = (idx - shift >= 0) & (idx - shift < n)
            g[vm] -= h[idx[vm] - shift]
            grad += g
        grad /= (len(near) * dx)
        h = _project(h - step * grad, n)
        c5 = _compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        step *= decay
    return best_h, best_c5

def _minimax_1d(a, b):
    nlen = len(a)
    candidates = [0.0, 1.0]
    for i in range(nlen):
        for j in range(i + 1, nlen):
            denom = a[i] - a[j]
            if abs(denom) > 1e-14:
                x = (b[j] - b[i]) / denom
                if 0.0 <= x <= 1.0:
                    candidates.append(x)

    best_val = float('inf')
    for x in candidates:
        val = float(np.max(a * x + b))
        if val < best_val:
            best_val = val
            best_x = x
    return best_x
```
