def smooth_obj_grad(h, tau, top_k=None):

    dx = 2.0 / n
    corr = compute_corr(h)
    if top_k is not None and top_k < len(corr):

        c_use = corr[idx]
    else:
        idx = np.arange(len(corr))
        c_use = corr
    
    max_c = c_use.max()
    shifted = np.clip(tau * (c_use - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    grad = np.zeros(n)
    for i in range(len(c_use)):
        pk = p[i]
        if pk < 1e-18:
            continue
        ki = idx[i]
        sh = ki - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            js = np.arange(j0, j1)
            grad[js] += pk * (1.0 - h[js - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            js = np.arange(j0b, j1b)
            grad[js] -= pk * h[js + sh]
    grad *= dx
    return obj, grad

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)
