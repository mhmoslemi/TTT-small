sum_j h[j+k]*h[j]

The terms involving h[i]:

- j = i: term is h[i+k]*h[i] (if 0≤i+k<n)

So the coefficient of h[i] in sum_j h[j+k]*h[j] is:
h[i-k]*[0≤i-k<n] + h[i+k]*[0≤i+k<n]

And the coefficient of h[i] in sum_j h[j+k] is:
[0≤i-k<n and 0≤i<n] = [0≤i-k<n] (since 0≤i<n is always true)

So the coefficient D_k of h[i] in corr[k] = sum h[j+k] - sum h[j+k]*h[j] is:

This is a concave quadratic in h[i]. The max of a concave quadratic and linear functions is still convex (actually, the max of linear functions is convex, and max with a concave function is... not necessarily convex).

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def get_softmax_gradient(h, tau, topM=0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    if topM > 0 and topM < (2 * n - 1):
        top_idx = np.argpartition(g, -topM)[-topM:]
        g_sub = g[top_idx]
        max_g = g_sub.max()
        shifted = np.clip(tau * (g_sub - max_g), -500, 500)
        exp_c = np.exp(shifted)
        weights = np.zeros(2 * n - 1)
        weights[top_idx] = exp_c / exp_c.sum()
    else:
        max_g = g.max()
        shifted = np.clip(tau * (g - max_g), -500, 500)
        exp_c = np.exp(shifted)
        weights = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-15:
            continue
        d = k - n + 1
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += w * (1.0 - h[:-d])
            grad[:-d] += w * (-h[d:])
        else:
            dd = -d
            grad[:n - dd] += w * (1.0 - h[dd:])
            grad[dd:] += w * (-h[:n - dd])
    return grad * dx

def softmax_optimize(h, n, tau, max_iters, time_limit, lr_init=0.15, topM=0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_obj = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(80, max_iters // 20)
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        cur_max = corr.max() * dx
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        grad = get_softmax_gradient(h, tau, topM)
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    return best_h, best_obj

def coordinate_refine(h, n, n_sweeps=5, time_limit=100.0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    grid = np.linspace(0.0, 1.0, 50)
    for sweep in range(n_sweeps):
        if time.time() - start > time_limit:
            break
        order = np.random.permutation(n)
        for i in order:
            if time.time() - start > time_limit:
                break
            d_coeffs = np.zeros(2 * n - 1)
            for k in range(2 * n - 1):
                d = k - n + 1
                if d == 0:
                    d_coeffs[k] = 1.0 - 2.0 * h[i]
                elif d > 0:
                    if i - d >= 0:
                        d_coeffs[k] += (1.0 - h[i - d])
                    if i + d < n:
                        d_coeffs[k] += (-h[i + d])
                else:
                    dd = -d
                    if i + dd < n:
                        d_coeffs[k] += (1.0 - h[i + dd])
                    if i - dd >= 0:
                        d_coeffs[k] += (-h[i - dd])
            # For k=0 (k index n-1), handle quadratic separately
            base_corr = corr - h[i] * d_coeffs
            # corr_new[k] = base_corr[k] + t * d_coeffs[k] for k != n-1
            # corr_new[n-1] = base_corr[n-1] + t*(1-t) = base_corr[n-1] + t - t^2
            # Evaluate on grid
            c5_vals = np.empty(len(grid))
            for gi, t in enumerate(grid):
                corr_new = base_corr + t * d_coeffs
                corr_new[n - 1] = base_corr[n - 1] + t * (1.0 - t)
                c5_vals[gi] = corr_new.max()
            best_gi = np.argmin(c5_vals)
            t_best = grid[best_gi]
            if abs(t_best - h[i]) > 1e-10:
                corr = base_corr + t_best * d_coeffs
                corr[n - 1] = base_corr[n - 1] + t_best * (1.0 - t_best)
                h[i] = t_best
        # Project after sweep
        h = project_to_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)
        cur_c5 = corr.max() * dx
        if cur_c5 < best_c5 - 1e-15:
            best_c5 = cur_c5
            best_h = h.copy()
    return best_h, best_c5

def subgrad_optimize(h, n, max_iters, time_limit, lr_init=0.004, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = np.correlate(h, 1.0 - h, mode='full').max() * dx
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    patience = 500
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            if d == 0:
                grad += (1.0 - 2.0 * h)
            elif d > 0:
                grad[d:] += (1.0 - h[:-d])
                grad[:-d] += (-h[d:])
            else:
                dd = -d
                grad[:n - dd] += (1.0 - h[dd:])
                grad[dd:] += (-h[:n - dd])
        grad *= dx / topK
        cur_lr = lr / (1.0 + it * 0.003)
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-13:
                break
        h = h_new
    return best_h, best_c5
