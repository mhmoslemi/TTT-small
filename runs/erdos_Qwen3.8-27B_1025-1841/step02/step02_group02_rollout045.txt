    lo = g.min() - 2.0

    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            sh_idx = np.arange(j0, j1) - sh
            grad[j0:j1] += pk * (1.0 - h[sh_idx])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            sh_idx = np.arange(j0b, j1b) + sh
            grad[j0b:j1b] -= pk * h[sh_idx]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_adam(h, tau, max_iters, time_limit, lr):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    _, best_obj = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    u = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    stall = 0

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-12:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        m = b1 * m + (1.0 - b1) * grad
        u = b2 * u + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        u_hat = u / (1.0 - b2 ** t)
        step = lr * m_hat / (np.sqrt(u_hat) + eps)
        h = project_to_feasible(h - step, n)

        if stall > 150:
            lr *= 0.5
            m *= 0.5
            u *= 0.25
            stall = 0
            if lr < 1e-12:
                break

    return best_h


def optimize_subgrad(h, max_iters, time_limit, lr):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    stall = 0

    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[np.arange(j0b, j1b) + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

        if stall > 400:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return max(0.0, budget_s - elapsed())

    h_best = initial_h_values.copy()
    n_best = len(h_best)
    c5_best = compute_c5(h_best)

    # Phase 1: Upsample to n=1200, multi-tau optimization
    n1 = 1200
    if remaining() > 250:
        h1 = interp_periodic(h_best, n1)
        h1 = project_to_feasible(h1, n1)
        tau_sched = [5.0, 10.0, 20.0, 40.0, 80.0, 160.0]
        lr_sched = [0.6, 0.4, 0.3, 0.2, 0.1, 0.05]
        t_per = min(40.0, remaining() * 0.28 / len(tau_sched))
        for i, tau in enumerate(tau_sched):
            if remaining() < 60:
                break
            h1 = optimize_adam(h1, tau, max_iters=20000, time_limit=t_per, lr=lr_sched[i])
        c5_1 = compute_c5(h1)
        if c5_1 < c5_best:
            c5_best = c5_1
            h_best = h1.copy()
            n_best = n1

    # Phase 2: Upsample to n=1800, multi-tau
    n2 = 1800
    if remaining() > 200:
        h2 = interp_periodic(h_best, n2)
        h2 = project_to_feasible(h2, n2)
        tau_sched2 = [10.0, 20.0, 40.0, 80.0, 160.0, 320.0]
        lr_sched2 = [0.4, 0.25, 0.15, 0.08, 0.04, 0.02]
        t_per2 = min(35.0, remaining() * 0.30 / len(tau_sched2))
        for i, tau in enumerate(tau_sched2):
            if remaining() < 50:
                break
            h2 = optimize_adam(h2, tau, max_iters=15000, time_limit=t_per2, lr=lr_sched2[i])
        c5_2 = compute_c5(h2)
        if c5_2 < c5_best:
            c5_best = c5_2
            h_best = h2.copy()
            n_best = n2

    # Phase 3: Subgradient refinement at current resolution
    if remaining() > 100:
        h3 = h_best.copy()
        h3, c5_3 = optimize_subgrad(h3, max_iters=200000, time_limit=min(remaining() * 0.3, 250.0), lr=0.004)
        if c5_3 < c5_best:
            c5_best = c5_3
            h_best = h3.copy()

    # Phase 4: Random perturbations + re-optimize
    if remaining() > 120:
        n_p = len(h_best)
        for trial in range(4):
            if remaining() < 40:
                break
            amp = 0.005 + 0.005 * trial
            perturb = amp * np.random.randn(n_p)
            h_p = project_to_feasible(h_best + perturb, n_p)
            h_p = optimize_adam(h_p, tau=40.0, max_iters=10000, time_limit=min(remaining() * 0.12, 50.0), lr=0.1)
            h_p = optimize_adam(h_p, tau=160.0, max_iters=10000, time_limit=min(remaining() * 0.08, 30.0), lr=0.03)
            c5_p = compute_c5(h_p)
            if c5_p < c5_best:
                c5_best = c5_p
                h_best = h_p.copy()

    # Phase 5: Final subgradient polish with small lr
    if remaining() > 40:
        h_f = h_best.copy()
        h_f, c5_f = optimize_subgrad(h_f, max_iters=200000, time_limit=remaining() * 0.45, lr=0.001)
        if c5_f < c5_best:
            c5_best = c5_f
            h_best = h_f.copy()

    # Phase 6: One more high-tau smooth pass if time allows
    if remaining() > 60:
        n_final = len(h_best)
        h_fin = h_best.copy()
        h_fin = optimize_adam(h_fin, tau=400.0, max_iters=20000, time_limit=min(remaining() * 0.35, 80.0), lr=0.01)
        c5_fin = compute_c5(h_fin)
        if c5_fin < c5_best:
            c5_best = c5_fin
            h_best = h_fin.copy()

    # Final feasibility check
    h_best = project_to_feasible(h_best, len(h_best))
    c5_final = compute_c5(h_best)
    if c5_final < c5_best:
        c5_best = c5_final

    return h_best, c5_best, len(h_best)
