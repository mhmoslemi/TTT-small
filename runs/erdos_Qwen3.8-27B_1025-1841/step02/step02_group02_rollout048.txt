    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    t = (lo + hi) / 2.0

    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    # Periodic interpolation
    h_ext = np.concatenate([h[-n//2:], h, h[:n//2]])

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(g - t, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    x_old_ext = np.concatenate([x_old - 2.0, x_old, x_old + 2.0])
    h_ext = np.concatenate([h, h, h])
    return np.interp(x_new, x_old_ext, h_ext)


def compute_c5_full(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr


def topk_gradient(h, corr, k_top, temperature):
    n = len(h)
    dx = 2.0 / n
    max_corr = corr.max()
    k = min(k_top, len(corr))
    top_idx = np.argpartition(corr, -k)[-k:]
    top_vals = corr[top_idx]
    weights = np.exp(temperature * (top_vals - max_corr))
    wsum = weights.sum()
    if wsum < 1e-30:
        weights = np.ones(k) / k
    else:
        weights /= wsum
    grad = np.zeros(n)
    for idx, w in zip(top_idx, weights):
        if w < 1e-12:
            continue
        sh = idx - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += w * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= w * h[j0b + sh:j1b + sh]
    grad *= dx
    return grad, max_corr * dx

    stall = 0
    sp = stall_pat

        y = h + beta * v
        corr_y = np.correlate(y, 1.0 - y, mode='full')
        grad, _ = topk_gradient(y, corr_y, k_top, temp_cur)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

        if (int)((time.time() - t0) % (eval_interval * 0.01) == 0):
            pass
        it = int((time.time() - t0) * 100)
        if it % eval_interval == 0:
            c5_now, _ = compute_c5_full(h)

                best_c5 = c5_now

                stall = 0

            v *= 0.2
            stall = 0

    hs = h.copy()
    hs[hs < thr] = 0.0
    hs[hs > 1.0 - thr] = 1.0
    return project_to_feasible(hs, len(hs))
