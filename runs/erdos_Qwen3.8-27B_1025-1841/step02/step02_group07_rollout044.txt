        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)

    h_sym = (h[:half] + h[n-half:][::-1].copy()) / 2.0
    full = np.empty(n)
    full[:half] = h_sym
    full[half:] = h_sym[::-1]
    return full

def interp_to(h_from, n_to):

    if n_from == n_to:
        return h_from.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h_from)

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    c_raw = np.correlate(h, 1.0 - h, mode='full')
    c = c_raw * dx
    m = np.max(c)
    exp_c = np.exp(c - m)
    w = exp_c / np.sum(exp_c)
    
    J = np.arange(n)[:, None]
    I = np.arange(n)[None, :]
    idx = n - 1 + I - J
    W = w[idx]
    
    term1 = W @ (1.0 - h)
    term2 = W.T @ h
    return dx * (term1 - term2)

def pgd_step(h, n, T, lr, symmetric=True):
    grad = compute_grad(h, T)
    if symmetric:
        # Project gradient onto symmetric subspace
        half = n // 2
        grad_sym = (grad[:half] + grad[n-half:][::-1].copy()) / 2.0
        grad_h_sym = np.zeros(n)
        grad_h_sym[:half] = grad_sym
        grad_h_sym[half:] = grad_sym[::-1]
        grad = grad_h_sym
    
    h_new = h - lr * grad
    h_new = project_feasible(h_new, n)
    if symmetric:
        h_new = symmetrize(h_new, n)
        h_new = project_feasible(h_new, n)
    return h_new

def pgd_optimize(h0, n, T_schedule, lr0, symmetric=True, max_iter=2000, deadline=None):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    lr = lr0
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        T = T_schedule[min(it // 100, len(T_schedule)-1)]
        
        # Backtracking line search
        alpha = lr
        f_cur = evaluate_c5(h)
        for bt in range(20):
            h_try = pgd_step(h, n, T, alpha, symmetric)
            f_try = evaluate_c5(h_try)
            if f_try < f_cur + 1e-12:
                break
            alpha *= 0.5
        
        h = h_try
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adapt lr
        if it % 100 == 99:
            lr *= 0.8
    
    return best_h, best_c5
