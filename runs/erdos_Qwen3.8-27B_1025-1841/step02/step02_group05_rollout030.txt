import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    # Same as before but more efficient
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    obj = dx * (max_c + np.log(s) / tau)
    
    # Gradient computation
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return obj, grad

def optimize_smooth(h, n, tau, time_limit, lr=None, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.4 / tau
    v = np.zeros(n)
    beta = momentum
    stall = 0
    
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 1500:
            lr *= 0.6
            v *= 0.2
            stall = 0
            if lr < 1e-12:
                break
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5

def optimize_subgrad(h, n, time_limit, lr=None):
    # Same as before
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.003
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 8000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:

        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5

        # Sort by fitness
        order = np.argsort(fitness)
        pop = [pop[i] for i in order]
        fitness = [fitness[i] for i in order]
        
        # Create new population
        new_pop = []
        new_fitness = []
        
        for i in range(pop_size):
            if time.time() - start > budget_s:
                break
            
            # Selection: tournament of 3
            idx1, idx2 = np.random.choice(range(min(10, pop_size)), size=2, replace=False)
            if fitness[idx1] < fitness[idx2]:
                p1 = pop[idx1]
            else:
                p1 = pop[idx2]
            
            if np.random.random() < 0.7:
                # Crossover with another selection
                idx3, idx4 = np.random.choice(range(min(10, pop_size)), size=2, replace=False)
                if fitness[idx3] < fitness[idx4]:
                    p2 = pop[idx3]
                else:
                    p2 = pop[idx4]
                
                # Blend crossover
                alpha = np.random.uniform(0.3, 0.7)
                child = alpha * p1 + (1 - alpha) * p2
            else:
                child = p1.copy()
            
            # Mutation
            if np.random.random() < 0.8:
                noise_scale = 0.1 * (1 - gen / max(n_gen, 1))
                child = child + np.random.randn(n) * noise_scale
            else:
                # Block mutation
                block_size = np.random.randint(10, 50)
                start_idx = np.random.randint(0, n - block_size)
                child[start_idx:start_idx+block_size] = np.random.rand(block_size)
            
            child = project_to_feasible(child, n)
            new_pop.append(child)
            new_fitness.append(compute_c5(child))
        
        # Combine and select
        combined_pop = pop + new_pop
        combined_fit = fitness + new_fitness
        combined_order = np.argsort(combined_fit)
        
        pop = [combined_pop[i] for i in combined_order[:pop_size]]
        fitness = [combined_fit[i] for i in combined_order[:pop_size]]
    
    return pop[0], fitness[0]

        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - (n - 1)
        lo1 = max(0, -sh)
        hi1 = min(n, n - sh)
        if lo1 < hi1:
            idx = np.arange(lo1, hi1)
            grad_h[idx] += pk * (1.0 - h[idx + sh])
        lo2 = max(0, sh)
        hi2 = min(n, n + sh)
        if lo2 < hi2:
            idx = np.arange(lo2, hi2)
            grad_h[idx] -= pk * h[idx - sh]
    grad_h *= dx
    return obj, grad_h


def optimize_smooth_anneal(h, n, time_limit, taus, base_lr=0.5, noise_every=3000, noise_scale=0.03):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    v = np.zeros(n)
    beta = 0.9
    
    for tau in taus:
        if time.time() - t0 > time_limit:
            break
        lr = base_lr / tau
        stall = 0
        
        while time.time() - t0 < time_limit:
            obj, grad = smooth_obj_grad(h, tau)
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            
            if noise_every > 0 and stall > noise_every:
                h = project_to_feasible(h + np.random.randn(n) * noise_scale / max(tau, 1.0), n)
                v *= 0.3
                stall = 0
            
            if stall > 4000:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            
            v = beta * v - lr * grad
            h = project_to_feasible(h + v, n)
    
    return best_h, best_c5
