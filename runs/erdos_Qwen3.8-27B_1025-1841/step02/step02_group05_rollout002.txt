                break
    return best_h, best_c5


def subgrad_polish(h, n, time_limit, lr=0.002, top_k=3, beta=0.85):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    stall = 0

    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = float(corr.max()) * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            lr *= 0.7
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break

        if top_k > 1:
            idx_sorted = np.argsort(corr)[::-1][:top_k]
            top_vals = corr[idx_sorted]
            w = np.exp(top_vals - top_vals[0])
            w /= w.sum()
            grad = np.zeros(n)
            for i in range(top_k):
                pk = float(w[i])
                sh = int(idx_sorted[i]) - n + 1
                i0 = max(0, -sh)
                i1 = min(n, n - sh)
                if i0 < i1:
                    grad[i0:i1] += pk * (1.0 - h[i0 + sh:i1 + sh])
                j0 = max(0, i0 + sh)
                j1 = min(n, i1 + sh)
                if j0 < j1:
                    grad[j0:j1] -= pk * h[j0 - sh:j1 - sh]
            grad *= dx
        else:
            k_star = int(np.argmax(corr))
            sh = k_star - n + 1
            grad = np.zeros(n)
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 < i1:
                grad[i0:i1] += (1.0 - h[i0 + sh:i1 + sh])
            j0 = max(0, i0 + sh)
            j1 = min(n, i1 + sh)
            if j0 < j1:
                grad[j0:j1] -= h[j0 - sh:j1 - sh]
            grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def remaining_time(start_time, budget_s):
    return max(0.0, budget_s - (time.time() - start_time))


def run(seed=42, budget_s=1300, **kwargs):
    global initial_h_values
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: n=2000, momentum smooth max with fixed tau levels
    n1 = 2000
    h1 = interp_periodic(h_init, n1)
    h1 = project_to_feasible(h1, n1)

    tau_schedule = [(200.0, 0.5, 90), (80.0, 0.4, 90), (25.0, 0.25, 90), (8.0, 0.12, 80), (3.0, 0.06, 60)]
    for tau, lr, tl in tau_schedule:
        if remaining_time(start_time, budget_s) < 50:
            break
        actual_tl = min(tl, remaining_time(start_time, budget_s) * 0.18)
        h1, c5_t = optimize_momentum(h1, n1, tau, time_limit=actual_tl, lr=lr, beta=0.92)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_h = h1.copy()

    # Phase 2: Subgradient polish at n=2000
    if remaining_time(start_time, budget_s) > 60:
        tl = min(60, remaining_time(start_time, budget_s) * 0.25)
        h_f, c5_f = subgrad_polish(best_h, n1, time_limit=tl, lr=0.002, top_k=5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 3: Upsample to n=4000
    if remaining_time(start_time, budget_s) > 180:
        n2 = 4000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)

        tau2_schedule = [(100.0, 0.4, 80), (30.0, 0.2, 70), (8.0, 0.1, 60)]
        for tau, lr, tl in tau2_schedule:
            if remaining_time(start_time, budget_s) < 50:
                break
            actual_tl = min(tl, remaining_time(start_time, budget_s) * 0.2)
            h2, c5_t = optimize_momentum(h2, n2, tau, time_limit=actual_tl, lr=lr, beta=0.92)
            if c5_t < best_c5:
                best_c5 = c5_t
                best_h = h2.copy()

        if remaining_time(start_time, budget_s) > 40:
            tl = min(50, remaining_time(start_time, budget_s) * 0.3)
            h_f, c5_f = subgrad_polish(h2, n2, time_limit=tl, lr=0.001, top_k=5)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 4: Try annealing at the best resolution found
    if remaining_time(start_time, budget_s) > 100:
        n3 = len(best_h)
        if n3 < 3000:
            n3 = 3000
            h3 = interp_periodic(best_h, n3)
        else:
            h3 = best_h.copy()
        h3 = project_to_feasible(h3, n3)
        tl = min(100, remaining_time(start_time, budget_s) * 0.5)
        h3, c5_3 = optimize_anneal(h3, n3, tau_max=50.0, tau_min=2.0, time_limit=tl, lr_start=0.3, beta=0.9, total_iters=30000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 5: Final subgradient polish
    if remaining_time(start_time, budget_s) > 30:
        n_f = len(best_h)
        tl = min(40, remaining_time(start_time, budget_s) * 0.6)
        h_f, c5_f = subgrad_polish(best_h, n_f, time_limit=tl, lr=0.0008, top_k=7)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
