        return h

def coord_descent(h, n, n_rounds, step_sizes, budget_check):
    best = h.copy()
    best_c = compute_c5(h)
    for rnd in range(n_rounds):
        if budget_check() < 10:
            break
        improved = False
        for j in range(n):
            if budget_check() < 5:
                break
            old = best[j]
            best_trial = old
            best_c_l = best_c
            cands = [0.0, 1.0, 0.5]
            for s in step_sizes:
                cands.append(max(0.0, old - s))
                cands.append(min(1.0, old + s))
            seen = set()
            for cand in cands:
                if cand in seen:
                    continue
                seen.add(cand)
                trial = best.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_l - 1e-14:
                    best_c_l = c
                    best_trial = cand
            if best_trial != old:
                best[j] = best_trial
                best = make_feasible(best, n)
                best_c = best_c_l
                improved = True
        if not improved:
            break
    return best, best_c

def block_perturb_search(h, n, rng, n_iterations, budget_check):
    best = h.copy()
    best_c = compute_c5(h)
    block_sizes = [4, 8, 12, 16, 24]
    for it in range(n_iterations):
        if budget_check() < 10:
            break
        bs = block_sizes[rng.randint(len(block_sizes))]
        start = rng.randint(0, n - bs)
        for attempt in range(3):
            pert = best[start:start+bs].copy()
            noise = rng.randn(bs) * 0.08
            pert = np.clip(pert + noise, 0, 1)
            trial = best.copy()
            trial[start:start+bs] = pert
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best_c = c
                best = trial.copy()
                break
    return best, best_c

def binary_local_search(h, n, rng, n_flips, budget_check):
    best = h.copy()
    best_c = compute_c5(h)
    for _ in range(n_flips):
        if budget_check() < 5:
            break
        idx = rng.randint(n)
        trial = best.copy()
        trial[idx] = 1.0 - trial[idx]
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c - 1e-14:
            best_c = c
            best = trial.copy()
    return best, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_n = n_init

    # Phase 1: SLSQP cleanup at initial resolution
    if remaining() > 80:
        h1 = slsqp_optimize(best_h, best_n, 500.0, 200, 1e-14)
        c1 = compute_c5(h1)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1

    # Phase 2: Intensive coordinate descent
    if remaining() > 200:
        n = len(best_h)
        coarse_steps = [0.5, 0.25, 0.1, 0.05, 0.02, 0.01, 0.005, 0.002]
        best_h, best_c5 = coord_descent(best_h, n, 20, coarse_steps, remaining)

    # Phase 3: Block perturbations to escape local minima
    if remaining() > 150:
        n = len(best_h)
        best_h, best_c5 = block_perturb_search(best_h, n, rng, 500, remaining)
        if remaining() > 40:
            h3 = slsqp_optimize(best_h, n, 800.0, 150, 1e-14)
            c3 = compute_c5(h3)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3

    # Phase 4: Fine coordinate descent
    if remaining() > 150:
        n = len(best_h)
        fine_steps = [0.01, 0.005, 0.002, 0.001, 0.0005]
        best_h, best_c5 = coord_descent(best_h, n, 30, fine_steps, remaining)

    # Phase 5: Try lower resolution (480) - faster per iteration, might find different min
    if remaining() > 150:
        h480 = upsample_h(best_h, 480)
        h480 = slsqp_optimize(h480, 480, 400.0, 200, 1e-13)
        c480 = compute_c5(h480)
        if c480 < best_c5:
            best_c5 = c480
            best_h = h480
            best_n = 480
        if remaining() > 60:
            best_h, best_c5 = coord_descent(best_h, len(best_h), 30,
                                            [0.1, 0.05, 0.02, 0.01, 0.005], remaining)

    # Phase 6: Try higher resolution (1920)
    if remaining() > 120:
        h1920 = upsample_h(best_h, 1920)
        h1920 = slsqp_optimize(h1920, 1920, 300.0, 100, 1e-13)
        c1920 = compute_c5(h1920)
        if c1920 < best_c5:
            best_c5 = c1920
            best_h = h1920
            best_n = 1920

    # Phase 7: Return to best, do SLSQP with very high T
    n = len(best_h)
    if remaining() > 50:
        h7 = slsqp_optimize(best_h, n, 2000.0, 100, 1e-15)
        c7 = compute_c5(h7)
        if c7 < best_c5:
            best_c5 = c7
            best_h = h7

    # Phase 8: Binary local search (flip individual bits)
    if remaining() > 30:
        best_h, best_c5 = binary_local_search(best_h, n, rng, 5000, remaining)

    # Phase 9: Final ultra-fine coordinate descent
    if remaining() > 20:
        n = len(best_h)
        ultra_steps = [0.002, 0.001, 0.0005, 0.0002, 0.0001]
        best_h, best_c5 = coord_descent(best_h, n, 20, ultra_steps, remaining)

    # Final: ensure feasibility and compute
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
