def linearize_step(h, n):
    """One step of linearization: solve LP to get new h."""
    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full') * dx
    # Linearize each f_k
    # f_k(h) = Σ_i h[i](1-h[i+k]) * dx (for shift k = j - (n-1))
    # Linearization at h_0: f_k(h) ≈ f_k(h_0) + Σ_i [(1-h_0[i+k]) - h_0[i-k]] (h[i] - h_0[i]) * dx

    # Objective: [0, ..., 0, 1] (minimize t)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x ≤ b_ub
    # For each shift k: Σ_i g_i^{(k)} h[i] - t ≤ -c_k

        if time.time() - start_time > budget_s - 60:

        if step_count % 2000 == 0:
            remaining = budget_s - (time.time() - start_time)
            if remaining < 60:
                break
        step_count += 1
        bs = int(rng.choice(block_sizes))
        start_idx = int(rng.randint(0, n - bs + 1))
        h_trial = h_curr.copy()
        noise = rng.randn(bs) * 0.15
        h_trial[start_idx:start_idx + bs] += noise
        h_trial = make_feasible(h_trial, n)
        new_c5 = compute_c5(h_trial)
        delta = new_c5 - curr_c5
        if delta < 0 or rng.random() < np.exp(-delta / t):
            h_curr = h_trial
            curr_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_curr.copy()
        t *= 0.9997
    return best_h, best_c5

def binary_snap(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        for val in [0.0, 1.0]:
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def coord_descent(h, n, budget_s, start_time, max_rounds=50):
    best_h = h.copy()
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 5:
                break
            old = best_h[j]
            best_cand = old
            best_vc = best_c
            cands = [0.0, 1.0, 0.25, 0.75]
            for d in [-0.15, -0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1, 0.15]:
                cands.append(max(0.0, min(1.0, old + d)))
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_vc - 1e-14:
                    best_vc = c
                    best_cand = cand
            if best_cand != old:
                best_h[j] = best_cand
                best_h = make_feasible(best_h, n)
                best_c = best_vc
                improved = True
        if not improved:
            break
    return best_h, best_c

def lp_refine(h, n, budget_s, start_time, max_iters=40):
    dx = 2.0 / n
    h_curr = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h_curr)
    n_shifts = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 30:
            break
        corr_vals = np.correlate(h_curr, 1.0 - h_curr, mode='full') * dx
        c_obj = np.zeros(n + 1)
        c_obj[-1] = 1.0
        A_ub = np.zeros((n_shifts, n + 1))
        b_ub = np.zeros(n_shifts)
        
        for j in range(n_shifts):
            s = j - (n - 1)
            c_k = corr_vals[j]
            g = np.zeros(n)
            if s == 0:
                g[:] = 1.0 - 2.0 * h_curr
            elif s > 0:
                g[:s] = -h_curr[s:2*s] if 2*s <= n else -h_curr[s:]
                mid_start = s
                mid_end = n - s
                if mid_start < mid_end:
                    g[mid_start:mid_end] = 1.0 - h_curr[2*s:n] - h_curr[:n-2*s]
                g[n-s:] = 1.0 - h_curr[:s]
            else:
                s2 = -s
                g[:s2] = 1.0 - h_curr[s2:2*s2] if 2*s2 <= n else 1.0 - h_curr[s2:]
                mid_start = s2
                mid_end = n - s2
                if mid_start < mid_end:
                    g[mid_start:mid_end] = 1.0 - h_curr[2*s2:n] - h_curr[:n-2*s2]
                g[n-s2:] = -h_curr[:s2]
            
            A_ub[j, :n] = g * dx
            A_ub[j, n] = -1.0
            b_ub[j] = -c_k
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        bounds_lp = [(0.0, 1.0)] * n + [(0.0, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds_lp, method='highs')
        except Exception:
            break
        
        if not res.success:
            break
        
        h_new = make_feasible(res.x[:n], n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new
        else:
            blend = 0.3
            h_trial = (1.0 - blend) * h_curr + blend * h_new
            h_trial = make_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-12:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h_curr = h_trial
            else:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    total_budget = budget_s + start_time
    
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)
    
    n = len(h_init)
    h_init = make_feasible(h_init, n)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Phase 1: SLSQP with soft-max at current resolution
    T_list = [(150, 500, 1e-12), (300, 400, 1e-12), (600, 300, 1e-13)]
    for T, maxit, ftol in T_list:
        if time.time() > total_budget - 200:
            break
        h_res = slsqp_solve(best_h, n, T, maxit, ftol)
        h_res = make_feasible(h_res, n)
        c5 = compute_c5(h_res)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res
    
    # Phase 2: LP linearization steps
    if time.time() < total_budget - 400:
        h_lp, c5_lp = lp_refine(best_h, n, total_budget, start_time, max_iters=40)
        if c5_lp < best_c5:
            best_c5 = c5_lp
            best_h = h_lp
    
    # Phase 3: SLSQP after LP
    if time.time() < total_budget - 350:
        h_res = slsqp_solve(best_h, n, 500, 300, 1e-13)
        h_res = make_feasible(h_res, n)
        c5 = compute_c5(h_res)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_res
    
    # Phase 4: Block simulated annealing
    if time.time() < total_budget - 250:
        h_sa, c5_sa = block_sa(best_h, n, total_budget, start_time, seed + 1)
        if c5_sa < best_c5:
            best_c5 = c5_sa
            best_h = h_sa
        # Quick SLSQP polish after SA
        if time.time() < total_budget - 100:
            h_res = slsqp_solve(best_h, n, 400, 200, 1e-13)
            h_res = make_feasible(h_res, n)
            c5 = compute_c5(h_res)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_res
    
    # Phase 5: Binary snap
    if time.time() < total_budget - 80:
        best_h, best_c5 = binary_snap(best_h, n)
    
    # Phase 6: Coordinate descent
    if time.time() < total_budget - 30:
        best_h, best_c5 = coord_descent(best_h, n, total_budget, start_time, max_rounds=40)
    
    # Phase 7: Try higher resolution
    if time.time() < total_budget - 200:
        n_high = n * 2
        h_high = upsample(best_h, n_high)
        h_high = slsqp_solve(h_high, n_high, 200, 300, 1e-12)
        h_high = make_feasible(h_high, n_high)
        c5_high = compute_c5(h_high)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high
            n = n_high
        if time.time() < total_budget - 100:
            h_high2 = slsqp_solve(best_h, n, 500, 200, 1e-13)
            h_high2 = make_feasible(h_high2, n)
            c5_h2 = compute_c5(h_high2)
            if c5_h2 < best_c5:
                best_c5 = c5_h2
                best_h = h_high2
    
    # Phase 8: Final coordinate descent
    if time.time() < total_budget - 15:
        best_h, best_c5 = coord_descent(best_h, len(best_h), total_budget, start_time, max_rounds=30)
    
    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
