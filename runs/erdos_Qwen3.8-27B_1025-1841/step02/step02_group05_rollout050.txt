def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr.max() * dx


def optimize_smooth(h, n, tau, time_limit, lr=None, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.5 / tau
    v = np.zeros(n)
    beta = momentum
    stall = 0
    for it in range(200000):
        if time.time() - start > time_limit:
            break
        obj, grad, c5_now = smooth_obj_grad(h, tau)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 1500:
            lr *= 0.6
            v *= 0.2
            stall = 0
            if lr < 1e-13:
                break
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def coordinate_descent(h, n, time_limit, top_k=200):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    for pass_i in range(100000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        idx = np.argsort(corr)[::-1][:top_k]
        corr_at = corr[idx]
        offsets = idx - (n - 1)
        for i in range(n):
            pos_idx = i + offsets
            neg_idx = i - offsets
            valid_pos = (pos_idx >= 0) & (pos_idx < n)
            valid_neg = (neg_idx >= 0) & (neg_idx < n)
            coeffs = np.zeros(top_k)
            vp = valid_pos.nonzero()[0]
            if len(vp) > 0:
                coeffs[vp] += (1.0 - h[pos_idx[vp]])
            vn = valid_neg.nonzero()[0]
            if len(vn) > 0:
                coeffs[vn] += (-h[neg_idx[vn]])
            if (np.abs(coeffs) < 1e-12).all():
                continue
            lo_t = 0.0
            hi_t = 1.0
            cur = h[i]
            for _ in range(28):
                m1 = lo_t + (hi_t - lo_t) / 3.0
                m2 = hi_t - (hi_t - lo_t) / 3.0
                g1 = (corr_at + (m1 - cur) * coeffs).max()
                g2 = (corr_at + (m2 - cur) * coeffs).max()
                if g1 < g2:
                    hi_t = m2
                else:
                    lo_t = m1
            h[i] = lo_t + (hi_t - lo_t) * 0.5
        h = project_to_feasible(h, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
    return best_h, best_c5


def optimize_subgrad(h, n, time_limit, lr=None, top_k=8):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.003
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 6000:
            lr *= 0.65
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        idx_sorted = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[idx_sorted]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for ii in range(top_k):
            k = idx_sorted[ii]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[ii] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[ii] * h[j0b + sh:j1b + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def tleft():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Smooth optimization with decreasing tau
    for tau in [80.0, 30.0, 12.0, 6.0]:
        if tleft() < 180:
            break
        tl = min(130, tleft() * 0.22)
        best_h, best_c5 = optimize_smooth(best_h, n, tau=tau, time_limit=tl, lr=0.4 / tau)

    # Phase 2: Coordinate descent polish
    if tleft() > 150:
        tl = min(220, tleft() * 0.4)
        h_cd, c5_cd = coordinate_descent(best_h, n, time_limit=tl, top_k=200)
        if c5_cd < best_c5:
            best_c5, best_h = c5_cd, h_cd.copy()

    # Phase 3: Subgradient polish
    if tleft() > 80:
        tl = min(100, tleft() * 0.2)
        h_sg, c5_sg = optimize_subgrad(best_h, n, time_limit=tl, lr=0.002, top_k=10)
        if c5_sg < best_c5:
            best_c5, best_h = c5_sg, h_sg.copy()

    # Phase 4: Higher resolution n=4000
    n2 = 4000
    if tleft() > 350:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        if tleft() > 300:
            tl = min(160, tleft() * 0.28)
            h2, c5_2 = optimize_smooth(h2, n2, tau=50.0, time_limit=tl, lr=0.3)
        if tleft() > 200:
            tl = min(100, tleft() * 0.18)
            h2, c5_2 = optimize_smooth(h2, n2, tau=15.0, time_limit=tl, lr=0.15)
        if tleft() > 180:
            tl = min(160, tleft() * 0.3)
            h2, c5_2 = coordinate_descent(h2, n2, time_limit=tl, top_k=200)
        if tleft() > 60:
            tl = min(60, tleft() * 0.15)
            h2, c5_2 = optimize_subgrad(h2, n2, time_limit=tl, lr=0.001, top_k=10)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 5: Final coordinate descent at best resolution
    if tleft() > 80:
        n_b = len(best_h)
        tl = min(120, tleft() * 0.4)
        h_f, c5_f = coordinate_descent(best_h, n_b, time_limit=tl, top_k=300)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Phase 6: Final subgradient
    if tleft() > 40:
        n_b = len(best_h)
        tl = min(60, tleft() * 0.5)
        h_f, c5_f = optimize_subgrad(best_h, n_b, time_limit=tl, lr=0.0008, top_k=12)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Phase 7: One more coord descent if time
    if tleft() > 50:
        n_b = len(best_h)
        tl = min(80, tleft() * 0.6)
        h_f, c5_f = coordinate_descent(best_h, n_b, time_limit=tl, top_k=300)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
