import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binary_from_continuous(h, n, rng):
    """Convert continuous h to binary with n/2 ones, preferring positions where h was closer to 1."""
    h = np.clip(h, 0, 1)
    target = n // 2
    # Sort by h value descending - take top n/2 as ones
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def thue_morse_binary(n, rng):
    """Thue-Morse based binary sequence."""
    tm = np.array([bin(i & (n-1)).count('1') % 2 for i in range(n)], dtype=float)
    # This works nicely when n is a power of 2
    n_ones = int(np.sum(tm))
    target = n // 2
    if n_ones != target:
        # Adjust
        if n_ones > target:
            ones_idx = np.where(tm == 1)[0]
            to_flip = rng.choice(ones_idx, size=n_ones - target, replace=False)
            tm[to_flip] = 0
        else:
            zeros_idx = np.where(tm == 0)[0]
            to_flip = rng.choice(zeros_idx, size=target - n_ones, replace=False)
            tm[to_flip] = 1
    return tm

def stochastic_ls(h, n, rng, n_iters, time_limit=None, start_time=None):
    """Stochastic local search with 2-bit flips."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    zero_list = list(np.where(cur_h == 0)[0])
    one_list = list(np.where(cur_h == 1)[0])
    
    for it in range(n_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        ia = rng.randint(len(zero_list))
        ib = rng.randint(len(one_list))
        a = zero_list[ia]
        b = one_list[ib]
        
        cur_h[a] = 1.0
        cur_h[b] = 0.0
        new_c5 = compute_c5(cur_h)
        
        if new_c5 <= cur_c5:
            cur_c5 = new_c5
            zero_list[ia] = b
            one_list[ib] = a
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[a] = 0.0
            cur_h[b] = 1.0
    
    return best_h, best_c5

def sa_search(h, n, rng, n_iters, T_init=0.01, T_final=0.0001, time_limit=None, start_time=None):
    """Simulated annealing with 2-bit flips."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_h = h.copy()
    cur_c5 = best_c5
    zero_list = list(np.where(cur_h == 0)[0])
    one_list = list(np.where(cur_h == 1)[0])
    
    for it in range(n_iters):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        progress = it / max(n_iters, 1)
        T = T_init * (T_final / T_init) ** progress
        
        ia = rng.randint(len(zero_list))
        ib = rng.randint(len(one_list))
        a = zero_list[ia]
        b = one_list[ib]
        
        cur_h[a] = 1.0
        cur_h[b] = 0.0
        new_c5 = compute_c5(cur_h)
        
        delta = new_c5 - cur_c5
        if delta <= 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            zero_list[ia] = b
            one_list[ib] = a
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = cur_h.copy()
        else:
            cur_h[a] = 0.0
            cur_h[b] = 1.0
    
    return best_h, best_c5

def deterministic_sweep(h, n, time_limit=None, start_time=None):
    """Deterministic coordinate descent: for each position, try all possible partners."""
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for i in range(n):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        if best_h[i] == 0:
            # Try flipping h[i] to 1, and h[j] to 0 for each j where h[j]=1
            ones = np.where(best_h == 1)[0]
            for j in ones:
                best_h[i] = 1.0
                best_h[j] = 0.0
                c5 = compute_c5(best_h)
                if c5 < best_c5:
                    best_c5 = c5
                else:
                    best_h[i] = 0.0
                    best_h[j] = 1.0
            # After trying all, if none improved, h[i] is already 0
            # If one improved, h[i] is 1 and some j is 0
        # If best_h[i] == 1, we already handled it when i was the "j" in a previous iteration
    
    return best_h, best_c5
