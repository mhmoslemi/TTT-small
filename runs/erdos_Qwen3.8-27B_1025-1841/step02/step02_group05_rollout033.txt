def project_to_feasible(g, n):
    target = n / 2.0

        else:

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    gmin = g.min()
    gmax = g.max()
    if gmin >= 1.0 - 1e-12:
        return g.copy()
    if gmax <= 1e-12:
        return g.copy()
    lo = gmin - 1.0
    hi = gmax + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    nm1 = n - 1
    for k_idx in range(2 * n - 1):
        pk = p[k_idx]
        if pk < 1e-13:
            continue
        sh = k_idx - nm1
        r1s = max(0, sh)
        r1e = min(n, n + sh)
        if r1s < r1e:
            idx = np.arange(r1s, r1e)
            grad[r1s:r1e] += pk * (1.0 - h[idx - sh])
        r2s = max(0, -sh)
        r2e = min(n, n - sh)
        if r2s < r2e:
            idx = np.arange(r2s, r2e)
            grad[r2s:r2e] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad
