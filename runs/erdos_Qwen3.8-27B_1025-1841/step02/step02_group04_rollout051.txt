                grad += w * subgrad_single(hx, shift, n)
            grad *= dx

        t_a += 1
        m_adam = b1 * m_adam + (1 - b1) * grad
        v_adam = b2 * v_adam + (1 - b2) * grad * grad
        m_hat = m_adam / (1 - b1 ** t_a)
        v_hat = v_adam / (1 - b2 ** t_a)
        step = lr * m_hat / (np.sqrt(v_hat) + eps_a)
        hx = hx - step
        hx = make_feasible(hx, n)

        if (it + 1) % 100 == 0:
            c5 = compute_c5(hx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hx.copy()

    c5 = compute_c5(hx)
    if c5 < best_c5:
        best_c5, best_h = c5, hx.copy()
    print(f"Adam: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: Nesterov subgradient on worst shift
    n = len(best_h)
    dx = 2.0 / n
    hx = best_h.copy()
    vel = np.zeros(n)
    beta = 0.85
    n_it2 = 3000
    lr_s2, lr_e2 = 0.05, 0.0003

    for it in range(n_it2):
        if time.time() - start_time > budget_s - 80:
            break
        frac = it / max(n_it2 - 1, 1)
        lr = lr_s2 * (lr_e2 / lr_s2) ** frac
        corr = np.correlate(hx, 1.0 - hx, mode='full')
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        grad = subgrad_single(hx, shift, n) * dx
        vel = beta * vel - lr * grad
        hx = make_feasible(hx + vel, n)
        if (it + 1) % 60 == 0:
            c5 = compute_c5(hx)
            if c5 < best_c5:
                best_c5, best_h = c5, hx.copy()

    c5 = compute_c5(hx)
    if c5 < best_c5:
        best_c5, best_h = c5, hx.copy()
    print(f"Nesterov: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: Alternating optimization on top shifts
    corr = np.correlate(best_h, 1.0 - best_h, mode='full')
    top_shifts = np.argsort(corr)[-6:]
    for si in top_shifts:
        if time.time() - start_time > budget_s - 60:
            break
        shift = si - (n - 1)
        hx = best_h.copy()
        vel = np.zeros(n)
        for it in range(1500):
            if time.time() - start_time > budget_s - 30:
                break
            lr = 0.012 * (0.4 ** (it / 300.0))
            grad = subgrad_single(hx, shift, n) * dx
            vel = 0.8 * vel - lr * grad
            hx = make_feasible(hx + vel, n)
        c5 = compute_c5(hx)
        if c5 < best_c5:
            best_c5, best_h = c5, hx.copy()
    print(f"AltShifts: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 4: SLSQP polish
    n = len(best_h)
    dx = 2.0 / n
    if n <= 512 and time.time() - start_time < budget_s - 50:
        for T_val in [12.0, 5.0, 2.0]:
            if time.time() - start_time > budget_s - 20:
                break
            _SLSQP_PARAMS['n'] = n
            _SLSQP_PARAMS['dx'] = dx
            _SLSQP_PARAMS['T'] = T_val
            bounds = [(0.0, 1.0)] * n
            constraints = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac}]
            try:
                res = minimize(_slsqp_obj, best_h, method='SLSQP', bounds=bounds,
                               constraints=constraints,
                               options={'maxiter': 1200, 'ftol': 1e-15, 'maxls': 60})
                if res.x is not None:
                    hx = make_feasible(res.x, n)
                    cx = compute_c5(hx)
                    if cx < best_c5:
                        best_c5, best_h = cx, hx
                        print(f"  SLSQP T={T_val}: C5={best_c5:.6f}", flush=True)
            except Exception:
                pass
    print(f"SLSQP: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 5: Local single-coordinate search
    n = len(best_h)
    hx = best_h.copy()
    for p in range(3):
        if time.time() - start_time > budget_s - 20:
            break
        current = compute_c5(hx)
        for i in range(n):
            if time.time() - start_time > budget_s - 12:
                break
            old = hx[i]
            hx[i] = 1.0 - old
            hx = make_feasible(hx, n)
            c5_new = compute_c5(hx)
            if c5_new < current - 1e-14:
                current = c5_new
            else:
                hx[i] = old
                hx = make_feasible(hx, n)
        if current < best_c5:
            best_c5, best_h = current, hx.copy()
    print(f"LocalSearch: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 6: Multi-start perturbation + Adam
    if time.time() - start_time < budget_s - 80:
        for ps in [0.04, 0.10]:
            if time.time() - start_time > budget_s - 40:
                break
            h_pert = best_h + rng.randn(len(best_h)) * ps
            h_pert = make_feasible(h_pert, len(best_h))
            # Short Adam
            n_p = len(h_pert)
            dx_p = 2.0 / n_p
            hx = h_pert.copy()
            m_a = np.zeros(n_p)
            v_a = np.zeros(n_p)
            t_ap = 0
            for it in range(1500):
                if time.time() - start_time > budget_s - 25:
                    break
                lr = 0.05 * (0.01 ** (it / 1500.0))
                corr = np.correlate(hx, 1.0 - hx, mode='full')
                mc = np.max(corr)
                shifted = np.clip(corr - mc, -50.0, 0.0)
                exp_s = np.exp(10.0 * shifted)
                total = np.sum(exp_s)
                if total < 1e-300:
                    k_max = int(np.argmax(corr))
                    shift = k_max - (n_p - 1)
                    grad = subgrad_single(hx, shift, n_p) * dx_p
                else:
                    weights = exp_s / total
                    top_idx = np.argsort(weights)[-15:]
                    grad = np.zeros(n_p)
                    for ki in top_idx:
                        w = weights[ki]
                        if w < 1e-13:
                            break
                        shift = ki - (n_p - 1)
                        grad += w * subgrad_single(hx, shift, n_p)
                    grad *= dx_p
                t_ap += 1
                m_a = 0.9 * m_a + 0.1 * grad
                v_a = 0.999 * v_a + 0.001 * grad * grad
                m_h = m_a / (1 - 0.9 ** t_ap)
                v_h = v_a / (1 - 0.999 ** t_ap)
                step = lr * m_h / (np.sqrt(v_h) + 1e-8)
                hx = make_feasible(hx - step, n_p)
            c5 = compute_c5(hx)
            if c5 < best_c5:
                best_c5, best_h = c5, hx.copy()
                print(f"  Perturb {ps}: C5={best_c5:.6f}", flush=True)
    print(f"MultiStart: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 7: Final fine Nesterov
    if time.time() - start_time < budget_s - 10:
        n = len(best_h)
        dx = 2.0 / n
        hx = best_h.copy()
        vel = np.zeros(n)
        for it in range(2000):
            if time.time() - start_time > budget_s - 5:
                break
            lr = 0.004 * (0.9995 ** it)
            corr = np.correlate(hx, 1.0 - hx, mode='full')
            k_max = int(np.argmax(corr))
            shift = k_max - (n - 1)
            grad = subgrad_single(hx, shift, n) * dx
            vel = 0.75 * vel - lr * grad
            hx = make_feasible(hx + vel, n)
            c5 = compute_c5(hx)
            if c5 < best_c5:
                best_c5, best_h = c5, hx.copy()
    print(f"Final: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
