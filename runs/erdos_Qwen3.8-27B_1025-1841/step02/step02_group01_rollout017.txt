import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:n//2]] = 1.0
    return h

def _lbfgsb_obj(h, *args):
    t, penalty_w = args
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0/t, 0.0)
    logits = np.clip(t * shifted, -50, 0)
    exp_l = np.exp(logits)
    obj = (mc + np.log(np.sum(exp_l)) / t) * dx
    s = np.sum(h)
    obj += penalty_w * (s - n / 2.0) ** 2
    return obj

def _lbfgsb_grad(h, *args):
    t, penalty_w = args
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0/t, 0.0)
    logits = np.clip(t * shifted, -50, 0)
    exp_l = np.exp(logits)
    exp_sum = np.sum(exp_l)
    if exp_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = exp_l / exp_sum
    j_idx = np.arange(n)[:, None]
    m_vals = np.arange(2 * n - 1) - (n - 1)
    jm = j_idx + m_vals[None, :]
    jm1 = j_idx - m_vals[None, :]
    valid_m = (jm >= 0) & (jm < n)
    valid_m1 = (jm1 >= 0) & (jm1 < n)
    h_safe = np.clip(jm, 0, n - 1)
    h_safe1 = np.clip(jm1, 0, n - 1)
    term_pos = np.where(valid_m, h[h_safe], 0.0)
    term_neg = np.where(valid_m1, h[h_safe1], 0.0)
    contrib = valid_m.astype(float) - term_pos - term_neg
    s = np.sum(h)
    grad = dx * (contrib @ w) + 2.0 * penalty_w * (s - n / 2.0)
    return grad

def _slsqp_obj(h, *args):
    t = args[0]
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0/t, 0.0)
    logits = np.clip(t * shifted, -50, 0)
    return (mc + np.log(np.sum(np.exp(logits))) / t) * dx

def _cons_fun(h, *args):
    n = args[0]
    return np.sum(h) - n / 2.0

def _cons_jac(h, *args):
    n = args[0]
    return np.ones(n)

def _run_lbfgsb(h0, n, t, penalty_w, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_lbfgsb_obj, h0, jac=_lbfgsb_grad,
                   method='L-BFGS-B', bounds=bounds,
                   args=(t, penalty_w),
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-7,
                            'iprint': -1})
    return res.x

def _run_slsqp(h0, n, t, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac, 'args': (n,)}]
    res = minimize(_slsqp_obj, h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   args=(t,),
                   options={'maxiter': maxiter, 'ftol': 1e-10, 'iprint': -1})
    return res.x

def _coord_descent(h, n, max_rounds, time_limit):
    start = time.time()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[j]
            for cand in [0.0, 1.0, 0.5, old_val+0.05, old_val-0.05,
                         old_val+0.1, old_val-0.1, old_val+0.2, old_val-0.2]:
                if cand < 0 or cand > 1:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial
                    improved = True
                    break
        if not improved:
            break
    return h, best_c5

    rng = np.random.RandomState(seed)
    
    best_c5 = float('inf')
    best_h = None
    
    # Get parent

    except Exception:

import numpy as np
from scipy.optimize import minimize
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)


def random_binary_h(n, rng):
    h = np.zeros(n)
    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h


def _lbfgsb_obj(h, t, penalty_w):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0 / t, 0.0)
    logits = np.clip(t * shifted, -50.0, 0.0)
    exp_l = np.exp(logits)
    obj = (mc + np.log(np.sum(exp_l)) / t) * dx
    s = np.sum(h)
    obj += penalty_w * (s - n / 2.0) ** 2
    return obj


def _lbfgsb_grad(h, t, penalty_w):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0 / t, 0.0)
    logits = np.clip(t * shifted, -50.0, 0.0)
    exp_l = np.exp(logits)
    exp_sum = np.sum(exp_l)
    if exp_sum < 1e-300:
        w = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        w = exp_l / exp_sum
    j_idx = np.arange(n)[:, None]
    m_vals = np.arange(2 * n - 1) - (n - 1)
    jm = j_idx + m_vals[None, :]
    jm1 = j_idx - m_vals[None, :]
    valid_m = (jm >= 0) & (jm < n)
    valid_m1 = (jm1 >= 0) & (jm1 < n)
    h_safe = np.clip(jm, 0, n - 1)
    h_safe1 = np.clip(jm1, 0, n - 1)
    term_pos = np.where(valid_m, h[h_safe], 0.0)
    term_neg = np.where(valid_m1, h[h_safe1], 0.0)
    contrib = valid_m.astype(float) - term_pos - term_neg
    s = np.sum(h)
    grad = dx * (contrib @ w) + 2.0 * penalty_w * (s - n / 2.0)
    return grad


def _lbfgsb_obj_wrap(h, *args):
    return _lbfgsb_obj(h, *args)


def _lbfgsb_grad_wrap(h, *args):
    return _lbfgsb_grad(h, *args)


def _slsqp_obj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -50.0 / t, 0.0)
    logits = np.clip(t * shifted, -50.0, 0.0)
    return (mc + np.log(np.sum(np.exp(logits))) / t) * dx


def _slsqp_obj_wrap(h, *args):
    return _slsqp_obj(h, *args)


def _cons_fun(h, n):
    return np.sum(h) - n / 2.0


def _cons_jac(h, n):
    return np.ones(n)


def _cons_fun_wrap(h, *args):
    return _cons_fun(h, args[0])


def _cons_jac_wrap(h, *args):
    return _cons_jac(h, args[0])


def _run_lbfgsb(h0, n, t, penalty_w, maxiter):
    bounds = [(0.0, 1.0)] * n
    res = minimize(_lbfgsb_obj_wrap, h0, jac=_lbfgsb_grad_wrap,
                   method='L-BFGS-B', bounds=bounds,
                   args=(t, penalty_w),
                   options={'maxiter': maxiter, 'ftol': 1e-14,
                            'gtol': 1e-7, 'iprint': -1, 'maxls': 50})
    return res.x


def _run_slsqp(h0, n, t, maxiter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun_wrap, 'jac': _cons_jac_wrap, 'args': (n,)}]
    res = minimize(_slsqp_obj_wrap, h0, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   args=(t,),
                   options={'maxiter': maxiter, 'ftol': 1e-10, 'iprint': -1})
    return res.x


def _coord_descent(h, n, time_limit):
    start = time.time()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    no_improve = 0
    while time.time() - start < time_limit:
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[j]
            cands = [0.0, 1.0, 0.5]
            for delta in [0.03, 0.07, 0.15, 0.25]:
                cands.append(old_val + delta)
                cands.append(old_val - delta)
            for cand in cands:
                if cand < -0.01 or cand > 1.01:
                    continue
                cand = max(0.0, min(1.0, cand))
                if abs(cand - old_val) < 1e-6:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial.copy()
                    old_val = cand
                    improved = True
                    break
        if not improved:
            no_improve += 1
            if no_improve >= 3:
                break
            # Try larger perturbations
            for j in range(n):
                if time.time() - start > time_limit:
                    break
                for _ in range(2):
                    cand = rng_cd_uniform(j, no_improve)
                    trial = h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-12:
                        best_c5 = c
                        h = trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                no_improve = 0
    return h, best_c5


def rng_cd_uniform(j, rnd):
    return (j * 7919 + rnd * 104729 + 12345) % 1000 / 1000.0
