def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0

    n = len(h)

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    # Softmax weights
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        # d(corr_k)/d(h[i]) = (1-h)[i+sh] - h[i-sh] (with valid indices)
        # (1-h)[i+sh] is valid when 0 <= i+sh < n, i.e., max(0,-sh) <= i < n-sh
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        # h[i-sh] is valid when 0 <= i-sh < n, i.e., sh <= i < n+sh
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    return obj, grad, corr, max_c, p

        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            lr *= 0.65
            v *= 0.25
            stall = 0
            if lr < 1e-16:
                break

        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        top_vals = corr[top_idx]
        max_tv = top_vals[0]
        w = np.exp(0.5 * (top_vals - max_tv))
        w /= w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            if w[i] < 1e-10:
                continue
            sh = k - n + 1
            j0 = max(0, -sh)
            j1 = min(n, n - sh)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 + sh:j1 + sh])
            j0b = max(0, sh)
            j1b = min(n, n + sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b - sh:j1b - sh]
        grad *= dx

        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def rem():
        return max(0.0, budget_s - (time.time() - start_time))

    # Phase 1: Annealed optimization (45% budget)
    if rem() > 350:
        tl = min(320, rem() * 0.42)
        h1, c5_1 = anneal_optimize(h_init, n, tl, tau_max=120, tau_min=2, lr_base=0.14, momentum=0.91)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Schedule refinement (20% budget)
    if rem() > 200:
        tl = min(170, rem() * 0.25)
        sched = [(40.0, 0.25), (12.0, 0.25), (5.0, 0.25), (2.5, 0.25)]
        h2, c5_2 = schedule_optimize(best_h, n, tl, sched, lr_base=0.06, momentum=0.93)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Multi-shift subgradient polish (15% budget)
    if rem() > 120:
        tl = min(110, rem() * 0.28)
        h3, c5_3 = subgrad_polish(best_h, n, tl, top_k=3, lr=0.0015, momentum=0.88)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase 4: Perturbation + re-optimize (12% budget)
    for trial in range(4):
        if rem() < 60:
            break
        sigma = 0.015 + 0.005 * trial
        pert = sigma * np.random.randn(n)
        h_pert = project_to_feasible(best_h + pert, n)
        tl = min(50, rem() * 0.35)
        h_p, c5_p = anneal_optimize(h_pert, n, tl, tau_max=50, tau_min=3, lr_base=0.1, momentum=0.9)
        if c5_p < best_c5:
            best_c5, best_h = c5_p, h_p.copy()

    # Phase 5: Higher resolution (try n=3000)
    if rem() > 200:
        n2 = 3000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(130, rem() * 0.4)
        h2, c5_2 = anneal_optimize(h2, n2, tl, tau_max=60, tau_min=3, lr_base=0.06, momentum=0.92)
        if rem() > 60:
            tl2 = min(80, rem() * 0.4)
            h2, c5_2 = subgrad_polish(h2, n2, tl2, top_k=3, lr=0.0008, momentum=0.87)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 6: Focus tightly on worst shifts at fine scale
    if rem() > 40:
        n_f = len(best_h)
        tl = min(50, rem() * 0.5)
        h_f, c5_f = subgrad_polish(best_h, n_f, tl, top_k=2, lr=0.0004, momentum=0.9)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Phase 7: Very tight final polish
    if rem() > 20:
        n_f = len(best_h)
        tl = min(25, rem() * 0.7)
        h_f, c5_f = subgrad_polish(best_h, n_f, tl, top_k=1, lr=0.0001, momentum=0.85)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
