def adam_optimize(h, n, tau, time_limit, lr0=0.1, max_iters=50000):
    h = project_to_feasible(h, n).copy()
    m = np.zeros(n)  # first moment
    v = np.zeros(n)  # second moment
    t = 0
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    start = time.time()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if time.time() - start > time_limit:
            break
        
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
        
        lr = lr0 / (1.0 + 0.01 * t)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)
    
    return best_h, best_obj
