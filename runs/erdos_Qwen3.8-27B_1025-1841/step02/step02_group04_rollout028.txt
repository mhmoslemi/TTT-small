                for m in range(k2, n):
                    if h[m] == 1 and h[m-k2] == 0:
                        candidates.add((m-k2, m))
        
        cand_list = list(candidates)
        if not cand_list:
            iters_since_improve += 1
            if iters_since_improve > stagnation_limit:
                break
            continue
        
        if len(cand_list) > n_candidates:
            rng_sel = np.random.choice(len(cand_list), n_candidates, replace=False)
            cand_list = [cand_list[i] for i in rng_sel]
        
        best_delta = 0
        best_pair = None
        for (a, b) in cand_list:
            if h[a] == h[b]:
                continue
            # Flip: h[a] and h[b] swap values
            h2 = h.copy()
            h2[a], h2[b] = h2[b], h2[a]
            c2 = np.correlate(h2, 1.0 - h2, mode='full')
            mx2 = np.max(c2)
            if mx2 < best_max - 1e-12:
                delta = mx2 - best_max
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (a, b)
                    best_corr = c2
        
        if best_pair is not None:
            h[best_pair[0]], h[best_pair[1]] = h[best_pair[1]], h[best_pair[0]]
            best_max = best_max + best_delta
            corr = best_corr if best_pair else corr
            iters_since_improve = 0
        else:
            iters_since_improve += 1
            if iters_since_improve > stagnation_limit:
                break
    
    return h, best_max * dx

def softgrad(h, T, top_k=60):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        grad = np.zeros(n)
        s = abs(shift)
        if s < n:
            if shift >= 0:
                grad[s:] += (1.0 - h[:n-s])
                grad[:n-s] += (-h[s:])
            else:
                grad[:n-s] += (1.0 - h[s:])
                grad[s:] += (-h[:n-s])
        return grad * dx
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-12:
            break
        shift = k_idx - (n - 1)
        s = abs(shift)
        if s < n:
            if shift >= 0:
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] += w * (-h[s:])
            else:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] += w * (-h[:n-s])
    return grad * dx

def gd_refine(h, n_iter=3000, T_start=50.0, T_end=1.0, lr=0.1, check_every=40):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        grad = softgrad(h, T, top_k=60)
        h = h - lr_t * grad
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _obj_smooth(h):
    T = _OPT_PARAMS['T']
    n = _OPT_PARAMS['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_PARAMS['n'] / 2.0

def _cons_jac(h):
    return np.ones(_OPT_PARAMS['n'])

def slsqp_refine(h, T_val, maxiter=500):
    n = len(h)
    _OPT_PARAMS['T'] = T_val
    _OPT_PARAMS['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj_smooth, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
    if res.x is not None:
        hx = make_feasible(res.x, n)
        return hx
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def local_search_fractions(h, n_max_iter=5000, top_lags=4, n_candidates=50, stagnation_limit=500):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_max = np.max(corr)
    iters_since_improve = 0
    
    for it in range(n_max_iter):
        top_idx = np.argsort(corr)[-top_lags:]
        
        ones_idx = np.where(h > 0.3)[0]
        zeros_idx = np.where(h < 0.7)[0]
        
        candidates = set()
        for ti in top_idx:
            k = ti - (n - 1)
            if k >= 0:
                for m in range(n - k):
                    if h[m+k] > 0.5 and h[m] < 0.5:
                        candidates.add((m, m+k))
            else:
                k2 = -k
                for m in range(k2, n):
                    if h[m] > 0.5 and h[m-k2] < 0.5:
                        candidates.add((m-k2, m))
        
        cand_list = list(candidates)
        if not cand_list:
            iters_since_improve += 1
            if iters_since_improve > stagnation_limit:
                break
            continue
        
        if len(cand_list) > n_candidates:
            sel = np.random.choice(len(cand_list), n_candidates, replace=False)
            cand_list = [cand_list[i] for i in sel]
        
        best_delta = 0
        best_pair = None
        best_corr = None
        for (a, b) in cand_list:
            h2 = h.copy()
            h2[a], h2[b] = h2[b], h2[a]
            h2 = np.clip(h2, 0, 1)
            c2 = np.correlate(h2, 1.0 - h2, mode='full')
            mx2 = np.max(c2)
            if mx2 < best_max - 1e-10:
                delta = mx2 - best_max
                if delta < best_delta:
                    best_delta = delta
                    best_pair = (a, b)
                    best_corr = c2
        
        if best_pair is not None:
            h[best_pair[0]], h[best_pair[1]] = h[best_pair[1]], h[best_pair[0]]
            best_max += best_delta
            corr = best_corr
            iters_since_improve = 0
        else:
            iters_since_improve += 1
            if iters_since_improve > stagnation_limit:
                break
    
    return h, best_max * dx

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    best_c5 = float('inf')
    best_h = None
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n_init = 320
        h0 = (np.random.rand(n_init) < 0.5).astype(float)
        n_ones = n_init // 2
        ones = np.where(h0 == 1)[0]
        zeros = np.where(h0 == 0)[0]
        if len(ones) > n_ones:
            h0[np.random.choice(ones, len(ones)-n_ones, replace=False)] = 0
        elif len(ones) < n_ones:
            h0[np.random.choice(ones if len(ones) < n_ones else zeros, n_ones-len(ones), replace=False)] = 1
        if len(ones) < n_ones:
            h0[np.random.choice(zeros, n_ones-len(ones), replace=False)] = 1
    
    n0 = len(h0)
    
    # Phase 1: Binary local search from parent
    h_bin = to_binary(h0, n0)
    c5_init = compute_c5(h_bin)
    best_c5 = min(best_c5, c5_init)
    best_h = h_bin.copy()
    print(f"Binary init: C5={c5_init:.6f}", flush=True)
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Multiple rounds of local search with different parameters
    for round_num in range(3):
        if time.time() - start_time > budget_s - 100:
            break
        n_iter = min(25000, int(remaining * 0.25 / max(1, 3-round_num) * 2))
        n_iter = max(n_iter, 5000)
        
        top_l = 3 + round_num
        n_cand = 60 + round_num * 30
        
        h_ls, c5_ls = local_search_binary(best_h, n_max_iter=n_iter, top_lags=top_l, 
                                           n_candidates=n_cand, stagnation_limit=3000)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            print(f" LS round {round_num}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
        else:
            print(f" LS round {round_num}: no improvement (best={best_c5:.6f}), t={time.time()-start_time:.0f}s", flush=True)
        
        # Random restart from current best
        if round_num > 0 and time.time() - start_time < budget_s - 120:
            h_pert = best_h.copy()
            n_flips = 5 + round_num * 3
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            flip_ones = np.random.choice(ones, min(n_flips, len(ones)), replace=False)
            flip_zeros = np.random.choice(zeros, min(n_flips, len(zeros)), replace=False)
            for i in flip_ones:
                h_pert[i] = 0
            for i in flip_zeros:
                h_pert[i] = 1
            h_ls2, c5_ls2 = local_search_binary(h_pert, n_max_iter=n_iter//2, top_lags=top_l,
                                                 n_candidates=n_cand, stagnation_limit=2000)
            if c5_ls2 < best_c5:
                best_c5 = c5_ls2
                best_h = h_ls2.copy()
                print(f"  Restart round {round_num}: C5={best_c5:.6f}", flush=True)
    
    # Phase 2: Continuous GD refinement on binary solution
    if time.time() - start_time < budget_s - 60:
        h_g, c5_g = gd_refine(best_h, n_iter=3000, T_start=60.0, T_end=0.5, lr=0.12, check_every=40)
        if c5_g < best_c5:
            best_c5 = c5_g
            best_h = h_g.copy()
        print(f"GD refine: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
        
        # Fractional local search
        if time.time() - start_time < budget_s - 50:
            h_fs, c5_fs = local_search_fractions(best_h, n_max_iter=3000, top_lags=4, 
                                                   n_candidates=50, stagnation_limit=400)
            if c5_fs < best_c5:
                best_c5 = c5_fs
                best_h = h_fs.copy()
                print(f"  Fractional LS: C5={best_c5:.6f}", flush=True)
    
    # Phase 3: SLSQP polish
    n_cur = len(best_h)
    if n_cur <= 480 and time.time() - start_time < budget_s - 40:
        for T_val in [10.0, 4.0, 2.0]:
            if time.time() - start_time > budget_s - 25:
                break
            try:
                hx = slsqp_refine(best_h, T_val, maxiter=400)
                cx = compute_c5(hx)
                if cx < best_c5:
                    best_c5 = cx
                    best_h = hx
            except Exception:
                pass
        print(f"SLSQP: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Phase 4: Try higher resolution
    for target_n in [480, 640]:
        if time.time() - start_time > budget_s - 120:
            break
        if target_n <= n_cur:
            continue
        h_up = upsample_h(best_h, target_n)
        c5_up = compute_c5(h_up)
        
        # Binary local search at higher res
        h_up_bin = to_binary(h_up, target_n)
        c5_upb = compute_c5(h_up_bin)
        if c5_upb < c5_up:
            h_up = h_up_bin
        
        if time.time() - start_time < budget_s - 80:
            h_ls3, c5_ls3 = local_search_binary(h_up, n_max_iter=15000, top_lags=4,
                                                 n_candidates=80, stagnation_limit=2500)
            if c5_ls3 < c5_up:
                h_up = h_ls3
                c5_up = c5_ls3
        
        if time.time() - start_time < budget_s - 50:
            h_g2, c5_g2 = gd_refine(h_up, n_iter=2000, T_start=40.0, T_end=0.5, lr=0.1, check_every=50)
            if c5_g2 < c5_up:
                h_up = h_g2
                c5_up = c5_g2
        
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            print(f"Res {target_n}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Phase 5: Final fine-tuning GD
    if time.time() - start_time < budget_s - 20:
        h_f, c5_f = gd_refine(best_h, n_iter=3000, T_start=5.0, T_end=0.1, lr=0.03, check_every=30)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()
        print(f"FinalGD: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Final: SLSQP if enough time
    n_final = len(best_h)
    if n_final <= 480 and time.time() - start_time < budget_s - 15:
        try:
            hx = slsqp_refine(best_h, 3.0, maxiter=300)
            cx = compute_c5(hx)
            if cx < best_c5:
                best_c5 = cx
                best_h = hx
        except Exception:
            pass
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
