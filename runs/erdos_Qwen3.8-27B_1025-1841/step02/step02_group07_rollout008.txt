            h_cd = coord_descent(h_s, n, n_sweeps=80, proj_every=8, deadline=deadline)
            c5_cd = evaluate_c5(h_cd)
            if c5_cd < best_c5:
                best_c5 = c5_cd
                best_h = h_cd

    # Phase 3: Simulated annealing restarts
    n_tries = 40
    for attempt in range(n_tries):
        if time.time() > deadline - 25:
            break
        temp = 0.12 * (1.0 - attempt / n_tries) + 0.02
        h_pert = best_h + np.random.randn(n) * temp
        h_pert = project(h_pert, n)
        h_new = coord_descent(h_pert, n, n_sweeps=80, proj_every=8, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

    # Phase 4: Binary snap + re-optimize
    if time.time() < deadline - 30:
        h_bin = snap_binary(best_h, n, thresh=0.25)
        h_reopt = coord_descent(h_bin, n, n_sweeps=100, proj_every=8, deadline=deadline)
        c5_re = evaluate_c5(h_reopt)
        if c5_re < best_c5:
            best_c5 = c5_re
            best_h = h_reopt

    # Phase 5: Double resolution
    if time.time() < deadline - 80:
        n2 = n * 2
        xf = np.linspace(0, 2, n, endpoint=False)
        xt = np.linspace(0, 2, n2, endpoint=False)
        h2 = np.interp(xt, xf, best_h)
        h2 = project(h2, n2)
        h2_cd = coord_descent(h2, n2, n_sweeps=60, proj_every=10, deadline=deadline)
        c5_2 = evaluate_c5(h2_cd)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2_cd
            n = n2

    # Phase 6: Final polish with long coordinate descent
    if time.time() < deadline - 15:
        n_best = len(best_h)
        h_final = coord_descent(best_h, n_best, n_sweeps=300, proj_every=5, deadline=deadline)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
