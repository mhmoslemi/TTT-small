    _gT = T
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons, 'jac': _cons_j}]
    r = minimize(_obj, h0, method='SLSQP', bounds=bnds,
                 constraints=cons, options={'maxiter': max_it, 'ftol': 1e-15})
    return make_feasible(r.x, n)

def cd_polish(h, n, t0, budget, rounds=4):
    best_c5 = compute_c5(h)
    deltas = [0.003, -0.003, 0.007, -0.007, 0.015, -0.015, 0.03, -0.03]
    for _ in range(rounds):
        if time.time() - t0 > budget - 20:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget - 20:
                break
            ov = h[j]
            for d in deltas:
                c = ov + d
                if c < 0 or c > 1:
                    continue
                tr = h.copy()
                tr[j] = c
                tr = make_feasible(tr, n)
                c5t = compute_c5(tr)
                if c5t < best_c5 - 1e-13:
                    best_c5 = c5t
                    h = tr
                    ov = c
                    improved = True
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                tr = h.copy()
                tr[j] = bv
                tr = make_feasible(tr, n)
                c5t = compute_c5(tr)
                if c5t < best_c5 - 1e-13:
                    best_c5 = c5t
                    h = tr
                    improved = True
        if not improved:
            break
    return h, best_c5

def targeted_lag_fix(h, n, t0, budget, n_rounds=5):
    """Identify problematic lags and try to reduce them."""
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for rnd in range(n_rounds):
        if time.time() - t0 > budget - 20:
            break
        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        dx = 2.0 / n
        top_lags = np.argsort(corr)[-5:]
        for li in top_lags:
            k = li - (n - 1)
            h_fwd = np.roll(best_h, -k)
            h_bwd = np.roll(best_h, k)
            grad = dx * ((1.0 - h_fwd) - h_bwd)
            for alpha in [0.005, 0.002, -0.005, -0.002]:
                h_try = best_h - alpha * grad
                h_try = make_feasible(h_try, n)
                c5t = compute_c5(h_try)
                if c5t < best_c5 - 1e-14:
                    best_c5 = c5t
                    best_h = h_try
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - t0)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    if tl() < 120:
        return best_h, best_c5, n0

    # Phase 1: Subgradient descent at original resolution
    h_sg, c5_sg = sg_descent(best_h, n0, t0, budget, max_it=25000)
    if c5_sg < best_c5:
        best_c5 = c5_sg
        best_h = h_sg.copy()

    # Phase 2: SLSQP multi-temperature at original res
    for T in [30, 100, 400, 1500]:
        if tl() < 40:
            break
        try:
            ho = slsqp(best_h, n0, T, 350)
            c5o = compute_c5(ho)
            if c5o < best_c5:
                best_c5 = c5o
                best_h = ho.copy()
        except Exception:
            pass

    # Phase 3: Targeted lag fix at original res
    if tl() > 50:
        hf, c5f = targeted_lag_fix(best_h, n0, t0, budget, n_rounds=8)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()

    # Phase 4: Coordinate descent at original res
    if tl() > 30:
        hc, c5c = cd_polish(best_h, n0, t0, budget, rounds=3)
        if c5c < best_c5:
            best_c5 = c5c
            best_h = hc.copy()

    # Phase 5: Increase to 2x resolution
    if tl() > 120:
        n1 = 2 * n0
        h_up = upsample_h(best_h, n1)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        else:
            best_h = h_up.copy()

        h_sg2, c5_sg2 = sg_descent(best_h, n1, t0, budget, max_it=15000)
        if c5_sg2 < best_c5:
            best_c5 = c5_sg2
            best_h = h_sg2.copy()

        for T in [50, 300, 1000]:
            if tl() < 35:
                break
            try:
                ho = slsqp(best_h, n1, T, 250)
                c5o = compute_c5(ho)
                if c5o < best_c5:
                    best_c5 = c5o
                    best_h = ho.copy()
            except Exception:
                pass

        if tl() > 40:
            hf2, c5f2 = targeted_lag_fix(best_h, n1, t0, budget, n_rounds=5)
            if c5f2 < best_c5:
                best_c5 = c5f2
                best_h = hf2.copy()

        if tl() > 30:
            hc2, c5c2 = cd_polish(best_h, min(n1, 1280), t0, budget, rounds=3)
            if c5c2 < best_c5:
                best_c5 = c5c2
                best_h = upsample_h(hc2, n1) if len(hc2) != n1 else hc2.copy()

    # Phase 6: Try 4x resolution if time allows
    if tl() > 200:
        n2 = 4 * n0
        h_up2 = upsample_h(best_h, n2)
        c5_up2 = compute_c5(h_up2)
        if c5_up2 < best_c5:
            best_c5 = c5_up2

        h_sg3, c5_sg3 = sg_descent(h_up2, n2, t0, budget, max_it=8000)
        if c5_sg3 < best_c5:
            best_c5 = c5_sg3
            best_h = h_sg3.copy()

        for T in [100, 500]:
            if tl() < 30:
                break
            try:
                ho = slsqp(best_h, n2, T, 200)
                c5o = compute_c5(ho)
                if c5o < best_c5:
                    best_c5 = c5o
                    best_h = ho.copy()
            except Exception:
                pass

        if tl() > 30:
            hc3, c5c3 = cd_polish(best_h, min(n2, 2560), t0, budget, rounds=2)
            if c5c3 < best_c5:
                best_c5 = c5c3
                best_h = upsample_h(hc3, n2) if len(hc3) != n2 else hc3.copy()

    # Phase 7: Random restart + optimize (if time)
    if tl() > 120:
        n_r = len(best_h)
        h_r = 0.8 * best_h + 0.2 * rng.rand(n_r)
        h_r = make_feasible(h_r, n_r)
        for T in [200, 800]:
            if tl() < 30:
                break
            try:
                ho = slsqp(h_r, n_r, T, 200)
                c5o = compute_c5(ho)
                if c5o < best_c5:
                    best_c5 = c5o
                    best_h = ho.copy()
            except Exception:
                pass

    # Final polish
    if tl() > 15:
        nf = len(best_h)
        try:
            ho = slsqp(best_h, nf, 2000, 150)
            c5o = compute_c5(ho)
            if c5o < best_c5:
                best_c5 = c5o
                best_h = ho.copy()
        except Exception:
            pass

    if tl() > 15:
        hc_f, c5_f = cd_polish(best_h, min(len(best_h), 1280), t0, budget, rounds=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = upsample_h(hc_f, len(best_h)) if len(hc_f) != len(best_h) else hc_f.copy()

    best_h = make_feasible(best_h, len(best_h))
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
