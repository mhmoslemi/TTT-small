import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def round_binary(h):
    n = len(h)
    k = n // 2
    idx = np.argpartition(h, n - k)[n - k:]
    hb = np.zeros(n)
    hb[idx] = 1.0
    return hb

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2.0, n_from, endpoint=False)
    xt = np.linspace(0, 2.0, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def sm_obj_grad(h, T, lam):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    Tc = T * dx * corr
    mx = Tc.max()
    exps = np.exp(Tc - mx)
    sx = exps.sum()
    w = exps / sx
    obj_sm = (mx + np.log(sx)) / T
    
    s = h.sum()
    target = n / 2.0
    pen = lam * (s - target) ** 2
    obj = obj_sm + pen
    
    # Gradient of soft-max part
    sm_g = np.zeros(n)
    for L in range(-(n-1), n):
        wi = w[L + n - 1]
        if wi < 1e-14:
            continue
        if L >= 0:
            sm_g[L:n] += wi * (1.0 - h[:n-L])
            sm_g[:n-L] -= wi * h[L:]
        else:
            m = -L
            sm_g[:n-m] += wi * (1.0 - h[m:])
            if n - 2*m > 0:
                sm_g[m:n-m] -= wi * h[:n-2*m]
    sm_g *= dx
    
    # Gradient of penalty
    pen_g = 2.0 * lam * (s - target)
    grad = sm_g + pen_g
    return obj, grad

def optimize_lbfgsb(h0, T, lam, deadline, maxiter=300):
    n = len(h0)
    def fun(x):
        obj, _ = sm_obj_grad(x, T, lam)
        return obj
    def fprime(x):
        _, g = sm_obj_grad(x, T, lam)
        return g
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun, h0, jac=fprime, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except:
        pass
    return h0

def proj_grad_desc(h0, T, lam, deadline, n_steps=200, lr0=0.05):
    n = len(h0)
    h = h0.copy()
    lr = lr0
    for step in range(n_steps):
        if time.time() > deadline:
            break
        obj, grad = sm_obj_grad(h, T, lam)
        # Backtracking line search
        for _ in range(15):
            h_new = project(h - lr * grad, n)
            obj_new, _ = sm_obj_grad(h_new, T, lam)
            if obj_new < obj:
                break
            lr *= 0.5
        else:
            break
        h = h_new
        lr = min(lr * 1.1, lr0 * 2)
    return h

import cvxpy as cp

def solve_linearized_lp(h0, n, deadline, max_iters=100):
    h = cp.Variable(n)
    t = cp.Variable()
    corr0 = np.correlate(h0, 1.0 - h0, mode='full')
    dx = 2.0 / n
    
    # Compute gradient of each corr[k] w.r.t. h at h0
    constraints = [h >= 0, h <= 1, cp.sum(h) == n/2]
    
    for k in range(2*n-1):
        L = k - (n-1)  # lag
        # grad at h0: for each j, (1-h0[j-L]) - h0[j+L] (valid parts)
        gj = np.zeros(n)
        if L >= 0:
            gj[L:n] += (1.0 - h0[:n-L])
            gj[:n-L] -= h0[L:]
        else:
            m = -L
            gj[:n-m] += (1.0 - h0[m:])
            if n-2*m > 0:
                gj[m:n-m] -= h0[:n-2*m]

        constraints.append(dx * (corr0[k] + gj @ (h - h0)) <= t)
    
    prob = cp.Problem(cp.Minimize(t), constraints)

        if h.value is not None and np.all(np.isfinite(h.value)):
            return project(np.array(h.value), n)
    except:
        pass
    return h0
