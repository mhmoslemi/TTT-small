from scipy.optimize import minimize
import time

def _cons_fun(x):
    return np.sum(x) - _N_GLOBAL / 2.0

def _cons_jac(x):
    return np.ones(_N_GLOBAL)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def obj_soft(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.clip(shifted, -700.0 / T, 0.0)
    soft = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return soft * dx

def run_slsqp(h, T, maxiter=800):
    global _N_GLOBAL
    n = len(h)
    _N_GLOBAL = n
    bounds = [(0.0, 1.0)] * n

                       bounds=bounds, constraints=cons,

        return make_feasible(res.x, n)
    except Exception:
        return h

            break

                trial[i] = cand
