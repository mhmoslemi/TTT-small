    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    y = h.copy()
    t = 1.0

def project_to_feasible(g, n):

    lo = g.min() - 2.0

        else:

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def fast_correlate(h):
    n = len(h)
    N = 2 * n
    fa = np.fft.rfft(h, N)
    fb = np.fft.rfft(1.0 - h, N)
    cirr = np.fft.irfft(fa * np.conj(fb), N)
    return np.concatenate([cirr[n+1:], cirr[:n]])
