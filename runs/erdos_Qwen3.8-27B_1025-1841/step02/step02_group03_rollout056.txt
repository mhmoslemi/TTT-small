            grad[lag:] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
        else:
            m = -lag
            grad[:n-m] += (1.0 - h[m:])
            grad[m:] -= h[:n-m]

    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)

    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = make_feasible(h_new, n)
    return h_new, momentum, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h = np.full(960, 0.5)

    n = len(h)
    h = make_feasible(h, n)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Subgradient with momentum (main workhorse)
    lr = 0.012
    beta = 0.85
    top_m = 5
    momentum = np.zeros(n)
    patience = 0
    max_patience = 400
    last_best = best_c5
    step_count = 0
    time_limit_1 = start_time + (budget_s - 280)

    while time.time() < time_limit_1:
        h, momentum, c5 = _subgrad_step(h, n, top_m, lr, momentum, beta)
        step_count += 1

        if step_count % 100 == 0:
            c5_exact = compute_c5(h)
            if c5_exact < best_c5 - 1e-10:
                best_c5 = c5_exact
                best_h = h.copy()
                patience = 0
                last_best = best_c5
                lr = min(lr * 1.03, 0.025)
            else:
                patience += 1
                if c5_exact > last_best * 1.004:
                    lr *= 0.93

        if patience > max_patience:
            n_pert = max(5, n // 35)
            idx = rng.choice(n, n_pert, replace=False)
            h = best_h.copy()
            h[idx] += rng.randn(n_pert) * 0.1
            h = make_feasible(h, n)
            momentum = np.zeros(n)
            patience = 0
            max_patience = int(max_patience * 1.3)
            top_m = int(rng.choice([1, 3, 5, 7, 10]))
            lr = 0.015

    best_h = make_feasible(best_h, n)
    best_c5 = compute_c5(best_h)

    # Phase 2: SLSQP polish at multiple temperatures
    for T in [150.0, 400.0]:
        if budget_s - (time.time() - start_time) < 55:
            break
        _G_N = n
        _G_T = T
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-13})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    # Phase 3: Focused subgradient at different top_m
    if budget_s - (time.time() - start_time) > 80:
        h = best_h.copy()
        momentum = np.zeros(n)
        for tm in [1, 3, 7]:
            if budget_s - (time.time() - start_time) < 30:
                break
            slr = 0.006
            for _ in range(5000):
                if budget_s - (time.time() - start_time) < 10:
                    break
                h, momentum, c5 = _subgrad_step(h, n, tm, slr, momentum, 0.9)
                if c5 < best_c5:
                    c5e = compute_c5(h)
                    if c5e < best_c5:
                        best_c5 = c5e
                        best_h = h.copy()

    # Phase 4: Coordinate descent
    if budget_s - (time.time() - start_time) > 60:
        for rnd in range(6):
            if budget_s - (time.time() - start_time) < 15:
                break
            improved = False
            for j in range(n):
                if budget_s - (time.time() - start_time) < 5:
                    break
                old = best_h[j]
                bc = best_c5
                bt = old
                trial_vals = [0.0, 1.0]
                for d in [-0.04, -0.015, 0.015, 0.04]:
                    trial_vals.append(max(0.0, min(1.0, old + d)))
                for cand in trial_vals:
                    if abs(cand - old) < 1e-12:
                        continue
                    trial = best_h.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n)
                    c = compute_c5(trial)
                    if c < bc - 1e-12:
                        bc = c
                        bt = cand
                if bt != old:
                    best_h[j] = bt
                    best_h = make_feasible(best_h, n)
                    best_c5 = bc
                    improved = True
            if not improved:
                break

    # Phase 5: Binary snap
    if budget_s - (time.time() - start_time) > 30:
        h = best_h.copy()
        cc = best_c5
        for j in range(n):
            if budget_s - (time.time() - start_time) < 5:
                break
            for val in [0.0, 1.0]:
                trial = h.copy()
                trial[j] = val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < cc - 1e-12:
                    h = trial
                    cc = c
        if cc < best_c5:
            best_h = h
            best_c5 = cc

    # Phase 6: Final SLSQP
    if budget_s - (time.time() - start_time) > 25:
        _G_N = n
        _G_T = 600.0
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_jac}]
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n)
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
