def project_to_feasible(g, n):

    return corr.max() * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

    """Compute gradient using softmax over top-M correlations."""
    dx = 2.0 / n
    corr = compute_corr(h)
    g = corr * dx

        g_top = g[part]
        max_g = g_top.max()
        shifted = np.clip(tau * (g_top - max_g), -500, 500)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()
    else:
        part = np.arange(2 * n - 1)
        p = np.ones(2 * n - 1) / (2 * n - 1)
    
    grad = np.zeros(n)
    for i in range(len(part)):
        pk = p[i]

            continue
        k = int(part[i])

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    if g.min() >= -1e-12 and g.max() <= 1.0 + 1e-12 and abs(g.sum() - target) < 1e-8:
        return np.clip(g, 0.0, 1.0)
    lo = g.min() - 1.5
    hi = g.max() + 1.5
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def gradient_for_shifts(h, n, top_indices, weights):
    dx = 2.0 / n
    grad = np.zeros(n)
    for i in range(len(top_indices)):
        w = weights[i]
        if w < 1e-14:
            continue
        k = int(top_indices[i])
        d = k - n + 1
        j0 = max(0, -d)
        j1 = min(n, n - d)
        if j0 < j1:
            grad[j0:j1] += w * (1.0 - h[j0 + d:j1 + d])
        j0b = max(0, d)
        j1b = min(n, n + d)
        if j0b < j1b:
            grad[j0b:j1b] -= w * h[j0b - d:j1b - d]
    return grad * dx


def optimize_smooth(h0, n, tau_start, tau_end, max_iters, time_limit,
                    lr=0.12, topM=0, momentum=0.9, nesterov=True):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_h = h.copy()
    best_obj = compute_c5(h)
    start = time.time()
    v = np.zeros(n)
    last_check = start

    for it in range(max_iters):
        if it % 20 == 0 and (time.time() - start) > time_limit:
            break

        frac = min(float(it) / max(1, max_iters), 1.0)
        tau = tau_start + (tau_end - tau_start) * frac
        lr_eff = lr * (1.0 - 0.5 * frac)

        corr = compute_corr(h)
        g = corr * dx
        max_g = g.max()

        if topM > 0 and topM < (2 * n - 1):
            part = np.argpartition(-g, topM - 1)[:topM]
            g_sub = g[part]
            shifted = np.clip(tau * (g_sub - g_sub.max()), -500, 500)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            idx_list = part
        else:
            shifted = np.clip(tau * (g - max_g), -500, 500)
            exp_c = np.exp(shifted)
            w = exp_c / exp_c.sum()
            idx_list = np.arange(2 * n - 1)

        if nesterov:
            h_look = project_to_feasible(h - momentum * v, n)
            corr_look = compute_corr(h_look)
            g_look = corr_look * dx
            if topM > 0 and topM < (2 * n - 1):
                part_l = np.argpartition(-g_look, topM - 1)[:topM]
                g_sub_l = g_look[part_l]
                shifted_l = np.clip(tau * (g_sub_l - g_sub_l.max()), -500, 500)
                exp_l = np.exp(shifted_l)
                w_l = exp_l / exp_l.sum()
                idx_l = part_l
            else:
                shifted_l = np.clip(tau * (g_look - g_look.max()), -500, 500)
                exp_l = np.exp(shifted_l)
                w_l = exp_l / exp_l.sum()
                idx_l = np.arange(2 * n - 1)
            grad = gradient_for_shifts(h_look, n, idx_l, w_l)
        else:
            grad = gradient_for_shifts(h, n, idx_list, w)

        v = momentum * v + lr_eff * grad
        h = project_to_feasible(h - v, n)

        if it % 100 == 0:
            cur = compute_c5(h)
            if cur < best_obj:
                best_obj = cur
                best_h = h.copy()

    cur = compute_c5(h)
    if cur < best_obj:
        best_obj = cur
        best_h = h.copy()
    return best_h, best_obj


def subgrad_topK(h0, n, K, max_iters, time_limit, lr=0.004, momentum=0.85):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_h = h.copy()
    best_obj = compute_c5(h)
    start = time.time()
    v = np.zeros(n)

    for it in range(max_iters):
        if it % 50 == 0 and (time.time() - start) > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_obj - 1e-14:
            best_obj = c5_cur
            best_h = h.copy()

        top_idx = np.argsort(corr)[-K:]
        w = np.ones(K) / K
        grad = gradient_for_shifts(h, n, top_idx, w)

        lr_eff = lr / (1.0 + it * 0.003)
        v = momentum * v + lr_eff * grad
        h = project_to_feasible(h - v, n)

    cur = compute_c5(h)
    if cur < best_obj:
        best_obj = cur
        best_h = h.copy()
    return best_h, best_obj


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def local_refine(h, n, time_limit, max_rounds=500):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_h = h.copy()
    best_obj = compute_c5(h)
    start = time.time()
    rng = np.random.RandomState(12345)

    for rnd in range(max_rounds):
        if (time.time() - start) > time_limit:
            break

        corr = compute_corr(h)
        g = corr * dx
        top1 = np.argmax(g)
        d = top1 - n + 1

        # Find the shift with max gradient contribution at each position
        # Try to reduce h[j] where (1-h[j+d]) is large, increase h[j] where h[j-d] is large
        j0 = max(0, -d)
        j1 = min(n, n - d)
        j0b = max(0, d)
        j1b = min(n, n + d)

        # Score: how much reducing h[j] helps vs increasing h[j]
        score = np.zeros(n)
        if j0 < j1:
            score[j0:j1] -= (1.0 - h[j0 + d:j1 + d])
        if j0b < j1b:
            score[j0b:j1b] -= h[j0b - d:j1b - d]

        # Pick a random position to decrease and transfer to a position to increase
        # (to maintain the sum constraint)
        dec_candidates = np.where(score < -0.1)[0]
        inc_candidates = np.where(score < -0.1)[0]

        if len(dec_candidates) < 2:
            continue

        for trial in range(50):
            j_dec = dec_candidates[rng.randint(len(dec_candidates))]
            j_inc = inc_candidates[rng.randint(len(inc_candidates))]
            if j_dec == j_inc:
                continue
            if h[j_dec] < 1e-6 or h[j_inc] > 1.0 - 1e-6:
                continue

            eps = min(h[j_dec] * 0.1, (1.0 - h[j_inc]) * 0.1, 0.005)
            if eps < 1e-7:
                continue

            h_trial = h.copy()
            h_trial[j_dec] -= eps
            h_trial[j_inc] += eps
            h_trial = project_to_feasible(h_trial, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_obj - 1e-12:
                h = h_trial
                best_h = h.copy()
                best_obj = c5_trial
                break

    return best_h, best_obj
