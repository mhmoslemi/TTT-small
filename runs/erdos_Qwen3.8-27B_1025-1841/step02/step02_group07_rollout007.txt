    n = len(h)

def project(h, n):
    target = n / 2.0
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    # Find tau such that sum(clip(h - tau, 0, 1)) = target
    lo, hi = -1.0, 2.0
    for _ in range(50):
        tau = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - tau, 0.0, 1.0))
        if s > target:
            lo = tau
        else:
            hi = tau
    tau = 0.5 * (lo + hi)
    h = np.clip(h - tau, 0.0, 1.0)
    # Fix rounding
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(-(np.minimum(h, 1.0 - 1e-15)))
        else:
            idx = np.argsort(h)
        for i in idx:
            if abs(diff) < 1e-14:
                break
            if diff > 0 and h[i] < 1.0:
                adj = min(diff, 1.0 - h[i])
                h[i] += adj
                diff -= adj
            elif diff < 0 and h[i] > 0.0:
                adj = max(diff, -h[i])
                h[i] += adj
                diff -= adj
    return h
