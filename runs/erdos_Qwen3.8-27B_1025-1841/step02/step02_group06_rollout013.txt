    """Compute full correlation of h with 1-h using FFT."""
    M = 1
    while M < 2*n - 1:

import numpy as np
import time

def fft_correlate(h):
    n = len(h)
    M = 1 << (2 * n - 1).bit_length()
    h_pad = np.zeros(M)
    h_pad[:n] = h
    rev = np.zeros(M)
    rev[:n] = (1.0 - h)[::-1]
    return np.fft.irfft(np.fft.rfft(h_pad) * np.fft.rfft(rev), M)[:2*n-1]

def project_feasible(g, n):
    target = n * 0.5
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(45):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(g - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - 0.5*(lo+hi), 0.0, 1.0)

def upsample(h, new_n):
    n = len(h)
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, h, period=2.0)

def weighted_grad(h, weights, lags, n, dx):
    grad = np.zeros(n)
    for i, k in enumerate(lags):
        w = weights[i]
        if w < 1e-12:
            continue
        d = abs(k - n + 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        else:
            grad[d:] += w * (1.0 - h[:n-d])
            grad[:n-d] -= w * h[d:]
    return grad * dx

def main_optimize(h0, n, time_limit, lr=0.08, topM=300, tau=25.0):
    start = time.time()
    dx = 2.0 / n
    h = project_feasible(h0.copy(), n)
    corr = fft_correlate(h) * dx
    best_c5 = corr.max()
    best_h = h.copy()
    
    v1 = np.zeros(n)  # momentum
    v2 = np.zeros(n)  # adaptive LR
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = 0
    stall = 0
    patience = 150
    
    while time.time() - start < time_limit:
        m += 1
        cur = corr.max()
        if cur < best_c5 - 1e-13:
            best_c5 = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Top-M selection
        if topM < 2*n - 1:
            top_idx = np.argpartition(corr, -topM)[-topM:]
            g_top = corr[top_idx]
        else:
            top_idx = np.arange(2*n-1)
            g_top = corr
        
        # Softmax weights
        shift = tau * (g_top - g_top.max())
        p = np.exp(np.clip(shift, -500, 500))
        p /= p.sum()
        
        # Gradient
        grad = weighted_grad(h, p, top_idx, n, dx)
        
        # Adam update
        v1 = beta1 * v1 + (1 - beta1) * grad
        v2 = beta2 * v2 + (1 - beta2) * grad * grad
        v1_hat = v1 / (1 - beta1**m)
        v2_hat = v2 / (1 - beta2**m)
        h_new = project_feasible(h - lr * v1_hat / (np.sqrt(v2_hat) + eps), n)
        
        # Adaptive learning rate
        if stall > patience:
            lr *= 0.75
            v1 *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        
        h = h_new
        corr = fft_correlate(h) * dx
    
    return best_h, best_c5

def subgrad_optimize(h0, n, time_limit, lr=0.002, topK=5):

            best_h = h.copy()
            stall = 0

            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
    
    return best_h, best_c5

import numpy as np
import time


def fft_correlate(h):
    n = len(h)
    M = 1 << (2 * n - 1).bit_length()
    h_pad = np.zeros(M)
    h_pad[:n] = h
    rev = np.zeros(M)
    rev[:n] = (1.0 - h)[::-1]
    return np.fft.irfft(np.fft.rfft(h_pad) * np.fft.rfft(rev), M)[:2 * n - 1]


def project_feasible(g, n, theta_hint=None):
    target = n * 0.5
    if theta_hint is not None and -1.0 < theta_hint < 1.5:
        lo = max(g.min() - 0.1, theta_hint - 0.3)
        hi = min(g.max() + 0.1, theta_hint + 0.3)
        if np.sum(np.clip(g - lo, 0, 1)) < target:
            lo = g.min() - 1.0
        if np.sum(np.clip(g - hi, 0, 1)) > target:
            hi = g.max() + 1.0
    else:
        lo = g.min() - 1.0
        hi = g.max() + 1.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        if np.sum(np.clip(g - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    theta = 0.5 * (lo + hi)
    return np.clip(g - theta, 0.0, 1.0), theta


def upsample(h, new_n):
    n = len(h)
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, h, period=2.0)


def weighted_grad(h, weights, lags, n, dx):
    grad = np.zeros(n)
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-12:
            continue
        d = lags[i] - (n - 1)
        if d == 0:
            grad += w * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += w * (1.0 - h[:n - d])
            grad[:n - d] -= w * h[d:]
        else:
            dn = -d
            grad[:n - dn] += w * (1.0 - h[dn:])
            grad[dn:] -= w * h[:n - dn]
    return grad * dx


def adam_optimize(h0, n, time_limit, lr=0.05, topM=400, tau_start=15.0, tau_end=80.0):
    start = time.time()
    dx = 2.0 / n
    h, theta = project_feasible(h0.copy(), n)
    corr = fft_correlate(h) * dx
    best_c5 = corr.max()
    best_h = h.copy()

    v1 = np.zeros(n)
    v2 = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = 0
    stall = 0
    patience = 200
    total_iters = max(1, int(time_limit / 0.15))

    while time.time() - start < time_limit:
        m += 1
        cur = corr.max()
        if cur < best_c5 - 1e-13:
            best_c5 = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        frac = min(1.0, m / total_iters)
        tau = tau_start + (tau_end - tau_start) * frac

        if topM < 2 * n - 1:
            top_idx = np.argpartition(corr, -topM)[-topM:]
            g_top = corr[top_idx]
        else:
            top_idx = np.arange(2 * n - 1)
            g_top = corr

        shift = tau * (g_top - g_top.max())
        p = np.exp(np.clip(shift, -500, 500))
        p /= p.sum()

        grad = weighted_grad(h, p, top_idx, n, dx)

        v1 = b1 * v1 + (1 - b1) * grad
        v2 = b2 * v2 + (1 - b2) * grad * grad
        v1h = v1 / (1 - b1 ** m)
        v2h = v2 / (1 - b2 ** m)
        h_new_t = h - lr * v1h / (np.sqrt(v2h) + eps)
        h, theta = project_feasible(h_new_t, n, theta)
        corr = fft_correlate(h) * dx

        if stall > patience:
            lr *= 0.72
            v1 *= 0.3
            v2 *= 0.3
            stall = 0
            if lr < 1e-10:
                break

    return best_h, best_c5


def subgrad_optimize(h0, n, time_limit, lr=0.001, topK=4):
    start = time.time()
    dx = 2.0 / n
    h, theta = project_feasible(h0.copy(), n)
    corr = fft_correlate(h) * dx
    best_c5 = corr.max()
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.85
    it = 0
    stall = 0
    patience = 400

    while time.time() - start < time_limit:
        it += 1
        cur = corr.max()
        if cur < best_c5 - 1e-13:
            best_c5 = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for k in top_idx:
            d = k - (n - 1)
            if d == 0:
                grad += 1.0 - 2.0 * h
            elif d > 0:
                grad[d:] += 1.0 - h[:n - d]
                grad[:n - d] -= h[d:]
            else:
                dn = -d
                grad[:n - dn] += 1.0 - h[dn:]
                grad[dn:] -= h[:n - dn]
        grad *= dx / topK

        cur_lr = lr / (1.0 + it * 0.002)
        v = beta * v - cur_lr * grad
        h, theta = project_feasible(h + v, n, theta)
        corr = fft_correlate(h) * dx

        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-13:
                break

    return best_h, best_c5
