def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_soft_obj(h, t):
    corr = compute_corr(h)
    # soft max
    shifted = corr - corr.max()
    log_sum = corr.max() + np.log(np.sum(np.exp(t * shifted))) / t
    return log_sum

def compute_gradient(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # softmax weights
    shifted = corr - corr.max()
    logits = t * shifted + t * corr.max()
    logits -= logits.max()
    p = np.exp(logits)
    p /= p.sum()
    # Now compute gradient
    # corr[k] = sum_i h[i] * (1-h[i + k - (n-1)])  -- need to get indices right
    # For full correlation: corr[k] for k=0,...,2n-2

    # If lag < 0: corr[k] = sum_{i=-lag}^{n-1} h[i]*(1-h[i+lag])
    
    # grad[j] = dx * sum_k p[k] * dcorr[k]/dh[j]
    grad = np.zeros(n)

            continue
        lag = k - (n - 1)
        if lag >= 0:

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)  # periodic interp
    # Make feasible
    h_new = np.clip(h_new, 0, 1)
    target = new_n / 2.0
    diff = target - np.sum(h_new)
    h_new += diff / new_n
    h_new = np.clip(h_new, 0, 1)
    # Fix sum again
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h_new)
        else:
            idx = np.argsort(h_new)
        for i in idx:
            if h_new[i] < 1.0 and diff > 0:
                h_new[i] = min(1.0, h_new[i] + diff)
                diff = target - np.sum(h_new)
            elif h_new[i] > 0.0 and diff < 0:
                h_new[i] = max(0.0, h_new[i] + diff)
                diff = target - np.sum(h_new)
            if abs(diff) < 1e-12:
                break
    return h_new

    target = n / 2.0

    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0 and h[i] < 1.0:

            elif diff < 0 and h[i] > 0.0:
                amt = max(diff, -h[i])
                h[i] += amt

def project(h, n):

    lo, hi = -1.0, 1.0
    for _ in range(60):

        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            hi = mid  # need to subtract more
        else:
            lo = mid  # need to subtract less
    return np.clip(h - (lo+hi)/2, 0.0, 1.0)

```python
def compute_grad_softmax(h, t):
    """Gradient of soft-max objective with temperature t"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    max_c = corr.max()
    p = np.exp(t * (corr - max_c))
    p /= p.sum()
    
    # Gradient: grad[j] = dx * sum_k p[k] * dcorr[k]/dh[j]
    grad = np.zeros(n)
    n_corr = len(corr)  # 2n-1
    for k in range(n_corr):
        pk = p[k]
        if pk < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # corr[k] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j] = (1-h[j+lag]) for j in [0, n-1-lag]
            #          - h[j-lag] for j in [lag, n-1]
            if n - lag > 0:
                idx1 = slice(0, n - lag)
                grad[idx1] += pk * (1.0 - h[idx1 + lag])
            if lag < n:
                idx2 = slice(lag, n)
                grad[idx2] -= pk * h[idx2 - lag]
        else:
            lp = -lag
            # corr[k] = sum_{i=lp}^{n-1} h[i]*(1-h[i-lp])
            # d/dh[j] = (1-h[j-lp]) for j in [lp, n-1]
            #          - h[j+lp] for j in [0, n-1-lp]
            idx1 = slice(lp, n)
            grad[idx1] += pk * (1.0 - h[idx1 - lp])
            if n - lp > 0:
                idx2 = slice(0, n - lp)
                grad[idx2] -= pk * h[idx2 + lp]
    
    grad *= dx
    return grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        for i in idx:
            if diff > 0 and h[i] < 1.0:
                amt = min(diff, 1.0 - h[i])
                h[i] += amt
                diff -= amt
            elif diff < 0 and h[i] > 0.0:
                amt = max(diff, -h[i])
                h[i] += amt
                diff -= amt
            if abs(diff) < 1e-14:
                break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def project_hcp(h, n):
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5*(lo+hi), 0.0, 1.0)

def softobjective(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    shifted = corr - mc
    val = mc + np.log(np.sum(np.exp(t * shifted))) / t
    return val * dx

def grad_softobj(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = corr.max()
    exp_t = np.exp(t * (corr - mc))
    p = exp_t / exp_t.sum()
    grad = np.zeros(n)
    n_corr = len(corr)
    for k in range(n_corr):
        pk = p[k]
        if pk < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                i1 = slice(0, n - lag)
                grad[i1] += pk * (1.0 - h[i1 + lag])
            if lag < n and lag > 0:
                i2 = slice(lag, n)
                grad[i2] -= pk * h[i2 - lag]
            elif lag == 0:
                grad[:] += pk * (1.0 - 2.0 * h)
        else:
            lp = -lag
            i1 = slice(lp, n)
            grad[i1] += pk * (1.0 - h[i1 - lp])
            if n - lp > 0:
                i2 = slice(0, n - lp)
                grad[i2] -= pk * h[i2 + lp]
    grad *= dx
    return grad

def proj_grad_descent(h, n, t, lr_init, n_iter, t0, budget_s):
    h = h.copy()
    lr = lr_init
    for it in range(n_iter):
        if time.time() - t0 > budget_s - 40:
            break
        g = grad_softobj(h, t)
        h_new = h - lr * g
        h_new = project_hcp(h_new, n)
        # Simple line search: accept if objective decreased
        f_old = softobjective(h, t)
        f_new = softobjective(h_new, t)
        retry = 0
        while f_new > f_old and retry < 10:
            lr *= 0.5
            h_new = h - lr * g
            h_new = project_hcp(h_new, n)
            f_new = softobjective(h_new, t)
            retry += 1
        if retry >= 10:
            break
        h = h_new
        # Adaptive lr
        if f_new < f_old * (1 - 1e-7):
            lr = min(lr * 1.02, 5.0 / t)
    return h

def coord_descent_polish(h, n, max_rounds, t0, budget_s):
    best_c5 = compute_c5(h)
    for rnd in range(max_rounds):
        if time.time() - t0 > budget_s - 20:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > budget_s - 20:
                break
            old = h[j]
            best_local = old
            best_local_c5 = best_c5
            candidates = [0.0, 1.0, 0.5,
                          old + 0.01, old - 0.01,
                          old + 0.03, old - 0.03,
                          old + 0.05, old - 0.05,
                          old + 0.005, old - 0.005]
            for c in candidates:
                if c < -1e-6 or c > 1.0 + 1e-6:
                    continue
                trial = h.copy()
                trial[j] = np.clip(c, 0.0, 1.0)
                trial = make_feasible(trial, n)
                c_val = compute_c5(trial)
                if c_val < best_local_c5 - 1e-12:
                    best_local_c5 = c_val
                    best_local = np.clip(c, 0.0, 1.0)
            if best_local != old:
                h[j] = best_local
                h = make_feasible(h, n)
                best_c5 = best_local_c5
                improved = True
        if not improved:
            break
    return h, best_c5

def random_binary_h(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.choice(n, n_ones, replace=False)
    h[idx] = 1.0
    return h

def block_binary_h(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    # Random block structure
    blocks = rng.randint(2, 20)
    block_sizes = np.zeros(blocks)
    for i in range(blocks):
        block_sizes[i] = rng.randint(n // blocks // 2, n // blocks + 2)
    total = block_sizes.sum()
    if total > n:
        block_sizes[-1] = max(1, n - block_sizes[:-1].sum())
    pos = 0
    for i in range(blocks):
        bs = min(int(block_sizes[i]), n - pos)
        if bs <= 0:
            break
        if rng.rand() < 0.5:
            h[pos:pos+bs] = 1.0
        pos += bs
    return make_feasible(h, n)

def slsqp_refine(h, n, t0, budget_s):
    if time.time() - t0 > budget_s - 60:
        return h, compute_c5(h)
    
    global _gN, _gT

    _gT = 80.0
    
    def obj(x):

        mc = corr.max()
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(_gT * shifted))) / _gT) * (2.0 / _gN)
    
    def obj_grad(x):
        return grad_softobj(x, _gT)
    
    def con_fun(x):
        return np.sum(x) - n / 2.0
    
    con = {'type': 'eq', 'fun': con_fun, 'jac': lambda x: np.ones(n)}
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=[con],
                       options={'maxiter': 300, 'ftol': 1e-12})
        if res.success or res.fun < compute_c5(h):
            return res.x, compute_c5(res.x)

        pass
    return h, compute_c5(h)
