    dx = 2.0 / n
    corr = fast_corr_binary(h)
    return corr.max() * dx

def project_to_feasible(g, n):

    lo = g.min() - 2.0
    hi = g.max() + 2.0

    order = np.argsort(h)[::-1]

```python
import numpy as np
import time

def fast_corr(h):
    """Cross-correlation of h with 1-h using FFT."""
    n = len(h)
    size = 2 * n
    H = np.fft.rfft(h, size)
    P = np.fft.rfft(1.0 - h, size)
    return np.fft.irfft(H * P, size)[:2 * n - 1]

def compute_c5(h):
    n = len(h)
    return fast_corr(h).max() * (2.0 / n)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def binary_round(h, n, rng):
    order = np.argsort(h)[::-1]
    h_bin = np.zeros(n, dtype=np.float64)
    h_bin[order[:n // 2]] = 1.0
    return h_bin

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def sa_binary(h_bin, n, time_budget, seed=42, T0=0.003, cool=0.99997, restart_patience=50000):
    """Simulated annealing on binary patterns with restarts."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    
    h = h_bin.copy()
    corr = fast_corr(h)
    c5 = corr.max() * dx
    best_c5 = c5
    best_h = h.copy()
    
    start = time.time()
    T = T0
    no_improve = 0
    
    while time.time() - start < time_budget:
        # Batch of iterations
        for _ in range(1000):
            if time.time() - start > time_budget:
                break
            ones_mask = (h == 1.0)
            zeros_mask = (h == 0.0)
            i = rng.choice(np.where(ones_mask)[0])
            j = rng.choice(np.where(zeros_mask)[0])
            
            h[i] = 0.0
            h[j] = 1.0
            
            corr = fast_corr(h)
            c5_new = corr.max() * dx
            
            delta = c5_new - c5
            if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                c5 = c5_new
                if c5_new < best_c5 - 1e-12:
                    best_c5 = c5_new
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
            
            T *= cool
        
        # Check for restart
        if no_improve > restart_patience:
            h = best_h.copy()
            corr = fast_corr(h)
            c5 = corr.max() * dx
            T = max(T0 * 0.5, T)
            no_improve = 0
    
    return best_h, best_c5
```

def compute_s(p, h, n):
    """Compute s[i] = sum_k p[k] * h[i+k-(n-1)] for i=0,...,n-1."""
    # This is the cross-correlation of p and h, evaluated at specific lags
    # Use FFT

    P = np.fft.rfft(p, size)
    H = np.fft.rfft(h, size)
    # Cross-correlation: ifft(conj(P) * H) gives corr where corr[k] = sum_i p[i] * h[i+k]
    # Actually: ifft(conj(fft(p)) * fft(h)) gives the cross-correlation
    # corr[k] = sum_i p[i] * h[(i+k) % size] (circular)
    # With zero-padding, this equals the linear cross-correlation
    corr = np.fft.irfft(np.conj(P) * H, size)
    # corr[k] = sum_i p[i] * h[i+k] for k = 0, ..., size-1
    # We want s[i] = sum_k p[k] * h[i+k-(n-1)]
    # Let j = i + k - (n-1), so k = j - i + (n-1)
    # s[i] = sum_j p[j-i+(n-1)] * h[j]
    # This is the cross-correlation of h and p at lag (n-1-i):
    # corr_hp[m] = sum_j h[j] * p[j+m]
    # s[i] = corr_hp[n-1-i]
    
    # Compute corr_hp = ifft(conj(fft(h)) * fft(p))
    Hp = np.fft.rfft(h, size)
    Pp = np.fft.rfft(p, size)
    corr_hp = np.fft.irfft(np.conj(Hp) * Pp, size)
    # corr_hp[m] = sum_j h[j] * p[j+m]
    # We need s[i] = sum_j h[j] * p[j + i + 1 - n] = corr_hp[i + 1 - n]
    # For i = 0: corr_hp[1-n] = corr_hp[size + 1 - n] (negative index wraps)
    # Actually in numpy, negative indices wrap: corr_hp[1-n] = corr_hp[size + 1 - n]
    
    s = np.array([corr_hp[(i + 1 - n) % size] for i in range(n)])
    return s

```python
def fictitious_play(h_init, n, time_budget, seed=42):
    """Fictitious play / mean-field approach to minimize max overlap."""
    rng = np.random.RandomState(seed)
    dx = 2.0 / n
    num_lags = 2 * n - 1
    
    # Initialize p uniformly
    p = np.ones(num_lags) / num_lags
    
    h = h_init.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    start = time.time()
    lr_p = 0.1  # Learning rate for p updates
    
    for it in range(1000000):
        if time.time() - start > time_budget:
            break
        
        # Step 1: Compute s[i] = sum_d p[d] * h[i+d-(n-1)]
        s = np.correlate(p, h, mode='full')[:n]
        
        # Step 2: Set h_new[i] = 1 for the n/2 indices with largest s[i]
        # (because we want to minimize sum_i (1-s[i]) * h[i], so put h=1 where s is largest)
        order = np.argsort(s)[::-1]
        h_new = np.zeros(n)
        h_new[order[:n//2]] = 1.0
        
        # Step 3: Compute overlaps for h_new
        corr = np.correlate(h_new, 1.0 - h_new, mode='full')
        overlaps = corr * dx  # These are the c_k values
        c5_new = overlaps.max()
        
        # Step 4: Update p (regret matching or simple gradient)
        # p_k should concentrate on the lags with highest overlap
        # Use: p_k = exp(tau * overlaps_k) / sum(exp(tau * overlaps))
        tau = 10.0  # Temperature
        shifted = overlaps - overlaps.max()
        p_new = np.exp(tau * shifted)
        p_new /= p_new.sum()
        
        # Blend
        p = (1 - lr_p) * p + lr_p * p_new
        
        # Update best
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5
```
