_OPT_TOP_IDX = None

def obj_func(x):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    if _OPT_TOP_IDX is not None:
        top = corr[_OPT_TOP_IDX]
    else:
        top = corr
    mc = np.max(top)
    shifted = top - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def con_func(x):
    return np.sum(x) - _OPT_N / 2.0

def con_jac(x):
    return np.ones(_OPT_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_N, _OPT_T, _OPT_TOP_IDX
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(480, 0.5)

    h_init = make_feasible(h_init, len(h_init))
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Coarse optimization at n=480 for speed
    n_c = 480
    h_c = upsample_h(best_h, n_c)

    for T in [30, 60, 100]:
        if remaining() < 60:
            break
        _OPT_N = n_c
        _OPT_T = T
        _OPT_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_c
        cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
        for restart in range(2):
            if remaining() < 40:
                break
            if restart == 0:
                h_start = h_c.copy()
            else:
                h_start = h_c + rng.randn(n_c) * 0.05
                h_start = make_feasible(h_start, n_c)
            try:
                res = minimize(obj_func, h_start, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-12})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_c)
                    h_c = upsample_h(best_h, n_c)
            except Exception:
                pass

    # Phase 2: Medium resolution n=768
    n_m = 768
    h_m = upsample_h(best_h, n_m)

    for T in [50, 100, 200]:
        if remaining() < 80:
            break
        _OPT_N = n_m
        _OPT_T = T
        _OPT_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_m
        cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
        for restart in range(2):
            if remaining() < 60:
                break
            if restart == 0:
                h_start = h_m.copy()
            else:
                h_start = h_m + rng.randn(n_m) * 0.03
                h_start = make_feasible(h_start, n_m)
            try:
                res = minimize(obj_func, h_start, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 250, 'ftol': 1e-12})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_m)
                    h_m = upsample_h(best_h, n_m)
            except Exception:
                pass

    # Phase 3: High resolution n=1280
    n_h = 1280
    h_h = upsample_h(best_h, n_h)

    for T in [80, 150, 300]:
        if remaining() < 90:
            break
        _OPT_N = n_h
        _OPT_T = T
        _OPT_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_h
        cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
        for restart in range(2):
            if remaining() < 70:
                break
            if restart == 0:
                h_start = h_h.copy()
            else:
                h_start = h_h + rng.randn(n_h) * 0.02
                h_start = make_feasible(h_start, n_h)
            try:
                res = minimize(obj_func, h_start, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 200, 'ftol': 1e-13})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_h)
                    h_h = upsample_h(best_h, n_h)
            except Exception:
                pass

    # Phase 4: Top-k lag focused optimization at current best
    n_curr = len(best_h)
    dx_curr = 2.0 / n_curr
    corr_curr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx_curr
    top_k = 20
    top_idx = np.argpartition(corr_curr, -top_k)[-top_k:]

    for T in [150, 400, 800]:
        if remaining() < 70:
            break
        _OPT_N = n_curr
        _OPT_T = T
        _OPT_TOP_IDX = top_idx
        bounds = [(0.0, 1.0)] * n_curr
        cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
        try:
            res = minimize(obj_func, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-13})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_curr)
                # Update top indices
                corr_curr = np.correlate(best_h, 1.0 - best_h, mode='full') * dx_curr
                top_idx = np.argpartition(corr_curr, -top_k)[-top_k:]
                _OPT_TOP_IDX = top_idx
        except Exception:
            pass

    # Phase 5: Very fine coordinate descent on all points
    n_curr = len(best_h)
    for round_num in range(60):
        if remaining() < 15:
            break
        improved = False
        for j in range(n_curr):
            if remaining() < 5:
                break
            old = best_h[j]
            best_c = best_c5
            best_trial = old
            cands = [0.0, 1.0, 0.5,
                     max(0.0, old - 0.15), max(0.0, old - 0.08), max(0.0, old - 0.03),
                     min(1.0, old + 0.15), min(1.0, old + 0.08), min(1.0, old + 0.03)]
            for cand in cands:
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_curr)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                best_h[j] = best_trial
                best_h = make_feasible(best_h, n_curr)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    # Phase 6: Binary snap pass
    n_curr = len(best_h)
    for j in range(n_curr):
        if remaining() < 5:
            break
        for val in [0.0, 1.0]:
            if abs(best_h[j] - val) < 1e-6:
                continue
            trial = best_h.copy()
            trial[j] = val
            trial = make_feasible(trial, n_curr)
            c = compute_c5(trial)
            if c < best_c5 - 1e-14:
                best_h = trial
                best_c5 = c

    # Phase 7: Pair-flip local search (maintain sum constraint)
    n_curr = len(best_h)
    zeros = np.where(best_h < 0.01)[0]
    ones = np.where(best_h > 0.99)[0]
    if len(zeros) > 0 and len(ones) > 0:
        for zi in range(min(len(zeros), 200)):
            if remaining() < 10:
                break
            z_idx = zeros[zi]
            for oi in range(min(len(ones), 50)):
                o_idx = ones[oi]
                trial = best_h.copy()
                trial[z_idx] = 1.0
                trial[o_idx] = 0.0
                trial = make_feasible(trial, n_curr)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c
                    zeros = np.where(best_h < 0.01)[0]
                    ones = np.where(best_h > 0.99)[0]
                    break

    # Phase 8: Final SLSQP polish
    if remaining() > 50:
        n_curr = len(best_h)
        _OPT_N = n_curr
        _OPT_T = 500.0
        _OPT_TOP_IDX = None
        bounds = [(0.0, 1.0)] * n_curr
        cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
        try:
            res = minimize(obj_func, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 150, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_curr)
        except Exception:
            pass

    # Phase 9: Final fine coordinate descent
    n_curr = len(best_h)
    for round_num in range(40):
        if remaining() < 5:
            break
        improved = False
        for j in range(n_curr):
            if remaining() < 3:
                break
            old = best_h[j]
            best_c = best_c5
            best_trial = old
            for cand in [0.0, 1.0,
                         max(0.0, old - 0.01), max(0.0, old - 0.005),
                         min(1.0, old + 0.01), min(1.0, old + 0.005)]:
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_curr)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_trial = cand
            if best_trial != old:
                best_h[j] = best_trial
                best_h = make_feasible(best_h, n_curr)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    # Phase 10: Try higher resolution if time allows
    if remaining() > 120:
        n_fine = 1920
        h_fine = upsample_h(best_h, n_fine)
        c5_fine = compute_c5(h_fine)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine
        # Quick SLSQP at high resolution
        if remaining() > 80:
            _OPT_N = n_fine
            _OPT_T = 200.0
            _OPT_TOP_IDX = None
            bounds = [(0.0, 1.0)] * n_fine
            cons = [{'type': 'eq', 'fun': con_func, 'jac': con_jac}]
            try:
                res = minimize(obj_func, best_h, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 100, 'ftol': 1e-13})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_fine)
            except Exception:
                pass

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
