    d = lag_idx - n + 1  # actual shift

    j1b = min(n, n + d)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        grad[idx] -= h[idx - d]
    return grad

import numpy as np
import time
import cvxpy as cp


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def get_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def grad_for_lag(h, lag_idx, n):
    d = lag_idx - n + 1
    grad = np.zeros(n)
    j0 = max(0, -d)
    j1 = min(n, n - d)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[idx] += (1.0 - h[idx + d])
    j0b = max(0, d)
    j1b = min(n, n + d)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        grad[idx] -= h[idx - d]
    return grad


def lp_step(h, n, M, solver_list):
    dx = 2.0 / n
    overlaps = get_overlaps(h)
    top_idx = np.argpartition(overlaps, -M)[-M:]
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    for li in top_idx:
        grad = grad_for_lag(h, li, n)
        val = overlaps[li]
        rhs_base = val - dx * np.dot(grad, h)
        constraints.append(t_var >= rhs_base + dx * (grad @ h_var))
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for sol in solver_list:
        try:
            prob.solve(solver=sol)
            if h_var.value is not None and not np.any(np.isnan(h_var.value)):
                return np.asarray(h_var.value), float(t_var.value)
        except Exception:
            continue
    return h, overlays_max(overlaps)


def overlays_max(ov):
    return float(ov.max())


def lp_optimize(h, n, M_schedule, iters_per_M, time_limit, start_time, solver_list):
    for M in M_schedule:
        for it in range(iters_per_M):
            if time.time() - start_time > time_limit:
                break
            h_new, t_val = lp_step(h, n, M, solver_list)
            h_new = project_to_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new <= compute_c5(h) + 1e-12:
                h = h_new
    return h


def subgrad_polish(h, n, max_iters, time_limit, start_time, lr_init=0.003, topK=3):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, -d)
            j1 = min(n, n - d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx + d])
            j0b = max(0, d)
            j1b = min(n, n + d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx - d]
        lr = lr_init / (1.0 + it * 0.008)
        h = project_to_feasible(h - lr * grad * dx / topK, n)
    return best_h, best_c5


def softmax_grad_optimize(h, n, tau, max_iters, time_limit, start_time, lr_init=0.15, topM=0):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    stall = 0
    patience = max(50, max_iters // 10)
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        overlaps = get_overlaps(h)
        if overlaps.max() < best_c5:
            best_c5 = overlaps.max()
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if topM > 0 and topM < len(overlaps):
            top_idx = np.argpartition(overlaps, -topM)[-topM:]
            seg = overlaps[top_idx]
            sh = np.clip(tau * (seg - seg.max()), -400, 400)
            e = np.exp(sh)
            p = e / e.sum()
            grad = np.zeros(n)
            for m in range(len(top_idx)):
                g = grad_for_lag(h, top_idx[m], n)
                grad += p[m] * g
        else:
            sh = np.clip(tau * (overlaps - overlaps.max()), -400, 400)
            e = np.exp(sh)
            p = e / e.sum()
            grad = np.zeros(n)
            for k in range(len(overlaps)):
                if p[k] > 1e-10:
                    grad += p[k] * grad_for_lag(h, k, n)
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
