            h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')

    cv = corr_values(h)
    cmax = np.max(cv)
    idx = np.where(cv >= cmax - thresh_frac)[0]
    return idx - (len(h) - 1), cmax

    """Subgradient of corr_shift with respect to h"""
    n = len(h)
    grad = np.zeros(n)
    s = int(round(shift))
    if abs(s) >= n:
        return grad
    if s >= 0:
        for i in range(n - s):
            grad[i] += (1 - h[i+s])
            grad[i+s] += (-h[i])
    else:
        s2 = -s
        for i in range(s2, n):
            grad[i] += (1 - h[i-s2])
            grad[i-s2] += (-h[i])
    return grad

    """Vectorized subgradient for a single shift"""
    n = len(h)
    grad = np.zeros(n)
    s = int(shift)
    if abs(s) >= n:

    if s >= 0:

        grad[s:] += (-h[:n-s])
    else:

        grad[s2:] += (1 - h[:n-s2])

    n = len(h)
