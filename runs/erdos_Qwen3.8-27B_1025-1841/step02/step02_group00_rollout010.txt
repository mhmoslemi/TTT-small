```python
    # grad_k = (1 - h[i+s_k] - h[i-s_k]) * dx where s_k = k - (n-1)
    # We need: sum_k weights[k] * grad_k
    
    # For each k with non-negligible weight, compute h[i+s_k] and h[i-s_k]
    # This is expensive if many k have significant weight
    # Optimization: for large T, only top few matter
    
    # Count significant shifts
    sig_mask = exp_shifted > 1e-10 * Z
    sig_indices = np.where(sig_mask)[0]
    
    grad = np.zeros(2*n-1)  # Will reduce to n
    grad_n = np.zeros(n)
    
    for k in sig_indices:
        s = k - (n-1)
        w = weights[k]
        # grad_k[i] = 1 - h[i+s] - h[i-s]
        g = np.ones(n)
        if s >= 0:
            g[:n-s] -= h[s:]
            g[s:] -= h[:n-s]
        else:
            g[-s:] -= h[:n+2*s]
            g[:n+2*s] -= h[-s:]
        grad_n += w * g
    
    grad_n *= dx
    return obj, grad_n
```

This is O(n * |sig_indices|) per gradient evaluation. For large T, |sig_indices| is small. For small T, it could be O(n), making it O(n²).

```python
def nesterov_step(h, n, T, top_k=20):
    """One Nesterov step: compute smoothed gradient and update"""
    dx = 2.0 / n
    target = n / 2.0
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    Z = np.sum(exp_s)
    weights = exp_s / Z
    
    # Top-k approximation for gradient
    if top_k < len(weights):
        top_idx = np.argsort(weights)[-top_k:]
        # Sort by weight descending
        top_idx = top_idx[np.argsort(weights[top_idx])[::-1]]
        w_top = weights[top_idx]
        w_top = w_top / np.sum(w_top)
    else:
        top_idx = np.arange(len(weights))
        w_top = weights
    
    grad_n = np.zeros(n)
    for k_idx, k in enumerate(top_idx):
        s = k - (n - 1)
        w = w_top[k_idx]
        g = 1.0
        if s >= 0:
            gp = np.zeros(n)
            gp[:n-s] = h[s:]
            gm = np.zeros(n)
            gm[s:] = h[:n-s]
            grad_n += w * (1.0 - gp - gm)
        else:
            gp = np.zeros(n)
            gp[-s:] = h[:n+2*s]
            gm = np.zeros(n)
            gm[:n+2*s] = h[-s:]
            grad_n += w * (1.0 - gp - gm)
    
    grad_n *= dx
    obj = (mc + np.log(Z) / T) * dx
    return obj, grad_n
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_OG_N = 0
_OG_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        if diff > 0:
            m = h < 1.0 - 1e-13
        else:
            m = h > 1e-13
        c = np.sum(m)
        if c > 0:
            h[m] = np.clip(h[m] + diff / c, 0.0, 1.0)
    return h

def project_box_sum(h, target):
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(h - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _slsqp_obj(h):
    n = _OG_N
    T = _OG_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = np.clip(corr - mc, -40.0, 0.0)
    return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx

def _slsqp_f(h):
    return np.sum(h) - _OG_N / 2.0

def _slsqp_j(h):
    return np.ones(_OG_N)

def slsqp_opt(h_start, n, T, max_iter):
    global _OG_N, _OG_T
    _OG_N = n
    _OG_T = T
    bds = [(0.0, 1.0)] * n
    con = [{'type': 'eq', 'fun': _slsqp_f, 'jac': _slsqp_j}]
    r = minimize(_slsqp_obj, h_start, method='SLSQP',
                 bounds=bds, constraints=con,
                 options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(r.x, n)

def nesterov_opt(h_start, n, T_list, lr_init, max_iter, tlim):
    t0 = time.time()
    target = n / 2.0
    dx = 2.0 / n
    h = h_start.astype(float)
    y = h.copy()
    t = 1.0
    lr = lr_init
    best_h = h.copy()
    best_c5 = compute_c5(h)
    seg = max(1, max_iter // len(T_list))

    for it in range(max_iter):
        if time.time() - t0 > tlim:
            break
        T = T_list[min(it // seg, len(T_list) - 1)]
        corr = np.correlate(y, 1.0 - y, mode='full')
        mc = np.max(corr)
        sh = corr - mc
        es = np.exp(T * sh)
        Z = np.sum(es)
        w = es / Z
        nk = min(len(w), max(10, len(w) // 40))
        ti = np.argsort(w)[-nk:]
        wt = w[ti]
        wt = wt / np.sum(wt)
        g = np.zeros(n)
        for ii in range(nk):
            s = ti[ii] - (n - 1)
            wt_i = wt[ii]
            gp = np.zeros(n)
            gm = np.zeros(n)
            if s >= 0:
                gp[:n - s] = y[s:]
                gm[s:] = y[:n - s]
            else:
                gp[-s:] = y[:n + 2 * s]
                gm[:n + 2 * s] = y[-s:]
            g += wt_i * (1.0 - gp - gm)
        g *= dx
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t_new - 1.0) / t_new
        y_new = h - lr * g
        h_new = project_box_sum(y_new, target)
        y = h_new + beta * (h_new - h)
        h = h_new
        t = t_new
        lr *= 0.9998
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n, tlim):
    t0 = time.time()
    best_c5 = compute_c5(h)
    deltas = [0.008, -0.008, 0.004, -0.004, 0.016, -0.016]
    for _ in range(4):
        if time.time() - t0 > tlim:
            break
        imp = False
        for j in range(n):
            if time.time() - t0 > tlim:
                break
            ov = h[j]
            for d in deltas:
                c = ov + d
                if c < 0.0 or c > 1.0:
                    continue
                tr = h.copy()
                tr[j] = c
                tr = make_feasible(tr, n)
                cv = compute_c5(tr)
                if cv < best_c5 - 1e-13:
                    best_c5 = cv
                    h = tr
                    ov = c
                    imp = True
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                tr = h.copy()
                tr[j] = bv
                tr = make_feasible(tr, n)
                cv = compute_c5(tr)
                if cv < best_c5 - 1e-13:
                    best_c5 = cv
                    h = tr
                    imp = True
        if not imp:
            break
    return h, best_c5
```
