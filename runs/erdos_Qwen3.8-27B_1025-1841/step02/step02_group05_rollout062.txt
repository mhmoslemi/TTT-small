```python
def lp_descent_step(h, n, T, max_alpha=1.0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx  # g_k values
    
    # Softmax weights
    max_g = g.max()
    w = np.exp(-(g - max_g) / T)
    w /= w.sum()
    
    # Compute weighted gradient c = sum_k w_k * grad_k
    c = np.zeros(n)
    for k in range(2*n-1):
        if w[k] < 1e-10:
            continue
        s = k - n + 1
        # grad_k[i] = (1-h[i+s]) for valid i, minus h[i-s] for valid i
        # More precisely:
        # For j in valid range (0≤j<n, 0≤j+s<n):
        #   ∂g_k/∂h[j] += (1-h[j+s])
        #   ∂g_k/∂h[j+s] -= h[j]
        j0 = max(0, -s)
        j1 = min(n, n-s)
        for j in range(j0, j1):
            c[j] += w[k] * (1.0 - h[j+s])
            c[j+s] -= w[k] * h[j]
    c *= dx
    
    # Solve LP: min c^T h_new s.t. 0<=h_new<=1, sum(h_new)=n/2
    from scipy.optimize import linprog
    result = linprog(c, A_eq=np.ones((1,n)), b_eq=n/2.0, bounds=(0,1), method='highs')
    if result.success:
        h_lp = result.x
    else:
        h_lp = h
    
    # Line search for step size
    best_alpha = 0.0
    best_obj = g.max()
    for alpha in np.linspace(0.1, 1.0, 20):
        h_try = h + alpha * (h_lp - h)
        h_try = np.clip(h_try, 0, 1)
        # Re-project to maintain sum
        # Actually the convex combination of two feasible points is feasible
        # (box constraint: convex combo of [0,1] is [0,1])
        # (sum constraint: convex combo of n/2 is n/2)
        # So h_try is already feasible!
        corr_try = np.correlate(h_try, 1.0-h_try, mode='full')
        obj = (corr_try * dx).max()
        if obj < best_obj:
            best_obj = obj
            best_alpha = alpha
    
    return h + best_alpha * (h_lp - h), best_obj
```

```python
import numpy as np
import time
from scipy.optimize import linprog

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_weighted_grad(h, n, weights):
    """Compute c = sum_k w_k * grad_k * dx"""
    dx = 2.0 / n
    c = np.zeros(n)
    for k in range(2 * n - 1):
        if weights[k] < 1e-12:
            continue
        s = k - n + 1
        j0 = max(0, -s)
        j1 = min(n, n - s)
        if j0 < j1:
            c[j0:j1] += weights[k] * (1.0 - h[j0 + s:j1 + s])
            c[j0 + s:j1 + s] -= weights[k] * h[j0:j1]
    c *= dx
    return c

def lp_direction(h, n, c):
    """Solve LP: min c^T h_lp s.t. 0<=h_lp<=1, sum(h_lp)=n/2. Return h_lp - h (direction)."""
    try:
        result = linprog(c, A_eq=np.ones((1, n)), b_eq=n/2.0, 
                        bounds=np.zeros((n, 2)),  # (0, None) for each var... 
                        # Actually bounds should be [(0,1)]*n
                        method='highs')
        if result.success:
            return result.x - h
    except:
        pass
    return None
```

```python
    best_c5 = compute_c5(h)
    best_h = h.copy()

    target = n / 2.0
    
    for it in range(n_iters_max):
        if time.time() - start > time_limit:
            break
        
        # Compute objective values
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        c5_now = g.max()
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
        
        # Temperature schedule
        frac = min(1.0, it / (0.8 * n_iters_max))
        T = T_init * (T_end / T_init) ** frac
        T = max(T, 1e-6 * c5_now)

        max_g = g.max()
        w = np.exp(-(g - max_g) / T)
        w_sum = w.sum()
        if w_sum < 1e-30:

            w[np.argmax(g)] = 1.0

                c[j0+s:j1+s] -= w[k] * h[j0:j1]

        order = np.argsort(c)
        h_lp = np.zeros(n)
        h_lp[order[:int(target)]] = 1.0
        
        # Line search
        best_alpha = 0.0
        best_obj = c5_now

                best_alpha = alpha

            # Try shrinking T
            T *= 0.5
            continue
            
        h = h + best_alpha * (h_lp - h)
    
    return best_h, best_c5
```

```python
def smooth_gradient_descent(h, n, tau, time_limit, lr=0.3, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    stall = 0
    
    for it in range(300000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        
        grad = np.zeros(n)
        for k in range(2*n-1):
            pk = p[k]
            if pk < 1e-12:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0-sh:j1-sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b+sh:j1b+sh]
        grad *= dx
        
        c5_now = max_c * dx
        if c5_now < best_c5 - 1e-12:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            lr *= 0.7
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5
```

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def weighted_gradient(h, w, n):
    dx = 2.0 / n
    c = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-12:
            continue
        s = k - n + 1
        i0 = max(0, -s)
        i1 = min(n, n - s)
        if i0 < i1:
            c[i0:i1] += wk * (1.0 - h[i0 + s:i1 + s])
        i0b = max(0, s)
        i1b = min(n, n + s)
        if i0b < i1b:
            c[i0b:i1b] -= wk * h[i0b - s:i1b - s]
    c *= dx
    return c


def lp_descent(h, n, time_limit, T_init=0.4, T_end=0.005, max_iters=100000):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    target = int(n / 2)
    stall = 0
    current_T = T_init * best_c5

    for it in range(max_iters):
        rem = time_limit - (time.time() - start)
        if rem <= 0:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        c5_now = g.max()

        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            current_T *= 0.4
            stall = 0
            if current_T < 1e-6 * best_c5:
                break

        frac = min(1.0, it / (0.75 * max_iters))
        T_sched = T_init * best_c5 * (T_end / T_init) ** frac
        current_T = max(current_T, T_sched, 1e-8 * best_c5)

        max_g = g.max()
        w = np.exp(-(g - max_g) / current_T)
        w_s = w.sum()
        if w_s < 1e-30:
            w = np.zeros(2 * n - 1)
            w[np.argmax(g)] = 1.0
        else:
            w /= w_s

        c = weighted_gradient(h, w, n)
        order = np.argsort(c)
        h_lp = np.zeros(n)
        h_lp[order[:target]] = 1.0
        direction = h_lp - h
        if np.abs(direction).max() < 1e-12:
            continue

        best_alpha = 0.005
        best_obj = c5_now
        for alpha in (0.002, 0.005, 0.01, 0.02, 0.04, 0.07, 0.1, 0.15,
                      0.2, 0.3, 0.4, 0.5, 0.65, 0.8, 1.0):
            h_t = h + alpha * direction
            corr_t = np.correlate(h_t, 1.0 - h_t, mode='full')
            obj = (corr_t * dx).max()
            if obj < best_obj:
                best_obj = obj
                best_alpha = alpha

        h = h + best_alpha * direction

    return best_h, best_c5


def smooth_grad_descent(h, n, tau, time_limit, lr, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    stall = 0

            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-12:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
        grad *= dx

        c5_now = max_c * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            lr *= 0.6
            v *= 0.2
            stall = 0
            if lr < 1e-14:
                break
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def subgrad_polish(h, n, time_limit, lr=0.001):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5
```
