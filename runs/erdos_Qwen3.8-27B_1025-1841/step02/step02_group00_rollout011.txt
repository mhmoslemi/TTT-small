    T = _gT
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    arg = np.clip(T * (corr - mc), -700.0, 0.0)
    log_sum = np.log(np.sum(np.exp(arg)))
    return (mc + log_sum / T) * dx

    return np.sum(x) - _gN / 2.0

    return np.ones(_gN)

    dx = 2.0 / n

    return float(np.max(corr) * dx)

def max_corr_fft(h):
    """Fast max of correlate(h, 1-h) using FFT"""
    n = len(h)
    N = 1 << (2*n - 1).bit_length()  # next power of 2 >= 2n-1
    # Actually, just use 2*n (or next power of 2)
    N = 2 * n
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(A * np.conj(B), N)
    return float(np.max(corr)) * (2.0 / n)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):

            break

        else:

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def slsqp_run(h_start, n, T, max_iter, ftol):
    global _gN, _gT
    _gN = n
    _gT = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_eq_jac}]
    res = minimize(_obj_slsqp, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': ftol})
    return make_feasible(res.x, n)

def fast_local_search(h, rng, n_iterations, time_budget, start_time):
    """Random swap local search using FFT for fast C5 evaluation"""
    n = len(h)
    h = h.copy()
    dx = 2.0 / n
    best_c5 = max_corr_fft(h)
    
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    accepted = 0
    for it in range(n_iterations):
        if time_budget - (time.time() - start_time) < 5:
            break
        # Random swap: pick one from ones, one from zeros
        if len(ones) == 0 or len(zeros) == 0:
            break
        j = ones[rng.randint(len(ones))]
        k = zeros[rng.randint(len(zeros))]
        
        # Swap: h[j] -> 0 (or reduce), h[k] -> 1 (or increase)
        old_j, old_k = h[j], h[k]

        new_c5 = max_corr_fft(h)
        if new_c5 < best_c5 - 1e-15:
            best_c5 = new_c5
            accepted += 1
            # Update ones/zeros tracking
        else:
            h[j], h[k] = old_j, old_k
    
    return h, best_c5
