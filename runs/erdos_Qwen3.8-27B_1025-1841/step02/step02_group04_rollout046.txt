    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac if T_start > 0 else T_end
        lr_t = lr * (1.0 - 0.6 * frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        if T * max(np.max(shifted), 0) > 450:
            T_eff = 450.0 / max(np.max(shifted), 1e-12)
        else:
            T_eff = T
        exp_s = np.exp(T_eff * shifted)
        total = np.sum(exp_s)
        weights = exp_s / total
        idx_top = np.argsort(weights)[-top_k:]
        g = np.zeros(n)
        for ki in idx_top:
            w = weights[ki]
            if w < 1e-12:
                break
            shift = ki - (n - 1)
            s = abs(shift)
            if shift >= 0:
                if s == 0:
                    g += (1.0 - 2.0 * h)
                else:
                    g[:n - s] += (1.0 - h[s:n])
                    g[s:] += (-h[:n - s])
            else:
                if s == 0:
                    g += (1.0 - 2.0 * h)
                else:
                    g[s:] += (1.0 - h[:n - s])
                    g[:n - s] += (-h[s:n])
            g *= w
        g *= 2.0 / n
        h = h - lr_t * g
        h = make_feasible(h, n)
        if (it + 1) % 80 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def thue_morse_h(n):
    tm = np.array([(i & (i >> 1) & (i >> 2) & (i >> 3) & (i >> 4)).sum() % 2 for i in range(n)], dtype=float)
    h = 1.0 - tm
    return make_feasible(h, n)

def discrete_ls(h, n_flips, seed):
    rng = np.random.RandomState(seed)
    n = len(h)
    h01 = (h > 0.5).astype(float).copy()
    best_c5 = compute_c5(h01)
    best_h01 = h01.copy()
    for it in range(n_flips):
        ones = np.where(h01 == 1)[0]
        zeros = np.where(h01 == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h01[i] = 0.0
        h01[j] = 1.0
        c5 = compute_c5(h01)
        if c5 < best_c5:
            best_c5 = c5
            best_h01 = h01.copy()
        else:
            h01[i] = 1.0
            h01[j] = 0.0
    return best_h01, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None

    if h_init is None:
        n_init = 320
        h_init = (rng.rand(n_init) < 0.5).astype(float)
        target = n_init // 2
        cur = int(h_init.sum())
        if cur > target:
            idx = np.where(h_init == 1)[0]
            h_init[rng.choice(idx, cur - target, replace=False)] = 0.0
        elif cur < target:
            idx = np.where(h_init == 0)[0]
            h_init[rng.choice(idx, target - cur, replace=False)] = 1.0
        h_init = make_feasible(h_init, n_init)

    n0 = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    print(f"Init: n={n0} C5={best_c5:.6f}", flush=True)

    # Phase 1: GD at initial resolution
    if time.time() - t0 < budget_s - 120:
        h1, c1 = gd_solve(best_h, 2000, 60.0, 0.5, 0.06, 50)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        print(f"GD1: C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 2: L-BFGS-B with multiple temperatures
    n_curr = len(best_h)
    for T_val in [40.0, 15.0, 7.0, 3.0, 1.5, 0.8, 0.4]:
        if time.time() - t0 > budget_s - 80:
            break
        h_new = lbfgs_solve(best_h, T_val, 8000.0, 60, 400)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5, best_h = c_new, h_new
            print(f"  LBFGS T={T_val}: C5={best_c5:.6f}", flush=True)
    print(f"LBFGS: C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 3: {0,1} rounding + discrete local search
    h01 = (best_h > 0.5).astype(float)
    h01 = make_feasible(h01, n0)
    c01 = compute_c5(h01)
    if c01 < best_c5:
        best_c5, best_h = c01, h01.copy()
    if time.time() - t0 < budget_s - 80:
        h01b, c01b = discrete_ls(best_h, 12000, seed + 1)
        if c01b < best_c5:
            best_c5, best_h = c01b, h01b.copy()
            print(f"  DiscLS: C5={best_c5:.6f}", flush=True)
    print(f"Disc: C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 4: Thue-Morse start + optimize
    if time.time() - t0 < budget_s - 120:
        h_tm = thue_morse_h(n0)
        c_tm = compute_c5(h_tm)
        h_tm2, c_tm2 = gd_solve(h_tm, 1500, 40.0, 0.3, 0.05, 50)
        c_tm2 = compute_c5(h_tm2)
        if c_tm2 < best_c5:
            best_c5, best_h = c_tm2, h_tm2.copy()
            print(f"  TM: C5={best_c5:.6f}", flush=True)
        h_tm3 = lbfgs_solve(h_tm2, 2.0, 8000.0, 60, 300)
        c_tm3 = compute_c5(h_tm3)
        if c_tm3 < best_c5:
            best_c5, best_h = c_tm3, h_tm3
            print(f"  TM+LBFGS: C5={best_c5:.6f}", flush=True)
    print(f"TM: C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 5: Upsample to 2x
    new_n = min(n0 * 2, 640)
    if time.time() - t0 < budget_s - 180:
        h_up = upsample(best_h, new_n)
        if time.time() - t0 < budget_s - 100:
            h_up, _ = gd_solve(h_up, 1500, 30.0, 0.3, 0.04, 50)
        for T_val in [8.0, 3.0, 1.0, 0.5]:
            if time.time() - t0 > budget_s - 60:
                break
            h_up2 = lbfgs_solve(h_up, T_val, 8000.0, 60, 300)
            if compute_c5(h_up2) < compute_c5(h_up):
                h_up = h_up2
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up.copy()
            print(f"  Up n={new_n}: C5={best_c5:.6f}", flush=True)
    print(f"Up: C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 6: Try 3x upsample
    new_n3 = min(n0 * 3, 960)
    if new_n3 > new_n and time.time() - t0 < budget_s - 150:
        h_up3 = upsample(best_h, new_n3)
        if time.time() - t0 < budget_s - 80:
            h_up3, _ = gd_solve(h_up3, 1000, 20.0, 0.3, 0.03, 40)
        for T_val in [5.0, 1.5, 0.5]:
            if time.time() - t0 > budget_s - 40:
                break
            h_up3b = lbfgs_solve(h_up3, T_val, 8000.0, 50, 250)
            if compute_c5(h_up3b) < compute_c5(h_up3):
                h_up3 = h_up3b
        c_up3 = compute_c5(h_up3)
        if c_up3 < best_c5:
            best_c5, best_h = c_up3, h_up3.copy()
            print(f"  Up3 n={new_n3}: C5={best_c5:.6f}", flush=True)
    print(f"Up3: C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 7: Multi-start from perturbations
    n_pts = len(best_h)
    if time.time() - t0 < budget_s - 80:
        for sc in [0.03, 0.08, 0.15]:
            if time.time() - t0 > budget_s - 40:
                break
            h_p = best_h + rng.randn(n_pts) * sc
            h_p = make_feasible(h_p, n_pts)
            h_r, c_r = gd_solve(h_p, 800, 15.0, 0.3, 0.04, 40)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  Pert {sc}: C5={best_c5:.6f}", flush=True)
            h_r2 = lbfgs_solve(h_r, 1.0, 8000.0, 50, 200)
            c_r2 = compute_c5(h_r2)
            if c_r2 < best_c5:
                best_c5, best_h = c_r2, h_r2
                print(f"  Pert+LB {sc}: C5={best_c5:.6f}", flush=True)
    print(f"Multi: C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 8: Final polish
    if time.time() - t0 < budget_s - 20:
        h_f = lbfgs_solve(best_h, 0.3, 12000.0, 60, 500)
        c_f = compute_c5(h_f)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
    if time.time() - t0 < budget_s - 5:
        h_ff, c_ff = gd_solve(best_h, 800, 1.5, 0.01, 0.015, 50)
        if c_ff < best_c5:
            best_c5, best_h = c_ff, h_ff.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)} C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
