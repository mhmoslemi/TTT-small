        h_start = (rng.rand(n0) < 0.5).astype(float)
        h_start = make_feasible(h_start, n0)

    best_h = h_start.copy()
    best_c5 = compute_c5(best_h)

    def elapsed():
        return time.time() - start_time

    def remaining():
        return budget_s - elapsed()

    # Phase 1: Optimize at native resolution with temperature annealing
    n_cur = len(best_h)
    temps_phase1 = [120.0, 60.0, 30.0, 15.0, 8.0, 4.0]
    for temp in temps_phase1:
        if remaining() < 80:
            break
        _G_N = n_cur
        _G_TEMP = temp
        try:
            res_x = slsqp_optimize(best_h, n_cur, temp, maxiter=400, ftol=1e-12)
            c = compute_c5(res_x)
            if c < best_c5:
                best_c5 = c
                best_h = res_x.copy()
        except Exception:
            pass

    # Phase 2: Perturbation restart at same resolution
    for _ in range(2):
        if remaining() < 100:
            break
        pert = best_h.copy()
        mask = rng.rand(n_cur) < 0.15
        pert[mask] += rng.randn(np.sum(mask)) * 0.1
        pert = make_feasible(pert, n_cur)
        for temp in [30.0, 12.0]:
            if remaining() < 60:
                break
            try:
                res_x = slsqp_optimize(pert, n_cur, temp, maxiter=250, ftol=1e-11)
                c = compute_c5(res_x)
                if c < best_c5:
                    best_c5 = c
                    best_h = res_x.copy()
                pert = res_x.copy()
            except Exception:
                pass

    # Phase 3: Upscale to higher resolutions
    for target_n in [512, 768]:
        if remaining() < 150:
            break
        h_up = upsample(best_h, target_n)
        if elapsed() - start_time > budget_s * 0.1:
            pass
        temps_up = [80.0, 40.0, 20.0, 10.0, 5.0]
        maxiters = [300, 250, 200, 150, 100]
        for idx, (temp, mi) in enumerate(zip(temps_up, maxiters)):
            if remaining() < 80:
                break
            try:
                res_x = slsqp_optimize(h_up, target_n, temp, maxiter=mi, ftol=1e-11)
                c = compute_c5(res_x)
                if c < best_c5:
                    best_c5 = c
                    best_h = res_x.copy()
                h_up = res_x.copy()
            except Exception:
                break

    # Phase 4: Subgradient mixer to escape local optima
    if remaining() > 120:
        n_m = len(best_h)
        h_mix, c_mix = subgradient_mixer(best_h, n_m, n_steps=300,
                                          time_limit=budget_s - 100, start_time=start_time)
        if c_mix < best_c5:
            best_c5 = c_mix
            best_h = h_mix.copy()
        # Re-optimize after mixing
        if remaining() > 60:
            try:
                res_x = slsqp_optimize(best_h, len(best_h), 15.0, maxiter=200, ftol=1e-11)
                c = compute_c5(res_x)
                if c < best_c5:
                    best_c5 = c
                    best_h = res_x.copy()
            except Exception:
                pass

    # Phase 5: Coordinate descent polish
    if remaining() > 40:
        n_cd = len(best_h)
        best_h, best_c5 = coord_descent(best_h, n_cd, max_sweeps=30,
                                        time_limit=budget_s - 30, start_time=start_time)

    # Phase 6: Try binary rounding
    h_bin = try_binary_round(best_h, len(best_h))
    c_bin = compute_c5(h_bin)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin.copy()
        # Polish the binary
        if remaining() > 30:
            best_h, best_c5 = coord_descent(best_h, len(best_h), max_sweeps=10,
                                            time_limit=budget_s - 20, start_time=start_time)

    # Final: one more SLSQP polish on the best
    if remaining() > 30:
        try:
            res_x = slsqp_optimize(best_h, len(best_h), 5.0, maxiter=100, ftol=1e-12)
            c = compute_c5(res_x)
            if c < best_c5:
                best_c5 = c
                best_h = res_x.copy()
        except Exception:
            pass

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
