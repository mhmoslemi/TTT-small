def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(v, target):
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(v + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-9:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()

        if time_left() < 80:
            break
        h_r = 0.7 * h2 + 0.3 * rng.rand(n2)
        h_r = make_feasible(h_r, n2)
        h_r = nesterov_loop(h_r, n2, [30.0, 200.0], [0.1, 0.05], 0.9, target2, 200, time_left)
        c_r = compute_c5(h_r)
        if c_r < best_c5:
            best_c5 = c_r
            best_h = h_r.copy()
            h2 = h_r.copy()

    # Phase 4: Binary-like restart
    if time_left() > 60:
        h_b = binary_project(h2, n2, rng)
        h_b = nesterov_loop(h_b, n2, [15.0, 100.0], [0.1, 0.05], 0.9, target2, 200, time_left)
        c_b = compute_c5(h_b)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = h_b.copy()

    # Phase 5: SLSQP polish at n=2048
    if time_left() > 50:
        h5 = upsample_h(best_h, n2)
        try:
            h5 = slsqp_opt(h5, n2, 300.0, 200)
            c5 = compute_c5(h5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h5.copy()
        except Exception:
            pass

    # Phase 6: Very high T Nesterov (near-exact max)
    if time_left() > 40:
        h6 = h2 = upsample_h(best_h, n2)
        for T, lr in [(5000.0, 0.02), (20000.0, 0.01)]:
            if time_left() < 30:
                break
            h6 = nesterov_loop(h6, n2, [T], [lr], 0.85, target2, 150, time_left)
            c6 = compute_c5(h6)
            if c6 < best_c5:
                best_c5 = c6
                best_h = h6.copy()
                h2 = h6.copy()

    # Phase 7: Targeted coordinate descent on worst shifts
    if time_left() > 25:
        nc = min(len(best_h), 2048)
        hc = upsample_h(best_h, nc)
        dx_c = 2.0 / nc
        for round_i in range(8):
            if time_left() < 10:
                break
            corr_c = np.correlate(hc, 1.0 - hc, mode='full')
            top_k = np.argsort(corr_c)[-3:]
            improved = False
            for ki in top_k:
                k = ki - (nc - 1)
                if k < 0:
                    k = -k
                if k == 0:
                    continue
                if k <= nc - 1:
                    overlap = hc[:nc - k] * (1.0 - hc[k:])
                    for trial_frac in [0.0, 0.3, 0.5, 0.7]:
                        if time_left() < 10:
                            break
                        indices = np.where(overlap > 0.1)[0]
                        if len(indices) == 0:
                            continue
                        idx_pick = indices[rng.randint(len(indices))]
                        trial = hc.copy()
                        trial[idx_pick + k] = min(1.0, trial[idx_pick + k] + 0.4)
                        trial[idx_pick] = max(0.0, trial[idx_pick] - 0.4 * trial_frac)
                        trial = make_feasible(trial, nc)
                        ct = compute_c5(trial)
                        if ct < best_c5 - 1e-12:
                            best_c5 = ct
                            hc = trial
                            improved = True
            if not improved:
                break
        if nc < len(best_h):
            if compute_c5(hc) < best_c5:
                best_c5 = compute_c5(hc)
                best_h = upsample_h(hc, len(best_h))
        else:
            if compute_c5(hc) < best_c5:
                best_c5 = compute_c5(hc)
                best_h = hc.copy()

    # Final
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
