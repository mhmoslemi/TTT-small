    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_c5 = c5_val(h)
    best_h = h.copy()
    start = time.time()
    t_adam = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_adam += 1
        obj, grad = lse_grad(h, tau, n_active)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t_adam)
        v_hat = v / (1 - beta2**t_adam)

        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def project(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
