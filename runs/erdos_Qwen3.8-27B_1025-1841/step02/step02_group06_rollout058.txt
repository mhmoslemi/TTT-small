    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)

            grad[j0:j1] -= h[j0-d:j1-d]

def proj_grad_opt(h, n, tau, max_iters, time_limit, lr_init=0.2, topM=200):

    h = project_to_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(50, max_iters // 20)

        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Get top M lags
        if topM > 0 and topM < (2*n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            w = np.exp(shifted)
            w /= w.sum()
            # Weighted gradient from top M
            grad = np.zeros(n)
            for i in range(len(top_idx)):
                pk = w[i]
                if pk < 1e-14:
                    continue
                k = top_idx[i]
                d = k - n + 1
                j0 = max(0, -d)
                j1 = min(n, n - d)
                if j0 < j1:
                    grad[j0:j1] += pk * (1.0 - h[j0+d:j1+d])
                j0b = max(0, d)
                j1b = min(n, n + d)
                if j0b < j1b:
                    grad[j0b:j1b] -= pk * h[j0b-d:j1b-d]
            grad *= dx
        else:
            max_g = g.max()
            shifted = np.clip(tau * (g - max_g), -500, 500)
            w = np.exp(shifted)
            w /= w.sum()
            grad = np.zeros(n)
            for i in range(len(w)):
                pk = w[i]
                if pk < 1e-10:
                    continue
                d = i - n + 1
                j0 = max(0, -d)
                j1 = min(n, n - d)
                if j0 < j1:
                    grad[j0:j1] += pk * (1.0 - h[j0+d:j1+d])
                j0b = max(0, d)
                j1b = min(n, n + d)
                if j0b < j1b:
                    grad[j0b:j1b] -= pk * h[j0b-d:j1b-d]
            grad *= dx
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.75
            v *= 0.4
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    
    return best_h, best_obj

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def fft_corr(h):
    """Compute np.correlate(h, 1-h, mode='full') using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    h1 = 1.0 - h
    m_rev = h1[::-1]
    H = np.fft.rfft(h, N)
    MR = np.fft.rfft(m_rev, N)
    result = np.fft.irfft(H * MR, N)
    return result[:2*n-1]

def compute_c5(h):
    n = len(h)
    return fft_corr(h).max() * (2.0 / n)

def compute_grad_topM(h, lags, weights, n, dx):
    """Compute weighted gradient from specific lags."""
    grad = np.zeros(n)
    for i in range(len(lags)):
        pk = weights[i]
        if pk < 1e-14:
            continue
        k = lags[i]
        d = k - n + 1
        # term1: (1-h[j+d]) for valid j
        j0 = max(0, -d)
        j1 = min(n, n - d)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0+d:j1+d])
        # term2: -h[j-d] for valid j
        j0b = max(0, d)
        j1b = min(n, n + d)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b-d:j1b-d]
    return grad * dx

def proj_grad_opt(h, n, tau, max_iters, time_limit, lr_init=0.15, topM=200):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = fft_corr(h)
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(50, max_iters // 20)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = fft_corr(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive topM
        if topM > 0:
            eff_topM = min(topM, 2*n-1)
            top_idx = np.argpartition(g, -eff_topM)[-eff_topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            w = np.exp(shifted)
            s = w.sum()
            if s < 1e-30:
                w = np.zeros_like(g_top)
                w[0] = 1.0
            else:
                w /= s
            grad = compute_grad_topM(h, top_idx, w, n, dx)
        else:

            shifted = np.clip(tau * (g - max_g), -500, 500)
            w = np.exp(shifted)
            w /= w.sum()
            grad = compute_grad_topM(h, np.arange(2*n-1), w, n, dx)

            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    
    return best_h, best_obj
