import numpy as np
import time
import math

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def adam_smooth_optimize(h, n, tau_start, tau_end, n_iter, time_limit, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8, kick_interval=200, kick_magnitude=0.02):
    """Adam optimizer with smooth max and adaptive tau."""
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    
    for it in range(n_iter):
        if time.time() - start > time_limit:
            break
        
        # Adaptive tau
        if it == 0:
            progress = 0.0
        else:
            progress = min(1.0, it / (n_iter * 0.8))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        obj, grad = smooth_obj_grad(h, tau)
        c5_now = compute_c5(h)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        effective_lr = lr
        if stall > kick_interval:
            effective_lr *= 0.5
            stall = 0
        
        h_new = h - effective_lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)
        
        # Periodic kick
        if it > 0 and it % (kick_interval * 3) == 0 and stall > kick_interval:
            noise = np.random.randn(n) * kick_magnitude
            h = project_to_feasible(h + noise, n)
            stall = 0
            m = np.zeros(n)
            v = np.zeros(n)
            t = 0
    
    return best_h, best_c5

            idx2 = np.arange(0, n - sh_neg)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    t = 0
    stall = 0

    while time.time() - start < time_limit:
        n_iter += 1
        progress = min(1.0, n_iter / max(1, int(time_limit / 0.03)))
        tau = tau_start * (tau_end / tau_start) ** progress

def fast_grad(h, tau, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shf = np.clip(tau * (corr - max_c), -500, 500)
    ec = np.exp(shf)
    s = ec.sum()
    p = ec / s
    top_idx = np.argsort(p)[::-1][:top_k]
    mask = p[top_idx] > 1e-8
    top_idx = top_idx[mask]
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        sh = k - n + 1
        if sh >= 0:
            grad[:n-sh] += pk * (1.0 - h[sh:])
            grad[sh:] -= pk * h[:n-sh]
        else:
            sn = -sh
            grad[sn:] += pk * (1.0 - h[:n-sn])
            grad[:n-sn] -= pk * h[sn:]
    return grad * dx
