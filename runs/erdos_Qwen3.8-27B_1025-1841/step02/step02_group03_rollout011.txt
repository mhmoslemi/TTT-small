```python
def swap_delta(h, p, q, n):
    k = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # p: 1->0
    i1 = p - k
    valid1 = (i1 >= 0) & (i1 < n)
    delta[valid1] -= (1 - h[i1[valid1]])
    
    j1 = p + k
    valid2 = (j1 >= 0) & (j1 < n)
    delta[valid2] += h[j1[valid2]]
    
    # q: 0->1

    valid3 = (i2 >= 0) & (i2 < n)
    delta[valid3] += (1 - h[i2[valid3]])
    
    j2 = q + k
    valid4 = (j2 >= 0) & (j2 < n)
    delta[valid4] -= h[j2[valid4]]
    
    return delta
```

```python
def delta_swap(h, p, q, n):
    lags = np.arange(-(n-1), n)
    delta = np.zeros(2*n-1)
    
    # Standard contributions
    idx1 = p - lags
    v1 = (idx1 >= 0) & (idx1 < n)
    delta[v1] -= (1.0 - h[idx1[v1]])
    
    idx2 = p + lags
    v2 = (idx2 >= 0) & (idx2 < n)
    delta[v2] += h[idx2[v2]]
    
    idx3 = q - lags
    v3 = (idx3 >= 0) & (idx3 < n)
    delta[v3] += (1.0 - h[idx3[v3]])
    
    idx4 = q + lags
    v4 = (idx4 >= 0) & (idx4 < n)
    delta[v4] -= h[idx4[v4]]
    
    # Fix for overlap at lag = p-q (cases 1&4 double-counted) and lag = q-p (cases 2&3)
    # At lag = p-q: correct value is -1, formula gives -(1-h[q]) + (-h[p]) = -1 + (-1) = -2
    # So add corrective +1
    # At lag = q-p: correct value is +1, formula gives h[q] + (1-h[p]) = 0 + 0 = 0
    # So add corrective +1
    
    lag_pq = p - q  # this is the lag where cases 1&4 overlap
    idx_pq = lag_pq + (n-1)  # convert to array index
    if 0 <= idx_pq < 2*n-1:
        delta[idx_pq] += 1.0
    
    lag_qp = q - p  # this is the lag where cases 2&3 overlap
    idx_qp = lag_qp + (n-1)
    if 0 <= idx_qp < 2*n-1:
        delta[idx_qp] += 1.0
    
    return delta
```
