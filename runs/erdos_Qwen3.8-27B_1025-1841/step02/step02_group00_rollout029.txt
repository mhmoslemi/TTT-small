    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def round_binary(h, n):
    n_ones = int(round(np.sum(h)))
    n_ones = max(1, min(n - 1, n_ones))
    idx = np.argsort(h)[::-1][:n_ones]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

def binary_sa(h_bin, n, n_iter=100000, T_init=0.008, T_end=0.0005, seed=42):
    rng = np.random.RandomState(seed)
    current = h_bin.copy()
    c_current = compute_c5(current)
    best = h_bin.copy()
    c_best = c_current
    one_pos = np.where(current == 1.0)[0].tolist()
    zero_pos = np.where(current == 0.0)[0].tolist()
    for it in range(n_iter):
        frac = it / max(1, n_iter)
        T = T_init * (T_end / T_init) ** frac
        a_i = rng.randint(len(one_pos))
        b_i = rng.randint(len(zero_pos))
        a = one_pos[a_i]
        b = zero_pos[b_i]
        current[a] = 0.0
        current[b] = 1.0
        c = compute_c5(current)
        delta = c - c_current
        if delta < 0 or (T > 1e-8 and rng.random() < np.exp(-delta / T)):
            c_current = c
            one_pos[a_i] = b
            zero_pos[b_i] = a
            if c < c_best:
                best = current.copy()
                c_best = c
        else:
            current[a] = 1.0
            current[b] = 0.0
    return best, c_best

def lbfhs_opt(h0, n, T, max_iter=400):
    dx = 2.0 / n
    target_sum = n / 2.0
    def obj(x):
        pen = 200.0 * (np.sum(x) - target_sum) ** 2
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = np.clip(T * (corr - mc), -500, 500)
        lse = mc + np.log(np.sum(np.exp(shifted))) / T
        return lse * dx + pen
    res = minimize(obj, h0, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def golden_ratio_pattern(n, shift=0.0):
    phi = (1.0 + 5.0 ** 0.5) / 2.0
    x = np.arange(n) / n
    vals = ((x * 2.0 + shift) * phi) % 1.0
    return (vals < 0.5).astype(float)

def block_pattern(n, block_size):
    h = np.zeros(n)
    for i in range(0, n, 2 * block_size):
        h[i:i + block_size] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        h0 = golden_ratio_pattern(640)
        h0 = make_feasible(h0, len(h0))

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: L-BFGS-B at initial resolution with multiple temperatures
    n0 = len(best_h)
    for T in [15.0, 40.0, 100.0, 300.0, 800.0]:
        if time_left() < 20:
            break
        h_opt = lbfhs_opt(best_h, n0, T, max_iter=400)
        c = compute_c5(h_opt)
        if c < best_c5:
            best_c5 = c
            best_h = h_opt.copy()

    # Phase 2: Binary SA at initial resolution
    if time_left() > 30:
        h_bin = round_binary(best_h, n0)
        h_bin, c_bin = binary_sa(h_bin, n0, n_iter=80000, T_init=0.006, T_end=0.0003, seed=seed)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()

    # Phase 3: Progressive resolution
    for n_pts in [1024, 2048]:
        if time_left() < 150:
            break
        h_curr = upsample_h(best_h, n_pts)
        c_curr = compute_c5(h_curr)
        if c_curr < best_c5:
            best_c5 = c_curr
            best_h = h_curr.copy()

        for T in [50.0, 150.0, 400.0, 1000.0]:
            if time_left() < 40:
                break
            h_opt = lbfhs_opt(h_curr, n_pts, T, max_iter=300)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()
                h_curr = h_opt.copy()

        if time_left() > 60:
            h_bin = round_binary(best_h, n_pts)
            n_iter = 80000 if n_pts <= 1024 else 50000
            h_bin, c_bin = binary_sa(h_bin, n_pts, n_iter=n_iter, T_init=0.005, T_end=0.0002, seed=seed + n_pts)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()

    # Phase 4: Diverse structured restarts
    n_best = len(best_h)
    patterns = []
    for shift in [0.0, 0.25, 0.5, 0.75, 0.12, 0.37, 0.61]:
        patterns.append(golden_ratio_pattern(n_best, shift))
    for bs in [2, 3, 4, 5, 7, 11]:
        patterns.append(block_pattern(n_best, bs))
    patterns.append((rng.rand(n_best) < 0.5).astype(float))

    for pat in patterns:
        if time_left() < 40:
            break
        pat = make_feasible(pat, n_best)
        for alpha in [0.3, 0.5, 0.7]:
            if time_left() < 25:
                break
            h_blend = alpha * best_h + (1.0 - alpha) * pat
            h_blend = make_feasible(h_blend, n_best)
            h_opt = lbfhs_opt(h_blend, n_best, 80.0, max_iter=250)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()

    # Phase 5: Final binary polish at multiple resolutions near best
    for n_try in [n_best, max(512, n_best // 2), min(4096, n_best * 2)]:
        if time_left() < 40:
            break
        if n_try == n_best:
            h_try = best_h.copy()
        else:
            h_try = upsample_h(best_h, n_try)
        h_bin = round_binary(h_try, n_try)
        n_iter = min(100000, max(20000, int(2 * time_left() * 1000 / max(1, n_try))))
        h_bin, c_bin = binary_sa(h_bin, n_try, n_iter=n_iter, T_init=0.004, T_end=0.0001, seed=seed + 7777)
        if c_bin < best_c5:
            best_c5 = c_bin
            best_h = h_bin.copy()

    # Phase 6: One more continuous round from best binary
    if time_left() > 30:
        n_f = len(best_h)
        for T in [200.0, 600.0]:
            if time_left() < 20:
                break
            h_opt = lbfhs_opt(best_h, n_f, T, max_iter=300)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()
        # Round and quick SA
        if time_left() > 20:
            h_bin = round_binary(best_h, n_f)
            h_bin, c_bin = binary_sa(h_bin, n_f, n_iter=40000, T_init=0.003, T_end=0.0001, seed=seed + 12345)
            if c_bin < best_c5:
                best_c5 = c_bin
                best_h = h_bin.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
