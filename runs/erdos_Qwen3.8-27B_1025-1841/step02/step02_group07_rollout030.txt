import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def lsq_obj_and_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    weights /= np.sum(weights)
    
    # Objective value
    obj = (m + np.log(np.sum(weights))) / T * dx + penalty * (np.sum(h) - n/2.0)**2
    
    # Gradient
    # d(corr[k])/dh[j] = (1 - h[j-(k-n+1)])_valid - h[j+(k-n+1)]_valid
    # where valid means the index is in [0, n-1]
    
    # B[j] = sum_k w[k] * h[j+k-(n-1)] (h=0 outside)
    # A_part[j] = sum_k w[k] * h[j-k+(n-1)] (h=0 outside)
    
    j_idx = np.arange(n)[:, None]
    k_idx = np.arange(2*n-1)[None, :]
    
    idx_B = j_idx + k_idx - (n-1)
    mask_B = (idx_B >= 0) & (idx_B < n)
    h_B = np.where(mask_B, h[np.clip(idx_B, 0, n-1)], 0.0)
    B = h_B @ weights  # (n,)
    
    idx_A = j_idx - k_idx + (n-1)
    mask_A = (idx_A >= 0) & (idx_A < n)
    h_A = np.where(mask_A, h[np.clip(idx_A, 0, n-1)], 0.0)
    A_part = h_A @ weights  # (n,)
    
    # grad[j] = dx * ((1 - A_part[j]) - B[j]) + 2*penalty*(sum(h) - n/2)
    s = np.sum(h)
    grad = dx * ((1.0 - A_part) - B) + 2.0 * penalty * (s - n/2.0)
    
    return obj, grad

    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    n1 = len(h_start)
    h_work = project(h_start, n1)
    
    T_schedule = [15.0, 30.0, 60.0, 120.0, 250.0]
    for T in T_schedule:

        penalty = 500.0
        maxiter = 300
        try:

                          jac=True, bounds=[(0.0, 1.0)]*n1,
                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project(res.x, n1)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h_work = h_new
        except Exception:
            pass
    
    # Phase 2: Upscale to higher resolution and optimize
    n2 = n1 * 2  # 672
    h_up = project(interp_to(best_h, n1, n2), n2)
    
    for T in [30.0, 60.0, 120.0, 250.0, 500.0]:

                          options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):

                if c5_new < best_c5:

                    best_h = h_new.copy()
                h_up = h_new
        except Exception:

    for attempt in range(50):

        # Structured perturbation
        rng = np.random.RandomState(seed + 10000 + attempt * 7)
        noise = rng.randn(n_use) * 0.1

                          jac=True, bounds=[(0.0, 1.0)]*n_use,
                          options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project(res.x, n_use)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except Exception:
            pass
    
    # Final: also try at lower resolution (downscale best and re-upscale)
    if time.time() < deadline - 30:
        n_low = n1 // 2
        h_low = project(interp_to(best_h, len(best_h), n_low), n_low)

            try:

                        best_h = h_new.copy()
                    h_low = h_new_low

                pass

    return (best_h, best_c5, len(best_h))

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def lsq_fun(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    lse = m + np.log(np.sum(np.exp(scaled - m)))
    return lse / T * dx + penalty * (np.sum(h) - n / 2.0) ** 2

def lsq_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T * dx
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    weights /= np.sum(weights)
    
    j = np.arange(n)
    k = np.arange(2 * n - 1)
    
    idx_P = j[:, None] - k[None, :] + (n - 1)
    mask_P = (idx_P >= 0) & (idx_P < n)
    P = np.where(mask_P, h[np.clip(idx_P, 0, n - 1)], 0.0) @ weights
    
    idx_Q = j[:, None] + k[None, :] - (n - 1)
    mask_Q = (idx_Q >= 0) & (idx_Q < n)
    Q = np.where(mask_Q, h[np.clip(idx_Q, 0, n - 1)], 0.0) @ weights
    
    s = np.sum(h)
    grad = dx * (1.0 - P - Q) + 2.0 * penalty * (s - n / 2.0)
    return grad

def lsq_wrapper(h, T, penalty):
    return lsq_fun(h, T, penalty), lsq_grad(h, T, penalty)

def lbgfb_opt(h0, n, T, penalty, maxiter, deadline):
    try:

        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

        dx = 2.0 / n

        scaled = corr * T * dx
        m = np.max(scaled)
        lse = m + np.log(np.sum(np.exp(scaled - m)))
        return lse / T * dx + penalty * (np.sum(h) - n / 2.0) ** 2
