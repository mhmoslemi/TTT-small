import numpy as np
import time
from scipy.optimize import minimize

def project_to_feasible(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_max_obj_grad(h, tau):
    """Smooth max of correlations with softplus-like temperature."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        # h[i] * (1-h[i+sh]) for valid i
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    return obj, grad

def topk_obj_grad(h, k):
    """Sum of top-k correlations (or all if k >= 2n-1)."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    # Find top-k threshold
    if k >= 2*n - 1:
        threshold = corr.min()
        weights = np.ones(2*n-1)
    else:
        sorted_idx = np.argsort(corr)[::-1]
        threshold = corr[sorted_idx[k-1]] if k > 1 else corr.max()
        # Soft assignment: weight = 1 if corr > threshold, 0 otherwise
        # For gradient, use soft version
        weights = (corr > threshold).astype(float)
        # If tie at boundary, split weight
        n_above = np.sum(corr > threshold)
        n_equal = np.sum(corr == threshold)
        if n_above < k < n_above + n_equal:
            need = k - n_above
            weights[corr == threshold] = need / n_equal
    
    obj = dx * np.sum(weights * corr)
    
    # Gradient (only shifts with weight > 0 contribute)
    grad = np.zeros(n)
    active = np.where(weights > 0)[0]
    for k_idx in active:
        w = weights[k_idx]
        sh = k_idx - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += w * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= w * h[idx + sh]
    grad *= dx
    return obj, grad

def frank_wolfe(h, n, obj_grad_fn, max_iters, time_limit, step_rule='d25'):
    """Frank-Wolfe method for the hypersimplex constraint."""
    h = project_to_feasible(h, n).copy()
    best_obj, best_grad = obj_grad_fn(h)
    best_h = h.copy()
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = obj_grad_fn(h)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
        
        # FW step: find s in {0,1}^n with sum = n/2 that minimizes <s, grad>
        # This means s[i] = 1 for the n/2 positions with smallest grad[i]
        n_half = n // 2
        threshold_idx = np.argpartition(grad, n_half)[n_half]
        s = (grad <= threshold_idx).astype(float)
        # Handle ties
        if s.sum() > n_half:
            tie_mask = (grad == threshold_idx)
            n_ties = tie_mask.sum()
            n_above = s.sum() - n_ties
            n_select = n_half - n_above
            if n_select < n_ties:
                tie_indices = np.where(tie_mask)[0]
                # Randomly select some of the ties
                perm = np.random.permutation(tie_indices)
                s[tie_indices] = 0
                s[perm[:n_select]] = 1
        elif s.sum() < n_half:
            # Shouldn't happen but handle it
            zero_mask = (s == 0)
            zero_indices = np.where(zero_mask)[0]
            perm = np.random.permutation(zero_indices)
            s[perm[:n_half - s.sum()]] = 1
        
        # Step size
        gap = np.dot(h - s, grad)
        if gap < 1e-10:
            break
        if step_rule == 'd25':
            gamma = 2.0 / (it + 2.5)
        elif step_rule == 'linesearch':
            # Simple line search
            best_gamma, best_val = 0.0, obj
            for gamma_try in [0.1, 0.2, 0.3, 0.5, 0.7, 1.0]:
                h_try = (1 - gamma_try) * h + gamma_try * s
                obj_try, _ = obj_grad_fn(h_try)
                if obj_try < best_val:
                    best_val = obj_try
                    best_gamma = gamma_try
            gamma = best_gamma
        else:
            gamma = 1.0 / (it + 1)
        
        h = (1 - gamma) * h + gamma * s
    
    # Project final
    h = project_to_feasible(h, n)
    return best_h, best_obj

    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
    
    # For each shift, add constraint t >= sum_i h_var[i] * (1-h_fixed[i+sh])
    for k in range(2*n - 1):
        sh = k - n + 1
        # overlap = sum_i h[i]*(1-h_fixed[i+sh]) for valid i
        # valid i: 0<=i<n and 0<=i+sh<n
        # So i in [max(0,-sh), min(n, n-sh))
        i0 = max(0, -sh)
        i1 = min(n, n - sh)
        if i0 < i1:
            coeffs = 1.0 - h_fixed[i0+sh:i1+sh]  # (1-h_fixed[i+sh]) for i in [i0,i1)
            # Wait, h_fixed[i+sh] for i in [i0, i1)
            # i+sh ranges from i0+sh to i1-1+sh
            # We need: sum_{i=i0}^{i1-1} h[i] * (1-h_fixed[i+sh])
            # = dot(h_var[i0:i1], 1-h_fixed[i0+sh:i1+sh])
            expr = cp.sum(h_var[i0:i1] * (1.0 - h_fixed[i0+sh:i1+sh]))
            constraints.append(t >= expr)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    prob.solve(solver=cp.ECOS)  # or CBC, GLPK
    
    if h_var.value is not None:
        return np.clip(h_var.value, 0, 1)
    return h_fixed.copy()

import numpy as np
import time
import cvxpy as cp


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            idx = np.arange(j0a, j1a)
            grad[j0a:j1a] += pk * (1.0 - h[idx + sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx - sh]
    grad *= dx
    return obj, grad


def optimize_adam(h, n, tau, max_iters, time_limit, initial_lr=0.2):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_to_feasible(h_new, n)
        if stall > 150:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
    return best_h, best_obj


def lp_alternate(h, n, max_iters, time_limit, damping=1.0):
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h_cur = h.copy()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        h_var = cp.Variable(n)
        t = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2]
        for k in range(2 * n - 1):
            sh = k - n + 1
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 >= i1:
                continue
            coeffs = 1.0 - h_cur[i0 + sh:i1 + sh]
            constraints.append(t >= cp.sum(h_var[i0:i1] * coeffs))
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iter=5000, abstol=1e-8)
        except Exception:
            try:
                prob.solve(solver=cp.CBC)
            except Exception:
                break
        if h_var.value is None:
            break
        h_lp = np.clip(h_var.value, 0.0, 1.0)
        h_new = (1.0 - damping) * h_cur + damping * h_lp
        h_new = project_to_feasible(h_new, n)
        c5_cur = compute_c5(h_new)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h_new.copy()
        h_cur = h_new
    return best_h, best_c5


def frank_wolfe_smooth(h, n, tau, max_iters, time_limit):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    n_half = n // 2

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
        thresh = np.partition(grad, n_half)[n_half]
        s = (grad <= thresh).astype(float)
        s_sum = s.sum()
        if s_sum > n_half:
            tie_mask = (grad == thresh)
            n_ties = tie_mask.sum()
            n_above = s_sum - n_ties
            n_select = n_half - n_above
            if n_select < n_ties:
                tie_idx = np.where(tie_mask)[0]
                perm = np.random.permutation(tie_idx)
                s[tie_idx] = 0.0
                s[perm[:n_select]] = 1.0
        elif s_sum < n_half:
            zero_idx = np.where(s == 0)[0]
            perm = np.random.permutation(zero_idx)
            s[perm[:n_half - int(s_sum)]] = 1.0
        gamma = 2.0 / (it + 2.5)
        h = (1.0 - gamma) * h + gamma * s
    return best_h, best_obj


def subgrad_polish(h, n, max_iters, time_limit, initial_lr=0.003):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            idx = np.arange(j0a, j1a)
            grad[j0a:j1a] += (1.0 - h[idx + sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= h[idx - sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 300:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
    return best_h, best_c5


def binary_local_search(h, n, max_iters, time_limit):
    start = time.time()
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    while h_bin.sum() > target:
        idx = np.where(h_bin == 1)[0]
        if len(idx) == 0:
            break
        h_bin[np.random.choice(idx)] = 0
    while h_bin.sum() < target:
        idx = np.where(h_bin == 0)[0]
        if len(idx) == 0:
            break
        h_bin[np.random.choice(idx)] = 1
    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = np.random.choice(ones)
        j = np.random.choice(zeros)
        h_bin[i] = 0
        h_bin[j] = 1
        c5_new = compute_c5(h_bin)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_bin.copy()
        else:
            h_bin[i] = 1
            h_bin[j] = 0
    return best_h, best_c5
