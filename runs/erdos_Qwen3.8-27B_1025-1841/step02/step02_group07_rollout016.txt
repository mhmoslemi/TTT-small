import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def coord_sweep(h, n, deadline, n_grid=200):
    """One full coordinate descent sweep."""
    target = n / 2.0
    dx = 2.0 / n
    lags = np.arange(2 * n - 1) - (n - 1)
    
    for i in range(n):
        if time.time() > deadline:
            break
        # Compute corr (O(n) using numpy)
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Compute a_k = d(corr[k])/d(h[i])
        a = np.zeros(2 * n - 1)
        idx1 = i - lags
        mask1 = (idx1 >= 0) & (idx1 < n)
        a[mask1] += (1.0 - h[idx1[mask1]])
        idx2 = i + lags
        mask2 = (idx2 >= 0) & (idx2 < n)
        a[mask2] -= h[idx2[mask2]]
        
        b = corr - a * h[i]

        x_grid = np.linspace(0.0, 1.0, n_grid)
        vals = np.max(a * x_grid[None, :] + b[:, None], axis=0)
        x_opt = x_grid[np.argmin(vals)]

    return h

    """One subgradient step targeting the max lag."""
    target = n / 2.0
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    
    # Subgradient
    lags_arr = np.arange(n)
    # d(corr[k_star])/d(h[p]) = (1-h[p-lag]) * I[0<=p-lag<n] - h[p+lag] * I[0<=p+lag<n]
    g = np.zeros(n)

    mask1 = (idx1 >= 0) & (idx1 < n)
    g[mask1] += (1.0 - h[idx1[mask1]])
    idx2 = lags_arr + lag
    mask2 = (idx2 >= 0) & (idx2 < n)
    g[mask2] -= h[idx2[mask2]]
    
    h_new = h - alpha * dx * g
    h_new = project(h_new, target)
    return h_new
