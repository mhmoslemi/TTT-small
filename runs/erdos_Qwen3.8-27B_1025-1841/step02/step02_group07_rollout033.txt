import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m))) - m) / T * dx + m / T * dx)
    # Simplified: (m + log(sum(exp(scaled-m))))/T * dx

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    weights = exp_vals / np.sum(exp_vals)
    
    grad = np.zeros(n)
    for l_idx in range(2 * n - 1):
        w = weights[l_idx]
        if w < 1e-14:
            continue
        l = l_idx - (n - 1)
        
        # First term: (1-h[j-l]) * I(0 <= j-l < n)
        if l >= 0:
            # j from l to n-1, j-l from 0 to n-1-l
            grad[l:] += w * (1.0 - h[:n-l])
        else:
            # j from 0 to n+l-1 (i.e., 0 to n-|l|-1), j-l from |l| to n-1
            jl = -l  # |l|
            grad[:n-jl] += w * (1.0 - h[jl:])
        
        # Second term: -h[j+l] * I(0 <= j+l < n)
        if l >= 0:
            # j from 0 to n-l-1, j+l from l to n-1
            grad[:n-l] -= w * h[l:]
        else:
            # j from -l to n-1, j+l from 0 to n-|l|-1
            jl = -l
            grad[jl:] -= w * h[:n-jl]
    
    grad *= dx
    return grad

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def optimize_lbfgsb(h0, n, T, deadline, maxiter=500, mu=100.0):
    """Optimize using L-BFGS-B with penalty for equality constraint."""
    target = n / 2.0
    
    def f(x):
        val = soft_max_obj(x, T)
        penalty = mu * (np.sum(x) - target) ** 2
        return val + penalty
    
    def df(x):
        g = soft_max_grad(x, T)
        g += 2 * mu * (np.sum(x) - target)
        return g
    
    try:
        res = minimize(f, h0, jac=df, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10,
                                'maxfun': maxiter * 20})
        if time.time() > deadline:
            return h0, False
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n), True
    except Exception as e:
        pass
    return h0, False

def sliding_window_move(h, n, window_size, deadline):
    """Try local moves: flip a window of values."""
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    rng = np.random.RandomState(42)
    
    for _ in range(200):
        if time.time() > deadline:
            break
        start = rng.randint(0, n - window_size)
        # Try complementing a window
        h_trial = best_h.copy()
        h_trial[start:start+window_size] = 1.0 - h_trial[start:start+window_size]
        h_trial = project(h_trial, n)
        c5 = evaluate_c5(h_trial)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_trial
    
    return best_h, best_c5

```python
def soft_max_grad_fast(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    w = exp_vals / np.sum(exp_vals)  # length 2n-1
    
    # A = convolve(w, 1-h, 'full')[n-1 : 2n-1]
    A = np.convolve(w, 1.0 - h, mode='full')[n-1 : 2*n-1]
    
    # B = correlate(w, h, 'full')[0:n][::-1]
    B = np.correlate(w, h, mode='full')[0:n][::-1]
    
    return dx * (A - B)
```
