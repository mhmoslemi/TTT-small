def evaluate_c5(h):

def softmax_obj_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    obj = (m + np.log(sum_exp)) / T * dx
    w = exp_s / sum_exp  # softmax weights
    
    # Analytical gradient
    grad = np.zeros(n)
    for i in range(n):
        total = 0.0
        for s in range(-(n-1), n):
            idx = n - 1 + s
            val = 0.0
            if 0 <= i + s < n:
                val += (1.0 - h[i + s])
            if 0 <= i - s < n:
                val -= h[i - s]
            total += w[idx] * val
        grad[i] = dx * total
    return obj, grad

Now let me also think about whether the objective function is correct.

The objective is: C₅ = max_k ∫ h(x)(1-h(x+k)) dx

In discrete form: C₅ = max_k corr[k] * dx where corr = np.correlate(h, 1-h, mode='full')

The soft-max approximation: (1/T) * log(sum_k exp(T * corr[k] * dx))

As T → ∞, this approaches max_k corr[k] * dx = C₅.

The gradient I derived is correct for this soft-max objective.

Now, one issue: the soft-max objective is smooth but the true C₅ is the max (non-smooth). The soft-max with high T approaches the max but the gradient becomes concentrated on the maximizing lags. This is actually good for optimization - it focuses the search on the worst lags.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[n-1+s]
        if ws < 1e-16:
            continue
        i_min = max(0, -s)
        i_max = min(n-1, n-1-s)
        if i_max >= i_min:
            grad[i_min:i_max+1] += ws * (1.0 - h[i_min+s:i_max+s+1])
        j_min = max(0, s)
        j_max = min(n-1, n-1+s)
        if j_max >= j_min:
            grad[j_min:j_max+1] -= ws * h[j_min-s:j_max-s+1]
    grad *= dx
    return grad

def project_sum(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, T, maxiter=300, deadline=None):
    target = n / 2.0
    
    def f(x):
        return softmax_obj(x, n, T)
    
    def fgrad(x):
        return softmax_grad(x, n, T)
    
    def c_eq(x):
        return np.sum(x) - target
    
    def c_eq_jac(x):
        return np.ones(n)
    
    try:
        res = minimize(f, h0, jac=fgrad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                      options={'ftol': 1e-14, 'maxiter': maxiter, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0

def l_bfgs_opt(h0, n, T, maxiter=2000, deadline=None):
    target = n / 2.0
    penalty = 5000.0
    
    def f(x):
        obj = softmax_obj(x, n, T)
        pen = penalty * (np.sum(x) - target) ** 2
        return obj + pen
    
    def fgrad(x):
        g = softmax_grad(x, n, T)
        g += 2.0 * penalty * (np.sum(x) - target)
        return g
    
    try:
        res = minimize(f, h0, jac=fgrad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': maxiter*2})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def softmax_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        ws = w[n-1+s]
        if ws < 1e-16:
            continue
        i_start = max(0, s)
        i_end = min(n, n + s)
        if i_end > i_start:
            grad[i_start:i_end] += ws * (1.0 - h[i_start-s:i_end-s])
        i_start2 = max(0, -s)
        i_end2 = min(n, n - s)
        if i_end2 > i_start2:
            grad[i_start2:i_end2] -= ws * h[i_start2+s:i_end2+s]
    grad *= dx
    return grad

def project_sum(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, T, maxiter=300):
    target = n / 2.0
    def f(x): return softmax_obj(x, n, T)
    def fgrad(x): return softmax_grad(x, n, T)
    def c_eq(x): return np.sum(x) - target
    def c_eq_jac(x): return np.ones(n)
    try:
        res = minimize(f, h0, jac=fgrad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                      options={'ftol': 1e-14, 'maxiter': maxiter, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0

def l_bfgs_opt(h0, n, T, maxiter=3000):
    target = n / 2.0
    penalty = 5000.0
    def f(x):
        return softmax_obj(x, n, T) + penalty * (np.sum(x) - target)**2
    def fgrad(x):
        return softmax_grad(x, n, T) + 2*penalty*(np.sum(x) - target)
    try:
        res = minimize(f, h0, jac=fgrad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0

def interp_to(h, n_from, n_to):
    if n_from == n_to:

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def softmax_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    w = exp_s / np.sum(exp_s)
    first = np.convolve(w, 1.0 - h, mode='full')[n-1:2*n-1]
    second = np.correlate(w, h, mode='full')[n-1::-1]
    return dx * (first - second)

def project_sum(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def slsqp_opt(h0, n, T, maxiter=300):
    target = n / 2.0
    def f(x):
        return softmax_obj(x, n, T)
    def fgrad(x):
        return softmax_grad(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    def c_eq_jac(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, jac=fgrad, method='SLSQP',
                      bounds=[(0.0, 1.0)] * n,
                      constraints=[{'type': 'eq', 'fun': c_eq, 'jac': c_eq_jac}],
                      options={'ftol': 1e-15, 'maxiter': maxiter, 'iprint': 0})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0

def l_bfgs_opt(h0, n, T, maxiter=5000):
    target = n / 2.0
    penalty = 10000.0
    def f(x):
        return softmax_obj(x, n, T) + penalty * (np.sum(x) - target) ** 2
    def fgrad(x):
        return softmax_grad(x, n, T) + 2.0 * penalty * (np.sum(x) - target)
    try:
        res = minimize(f, h0, jac=fgrad, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-17, 'gtol': 1e-13, 'maxfun': maxiter * 3})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n)
    except Exception:
        pass
    return h0

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 25

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(best_h)

    # Phase 1: SLSQP with increasing temperature for progressive refinement
    for T in [15.0, 40.0, 100.0, 250.0, 600.0]:
        if time.time() > deadline - 60:
            break
        h_work = slsqp_opt(best_h, n0, T, maxiter=400)
        c5 = evaluate_c5(h_work)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 2: L-BFGS-B for faster convergence at large scale
    for T in [300.0, 800.0]:
        if time.time() > deadline - 60:
            break
        h_work = l_bfgs_opt(best_h, n0, T, maxiter=5000)
        c5 = evaluate_c5(h_work)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 3: Try higher resolutions
    for n_new in [420, 504, 600]:
        if time.time() > deadline - 180:
            break
        h_i = project_sum(interp_to(best_h, len(best_h), n_new), n_new)
        for T in [80.0, 250.0, 600.0]:
            if time.time() > deadline - 60:
                break
            h_work = slsqp_opt(h_i, n_new, T, maxiter=300)
            c5 = evaluate_c5(h_work)
            if c5 < best_c5:
                best_c5, best_h = c5, h_work.copy()
            h_i = h_work

    # Phase 4: Aggressive random restarts with diverse perturbations
    n_b = len(best_h)
    n_restarts = 300
    for att in range(n_restarts):
        if time.time() > deadline - 20:
            break
        r = np.random.randn(n_b)
        if att % 5 == 0:
            h_p = best_h + 0.02 * r
        elif att % 5 == 1:
            fr = 1.0 / (1.0 + np.abs(np.fft.fftfreq(n_b)) * 8.0)
            h_p = best_h + 0.06 * r * fr
        elif att % 5 == 2:
            h_p = best_h + 0.008 * np.cumsum(r)
        elif att % 5 == 3:
            # Block perturbation
            block = max(1, n_b // 20)
            h_p = best_h.copy()
            for b in range(0, n_b - 1, block):
                h_p[b:b+block] += 0.04 * np.random.randn(block)
        else:
            # Antisymmetric perturbation (exploits h -> 1-h symmetry)
            h_p = best_h + 0.03 * (r - r[::-1])

        h_p = project_sum(h_p, n_b)
        T_use = [150.0, 400.0, 800.0, 100.0, 200.0][att % 5]
        h_new = slsqp_opt(h_p, n_b, T_use, maxiter=120)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5, best_h = c5, h_new.copy()

    # Phase 5: L-BFGS refinement on best found
    if time.time() < deadline - 40:
        h_work = l_bfgs_opt(best_h, n_b, 500.0, maxiter=5000)
        c5 = evaluate_c5(h_work)
        if c5 < best_c5:
            best_c5, best_h = c5, h_work.copy()

    # Phase 6: Final polish with very high T
    if time.time() < deadline - 10:
        h_f = slsqp_opt(best_h, len(best_h), 3000.0, maxiter=300)
        c5 = evaluate_c5(h_f)
        if c5 < best_c5:
            best_c5, best_h = c5, h_f.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
