```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    """Compute smooth approximation to max_k corr_k and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        # Correct gradient: d(corr_k)/dh[m] = (1-h[m+sh])*I(0<=m+sh<n) - h[m-sh]*I(0<=m-sh<n)
        # Term 1: (1-h[m+sh]) for m in [max(0,-sh), min(n, n-sh))
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        # Term 2: -h[m-sh] for m in [max(0, sh), min(n, n+sh))
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    return obj, grad
```

```python
def multishift_subgrad(h, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    c5 = max_c * dx
    idx_sorted = np.argsort(corr)[::-1][:top_k]
    top_vals = corr[idx_sorted]
    w = np.exp(top_vals - top_vals[0])
    w /= w.sum()
    grad = np.zeros(n)
    for i, k in enumerate(top_k):
        sh = k - n + 1
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += w[i] * (1.0 - h[j0 + sh:j1 + sh])

        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= w[i] * h[j0b - sh:j1b - sh]
    grad *= dx
    return c5, grad
```

```python
    n = len(h)
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    
    while time.time() - start < time_limit:
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)

            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
    
    return best_h, best_c5
```

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)

    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    return obj, grad


def adam_smooth(h, tau, time_limit, lr=0.005, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0

    while time.time() - start < time_limit:
        obj, grad = smooth_obj_grad(h, tau)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)

        if t % 15 == 0:
            c5_now = compute_c5(h)
            if c5_now < best_c5 - 1e-14:
                best_c5 = c5_now
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        if stall > 400:
            lr *= 0.5
            stall = 0
            m *= 0.5
            if lr < 1e-12:
                break

    c5_now = compute_c5(h)
    if c5_now < best_c5:
        best_c5 = c5_now
        best_h = h.copy()
    return best_h, best_c5


def subgrad_polish(h, time_limit, lr=0.001, top_k=3):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.8
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        idx_sorted = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[idx_sorted]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(idx_sorted):
            sh = k - n + 1
            j0 = max(0, -sh)
            j1 = min(n, n - sh)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 + sh:j1 + sh])
            j0b = max(0, sh)
            j1b = min(n, n + sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b - sh:j1b - sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    c5_now = compute_c5(h)
    if c5_now < best_c5:
        best_c5 = c5_now
        best_h = h.copy()
    return best_h, best_c5
```
