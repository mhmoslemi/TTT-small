def fft_correlate(a, b):
    n = len(a)
    m = len(b)
    size = n + m - 1
    fft_size = 1
    while fft_size < size:
        fft_size *= 2
    fa = np.fft.rfft(a, fft_size)
    fb = np.fft.rfft(b[::-1].copy(), fft_size)
    corr = np.fft.irfft(fa * fb, fft_size)[:size]
    return corr


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
