def focused_gd(h, n_iter, T_start, T_end, lr, top_k, check_every):
    """GD focusing gradient on top-k lags."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5*frac)
        
        corr = corr_vector(h)
        # Top-k lags with softmax weight
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        c_top = corr[top_idx]
        mc = c_top.max()
        shifted = T * (c_top - mc)
        exp_s = np.exp(shifted)
        weights = exp_s / exp_s.sum()
        
        # Compute gradient as weighted sum of subgradients at top lags
        grad = np.zeros(n)
        for idx, k_idx in enumerate(top_idx):
            if weights[idx] < 1e-8:
                continue
            s = k_idx - (n - 1)  # actual shift
            # subgradient of sum_p h[p]*(1-h[p+s]) w.r.t. h
            if s >= 0:
                grad[s:] += weights[idx] * (1.0 - h[:n-s])
                grad[:n-s] += weights[idx] * (-h[s:])
            else:
                t = -s
                grad[:n-t] += weights[idx] * (1.0 - h[t:])
                grad[t:] += weights[idx] * (-h[:n-t])
        
        h = h - lr_t * grad * dx
        h = make_feasible_h(h, n)
        
        if (it+1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
