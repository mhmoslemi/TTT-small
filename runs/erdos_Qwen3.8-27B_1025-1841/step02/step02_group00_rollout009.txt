def smooth_grad(h, T, n):

    mask2 = (idx2 >= 0) & (idx2 < n)

def smooth_grad(h, T, N):
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = T * (c - m)
    p = np.exp(shifted)
    p /= p.sum()
    
    # s_k = k - (N-1)
    grad = np.zeros(N)
    chunk = 256  # process 256 j's at a time
    
    for start in range(0, N, chunk):
        end = min(start + chunk, N)
        J = np.arange(start, end)  # shape (chunk,)
        
        # First term: for each j, sum p[k]*(1-h[j-k+N-1]) over valid k
        # u = j-k+N-1, valid when 0<=u<N and 0<=k<=2N-2
        # k = j+N-1-u, valid when 0<=j+N-1-u<=2N-2 and 0<=u<N
        # u ranges: max(0, j+N-1-(2N-2)) = max(0, j-N+1) to min(N-1, j+N-1)
        #         = max(0, j-N+1) to N-1 (since j<N, j+N-1<2N-1, so min is N-1)
        
        # Let me just compute directly with broadcasting
        # For J (shape chunk), k (shape 2N-1):
        # u = J[:,None] - k[None,:] + (N-1)  → shape (chunk, 2N-1)
        # v = J[:,None] + k[None,:] - (N-1)  → shape (chunk, 2N-1)
        
        k_idx = np.arange(2 * N - 1)
        u = J[:, None] - k_idx[None, :] + (N - 1)  # shape (chunk, 2N-1)
        v = J[:, None] + k_idx[None, :] - (N - 1)  # shape (chunk, 2N-1)
        
        mask_u = (u >= 0) & (u < N)
        mask_v = (v >= 0) & (v < N)
        
        # First term: p[k] * (1-h[u]) for valid
        term1 = np.where(mask_u, p[None, :] * (1.0 - h[np.clip(u, 0, N-1)]), 0.0)
        # Second term: p[k] * h[v] for valid
        term2 = np.where(mask_v, p[None, :] * h[np.clip(v, 0, N-1)], 0.0)
        
        grad[start:end] = dx * (term1.sum(axis=1) - term2.sum(axis=1))
    
    return grad

def full_obj(h, T, N):
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    shifted = T * (c - m)
    logsum = np.log(np.sum(np.exp(shifted)))
    f_val = (m + logsum / T) * dx
    # Penalty for integral constraint
    s = np.sum(h) - N / 2.0
    f_val += 1e4 * s * s
    return f_val
