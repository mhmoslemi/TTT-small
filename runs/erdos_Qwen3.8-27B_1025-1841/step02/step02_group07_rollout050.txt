    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        # Distribute d among non-saturated entries proportional to distance to boundary
        if d > 0:
            room = 1.0 - h[mask]
        else:
            room = h[mask]
        total_room = np.sum(room)
        if total_room < 1e-12:
            break
        adj = d * room / total_room
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def obj_grad(h, T, penalty=1000.0):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    weights = exp_s / sum_exp
    obj_sm = (m + np.log(sum_exp)) / T
    # Penalty
    dev = np.sum(h) - n / 2.0
    obj = obj_sm + penalty * dev * dev
    
    # Gradient of soft-max part
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = weights[idx] * dx
        if w < 1e-14:
            continue
        k = idx - n + 1
        if k >= 0:
            if n - k > 0:
                js = np.arange(k, n)
                grad[js] += w * (1.0 - h[js - k])
            if k < n:
                js2 = np.arange(0, n - k)
                grad[js2] -= w * h[js2 + k]
        else:
            k2 = -k
            if n - k2 > 0:
                js = np.arange(0, n - k2)
                grad[js] += w * (1.0 - h[js + k2])
            if k2 < n:
                js2 = np.arange(k2, n)
                grad[js2] -= w * h[js2 - k2]
    
    # Gradient of penalty
    grad += 2.0 * penalty * dev * np.ones(n)
    
    return obj, grad

        grad[js] += (1.0 - h[js + k2])
        js2 = np.arange(k2, n)
        grad[js2] -= h[js2 - k2]
    return h - lr * grad

    if k >= 0:
        if k < n:

            grad[js] += (1.0 - h[js - k])
