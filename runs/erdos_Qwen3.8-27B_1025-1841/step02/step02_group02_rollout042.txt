import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(g - mid, 0.0, 1.0)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5_val(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def snap_to_binary(h, n):
    """Round to {0,1} with sum = n/2, minimizing L2 distance."""
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin


def smooth_obj_grad(h, tau, dx):
    n = len(h)
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, max_iters, time_limit, lr=0.3):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau, dx)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_obj_grad(h, tau, dx)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
    return best_h, best_obj


def compute_d_i(h, i, n):
    """Compute d_i[k] for all k: derivative of corr[k] w.r.t. h[i] (linear part, k!=0)."""
    d = np.zeros(2 * n - 1)
    k1s = max(0, i - n + 1)
    k1e = min(2 * n - 2, i)
    if k1s <= k1e:
        d[k1s:k1e + 1] = 1.0 - h[i - k1s:i + 1]
    k2s = max(0, -i)
    k2e = min(2 * n - 2, n - 1 - i)
    if k2s <= k2e:
        d[k2s:k2e + 1] -= h[i + k2s:i + k2e + 1]
    return d


def binary_local_search(h_bin, n, max_iters, time_limit, rng=None):
    """Greedy pair-swap local search on {0,1} solution."""
    dx = 2.0 / n
    if rng is None:
        rng = np.random.RandomState(42)
    h_bin = h_bin.copy()
    corr = compute_corr(h_bin)
    best_c5 = corr.max() * dx
    best_h = h_bin.copy()
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        ones = np.where(h_bin == 1)[0]
        zeros = np.where(h_bin == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        np.random.seed(it)
        i = ones[rng.randint(len(ones))]
        # Compute effect of setting h[i]=0
        d = compute_d_i(h_bin, i, n)
        corr_i0 = corr + d
        max_i0 = corr_i0.max()
        k_star = np.argmax(corr_i0)
        # Find best j in zeros to set to 1
        best_j = -1
        best_val = max_i0
        # Sample a subset of zeros for speed
        n_sample = min(len(zeros), 200)
        sample_idx = rng.choice(len(zeros), size=n_sample, replace=False)
        for si in sample_idx:
            j = zeros[si]
            # e_j[k_star]: effect of setting h[j]=1 on corr[k_star]
            ek = 0.0
            ks = k_star - n + 1
            if 0 <= j - ks < n:
                ek += (1.0 - h_bin[j - ks])
            if 0 <= j + ks < n:
                ek -= h_bin[j + ks]
            cand = corr_i0[k_star] + ek
            if cand < best_val:
                best_val = cand
                best_j = j
        if best_j >= 0 and best_val < corr.max() - 1e-14:
            j = best_j
            # Apply swap: h[i] 1->0, h[j] 0->1
            corr = corr + d
            e = compute_d_i(h_bin, j, n)
            corr = corr + e
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            improved = True
            c5_new = corr.max() * dx
            if c5_new < best_c5 - 1e-15:
                best_c5 = c5_new
                best_h = h_bin.copy()
        if not improved and it > 100:
            # Try random restarts of the search
            continue
    return best_h, best_c5


def subgrad_refine(h, n, max_iters, time_limit):
    """Subgradient method with decreasing step size."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5_val(h)
    best_h = h.copy()
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= h[idx + sh]
        grad *= dx
        lr = 0.05 / np.sqrt(it + 1)
        h = project_to_feasible(h - lr * grad, n)
    return best_h, best_c5
