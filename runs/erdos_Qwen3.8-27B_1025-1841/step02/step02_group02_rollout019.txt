import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

        if s > target:
            lo = mid
        else:
            hi = mid

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    return corr.max() * (2.0 / n)

    """Compute smooth max objective and gradient using top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_vals = np.partition(corr, -top_k)[-top_k:]
    max_c = corr.max()
    # Softmax over top-k
    shifted = tau * (max_vals - max_c)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
