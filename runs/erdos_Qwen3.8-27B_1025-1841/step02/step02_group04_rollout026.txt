        else:
            mask = h > 0.0
            if np.any(mask):
                h[mask] += diff * h[mask] / np.sum(h[mask])
                h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def _subgrad_single(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def soft_gradient(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        return _subgrad_single(h, shift) * dx
    weights = exp_s / total
    idx = np.argsort(weights)[::-1][:top_k]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-12:
            break
        shift = k_idx - (n - 1)
        grad += w * _subgrad_single(h, shift)
    return grad * dx

def momentum_gd(h, n_iter, T_start, T_end, lr_start, lr_end, momentum=0.92, top_k_start=80, top_k_end=30, check_every=25):
    n = len(h)
    h = h.astype(float).copy()
    vel = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start + (lr_end - lr_start) * frac
        tk = int(round(top_k_start + (top_k_end - top_k_start) * frac))
        grad = soft_gradient(h, T, top_k=tk)
        vel = momentum * vel - lr * grad
        h = h + vel
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _obj_func(h):
    T = _OPT_PARAMS['T']
    n = _OPT_PARAMS['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_PARAMS['n'] / 2.0

def _cons_jac(h):
    return np.ones(_OPT_PARAMS['n'])

def direct_subgrad_polish(h, n_iter=2000, lr=0.005, tol=1e-8):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    prev_c5 = best_c5
    stuck = 0
    for it in range(n_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        grad = _subgrad_single(h, shift) * dx
        h_try = h - lr * grad
        h_try = make_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-10:
            h = h_try
            best_c5 = c5_try
            best_h = h.copy()
            stuck = 0
        else:
            stuck += 1
            lr *= 0.95
            if stuck > 200:
                break
        if lr < 1e-6:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_c5 = float('inf')
    best_h = None
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    
    n0 = 320
    if h0 is not None:
        n0 = len(h0)
    else:
        h0 = (rng.rand(n0) < 0.5).astype(float)
        n_ones = n0 // 2
        ones_idx = np.where(h0 == 1)[0]
        zeros_idx = np.where(h0 == 0)[0]
        if len(ones_idx) > n_ones:
            h0[rng.choice(ones_idx, len(ones_idx) - n_ones, replace=False)] = 0.0
        elif len(ones_idx) < n_ones:
            h0[rng.choice(zeros_idx, n_ones - len(ones_idx), replace=False)] = 1.0
    
    h0 = make_feasible(h0, n0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    print(f"Initial: n={n0}, C5={best_c5:.6f}", flush=True)
    
    def elapsed():
        return time.time() - start_time
    
    # Phase 1: Aggressive momentum GD
    if elapsed() < budget_s - 250:
        h1, c1 = momentum_gd(best_h, n_iter=3500, T_start=50.0, T_end=1.0,
                            lr_start=0.12, lr_end=0.005, momentum=0.93,
                            top_k_start=100, top_k_end=40, check_every=25)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        print(f"GD1: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    
    # Phase 2: SLSQP polishing
    n_pts = len(best_h)
    if n_pts <= 400:
        for T_val in [25.0, 10.0, 4.0, 2.0]:
            if elapsed() > budget_s - 80:
                break
            _OPT_PARAMS['T'] = T_val
            _OPT_PARAMS['n'] = n_pts
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
            try:
                res = minimize(_obj_func, best_h, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 1000, 'ftol': 1e-15})
                if res.x is not None:
                    hx = make_feasible(res.x, n_pts)
                    cx = compute_c5(hx)
                    if cx < best_c5:
                        best_c5, best_h = cx, hx
            except Exception:
                pass
        print(f"SLSQP: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    
    # Phase 3: Fine momentum GD
    if elapsed() < budget_s - 150:
        h3, c3 = momentum_gd(best_h, n_iter=4000, T_start=5.0, T_end=0.1,
                            lr_start=0.04, lr_end=0.001, momentum=0.9,
                            top_k_start=50, top_k_end=20, check_every=20)
        if c3 < best_c5:
            best_c5, best_h = c3, h3.copy()
        print(f"GD2: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    
    # Phase 4: Direct subgradient polish on active constraint
    if elapsed() < budget_s - 100:
        h4, c4 = direct_subgrad_polish(best_h, n_iter=3000, lr=0.003)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
        print(f"SubGrad: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    
    # Phase 5: Multi-start perturbations
    if elapsed() < budget_s - 200:
        n_pts = len(best_h)
        for pert_scale in [0.03, 0.07, 0.12, 0.2]:
            if elapsed() > budget_s - 100:
                break
            h_pert = best_h + rng.randn(n_pts) * pert_scale
            h_pert = make_feasible(h_pert, n_pts)
            h_r, c_r = momentum_gd(h_pert, n_iter=2500, T_start=40.0, T_end=0.5,
                                   lr_start=0.1, lr_end=0.005, momentum=0.93,
                                   top_k_start=80, top_k_end=30, check_every=30)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  Pert {pert_scale}: C5={best_c5:.6f}", flush=True)
        print(f"MultiStart: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    
    # Phase 6: Upsample and optimize
    for new_n in [400, 480]:
        if elapsed() > budget_s - 180:
            break
        old_x = np.linspace(0.0, 2.0, len(best_h))
        new_x = np.linspace(0.0, 2.0, new_n)
        h_up = np.interp(new_x, old_x, best_h)
        h_up = make_feasible(h_up, new_n)
        
        h_opt, c_opt = momentum_gd(h_up, n_iter=2500, T_start=20.0, T_end=0.5,
                                   lr_start=0.06, lr_end=0.003, momentum=0.92,
                                   top_k_start=60, top_k_end=25, check_every=25)
        # SLSQP if manageable
        if new_n <= 400 and elapsed() < budget_s - 80:
            _OPT_PARAMS['T'] = 4.0
            _OPT_PARAMS['n'] = new_n
            bounds = [(0.0, 1.0)] * new_n
            cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
            try:
                res = minimize(_obj_func, h_opt, method='SLSQP', bounds=bounds,
                              constraints=cons, options={'maxiter': 400, 'ftol': 1e-14})
                if res.x is not None:
                    hx = make_feasible(res.x, new_n)
                    cx = compute_c5(hx)
                    if cx < c_opt:
                        h_opt, c_opt = hx, cx
            except Exception:
                pass
        c_opt = compute_c5(h_opt)
        if c_opt < best_c5:
            best_c5, best_h = c_opt, h_opt.copy()
        print(f"Res {new_n}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    
    # Phase 7: Final fine-tune
    if elapsed() < budget_s - 40:
        n_pts = len(best_h)
        h_f, c_f = momentum_gd(best_h, n_iter=3000, T_start=2.0, T_end=0.05,
                              lr_start=0.01, lr_end=0.0001, momentum=0.85,
                              top_k_start=30, top_k_end=10, check_every=15)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
        
        if elapsed() < budget_s - 20:
            h_s, c_s = direct_subgrad_polish(best_h, n_iter=2000, lr=0.002)
            if c_s < best_c5:
                best_c5, best_h = c_s, h_s.copy()
        print(f"Final: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    
    # Final SLSQP if small enough
    n_pts = len(best_h)
    if n_pts <= 350 and elapsed() < budget_s - 15:
        _OPT_PARAMS['T'] = 1.5
        _OPT_PARAMS['n'] = n_pts
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
        try:
            res = minimize(_obj_func, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': 500, 'ftol': 1e-15})
            if res.x is not None:
                hx = make_feasible(res.x, n_pts)
                cx = compute_c5(hx)
                if cx < best_c5:
                    best_c5, best_h = cx, hx
        except Exception:
            pass
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
