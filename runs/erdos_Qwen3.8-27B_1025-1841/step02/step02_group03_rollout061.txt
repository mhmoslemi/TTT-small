    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in top_lags:

def smooth_grad_obj(h, n, T):
    """Compute the value and gradient of the soft-max smoothed objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(T * shifted)
    weights /= np.sum(weights)
    
    # Objective value
    obj_val = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    
    # Gradient
    grad = np.zeros(n)
    for k_idx in range(len(corr)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        l = k_idx - (n - 1)
        if l >= 0:
            if n - l > 0:
                grad[l:] += w * (1.0 - h[:n-l])
                grad[:n-l] -= w * h[l:]
        else:
            l2 = -l
            if n - l2 > 0:
                grad[:n-l2] += w * (1.0 - h[l2:])
                grad[l2:] -= w * h[:n-l2]
    grad *= dx
    
    return obj_val, grad
