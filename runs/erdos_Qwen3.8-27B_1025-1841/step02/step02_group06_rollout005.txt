```python
import numpy as np
import time

def correlate_fft(a, b):
    """Compute np.correlate(a, b, mode='full') using FFT."""
    n = len(a)
    m = len(b)
    N = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b[::-1], N)
    return np.fft.irfft(fa * fb, N)[:n+m-1]

def project_to_feasible(g, n):
    """Project g onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_fft(h, 1.0 - h)
    return (corr * dx).max()

def grad_single_lag(h, d, n):
    """Gradient of c[d] = sum h[i]*(1-h[i+d]) with respect to h.
    d is the lag in samples (can be negative)."""
    grad = np.zeros(n)
    # c[d] = sum_{i} h[i] * (1-h[i+d]) for valid i
    # d >= 0: i ranges from 0 to n-1-d
    # d < 0: i ranges from -d to n-1, and h[i+d] = h[i-|d|]
    
    if d >= 0:
        i_max = n - d
        for i in range(i_max):
            # dc/dh[i] += (1 - h[i+d])
            # dc/dh[i+d] += -h[i]
            grad[i] += (1.0 - h[i + d])
            grad[i + d] -= h[i]
    else:
        dd = -d
        for i in range(dd, n):
            # c[d] includes term h[i] * (1-h[i-dd]) (since i+d = i-dd)
            grad[i] += (1.0 - h[i - dd])
            grad[i - dd] -= h[i]
    return grad
```

```python
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    
    v = np.zeros(n)

    stall = 0
    patience = 500

            break
        
        # Compute correlation
        corr = correlate_fft(h, 1.0 - h)
        c = corr * dx
        cur_c5 = c.max()
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive tau
        tau = tau_start * (tau_end / tau_start) ** (it / max_iters)
        
        # Select top K lags
        top_idx = np.argpartition(c, -topK)[-topK:]
        c_top = c[top_idx]
        
        # Softmax weights
        shifted = np.clip(tau * (c_top - c_top.max()), -500, 500)
        p = np.exp(shifted)
        p /= p.sum()
        
        # Compute gradient (weighted sum of per-lag gradients)
        grad = np.zeros(n)
        for i in range(topK):
            if p[i] < 1e-15:
                continue
            d = top_idx[i] - n + 1  # convert from correlate index to lag
            grad += p[i] * grad_for_lag(h, d, n)
        grad *= dx
        
        # Nesterov momentum
        v = beta * v - lr_init * grad / (1 + it * 0.01)
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr_init *= 0.5
            v *= 0.3
            stall = 0
            if lr_init < 1e-8:
                break
        
        h = h_new
    
    return best_h, best_c5
```

```python
import numpy as np
import time

def correlate_fft(a, b):
    n = len(a)
    m = len(b)
    N = 1 << (n + m - 2).bit_length()
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b[::-1], N)
    return np.fft.irfft(fa * fb, N)[:n + m - 1]

def project_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 5.0
    hi = g.max() + 5.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def lag_gradient(h, d, n):
    """Gradient of sum_{valid i} h[i]*(1-h[i+d]) w.r.t. h"""
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            idx = np.arange(m)
            grad[idx] += (1.0 - h[idx + d])
            grad[idx + d] -= h[idx]
    else:
        dd = -d
        m = n - dd
        if m > 0:
            idx = np.arange(dd, n)
            grad[idx] += (1.0 - h[idx - dd])
            grad[idx - dd] -= h[idx]
    return grad

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = correlate_fft(h, 1.0 - h)
    return float(np.max(corr) * dx)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n)
    x_new = np.arange(target_n) * (n / target_n)
    return np.interp(x_new, x_old, h, period=n)

def softmax_optimize(h, n, time_limit, tau_start=30.0, tau_end=200.0, topK=30, lr=0.08, momentum=0.9):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    corr = correlate_fft(h, 1.0 - h)
    c = corr * dx
    best_c5 = float(c.max())
    best_h = h.copy()
    t0 = time.time()
    v = np.zeros(n)
    stall = 0
    patience = 300
    it = 0
    while time.time() - t0 < time_limit:
        it += 1
        # Adaptive tau
        frac = min(1.0, it / 5000.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        
        # Get top K lags
        top_idx = np.argpartition(c, -topK)[-topK:]
        c_top = c[top_idx]
        
        # Softmax
        shifted = np.clip(tau * (c_top - c_top.max()), -300, 300)
        p = np.exp(shifted)
        p /= p.sum()
        
        # Gradient
        grad = np.zeros(n)
        for i in range(topK):
            if p[i] < 1e-10:
                continue
            d = top_idx[i] - n + 1
            grad += p[i] * lag_gradient(h, d, n)
        grad *= dx
        
        # Update
        lr_cur = lr / (1.0 + 0.002 * it)
        v = momentum * v - lr_cur * grad
        h_new = project_feasible(h + v, n)
        
        # Evaluate
        corr = correlate_fft(h_new, 1.0 - h_new)
        c = corr * h_new.shape[0] * dx / h_new.shape[0]  # just corr * dx
        c = corr * dx
        cur = float(c.max())
        if cur < best_c5 - 1e-13:
            best_c5 = cur
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > patience:
            lr *= 0.5
            v *= 0.2
            stall = 0
            if lr < 1e-10:
                break
        
        h = h_new
    
    return best_h, best_c5
```

But the inner problem is: min_h sum_k p_k sum_i h[i](1-h[i+d_k])
= min_h sum_i h[i] (1 - sum_k p_k h[i+d_k])
= min_h sum_i h[i] - sum_k p_k sum_i h[i]h[i+d_k]

The first term is constant (n/2 or 1 in continuous). So we're minimizing -sum_k p_k R(k) = sum_k p_k R(k) is maximized, where R(k) is the weighted autocorrelation.

max_h sum_k p_k R(k) subject to constraints.

This is still non-convex (product of h with shifted h). But for fixed p, the objective is a quadratic form in h:
sum_k p_k sum_i h[i]h[i+d_k] = h^T M(h) p h (where M is some matrix)

Actually it's h^T A h where A is a symmetric matrix with A[i,j] = sum_k p_k I[j-i = d_k or i-j = d_k]. This is a quadratic optimization with box and sum constraints.

We can solve this with:
- Projected gradient (for the quadratic)
- Or CVXPY if we use the SDP relaxation
- Or coordinate descent

For the coordinate descent on the quadratic: fix all h except h[j], optimize h[j]:
h[j] * (sum_{i≠j} A[j,i] h[i]) + h[j]² * A[j,j] + const

The optimal h[j] = clip(-0.5 * sum_{i≠j} A[j,i] h[i] / A[j,j], 0, 1)

Then project to maintain the sum constraint.

This is a cyclic coordinate descent on the quadratic subproblem, alternating with updating p.

This is essentially a block coordinate descent / alternating optimization approach. Let me implement this:

```python
import numpy as np
import time

def correlate_fft(a, b):
    n, m = len(a), len(b)
    N = 1 << (n + m - 2).bit_length()
    return np.fft.irfft(np.fft.rfft(a, N) * np.fft.rfft(b[::-1], N), N)[:n+m-1]

def project_feasible(g, n):
    target = n / 2.0
    lo, hi = g.min() - 5, g.max() + 5
    for _ in range(80):
        mid = (lo + hi) / 2
        if np.sum(np.clip(g - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2, 0, 1)

def lag_grad(h, d, n):
    g = np.zeros(n)
    if d >= 0 and d < n:
        m = n - d
        idx = np.arange(m)
        g[idx] += 1.0 - h[idx + d]
        g[idx + d] -= h[idx]
    elif d < 0 and -d < n:
        dd = -d
        m = n - dd
        idx = np.arange(dd, n)
        g[idx] += 1.0 - h[idx - dd]
        g[idx - dd] -= h[idx]
    return g

def eval_c5(h):
    return float(np.max(correlate_fft(h, 1.0 - h)) * (2.0 / len(h)))

def interp_up(h, target_n):
    n = len(h)
    if n >= target_n:
        return h[:target_n].copy()
    x_old = np.arange(n)
    x_new = np.arange(target_n) * n / target_n
    return np.interp(x_new, x_old, h, period=n)

def main_optimize(h, n, tlimit, tau0=50, tau1=300, topK=30, lr0=0.12, mom=0.91):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    corr = correlate_fft(h, 1 - h)
    c = corr * dx
    best, best_h = float(c.max()), h.copy()
    t0 = time.time()
    v = np.zeros(n)
    lr = lr0
    stall = 0
    it = 0
    while time.time() - t0 < tlimit:
        it += 1
        frac = min(1.0, it / 8000.0)
        tau = tau0 * (tau1 / tau0) ** frac
        tk = np.argpartition(c, -topK)[-topK:]
        ct = c[tk]
        sh = np.clip(tau * (ct - ct.max()), -300, 300)
        p = np.exp(sh); p /= p.sum()
        g = np.zeros(n)
        for i in range(topK):
            if p[i] < 1e-10: continue
            g += p[i] * lag_grad(h, tk[i] - n + 1, n)
        g *= dx
        v = mom * v - (lr / (1 + 0.001*it)) * g
        h = project_feasible(h + v, n)
        corr = correlate_fft(h, 1 - h)
        c = corr * dx
        cur = float(c.max())
        if cur < best - 1e-13:
            best, best_h, stall = cur, h.copy(), 0
        else:
            stall += 1
        if stall > 400:
            lr *= 0.5; v *= 0.2; stall = 0
            if lr < 1e-10: break
    return best_h, best

def subgrad_polish(h, n, tlimit, topK=5, lr0=0.002):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    corr = correlate_fft(h, 1 - h)
    c = corr * dx
    best, best_h = float(c.max()), h.copy()
    t0 = time.time()
    v = np.zeros(n)

        g *= dx / topK
        v = 0.85 * v - (lr / (1 + 0.001*it)) * g
        h = project_feasible(h + v, n)
        corr = correlate_fft(h, 1 - h)
        c = corr * dx
        cur = float(c.max())
        if cur < best - 1e-13:

        else:
            stall += 1
        if stall > 300:

            if lr < 1e-12: break
    return best_h, best

def local_flip_search(h, n, tlimit, rng):
    dx = 2.0 / n
    h = project_feasible(h.copy(), n)
    corr = correlate_fft(h, 1 - h)
    c = corr * dx
    best, best_h = float(c.max()), h.copy()
    t0 = time.time()

        j = rng.randint(n)
        old = h[j]
        # Try setting to 0 or 1 (binary flip)
        for new_v in [0.0, 1.0]:
            if abs(new_v - old) < 1e-6: continue
            h[j] = new_v
            # Adjust sum: change another element
            s = np.sum(h) - n / 2
            if s != 0:
                # Find a partner to adjust
                if s > 0:  # too much sum, need to decrease somewhere
                    # Find the largest element that's > 0
                    candidates = np.where(h > 0.01)[0]
                    if len(candidates) > 0:
                        k = candidates[rng.randint(len(candidates))]
                        adj = min(s, h[k])
                        h[k] -= adj
                        # If we over-corrected, fix
                        if h[k] < 0: h[k] = 0
                else:  # need to increase
                    candidates = np.where(h < 0.99)[0]
                    if len(candidates) > 0:
                        k = candidates[rng.randint(len(candidates))]
                        adj = min(-s, 1.0 - h[k])
                        h[k] += adj
            corr = correlate_fft(h, 1 - h)
            cur = float(np.max(corr) * dx)
            if cur < best - 1e-13:
                best = cur
                best_h = h.copy()
            h[j] = old  # revert
        # Occasionally try intermediate values
        if rng.rand() < 0.3:
            new_v = rng.uniform(0, 1)
            h[j] = new_v
            s = np.sum(h) - n / 2
            if s != 0:
                ...  # same adjustment
            corr = correlate_fft(h, 1 - h)
            cur = float(np.max(corr) * dx)
            if cur < best - 1e-13:
                best = cur
                best_h = h.copy()
            h[j] = old
    
    return best_h, best
```

Actually, the projection changes many coordinates, not just one, so it might undo the local improvement. A better approach for local search:

1. Pick coordinate j.
2. Try new value v.
3. Adjust: distribute the sum error (v - old) equally among all other coordinates (or among a random subset).
4. Clip to [0,1].
5. If sum is still off, do a second pass.
6. Evaluate.

This is O(n) per trial due to the adjustments. For 5000 trials at n=7200, that's 36M operations, about 1-2 seconds. Plus the FFT evaluations. Should be fine.
