import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def find_optimal_t(A, B):
    t_lo, t_hi = 0.0, 1.0
    for _ in range(50):
        t_mid = (t_lo + t_hi) * 0.5
        vals = A * t_mid + B
        mx = np.max(vals)
        tol = 1e-10 * (1.0 + abs(mx))
        active = A[np.abs(vals - mx) <= tol]
        if len(active) > 0:
            amax = np.max(active)
            amin = np.min(active)
            if amax < -1e-12:
                t_lo = t_mid
            elif amin > 1e-12:
                t_hi = t_mid
            else:
                break
        else:
            break
    return (t_lo + t_hi) * 0.5

def coord_descent(h, n_passes=100, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(0)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    lags = np.arange(2 * n - 1) - (n - 1)
    elapsed_inner = 0.0
    for p in range(n_passes):
        improved = False
        order = rng.permutation(n)
        for j in order:
            old = h[j]
            jp = j + lags
            jm = j - lags
            vp = (jp >= 0) & (jp < n)
            vm = (jm >= 0) & (jm < n)
            A = np.zeros(2 * n - 1)
            A[vp] += (1.0 - h[jp[vp]])
            A[vm] -= h[jm[vm]]
            B = corr - A * old
            t_opt = find_optimal_t(A, B)
            if abs(t_opt - old) > 1e-10:
                h[j] = t_opt
                corr += (t_opt - old) * A
                improved = True
        h = make_feasible(h, n)
        corr = np.correlate(h, 1.0 - h, mode='full')
        if not improved:
            break
    return h

def random_binary(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    h = make_feasible(h, n)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def try_block_moves(h, n_moves=20, rng=None):
    n = len(h)
    if rng is None:
        rng = np.random.RandomState(1)
    best = h.copy()
    best_c = compute_c5(best)
    for _ in range(n_moves):
        block_len = rng.randint(4, max(5, n // 10))
        start = rng.randint(0, n - block_len)
        trial = best.copy()
        if rng.rand() < 0.5:
            trial[start:start+block_len] = 1.0 - trial[start:start+block_len]
        else:
            trial[start:start+block_len] = rng.rand(block_len)
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c:
            best = trial
            best_c = c
    return best, best_c
