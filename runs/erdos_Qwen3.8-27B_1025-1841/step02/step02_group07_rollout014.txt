def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0, 1.0)

    """Compute f_k(h) and its gradient."""
    n = len(h)
    # f_k = sum_i h[i] * (1 - h[i+k]) for valid i
    # k can be negative
    if k >= 0:
        idx_i = np.arange(0, n - k)
        idx_j = idx_i + k
    else:
        idx_i = np.arange(-k, n)
        idx_j = idx_i + k
    f_k = np.sum(h[idx_i] * (1.0 - h[idx_j]))
    
    # Gradient
    grad = np.zeros(n)
    # From h[i]*(1-h[i+k]): d/dh[j] = (1-h[j+k]) if j+k valid
    # From h[i+k]*(1-h[i]) where i+k=j: d/dh[j] = -h[j-k] if j-k valid
    if k >= 0:
        # h[j+k] valid when j < n-k
        valid_j = np.arange(0, n - k)
        grad[valid_j] += (1.0 - h[valid_j + k])
        # h[j-k] valid when j >= k
        valid_j2 = np.arange(k, n)
        grad[valid_j2] -= h[valid_j2 - k]

        k2 = -k
        # h[j-k] = h[j+k2] valid when j < n-k2
        valid_j = np.arange(0, n - k2)
        grad[valid_j] += (1.0 - h[valid_j + k2])
        # h[j+k] = h[j-k2] valid when j >= k2
        valid_j2 = np.arange(k2, n)
        grad[valid_j2] -= h[valid_j2 - k2]
    
    return f_k, grad

    lr = lr0

                lr *= 0.5
            elif c5 < best_c5 * 0.99:
                lr *= 1.1
    
    return best_h, best_c5

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0, 1.0)
    return h

def smax_obj_and_grad(h, T, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    weights = exp_vals / np.sum(exp_vals)
    obj = (m + np.log(np.sum(exp_vals)) / T)
    
    # Penalty for sum constraint
    s = np.sum(h)
    target = n / 2.0
    obj += penalty * (s - target) ** 2
    
    # Gradient of soft max part
    grad = np.zeros(n)
    for kk in range(2 * n - 1):
        w = weights[kk]
        if w < 1e-14:
            continue
        k = kk - (n - 1)
        if k >= 0:
            idx1 = np.arange(0, n - k)
            grad[idx1] += w * (1.0 - h[idx1 + k])
            idx2 = np.arange(k, n)

            grad[idx1] += w * (1.0 - h[idx1 - k2])
            idx2 = np.arange(0, n - k2)
            grad[idx2] -= w * h[idx2 + k2]
    grad *= dx
    
    # Gradient of penalty
    grad += 2.0 * penalty * (s - target)
    
    return obj, grad

    lr = 0.05
    momentum = 0.5
    v = np.zeros(n)

        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        k_idx = int(np.argmax(corr))

            g[idx1] += (1.0 - h[idx1 - k2])

        h_new = h - lr * v

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0, 1.0)
    return h

def smax_obj(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    obj = m + np.log(np.sum(exp_vals)) / T
    s = np.sum(x)
    obj += penalty * (s - n / 2.0) ** 2
    return obj

def smax_grad(x, T, penalty):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for kk in range(2 * n - 1):
        w = weights[kk]
        if w < 1e-12:
            continue
        k = kk - (n - 1)
        if k > 0:
            idx1 = np.arange(0, n - k)
            grad[idx1] += w * (1.0 - x[idx1 + k])
            idx2 = np.arange(k, n)
            grad[idx2] -= w * x[idx2 - k]
        elif k == 0:
            grad += w * (1.0 - 2.0 * x)
        else:
            k2 = -k
            idx1 = np.arange(k2, n)
            grad[idx1] += w * (1.0 - x[idx1 - k2])
            idx2 = np.arange(0, n - k2)
            grad[idx2] -= w * x[idx2 + k2]
    grad *= dx
    s = np.sum(x)
    grad += 2.0 * penalty * (s - n / 2.0)
    return grad

def subgrad_h(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    k_idx = int(np.argmax(corr))
    k = k_idx - (n - 1)
    g = np.zeros(n)
    if k > 0:
        idx1 = np.arange(0, n - k)
        g[idx1] = (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        g[idx2] -= h[idx2 - k]
    elif k == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        k2 = -k
        idx1 = np.arange(k2, n)
        g[idx1] = (1.0 - h[idx1 - k2])
        idx2 = np.arange(0, n - k2)
        g[idx2] -= h[idx2 + k2]
    return g * dx

def subgrad_refine(h, n, time_limit, max_iter, lr0, mom0):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = lr0
    mom = mom0
    v = np.zeros(n)
    stag = 0
    for it in range(max_iter):
        if time.time() > time_limit:
            break
        g = subgrad_h(h, n)
        v = mom * v + (1.0 - mom) * g
        h = project(h - lr * v, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stag = 0
        else:
            stag += 1
        if stag > 800:
            lr *= 0.6
            stag = 0
        elif it % 2000 == 1999 and stag < 200:
            lr = min(lr * 1.05, 1.5)
        if it > 8000:
            mom = max(0.0, mom - 0.00005)
    return best_h, best_c5
