def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

        if mask.sum() == 0:
            break

                break
            j_best = candidates_j[rng.randint(len(candidates_j))]
            k_best = candidates_k[rng.randint(len(candidates_k))]
            if j_best == k_best:
                continue
        
        # Determine max delta
        max_delta = min(1.0 - h[j_best], h[k_best])
        
        # Try several deltas
        improved = False
        for frac in [1.0, 0.5, 0.25, 0.1, 0.5, 0.75]:
            delta = max_delta * frac
            if delta < 1e-12:
                continue
            h_trial = h.copy()
            h_trial[j_best] += delta
            h_trial[k_best] -= delta
            corr_trial = compute_corr(h_trial)
            max_trial = corr_trial.max()
            if max_trial < corr.max() - 1e-14:
                h = h_trial
                corr = corr_trial
                if max_trial * dx < best_c5:
                    best_c5 = max_trial * dx
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            # Try a random perturbation to escape
            if rng.random() < 0.3:
                j_r = rng.randint(n)
                k_r = rng.randint(n)
                if j_r != k_r and h[j_r] < 1.0 and h[k_r] > 0.0:
                    delta = min(1.0 - h[j_r], h[k_r]) * rng.uniform(0.1, 0.5)
                    h_trial = h.copy()
                    h_trial[j_r] += delta
                    h_trial[k_r] -= delta
                    corr_trial = compute_corr(h_trial)
                    if corr_trial.max() < corr.max() + 5.0:  # allow some degradation
                        h = h_trial
                        corr = corr_trial
            else:
                # Stuck, try binary projection and restart
                h_bin = (h > 0.5).astype(float)
                h_bin = make_feasible(h_bin, n)
                if compute_c5(h_bin) < best_c5:
                    h = h_bin
                    corr = compute_corr(h)
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
                else:
                    break
    
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0, 2.0, new_n, endpoint=False)
    return make_feasible(np.interp(new_x, old_x, h, period=2.0), new_n)
