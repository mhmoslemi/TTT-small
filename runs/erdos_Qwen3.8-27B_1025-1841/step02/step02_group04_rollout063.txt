import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_val_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    s = e.sum()
    w = e / s
    val = (mc + np.log(s) / T) * dx
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = T * w[idx]
        if wi < 1e-10:
            continue
        sh = idx - (n - 1)
        if sh > 0:
            g[sh:] += wi * (1.0 - h[:n - sh])
            g[:n - sh] -= wi * h[sh:]
        elif sh < 0:
            g[:n + sh] += wi * (1.0 - h[-sh:])
            g[-sh:] -= wi * h[:n + sh]
        else:
            g += wi * (1.0 - 2.0 * h)
    return val, g * dx

def run_lbfgs(h0, T, maxiter=3000, deadline=None):
    n = len(h0)
    brds = [(0.0, 1.0)] * n
    
    def f(x):
        if deadline is not None and time.time() > deadline:
            raise ValueError("timeout")
        return lse_val_grad(x, T)[0]
    
    def fprime(x):
        if deadline is not None and time.time() > deadline:
            raise ValueError("timeout")
        return lse_val_grad(x, T)[1]
    
    try:
        res = minimize(f, h0, jac=fprime, method='L-BFGS-B',
                       bounds=brds, options={'maxiter': maxiter, 'ftol': 1e-15,
                                              'gtol': 1e-12, 'maxls': 50})
        return make_feasible(res.x, n)
    except Exception:
        return h0

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    ox = np.linspace(0, 2, old_n)
    nx = np.linspace(0, 2, new_n)
    h_new = np.interp(nx, ox, h)
    return make_feasible(h_new, new_n)

    best_c5 = float(compute_c5(h))

        return time.time() - t0

        print(f"L-BFGS T=8: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        m = (h < 1.0) if d > 0 else (h > 0.0)
        c = m.sum()
        if c == 0:
            break
        h[m] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def lse_vg(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    e = np.exp(T * (corr - mc))
    s = e.sum()
    w = e / s
    val = (mc + np.log(s) / T) * dx
    g = np.zeros(n)
    for idx in range(2 * n - 1):
        wi = T * w[idx]
        if wi < 1e-10:
            continue
        sh = idx - (n - 1)
        if sh > 0:
            g[sh:] += wi * (1.0 - h[:n - sh])
            g[:n - sh] -= wi * h[sh:]
        elif sh < 0:
            g[:n + sh] += wi * (1.0 - h[-sh:])
            g[-sh:] -= wi * h[:n + sh]
        else:
            g += wi * (1.0 - 2.0 * h)
    return val, g * dx

def do_lbfgs(h0, T, maxiter, deadline):
    n = len(h0)
    brd = [(0.0, 1.0)] * n
    arg = (T,)
    def f(x):
        if time.time() > deadline:
            raise ValueError("t/o")
        return lse_vg(x, T)[0]
    def fp(x):
        if time.time() > deadline:
            raise ValueError("t/o")
        return lse_vg(x, T)[1]
    try:
        r = minimize(f, h0, jac=fp, method='L-BFGS-B', bounds=brd,
                     options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-11})
        return make_feasible(r.x, n)
    except Exception:
        return h0

def upsample(h, nn):
    on = len(h)
    if nn == on:
        return h.copy()
    h2 = np.interp(np.linspace(0, 2, nn), np.linspace(0, 2, on), h)
    return make_feasible(h2, nn)

def _lg(h, T):

    c = np.correlate(h, 1.0 - h, mode='full')
    mc = c.max()
    e = np.exp(T * (c - mc))

For a "tight" distribution where all c_k are close to mc: s ≈ 2n-1, LSE ≈ mc + log(639)/3 ≈ mc + 5.5. The "error" is 5.5 in correlation units = 5.5*0.00625 ≈ 0.034 in C5 units. That's significant!

For a "peaked" distribution where only 1-2 c_k are near mc: s ≈ 1-2, LSE ≈ mc + log(1.5)/3 ≈ mc + 0.14. Error: 0.14*0.00625 ≈ 0.0009. Very small.

So the LSE approximation quality depends on how "spread out" the c_k values are. At T=3, if the correlations are fairly uniform, the LSE is a poor upper bound on the max. At T=200, it's always a good upper bound regardless.

This means: for the optimization to converge to the true C5 minimum, I should use relatively high T (50-200). At low T (3-10), the LSE is a smoother but less accurate objective, which might lead the optimizer in the wrong direction.

BUT, the low-T LSE can help escape local minima because it's smoother (the gradient doesn't focus on just one shift). So the strategy should be:
1. Use low T for exploration (escape local minima)
2. Use high T for exploitation (refine the solution)

This is exactly what my multi-T schedule does. Good.

But I should be aware that the intermediate T solutions might have a "worse" LSE value than the true C5. I should always track the true C5 (using max, not LSE) and only accept improvements in true C5.
