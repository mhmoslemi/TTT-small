from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    """Project h to satisfy sum(h) = n/2 while keeping h in [0,1]."""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_shifts(h, n):
    """Compute f_k for all shifts k."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # This gives the actual integral values

def golden_section_1d(f, a, b, tol=1e-10, max_iter=60):
    """Golden section search for minimum of f on [a,b]."""
    gr = (np.sqrt(5) - 1) / 2  # ~0.618
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(max_iter):
        if b - a < tol:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    return (a + b) / 2

def two_point_update(h, j, m, n, deadline):
    """Optimize h[j] and h[m] keeping their sum fixed."""
    s = h[j] + h[m]

        return h

        h[j] = x
        h[m] = s - x

        return np.max(corr)
    
    # Only do this if we have time

        return h
    
    x_star = golden_section_1d(obj, lo, hi, tol=1e-8, max_iter=50)
    h[j] = x_star
    h[m] = s - x_star
    return h

def cyclic_coordinate_descent(h, n, n_passes, deadline):
    """Run cyclic two-point coordinate descent."""
    h = h.copy()

            if time.time() > deadline:

            m = (j + 1) % n  # Adjacent pair
            h = two_point_update(h, j, m, n, deadline)
