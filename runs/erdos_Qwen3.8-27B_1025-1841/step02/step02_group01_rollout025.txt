def to_binary(h, n):

    h = h.copy()

    target = n // 2

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    target = n // 2
    order = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[order[:target]] = 1.0
    return h_bin

def binary_sa(h_init, start_time, time_limit, rng, max_rounds=1000):
    n = len(h_init)
    h = h_init.copy()
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    T = 0.08
    T_min = 0.0003
    cooling = 0.9995
    n_trials = 300
    for rnd in range(max_rounds):
        if time.time() - start_time > time_limit:
            break
        if T < T_min:
            break
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        for _ in range(n_trials):
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i] = 0.0
            h[j] = 1.0
            new_c5 = compute_c5(h)
            delta = new_c5 - c5
            if delta < -1e-14 or (T > T_min and rng.random() < np.exp(-max(delta, 0.0) / T)):
                c5 = new_c5
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            else:
                h[i] = 1.0
                h[j] = 0.0
        T *= cooling
    return best_h, best_c5

        return soft * dx

        return np.sum(h) - target

    def con_jac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    for restart in range(6):
        if time.time() - start_time > time_limit - 20:
            break
        if restart == 0:
            h_start = h_init.copy()
        elif restart == 1:
            h_start = best_h + rng.randn(n) * 0.02
            h_start = make_feasible(h_start, n)
        elif restart == 2:
            h_start = best_h + rng.randn(n) * 0.05
            h_start = make_feasible(h_start, n)
        else:
            h_start = to_binary(best_h, n) + rng.randn(n) * 0.1
            h_start = make_feasible(h_start, n)
        try:
            res = minimize(soft_obj, h_start, method='SLSQP',

            if res.x is not None:
                xc = make_feasible(res.x, n)
                true_c5 = compute_c5(xc)
                if true_c5 < best_c5:
                    best_c5 = true_c5
                    best_h = xc.copy()
        except Exception:
            pass
    return best_h, best_c5

def coord_descent(h, n, start_time, time_limit):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for outer in range(80):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > time_limit:
                break
            old = best_h[j]
            cands = [0.0, 1.0, 0.5, old + 0.1, old - 0.1, old + 0.04, old - 0.04, old + 0.01, old - 0.01]
            for cv in cands:
                if cv < 0.0 or cv > 1.0:
                    continue
                trial = best_h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    best_h = trial
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
