    n = len(a)

            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                best_c5 = c
                best_h = trial

    # Phase 7: Pairwise moves (swap two active coords)
    active = np.where((best_h > 0.05) & (best_h < 0.95))[0]
    if len(active) > 1 and time_left() > 40:
        for _ in range(3):
            if time_left() < 15:
                break
            idx_i = active[rng.randint(len(active))]
            idx_j = active[rng.randint(len(active))]
            if idx_i == idx_j:
                continue
            trial = best_h.copy()
            trial[idx_i], trial[idx_j] = 1.0 - trial[idx_j], 1.0 - trial[idx_i]
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c5 - 1e-15:
                best_c5 = c
                best_h = trial

    # Phase 8: Final SLSQP polish
    if time_left() > 60:
        n = len(best_h)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eqf, 'jac': eqj, 'args': (n,)}]
        try:
            res = minimize(slq_obj, best_h, method='SLSQP',
                         bounds=bounds, constraints=cons, args=(n, 3000.0),
                         options={'maxiter': 300, 'ftol': 1e-16})
            cand = make_feasible(res.x, n)
            c = compute_c5(cand)
            if c < best_c5:
                best_c5 = c
                best_h = cand
        except Exception:
            pass

    # Phase 9: Final fine coordinate descent
    n = len(best_h)
    active = np.where((best_h > 0.02) & (best_h < 0.98))[0]
    if len(active) == 0:
        active = np.arange(n)
    for rnd in range(60):
        if time_left() < 8:
            break
        improved = False
        for j in active:
            if time_left() < 3:
                break
            old = best_h[j]
            bc = best_c5
            bv = old
            for cv in [0.0, 1.0,
                       max(0.0, old - 0.02), max(0.0, old - 0.005),
                       min(1.0, old + 0.02), min(1.0, old + 0.005)]:
                trial = best_h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-16:
                    bc = c
                    bv = cv
            if bv != old:
                best_h[j] = bv
                best_h = make_feasible(best_h, n)
                best_c5 = bc
                improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
