So minimizing max_k overlap means maximizing the minimum number of "agreeing" 1s across all shifts.

For a binary function, this is related to the autocorrelation of h. overlap_at_k = n/2 - autocorr_k. We want to minimize max_k (n/2 - autocorr_k) * dx = (n/2 - min_k autocorr_k) * dx.

So we want to maximize min_k autocorr_k, i.e., make the autocorrelation as flat as possible (as close to n/2 for all k).

For a binary sequence of length n with n/2 ones, the autocorrelation at lag 0 is n/2. At other lags, we want it to be as close to n/2 as possible.

This is related to the theory of balanced sequences / difference sets! A (v, k, λ) difference set would give constant autocorrelation. For v=n, k=n/2, we'd need a difference set with constant λ.

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = T * (corr - m)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / T

def soft_max_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    m = np.max(corr)
    shifted = T * (corr - m)
    shifted = np.clip(shifted, -500, 500)
    p = np.exp(shifted)
    p /= np.sum(p)
    
    grad = np.zeros(n)
    for k in range(2*n-1):
        wk = T * p[k]
        if wk < 1e-14:
            continue
        off = k - (n-1)
        # First term
        j_lo = max(0, -off)
        j_hi = min(n-1, n-1-off)
        if j_lo <= j_hi:
            js = np.arange(j_lo, j_hi+1)
            grad[js] += wk * (1.0 - x[js + off])
        # Second term
        j_lo2 = max(0, off)
        j_hi2 = min(n-1, n-1+off)
        if j_lo2 <= j_hi2:
            js2 = np.arange(j_lo2, j_hi2+1)
            grad[js2] -= wk * x[js2 - off]
    
    return grad

    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):

        else:

        cnt = np.sum(mask)

            break

        h = np.clip(h, 0.0, 1.0)

    target = n / 2.0
    
    def obj(x):
        return soft_max_obj(x, n, T)
    
    def obj_grad(x):
        return soft_max_grad(x, n, T)
    
    def cons_eq(x):
        return np.sum(x) - target

        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       options={'ftol': 1e-12, 'maxiter': maxiter})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

            f_old = soft_max_obj(h, n, T)
            if f_new < f_old:
                h = h_trial
                break

        else:

    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
