import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)
    return h

def grad_at_shift(h, shift):
    n = len(h)
    grad = np.zeros(n)
    s = abs(shift)
    if s >= n:
        return grad
    if s == 0:
        grad[:] = 1.0 - 2.0 * h
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += -h[:n - s]
    return grad

def compute_soft_grad(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = np.max(corr)
    shifted = corr - max_c
    max_shifted = np.max(shifted)
    if max_shifted > 0:
        T = min(T, 400.0 / max_shifted)
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        return grad_at_shift(h, shift) * dx
    weights = exp_s / total
    if top_k is not None and top_k < len(weights):
        idx_sorted = np.argsort(weights)[-top_k:]
        mask = np.zeros(len(weights), dtype=bool)
        mask[idx_sorted] = True
        weights *= mask
        s = np.sum(weights)
        if s > 0:
            weights /= s
    grad = np.zeros(n)
    nz = np.nonzero(weights > 1e-10)[0]
    for k_idx in nz:
        shift = int(k_idx) - (n - 1)
        grad += weights[k_idx] * grad_at_shift(h, shift)
    return grad * dx

def gd_linesearch(h, n_iter, T_start, T_end, top_k=30, time_limit=600.0):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    alpha = 0.8
    t0 = time.time()
    stuck = 0
    for it in range(n_iter):
        if time.time() - t0 > time_limit:
            break
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        grad = compute_soft_grad(h, T, top_k=top_k)
        a = alpha
        improved = False
        for _ in range(25):
            h_trial = make_feasible(h - a * grad, n)
            c5_trial = compute_c5(h_trial)
            if c5_trial < best_c5 - 1e-11:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial
                alpha = min(a * 1.3, 3.0)
                stuck = 0
                improved = True
                break
            a *= 0.45
            if a < 1e-10:
                break
        if not improved:
            stuck += 1
            alpha = max(alpha * 0.6, 0.005)
            if stuck > 500:
                break
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
