   - Actually h[i]*(1-h[i+k*]) is linear in h[i]! Not quadratic!
   
Wait: h[i] * (1 - h[i+k]). If h[i+k] is fixed, this is h[i] * (1 - h[i+k]) which is LINEAR in h[i]. The minimum over h[i] ∈ [0,1] is at h[i] = 0 (giving 0) or h[i] = 1 (giving 1-h[i+k]).

So to minimize the contribution to a specific lag, set h[i] = 0. But that's too aggressive because it affects other lags.

This explains why the problem is hard - it's a max of linear functions, which is a convex optimization problem (minimize a convex function). The soft-max is a smooth approximation.

OK so the problem is: minimize max_k f_k(h) where f_k(h) = ∑_i h[i]*(1-h[i+k])*dx

Each f_k is linear in h (bilinear actually - h[i]*(1-h[j]) is bilinear). So the overall objective is a maximum of bilinear functions, which is NOT convex. That's why it's hard.

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_full(h_half):
    n = len(h_half)
    h_full = np.empty(2 * n)
    h_full[:n] = h_half
    h_full[n:] = 1.0 - h_half[::-1]
    return h_full

def soft_max_obj(h_half, T):
    full = make_full(h_half)
    n = len(full)
    dx = 2.0 / n
    corr = np.correlate(full, 1.0 - full, mode='full') * dx
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T)

def clip01(h):
    return np.clip(h, 0.0, 1.0)

def lbb_optimize(h_half, T, maxiter, deadline):
    n = len(h_half)
    def obj(x):
        return soft_max_obj(x, T)
    try:
        res = minimize(obj, h_half, method='L-BFGS-B',
                      bounds=[(0.0, 1.0)] * n,
                      options={'maxiter': maxiter, 'ftol': 1e-15, 'gtol': 1e-10, 'maxls': 40})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return clip01(res.x)
    except Exception:
        pass
    return h_half

def coordinate_descent(h_half, deadline):
    n = len(h_half)
    best_c5 = evaluate_c5(make_full(h_half))
    steps_coarse = [0.03, 0.015, 0.007]
    steps_fine = [0.003, 0.001]
    all_steps = steps_coarse + steps_fine
    for step in all_steps:
        improved = True
        while improved:
            improved = False
            for i in range(n):
                if time.time() > deadline:
                    return h_half, best_c5
                for sign in (1.0, -1.0):
                    new_val = h_half[i] + sign * step
                    if new_val < 0.0 or new_val > 1.0:
                        continue
                    old = h_half[i]
                    h_half[i] = new_val
                    c5 = evaluate_c5(make_full(h_half))
                    if c5 < best_c5:
                        best_c5 = c5
                        improved = True
                    else:
                        h_half[i] = old
    return h_half, best_c5

def interp_half(h_half, n_to):
    if len(h_half) == n_to:
        return h_half.copy()
    xf = np.linspace(0, 1, len(h_half), endpoint=False)
    xt = np.linspace(0, 1, n_to, endpoint=False)
    return clip01(np.interp(xt, xf, h_half))

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
        n_full_init = len(h_init)
        n_half_init = n_full_init // 2
    else:
        n_half_init = 168
        h_init = np.full(2 * n_half_init, 0.5)

    h_half = h_init[:n_half_init].copy()
    h_half = clip01(h_half)

    best_h_half = h_half.copy()
    best_c5 = evaluate_c5(make_full(best_h_half))

    # Phase 1: Symmetric optimization with annealing T schedule
    T_schedule = [30.0, 80.0, 200.0, 500.0, 1200.0]
    for T in T_schedule:
        if time.time() > deadline - 100:
            break
        iters = 600 if T < 100 else 400
        h_new = lbb_optimize(best_h_half, T, iters, deadline)
        c5_new = evaluate_c5(make_full(h_new))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h_half = h_new.copy()

    # Phase 2: Coordinate descent
    if time.time() < deadline - 30:
        best_h_half, best_c5 = coordinate_descent(best_h_half, deadline - 15)

    # Phase 3: Try higher resolution
    if time.time() < deadline - 80:
        n_half_hi = 210
        h_hi = interp_half(best_h_half, n_half_hi)
        for T in [100.0, 300.0]:
            if time.time() > deadline - 40:
                break
            h_hi = lbb_optimize(h_hi, T, 500, deadline)
        c5_hi = evaluate_c5(make_full(h_hi))
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h_half = h_hi.copy()
        if time.time() < deadline - 30:
            best_h_half, best_c5 = coordinate_descent(best_h_half, deadline - 15)

    # Phase 4: Random restarts in symmetric space
    n_best = len(best_h_half)
    for attempt in range(25):
        if time.time() > deadline - 25:
            break
        scale = 0.04 + 0.02 * (attempt % 3)
        h_pert = best_h_half + np.random.randn(n_best) * scale
        h_pert = clip01(h_pert)
        T = 200.0 + 100.0 * (attempt % 3)
        h_new = lbb_optimize(h_pert, T, 300, deadline)
        c5_new = evaluate_c5(make_full(h_new))
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h_half = h_new.copy()

    # Phase 5: Try asymmetric optimization from best symmetric point
    if time.time() < deadline - 60:
        n_full = len(make_full(best_h_half))
        h_full = make_full(best_h_half)
        target = n_full / 2.0

        def obj_full(x):
            corr = np.correlate(x, 1.0 - x, mode='full')
            dx = 2.0 / n_full
            c = corr * dx
            T = 300.0
            sc = c * T
            m = np.max(sc)
            return float((m + np.log(np.sum(np.exp(sc - m)))) / T)

        def con_full(x):
            return np.sum(x) - target

        try:
            res = minimize(obj_full, h_full, method='SLSQP',
                          bounds=[(0.0, 1.0)] * n_full,
                          constraints=[{'type': 'eq', 'fun': con_full}],
                          options={'maxiter': 300, 'ftol': 1e-14})
            if res.x is not None and np.all(np.isfinite(res.x)):
                hf = clip01(res.x)
                # Fix integral constraint
                s = np.sum(hf)
                diff = target - s
                mask = (hf > 0.01) & (hf < 0.99)
                if np.sum(mask) > 0:
                    hf[mask] += diff / np.sum(mask)
                    hf = clip01(hf)
                c5_asym = evaluate_c5(hf)
                if c5_asym < best_c5:
                    best_c5 = c5_asym
                    best_h_half = hf[:n_full // 2].copy()
        except Exception:
            pass

    # Final coordinate descent
    if time.time() < deadline - 5:
        best_h_half, best_c5 = coordinate_descent(best_h_half, deadline - 2)

    final_h = make_full(best_h_half)
    final_c5 = evaluate_c5(final_h)
    return (final_h, final_c5, len(final_h))
