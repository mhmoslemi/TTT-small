    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        corrs = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corrs)
        s = k_star - n + 1
        grad = np.zeros(n)
        i0 = max(0, -s)
        i1 = min(n - 1, n - 1 - s)
        if i0 <= i1:
            idx = np.arange(i0, i1 + 1)
            grad[idx] += (1.0 - h[idx + s])
            grad[idx + s] -= h[idx]
        grad *= dx_f
        v = 0.8 * v - lr * grad
        h = project_to_feasible(h + v, n)

        if it % 300 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-15:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 15:
                lr *= 0.5
                v *= 0.3
                stall = 0
                if lr < 1e-14:
                    break

    c5_cur = compute_c5(h)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h.copy()
    return best_h, best_c5


def binary_local_search(h, max_iters, deadline):
    n = len(h)
    h_bin = (h >= 0.5).astype(float)
    target = n // 2
    ones = int(h_bin.sum())
    if ones > target:
        idx1 = np.where(h_bin == 1)[0]
        order = np.argsort(h[idx1])
        for i in range(ones - target):
            h_bin[idx1[order[i]]] = 0
    elif ones < target:
        idx0 = np.where(h_bin == 0)[0]
        order = np.argsort(-h[idx0])
        for i in range(target - ones):
            h_bin[idx0[order[i]]] = 1

    best_c5 = compute_c5(h_bin)
    best_h = h_bin.copy()
    rng = np.random.RandomState(123)
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j or h_bin[i] == h_bin[j]:
            continue
        old_i, old_j = h_bin[i], h_bin[j]
        h_bin[i], h_bin[j] = old_j, old_i
        c5_try = compute_c5(h_bin)
        if c5_try < best_c5 - 1e-15:
            best_c5 = c5_try
            best_h = h_bin.copy()
            no_improve = 0
        else:
            h_bin[i], h_bin[j] = old_i, old_j
            no_improve += 1
        if no_improve > 50000:
            break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    end_time = start_time + budget_s - 5.0
    np.random.seed(seed)

    h_init = initial_h_values.copy()
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    def remaining():
        return max(0.0, end_time - time.time())

    # Phase 1: n=2500, multi-tau continuation
    n1 = 2500
    h1 = upsample(best_h, n1)
    h1 = project_to_feasible(h1, n1)

    phases = [(5, 0.12), (15, 0.13), (40, 0.13), (100, 0.12)]
    for tau, frac in phases:
        if remaining() < 50:
            break
        lr = n1 / (4.0 * (1.0 + tau))
        deadline = time.time() + remaining() * frac
        h1, c5 = optimize_momentum(h1, tau, max_iters=500000, deadline=deadline, lr=lr, beta=0.9)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h1.copy()

    # Phase 2: n=5000 refinement
    n2 = 5000
    if remaining() > 200:
        h2 = upsample(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        for tau, frac in [(30, 0.15), (100, 0.12)]:
            if remaining() < 50:
                break
            lr = n2 / (5.0 * (1.0 + tau))
            deadline = time.time() + remaining() * frac
            h2, c5 = optimize_momentum(h2, tau, max_iters=300000, deadline=deadline, lr=lr, beta=0.9)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2.copy()

    # Phase 3: Alternative starting point (block pattern) at n=2500
    if remaining() > 150:
        n3 = 2500
        h3 = np.zeros(n3)
        # 6-block pattern: 1,0,1,0,1,0 with ratio ~5:4
        seg1 = n3 * 5 // 9
        seg0 = n3 * 4 // 9
        pos = 0
        for i in range(6):
            if i % 2 == 0:
                length = seg1
            else:
                length = seg0
            if i % 2 == 0:
                h3[pos:pos + length] = 1.0
            pos += length
        # Fix the last segment to fill exactly
        if pos < n3:
            h3[pos:] = 1.0 if (5 % 2 == 0) else 0.0
        h3 = project_to_feasible(h3, n3)
        for tau, frac in [(20, 0.15), (60, 0.12)]:
            if remaining() < 40:
                break
            lr = n3 / (5.0 * (1.0 + tau))
            deadline = time.time() + remaining() * frac
            h3, c5 = optimize_momentum(h3, tau, max_iters=300000, deadline=deadline, lr=lr, beta=0.85)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3.copy()

    # Phase 4: Binary local search on best solution
    if remaining() > 80:
        h_bs, c5_bs = binary_local_search(best_h, max_iters=800000,
                                          deadline=time.time() + remaining() * 0.35)
        if c5_bs < best_c5:
            best_c5 = c5_bs
            best_h = h_bs.copy()

    # Phase 5: Subgradient polish
    if remaining() > 30:
        h_f, c5_f = subgrad_polish(best_h, max_iters=300000,
                                   deadline=time.time() + remaining() * 0.5, lr=0.002)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: Final high-res attempt
    if remaining() > 100:
        n_final = 4000
        h_fin = upsample(best_h, n_final)
        h_fin = project_to_feasible(h_fin, n_final)
        lr = n_final / (5.0 * 50.0)
        deadline = time.time() + remaining() * 0.7
        h_fin, c5_fin = optimize_momentum(h_fin, 50, max_iters=200000,
                                          deadline=deadline, lr=lr, beta=0.9)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Ensure feasibility and final evaluation
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
