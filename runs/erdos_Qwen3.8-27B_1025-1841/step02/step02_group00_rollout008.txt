```python
def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    
    # Compute correlation
    corr = np.correlate(h, g, 'full')  # length 2n-1
    
    # Softmax weights
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    w = exp_c / exp_c.sum()
    
    # Gradient
    P1 = np.convolve(w, g, 'full')[n-1 : n-1+n]
    P2 = np.convolve(h, w[::-1], 'full')[n-1 : n-1+n]
    grad = P1 - h * w[n-1] - P2
    grad *= dx  # chain rule for the dx factor
    
    return grad, float(np.max(corr) * dx)
```

Wait, I need to double-check the dx factor. The objective is max(corr) * dx. The soft approximation is (max_c + log_sum/T) * dx. The gradient of this w.r.t. h is dx * (gradient of soft_max of corr).

The soft_max gradient: d/dh [log(Σ exp(T*corr))/T] = Σ w_k * d(corr[k])/d(h) / 1... 

Actually: f = (1/T) * log(Σ_k exp(T * corr_k))

= Σ_k w_k * d(corr_k)/d(h_j)

where w_k = exp(T*corr_k) / Σ exp(T*corr_k)

So the gradient of f w.r.t. h is: Σ_k w_k * d(corr_k)/d(h_j) = grad (as computed above)

And the gradient of the objective f*dx is: dx * grad

Great. Now let me also handle the constraint sum(h) = n/2. I'll use Lagrange multiplier or projection.

**Projection onto {0 ≤ h ≤ 1, sum(h) = n/2}:**
1. Clip to [0,1]
2. If sum > n/2, reduce from largest values
3. If sum < n/2, increase from smallest values
4. Iterate until converged

Actually the standard water-filling approach:
- Find λ such that sum(clip(h - λ, 0, 1)) = n/2
- Projected h = clip(h - λ, 0, 1)

This can be done by bisection on λ.

import numpy as np
from scipy.fft import fft, ifft
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_onto_constraints(v, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(v - lam, 0.0, 1.0)

def compute_softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    Z = np.sum(exp_c)
    w = exp_c / Z
    P1 = np.convolve(w, g, 'full')[n-1 : 2*n-1]
    P2 = np.convolve(h, w[::-1], 'full')[n-1 : 2*n-1]
    grad = (P1 - h * w[n-1] - P2) * dx
    obj = (max_c + np.log(Z) / T) * dx
    return grad, obj, float(np.max(corr) * dx)

def pgd_optimize(h_start, n, T_init, T_final, n_iter, momentum_init=0.9):
    h = h_start.copy()
    v = np.zeros(n)
    mu = momentum_init
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 1.0 / T_init
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_init * (T_final / T_init) ** frac
        mu_curr = momentum_init * (1.0 - 0.3 * frac)
        lr_curr = 1.0 / max(T, 1.0)
        
        grad, soft_obj, c5_raw = compute_softmax_grad(h, T)
        
        v = mu_curr * v - lr_curr * grad
        h_new = project_onto_constraints(h + v, n)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_c5

def binary_polish(h, n, rng, n_rounds=5):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for rnd in range(n_rounds):
        improved = False
        for j in range(n):
            for bv in (0.0, 1.0):
                if abs(h[j] - bv) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = bv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-10:
                    best_c5 = c
                    best_h = trial.copy()
                    h = trial.copy()
                    improved = True
            if improved:
                break
        if not improved:
            break
    return best_h, best_c5

def block_moves(h, n, rng, n_trials=200, block_sizes=(5, 10, 20)):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for _ in range(n_trials):
        bs = rng.choice(block_sizes)
        start = rng.randint(0, max(n - bs, 1))
        trial = h.copy()
        if rng.rand() < 0.5:
            trial[start:start+bs] = 1.0
        else:
            trial[start:start+bs] = 0.0
        trial = make_feasible(trial, n)
        c = compute_c5(trial)
        if c < best_c5 - 1e-10:
            best_c5 = c
            best_h = trial.copy()
    return best_h, best_c5

        if c5_opt < best_c5:
