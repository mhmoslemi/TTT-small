def project_to_feasible(g, n):
    """Project g to {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

    m = len(b)
    size = n + m - 1
    # Pad to next power of 2 for efficiency

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def compute_grad_topM(h, p_top, top_idx, n, dx):
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_top[i]
        if pk < 1e-15:
            continue
        k = int(top_idx[i])
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx


def optimize_h(h0, n, tau, max_iters, time_limit, lr_init=0.05, topM=200, use_adam=True):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0
    lr = lr_init
    stall = 0
    patience = 2000

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_c5 - 1e-14:
            best_c5 = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        eff_topM = topM if topM > 0 else (2 * n - 1)
        eff_topM = min(eff_topM, 2 * n - 1)
        if eff_topM < (2 * n - 1):
            top_idx = np.argpartition(g, -eff_topM)[-eff_topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            if s < 1e-30:
                p_top = np.ones(len(top_idx)) / len(top_idx)
            else:
                p_top = exp_c / s
            grad = compute_grad_topM(h, p_top, top_idx, n, dx)
        else:
            max_g = g.max()
            shifted = np.clip(tau * (g - max_g), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            if s < 1e-30:
                p = np.ones(len(g)) / len(g)
            else:
                p = exp_c / s
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                pk = p[k]
                if pk < 1e-15:
                    continue
                d = k - n + 1
                j0 = max(0, d)
                j1 = min(n, n + d)
                if j0 < j1:
                    idx = np.arange(j0, j1)
                    grad[idx] += pk * (1.0 - h[idx - d])
                j0b = max(0, -d)
                j1b = min(n, n - d)
                if j0b < j1b:
                    idx = np.arange(j0b, j1b)
                    grad[idx] -= pk * h[idx + d]
            grad *= dx

        if use_adam:
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        else:
            step = lr * grad

        h_new = project_to_feasible(h - step, n)

        if stall > patience:
            lr *= 0.8
            if use_adam:
                m *= 0.5
                v *= 0.5
            stall = 0
            if lr < 1e-12:
                break

        if it % 2000 == 1999:
            lr *= 0.97

        h = h_new

    return best_h, best_c5


def make_cos2(n):
    x = np.arange(n) * (2.0 / n)
    h = np.cos(np.pi * x) ** 2
    return project_to_feasible(h, n)


def make_parametric(n, a=1.0, b=0.0, phase=0.0):
    x = np.arange(n) * (2.0 / n)
    h = 0.5 * (1.0 + a * np.cos(2.0 * np.pi * x + phase) + b * np.cos(4.0 * np.pi * x))
    return project_to_feasible(np.clip(h, 0.0, 1.0), n)


def interp_up(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h)


def subgrad_polish(h0, n, time_limit, lr_init=0.004, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    count = 0

    for it in range(100000):
        if time.time() - start > time_limit:
            break
        count += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = int(ki) - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK

        cur_lr = lr / (1.0 + it * 0.008)
        h_new = project_to_feasible(h - cur_lr * grad, n)
        h = h_new

        if it % 500 == 499:
            lr *= 0.95

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def _tl():
        return max(0.0, budget_s - (time.time() - start_time))

    n_base = 3600
    best_c5 = float('inf')
    best_h = None

    # Starting point A: cos^2(pi*x)
    h_a = make_cos2(n_base)
    c5_a = compute_c5(h_a)
    best_c5 = c5_a
    best_h = h_a.copy()

    # Starting point B: initial_h_values
    h_b = initial_h_values.copy()
    c5_b = compute_c5(h_b)
    if c5_b < best_c5:
        best_c5 = c5_b
        best_h = h_b.copy()

    # Starting point C: parametric with 4th harmonic
    h_c = make_parametric(n_base, a=0.95, b=0.1, phase=0.0)
    c5_c = compute_c5(h_c)
    if c5_c < best_c5:
        best_c5 = c5_c
        best_h = h_c.copy()

    # Starting point D: another parametric
    h_d = make_parametric(n_base, a=1.0, b=-0.05, phase=0.5)
    c5_d = compute_c5(h_d)
    if c5_d < best_c5:
        best_c5 = c5_d
        best_h = h_d.copy()

    # Phase 1: Optimize best starting point at base resolution
    if _tl() > 120:
        for tau in [8.0, 20.0, 50.0]:
            if _tl() < 50:
                break
            tl = min(_tl() * 0.18, 150.0)
            h_opt, c5_opt = optimize_h(best_h, n_base, tau,
                                       max_iters=20000, time_limit=tl,
                                       lr_init=0.04, topM=500, use_adam=True)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 2: Also optimize from cos2 specifically
    if _tl() > 80:
        h_cos = make_cos2(n_base)
        c5_cos = compute_c5(h_cos)
        if c5_cos + 0.02 > best_c5:
            for tau in [15.0, 40.0]:
                if _tl() < 40:
                    break
                tl = min(_tl() * 0.15, 120.0)
                h_opt, c5_opt = optimize_h(h_cos, n_base, tau,
                                           max_iters=15000, time_limit=tl,
                                           lr_init=0.03, topM=400, use_adam=True)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()

    # Phase 3: Higher resolution
    n2 = n_base * 2
    if _tl() > 100:
        h_up = interp_up(best_h, n2)
        h_up = project_to_feasible(h_up, n2)
        for tau in [15.0, 40.0]:
            if _tl() < 50:
                break
            tl = min(_tl() * 0.22, 200.0)
            h_opt, c5_opt = optimize_h(h_up, n2, tau,
                                       max_iters=8000, time_limit=tl,
                                       lr_init=0.025, topM=300, use_adam=True)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 4: Subgradient polish at best resolution
    if _tl() > 50:
        n_f = len(best_h)
        tl = min(_tl() * 0.25, 200.0)
        h_f, c5_f = subgrad_polish(best_h, n_f, time_limit=tl, lr_init=0.003, topK=5)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Randomized restarts
    if _tl() > 60:
        for attempt in range(4):
            if _tl() < 25:
                break
            n_p = n_base
            h_cur = best_h
            if len(h_cur) != n_p:
                h_cur = interp_up(h_cur, n_p)
                h_cur = project_to_feasible(h_cur, n_p)
            noise_scale = 0.015 * (attempt + 1)
            h_p = h_cur + noise_scale * np.random.randn(n_p)
            h_p = project_to_feasible(h_p, n_p)
            tl = min(_tl() * 0.18, 80.0)
            h_opt, c5_opt = optimize_h(h_p, n_p, 25.0,
                                       max_iters=8000, time_limit=tl,
                                       lr_init=0.03, topM=300, use_adam=True)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 6: Final polish
    if _tl() > 20:
        n_final = len(best_h)
        tl = min(_tl() * 0.4, 150.0)
        h_fin, c5_fin = subgrad_polish(best_h, n_final, time_limit=tl, lr_init=0.001, topK=3)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
