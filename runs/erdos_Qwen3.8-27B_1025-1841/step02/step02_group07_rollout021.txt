import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def autocorr(h):
    """Compute full autocorrelation of h"""
    n = len(h)
    f = np.fft.rfft(h, n=4*n)
    ac = np.fft.irfft(f * np.conj(f))[:2*n]
    return ac  # ac[n-1+k] = sum_i h[i]*h[i+k] for lag k

def compute_c_values(h):
    """Compute all c[k] values (before multiplying by dx)"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr

def gradient_c_k(h, k):
    """Compute gradient of c[k] with respect to h
    k is the index into the full correlation (0 to 2n-2)
    Lag = k - (n-1)
    """
    n = len(h)
    lag = k - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        # c[lag] = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
        # d/dh[j]: 
        #   if 0<=j<=n-1-lag: (1-h[j+lag])
        #   if lag<=j<=n-1: -h[j-lag]
        valid1 = slice(0, n - lag)
        valid2 = slice(lag, n)
        g[valid1] += (1.0 - h[lag:n])  # h[j+lag] for j in 0..n-1-lag
        g[valid2] -= h[0:n-lag]  # h[j-lag] for j in lag..n-1
    else:
        # lag < 0, let k' = -lag
        kp = -lag
        # c[-kp] = sum_{i=kp}^{n-1} h[i]*(1-h[i-kp])
        # d/dh[j]:
        #   if kp<=j<=n-1: (1-h[j-kp])
        #   if 0<=j<=n-1-kp: -h[j+kp]
        valid1 = slice(kp, n)
        valid2 = slice(0, n - kp)
        g[valid1] += (1.0 - h[0:n-kp])  # h[j-kp] for j in kp..n-1
        g[valid2] -= h[kp:n]  # h[j+kp] for j in 0..n-1-kp
    return g

def soft_max(c, T):
    """Soft approximation of max(c)"""
    m = np.max(c)
    return m + np.log(np.sum(np.exp(c - m))) / T

def soft_max_grad(c, T):
    """Gradient of soft_max with respect to c"""
    m = np.max(c)
    exp_c = np.exp(c - m)
    weights = exp_c / np.sum(exp_c) / T
    return weights

def subgradient_opt(h0, n, max_iters, deadline, alpha=0.05, beta=1.0):
    """Subgradient method on the exact max objective (minimizing)"""
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_c_values(h) * dx
        k_star = np.argmax(c)
        c5 = c[k_star]
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient of c[k_star] * dx
        g = gradient_c_k(h, k_star) * dx
        
        # Step size: alpha / (beta + t)
        step = alpha / (beta + t ** 0.5)
        
        h_new = h - step * g
        h = project(h_new, n)
        
        # Occasionally try larger steps or perturbations
        if t % 50 == 0 and t > 0:
            h_pert = h + np.random.randn(n) * 0.02
            h_pert = project(h_pert, n)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert
    
    return best_h, best_c5

def smooth_opt(h0, n, T, max_iters, deadline, alpha=0.01):
    """Gradient descent on smoothed objective"""
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for t in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_c_values(h)
        sm = soft_max(c, T) * dx
        weights = soft_max_grad(c, T)
        
        # Gradient: sum of weights * gradient of c[k]
        g = np.zeros(n)
        for k in range(len(c)):
            if weights[k] > 1e-6:
                g += weights[k] * gradient_c_k(h, k)
        g *= dx
        
        # Adaptive step size
        step = alpha / (1.0 + t * 0.01)
        h_new = h - step * g
        h = project(h_new, n)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    # Bisection on lambda for projection onto {0<=h<=1, sum=n/2}
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def soft_max(c, T):
    m = np.max(c)
    return m + np.log(np.sum(np.exp(c - m))) / T

def compute_c(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_soft_max(h, n, T):
    """Gradient of soft_max(c, T) * (2/n)"""
    dx = 2.0 / n
    c = compute_c(h)
    m = np.max(c)
    exp_c = np.exp((c - m) * T)
    p = exp_c / np.sum(exp_c)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            ns = n - lag
            if ns > 0:
                g[:ns] += p[k] * (1.0 - h[lag:])
                g[lag:] -= p[k] * h[:ns]
        else:
            lp = -lag
            ns = n - lp
            if ns > 0:
                g[lp:] += p[k] * (1.0 - h[:ns])
                g[:ns] -= p[k] * h[lp:]
    return g * dx

def interp_to(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xf = np.linspace(0, 2, n_src, endpoint=False)
    xt = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xt, xf, h_src)

def fista_smooth(h0, n, T_start, T_end, max_iters, deadline, lr=None):
    """FISTA with temperature annealing"""
    h = h0.copy()
    h_prev = h0.copy()
    t = 1
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    if lr is None:
        lr = 0.1 / T_start
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Anneal temperature
        T = T_start + (T_end - T_start) * (it / max_iters)
        current_lr = 0.1 / max(T, 1.0)
        
        # Nesterov momentum
        if t > 1:
            mu = (t - 1) / (t + 2.0)
            y = h + mu * (h - h_prev)
        else:
            y = h
        
        # Evaluate y to ensure it's in [0,1]
        y = np.clip(y, 0.0, 1.0)
        
        # Compute gradient at y
        c = compute_c(y)
        m_c = np.max(c)
        exp_c = np.exp((c - m_c) * T)
        p = exp_c / np.sum(exp_c)
        
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                ns = n - lag
                if ns > 0:
                    g[:ns] += p[k] * (1.0 - y[lag:])
                    g[lag:] -= p[k] * y[:ns]
            else:
                lp = -lag
                ns = n - lp
                if ns > 0:
                    g[lp:] += p[k] * (1.0 - y[:ns])
                    g[:ns] -= p[k] * y[lp:]
        g *= 2.0 / n
        
        h_prev = h
        h = project(y - current_lr * g, n)
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        t += 1
    
    return best_h, best_c5

def subgradient_exact(h0, n, max_iters, deadline, alpha=0.02):
    """Exact subgradient minimization"""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    
    for t in range(max_iters):
        if time.time() > deadline:
            break
        
        c = compute_c(h)
        k_star = np.argmax(c)
        c5 = c[k_star] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient of c[k_star]
        lag = k_star - (n - 1)
        g = np.zeros(n)
        if lag >= 0:
            ns = n - lag
            if ns > 0:
                g[:ns] += (1.0 - h[lag:])
                g[lag:] -= h[:ns]
        else:
            lp = -lag
            ns = n - lp
            if ns > 0:
                g[lp:] += (1.0 - h[:ns])
                g[:ns] -= h[lp:]
        
        # Step size
        step = alpha / (1.0 + 0.1 * np.sqrt(t))
        h = project(h - step * g * dx, n)
        
        # Random perturbation every 100 steps
        if t > 0 and t % 100 == 0:
            h_pert = h + np.random.randn(n) * 0.03
            h_pert = project(h_pert, n)
            c5_pert = evaluate_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h = h_pert
    
    return best_h, best_c5

def slsqp_refine(h0, n, T, deadline):
    """SLSQP refinement with smooth objective"""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(c - m)))) / T * dx
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target}],
                       options={'ftol': 1e-15, 'maxiter': 200})
        if time.time() > deadline:
            return h0, evaluate_c5(h0)
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h0, evaluate_c5(h0)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 20
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)
    
    n_start = len(h_start)

    best_c5 = evaluate_c5(best_h)
    
    # Phase 1: FISTA with smooth objective at initial resolution
    n = n_start
    h_work = best_h.copy()
    max_iters_p1 = 500
    
    T_list = [5.0, 20.0, 100.0, 500.0, 2000.0]
    for T in T_list:
        if time.time() > deadline - 100:
            break
        h_work, c5 = fista_smooth(h_work, n, T, T * 2, max_iters_p1, deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
    
    # Phase 2: Exact subgradient
    if time.time() < deadline - 60:
        h_sg, c5_sg = subgradient_exact(best_h, n, max_iters=2000, deadline=deadline - 60)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Phase 3: SLSQP refinement
    if time.time() < deadline - 30:
        for T in [50.0, 200.0]:
            if time.time() > deadline - 20:

            h_sl, c5_sl = slsqp_refine(best_h, n, T, deadline - 15)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

    for n_high in [672, 504]:
        if time.time() > deadline - 30:
            break
        h_high = interp_to(best_h, len(best_h), n_high)
        h_high = project(h_high, n_high)
        
        for T in [20.0, 100.0]:
            if time.time() > deadline - 25:

            h_high, c5_high = fista_smooth(h_high, n_high, T, T * 3, 200, deadline - 15)
            # Project back to original resolution
            h_back = interp_to(h_high, n_high, len(best_h))
            h_back = project(h_back, len(best_h))
            c5_back = evaluate_c5(h_back)
            if c5_back < best_c5:
                best_c5 = c5_back
                best_h = h_back.copy()
            # Also check the high-res solution directly
            if c5_high < best_c5:

            best_h = h_final.copy()

    return (best_h, best_c5, len(best_h))

∂c[lag≥0]/∂h[j] = (1-h[j+lag]) - h[j-lag] (with indicator functions)
So |∂c[k]/∂h[j]| ≤ 2

The Hessian of c[k] with respect to h is:

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_c_at_lag(h, n, k_idx):
    lag = k_idx - (n - 1)
    g = np.zeros(n)
    if lag >= 0:
        ns = n - lag
        if ns > 0:
            g[:ns] += (1.0 - h[lag:])
            g[lag:] -= h[:ns]
    else:
        lp = -lag
        ns = n - lp
        if ns > 0:
            g[lp:] += (1.0 - h[:ns])
            g[:ns] -= h[lp:]
    return g

def grad_soft(h, n, T):
    dx = 2.0 / n
    c = compute_c(h)
    m = np.max(c)
    exp_c = np.exp((c - m) * T)
    p = exp_c / np.sum(exp_c)
    g = np.zeros(n)
    active = p > 1e-10
    for k in np.where(active)[0]:
        g += p[k] * grad_c_at_lag(h, n, k)
    return g * dx

def fista_smooth(h0, n, T_start, T_end, max_iters, deadline):
    h = h0.copy()
    h_prev = h0.copy()
    t_count = 1
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        frac = it / max(max_iters - 1, 1)
        T = T_start + (T_end - T_start) * frac
        lr = 0.05 / max(T, 0.5)
        mu = (t_count - 1) / (t_count + 2.0)
        y = h + mu * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        g = grad_soft(y, n, T)
        h_prev = h.copy()
        h = project(y - lr * g, n)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        t_count += 1
    return best_h, best_c5

def subgradient_exact(h0, n, max_iters, deadline, alpha=0.015):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    avg_h = h.copy() * 0.0
    for t in range(max_iters):
        if time.time() > deadline:
            break
        c = compute_c(h)
        k_star = np.argmax(c)
        c5 = c[k_star] * dx
        if t % 10 == 0:
            avg_h += h
            avg_h = project(avg_h / (t // 10 + 1), n)
            c5_avg = evaluate_c5(avg_h)
            if c5_avg < best_c5:
                best_c5 = c5_avg
                best_h = avg_h.copy()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        g = grad_c_at_lag(h, n, k_star)
        step = alpha / (1.0 + 0.15 * np.sqrt(t))
        h = project(h - step * g * dx, n)
        if t > 0 and t % 200 == 0:
            h_pert = h + np.random.randn(n) * 0.04
            h_pert = project(h_pert, n)
            c5_p = evaluate_c5(h_pert)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_pert.copy()
                h = h_pert
    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def slsqp_refine(h0, n, T, deadline):
    dx = 2.0 / n
    target = n / 2.0
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        return (m + np.log(np.sum(np.exp(c - m)))) / T * dx
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        m = np.max(c)
        exp_c = np.exp((c - m) * T)
        p = exp_c / np.sum(exp_c)
        g = np.zeros(n)
        for k in range(2 * n - 1):
            if p[k] < 1e-10:
                continue
            g += p[k] * grad_c_at_lag(x, n, k)
        return g * dx / T
    def cons_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_eq}],
                       options={'ftol': 1e-15, 'maxiter': 300})
        if time.time() > deadline:
            return h0, evaluate_c5(h0)
        if res.x is not None and np.all(np.isfinite(res.x)):
            h_new = project(res.x, n)
            return h_new, evaluate_c5(h_new)
    except Exception:
        pass
    return h0, evaluate_c5(h0)

def interp_to(h_src, n_src, n_dst):
    if n_src == n_dst:
        return h_src.copy()
    xf = np.linspace(0, 2, n_src, endpoint=False)
    xt = np.linspace(0, 2, n_dst, endpoint=False)
    return np.interp(xt, xf, h_src)

def binary_local_search(h0, n, max_iters, deadline):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    c = compute_c(h) * (2.0 / n)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        k_star = np.argmax(c)
        g = grad_c_at_lag(h, n, k_star) * (2.0 / n)
        idx = np.argmax(g)
        step = 0.05
        h[idx] = np.clip(h[idx] - step, 0.0, 1.0)
        h = project(h, n)
        c = compute_c(h) * (2.0 / n)
        c5 = np.max(c)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        else:
            h = best_h.copy()
            idx = np.argmin(g)
            h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
            h = project(h, n)
            c = compute_c(h) * (2.0 / n)
            c5 = np.max(c)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            else:
                h = best_h.copy()
    return best_h, best_c5
