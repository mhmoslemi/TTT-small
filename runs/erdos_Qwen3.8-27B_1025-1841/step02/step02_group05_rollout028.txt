def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):

            lo = mid
        else:
            hi = mid

    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def coord_update(h, i, corr, n):
    """Find optimal value for h[i] given all other coordinates fixed."""
    # s ranges from -(n-1) to (n-1)
    # a_s = 1 - h[i-s] - h[i+s] (with bounds)
    s = np.arange(-(n-1), n)  # shifts
    # For each shift s, compute coefficient of h[i] in corr[s+n-1]
    # h[i] appears as h[j] when j=i (shift s means h[i-s] in second factor)
    # and as (1-h[j]) when j-s=i, i.e., j=i+s

    # Coefficient from second term: h[i+s]*(1-h[i]) -> coeff is -h[i+s] if valid
    
    idx1 = i - s  # indices for h[i-s], valid when 0 <= i-s < n
    idx2 = i + s  # indices for h[i+s], valid when 0 <= i+s < n

    valid2 = (idx2 >= 0) & (idx2 < n)

    a[valid1] += 1.0 - h[idx1[valid1]]
    a[valid2] -= h[idx2[valid2]]
    
    # Now corr[s] = a[s]*h[i] + b[s] where b[s] = corr[s] - a[s]*h[i]

            hi = mid
        elif a[active] < 0:
            lo = mid
        else:
            break

                h = project_to_feasible(interp_periodic(h_best_curr, n), n)
                corr = compute_corr(h)
            if n_best_curr != n_hi:
                h_hi = project_to_feasible(interp_periodic(h_best_curr, n_hi), n_hi)
                corr_hi = compute_corr(h_hi)

        sweep_count += 1

    # Final polish: single-coordinate sweep (higher precision)
    if _remaining() > 30:
        n_f = len(best_h)
        h_f = best_h.copy()
        corr_f = compute_corr(h_f)
        tl_polish = min(25, _remaining() * 0.4)
        t_polish_start = time.time()
        while time.time() - t_polish_start < tl_polish:
            order = np.random.permutation(n_f)
            h_f, corr_f = single_sweep(h_f, corr_f, n_f, order, n_bs=30)
            h_f = np.clip(h_f, 0.0, 1.0)
            h_f = project_to_feasible(h_f, n_f)
            corr_f = compute_corr(h_f)
            c5_f = corr_f.max() * (2.0 / n_f)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Final: ensure feasibility and compute exact C5
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
