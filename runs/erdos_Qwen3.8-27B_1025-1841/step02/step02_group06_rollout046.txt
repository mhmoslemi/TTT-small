        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def lse_gradient(h, beta, n, dx, topM=None):
    corr = compute_corr(h)
    g = corr * dx
    max_g = g.max()
    shifted = np.clip(beta * (g - max_g), -500, 500)
    exp_vals = np.exp(shifted)
    if topM is not None and topM < len(g):
        top_idx = np.argpartition(exp_vals, -topM)[-topM:]
        mask = np.zeros(len(g), dtype=bool)
        mask[top_idx] = True
        exp_vals = exp_vals * mask
    total = exp_vals.sum()
    if total < 1e-300:
        p = np.zeros(len(g))
        p[np.argmax(g)] = 1.0
    else:
        p = exp_vals / total
    grad = np.zeros(n)
    active = np.where(p > 1e-15)[0]
    for k in active:
        pk = p[k]
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx, max_g


def lse_optimize(h, n, beta, max_iters, time_limit, lr_init=0.1, topM=None, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_obj = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    patience = max(150, max_iters // 8)
    lr = lr_init
    dx = 2.0 / n

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        grad, cur_max = lse_gradient(h, beta, n, dx, topM)
        if cur_max < best_obj - 1e-14:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = momentum * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > patience:
            lr *= 0.55
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    return best_h, best_obj


def subgrad_focused(h, n, max_iters, time_limit, lr_init=0.003, topK=4):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.82
    stall = 0
    patience = 300

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_c5 - 1e-14:
            best_c5 = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        top_idx = np.argsort(g)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK
        cur_lr = lr_init / (1.0 + 0.008 * it)
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > patience:
            lr_init *= 0.6
            v *= 0.4
            stall = 0
            if lr_init < 1e-14:
                break
        h = h_new
    return best_h, best_c5


def local_search_binary(h, n, time_limit):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    at_one = np.where(h > 0.95)[0]
    at_zero = np.where(h < 0.05)[0]
    np.random.shuffle(at_one)
    np.random.shuffle(at_zero)
    trials = 0
    while time.time() - start < time_limit and trials < 5000:
        trials += 1
        if len(at_one) == 0 or len(at_zero) == 0:
            break
        i = at_one[trials % len(at_one)]
        h[i] = 0.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            at_zero = np.where(h < 0.05)[0]
        else:
            h[i] = 1.0
        j = at_zero[trials % len(at_zero)]
        h[j] = 1.0
        c5_new = compute_c5(h)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h.copy()
            at_one = np.where(h > 0.95)[0]
        else:
            h[j] = 0.0
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Quick LSE at initial resolution
    if remaining() > 80:
        for beta in [50.0, 150.0]:
            if remaining() < 20:
                break
            tl = min(remaining() * 0.25, 100.0)
            h1, _ = lse_optimize(best_h, n_init, beta, max_iters=15000,
                                 time_limit=tl, lr_init=0.12, topM=0, momentum=0.92)
            c5_1 = compute_c5(h1)
            if c5_1 < best_c5:
                best_c5 = c5_1
                best_h = h1.copy()

    # Phase 2: Double resolution - main optimization push
    n2 = n_init * 2
    if remaining() > 120:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        for beta in [100.0, 300.0, 600.0]:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.22, 160.0)
            h2, _ = lse_optimize(h2, n2, beta, max_iters=10000,
                                 time_limit=tl, lr_init=0.09, topM=200, momentum=0.9)
            c5_2 = compute_c5(h2)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2.copy()

    # Phase 3: Quadruple resolution - high-beta focused optimization
    n3 = n_init * 4
    if remaining() > 100:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        for beta in [200.0, 500.0, 1000.0, 2000.0]:
            if remaining() < 25:
                break
            tl = min(remaining() * 0.28, 200.0)
            h3, _ = lse_optimize(h3, n3, beta, max_iters=6000,
                                 time_limit=tl, lr_init=0.07, topM=30, momentum=0.88)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()

    # Phase 4: Subgradient polish at current best resolution
    if remaining() > 50:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = subgrad_focused(h_f, n_f, max_iters=80000,
                                     time_limit=remaining() * 0.25, lr_init=0.002, topK=4)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Binary local search (flip 1<->0 at boundaries)
    if remaining() > 40:
        n_ls = len(best_h)
        h_ls = best_h.copy()
        h_ls, c5_ls = local_search_binary(h_ls, n_ls, time_limit=remaining() * 0.15)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()

    # Phase 6: Perturbation + re-optimization at high res
    if remaining() > 60:
        n_p = len(best_h)
        for attempt in range(3):
            if remaining() < 25:
                break
            noise = 0.03 * (attempt + 1)
            h_p = best_h + noise * np.random.randn(n_p)
            h_p = project_to_feasible(h_p, n_p)
            tl = min(remaining() * 0.2, 80.0)
            h_p, _ = lse_optimize(h_p, n_p, beta=500.0, max_iters=4000,
                                   time_limit=tl, lr_init=0.08, topM=30, momentum=0.9)
            c5_p = compute_c5(h_p)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 7: Final ultra-focused LSE
    if remaining() > 20:
        n_fin = len(best_h)
        h_fin = best_h.copy()
        h_fin, _ = lse_optimize(h_fin, n_fin, beta=3000.0, max_iters=3000,
                                 time_limit=remaining() * 0.35, lr_init=0.04, topM=10, momentum=0.85)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility enforcement
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
