def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        mask = h < 1.0 - 1e-14 if diff > 0 else h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

def upsample_h(h, new_n):

            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
