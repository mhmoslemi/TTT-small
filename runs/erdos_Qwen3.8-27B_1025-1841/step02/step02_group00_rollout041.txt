import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def compute_c5_fft(h):
    """Compute all f_k values using FFT, return max * dx."""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    # Cross-correlation via FFT
    N = 2 * n
    F_h = fft(h, N)
    F_b = fft(b, N)
    corr = np.real(ifft(F_h * np.conj(F_b)))
    # corr[m] corresponds to lag k = m - (n-1)
    # We want max over all lags
    return float(np.max(corr) * dx), corr * dx

def compute_c5_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Water-filling
    if diff > 0:
        # Need to increase sum by diff
        # Increase from smallest values first
        order = np.argsort(h)
        h[order] += 0.0  # just to establish order
        # Binary search for threshold
        lo, hi = 0.0, 1.0
        for _ in range(60):
            mid = (lo + hi) / 2
            # New sum if we set all values below mid to mid, keep others
            new_h = np.where(h < mid, mid, h)
            new_sum = np.sum(new_h)
            if new_sum < target:
                lo = mid
            else:
                hi = mid
        # Use lo (slightly below target)
        h = np.where(h < lo, lo, h)
        # Fix remaining
        rem = target - np.sum(h)
        if rem > 0:
            # Add to entries that are still < 1
            mask = h < 1.0
            if np.sum(mask) > 0:
                h[mask] += rem / np.sum(mask)
                h = np.clip(h, 0, 1)
    else:
        # Need to decrease sum by |diff|
        order = np.argsort(-h)
        lo, hi = 0.0, 1.0
        for _ in range(60):

def upsample_h(h, new_n):

def _next_pow2(n):
    return 1 << (n - 1).bit_length()

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_fft(h):
    n = len(h)
    b = 1.0 - h
    N = _next_pow2(2 * n - 1)
    Fh = fft(h, N)
    Fb = fft(b, N)
    corr = np.real(ifft(Fh * np.conj(Fb)))[:2 * n - 1]
    return corr

def make_feasible(h, n):

    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    best_h = h.copy()
    dx = 2.0 / n
