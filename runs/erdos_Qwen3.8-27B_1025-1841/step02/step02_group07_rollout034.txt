def iterative_top_m(h, n, top_m_count, lr, max_iter, deadline):
    """Iteratively minimize the sum of top-m overlaps."""

    for it in range(max_iter):

            break
        overlaps, corr = compute_overlaps(h)

    top_idx = np.argsort(overlaps)[::-1][:top_m_count]

    one_minus_h = 1.0 - h
    for km in k_vals:
        if km >= 0:
            # (1-h[i+km]) for i = 0..n-1-km
            grad[:n-km] += one_minus_h[km:]

            grad[:n-km_abs] -= h[km_abs:]

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:

# Helper functions (top-level, no closures)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def gradient_top_m(h, m):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1.0 - h, mode='full') * dx
    top_idx = np.argsort(overlaps)[::-1][:m]
    k_vals = (top_idx - (n - 1)).astype(int)
    grad = np.zeros(n)
    omh = 1.0 - h
    for km in k_vals:
        if km >= 0:
            grad[:n-km] += omh[km:]
            grad[km:] -= h[:n-km]
        else:
            ka = -km
            grad[ka:] += omh[:n-ka]
            grad[:n-ka] -= h[ka:]
    return grad * dx

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

    """Cycling gradient descent through different m values."""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = lr_init
    
    for cycle in range(100):
        if time.time() > deadline:
            break
        no_improve_total = 0
        for m in m_seq:
            if time.time() > deadline:
                break
            lr_local = lr
            stuck = 0
            for it in range(500):
                if time.time() > deadline:
                    break
                grad = gradient_top_m(h, m)
                improved = False
                for sf in [1.0, 0.5, 0.25, 0.1, 0.05]:
                    h_try = project(h - lr_local * sf * grad, n)
                    c5_try = evaluate_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try
                        improved = True
                        stuck = 0
                        break
                if not improved:
                    stuck += 1
                    lr_local *= 0.5
                    if lr_local < 1e-10 or stuck >= stuck_limit:
                        break
            no_improve_total += stuck
        if no_improve_total >= stuck_limit * len(m_seq):
            break
    
    return best_h, best_c5

        if len(ones) >= excess:
            flip = rng.choice(ones, size=excess, replace=False)
            h[flip] = 0
    elif excess < 0:

        if len(zeros) >= -excess:

            h[flip] = 1
