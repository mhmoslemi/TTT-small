            tl = min(rem() * 0.12, 80.0)
            hp, cp = optimize_main(hp, n_final, t0, tl, lr0=0.1,
                                    tau0=30.0, tau1=200.0, topM_base=100, ramp=0)
            if cp < best_c5:
                best_c5 = cp
                best_h = hp.copy()

    # Phase 8: Final subgradient polish
    if rem() > 20:
        tl = min(rem() * 0.5, 100.0)
        hf, cf = subgrad_polish(best_h, n_final, t0, tl, K=3, lr0=0.002)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    best_h = project_to_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n_final
