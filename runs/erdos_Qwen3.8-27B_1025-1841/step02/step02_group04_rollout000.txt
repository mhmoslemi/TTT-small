    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    if abs(np.sum(h) - target) > 1e-10:

def subgrad_at_max(h, top_k=5):
    """Compute weighted gradient from top-k correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Get top-k indices
    idx = np.argsort(corr)[-top_k:][::-1]
    mc = corr[idx[0]]
    # Weight by soft gap
    gaps = mc - corr[idx]
    weights = np.exp(-gaps * 10.0)  # temperature for weighting
    weights /= np.sum(weights)
