def project_to_feasible(g, n):
    """Project g onto {h : 0<=h<=1, sum(h)=n/2} using bisection on sum."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid

    lo = g.min() - 2.0

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(exp_c.sum()) / tau)
    return obj, grad

def topk_subgrad(h, kt):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    top = np.argsort(corr)[-kt:]
    grad = np.zeros(n)
    for k in top:
        sh = k - n + 1
        j0, j1 = max(0, sh), min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[np.arange(j0, j1) - sh])
        j0b, j1b = max(0, -sh), min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[np.arange(j0b, j1b) + sh]
    grad /= kt
    grad *= dx
    return grad, corr.max() * dx
