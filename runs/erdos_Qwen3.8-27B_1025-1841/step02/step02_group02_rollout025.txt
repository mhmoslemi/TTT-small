            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= h[idx + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

        if stall > 150:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()
    n_work = n_init

    # Phase 1: Adaptive tau at parent resolution, multiple restarts
    n_restarts = 5
    restart_budget = remaining() * 0.35
    per_restart = restart_budget / n_restarts

    for r in range(n_restarts):
        if remaining() < 60:
            break
        if r == 0:
            h_cur = h_init.copy()
            tau_s, tau_e = 8.0, 60.0
            lr_init = 0.3
        elif r == 1:
            h_cur = h_init.copy()
            h_cur += 0.03 * np.random.randn(n_work)
            h_cur = project_to_feasible(h_cur, n_work)
            tau_s, tau_e = 15.0, 80.0
            lr_init = 0.2
        elif r == 2:
            h_cur = best_h.copy()
            h_cur += 0.02 * np.random.randn(n_work)
            h_cur = project_to_feasible(h_cur, n_work)
            tau_s, tau_e = 30.0, 120.0
            lr_init = 0.1
        elif r == 3:
            # Symmetry-enforced start
            h_sym = 0.5 * (h_init + h_init[::-1])
            h_sym = project_to_feasible(h_sym, n_work)
            h_cur = h_sym
            tau_s, tau_e = 10.0, 70.0
            lr_init = 0.25
        else:
            h_cur = best_h.copy()
            h_cur += 0.01 * np.random.randn(n_work)
            h_cur = project_to_feasible(h_cur, n_work)
            tau_s, tau_e = 40.0, 150.0
            lr_init = 0.08

        tl = min(per_restart, remaining() * 0.4)
        h_cur, c5_cur = adaptive_smooth(h_cur, n_work, tau_s, tau_e, 8000, tl, lr_init)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

    # Phase 2: Higher resolution
    n_high = n_work * 2
    if remaining() > 200:
        h_high = interp_periodic(best_h, n_high)
        h_high = project_to_feasible(h_high, n_high)
        tl = min(remaining() * 0.18, 250)
        h_high, c5_high = adaptive_smooth(h_high, n_high, 20.0, 100.0, 4000, tl, 0.15)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high.copy()
            n_work = n_high

    # Phase 3: Fine-tuning with high tau
    if remaining() > 80:
        h_fine = best_h.copy()
        tl = min(remaining() * 0.25, 300)
        h_fine, c5_fine = adaptive_smooth(h_fine, n_work, 60.0, 250.0, 6000, tl, 0.06)
        if c5_fine < best_c5:
            best_c5 = c5_fine
            best_h = h_fine.copy()

    # Phase 4: Subgradient polish
    if remaining() > 40:
        h_sub = best_h.copy()
        h_sub, c5_sub = subgrad_polish(h_sub, n_work, 200000, remaining() * 0.35, 1.0)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub.copy()

    # Phase 5: Ultra-fine subgrad
    if remaining() > 20:
        h_f = best_h.copy()
        h_f, c5_f = subgrad_polish(h_f, n_work, 100000, remaining() * 0.4, 0.3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
