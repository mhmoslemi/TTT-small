    target = n / 2.0

            break

    dx = 2.0 / n

import numpy as np
import time
import math
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T)

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    p = exp_s / np.sum(exp_s)
    
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        # (1-h[j+s]) for j+s in [0,n-1]
        a = max(0, -s)
        b = min(n, n - s)
        if b > a:
            g[a:b] += pk * (1.0 - h[a+s:b+s])
        # -h[j-s] for j-s in [0,n-1]
        a2 = max(0, s)
        b2 = min(n, n + s)
        if b2 > a2:
            g[a2:b2] -= pk * h[a2-s:b2-s]
    g *= dx
    return g

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(200):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return project_feasible(np.interp(xt, xf, h), n_to)

        return soft_c5(x, T)

import numpy as np
import time
import math
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_c5(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T)

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    p = exp_s / np.sum(exp_s)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        a = max(0, -s)
        b = min(n, n - s)
        if b > a:
            g[a:b] += pk * (1.0 - h[a+s:b+s])
        a2 = max(0, s)
        b2 = min(n, n + s)
        if b2 > a2:
            g[a2:b2] -= pk * h[a2-s:b2-s]
    g *= dx
    return g

def project_feasible(h, n):
    target = n * 0.5
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        mask = (h < 1.0 - 1e-15) if d > 0 else (h > 1e-15)
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)

    return project_feasible(np.interp(xt, xf, h), n_to)

def coordinate_refine(h0, n, T, n_rounds, deadline):
    """Cyclic coordinate descent on the soft-max objective."""
    h = h0.copy()
    dx = 2.0 / n
    for rnd in range(n_rounds):

            break
        for i in range(n):
            if time.time() > deadline:

            # Optimize h[i] in [0,1] while keeping sum fixed

            best_val = soft_c5(h, T)

                h[i] = trial

                diff = old - trial
                # Spread across all others
                h_rest = h.copy()
                h_rest[i] = 0.0
                # Find a good partner
                if diff > 0:

                        mask[partner] = False

            # Revert if no improvement
            if soft_c5(h, T) > soft_c5(h if h[i] != trial else h, T):
