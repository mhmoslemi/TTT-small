```python
def subgradient_step(h, n, dx, top_m=20, step_size=0.01):
    corr = compute_all_correlations(h, n, dx)
    # Find top M shifts
    top_indices = np.argsort(corr)[-top_m:]
    # Compute subgradient: average of gradients of top M
    grad = np.zeros(n)
    for k_idx in top_indices:
        k = (k_idx - (n-1))  # actual shift
        for i in range(n):
            i_plus = i + k
            i_minus = i - k
            if 0 <= i_plus < n:
                grad[i] += (1 - h[i_plus]) * dx
            if 0 <= i_minus < n:
                grad[i] -= h[i_minus] * dx
    grad /= top_m
    # Normalize
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-12:
        return h
    # Move in negative gradient direction
    h_new = h - step_size * grad / gnorm
    # Project: maintain sum constraint and bounds
    h_new = make_feasible(h_new, n)
    return h_new
```

```python
        if s >= 0:
            grad[:n-s] += wm * dx * (1.0 - h[s:])
            grad[s:] += -wm * dx * h[:n-s]
        else:
            sa = -s
            grad[sa:] += wm * dx * (1.0 - h[:n-sa])
            grad[:n-sa] += -wm * dx * h[sa:]
    return soft_val, grad

def slsqp_opt(h0, t=80.0, maxiter=400):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0,
             'jac': lambda x: np.ones(n)}]
    def obj(x):
        return soft_obj_and_grad(x, t)[0]
    def obj_grad(x):
        return soft_obj_and_grad(x, t)
    try:
        res = minimize(obj_grad, h0, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or res.x is not None:
            h_res = make_feasible(res.x, n)
            return h_res
    except Exception:
        pass
    return None

def subgradient_refine(h, max_time=300.0, top_m=30, lr_init=0.01):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_curr = h.copy()
    momentum = np.zeros(n)
    beta = 0.9
    lr = lr_init
    start = time.time()
    stall = 0
    while time.time() - start < max_time and stall < 50:
        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full') * dx
        a_max = np.max(corr)
        top_idx = np.argsort(corr)[-top_m:]
        grad = np.zeros(n)
        for m in top_idx:
            s = m - (n - 1)
            if s >= 0:
                grad[:n-s] += (1.0 - h_curr[s:])
                grad[s:] += -h_curr[:n-s]
            else:
                sa = -s
                grad[sa:] += (1.0 - h_curr[:n-sa])
                grad[:n-sa] += -h_curr[sa:]
        grad *= dx / top_m
        momentum = beta * momentum + lr * grad
        h_new = h_curr - momentum
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-9:
            best_h = h_new.copy()
            best_c5 = c5_new
            stall = 0
            h_curr = h_new
        else:
            stall += 1
            lr *= 0.9
            h_curr = best_h.copy()
        if time.time() - start < max_time:
            pass
    return best_h, best_c5

def random_binary_h(n, rng):
    h = np.zeros(n)
    indices = rng.permutation(n)
    h[indices[:n//2]] = 1.0
    return h

def random_structured_h(n, rng):
    """Create h with block structure for diversity."""
    n_blocks = rng.randint(4, 16)
    block_size = n // n_blocks
    h = np.zeros(n)
    for b in range(n_blocks):
        if rng.rand() < 0.5:
            h[b*block_size:(b+1)*block_size] = 1.0
    return make_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    elapsed = lambda: time.time() - start_time

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        h0 = random_binary_h(320, rng)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP at n=400 with analytic gradient
    n1 = 400
    h1 = upsample_h(best_h, n1)
    c1 = compute_c5(h1)
    if c1 < best_c5:
        best_c5 = c1
        best_h = h1

    # Multiple SLSQP restarts at n=400
    for r in range(6):
        if elapsed() > budget_s * 0.45:
            break
        if r == 0:
            h_start = h1.copy()
        elif r == 1:
            h_start = best_h.copy()
            if len(h_start) != n1:
                h_start = upsample_h(best_h, n1)
            h_start = h_start + rng.randn(n1) * 0.03
            h_start = make_feasible(h_start, n1)
        elif r == 2:
            h_start = random_binary_h(n1, rng)
        elif r == 3:
            h_start = random_structured_h(n1, rng)
        elif r == 4:
            h_start = upsample_h(best_h, n1) + rng.randn(n1) * 0.08
            h_start = make_feasible(h_start, n1)
        else:
            h_start = best_h.copy()
            if len(h_start) != n1:
                h_start = upsample_h(best_h, n1)
            # Flip a random block
            block_size = n1 // 8
            start_idx = rng.randint(0, n1 - block_size)
            h_start[start_idx:start_idx+block_size] = 1.0 - h_start[start_idx:start_idx+block_size]
            h_start = make_feasible(h_start, n1)

        for t_val in [80.0, 200.0]:
            if elapsed() > budget_s * 0.45:
                break
            result = slsqp_opt(h_start, t=t_val, maxiter=350)
            if result is not None:
                c = compute_c5(result)
                if c < best_c5:
                    best_c5 = c
                    best_h = result.copy()

    # Phase 2: Subgradient refinement at n=400
    n_sub = len(best_h)
    remaining = budget_s - elapsed() - 60
    if remaining > 30:
        h_sub, c5_sub = subgradient_refine(best_h, max_time=min(remaining, 300), top_m=40, lr_init=0.005)
        if c5_sub < best_c5:
            best_c5 = c5_sub
            best_h = h_sub

    # Phase 3: Try higher resolution n=500
    remaining = budget_s - elapsed() - 30
    if remaining > 60:
        n2 = 500
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2
        for r in range(3):
            if elapsed() > budget_s - 40:
                break
            if r == 0:
                h_start = h2.copy()
            else:
                h_start = upsample_h(best_h, n2) + rng.randn(n2) * 0.03
                h_start = make_feasible(h_start, n2)
            result = slsqp_opt(h_start, t=100.0, maxiter=250)
            if result is not None:
                c = compute_c5(result)
                if c < best_c5:
                    best_c5 = c
                    best_h = result.copy()

    # Phase 4: Final subgradient polish
    remaining = budget_s - elapsed() - 10
    if remaining > 20:
        h_pol, c5_pol = subgradient_refine(best_h, max_time=remaining * 0.6, top_m=50, lr_init=0.003)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
