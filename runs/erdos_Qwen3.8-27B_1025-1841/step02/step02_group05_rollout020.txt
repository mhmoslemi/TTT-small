def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

    current_tau = tau_schedule[0]

        if tau_idx < len(tau_schedule) and it >= tau_schedule[tau_idx + 1] if tau_idx + 1 < len(tau_schedule) else False:
            pass  # handled below

            v *= 0.3
            stall = 0

        y_proj = project_to_feasible(y, n)
        _, grad_y = smooth_obj_grad(y_proj, current_tau)
        h_new = y_proj - lr * grad_y
        h_new = project_to_feasible(h_new, n)
        h_prev = h
        h = h_new
    
    return best_h, best_c5

        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)
