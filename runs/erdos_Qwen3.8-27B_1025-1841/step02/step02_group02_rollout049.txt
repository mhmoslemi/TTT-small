        return h.copy()
    step_old = 2.0 / n
    idx = np.arange(target_n) * (2.0 / target_n) / step_old
    i0 = np.floor(idx).astype(np.int64) % n
    i1 = (i0 + 1) % n
    frac = idx - np.floor(idx)
    return h[i0] * (1.0 - frac) + h[i1] * frac

def compute_corr(h):
    n = len(h)
    return fftconvolve(h, np.flip(1.0 - h), mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_grad(h, tau, top_m=300):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -50, 50)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    nc = len(corr)
    if top_m < nc:
        order = np.argpartition(p, -top_m)[-top_m:]
    else:
        order = np.arange(nc)
    
    L_all = np.arange(nc) - (n - 1)
    grad = np.zeros(n)
    
    for k in order:
        pk = p[k]
        if pk < 1e-14:
            continue
        Lk = L_all[k]
        if Lk >= 0:
            i1 = np.arange(0, n - Lk)
            grad[i1] += pk * (1.0 - h[i1 + Lk])
            i2 = np.arange(Lk, n)
            grad[i2] -= pk * h[i2 - Lk]
        else:
            M = -Lk
            i1 = np.arange(M, n)
            grad[i1] += pk * (1.0 - h[i1 - M])
            i2 = np.arange(0, n - M)
            grad[i2] -= pk * h[i2 + M]
    
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def optimize_smooth(h_in, tau, max_iters, time_limit, lr=0.2, momentum=0.9):
    n = len(h_in)
    h = project_to_feasible(h_in, n).copy()
    t0 = time.time()
    
    obj, grad = smooth_obj_grad(h, tau)
    best_obj = obj
    best_h = h.copy()
    
    v = np.zeros(n)
    y = h.copy()
    stall = 0
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        v = momentum * v - lr * grad
        y = h + v  # Nesterov lookahead
        h = project_to_feasible(y, n)
        
        if stall > 60:
            lr *= 0.5
            v = 0.8 * v
            stall = 0
            if lr < 1e-10:
                break
    
    c5 = compute_c5(best_h)
    return best_h, c5

def optimize_subgrad(h_in, max_iters, time_limit, lr=0.005):
    n = len(h_in)
    dx = 2.0 / n
    h = project_to_feasible(h_in, n).copy()
    t0 = time.time()
    
    corr = compute_corr(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        k_star = np.argmax(corr)
        L = k_star - (n - 1)
        grad = np.zeros(n)
        if L >= 0:
            i1 = np.arange(0, n - L)
            grad[i1] = (1.0 - h[i1 + L])
            i2 = np.arange(L, n)
            grad[i2] -= h[i2 - L]
        else:
            M = -L
            i1 = np.arange(M, n)
            grad[i1] = (1.0 - h[i1 - M])
            i2 = np.arange(0, n - M)
            grad[i2] -= h[i2 + M]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        
        if stall > 200:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break
    
    return best_h, best_c5

def local_search(h_in, time_limit, max_rounds=500):
    """Try small perturbations to reduce C5."""
    n = len(h_in)
    dx = 2.0 / n
    h = project_to_feasible(h_in, n).copy()
    t0 = time.time()
    
    corr = compute_corr(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    
    for rnd in range(max_rounds):
        if time.time() - t0 > time_limit:
            break
        
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
        
        k_star = np.argmax(corr)
        L = k_star - (n - 1)
        
        # Compute gradient of active constraint
        if L >= 0:
            d = np.zeros(n)
            d[:n-L] = (1.0 - h[L:])  # (1-h[i+L]) for i in [0, n-1-L]
            d[L:] -= h[:n-L]  # -h[i-L] for i in [L, n-1]
        else:
            M = -L
            d = np.zeros(n)
            d[M:] = (1.0 - h[:-M])  # (1-h[i-M]) for i in [M, n-1]
            d[:-M] -= h[M:]  # -h[i+M] for i in [0, n-1-M]
        
        # Try a few directions
        for step in [0.005, 0.002, 0.001]:
            h_trial = project_to_feasible(h - step * d * dx, n)
            c = compute_c5(h_trial)
            if c < best_c5:
                h = h_trial
                best_c5 = c
                best_h = h.copy()
                break
    
    return best_h, best_c5
