import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 0
_OPT_T = 1.0
_OPT_K = 5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(h)
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            if diff > 0:
                add = min(1.0 - h[i], diff)
                h[i] += add
                diff -= add
            else:
                sub = min(h[i], -diff)
                h[i] -= sub
                diff += sub
            if abs(diff) < 1e-12:
                break
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def obj_lse(h):
    """Log-sum-exp objective."""
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = get_corr(h)
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def obj_topk(h):
    """Top-K average objective (differentiable approximation)."""
    n = _OPT_N
    K = _OPT_K
    dx = 2.0 / n
    corr = get_corr(h)
    k = min(K, len(corr))
    top = np.sort(corr)[-k:]
    return np.mean(top) * dx

def constraint_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def constraint_jac(h):
    return np.ones(_OPT_N)

def slsqp_optimize(h_start, n, T, max_iter, use_topk=False):
    global _OPT_N, _OPT_T, _OPT_K
    _OPT_N = n
    _OPT_T = T
    if use_topk:
        _OPT_K = min(10, len(get_corr(h_start)))
        obj = obj_topk
    else:
        obj = obj_lse
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return make_feasible(h_start, n)

def gradient_for_lag(h, lag):
    """Compute gradient of the overlap at a given lag (positive or negative integer)."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        lag2 = -lag
        if lag2 < n:
            g[lag2:] += (1.0 - h[:n-lag2])
            g[:n-lag2] -= h[lag2:]
    return g

def topk_gradient_step(h, K, lr):
    """One step of gradient descent on the top-K objective."""
    n = len(h)
    corr = get_corr(h)
    k = min(K, len(corr))
    top_idx = np.argpartition(corr, -k)[-k:]
    
    g = np.zeros(n)
    for m in top_idx:
        lag = m - (n - 1)
        g += gradient_for_lag(h, lag)
    g /= k
    
    h_new = h - lr * g
    return make_feasible(h_new, n), g

def gradient_descent(h, n, K, n_iter, lr_init, lr_decay, time_limit):
    """Run gradient descent on top-K objective with time limit."""
    h = h.copy()
    lr = lr_init

            h = h_new
        else:
            # Reduce step size
            lr *= 0.9

    return best_h, best_c5

def coord_descent(h, n, n_rounds, time_limit):
    """Coordinate descent: try changing one coordinate at a time."""
    h = h.copy()

                best_c5 = best_c
                improved = True
        if not improved:
            break
    
    return h, best_c5

    # weights = corr^(p-1)
    corr_pm1 = corr ** (p-1)
    W = S ** (1.0 - 1.0/p)

        if w < 1e-10:
