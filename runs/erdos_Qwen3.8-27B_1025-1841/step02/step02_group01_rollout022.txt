    n = len(h)
    h = h.copy()
    corr_current = np.correlate(h, 1.0 - h, mode='full')
    n_corr = 2 * n - 1
    m_arr = np.arange(n_corr)
    
    prev_max = np.max(corr_current)
    
    for p in range(max_passes):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 5:
                break
        max_improvement = 0.0
        
        for j in range(n):
            j_val = h[j]
            IA = (m_arr >= j) & (m_arr <= j + n - 1)
            IB = (m_arr >= n - 1 - j) & (m_arr <= 2 * n - 2 - j)
            
            A = np.zeros(n_corr)
            i_A = j - m_arr + n - 1
            i_B = j + m_arr + 0
            i_B = j + m_arr - n + 1
            
            if np.any(IA):
                A[IA] = 1.0 - h[i_A[IA]]
            if np.any(IB):
                A[IB] -= h[i_B[IB]]
            
            B = corr_current - j_val * A
            v_new = solve_1d_convex(A, B)
            
            if abs(v_new - j_val) > 1e-12:
                new_max = np.max(A * v_new + B)
                old_max = np.max(corr_current)
                if new_max < old_max - 1e-14:
                    h[j] = v_new
                    corr_current = corr_current + (v_new - j_val) * A
                    max_improvement = max(max_improvement, old_max - new_max)
        
        h = project_feasible(h, n)
        corr_current = np.correlate(h, 1.0 - h, mode='full')
        cur_max = np.max(corr_current)
        
        if max_improvement < tol or cur_max > prev_max + 1e-10:
            if cur_max > prev_max + 1e-10:
                h = project_feasible(h, n)
            break
        prev_max = cur_max
    
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.permutation(n)
    h[idx[:n_ones]] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    best_c5 = float('inf')
    best_h = None
    
    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    
    if h0 is None:
        h0 = random_binary_h(320, rng)
        h0 = project_feasible(h0, len(h0))
    
    n0 = len(h0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent on initial resolution
    h_cd = coord_descent_exact(h0, max_passes=300, time_limit=budget_s, start_time=start_time)
    h_cd = project_feasible(h_cd, n0)
    c5_cd = compute_c5(h_cd)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()
    
    # Phase 2: Perturbation + coordinate descent at same resolution
    elapsed = time.time() - start_time
    n_perturb = min(8, int((budget_s - elapsed - 120) / 40))
    n_perturb = max(n_perturb, 3)
    
    for p in range(n_perturb):
        if time.time() - start_time > budget_s - 60:
            break
        scale = 0.03 + 0.02 * p
        h_pert = best_h + rng.randn(n0) * scale
        h_pert = project_feasible(h_pert, n0)
        h_pert = coord_descent_exact(h_pert, max_passes=100, time_limit=budget_s, start_time=start_time)
        h_pert = project_feasible(h_pert, n0)
        c5_pert = compute_c5(h_pert)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
    
    # Phase 3: Binary restart + coordinate descent
    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        for p in range(3):
            if time.time() - start_time > budget_s - 40:
                break
            h_bin = random_binary_h(n0, rng)
            h_bin = project_feasible(h_bin, n0)
            h_bin = coord_descent_exact(h_bin, max_passes=150, time_limit=budget_s, start_time=start_time)
            h_bin = project_feasible(h_bin, n0)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
    
    # Phase 4: Try higher resolutions
    elapsed = time.time() - start_time
    higher_res = [480, 640]
    for new_n in higher_res:
        if time.time() - start_time > budget_s - 120:
            break
        h_hi = upsample_h(best_h, new_n)
        h_hi = coord_descent_exact(h_hi, max_passes=100, time_limit=budget_s, start_time=start_time)
        h_hi = project_feasible(h_hi, new_n)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
        
        # Perturbation at higher res
        if time.time() - start_time > budget_s - 60:
            break
        h_hi2 = best_h + rng.randn(new_n) * 0.02
        h_hi2 = project_feasible(h_hi2, new_n)
        h_hi2 = coord_descent_exact(h_hi2, max_passes=80, time_limit=budget_s, start_time=start_time)
        h_hi2 = project_feasible(h_hi2, new_n)
        c5_hi2 = compute_c5(h_hi2)
        if c5_hi2 < best_c5:
            best_c5 = c5_hi2
            best_h = h_hi2.copy()
    
    # Phase 5: SLSQP polish with log-sum-exp smoothing
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        n_pts = len(best_h)
        h_start = best_h.copy()
        dx = 2.0 / n_pts
        
        def slsqp_obj(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            mc = np.max(corr)
            t = 80.0
            shifted = corr - mc
            return (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx
        
        def slsqp_jac(h):
            corr = np.correlate(h, 1.0 - h, mode='full')
            mc = np.max(corr)
            t = 80.0
            shifted = corr - mc
            weights = np.exp(t * shifted)
            weights /= np.sum(weights)
            # gradient of mc + logsumexp(t*(corr-mc))/t wrt h
            # d corr[m]/d h[j] = A_j[m] (from our earlier derivation)
            # But computing full Jacobian is expensive; use numerical
            # For SLSQP, let's just use the function and let it approximate
            return None
        
        bounds = [(0.0, 1.0)] * n_pts
        target = n_pts / 2.0
        cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target, 'jac': lambda h: np.ones(n_pts)}]
        
        try:
            res = minimize(slsqp_obj, h_start, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 200, 'ftol': 1e-12})
            if res.success or hasattr(res, 'x'):
                h_res = project_feasible(res.x, n_pts)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
        except Exception:
            pass
    
    # Phase 6: Final coordinate descent polish
    if time.time() - start_time < budget_s - 10:
        n_pts = len(best_h)
        h_final = coord_descent_exact(best_h, max_passes=200, time_limit=budget_s, start_time=start_time)
        h_final = project_feasible(h_final, n_pts)
        c5_final = compute_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
