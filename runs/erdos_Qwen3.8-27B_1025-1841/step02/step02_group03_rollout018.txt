    return np.ones(_N)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:

        if diff > 0:
            mask = h < 1.0
        else:

        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    # Convert to binary
    h_bin = (best_h > 0.5).astype(float)
    # Fix count
    target = n // 2
    n_ones = int(np.sum(h_bin))
    if n_ones > target:
        # Remove excess ones (those with h closest to 0.5 from above)
        ones_idx = np.where(h_bin == 1)[0]
        sorted_by_val = ones_idx[np.argsort(best_h[ones_idx])]
        for idx in sorted_by_val[:n_ones - target]:
            h_bin[idx] = 0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        sorted_by_val = zeros_idx[np.argsort(-best_h[zeros_idx])]
        for idx in sorted_by_val[:target - n_ones]:
            h_bin[idx] = 1
    
    c5_bin = compute_c5(h_bin)
    if c5_bin < best_c5:
        best_h = h_bin
        best_c5 = c5_bin
    
    # Local search: try swaps
    for rnd in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        ones = np.where(best_h == 1)[0]
        zeros = np.where(best_h == 0)[0]
        n_trials = min(len(ones) * len(zeros), 500)
        for _ in range(n_trials):
            if time.time() > deadline:
                break
            i = rng.choice(ones)
            j = rng.choice(zeros)
            trial = best_h.copy()
            trial[i] = 0
            trial[j] = 1
            c = compute_c5(trial)
            if c < best_c5 - 1e-14:
                best_h = trial
                best_c5 = c
                improved = True
                ones = np.where(best_h == 1)[0]
                zeros = np.where(best_h == 0)[0]
        if not improved:
            break
    return best_h, best_c5

def coord_descent(h, n, max_rounds, deadline):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_v = old_val
            for d in [-0.15, -0.08, -0.04, -0.02, -0.01, 0.01, 0.02, 0.04, 0.08, 0.15]:
                cv = max(0.0, min(1.0, old_val + d))
                if abs(cv - old_val) < 1e-10:

                trial = best_h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = cv
            if best_v != old_val:
                best_h = best_h.copy()
                best_h[j] = best_v
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return best_h, best_c5

from scipy.optimize import minimize
import time

_N = 960
_T = 100.0
_DX = 2.0 / 960

def _obj(x):
    n = _N
    dx = _DX
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    mc = np.max(corr)
    z = _T * (corr - mc)
    z = np.clip(z, -50, 0)
    return mc + np.log(np.sum(np.exp(z))) / _T

def _con_f(x):
    return np.sum(x) - _N / 2.0

def _con_j(x):
    return np.ones(_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    best_h = h.copy()
