        for i in idx:

            else:
                adj = max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        if k < n:
            g[:n-k] += (1.0 - h[k:])
            g[k:] += -h[:n-k]
    else:
        s = -k
        if s < n:
            g[s:] += (1.0 - h[:n-s])
            g[:n-s] += -h[s:]
    return g

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            idx = idx[np.argsort(1.0 - h[idx])[::-1]]
        else:
            idx = np.where(h > 0.0)[0]
            idx = idx[np.argsort(h[idx])[::-1]]
        for i in idx:
            if diff > 0:
                adj = min(diff, 1.0 - h[i])
            else:
                adj = max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

def subgrad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if 0 <= k < n:
        g[:n-k] += (1.0 - h[k:])
        g[k:] += -h[:n-k]
    elif -n < k < 0:
        s = -k
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] += -h[s:]
    return g

def softmax_obj_grad(h, T, top_k=40):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    obj = (cmax + np.log(total) / T) * dx
    weights = exp_s / total
    top_idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for i in top_idx:
        w = weights[i]
        if w < 1e-10:
            break
        shift = i - (n - 1)
        grad += w * subgrad_shift(h, shift)
    grad *= dx
    return obj, grad

def nesterov_opt(h, n_iter, T_start, T_end, lr_start, lr_end, beta=0.9, 
                  top_k=40, check_every=20, sa_T=0.0):
    n = len(h)
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        _, grad = softmax_obj_grad(h, T, top_k)
        v = beta * v - lr * grad
        h_new = h + v
        h_new = make_feasible(h_new, n)
        
        if sa_T > 0 and it % check_every == 0:
            c_new = compute_c5(h_new)
            c_cur = compute_c5(h)
            if c_new < best_c5:
                best_c5 = c_new
                best_h = h_new.copy()
            # Simulated annealing: accept worse with probability

            else:

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            adj = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

def subgrad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if 0 <= k < n:
        g[:n-k] += (1.0 - h[k:])
        g[k:] += -h[:n-k]
    elif -n < k < 0:
        s = -k
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] += -h[s:]
    return g

def softmax_grad(h, T, top_k=40):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    top_idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for i in top_idx:
        w = weights[i]
        if w < 1e-10:
            break
        shift = i - (n - 1)
        grad += w * subgrad_shift(h, shift)
    grad *= dx
    return grad

    return np.sum(h) - _PARAMS['n'] / 2.0

def _eq_con_jac(h):
    return np.ones(_PARAMS['n'])

def nesterov_gd(h, n_iter, T_start, T_end, lr, beta=0.9, top_k=40, check_every=20):
    n = len(h)
    h = h.astype(np.float64).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.7 * frac)
        grad = softmax_grad(h, T, top_k)
        v = beta * v - lr_t * grad
        h_new = h + v
        h_new = make_feasible(h_new, n)
        h = h_new
        if (it + 1) % check_every == 0:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5

def upsample(h, new_n):

    new_x = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]  # most room first

            idx = np.argsort(h)[::-1]  # most value first

            diff -= adj
            if abs(diff) < 1e-15:

import numpy as np
from scipy.optimize import minimize
import time

_PARAMS = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=np.float64), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(6):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
        else:
            mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.argsort(1.0 - h)[::-1]
        else:
            idx = np.argsort(h)[::-1]
        for i in idx:
            adj = min(diff, 1.0 - h[i]) if diff > 0 else max(diff, -h[i])
            h[i] += adj
            diff -= adj
            if abs(diff) < 1e-14:
                break
    return h

def subgrad_shift(h, k):
    n = len(h)
    g = np.zeros(n)
    if 0 <= k < n:
        g[:n - k] += (1.0 - h[k:])
        g[k:] += -h[:n - k]
    elif -n < k < 0:
        s = -k
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] += -h[s:]
    return g

def softmax_grad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    top_idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for i in top_idx:
        w = weights[i]
        if w < 1e-11:
            break
        shift = i - (n - 1)
        grad += w * subgrad_shift(h, shift)
    grad *= dx
    return grad

def _sm_obj(h):
    n = _PARAMS['n']
    T = _PARAMS['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = corr - cmax
    return (cmax + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _sm_grad(h):
    return softmax_grad(h, _PARAMS['T'], top_k=_PARAMS['top_k'])

def _eq_con(h):
    return np.sum(h) - _PARAMS['n'] / 2.0

def _eq_con_jac(h):
    return np.ones(_PARAMS['n'])

def nesterov_gd(h, n_iter, T_start, T_end, lr, beta=0.9, top_k=50, check_every=15):
    n = len(h)
    h = h.astype(np.float64).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.7 * frac)
        grad = softmax_grad(h, T, top_k)
        v = beta * v - lr_t * grad
        h_new = h + v
        h_new = make_feasible(h_new, n)
        h = h_new
        if (it + 1) % check_every == 0:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
