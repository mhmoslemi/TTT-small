def project_to_feasible(g, n):

def fft_correlate(h, n):
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * np.conj(G), N)
    # Extract full correlation (2*n-1 elements)
    # corr[0] is zero-lag, corr[1:n-1] are positive lags
    # corr[N-n+1:N] are negative lags (reversed)
    full = np.concatenate([corr[N-n+1:], corr[:n]])
    return full  # length 2*n-1, same as np.correlate(h, 1-h, 'full')

    dx = 2.0 / n

    return corr.max() * dx, corr * dx

Wait, I realize there might be an issue with the FFT approach. Let me verify that the "wrap-around" doesn't cause problems.

With N ≥ 2n-1, the zero-padding ensures that the circular correlation equals the linear correlation for the first n and last n-1 elements (or vice versa). The wrap-around terms (which would be non-zero in a truly circular correlation) are zero because of the padding.

Specifically, full[m] = Σ_i h[i] * (1-h)[(i-m) mod N]
For m=0: Σ_{i=0}^{n-1} h[i]*(1-h)[i] ✓ (all indices in range)
For m=1: Σ_{i=0}^{n-1} h[i]*(1-h)[i-1] where (i-1) mod N:
  - For i=0: (1-h)[-1 mod N] = (1-h)[N-1] = 0 (since h is zero-padded)
  - For i=1 to n-1: (1-h)[i-1] ✓
  So full[1] = Σ_{i=1}^{n-1} h[i]*(1-h)[i-1] ✓

For m=N-1 (i.e., m=-1): Σ_{i=0}^{n-1} h[i]*(1-h)[(i+1) mod N]
  - For i=0 to n-2: (1-h)[i+1] ✓
  - For i=n-1: (1-h)[n mod N] = 0 (zero-padded)
  So full[N-1] = Σ_{i=0}^{n-2} h[i]*(1-h)[i+1] ✓

Great, so the FFT approach works correctly with zero-padding.

Now, for the gradient:
grad[j] = dx * [forward[j] - backward[j]]

where:
forward[j] = Σ_{d=0}^{n-1-j} p[d] * (1-h)[j+d]  (cross-correlation of p and (1-h))
backward[j] = Σ_{d=0}^{j} p[d] * h[j-d]  (convolution of p and h)

Using FFT:
forward = ifft(conj(fft(p, N)) * fft(1-h, N)).real[:n]
backward = ifft(fft(p, N) * fft(h, N)).real[:n]

Wait, I need to double-check the cross-correlation formula.

(a ⋆ b)[j] = Σ_k a[k] * b[k+j] (for valid indices)

In frequency domain: FFT(a ⋆ b) = conj(FFT(a)) * FFT(b) (for circular, with zero-padding for linear)

So: forward[j] = (p ⋆ (1-h))[j] = ifft(conj(fft(p, N)) * fft(1-h, N))[j]

And for convolution:
(a * b)[j] = Σ_k a[k] * b[j-k]
FFT(a * b) = FFT(a) * FFT(b)
backward[j] = (p * h)[j] = ifft(fft(p, N) * fft(h, N))[j]

So:
P = np.fft.rfft(p, N)
H_forw = np.fft.rfft(1.0 - h, N)
H_back = np.fft.rfft(h, N)
forward = np.fft.irfft(np.conj(P) * H_forw, N)[:n]
backward = np.fft.irfft(P * H_back, N)[:n]
grad = dx * (forward - backward)

This is O(N log N) per iteration!

Now let me put together the full algorithm. I'll use:
1. Start from initial_h_values, interpolate to n=8192
2. Soft-max optimization with increasing tau, using FFT-based gradient
3. Subgradient polish with top-K lags
4. Try higher resolutions

Let me also think about whether to use the full softmax (all lags) or top-M lags. For n=8192, the full softmax requires an FFT of size ~16384, which is very fast. The gradient computation also uses FFT. So we can afford to use ALL lags in the softmax, which gives a smoother objective and better gradients.

Let me write the complete code:

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def get_N(n):
    N = 1
    while N < 2 * n - 1:
        N *= 2
    return N


def fft_corr(h, N):
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(np.conj(H) * G, N)


def compute_c5(h, n):
    N = get_N(n)
    corr = fft_corr(h, N) * (2.0 / n)
    return corr.max()


def compute_gradient(h, p, n, N):
    dx = 2.0 / n
    P = np.fft.rfft(p, N)
    H1 = np.fft.rfft(1.0 - h, N)
    H0 = np.fft.rfft(h, N)
    forward = np.fft.irfft(np.conj(P) * H1, N)[:n]
    backward = np.fft.irfft(P * H0, N)[:n]
    return dx * (forward - backward)


def softmax_projected_gradient(h, n, tau, time_limit, lr_init=0.1, topM=0):
    N = get_N(n)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    
    corr = fft_corr(h, N) * dx
    best_obj = corr.max()
    best_h = h.copy()
    
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = 200
    
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        
        corr = fft_corr(h, N) * dx
        cur_max = corr.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute softmax weights
        if topM > 0 and topM < len(corr):
            top_idx = np.argpartition(corr, -topM)[-topM:]
            g_top = corr[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            exp_c = np.exp(shifted)
            p_top = exp_c / exp_c.sum()
            p = np.zeros(len(corr))
            p[top_idx] = p_top
        else:
            max_g = corr.max()
            shifted = np.clip(tau * (corr - max_g), -500, 500)
            exp_c = np.exp(shifted)
            p = exp_c / exp_c.sum()
        
        grad = compute_gradient(h, p, n, N)
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        
        h = h_new
    
    return best_h, best_obj


def subgradient_polish(h, n, time_limit, lr_init=0.005, topK=5):
    N = get_N(n)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    patience = 500
    
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        
        corr = fft_corr(h, N) * dx
        c5_cur = corr.max()
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - (n - 1)  # convert to lag
            # value(d) = sum_{i valid} h[i]*(1-h[i+d])*dx
            # gradient for this lag:
            # For d >= 0: d in [0, n-1] means lag index in [n-1, 2n-2]
            # For d < 0: d in [-(n-1), -1] means lag index in [0, n-2]
            
            # Actually, let me use the direct formula.
            # The lag corresponding to index ki in the full correlation:
            # full_corr[k] corresponds to lag d = k - (n-1)
            # For d >= 0 (k >= n-1): value = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])*dx
            #   d(v)/d(h[j]) = dx*(1-h[j+d]) for j <= n-1-d, and -dx*h[j-d] for j >= d
            # For d < 0 (k < n-1), let d' = -d > 0: value = sum_{i=d'}^{n-1} h[i]*(1-h[i-d'])*dx
            #   Let j = i: d(v)/d(h[j]) = dx*(1-h[j-d']) for j >= d', and -dx*h[j+d'] for j <= n-1-d'
            
            if d >= 0:
                j0 = 0
                j1 = n - d  # j from 0 to n-1-d
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx + d]) * dx
                j0b = d
                j1b = n  # j from d to n-1
                idx2 = np.arange(j0b, j1b)
                grad[idx2] -= h[idx2 - d] * dx
            else:
                dp = -d
                j0 = dp
                j1 = n  # j from dp to n-1
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - dp]) * dx
                j0b = 0
                j1b = n - dp  # j from 0 to n-1-dp
                idx2 = np.arange(j0b, j1b)
                grad[idx2] -= h[idx2 + dp] * dx
        
        cur_lr = lr / (1.0 + it * 0.003)
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break
        
        h = h_new
    
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    h_init = initial_h_values.copy()
    n_init = len(h_init)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))
    
    # Start at n=8192
    n = 8192
    x_old = np.arange(n_init) * (2.0 / n_init)
    x_new = np.arange(n) * (2.0 / n)
    h = np.interp(x_new, x_old, h_init, period=2.0)
    h = project_to_feasible(h, n)
    
    best_c5 = compute_c5(h, n)
    best_h = h.copy()
    
    # Phase 1: Soft-max optimization at n=8192 with increasing tau
    tau_schedule = [5, 10, 20, 50, 100, 200, 400]
    for tau in tau_schedule:
        if remaining() < 40:
            break
        tl = min(remaining() * 0.2, 150.0)
        h_cur, _ = softmax_projected_gradient(h, n, tau, tl, lr_init=0.15, topM=0)
        c5_cur = compute_c5(h_cur, n)
        if c5_cur < best_c5:

            best_h = h_cur.copy()
    
    # Phase 2: Subgradient polish
    if remaining() > 60:

        h_cur, c5_cur = subgradient_polish(h_cur, n, remaining() * 0.25, lr_init=0.004, topK=3)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
    
    # Phase 3: Try n=16384
    if remaining() > 120:
        n2 = 16384
        x_cur = np.arange(n) * (2.0 / n)

        h2 = np.interp(x_new2, x_cur, best_h, period=2.0)
        h2 = project_to_feasible(h2, n
