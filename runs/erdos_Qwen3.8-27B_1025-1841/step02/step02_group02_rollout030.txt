import numpy as np
import time
from scipy.signal import fftconvolve

def project_to_feasible(g, n):
    target = n / 2.0
    lo, hi = g.min() - 2.0, g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def softmax_obj_grad(h, tau, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    # Gradient using FFT
    T1 = fftconvolve(1.0 - h, p, mode='full')[n-1:2*n-1]
    T2 = np.correlate(h, p, mode='full')[n-1:2*n-1]
    grad = dx * (T1 - T2)
    return obj, grad

def optimize_adam(h, n, tau_schedule, time_limit, lr=0.1):
    """Optimize with Adam and adaptive tau."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0
    stall = 0
    best_stall = 0
    
    for tau in tau_schedule:
        if time.time() - start > time_limit:
            break
        tau_start = time.time()
        while time.time() - tau_start < time_limit * 0.1 and time.time() - start < time_limit:
            if time.time() - start > time_limit:
                break
            t += 1
            obj, grad = softmax_obj_grad(h, tau, dx)
            
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-15:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h_new = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
            
            if stall > 100:
                lr *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
            
            h = h_new
    
    return best_h, best_c5

One more thought: for the pair-swap local search, I need an efficient way to evaluate the change in C₅ when I swap two values. Instead of recomputing the full correlation (O(n²)), I can compute the change in each corr[k] analytically:

If I change h[i] by +δ and h[j] by -δ:
Δcorr[k] = δ * [(1-h[i-s_k])_{valid} - h[i+s_k]_{valid}] - δ * [(1-h[j-s_k])_{valid} - h[j+s_k]_{valid}]

where s_k = k-(n-1).

But I only care about whether the max decreases. So I need to know:
1. The current max value and which k achieves it
2. Whether the swap reduces the max

For a swap that reduces the active k's correlation, I need:
Δcorr[k*] < 0 where k* is the argmax.

This is: δ * [(1-h[i-s*]) - h[i+s*] - (1-h[j-s*]) + h[j+s*]] < 0 (with valid indices)
= δ * [(h[j-s*] - h[i-s*]) - (h[i+s*] - h[j+s*])] < 0

For δ > 0 (increasing h[i], decreasing h[j]):
Need: (h[j-s*] - h[i-s*]) < (h[i+s*] - h[j+s*])

This is a simple check! I can evaluate many candidate swaps cheaply.

But I also need to ensure no OTHER correlation increases above the current max. This is harder in general, but if the swap is small (δ is small), it's unlikely to cause another k to exceed the max.

For the local search, I'll use:
- Random pairs (i, j)
- Small δ (e.g., 0.1 or 0.05)
- Accept if it reduces the max correlation
- Also try to reduce the "second max" if the max is already at a plateau

Actually, for simulated annealing style, I'll accept moves that don't increase the max by more than a temperature T, with T decreasing over time.

import numpy as np
import time
from scipy.signal import fftconvolve


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def fft_gradient(h, p, dx):
    n = len(h)
    T1 = fftconvolve(1.0 - h, p, mode='full')[n - 1:2 * n - 1]
    T2 = np.correlate(h, p, mode='full')[n - 1:2 * n - 1]
    return dx * (T1 - T2)


def softmax_obj_grad(h, tau, dx):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = fft_gradient(h, p, dx)
    return obj, grad, corr


def optimize_adam_fft(h, n, time_limit, tau_schedule=None, base_lr=0.1):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if tau_schedule is None:
        tau_schedule = [30.0, 20.0, 15.0, 10.0, 8.0, 6.0]
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    t = 0
    lr = base_lr

    for tau in tau_schedule:
        if time.time() - start > time_limit:
            break
        tau_frac = 1.0 / len(tau_schedule)
        tau_budget = time_limit * tau_frac
        tau_start = time.time()
        stall = 0
        while time.time() - tau_start < tau_budget and time.time() - start < time_limit:
            t += 1
            obj, grad, corr = softmax_obj_grad(h, tau, dx)
            c5_cur = corr.max() * dx
            if c5_cur < best_c5 - 1e-15:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1

            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad ** 2
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_to_feasible(h - step, n)

            if stall > 60:
                lr *= 0.5
                m *= 0.5
                stall = 0
                if lr < 1e-12:
                    break

    return best_h, best_c5


def optimize_subgrad_fft(h, n, time_limit, lr=0.02):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    stall = 0

    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        s = k_star - (n - 1)
        grad = np.zeros(n)
        # (1-h[i-s]) for valid i-s
        j0 = max(0, s)
        j1 = min(n, n + s)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[np.arange(j0, j1) - s])
        # -h[i+s] for valid i+s
        j0b = max(0, -s)
        j1b = min(n, n - s)
        if j0b < j1b:
            grad[j0b:j1b] -= h[np.arange(j0b, j1b) + s]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

        if stall > 200:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break

    return best_h, best_c5


def local_search_pairs(h, n, time_limit, rng):
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    delta = 0.1

    while time.time() - start < time_limit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if h[i] < delta or h[j] < delta:
            continue
        # Try h[i] += delta, h[j] -= delta
        h[i] += delta
        h[j] -= delta
        corr = compute_corr(h)
        c5_new = corr.max() * dx
        if c5_new <= best_c5 + 1e-15:
            best_c5 = min(best_c5, c5_new)
            best_h = h.copy()
        else:
            h[i] -= delta
            h[j] += delta
        # Also try reverse
        h[i] -= delta
        h[j] += delta
        corr = compute_corr(h)
        c5_new = corr.max() * dx
        if c5_new <= best_c5 + 1e-15:
            best_c5 = min(best_c5, c5_new)
            best_h = h.copy()
        else:
            h[i] += delta
            h[j] -= delta

    return best_h, best_c5


def local_search_smart(h, n, time_limit, rng):
    dx = 2.0 / n
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    delta = 0.05
    accepted = 0

    while time.time() - start < time_limit:
        # Focus on pairs that can reduce the active correlation
        k_star = np.argmax(corr)
        s = k_star - (n - 1)

        # Find i where increasing h[i] helps (reduces corr at k_star)
        # and j where decreasing h[j] helps
        # delta_corr at k_star for h[i]+=d: d*(1-h[i-s]) - d*h[i+s]
        # For this to be negative: (1-h[i-s]) < h[i+s], i.e., h[i+s] > h[i-s] (roughly)

        # Random approach: sample i, j and check
        batch = 50
        indices = rng.randint(0, n, size=batch)
        for i in indices:
            if h[i] < delta:
                continue
            for _ in range(3):
                j = rng.randint(n)
                if j == i or h[j] < delta:
                    continue
                # Evaluate: h[i]+=delta, h[j]-=delta
                old_i, old_j = h[i], h[j]
                h[i] = old_i + delta
                h[j] = old_j - delta
                corr_new = compute_corr(h)
                c5_new = corr_new.max() * dx
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h.copy()
                    corr = corr_new
                    accepted += 1
                else:
                    h[i] = old_i
                    h[j] = old_j

        # Occasionally try larger moves
        if accepted > 50:
            delta *= 0.7

    return best_h, best_c5
