```python
def compute_gradient(h, tau, top_k=20):
    n = len(h)
    dx = 2.0/n
    corr = np.correlate(h, 1.0-h, mode='full')
    max_c = corr.max()
    # Compute softmax weights
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -500, 50)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # Only use top-k shifts
    top_idx = np.argsort(p)[::-1][:top_k]
    top_p = p[top_idx]
    top_p = top_p / top_p.sum()  # renormalize
    
    # Gradient
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        sh = k - n + 1
        # Term 1: (1-h[j+sh]) for valid j+sh
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += top_p[i] * (1.0 - h[j0+sh:j1+sh])
        # Term 2: -h[j-sh] for valid j-sh  
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= top_p[i] * h[j0b-sh:j1b-sh]
    
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad
```

def smooth_grad(h, tau, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 50)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # Top-k shifts
    top_idx = np.argsort(p)[::-1][:top_k]
    top_p = p[top_idx]
    top_p = top_p / top_p.sum()
    
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        pk = top_p[i]
        sh = k - n + 1
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0+sh:j1+sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b-sh:j1b-sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, corr
