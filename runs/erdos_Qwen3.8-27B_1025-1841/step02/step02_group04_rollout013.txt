import numpy as np
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_objective(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Use log-sum-exp with stabilization
    max_shifted = np.max(shifted)  # 0
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def softmax_grad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n-s] += 1.0 - h[s:]  # d/dh[i] of h[i]*(1-h[i+s])
            grad[s:] += -h[:n-s]        # d/dh[i+s] of h[i]*(1-h[i+s])
        else:
            s2 = -s
            grad[s2:] += 1.0 - h[:n-s2]  # d/dh[i+s2] of h[i+s2]*(1-h[i])
            grad[:n-s2] += -h[s2:]       # d/dh[i] of h[i+s2]*(1-h[i])
        return grad * dx
    
    weights = exp_s / total
    # Get top_k weighted shifts
    idx = np.argsort(weights)[::-1][:top_k]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-8:
            break
        s = k_idx - (n - 1)
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] += w * (-h[:n-s])
        else:
            s2 = -s
            grad[s2:] += w * (1.0 - h[:n-s2])
            grad[:n-s2] += w * (-h[s2:])
    return grad * dx

def gd_optimize(h, n_iter=3000, T_start=50.0, T_end=0.5, lr_start=0.15, lr_end=0.01, check_every=30):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        grad = softmax_grad(h, T)
        h = h - lr * grad
        h = make_feasible(h, n)
        
        if (it + 1) % check_every == 0:

                best_h = h.copy()

    return best_h, best_c5

def sqp_refine(h0, n, max_iter=30, top_k=20):
    """SQP: linearize constraints at current point, solve LP"""
    dx = 2.0 / n
    h = h0.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iter):
        corr = np.correlate(h, 1.0 - h, mode='full')
        
        # Select tightest constraints
        idx = np.argsort(corr)[-top_k:]
        
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        constraints = [
            cp.sum(h_var) == n / 2.0,
            h_var >= 0,
            h_var <= 1,
        ]
        
        for k in idx:
            s = k - (n - 1)
            if s >= 0:
                # f = sum_{i=0}^{n-1-s} h[i]*(1-h[i+s])
                # lin: sum (1-h0[i+s])*h[i] - h0[i]*h[i+s] + const
                a = 1.0 - h[s:]      # coefficients for h[0..n-1-s]

                const = np.dot(h[:n-s], h[s:])
                
                expr = a @ h_var[:n-s] + b @ h_var[s:]
                constraints.append(expr * dx + const * dx <= t_var)
            else:
                s2 = -s
                # f = sum_{i=0}^{n-1-s2} h[i+s2]*(1-h[i])
                a = -h[s2:]          # coefficients for h[0..n-1-s2] (the -h[i+s2]*(1)... wait

                # const = sum h0[i]*h0[i+s2]
                a = -h[s2:]          # for h[0..n-1-s2]
                b = 1.0 - h[:n-s2]  # for h[s2..n-1]

                expr = a @ h_var[:n-s2] + b @ h_var[s2:]

        prob = cp.Problem(cp.Minimize(t_var), constraints)

            except Exception:

                except Exception:
                    break

            break

        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()

                break
        h = h_new
    
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_grad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        s = k_max - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            s2 = -s
            grad[s2:] += (1.0 - h[:n-s2])
            grad[:n-s2] += (-h[s2:])
        return grad * dx
    weights = exp_s / total
    idx = np.argsort(weights)[::-1][:top_k]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-8:
            break
        s = k_idx - (n - 1)
        if s >= 0:
