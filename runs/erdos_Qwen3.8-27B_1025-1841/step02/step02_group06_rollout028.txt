def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def compute_grad_toph(h, top_idx, n, dx, p=None):
    """Compute gradient from top peaks only."""
    n = len(h)
    grad = np.zeros(n)
    m = len(top_idx)
    if p is None:
        p = np.ones(m) / m
    for i in range(m):
        pk = p[i]
        if pk < 1e-10:
            continue
        d = top_idx[i] - (n - 1)
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx

                     sigma_smooth=0.0, momentum=0.85):

    g = corr * dx

    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = momentum
    stall = 0
    patience = max(50, max_iters // 10)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -300, 300)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p_top = exp_c / s
            grad = compute_grad_toph(h, top_idx, n, dx, p_top)
        else:
            top_idx = np.arange(2 * n - 1)

This is non-convex because of the h(1-h) term. However, we can introduce a variable u_k = ∫ h(x)(1-h(x+k)) dx and try to minimize max_k u_k.

Alternatively, we can use the fact that for h ∈ [0,1]:
h(x)(1-h(x+k)) ≤ h(x) * (1-h(x+k))

And we know that h(1-h') ≤ min(h, 1-h'). This gives an upper bound but not a tight formulation.

Another approach: Since h ∈ [0,1], we can write h = 1/2 + f where f ∈ [-1/2, 1/2].
Then h(x)(1-h(x+k)) = (1/2+f(x))(1/2-f(x+k)) = 1/4 + f(x)/2 - f(x+k)/2 - f(x)f(x+k)

The integral becomes:
∫ h(x)(1-h(x+k)) dx = 1/4 * (measure of domain) + 1/2∫f(x)dx - 1/2∫f(x+k)dx - ∫f(x)f(x+k)dx

If we work on [0,2] with periodic extension, ∫f = 0 and ∫f(x+k) = ∫f = 0, so:
= 1/2 - ∫ f(x)f(x+k) dx (where the 1/4*2 = 1/2 comes from the domain length of 2)

Wait, that's not right. The domain of integration depends on k. Let me not go down this path.

Let me instead try a **penalty method**: minimize t + λ * Σ_k (g_k - t)₊² where g_k are the correlation values. This makes the problem smooth and we can use standard optimization.

Actually, I think the most impactful improvement would be to:
1. Use a better initial condition (not just the parent)
2. Use a multi-start approach
3. Be more aggressive in the final local optimization

Let me also try a **simulated annealing** approach on the "temperature" parameter τ. Start with high τ (broad focus) and gradually decrease to low τ (narrow focus on worst peak).

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def compute_grad_toph(h, top_idx, n, dx, p=None):
    n = len(h)
    grad = np.zeros(n)
    m = len(top_idx)
    if p is None:
        p = np.ones(m) / m
    for i in range(m):
        pk = p[i]

        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx

def optimize_anneal(h, n, tau_start, tau_end, max_iters, time_limit, 
                    lr_init=0.12, topM=200, momentum=0.85):
    """Optimization with annealing on tau (temperature)."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = momentum
    stall = 0
    patience = max(50, max_iters // 10)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # Anneal tau from tau_start to tau_end
        progress = min(1.0, it / max_iters)
        tau = tau_start * (tau_end / tau_start) ** progress
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()

            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -300, 300)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p_top = exp_c / s
        else:

            max_g = g.max()

            lr *= 0.75
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
    
    return best_h, best_obj
