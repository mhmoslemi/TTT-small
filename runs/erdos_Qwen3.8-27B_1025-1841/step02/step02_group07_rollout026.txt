def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    """Project y onto {0≤h≤1, ∑h=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y + mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y + (lo + hi) / 2, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute remaining difference
        free = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(free):
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

    corr = np.correlate(h, 1.0 - h, mode='full')
    dx = 2.0 / len(h)

    m = np.max(scaled)
    exp_vals = np.exp(scaled - m)
    logsum = np.log(np.sum(exp_vals))
    obj = (m + logsum) / T * dx
    weights = exp_vals / np.sum(exp_vals)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(y, n):
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(y + (lo + hi) / 2, 0.0, 1.0)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    ev = np.exp(scaled - m)
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    for c in range(2 * n - 1):
        wc = w[c]
        if wc < 1e-9:
            continue
        lag = c - (n - 1)
        if lag >= 0:
            grad[:n - lag] += wc * (1.0 - h[lag:])
            grad[lag:] -= wc * h[:n - lag]
        else:
            grad[-lag:] += wc * (1.0 - h[:n + lag])
            grad[:n + lag] -= wc * h[-lag:]
    return grad * dx

def lbfgs_opt(h, n, T, maxiter, deadline):
    def f(x):
        return smooth_obj(x, T)
    def g(x):
        return smooth_grad(x, T)
    try:
        res = minimize(f, h, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-18,
                                'gtol': 1e-14, 'maxls': 50})
        if time.time() > deadline:
            return h
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

    h = h.copy()
    bh = h.copy()
    bv = evaluate_c5(h)
    alpha = 0.002
    stall = 0
    for s in range(steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        mx = np.max(corr)
        thresh = mx - 0.015
        active = np.where(corr >= thresh)[0]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        dx = 2.0 / n
        grad = np.zeros(n)
        for c in active:
            lag = c - (n - 1)
            t = np.zeros(n)
            if lag >= 0:
                t[:n - lag] = 1.0 - h[lag:]
                t[lag:] = -h[:n - lag]
            else:
                t[-lag:] = 1.0 - h[:n + lag]
                t[:n + lag] = -h[-lag:]
            grad += t
        grad *= dx / len(active)
        h_new = project_feasible(h - alpha * grad, n)
        v = evaluate_c5(h_new)
        if v < bv - 1e-10:
            bv = v
            bh = h_new.copy()
            alpha = min(alpha * 1.03, 0.02)
            stall = 0
        else:
            stall += 1
            alpha *= 0.85
            if stall > 50:
                alpha = 0.001
                stall = 0
        h = h_new
    return bh, bv
