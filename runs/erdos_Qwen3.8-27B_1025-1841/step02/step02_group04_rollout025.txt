                new_corr = np.correlate(h, 1.0 - h, mode='full')
                new_max = np.max(new_corr)
                gain = cur_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_swap = (i, j)
                h[i] = 1.0
                h[j] = 0.0
            
            if best_swap is not None:
                i, j = best_swap
                h[i] = 0.0
                h[j] = 1.0
                # Update the sets
                idx_i = np.where(ones == i)[0]
                idx_j = np.where(zeros == j)[0]
                ones = np.delete(ones, idx_i[0])
                zeros = np.delete(zeros, idx_j[0])
                ones = np.append(ones, j)
                zeros = np.append(zeros, i)
                
                corr = np.correlate(h, 1.0 - h, mode='full')
                cur_max = np.max(corr)
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
            else:
                break  # No improvement found
        else:
            # Random swap
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = np.max(new_corr)
            
            if new_max <= cur_max:
                # Accept (even if equal, for exploration)
                idx_i = np.where(ones == i)[0]
