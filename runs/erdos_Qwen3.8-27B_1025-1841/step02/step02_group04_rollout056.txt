import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_step(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def soft_gradient(h, T, top_k=60):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    if T < 1e-4:
        idx = np.where(corr == mc)[0]
        idx = idx[0]
        shift = idx - (n - 1)
        return subgrad_step(h, shift) * dx
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        return subgrad_step(h, shift) * dx
    weights = exp_s / total
    if top_k is not None and top_k < len(weights):
        thresh = np.sort(weights)[-top_k]
        weights[weights < thresh] = 0.0
        s = np.sum(weights)
        if s > 0:
            weights /= s
    grad = np.zeros(n)
    nz = weights > 1e-10
    for k_idx in np.where(nz)[0]:
        grad += weights[k_idx] * subgrad_step(h, k_idx - (n - 1))
    return grad * dx

def adam_optimize(h, n_iter, T_start, T_end, lr=0.05, top_k=60, check_every=30):
    n = len(h)
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = (it + 1) / n_iter
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 + 0.5 * (1.0 - frac))
        grad = soft_gradient(h, T, top_k)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

# SLSQP objective and constraints (top-level, no closures)
_OPT_T = 5.0
_OPT_n = 320

def _obj(h):
    n = _OPT_n
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _grad_obj(h):
    n = _OPT_n
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    grad = np.zeros(n)
    for k_idx in range(len(weights)):
        w = weights[k_idx]
        if w < 1e-10:
            continue
        grad += w * subgrad_step(h, k_idx - (n - 1))
    return grad * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_n / 2.0

def _cons_jac(h):
    return np.ones(_OPT_n)

def slsqp_optimize(h, T_val, maxiter=500):
    n = len(h)
    global _OPT_T, _OPT_n
    _OPT_T = T_val
    _OPT_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=_grad_obj,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            cx = compute_c5(hx)
            return hx, cx
    except Exception:
        pass
    return h, compute_c5(h)

def upsample_h(h, new_n):

    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h_p = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(h_p)
        if s < target:
            lo = mid
        else:
            hi = mid
    h_p = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(h_p)
    if abs(diff) > 1e-13:
        mask = (h_p > 1e-12) & (h_p < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h_p[mask] += diff / cnt
            h_p = np.clip(h_p, 0.0, 1.0)
    return h_p

def subgrad_step(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    return grad

def soft_gradient(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc
    if T < 1e-3:
        idx = int(np.argmax(corr))
        return subgrad_step(h, idx - (n - 1)) * dx
    exp_s = np.exp(T * shifted)
    total = float(np.sum(exp_s))
    if total < 1e-300:
        idx = int(np.argmax(corr))
        return subgrad_step(h, idx - (n - 1)) * dx
    weights = exp_s / total
    if top_k is not None and top_k < len(weights):
        order = np.argsort(weights)
        weights[order[:len(weights) - top_k]] = 0.0
        s = float(np.sum(weights))
        if s > 0:
            weights /= s
    grad = np.zeros(n)
    nz = np.where(weights > 1e-10)[0]
    for k in nz:
        grad += weights[k] * subgrad_step(h, int(k) - (n - 1))
    return grad * dx

def adam_opt(h, n_iter, T_start, T_end, lr=0.05, top_k=80, check_every=30):
    n = len(h)
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = (it + 1) / n_iter
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 + 0.5 * (1.0 - frac))
        grad = soft_gradient(h, T, top_k)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** (it + 1))
        v_hat = v / (1.0 - b2 ** (it + 1))
        h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = project_h(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

_O_T = 5.0
_O_n = 320

def _obj(h):
    n = _O_n
    T = _O_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _obj_grad(h):
    n = _O_n
    T = _O_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = float(np.sum(exp_s))
    weights = exp_s / total
    grad = np.zeros(n)

    for k in nz:
        grad += weights[k] * subgrad_step(h, int(k) - (n - 1))
    return grad * dx

def _cons_fun(h):
    return np.sum(h) - _O_n / 2.0

def _cons_jac(h):
    return np.ones(_O_n)

def slsqp_opt(h, T_val, maxiter=500):
    global _O_T, _O_n
    n = len(h)
    _O_T = T_val
    _O_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=_obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = project_h(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_h(h_new, new_n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_h(h, n):
    target = n / 2.0
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        hp = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(hp)
        if s < target:
            lo = mid
        else:
            hi = mid
    hp = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - np.sum(hp)
    if abs(diff) > 1e-13:
        mask = (hp > 1e-12) & (hp < 1.0 - 1e-12)
        cnt = int(np.sum(mask))
        if cnt > 0:
            hp[mask] += diff / cnt
            hp = np.clip(hp, 0.0, 1.0)
    return hp

def subgrad_step(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        g[:n - s] += (1.0 - h[s:])
        g[s:] -= h[:n - s]
    return g

def soft_gradient(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    sh = corr - mc
    if T < 1e-3:
        idx = int(np.argmax(corr))
        return subgrad_step(h, idx - (n - 1)) * dx
    es = np.exp(T * sh)
    tot = float(np.sum(es))
    if tot < 1e-300:
        idx = int(np.argmax(corr))
        return subgrad_step(h, idx - (n - 1)) * dx
    w = es / tot
    if top_k is not None and top_k < len(w):
        order = np.argsort(w)
        w[order[:len(w) - top_k]] = 0.0
        s = float(np.sum(w))
        if s > 0:
            w /= s
    g = np.zeros(n)
    nz = np.where(w > 1e-11)[0]
    for k in nz:
        g += w[k] * subgrad_step(h, int(k) - (n - 1))
    return g * dx

    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = (it + 1) / n_iter
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 + 0.5 * (1.0 - frac))
        grad = soft_gradient(h, T, top_k)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        mh = m / (1.0 - b1 ** (it + 1))
        vh = v / (1.0 - b2 ** (it + 1))
        h = h - lr_t * mh / (np.sqrt(vh) + eps)
        h = project_h(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

_O_T = 5.0
_O_n = 320

def _obj(h):
    n = _O_n
    T = _O_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    sh = corr - mc
    return (mc + np.log(np.sum(np.exp(T * sh))) / T) * dx

def _obj_grad(h):
    n = _O_n
    T = _O_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = float(np.max(corr))
    sh = corr - mc
    es = np.exp(T * sh)
    tot = float(np.sum(es))
    w = es / tot
    g = np.zeros(n)
    nz = np.where(w > 1e-11)[0]
    for k in nz:
        g += w[k] * subgrad_step(h, int(k) - (n - 1))
    return g * dx

def _cons_fun(h):
    return np.sum(h) - _O_n / 2.0

def _cons_jac(h):
    return np.ones(_O_n)

def slsqp_opt(h, T_val, maxiter=500):
    global _O_T, _O_n
    n = len(h)
    _O_T = T_val
    _O_n = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, jac=_obj_grad,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = project_h(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_h(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n_init = 320
        h = (rng.rand(n_init) < 0.5).astype(float)
        tgt = n_init // 2
        ones_i = np.where(h == 1)[0]
        zeros_i = np.where(h == 0)[0]
        if len(ones_i) > tgt:
            h[rng.choice(ones_i, len(ones_i) - tgt, replace=False)] = 0.0
        elif len(ones_i) < tgt:
            h[rng.choice(zeros_i, tgt - len(ones_i), replace=False)] = 1.0

    n0 = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    print(f"Init n={n0} C5={best_c5:.6f}", flush=True)

    # Phase 1: Adam at initial resolution
    if budget_s - (time.time() - t0) > 120:
        h1, c1 = adam_opt(best_h, 5000, 50.0, 0.5, lr=0.05, top_k=80, check_every=25)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
    print(f"P1 n={n0} C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 2: SLSQP at initial resolution
    if n0 <= 400 and budget_s - (time.time() - t0) > 80:
        for Tv in [10.0, 3.0, 0.8]:
            if budget_s - (time.time() - t0) < 30:
                break
            hx, cx = slsqp_opt(best_h, Tv, 500)
            if cx < best_c5:
                best_c5, best_h = cx, hx
    print(f"P2 n={n0} C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 3: Upsample and optimize
    for mult in [2, 3]:
        new_n = n0 * mult
        if new_n > 1200 or budget_s - (time.time() - t0) < 130:
            break
        h_up = upsample_h(best_h, new_n)
        if budget_s - (time.time() - t0) > 90:
            h_up, _ = adam_opt(h_up, 3500, 30.0, 0.3, lr=0.04, top_k=60, check_every=30)
        if new_n <= 500 and budget_s - (time.time() - t0) > 50:
            h_up, _ = slsqp_opt(h_up, 2.0, 400)
        cu = compute_c5(h_up)
        if cu < best_c5:
            best_c5, best_h = cu, h_up
        print(f"P3 n={new_n} C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 4: Multi-start perturbation
    np_ = len(best_h)
    for ps in [0.02, 0.06, 0.12]:
        if budget_s - (time.time() - t0) < 80:
            break
        hp = best_h + rng.randn(np_) * ps
        hp = project_h(hp, np_)
        hr, cr = adam_opt(hp, 2500, 20.0, 0.3, lr=0.04, top_k=60, check_every=40)
        if cr < best_c5:
            best_c5, best_h = cr, hr
            print(f"  MS {ps}: C5={best_c5:.6f}", flush=True)
    print(f"P4 C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 5: Final SLSQP
    np_ = len(best_h)
    if np_ <= 500 and budget_s - (time.time() - t0) > 40:
        for Tv in [1.0, 0.3]:
            if budget_s - (time.time() - t0) < 20:
                break
            hx, cx = slsqp_opt(best_h, Tv, 500)
            if cx < best_c5:
                best_c5, best_h = cx, hx
    print(f"P5 C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 6: Final fine Adam
    if budget_s - (time.time() - t0) > 40:
        hf, cf = adam_opt(best_h, 3000, 1.0, 0.01, lr=0.01, top_k=40, check_every=20)
        if cf < best_c5:
            best_c5, best_h = cf, hf
    print(f"P6 C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    # Phase 7: Extra SLSQP pass if time allows
    np_ = len(best_h)
    if np_ <= 512 and budget_s - (time.time() - t0) > 30:
        hx, cx = slsqp_opt(best_h, 0.5, 400)
        if cx < best_c5:
            best_c5, best_h = cx, hx
    print(f"P7 C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)

    best_h = project_h(best_h, len(best_h))
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5
    print(f"DONE n={len(best_h)} C5={best_c5:.6f} t={time.time()-t0:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
