            adj = min(adj, max_adj.min() if len(max_adj) > 0 else adj)

        else:

            adj = max(adj, -max_adj.min() if len(max_adj) > 0 else adj)

def softmax_grad(h, T):
    """Compute gradient of soft-max objective"""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    exp_shifted = np.exp(T * (corr - mc))
    w = exp_shifted / np.sum(exp_shifted)
    
    n_half = n - 1
    A = np.zeros(n)
    B = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - n_half
        if lag >= 0:
            if n - lag > 0:
                A[:n-lag] += wk * h[lag:]
            B[lag:] += wk * h[:n-lag]
        else:
            ll = -lag
            if n - ll > 0:
                A[ll:] += wk * h[:n-ll]
            B[:n-ll] += wk * h[ll:]
    
    grad = dx * (1.0 - A - B)
    return grad

    h = h0.copy()
    h_prev = h0.copy()
    beta = 0.0
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    step = step0
    
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        
        # Nesterov momentum
        beta_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * beta * beta))
        y = h + (beta / beta_new - 1.0) * (h - h_prev)
        y = project_feasible(y, n)

            best_c5 = c5_new

            h_new = h  # backtrack
            c5_new = compute_c5(h_new)
        
        h_prev = h
        h = h_new
        beta = beta_new
    
    return best_h, best_c5

Wait, I need to double-check the Nesterov implementation. The standard Nesterov accelerated gradient:
- x_0, x_{-1} = x_0, x_0
- For k = 0, 1, 2, ...:
  - t_k = (1 + sqrt(1 + 4*t_{k-1}²)) / 2
  - y_k = x_k + ((t_{k-1} - 1) / t_k) * (x_k - x_{k-1})
  - x_{k+1} = proj(y_k - (1/L) * grad(y_k))

Where L is the Lipschitz constant of the gradient. For our problem, L depends on T and n.

Actually, for a projected gradient method with momentum, a simpler approach is:
- v_0 = 0
- For k:
  - y = x + v (momentum)
  - g = grad(y)
  - x_new = proj(x - step * g)
  - v = (x_new - x) * momentum_factor

corr[k] = Σ_i h[i] · (1 - h[i + k - (n-1)])

For a fixed k, d(corr[k])/d(h[j]):
- Term 1: h[j] appears in h[i] when i=j: contributes (1 - h[j + k - (n-1)])
- Term 2: h[j] appears in (1-h[i+k-(n-1)]) when i+k-(n-1) = j, i.e., i = j - k + (n-1): contributes -h[j - k + (n-1)]

So d(corr[k])/d(h[j]) = (1 - h[j + lag_k]) - h[j - lag_k]

where lag_k = k - (n-1), and we interpret h[m] = 0 for m outside [0, n-1].

The soft-max gradient:
d/dh[j] [Σ_k w_k · corr[k]] = Σ_k w_k · [(1 - h[j+lag_k]) - h[j-lag_k]]

= Σ_k w_k - Σ_k w_k · h[j+lag_k] - Σ_k w_k · h[j-lag_k]

= 1 - A[j] - B[j]

where A[j] = Σ_k w_k · h[j+k-(n-1)] and B[j] = Σ_k w_k · h[j-k+(n-1)]

Note that B[j] = Σ_k w_k · h[j-(k-(n-1))] = Σ_l w_{l+(n-1)} · h[j-l] where l = k-(n-1)

And A[j] = Σ_l w_{l+(n-1)} · h[j+l]

So A and B are "cross-correlations" of h with the weight sequence (one in each direction).
