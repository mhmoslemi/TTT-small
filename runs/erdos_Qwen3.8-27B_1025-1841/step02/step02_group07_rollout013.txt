    corr = np.correlate(hf, 1.0 - hf, mode='full')
    sm = _softmax_val(corr, T) * (2.0 / nf)
    s = np.sum(x) - nh / 2.0
    return sm + pen * s * s

def obj_full(x, T, pen):
    n = len(x)
    corr = np.correlate(x, 1.0 - x, mode='full')
    sm = _softmax_val(corr, T) * (2.0 / n)
    s = np.sum(x) - n / 2.0
    return sm + pen * s * s

def project(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        d = target - np.sum(h)
        if abs(d) < 1e-15:
            break
        mask = h < 1.0 if d > 0 else h > 0.0
        c = np.sum(mask)
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def project_sym(hr, nh):
    t = nh / 2.0
    h = np.clip(hr, 0.0, 1.0)
    for _ in range(1000):
        d = t - np.sum(h)
        if abs(d) < 1e-15:
            break
        mask = h < 1.0 if d > 0 else h > 0.0
        c = np.sum(mask)
        if c == 0:
            break
        h[mask] += d / c
        h = np.clip(h, 0.0, 1.0)
    return h

def expand_sym(hr):
    return np.concatenate([hr, hr[::-1]])

def symmetrize(hf):
    nh = len(hf) // 2
    return 0.5 * (hf[:nh] + hf[nh:][::-1])

def interp_fn(h, nf, nt):
    if nf == nt:
        return h.copy()
    xf = np.linspace(0, 2, nf, endpoint=False)
    xt = np.linspace(0, 2, nt, endpoint=False)
    return np.interp(xt, xf, h)

def subgrad_polish(h, n, lr, n_steps, deadline):
    target = n / 2.0
    h = h.copy()
    for _ in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k = np.argmax(corr)
        shift = k - (n - 1)
        g = np.zeros(n)
        if shift >= 0:
            i0, i1 = 0, n - shift
        else:
            i0, i1 = -shift, n
        if i1 > i0:
            g[i0:i1] += 1.0
            vi = np.arange(i0, i1)
            j1 = vi
            j2 = vi + shift
            m1 = (j2 >= 0) & (j2 < n)
            g[j1[m1]] -= h[j2[m1]]
            g[j2[m1]] -= h[j1[m1]]
        g += 200.0 * (np.sum(h) - target)
        h = project(h - lr * g, n)
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_init = np.array(initial_h_values, dtype=float).copy()
    else:
        h_init = np.full(336, 0.5)

    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(best_h)

    # Symmetric start
    h_red = symmetrize(h_init)
    h_red = project_sym(h_red, n // 2)
    hs = expand_sym(h_red)
    cs = evaluate_c5(hs)
    if cs < best_c5:
        best_c5 = cs
        best_h = hs.copy()

    PEN = 1e5

    # Phase 1: Symmetric L-BFGS-B annealing
    for T in [100.0, 60.0, 40.0, 25.0, 18.0, 12.0, 8.0, 6.0, 4.0, 3.0]:
        if time.time() > deadline - 40:
            break
        res = minimize(obj_sym, h_red, args=(T, PEN), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * (n // 2),
                       options={'maxiter': 400, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None:
            h_red = project_sym(res.x, n // 2)
            ht = expand_sym(h_red)
            c5 = evaluate_c5(ht)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ht.copy()

    # Phase 2: Full-space L-BFGS-B
    hf = best_h.copy()
    for T in [40.0, 25.0, 15.0, 10.0, 7.0, 5.0, 3.0]:
        if time.time() > deadline - 40:
            break
        res = minimize(obj_full, hf, args=(T, PEN), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': 300, 'ftol': 1e-15, 'gtol': 1e-10})
        if res.x is not None:
            hf = project(res.x, n)
            c5 = evaluate_c5(hf)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hf.copy()

    # Phase 3: Subgradient polish on best
    h_sub = subgrad_polish(best_h, n, 0.002, 500, deadline)
    c5 = evaluate_c5(h_sub)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_sub.copy()

    # Phase 4: Random restarts (symmetric)
    nh = n // 2
    for attempt in range(50):
        if time.time() > deadline - 25:
            break
        sc = 0.015 + 0.07 * np.sqrt(attempt / 50.0)
        hp = h_red + np.random.randn(nh) * sc
        hp = project_sym(hp, nh)
        res = minimize(obj_sym, hp, args=(8.0, PEN), method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * nh,
                       options={'maxiter': 120, 'ftol': 1e-13})
        if res.x is not None:
            hn = project_sym(res.x, nh)
            hfn = expand_sym(hn)
            c5 = evaluate_c5(hfn)
            if c5 < best_c5:
                best_c5 = c5
                best_h = hfn.copy()
                h_red = hn.copy()

    # Phase 5: Higher resolution
    for nn in [504, 672]:
        if time.time() > deadline - 80:
            break
        hi = interp_fn(best_h, n, nn)
        hi = project(hi, nn)
        hr = symmetrize(hi)
        hr = project_sym(hr, nn // 2)
        for T in [30.0, 15.0, 10.0, 6.0, 4.0]:
            if time.time() > deadline - 40:
                break
            res = minimize(obj_sym, hr, args=(T, PEN), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * (nn // 2),
                           options={'maxiter': 200, 'ftol': 1e-14})
            if res.x is not None:
                hr = project_sym(res.x, nn // 2)
        hfn = expand_sym(hr)
        c5 = evaluate_c5(hfn)
        if c5 < best_c5:
            best_c5 = c5
            best_h = hfn.copy()
            n = nn
            h_red = hr.copy()

    # Phase 6: Full-space at new resolution
    if time.time() < deadline - 40:
        hf2 = best_h.copy()
        for T in [20.0, 10.0, 6.0]:
            if time.time() > deadline - 25:
                break
            res = minimize(obj_full, hf2, args=(T, PEN), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': 200, 'ftol': 1e-14})
            if res.x is not None:
                hf2 = project(res.x, n)
                c5 = evaluate_c5(hf2)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = hf2.copy()

    # Phase 7: Final subgradient polish
    if time.time() < deadline - 15:
        h_sub2 = subgrad_polish(best_h, n, 0.001, 2000, deadline)
        c5 = evaluate_c5(h_sub2)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sub2.copy()

    # Final: ensure constraint is met
    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
