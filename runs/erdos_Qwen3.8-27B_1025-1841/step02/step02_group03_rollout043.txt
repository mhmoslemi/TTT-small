        cnt = 0
        while x:
            cnt ^= 1  # XOR with LSB
            x >>= 1
        b[i] = 1.0 if cnt else 0.0

import numpy as np
import time

def compute_c5_binary(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_binary_from_continuous(h, n):
    target = n // 2
    idx = np.argsort(-h)
    b = np.zeros(n, dtype=np.float64)
    b[idx[:target]] = 1.0
    return b

def generate_thue_morse(n):
    b = np.zeros(n, dtype=np.float64)
    for i in range(n):
        p = 0
        x = i
        while x:
            p ^= (x & 1)
            x >>= 1
        b[i] = 1.0 if p else 0.0
    return b

def upsample_binary(b, new_n, rng):
    old_n = len(b)
    reps = max(1, new_n // old_n)
    b_new = np.tile(b, reps)[:new_n].copy()
    cnt = int(b_new.sum())
    target = new_n // 2
    if cnt > target:
        one_idx = np.where(b_new == 1)[0]
        sel = rng.choice(len(one_idx), size=cnt - target, replace=False)
        b_new[one_idx[sel]] = 0
    elif cnt < target:
        zero_idx = np.where(b_new == 0)[0]
        sel = rng.choice(len(zero_idx), size=target - cnt, replace=False)
        b_new[zero_idx[sel]] = 1
    return b_new

def simulated_annealing(b, time_limit, rng, T_init=0.004, T_min=0.0001, cool=0.99999):
    n = len(b)
    current = b.copy().astype(np.float64)
    current_c5 = compute_c5_binary(current)
    best = current.copy()
    best_c5 = current_c5

    start = time.time()
    T = T_init
    stall = 0
    max_stall = 60000

    while time.time() - start < time_limit:
        one_pos = np.where(current == 1)[0]
        zero_pos = np.where(current == 0)[0]

        i = one_pos[rng.randint(len(one_pos))]
        j = zero_pos[rng.randint(len(zero_pos))]

        if i == j:
            continue

        current[i] = 0
        current[j] = 1
        new_c5 = compute_c5_binary(current)
        delta = new_c5 - current_c5

        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best = current.copy()
                best_c5 = new_c5
                stall = 0
            else:
                stall += 1
        else:
            current[i] = 1
            current[j] = 0
            stall += 1

        T = max(T_min, T * cool)

        if stall > max_stall:
            current = best.copy()
            current_c5 = best_c5
            T = T_init
            stall = 0

    return best, best_c5

def greedy_polish(b, time_limit, rng):
    n = len(b)
    current = b.copy().astype(np.float64)
    current_c5 = compute_c5_binary(current)

    start = time.time()
    while time.time() - start < time_limit:
        one_pos = np.where(current == 1)[0]
        zero_pos = np.where(current == 0)[0]
        i = one_pos[rng.randint(len(one_pos))]
        j = zero_pos[rng.randint(len(zero_pos))]
        if i == j:
            continue
        current[i] = 0
        current[j] = 1
        new_c5 = compute_c5_binary(current)
        if new_c5 < current_c5 - 1e-15:
            current_c5 = new_c5
        else:
            current[i] = 1
            current[j] = 0

    return current, current_c5

def systematic_descent(b, rng, max_sweeps=50):
    n = len(b)
    current = b.copy().astype(np.float64)
    current_c5 = compute_c5_binary(current)

    for sweep in range(max_sweeps):
        improved = False
        one_pos = np.where(current == 1)[0].tolist()
        for i in one_pos:
            zero_pos = np.where(current == 0)[0]
            best_j = -1
            best_c = current_c5
            for j in zero_pos:
                if j == i:
                    continue
                current[i] = 0
                current[j] = 1
                c = compute_c5_binary(current)
                if c < best_c - 1e-15:
                    best_c = c
                    best_j = j
                current[i] = 1
                current[j] = 0
            if best_j >= 0:
                current[i] = 0
                current[best_j] = 1
                current_c5 = best_c
                improved = True
        if not improved:
            break

    return current, current_c5
