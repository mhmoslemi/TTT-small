import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def smooth_max_grad(h, tau):
    """Compute smooth max of correlation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        # h[i] * (1 - h[i+sh]) contributes to correlation at shift sh
        # d/dh[i] of h[i]*(1-h[i+sh]) = 1 - h[i+sh]
        # d/dh[i+sh] of h[i]*(1-h[i+sh]) = -h[i]
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    return obj, grad

def optimize_smooth(h, tau, time_limit, max_iters, initial_lr=0.3):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_max_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_max_grad(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > 100:
            lr *= 0.5
            v = 0.5 * v
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    
    return best_h

def optimize_subgrad(h, time_limit, max_iters, initial_lr=0.005):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        k_star = int(np.argmax(corr))
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= h[idx + sh]
        grad *= dx
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > 300:
            lr *= 0.5
            v = 0.5 * v
            stall = 0
            if lr < 1e-14:
                break
        h = h_new
    
    return best_h, best_c5

def binary_projection(h):
    """Project to nearest binary solution (0 or 1) with correct sum."""
    n = len(h)
    target = n // 2
    # Sort by value, keep top n/2 as 1
    thresh = np.sort(h)[:n - target]
    h_bin = (h >= thresh).astype(float)
    # Fix if count is wrong
    cnt = h_bin.sum()
    if cnt > target:
        # Flip some 1s to 0s (the ones closest to threshold)
        ones = np.where(h_bin == 1)[0]
        ones_sorted = ones[np.argsort(h[ones])]
        for i in range(int(cnt - target)):
            h_bin[ones_sorted[i]] = 0
    elif cnt < target:
        zeros = np.where(h_bin == 0)[0]
        zeros_sorted = zeros[np.argsort(-h[zeros])]
        for i in range(int(target - cnt)):
            h_bin[zeros_sorted[i]] = 1
    return h_bin

        h_cur = interp_periodic(best_h, target_n)
        h_cur = project_to_feasible(h_cur, target_n)

        taus = [30.0, 50.0, 80.0, 120.0]
        for tau in taus:
            if remaining() < 60:

            tl = min(remaining() * 0.2, 120)
            h_cur = optimize_smooth(h_cur, tau, tl, max_iters=20000, initial_lr=0.2)
            c5_cur = compute_c5(h_cur)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_cur.copy()
    
    # Phase 2: Try binary projection and re-optimize
    if remaining() > 100:
        n_b = len(best_h)
        h_b = binary_projection(best_h)
        c5_b = compute_c5(h_b)
        if c5_b < best_c5:
            best_c5 = c5_b
            best_h = h_b.copy()
        else:
            # Blend binary with current
            h_b_blend = 0.3 * h_b + 0.7 * best_h
            h_b_blend = project_to_feasible(h_b_blend, n_b)
            c5_blend = compute_c5(h_b_blend)
            if c5_blend < best_c5:
                best_c5 = c5_blend
                best_h = h_b_blend.copy()
        
        # Optimize from binary start
        if remaining() > 80:
            h_b2 = binary_projection(best_h)
            h_b2 = optimize_smooth(h_b2, 50.0, remaining() * 0.3, max_iters=15000, initial_lr=0.3)
            c5_b2 = compute_c5(h_b2)
            if c5_b2 < best_c5:
                best_c5 = c5_b2
                best_h = h_b2.copy()
    
    # Phase 3: Subgradient refinement on best
    if remaining() > 60:
        h_f = best_h.copy()
        h_f, c5_f = optimize_subgrad(h_f, remaining() * 0.4, max_iters=100000, initial_lr=0.003)

            best_h = h_f.copy()
    
    # Phase 4: Try perturbation and re-optimize
    if remaining() > 80:
        h_pert = best_h + 0.01 * np.random.randn(len(best_h))

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def smooth_obj_grad(h, tau, thresh=1e-5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    active = np.where(p > thresh)[0]
    for k in active:
        pk = p[k]
        sh = k - (n - 1)
        if sh >= 0:
            grad[:n - sh] += pk * (1.0 - h[sh:])
            grad[sh:] -= pk * h[:n - sh]
        else:
            sp = -sh
            grad[sp:] += pk * (1.0 - h[:n - sp])
            grad[:n - sp] -= pk * h[sp:]
    grad *= dx
    return obj, grad

    if cnt > target:
        ones = np.where(h_bin == 1)[0]
        order = ones[np.argsort(h[ones])]
        h_bin[order[:cnt - target]] = 0
    elif cnt < target:
        zeros = np.where(h_bin == 0)[0]
        order = zeros[np.argsort(-h[zeros])]
        h_bin[order[:target - cnt]] = 1
    return h_bin
