from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    # Sort by value, keep top n/2 as 1
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_sa(h_bin, n_iter, start_temp, end_temp, rng):
    """Simulated annealing on binary problem."""
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.astype(float).copy()
    corr = np.correlate(h, 1-h, mode='full')
    current_max = np.max(corr)
    best_h = h.copy()
    best_max = current_max
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    
    for it in range(n_iter):
        temp = start_temp * (end_temp / start_temp) ** (it / max(n_iter-1, 1))
        if temp < 1e-8:
            break
            
        # Random pair swap
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
            
        h[i] = 0
        h[j] = 1
        new_corr = np.correlate(h, 1-h, mode='full')
        new_max = np.max(new_corr)
        
        delta = new_max - current_max
        if delta <= 0 or rng.random() < np.exp(-delta * dx / temp):
            current_max = new_max
            # Update sets
            # i was 1, now 0: move from ones to zeros
            # j was 0, now 1: move from zeros to ones
            ones = np.where(h > 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        else:
            h[i] = 1
            h[j] = 0
    
    return best_h, best_max * dx

When h[i] changes from 1 to 0:

def binary_sa_fast(h_bin, n_iter, start_temp, end_temp, rng):
    n = len(h_bin)
    dx = 2.0 / n
    h = h_bin.copy()  # 0s and 1s
    # Initialize correlation
    C = np.zeros(2*n - 1)
    # C[k] corresponds to shift s = k - (n-1)
    # C[k] = sum_i h[i] * (1 - h[i+s]) for valid i
    for s in range(-(n-1), n):
        total = 0.0
        if s >= 0:
            # i ranges from 0 to n-1-s
            total = np.sum(h[s:] * (1 - h[:n-s]))
        else:
            # s negative, let t = -s
            # i ranges from -s to n-1
            t = -s
            total = np.sum(h[:n-t] * (1 - h[t:]))
        C[s + n - 1] = total
    
    current_max = np.max(C)
    best_h = h.copy()
    best_max = current_max
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]

        if temp < 1e-10:
            break

        j = zeros[rng.randint(len(zeros))]

        # ΔC(s) = -(1-h[i+s]) + h[i-s] + (1-h[j+s]) - h[j-s]
        # for valid s (where both i and i+s are in range, AND j and j+s are in range)
        # Actually, the formula works for all s, we just need to check bounds
        
        # For the term involving i: valid when i ∈ [0,n-1] and i+s ∈ [0,n-1]
        # For the term involving j: valid when j ∈ [0,n-1] and j+s ∈ [0,n-1]
        
        # Let's compute for all s from -(n-1) to n-1
        s_range = np.arange(-(n-1), n)  # length 2n-1
        
        # h[i+s] is valid when 0 <= i+s <= n-1, i.e., -i <= s <= n-1-i
        # h[i-s] is valid when 0 <= i-s <= n-1, i.e., i-(n-1) <= s <= i
        # Similarly for j
        
        # change_i[s] = -(1-h[i+s]) + h[i-s] for valid s
        # change_j[s] = (1-h[j+s]) - h[j-s] for valid s
        
        delta = np.zeros(2*n-1)

        s_min_i1 = max(-(n-1), -i)

            idx = slice(s_min_i1 + n - 1, s_max_i1 + n - 1 + 1)
            delta[idx] -= (1 - h[i + s_range[idx]])

        s_min_i2 = max(-(n-1), i-(n-1))
        s_max_i2 = min(n-1, i)
        if s_min_i2 <= s_max_i2:
            idx = slice(s_min_i2 + n - 1, s_max_i2 + n - 1 + 1)
            delta[idx] += h[i - s_range[idx]]
        
        # For j:
        # Term 3: +(1-h[j+s]) valid when 0<=j+s<=n-1
        s_min_j1 = max(-(n-1), -j)
        s_max_j1 = min(n-1, n-1-j)
        if s_min_j1 <= s_max_j1:
            idx = slice(s_min_j1 + n - 1, s_max_j1 + n - 1 + 1)
            delta[idx] += (1 - h[j + s_range[idx]])
        
        # Term 4: -h[j-s] valid when 0<=j-s<=n-1
        s_min_j2 = max(-(n-1), j-(n-1))
        s_max_j2 = min(n-1, j)
        if s_min_j2 <= s_max_j2:
            idx = slice(s_min_j2 + n - 1, s_max_j2 + n - 1 + 1)
            delta[idx] -= h[j - s_range[idx]]
        
        new_C = C + delta
        new_max = np.max(new_C)

            h[j] = 1

            zeros = np.where(h == 0)[0]
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
        # else reject, C unchanged
