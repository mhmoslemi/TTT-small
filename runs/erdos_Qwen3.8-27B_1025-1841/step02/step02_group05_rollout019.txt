def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    # Final adjustment for exact sum
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Add/subtract uniformly from interior points
        mask = (result > 0.01) & (result < 0.99)
        if mask.sum() > 0:
            adj = diff / mask.sum()
            result[mask] = np.clip(result[mask] + adj, 0, 1)
    return result

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

    n = len(a)

    fa = np.fft.rfft(a, size)

import numpy as np
import time
import math

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def fft_full_correlate(a, b):
    n, m = len(a), len(b)
    size = n + m - 1
    return np.fft.irfft(np.fft.rfft(a, size) * np.fft.rfft(b[::-1], size), size)

def smooth_grad_fft(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = fft_full_correlate(h, 1.0 - h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -700, 700)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + math.log(s) / tau)
    csum = np.concatenate([[0.0], np.cumsum(p)])
    T1 = csum[n:2*n] - csum[0:n]
    p_half = p[n-1:]
    h_padded = np.concatenate([h, np.zeros(n-1)])
    T2_full = fft_full_correlate(h_padded, p_half)
    T2 = T2_full[n-1:2*n-1]
    size3 = 2*n - 1
    T3_full = np.fft.irfft(np.fft.rfft(p_half, size3) * np.fft.rfft(h, size3), size3)
    T3 = T3_full[0:n]
    grad = dx * (T1 - T2 - T3)
    return obj, grad, corr

def optimize_momentum_fft(h, n, tau, time_limit, lr=0.05, momentum=0.9, lr_decay=0.95, patience=3000):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    while time.time() - start < time_limit:
        obj, grad, corr = smooth_grad_fft(h, tau)
        c5_now = corr.max() * (2.0 / n)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > patience:
            lr *= lr_decay
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5

def optimize_subgrad_fft(h, n, time_limit, lr=0.002):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    while time.time() - start < time_limit:
        corr = fft_full_correlate(h, 1.0 - h)
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.full(n, 1.0)
        if s >= 0:
            grad[s:] -= h[s:]
            grad[:n-s] -= h[:n-s]
        else:
            idx = -s
            grad[:idx] -= h[:idx]
            grad[idx:] -= h[idx:]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5

def coord_descent_polish(h, n, time_limit, step=0.01):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    n_iter = 0
    while time.time() - start < time_limit:
        improve = False
        corr = fft_full_correlate(h, 1.0 - h)
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        for i in range(n):
            if time.time() - start > time_limit:
                break
            if h[i] <= 0.001 or h[i] >= 0.999:
                continue
            delta = -step
            h_new = h.copy()
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
            h_new = project_to_feasible(h_new, n)
            corr_new = fft_full_correlate(h_new, 1.0 - h_new)
            c5_new = corr_new.max() * dx
            if c5_new < best_c5 - 1e-14:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                improve = True
            else:
                delta = step
                h_new[i] = np.clip(h_new[i] + 2*delta, 0, 1)
                h_new = project_to_feasible(h_new, n)

def multishift_subgrad_fft(h, n, time_limit, lr=0.003, top_k=3):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    while time.time() - start < time_limit:
        corr = fft_full_correlate(h, 1.0 - h)
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            lr *= 0.65
            v *= 0.35
            stall = 0
            if lr < 1e-14:
                break
        idx_top = np.argsort(corr)[::-1][:top_k]
        vals = corr[idx_top]
        w = np.exp(vals - vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for wi, kk in zip(w, idx_top):
            s = int(kk) - (n - 1)
            g = np.full(n, 1.0)
            if s >= 0:
                g[s:] -= h[s:]
                g[:n-s] -= h[:n-s]
            else:
                idx = -s
                g[:idx] -= h[:idx]
                g[idx:] -= h[idx:]
            grad += wi * g
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5
