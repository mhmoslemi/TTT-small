    h = np.clip(h, 0.0, 1.0)

    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) / 2
        h_trial = np.clip(h + mid, 0.0, 1.0)

        if s < target_sum:

    """Compute gradient of smooth max of corr values"""
    mc = np.max(corr)
    shifted = t * (corr - mc)
    # For numerical stability
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    w /= w.sum()

def proj_feasible(h, target_sum):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -1.5, 1.5
    for _ in range(50):

        ht = np.clip(h + mid, 0.0, 1.0)
        if ht.sum() < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target_sum - h.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smax_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = t * (corr - mc)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n), mc * dx
    w /= ws
    active = np.where(w > 1e-12)[0]
    grad = np.zeros(n)
    for m in active:
        k = int(m) - (n - 1)
        wm = w[m]
        if k >= 0:
            grad[k:] += wm * (1.0 - h[:n-k])
            grad[:n-k] -= wm * h[k:]
        else:
            kp = -k
            grad[:n-kp] += wm * (1.0 - h[kp:])
            grad[kp:] -= wm * h[:n-kp]
    return dx * grad, mc * dx
