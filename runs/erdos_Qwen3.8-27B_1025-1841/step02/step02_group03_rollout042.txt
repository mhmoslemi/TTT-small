        n_try = min(400, len(zeros) * len(ones))
        zi = rng.choice(len(zeros), size=min(n_try, len(zeros)), replace=False)
        oi = rng.choice(len(ones), size=min(n_try, len(ones)), replace=False)
        n_test = min(len(zi), len(oi))
        
        best_delta = None
        best_new_max = cur_max
        best_a = -1
        best_b = -1
        
        for idx in range(n_test):
            a = int(zeros[zi[idx]])
            b = int(ones[oi[idx]])
            am = a - m_vals
            ap = a + m_vals
            bm = b - m_vals
            bp = b + m_vals
            amv = (am >= 0) & (am < n)
            apv = (ap >= 0) & (ap < n)
            bm = bm  # reuse
            bmv = (bm >= 0) & (bm < n)
            bpv = (bp >= 0) & (bp < n)
            delta = np.zeros(2 * n - 1)
            if np.any(amv):
                delta[amv] += 1.0 - h_bin[am[amv]]
            if np.any(apv):
                delta[apv] -= h_bin[ap[apv]]
            if np.any(bmv):
                delta[bmv] -= 1.0 - h_bin[bm[bmv]]
            if np.any(bpv):
                delta[bpv] += h_bin[bp[bpv]]
            new_max = float(np.max(corr + delta))
            if new_max < best_new_max - 1e-14:
                best_new_max = new_max
                best_delta = delta
                best_a = a
                best_b = b
        
        if best_a < 0:
            break
        h_bin[best_a] = 1.0
        h_bin[best_b] = 0.0
        corr = corr + best_delta
        cur_max = best_new_max
        best_c = cur_max * dx
    
    return h_bin, best_c
