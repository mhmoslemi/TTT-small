import time

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def project_to_feasible(g, n):

    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def make_binary(h, n, rng):
    """Convert h to binary with exactly n//2 ones."""
    target = n // 2
    # Sort by value, take top target
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def sa_search(h_start, n, time_limit, rng, temp_init=None, cool_rate=0.99998, reheat_period=20000):
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    
    # Ensure exactly n//2 ones
    ones_count = int(np.round(h.sum()))
    target = n // 2
    if ones_count > target:
        idx = np.where(h > 0.5)[0]
        perm = rng.permutation(len(idx))
        for i in range(ones_count - target):
            h[idx[perm[i]]] = 0.0
    elif ones_count < target:
        idx = np.where(h < 0.5)[0]
        perm = rng.permutation(len(idx))
        for i in range(target - ones_count):
            h[idx[perm[i]]] = 1.0
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = corr.max() * dx
    best_c5 = current_c5

def smooth_refine(h, n, tau, time_limit, lr=0.1, momentum=0.9):
    """Smooth max optimization for fine-tuning."""
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    dx = 2.0 / n
    
    for _ in range(100000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

            lr *= 0.8
            v *= 0.5
            stall = 0
            if lr < 1e-14:

        # Softmax gradient
        shifted = np.clip(tau * (corr - max_c), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-15:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:

        grad *= dx

        h = project_to_feasible(h + v, n)

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def make_binary(h, n, rng):
    target = n // 2
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result


def sa_search(h_start, n, time_limit, rng, temp_init=None):
    dx = 2.0 / n
    h = h_start.astype(float).copy()
    ones_count = int(np.round(h.sum()))
    target = n // 2
    if ones_count > target:
        idx = np.where(h > 0.5)[0]
        perm = rng.permutation(len(idx))
        for i in range(ones_count - target):
            h[idx[perm[i]]] = 0.0
    elif ones_count < target:
        idx = np.where(h < 0.5)[0]
        perm = rng.permutation(len(idx))
        for i in range(target - ones_count):
            h[idx[perm[i]]] = 1.0

    corr = np.correlate(h, 1.0 - h, mode='full')
    current_c5 = corr.max() * dx
    best_c5 = current_c5
    best_h = h.copy()

    if temp_init is None:
        temp = 0.04 * current_c5
    else:
        temp = temp_init

    start = time.time()
    cool = 0.999995
    reheat_every = 30000
    count = 0

    while time.time() - start < time_limit:
        i = rng.integers(n)
        while h[i] != 1.0:
            i = rng.integers(n)
        j = rng.integers(n)
        while h[j] != 0.0:
            j = rng.integers(n)

        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = corr.max() * dx
        delta = new_c5 - current_c5

        if delta <= 0 or (delta / temp < 50 and rng.random() < np.exp(-delta / temp)):
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0

        temp *= cool
        count += 1
        if count % reheat_every == 0:
            temp = max(temp, 0.02 * best_c5)

    return best_h, best_c5


def sa_targeted(h_start, n, time_limit, rng, temp_init=None):

def smooth_refine(h, n, tau, time_limit, lr=0.1, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    dx = 2.0 / n

    for _ in range(50000):
        if time.time() - start > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx

        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 800:
            lr *= 0.8
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break

        shifted = np.clip(tau * (corr - max_c), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s

        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-15:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
        grad *= dx

        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def smooth_refine_multitau(h, n, taus, time_limit, rng):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    tau = taus[0]

    while time.time() - start < time_limit:
        tl_chunk = min(30.0, time_limit - (time.time() - start))

            break
        h_cur, c5_cur = smooth_refine(h, n, tau=tau, time_limit=tl_chunk, lr=0.05 / tau * 10, momentum=0.92)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()
        tau = rng.choice(taus)

    return best_h, best_c5
