def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        h = np.clip(h, 0.0, 1.0)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    h = np.clip(h.astype(float).copy(), 0.0, 1.0)

            break

        else:
            mask = h > 0.0

            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
