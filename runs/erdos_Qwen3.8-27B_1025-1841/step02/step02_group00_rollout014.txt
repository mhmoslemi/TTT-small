def binary_local_search(h, time_limit):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    # Round to binary
    threshold = 0.5
    h[h > threshold] = 1.0
    h[h <= threshold] = 0.0
    # Fix sum
    target = n // 2
    ones = np.sum(h == 1)
    if ones > target:
        # Remove excess ones
        idx = np.where(h == 1)[0]
        for i in idx[ones-target:]:
            h[i] = 0
    elif ones < target:
        idx = np.where(h == 0)[0]
        for i in idx[:target-ones]:
            h[i] = 1
    
    c5 = max(np.correlate(h, 1-h, mode='full')) * dx
    best_c5 = c5
    start = time.time()
    
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    
    improvement = True
    while improvement and time.time() - start < time_limit:
        improvement = False
        best_c5_new = best_c5
        best_swap = None
        for i in ones_idx:
            for j in zeros_idx:
                h[i] = 0
                h[j] = 1
                c = np.max(np.correlate(h, 1-h, mode='full')) * dx
                h[j] = 0
                h[i] = 1
                if c < best_c5_new - 1e-12:
                    best_c5_new = c
                    best_swap = (i, j)
        if best_swap is not None:
            i, j = best_swap
            h[i] = 0
            h[j] = 1
            ones_idx = np.where(h == 1)[0]
            zeros_idx = np.where(h == 0)[0]
            best_c5 = best_c5_new
            improvement = True
    
    return h, best_c5

                corr = corr + g[j] - g[i]
                h[i] = 0
                h[j] = 1
                update_g_pos(g, h, n, i)
                update_g_pos(g, h, n, j)
                current_max = float(np.max(corr))
                if current_max * dx < best_c5:
                    best_c5 = current_max * dx
                    best_h = h.copy()
                round_improved = True
                no_imp = 0
                # Update zeros list
                zeros = zeros[zeros != j]
                ones = np.where(h == 1)[0]  # i was removed
        if not round_improved:
            no_imp += 1
            if no_imp < 5:
                # Random perturbation
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                if len(ones) > 0 and len(zeros) > 0:
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0
                    h[j] = 1
                    corr = compute_corr(h)
                    g = compute_g(h, n)
                    current_max = float(np.max(corr))
    
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5

def slsqp_optimize(h, n, T, max_iter, time_check=None):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    
    def obj(h_vals):
        corr = compute_corr(h_vals)
        m = np.max(corr)
        s = np.exp(T * (corr - m))
        return float((m + np.log(s.sum()) / T) * (2.0 / n))
    
    def obj_grad(h_vals):
        # Numerical gradient (expensive but simple)
        # Actually let's use the analytical approach via the smooth weight
        corr = compute_corr(h_vals)
        m = np.max(corr)
        s = np.exp(T * (corr - m))
        w = s / s.sum()
        # grad of smooth_max w.r.t. corr is w
        # grad of corr[k] w.r.t. h[j]:
        # corr[k_idx] = sum_i h[i+k](1-h[i]) where k = k_idx-(n-1)
        # d/dh[j]: (1-h[j-k])*I(0≤j-k<n) - h[j+k]*I(0≤j+k<n)
        # This is exactly g[j, k_idx]!
        g = compute_g(h_vals, n)  # (n, 2n-1)
        # grad[j] = sum_k w[k] * g[j,k] = w @ g[j, :]... 
        # Actually: grad[j] = sum_k w[k] * (d corr[k]/d h[j]) = sum_k w[k] * g[j,k]
        grad = g @ w  # (n,) since g is (n, 2n-1) and w is (2n-1,)
        return grad * (2.0 / n)
    
    def cons_fun(h_vals):
        return np.sum(h_vals) - n / 2.0
    
    def cons_jac(h_vals):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons,
                   jac=obj_grad,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

_G_T = 1.0
