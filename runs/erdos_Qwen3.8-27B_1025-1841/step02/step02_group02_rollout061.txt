def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def coord_min_1d(C, D):
    """Minimize max_s(C[s] + D[s]*t) over t in [0,1] using bisection on value."""
    # Lower bound: max(C) (at t=0) or max(C+D) (at t=1), take min
    v0 = np.max(C)
    v1 = np.max(C + D)
    
    # Bisection on optimal value
    lo_v = min(v0, v1)
    hi_v = max(v0, v1)
    
    # Actually, the min of max of lines is >= max of each line's min over [0,1]
    # And <= min(v0, v1)
    # Let's just use ternary search, it's simpler and works for convex functions
    
    lo_t, hi_t = 0.0, 1.0
    for _ in range(80):
        m1 = lo_t + (hi_t - lo_t) / 3.0
        m2 = hi_t - (hi_t - lo_t) / 3.0
        f1 = np.max(C + D * m1)
        f2 = np.max(C + D * m2)
        if f1 < f2:
            hi_t = m2
        else:
            lo_t = m1
    return (lo_t + hi_t) / 2.0

def interp_periodic(h, target_n):
    n = len(h)

        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:

            grad[j0:j1] += pk * (1.0 - h[jj - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            jj = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[jj + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def coord_min_1d(C, D):
    lo_t, hi_t = 0.0, 1.0
    for _ in range(60):
        m1 = lo_t + (hi_t - lo_t) / 3.0
        m2 = hi_t - (hi_t - lo_t) / 3.0
        f1 = np.max(C + D * m1)
        f2 = np.max(C + D * m2)
        if f1 < f2:
            hi_t = m2
        else:
            lo_t = m1
    return (lo_t + hi_t) / 2.0


def coordinate_descent(h, max_sweeps, time_limit, start_time=None, verbose=False):
    n = len(h)
    h = h.copy()
    corr = compute_corr(h)
    best_c5 = corr.max() * (2.0 / n)
    best_h = h.copy()
    if start_time is None:
        start_time = time.time()
    total = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        idx = np.arange(n)
        np.random.shuffle(idx)
        for j in idx:
            if time.time() - start_time > time_limit:
                break
            s_lo = max(0, n - 1 - j)
            s_hi = min(total - 1, n - 1 + j)
            s_range = np.arange(s_lo, s_hi + 1)
            idx1 = j - s_range + (n - 1)
            idx2 = j + s_range - (n - 1)
            D = np.zeros(total)
            D[s_range] = (1.0 - h[idx1]) - h[idx2]
            C = corr - h[j] * D
            t_new = coord_min_1d(C, D)
            delta = t_new - h[j]
            if abs(delta) > 1e-14:
                corr += delta * D
                h[j] = t_new
        h = project_to_feasible(h, n)
        corr = compute_corr(h)
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            jj = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[jj - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            jj = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[jj + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr if initial_lr is not None else 0.3
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    for _ in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > 80:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    return best_h, best_obj

    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
