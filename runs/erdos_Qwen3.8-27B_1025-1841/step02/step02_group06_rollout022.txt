        # For shift d:
        # Terms where i = j: contributes h[j]*(1-h[j+d]) -> derivative w.r.t. h[j] is (1-h[j+d])
        #   (valid if 0<=j<n and 0<=j+d<n)
        # Terms where i+d = j, i.e., i = j-d: contributes h[j-d]*(1-h[j]) -> derivative w.r.t. h[j] is -h[j-d]
        #   (valid if 0<=j-d<n and 0<=j<n)
        
        # So df_d/dh[j] = (1-h[j+d]) - h[j-d] (with appropriate boundary handling)
        
        j0 = max(0, d)
        j1 = min(n, n + d)
        # For j in [j0, j1): h[j] appears as the first factor (i=j), and h[j+d] is the second
        # df/dh[j] gets +(1-h[j+d])
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += (1.0 - h[idx + d])
        
        # For j where j-d is a valid index: h[j-d]*(1-h[j]) -> df/dh[j] gets -h[j-d]
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= h[idx + d]
    
    return grad * dx, g.max()

def nesterov_projected_grad(h, n, max_iters, time_limit, topK=3, lr_init=0.1, beta=0.9):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    v = np.zeros(n)
    lr = lr_init
    stall = 0
    patience = 500
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Nesterov lookahead
        h_lookahead = project_to_feasible(h + beta * v, n)
        grad, cur_max = compute_gradient_topK(h_lookahead, n, dx, topK)
        
        if cur_max < best_c5 - 1e-12:
            best_c5 = cur_max
            best_h = h_lookahead.copy()
            stall = 0
        else:
            stall += 1
        
        # Update
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        
        # Check actual objective
        if it % 50 == 0:
            actual_c5 = compute_c5(h)
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h.copy()
        
        # Learning rate decay
        if stall > patience:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
    
    return best_h, best_c5

Wait, I realize there might be an issue with the Nesterov lookahead + projection. The projection might destroy the Nesterov property. Let me use a simpler approach: just momentum (not full Nesterov) with a good learning rate schedule.

Let me also add the **coordinate descent** step, which I think could be very effective for the final polish.

**Coordinate Descent via Pair Swaps:**

For the top shift k*:
- Compute grad_j = (1-h[j+k*]) - h[j-k*] for valid j
- Find j_min = argmin grad_j (among h[j] < 1)
- Find j_max = argmax grad_j (among h[j] > 0)
- Swap: increase h[j_min] by δ, decrease h[j_max] by δ
- Choose δ by line search (or fixed small step)

This is O(n) per swap step. We can do many swap steps cheaply.

Let me implement this as a separate phase after the gradient optimization.

Actually, I realize that for the coordinate descent to be effective, I need to consider ALL active shifts, not just the top one. Because reducing f_{k*} might increase f_{k'} for another shift.

A better criterion: use the **smooth maximum** (log-sum-exp or the average of top-K) as the objective for the coordinate descent.

Let me implement a version where:
1. Compute top-K shifts
2. Compute the gradient of their average: g_j = (1/K) ∑_{k∈topK} [(1-h[j+k]) - h[j-k]]
3. Find the best swap: j_min = argmin g_j, j_max = argmax g_j
4. Perform the swap with a step size that ensures the max doesn't increase

This is O(K·n) per swap, which is very fast.

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def gradient_topK(h, n, dx, K=3):
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    top_idx = np.argsort(g)[-K:]
    grad = np.zeros(n)
    for k in top_idx:
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += (1.0 - h[idx + d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= h[idx + d]
    return grad * dx, g, top_idx

def coordinate_descent(h, n, time_limit, topK=3, max_rounds=100000):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        
        if cur_max < best_c5 - 1e-12:
            best_c5 = cur_max
            best_h = h.copy()
        
        top_idx = np.argsort(g)[-topK:]
        
        # Compute combined gradient for top-K
        grad = np.zeros(n)
        for k in top_idx:
            d = k - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx + d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad /= topK
        
        # Find best swap
        # We want to decrease grad at the max-grad position and increase at the min-grad position
        # Actually, we want to move h in the direction -grad to decrease the objective
        # So increase h where grad is most negative, decrease h where grad is most positive
        
        # But we need to maintain sum constraint: increase one, decrease another
        # Best: j_inc = argmin grad (where -grad is largest), j_dec = argmax grad
        
        # With feasibility: h[j_inc] < 1, h[j_dec] > 0
        feasible_inc = np.where((h < 1.0) & (grad < 0))[0]
        feasible_dec = np.where((h > 0.0) & (grad > 0))[0]
        
        if len(feasible_inc) == 0 or len(feasible_dec) == 0:
            # No improvement possible with this top-K
            break
        
        j_inc = feasible_inc[np.argmin(grad[feasible_inc])]
        j_dec = feasible_dec[np.argmax(grad[feasible_dec])]
        
        # Step size: limited by feasibility
        delta = min(0.5, (1.0 - h[j_inc]), h[j_dec])
        # Try a fraction of this
        for frac in [1.0, 0.5, 0.25, 0.1]:
            h_try = h.copy()
            h_try[j_inc] += frac * delta
            h_try[j_dec] -= frac * delta
            c5_try = compute_c5(h_try)
            if c5_try < cur_max:
                h = h_try
                break
    
    return best_h, best_c5
