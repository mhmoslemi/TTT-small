    n = len(h)

        grad[:n-s] += (1.0 - h[s:])

        s2 = -s
        grad[:n-s2] += (1.0 - h[s2:])
        grad[s2:] -= h[:n-s2]

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    sorted_idx = np.argsort(corr)[::-1]

            grad[:n-s] += weight * (1.0 - h[s:])
            grad[s:] -= weight * h[:n-s]

            s2 = -s
            grad[:n-s2] += weight * (1.0 - h[s2:])
            grad[s2:] -= weight * h[:n-s2]
        total_weight += weight
    
    if total_weight > 0:
        grad /= total_weight
    return grad * dx
