        s = np.sum(np.clip(g - mid, 0.0, 1.0))

        else:

def smooth_grad(h, corr, tau, n, p_keep=100):
    """Compute gradient of smooth max. Only use top p_keep shifts for speed."""
    dx = 2.0 / n
    max_c = corr.max()
    shifts = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifts)
    s = exp_c.sum()
    p = exp_c / s
    
    # Only keep top p_keep shifts where p is significant
    if p_keep < len(p):
        top_idx = np.argsort(p)[-p_keep:]
        p_supp = p[top_idx]
        p_supp = p_supp / p_supp.sum()
        p_dict = {int(idx): p_supp[k] for k, idx in enumerate(top_idx)}
    else:
        p_dict = {k: p[k] for k in range(len(p)) if p[k] > 1e-15}
    
    grad = np.ones(n) * sum(p_dict.values())  # = 1 (since p sums to 1)
    # Actually grad starts at sum_k p[k] = 1 for each j

        j1 = min(n, n - sh)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5_full(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def smooth_gradient(h, corr, tau, n, p_keep=200):
    dx = 2.0 / n
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    if p_keep < len(p):
        top_idx = np.argsort(p)[-p_keep:]
        pk_arr = p[top_idx]
        pk_arr = pk_arr / pk_arr.sum()
        p_list = list(zip(top_idx, pk_arr))
    else:
        p_list = [(k, p[k]) for k in range(len(p)) if p[k] > 1e-14]
    grad = np.ones(n)
    for k, pk in p_list:
        if pk < 1e-15:
            continue
        sh = int(k) - n + 1
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] -= pk * h[j0 + sh:j1 + sh]
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    return obj, grad


def optimize_smooth_adam(h, n, time_limit):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5_full(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    tau_start, tau_end = 3.0, 100.0
    anneal_rate = 0.998
    tau = tau_start
    while time.time() - start < time_limit:
        t += 1
        tau *= anneal_rate
        if tau < tau_end:
            tau = tau_end
        corr = np.correlate(h, 1.0 - h, mode='full')
        pk = min(300, max(50, int(2 * n / max(1, tau))))
        _, grad = smooth_gradient(h, corr, tau, n, p_keep=pk)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        lr = 0.15 / (1.0 + 0.005 * t)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)
    return best_h, best_c5


def optimize_subgrad(h, n, time_limit, lr_init=0.003):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5_full(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    stall = 0
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.ones(n)
        j0 = max(0, -sh)
        j1 = min(n, n - sh)
        if j0 < j1:
            grad[j0:j1] -= h[j0 + sh:j1 + sh]
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b - sh:j1b - sh]
        grad *= dx
        h = project_to_feasible(h - lr * grad, n)
        if stall > 300:
            lr *= 0.7
            stall = 0
            if lr < 1e-10:
                break
    return best_h, best_c5


def coord_descent(h, n, time_limit):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5_full(h)
    best_h = h.copy()
    start = time.time()
    s_vals = np.arange(-(n - 1), n)
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        for j in range(n):
            mask1 = (s_vals >= -j) & (s_vals < n - j)
            mask2 = (s_vals > j - n) & (s_vals <= j)
            js = np.clip(j + s_vals, 0, n - 1)
            jm = np.clip(j - s_vals, 0, n - 1)
            h_js = np.where(mask1, h[js], 0.0)
            h_jm = np.where(mask2, h[jm], 0.0)
            d = np.where(mask1, 1.0 - h_js, 0.0) - np.where(mask2, h_jm, 0.0)
            a = corr - h[j] * d
            x_lo, x_hi = 0.0, 1.0
            f_lo = np.max(a)
            f_hi = np.max(a + d)
            if f_lo <= f_hi:
                h[j] = 0.0
                continue
            if f_hi < f_lo:
                h[j] = 1.0
                continue
            for _ in range(20):
                x_mid = 0.5 * (x_lo + x_hi)
                f_mid = np.max(a + x_mid * d)
                if f_mid < f_lo:
                    x_lo, f_lo = x_mid, f_mid
                else:
                    x_hi = x_mid
            h[j] = x_lo
        h = np.clip(h, 0.0, 1.0)
        h = project_to_feasible(h, n)
        c5 = compute_c5_full(h)
        if c5 < best_c5 - 1e-15:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
