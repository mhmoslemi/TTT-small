        else:

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()

            if c5_4b < c5_4:
                c5_4, h4 = c5_4b, h4
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4

    # Phase 5: Multi-start at best resolution
    n_best = len(best_h)
    if rem() > 60:
        n_starts = max(2, min(5, int(rem() / 50)))
        best_h, best_c5 = multi_start_polish(best_h, n_best, n_starts, rem() * 0.25, seed=seed + 1)

    # Phase 6: Final subgradient polish
    if rem() > 15:
        hf, c5_f = subgrad_polish(best_h, len(best_h), 100000, rem() * 0.5, lr=0.001)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, hf

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
