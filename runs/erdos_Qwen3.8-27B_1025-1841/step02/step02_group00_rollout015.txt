```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        for i in range(n):
            if diff > 0 and h[i] < 1.0:
                adj = min(1.0 - h[i], diff)
                h[i] += adj
                diff -= adj
                if abs(diff) < 1e-12:
                    break
            elif diff < 0 and h[i] > 0.0:
                adj = min(h[i], -diff)
                h[i] -= adj
                diff += adj
                if abs(diff) < 1e-12:
                    break
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

# SLSQP optimization with smooth max
_OPT_N = 0
_OPT_T = 1.0

def _obj_fn(h):
    n = _OPT_N
    T = _OPT_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    log_sum = np.log(np.sum(np.exp(T * shifted)))
    return (mc + log_sum / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_N / 2.0

def _cons_jac(h):
    return np.ones(_OPT_N)

def _slsqp(h_start, n, T, max_iter):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_obj_fn, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

- Objective: minimize t

= t - f_s(h⁰) + Σ_j g_s[j] * h⁰_j

where g_s = ∇h f_s at h⁰.

So the constraint is: Σ_j g_s[j] * h_j - t ≤ constant_s

        accepted = False
        for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
            h_trial = alpha * h_lp + (1 - alpha) * h_arr
            h_trial = make_feasible(h_trial, n)
            c_trial = compute_c5(h_trial)
            if c_trial < best_c5 - 1e-14:
                h = h_trial
                best_c5 = c_trial
                accepted = True
                break
            elif c_trial < best_c5:
                # Accept even tiny improvement
                h = h_trial
                best_c5 = c_trial
                accepted = True
                break
        
        if not accepted:
            break
    
    return h, best_c5
```

Hmm wait, there's an issue with the SCA-LP. The linearization of a bilinear function does NOT provide a valid upper bound. So the constraint "linearized_f_s ≤ t" does NOT imply "f_s ≤ t". The LP solution might not actually reduce C₅.

That's why I added the line search / verification step. If the LP solution doesn't actually reduce C₅, I reject it and stop.

But this means SCA-LP might only give one or two iterations of improvement before stalling. That limits its usefulness.

Alternative: instead of linearizing ALL shifts, only linearize the TOP few (say top 10) shifts. This makes the LP less restrictive and might find a better direction.

Or: use the subgradient of the max (just the worst shift) with a projected gradient step.

Let me go with a hybrid:
- Compute all f_s
- Take the top 5 shifts
- Linearize only those 5
- Solve the smaller LP (5 + n + 1 constraints instead of 2n-1 + n + 1)
- Verify and accept if improved

This is much faster per iteration and more likely to find useful directions.

Actually, for the top-5 shifts LP, I only need 5 gradient vectors, each O(n) to compute. The LP has 5 + 2n + 1 constraints and n+1 variables. For n=1280: 5 + 2561 + 1 ≈ 2567 constraints, 1281 variables. Very fast.
