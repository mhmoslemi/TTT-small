import numpy as np
from scipy.optimize import minimize
import time

_OPT_N = 0
_OPT_T = 1.0
_OPT_LAMBDA = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _obj_lse(h):
    n = _OPT_N
    T = _OPT_T
    lam = _OPT_LAMBDA
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    lse = (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    if lam > 0:
        lse += lam * (np.sum(h) - n / 2.0) ** 2
    return lse

def _slsqp_optimize(h_start, n, T, max_iter, lam=0.0):
    global _OPT_N, _OPT_T, _OPT_LAMBDA
    _OPT_N = n
    _OPT_T = T
    _OPT_LAMBDA = lam
    bounds = [(0.0, 1.0)] * n
    res = minimize(_obj_lse, h_start, method='L-BFGS-B',
                   bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': 1e-16, 'gtol': 1e-14})
    return make_feasible(res.x, n)

def _slsqp_constrained(h_start, n, T, max_iter):
    global _OPT_N, _OPT_T, _OPT_LAMBDA
    _OPT_N = n
    _OPT_T = T
    _OPT_LAMBDA = 0.0

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    res = minimize(_obj_lse, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def _coord_descent(h, n, time_budget, start_time):
    # Coordinate descent with adaptive step sizes
    h = h.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)

             0.002, -0.002, 0.08, -0.08, 0.03, -0.03]
    rounds = 0
    while time.time() - start_time < time_budget and rounds < 8:
        improved = False
        for j in range(n):
            if time.time() - start_time > time_budget:
                break
            old_val = h[j]

                cand = old_val + delta

                trial = h.copy()
                trial[j] = cand

                    old_val = cand

    return h, best_c5

def _pairwise_exchange(h, n, time_budget, start_time):
    # Pairwise: try h[i]+d, h[j]-d to preserve sum
    h = h.copy()
    best_c5 = compute_c5(h)
    d = 0.02
    rng = np.random.RandomState(123)
    attempts = 0
    while time.time() - start_time < time_budget and attempts < 50000:

            continue
        attempts += 1
        if h[i] + d > 1.0 or h[j] - d < 0.0:
            continue
        trial = h.copy()
        trial[i] += d
        trial[j] -= d
        c = compute_c5(trial)
        if c < best_c5 - 1e-14:
            best_c5 = c
            h = trial
    return h, best_c5

def _fourier_init(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5
    n_modes = 12
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.01, 0.15) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * x / 2.0 + phase)
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, n)

def _periodic_init(n, period_frac, duty, offset):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    period = 2.0 * period_frac
    on_len = period * duty
    for i in range(n):
        xm = (x[i] - offset) % period
        if xm < on_len:
            h[i] = 1.0
    return make_feasible(h, n)
