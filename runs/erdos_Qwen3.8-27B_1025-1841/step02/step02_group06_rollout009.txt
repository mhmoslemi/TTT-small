        if s > target:
            lo = mid
        else:

    t = (lo + hi) / 2.0

def get_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def compute_grad_parts(h, d, n):
    """Compute the gradient contribution for a single lag d.
    d/dh[j] of corr at lag d = (1-h[j+d]) - h[j-d] (with bounds)"""
    n_half = len(h)
    grad = np.zeros(n_half)
    j0 = max(0, d)
    j1 = min(n_half, n_half + d)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[idx] += (1.0 - h[idx - d])
    j0b = max(0, -d)
    j1b = min(n_half, n_half - d)
    if j0b < j1b:
        idxb = np.arange(j0b, j1b)
        grad[idxb] -= h[idxb + d]
    return grad

def adam_smooth_max(h, n, time_limit, max_iters, tau_start, tau_end, topM, lr_start):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = get_c5(h)
    best_h = h.copy()
    start = time.time()
    
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr_start
    stall = 0
    patience = 300
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_c5 = g.max()
        
        if cur_c5 < best_c5 - 1e-13:
            best_c5 = cur_c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive tau
        progress = min(1.0, it / (max_iters * 0.8))
        tau = tau_start + (tau_end - tau_start) * (progress ** 0.5)
        
        # Select top M
        if topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_sel = g[top_idx]
        else:
            top_idx = np.arange(len(g))
            g_sel = g
        
        # Softmax weights
        m = g_sel.max()
        shifted = np.clip(tau * (g_sel - m), -400, 400)
        exp_w = np.exp(shifted)
        p = exp_w / exp_w.sum()
        
        # Weighted gradient
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            if p[i] < 1e-15:
                continue
            k = int(top_idx[i])
            d = k - n + 1
            grad += p[i] * compute_grad_parts(h, d, n)
        grad *= dx
        
        # Adam
        m_v = beta1 * m_v + (1 - beta1) * grad
        v_v = beta2 * v_v + (1 - beta2) * grad * grad
        m_hat = m_v / (1 - beta1 ** t)
        v_hat = v_v / (1 - beta2 ** t)
        cur_lr = lr * (1.0 / (1.0 + 0.001 * t))
        step = -cur_lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = project_to_feasible(h + step, n)
        
        if stall > patience:
            lr *= 0.75
            m_v *= 0.3
            v_v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
        
        h = h_new
    
    return best_h, best_c5

out[k] = sum_j h[j+k] * conj(v[j]) for k = 0, ..., len(h)+len(v)-2

Actually, from numpy docs:
"correlate(a, v) = sum_n a[n+k] * conj(v[n])"

where k ranges from -(len(v)-1) to (len(a)-1), stored in positions 0 to len(a)+len(v)-2.

So position k in the output corresponds to shift (k - len(v) + 1).

For a=h, v=1-h, len=a=n:

This is: sum_j h[j+d] * (1-h[j]) = sum_j h[j+d] - sum_j h[j+d]*h[j]

For the convolution interpretation:
conv(h, s_flip)[m] = sum_j h[j] * s_flip[m-j] = sum_j h[j] * s[n-1-(m-j)]

Setting n' = n (length of s_flip):
conv_full[m] for m = 0, ..., 2n-2

= sum_i h[i] * s[i-d] = sum_i h[i] * s_flip[n-1-(i-d)] = sum_i h[i] * s_flip[d+n-1-i]

This is conv(h, s_flip)[d+n-1] = conv_full[d+n-1]

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    return np.clip(g - t, 0.0, 1.0)


def correlate_fast(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    S = np.fft.rfft((1.0 - h)[::-1], N)
    conv = np.fft.irfft(H * S, N)[:2 * n - 1]
    return conv[::-1]


def get_c5(h):
    n = len(h)
    corr = correlate_fast(h)
    return corr.max() * (2.0 / n)


def grad_single_lag(h, d, n):
    grad = np.zeros(n)
    j0 = max(0, d)
    j1 = min(n, n + d)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[idx] += (1.0 - h[idx - d])
    j0b = max(0, -d)
    j1b = min(n, n - d)
    if j0b < j1b:
        idxb = np.arange(j0b, j1b)
        grad[idxb] -= h[idxb + d]
    return grad


def adam_smooth_max(h, n, time_limit, max_iters, tau_start, tau_end, topM, lr_start=0.12):

    h = project_to_feasible(h, n).copy()
    best_c5 = get_c5(h)
    best_h = h.copy()

    v_v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    lr = lr_start
    stall = 0
    patience = 300

    for it in range(max_iters):
        if time.time() - start > time_limit:

        t += 1

        corr = correlate_fast(h)
        g = corr * dx
        cur_c5 = g.max()

            best_c5 = cur_c5
            best_h = h.copy()

def coord_descent_refine(h, n, time_limit, max_rounds=20000):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = get_c5(h)
    best_h = h.copy()
    best_corr = correlate_fast(h)
    start = time.time()

    for rd in range(max_rounds):
        if time.time() - start > time_limit:
            break

        top_k = np.argsort(best_corr)[-8:]
        for k in top_k:
            if time.time() - start > time_limit:
                break
            d = k - n + 1
            glag = grad_single_lag(h, d, n) * dx
            pos = glag.argmax()
            if glag[pos] < 1e-10:

            if abs(delta) < 1e-8:

                h = h_try.copy()
                best_corr = correlate_fast(h)
                break

def interp_periodic(h, target_n):
    n = len(h)

        return h.copy()
    x_old = np.arange(n) * (2.0 / n)

    h_ext = np.concatenate([h, h])
    x_ext = np.concatenate([x_old, x_old + 2.0])
    return np.interp(x_new, x_ext, h_ext)


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = get_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Adam smooth max at initial resolution, moderate tau
    tl = min(remaining() * 0.25, 250.0)
    if tl > 40:
        h1, c5_1 = adam_smooth_max(best_h, n_init, tl, max_iters=50000,
                                   tau_start=5.0, tau_end=60.0, topM=120, lr_start=0.15)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: High tau refinement at initial resolution
    tl = min(remaining() * 0.15, 180.0)
    if tl > 30:
        h2, c5_2 = adam_smooth_max(best_h, n_init, tl, max_iters=40000,
                                   tau_start=40.0, tau_end=200.0, topM=40, lr_start=0.06)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Coordinate descent refinement
    tl = min(remaining() * 0.1, 90.0)
    if tl > 20:
        h3, c5_3 = coord_descent_refine(best_h, n_init, tl, max_rounds=15000)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Double resolution
    if remaining() > 200:
        n2 = n_init * 2
        h2_init = interp_periodic(best_h, n2)
        h2_init = project_to_feasible(h2_init, n2)
        tl = min(remaining() * 0.25, 300.0)
        if tl > 40:
            h4, c5_4 = adam_smooth_max(h2_init, n2, tl, max_iters=25000,
                                       tau_start=8.0, tau_end=80.0, topM=100, lr_start=0.12)
            if c5_4 < best_c5:
                best_c5 = c5_4
                best_h = h4.copy()

        tl = min(remaining() * 0.12, 120.0)
        if tl > 20:
            h4b, c5_4b = adam_smooth_max(best_h, len(best_h), tl, max_iters=20000,
                                         tau_start=60.0, tau_end=300.0, topM=30, lr_start=0.04)
            if c5_4b < best_c5:
                best_c5 = c5_4b
                best_h = h4b.copy()

    # Phase 5: Triple resolution (if time permits)
    if remaining() > 180:
        n3 = n_init * 3
        h3_init = interp_periodic(best_h, n3)
        h3_init = project_to_feasible(h3_init, n3)
        tl = min(remaining() * 0.2, 250.0)
        if tl > 40:
            h5, c5_5 = adam_smooth_max(h3_init, n3, tl, max_iters=15000,
                                       tau_start=10.0, tau_end=100.0, topM=80, lr_start=0.1)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5.copy()

    # Phase 6: Coordinate descent at best resolution
    if remaining() > 50:
        n_f = len(best_h)
        tl = min(remaining() * 0.15, 100.0)
        h6, c5_6 = coord_descent_refine(best_h, n_f, tl, max_rounds=10000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Structured restarts
    if remaining() > 100:
        for attempt in range(4):
            if remaining() < 50:
                break
            n_p = len(best_h)
            h_p = best_h.copy()
            seg_len = max(3, n_p // 60)
            n_segs = 2 + attempt
            for _ in range(n_segs):
                sp = np.random.randint(0, max(1, n_p - seg_len))
                h_p[sp:sp + seg_len] = 1.0 - h_p[sp:sp + seg_len]
            h_p = project_to_feasible(h_p, n_p)
            tl = min(remaining() * 0.18, 120.0)
            h_p, c5_p = adam_smooth_max(h_p, n_p, tl, max_iters=8000,
                                         tau_start=15.0, tau_end=120.0, topM=60, lr_start=0.1)
            if c5_p < best_c5:
                best_c5 = c5_p
                best_h = h_p.copy()

    # Phase 8: Final high-tau Adam
    if remaining() > 40:
        n_f = len(best_h)
        tl = min(remaining() * 0.25, 150.0)
        h8, c5_8 = adam_smooth_max(best_h, n_f, tl, max_iters=15000,
                                    tau_start=80.0, tau_end=400.0, topM=15, lr_start=0.03)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Phase 9: Final coordinate descent
    if remaining() > 25:
        n_f = len(best_h)
        tl = min(remaining() * 0.3, 80.0)
        h9, c5_9 = coord_descent_refine(best_h, n_f, tl, max_rounds=8000)
        if c5_9 < best_c5:
            best_c5 = c5_9
            best_h = h9.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = get_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
