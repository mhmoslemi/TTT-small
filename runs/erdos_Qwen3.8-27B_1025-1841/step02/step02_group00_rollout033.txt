```python
def corr_gradient(h, n, lag_idx):
    """Compute gradient of correlation at lag_idx (in full correlation array) w.r.t. h."""
    L = lag_idx - (n - 1)  # actual lag (can be negative)
    grad = np.zeros(n)
    if L >= 0:
        # corr = sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
        # d/dh[j] = (1-h[j+L]) for 0 <= j <= n-1-L
        #           - h[j-L] for L <= j <= n-1
        j_range = np.arange(0, n - L)
        grad[j_range] += 1.0 - h[j_range + L]
        if L > 0:
            j_range2 = np.arange(L, n)
            grad[j_range2] -= h[j_range2 - L]
    else:
        M = -L
        # corr = sum_{i=0}^{n-1-M} h[i+M]*(1-h[i])
        # d/dh[j]:
        #   j appears as h[i+M] when i=j-M, valid if 0 <= j-M <= n-1-M => M <= j <= n-1
        #     contribution: (1-h[j-M])
        #   j appears as h[i] (in 1-h[i]) when i=j, valid if 0 <= j <= n-1-M => j <= n-1-M
        #     contribution: -h[j+M]
        j_range = np.arange(M, n)
        grad[j_range] += 1.0 - h[j_range - M]
        j_range2 = np.arange(0, n - M)
        grad[j_range2] -= h[j_range2 + M]
    return grad
```

```python
def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        n_fix = min(50, n)
        if diff > 0:
            idx = np.argsort(h)[-n_fix:]
        else:
            idx = np.argsort(h)[:n_fix]
        per = diff / len(idx)
        h[idx] = np.clip(h[idx] + per, 0.0, 1.0)

    old_n = len(h)
```
