import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def grad_lags(h, lags, n, dx):
    grad = np.zeros(n)
    for k in lags:
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - d:j1 - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + d:j1b + d]
    return grad * dx


def adam_opt(h, n, tau, max_iter, time_limit, lr=0.1, topM=100):
    dx = 2.0 / n
    hw = project_to_feasible(h, n).copy()
    corr = compute_corr(hw)
    g = corr * dx
    best = g.max()
    best_h = hw.copy()
    t0 = time.time()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2, eps = 0.85, 0.999, 1e-8
    warmup = min(60, max_iter // 8)

    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        corr = compute_corr(hw)
        g = corr * dx
        if g.max() < best - 1e-14:
            best = g.max()
            best_h = hw.copy()

        t = it + 1
        if t <= warmup:
            lr_t = lr * t / warmup
        else:
            frac = (t - warmup) / max(1, max_iter - warmup)
            lr_t = lr * 0.5 * (1.0 + np.cos(np.pi * frac))

        if topM and topM < 2 * n - 1:
            top_idx = np.argpartition(g, -topM)[-topM:]
        else:
            top_idx = np.arange(2 * n - 1)

        grad = grad_lags(hw, top_idx, n, dx)

        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * grad * grad
        mh = m_v / (1.0 - b1 ** t)
        vh = v_v / (1.0 - b2 ** t)
        hw = project_to_feasible(hw - lr_t * mh / (np.sqrt(vh) + eps), n)

    return best_h, best


def interp_h(h, new_n):
    n = len(h)
    if n == new_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(new_n) * (2.0 / new_n)
    return np.interp(x_new, x_old, h, period=2.0)


def local_search(h, n, time_limit, n_trials=8000):
    best_c5 = compute_c5(h)
    best_h = h.copy()
    t0 = time.time()
    for _ in range(n_trials):
        if time.time() - t0 > time_limit:
            break
        n_pert = np.random.randint(1, 15)
        idx = np.random.choice(n, n_pert, replace=False)
        h_try = best_h.copy()
        h_try[idx] += np.random.uniform(-0.8, 0.8, n_pert)
        h_try = project_to_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-12:
            best_c5 = c5_try
            best_h = h_try.copy()
    return best_h, best_c5


def fourier_start(n, n_modes=12, seed=0):
    rng = np.random.RandomState(seed)
    x = np.arange(n) * (2.0 / n)
    h = 0.5
    for m in range(1, n_modes + 1):
        a = rng.uniform(-0.15, 0.15)
        b = rng.uniform(-0.15, 0.15)
        h = h + a * np.cos(m * np.pi * x) + b * np.sin(m * np.pi * x)
    return project_to_feasible(h, n)


def periodic_start(n, n_blocks=4, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    block_size = n // n_blocks
    on = n // (2 * n_blocks)
    for i in range(n_blocks):
        start = i * block_size + rng.randint(0, max(1, block_size // 4))
        h[start:start + on] = 1.0
    return project_to_feasible(h, n)
