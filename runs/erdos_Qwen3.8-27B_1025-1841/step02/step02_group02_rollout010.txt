```python
import numpy as np
import time

def project_to_feasible(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2}."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    """Compute smooth max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = compute_grad_smooth(h, p, n)
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def compute_grad_smooth(h, p, n):
    w = p
    A = 1.0 - h
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        pi = w[idx]
        if pi < 1e-15:
            continue
        m = idx - (n - 1)
        i0 = max(0, -m)
        i1 = min(n, n - m)
        if i0 < i1:
            grad[i0:i1] += pi * A[i0 + m:i1 + m]
        j0 = max(0, m)
        j1 = min(n, n + m)
        if j0 < j1:
            grad[j0:j1] -= pi * h[j0 - m:j1 - m]
    return grad

def adam_projected(h, n, tau, max_iters, time_limit, lr=0.01):
    """Adam optimizer with projection."""
    start = time.time()
    h = project_to_feasible(h, n).copy()
    best_obj, grad = smooth_obj_grad(h, tau)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        
        h_new = h - step
        h_new = project_to_feasible(h_new, n)
        
        if stall > 100:
            lr *= 0.5
            m *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    
    return best_h, best_obj

def subgrad_refine(h, n, max_iters, time_limit, lr=0.005):
    """Subgradient method on exact max."""
    start = time.time()
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    stall = 0
    last_improve = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
            last_improve = it
        else:
            stall += 1
        
        k_star = np.argmax(corr)
        m_shift = k_star - n + 1
        grad = np.zeros(n)
        i0 = max(0, -m_shift)
        i1 = min(n, n - m_shift)
        if i0 < i1:
            grad[i0:i1] += (1.0 - h[i0 + m_shift:i1 + m_shift])
        j0 = max(0, m_shift)
        j1 = min(n, n + m_shift)
        if j0 < j1:
            grad[j0:j1] -= h[j0 - m_shift:j1 - m_shift]
        grad *= dx
```

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def compute_grad_smooth(h, p, n):
    A = 1.0 - h
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        pi = p[idx]
        if pi < 1e-14:
            continue
        m = idx - (n - 1)
        i0 = max(0, -m)
        i1 = min(n, n - m)
        if i0 < i1:
            grad[i0:i1] += pi * A[i0 + m:i1 + m]
        j0 = max(0, m)
        j1 = min(n, n + m)
        if j0 < j1:
            grad[j0:j1] -= pi * h[j0 - m:j1 - m]
    return grad


def smooth_obj_grad(h, tau):

    p = exp_c / s
    grad = compute_grad_smooth(h, p, n) * dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad
