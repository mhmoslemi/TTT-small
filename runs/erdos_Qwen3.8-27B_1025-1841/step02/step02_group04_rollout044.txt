from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_grad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return subgrad_single(h, k_max - (n-1)) * dx
    weights = exp_s / total
    # Use top_k for efficiency
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-12:

        shift = k_idx - (n - 1)

    s = abs(shift)

    grad = np.zeros(n)

        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

        h[i], h[j] = h[j], h[i]
        
        # Compute new C5
        corr_new = np.correlate(h, 1.0 - h, mode='full') * dx
        max_new = np.max(corr_new)
        
        delta = max_new - max_best
        
        if delta < -1e-12 or rng.rand() < np.exp(-delta / (T * dx)):
            # Accept
            max_best = max_new
            corr_best = corr_new
            if max_best < best_c5:
                best_c5 = max_best
                best_h = h.copy()
        else:
            # Reject
            h[i], h[j] = h[j], h[i]
    
    return best_h, best_c5

def continuous_local_search(h, n_steps, step_size=0.1, rng=None):
    """Continuous local search: pick two positions, move one up and one down."""
    if rng is None:
        rng = np.random.RandomState(42)
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_steps):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        eps = step_size * (1.0 - 0.8 * it / n_steps)
        
        # Try moving h[i] up and h[j] down
        h_new = h.copy()
        h_new[i] = min(1.0, h[i] + eps)
        h_new[j] = max(0.0, h[j] - eps)
        
        # Check if sum is preserved
        diff = (h_new[i] - h[i]) - (h[j] - h_new[j])
        if abs(diff) > 1e-10:
            h_new[i] = h[i] + (eps - diff / 2)
            h_new[j] = h[j] - (eps + diff / 2)
            h_new = np.clip(h_new, 0, 1)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
    
    return best_h, best_c5
