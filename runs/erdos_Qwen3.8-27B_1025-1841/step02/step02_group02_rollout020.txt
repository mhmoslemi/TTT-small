import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        p = np.zeros(len(corr))
        p[int(np.argmax(corr))] = 1.0
    else:
        p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b:j1b]
    grad *= dx
    obj = dx * (max_c + np.log(max(s, 1e-300)) / tau)
    return obj, grad, p

def fast_smooth_grad(h, tau, top_k=200):
    """Faster gradient computation using only top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # Only compute gradient for significant p values
    threshold = 1e-10
    significant = np.where(p > threshold)[0]
    if len(significant) > top_k:
        # Keep only top_k
        top_idx = np.argsort(p)[::-1][:top_k]
        p_sub = p[top_idx]
        p_sub = p_sub / p_sub.sum()  # renormalize
        significant = top_idx
    
    grad = np.zeros(n)
    for k in significant:
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b:j1b]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

    for it in range(100000):

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_obj_grad(h, tau, top_k=500):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s

    mask = p > 1e-12
    if mask.sum() > top_k:
        idx = np.where(mask)[0]
        order = np.argsort(p[idx])[::-1][:top_k]
        mask = np.zeros(len(p), dtype=bool)
        mask[idx[order]] = True
        p_sub = p[mask].sum()
        p[mask] = p[mask] / p[mask].sum()

    grad = np.zeros(n)
    for k in np.where(mask)[0]:
        pk = p[k]
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 + sh:j1 + sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b:j1b]
    grad *= dx
    obj = dx * (max_c + np.log(max(s, 1e-300)) / tau)
    return obj, grad


def optimize_fista(h, n, tau, time_limit, initial_lr=None):
    dx = 2.0 / n
    x = project_to_feasible(h, n).copy()
    y = x.copy()
    x_prev = x.copy()
    t_k = 1.0
    best_c5 = compute_c5(x)
    best_h = x.copy()
    start = time.time()

    if initial_lr is None:
        lr = 0.8 / (2.0 * tau * dx)
    else:
        lr = initial_lr

    stall = 0
    best_stall_c5 = best_c5

    for it in range(200000):
        if time.time() - start > time_limit:
            break

        obj, grad = smooth_obj_grad(y, tau)

        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        mu = (t_k - 1.0) / t_new
        y_next = y + mu * (y - x_prev)
        x_candidate = y_next - lr * smooth_obj_grad(y_next, tau)[1]
        x_new = project_to_feasible(x_candidate, n)

        c5_cur = compute_c5(x_new)

def optimize_subgrad(h, n, time_limit, initial_lr=None):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.005 if initial_lr is None else initial_lr
    v = np.zeros(n)
    beta = 0.8
    stall = 0

    for it in range(200000):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        top_k = 10
        k_indices = np.argsort(corr)[::-1][:top_k]
        k_indices = k_indices[corr[k_indices] > corr.max() - 1e-8 / dx]
        if len(k_indices) == 0:
            k_indices = [np.argmax(corr)]

        grad = np.zeros(n)
        for k in k_indices:
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[j0 + sh:j1 + sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= h[j0b:j1b]
        grad *= dx / max(1, len(k_indices))

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 300:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break
        h = h_new

    return best_h, best_c5
