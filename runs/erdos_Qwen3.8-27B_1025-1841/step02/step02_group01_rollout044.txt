import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    amax = T * np.max(shifted)
    return (mc + np.log(np.sum(np.exp(T * shifted - amax))) / T + amax / T) * dx

def softmax_obj_simple(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def coord_descent_polish(h, max_rounds=10, time_limit=None, start_time=None):
    n = len(h)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    candidates_steps = [0.0, 1.0, -0.1, 0.1, -0.05, 0.05, -0.02, 0.02]
    
    for rnd in range(max_rounds):
        if time_limit and start_time and time.time() - start_time > time_limit - 15:
            break
        improved = False
        order = np.random.permutation(n)
        for j in order:
            if time_limit and start_time and time.time() - start_time > time_limit - 15:
                break
            old = h[j]
            for delta in candidates_steps:
                v = old + delta
                if v < 0.0 or v > 1.0:
                    continue
                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial
                    old = v
                    improved = True
            # Try exact 0 and 1
            for v in [0.0, 1.0]:
                if abs(v - h[j]) < 1e-12:
                    continue
                trial = h.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-12:
                    best_c5 = c
                    h = trial
                    improved = True
        if not improved:
            break
    return h, best_c5

def simulated_annealing(h, n_steps, T_init, T_final, time_limit, start_time):
    n = len(h)

            break
        progress = step / max(n_steps - 1, 1)
        T = T_init * (T_final / T_init) ** progress
        
        # Random perturbation
        trial = curr_h.copy()
        r = np.random.random()
        if r < 0.3:
            # Binary flip
            j = np.random.randint(n)
            trial[j] = 1.0 - trial[j]
        elif r < 0.5:
            # Swap
            j1, j2 = np.random.choice(n, 2, replace=False)
            trial[j1], trial[j2] = trial[j2], trial[j1]
        elif r < 0.7:
            # Set to 0 or 1
            j = np.random.randint(n)
            trial[j] = np.random.random()
        elif r < 0.85:
            # Gaussian perturbation
            j1, j2 = np.random.choice(n, 2, replace=False)
            d = np.random.uniform(-0.3, 0.3)
            trial[j1] += d
            trial[j2] -= d
        else:
            # Block flip
            block_size = np.random.randint(3, 10)
            j = np.random.randint(n)
            for k in range(block_size):
                idx = (j + k) % n
                trial[idx] = 1.0 - trial[idx]
        
        trial = make_feasible(trial, n)
        trial_c5 = compute_c5(trial)
        
        delta = trial_c5 - curr_c5
        if delta < 0 or np.random.random() < np.exp(-delta / max(T, 1e-10)):
            curr_h = trial
            curr_c5 = trial_c5
            if trial_c5 < best_c5:
                best_c5 = trial_c5
                best_h = trial.copy()
    
    return best_h, best_c5

def thue_morse_h(n, rng):
    tm = np.zeros(n)
    for i in range(n):
        cnt = bin(i).count('1')
        tm[i] = 1.0 if cnt % 2 == 0 else 0.0
    # Adjust to have exactly n/2 ones
    target = n // 2
    ones = np.where(tm == 1.0)[0]
    zeros = np.where(tm == 0.0)[0]
    n_ones = len(ones)
    if n_ones > target:
        to_flip = rng.choice(ones, n_ones - target, replace=False)
        tm[to_flip] = 0.0
    elif n_ones < target:
        to_flip = rng.choice(zeros, target - n_ones, replace=False)
        tm[to_flip] = 1.0
    return tm
