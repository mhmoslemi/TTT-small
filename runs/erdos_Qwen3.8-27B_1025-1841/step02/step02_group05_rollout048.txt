    # The delta is non-zero only at specific shifts
    # For a (1->0):
    #   First factor: shifts s where 0<=a+s<n, i.e., -a <= s <= n-1-a
    #   delta = -(1-h[a+s]) at those shifts
    #   Second factor: shifts s where 0<=a-s<n, i.e., a-(n-1) <= s <= a
    #   delta = +h[a-s] at those shifts
    # For b (0->1):
    #   First factor: shifts s where 0<=b+s<n, i.e., -b <= s <= n-1-b
    #   delta = +(1-h[b+s])
    #   Second factor: shifts s where 0<=b-s<n, i.e., b-(n-1) <= s <= b
    #   delta = -h[b-s]

    delta = np.zeros(2*n-1)
    
    # a first factor: s from max(-a, -(n-1)) to min(n-1-a, n-1)

        corr = compute_corr(h)
        max_c = corr.max()
        c5_now = max_c * dx
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-13:
                break
        
        # Smoothed gradient (log-sum-exp based)
        shifted = np.clip(tau * (corr - max_c), -500, 5
