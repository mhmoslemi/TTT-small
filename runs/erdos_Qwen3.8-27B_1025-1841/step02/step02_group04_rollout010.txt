    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-10:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt

def grad_corr_at_shift(h, k_idx):
    """Gradient of corr[k_idx] w.r.t. h, where k_idx is index in full correlation."""
    n = len(h)
    s = k_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:])
        grad[s:] -= h[:m]
    else:
        sa = -s
        m = n - sa
        grad[sa:] += (1.0 - h[:m])
        grad[:m] -= h[sa:]
    return grad

    """Nesterov ACGrad with soft-max gradient on top-k shifts."""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        # Nesterov lookahead
        y = h + momentum * v
        y = np.clip(y, 0, 1)
        
        # Compute gradient at y
        corr = np.correlate(y, 1.0 - y, mode='full')
        mc = np.max(corr)
        shifted = (corr - mc) * T
        shifted = np.clip(shifted, -50, 0)
        exp_s = np.exp(shifted)
        total = np.sum(exp_s)

                new_weights = np.zeros_like(weights)
                new_weights[top_idx] = weights[top_idx]
                new_weights /= np.sum(new_weights)
                weights = new_weights
            grad = np.zeros(n)
            nz = np.where(weights > 1e-8)[0]

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-11:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_corr_shift(h, k_idx):
    n = len(h)
    s = k_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        m = n - s
        grad[:m] += (1.0 - h[s:])
        grad[s:] -= h[:m]
    else:
        sa = -s
        m = n - sa
        grad[sa:] += (1.0 - h[:m])
        grad[:m] -= h[sa:]
    return grad

def nesterov_opt(h, n_iters, top_k, T, lr, momentum, time_limit):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iters):
        if time.time() > time_limit:
            break
        y = np.clip(h + momentum * v, 0.0, 1.0)
        corr = np.correlate(y, 1.0 - y, mode='full')
        mc = np.max(corr)
        shifted = np.clip((corr - mc) * T, -50.0, 0.0)
        exp_s = np.exp(shifted)
        tot = np.sum(exp_s)
        if tot < 1e-300:
            kmax = int(np.argmax(corr))
            grad = grad_corr_shift(y, kmax)
        else:
            w = exp_s / tot
            if top_k < len(w):
                idx = np.argsort(w)[-top_k:]
                w2 = np.zeros_like(w)
                w2[idx] = w[idx]
                w2 /= np.sum(w2)
                w = w2
            grad = np.zeros(n)
            nz = np.where(w > 1e-9)[0]
            for ki in nz:
                grad += w[ki] * grad_corr_shift(y, ki)
        grad *= dx
        h_new = make_feasible(y - lr * grad, n)
        v = h_new - h
        h = h_new
        if (it + 1) % 200 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def obj_lse(hv, T, n):
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(corr)
    sh = np.clip((corr - mc) * T, -50.0, 0.0)
    return (mc + np.log(np.sum(np.exp(sh))) / T) * (2.0 / n)

def slsqp_refine(h, T, maxiter, time_limit):
    n = len(h)
    def f(hv):
        return obj_lse(hv, T, n)
    def g(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        mc = np.max(corr)
        sh = np.clip((corr - mc) * T, -50.0, 0.0)
        ew = np.exp(sh)
        tot = np.sum(ew)
        if tot < 1e-300:
            kmax = int(np.argmax(corr))
            return grad_corr_shift(hv, kmax) * (2.0 / n)
        w = ew / tot
        grad = np.zeros(n)
        nz = np.where(w > 1e-8)[0]
        for ki in nz:
            grad += w[ki] * grad_corr_shift(hv, ki)
        return grad * (2.0 / n)
    def cons_f(hv):
        return np.sum(hv) - n / 2.0
    def cons_j(hv):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_f, 'jac': cons_j}]
    try:
        if time.time() > time_limit:
            return h, compute_c5(h)
        res = minimize(f, h, jac=g, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if time.time() > time_limit:
            return h, compute_c5(h)
        if res.x is not None and res.x.min() >= -0.01:
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)

def local_search(h, n_steps, delta, seed, time_limit):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for step in range(n_steps):
        if time.time() > time_limit:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        if j == i:
            j = (j + 1) % n
        direction = 1.0 if h[i] < 0.5 else -1.0
        new_h = h.copy()
        new_h[i] = np.clip(h[i] + direction * delta, 0.0, 1.0)
        diff = h[i] - new_h[i]
        new_h[j] = np.clip(new_h[j] - diff, 0.0, 1.0)
        new_h = make_feasible(new_h, n)
        c5_new = compute_c5(new_h)
        if c5_new < best_c5 - 1e-12:
            h = new_h
            best_c5 = c5_new
            best_h = h.copy()
    return best_h, best_c5

        if c2 < c:
            h, c = h2, c2
    return h, c
