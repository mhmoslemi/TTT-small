def project_to_feasible(g, n):
    """Project g to {h: 0<=h<=1, sum(h)=n/2}"""

    n = len(a)
    m = len(b)
    size = n + m - 1
    # For correlation, we need to correlate a with flipped b
    b_flip = b[::-1]
    A = np.fft.rfft(a, size=2*size)
    B = np.fft.rfft(b_flip, size=2*size)

    if n == target_n:

    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_max_grad(h, tau):
    """Compute smooth max of correlation and its gradient"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    return obj, grad

def adam_optimize(h, n, tau, max_iters, time_limit, lr=0.1, verbose=False):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    c5_init, _ = compute_c5(h)
    best_c5 = c5_init
    best_h = h.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    t = 0
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        
        obj, grad = smooth_max_grad(h, tau)
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, n)
        
        c5_cur, _ = compute_c5(h_new)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def upsample(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))

def find_offset(z, n):
    target = n / 2.0
    lo = -20.0
    hi = 20.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = sigmoid(z + mid).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    return (lo + hi) / 2.0

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    return obj, grad

        sd = h * (1.0 - h)
        grad_z = grad_h * sd
        m = b1 * m + (1 - b1) * grad_z
