        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += w * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= w * h[idx + sh]
    grad *= dx
    c5 = corr.max() * dx
    return c5, grad


def run_alg_smooth(h, n, time_limit, tau_start, tau_end, K, lr_init, beta, stall_patience):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = topk_grad(h, tau_start, K)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        frac = min(1.0, it / 5000.0)
        tau = tau_start * (tau_end / tau_start) ** frac
        obj, grad = topk_grad(h, tau, K)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > stall_patience:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    return best_h, best_obj


def run_alg_subgrad(h, n, time_limit, K, sharpness, lr_init, beta, stall_patience):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    stall = 0
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        c5_cur, grad = subgrad_topk(h, K, sharpness)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > stall_patience:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break
        h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: top-K smooth at initial resolution
    if rem() > 250:
        h1 = h_init.copy()
        h1, _ = run_alg_smooth(h1, n_init, rem() * 0.18, 50.0, 10.0, 12, 0.4, 0.9, 120)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: subgrad at initial resolution (top 5)
    if rem() > 300:
        h2 = best_h.copy()
        h2, c5_2 = run_alg_subgrad(h2, n_init, rem() * 0.15, 5, 8.0, 0.015, 0.82, 400)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: upgrade to n=1800
    n3 = 1800
    if rem() > 300:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        h3, _ = run_alg_smooth(h3, n3, rem() * 0.18, 60.0, 15.0, 15, 0.25, 0.9, 100)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: subgrad at n=1800
    n_cur = len(best_h)
    if rem() > 200:
        h4 = best_h.copy()
        h4, c5_4 = run_alg_subgrad(h4, n_cur, rem() * 0.15, 4, 12.0, 0.008, 0.80, 500)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: upgrade to n=3600
    n5 = 3600
    if rem() > 200:
        h5 = interp_periodic(best_h, n5)
        h5 = project_to_feasible(h5, n5)
        h5, _ = run_alg_smooth(h5, n5, rem() * 0.18, 80.0, 20.0, 20, 0.15, 0.9, 80)
        c5_5 = compute_c5(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: subgrad at n=3600
    n_cur = len(best_h)
    if rem() > 150:
        h6 = best_h.copy()
        h6, c5_6 = run_alg_subgrad(h6, n_cur, rem() * 0.15, 3, 15.0, 0.004, 0.78, 600)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: random restarts at best resolution
    n_final = len(best_h)
    if rem() > 150:
        for trial in range(8):
            if rem() < 40:
                break
            sigma = 0.005 + 0.002 * trial
            pert = np.random.randn(n_final) * sigma
            hr = best_h + pert
            hr = project_to_feasible(hr, n_final)
            tl = min(rem() * 0.12, 80.0)
            hr, _ = run_alg_smooth(hr, n_final, tl, 40.0, 10.0, 10, 0.2, 0.88, 80)
            c5r = compute_c5(hr)
            if c5r < best_c5:
                best_c5 = c5r
                best_h = hr.copy()

    # Phase 8: final subgrad polish
    if rem() > 60:
        h8 = best_h.copy()
        h8, c5_8 = run_alg_subgrad(h8, len(best_h), rem() * 0.35, 2, 20.0, 0.001, 0.75, 800)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
