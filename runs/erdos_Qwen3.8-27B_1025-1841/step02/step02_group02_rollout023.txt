def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def corr_grad_index(h, m):
    n = len(h)
    L = m - (n - 1)
    g = np.zeros(n)
    j0 = max(L, 0)
    j1 = min(n + L, n)
    if j0 < j1:
        g[j0:j1] += (1.0 - h[j0 - L:j1 - L])
    j0b = max(0, -L)
    j1b = min(n, n - L)
    if j0b < j1b:
        g[j0b:j1b] -= h[j0b + L:j1b + L]
    return g

    corr = compute_corr(h)
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = []
    
    for m in active_indices:
        g = corr_grad_index(h, m)
        a = corr[m] - np.dot(g, h)
        constraints.append(g @ h_var + a <= t)
    
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    constraints.append(cp.sum(h_var) == n / 2.0)
    
    prob = cp.Problem(cp.Minimize(t), constraints)
    
    if solver is None:
        solvers = [cp.GLOP, cp.ECOS, cp.SCS]
    else:
        solvers = [solver]
    
    last_err = None
    for sol in solvers:
        try:
            prob.solve(solver=sol)
            if h_var.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                return np.clip(h_var.value, 0.0, 1.0), t.value
        except Exception as e:

def lp_optimize(h, n, max_iters, time_limit, top_k=200, growth_rate=5):
    """LP-based optimization with adaptive active set."""
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Initialize active set
    corr = compute_corr(h)
    active = np.argsort(corr)[-top_k:]
    
    for it in range(max_iters):

            break
        
        new_h, t_val = solve_lp(h, n, active)

import numpy as np
import time
import cvxpy as cp


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def corr_grad_index(h, m):
    n = len(h)
    L = m - (n - 1)
    g = np.zeros(n)
    j0 = max(L, 0)
    j1 = min(n + L, n)
    if j0 < j1:
        g[j0:j1] += (1.0 - h[j0 - L:j1 - L])
    j0b = max(0, -L)
    j1b = min(n, n - L)
    if j0b < j1b:
        g[j0b:j1b] -= h[j0b + L:j1b + L]
    return g


def solve_lp(h, n, active_indices):
    dx = 2.0 / n
    corr = compute_corr(h)
    h_var = cp.Variable(n)
    t = cp.Variable()
    constraints = []
    for m in active_indices:
        g = corr_grad_index(h, m)
        a = dx * g
        b = dx * (corr[m] - np.dot(g, h))
        constraints.append(a @ h_var + b <= t)
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    constraints.append(cp.sum(h_var) == n / 2.0)
    prob = cp.Problem(cp.Minimize(t), constraints)
    for sol in [cp.GLOP, cp.ECOS, cp.SCS, cp.CBC]:
        try:
            prob.solve(solver=sol)
            if h_var.value is not None and prob.status in ['optimal', 'optimal_inaccurate']:
                return np.clip(np.asarray(h_var.value).flatten(), 0.0, 1.0), t.value
        except Exception:
            continue
    return None, None


def lp_optimize(h, n, max_iters, time_limit, top_k=200):
    start = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = compute_corr(h)
    active = np.argsort(corr)[-top_k:]
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        new_h, t_val = solve_lp(h, n, active)
        if new_h is None:
            active = np.argsort(corr)[-max(50, top_k // 2):]
            stall += 1
            if stall > 10:
                break
            continue
        new_h = project_to_feasible(new_h, n)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5 - 1e-12:
            best_c5 = new_c5
            best_h = new_h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 30:
                break
        new_corr = compute_corr(new_h)
        threshold = new_c5 / (2.0 / n)
        violated = np.where(new_corr > threshold + 0.5)[0]
        if len(violated) > 0:
            active = np.unique(np.concatenate([active, violated]))
            if len(active) > top_k + 100:
                av = new_corr[active]
                keep = np.argsort(av)[-top_k:]
                active = active[keep]
        corr = new_corr
        h = new_h
    return best_h, best_c5


def optimize_smooth(h, n, tau, max_iters, time_limit, initial_lr=None):
    start = time.time()
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = 0.1 if initial_lr is None else initial_lr
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * (2.0 / n)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -400, 400)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        grad = np.zeros(n)
        dx = 2.0 / n
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-14:
                continue
            g = corr_grad_index(h, k)
            grad += pk * g
        grad *= dx
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > 60:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    return best_h, best_c5
