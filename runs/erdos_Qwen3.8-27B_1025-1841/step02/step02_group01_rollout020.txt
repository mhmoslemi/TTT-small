def optimal_delta(base, a, h_j):
    # Find delta that minimizes max_s (base[s] + delta * a[s])
    # subject to 0 <= h_j + delta <= 1
    # i.e., -h_j <= delta <= 1 - h_j
    
    delta_lo = -h_j
    delta_hi = 1.0 - h_j
    
    if delta_lo >= delta_hi:
        return 0.0
    
    # f(delta) = max_s (base[s] + delta * a[s]) is convex
    # Minimum is where subgradient crosses 0
    # Use bisection on subgradient
    
    def subgrad(delta):
        vals = base + delta * a
        return np.max(a * (vals == np.max(vals)))
        # Actually, for convex f, subgradient is the set of a_s where s is argmax
        # A single subgradient is a_s for the argmax s
    
    # Simpler: ternary search
    for _ in range(50):
        m1 = delta_lo + (delta_hi - delta_lo) / 3
        m2 = delta_hi - (delta_hi - delta_lo) / 3
        f1 = np.max(base + m1 * a)
        f2 = np.max(base + m2 * a)
        if f1 < f2:
            delta_hi = m2
        else:
            delta_lo = m1
    
    return (delta_lo + delta_hi) / 2

            h[idx] = 0
        elif ones < n_half:
            idx = rng.choice(np.where(h==0)[0], n_half - ones, replace=False)
            h[idx] = 1
        h = project_sum(h, 320)
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Phase 1: Coordinate descent at parent resolution (n=320)
    n = len(h)
    h_work = h.copy()
    max_passes_p1 = 40
    for p in range(max_passes_p1):
        if time.time() - start_time > budget_s * 0.08:
            break
        h_work, _ = coord_descent_pass(h_work, n)
        h_work = project_sum(h_work, n)
        c5 = compute_c5(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()
    
    # Phase 2: SLSQP at n=320 for a different landscape view
    if time.time() - start_time < budget_s * 0.15:
        h_slq = slsqp_polish(best_h, 320, maxiter=300)
        c5 = compute_c5(h_slq)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_slq.copy()
    
    # Phase 3: Multi-resolution coordinate descent
    resolutions = [480, 640, 960, 1280, 1920]
    for n_r in resolutions:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 80
        if remaining < 30:
            break
        
        h_work = upsample_h(best_h, n_r)
        
        # Determine passes based on time
        # Estimate: ~0.3 * (n_r/320)^2 seconds per pass roughly
        est_pass_time = 0.02 * (n_r / 320.0) ** 2
        n_passes = int(remaining / (est_pass_time + 0.01))
        n_passes = max(5, min(n_passes, 500))
        
        improved_last = 0
        for p in range(n_passes):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 80:
                break
            
            h_work, _ = coord_descent_pass(h_work, n_r)
            h_work = project_sum(h_work, n_r)
            c5 = compute_c5(h_work)
            
            if c5 < best_c5 - 1e-10:
                best_c5 = c5
                best_h = h_work.copy()
                improved_last = 0
            else:
                improved_last += 1
            
            # Early stop if no improvement for 15 passes
            if improved_last >= 15 and p > 10:
                break
    
    # Phase 4: Random restarts at best resolution
    n_best = len(best_h)
    elapsed = time.time() - start_time
    if budget_s - elapsed > 100:
        n_try = min(n_best, 960)  # use slightly lower res for speed on restarts
        h_try = upsample_h(best_h, n_try) if n_best != n_try else best_h.copy()
        
        for attempt in range(8):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 30:
                break
            
            noise_scale = 0.03 + 0.02 * attempt
            h_r = best_h.copy()
            if n_best != n_try:
                h_r = upsample_h(best_h, n_try)
            h_r = h_r + rng.randn(n_try) * noise_scale
            h_r = project_sum(h_r, n_try)
            
            n_r = len(h_r)
            max_p = int((budget_s - 40 - (time.time() - start_time)) / (0.02 * (n_r/320.0)**2 + 0.01))
            max_p = max(5, min(max_p, 100))
            
            for p in range(max_p):
                if time.time() - start_time > budget_s - 30:
                    break
                h_r, _ = coord_descent_pass(h_r, n_r)
                h_r = project_sum(h_r, n_r)
                c5 = compute_c5(h_r)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_r.copy()
    
    # Phase 5: Final SLSQP polish at best resolution (if time allows)
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60 and len(best_h) <= 640:
        h_final = slsqp_polish(best_h, len(best_h), maxiter=200)
        c5 = compute_c5(h_final)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
    
    # Final coordinate descent polish
    elapsed = time.time() - start_time
    if budget_s - elapsed > 20:
        n_f = len(best_h)
        h_f = best_h.copy()
        n_final_passes = int((budget_s - 10 - (time.time() - start_time)) / (0.02 * (n_f/320.0)**2 + 0.01))
        n_final_passes = max(2, min(n_final_passes, 50))
        for p in range(n_final_passes):
            if time.time() - start_time > budget_s - 5:
                break
            h_f, _ = coord_descent_pass(h_f, n_f)
            h_f = project_sum(h_f, n_f)
            c5 = compute_c5(h_f)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_f.copy()
    
    # Ensure validity
    best_h = project_sum(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
