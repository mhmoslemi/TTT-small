import numpy as np

def project_to_feasible(g, n):
    # Bisection for water-filling
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):  # more iterations for precision
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def gradient_for_shift(h, d, n):
    """Gradient of f_d(h) = sum_i h[i]*(1-h[i+d]) with respect to h."""
    grad = np.zeros(n)
    j0 = max(0, d)
    j1 = min(n, n + d)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[idx] += (1.0 - h[idx - d])
    j0b = max(0, -d)
    j1b = min(n, n - d)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        grad[idx] -= h[idx + d]
    return grad

    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    obj = (corr * dx).max()
    best_obj = obj
    best_h = h.copy()
    start = time.time()

    beta = 0.9

    patience = max(200, max_iters // 20)
    lr = lr_init

        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if use_topk:
            # Top-K subgradient
            top_idx = np.argpartition(corr, -topK)[-topK:]
            top_idx = top_idx[np.argsort(corr[top_idx])[::-1]]  # sort descending
            grad = np.zeros(n)

            grad *= dx / topK

                p_top = exp_c / s

                    d = top_idx[i] - n + 1

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def gradient_shift(h, d, n):
    grad = np.zeros(n)
    j0 = max(0, d)
    j1 = min(n, n + d)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[idx] += (1.0 - h[idx - d])
    j0b = max(0, -d)
    j1b = min(n, n - d)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        grad[idx] -= h[idx + d]
    return grad


def softmax_optimize(h, n, tau, max_iters, time_limit, lr_init, topM=0):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_obj = (corr * dx).max()
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(200, max_iters // 20)
    lr = lr_init

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p_top = exp_c / s
            grad = np.zeros(n)
            for i in range(len(top_idx)):
                if p_top[i] < 1e-15:
                    continue
                d = top_idx[i] - n + 1
                grad += p_top[i] * gradient_shift(h, d, n)
        else:
            max_g = g.max()
            shifted = np.clip(tau * (g - max_g), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p = exp_c / s
            grad = np.zeros(n)
            for k in range(2 * n - 1):
                if p[k] < 1e-15:
                    continue
                d = k - n + 1
                grad += p[k] * gradient_shift(h, d, n)
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > patience:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
    return best_h, best_obj


def topk_optimize(h, n, max_iters, time_limit, lr_init, topK=2):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_obj = (corr * dx).max()
    best_h = h.copy()

            stall = 0
        else:
            stall += 1

        top_idx = np.argpartition(corr, -topK)[-topK:]

            grad += gradient_shift(h, d, n)

            v *= 0.2
            stall = 0

    x_old = np.arange(n) * (2.0 / n)
