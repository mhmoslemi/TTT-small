        h0 = np.full(320, 0.5)

    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    n0 = len(best_h)
    resolutions = sorted(set([n0, n0 * 2, n0 * 4, n0 * 8]))
    resolutions = [r for r in resolutions if r <= 5120]

    for res_idx, n in enumerate(resolutions):
        elapsed = time.time() - start_time
        remaining = deadline - elapsed
        if remaining < 30:
            break
        n_res = len(resolutions)
        frac = (n0 ** (1.0)) / sum(r ** 1.0 for r in resolutions)
        alloc = remaining * frac * 1.2
        if res_idx == n_res - 1:
            alloc = remaining * 0.9
        sweep_deadline = time.time() + alloc

        if n != len(best_h):
            h = upsample_h(best_h, n)
        else:
            h = best_h.copy()

        h, c5 = coord_descent(h, n, deadline=sweep_deadline, max_sweeps=500)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    n_best = len(best_h)

    # Try perturbed restarts
    remaining = deadline - time.time()
    if remaining > 60:
        n_restarts = min(4, int(remaining // 100) + 1)
        for att in range(n_restarts):
            if time.time() > deadline - 30:
                break
            h_try = best_h + rng.randn(n_best) * 0.03 * (att + 1)
            h_try = make_feasible(h_try, n_best)
            sub_deadline = min(deadline - 30, time.time() + remaining / (n_restarts + 1))
            h_try, c5_try = coord_descent(h_try, n_best, deadline=sub_deadline, max_sweeps=200)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

    # Try binary local search at best resolution
    remaining = deadline - time.time()
    if remaining > 40:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = random_binary_h(n_best, rng)
        # Use best_h as guide
        h_bin = np.where(best_h > 0.5, 1.0, 0.0)
        n_ones = int(np.sum(h_bin))
        if n_ones > n_best // 2:
            ones_idx = np.where(h_bin == 1.0)[0]
            to_flip = rng.choice(ones_idx, n_ones - n_best // 2, replace=False)
            h_bin[to_flip] = 0.0
        elif n_ones < n_best // 2:
            zeros_idx = np.where(h_bin == 0.0)[0]
            to_flip = rng.choice(zeros_idx, n_best // 2 - n_ones, replace=False)
            h_bin[to_flip] = 1.0
        h_bin, c5_bin = coord_descent(h_bin, n_best, deadline=min(deadline - 15, time.time() + remaining * 0.4), max_sweeps=200)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()

    # Final SLSQP polish
    remaining = deadline - time.time()
    if remaining > 50 and best_h is not None:
        n = len(best_h)
        dx = 2.0 / n
        t_param = 80.0

        def obj_func(h_x):
            c = np.correlate(h_x, 1.0 - h_x, mode='full')
            mc = c.max()
            shifted = c - mc
            soft = mc + np.log(np.sum(np.exp(t_param * shifted))) / t_param
            return soft * dx

        def obj_grad(h_x):
            c = np.correlate(h_x, 1.0 - h_x, mode='full')
            mc = c.max()
            shifted = c - mc
            w = np.exp(t_param * shifted)
            w /= w.sum()
            k_all = np.arange(len(c))
            grad = np.zeros(n)
            for j in range(0, n, max(1, n // 200)):
                idx1 = j + n - 1 - k_all
                v1 = (idx1 >= 0) & (idx1 < n)
                idx2 = j + k_all - n + 1
                v2 = (idx2 >= 0) & (idx2 < n)
                b = np.zeros(len(c))
                b[v1] = 1.0 - h_x[idx1[v1]]
                b[v2] -= h_x[idx2[v2]]
                grad[j] = dx * np.dot(w, b)
            return grad

        def eq_fun(h_x):
            return np.sum(h_x) - n / 2.0

        def eq_jac(h_x):
            return np.ones(n)

        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}]
        slp_deadline = min(deadline - 5, time.time() + remaining * 0.6)
        try:
            res = minimize(obj_func, best_h, method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-13})
            h_polish = make_feasible(res.x, n)
            c5_polish = compute_c5(h_polish)
            if c5_polish < best_c5:
                best_c5 = c5_polish
                best_h = h_polish
        except Exception:
            pass

    # One final coordinate descent pass
    remaining = deadline - time.time()
    if remaining > 20:
        best_h, c5_final_cd = coord_descent(best_h, len(best_h), deadline=deadline - 5, max_sweeps=50)
        if c5_final_cd < best_c5:
            best_c5 = c5_final_cd
            best_h = best_h

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
