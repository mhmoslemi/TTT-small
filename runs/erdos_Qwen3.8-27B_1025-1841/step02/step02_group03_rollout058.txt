Average per lag = n^2/4 / (2(n-1)) ≈ n/8.
So average corr * dx ≈ (n/8) * (2/n) = 1/4 = 0.25.

import numpy as np
from scipy.optimize import minimize
import time

_N = 960
_T = 50.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _obj_grad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = float(np.max(corr))
    shifted = corr - mc
    exp_vals = np.exp(np.clip(T * shifted, -700.0, 0.0))
    sum_exp = np.sum(exp_vals)
    if sum_exp < 1e-300:
        return mc * dx, np.zeros(n)
    obj = (mc + np.log(sum_exp) / T) * dx
    weights = exp_vals / sum_exp
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        w = weights[m]
        if w < 1e-10:
            continue
        shift = m - (n - 1)
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - x[shift:])
            grad[shift:] -= w * x[:n - shift]
        else:
            s = -shift
            grad[s:] += w * (1.0 - x[:n - s])
            grad[:n - s] -= w * x[s:]
    grad *= dx
    return obj, grad

def _obj_only(x):
    return _obj_grad(x, _N, _T)[0]

def _grad_fn(x):
    return _obj_grad(x, _N, _T)[1]

def _c_fun(x):
    return np.sum(x) - _N / 2.0

def _c_jac(x):
    return np.ones(_N)

def _slsqp(h_start, n, T, maxiter):
    global _N, _T
    _N = n
    _T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _c_fun, 'jac': _c_jac}]
    try:
        res = minimize(_obj_only, h_start, jac=_grad_fn,
                       method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except Exception:
        return h_start.copy()

def _binary_snap(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        for val in (0.0, 1.0):
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def _coord_descent(h, n, t0, time_limit=None):
    best = h.copy()
    best_c = compute_c5(h)
    end_time = t0 + (time_limit if time_limit else 60.0)
    for rnd in range(10):
        if time.time() > end_time:
            break
        improved = False
        for j in range(n):
            if time.time() > end_time:
                break
            old = best[j]
            bc = best_c
            bt = old
            for cand in (0.0, 1.0, 0.5):
                if abs(cand - old) < 1e-12:
                    continue
                trial = best.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-14:
                    bc = c
                    bt = cand
            if bt != old:
                best = best.copy()
                best[j] = bt
                best = make_feasible(best, n)
                best_c = bc
                improved = True
        if not improved:
            break
    return best, best_c

def _subgradient_step(h, n, step_size, t0):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_val = float(np.max(corr))
    active = np.where(corr >= max_val - 1e-10)[0]
    grad = np.zeros(n)
    weights = np.ones(len(active)) / len(active)
    for idx, m in enumerate(active):
        shift = m - (n - 1)
        w = weights[idx]
        if shift >= 0:
            grad[:n - shift] += w * (1.0 - h[shift:])
            grad[shift:] -= w * h[:n - shift]
        else:
            s = -shift
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] -= w * h[s:]
    grad *= dx
    h_new = h - step_size * grad
    h_new = make_feasible(h_new, n)
    return h_new, compute_c5(h_new)
