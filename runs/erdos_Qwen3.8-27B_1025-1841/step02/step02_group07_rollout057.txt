    corr = np.correlate(h, 1.0 - h, mode='full')
    obj, _ = soft_max_and_probs(corr, T)
    return obj * dx

def obj_grad_sym(h_r, T):
    full = sym_to_full(h_r)
    n = len(full)
    dx = 2.0 / n
    corr = np.correlate(full, 1.0 - full, mode='full')
    obj, probs = soft_max_and_probs(corr, T)
    grad_full = np.zeros(n)
    center = n - 1
    for idx in range(len(corr)):
        p = probs[idx]
        if p < 1e-15:
            continue
        lag = idx - center
        if lag >= 0:
            if lag > 0:
                j1 = np.arange(lag, n)
                grad_full[j1] += p * (1.0 - full[j1 - lag])
            j2 = np.arange(0, n - lag)
            grad_full[j2] -= p * full[j2 + lag]
        else:
            ll = -lag
            j1 = np.arange(0, n - ll)
            grad_full[j1] += p * (1.0 - full[j1 + ll])
            j2 = np.arange(ll, n)
            grad_full[j2] -= p * full[j2 - ll]
    grad_full *= dx
    half = len(h_r)
    grad_r = grad_full[:half].copy()
    for i in range(half):
        grad_r[i] += grad_full[n - 1 - i]
    return obj * dx, grad_r

def lfgb_sym(h_r, T, maxiter=500, deadline=None):
    def f(x):
        val, _ = obj_grad_sym(x, T)
        return val
    def g(x):
        _, gr = obj_grad_sym(x, T)
        return gr
    try:
        res = minimize(f, h_r.copy(), method='L-BFGS-B', jac=g,
                       bounds=[(0.0, 1.0)] * len(h_r),
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 5000})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sym(res.x)
    except Exception:
        pass
    return h_r.copy()

def lfgb_full(h, T, maxiter=300, deadline=None):
    def f(x):
        return obj_soft(x, T)
    def g(x):
        return grad_analytic(x, T)
    try:
        res = minimize(f, h.copy(), method='L-BFGS-B', jac=g,
                       bounds=[(0.0, 1.0)] * len(h),
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-12, 'maxfun': 5000})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_full(res.x, len(h))
    except Exception:
        pass
    return h.copy()

def coord_descent(h, T, n_sweeps=5, deadline=None):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    for sweep in range(n_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        for i in range(n):
            if deadline is not None and time.time() > deadline:
                break
            lo = max(0.0, h[i] - 0.15)
            hi = min(1.0, h[i] + 0.15)
            if lo >= hi:
                continue
            def f1d(t):
                h[i] = t
                c = np.correlate(h, 1.0 - h, mode='full')
                obj, _ = soft_max_and_probs(c, T)
                h[i] = t  # keep
                return obj * dx
            saved = h[i]
            try:
                res = minimize_scalar(f1d, bounds=(lo, hi), method='bounded',
                                      options={'maxiter': 30})
                if res.success and res.fun < f1d(saved):
                    h[i] = res.x
            except Exception:
                h[i] = saved
    return project_full(h, n)

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10.0

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(336, 0.5)
    n0 = len(h0)
    h0 = project_full(h0, n0)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Symmetric optimization
    h_sym = make_symmetric(best_h)
    n_s = len(h_sym)
    h_sym = project_full(h_sym, n_s)
    c5_s = evaluate_c5(h_sym)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_sym.copy()

    half = n_s // 2
    h_r = h_sym[:half].copy()
    h_r = project_sym(h_r)

    sym_temps = [50.0, 100.0, 200.0, 400.0, 800.0, 1500.0]
    for T in sym_temps:
        if time.time() > deadline - 120:
            break
        remaining = max(30, (deadline - time.time()) * 0.4)
        maxiter = min(600, max(100, int(remaining * 8)))
        h_r = lfgb_sym(h_r, T, maxiter=maxiter, deadline=deadline)
        full = sym_to_full(h_r)
        c5_new = evaluate_c5(full)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = full.copy()

    # Phase 2: Coordinate descent polish on symmetric
    if time.time() < deadline - 60:
        full = sym_to_full(project_sym(best_h[:half]))
        full = coord_descent(full, 300.0, n_sweeps=8, deadline=deadline)
        full_sym = make_symmetric(full)
        full_sym = project_full(full_sym, n_s)
        c5_cd = evaluate_c5(full_sym)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = full_sym.copy()

    # Phase 3: Non-symmetric L-BFGS-B refinement
    if time.time() < deadline - 60:
        h_work = best_h.copy()
        for T in [100.0, 300.0, 600.0]:
            if time.time() > deadline - 30:
                break
            maxiter = min(300, max(50, int((deadline - time.time() - 20) * 3)))
            h_work = lfgb_full(h_work, T, maxiter=maxiter, deadline=deadline)
            c5_new = evaluate_c5(h_work)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_work.copy()

    # Phase 4: Coordinate descent on best (non-symmetric)
    if time.time() < deadline - 30:
        h_cd = coord_descent(best_h, 300.0, n_sweeps=5, deadline=deadline)
        c5_cd = evaluate_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: Random restarts
    n_best = len(best_h)
    rng = np.random.RandomState(seed + 777)
    n_restarts = max(3, int((deadline - time.time() - 10) / 8.0))
    for attempt in range(min(n_restarts, 25)):
        if time.time() > deadline - 5:
            break
        scale = 0.04 + 0.04 * (attempt % 3)
        h_pert = best_h + rng.randn(n_best) * scale
        h_pert = project_full(h_pert, n_best)
        T_r = 200.0 + 100.0 * (attempt % 3)
        maxiter_r = min(200, max(50, int((deadline - time.time() - 5) * 2)))
        h_new = lfgb_full(h_pert, T_r, maxiter=maxiter_r, deadline=deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 6: Try a different resolution from the best solution
    if time.time() < deadline - 40:
        for n_try in [448, 504]:
            if time.time() > deadline - 20:
                break
            h_hi = interp_to(best_h, n_best, n_try)
            h_hi = project_full(h_hi, n_try)
            h_sym_hi = make_symmetric(h_hi)
            h_sym_hi = project_full(h_sym_hi, n_try)
            half_hi = n_try // 2
            h_r_hi = h_sym_hi[:half_hi].copy()
            h_r_hi = project_sym(h_r_hi)
            for T in [200.0, 500.0]:
                if time.time() > deadline - 10:
                    break
                h_r_hi = lfgb_sym(h_r_hi, T, maxiter=200, deadline=deadline)
                full_hi = sym_to_full(h_r_hi)
                c5_hi = evaluate_c5(full_hi)
                if c5_hi < best_c5:
                    best_c5 = c5_hi
                    best_h = full_hi.copy()
                    n_best = n_try

    # Final: one more coordinate descent
    if time.time() < deadline - 5:
        h_cd = coord_descent(best_h, 500.0, n_sweeps=3, deadline=deadline)
        c5_cd = evaluate_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
