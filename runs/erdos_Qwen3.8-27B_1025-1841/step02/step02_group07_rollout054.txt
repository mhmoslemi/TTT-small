    m = np.max(scaled)
    exps = np.exp(scaled - m)
    probs = exps / np.sum(exps)

    B = np.zeros(n)  # sum_m probs[m] * h[j - m + (n-1)] for valid

def project(h, n):

    h = np.asarray(h, dtype=float)

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h_p = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    d = target - np.sum(h_p)
    if abs(d) > 1e-13:
        if d > 0:

        else:
            msk = h_p > 1e-13
        cnt = msk.sum()
        if cnt > 0:
            h_p[msk] += d / cnt
            np.clip(h_p, 0.0, 1.0, out=h_p)

            if obj_new > obj + 1e-14:
                lr *= 0.5
            else:
                lr = min(lr * 1.1, 2.0)
            h = h_new
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
