                        h_up = ht2.copy()
                except Exception:
                    pass

        # L-BFGS-B with penalty at this resolution
        if elapsed() < budget_s - 50:
            def _pen2(h2):
                h2c = np.clip(h2, 0.0, 1.0)
                pen = (np.sum(h2c) - n2 / 2.0) ** 2 * 80.0
                return softmax_obj(h2c, n2, 100) + pen
            try:
                res2b = minimize(_pen2, np.clip(h_up, 0.001, 0.999), method='L-BFGS-B',
                               bounds=[(0.0, 1.0)] * n2,
                               options={'maxiter': 250, 'ftol': 1e-15})
                ht2b = make_feasible(res2b.x, n2)
                c5t2b = compute_c5(ht2b)
                if c5t2b < best_c5:
                    best_c5 = c5t2b
                    best_h = ht2b.copy()
                    h_up = ht2b.copy()
            except Exception:
                pass

    # Phase 4: Simulated annealing with block swap moves
    n3 = len(best_h)
    h_sa = best_h.copy()
    c_sa = compute_c5(h_sa)
    T_sa = 0.0004
    for it in range(6000):
        if elapsed() > budget_s - 50:
            break
        blen = rng.randint(1, 4)
        i = rng.randint(0, n3 - blen + 1)
        j = rng.randint(0, n3 - blen + 1)
        if abs(i - j) < blen:
            continue
        h_new = h_sa.copy()
        h_new[i:i+blen], h_new[j:j+blen] = h_sa[j:j+blen].copy(), h_sa[i:i+blen].copy()
        c_new = compute_c5(h_new)
        if c_new <= c_sa + T_sa * 0.1:
            if c_new < c_sa:
                h_sa = h_new
                c_sa = c_new
                if c_sa < best_c5:
                    best_c5 = c_sa
                    best_h = h_sa.copy()
            else:
                if rng.rand() < np.exp(-(c_new - c_sa) / T_sa):
                    h_sa = h_new
                    c_sa = c_new
        T_sa *= 0.9997

    # Phase 5: Intensive coordinate descent
    n4 = len(best_h)
    offsets = [0.0005, 0.001, 0.002, 0.005, 0.01, 0.02, 0.05]
    for rd in range(50):
        if elapsed() > budget_s - 5:
            break
        imp = False
        for j in range(n4):
            if elapsed() > budget_s - 3:
                break
            ov = best_h[j]
            cands = [0.0, 1.0]
            for s in offsets:
                cands.append(ov + s)
                cands.append(ov - s)
            cands.append(ov * 0.5)
            for c in cands:
                if c < 0 or c > 1:
                    continue
                trial = best_h.copy()
                trial[j] = c
                trial = make_feasible(trial, n4)
                cv = compute_c5(trial)
                if cv < best_c5 - 1e-14:
                    best_c5 = cv
                    best_h = trial
                    imp = True
                    break
        if not imp and rd > 10:
            break

    # Phase 6: Final L-BFGS-B polish
    if elapsed() < budget_s - 15:
        n5 = len(best_h)
        def _pen5(h5):
            h5c = np.clip(h5, 0.0, 1.0)
            pen = (np.sum(h5c) - n5 / 2.0) ** 2 * 200.0
            return softmax_obj(h5c, n5, 150) + pen
        try:
            res5 = minimize(_pen5, np.clip(best_h, 0.001, 0.999), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n5,
                           options={'maxiter': 300, 'ftol': 1e-16})
            h5 = make_feasible(res5.x, n5)
            c5_5 = compute_c5(h5)
            if c5_5 < best_c5:
                best_c5 = c5_5
                best_h = h5
        except Exception:
            pass

    # Final tiny coordinate descent
    if elapsed() < budget_s - 3:
        n6 = len(best_h)
        for j in range(n6):
            if elapsed() > budget_s - 1:
                break
            ov = best_h[j]
            for s in [0.0002, 0.0005, 0.001]:
                for sign in [1.0, -1.0]:
                    c = ov + sign * s
                    if c < 0 or c > 1:
                        continue
                    trial = best_h.copy()
                    trial[j] = c
                    trial = make_feasible(trial, n6)
                    cv = compute_c5(trial)
                    if cv < best_c5 - 1e-15:
                        best_c5 = cv
                        best_h = trial

    best_h = make_feasible(best_h, len(best_h))
    fc5 = compute_c5(best_h)
    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
