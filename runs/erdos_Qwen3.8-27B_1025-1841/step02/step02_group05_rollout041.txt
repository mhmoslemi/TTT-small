def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

    idx = np.argpartition(corr, -k)[-k:]

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_subgrad_k1(corr, h, n):
    kk = np.argmax(corr)
    sh = kk - n + 1
    subgrad = np.zeros(n)
    if sh >= 0:
        if n - sh > 0:
            subgrad[:n - sh] += (1.0 - h[sh:n])
        if sh < n:
            subgrad[sh:] -= h[:n - sh]
    else:
        s = -sh
        if n - s > 0:
            subgrad[s:] += (1.0 - h[:n - s])
        if s < n:
            subgrad[:n - s] -= h[s:n]
    return subgrad

def compute_subgrad_topk(corr, h, n, k, p_power):
    if k <= 1:
        return compute_subgrad_k1(corr, h, n)
    idx = np.argpartition(corr, -k)[-k:]
    top_corr = corr[idx]
    max_tc = top_corr.max()
    w = np.exp(p_power * (top_corr - max_tc))
    w /= w.sum()
    subgrad = np.zeros(n)
    for i in range(k):
        kk = idx[i]
        sh = kk - n + 1
        wi = w[i]
        if wi < 1e-10:
            continue
        if sh >= 0:
            if n - sh > 0:
                subgrad[:n - sh] += wi * (1.0 - h[sh:n])
            if sh < n:
                subgrad[sh:] -= wi * h[:n - sh]
        else:
            s = -sh
            if n - s > 0:
                subgrad[s:] += wi * (1.0 - h[:n - s])
            if s < n:
                subgrad[:n - s] -= wi * h[s:n]
    return subgrad

    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    alpha = alpha_init
    v = np.zeros(n)
    stall = 0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        frac = min(1.0, elapsed / time_limit)
        k = max(k_min, int(k_start * (1.0 - frac) ** 1.2))
        p_power = 1.0 + 6.0 * frac * frac

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * (2.0 / n)

        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > stall_limit:
            alpha *= alpha_decay
            v *= 0.2
            stall = 0
            if alpha < 1e-10:
                break

        subgrad = compute_subgrad_topk(corr, h, n, k, p_power)
        v = momentum * v - alpha * subgrad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5
