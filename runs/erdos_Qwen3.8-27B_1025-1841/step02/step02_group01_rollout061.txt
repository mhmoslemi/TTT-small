            if diff2 > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff2 / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def project_onto_constraint(h, n):
    """Project h onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda: find lambda s.t. sum(clip(h - lambda, 0, 1)) = target
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = proj.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0.0, 1.0)

def soft_max_value_and_weights(corr, temp):
    mc = corr.max()
    shifted = corr - mc
    ex = np.exp(temp * shifted)
    ex_sum = ex.sum()
    w = ex / ex_sum
    val = mc + np.log(ex_sum) / temp
    return val, w

def smooth_max_gradient(h, w):
    """Compute gradient of smooth max of correlations w.r.t. h.
    w: weights for each correlation lag (length 2n-1)
    """
    n = len(h)
    # Part 1: sum_m w[m] * 1[0 <= j-(m-n+1) < n]
    # = sum_m w[m] * 1[n-1-j <= m <= n-1+j] * 1[0<=m<=2n-2]
    cumw = np.concatenate([[0.0], np.cumsum(w)])
    part1 = np.zeros(n)
    for j in range(n):
        lo = max(0, n - 1 - j)
        hi = min(2 * n - 2, n - 1 + j)
        part1[j] = cumw[hi + 1] - cumw[lo]
    
    # Part 2: A[j] = sum_m w[m] * h[j-m+n-1] = convolve(h, w, 'full')[j+n-1]
    conv_hw = convolve(h, w, 'full', method='auto')
    A = conv_hw[n-1 : 2*n-1]  # indices n-1 to 2n-2, length n
    
    # Part 3: C[j] = sum_m w[m] * h[j+m-n+1] = convolve(h, w[::-1], 'full')[j+n-1]
    conv_hwf = convolve(h, w[::-1], 'full', method='auto')
    C = conv_hwf[n-1 : 2*n-1]
    
    return part1 - A - C

def slsqp_smooth_max(h, n, temp, max_iter=300):
    """Optimize smooth max of correlations using SLSQP."""
    dx = 2.0 / n
    
    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        val, _ = soft_max_value_and_weights(corr, temp)
        return val * dx
    
    def grad(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        _, w = soft_max_value_and_weights(corr, temp)
        g = smooth_max_gradient(hv, w)
        return g * dx
    
    bounds = [(0.0, 1.0)] * n
    constraint = {'type': 'eq', 'fun': lambda hv: hv.sum() - n/2.0, 'jac': lambda hv: np.ones(n)}
    
    res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds,
                   constraints=[constraint], options={'maxiter': max_iter, 'ftol': 1e-12})
    return res.x

def coord_descent_polish(h, n, max_passes=30, time_limit=None, start_time=None):
    """Coordinate descent: for each coordinate, try small changes."""
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    steps = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for p in range(max_passes):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > budget_s_remaining:  # handled by caller
                break
        improved = False
        for j in range(n):
            for s in steps:
                for direction in (+1, -1):
                    h_trial = h.copy()
                    h_trial[j] = np.clip(h[j] + direction * s, 0.0, 1.0)
                    h_trial = make_feasible(h_trial, n)
                    c = compute_c5(h_trial)
                    if c < best_c5 - 1e-12:
                        best_c5 = c
                        h = h_trial.copy()
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    return h, best_c5

2. For each coordinate j, compute how the correlation changes as a function of h[j]
3. Find the optimal h[j] directly

When we change h[j] by δ (and adjust the sum by distributing -δ/n to all other coordinates, or to a single other coordinate), the correlation changes in a specific way.

Actually, the issue is that changing h[j] and redistributing the sum error to all other coordinates changes all of them, which changes all correlations. This makes the 1D analysis complex.

Alternative: handle the sum constraint separately. Optimize h[j] ignoring the sum constraint, then project. This won't be exact but might work as a heuristic.

Δcorr[m] = δ * [(1-h[j-s_m]) - h[j'+s_m]... ] (complex)

import numpy as np
from scipy.optimize import minimize
from scipy.signal import convolve
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h += diff / n
        h = np.clip(h, 0.0, 1.0)
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-12:
            if diff2 > 0:
                mask = h < 1.0 - 1e-9
            else:
                mask = h > 1e-9
            cnt = mask.sum()
            if cnt > 0:
                h[mask] += diff2 / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def binary_snap(h, n, rng):
    h_b = (h > 0.5).astype(float)

    cur_ones = h_b.sum()

        idx = np.where(h_b == 1)[0]
        flip = rng.choice(idx, int(cur_ones - n_ones), replace=False)
        h_b[flip] = 0.0

        idx = np.where(h_b == 0)[0]
        flip = rng.choice(idx, int(n_ones - cur_ones), replace=False)
        h_b[flip] = 1.0
    return h_b

    mc = corr.max()
    shifted = corr - mc
    ex = np.exp(temp * shifted)
    ex_sum = ex.sum()
    w = ex / ex_sum

def smooth_max_grad(h, w):
    n = len(h)

    part1 = np.zeros(n)
    for j in range(n):
        lo = max(0, n - 1 - j)
        hi = min(2 * n - 2, n - 1 + j)

    conv_hwf = convolve(h, w[::-1], 'full', method='auto')
    C = conv_hwf[n - 1:2 * n - 1]
    return part1 - A - C

def slsqp_solve(h, n, temp, max_iter):
    dx = 2.0 / n

    def obj(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        val, _ = soft_max_val_weights(corr, temp)
        return val * dx

    def grad(hv):
        corr = np.correlate(hv, 1.0 - hv, mode='full')
        _, w = soft_max_val_weights(corr, temp)
        g = smooth_max_grad(hv, w)
        return g * dx

    one_vec = np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda hv: hv.sum() - n / 2.0, 'jac': lambda hv: one_vec}]
    res = minimize(obj, h, method='SLSQP', jac=grad, bounds=bounds,
                   constraints=constraints, options={'maxiter': max_iter, 'ftol': 1e-14})
    return res.x

    dx = 2.0 / n
    lr = lr_init
    best_c5 = compute_c5(h)

            break

        val, w = soft_max_val_weights(corr, temp)

        h_trial = h - lr * g
        h_trial = np.clip(h_trial, 0.0, 1.0)

        if c_trial < compute_c5(h) - 1e-12:
