```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-14:
            if diff > 0:
                idx = np.argsort(h)[-int(abs(diff))+1:]
            else:
                idx = np.argsort(h)[:int(abs(-diff))+1:]
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=None)
    return make_feasible(h_new, new_n)

def corr_gradient(h, L):
    """Gradient of corr(L) w.r.t. h. L is the lag (can be negative)."""
    n = len(h)
    grad = np.zeros(n)
    if L >= 0:
        # For j in [L, n-1]: grad[j] += (1 - h[j-L])
        # For j in [0, n-1-L]: grad[j] -= h[j+L]
        if n - L > 0:
            grad[L:] += (1.0 - h[:n-L])
            grad[:n-L] -= h[L:]
    else:
        Lp = -L
        if n - Lp > 0:
            grad[:n-Lp] += (1.0 - h[Lp:])
            grad[Lp:] -= h[:n-Lp]
    return grad

def subgradient_step(h, lr, top_k=3, softmax_temp=50.0):
    """One subgradient step with weighted active constraint gradients."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_corr = np.max(corr)
    
    # Weight active constraints
    shifted = corr - max_corr
    weights = np.exp(softmax_temp * shifted / max(1e-10, abs(max_corr)))
    weights /= np.sum(weights)
    
    # Compute weighted gradient
    center = n - 1
    grad = np.zeros(n)
    # Only use top-k for efficiency
    top_idx = np.argsort(corr)[-top_k:]
    top_weights = weights[top_idx]
    top_weights /= np.sum(top_weights)
    for i, m in enumerate(top_idx):
        L = m - center
        grad += top_weights[i] * corr_gradient(h, L)
    
    h_new = h - lr * grad
    h_new = make_feasible(h_new, n)
    return h_new, max_corr

def subgradient_optimize(h, max_time=100, lr_init=0.01, top_k=5):
    """Projected subgradient optimization."""
    n = len(h)
    start = time.time()
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    lr = lr_init
    patience = 0
    max_patience = 200
    
    while time.time() - start < max_time:
        h_new, _ = subgradient_step(h_cur, lr, top_k=top_k)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_cur = h_new
            patience = 0
        else:
            h_cur = h_new  # still move, for escape
            patience += 1
            lr *= 0.95
        
        if patience > max_patience:
            lr = lr_init * 0.3
            patience = 0
    
    return best_h, best_c5

        improved = False
        for j in range(n):

                return h_cur, best_c5
            
            # Try different values for h[j]
            old = h_cur[j]
            candidates = [0.0, 1.0, 0.5, old-0.1, old-0.05, old+0.05, old+0.1]

            best_c = best_c5

                delta = cv - old
                trial[j] = cv

                if abs(delta) > 1e-12:

                    trial[~(np.arange(n) == j)] -= delta / (n - 1)

                best_c5 = best_c
                improved = True

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(3):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = float(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def lag_gradient(h, L):
    n = len(h)
    grad = np.zeros(n)
    if L >= 0:
        if n - L > 0:
            grad[L:] += (1.0 - h[:n-L])
            grad[:n-L] -= h[L:]
    else:
        Lp = -L
        if n - Lp > 0:
            grad[:n-Lp] += (1.0 - h[Lp:])
            grad[Lp:] -= h[:n-Lp]
    return grad

def soft_max_objective(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    val = m + np.log(np.sum(np.exp(T * (corr - m)))) / T
    return val * dx

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = T * (corr - m)
    weights = np.exp(shifted - np.max(shifted))
    weights /= np.sum(weights)
    center = n - 1
    # Compute weighted gradient
    grad = np.zeros(n)
    # Use top indices for efficiency
    top_k = min(20, len(corr))
    top_idx = np.argsort(corr)[-top_k:]
    top_w = weights[top_idx]
    top_w /= np.sum(top_w)
    for i, m_idx in enumerate(top_idx):
        L = m_idx - center
        grad += top_w[i] * lag_gradient(h, L)
    return grad * dx

def subgrad_optimize(h, max_time, lr_init=0.005):
    n = len(h)
    start = time.time()
    h_cur = h.copy()
    best_h = h_cur.copy()
    best_c5 = compute_c5(h_cur)
    lr = lr_init
    no_improve = 0
    
    while time.time() - start < max_time:
        _, g = soft_max_grad(h_cur, n, T=100.0)
        h_new = h_cur - lr * g
        h_new = make_feasible(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        h_cur = h_new
        if no_improve > 100:
            lr *= 0.7
            no_improve = 0
        elif no_improve % 50 == 0:
            lr *= 0.98
        
        if lr < 1e-8:
            break
    
    return best_h, best_c5

def slsqp_optimize(h, max_time, T=300.0, maxiter=200):
    n = len(h)
    start = time.time()
    
    def obj(x):
        return soft_max_objective(x, n, T)
    
    def obj_grad(x):
        return soft_max_grad(x, n, T)
    
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    try:
        res = minimize(obj, h, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        c5 = compute_c5(res.x)
        if c5 < best_c5:
            best_h = res.x.copy()
            best_c5 = c5
    except Exception:
        pass
    
    # Second pass with higher T if time allows
    if time.time() - start < max_time - 20:
        T2 = 800.0
        def obj2(x):
            return soft_max_objective(x, n, T2)
        def obj2_grad(x):
            return soft_max_grad(x, n, T2)
        try:
            res2 = minimize(obj2, best_h, jac=obj2_grad, method='SLSQP',
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 100, 'ftol': 1e-12})
            c5_2 = compute_c5(res2.x)
            if c5_2 < best_c5:
                best_h = res2.x.copy()
                best_c5 = c5_2
        except Exception:
            pass
    
    return best_h, best_c5

def coord_descent(h, max_time=40):
    n = len(h)
    start = time.time()
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    
    sweeps = 0
    while time.time() - start < max_time - 5:
        sweeps += 1
        improved = False
        for j in range(n):
            if time.time() - start > max_time - 3:
                return h_cur, best_c5
            old = h_cur[j]
            if old < 1e-10 or old > 1.0 - 1e-10:
                # Try moving toward 0.5
                candidates = [min(1.0, old + 0.1), max(0.0, old - 0.1), 0.5]
            else:
                candidates = [0.0, 1.0, old - 0.05, old + 0.05]
            
            best_val = old
            best_c = best_c5
            for cv in candidates:
                cv = float(np.clip(cv, 0.0, 1.0))
                if abs(cv - old) < 1e-10:
                    continue
                trial = h_cur.copy()
                delta = cv - old
                trial[j] = cv
                others = slice(None)  # all indices except j
                trial[~(np.arange(n) == j)] -= delta / (n - 1)
                trial = np.clip(trial, 0.0, 1.0)
                c = compute_c5(trial)
                if c < best_c - 1e-12:
                    best_c = c
                    best_val = cv
            
            if best_val != old:
                h_cur[j] = best_val
                delta = best_val - old
                mask = np.ones(n, dtype=bool)
                mask[j] = False
                h_cur[mask] -= delta / (n - 1)
                h_cur = np.clip(h_cur, 0.0, 1.0)
                h_cur = make_feasible(h_cur, n)
                best_c5 = best_c
                improved = True
        
        if not improved:
            break
    
    return h_cur, best_c5

def random_binary_h(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        to_flip = rng.choice(ones, len(ones) - target, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < target:
        to_flip = rng.choice(zeros, target - len(ones), replace=False)
        h[to_flip] = 1.0
    return h
