    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

def to_binary(h, n):

    h = h.astype(float)

import numpy as np
from scipy.optimize import minimize
from math import comb
from itertools import combinations
import time

_G_N = 960
_G_T = 200.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    target = n // 2
    idx = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[idx[:target]] = 1.0
    return b

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def periodic_search(n, max_p=24, sample_cap=200000):
    best_c = float('inf')
    best_b = None
    for p in range(4, max_p + 2):
        if n % p != 0 or p % 2 != 0:
            continue
        half = p // 2
        total = comb(p, half)
        if total > sample_cap:
            rng_local = np.random.RandomState(12345)
            indices = rng_local.choice(total, size=sample_cap, replace=False)
            all_pats = None
        else:
            all_pats = list(combinations(range(p), half))
            indices = range(total)
        
        rng_s = np.random.RandomState(99)
        count = 0
        for idx in indices:
            if all_pats is not None:
                pat = all_pats[idx]
            else:
                pat = tuple(sorted(rng_s.choice(p, half, replace=False).tolist()))
            bp = np.zeros(p)
            for i in pat:
                bp[i] = 1.0
            b = np.tile(bp, n // p)
            c = compute_c5(b)
            if c < best_c:
                best_c = c
                best_b = b.copy()
            count += 1
    return best_b, best_c

def _slsqp_obj(x):
    n = _G_N
    dx = 2.0 / n
    T = _G_T
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_eq(x):
    return np.sum(x) - _G_N / 2.0

def _con_eq_jac(x):
    return np.ones(_G_N)

def binary_sa(b_init, n, max_time, seed):
    rng_sa = np.random.RandomState(seed)
    b_curr = b_init.copy()
    n_ones = int(np.sum(b_curr))
    target = n // 2
    if n_ones > target:

        rng_sa.shuffle(ones_idx)
        for i in range(n_ones - target):

    elif n_ones < target:
        zeros_idx = np.where(b_curr == 0.0)[0]
        rng_sa.shuffle(zeros_idx)
        for i in range(target - n_ones):
            b_curr[zeros_idx[i]] = 1.0
    
    c_curr = compute_c5(b_curr)
    c_best = c_curr
    b_best = b_curr.copy()
    
    ones_list = list(np.where(b_curr == 1.0)[0])
    zeros_list = list(np.where(b_curr == 0.0)[0])

        b_curr[ii] = 0.0
        b_curr[jj] = 1.0
        c_try = compute_c5(b_curr)
        delta = c_try - c_curr

            c_curr = c_try

            zeros_list[j_p] = ii
            if c_try < c_best:
                c_best = c_try
                b_best = b_curr.copy()
        else:
            b_curr[ii] = 1.0
            b_curr[jj] = 0.0
        T *= cool
    
    return b_best, c_best
