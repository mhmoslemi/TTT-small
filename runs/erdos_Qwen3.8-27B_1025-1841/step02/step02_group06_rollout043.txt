        else:
            hi = mid

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def logsumexp_smooth_c5(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    return (1.0/tau) * np.log(np.sum(np.exp(tau * (vals - m)))) + m

def logsumexp_smooth_grad(h, n, tau):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = vals.max()
    shifted = np.exp(tau * (vals - m))
    p = shifted / shifted.sum()
    grad = np.zeros(n)
    for k in range(len(p)):
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx

    y = h.copy()

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_topk(h, n, dx, corr_vals, tau, topM):
    """Compute gradient using topM components of softmax-weighted correlation."""
    m = corr_vals.max()
    shifted = np.exp(np.clip(tau * (corr_vals - m), -500, 500))
    p = shifted / shifted.sum()
    
    # Get top M indices by p value (which correlates with corr_vals)
    if topM > 0 and topM < len(p):
        top_idx = np.argpartition(p, -topM)[-topM:]
    else:
        top_idx = np.where(p > 1e-8)[0]
    
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx

def nesterov_smooth(h, n, time_limit, tau_start=10.0, tau_end=100.0, lr=0.3, topM=300):
    """Nesterov-accelerated projected gradient on log-sum-exp smoothed objective."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    y = h.copy()
    v = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    stall = 0
    patience = 5000
    
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / time_limit)
        tau = tau_start * (tau_end / tau_start) ** progress
        cur_lr = lr / (1.0 + 0.003 * it)
        
        # Nesterov extrapolation
        y = h + beta * (h - y)
        y = np.clip(y, 0.0, 1.0)
        
        # Gradient at y
        corr_y = compute_corr(y)
        vals = corr_y * dx
        grad = gradient_topk(y, n, dx, vals, tau, topM)
        
        # Update
        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)
        y = h_new
        h = h_new
        
        # Check improvement (every 100 iterations to save time)
        if it % 100 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > patience // 100:
                v *= 0.5
                cur_lr *= 0.5
                stall = 0
        
        it += 1
    
    # Final check
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def make_random_step(n, rng):
    """Create a random binary step function with n/2 ones."""
    positions = rng.choice(n, size=n//2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h

def make_vander_corput_step(n):
    """Create a step function based on van der Corput sequence."""
    half = n // 2
    # Van der Corput sequence in [0, 1)
    def vdc(k):
        s = 0
        d = 0.5
        k = k + 1
        while k > 0:
            s += d * (k % 2)
            k //= 2
            d /= 2
        return s
    # Place ones at the first 'half' van der Corput positions
    vp = np.array([vdc(i) for i in range(n)])
    order = np.argsort(vp)
    h = np.zeros(n)
    h[order[:half]] = 1.0
    return h

def binarize_and_reoptimize(h, n, time_limit):
    """Binarize h, then re-optimize continuously from the binary solution."""
    h_bin = (h > 0.5).astype(float)
    # Ensure exactly n/2 ones
    n_ones = int(h_bin.sum())
    if n_ones > n // 2:
        # Remove excess ones (smallest h values among ones)
        one_idx = np.where(h_bin == 1.0)[0]
        remove = one_idx[np.argsort(h[one_idx])[:n_ones - n//2]]
        h_bin[remove] = 0.0
    elif n_ones < n // 2:
        zero_idx = np.where(h_bin == 0.0)[0]
        add = zero_idx[np.argsort(-h[zero_idx])[:n//2 - n_ones]]
        h_bin[add] = 1.0
    # Now re-optimize from this binary start
    h_opt, c5 = nesterov_smooth(h_bin, n, time_limit, tau_start=50, tau_end=200, lr=0.2, topM=200)
    return h_opt, c5

Wait, I realize there might be an issue with the Nesterov implementation. The standard form is:

```
y = x + (beta_t - 1) * (x - x_prev)   # where beta_t = (1+sqrt(1+4*beta_t²))/2
grad = ∇f(y)
x_new = x - α * grad
```

Or the simpler "heavy ball with lookahead":
```
y = x + β * (x - y_prev)   # extrapolate
grad = ∇f(y)
v = β * v - α * grad       # momentum
x_new = x + v              # update (or project)
```

The second form is what I'll use. The key is that the gradient is evaluated at y (the extrapolated point), not at x.

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * dx


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def gradient_topk(h, n, dx, corr_vals, tau, topM):
    m = corr_vals.max()
    shifted = np.exp(np.clip(tau * (corr_vals - m), -500, 500))
    p = shifted / shifted.sum()
    if topM > 0 and topM < len(p):
        top_idx = np.argpartition(p, -topM)[-topM:]
    else:
        top_idx = np.where(p > 1e-8)[0]
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx


def nesterov_smooth(h, n, time_limit, tau_start=10.0, tau_end=100.0, lr=0.25, topM=300, check_interval=50):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    y = h.copy()
    v = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    it = 0
    stall = 0
    patience = 3000
    lr_schedule_rate = 0.002

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / max(time_limit, 1e-6))
        tau = tau_start * (tau_end / tau_start) ** progress
        cur_lr = lr / (1.0 + lr_schedule_rate * it)

        y_new = h + beta * (h - y)
        y_new = np.clip(y_new, 0.0, 1.0)

        corr_y = compute_corr(y_new)
        vals = corr_y * dx
        grad = gradient_topk(y_new, n, dx, vals, tau, topM)

        v = beta * v - cur_lr * grad
        h_new = project_to_feasible(h + v, n)
        y = h
        h = h_new

        if it % check_interval == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
                stall = 0
            else:
                stall += check_interval
            if stall > patience:
                v *= 0.3
                it = 0
                stall = 0
                if cur_lr < 1e-6:
                    break

        it += 1

    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def make_random_step(n, rng):
    positions = rng.choice(n, size=n // 2, replace=False)
    h = np.zeros(n)
    h[positions] = 1.0
    return h


def make_vander_corput_step(n):
    half = n // 2
    vp = np.zeros(n)
    for i in range(n):
        s = 0.0
        d = 0.5
        k = i + 1
        while k > 0:
            s += d * (k % 2)
            k //= 2
            d /= 2.0
        vp[i] = s
    order = np.argsort(vp)
    h = np.zeros(n)
    h[order[:half]] = 1.0
    return h


def binarize(h):
    n = len(h)
    h_bin = (h > 0.5).astype(float)
    n_ones = int(h_bin.sum())
    target = n // 2
    if n_ones > target:
        one_idx = np.where(h_bin == 1.0)[0]
        remove = one_idx[np.argsort(h[one_idx])[:n_ones - target]]
        h_bin[remove] = 0.0
    elif n_ones < target:
        zero_idx = np.where(h_bin == 0.0)[0]
        add = zero_idx[np.argsort(-h[zero_idx])[:target - n_ones]]
        h_bin[add] = 1.0
    return h_bin
