            delta_hi = delta_mid

import numpy as np
from scipy.optimize import minimize
import time

_N_G = 1
_T_G = 50.0

def _obj(h):
    n = _N_G
    t = _T_G
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(t * shifted))) / t
    return soft * dx

def _con_fun(h):
    return np.sum(h) - _N_G / 2.0

def _con_jac(h):
    return np.ones(_N_G)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-14:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def slsqp_run(h_init, n, maxiter, t, ftol=1e-12):
    global _N_G, _T_G
    _N_G = n
    _T_G = t
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h_init, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        out = res.x
        return make_feasible(out, n)
    except Exception:
        return h_init.copy()

def two_coord_descent(h, n, n_iters, rng, time_limit):
    dx_val = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = float(np.max(corr))
    lags = np.arange(-(n - 1), n)
    accepted = 0
    recompute_every = 100
    for it in range(n_iters):
        if time.time() > time_limit:
            break
        p = rng.randint(n)
        q = rng.randint(n)
        if p == q:
            continue
        idx1 = q - lags
        idx2 = p - lags
        idx3 = p + lags
        idx4 = q + lags
        B = (np.where((idx1 >= 0) & (idx1 < n), h[np.clip(idx1, 0, n - 1)], 0.0) -
             np.where((idx2 >= 0) & (idx2 < n), h[np.clip(idx2, 0, n - 1)], 0.0) -
             np.where((idx3 >= 0) & (idx3 < n), h[np.clip(idx3, 0, n - 1)], 0.0) +
             np.where((idx4 >= 0) & (idx4 < n), h[np.clip(idx4, 0, n - 1)], 0.0))
        dlo, dhi = -h[p], h[q]
        if dlo >= dhi - 1e-15:
            continue
        for _ in range(30):
            dmid = (dlo + dhi) / 2.0
            vals = corr + dmid * B
            mx = np.max(vals)
            act_mask = vals >= mx - 1e-13
            if not np.any(act_mask):
                break
            act_B = B[act_mask]
            if np.all(act_B > 1e-11):
                dhi = dmid
            elif np.all(act_B < -1e-11):
                dlo = dmid
            else:
                break
        dopt = (dlo + dhi) / 2.0
        if abs(dopt) < 1e-14:
            continue
        new_max = float(np.max(corr + dopt * B))
        if new_max < current_max - 1e-13:
            h[p] = np.clip(h[p] + dopt, 0.0, 1.0)
            h[q] = np.clip(h[q] - dopt, 0.0, 1.0)
            corr = corr + dopt * B
            current_max = new_max
            accepted += 1
            if accepted % recompute_every == 0:
                diff = n / 2.0 - np.sum(h)
                if abs(diff) > 1e-12:
                    h += diff / n
                    h = np.clip(h, 0.0, 1.0)
                corr = np.correlate(h, 1.0 - h, mode='full')
                current_max = float(np.max(corr))
    h = make_feasible(h, n)
    return h, compute_c5(h)

def projected_subgradient(h, n, n_iters, rng, time_limit):
    dx_val = 2.0 / n
    current_c5 = compute_c5(h)
    step0 = 0.05
    for it in range(n_iters):
        if time.time() > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        m_star = np.argmax(corr)
        lag = m_star - (n - 1)
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            n1 = j - lag
            if 0 <= n1 < n:
                g += (1.0 - h[n1])
            n2 = j + lag
            if 0 <= n2 < n:
                g -= h[n2]
            grad[j] = g
        alpha = step0 / (1.0 + 0.01 * it)
        h_trial = h - alpha * grad
        h_trial = make_feasible(h_trial, n)
        c_trial = compute_c5(h_trial)
        if c_trial < current_c5 - 1e-14:
            h = h_trial
            current_c5 = c_trial
        else:
            step0 *= 0.9
    return h, current_c5

    best_c5 = compute_c5(h0)

    # Phase 1: SLSQP at n=640
    n1 = 640
    h640 = upsample_h(best_h, n1)
    c640 = compute_c5(h640)
    if c640 < best_c5:
        best_c5 = c640
        best_h = h640.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        res_x = slsqp_run(h640, n1, 800, 50)
        c5 = compute_c5(res_x)
        if c5 < best_c5:

    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        h_pert = best_h + rng.randn(n1) * 0.02
        h_pert = make_feasible(h_pert, n1)
        res_x = slsqp_run(h_pert, n1, 600, 80)
        c5 = compute_c5(res_x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = res_x.copy()

    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        res_x = slsqp_run(best_h, n1, 600, 120)
        c5 = compute_c5(res_x)
        if c5 < best_c5:
            best_c5 = c5
            best_h = res_x.copy()

    # Phase 2: 2-coordinate descent
    elapsed = time.time() - start_time
    n_iters_2c = 80000
    if budget_s - elapsed > 60:

    # Phase 4: More 2-coordinate descent
    elapsed = time.time() - start_time

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    n_final = len(best_h)

    return best_h, final_c5, n_final
