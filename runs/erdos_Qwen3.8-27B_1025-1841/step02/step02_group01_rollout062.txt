        h1024 = coordinate_descent(h1024, n_passes=10, time_budget=max_t4, t0=start_time, track_best=True)
        c5_1024 = compute_c5(h1024)
        if c5_1024 < best_c5:
            best_c5 = c5_1024
            best_h = h1024.copy()

    # Phase 5: Multiple perturbation restarts
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 20
    if remaining > 40:
        n_restarts = min(6, int(remaining / 25))
        for r in range(n_restarts):
            elapsed = time.time() - start_time
            if budget_s - elapsed < 25:
                break
            n = len(best_h)
            noise_scale = 0.02 + 0.01 * r
            h_start = best_h + rng.randn(n) * noise_scale
            h_start = make_feasible(h_start, n)
            max_tr = min(20, (budget_s - elapsed - 5) * 0.6)
            h_try = coordinate_descent(h_start, n_passes=10, time_budget=max_tr, t0=start_time, track_best=True)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()

            # Also try a random binary start at current resolution
            if budget_s - elapsed > 30 and n <= 512:
                h_bin = random_binary_h(n, rng)
                max_tr2 = min(15, (budget_s - elapsed - 5) * 0.5)
                h_try2 = coordinate_descent(h_bin, n_passes=8, time_budget=max_tr2, t0=start_time, track_best=True)
                c5_try2 = compute_c5(h_try2)
                if c5_try2 < best_c5:
                    best_c5 = c5_try2
                    best_h = h_try2.copy()

    # Phase 6: Final intensive polish at best resolution
    elapsed = time.time() - start_time
    if budget_s - elapsed > 15:
        final_t = (budget_s - elapsed - 3) * 0.8
        best_h = coordinate_descent(best_h, n_passes=30, time_budget=final_t, t0=start_time, track_best=True)
        best_c5 = compute_c5(best_h)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
