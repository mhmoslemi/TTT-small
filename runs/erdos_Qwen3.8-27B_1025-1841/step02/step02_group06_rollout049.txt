        h8, c8 = nesterov_subgrad(best_h, nf, remaining() * 0.3, topK_start=1, topK_end=2, lr_init=0.004)
        if c8 < best_c5:
            best_c5, best_h = c8, h8.copy()

    # Phase 9: Final local polish
    if remaining() > 15:
        nf = len(best_h)
        h9, c9 = local_polish(best_h, nf, remaining() * 0.35, max_rounds=5000)
        if c9 < best_c5:
            best_c5, best_h = c9, h9.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
