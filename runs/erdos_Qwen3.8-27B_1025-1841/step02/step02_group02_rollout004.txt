```python
def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in np.where(p > 1e-10)[0]:
        sh = int(k) - n + 1
        pk = p[k]
        # First term: j in [max(0,-sh), min(n, n-sh))
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            grad[j0a:j1a] += pk * (1.0 - h[j0a + sh : j1a + sh])
        # Second term: j in [max(0, sh), min(n, sh+n))
        j0b = max(0, sh)
        j1b = min(n, sh + n)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh : j1b - sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad
```

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    sig = np.where(p > 1e-10)[0]
    for k in sig:
        sh = int(k) - n + 1
        pk = p[k]
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            grad[j0a:j1a] += pk * (1.0 - h[j0a + sh:j1a + sh])
        j0b = max(0, sh)
        j1b = min(n, sh + n)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def topk_grad(h, topk):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    if topk >= len(corr):
        top_idx = np.arange(len(corr))
    else:
        top_idx = np.argpartition(corr, -topk)[-topk:]
    grad = np.zeros(n)
    for k in top_idx:
        sh = int(k) - n + 1
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            grad[j0a:j1a] += (1.0 - h[j0a + sh:j1a + sh])
        j0b = max(0, sh)
        j1b = min(n, sh + n)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b - sh:j1b - sh]
    grad *= dx / float(topk)
    return grad, corr.max() * dx


def optimize_smooth(h, n, tau, lr, max_iters, time_limit, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    best_stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_c5 - 1e-15:
            best_c5 = obj
            best_h = h.copy()
            stall = 0
            best_stall = 0
        else:
            stall += 1
            best_stall += 1

        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)

        if stall > 30:
            lr *= 0.5
            v *= 0.4
            stall = 0
            if lr < 1e-11:
                break

    return best_h, best_c5


def optimize_topk(h, n, topk, lr, max_iters, time_limit, momentum=0.8):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    avg_h = h.copy() * (1.0 / max(max_iters, 1))

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        grad, c5_cur = topk_grad(h, topk)
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
        avg_h += (n / (n + it + 1)) * (h - avg_h)

        if stall > 80:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-13:
                break

    c5_avg = compute_c5(avg_h)
    if c5_avg < best_c5:
        return avg_h, c5_avg
    return best_h, best_c5
```
