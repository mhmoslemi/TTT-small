```python
import numpy as np
import time

def project_to_feasible(g, n):
    """Project g onto {h: 0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def fft_correlation(h):
    """Compute full correlation of h with (1-h) using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.fft(h, N)
    # autocorrelation of h
    ac = np.real(np.fft.ifft(H * np.conj(H)))
    # full_corr[m] = sum(h) - autocorr(d) where d = m - (n-1)
    sh = np.sum(h)
    full_corr = np.empty(2 * n - 1)
    # d = 0: m = n-1
    full_corr[n-1] = sh - ac[0]
    # d > 0: m = n-1+d
    for d in range(1, n):

    # d < 0: m = n-1+d, autocorr(-d) = ac[N-d]
    for d in range(1, n):
        full_corr[n-1-d] = sh - ac[N-d]
    return full_corr

    n = len(h)
    corr = fft_correlation(h)
    return corr.max() * (2.0 / n)

    for k in range(n_lags):
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - (n - 1)
        # First term: (1-h[a-d]) for valid a
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        # Second term: -h[a+d] for valid a
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx
```

```python
    grad = np.zeros(n)
```
