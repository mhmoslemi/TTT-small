    def soft_obj(h):
        dx = 2.0 / len(h)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)

        ex = np.exp(t * shifted)
        se = np.sum(ex)

        return (mc + np.log(se) / t) * dx

    return compute_c5(h)

        return np.ones(n)
