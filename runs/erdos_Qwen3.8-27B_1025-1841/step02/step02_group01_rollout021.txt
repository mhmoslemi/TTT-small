```python
def softmax_gradient(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    shifted = corr - mc
    weights = np.exp(t * shifted)
    wsum = np.sum(weights)
    weights /= wsum
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = weights[k]
        if wk < 1e-15:
            continue
        s = k - (n - 1)  # shift
        
        # Forward: d/dh[j] of h[j]*(1-h[j+s]) = (1-h[j+s]) for j where 0<=j<n and 0<=j+s<n
        j_min = max(0, -s)
        j_max = min(n, n - s)
        if j_max > j_min:
            idx = np.arange(j_min, j_max)
            grad[idx] += wk * (1.0 - h[idx + s])
        
        # Backward: d/dh[j] of h[j-s]*(1-h[j]) = -h[j-s] for j where 0<=j<n and 0<=j-s<n
        j_min2 = max(0, s)
        j_max2 = min(n, n + s)
        if j_max2 > j_min2:
            idx = np.arange(j_min2, j_max2)
            grad[idx] += wk * (-h[idx - s])
    
    return dx * grad
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    idx = np.arange(n)
    ones = np.sort(rng.choice(n, n // 2, replace=False))
    h = np.zeros(n)
    h[ones] = 1.0
    return h

def softmax_obj_and_grad(h, t):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    mc = np.max(corr)
    shifted = corr - mc
    exps = np.exp(t * shifted)
    wsum = np.sum(exps)
    weights = exps / wsum
    soft_val = (mc + np.log(wsum) / t) * dx
    grad = np.zeros(n)
    active_mask = weights > 1e-12
    for k in np.where(active_mask)[0]:
        wk = weights[k]
        if wk < 1e-14:
            continue
        s = k - (n - 1)
        j_min = max(0, -s)
        j_max = min(n, n - s)
        if j_max > j_min:
            idx = np.arange(j_min, j_max)
            grad[idx] += wk * (1.0 - h[idx + s])
        j_min2 = max(0, s)
        j_max2 = min(n, n + s)
        if j_max2 > j_min2:
            idx = np.arange(j_min2, j_max2)
            grad[idx] += wk * (-h[idx - s])
    return soft_val, dx * grad

def slsqp_optimize(h, n, t, maxiter, start_time, budget_s):
    global _G_N, _G_T
    _G_N = n
    _G_T = t
    bounds = [(0.0, 1.0)] * n
    con = {'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
           'jac': lambda x: np.ones(n)}
    res = minimize(_slsqp_obj, h, jac=_slsqp_grad, method='SLSQP',
                   bounds=bounds, constraints=[con],
                   options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

def _slsqp_obj(h):
    v, _ = softmax_obj_and_grad(h, _G_T)
    return v

def _slsqp_grad(h):
    _, g = softmax_obj_and_grad(h, _G_T)
    return g

def subgradient_step(h, lr, momentum_vec, mu):
    n = len(h)
    corr = compute_corr(h)
    mc = np.max(corr)
    # Use top-k shifts with exponential weighting
    sorted_idx = np.argsort(corr)[::-1]
    top_k = sorted_idx[:8]
    exps = np.exp(50.0 * (corr[top_k] - mc))
    weights = exps / np.sum(exps)
    
    grad = np.zeros(n)
    for kk in range(len(top_k)):
        k = top_k[kk]
        wk = weights[kk]
        if wk < 1e-10:
            continue
        s = k - (n - 1)
        j_min = max(0, -s)
        j_max = min(n, n - s)
        if j_max > j_min:
            idx = np.arange(j_min, j_max)
            grad[idx] += wk * (1.0 - h[idx + s])
        j_min2 = max(0, s)
        j_max2 = min(n, n + s)
        if j_max2 > j_min2:
            idx = np.arange(j_min2, j_max2)
            grad[idx] += wk * (-h[idx - s])
    
    momentum_vec = mu * momentum_vec + (1 - mu) * grad
    h_new = h - lr * momentum_vec
    h_new = make_feasible(h_new, n)
    return h_new, momentum_vec

def coord_descent_polish(h, n, start_time, budget_s, max_passes=20):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    for p in range(max_passes):
        if time.time() - start_time > budget_s - 15:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 15:
                break
            old_val = h[j]
            candidates = [0.0, 1.0, old_val + 0.02, old_val - 0.02,
                          old_val + 0.05, old_val - 0.05,
                          old_val + 0.01, old_val - 0.01, 0.5]
            best_candidate = old_val
            best_trial_c5 = best_c5
            for cv in candidates:
                if cv < 0 or cv > 1:
                    continue
                trial = h.copy()
                trial[j] = cv
                # Adjust to maintain sum constraint
                diff = n / 2.0 - np.sum(trial)
                if abs(diff) < 1e-15:
                    pass
                else:
                    # Spread adjustment to nearest non-bound coordinates
                    others = np.where(np.arange(n) != j)[0]
                    non_bound = np.where((trial[others] > 0.01) | (trial[others] < 0.99))[0]
                    if len(non_bound) == 0:
                        non_bound = others
                    else:
                        non_bound = others[non_bound]
                    if len(non_bound) > 0:
                        trial[non_bound] += diff / len(non_bound)
                        trial = np.clip(trial, 0.0, 1.0)
                        # Final fix
                        diff2 = n / 2.0 - np.sum(trial)
                        if abs(diff2) > 1e-12:
                            mask = trial < 0.99 if diff2 > 0 else trial > 0.01
                            if np.sum(mask) > 0:
                                trial[mask] += diff2 / np.sum(mask)
                                trial = np.clip(trial, 0.0, 1.0)
                c = compute_c5(trial)
                if c < best_trial_c5 - 1e-12:
                    best_trial_c5 = c
                    best_candidate = cv
            if best_candidate != old_val:
                h[j] = best_candidate
                diff = n / 2.0 - np.sum(h)
                if abs(diff) > 1e-12:
                    others = np.where(np.arange(n) != j)[0]
                    non_bound = np.where((h[others] > 0.01) | (h[others] < 0.99))[0]
                    if len(non_bound) > 0:
                        h[others[non_bound]] += diff / len(non_bound)
                        h = np.clip(h, 0.0, 1.0)
                        h = make_feasible(h, n)
                    else:
                        h = make_feasible(h, n)
                best_c5 = best_trial_c5
                improved = True
        if not improved:
            break
    return h, best_c5

def binary_local_search(h, n, rng, start_time, budget_s, max_iters=5000):
    """Local search on binary solutions using pair swaps."""
    h = np.round(h)
    h = make_feasible(h, n)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    best_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 15:
            break
        # Try a random pair swap
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        h[i] = 0.0
        h[j] = 1.0
        c = compute_c5(h)
        if c < best_c5 - 1e-12:
            best_c5 = c
            # Update sets
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return h, best_c5
