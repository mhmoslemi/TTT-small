def project_to_feasible(h, n):

def smooth_max_obj(x, n, T):

    corr = np.correlate(x, 1.0 - x, mode='full')
    vals = corr * dx
    scaled = T * vals
    m = np.max(scaled)

def smooth_max_grad(x, n, T):
