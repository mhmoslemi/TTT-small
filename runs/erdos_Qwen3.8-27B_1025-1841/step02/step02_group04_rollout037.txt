    """Project h onto {x: 0<=x_i<=1, sum(x)=target_sum}"""
    n = len(h)
    # Bisection for tau
    lo = h.min() - 1.0

    for _ in range(100):

        s = proj.sum()
        if s > target_sum:

        else:

def compute_gradient_fast(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = exp_s.sum()
    obj = (mc + np.log(total) / T) * dx
    
    weights = exp_s / total
    # P[s] for s=0..n-1: weights[s + n - 1]
    # N[s] for s=1..n-1: weights[n - 1 - s]
    P_pos = weights[(n-1):(2*n-1)]  # length n
    N_pos = weights[n-2:-1:-1] if n > 1 else np.zeros(0)  # length n-1
    
    # M[s] = P[s] + N[s] for s=1..n-1
    M = P_pos[1:] + N_pos  # length n-1
    
    # Kernel k: length n, k[0]=0, k[s]=M[s-1] for s=1..n-1
    k = np.zeros(n)
    k[1:] = M
    
    # FFT convolution
    N_fft = 2 * n  # next power of 2 or just 2n
    H = np.fft.rfft(h, N_fft)
    K = np.fft.rfft(k, N_fft)
    conv_h = np.fft.irfft(H * K, N_fft)[:n]  # (h * k)[j] for j=0..n-1
    
    h_rev = h[::-1]
    H_rev = np.fft.rfft(h_rev, N_fft)
    conv_hrev = np.fft.irfft(H_rev * K, N_fft)[:n]  # (h_rev * k)[j] for j=0..n-1
    
    # A[j] = conv_h[j], B[j] = conv_hrev[n-1-j]
    conv = conv_h + conv_hrev[::-1]
    
    # C[j] = P[0] + sum_{s=1}^{j} P[s] + sum_{s=1}^{n-1-j} N[s]
    P_cum = np.cumsum(P_pos)  # P_cum[j] = sum_{s=0}^{j} P[s]
    N_cum = np.cumsum(N_pos)  # N_cum[j] = sum_{s=0}^{j} N[s] = sum_{s=1}^{j+1} N[s]
    
    C = np.zeros(n)
    for j in range(n):
        C[j] = P_pos[0]
        if j >= 1:
            C[j] += P_cum[j-1]  # sum_{s=1}^{j} P[s]
        if n-1-j >= 1:
            C[j] += N_cum[n-2-j]  # sum_{s=1}^{n-1-j} N[s]
    
    grad = (C - 2.0 * P_pos[0] * h - conv) * dx
    return obj, grad

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def project_convex(h, target_sum):
    n = len(h)
    lo = h.min() - 1.0
    hi = h.max() + 0.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)

    else:
        N_pos = np.zeros(0)

    k = np.zeros(n)
    k[1:] = M
    
    N_fft = 1
    while N_fft < 2 * n:

    H = np.fft.rfft(h, N_fft)
    K = np.fft.rfft(k, N_fft)
    conv_h = np.fft.irfft(H * K, N_fft)[:n]
    
    h_rev = h[::-1]
    H_rev = np.fft.rfft(h_rev, N_fft)
    conv_hrev = np.fft.irfft(H_rev * K, N_fft)[:n]

            C[j] += P_cum[j-1]
        for j in range(n-1):
            C[j] += N_cum[n-2-j]

    """FISTA with projection onto {0<=h<=1, sum(h)=n/2}"""
    n = len(h_init)
    target = n / 2.0
    h = h_init.copy()
    z = h.copy()
    t_k = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_obj = soft_obj_and_grad(h, T)[0]
    
    if lr is None:
        lr = 0.1 / T  # initial guess
    
    for it in range(n_iter):
        # Compute gradient at z
        obj_z, grad_z = soft_obj_and_grad(z, T)
        
        # FISTA update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        z = h + ((t_k - 1.0) / t_new) * (h - z)
        z = project_convex(z, target)
        
        obj_z, grad_z = soft_obj_and_grad(z, T)
        h_new = z - lr * grad_z
        h_new = project_convex(h_new, target)
        
        # Check improvement
        obj_new = soft_obj_and_grad(h_new, T)[0]
        if obj_new > obj_z + 1e-10:
            # Backtrack
            h_new = z
            lr *= 0.5
        
        h = h_new
        t_k = t_new
        
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5
