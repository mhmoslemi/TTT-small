```python
def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def flip_delta(corr, h, i, j, n):
    """Compute change in corr when flipping h[i] 1->0 and h[j] 0->1."""
    # Only compute for top shifts
    # For now, compute all
    delta = np.zeros(2*n-1)
    # h[i] 1->0:
    # For s >= 0, s <= n-1-i: delta at index (n-1-s) -= (1-h[i+s])
    for s in range(n-i):  # s from 0 to n-1-i
        idx = n-1-s
        delta[idx] -= (1.0 - h[i+s])
    # For s < 0, s >= -i: i.e., -i <= s < 0
    for s in range(-i, 0):  # s from -i to -1
        idx = n-1-s  # index in full correlation
        delta[idx] -= (1.0 - h[i+s])
    
    # h[j] 0->1:
    # For s >= 0, s <= j: delta -= h[j-s]
    for s in range(j+1):
        idx = n-1-s
        delta[idx] -= h[j-s]
    # For s < 0, s >= -j: delta += (1-h[j+s])
    for s in range(-j, 0):
        idx = n-1-s
        delta[idx] += (1.0 - h[j+s])
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx), corr

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    h = h + (target - np.sum(h)) / n
    h = np.clip(h, 0.0, 1.0)
    # Fix remaining
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def to_binary(h, n):
    """Convert h to binary with exactly n/2 ones."""
    h_bin = (h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(h_bin == 1)[0]
        # Remove ones at positions where h was closest to 0.5
        order = np.argsort(h[ones_idx])[::-1][:n_ones-target]  # worst ones
        h_bin[ones_idx[order]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h_bin == 0)[0]
        order = np.argsort(h[zeros_idx])[:target-n_ones]  # worst zeros
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def swap_delta_topK(corr, h, i, j, topK_idx, n):
    """Compute change in top-K correlations for swap i(1->0), j(0->1)."""
    deltas = np.zeros(len(topK_idx))
    for k_idx, s in enumerate(topK_idx):
        delta = 0.0
        if s >= 0:
            # h[i] as h[a], a=i: valid if i <= n-1-s
            if i <= n - 1 - s:
                delta -= (1.0 - h[i + s])
            # h[i] as h[a+s], a=i-s: valid if s <= i and i-s <= n-1-s
            if s <= i and i - s <= n - 1 - s:
                delta += h[i - s]  # (1-h[i]) goes 0->1
            # h[j] as h[a], a=j: valid if j <= n-1-s
            if j <= n - 1 - s:
                delta += (1.0 - h[j + s])
            # h[j] as h[a+s], a=j-s: valid if s <= j and j-s <= n-1-s
            if s <= j and j - s <= n - 1 - s:
                delta -= h[j - s]  # (1-h[j]) goes 1->0
        else:
            t = -s
            # h[i] as h[b+t], b=i-t: valid if t<=i and i-t>=0, i-t<=n-1-t
            if t <= i and i + t < n:  # h[i-t] factor
                if t <= i:
                    delta -= (1.0 - h[i - t])
            # h[i] as h[b], b=i: valid if t<=i
            if t <= i and i + t < n:
                delta += h[i + t]
            # h[j] as h[b+t], b=j-t: valid if t<=j
            if t <= j:
                delta += (1.0 - h[j - t])
            # h[j] as h[b], b=j: valid if t<=j
            if t <= j and j + t < n:
                delta -= h[j + t]
        deltas[k_idx] = delta
    return deltas
```

```python
        h_bin[zeros_idx[order]] = 1.0
    return h_bin

def swap_corr_delta(h, i, j, s, n):
    """Delta in corr[s] when h[i]:1->0, h[j]:0->1. Binary h."""
    if s == 0:
        return 0.0
    d = 0.0
    if s > 0:
        if i <= n - 1 - s:
            d -= (1.0 - h[i + s])
        if i >= s:
            d += h[i - s]
        if j <= n - 1 - s:
            d += (1.0 - h[j + s])
        if j >= s:
            d -= h[j - s]
    else:
        t = -s
        if i >= t:
            d -= (1.0 - h[i - t])
        if i + t < n:
            d += h[i + t]
        if j >= t:
            d += (1.0 - h[j - t])
        if j + t < n:
            d -= h[j + t]
    if abs(i - j) == abs(s):
        d -= (1.0 if j > i else -1.0)
    return d

def binary_sa(h, n, n_iter, T_start, T_end, seed, top_k=20):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    cur_max = np.max(corr)
    best_cost = cur_max
    best_h = h.copy()
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    dx = 2.0 / n
    n_eval = 0
    for it in range(n_iter):
        T = T_start * (T_end / T_start) ** (it / max(n_iter - 1, 1))
        # Get top shifts
        top_idx = np.argsort(corr)[-top_k:][::-1]
        # Try guided swaps: fix worst shift
        s_star = top_idx[0] - (n - 1)
        # Find critical pairs for s_star
        if s_star > 0:
            crit_i = [i for i in ones if i + s_star < n and h[i + s_star] == 0]
        elif s_star < 0:
            crit_i = [i for i in ones if i + s_star >= 0 and h[i + s_star] == 0]
        else:
            crit_i = list(ones)
        # Pick a candidate swap
        if len(crit_i) > 0 and rng.rand() < 0.6:
            i = crit_i[rng.randint(len(crit_i))]
            j = zeros[rng.randint(len(zeros))]
        else:
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        # Evaluate new max
        h[i], h[j] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_max = np.max(new_corr)
        n_eval += 1
        delta = (new_max - cur_max) * dx
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            cur_max = new_max
            corr = new_corr
            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if cur_max < best_cost:
                best_cost = cur_max
                best_h = h.copy()
        else:
            h[i], h[j] = 1.0, 0.0
    return best_h, best_cost * dx

def greedy_binary(h, n, max_rounds, seed):
    rng = np.random.RandomState(seed)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_cost = np.max(corr)
    best_h = h.copy()
    dx = 2.0 / n
    for rnd in range(max_rounds):
        top_idx = np.argsort(corr)[-10:][::-1]
        s_star = top_idx[0] - (n - 1)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        best_delta = 0
        best_pair = None
        # Find swap that most reduces worst shift
        if s_star >= 0:
            cand_i = [i for i in ones if i + s_star < n and h[i + s_star] == 0]
        else:
            cand_i = [i for i in ones if i + s_star >= 0 and h[i + s_star] == 0]
        if not cand_i:
            break
        # Sample up to 200 candidate swaps
        n_cand = min(len(cand_i), 200)
        sampled_i = rng.choice(cand_i, n_cand, replace=False) if len(cand_i) > n_cand else np.array(cand_i)
        for i in sampled_i:
            # Best j to pair with i for this shift
            if s_star >= 0:
                good_j = [j for j in zeros if j - s_star >= 0 and j - s_star < n]
            else:
                good_j = [j for j in zeros if j + s_star >= 0 and j + s_star < n]
            if not good_j:
                continue
            j_pool = rng.choice(good_j, min(len(good_j), 20), replace=False)
            for j in j_pool:
                d = swap_corr_delta(h, int(i), int(j), s_star, n)
                if d < best_delta:
                    best_delta = d
                    best_pair = (int(i), int(j))
        if best_pair is None:
            break
        i, j = best_pair
        h[i], h[j] = 0.0, 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        cur_max = np.max(corr)
        if cur_max < best_cost:
            best_cost = cur_max
            best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = 1.0, 0.0
            corr = np.correlate(h, 1.0 - h, mode='full')
    return best_h, best_cost * dx

def _obj_func(h):
    T = _OPT_PARAMS['T']
    n = _OPT_PARAMS['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _cons_fun(h):
    return np.sum(h) - _OPT_PARAMS['n'] / 2.0

def _cons_jac(h):
    return np.ones(_OPT_PARAMS['n'])

_OPT_PARAMS = {}

def slsqp_refine(h, n, T, maxiter=300):
    _OPT_PARAMS['T'] = T
    _OPT_PARAMS['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_func, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.success or res.x is not None:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h

def proj_gd(h, n_iter=2000, T_start=20.0, T_end=0.5, lr=0.05, check_every=50):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        exp_s = np.exp(T * shifted)
        total = np.sum(exp_s)
        if total < 1e-300:
            break
        weights = exp_s / total
        grad = np.zeros(n)
        top_k = np.argsort(weights)[-30:][::-1]
        for k_idx in top_k:
            w = weights[k_idx]
            if w < 1e-12:
                break
            s = k_idx - (n - 1)
            if s >= 0:
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] += w * (-h[:n - s])
            else:
                t = -s
                grad[t:] += w * (1.0 - h[:n - t])
                grad[:n - t] += w * (-h[t:])
        h = h - lr_t * grad * dx
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)[0]
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)[0]
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    # Get initial
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        n_init = 320
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        n_ones = int(np.sum(h0))
        target = n_init // 2
        if n_ones > target:
            ones = np.where(h0 == 1)[0]
            h0[rng.choice(ones, n_ones - target, replace=False)] = 0.0
        elif n_ones < target:
            zeros = np.where(h0 == 0)[0]
            h0[rng.choice(zeros, target - n_ones, replace=False)] = 1.0
    n0 = len(h0)
    best_c5 = compute_c5(h0)[0]
    best_h = h0.copy()
    print(f"Initial n={n0}: C5={best_c5:.6f}", flush=True)

    # Phase 1: Binary local search at n0
    h_bin = to_binary(best_h, n0)
    c_bin = compute_c5(h_bin)[0]
    print(f"Binary n={n0}: C5={c_bin:.6f}", flush=True)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin.copy()

    # Greedy rounds
    if time.time() - start_time < budget_s - 200:
        h_g, c_g = greedy_binary(h_bin, n0, max_rounds=500, seed=seed)
        if c_g < best_c5:
            best_c5 = c_g
            best_h = h_g.copy()
        print(f"Greedy n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Simulated annealing (multiple runs with different params)
    sa_configs = [
        (80000, 0.5, 0.01),
        (60000, 0.3, 0.005),
        (40000, 0.1, 0.001),
    ]
    for (ni, ts, te) in sa_configs:
        if time.time() - start_time > budget_s - 150:
            break
        h_sa, c_sa = binary_sa(best_h, n0, ni, ts, te, seed=seed + 1)
        if c_sa < best_c5:
            best_c5 = c_sa
            best_h = h_sa.copy()
            print(f"SA({ni},{ts},{te}) n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: Continuous GD from best binary
    if time.time() - start_time < budget_s - 100:
        h_gd, c_gd = proj_gd(best_h, n_iter=3000, T_start=15.0, T_end=0.5, lr=0.06, check_every=40)
        if c_gd < best_c5:
            best_c5 = c_gd
            best_h = h_gd.copy()
        print(f"GD n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: SLSQP refinement
    if n0 <= 512 and time.time() - start_time < budget_s - 60:
        for T in [4.0, 1.5, 0.5]:
            if time.time() - start_time > budget_s - 25:
                break
            h_s = slsqp_refine(best_h, n0, T, maxiter=400)
            c_s = compute_c5(h_s)[0]
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()
        print(f"SLSQP n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 4: Try binary search at n0 with more aggressive annealing
    if time.time() - start_time < budget_s - 80:
        h_bin2 = to_binary(best_h, n0)
        # Perturb and re-search
        for attempt in range(3):
            if time.time() - start_time > budget_s - 50:
                break
            h_pert = h_bin2.copy()
            n_swap = n0 // 20
            ones = np.where(h_pert == 1)[0]
            zeros = np.where(h_pert == 0)[0]
            for _ in range(n_swap):
                i = ones[rng.randint(len(ones))]
                j = zeros[rng.randint(len(zeros))]
                h_pert[i], h_pert[j] = 0.0, 1.0
                ones = np.where(h_pert == 1)[0]
                zeros = np.where(h_pert == 0)[0]
            h_p = to_binary(h_pert, n0)
            h_r, c_r = binary_sa(h_p, n0, 30000, 0.4, 0.005, seed=seed+10+attempt)
            if c_r < best_c5:
                best_c5 = c_r
                best_h = h_r.copy()
                print(f"  PerturbSA {attempt}: C5={best_c5:.6f}", flush=True)

    # Phase 5: Upsample to 2*n0 and optimize
    n2 = min(n0 * 2, 640)
    if n2 > n0 and time.time() - start_time < budget_s - 100:
        h_up = upsample(best_h, n2)
        h_bin_up = to_binary(h_up, n2)
        c_up0 = compute_c5(h_bin_up)[0]
        print(f"Up n={n2}: C5={c_up0:.6f}, t={time.time()-start_time:.0f}s", flush=True)
        if time.time() - start_time < budget_s - 60:
            h_s, c_s = binary_sa(h_bin_up, n2, 40000, 0.4, 0.005, seed=seed+100)
            if c_s < c_up0:
                h_bin_up = h_s
                c_up0 = c_s
            if time.time() - start_time < budget_s - 40:
                h_gd2, c_gd2 = proj_gd(h_bin_up, n_iter=2000, T_start=10.0, T_end=0.3, lr=0.04, check_every=50)
                if c_gd2 < c_up0:
                    h_bin_up, c_up0 = h_gd2, c_gd2
        if c_up0 < best_c5:
            best_c5 = c_up0
            best_h = h_bin_up.copy()
            print(f"Up n={n2} best: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 6: Final fine-tuning GD
    n_final = len(best_h)
    if time.time() - start_time < budget_s - 20:
        h_f, c_f = proj_gd(best_h, n_iter=2000, T_start=3.0, T_end=0.1, lr=0.02, check_every=30)
        if c_f < best_c5:
            best_c5 = c_f
            best_h = h_f.copy()
        # SLSQP at end
        if n_final <= 512 and time.time() - start_time < budget_s - 10:
            h_s = slsqp_refine(best_h, n_final, 1.0, maxiter=200)
            c_s = compute_c5(h_s)[0]
            if c_s < best_c5:
                best_c5 = c_s
                best_h = h_s.copy()

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)[0]
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
```
