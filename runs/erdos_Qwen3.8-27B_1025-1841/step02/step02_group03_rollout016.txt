def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_obj_grad(h, T):
    """Compute soft-max objective and gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.minimum(T * (corr - cmax), 0.0)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    w = exp_s / Z
    
    obj = (cmax + np.log(Z) / T) * dx
    
    # Gradient
    # S1[i] = sum_{k=n-1-i}^{2n-2-i} w[k]
    wp = np.zeros(2 * n)
    wp[1:] = np.cumsum(w)
    i_idx = np.arange(n)
    S1 = wp[2*n - 1 - i_idx] - wp[n - 1 - i_idx]
    
    # PartB[i] = sum_j h[j] * w[j + n - 1 - i]
    # Using broadcasting:
    cols = np.arange(n)[None, :]
    rows_B = (n - 1 - np.arange(n)[:, None])
    idx_B = cols + rows_B  # shape (n, n): idx_B[i,j] = j + n - 1 - i
    PartB = w[idx_B].dot(h)
    
    # PartC[i] = sum_j h[j] * w[i + n - 1 - j]
    rows_C = np.arange(n)[:, None]
    cols_C = (n - 1 - np.arange(n)[None, :])
    idx_C = rows_C + cols_C  # shape (n, n): idx_C[i,j] = i + n - 1 - j
    PartC = w[idx_C].dot(h)
    
    grad = dx * (S1 - PartB - PartC)
    return obj, grad

import numpy as np
from scipy.optimize import minimize
import time

_G_T = 50.0
_G_N = 960

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_max_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = np.max(corr)
    shifted = np.minimum(T * (corr - cmax), 0.0)
    exp_s = np.exp(shifted)
    Z = np.sum(exp_s)
    w = exp_s / Z
    obj = (cmax + np.log(Z) / T) * dx
    wp = np.zeros(2 * n)
    wp[1:] = np.cumsum(w)
    i_idx = np.arange(n)
    S1 = wp[2 * n - 1 - i_idx] - wp[n - 1 - i_idx]
    cols = np.arange(n)[None, :]
    rows_B = (n - 1 - np.arange(n)[:, None])
    idx_B = cols + rows_B
    PartB = w[idx_B].dot(h)
    rows_C = np.arange(n)[:, None]
    cols_C = (n - 1 - np.arange(n)[None, :])
    idx_C = rows_C + cols_C
    PartC = w[idx_C].dot(h)
    grad = dx * (S1 - PartB - PartC)
    return obj, grad

def _obj_func(x):
    return soft_max_obj_grad(x, _G_T)[0]

def _jac_func(x):
    return soft_max_obj_grad(x, _G_T)[1]

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def binary_snap(h, n, timeout_s=20.0):
    t0 = time.time()
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        if time.time() - t0 > timeout_s:
            break
        for val in [0.0, 1.0]:
            if abs(best[j] - val) < 1e-7:
                continue
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-13:
                best = trial
                best_c = c
    return best, best_c

def coord_descent(h, n, timeout_s=60.0):
    t0 = time.time()
    best = h.copy()
    best_c = compute_c5(h)
    cands_global = [0.0, 1.0, 0.5]
    for _round in range(50):
        if time.time() - t0 > timeout_s:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > timeout_s:
                break
            old = best[j]
            best_trial = old
            best_cj = best_c
            cands = cands_global + [
                max(0.0, old - 0.1), max(0.0, old - 0.05), max(0.0, old - 0.01),
                min(1.0, old + 0.01), min(1.0, old + 0.05), min(1.0, old + 0.1)
            ]
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = best.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_cj - 1e-13:
                    best_cj = c
                    best_trial = cand
            if best_trial != old:
                best[j] = best_trial
                best = make_feasible(best, n)
                best_c = best_cj
                improved = True
        if not improved:
            break
    return best, best_c
