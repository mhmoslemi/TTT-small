import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return (corr.max() * dx, corr)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    
    obj = (max_c + np.log(s) / tau) * dx
    return obj, grad

def nn_interpolate(h, target_n):
    """Nearest-neighbor interpolation (for step functions)."""
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = (np.arange(n) + 0.5) / n  # centers
    x_new = (np.arange(target_n) + 0.5) / target_n
    # Find nearest old point for each new point
    idx = np.searchsorted(x_old, x_new)
    idx = np.clip(idx - 1, 0, n - 1)
    # Actually for nearest neighbor, just use np.interp with method
    # But np.interp doesn't do nearest. Let's do it manually.
    # x_old and x_new are uniformly spaced, so:
    # new_index i corresponds to old_index i * n / target_n (approximately)
    idx = (np.arange(target_n) * n / target_n).astype(int)
    idx = np.clip(idx, 0, n - 1)
    return h[idx]

def subgrad_single_peak(h, n, time_limit, lr=None):

    best_c5, _ = compute_c5(h)

    stall = 0

        if time.time() - start > time_limit:
            break

        c5_cur = corr.max() * dx
        
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        sh = k_star - n + 1
        
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= h[idx + sh]
        grad *= dx
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > 500:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        
        h = h_new
    
    return best_h, best_c5

def coord_descent(h, n, time_limit):

    h = project_to_feasible(h, n).copy()

    start = time.time()

        improved = False
        for i in range(n):
            if time.time() - start > time_limit:
                break
            orig = h[i]
            for step in step_sizes:
                for delta in [step, -step]:
                    new_val = orig + delta
                    if 0 <= new_val <= 1:
                        h[i] = new_val
                        # Need to maintain constraint: adjust another point
                        # For simplicity, skip constraint maintenance here

                        c5_temp = compute_c5(h)[0]
                        if c5_temp < best_c5 - 1e-10:
                            best_c5 = c5_temp
                            improved = True
                        else:
                            h[i] = orig
                    h[i] = orig
            
            if i % 50 == 0:
                h = project_to_feasible(h, n)
        
        if not improved:
            break
    
    h = project_to_feasible(h, n)
    best_c5 = compute_c5(h)[0]
    return h, best_c5
