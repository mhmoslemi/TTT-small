def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def sa_binary(h_start, n, n_iter, T_start, T_end, seed):
    h = np.round(h_start).clip(0, 1)
    h = make_feasible(h, n)
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) != n // 2:
        h = np.zeros(n)
        h[:n//2] = 1.0
        h[n//2:] = 0.0
        rng2 = np.random.RandomState(seed + 7)
        perm = rng2.permutation(n)
        h[perm[:n//2]] = 1.0
        h[perm[n//2:]] = 0.0
    dx = 2.0 / n
    corr = compute_corr(h)
    best_c5 = dx * np.max(corr)
    best_h = h.copy()
    rng = np.random.RandomState(seed)
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        oi = rng.randint(len(ones))
        zi = rng.randint(len(zeros))
        a = ones[oi]
        b = zeros[zi]
        h[a] = 0.0
        h[b] = 1.0
        ones[oi] = b
        zeros[zi] = a
        corr_new = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = dx * np.max(corr_new)
        delta = new_c5 - best_c5
        if delta < 0 or rng.rand() < np.exp(-delta / max(T, 1e-10)):
            corr = corr_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[a] = 1.0
            h[b] = 0.0
            ones[oi] = a
            zeros[zi] = b
        if it % 5000 == 0 and it > 0:
            T *= 0.95
    return best_h, best_c5

def pgd_refine(h0, n, max_iter, T, lr, seed):
    h = h0.copy()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    corr = compute_corr(h)
    for it in range(max_iter):
        grad = compute_grad(h, corr, dx, T)
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, n)
        corr_new = compute_corr(h_new)
        c5_new = dx * np.max(corr_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
        corr = corr_new
        if it % 20 == 19:
            lr *= 0.95
    return best_h, best_c5

def coord_descent(h, n, max_rounds, time_limit):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step_sizes = [0.02, -0.02, 0.01, -0.01, 0.005, -0.005, 0.05, -0.05]
    for rnd in range(max_rounds):
        if time.time() > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() > time_limit:
                break
            old = h[j]
            for d in step_sizes:
                tri = h.copy()
                tri[j] = np.clip(old + d, 0.0, 1.0)
                tri = make_feasible(tri, n)
                c = compute_c5(tri)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    h = tri
                    old = tri[j]
                    improved = True
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-10:
                    continue
                tri = h.copy()
                tri[j] = bv
                tri = make_feasible(tri, n)
                c = compute_c5(tri)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    h = tri
                    improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def _time_left():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: SA at higher resolution
    n_sa = 4096
    h_up = upsample_h(best_h, n_sa)
    c5_up = compute_c5(h_up)
    if c5_up < best_c5:
        best_c5 = c5_up
        best_h = h_up.copy()

    # Binary initialization from upsampled
    h_bin = (h_up > 0.5).astype(float)
    h_bin = make_feasible(h_bin, n_sa)

    # Multiple SA restarts
    sa_configs = [
        (h_bin, 60000, 0.005, 0.00005, seed),
        (h_up.copy(), 50000, 0.004, 0.00004, seed + 1),
    ]
    # Random restart
    h_rand = np.zeros(n_sa)
    inds = rng.choice(n_sa, n_sa//2, replace=False)
    h_rand[inds] = 1.0
    sa_configs.append((h_rand, 50000, 0.006, 0.00006, seed + 2))

    for cfg in sa_configs:
        if _time_left() < 60:
            break
        h_s, c5_s = sa_binary(cfg[0], n_sa, cfg[1], cfg[2], cfg[3], cfg[4])
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s.copy()

    # Phase 2: PGD refinement at n_sa
    if _time_left() > 120:
        n_pgd = 2048
        h_pgd_start = upsample_h(best_h, n_pgd)
        c5_pgd_start = compute_c5(h_pgd_start)
        if c5_pgd_start < best_c5:
            best_c5 = c5_pgd_start
            best_h = h_pgd_start.copy()

        for T_pgd in [50.0, 200.0, 500.0]:
            if _time_left() < 40:
                break
            h_pg, c5_pg = pgd_refine(h_pgd_start, n_pgd, 80, T_pgd, 0.002, seed)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                h_pgd_start = h_pg.copy()

    # Upsample back to n_sa for final work
    best_h = upsample_h(best_h, n_sa)
    c5_check = compute_c5(best_h)
    if c5_check < best_c5:
        best_c5 = c5_check
        best_h = best_h.copy()

    # Phase 3: PGD at n_sa
    if _time_left() > 100:
        for T_pg in [100.0, 400.0, 1000.0]:
            if _time_left() < 30:
                break
            h_pg, c5_pg = pgd_refine(best_h, n_sa, 50, T_pg, 0.001, seed + 10)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    # Phase 4: Coordinate descent polish (on subset for speed)
    if _time_left() > 60:
        cd_n = n_sa
        h_cd = best_h.copy()
        time_limit_cd = start_time + _time_left() - 10
        h_cd, c5_cd = coord_descent(h_cd, cd_n, 4, time_limit_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 5: One final PGD pass with high T
    if _time_left() > 30:
        h_fin, c5_fin = pgd_refine(best_h, n_sa, 30, 2000.0, 0.0005, seed + 20)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
