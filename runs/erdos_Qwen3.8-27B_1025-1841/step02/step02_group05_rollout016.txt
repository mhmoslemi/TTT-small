        j1b = min(n, n - sh)
        if j0b < j1b:

    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    stall_thresh = 3000
    
    while time.time() - start < time_limit:
        t += 1
        obj, grad, c5_now = smooth_grad(h, tau)
        
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > stall_thresh:
            lr *= 0.5
            m *= 0.05
            v *= 0.05
            stall = 0
            if lr < 1e-14:
                break
        
        # Adam update
        bias1 = 1 - beta1 ** t
        bias2 = 1 - beta2 ** t
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / bias1
        v_hat = v / bias2
        
        # Scale learning rate for early steps
        lr_eff = lr * min(1.0, t / warmup)
        
        h_new = h - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)
    
    return best_h, best_c5

def subgrad_polish(h, n, time_limit, lr=0.001, top_k=3):

    best_c5 = compute_c5(h)
    best_h = h.copy()

            v *= 0.3
            stall = 0
            if lr < 1e-16:

        idx_top = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[idx_top]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()

            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        s = 1.0
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    thresh = 1e-12
    for k in range(len(corr)):
        pk = p[k]
        if pk < thresh:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx * tau
    return obj, grad, corr.max() * dx

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        s = 1.0
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    thresh = 1e-13
    for k in range(len(corr)):
        pk = p[k]
        if pk < thresh:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx * tau
    return obj, grad, corr.max() * dx


def adam_smooth(h, n, tau, time_limit, lr=0.5, warmup=80):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    b1, b2, eps = 0.9, 0.999, 1e-8

    while time.time() - start < time_limit:
        t += 1
        _, grad, c5_now = smooth_grad(h, tau)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 2000:
            lr *= 0.5
            m *= 0.01
            v *= 0.01
            stall = 0
            if lr < 1e-15:
                break
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        lr_eff = lr * min(1.0, t / warmup)
        h_new = h - lr_eff * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)
    return best_h, best_c5


def subgrad_polish(h, n, time_limit, lr=0.0005, top_k=3):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    stall = 0
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            lr *= 0.7
            v *= 0.3
            stall = 0
            if lr < 1e-16:
                break
        idx_top = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[idx_top]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(idx_top):
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b + sh:j1b + sh]
        grad *= dx
        v = 0.85 * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5
