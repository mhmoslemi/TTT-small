import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    """Compute smooth max objective and gradient using vectorized approach."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    
    # Compute gradient: grad[i] = dx * sum_k p[k] * ((1-h[i+s_k]) - h[i-s_k])
    # where s_k = k - (n-1)
    grad = np.zeros(n)
    # s ranges from -(n-1) to n-1
    for s in range(-(n-1), n):
        pk = p[s + n - 1]
        if pk < 1e-14:
            continue
        if s >= 0:
            # h[i+s] valid for i in [0, n-s-1]
            # h[i-s] valid for i in [s, n-1]
            grad[:n-s] += pk * (1.0 - h[s:n])
            grad[s:] -= pk * h[:n-s]
        else:
            t = -s
            # h[i+s] = h[i-t] valid for i in [t, n-1]
            # h[i-s] = h[i+t] valid for i in [0, n-t-1]
            grad[t:] += pk * (1.0 - h[:n-t])
            grad[:n-t] -= pk * h[t:n]
    grad *= dx
    return obj, grad

import numpy as np
import time


def project_to_feasible(g):
    n = len(g)
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400.0, 400.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        sh = k - n + 1
        if sh >= 0:
            L = n - sh
            grad[:L] += pk * (1.0 - h[sh:sh + L])
            grad[sh:sh + L] -= pk * h[:L]
        else:
            t = -sh
            L = n - t
            grad[t:t + L] += pk * (1.0 - h[:L])
            grad[:L] -= pk * h[t:t + L]
    grad *= dx
    return obj, grad


def adam_refine(h, tau, time_limit, lr=0.05):
    n = len(h)
    h = project_to_feasible(h).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t_step = 0
    stall = 0
    while time.time() - start < time_limit:
        t_step += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 400:
                lr *= 0.5
                if lr < 1e-7:
                    break
                stall = 0
                m *= 0.3
                v *= 0.1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t_step)
        v_hat = v / (1.0 - beta2 ** t_step)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new)
    return best_h, best_obj

def interp_up(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    return np.interp(x_new, x_old, h, period=2.0)

        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
        h2s = symmetrize(h2)
        c5_2s = compute_c5(h2s)
        if c5_2s < best_c5:
            best_c5 = c5_2s
            best_h = h2s.copy()

    # Phase 3: subgradient polish at best resolution
    elapsed = time.time() - start_time
    tl = max(0.0, budget_s * 0.68 - elapsed)
    if tl > 15:
        h3 = best_h.copy()
        h3, c5_3 = subgrad_refine(h3, tl, lr=0.0015)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: fine Adam at high tau for polishing
    elapsed = time.time() - start_time
    tl = max(0.0, budget_s * 0.84 - elapsed)
    if tl > 20:
        h4 = best_h.copy()
        h4, _ = adam_refine(h4, tau=120.0, time_limit=tl, lr=0.008)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: final subgradient with very small lr
    elapsed = time.time() - start_time
    tl = max(0.0, budget_s - elapsed - 15)
    if tl > 10:
        h5 = best_h.copy()
        h5, c5_5 = subgrad_refine(h5, tl, lr=0.0005)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Final: project and evaluate
    best_h = project_to_feasible(best_h)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
