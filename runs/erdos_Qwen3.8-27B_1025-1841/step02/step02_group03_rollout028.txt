import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n, rng):
    h = np.round(h).astype(float)
    num_ones = int(np.sum(h))
    target = n // 2
    if num_ones > target:
        ones = np.where(h == 1)[0]
        idx = rng.choice(ones, size=num_ones - target, replace=False)
        h[idx] = 0.0
    elif num_ones < target:
        zeros = np.where(h == 0)[0]
        idx = rng.choice(zeros, size=target - num_ones, replace=False)
        h[idx] = 1.0
    return h

def _slsqp_obj(x):
    n = _G_N
    T = _G_T
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)

def _slsqp_con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _slsqp_con_jac(x):
    return np.ones(_G_N)

def slsqp_polish(h, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_con_fun, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h.copy(), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(res.x, n)
    except Exception:
        return h.copy()

    start = time.time()
    h_curr = h.copy()
    curr_c5 = compute_c5(h_curr)
    best_c5 = curr_c5

    T = T_start

    stall_limit = 5000

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if it % 10000 == 0 and time.time() - start > time_limit:
            break

        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        center = n - 1
        u = rng.random()

        if u < 0.72:
            corr_work = corr.copy()
            corr_work[center] = -1.0
            top_k = min(5, len(corr_work))
            top_idxs = np.argpartition(corr_work, -top_k)[-top_k:]
            max_idx = int(top_idxs[np.argmax(corr_work[top_idxs])])
            d = max_idx - center

            if d > 0:

def thue_morse(n, rng):

    for i in range(1, n):
        if i % 2 == 0:
            t[i] = t[i // 2]
        else:
            t[i] = 1.0 - t[(i - 1) // 2]
