    """Generate an m-sequence (maximum length LFSR sequence) of period 2^m - 1."""
    # Use a primitive polynomial
    # For m=10: x^10 + x^7 + 1
    # For m=11: x^11 + x^9 + x^6 + x^5 + 1

        4: [3, 0],  # x^4 + x^3 + 1
        10: [7, 0],  # x^10 + x^7 + 1
        11: [9, 6, 5, 0],  # x^11 + x^9 + x^6 + x^5 + 1
    }
    if m not in taps:

        # For now, use a known one
        raise ValueError(f"No primitive polynomial for m={m}")
    
    L = 2**m - 1
    register = np.ones(m, dtype=int)  # Start with all 1s

        # Compute new bit

        # Shift register

_G_N = 0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def _obj(h):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    s = corr - mc
    s = np.clip(s, -200.0 / T, 50.0)
    return (mc + np.log(np.sum(np.exp(T * s))) / T) * dx

def _cons(h):
    return np.sum(h) - _G_N / 2.0

def _cons_jac(h):
    return np.ones(_G_N)

    _G_T = T

                   constraints=[{'type': 'eq', 'fun': _cons, 'jac': _cons_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-13})
    return make_feasible(res.x, n)
