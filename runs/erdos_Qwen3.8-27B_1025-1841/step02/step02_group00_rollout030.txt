        improved = False
        for j in range(n):
            if start_time is not None and time.time() - start_time > time_limit:
                break
            old_val = h_cd[j]
            for delta in [0.01, -0.01, 0.02, -0.02, 0.005, -0.005]:
                cand = old_val + delta
                if cand < 0 or cand > 1:
                    continue
                trial = h_cd.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    h_cd = trial
                    old_val = cand
                    improved = True
            for bv in (0.0, 1.0):
                if abs(bv - h_cd[j]) < 1e-9:
                    continue
                trial = h_cd.copy()
                trial[j] = bv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-13:
                    best_c5 = c
                    h_cd = trial
                    improved = True
        if not improved:
            break
    return h_cd, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tl():
        return budget_s - (time.time() - start_time)

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    best_h = h0.copy()
    best_c5 = compute_c5(h0)

    # Phase 1: Projected gradient at increasing resolutions
    for n_pts in [1024, 2048]:
        if tl() < 80:
            break
        h_curr = upsample_h(best_h, n_pts)
        c5_curr = compute_c5(h_curr)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h_curr.copy()

        # Top-k subgradient with momentum (exploitation)
        for momentum in [0.85, 0.95]:
            if tl() < 40:
                break
            lr_base = 0.003 if n_pts <= 1024 else 0.001
            h_pg, c5_pg = pgd_nesterov(h_curr, n_pts, 'topk', 100, lr_base,
                                        momentum, max_iters=2000,
                                        start_time=start_time, time_limit=tl() - 10)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                h_curr = h_pg.copy()

        # Full softmax gradient (exploration)
        for T in [100, 300]:
            if tl() < 30:
                break
            lr_base = 0.002 if n_pts <= 1024 else 0.0008
            h_pg, c5_pg = pgd_nesterov(h_curr, n_pts, 'full', T, lr_base,
                                        0.9, max_iters=1000,
                                        start_time=start_time, time_limit=tl() - 10)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()
                h_curr = h_pg.copy()

        # SLSQP refinement
        if tl() > 80:
            try:
                h_opt = slsqp_optimize(h_curr, n_pts, 200, 200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_curr = h_opt.copy()
            except Exception:
                pass

        if tl() > 50:
            try:
                h_opt = slsqp_optimize(h_curr, n_pts, 800, 150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

        # Random restart blended with best
        if tl() > 60:
            h_rand = 0.6 * upsample_h(best_h, n_pts) + 0.4 * rng.rand(n_pts)
            h_rand = make_feasible(h_rand, n_pts)
            h_pg, c5_pg = pgd_nesterov(h_rand, n_pts, 'topk', 100, 0.002,
                                        0.9, max_iters=800,
                                        start_time=start_time, time_limit=tl() - 10)
            if c5_pg < best_c5:
                best_c5 = c5_pg
                best_h = h_pg.copy()

    # Phase 2: Coordinate descent polish
    n_cd = len(best_h)
    h_cd, c5_cd = coord_descent(best_h, n_cd, max_rounds=5,
                                 start_time=start_time, time_limit=tl() - 20)
    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Final SLSQP at high T
    if tl() > 20:
        try:
            h_opt = slsqp_optimize(best_h, len(best_h), 2000, 80)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
