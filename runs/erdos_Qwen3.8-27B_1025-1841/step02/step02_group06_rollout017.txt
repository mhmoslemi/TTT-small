                    j0b = max(0, -d)
                    j1b = min(n, n - d)
                    if j0b < j1b:
                        idx = np.arange(j0b, j1b)
                        grad[idx] -= p[k] * h[idx + d]
                grad *= dx
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_to_feasible(h_new, n)
            h = h_new
    
    return best_h, best_c5

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    if g.min() >= -1e-12 and g.max() <= 1 + 1e-12 and abs(g.sum() - target) < 1e-9:
        return np.clip(g, 0.0, 1.0)
    lo = g.min() - 3.0
    hi = g.max() + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def compute_grad_topM(h, p_top, top_idx, n, dx):
    grad = np.zeros(n)
    for idx_i in range(len(top_idx)):
        pk = p_top[idx_i]
        if pk < 1e-15:
            continue
        k = top_idx[idx_i]
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx


def adam_optimize(h, n, tau_list, topM_list, time_limit, lr=0.12, seed=42):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1 = 0.85
    beta2 = 0.999
    eps = 1e-8

    for phase in range(len(tau_list)):
        tau = tau_list[phase]
        topM = topM_list[phase] if phase < len(topM_list) else topM_list[-1]
        phase_iters = 50000
        for it in range(phase_iters):
            if time.time() - start > time_limit:
                return best_h, best_c5
            t += 1
            corr = compute_corr(h)
            g = corr * dx
            max_g = g.max()
            if max_g < best_c5 - 1e-13:
                best_c5 = max_g
                best_h = h.copy()

            nm1 = 2 * n - 1
            if topM > 0 and topM < nm1:
                top_idx = np.argpartition(g, -topM)[-topM:]
                g_top = g[top_idx]
                shifted = np.clip(tau * (g_top - g_top.max()), -500, 500)
                exp_c = np.exp(shifted)
                s = exp_c.sum()
                p_top = exp_c / s
                grad = compute_grad_topM(h, p_top, top_idx, n, dx)
            else:
                shifted = np.clip(tau * (g - max_g), -500, 500)
                exp_c = np.exp(shifted)
                s = exp_c.sum()
                p = exp_c / s
                grad = np.zeros(n)
                for k in range(nm1):
                    if p[k] < 1e-15:
                        continue
                    d = k - n + 1
                    j0 = max(0, d)
                    j1 = min(n, n + d)
                    if j0 < j1:
                        idx = np.arange(j0, j1)
                        grad[idx] += p[k] * (1.0 - h[idx - d])
                    j0b = max(0, -d)
                    j1b = min(n, n - d)
                    if j0b < j1b:
                        idx = np.arange(j0b, j1b)
                        grad[idx] -= p[k] * h[idx + d]
                grad *= dx

            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            h_new = project_to_feasible(h - step, n)
            h = h_new
    return best_h, best_c5


def subgrad_optimize(h, n, time_limit, lr_init=0.004, topK=5):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    patience = 300
    lr = lr_init
    it = 0
    while time.time() - start < time_limit:
        it += 1
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > patience:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break

        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK
        cur_lr = lr / (1.0 + it * 0.003)
        v = beta * v - cur_lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def local_perturb_opt(h, n, time_limit, n_pert=8):
    """Try perturbing small random blocks and keep improvements."""
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)

    start = time.time()

    dx = 2.0 / n

        i0 = rng.randint(0, max(1, n - n_pert))
        idx = np.arange(i0, min(n, i0 + n_pert))

        if c5_trial < best_c5 - 1e-12:

            h = h_trial.copy()
