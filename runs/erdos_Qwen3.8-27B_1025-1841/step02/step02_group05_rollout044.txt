        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def grad_for_shift(h, sh, n):
    grad = np.zeros(n)
    j0 = max(0, sh)
    j1 = min(n, sh + n)
    if j0 < j1:
        grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
    j0b = max(0, -sh)
    j1b = min(n, n - sh)
    if j0b < j1b:
        grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
    return grad


def optimize_focused(h, time_limit, top_k=10, tau_start=60.0, lr=None, momentum=0.9, tau_min=3.0, k_max=30):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.3 / tau_start
    v = np.zeros(n)
    beta = momentum
    tau = tau_start
    stall = 0
    it = 0
    while time.time() - start < time_limit:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_use = min(top_k + stall // 150, k_max)
        idx_top = np.argpartition(corr, -k_use)[-k_use:]
        top_corr = corr[idx_top]
        max_c = top_corr.max()
        shifted = tau * (top_corr - max_c)
        max_shift = shifted.max()
        exp_c = np.exp(shifted - max_shift)
        s = exp_c.sum()
        p = exp_c / s
        grad = np.zeros(n)
        for i, k in enumerate(idx_top):
            if p[i] < 1e-14:
                continue
            sh = k - n + 1
            grad += p[i] * grad_for_shift(h, sh, n)
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 250:
            tau = max(tau_min, tau * 0.72)
            lr = 0.3 / tau
            v *= 0.25
            stall = 0
        elif stall > 2500:
            break
    return best_h, best_c5


def optimize_adaptive_tau(h, time_limit, phases):
    n = len(h)
    h_cur = h.copy()
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    for phase in phases:
        el = time.time() - (0)
        if time.time() - phase['start'] > time_limit:
            break
        tl = min(phase['dur'], time_limit - (time.time() - phase['start']))
        if tl <= 5:
            break
        h_cur, c5 = optimize_focused(
            h_cur, tl, top_k=phase['k'], tau_start=phase['tau'],
            lr=phase.get('lr', 0.3 / phase['tau']),
            momentum=phase.get('mom', 0.9),
            tau_min=phase.get('tau_min', 3.0), k_max=phase.get('kmax', 30)
        )
        if c5 < best_c5:
            best_c5, best_h = c5, h_cur.copy()
    return best_h, best_c5


def perturbed_restart(h, n, scale=0.03, rng=None):
    if rng is None:
        rng = np.random
    h_p = h + rng.standard_normal(n) * scale
    return project_to_feasible(h_p, n)


def try_binary_and_optimize(h, n, time_limit):
    h_bin = np.where(h > 0.5, 1.0, 0.0)
    h_bin = project_to_feasible(h_bin, n)
    c5_bin = compute_c5(h_bin)
    if time_limit > 30:
        h_opt, c5_opt = optimize_focused(
            h_bin, time_limit, top_k=10, tau_start=40.0,
            lr=0.2, momentum=0.9, tau_min=5.0, k_max=25
        )
        if c5_opt < c5_bin:
            return h_opt, c5_opt
    return h_bin, c5_bin


def soft_binary_push(h, n, amount=0.3):
    h_push = h + amount * (np.where(h > 0.5, 1.0 - h, -(h - 0.0)))
    return project_to_feasible(h_push, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    n = n_init
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase A: Coarse exploration at lower resolution
    if rem() > 120:
        n_coarse = max(400, n // 5)
        h_c = interp_periodic(best_h, n_coarse)
        h_c = project_to_feasible(h_c, n_coarse)
        tl = min(80, rem() * 0.12)
        h_c, c5_c = optimize_focused(h_c, tl, top_k=6, tau_start=50.0, lr=0.3, momentum=0.88, tau_min=8.0, k_max=15)
        h_c_up = interp_periodic(h_c, n)
        h_c_up = project_to_feasible(h_c_up, n)
        c5_up = compute_c5(h_c_up)
        if c5_up < best_c5:
            best_c5, best_h = c5_up, h_c_up.copy()

    # Phase B: Fine optimization from best
    if rem() > 100:
        tl = min(150, rem() * 0.22)
        h_b, c5_b = optimize_focused(best_h, tl, top_k=8, tau_start=100.0, lr=0.25, momentum=0.9, tau_min=15.0, kmax=20)
        if c5_b < best_c5:
            best_c5, best_h = c5_b, h_b.copy()

    # Phase C: Medium tau
    if rem() > 80:
        tl = min(120, rem() * 0.18)
        h_c, c5_c = optimize_focused(best_h, tl, top_k=14, tau_start=35.0, lr=0.15, momentum=0.9, tau_min=6.0, kmax=40)
        if c5_c < best_c5:
            best_c5, best_h = c5_c, h_c.copy()

    # Phase D: Low tau, tight focus
    if rem() > 60:
        tl = min(100, rem() * 0.16)
        h_d, c5_d = optimize_focused(best_h, tl, top_k=5, tau_start=15.0, lr=0.1, momentum=0.93, tau_min=3.0, kmax=12)
        if c5_d < best_c5:
            best_c5, best_h = c5_d, h_d.copy()

    # Phase E: Soft binary push then optimize
    if rem() > 60:
        h_sb = soft_binary_push(best_h, n, amount=0.25)
        tl = min(50, rem() * 0.2)
        h_e, c5_e = optimize_focused(h_sb, tl, top_k=10, tau_start=40.0, lr=0.2, momentum=0.9, tau_min=5.0, kmax=25)
        if c5_e < best_c5:
            best_c5, best_h = c5_e, h_e.copy()

    # Phase F: True binary snap
    if rem() > 50:
        tl = min(40, rem() * 0.2)
        h_bin, c5_bin = try_binary_and_optimize(best_h, n, tl)
        if c5_bin < best_c5:
            best_c5, best_h = c5_bin, h_bin.copy()

    # Phase G: Multiple perturbations
    for r in range(5):
        if rem() > 45:
            scale = 0.015 + 0.01 * r
            h_p = perturbed_restart(best_h, n, scale=scale)
            tl = min(40, rem() * 0.2)
            h_g, c5_g = optimize_focused(h_p, tl, top_k=10, tau_start=35.0, lr=0.18, momentum=0.9, tau_min=5.0, kmax=30)
            if c5_g < best_c5:
                best_c5, best_h = c5_g, h_g.copy()

    # Phase H: Higher resolution n=4000
    if rem() > 130:
        n2 = 4000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl1 = min(90, rem() * 0.3)
        h2, c5_2 = optimize_focused(h2, tl1, top_k=10, tau_start=80.0, lr=0.22, momentum=0.9, tau_min=10.0, kmax=25)
        if rem() > 50:
            tl2 = min(50, rem() * 0.25)
            h2, c5_2 = optimize_focused(h2, tl2, top_k=6, tau_start=18.0, lr=0.1, momentum=0.93, tau_min=4.0, kmax=15)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase I: Very high resolution n=6000 if lots of time left
    if rem() > 100:
        n3 = 6000
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        tl3 = min(70, rem() * 0.35)
        h3, c5_3 = optimize_focused(h3, tl3, top_k=8, tau_start=50.0, lr=0.18, momentum=0.9, tau_min=8.0, kmax=20)
        if rem() > 30:
            tl4 = min(30, rem() * 0.3)
            h3, c5_3 = optimize_focused(h3, tl4, top_k=5, tau_start=12.0, lr=0.08, momentum=0.93, tau_min=3.0, kmax=10)
        if c5_3 < best_c5:
            best_c5, best_h = c5_3, h3.copy()

    # Phase J: Final polish
    if rem() > 25:
        n_f = len(best_h)
        tl_f = min(20, rem() * 0.5)
        h_f, c5_f = optimize_focused(best_h, tl_f, top_k=3, tau_start=8.0, lr=0.06, momentum=0.94, tau_min=3.0, kmax=8)
        if c5_f < best_c5:
            best_c5, best_h = c5_f, h_f.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
