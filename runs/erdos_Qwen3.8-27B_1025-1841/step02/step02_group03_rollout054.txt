    mc = np.max(corr)

        shifted = corr - mc
        val = mc + np.log(np.sum(np.exp(T * shifted))) / T
    else:
        val = mc
    return val * dx

def slsqp_polish(h, n, T, maxiter, t0, time_limit):
    def obj(x):
        return _glb_obj(x, T, n)
    
    def con_fun(x):
        return _sum_con(x, n)
    
    def con_jac(x):
        return _sum_jac(x, n)

        c5 = compute_c5(res.x)
        return res.x, c5
    except Exception:
        return h, compute_c5(h)
