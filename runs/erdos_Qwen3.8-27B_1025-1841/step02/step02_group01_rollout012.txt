import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def softmax_loss(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    return (mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T) * dx

def slsqp_optimize(h, n, T, maxiter=400):
    def obj(x):
        return softmax_loss(x, T)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-10})
    return res.x

def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-np.clip(z, -50, 50)))

def reparam_obj(z, n, T, lam):
    h = sigmoid(z)
    loss = softmax_loss(h, T)
    penalty = lam * (np.sum(h) - n / 2.0) ** 2
    return loss + penalty

def lbfgs_optimize(h, n, T, lam=5.0, maxiter=300):
    z = np.log(h / (1.0 - np.clip(h, 1e-10, 1.0 - 1e-10)))
    z = np.clip(z, -10, 10)
    res = minimize(reparam_obj, z, args=(n, T, lam), method='L-BFGS-B',
                   options={'maxiter': maxiter, 'ftol': 1e-12, 'gtol': 1e-8})
    h_opt = sigmoid(res.x)
    return make_feasible(h_opt, n)

def bit_flip_search(h, n, start_time, time_limit, rng, max_rounds=5000):
    best = h.copy()
    # Round to binary
    b = (best >= 0.5).astype(float)
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        ones = np.where(b == 1)[0]
        flip = rng.choice(ones, n_ones - target, replace=False)
        b[flip] = 0.0
    elif n_ones < target:
        zeros = np.where(b == 0)[0]
        flip = rng.choice(zeros, target - n_ones, replace=False)
        b[flip] = 1.0
    best = b
    best_c5 = compute_c5(best)
    if best_c5 >= compute_c5(h):
        best = h.copy()
        best_c5 = compute_c5(best)

    ones = np.where(best >= 0.5)[0]
    zeros = np.where(best < 0.5)[0]
    stall = 0

    for it in range(max_rounds):
        if time.time() - start_time > time_limit - 30:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        improved = False
        n_tries = min(50, max(10, 200000 // max(1, it)))
        for _ in range(n_tries):
            if time.time() - start_time > time_limit - 30:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            trial = best.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            c = compute_c5(trial)
            if c < best_c5 - 1e-12:
                best = trial
                best_c5 = c
                ones = np.where(best >= 0.5)[0]
                zeros = np.where(best < 0.5)[0]
                improved = True
                break
            # Occasionally accept slightly worse (annealing)
            elif it > 500 and c < best_c5 + 0.0005 * (1.0 - it / max_rounds) and rng.rand() < 0.1:
                best = trial
                best_c5 = c
                ones = np.where(best >= 0.5)[0]
                zeros = np.where(best < 0.5)[0]
        if improved:
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break

    return best, best_c5
