def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):

    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

                grad[j2] += wk * (-h[j2 - p])
    return grad

def pg_optimize(h_start, n, T, max_iter=200, time_limit=50, momentum=0.8):
    h = h_start.copy()
    h_prev = h.copy()
    t_acc = 1.0
    t0 = time.time()
    best_h = h.copy()
    best_f = lse_obj(h, n, T)
    alpha = 2.0 / (T * (2.0 / n) * n)
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        y = h + ((t_acc - 1.0) / (t_acc + 1.0)) * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        grad = lse_grad(y, n, T)
        f_cur = lse_obj(h, n, T)
        a = alpha
        for _ in range(25):
            h_new = h - a * grad
            h_new = make_feasible(h_new, n)
            f_new = lse_obj(h_new, n, T)
            if f_new < f_cur - 1e-16:
                break
            a *= 0.5
        else:
            break
        h_prev = h
        h = h_new
        t_acc = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t_acc * t_acc))
        alpha = a * 1.2
        if f_new < best_f:
            best_f = f_new
            best_h = h.copy()
    return best_h

def slsqp_opt(h_start, n, T, max_iter=80, time_limit=50):
    def obj_fn(x):
        return lse_obj(x, n, T)
    def con_fun(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    res = minimize(obj_fn, h_start, method='SLSQP',
                   bounds=bounds,
                   constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                   options={'maxiter': max_iter, 'ftol': 1e-13})
    return make_feasible(res.x, n)

def fourier_perturb(h, n, rng, amp=0.15):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    noise = np.zeros(n)
    for k in range(1, 8):
        freq = 2.0 * np.pi * k / 2.0
        a = rng.randn() * amp
        b = rng.randn() * amp
        noise += a * np.cos(freq * x) + b * np.sin(freq * x)
    return np.clip(h + noise, 0.0, 1.0)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start_time)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)
    h0 = make_feasible(h0, len(h0))
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 1: Optimize at n=1280
    n1 = 1280
    h1 = upsample_h(best_h, n1)
    c5_1 = compute_c5(h1)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    for T in [150.0, 400.0, 1000.0, 3000.0]:
        if time_left() < 35:
            break
        h_opt = pg_optimize(h1, n1, T, max_iter=150, time_limit=25)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
        h1 = h_opt if c5_opt < c5_1 else h1
        if c5_opt < best_c5:
            c5_1 = c5_opt

    if time_left() > 25:
        h_opt = slsqp_opt(h1, n1, 8000.0, max_iter=60, time_limit=20)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            h1 = h_opt.copy()

    # Phase 2: Go to n=2560
    n2 = 2560
    if time_left() > 120:
        h2 = upsample_h(best_h, n2)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

        for T in [200.0, 600.0, 2000.0]:
            if time_left() < 35:
                break
            h_opt = pg_optimize(h2, n2, T, max_iter=100, time_limit=25)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h2 = h_opt if c5_opt < c5_2 else h2
            if c5_opt < best_c5:
                c5_2 = c5_opt

        if time_left() > 30:
            h_opt = slsqp_opt(h2, n2, 15000.0, max_iter=50, time_limit=25)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h2 = h_opt.copy()

    # Phase 3: Try n=4096
    n3 = 4096
    if time_left() > 150:
        h3 = upsample_h(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

        for T in [400.0, 1500.0]:
            if time_left() < 35:
                break
            h_opt = pg_optimize(h3, n3, T, max_iter=60, time_limit=25)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h3 = h_opt if c5_opt < c5_3 else h3
            if c5_opt < best_c5:
                c5_3 = c5_opt

        if time_left() > 30:
            h_opt = slsqp_opt(h3, n3, 30000.0, max_iter=40, time_limit=25)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                h3 = h_opt.copy()

    # Phase 4: Structured restarts at best resolution
    n_best = len(best_h)
    if time_left() > 80:
        for restart in range(4):
            if time_left() < 30:
                break
            h_r = fourier_perturb(best_h, n_best, rng, amp=0.12 + 0.03 * restart)
            h_r = make_feasible(h_r, n_best)
            h_opt = pg_optimize(h_r, n_best, 300.0, max_iter=80, time_limit=18)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            if time_left() > 15:
                h_opt2 = slsqp_opt(h_opt, n_best, 20000.0, max_iter=30, time_limit=12)
                c5_opt2 = compute_c5(h_opt2)
                if c5_opt2 < best_c5:
                    best_c5 = c5_opt2
                    best_h = h_opt2.copy()

    # Phase 5: Random perturbation restart
    if time_left() > 40:
        h_r = np.clip(best_h + rng.randn(n_best) * 0.08, 0.0, 1.0)
        h_r = make_feasible(h_r, n_best)
        h_opt = pg_optimize(h_r, n_best, 200.0, max_iter=60, time_limit=15)
        c5_opt = compute_c5(h_opt)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 6: Final high-T SLSQP polish
    if time_left() > 25:
        n_fin = len(best_h)
        try:
            h_opt = slsqp_opt(best_h, n_fin, 100000.0, max_iter=40, time_limit=20)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 7: Coordinate descent polish at reduced res if needed
    n_cd = min(n_best, 2560)
    if _time_left_check(start_time, budget_s) > 20 and n_cd < n_best:
        h_cd = upsample_h(best_h, n_cd)
        c5_cd = compute_c5(h_cd)
        for _cd in range(3):
            if time_left() < 15:
                break
            improved = False
            step_sizes = [0.02, -0.02, 0.01, -0.01]
            for j in range(n_cd):
                if time_left() < 15:
                    break
                for dv in step_sizes:
                    cand = h_cd[j] + dv
                    if cand < 0.0 or cand > 1.0:
                        continue
                    trial = h_cd.copy()
                    trial[j] = cand
                    trial = make_feasible(trial, n_cd)
                    c = compute_c5(trial)
                    if c < c5_cd - 1e-13:
                        c5_cd = c
                        h_cd = trial
                        improved = True
            if not improved:
                break
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = upsample_h(h_cd, n_best)

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)

def _time_left_check(start_time, budget_s):
    return budget_s - (time.time() - start_time)
