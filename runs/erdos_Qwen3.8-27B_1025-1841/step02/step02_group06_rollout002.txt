```python
def solve_lp_with_lags(h_old, n, active_lags, solver='CBC'):
    dx = 2.0 / n
    one_minus_h = 1.0 - h_old
    
    h = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2.0]
    
    for k in active_lags:
        d = k - (n - 1)
        # corr_k(lin) = sum_i h[i] * one_minus_h[i-d] for valid i
        if d >= 0:

            # constraint: h[d:n] @ coeff <= t/dx
            constraints.append(cp.sum(h[d:] * coeff) <= t / dx)
        else:

            coeff = one_minus_h[-d:]  # length n+d = n+|d|

            constraints.append(cp.sum(h[:n+d] * coeff) <= t / dx)

    if h.value is None:
```

```python
import numpy as np
import time
import cvxpy as cp

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def gradient_topM(h, p_top, top_idx, n, dx):
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_top[i]
        if pk < 1e-15:
            continue
        k = top_idx[i]
        d = k - n + 1
        if d >= 0:
            # For i from d to n-1: h[i]*(1-h[i-d])
            # ∂/∂h[m]: (1-h[m-d]) for m in [d, n-1], minus h[m+d] for m in [0, n-1-d]
            idx1 = np.arange(d, n)
            grad[idx1] += pk * (1.0 - h[idx1 - d])
            idx2 = np.arange(0, n - d)
            grad[idx2] -= pk * h[idx2 + d]
        else:
            # For i from 0 to n-1+d: h[i]*(1-h[i-d]) where i-d in [0, n-1]
            dd = -d
            idx1 = np.arange(0, n - dd)
            grad[idx1] += pk * (1.0 - h[idx1 + dd])
            idx2 = np.arange(dd, n)
            grad[idx2] -= pk * h[idx2 - dd]
    return grad * dx

def softmax_project_grad(h, n, tau, max_iters, time_limit, lr_init=0.2, topM=50):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(100, max_iters // 20)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            shifted = np.clip(tau * (g_top - g_top.max()), -500, 500)
            exp_c = np.exp(shifted)
            p_top = exp_c / exp_c.sum()
        else:
            top_idx = np.arange(2 * n - 1)
            g_top = g
            shifted = np.clip(tau * (g - g.max()), -500, 500)
            exp_c = np.exp(shifted)
            p_top = exp_c / exp_c.sum()
        
        grad = gradient_topM(h, p_top, top_idx, n, dx)
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    
    return best_h, best_obj

def solve_lp_erdos(h_lin, n, active_lags, solver='CBC'):
    """Solve LP: minimize t s.t. linearized corr_k <= t/dx for k in active_lags"""
    dx = 2.0 / n
    omh = 1.0 - h_lin
    
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n / 2.0]
    
    for k in active_lags:
        d = k - (n - 1)
        if d >= 0:
            coeff = omh[:n - d]
            constraints.append(cp.sum(h_var[d:] * coeff) <= t_var / dx)
        else:
            dd = -d
            coeff = omh[dd:]
            constraints.append(cp.sum(h_var[:n - dd] * coeff) <= t_var / dx)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    try:
        prob.solve(solver=solver)
    except:
        try:
            prob.solve(solver='CLARABEL')
        except:
            return h_lin.copy()
    
    if h_var.value is None:
        return h_lin.copy()
    
    return project_to_feasible(h_var.value, n)

def cutting_plane_optimize(h_init, n, time_limit, K=50, max_iters=500, solver='CBC', blend=1.0):
    start = time.time()
    h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * (2.0 / n)
        
        # Select active lags: top K by value
        if K < 2 * n - 1:
            active = np.argpartition(g, -K)[-K:]
        else:
            active = np.arange(2 * n - 1)
        
        # Solve LP
        h_lp = solve_lp_erdos(h, n, active, solver=solver)
        
        # Blend
        if blend < 1.0:
            h_new = (1 - blend) * h + blend * h_lp
            h_new = project_to_feasible(h_new, n)
        else:
            h_new = h_lp
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Check convergence
        if abs(c5_new - compute_c5(h)) < 1e-10:
            h = h_new
            # Try with slightly different K
            break
        
        h = h_new
    
    return best_h, best_c5
```

corr[k] = sum_i h[i] * (1 - h[i-d]) where d = k - (n-1), and the sum is over i where both i and i-d are in [0, n).

For d >= 0: valid i are d, d+1, ..., n-1 (so that i-d is in [0, n-1-d])
  corr[k] = sum_{i=d}^{n-1} h[i] * (1 - h[i-d])

For d < 0 (let dd = -d): valid i are 0, 1, ..., n-1-dd (so that i+dd is in [0, n-1])
  corr[k] = sum_{i=0}^{n-1-dd} h[i] * (1 - h[i+dd])

∂corr[k]/∂h[m]:

For d >= 0:
- If m ∈ [d, n-1]: m appears in h[i] term, gives (1 - h[m-d])
- If m ∈ [0, n-1-d]: m appears in h[i-d] term (i = m+d), gives -h[m+d]

So: ∂corr[k]/∂h[m] = [1-h[m-d]]·1_{d≤m≤n-1} - h[m+d]·1_{0≤m≤n-1-d}

For d < 0 (dd = -d):
- If m ∈ [0, n-1-dd]: m appears in h[i] term, gives (1 - h[m+dd])
- If m ∈ [dd, n-1]: m appears in h[i+dd] term (i = m-dd), gives -h[m-dd]

So: ∂corr[k]/∂h[m] = [1-h[m+dd]]·1_{0≤m≤n-1-dd} - h[m-dd]·1_{dd≤m≤n-1}

```python
from scipy.optimize import linprog

def solve_lp_fast(h_lin, n, active_lags):
    dx = 2.0 / n
    omh = 1.0 - h_lin
    K = len(active_lags)
    
    # Build A_ub: K x (n+1)
    A_ub = np.zeros((K, n + 1))
    A_ub[:, :n] = 0  # will fill
    A_ub[:, n] = -1.0 / dx
    b_ub = np.zeros(K)
    
    for i, k in enumerate(active_lags):
        d = k - (n - 1)
        if d >= 0:
            A_ub[i, d:] = omh[:n - d]
        else:
            dd = -d
            A_ub[i, :n - dd] = omh[dd:]
    
    # A_eq: 1 x (n+1)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Objective: minimize t
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    # Bounds
    bounds = [(0, 1)] * n + [(None, None)]
    
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if not result.success:
        return h_lin.copy()
    
    h_new = result.x[:n]
    return project_to_feasible(h_new, n)
```
