import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15 * n:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        adj = diff / cnt
        h[mask] += adj
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_dx = 2.0 / old_n
    new_dx = 2.0 / new_n
    old_x = np.arange(old_n) * old_dx
    new_x = np.arange(new_n) * new_dx
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def subgrad_at_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] -= h[:n - s]
    return grad

def compute_softgrad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        return subgrad_at_shift(h, shift) * dx
    weights = exp_s / total
    if top_k is not None and top_k < len(weights):
        idx = np.argsort(weights)[-top_k:]
        mw = np.zeros_like(weights)
        mw[idx] = 1.0
        weights = weights * mw
        s_arr = np.sum(weights)
        if s_arr > 1e-300:
            weights /= s_arr
        else:
            return np.zeros(n)
    grad = np.zeros(n)
    active = np.where(weights > 1e-12)[0]
    for k_idx in active:
        grad += weights[k_idx] * subgrad_at_shift(h, k_idx - (n - 1))
    return grad * dx

def gd_run(h, n_iter, T_start, T_end, lr, top_k_start=60, top_k_end=5, check_every=10):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / max(T_start, 1e-10)) ** frac
        lr_t = lr * (1.0 - 0.8 * frac ** 0.7)
        tk = int(top_k_start + (top_k_end - top_k_start) * frac)
        tk = max(tk, 3)
        grad = compute_softgrad(h, T, top_k=tk)
        h = h - lr_t * grad
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5

def slsqp_obj(hv, T_val, n):
    dx = 2.0 / n
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T_val * shifted))) / T_val) * dx

def slsqp_grad(hv, T_val, n):
    return compute_softgrad(hv, T_val, top_k=80)

def slsqp_cons_fun(hv, n):
    return np.sum(hv) - n / 2.0

def slsqp_cons_jac(hv, n):
    return np.ones(n)

def slsqp_run(h, T_val, maxiter=600):
    n = len(h)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: slsqp_cons_fun(x, n),
             'jac': lambda x: slsqp_cons_jac(x, n)}]
    try:
        res = minimize(slsqp_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons,
                      args=(T_val, n),
                      options={'maxiter': maxiter, 'ftol': 1e-16, 'iprint': 0})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            cx = compute_c5(hx)
            return hx, cx
    except Exception:
        pass
    return h.copy(), compute_c5(h)
