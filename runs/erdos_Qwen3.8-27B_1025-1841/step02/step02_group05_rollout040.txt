import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau, topk_frac):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s

    n_shifts = len(p)
    n_use = max(1, int(n_shifts * topk_frac))
    top_idx = np.argsort(p)[::-1][:n_use]
    p_use = p[top_idx].copy()
    p_use /= p_use.sum()

    grad = np.zeros(n)
    n_minus_1 = n - 1
    for i in range(n_use):
        k = top_idx[i]
        pk = p_use[i]
        sh = k - n_minus_1
        j_lo = max(0, -sh)
        j_hi = min(n, n - sh)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += pk * (1.0 - h[j_lo + sh:j_hi + sh])
        j_lo2 = max(0, sh)
        j_hi2 = min(n, n + sh)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= pk * h[j_lo2 - sh:j_hi2 - sh]

    grad *= dx
    obj = max_c + np.log(s) / tau
    return obj, grad


def adaptive_topk(tau):
    if tau > 30:
        return 0.05
    elif tau > 15:
        return 0.08
    elif tau > 8:
        return 0.12
    else:
        return 0.25


def optimize_momentum(h, n, tau, time_limit, lr_init, momentum, min_lr):
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = momentum
    stall = 0
    tkg = adaptive_topk(tau)

    while time.time() - start < time_limit:
        obj, grad = smooth_obj_grad(h, tau, tkg)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2500:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < min_lr:
                break

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def binary_round(h, n):
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:n // 2]] = 1.0
    return h_bin


def subgrad_optimize(h, n, time_limit, lr_init, top_k=8):
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    dx = 2.0 / n

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break

        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()

        grad = np.zeros(n)
        n_minus_1 = n - 1
        for i in range(top_k):
            k = top_idx[i]
            wi = w[i]
            sh = k - n_minus_1
            j_lo = max(0, -sh)
            j_hi = min(n, n - sh)
            if j_lo < j_hi:
                grad[j_lo:j_hi] += wi * (1.0 - h[j_lo + sh:j_hi + sh])
            j_lo2 = max(0, sh)
            j_hi2 = min(n, n + sh)
            if j_lo2 < j_hi2:
                grad[j_lo2:j_hi2] -= wi * h[j_lo2 - sh:j_hi2 - sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5
