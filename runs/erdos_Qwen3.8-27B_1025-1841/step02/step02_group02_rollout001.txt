    """Subgradient optimization focused on worst shifts."""
    h = project_to_feasible(h.copy(), n)
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    v = np.zeros(n)
    beta = 0.8
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Get top top_m shifts
        top_idx = np.argsort(corr)[-top_m:][::-1]
        
        grad = np.zeros(n)
        for k in top_idx:
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, n + sh)
            if j0 < j1:
                grad[j0:j1] += (1.0 - h[j0-sh:j1-sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= h[j0b+sh:j1b+sh]
        grad *= dx / top_m  # average over top shifts
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > 500:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        
        h = h_new
    
    return best_h, best_c5

- result[j] = h[i0]*(1-t) + h[i1]*t

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    frac = np.arange(target_n) * (n / float(target_n))
    i0 = np.floor(frac).astype(int) % n
    i1 = (i0 + 1) % n
    t = frac - np.floor(frac)
    return h[i0] * (1.0 - t) + h[i1] * t


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_grad_topm(h, n, tau, top_m=50):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    nc = len(corr)
    if top_m >= nc:
        top_idx = np.arange(nc)
    else:
        top_idx = np.argsort(corr)[-top_m:][::-1]
    shifted = tau * (corr[top_idx] - max_c)
    exp_t = np.exp(np.clip(shifted, -400, 400))
    s_t = exp_t.sum()
    p_t = exp_t / s_t
    grad = np.zeros(n)
    for i, k in enumerate(top_idx):
        pk = p_t[i]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, n + sh)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return max_c * dx, grad

    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.85, 0.999
    eps = 1e-8
    t_count = 0
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        _, grad = smooth_grad_topm(h, n, tau, top_m)
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        t_count += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t_count)
        v_hat = v / (1.0 - beta2 ** t_count)
        h_new = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        if stall > 150:
            lr *= 0.75
            m *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    return best_h, best_c5
