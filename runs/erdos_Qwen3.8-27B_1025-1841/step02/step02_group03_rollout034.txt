            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Deterministic + targeted again
    if remaining() > 100:
        h5, c5_5 = _det_search(best_h, n, k_arr, rng, t0, budget_s, max_rounds=40, n_trials=3000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
        if remaining() > 60:
            h5b, c5_5b = _targeted_search(best_h, n, k_arr, rng, t0, budget_s, max_rounds=30)
            if c5_5b < best_c5:
                best_c5 = c5_5b
                best_h = h5b.copy()

    # Phase 6: SLSQP polish
    if remaining() > 80:
        h6, c5_6 = _slsqp_polish(best_h, n, t0, budget_s)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Try higher resolution
    if remaining() > 300:
        n2 = n * 2
        x_old = np.linspace(0.0, 2.0, n)
        x_new = np.linspace(0.0, 2.0, n2)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = _make_binary(h_up, n2)
        k_arr2 = np.arange(2 * n2 - 1) - (n2 - 1)
        h7, c5_7 = _sa_phase(h_up, n2, k_arr2, rng, t0, budget_s, T_start=4.0, T_end=0.01, min_remaining=100)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
        if remaining() > 80:
            h7b, c5_7b = _det_search(best_h, n2, k_arr2, rng, t0, budget_s, max_rounds=30, n_trials=2000)
            if c5_7b < best_c5:
                best_c5 = c5_7b
                best_h = h7b.copy()

    # Final validation
    best_c5 = _compute_c5(best_h)
    return best_h, best_c5, len(best_h)
