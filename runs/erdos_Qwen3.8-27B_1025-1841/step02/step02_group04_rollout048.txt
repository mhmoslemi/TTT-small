        mask = h_bin == 1
        order = np.argsort(h[mask])[::-1]
        h_bin[mask[order[:excess]]] = 0
    elif h_bin.sum() < target:
        deficit = target - h_bin.sum()
        mask = h_bin == 0
        order = np.argsort(h[mask])[::-1]
        h_bin[mask[order[:deficit]]] = 1
    return h_bin

def local_search_binary(h_bin, max_iters=500, patience=80):
    n = len(h_bin)
    h_bin = h_bin.copy()
    positions_ones = np.where(h_bin == 1)[0]
    positions_zeros = np.where(h_bin == 0)[0]
    ov = compute_overlaps(h_bin)
    curr_max = ov.max()
    best_c5 = float(curr_max)
    best_h = h_bin.copy()
    stall = 0
    for it in range(max_iters):
        improved = False
        ov = compute_overlaps(h_bin)
        curr_max = ov.max()
        active_shifts = np.where(ov > curr_max - 0.02)[0]
        for k_idx in active_shifts[:5]:
            k = k_idx - (n - 1)
            ones = np.where(h_bin == 1)[0]
            zeros = np.where(h_bin == 0)[0]
            best_improvement = 0
            best_pair = None
            for i in zeros:
                if i + k < 0 or i + k >= n:
                    continue
                if h_bin[i + k] != 1:
                    continue
                for j in ones:
                    if j == i:
                        continue
                    old_max = ov.max()
                    # compute new overlap at affected shifts
                    del_i = -h_bin[np.clip(i+k, 0, n-1)] if 0 <= i+k < n else 0.0
                    del_i2 = (1 - h_bin[i-k]) if 0 <= i-k < n else 0.0
                    del_j = h_bin[np.clip(j+k, 0, n-1)] if 0 <= j+k < n else 0.0
                    del_j2 = -(1 - h_bin[j-k]) if 0 <= j-k < n else 0.0
                    new_at_k = ov[k_idx] + del_i + del_i2 + del_j + del_j2
                    if new_at_k < curr_max - 1e-9 and (curr_max - new_at_k) > best_improvement:
                        best_improvement = curr_max - new_at_k
                        best_pair = (i, j)
            if best_pair is not None:
                i2, j2 = best_pair
                h_bin[i2] = 1.0
                h_bin[j2] = 0.0
                improved = True
                break
        if improved:
            stall = 0
            new_c5 = compute_c5(h_bin)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_bin.copy()
        else:
            stall += 1
            if stall > patience:
                break
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def elapsed():
        return time.time() - start_time

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None or len(h0) < 10:
        n = 320
        h0 = np.zeros(n)
        h0[:n//2] = 1.0
        h0 = make_feasible(h0, n)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    n0 = len(best_h)
    print(f"Initial: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Adam GD with soft-max
    if elapsed() < budget_s - 100:
        h1, c1 = gradient_descend_adam(best_h, n_iter=3000, T_start=60.0, T_end=0.3, lr=0.5, top_k=80)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        print(f"Phase1 Adam: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 2: Top-3 focus
    if elapsed() < budget_s - 100:
        h2, c2 = top_k_focus_descent(best_h, n_iter=2000, k_count=3, lr=0.3)
        if c2 < best_c5:
            best_c5, best_h = c2, h2.copy()
        print(f"Phase2 Top3: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 3: Binary local search
    if elapsed() < budget_s - 100:
        h_bin = binary_project(best_h)
        c_bin = compute_c5(h_bin)
        h_bin, c_bin2 = local_search_binary(h_bin, max_iters=400, patience=60)
        if c_bin2 < best_c5:
            best_c5, best_h = c_bin2, h_bin.astype(float).copy()
        print(f"Phase3 Binary: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 4: More Adam GD after binary
    if elapsed() < budget_s - 80:
        h4, c4 = gradient_descend_adam(best_h, n_iter=3000, T_start=30.0, T_end=0.2, lr=0.4, top_k=100)
        if c4 < best_c5:
            best_c5, best_h = c4, h4.copy()
        print(f"Phase4 Adam2: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 5: Try different resolutions
    for new_n in [160, 256, 480, 640]:
        if elapsed() > budget_s - 120:
            break
        h_up = upsample(best_h, new_n)
        c_up_start = compute_c5(h_up)
        h_r, c_r = gradient_descend_adam(h_up, n_iter=2000, T_start=40.0, T_end=0.2, lr=0.4, top_k=60)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()
        h_t, c_t = top_k_focus_descent(best_h if c_r < c_up_start else h_up, n_iter=1500, k_count=4, lr=0.25)
        if c_t < best_c5:
            best_c5, best_h = c_t, h_t.copy()
        print(f"Phase5 n={new_n}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 6: Multi-start perturbation
    if elapsed() < budget_s - 80:
        n_pts = len(best_h)
        for ps in [0.05, 0.12, 0.2]:
            if elapsed() > budget_s - 30:
                break
            h_p = best_h + rng.randn(n_pts) * ps
            h_p = make_feasible(h_p, n_pts)
            h_r, c_r = gradient_descend_adam(h_p, n_iter=2000, T_start=20.0, T_end=0.3, lr=0.4, top_k=60)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  Perturb {ps}: C5={best_c5:.6f}", flush=True)

    # Phase 7: Final fine-tuning
    if elapsed() < budget_s - 20:
        n_pts = len(best_h)
        h_f, c_f = top_k_focus_descent(best_h, n_iter=3000, k_count=3, lr=0.1)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
        h_g, c_g = gradient_descend_adam(best_h, n_iter=2000, T_start=2.0, T_end=0.05, lr=0.1, top_k=120)
        if c_g < best_c5:
            best_c5, best_h = c_g, h_g.copy()

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
