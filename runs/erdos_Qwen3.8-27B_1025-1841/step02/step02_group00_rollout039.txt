    target = n / 2.0

One more consideration: the stability of the softmax for large T. When T is large, the softmax becomes very peaked, and np.exp(T*(corr-mc)) will underflow for non-maximum entries. I should use the log-sum-exp trick:

log(Σ exp(T*(c_i - c_max))) = log_sum_exp(T*(c - c_max))

And the softmax weight w[s] = exp(T*(c[s]-c_max)) / Σ exp(T*(c'-c_max))

For numerical stability:
max_shift = T*(corr - mc)  # this has max = 0
w = exp(max_shift) / sum(exp(max_shift))

Since max_shift ≤ 0, exp(max_shift) ≤ 1, so no overflow. Underflow is fine (just gives 0 for very non-maximum entries).

def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break

            idx = np.argmin(h)
        else:
            idx = np.argmax(h)

    # For stability, shifted <= 0
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    smooth_val = mc + np.log(sum_exp) / T
    w = exp_shifted / sum_exp  # softmax weights, length 2n-1
    
    # Gradient
    b_rev = w[::-1]
    A = np.convolve(1.0 - h, b_rev, mode='full')[n-1:2*n-1]
    B = np.convolve(h, w, mode='full')[n-1:2*n-1]
    grad = dx * (A - B)
    return smooth_val, grad

def penalty_obj_grad(h, T, penalty):
    n = len(h)
    val, grad = smooth_max_and_grad(h, T)
    mass_viol = np.sum(h) - n / 2.0
    val += penalty * mass_viol ** 2
    grad += 2.0 * penalty * mass_viol
    return val, grad

def lbfgsb_optimize(h_start, n, T, penalty, max_iter=500, ftol=1e-16):
    def fun(h):
        return penalty_obj_grad(h, T, penalty)[0]
    
    def fun_and_grad(h):
        v, g = penalty_obj_grad(h, T, penalty)
        return v, g
    
    bounds = [(0.0, 1.0)] * n
    res = minimize(fun_and_grad, h_start, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': max_iter, 'ftol': ftol, 'gtol': 1e-10, 'maxls': 50, 'maxfun': max_iter * 2})
    return make_feasible(res.x, n)

def coordinate_descent(h, n, max_rounds=5, tol=1e-10):
    """Coordinate descent with golden section search on each coordinate."""
    best = compute_c5(h)
    h = h.copy()
    
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            # Golden section search on h[j] in [0, 1]
            lo, hi = 0.0, 1.0
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            a = hi - gr * (hi - lo)
            b = lo + gr * (hi - lo)
            fa = compute_c5_with_update(h, j, a)
            fb = compute_c5_with_update(h, j, b)
            for _ in range(40):
                if fa < fb:
                    hi = b
                    b = a
                    fb = fa
                    a = hi - gr * (hi - lo)
                    fa = compute_c5_with_update(h, j, a)
                else:
                    lo = a
                    a = b
                    fa = fb
                    b = lo + gr * (hi - lo)
                    fb = compute_c5_with_update(h, j, b)
            best_j = (lo + hi) / 2.0
            # Also check endpoints
            candidates = [0.0, 1.0, best_j, h[j]]
            best_c = compute_c5(h)
            best_val = h[j]
            for cv in candidates:
                h[j] = cv
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_val = cv
            h[j] = best_val
            h = make_feasible(h, n)
            if compute_c5(h) < best - tol:
                best = compute_c5(h)
                improved = True
        if not improved:
            break
    return h, best

def compute_c5_with_update(h, j, new_val):
    """Compute C5 after setting h[j] = new_val, without full recomputation."""
    # This is expensive if we recompute full correlation each time
    # For simplicity, just set and compute
    old = h[j]
    h[j] = new_val
    c = compute_c5(h)
    h[j] = old
    return c

    #   if j is second index: h[j-off]*(1-h[j])

    #   if j is first index: new_val*(1-h[j+off])
    #   if j is second index: h[j-off]*(1-new_val)

    # For corr[s] with off = s-(n-1):

    # Terms where j is the "second" index h[i+off]: h[j-off]*(-delta) for valid j-off
    
    max_corr = np.max(corr).copy()

    # Affected s range: 
    # As first index: s from n-1-j to 2n-2-j (n values)
    # As second index: s from j to n-1+j (n values)
    # Union: min(n-1-j, j) to max(2n-2-j, n-1+j)
    
    s_lo = min(n-1-j, j)
    s_hi = max(2n-2-j, n-1+j)
    s_hi = min(s_hi, 2*n-2)
    
    for s in range(s_lo, s_hi + 1):
        off = s - (n + 1)  # Wait, off = s - (n-1)
        off = s - (n - 1)
        change = 0.0
        # j as first index: j + off valid?

            change += delta * (1.0 - h[j + off])
        # j as second index: j - off valid?
        if 0 <= j - off < n:
            change += h[j - off] * (-delta)
        corr[s] += change
    
    result = np.max(corr) * (2.0 / n)
    
    # Revert
    for s in range(s_lo, s_hi + 1):
        off = s - (n - 1)
        change = 0.0
        if 0 <= j + off < n:
            change += delta * (1.0 - h[j + off])
        if 0 <= j - off < n:
            change += h[j - off] * (-delta)
        corr[s] -= change
    
    return result

def compute_c5(h):

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    mc = np.max(corr)
    shifted = (corr - mc) * T
    exp_shifted = np.exp(shifted)
    sum_exp = np.sum(exp_shifted)
    smooth_val = mc + np.log(sum_exp) / T
    w = exp_shifted / sum_exp
    b_rev = w[::-1]
    A = np.convolve(1.0 - h, b_rev, mode='full')[n-1:2*n-1]
    B = np.convolve(h, w, mode='full')[n-1:2*n-1]
    grad = dx * (A - B)
    return smooth_val, grad

    n = len(h)
    val, grad = smooth_max_and_grad(h, T)
    mv = np.sum(h) - n / 2.0
    val += penalty * mv * mv
    grad = grad + 2.0 * penalty * mv
    return val, grad

    dx = 2.0 / n
    best = np.max(corr) * dx
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for rnd in range(max_rounds):
        improved = False
        for j in range(n):
            lo, hi = 0.0, 1.0
            for _ in range(gs_iters):
                a = hi - gr * (hi - lo)
                b = lo + gr * (hi - lo)
                fa = cd_eval(h, corr, j, a, n)
                fb = cd_eval(h, corr, j, b, n)
                if fa <= fb:
                    hi = b
                else:
                    lo = a
            opt = (lo + hi) / 2.0
            candidates = [0.0, 1.0, opt, h[j]]
            best_c = best
            best_val = h[j]
            for cv in candidates:
                if abs(cv - h[j]) < 1e-12:
                    continue
                c = cd_eval(h, corr, j, cv, n)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cv
            if best_val != h[j]:
                cd_apply(h, corr, j, best_val, n)
                h = make_feasible(h, n)
                corr = np.correlate(h, 1.0 - h, mode='full')
                best = best_c
                improved = True
        if not improved:
            break
    return h, best

def beatty_init(n, alpha):
    x = np.arange(n) * alpha
    h = (x - np.floor(x) < 0.5).astype(float)
    return make_feasible(h, n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    best_c5 = float('inf')
    best_h = None
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    
    N = 1024
    
    if h0 is not None:
        h_start = upsample_h(h0, N)
    else:
        h_start = beatty_init(N, 0.6180339887)
    
    best_h = h_start.copy()
    best_c5 = compute_c5(h_start)
    
    # L-BFGS-B with annealing
    schedule = [(10.0, 50.0, 300), (30.0, 200.0, 300), (100.0, 1000.0, 400), (300.0, 5000.0, 400), (1000.0, 20000.0, 500)]
    for T, pen, iters in schedule:
        if time_left() < 30:
            break
        h_try = lbfgsb_opt(h_start, N, T, pen, iters)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            h_start = h_try.copy()
        else:
            h_start = h_try.copy()
    
    # Coordinate descent polish
    if time_left() > 60:
        h_cd, c5_cd = coordinate_descent(h_start, N, max_rounds=5, gs_iters=35)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
            h_start = h_cd.copy()
    
    # Try other initializations
    alphas = [0.6180339887, 0.7071067811, 0.5 + 0.1 * np.pi, 0.314159265]
    for alpha in alphas:
        if time_left() < 80:
            break
        h_rand = beatty_init(N, alpha)
        h_blend = 0.5 * best_h + 0.5 * h_rand
        h_blend = make_feasible(h_blend, N)
        
        h_try = lbfgsb_opt(h_blend, N, 50.0, 500.0, 300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        
        if time_left() > 40:
            h_cd2, c5_cd2 = coordinate_descent(h_try, N, max_rounds=3, gs_iters=30)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2.copy()
    
    # Random restart
    if time_left() > 100:
        h_r = 0.6 * best_h + 0.4 * rng.rand(N)
        h_r = make_feasible(h_r, N)
        h_try = lbfgsb_opt(h_r, N, 30.0, 200.0, 300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        if time_left() > 50:
            h_cd3, c5_cd3 = coordinate_descent(h_try, N, max_rounds=3, gs_iters=30)
            if c5_cd3 < best_c5:
                best_c5 = c5_cd3
                best_h = h_cd3.copy()
    
    # Upsample to higher resolution and polish
    if time_left() > 150:
        N2 = 2048
        h_up = upsample_h(best_h, N2)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
        
        h_try = lbfgsb_opt(h_up, N2, 200.0, 5000.0, 500)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
        
        if time_left() > 80:
            h_cd4, c5_cd4 = coordinate_descent(h_try, N2, max_rounds=3, gs_iters=30)
            if c5_cd4 < best_c5:
                best_c5 = c5_cd4
                best_h = h_cd4.copy()
    
    # Final CD at original resolution if we ended up at higher res
    if len(best_h) != N and time_left() > 30:
        h_back = upsample_h(best_h, N)
        h_cd5, c5_cd5 = coordinate_descent(h_back, N, max_rounds=5, gs_iters=40)
        if c5_cd5 < best_c5:
            best_c5 = c5_cd5
            best_h = h_cd5.copy()
    
    # Final L-BFGS-B polish with very high T
    if time_left() > 20:
        n_fin = len(best_h)
        h_try = lbfgsb_opt(best_h, n_fin, 2000.0, 50000.0, 300)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
