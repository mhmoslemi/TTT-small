import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(55):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def made_gradient(h, weights, n, dx):
    n_corr = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_corr):
        wk = weights[k]
        if wk < 1e-18:
            continue
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += wk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= wk * h[idx + d]
    return grad * dx


def grade_optimize(h, n, tau_start, tau_end, max_iters, time_limit, lr_init=0.15, topM=0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(60, max_iters // 20)
    n_corr = 2 * n - 1

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        frac = it / max(max_iters - 1, 1)
        tau = tau_start + (tau_end - tau_start) * frac

        max_g = g.max()
        if topM > 0 and topM < n_corr:
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p_top = exp_c / s
            w = np.zeros(n_corr)
            w[top_idx] = p_top
            grad = made_gradient(h, w, n, dx)
        else:
            shifted = np.clip(tau * (g - max_g), -500, 500)
            exp_c = np.exp(shifted)
            s = exp_c.sum()
            p = exp_c / s
            grad = made_gradient(h, p, n, dx)

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        tl = min(remaining() * 0.3, 250.0)
        h_f, c5_f = coord_descent_polish(h_f, n_f, max_sweeps=2000, time_limit=tl)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Try higher resolution if time permits
    n4 = n_init * 8
    if remaining() > 100 and n4 <= 32000:
        h4 = interp_periodic(best_h, n4)
        h4 = project_to_feasible(h4, n4)
        tl = min(remaining() * 0.3, 200.0)
        h4, _ = grade_optimize(h4, n4, tau_start=30.0, tau_end=200.0,
                               max_iters=5000, time_limit=tl, lr_init=0.06, topM=150)
        c5_4 = compute_c5(h4)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 6: Final coordinate descent
    if remaining() > 30:
        n_final = len(best_h)
        h_final = best_h.copy()
        tl = min(remaining() * 0.4, 200.0)
        h_final, c5_final = coord_descent_polish(h_final, n_final, max_sweeps=3000, time_limit=tl)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final projection and check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
