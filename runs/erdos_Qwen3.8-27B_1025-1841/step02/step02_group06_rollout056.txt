```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def compute_smoothed_objective(h, tau):
    corr = compute_corr(h) * (2.0 / len(h))
    m = corr.max()
    shifted = np.clip(tau * (corr - m), -500, 500)
    return np.log(np.exp(shifted).sum()) / tau + m

def compute_smoothed_grad(h, tau, n, dx):
    corr = compute_corr(h)
    g = corr * dx
    m = g.max()
    shifted = np.clip(tau * (g - m), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad

def nesterov_optimize(h, n, tau, max_iters, time_limit, lr_init=0.1):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    h_prev = h.copy()
    corr = compute_corr(h)
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    mu = 0.9
    stall = 0
    patience = max(200, max_iters // 10)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        cur_mu = (1 + np.sqrt(1 + 4 * mu**2)) / 2
        h_y = h + (mu / cur_mu) * (h - h_prev)
        h_y = np.clip(h_y, 0, 1)
        h_y = project_to_feasible(h_y, n)
        
        grad = compute_smoothed_grad(h_y, tau, n, dx)
        h_new = project_to_feasible(h_y - lr * grad, n)
        
        corr = compute_corr(h_new)
        c5_cur = (corr * dx).max()
        if c5_cur < best_obj - 1e-15:
            best_obj = c5_cur
            best_h = h_new.copy()
            stall = 0
        else:
            stall += 1
        
        h_prev = h
        h = h_new
        
        if stall > patience:
            lr *= 0.7
            mu *= 0.95
            stall = 0
            if lr < 1e-12:
                break
    
    return best_h, best_obj
```

```python
def flip_effect(h, i0, n):
    """Return the change in all 2n-1 correlation values if h[i0] is flipped (0↔1)."""
    delta = 1.0 - 2.0 * h[i0]  # +1 if 0->1, -1 if 1->0
    change = np.zeros(2*n - 1)
    
    # h[i0] as first factor: delta * (1-h[i0+d]) for d where 0<=i0+d<n
    d_start1 = max(0 - i0, -(n-1))  # i0+d >= 0
    d_end1 = min(n-1-i0, n-1)       # i0+d <= n-1
    if d_start1 <= d_end1:
        d_vals = np.arange(d_start1, d_end1 + 1)
        k_vals = (n-1) - d_vals
        change[k_vals] += delta * (1.0 - h[i0 + d_vals])
    
    # h[i0] as second factor: h[i0-d] * (-delta) for d where 0<=i0-d<n
    # i0-d >= 0 → d <= i0
    # i0-d <= n-1 → d >= i0-(n-1)
    d_start2 = max(i0 - (n-1), -(n-1))
    d_end2 = min(i0, n-1)
    if d_start2 <= d_end2:
        d_vals = np.arange(d_start2, d_end2 + 1)
        k_vals = (n-1) - d_vals
        change[k_vals] += -delta * h[i0 - d_vals]
    
    return change
```

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_grad(h, tau, n, dx):
    corr = compute_corr(h)
    g = corr * dx
    m = g.max()
    shifted = np.clip(tau * (g - m), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        s = 1.0
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad


def nesterov_smooth(h, n, tau, max_iters, time_limit, lr=0.1):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    h_prev = h.copy()
    t = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    mu = 0.9
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_new = 1 + 0.75 * (1 + np.sqrt(1 + 4 * t * t))
        beta = (1 - t) / t_new
        h_y = h + beta * (h - h_prev)
        h_y = project_to_feasible(h_y, n)
        t = t_new
        
        grad = smooth_grad(h_y, tau, n, dx)
        h_new = project_to_feasible(h_y - lr / t * grad, n)
        
        c5_cur = compute_c5(h_new)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_new.copy()
        
        h_prev = h
        h = h_new
    
    return best_h, best_c5

def interp_periodic(h, target_n):

    if n == target_n:

def subgrad_polish(h, n, max_iters, time_limit, topK=3):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.005
    v = np.zeros(n)
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK
        
        cur_lr = lr / (1.0 + 0.01 * it)
        v = 0.85 * v - cur_lr * grad
        h = project_to_feasible(h + v, n)
        
        if stall > 300:
            lr *= 0.7
            v *= 0.5
            stall = 0
    
    return best_h, best_c5
```

Left: k = (n-1) - d, for d in [d_lo, d_hi], d ≠ 0
Right: k = (n-1) - d', for d' in [r_lo, r_hi], d' ≠ 0

Left range: d in [max(-i, 1-n), min(n-1-i, n-1)]
Right range: d in [max(i-n+1, 1-n), min(i, n-1)]

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_grad(h, tau, n, dx):
    corr = compute_corr(h)
    g = corr * dx
    m = g.max()
    shifted = np.clip(tau * (g - m), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        s = 1.0
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad


def nesterov_smooth(h, n, tau, max_iters, time_limit, lr=0.1):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()

    t_val = 1.0

    best_h = h.copy()
    start = time.time()
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t_new = 1 + 0.75 * (1 + np.sqrt(1 + 4 * t_val * t_val))
        beta = (1 - t_val) / t_new
        h_y = h + beta * (h - h_prev)
        h_y = project_to_feasible(h_y, n)
        t_val = t_new
        grad = smooth_grad(h_y, tau, n, dx)
        h_new = project_to_feasible(h_y - lr / t_val * grad, n)
        c5_cur = compute_c5(h_new)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_new.copy()
        h_prev = h
        h = h_new
    return best_h, best_c5
