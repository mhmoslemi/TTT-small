import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def circular_corr_max(h):
    """Compute max of circular cross-correlation of h and (1-h)."""
    # Use FFT for efficiency
    n = len(h)
    H = np.fft.rfft(h)
    G = np.fft.rfft(1.0 - h)
    # Circular correlation
    corr = np.fft.irfft(H * np.conj(G), n=n)
    return np.max(corr)

def compute_c5_fft(h):
    n = len(h)

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.arange(old_n) * (2.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def circular_corr(h):
    """Circular cross-correlation of h and (1-h)."""
    n = len(h)
    # corr[k] = sum_m h[m] * (1-h[(m+k)%n])
    # = sum_m h[m] - sum_m h[m]*h[(m+k)%n]
    # = sum(h) - autocorr_h[k]
    # autocorr_h[k] = sum_m h[m]*h[(m+k)%n] = IFFT(|FFT(h)|^2)[k] (with appropriate normalization)
    H = np.fft.rfft(h)
    auto = np.fft.irfft(H * np.conj(H), n=n)
    return np.sum(h) - auto

def swap_delta(h, corr, i, j, n):
    """Compute the change in correlation when swapping h[i] (1->0) and h[j] (0->1)."""
    ar = np.arange(n)
    # Effect of h[i]: 1->0
    # corr[k] changes by: -(1-h[i+k]) + h[i-k]
    delta_i = -(1.0 - h[(i + ar) % n]) + h[(i - ar) % n]
    # Effect of h[j]: 0->1
    # corr[k] changes by: +(1-h[j+k]) - h[j-k]
    delta_j = (1.0 - h[(j + ar) % n]) - h[(j - ar) % n]
    return delta_i + delta_j

def binary_local_search(h, n, max_iters=100000, rng=None, time_limit=None, start_time=None, best_c5=None):
    """Swap-based local search on binary h."""
    if rng is None:
        rng = np.random.RandomState(0)
    
    # Ensure binary
    h = (h >= 0.5).astype(float)
    h = make_feasible(h, n)
    
    # Get initial positions
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    corr = circular_corr(h)
    current_c5 = np.max(corr) * (2.0/n)
    best_h = h.copy()
    best_c5_val = current_c5
    
    # Temperature for simulated annealing
    T = (best_c5_val if best_c5_val is not None else current_c5) * 0.1
    T_min = 1e-6
    
    for it in range(max_iters):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        delta = swap_delta(h, corr, i, j, n)
        new_corr = corr + delta
        new_max = np.max(new_corr)
        new_c5 = new_max * (2.0/n)
        
        # Simulated annealing acceptance
        if new_c5 < current_c5 or (T > T_min and rng.rand() < np.exp(-(new_c5 - current_c5) / T)):
            # Accept
            h[i], h[j] = h[j], h[i]
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            corr = new_corr
            current_c5 = new_c5
            
            if current_c5 < best_c5_val:
                best_c5_val = current_c5
                best_h = h.copy()
