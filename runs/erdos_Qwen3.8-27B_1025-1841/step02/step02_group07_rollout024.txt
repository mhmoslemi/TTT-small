```
def project_bounded_simplex(h, S):
    n = len(h)
    # Start with no active bounds
    upper = np.ones(n)
    lower = np.zeros(n)
    
    # Water filling
    # We want to find t such that sum(clip(h - t, 0, 1)) = S
    # clip(h-t, 0, 1) = min(max(h-t, 0), 1)
    
    # Binary search on t
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.sum(np.clip(h - mid, 0, 1))
        if val > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    return np.clip(h - t, 0, 1)
```

```python
def evaluate_c5(h):

    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    h = np.array(h, dtype=float).copy()

    hi = h.max() + 1.0

        if val > target_sum:
            lo = mid
        else:

    theta = (lo + hi) / 2.0
    return np.clip(h - theta, 0.0, 1.0)

    """Compute subgradient of max_k corr[k]*dx at h"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    dk = k_star - (n - 1)

    lo2 = max(0, -dk)
    hi2 = min(n, n - dk)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2+dk:hi2+dk]
    
    return g * dx, float(np.max(corr) * dx)

            if best_f < best_c5:
                best_c5 = best_f
                best_h = h.copy()
        
        lr *= decay
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_bounded(h, target_sum):
    h = np.asarray(h, dtype=float)
    lo = h.min() - 1.0
    hi = h.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if val > target_sum:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    return np.clip(h - theta, 0.0, 1.0)

def compute_correction(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    dk = k_star - (n - 1)
    g = np.zeros(n)
    lo1 = max(0, dk)
    hi1 = min(n, dk + n)
    if lo1 < hi1:
        g[lo1:hi1] += (1.0 - h[lo1 - dk:hi1 - dk])
    lo2 = max(0, -dk)
    hi2 = min(n, n - dk)
    if lo2 < hi2:
        g[lo2:hi2] -= h[lo2 + dk:hi2 + dk]
    return g

def compute_smooth_correction(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    w = exp_s / np.sum(exp_s)
    active = np.where(w > 1e-10)[0]
    g = np.zeros(n)
    for k in active:
        dk = k - (n - 1)
        wdk = w[k]
        lo1 = max(0, dk)
        hi1 = min(n, dk + n)
        if lo1 < hi1:
            g[lo1:hi1] += wdk * (1.0 - h[lo1 - dk:hi1 - dk])
        lo2 = max(0, -dk)
        hi2 = min(n, n - dk)
        if lo2 < hi2:
            g[lo2:hi2] -= wdk * h[lo2 + dk:hi2 + dk]
    return g

def subgrad_optimize(h, n, deadline, lr_init=3.0, max_iter=200000):
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = lr_init
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = compute_correction(h)
        improved = False
        for trial_lr in [lr, lr * 0.5, lr * 0.25, lr * 0.125]:
            h_trial = project_bounded(h - trial_lr * g, target)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:
                h = h_trial
                best_h = h.copy()
                best_c5 = c5_trial
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            lr *= 0.85
            if no_improve > 200:
                break
        else:
            lr = min(lr * 1.02, 20.0)
            no_improve = 0

    return best_h, best_c5

def smooth_optimize(h, n, T, deadline, lr_init=1.0, max_iter=10000):
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = lr_init
    no_improve = 0

    for it in range(max_iter):

            break

        improved = False

            h_trial = project_bounded(h - trial_lr * g, target)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < best_c5:

                best_h = h.copy()
                best_c5 = c5_trial
                improved = True
                no_improve = 0
                break

            lr *= 0.8
            if no_improve > 100:
                break
        else:
            lr = min(lr * 1.01, 10.0)

def slsqp_polish(h0, n, deadline, T=50.0):
    target = n / 2.0
    dx = 2.0 / n

    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        s = T * c * dx
        m = np.max(s)
        return float((m + np.log(np.sum(np.exp(s - m)))) / T * dx)

    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full')
        s = T * c * dx
        m = np.max(s)
        w = np.exp(s - m)
        w /= np.sum(w)
        active = np.where(w > 1e-10)[0]
        g = np.zeros(n)
        for k in active:
            dk = k - (n - 1)
            wdk = w[k]
            lo1 = max(0, dk)
            hi1 = min(n, dk + n)
            if lo1 < hi1:
                g[lo1:hi1] += wdk * (1.0 - x[lo1 - dk:hi1 - dk])
            lo2 = max(0, -dk)
            hi2 = min(n, n - dk)
            if lo2 < hi2:
                g[lo2:hi2] -= wdk * x[lo2 + dk:hi2 + dk]
        return g * dx * T / T

    try:
        res = minimize(obj, h0, jac=obj_grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                                     'jac': lambda x: np.ones(n)}],
                       options={'maxiter': 300, 'ftol': 1e-15})
        if time.time() < deadline and res.x is not None and np.all(np.isfinite(res.x)):
            return project_bounded(res.x, target)
    except Exception:
        pass
    return h0

def interp_resample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
```
