```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_value_and_weights(corr, T):
    """Compute soft-max value and weights for a correlation array."""
    scaled = corr * T
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    value = (m + np.log(sum_exp)) / T
    weights = exp_s / sum_exp
    return value, weights

def soft_max_grad(h, T):
    """Compute gradient of soft_max objective w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    value, weights = soft_max_value_and_weights(corr, T)
    
    grad = np.zeros(n)
    
    # For shift s >= 0 (correlation index s + n - 1):
    for s in range(n):
        w = weights[s + n - 1]
        if w < 1e-12:
            continue
        grad[s:]   += w * (1.0 - h[:n-s])
        grad[:n-s] -= w * h[s:]
    
    # For shift s < 0, let sp = -s > 0 (correlation index n - 1 - sp):
    for sp in range(1, n):
        w = weights[n - 1 - sp]
        if w < 1e-12:
            continue
        grad[:n-sp] += w * (1.0 - h[sp:])
        grad[sp:]   -= w * h[:n-sp]
    
    return grad, value

    # Better: use the penalty method with L-BFGS-B.

    def obj(x):
        val, _ = soft_max_value_and_weights(np.correlate(x, 1-x, mode='full') * (2.0/n), T)
        penalty = 1e6 * (np.sum(x) - target)**2

        val, _ = soft_max_value_and_weights(np.correlate(x, 1-x, mode='full') * (2.0/n), T)
        return val
    
    def g(x):
        grad, _ = soft_max_grad(x, T)
        return grad
    
    def c_eq(x):
        return np.sum(x) - target
    
    try:

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],

        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14 * n:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def sm_value_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    scaled = corr * T
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    value = (m + np.log(sum_exp)) / T
    w = exp_s / sum_exp
    grad = np.zeros(n)
    for s in range(n):
        wi = w[s + n - 1]
        if wi < 1e-10:
            continue
        grad[s:] += wi * (1.0 - h[:n - s])
        grad[:n - s] -= wi * h[s:]
    for sp in range(1, n):
        wi = w[n - 1 - sp]
        if wi < 1e-10:
            continue
        grad[:n - sp] += wi * (1.0 - h[sp:])
        grad[sp:] -= wi * h[:n - sp]
    return value, grad

def corr_grad_single(h, k_lag):
    """Gradient of a single correlation term (with dx factor)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if k_lag >= 0:
        s = k_lag
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] -= h[s:]
    else:
        sp = -k_lag
        grad[:n - sp] += (1.0 - h[sp:])
        grad[sp:] -= h[:n - sp]
    return grad * dx

def slsqp_step(h0, T, maxiter, deadline):
    n = len(h0)
    target = n / 2.0
    def f(x):
        v, _ = sm_value_grad(x, T)
        return v
    def g(x):
        _, gr = sm_value_grad(x, T)
        return gr
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-12, 'maxiter': maxiter, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def subgradient_polish(h, n_steps, deadline):
    n = len(h)
    h = h.copy()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = int(np.argmax(corr))
        k_lag = k_idx - (n - 1)
        grad = corr_grad_single(h, k_lag)
        lr = 0.02 * (1.0 - step / max(n_steps, 1))
        h_new = h - lr * grad
        h_new = project(h_new, n)
        if evaluate_c5(h_new) <= evaluate_c5(h) + 1e-12:
            h = h_new
        else:
            h_new2 = h - lr * 0.3 * grad
            h_new2 = project(h_new2, n)
            if evaluate_c5(h_new2) <= evaluate_c5(h) + 1e-12:
                h = h_new2
    return h

def interp_reshape(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return project(np.interp(xt, xf, h), n_to)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(h_start)

    # Phase 1: Anneal at original resolution
    temps_p1 = [300, 150, 80, 40, 20, 10, 5, 3, 2, 1, 0.5, 0.3]
    for T in temps_p1:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time()
        maxiter = min(250, max(30, int(remaining * 0.04)))
        h_new = slsqp_step(best_h, T, maxiter, deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 2: Subgradient polish
    if time.time() < deadline - 30:
        h_new = subgradient_polish(best_h, 200, deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 3: Multi-resolution
    for n_try in sorted([252, 504, 400], key=lambda x: abs(x - n0)):
        if time.time() > deadline - 50:
            break
        h_try = interp_reshape(best_h, len(best_h), n_try)
        for T in [50, 20, 10, 5, 2, 1, 0.5]:
            if time.time() > deadline - 25:
                break
            maxiter = min(200, max(20, int((deadline - time.time()) * 0.03)))
            h_new = slsqp_step(h_try, T, maxiter, deadline)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_try = h_new
        if time.time() < deadline - 20:
            h_try = subgradient_polish(h_try, 100, deadline)
            c5_new = evaluate_c5(h_try)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_try.copy()

    # Phase 4: Diverse random restarts
    n_best = len(best_h)
    for attempt in range(80):
        if time.time() > deadline - 15:
            break
        strat = attempt % 6
        if strat == 0:
            h_pert = best_h + np.random.randn(n_best) * 0.04
        elif strat == 1:
            h_pert = best_h + np.random.randn(n_best) * 0.10
        elif strat == 2:
            h_pert = best_h.copy()
            bl = max(1, n_best // 12)
            i1 = np.random.randint(0, n_best // 2)
            i2 = np.random.randint(n_best // 2, n_best - bl)
            tmp = h_pert[i1:i1+bl].copy()
            h_pert[i1:i1+bl] = h_pert[i2:i2+bl]
            h_pert[i2:i2+bl] = tmp
        elif strat == 3:
            h_pert = best_h.copy()
            bl = np.random.randint(max(2, n_best // 25), max(3, n_best // 6))
            i = np.random.randint(0, n_best - bl)
            h_pert[i:i+bl] = 1.0 - best_h[i:i+bl]
        elif strat == 4:
            ht = np.fft.rfft(best_h - np.mean(best_h))
            noise = np.random.randn(len(ht)) * 0.08
            ht_pert = ht + noise
            h_pert = np.fft.irfft(ht_pert, n=n_best) + np.mean(best_h)
        else:
            h_pert = best_h + np.convolve(np.random.randn(n_best) * 0.06,
                                          np.ones(7) / 7, mode='same')
        h_pert = project(h_pert, n_best)
        T_use = [5, 10, 3, 20, 5, 10][attempt % 6]
        h_new = slsqp_step(h_pert, T_use, min(150, max(10, int((deadline - time.time()) * 0.02))), deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Final subgradient + SLSQP polish
    if time.time() < deadline - 10:
        h_new = subgradient_polish(best_h, 300, deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    if time.time() < deadline - 5:
        h_new = slsqp_step(best_h, 0.5, 100, deadline)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
