    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # For numerical stability

    val = (mc + log_sum / T) * dx

    v2 = (idx - s >= 0) & (idx - s < n)
    if np.any(v2):

    grad *= dx
    return val, grad

    dx = 2.0 / n
    momentum = np.zeros(n)
    best_h = h.copy()

    # Polyak: need lower bound on optimal value
