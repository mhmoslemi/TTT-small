def compute_c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_idx = np.argmax(corr)
    c5 = corr[max_idx] * dx
    s = max_idx - (n - 1)  # the shift
    # gradient of corr at shift s w.r.t. h
    g = np.zeros(n)
    if s >= 0:
        # valid i: s to n-1
        # d/dh[j] of sum_{i=s}^{n-1} h[i](1-h[i-s])

        #   - h[j+s] for j in [0, n-1-s] (second term)
        for j in range(n):
            val = 0.0
            if j >= s:  # first term: h[j] appears as h[i] with i=j, need j-s in [0,n-1]
                # j-s >= 0 since j >= s
                val += (1.0 - h[j - s])
            if j < n - s:  # second term: h[j] appears as h[i-s] with i=j+s, need i in [s, n-1]
                # i = j+s, need s <= j+s <= n-1, i.e., j >= 0 and j <= n-1-s
                val -= h[j + s]
            g[j] = val
    else:

        # valid i: 0 to n-1+s = 0 to n-1-t

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_full(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0).astype(float)
    target = n / 2.0
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_sym(h_full, n):
    """Project to symmetric + feasible."""
    n_half = n // 2
    h_sym = 0.5 * (h_full + h_full[::-1])
    return project_full(h_sym, n)

def grad_at_shift(h, n, s):
    """Gradient of corr_s w.r.t. h, times dx."""
    dx = 2.0 / n
    n_full = 2 * n - 1
    g = np.zeros(n)

        for j in range(n):
            if j >= s:
                g[j] += (1.0 - h[j - s])
            if j < n - s:
                g[j] -= h[j + s]
    else:

        for j in range(n):
            if j <= n - 1 - t:
                g[j] += (1.0 - h[j + t])
            if j >= t:
                g[j] -= h[j - t]

def smooth_grad(h, n, T):
    """Smooth gradient using log-sum-exp."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T

    obj = (m + np.log(np.sum(w))) / T * dx

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_full(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift_fast(h, n, s):
    dx = 2.0 / n
    g = np.zeros(n)
    if s >= 0:
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        t = -s
        g[:n - t] += (1.0 - h[t:])
        g[t:] -= h[:n - t]
    return g * dx

def hard_obj_grad(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    s = k_max - (n - 1)
    obj = corr[k_max] * dx
    g = grad_shift_fast(h, n, s)
    return obj, g

def smooth_obj_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    log_sum = m + np.log(np.sum(exps))
    obj = log_sum / T * dx
    w = exps / np.sum(exps)
    g = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s = k - (n - 1)
        g += w[k] * grad_shift_fast(h, n, s)
    return obj, g

def expand_sym(h_r, n):
    nh = n // 2
    h_f = np.empty(n)
    h_f[:nh + 1] = h_r
    h_f[nh + 1:] = h_r[:nh][::-1]
    return h_f

def reduce_sym(h_f, n):
    nh = n // 2
    return h_f[:nh + 1].copy()

def reduce_grad(g_f, n):
    nh = n // 2
    g_r = np.empty(nh + 1)
    g_r[nh] = g_f[nh]
    g_r[:nh] = g_f[:nh] + g_f[nh:][::-1]
    return g_r

def project_red(h_r, n):
    nh = n // 2
    h_r = np.clip(h_r, 0.0, 1.0).astype(float)

            mask = h_r > 1e-14

    h_r = reduce_sym(h_f, n)

    def obj_fn(x):
        hf = expand_sym(project_red(x, n), n)
        return evaluate_c5(hf)

    except Exception:

        try_optimize(best_h, n3, time_alloc_3)

    # Random restarts at best resolution
    n_best = len(best_h)
    n_half_best = n_best // 2
    for attempt in range(40):
        if time.time() > deadline - 15:
            break
        remaining = deadline - time.time()
        if remaining < 30:
            break
        t_per = min(40, remaining * 0.15)

        # Perturb best
        hr = reduce_sym(best_h, n_best)
        noise_scale = 0.06 + 0.04 * (attempt % 3)
        hr_pert = hr + np.random.randn(n_half_best + 1) * noise_scale
        hr_pert = project_red(hr_pert, n_best)

        hr_p, c5_p = smooth_descend(hr_pert, n_best, 80.0, 2000, 0.006, min(deadline, time.time() + t_per))
        hr_p, c5_p = hard_descend(hr_p, n_best, 3000, 0.003, min(deadline, time.time() + t_per))
        hf_p = expand_sym(hr_p, n_best)
        hf_p = project_full(hf_p, n_best)
        c5_p = evaluate_c5(hf_p)
        if c5_p < best_c5:
            best_c5 = c5_p
            best_h = hf_p.copy()

    # Structured starts
    for sv in range(3):
        if time.time() > deadline - 30:
            break
        hs = structured_start(n_best, sv)
        hs = 0.5 * (hs + hs[::-1])
        hs = project_full(hs, n_best)
        hr_s = reduce_sym(hs, n_best)
        hr_s, _ = smooth_descend(hr_s, n_best, 30.0, 2000, 0.01, min(deadline, time.time() + 60))
        hr_s, _ = hard_descend(hr_s, n_best, 3000, 0.003, min(deadline, time.time() + 60))
        hf_s = expand_sym(hr_s, n_best)
        hf_s = project_full(hf_s, n_best)
        c5_s = evaluate_c5(hf_s)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = hf_s.copy()

    # Final SLSQP polish
    if time.time() < deadline - 10:
        best_h = slsqp_polish(best_h, len(best_h), deadline)
        best_c5 = evaluate_c5(best_h)

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
