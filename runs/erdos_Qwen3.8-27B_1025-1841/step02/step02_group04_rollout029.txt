import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max(corr, T):
    m = np.max(corr)
    return m + np.log(np.sum(np.exp((corr - m) * T))) / T

def smax_grad(corr, T):
    m = np.max(corr)
    shifted = corr - m
    exp_s = np.exp(shifted * T)
    weights = exp_s / np.sum(exp_s)
    return weights

def shift_grad(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def nesterov_grad_descent(h, n_iter=5000, T_start=50.0, T_end=1.0, lr_init=0.05, 
                           top_k=30, check_every=40):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_init * (1.0 - 0.5 * frac)
        
        # momentum position
        h_mom = h - 0.9 * v  # Nesterov look-ahead
        
        # Compute gradient at momentum position
        h_mom = np.clip(h_mom, 0.0, 1.0)
        h_mom = make_feasible(h_mom, n)
        corr = np.correlate(h_mom, 1.0 - h_mom, mode='full')
        weights = smax_grad(corr, T)
        
        # Only use top-k shifts for efficiency
        idx = np.argsort(weights)[-top_k:][::-1]
        grad = np.zeros(n)
        for k_idx in idx:
            w = weights[k_idx]
            if w < 1e-10:
                break
            shift = k_idx - (n - 1)
            grad += w * shift_grad(h_mom, shift)
        grad *= dx
        
        v = 0.9 * v - lr * grad
        h = h + v
        h = make_feasible(h, n)
        
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_max_weights(corr, T):
    m = np.max(corr)
    shifted = corr - m
    exp_s = np.exp(shifted * T)
    total = np.sum(exp_s)
    if total < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
        return w
    return exp_s / total

def shift_grad(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def obj_smooth_max(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    return (m + np.log(np.sum(np.exp((corr - m) * T))) / T) * dx

def con_sum_eq(x, n):
    return np.sum(x) - n / 2.0

def con_sum_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, T_val, maxiter=400):
    def obj(x):
        return obj_smooth_max(x, T_val, n)
    def con_f(x):
        return con_sum_eq(x, n)
    def con_j(x):
        return con_sum_jac(x, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
    res = minimize(obj, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
    if res.x is not None:
        return make_feasible(res.x, n)
    return h
