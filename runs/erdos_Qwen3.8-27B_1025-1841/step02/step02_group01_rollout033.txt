def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    # Pad for interpolation
    h_pad = np.concatenate([h[-1:], h, h[:1]])
    x_pad = np.concatenate([[old_x[0]-2.0/old_n], old_x, [old_x[-1]+2.0/old_n]])
    h_new = np.interp(new_x, x_pad, h_pad)
    return make_feasible(h_new, new_n)

def soft_max_obj(h, temp):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(temp * shifted))) / temp
    return soft * dx

def slsqp_run(h_init, n, temp, maxiter=800, ftol=1e-14):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        soft = mc + np.log(np.sum(np.exp(temp * shifted))) / temp
        return soft * dx
    
    def cfun(h):
        return np.sum(h) - target
    
    def cjac(h):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    
    try:
        res = minimize(obj, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})

        return h_init

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_pad = np.concatenate([h[-1:], h, h[:1]])
    x_pad = np.concatenate([[old_x[0] - 2.0/old_n], old_x, [old_x[-1] + 2.0/old_n]])
    h_new = np.interp(new_x, x_pad, h_pad)
    return make_feasible(h_new, new_n)

def slsqp_run(h_init, n, temp, maxiter=600, ftol=1e-14):
    dx = 2.0 / n
    target = n / 2.0

    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        soft = mc + np.log(np.sum(np.exp(temp * shifted))) / temp
        return soft * dx

    def cfun(h):
        return np.sum(h) - target

    def cjac(h):
        return np.ones(n)

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h_init, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h_init

    target = n / 2.0
    best_c5 = compute_c5(h)
    h = h.copy()
    idx = np.arange(n)

    while (time.time() - t0) < time_limit:
        improved = False
        for j in range(n):
            if (time.time() - t0) >= time_limit:
                return h, best_c5
            old_val = h[j]
            candidates = set()
            for d in [-0.15, -0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1, 0.15]:
                v = old_val + d
                if 0.0 <= v <= 1.0:
                    candidates.add(round(v, 6))
            candidates.add(round(0.0, 6))
            candidates.add(round(1.0, 6))
            candidates.add(round(old_val, 6))

            best_c = best_c5
            best_v = old_val
            for v in candidates:
                trial = h.copy()
                trial[j] = v
                diff = target - trial.sum()
                if abs(diff) > 1e-14:
                    if diff > 0:
                        mask = (trial < 1.0) & (idx != j)
                    else:
                        mask = (trial > 0.0) & (idx != j)
                    cnt = mask.sum()
                    if cnt > 0:
                        trial[mask] += diff / cnt
                    trial = np.clip(trial, 0.0, 1.0)
                c = compute_c5(trial)
                if c < best_c - 1e-13:
                    best_c = c
                    best_v = v

            if best_v != old_val:
                h[j] = best_v
                diff = target - h.sum()
                if abs(diff) > 1e-14:

                    if cnt > 0:

                    h = np.clip(h, 0.0, 1.0)
                if best_c < best_c5:

                    improved = True
