import time

    h = h + diff / n

        else:
            mask = h > 0.0
            cnt = np.sum(mask)
        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def binary_sa(h_init, n, max_iters, rng, T0=0.3, T1=0.01):
    """Simulated annealing on binary patterns with swaps"""
    h = h_init.copy().astype(float)
    # Make binary
    thresh = np.median(h)
    h[h > thresh] = 1.0
    h[h <= thresh] = 0.0
    # Fix count
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        excess = rng.choice(ones, len(ones) - target, replace=False)
        h[excess] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    elif len(ones) < target:
        deficit = rng.choice(zeros, target - len(ones), replace=False)
        h[deficit] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    curr_val = np.max(corr)
    best_val = curr_val
    best_h = h.copy()
    
    for it in range(max_iters):
        frac = it / max_iters
        T = T0 * (T1 / T0) ** frac
        if T < 0.001:
            T = 0.001
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        
        h[i], h[j] = h[j], h[i]
        corr_new = np.correlate(h, 1.0 - h, mode='full')

    dx = 2.0 / n
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()

        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac

        mc = np.max(corr)

            weights = np.zeros(len(corr))
            weights[k_max] = 1.0

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def downsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n, dtype=float)
    new_x = np.linspace(0, old_n - 1, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def binary_sa(h_init, n, max_iters, rng, T0=0.5, T1=0.005):
    h = h_init.copy().astype(float)
    h[h < 0.5] = 0.0
    h[h >= 0.5] = 1.0
    target = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        sel = rng.choice(ones, len(ones) - target, replace=False)
        h[sel] = 0
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    elif len(ones) < target:
        sel = rng.choice(zeros, target - len(ones), replace=False)
        h[sel] = 1
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
    dx = 2.0 / n
    curr = np.max(np.correlate(h, 1.0 - h, mode='full'))
    best = curr
    best_h = h.copy()
    one_count = len(ones)
    for it in range(max_iters):
        T = T0 * (T1 / T0) ** (it / max_iters)
        i = ones[rng.randint(one_count)]
        j = zeros[rng.randint(one_count)]
        h[i], h[j] = h[j], h[i]
        new = np.max(np.correlate(h, 1.0 - h, mode='full'))
        delta = new - curr
        if delta <= 0 or rng.random() < np.exp(-delta / T):
            curr = new
            ones[i] = j
            zeros[j] = i
            if new < best:
                best = new
                best_h = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
    return best_h, best * dx

def soft_gd(h_init, n, n_iter, T_start=40.0, T_end=1.0, lr=0.12, momentum=0.92, top_k=60):
    dx = 2.0 / n
    h = h_init.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.4 * frac)
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        ms = np.max(shifted)
        exp_s = np.exp(T * (shifted - ms))
        total = np.sum(exp_s)
        if total < 1e-300:
            idx = [int(np.argmax(corr))]
            weights = np.array([1.0])
        else:
            w_all = exp_s / total
            idx = np.argsort(w_all)[-top_k:][::-1]
            weights = w_all[idx]
        grad = np.zeros(n)
        for ki in range(len(idx)):
            w = weights[ki]
            if w < 1e-10:
                break
            shift = idx[ki] - (n - 1)
            if shift >= 0:
                grad[shift:] += w * (1.0 - h[:n - shift])
                grad[:n - shift] += w * (-h[shift:])
            else:
                s = -shift
                grad[:n - s] += w * (1.0 - h[s:])
                grad[s:] += w * (-h[:n - s])
        grad *= dx
        v = momentum * v + lr_t * grad
        h_new = make_feasible(h - v, n)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h = h_new
    return best_h, best_c5

def _opt_obj(h):
    T = _OPT['T']
    n = _OPT['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    ms = np.max(shifted)
    return (mc + np.log(np.sum(np.exp(T * (shifted - ms))) + np.exp(-T * ms)) - ms) * dx

def _opt_cons(h):
    return np.sum(h) - _OPT['n'] / 2.0

def _opt_jac(h):
    return np.ones(_OPT['n'])

_OPT = {}

def slsqp_opt(h, n, T, maxiter=400):
    _OPT['T'] = T
    _OPT['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _opt_cons, 'jac': _opt_jac}]
    try:
        res = minimize(_opt_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)
