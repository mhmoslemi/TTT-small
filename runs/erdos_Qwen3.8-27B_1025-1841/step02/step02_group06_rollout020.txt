            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]

def adam_project_optimize(h, n, time_limit, tau_start, tau_end, lr_init, topM, perturb_interval=500, perturb_scale=0.03):
    """Adam optimizer with projection, adaptive tau, and periodic perturbation"""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr(h)
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)  # First moment
    v = np.zeros(n)  # Second moment
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    stall = 0
    patience = 200
    
    while time.time() - start < time_limit:
        t += 1
        # Adaptive tau
        frac = min(1.0, t / max(1, int(time_limit * 2)))  # This isn't great
        # Better: based on iteration count relative to some target
        tau = tau_start * (tau_end / tau_start) ** min(1.0, t / 5000.0)
        
        # Learning rate with cosine decay
        lr_t = lr_init * 0.5 * (1 + np.cos(np.pi * min(1.0, t / 10000.0)))
        
        # Compute gradient
        corr = compute_corr(h)
        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        grad = softmax_gradient(h, corr, dx, tau, topM)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad ** 2
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, n)
        
        # Periodic perturbation
        if t % perturb_interval == 0:
            h_new = h_new + perturb_scale * np.random.randn(n)
            h_new = project_to_feasible(h_new, n)
            stall = 0
        
        if stall > patience:
            perturb_scale *= 1.5
            if perturb_scale > 0.5:
                break
            stall = 0
        
        h = h_new
    
    return best_h, best_obj

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

    return np.correlate(h, 1.0 - h, mode='full')


def compute_topm_gradient(h, corr, dx, tau, topM):
    n = len(h)
    g = corr * dx
    if topM > 0 and topM < (2 * n - 1):
        top_idx = np.argpartition(g, -topM)[-topM:]
        g_sel = g[top_idx]
    else:
        top_idx = np.arange(len(g))
        g_sel = g
    max_g = g_sel.max()
    shifted = np.clip(tau * (g_sel - max_g), -500, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        p = np.ones(len(g_sel)) / len(g_sel)
    else:
        p = exp_c / s
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p[i]
        if pk < 1e-12:
            continue
        k = top_idx[i]
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])

        n_final = len(best_h)
        tl_f = remaining() * 0.6
        h_f = momentum_optimize(best_h, n_final, tl_f,
                                tau_start=150, tau_end=500,
                                lr_init=0.02, topM=50, beta=0.9,
                                perturb_every=1000, perturb_scale=0.005)
        if h_f[1] < best_c5:
            best_c5 = h_f[1]
            best_h = h_f[0].copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
