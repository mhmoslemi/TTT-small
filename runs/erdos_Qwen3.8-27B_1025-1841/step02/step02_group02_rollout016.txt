import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def lse_grad(h, tau, top_k=300):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    cmax = corr.max()
    shifted = np.clip(tau * (corr - cmax), -500, 0)
    exps = np.exp(shifted)
    S = exps.sum()
    p = exps / S
    if top_k < len(p):
        top_idx = np.argsort(p)[-top_k:]
    else:
        top_idx = np.arange(len(p))
    grad = np.zeros(n)
    a = np.arange(n)
    for k in top_idx:
        pk = p[k]
        if pk < 1e-14:
            continue
        s = k - (n - 1)
        m1 = (a - s >= 0) & (a - s < n)
        m2 = (a + s >= 0) & (a + s < n)
        if m1.any():
            grad[m1] += pk * (1.0 - h[a[m1] - s])
        if m2.any():
            grad[m2] -= pk * h[a[m2] + s]
    grad *= dx
    obj = dx * (cmax + np.log(S) / tau)
    return obj, grad


def fista_opt(h, tau, time_limit, step=None, top_k=300):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    y = h.copy()
    t_param = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    if step is None:
        step = 1.0 / (tau * dx)
    start = time.time()
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        obj, grad = lse_grad(y, tau, top_k)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_param * t_param)) / 2.0
        x_new = project_to_feasible(y - step * grad, n)
        y = x_new + ((t_param - 1.0) / t_new) * (x_new - y)
        t_param = t_new
        c5 = compute_c5(x_new)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = x_new.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            break
    return best_h, best_c5


def subgrad_polish(h, time_limit, lr_init=0.01):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    a = np.arange(n)
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)

        grad = np.zeros(n)

            grad[m1] += (1.0 - h[a[m1] - s])
        if m2.any():
