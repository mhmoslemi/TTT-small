def project_to_feasible(g, n):
    """Project g onto {h : 0<=h<=1, sum(h)=n/2} via bisection on threshold."""
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix: adjust to make sum exactly n/2
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        # Add/subtract uniformly from non-boundary points
        mask = (result > 0) & (result < 1)
        if mask.any():
            result[mask] += diff / mask.sum()
            result = np.clip(result, 0, 1)
    return result

def compute_c5(h):

    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-13:
            continue
        sh = k - n + 1
        # grad[i] += pk * (1-h[i+sh]) for valid i
        i0 = max(0, -sh)
        i1 = min(n, n - sh)
        if i0 < i1:
            grad[i0:i1] += pk * (1.0 - h[i0+sh:i1+sh])
        # grad[i] -= pk * h[i-sh] for valid i
        i0b = max(0, sh)
        i1b = min(n, sh + n)
        if i0b < i1b:
            grad[i0b:i1b] -= pk * h[i0b-sh:i1b-sh]
    grad *= dx
    
    obj = dx * (max_c + np.log(exp_s.sum()) / tau)
    return obj, grad

def optimize_smooth(h, n, tau, time_limit, lr, momentum=0.9):
    h = project_to_feasible(h.copy(), n)

    best_h = h.copy()

def optimize_subgrad(h, n, time_limit, lr=0.003, momentum=0.85):

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    s = tau * (corr - max_c)
    exp_s = np.exp(s)
    p = exp_s / exp_s.sum()
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        pk = p[k]
        sh = k - n + 1
        i0 = max(0, -sh)
        i1 = min(n, n - sh)
        if i0 < i1:
            grad[i0:i1] += pk * (1.0 - h[i0 + sh:i1 + sh])
        i0b = max(0, sh)
        i1b = min(n, sh + n)
        if i0b < i1b:
            grad[i0b:i1b] -= pk * h[i0b - sh:i1b - sh]
    grad *= dx
    obj = dx * (max_c + np.log(exp_s.sum()) / tau)
    return obj, grad

        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
        c5 = compute_c5(h)

            best_h = h.copy()
            stall = 0

    return best_h, best_c5

        tl = min(50, remaining() * 0.4)
        h2, c5_2 = optimize_smooth(h2, n2, tau=10.0, time_limit=tl * 0.5,
                                   lr=0.05, momentum=0.92)
        if remaining() > 20:
            h2, c5_2 = optimize_subgrad_multi(h2, n2, time_limit=min(30, remaining() * 0.4),
                                              lr=0.002, momentum=0.85, top_k=3)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
