import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    for _ in range(300):
        d = target - np.sum(h)
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_at(h, n, tol_frac=0.005):
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    if max_val < 1e-12:
        return np.zeros(n), 1, max_val
    threshold = max_val - tol_frac * max_val
    active = np.where(corr >= threshold)[0]
    sg = np.zeros(n)
    for k in active:
        s = k - (n - 1)
        if s >= 0:
            sg[:n - s] += (1.0 - h[s:])
            sg[s:] -= h[:n - s]
        else:
            t = -s
            sg[t:] += (1.0 - h[:n - t])
            sg[:n - t] -= h[t:]
    sg /= len(active)
    return sg, len(active), max_val

def accel_subgrad(h0, n, wall_time, lr_scale=0.3):
    target = n / 2.0
    h = h0.astype(float).copy()
    v = np.zeros(n)
    best_h = h.copy()
    best_f = evaluate_c5(h)
    it = 0
    t_start = time.time()
    momentum = 0.9
    while time.time() - t_start < wall_time:
        beta = (1.0 + np.sqrt(1.0 + 4.0 * (it + 1) ** 2)) / 2.0
        z = h + (1.0 - (it + 1) / beta) * v
        z = np.clip(z, 0.0, 1.0)
        sg, na, mv = subgrad_at(z, n)
        lr = lr_scale / (np.sqrt(it + 1.0) + 1.0)
        h_new = z - lr * sg
        h_new = project_sum(h_new, target)
        f_new = evaluate_c5(h_new)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        v = h_new - h
        h = h_new
        it += 1
    return best_h, best_f

def slsqp_polish(h0, n, maxiter=300, T=300.0):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        m = np.max(c)
        return float(m + np.log(np.sum(np.exp(T * (c - m)))) / T)
    def con_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, target)
    except Exception:
        pass
    return h0

def interp_resample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2.0, n_from, endpoint=False)
    xt = np.linspace(0, 2.0, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12.0

    if 'initial_h_values' in globals() and initial_h_values is not None:

    # Phase 3: Multi-resolution subgradient + SLSQP

        resolutions.append(n0 * 2)

    seen = set()
    for n_r in resolutions:
        if n_r in seen or n_r < 14 or n_r > 672:

        seen.add(n_r)
        if time.time() > deadline - 60:

        elif n_r < len(best_h):

        else:
            h_work = interp_resample(best_h, len(best_h), n_r)

        t_rem = deadline - time.time() - 40

            best_h = h_sg2.copy()

        if time.time() < deadline - 40 and n_r <= 336:
            h_pol = slsqp_polish(h_sg2, n_r, maxiter=250, T=350.0)
            c5_pol = evaluate_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    # Phase 4: Random restarts at best resolution
    n_best = len(best_h)
    attempts = 0

        h_pert = best_h + np.random.randn(n_best) * 0.05
        h_pert = project_sum(h_pert, n_best / 2.0)

        if t_rem < 20:

    # Final polish

        c5_fin = evaluate_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
