from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0 - 1e-10
        else:
            mask = h > 1e-10
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    obj = (mc + np.log(total) / T) * dx
    weights = exp_s / total
    # Gradient: d/dh of (mc + log(sum exp(T*(corr_k - mc)))) * dx
    # = dx * sum_k weights_k * d(corr_k)/dh
    grad = np.zeros(n)
    full_n = len(corr)
    for k_idx in range(full_n):
        if weights[k_idx] < 1e-10:
            continue
        shift = k_idx - (n - 1)
        s = abs(shift)
        if s >= n:
            continue
        if shift >= 0:
            grad[s:] += weights[k_idx] * (1.0 - h[:n-s])
            grad[:n-s] += weights[k_idx] * (-h[s:])
        else:
            grad[:n-s] += weights[k_idx] * (1.0 - h[s:])
            grad[s:] += weights[k_idx] * (-h[:n-s])
    grad *= dx
    return obj, grad

def adam_softmax(h, n_iter, T_start, T_end, lr, beta1=0.9, beta2=0.999, eps=1e-8, check_every=20):
    n = len(h)
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        obj, grad = softmax_obj_and_grad(h, T)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def subgrad_descent(h, n_iter, lr, check_every=50):

def coordinate_descent(h, n_iter=200):
    """For each coordinate, try to optimally adjust it."""
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)

        improved = False
        for i in range(n):

                h[i] = cv
                c5 = compute_c5(h)
                if c5 < best_local:

                    best_val = cv

                improved = True
        c5 = compute_c5(h)

        if not improved:

    return h, best_c5

    dx = 2.0 / n
    def obj_and_grad(x):

        mc = np.max(corr)
        shifted = corr - mc
        exp_s = np.exp(T * shifted)

            shift = k_idx - (n - 1)

        return f, grad

    return x, compute_c5(x)

- Term where j=i: δ * (1 - h[i+s]) [if i+s is valid]
- Term where j+s=i, i.e., j=i-s: h[i-s] * (-δ) [if i-s is valid]

So Δcorr[s] = δ*(1-h[i+s]) - δ*h[i-s] (with appropriate boundary handling)

This means I can update all correlations in O(n) time per flip! So a full pass is O(n²) per iteration.

            hx, cx = slsqp_opt(h_up, T=20.0, maxiter=600)
            if cx < c_up:
                c_up, h_up = cx, hx
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up.copy()
        print(f"Res {new_n}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 7: Multi-start perturbation
    if remaining() > 120:
        n_pts = len(best_h)
        for pert in [0.03, 0.08, 0.15, 0.25]:
            if remaining() < 60:
                break
            h_p = best_h + rng.randn(n_pts) * pert
            h_p = make_feasible(h_p, n_pts)
            h_r, c_r = adam_optimize(h_p, 2500, T_start=300.0, T_end=0.3, lr=0.05, check_every=40)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  Perturb {pert}: C5={best_c5:.6f}", flush=True)

    # Phase 8: Final SLSQP
    n_pts = len(best_h)
    if n_pts <= 512 and remaining() > 60:
        hx, cx = slsqp_opt(best_h, T=8.0, maxiter=800)
        if cx < best_c5:
            best_c5, best_h = cx, hx
        print(f"FinalSLSQP: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 9: Final fine Adam
    if remaining() > 40:
        n_pts = len(best_h)
        h_f, c_f = adam_optimize(best_h, 3000, T_start=5.0, T_end=0.01, lr=0.01, check_every=20)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()

    # Phase 10: Second binary local search on final
    if remaining() > 30:
        h_bin2 = (best_h > 0.5).astype(float)
        h_bin2 = make_feasible(h_bin2, len(best_h))
        h_bs2, c_bs2 = binary_local_search(h_bin2, max_rounds=50)
        if c_bs2 < best_c5:
            best_c5, best_h = c_bs2, h_bs2.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
