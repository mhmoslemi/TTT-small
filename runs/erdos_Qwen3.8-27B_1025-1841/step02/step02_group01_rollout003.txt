def coord_descent(h, n, max_sweeps=30, time_limit=None, start_time=0):
    best = h.copy()
    c5 = compute_c5(best)
    half = n // 2

            break
        improved = False
        # Shuffle order
        order = np.random.permutation(n)
        for j in order:
            if time_limit is not None and time.time() - start_time > time_limit:
                break
            l = (j + half) % n
            if l == j:
                l = (j + 1) % n
            c_sum = best[j] + best[l]
            lo = max(0.0, c_sum - 1.0)
            hi = min(1.0, c_sum)
            if hi - lo < 1e-10:
                continue
            # Evaluate at several points
            n_conds = 15
            candidates = np.linspace(lo, hi, n_conds)
            best_v = best[j]
            best_c = c5
            for v in candidates:
                if abs(v - best[j]) < 1e-9:
                    continue
                trial = best.copy()
                trial[j] = v
                trial[l] = c_sum - v
                if trial[l] < -1e-12 or trial[l] > 1.0 + 1e-12:
                    continue
                tc = compute_c5(trial)
                if tc < best_c - 1e-13:
                    best_c = tc
                    best_v = v
            if best_v != best[j]:
                best[j] = best_v
                best[l] = c_sum - best_v
                c5 = best_c
                improved = True
        if not improved:
            break
    
    return best, c5

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:

        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:

        h[mask] += diff / cnt

def upsample(h, new_n):
