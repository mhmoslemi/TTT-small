    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx  # shape (2n-1,)
    max_f = f.max()
    exp_f = np.exp(tau * (f - max_f))

    obj = (max_f + np.log(exp_f.sum())) / tau  # = (1/tau) log sum exp(tau*f)

            h_r = best_h + ns * np.random.randn(n_r)
            h_r = project_to_feasible(h_r, n_r)
            tl = min(remaining() * 0.15, 50.0)
            h_r, _ = optimize_nesterov(h_r, n_r, 40.0, tl,
                                        topM=min(200, 2 * n_r - 1), lr_init=0.1)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

            # Also try 1-h of the perturbed solution
            h_rc = 1.0 - h_r
            h_rc = project_to_feasible(h_rc, n_r)
            c5_rc = compute_c5(h_rc)
            if c5_rc < best_c5:
                best_c5 = c5_rc
                best_h = h_rc.copy()

    # Phase 7: Final short Nesterov pass
    if remaining() > 30:
        n_final = len(best_h)
        tl = min(remaining() * 0.3, 60.0)
        h_fin, _ = optimize_nesterov(best_h, n_final, 60.0, tl,
                                      topM=min(100, 2 * n_final - 1), lr_init=0.05)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final subgradient micro-polish
    if remaining() > 15:
        n_final = len(best_h)
        h_fin, c5_fin = subgrad_polish(best_h, n_final, min(remaining() * 0.4, 40.0),
                                        topK=3, lr_init=0.001)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
