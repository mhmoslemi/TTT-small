import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def subgrad_for_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n-s] += (1.0 - h[s:])
        grad[s:] += -h[:n-s]
    else:
        grad[:n-s] += -h[s:]
        grad[s:] += (1.0 - h[:n-s])
    return grad

def softmax_obj_grad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    obj = (mc + np.log(max(total, 1e-300)) / T) * dx
    weights = exp_s / total
    if top_k < len(weights):
        top_idx = np.argsort(weights)[-top_k:]
    else:
        top_idx = np.arange(len(weights))
    grad = np.zeros(n)
    for ki in top_idx:
        w = weights[ki]
        if w < 1e-10:
            continue
        shift = ki - (n - 1)
        grad += w * subgrad_for_shift(h, shift)
    grad *= dx
    return obj, grad

def momentum_gd(h, n_iter, T_start, T_end, lr_start, lr_end, momentum=0.9, top_k=50, check_every=200):
    n = len(h)
    h = h.astype(float).copy()
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        mu = momentum * (0.9995 ** it)
        _, grad = softmax_obj_grad(h, T, top_k=top_k)
        v = mu * v - lr * grad
        h = make_feasible(h + v, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_grad(x, n, T, top_k=60):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    weights = exp_s / total
    if top_k < len(weights):
        top_idx = np.argsort(weights)[-top_k:]
    else:
        top_idx = np.arange(len(weights))
    grad = np.zeros(n)
    for ki in top_idx:
        w = weights[ki]
        if w < 1e-10:
            continue
        shift = ki - (n - 1)
        grad += w * subgrad_for_shift(x, shift)
    grad *= dx
    return grad

def _slsqp_cons_fun(x, n):
    return np.sum(x) - n / 2.0

def _slsqp_cons_jac(x, n):
    return np.ones(n)

def slsqp_optimize(h, n, T, maxiter=400):
    bounds = [(0.0, 1.0)] * n
    cfun = lambda x: _slsqp_cons_fun(x, n)
    cjac = lambda x: _slsqp_cons_jac(x, n)
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(_slsqp_obj, h, jac=lambda x: _slsqp_grad(x, n, T),
                      method='SLSQP', bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h

def coord_descent_fast(h, n_sweeps, T):

    dx = 2.0 / n
    h = h.astype(float).copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    best_h = h.copy()
    best_c5 = compute_c5(h)

    def _obj_from_corr(c):
        mc = np.max(c)
        sh = c - mc
        return mc + np.log(np.sum(np.exp(T * sh))) / T

    current_obj = _obj_from_corr(corr)

    for sweep in range(n_sweeps):
        improved = 0
        for i in range(n):
            p = n - 1 - i
            t_old = h[i]
            best_t = t_old
            best_obj = _obj_from_corr(corr)
            # Try candidates around current
            if t_old < 0.1:
                candidates = [0.0, 0.05, 0.1, 0.15, 0.2]
            elif t_old > 0.9:
                candidates = [1.0, 0.95, 0.9, 0.85, 0.8]
            else:
                candidates = [t_old - 0.15, t_old - 0.08, t_old - 0.03,
                             t_old + 0.03, t_old + 0.08, t_old + 0.15, 0.0, 1.0]
            for tc in candidates:
                tc = max(0.0, min(1.0, tc))

                delta = tc - t_old
                corr[p:p+n] += delta * (1.0 - 2.0 * h[p:p+n])
                new_obj = _obj_from_corr(corr)
                if new_obj < best_obj:
                    best_obj = new_obj
                    best_t = tc
                corr[p:p+n] -= delta * (1.0 - 2.0 * h[p:p+n])
            if abs(best_t - t_old) > 1e-8:
                delta = best_t - t_old
                corr[p:p+n] += delta * (1.0 - 2.0 * h[p:p+n])
                h[i] = best_t

                improved += 1
        if (sweep + 1) % 20 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:

            break

    c5 = compute_c5(h)
    if c5 < best_c5:

        best_h = h.copy()
