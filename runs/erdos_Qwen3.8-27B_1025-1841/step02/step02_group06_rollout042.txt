def project_to_feasible(g, n):
    # Project g onto {h : 0 ≤ h ≤ 1, sum(h) = n/2}
    # Using water-filling (sort-based) for speed
    target = n / 2.0
    # Binary search (simple and reliable)
    lo = g.min() - 1.0
    hi = g.max() + 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):

def softmax_gradient_step(h, corr, n, dx, tau, topK, lr, v, beta):
    """One step of softmax gradient descent with Nesterov momentum."""
    g = corr * dx
    if topK < len(g):
        # Use top-K lags
        top_idx = np.argpartition(g, -topK)[-topK:]
        g_top = g[top_idx]
    else:
        top_idx = np.arange(len(g))
        g_top = g
    
    max_g = g_top.max()
    shifted = np.clip(tau * (g_top - max_g), -500, 500)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    
    # Compute gradient
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p[i]
        if pk < 1e-10:

        k = top_idx[i]
        d = k - n + 1
        # First term: (1-h[j-d]) for valid j-d
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:

            grad[idx] += pk * (1.0 - h[idx - d])

    grad *= dx

    v = beta * v - lr * grad
