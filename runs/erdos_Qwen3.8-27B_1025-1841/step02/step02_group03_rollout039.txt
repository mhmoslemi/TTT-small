import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def obj_softmax(x, n, T):
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)

def obj_focused(x, active, n):
    corr = np.correlate(x, 1.0 - x, mode='full')
    return np.sum(corr[active]) * (2.0 / n)

def sum_con(x, n):
    return np.sum(x) - n / 2.0

def sum_jac(x, n):
    return np.ones(n)

def do_slsqp(h, n, T, maxiter=400, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_con, 'jac': sum_jac, 'args': (n,)}]
    try:
        res = minimize(obj_softmax, h, args=(n, T), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = make_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def do_focused_slsqp(h, n, active, maxiter=200, ftol=1e-14):

    try:
        res = minimize(obj_focused, h, args=(active, n), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        h_new = make_feasible(res.x, n)
        return h_new, compute_c5(h_new)
    except Exception:
        return h, compute_c5(h)

def binary_snap(h, n):
    best = h.copy()
    best_c = compute_c5(h)
    for j in range(n):
        for val in [0.0, 1.0]:
            if abs(val - best[j]) < 1e-10:
                continue
            trial = best.copy()
            trial[j] = val
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-14:
                best = trial
                best_c = c
    return best, best_c

def coord_descent(h, n, max_rounds=10, time_limit=60.0):
    start_t = time.time()
    best_c = compute_c5(h)
    for r in range(max_rounds):
        if time.time() - start_t > time_limit:
            break
        improved = False
        for j in range(n):
            old = h[j]
            best_trial = old
            best_c_local = best_c
            cands = [0.0, 1.0]
            for d in [0.02, 0.05, 0.1, 0.2, 0.3]:
                cands.append(max(0.0, min(1.0, old - d)))
                cands.append(max(0.0, min(1.0, old + d)))
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_local - 1e-14:
                    best_c_local = c
                    best_trial = cand
            if best_trial != old:
                h = h.copy()
                h[j] = best_trial
                h = make_feasible(h, n)
                best_c = best_c_local
                improved = True
        if not improved:
            break
    return h, best_c
