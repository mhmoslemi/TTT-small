import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def obj_and_grad(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    wsum = np.sum(exps)
    weights = exps / wsum
    obj = (m + np.log(wsum)) / T * dx
    grad = np.zeros(n)
    for lag in range(-(n - 1), n):
        wi = weights[lag + n - 1]
        if wi < 1e-14:
            continue
        if lag >= 0:
            grad[lag:] += wi * (1.0 - x[:n - lag])
            grad[:n - lag] -= wi * x[lag:n]
        else:
            l = -lag
            grad[:n - l] += wi * (1.0 - x[l:])
            grad[l:] -= wi * x[:n - l]
    grad *= dx
    return obj, grad

def project(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def opt_lbbox(h0, T, maxiter=600):
    n = len(h0)
    try:
        res = minimize(lambda x: obj_and_grad(x, T)[0], h0,
                       jac=lambda x: obj_and_grad(x, T)[1],
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'gtol': 1e-9, 'maxls': 50})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def proj_grad_fine(h0, T, n_steps, lr, deadline):
    n = len(h0)
    h = h0.copy()
    step_size = lr
    for step in range(n_steps):
        if deadline and time.time() > deadline:
            break
        _, g = obj_and_grad(h, T)
        h = h - step_size * g
        h = project(h, n)
        step_size *= 0.999
    return h

    t0 = time.time()

    if 'initial_h_values' in globals() and initial_h_values is not None:

        h_start = np.full(168, 0.5)

    n_start = len(h_start)
    best_h = project(h_start, n_start)
    best_c5 = evaluate_c5(best_h)

    n_cur = n_start
    # Phase 1: Anneal T from low to high
    for T in [5.0, 20.0, 80.0, 300.0, 1000.0]:
        if time.time() > deadline - 120:

        h_new = opt_lbbox(best_h, T, maxiter=1000)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:

    # Phase 2: Modulated restarts at current resolution

    # Phase 3: Structured starts

    # Phase 5: Fine-tune with projected gradient at high T

        h_fine = proj_grad_fine(best_h, 2000.0, int((deadline - time.time() - 10) * 8), 0.003, deadline)

            best_h = h_fine.copy()

    # Phase 6: One more SLSQP-style pass (L-BFGS-B on penalized problem)

        try:
            res = minimize(lambda x: obj_and_grad(x, 800.0)[0] + mu * (np.sum(x) - target) ** 2,

                c5 = evaluate_c5(h_new)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()

            pass

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
