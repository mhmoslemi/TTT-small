def direct_max_opt(h0, n, n_iters=100, deadline=None):
    """
    Direct approach: find the active lag, minimize it, repeat.
    """
    h = h0.copy()
    dx = 2.0 / n
    
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_vals = corr * dx
        kstar = np.argmax(c5_vals)
        
        # Compute gradient of c(k*) with respect to h
        # c[k] = sum_i h[i] * (1-h[i+k-(n-1)]) ... 
        # Let me use the formula derived earlier
        # For the 'full' correlate, index m corresponds to lag k = m - (n-1)
        k = kstar - (n - 1)  # actual lag
        
        # dc_k/dh_j for lag k:
        # h_j appears as h[n+k] when n = j-k: contributes (1-h[j-k]) if valid
        # h_j appears as h[n] when n = j: contributes -h[j+k] if valid
        grad = np.zeros(n)

            # First term: j ranges from k to n-1, contribution (1-h[j-k])
            for j in range(k, n):
                grad[j] += (1.0 - h[j-k])
            # Second term: j ranges from 0 to n-1-k, contribution -h[j+k]
            for j in range(0, n-k):
                grad[j] -= h[j+k]
        else:
            # k < 0, let k' = -k
            kprime = -k
            # c_{-k'} = sum_m h_m * (1-h_{m+k'})
            # First term (h_j as h_m): j from 0 to n-1-k', contribution (1-h[j+k'])
            for j in range(0, n - kprime):
                grad[j] += (1.0 - h[j+kprime])
            # Second term (h_j as h_{m+k'}): m = j-k', m from 0 to n-1-k', 
            # so j from k' to n-1, contribution -h[j-k']
            for j in range(kprime, n):
                grad[j] -= h[j-kprime]

        # Find step size via line search
        best_step = 1.0
        best_obj = evaluate_c5(h)

            h_try = project_to_feasible(h_try, n)
            obj = evaluate_c5(h_try)

                best_step = alpha

            for alpha in [0.001, 0.005, 0.002, 0.003, 0.0005]:
                h_try = h - alpha * grad
                h_try = project_to_feasible(h_try, n)
                obj = evaluate_c5(h_try)
                if obj < best_obj:
                    best_obj = obj
                    best_step = alpha

        h_new = project_to_feasible(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < evaluate_c5(h):
            h = h_new
        else:
            # Try with symmetry
            h_sym = apply_symmetry(h)
            h_sym = project_to_feasible(h_sym, n)
            h_new_sym = h_sym - best_step * 0.5 * grad
            h_new_sym = project_to_feasible(h_new_sym, n)
            c5_sym = evaluate_c5(h_new_sym)
            if c5_sym < evaluate_c5(h):
                h = h_new_sym
            # If nothing works, break
            if evaluate_c5(h) < best_obj * 1.001:
                break
    
    return h

def near_active_opt(h0, n, epsilon, maxiter=200, deadline=None):
    """
    Optimize using average gradient over near-active lags.
    """
    h = h0.copy()
    dx = 2.0 / n
    
    for it in range(maxiter):
        if deadline is not None and time.time() > deadline:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_vals = corr * dx
        cmax = np.max(c5_vals)
        
        # Find near-active lags
        active_threshold = cmax - epsilon
        active_mask = c5_vals >= active_threshold
        active_indices = np.where(active_mask)[0]
        
        if len(active_indices) == 0:
            active_indices = [np.argmax(c5_vals)]
        
        # Compute average gradient over active lags
        grad = np.zeros(n)
        for m in active_indices:
            k = m - (n - 1)
            g = np.zeros(n)
            if k >= 0:
                for j in range(k, n):
                    g[j] += (1.0 - h[j-k])
                for j in range(0, n-k):
                    g[j] -= h[j+k]
            else:
                kprime = -k
                for j in range(0, n - kprime):
                    g[j] += (1.0 - h[j+kprime])
                for j in range(kprime, n):
                    g[j] -= h[j-kprime]
            grad += g
        
        grad /= len(active_indices)
        
        # Line search
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        for alpha in [0.01, 0.05, 0.1, 0.05, 0.02, 0.005]:
            h_try = h - alpha * grad
            h_try = project_to_feasible(h_try, n)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
        
        h = best_h
    
    return h

    return float(np.max(corr) * dx)

def softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def project_to_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k, n):
    """Compute dc_k/dh for a specific lag k (where k = m - (n-1) for full correlate index m)"""
    grad = np.zeros(n)
    if k >= 0:
        # c_k = sum_{n=0}^{n-1-k} h[n+k] * (1-h[n])
        # dc_k/dh[j] = (1-h[j-k]) for j in [k, n-1] (first factor)
        #              - h[j+k] for j in [0, n-1-k] (second factor)
        j1 = np.arange(k, n)
        grad[j1] += (1.0 - h[j1 - k])
        j2 = np.arange(0, n - k)
        grad[j2] -= h[j2 + k]
    else:

        grad[j1] += (1.0 - h[j1 + kp])

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

def project_to_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lag_gradient(h, k, n):
    grad = np.zeros(n)
    if k >= 0:
        j1 = np.arange(k, n)
        grad[j1] += (1.0 - h[j1 - k])
        j2 = np.arange(0, n - k)
        grad[j2] -= h[j2 + k]
    else:
        kp = -k
        j1 = np.arange(0, n - kp)
        grad[j1] += (1.0 - h[j1 + kp])
        j2 = np.arange(kp, n)
        grad[j2] -= h[j2 - kp]
    return grad

def slsqp_opt(h0, n, T, maxiter=300, deadline=None):
    def f(x):
        return softmax_obj(x, n, T)
    def c_eq(x):
        return np.sum(x) - n / 2.0
    def jac_eq(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq, 'jac': jac_eq}],
                       options={'ftol': 1e-13, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_to_feasible(res.x, n)
    except Exception:
        pass
    return h0

def near_active_opt(h0, n, epsilon, maxiter, deadline, step_init=0.1):
    h = h0.copy()
    dx = 2.0 / n
    step = step_init
    stagnant = 0
    best_local = evaluate_c5(h)

    for it in range(maxiter):
        if deadline is not None and time.time() > deadline:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_vals = corr * dx
        cmax = np.max(c5_vals)

        active_mask = c5_vals >= (cmax - epsilon)
        active_indices = np.where(active_mask)[0]

        grad = np.zeros(n)
        for m in active_indices:
            k = m - (n - 1)
            grad += lag_gradient(h, k, n)
        grad /= len(active_indices)

        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break

        grad_normalized = grad / gnorm * gnorm

        improved = False
        for _ in range(25):
            h_try = h - step * grad
            h_try = project_to_feasible(h_try, n)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_local - 1e-14:
                best_local = c5_try
                h = h_try
                improved = True
                step = min(step * 1.3, 1.0)
                stagnant = 0
                break
            else:
                step *= 0.5
                if step < 1e-8:
                    break

        if not improved:
            stagnant += 1
            step = step_init
            if stagnant > 50:
                break

    return h

def symmetry_refine(h, n, deadline):
    h_sym = 0.5 * (h + h[::-1])
    h_sym = project_to_feasible(h_sym, n)
    c5_sym = evaluate_c5(h_sym)
    c5_orig = evaluate_c5(h)
    if c5_sym < c5_orig:
        return h_sym, c5_sym
    return h, c5_orig

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
