```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr_fft(h):
    n = len(h)
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    H = np.fft.rfft(h, N)
    V = np.fft.rfft((1.0 - h)[::-1], N)
    corr = np.fft.irfft(H * V, N)[:size]
    return corr

def compute_c5(h):
    n = len(h)
    corr = compute_corr_fft(h)
    return corr.max() * (2.0 / n)

def compute_gradient(h, p, n, dx):
    """Compute gradient of sum_k p[k] * corr[k] with respect to h.
    corr[k] = sum_j h[j+d] * (1-h[j]) where d = k-(n-1)
    ∂corr[k]/∂h[j] = (1-h[j-d]) * I(j in range1) - h[j+d] * I(j in range2)
    where range1: max(0,d) <= j <= min(n-1, n-1+d)
    range2: max(0,-d) <= j <= min(n-1, n-1-d)
    """
    grad = np.zeros(n)
    for k in range(len(p)):
        pk = p[k]
        if pk < 1e-15:
            continue
        d = k - n + 1
        # First term: (1-h[j-d]) for j in [max(0,d), min(n, n+d))
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0-d:j1-d])
        # Second term: -h[j+d] for j in [max(0,-d), min(n, n-d))
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b+d:j1b+d]
    return grad * dx
```

2. **Main optimization**: Adaptive-tau softmax gradient with Nesterov momentum
3. **Resolution refinement**: Gradually increase resolution
4. **Polish**: Subgradient method on top-K shifts
5. **Local search**: Random perturbations

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr_fft(h):
    n = len(h)
    size = 2 * n - 1
    N = 1
    while N < size:
        N *= 2
    H = np.fft.rfft(h, N)
    V = np.fft.rfft((1.0 - h)[::-1], N)
    return np.fft.irfft(H * V, N)[:size]

def compute_c5(h):
    n = len(h)
    return compute_corr_fft(h).max() * (2.0 / n)

def softmax_weights(corr, tau, topM=0):
    n = len(corr) // 2 + 1  # not needed, corr has 2n-1 elements
    g = corr * (2.0 / len(corr)//2 + 1)  # this is wrong
    # Actually, the "value" at shift k is corr[k] * dx where dx = 2/n
    # and n = (len(corr)+1)//2
    n = (len(corr) + 1) // 2
    dx = 2.0 / n
    g = corr * dx
    if topM > 0 and topM < len(g):
        top_idx = np.argpartition(g, -topM)[-topM:]
        max_g = g[top_idx].max()
        exp_top = np.exp(tau * (g[top_idx] - max_g))
        p = np.zeros(len(g))
        p[top_idx] = exp_top / exp_top.sum()
    else:
        max_g = g.max()
        exp_all = np.exp(np.clip(tau * (g - max_g), -500, 500))
        p = exp_all / exp_all.sum()
    return p

def compute_grad_from_weights(h, p, n):
    """Compute gradient given softmax weights p over correlation shifts."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(p)):
        pk = p[k]
        if pk < 1e-15:
            continue
        # c[k] = sum_n h[n+k]*(1-h[n]) for valid n
        # ∂c[k]/∂h[j] = (1-h[j-k])*I(max(0,k)<=j<min(n,n+k)) - h[j+k]*I(max(0,-k)<=j<min(n,n-k))
        j0 = max(0, k)
        j1 = min(n, n + k)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0-k:j1-k])
        j0b = max(0, -k)
        j1b = min(n, n - k)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b+k:j1b+k]
    return grad * dx

def optimize_softmax(h, n, max_iters, time_limit, tau_start, tau_end, topM=0, lr_init=0.1):
    """Main optimization loop with adaptive tau."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr_fft(h)
    best_c5 = corr.max() * dx
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    lr = lr_init
    stall = 0
    patience = max(50, max_iters // 10)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if (it + 1) % 10 == 0:
            corr = compute_corr_fft(h)
            cur_c5 = corr.max() * dx
            if cur_c5 < best_c5 - 1e-15:
                best_c5 = cur_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        
        # Adaptive tau
        progress = (it + 1) / max_iters
        tau = tau_start + (tau_end - tau_start) * progress
        
        corr = compute_corr_fft(h)
        p = softmax_weights(corr, tau, topM)
        grad = compute_grad_from_weights(h, p, n)
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        h = h_new
    
    # Final check
    corr = compute_corr_fft(h)
    cur_c5 = corr.max() * dx
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = h.copy()
    
    return best_h, best_c5
