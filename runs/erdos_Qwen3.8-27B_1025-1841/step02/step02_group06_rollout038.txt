def project_to_feasible(g, n):
    """Project g onto {h: 0≤h≤1, sum(h)=n/2}"""
    target = n / 2.0
    lo = g.min() - 1.0
    hi = g.max() + 1.0

        else:

    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_overlaps(h):
    """Compute all 2n-1 overlap values using FFT"""
    n = len(h)
    dx = 2.0 / n
    # np.correlate(h, 1-h, 'full') = convolve(h, (1-h)[::-1], 'full')
    corr = fftconvolve(h, (1.0 - h)[::-1])
    return corr * dx

def compute_c5(h):
    return compute_overlaps(h).max()

def vectorized_gradient(h, w_active, shifts, n, dx):
    """Compute gradient contribution from active shifts.
    w_active: array of weights for the active shifts
    shifts: array of shift values d (can be negative)
    """
    grad = np.zeros(n)
    for i in range(len(w_active)):
        d = shifts[i]
        w = w_active[i]
        if abs(w) < 1e-15:
            continue
        if d >= 0:
            # (1-h[i+d]) for i in [0, n-d-1]
            grad[:n-d] += w * (1.0 - h[d:])
            # -h[i-d] for i in [d, n-1]
            grad[d:] -= w * h[:n-d]
        else:
            dp = -d  # positive
            # (1-h[i-dp]) for i in [dp, n-1]
            grad[dp:] += w * (1.0 - h[:n-dp])
            # -h[i+dp] for i in [0, n-dp-1]
            grad[:n-dp] -= w * h[dp:]
    return grad * dx

def optimize_level(h, n, tau, max_iters, time_limit, lr_init=0.1, M_max=200, seed=42):
    """Main optimization loop at a given resolution"""
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    ov = compute_overlaps(h)
    best_obj = ov.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(200, max_iters // 10)
    it = 0
    
    while it < max_iters:
        if time.time() - start > time_limit:
            break
        it += 1
        
        ov = compute_overlaps(h)
        cur_max = ov.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Softmax weights over top-M shifts
        M = min(M_max, 2 * n - 1)
        # Get top M shifts
        top_idx = np.argpartition(ov, -M)[-M:]
        ov_top = ov[top_idx]
        max_ov = ov_top.max()
        shifted = tau * (ov_top - max_ov)
        shifted = np.clip(shifted, -500, 500)
        exp_vals = np.exp(shifted)
        p = exp_vals / exp_vals.sum()
        
        # Compute gradient
        shifts = top_idx - (n - 1)
        grad = vectorized_gradient(h, p, shifts, n, dx)
        
        # Heavy ball update
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        # Learning rate adaptation
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        # Slowly increase beta (momentum)
        if it < patience:
            beta = 0.9 + 0.05 * (it / patience)
        
        h = h_new
    
    return best_h, best_obj
