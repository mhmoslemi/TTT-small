So `np.correlate(a, v, 'full')[k]` = sum over valid i of `a[i + k - (len(v)-1)] * v[i]`

For same-length arrays (both length n):
`corr[k] = sum_{i: 0<=i<n, 0<=i+k-(n-1)<n} h[i+k-(n-1)] * (1-h[i])`

The index k ranges from 0 to 2n-2.
- k=0: only i=0 contributes: h[0] * (1-h[0])
- k=n-1: all i contribute: sum_i h[i] * (1-h[i]) = n*0.5 - sum h[i]^2 (if mean is 0.5)
- k=2n-2: only i=n-1 contributes: h[n-1] * (1-h[n-1])

So the "center" of the correlation (k=n-1) gives sum h[i](1-h[i]), which is the "self-overlap" (shift 0 in the continuous problem).

The shifts in the continuous problem correspond to:
- Continuous shift k correspond to discrete shift of k/dx samples
- For k=0: corr[n-1] = sum h[i](1-h[i])
- For k=dx (1 sample shift): corr[n] or corr[n-2] (depending on direction)

Actually, the problem says max over all k of the integral. In the discrete setting, we just take max over all correlation values. The "full" correlation includes all possible shifts from -(n-1) to +(n-1), which in the continuous problem corresponds to shifts from -(n-1)*dx to +(n-1)*dx. But the function is only defined on [0,2], so really the shifts should be from -2 to 2, i.e., from -(n-1) to +(n-1) in discrete units. So the full correlation is exactly what we want.

One more idea: instead of pure L-BFGS-B, I can use the **analytic gradient** of the soft-max objective with respect to g. Since h is a linear function of g (in the symmetric parameterization), the gradient is:

d(f)/dg[i] = sum_j (d(f)/dh[j]) * (dh[j]/dg[i])

The Jacobian dh/dg is simple:
- dh[j]/dg[i] = 1 if j == i
- dh[j]/dg[i] = -1 if j == m-1-i (from h[m:2m] = 1-g[::-1])
- dh[j]/dg[i] = -1 if j == 2m+i (from h[2m:3m] = 1-g)
- dh[j]/dg[i] = -1 if j == 3m+m-1-i = 4m-1-i (from h[3m:] = g[::-1])

Wait: h[3m:4m] = g[::-1], so h[3m+j] = g[m-1-j]. So dh[3m+j]/dg[i] = 1 if m-1-j = i, i.e., j = m-1-i.

So:
dh[j]/dg[i] = 1 if j == i OR j == 3m + m-1-i = 4m-1-i
dh[j]/dg[i] = -1 if j == m-1-i OR j == 2m+i
dh[j]/dg[i] = 0 otherwise

This is a sparse Jacobian with 4 non-zero entries per column.

So the gradient wrt g is:
dg[i] = grad_h[i] + grad_h[4m-1-i] - grad_h[m-1-i] - grad_h[2m+i]

where grad_h = d(soft_obj)/dh.

And grad_h can be computed as:
grad_h = sum_k weights[k] * d(corr[k])/dh

This gives us an analytic gradient! Let me implement this.

def soft_grad_h(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(np.clip(T * shifted, -500, 500))
    total = np.sum(exp_s)
    weights = exp_s / total
    
    grad_h = np.zeros(n)
    for k in range(2*n - 1):
        if weights[k] < 1e-12:
            continue
        i_lo = max(0, n - 1 - k)
        i_hi = min(n - 1, n - 1 + k)
        h_idx = np.arange(i_lo, i_hi + 1) + k - (n - 1)

        grad_h[h_idx] += w * (1.0 - h[i_lo:i_hi+1])
        grad_h[i_lo:i_hi+1] += w * (-h[h_idx])
    
    grad_h *= dx
    return grad_h

    m = len(g)
    n = 4 * m
    h = np.empty(n)
    h[:m] = g
    h[m:2*m] = 1.0 - g[::-1]
    h[2*m:3*m] = 1.0 - g
    h[3*m:] = g[::-1]
    return h

    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_grad_h(h, T):
    n = len(h)
    dx = 2.0 / n

def make_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_descent(h, n_iter=5000, lr_start=0.05, lr_end=0.001):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best = c5_of_h(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n-s] += (1.0 - h[s:])
            grad[s:] += (-h[:n-s])
        else:
            s2 = -s
            grad[:n-s2] += (-h[s2:])
            grad[s2:] += (1.0 - h[:n-s2])
        h = h - lr * grad * dx
        h = make_feasible(h, n)
        c = c5_of_h(h)
        if c < best:
            best = c
            best_h = h.copy()
    return best_h, best

from scipy.optimize import minimize

def build_h_sym(g):
    m = len(g)
    n = 4 * m
    h = np.empty(n)
    h[:m] = g
    h[m:2*m] = 1.0 - g[::-1]
    h[2*m:3*m] = 1.0 - g
    h[3*m:] = g[::-1]
    return h

def c5_of_h(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_obj_g(g, T):
    m = len(g)
    n = 4 * m
    dx = 2.0 / n
    h = build_h_sym(g)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(np.clip(T * shifted, -500, 500))))) / T * dx

def soft_grad_h(h, T, top_k=200):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

        w2 = np.zeros_like(weights)
        w2[idx] = weights[idx]

    grad_h = np.zeros(n)

        grad_h[h_idx] += w * (1.0 - h[i_lo:i_hi + 1])
        grad_h[i_lo:i_hi + 1] += w * (-h[h_idx])

def make_feasible(h, n):

    diff = target - np.sum(h)

        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_descent(h, n_iter, lr_start, lr_end):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best = c5_of_h(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = int(np.argmax(corr))
        s = k_star - (n - 1)
        grad = np.zeros(n)
        if s >= 0:
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] += (-h[:n - s])
        else:
            s2 = -s
            grad[:n - s2] += (-h[s2:])
            grad[s2:] += (1.0 - h[:n - s2])
        h = h - lr * grad * dx
        h = make_feasible(h, n)
        c = c5_of_h(h)
        if c < best:
            best = c
            best_h = h.copy()
    return best_h, best

def sym_optimize(m, g0, T_list, max_iter=400):
    best_g = g0.copy()

            cg = np.clip(res.x, 0, 1)
            vg = soft_obj_g(cg, T)
            if vg < best_val:

                best_g = cg

            pass
    return best_g
