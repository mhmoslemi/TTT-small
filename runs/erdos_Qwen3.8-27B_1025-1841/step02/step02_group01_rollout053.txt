            best_c5 = c5_2
            best_h = h2
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Phase 3: Upsample to n=640 and coordinate descent
    if remaining > 100:
        n2 = 640
        h_up = upsample_h(best_h, n2)
        t_limit3 = min(300, remaining * 0.45)
        h3, c5_3 = coord_descent(h_up, max_sweeps=200, time_limit=t_limit3,
                                 start_time=start_time, rng=rng)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Phase 4: SLSQP at higher resolution
    if remaining > 40:
        n_cur = len(best_h)
        t_limit4 = min(50, remaining * 0.2)
        h4, c5_4 = slsqp_refine(best_h, n_cur, max_topt=t_limit4,
                                 time_limit=t_limit4, start_time=start_time)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Phase 5: Try different initializations for diversity
    if remaining > 80:
        n_try = len(best_h)
        for attempt in range(3):
            if time.time() - start_time > budget_s - 30:
                break
            if attempt == 0:
                h_try = perturbed_h(best_h, 0.08, rng)
            elif attempt == 1:
                h_try = random_binary_h(n_try, rng)
            else:
                h_try = perturbed_h(best_h, 0.15, rng)
            t_lim = min(60, (budget_s - 30) - (time.time() - start_time))
            if t_lim < 15:
                break
            h_try, c5_try = coord_descent(h_try, max_sweeps=100,
                                          time_limit=t_lim, start_time=start_time, rng=rng)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Phase 6: Final coordinate descent polish at best resolution
    if remaining > 20:
        n_final = len(best_h)
        t_lim_final = min(remaining - 10, 120)
        h_final, c5_final = coord_descent(best_h, max_sweeps=300,
                                          time_limit=t_lim_final,
                                          start_time=start_time, rng=rng)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    
    # Phase 7: Try upsample to very high resolution if time
    if remaining > 60:
        n_hi = 960
        h_hi = upsample_h(best_h, n_hi)
        t_lim_hi = min(remaining - 10, 80)
        h_hi, c5_hi = coord_descent(h_hi, max_sweeps=100,
                                     time_limit=t_lim_hi, start_time=start_time, rng=rng)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi
    
    # Phase 8: Final SLSQP if time
    if time.time() - start_time < budget_s - 15:
        n_cur = len(best_h)
        try:
            dx = 2.0 / n_cur
            bounds = [(0.0, 1.0)] * n_cur
            target_sum = n_cur / 2.0
            def obj_final(v):
                return softmax_objective(v, n_cur, 200.0)
            def con_f(v):
                return np.sum(v) - target_sum
            def con_j(v):
                return np.ones(n_cur)
            cons = [{'type': 'eq', 'fun': con_f, 'jac': con_j}]
            res = minimize(obj_final, best_h.copy(), method='SLSQP',
                          bounds=bounds, constraints=cons,
                          options={'maxiter': 150, 'ftol': 1e-12})
            if hasattr(res, 'x'):
                hx = make_feasible(res.x, n_cur)
                c = compute_c5(hx)
                if c < best_c5:
                    best_c5 = c
                    best_h = hx
        except Exception:
            pass
    
    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)
    
    return best_h, best_c5, len(best_h)
