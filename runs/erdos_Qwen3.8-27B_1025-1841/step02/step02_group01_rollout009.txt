            grad[m:] += -wi * h[:n-m]
        else:
            mm = -m
            grad[mm:] += wi * (1.0 - h[:n-mm])
            grad[:n-mm] += -wi * h[mm:]
    
    grad *= dx
    return val * dx, grad

    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project(h_new, new_n)

    val, grad = smooth_max_grad(h, n, top_k, temp)
    # Backtracking line search
    best_h = h.copy()
    best_val = val
    step = lr

        if v_try < best_val:
            best_h = h_try
            best_val = v_try
        step *= 0.5
    return best_h, best_val

    h_new = np.interp(new_x, old_x, h)
    return project_simplex(h_new, new_n)

def pgd_optimize(h, n, start_time, budget_s, max_iters, top_k, temps, lrs, rng):
    phases = len(temps)
    per_phase = max_iters // phases
    for ph in range(phases):
        temp = temps[ph]
        lr = lrs[ph]
        n_it = per_phase if ph < phases - 1 else max_iters - ph * per_phase
        for it in range(n_it):
            if time.time() - start_time > budget_s - 25:
                return h
            val, grad = smooth_obj_grad(h, n, top_k, temp)
            step = lr
            improved = False
            for ls in range(8):
                h_new = project_simplex(h - step * grad, n)
                val_new, _ = smooth_obj_grad(h_new, n, top_k, temp)
                if val_new < val:
                    h = h_new
                    val = val_new
                    improved = True
                    break
                step *= 0.5
            if not improved:
                pass
        if ph > 0 and ph % 3 == 0:
            h = project_simplex(h + rng.randn(n) * 0.015, n)
    return h

def coord_polish(h, n, max_sweeps, start_time, budget_s):
    for _ in range(max_sweeps):
        if time.time() - start_time > budget_s - 10:
            break
        improved = False
        for j in range(n):
            if time.time() - start_time > budget_s - 10:
                break
            old = h[j]
            cands = [old - 0.03, old - 0.01, old + 0.01, old + 0.03, 0.0, 1.0, 0.5]
            best_c5 = compute_c5(h)
            best_v = old
            for c in cands:
                if c < -1e-9 or c > 1 + 1e-9:
                    continue
                h[j] = c
                h = make_feasible(h, n)
                c5 = compute_c5(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_v = c
                    improved = True
            h[j] = best_v
            h = make_feasible(h, n)
        if not improved:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = None

    if h_init is None:
        n0 = 320
        h_init = np.full(n0, 0.5)
        x = np.linspace(0, 2, n0)
        h_init += 0.25 * np.sin(2 * np.pi * x) + 0.15 * np.sin(4 * np.pi * x + 0.5)
        h_init = make_feasible(h_init, n0)

    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: PGD at multiple resolutions
    plan = [
        # (n, max_iters, top_k, temps, lrs)
        (320, 3000, 8, [60, 40, 25, 15, 10, 7, 5, 4, 3, 2], [0.03, 0.025, 0.02, 0.015, 0.012, 0.008, 0.005, 0.004, 0.003, 0.002]),
        (480, 2500, 10, [50, 35, 22, 14, 9, 6, 4, 3, 2], [0.025, 0.02, 0.015, 0.012, 0.009, 0.006, 0.004, 0.003, 0.002]),
        (640, 2000, 12, [40, 28, 18, 12, 8, 5, 4, 3, 2], [0.02, 0.016, 0.012, 0.01, 0.007, 0.005, 0.003, 0.002, 0.001]),
        (960, 1500, 15, [35, 24, 15, 10, 7, 5, 3, 2], [0.015, 0.012, 0.009, 0.007, 0.005, 0.003, 0.002, 0.001]),
    ]

    for n_pts, max_it, tk, temps, lrs in plan:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 40
        if remaining < 50:
            break

        h_cur = upsample_h(best_h, n_pts)
        h_cur = pgd_optimize(h_cur, n_pts, start_time, budget_s - 40, max_it, tk, temps, lrs, rng)

        # Evaluate
        c5_cur = compute_c5(h_cur)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h_cur.copy()

        # Second pass with slightly perturbed start
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 40
        if remaining > 120:
            h_pert = project_simplex(h_cur + rng.randn(n_pts) * 0.02, n_pts)
            max_it2 = max(500, max_it // 3)
            h_pert = pgd_optimize(h_pert, n_pts, start_time, budget_s - 40, max_it2, tk, temps[2:], lrs[2:], rng)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()

    # Phase 2: Coordinate polish at moderate resolution
    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        n_pol = min(480, len(best_h))
        h_pol = upsample_h(best_h, n_pol)
        h_pol = coord_polish(h_pol, n_pol, 15, start_time, budget_s - 30)
        c5_pol = compute_c5(h_pol)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol.copy()

    # Phase 3: Fine PGD on best resolution found
    elapsed = time.time() - start_time
    if budget_s - elapsed > 60:
        n_fin = len(best_h)
        h_fin = best_h.copy()
        # Fine-tune with very low temperature
        fine_temps = [3, 2, 1.5, 1]
        fine_lrs = [0.002, 0.001, 0.0005, 0.0003]
        h_fin = pgd_optimize(h_fin, n_fin, start_time, budget_s - 20, 800, 5, fine_temps, fine_lrs, rng)
        c5_fin = compute_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Phase 4: Final coordinate micro-polish
    elapsed = time.time() - start_time
    if budget_s - elapsed > 20:
        n_mp = min(320, len(best_h))
        n_mp_final = len(best_h)
        h_mp = best_h.copy()
        # Only polish a subset of coordinates for speed at high n
        step = max(1, n_mp_final // 256)
        idxs = list(range(0, n_mp_final, step))
        for j in idxs:
            if time.time() - start_time > budget_s - 5:
                break
            old = h_mp[j]
            for c in [old - 0.01, old + 0.01, old - 0.005, old + 0.005]:
                if c < -1e-9 or c > 1 + 1e-9:
                    continue
                h_mp[j] = c
                h_mp = make_feasible(h_mp, n_mp_final)
                c5_t = compute_c5(h_mp)
                if c5_t < best_c5 - 1e-13:
                    best_c5 = c5_t
                else:
                    h_mp[j] = old
                    h_mp = make_feasible(h_mp, n_mp_final)
                    break
            else:
                h_mp = make_feasible(h_mp, n_mp_final)
                c5_t = compute_c5(h_mp)
                if c5_t < best_c5:
                    best_c5 = c5_t
        best_h = make_feasible(h_mp, n_mp_final)

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    best_h = make_feasible(best_h, len(best_h))

    return best_h, best_c5, len(best_h)
