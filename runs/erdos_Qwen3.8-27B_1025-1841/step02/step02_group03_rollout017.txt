def project_constraint(h, n):
    """Project h onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    # Final adjustment for numerical precision
    diff = target - np.sum(proj)
    if abs(diff) < 1e-10:
        return proj
    # Spread the difference among non-saturated entries
    mask = (proj > 1e-10) & (proj < 1 - 1e-10)
    if np.sum(mask) > 0:
        proj[mask] += diff / np.sum(mask)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def coord_optimal(h, j, n, top_k=20):
    """Find optimal h[j] to minimize max of top correlations"""
    # Compute current correlations
    v = 1.0 - h
    corr = np.correlate(h, v, mode='full')
    
    # For lag k, corr_k as a function of h[j] = t:
    # corr_k = (terms not involving h[j]) + t * a_k - t * b_k (approximately)
    # where a_k = coefficient of h[j] in the first factor
    # and b_k = coefficient of h[j] in the second factor (from 1-h[...])
    
    # More precisely:
    # corr_k = ∑_i h[i] * (1-h[i-shift_k])
    # h[j] appears as:
    # - h[i] when i=j: contributes t * (1-h[j-shift_k])
    # - h[i-shift_k] when i-shift_k=j, i.e., i=j+shift_k: contributes h[j+shift_k] * (1-t)
    
    # So: corr_k(t) = C_k + t*(1-h[j-shift_k]) - t*h[j+shift_k]
    #              = C_k + t*((1-h[j-shift_k]) - h[j+shift_k])
    # where C_k = corr_k - h[j]*((1-h[j-shift_k]) - h[j+shift_k])
    
    # We want to minimize max_k [C_k + t * D_k] over t ∈ [0,1]
    # where D_k = (1-h[j-shift_k]) - h[j+shift_k] (with appropriate bounds)
    
    # Get top correlations
    top_idx = np.argsort(corr)[-top_k:]
    
    # Compute C_k and D_k for each top correlation
    Cs = np.zeros(top_k)
    Ds = np.zeros(top_k)
    h_j = h[j]
    for idx, k in enumerate(top_idx):
        shift = k - (n - 1)
        a = 0.0  # coefficient from first factor: (1-h[j-shift])
        b = 0.0  # coefficient from second factor: -h[j+shift]
        if 0 <= j - shift < n:
            a = 1.0 - h[j - shift]
        if 0 <= j + shift < n:
            b = h[j + shift]
        D_k = a - b
        # C_k = corr_k - h[j] * D_k
        C_k = corr[k] - h_j * D_k
        Cs[idx] = C_k
        Ds[idx] = D_k
    
    # Now minimize max_i [C_i + t * D_i] for t in [0, 1]
    # This is a 1D problem. The max of linear functions is convex in t.
    # Find the minimum by checking where the "envelope" has its minimum.
    
    # Evaluate at t=0, t=1, and at intersection points
    candidates = [0.0, 1.0]
    
    # Find intersection points of pairs of lines
    for i in range(top_k):
        for l in range(i+1, top_k):
            if abs(Ds[i] - Ds[l]) < 1e-12:
                continue
            t_star = (Cs[l] - Cs[i]) / (Ds[i] - Ds[l])
            if 0.0 < t_star < 1.0:
                candidates.append(t_star)
    
    # Also add the current value
    candidates.append(h_j)
    
    # Evaluate the max at each candidate
    best_t = h_j
    best_val = np.max(Cs + h_j * Ds)
    for t in candidates:
        t = max(0.0, min(1.0, t))
        val = np.max(Cs + t * Ds)
        if val < best_val:
            best_val = val
            best_t = t
    
    return best_t

    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        n_pts = len(best_h)
        for r in range(2):
            remaining = budget_s - (time.time() - start_time)
            if remaining < 10:
                break
            improved = False
            for j in range(n_pts):
                remaining = budget_s - (time.time() - start_time)
                if remaining < 5:
                    break
                new_val = coord_minimax(best_h, j, n_pts, top_k=30)
                if abs(new_val - best_h[j]) > 1e-10:
                    trial = best_h.copy()
                    trial[j] = new_val
                    trial = make_feasible(trial, n_pts)
                    c = compute_c5(trial)
                    if c < best_c5 - 1e-15:
                        best_h = trial
                        best_c5 = c
                        improved = True
                        # Also try neighbors
                        for dv in [-0.005, 0.005]:
                            nv = new_val + dv
                            if 0 <= nv <= 1:
                                tr2 = best_h.copy()
                                tr2[j] = nv
                                tr2 = make_feasible(tr2, n_pts)
                                c2 = compute_c5(tr2)
                                if c2 < best_c5 - 1e-15:
                                    best_h = tr2
                                    best_c5 = c2
            if not improved:
                break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
