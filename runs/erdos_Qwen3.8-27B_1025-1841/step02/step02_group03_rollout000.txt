        best_h, best_c5 = coord_descent(best_h, n0, n_cd, rng)

    # Phase 3: Pair swap at initial resolution
    if remaining() > 150:
        n_ps = int(remaining() * 2)
        best_h, best_c5 = pair_swap_search(best_h, n0, n_ps, rng)

    # Phase 4: Try higher resolution
    if remaining() > 180:
        for n_pts in [1920, 2880]:
            if remaining() < 150:
                break
            h_res = upsample_h(best_h, n_pts)
            c_res = compute_c5(h_res)
            _GN = n_pts
            _GT = 200.0
            bounds = [(0.0, 1.0)] * n_pts
            cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
            try:
                res = minimize(_slsqp_obj, h_res, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 400, 'ftol': 1e-15})
                h_try = make_feasible(res.x, n_pts)
                c_try = compute_c5(h_try)
                if c_try < best_c5:
                    best_c5 = c_try
                    best_h = h_try
            except Exception:
                pass
            if remaining() > 120 and c_try < best_c5 + 1e-6:
                n_cd = min(1000, int(remaining() * 1.2))
                best_h, best_c5 = coord_descent(best_h, n_pts, n_cd, rng)
            if remaining() > 60 and c_try < best_c5 + 1e-6:
                n_ps = int(remaining() * 1.5)
                best_h, best_c5 = pair_swap_search(best_h, n_pts, n_ps, rng)

    # Phase 5: Final coordinate descent
    n_pts = len(best_h)
    if remaining() > 60:
        n_cd = min(5000, int(remaining() * 2))
        best_h, best_c5 = coord_descent(best_h, n_pts, n_cd, rng)

    # Phase 6: Final pair swaps
    if remaining() > 30:
        n_ps = int(remaining() * 3)
        best_h, best_c5 = pair_swap_search(best_h, n_pts, n_ps, rng)

    # Phase 7: Final SLSQP polish
    if remaining() > 50:
        _GN = n_pts
        _GT = 1000.0
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _sum_con, 'jac': _sum_jac}]
        try:
            res = minimize(_slsqp_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-16})
            h_try = make_feasible(res.x, n_pts)
            c_try = compute_c5(h_try)
            if c_try < best_c5:
                best_c5 = c_try
                best_h = h_try
        except Exception:
            pass

    # Phase 8: Final coordinate descent
    if remaining() > 20:
        n_cd = min(3000, int(remaining() * 2))
        best_h, best_c5 = coord_descent(best_h, n_pts, n_cd, rng)

    best_h = make_feasible(best_h, n_pts)
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, n_pts
