    c = np.correlate(h, v, mode='full')

    p /= np.sum(p)

    # grad[i] = dx * sum_k p[k] * (v[i + offset_k] - h[i - offset_k])

```python
def softmax_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p /= np.sum(p)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        if p[k] < 1e-15:
            continue
        offset = k - (n - 1)
        # Term A: (1 - h[i+offset]) for valid i: 0 <= i+offset < n
        i1s = max(0, -offset)
        i1e = min(n-1, n-1-offset)
        if i1s <= i1e:
            idx = np.arange(i1s, i1e+1)
            grad[i1s:i1e+1] += p[k] * (1.0 - h[i1s+offset:i1e+offset+1])
        # Term B: -h[i-offset] for valid i: 0 <= i-offset < n
        i2s = max(0, offset)
        i2e = min(n-1, n-1+offset)
        if i2s <= i2e:
            grad[i2s:i2e+1] -= p[k] * h[i2s-offset:i2e-offset+1]
    
    grad *= dx
    return grad
```

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr * dx
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def compute_grad(h, T, eps=1e-6):
    n_local = len(h)
    f0 = softmax_obj(h, T)
    g = np.zeros(n_local)
    for i in range(n_local):
        if h[i] + eps <= 1.0:
            h_p = h.copy()
            h_p[i] += eps
            g[i] = (softmax_obj(h_p, T) - f0) / eps
        elif h[i] - eps >= 0.0:
            h_m = h.copy()
            h_m[i] -= eps
            g[i] = (f0 - softmax_obj(h_m, T)) / eps
    return g

def adam_loop(h0, T, lr, max_steps, deadline, check_interval=50):
    n = len(h0)
    h = h0.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    b1, b2, eps_a = 0.9, 0.999, 1e-8
    
    for t in range(1, max_steps + 1):
        if time.time() > deadline:
            break
        grad = compute_grad(h, T, eps=1e-7)
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_a)
        h_new = project(h_new, n)
        
        if t % check_interval == 0:
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
        h = h_new
    
    # Final check
    c5 = evaluate_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5
