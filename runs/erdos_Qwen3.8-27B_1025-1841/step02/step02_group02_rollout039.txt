import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def adam_smooth(h, n, tau_start, tau_end, max_iters, time_limit, initial_lr=0.1):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj = compute_c5(h) * n / 2.0  # convert to same scale
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = initial_lr
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Adaptive tau
        progress = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        obj, grad = smooth_obj_grad(h, tau)
        
        # Track best by true C5
        c5_true = compute_c5(h)
        if c5_true < best_obj - 1e-12:
            best_obj = c5_true
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Adam update
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_new = project_to_feasible(h_new, n)
        
        # Learning rate decay
        if stall > 100:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        
        h = h_new
    
    return best_h, best_obj

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, it / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** progress
        obj, grad = smooth_obj_grad(h, tau)
        c5_true = compute_c5(h)
        if c5_true < best_c5 - 1e-13:
            best_c5 = c5_true
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        t = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        if stall > 150:
            lr *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        if stall > 300 and lr < 1e-6:
            break
    return best_h, best_c5

def subgrad_refine(h, n, max_iters, time_limit, lr=0.005):

    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    stall = 0

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:

            best_h = h.copy()
            stall = 0
        else:

        j1 = min(n, sh + n)

        j0b = max(0, -sh)

            v *= 0.5

def fast_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()
    
    # FFT size
    L = 1
    while L < 2 * (2*n - 1) - 1:
        L *= 2
    # Actually we need L >= (2n-1) + n - 1 = 3n-2 for the convolution
    # Let's just use L = 4*n to be safe
    L = 4 * n
    
    H = np.fft.rfft(h, L)
    P = np.fft.rfft(p, L)
    
    # Cross-correlation R_ph[m] = sum_i p[i+m] * h[i] = ifft(conj(FFT(p)) * FFT(h))
    # Wait, let me use the correct formula.
    # R_ab[m] = sum_n a[n+m] * b[n]
    # R_ab = ifft(FFT(a) * conj(FFT(b)))
    # So R_ph[m] = sum_n p[n+m] * h[n] = ifft(FFT(p) * conj(FFT(h)))
    
    R_ph = np.fft.irfft(H * np.conj(P), L)  # Wait, this gives ifft(FFT(h) * conj(FFT(p)))
    # = sum_n h[n+m] * p[n]... that's the cross-correlation of h and p, not p and h.
    
    # Let me be more careful.
    # ifft(FFT(a) * conj(FFT(b))) [m] = sum_n a[n+m] * b[n] (for real a, b, with zero-padding)
    
    # I want R_ph[m] = sum_i p[i+m] * h[i]
    # This is ifft(FFT(p) * conj(FFT(h))) [m] = sum_n p[n+m] * h[n] ✓
    
    R_ph = np.fft.irfft(P * np.conj(H), L)[:2*n-1]
    
    # Convolution (h*p)[m] = sum_i h[i] * p[m-i] = ifft(FFT(h) * FFT(p)) [m]
    conv_hp = np.fft.irfft(H * P, L)[:3*n-2]
    # We need conv_hp[n-1+j] for j=0,...,n-1, so we need indices n-1 to 2n-2
    # conv_hp has length 3n-2 (indices 0 to 3n-3), and n-1+j ranges from n-1 to 2n-2, which is fine.
    
    A = R_ph[n-1::-1]  # A[j] = R_ph[n-1-j] for j=0,...,n-1
    B = conv_hp[n-1:2*n-1]  # B[j] = conv_hp[n-1+j] for j=0,...,n-1
    
    grad = dx * (1.0 - A - B)
    obj = dx * (max_c + np.log(exp_c.sum()) / tau)
    return obj, grad

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def fast_smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # FFT size: need L >= (2n-1) + n - 1 = 3n - 2
    L = 1
    while L < 3 * n - 1:
        L *= 2
    
    H = np.fft.rfft(h, L)
    P = np.fft.rfft(p, L)
    
    # R_ph[m] = sum_i p[i+m] * h[i]
    R_ph = np.fft.irfft(P * np.conj(H), L)
    
    # (h*p)[m] = sum_i h[i] * p[m-i]
    conv_hp = np.fft.irfft(H * P, L)
    
    # A[j] = R_ph[n-1-j]
    A = R_ph[n-1::-1]  # indices n-1, n-2, ..., 0 → length n
    # B[j] = conv_hp[n-1+j]
    B = conv_hp[n-1:2*n-1]  # indices n-1, n, ..., 2n-2 → length n
    
    grad = dx * (1.0 - A - B)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def adam_optimize(h, n, tau_start, tau_end, max_iters, time_limit, lr=0.1):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        progress = min(1.0, (it + 1) / max(1, max_iters))
        tau = tau_start * (tau_end / tau_start) ** progress
        obj, grad = fast_smooth_grad(h, tau)
        c5_true = compute_c5(h)
        if c5_true < best_c5 - 1e-13:
            best_c5 = c5_true
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        t = it + 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)
        if stall > 200:
            lr *= 0.5
            stall = 0
            if lr < 1e-12:
                break
    return best_h, best_c5

def subgrad_refine(h, n, max_iters, time_limit, lr=0.005):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= h[idx + sh]
        grad *= dx
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 300:
            lr *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-14:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))
    
    h_init = initial_h_values.copy()

        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Phase 2: Adam at n=2500
    n2 = 2500
    if remaining() > 100:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(remaining() * 0.25, 150)
        h2, c5_2 = adam_optimize(h2, n2, 20.0, 100.0, 5000, tl, lr=0.08)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Phase 3: Adam at n=4000 with even lower tau

    if remaining() > 100:

        h3 = project_to_feasible(h3, n3)
        tl = min(remaining() * 0.2, 120)

        if c5_3 < best_c5:

            best_h = h3.copy()

        if c5_f < best_c5:

        best_c5 = final_c5
