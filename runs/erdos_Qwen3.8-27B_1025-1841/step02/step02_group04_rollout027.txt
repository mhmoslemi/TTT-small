                            new_max = float(np.max(new_corr))
                            delta = new_max - current_max
                            if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
                                corr = new_corr
                                current_max = new_max
                                if i in ones:
                                    ones.remove(i)
                                ones.append(j)
                                if j in zeros:
                                    zeros.remove(j)
                                zeros.append(i)
                                if new_max < best_max:
                                    best_max = new_max
                                    best_h = h.copy()
                                    no_improve = 0
                            else:
                                h[i] = 1.0
                                h[j] = 0.0
                                no_improve += 1
                            continue
        if not ones or not zeros:
            break
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        h[i] = 0.0
        h[j] = 1.0
        new_corr = corr_vec(h)
        new_max = float(np.max(new_corr))
        delta = new_max - current_max
        if delta <= 0 or rng.random() < np.exp(-delta / max(T, 1e-10)):
            corr = new_corr
            current_max = new_max
            if i in ones:
                ones.remove(i)
            ones.append(j)
            if j in zeros:
                zeros.remove(j)
            zeros.append(i)
            if new_max < best_max:
                best_max = new_max
                best_h = h.copy()
                no_improve = 0
        else:
            h[i] = 1.0
            h[j] = 0.0
            no_improve += 1
    return best_h, best_max * dx

def softgrad_step(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        return np.zeros(n)
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-10:
            break
        shift = k_idx - (n - 1)
        s = abs(shift)
        if s >= n:
            continue
        if shift >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] += w * (-h[s:])
        else:
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] += w * (-h[:n - s])
    return grad * dx

def continuous_refine(h, n_iter=500, T=5.0, lr=0.03):
    n = len(h)
    h = h.astype(float).copy()
    dx = 2.0 / n
    target = n / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        T_t = max(T * (0.5 ** (it / max(n_iter, 1))), 0.5)
        lr_t = lr * (1.0 - 0.5 * it / n_iter)
        grad = softgrad_step(h, T_t)
        h = h - lr_t * grad
        h = np.clip(h, 0.0, 1.0)
        h = h / max(h.sum(), 1e-10) * target
        h = np.clip(h, 0.0, 1.0)
        if (it + 1) % 50 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is not None:
        n0 = len(h0)
        c_cont = compute_c5(h0)
        h_bin = make_binary_feasible(h0, n0, rng)
        c_bin = compute_c5(h_bin)
        best_h = h0.copy() if c_cont <= c_bin else h_bin.copy()
        best_c5 = min(c_cont, c_bin)
        print(f"Initial: n={n0}, C5(cont)={c_cont:.6f}, C5(bin)={c_bin:.6f}", flush=True)
    else:
        n0 = 512
        h0 = (rng.rand(n0) < 0.5).astype(float)
        h0 = make_binary_feasible(h0, n0, rng)
        best_h = h0.copy()
        best_c5 = compute_c5(h0)
        print(f"Random init: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Binary SA at n=1024 from upsampled parent
    n1 = 1024
    h_up1 = upsample_binary(best_h, n1, rng)
    c1 = compute_c5(h_up1)
    print(f"Upsampled n={n1}: C5={c1:.6f}", flush=True)
    if c1 < best_c5:
        best_h, best_c5 = h_up1.copy(), c1

    remaining = budget_s - elapsed() - 200
    if remaining > 20:
        t_alloc = min(remaining, 400)
        h_sa, c_sa = binary_sa(h_up1, time_budget=t_alloc, seed=seed, T_start=0.4, T_end=0.003, patience=50000)
        if c_sa < best_c5:
            best_h, best_c5 = h_sa.copy(), c_sa
        print(f"SA n={n1}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 2: Binary SA at n=2048
    if elapsed() < budget_s - 150:
        n2 = 2048
        h_up2 = upsample_binary(best_h, n2, rng)
        c2 = compute_c5(h_up2)
        print(f"Upsampled n={n2}: C5={c2:.6f}", flush=True)
        if c2 < best_c5:
            best_h, best_c5 = h_up2.copy(), c2
        remaining = budget_s - elapsed() - 100
        if remaining > 20:
            t_alloc = min(remaining, 500)
            h_sa2, c_sa2 = binary_sa(h_up2, time_budget=t_alloc, seed=seed + 7, T_start=0.3, T_end=0.002, patience=60000)
            if c_sa2 < best_c5:
                best_h, best_c5 = h_sa2.copy(), c_sa2
            print(f"SA n={n2}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 3: Multi-start at best resolution
    if elapsed() < budget_s - 80:
        n_pts = len(best_h)
        n_restarts = 3
        time_per = (budget_s - 80 - elapsed()) / n_restarts
        for r in range(n_restarts):
            if elapsed() > budget_s - 40:
                break
            h_pert = best_h + rng.randn(n_pts) * 0.08
            h_pert = make_binary_feasible(h_pert, n_pts, rng)
            h_r, c_r = binary_sa(h_pert, time_budget=time_per, seed=seed + 500 + r * 31, T_start=0.5, T_end=0.005, patience=30000)
            if c_r < best_c5:
                best_h, best_c5 = h_r.copy(), c_r
                print(f"  Restart {r}: C5={best_c5:.6f}", flush=True)

    # Phase 4: Continuous refinement
    if elapsed() < budget_s - 50:
        n_pts = len(best_h)
        h_ref = best_h.astype(float).copy()
        h_ref = np.clip(h_ref + rng.randn(n_pts) * 0.015, 0, 1)
        target_sum = n_pts / 2.0
        h_ref = h_ref / max(h_ref.sum(), 1e-10) * target_sum
        h_ref = np.clip(h_ref, 0, 1)
        h_cr, c_cr = continuous_refine(h_ref, n_iter=400, T=4.0, lr=0.025)
        if c_cr < best_c5:
            best_h, best_c5 = h_cr, c_cr
            print(f"Continuous: C5={best_c5:.6f}", flush=True)

    # Phase 5: Final tight GD
    if elapsed() < budget_s - 20:
        h_f, c_f = continuous_refine(best_h.copy(), n_iter=300, T=2.0, lr=0.01)
        if c_f < best_c5:
            best_h, best_c5 = h_f, c_f
        print(f"FinalGD: C5={best_c5:.6f}", flush=True)

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
