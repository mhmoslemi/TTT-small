def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def compute_gradient(h, p_weights, top_idx, n, dx):
    """Compute gradient using given weights for top indices."""
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_weights[i]
        if pk < 1e-15:
            continue
        k = top_idx[i]
        d = k - n + 1
        # (1-h)[i-d] part: contribution to grad[i] from (1-h[i-d])

        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx

def fourier_smooth(h, n, keep_fraction=0.5):

    H_smooth[-cutoff:] = H[-cutoff:]
    h_smooth = np.real(np.fft.ifft(H_smooth))
    return project_to_feasible(h_smooth, n)

    """
    h: array of length n
    p: probability weights for each of the 2n-1 correlation lags
    Returns: gradient of the weighted sum of correlations
    """

    # K index n-1+d corresponds to weight for shift d

    # K[n-1] = 2*p[n-1]
    K = np.zeros(2*n - 1)
    for d in range(n):
        if d == 0:
            K[n-1] = 2.0 * p[n-1]
        else:
            K[n-1+d] = p[n-1+d] + p[n-1-d]
            K[n-1-d] = p[n-1+d] + p[n-1-d]
    
    # Compute convolution using FFT

    # Actually, we need length >= n + (2n-1) - 1 = 3n-2

        fft_len *= 2
