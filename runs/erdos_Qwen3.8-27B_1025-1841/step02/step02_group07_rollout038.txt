def objective_and_grad(h, n, T, mu):
    dx = 2.0 / n
    # Compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    # Softmax
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    sum_exp = np.sum(exp_scaled)
    f = (m + np.log(sum_exp)) / T * dx + mu * (np.sum(h) - n/2.0)**2
    
    # Gradient of softmax part
    p = exp_scaled / sum_exp  # softmax weights
    # grad_softmax[i] = dx * sum_k p_k * d(corr[k])/d(h[i])
    
    # Efficient gradient computation
    # For shift s (where s = k - (n-1)), corr[n-1+s] = sum_j h[j]*(1-h[j+s])
    # d(corr[n-1+s])/d(h[i]) = (1-h[i+s])*I[valid] - h[i-s]*I[valid]
    
    # Vectorized: for each shift s, contribution to grad is:
    #   p[n-1+s] * dx * [(1-h[i+s]) for valid i - h[i-s] for valid i]
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        idx = n - 1 + s  # index into corr
        w = p[idx] * dx
        if s >= 0:
            # (1-h[i+s]) valid for i in [0, n-1-s]
            # -h[i-s] valid for i in [s, n-1]
            grad[:n-s] += w * (1.0 - h[s:n])  # (1-h[i+s]) for i=0,...,n-1-s
            grad[s:] += -w * h[:n-s]  # -h[i-s] for i=s,...,n-1
        else:
            s_abs = -s
            # (1-h[i+s]) = (1-h[i-s_abs]) valid for i in [s_abs, n-1]
            # -h[i-s] = -h[i+s_abs] valid for i in [0, n-1-s_abs]
            grad[s_abs:] += w * (1.0 - h[:n-s_abs])  # (1-h[i-s_abs]) for i=s_abs,...,n-1
            grad[:n-s_abs] += -w * h[s_abs:n]  # -h[i+s_abs] for i=0,...,n-1-s_abs
    
    # Add penalty gradient
    grad += 2.0 * mu * (np.sum(h) - n/2.0)
    
    return f, grad

    exp_scaled = np.exp(scaled - m)
    sum_exp = np.sum(exp_scaled)
    f = (m + np.log(sum_exp)) / T * dx
    p = exp_scaled / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        idx = n - 1 + s
        w = p[idx] * dx
        if abs(w) < 1e-15:
            continue
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += w * (1.0 - h[s:])
            if n - s > 0:
                grad[s:] += -w * h[:n-s]
        else:
            sa = -s
            if n - sa > 0:
                grad[sa:] += w * (1.0 - h[:n-sa])
                grad[:n-sa] += -w * h[sa:]
    
    return f, grad, p

def lbbfgs_step(h0, n, T, mu, maxiter=2000, deadline=None, maxfun=5000):
    """L-BFGS-B optimization with penalty for sum constraint."""
    target = n / 2.0
    
    def fun_and_grad(x):
        f_soft, g_soft, _ = corr_and_grad_softmax(x, n, T)
        penalty = mu * (np.sum(x) - target) ** 2
        g_pen = 2.0 * mu * (np.sum(x) - target)
        return f_soft + penalty, g_soft + g_pen
    
    try:
        res = minimize(fun_and_grad, h0, method='L-BFGS-B',
                       jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'maxfun': maxfun,
                                'ftol': 1e-16, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0.copy()

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def projected_gradient(h, n, T, lr, n_steps, deadline):
    """Project gradient descent with adaptive step size."""
    h = h.copy()
    for step in range(n_steps):
        if time.time() > deadline:
            break
        f, g, _ = corr_and_grad_softmax(h, n, T)
        # Also add constraint gradient
        target = n / 2.0
        g += 2.0 * 100.0 * (np.sum(h) - target)
        
        # Adaptive step size (backtracking)
        step_size = lr
        for _ in range(20):
            h_new = project(h - step_size * g, n)
            f_new, _, _ = corr_and_grad_softmax(h_new, n, T)
            if f_new < f:
                break
            step_size *= 0.5
        else:
            h_new = project(h - step_size * g, n)
        h = h_new
    return h

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    sum_exp = np.sum(exp_s)
    f = (m + np.log(sum_exp)) * dx / T
    p = exp_s / sum_exp
    
    grad = np.zeros(n)
    for s in range(-(n-1), n):
        idx = n - 1 + s
        w = p[idx] * dx
        if w < 1e-15:
            continue
        if s >= 0:
            if n - s > 0:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] += -w * h[:n-s]
        else:
            sa = -s
            if n - sa > 0:
                grad[sa:] += w * (1.0 - h[:n-sa])
                grad[:n-sa] += -w * h[sa:]
    
    return f, grad

def lbbfgs(h0, n, T, mu, maxiter, maxfun):
    target = n / 2.0
    def fun_g(x):
        f, g = softmax_obj_grad(x, n, T)
        pen = mu * (np.sum(x) - target) ** 2
        gp = 2.0 * mu * (np.sum(x) - target)
        return f + pen, g + gp
    try:
        res = minimize(fun_g, h0, method='L-BFGS-B', jac=True,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'maxfun': maxfun,
                                'ftol': 1e-18, 'gtol': 1e-12, 'iprint': -1})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0.copy()

def interp_resample(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0.0, 2.0, n_from, endpoint=False)
    xt = np.linspace(0.0, 2.0, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)
    
    best_h = project_feasible(h_start, len(h_start))
    best_c5 = evaluate_c5(best_h)
    
    n0 = len(best_h)
    
    # Phase 1: Optimize at current and higher resolutions
    resolutions = sorted(set([n0, 500, 672, 840]))
    
    for n in resolutions:
        if time.time() > deadline - 60:
            break
        if n < n0:
            continue
        
        if n == n0:
            h_work = best_h.copy()
        else:
            h_work = project_feasible(interp_resample(best_h, n0, n), n)
        
        # Try increasing T and mu
        for T, mu in [(10.0, 1.0), (30.0, 5.0), (100.0, 10.0), (300.0, 50.0), (1000.0, 100.0)]:
            if time.time() > deadline - 30:

            remaining = deadline - time.time()
            if remaining < 20:

            maxiter = min(3000, int(remaining * 3))
            maxfun = min(10000, int(remaining * 5))
            
            h_new = lbbfgs(h_work, n, T, mu, maxiter=maxiter, maxfun=maxfun)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:

            h_work = h_new
