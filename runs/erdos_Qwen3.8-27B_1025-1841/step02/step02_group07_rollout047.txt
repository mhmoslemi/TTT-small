from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Project to binary while maintaining sum = n/2."""
    target = int(n / 2)
    h_sorted = np.sort(h)[::-1]  # descending
    # Take top 'target' as 1, rest as 0
    binary = np.zeros(n)
    # Use the original ordering
    idx = np.argsort(h)[::-1][:target]
    binary[idx] = 1.0
    return binary

def binary_local_search(h_bin, n, max_iters=100, deadline=None):
    """Greedy local search: flip pairs (0->1, 1->0) to reduce C5."""
    h = h_bin.copy().astype(float)
    c5 = evaluate_c5(h)
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        # Try all pairs
        best_delta = 0
        best_pair = None
        for i in zeros:

                old_i, old_j = h[i], h[j]
                h[i] = 1.0
                h[j] = 0.0

                h[i] = old_i
                h[j] = old_j
                delta = new_c5 - c5
                if delta < best_delta - 1e-12:

                    best_pair = (i, j)
        
        if best_pair is not None:

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            improved = True
        
        if not improved:
            break
    return h, c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)

def soft_max_obj(h, n, T, penalty=1e4):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    val = (m + np.log(np.sum(np.exp(corr - m + 1)) - 1))  # log-sum-exp trick
    # Actually let me be more careful

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        mask = (h < 1.0) if d > 0 else (h > 0.0)
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
