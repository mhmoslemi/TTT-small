    corr = np.correlate(h, 1.0 - h, mode='full')

    s = k_max - (n - 1)  # actual shift

    # q_k = sum_i h[i] * (1 - h[i+s]) for valid i
    # d/dh[j] q_k = (1 - h[j+s]) [if j+s in range] - h[j-s] [if j-s in range]
    grad = np.zeros(n)
    # First part: (1 - h[j+s]) for j+s in [0, n-1]
    if s >= 0:
        for j in range(n - s):
            grad[j] += (1 - h[j + s])
    else:
        for j in range(-s, n):
            grad[j] += (1 - h[j + s])  # j+s >= 0
    # Second part: -h[j-s] for j-s in [0, n-1]  
    if s >= 0:
        for j in range(s, n):
            grad[j] -= h[j - s]
    else:
        for j in range(n + s):
            grad[j] -= h[j - s]  # j-s >= 0
    
    return grad

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)

def lse_objective_and_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    exp_vals = np.exp(T * (corr - mc))
    sum_exp = np.sum(exp_vals)
    w = exp_vals / sum_exp
    F = mc + np.log(sum_exp) / T
    
    grad = np.zeros(n)

def projected_gradient_step(h, n, T, alpha, momentum=None, v=None):
    """One step of (momentum) projected gradient descent."""
    f, g = lse_objective_and_grad(h, n, T)
    h = make_feasible(h, n)
    if momentum is not None and v is not None:
        v_new = momentum * v - alpha * g
        h_new = h + v_new
    else:
        h_new = h - alpha * g
    h_new = make_feasible(h_new, n)
    return h_new, f, v_new if momentum is not None else None

                    best_val = c
            if best_val != h[j]:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c5 = best_local
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    best_c5 = float('inf')
    best_h = None

    h0 = None
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        pass
    if h0 is None:
        h0 = random_binary(320, rng)
    else:
        h0 = make_feasible(h0, len(h0))

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    def elapsed():
        return time.time() - t0

    def try_lbfgs(h_start, n, T, maxiter=300):
        nonlocal best_c5, best_h
        if elapsed() > budget_s - 30:
            return
        def fun(x):
            return lse_obj(x, n, T)
        def fun_g(x):
            return lse_obj_grad(x, n, T)
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(fun_g, h_start, method='L-BFGS-B',
                          bounds=bounds, jac=True,
                          options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-8})
            h_res = make_feasible(res.x, n)
            c5v = compute_c5(h_res)
            if c5v < best_c5:
                best_c5 = c5v
                best_h = h_res.copy()
        except Exception:
            pass

    # Phase 1: Multi-start at n=512 with increasing temperature
    n1 = 512
    h_cur = upsample_h(best_h, n1)
    c5_512 = compute_c5(h_cur)
    if c5_512 < best_c5:
        best_c5 = c5_512
        best_h = h_cur.copy()

    temps_512 = [12, 20, 35, 50, 80, 120, 180, 250]
    for T in temps_512:
        if elapsed() > budget_s - 100:
            break
        # Multiple restarts at this temperature
        n_restarts = 3 if T <= 50 else 2
        for r in range(n_restarts):
            if elapsed() > budget_s - 60:
                break
            if r == 0:
                h_start = best_h.copy()
                if len(h_start) != n1:
                    h_start = upsample_h(best_h, n1)
            elif r == 1:
                h_start = best_h.copy()
                if len(h_start) != n1:
                    h_start = upsample_h(best_h, n1)
                h_start = h_start + rng.randn(n1) * 0.03
                h_start = make_feasible(h_start, n1)
            else:
                h_start = random_binary(n1, rng)
            try_lbfgs(h_start, n1, T, maxiter=400 if T <= 50 else 200)

    # Phase 2: Try n=640
    n2 = 640
    if elapsed() < budget_s - 200:
        h640 = upsample_h(best_h, n2)
        c5_640 = compute_c5(h640)
        if c5_640 < best_c5:
            best_c5 = c5_640
            best_h = h640.copy()
        temps_640 = [20, 40, 70, 110, 160]
        for T in temps_640:
            if elapsed() > budget_s - 100:
                break
            for r in range(2):
                if elapsed() > budget_s - 60:
                    break
                if r == 0:
                    h_start = best_h.copy()
                else:
                    h_start = best_h + rng.randn(n2) * 0.02
                    h_start = make_feasible(h_start, n2)
                try_lbfgs(h_start, n2, T, maxiter=250)

    # Phase 3: Coordinate descent polish
    n_pts = len(best_h)
    remaining = budget_s - elapsed() - 20
    if remaining > 5:
        best_h, best_c5 = coordinate_polish(best_h, n_pts,
                                            time_limit=budget_s - 15,
                                            start_time=t0)

    # Phase 4: Final L-BFGS-B with very high T
    if elapsed() < budget_s - 50:
        n_pts = len(best_h)
        try_lbfgs(best_h, n_pts, T=300, maxiter=200)

    # Phase 5: Final coordinate polish
    if elapsed() < budget_s - 10:
        n_pts = len(best_h)
        best_h, best_c5 = coordinate_polish(best_h, n_pts,
                                            time_limit=budget_s - 5,
                                            start_time=t0)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
