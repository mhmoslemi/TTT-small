def lp_linearize(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute gradients of corr[k] w.r.t. h[j] at current h
    # g[k, j] = d(corr[k])/d(h[j])
    g = np.zeros((2*n-1, n))
    for j in range(n):
        for m in range(2*n-1):
            s = m - n + 1
            dc = 0.0
            if 0 <= j - s < n:
                dc += (1.0 - h[j - s])
            if 0 <= j + s < n:
                dc -= h[j + s]
            g[m, j] = dc
    
    # LP variables
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n / 2.0
    ]
    
    for m in range(2*n-1):
        # Linearized: corr[m] + sum_j g[m,j]*(h[j]-h_cur[j]) <= t/dx
        linear = corr[m] + g[m, :] @ (h_var - h)
        constraints.append(linear * dx <= t_var)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    prob.solve(solver=cp.ECOS, max_iters=10000, absopt=1e-8)
    
    if h_var.value is not None:
        return h_var.value.copy()
    return h

    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 25

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = project_h(h_start.copy(), len(h_start))
    best_c5 = evaluate_c5(best_h)
    n0 = len(best_h)

    # Phase 1: L-BFGS with increasing T at original resolution
    for T in [30.0, 80.0, 150.0, 300.0, 500.0]:
        if time.time() > deadline - 100:
            break
        h_new = lbfgs_opt(best_h, n0, T, maxiter=400)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new

    # Phase 2: SLSQP polish
    for T in [200.0, 400.0]:
        if time.time() > deadline - 80:
            break
        h_new = slsqp_opt(best_h, n0, T, maxiter=250)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new

    # Phase 3: Try higher resolution n=420
    if time.time() < deadline - 150:
        n_hi = 420
        h_hi = project_h(interp_to(best_h, n0, n_hi), n_hi)
        for T in [80.0, 200.0, 400.0]:
            if time.time() > deadline - 60:
                break
            h_new = lbfgs_opt(h_hi, n_hi, T, maxiter=300)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h_new
                n_hi = len(h_new)
        if len(best_h) == n_hi and time.time() < deadline - 40:
            h_new = slsqp_opt(best_h, n_hi, 300.0, maxiter=200)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h_new

    # Phase 4: LP linearization at current best resolution
    n_cur = len(best_h)
    if n_cur <= 420 and time.time() < deadline - 30:
        h_lp = lp_phase(best_h, n_cur, n_iters=4)
        c5_lp = evaluate_c5(h_lp)
        if c5_lp < best_c5 - 1e-10:
            best_c5 = c5_lp
            best_h = h_lp
            n_cur = len(best_h)

    # Phase 5: Random restarts with multi-scale perturbation
    n_cur = len(best_h)
    sigmas = [0.02, 0.04, 0.06, 0.08, 0.12, 0.15]
    for attempt in range(30):
        if time.time() > deadline - 20:
            break
        sigma = sigmas[attempt % len(sigmas)]
        h_pert = best_h + np.random.randn(n_cur) * sigma
        h_pert = project_h(h_pert, n_cur)
        h_new = lbfgs_opt(h_pert, n_cur, 200.0, maxiter=200)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new

    # Phase 6: Try truly random starts
    for r in range(5):
        if time.time() > deadline - 20:
            break
        h_rand = project_h(np.random.rand(n_cur), n_cur)
        h_new = lbfgs_opt(h_rand, n_cur, 60.0, maxiter=300)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new

    # Phase 7: Final high-T polish
    if time.time() < deadline - 10:
        h_new = lbfgs_opt(best_h, len(best_h), 800.0, maxiter=300)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new
        h_new = slsqp_opt(best_h, len(best_h), 600.0, maxiter=200)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
