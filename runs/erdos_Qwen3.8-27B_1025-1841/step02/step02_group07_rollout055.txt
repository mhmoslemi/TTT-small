```python
        # d(corr_j)/dh[m] = (1-h[m+shift]) * [0<=m+shift<N] - h[m-shift] * [0<=m-shift<N]
        vp = (idx + shift)
        vm = (idx - shift)
        mask_p = (vp >= 0) & (vp < N)
        mask_m = (vm >= 0) & (vm < N)
        if np.any(mask_p):
            grad[mask_p] += w[j] * (1.0 - h[vp[mask_p]])
        if np.any(mask_m):
            grad[mask_m] -= w[j] * h[vm[mask_m]]
    
    grad *= dx
    return obj, grad
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(2000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj_grad(h, T):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    sum_exps = np.sum(exps)
    w = exps / sum_exps
    obj = (m + np.log(sum_exps)) / T * dx
    grad = np.zeros(N)
    idx = np.arange(N)
    for j in range(2 * N - 1):
        if w[j] < 1e-14:
            continue
        shift = j - (N - 1)
        vp = idx + shift
        vm = idx - shift
        mask_p = (vp >= 0) & (vp < N)
        mask_m = (vm >= 0) & (vm < N)
        if np.any(mask_p):
            grad[mask_p] += w[j] * (1.0 - h[vp[mask_p]])
        if np.any(mask_m):
            grad[mask_m] -= w[j] * h[vm[mask_m]]
    grad *= dx
    return obj, grad

def slsqp_analytic(h0, N, T, maxiter, deadline):
    target = N / 2.0
    def f(x):
        return soft_max_obj_grad(x, T)[0]
    def g(x):
        return soft_max_obj_grad(x, T)[1]
    def c_eq(x):
        return np.sum(x) - target
    dc_eq = np.ones(N)
    constraint = {'type': 'eq', 'fun': c_eq, 'jac': lambda x: dc_eq}
    try:
        res = minimize(f, h0.copy(), jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * N,
                       constraints=[constraint],
                       options={'ftol': 1e-12, 'maxiter': maxiter, 'disp': False})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, target)
    except Exception:
        pass
    return h0

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def coord_descend(h, N, deadline, n_sweeps=3):
    target = N / 2.0
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        c5_cur = evaluate_c5(h)
        changed = False
        for i in range(N):
            if time.time() > deadline:
                return h
            orig = h[i]
            for delta in [0.02, 0.01, -0.01, -0.02, 0.04, -0.04]:
                cand = h.copy()
                cand[i] = np.clip(orig + delta, 0.0, 1.0)
                s = np.sum(cand)
                if s != target:
                    continue
                c5_try = evaluate_c5(cand)
                if c5_try < c5_cur - 1e-10:
                    h = cand
                    c5_cur = c5_try
                    changed = True
                    break
    return h

def block_perturb(h, N, rng):
    h_p = h.copy()
    start = rng.randint(0, N - 40)
    width = rng.randint(20, 60)
    amp = rng.uniform(0.05, 0.15)
    sig = rng.choice([1, -1])
    for i in range(start, min(start + width, N)):
        t = (i - start) / max(width - 1, 1)
        h_p[i] += sig * amp * 4.0 * t * (1 - t)
    return project(h_p, N / 2.0)
```
