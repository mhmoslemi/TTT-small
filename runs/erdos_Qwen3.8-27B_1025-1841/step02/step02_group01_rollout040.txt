    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n)

    h_new = np.interp(new_x, old_x, h)

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Numerical stability
    soft = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return soft * dx

import numpy as np
from scipy.optimize import minimize
import time

_G_T = 40.0
_G_N = 320

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _f_and_g(h):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    esum = exp_s.sum()
    p = exp_s / esum
    obj = (mc + np.log(esum) / T) * dx
    first_arr = np.correlate(p, 1.0 - h, mode='full')
    second_arr = np.convolve(p, h, mode='full')
    first_part = first_arr[n - 1::-1][:n]
    second_part = second_arr[n - 1:2 * n - 1]
    grad = dx * (first_part - second_part)
    return obj, grad

def _con_fun(h):
    return np.sum(h) - _G_N / 2.0

def _con_jac(h):
    return np.ones(_G_N)

def _slsqp_optimize(h_start, n, T, maxiter):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_f_and_g, h_start, jac=True, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13, 'disp': False})
        if res.x is not None:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return None

def _coord_descent(h, n, c5_budget_s, start_time):
    improved = True
    passes_done = 0
    steps_list = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    while improved and passes_done < 60:
        passes_done += 1
        improved = False
        step_idx = min(passes_done - 1, len(steps_list) - 1)
        if passes_done > len(steps_list):
            step_idx = len(steps_list) - 1
        step = steps_list[step_idx]
        for j in range(n):
            if time.time() - start_time > c5_budget_s:
                return h
            old_val = h[j]
            candidates = [old_val - step, old_val + step]
            if step < 0.01:
                candidates.extend([0.0, 1.0])
            best_c_here = compute_c5(h)
            best_val = old_val
            for candidate in candidates:
                if candidate < -1e-9 or candidate > 1.0 + 1e-9:
                    continue
                candidate = min(max(candidate, 0.0), 1.0)
                trial = h.copy()
                trial[j] = candidate
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_here - 1e-14:
                    best_c_here = c
                    best_val = candidate
                    improved = True
            if best_val != old_val:
                h[j] = best_val
                h = make_feasible(h, n)
    return h

    global _G_N, _G_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    if h_parent is not None:

        n_parent = len(best_h)
    else:
        n_parent = 320
        best_h = (rng.rand(n_parent) < 0.5).astype(float)
        best_h = make_feasible(best_h, n_parent)
        best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP optimization at increasing resolutions
    resolutions = [
        (n_parent, [15, 30, 60, 120, 240]),
        (int(n_parent * 1.5), [15, 30, 60, 120, 240]),
        (int(n_parent * 2), [15, 30, 60, 120, 240]),
        (int(n_parent * 3), [15, 30, 60, 120]),
    ]

    for n_target, temps in resolutions:
        if time.time() - start_time > budget_s * 0.65:
            break
        h_work = upsample_h(best_h, n_target)
        c_work = compute_c5(h_work)

        for T in temps:
            if time.time() - start_time > budget_s * 0.7:
                break
            elapsed = time.time() - start_time
            remain = budget_s - elapsed - 60
            if remain < 5:
                break
            # Scale maxiter based on remaining time and n
            maxiter = min(400, max(50, int(remain * 20 / (n_target / 320.0))))
            result = _slsqp_optimize(h_work, n_target, T, maxiter)
            if result is not None:
                c = compute_c5(result)
                if c < best_c5:
                    best_c5 = c
                    best_h = result.copy()
                    h_work = result.copy()
                elif c < c_work - 1e-8:
                    h_work = result.copy()
                    c_work = c

    # Phase 2: Iterated local search with random perturbations
    n_work = len(best_h)
    nils = 0
    while time.time() - start_time < budget_s * 0.85 and nils < 20:
        nils += 1
        perturbation_size = 0.03 * (1 + 0.5 * np.exp(-nils * 0.3))
        h_pert = best_h + rng.randn(n_work) * perturbation_size
        h_pert = make_feasible(h_pert, n_work)
        for T in [30, 80]:
            if time.time() - start_time > budget_s * 0.85:
                break
            maxiter = min(200, max(30, int((budget_s * 0.85 - time.time()) * 10 / (n_work / 320.0))))
            result = _slsqp_optimize(h_pert, n_work, T, maxiter)
            if result is not None:
                c = compute_c5(result)
                if c < best_c5:
                    best_c5 = c
                    best_h = result.copy()

    # Phase 3: Coordinate descent polish (most of remaining time)
    n_pts = len(best_h)
    cd_budget = budget_s - 15
    best_h = _coord_descent(best_h, n_pts, cd_budget, start_time)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
