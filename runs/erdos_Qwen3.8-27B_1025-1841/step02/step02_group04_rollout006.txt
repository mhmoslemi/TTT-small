    ctx = {'T': T, 'n': n}
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_constraint, 'jac': _eq_constraint_jac}]
    res = minimize(_slsqp_obj_wrap, h, jac=_slsqp_grad_wrap, args=(ctx,),
                   method='SLSQP', bounds=bounds, constraints=cons,
                   options={'maxiter': maxiter, 'ftol': 1e-14})
    return res.x

def nesterov_gd(h, n_iter, T_start, T_end, lr_start, lr_end, mu, momentum=0.85):
    n = len(h)
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        h_look = h - v * lr
        h_look = np.clip(h_look, 0.0, 1.0)
        g = smooth_grad(h_look, T, n) + 2.0 * mu * (np.sum(h_look) - n / 2.0) * np.ones(n)
        v = momentum * v + lr * g
        h = h - v
        h = make_feasible(h, n)
        if (it + 1) % 100 == 0:
            c = compute_c5(h)
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
    c = compute_c5(h)
    if c < best_c5:
        best_c5 = c
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    elapsed = lambda: time.time() - t0
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        n0 = 320
        h0 = (rng.rand(n0) < 0.5).astype(float)
        n_ones = n0 // 2
        ones = np.where(h0 == 1)[0]
        zeros = np.where(h0 == 0)[0]
        if len(ones) > n_ones:
            h0[rng.choice(ones, len(ones) - n_ones, replace=False)] = 0.0
        elif len(ones) < n_ones:
            h0[rng.choice(zeros, n_ones - len(ones), replace=False)] = 1.0

    n0 = len(h0)
    best_h = make_feasible(h0, n0)
    best_c5 = compute_c5(best_h)
    print(f"Init n={n0} C5={best_c5:.6f}", flush=True)

    # Phase 1: L-BFGS-B annealing at initial resolution
    T_schedule = [60.0, 30.0, 15.0, 8.0, 4.0, 2.0, 1.0, 0.5]
    for T in T_schedule:
        if elapsed() > budget_s - 150:
            break
        h_new = lbfgsb_optimize(best_h, T, n0, mu=300.0, maxiter=350)
        h_new = make_feasible(h_new, n0)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new.copy()
        print(f"  LB(T={T:.1f}) n={n0}: C5={best_c5:.6f} t={elapsed():.0f}s", flush=True)

    # Phase 2: SLSQP direct constraint refinement
    for T in [4.0, 2.0, 1.0, 0.5]:
        if elapsed() > budget_s - 100:
            break
        try:
            h_new = slsqp_optimize(best_h, T, n0, maxiter=400)
            h_new = make_feasible(h_new, n0)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()
        except Exception:
            pass
    print(f"SLSQP n={n0}: C5={best_c5:.6f} t={elapsed():.0f}s", flush=True)

    # Phase 3: Nesterov GD for additional polishing
    if elapsed() < budget_s - 100:
        h_gd, c_gd = nesterov_gd(best_h, 3000, 5.0, 0.3, 0.05, 0.005, 200.0)
        if c_gd < best_c5:
            best_c5, best_h = c_gd, h_gd.copy()
        print(f"GD n={n0}: C5={best_c5:.6f} t={elapsed():.0f}s", flush=True)

    # Phase 4: Upsample to 2x resolution
    n1 = min(n0 * 2, 640)
    if n1 > n0 and elapsed() < budget_s - 200:
        h_up = upsample_h(best_h, n1)
        c_up_init = compute_c5(h_up)
        for T in [30.0, 10.0, 4.0, 1.0, 0.5]:
            if elapsed() > budget_s - 120:
                break
            h_new = lbfgsb_optimize(h_up, T, n1, mu=300.0, maxiter=250)
            h_new = make_feasible(h_new, n1)
            c = compute_c5(h_new)
            if c < c_up_init:
                c_up_init = c
                h_up = h_new
        if c_up_init < best_c5:
            best_c5, best_h = c_up_init, h_up.copy()
        print(f"2x n={n1}: C5={best_c5:.6f} t={elapsed():.0f}s", flush=True)

    # Phase 5: Try 3x resolution
    n2 = min(n0 * 3, 960)
    if n2 > n1 and elapsed() < budget_s - 200:
        h_up2 = upsample_h(best_h, n2)
        c_up2 = compute_c5(h_up2)
        for T in [20.0, 8.0, 3.0, 1.0]:
            if elapsed() > budget_s - 100:
                break
            h_new = lbfgsb_optimize(h_up2, T, n2, mu=300.0, maxiter=150)
            h_new = make_feasible(h_new, n2)
            c = compute_c5(h_new)
            if c < c_up2:
                c_up2 = c
                h_up2 = h_new
        if c_up2 < best_c5:
            best_c5, best_h = c_up2, h_up2.copy()
        print(f"3x n={n2}: C5={best_c5:.6f} t={elapsed():.0f}s", flush=True)

    # Phase 6: Multi-start from perturbations
    n_cur = len(best_h)
    if elapsed() < budget_s - 100:
        for scale in [0.03, 0.08, 0.15, 0.3]:
            if elapsed() > budget_s - 50:
                break
            h_p = best_h + rng.randn(n_cur) * scale
            h_p = make_feasible(h_p, n_cur)
            h_new = lbfgsb_optimize(h_p, 5.0, n_cur, mu=300.0, maxiter=200)
            h_new = make_feasible(h_new, n_cur)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5, best_h = c, h_new.copy()
                print(f"  MS({scale}): C5={best_c5:.6f}", flush=True)
        print(f"MultiStart: C5={best_c5:.6f} t={elapsed():.0f}s", flush=True)

    # Phase 7: Final SLSQP polish at best resolution
    if n_cur <= 640 and elapsed() < budget_s - 40:
        for T in [2.0, 0.5]:
            if elapsed() > budget_s - 20:
                break
            try:
                h_new = slsqp_optimize(best_h, T, n_cur, maxiter=500)
                h_new = make_feasible(h_new, n_cur)
                c = compute_c5(h_new)
                if c < best_c5:
                    best_c5, best_h = c, h_new.copy()
            except Exception:
                pass

    # Phase 8: Ultra-fine L-BFGS-B
    if elapsed() < budget_s - 15:
        h_new = lbfgsb_optimize(best_h, 0.3, len(best_h), mu=500.0, maxiter=300)
        h_new = make_feasible(h_new, len(best_h))
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5, best_h = c, h_new.copy()

    # Final
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE n={len(best_h)} C5={best_c5:.6f} t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
