import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_single_shift(h, k_idx, n, dx):
    """Gradient of corr[k_idx]*dx with respect to h."""
    sh = k_idx - n + 1
    grad = np.zeros(n)
    j0 = max(0, sh)
    j1 = min(n, sh + n)
    if j0 < j1:
        idx = np.arange(j0, j1)
        grad[j0:j1] += (1.0 - h[idx - sh])
    j0b = max(0, -sh)
    j1b = min(n, n - sh)
    if j0b < j1b:
        idx = np.arange(j0b, j1b)
        grad[j0b:j1b] -= h[idx + sh]
    grad *= dx
    return grad

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s

        if pk < 1e-12:

        grad += pk * grad_single_shift(h, k, n, dx)
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

    best_c5 = compute_c5(h)
    best_h = h.copy()

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, time_limit, lr=None, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.3 / tau
    v = np.zeros(n)
    beta = momentum
    stall = 0

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            lr *= 0.5
            v *= 0.3
            stall = 0
            if lr < 1e-13:
                break

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def optimize_multishift(h, n, time_limit, lr=None, eps_frac=0.04):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.004
    v = np.zeros(n)
    beta = 0.82
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            lr *= 0.5
            v *= 0.2
            stall = 0
            if lr < 1e-15:
                break

        threshold = max_c * (1.0 - eps_frac)
        active = np.where(corr >= threshold)[0]
        active_vals = corr[active]
        w = np.exp(active_vals - active_vals[0])
        w /= w.sum()

        grad = np.zeros(n)
        for idx in range(len(active)):
            k = active[idx]
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[idx] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[idx] * h[j0b + sh:j1b + sh]
        grad *= dx

        gnorm = np.linalg.norm(grad)
        step = lr / (gnorm + 1e-15)
        v = beta * v - step * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def optimize_single_subgrad(h, n, time_limit, lr=None):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.003
    v = np.zeros(n)
    beta = 0.75
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 8000:
            lr *= 0.4
            v *= 0.2
            stall = 0
            if lr < 1e-15:
                break

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
        grad *= dx

        gnorm = np.linalg.norm(grad)
        step = lr / (gnorm + 1e-15)
        v = beta * v - step * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def optimize_adaptive_tau(h, n, time_limit):
    """Smooth max with adaptive decreasing tau."""
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    tau = 40.0
    lr = 0.1
    v = np.zeros(n)
    beta = 0.88
    stall = 0

    while time.time() - start < time_limit:
        obj, grad = smooth_obj_grad(h, tau)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
            if tau > 3.0:
                tau *= 0.97
            lr *= 0.999
        else:
            stall += 1

        if stall > 4000:
            lr *= 0.4
            v *= 0.2
            stall = 0
            if lr < 1e-13:
                if tau > 3.0:
                    tau *= 0.85
                else:
                    break

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def perturb_restart(h, n, sigma):
    rng = np.random.default_rng()
    noise = rng.normal(0, sigma, n)
    h_new = h + noise
    h_new = project_to_feasible(h_new, n)
    return h_new
