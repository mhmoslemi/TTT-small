def logsumexp_max(corr, T):
    """Smoothed max of corr using log-sum-exp."""
    mc = np.max(corr)
    if T < 1e-10:

    shifted = corr - mc
    return mc + np.log(np.sum(np.exp(T * shifted))) / T

def soft_gradient(h, T, top_k=60):
    """Gradient of the smoothed max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = corr - mc

    total = np.sum(exp_s)

        shift = k_max - (n - 1)

        shift = k_idx - (n - 1)
        grad += w * _subgrad_single(h, shift)
    return grad * dx

def cons_fun(h, n):
    return np.sum(h) - n / 2.0

def cons_jac(h, n):
    return np.ones(n)

def nesterov_optimize(h, n_iter=5000, T_start=200.0, T_end=1.0, lr=0.12, momentum=0.9):
    n = len(h)
    h_cur = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h_cur)
    best_h = h_cur.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.7 * frac)
        
        h_look = make_feasible(h_cur + v, n)
        grad = smooth_obj_grad(h_look, T, top_k=80)
        v = momentum * v - lr_t * grad
        h_new = make_feasible(h_cur + v, n)
        h_cur = h_new
        
        if (it + 1) % 200 == 0:
            c5 = compute_c5(h_cur)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_cur.copy()
    
    c5 = compute_c5(h_cur)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h_cur.copy()
    return best_h, best_c5

def upsample(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

# For L-BFGS-B, we need top-level functions (no closures)
# We'll use a global parameter dict
_PARAMS = {}

def _lbf_obj(h):
    return smooth_obj_val(h, _PARAMS['T'])

def _lbf_grad(h):
    return smooth_obj_grad(h, _PARAMS['T'], top_k=100)

def _lbf_cons_f(h):
    return cons_fun(h, _PARAMS['n'])

def _lbf_cons_jac(h):
    return cons_jac(h, _PARAMS['n'])

def _lbf_cons():
    return [{'type': 'eq', 'fun': _lbf_cons_f, 'jac': _lbf_cons_jac}]
