Let me also think about what the optimal solution looks like. The problem is to minimize the maximum of overlap(k) = ∫ h(x)(1-h(x+k)) dx over all k.

Intuitively, we want h to be such that for every shift k, the overlap between h and (1-h shifted by k) is small and roughly equal (flat profile of overlaps).

The integral constraint ∫h = 1 means the "average" of h is 1/2 (since the domain has length 2).

A constant h = 1/2 would give overlap(k) = ∫ (1/2)(1/2) dx over the overlapping region, which for shift k is (1/2)(1/2)(2-|k|) = (2-|k|)/4. The max is at k=0: overlap(0) = 2/4 = 1/2. So C5 = 1/2 for constant h.

We can do much better than 1/2. The target is 0.38085.

For a two-level function (h takes values 0 and 1), the overlap becomes:
overlap(k) = |{x: h(x)=1, h(x+k)=0}| * dx

This is related to the "disagreement" between h and its shift. The constraint is |{x: h(x)=1}| = 1/dx = n/2.

For a two-level function, this is a combinatorial optimization problem. The smooth approach effectively finds a "soft" version and then we can threshold.

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def fft_convolve(a, b):
    n = len(a) + len(b) - 1
    size = 1
    while size < n:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    return np.fft.irfft(fa * fb, size)[:n]

def smooth_grad_fft(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    # Gradient via FFT
    a1 = 1.0 - h  # length n
    a2 = h  # length n
    conv1 = fft_convolve(a1, p)  # length 3n-2
    conv2 = fft_convolve(a2, p)  # length 3n-2

    grad = dx * (conv1[n-1:2*n-1][::-1] - conv2[n-1:2*n-1])
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    # Periodic interpolation
    # Tile to handle wrap-around
    h_tiled = np.concatenate([h[-n//10:], h, h[:n//10]])
    x_tiled = np.concatenate([np.arange(-n//10, 0) * (2.0/n), x_old, np.arange(n, n+n//10) * (2.0/n)])
    return np.interp(x_new, x_tiled, h_tiled)

def adam_smooth(h, n, tau, max_iters, time_limit, lr=0.1, beta1=0.9, beta2=0.999):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_grad_fft(h, tau)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_grad_fft(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        lr_t = lr / (1 + it * 1e-4)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_to_feasible(h_new, n)
        
        if stall > 200:

            m *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-8:

        h = h_new
    return best_h, best_obj

def subgrad_refine(h, n, max_iters, time_limit, lr=0.01):

    best_c5 = compute_c5(h)
    best_h = h.copy()

            best_h = h.copy()
            stall = 0
        else:

        j1 = min(n, sh + n)
        if j0 < j1:

        grad *= (2.0 / n)
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > 500:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    return best_h, best_c5

    target = n / 2.0

        s = np.sum(np.clip(g - mid, 0.0, 1.0))

            lo = mid
        else:
            hi = mid

overlap(k) = sum_i h[i] * (1-h[i+s_k]) for valid i, where s_k = k-(n-1).

For s >= 0: overlap = sum_{i=0}^{n-1-s} h[i] - sum_{i=0}^{n-1-s} h[i]*h[i+s]
The second term is the "truncated" autocorrelation. The full autocorrelation at lag s is sum_{i=0}^{n-1-s} h[i]*h[i+s]. So for s >= 0, the truncated version IS the full autocorrelation!

For s < 0 (let s = -t, t > 0): overlap = sum_{i=t}^{n-1} h[i] - sum_{i=t}^{n-1} h[i]*h[i-t]
The second term: sum_{i=t}^{n-1} h[i]*h[i-t]. Let j = i-t: sum_{j=0}^{n-1-t} h[j+t]*h[j] = autocorrelation at lag t.

So:
- For s >= 0: overlap(s) = sum_{i=0}^{n-1-s} h[i] - autocorr[s]
- For s < 0 (s=-t): overlap(-t) = sum_{i=t}^{n-1} h[i] - autocorr[t]

Where autocorr[t] = sum_{i=0}^{n-1-t} h[i]*h[i+t] (which is np.correlate(h,h,'full')[n-1+t]).

    a1 = (1.0 - h)[::-1]
    conv1 = fft_convolve(a1, p)
    conv2 = fft_convolve(h, p)
    grad = dx * (conv1[n - 1:2 * n - 1][::-1] - conv2[n - 1:2 * n - 1])
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def multi_subgrad_obj(h, tau_w):
    n = len(h)
    dx = 2.0 / n
    corr = compute_overlaps_fft(h)
    max_c = corr.max()
    shifted = np.clip(tau_w * (corr - max_c), -400, 0)
    weights = np.exp(shifted)
    weights /= weights.sum()
    c5 = max_c * dx
    a1 = (1.0 - h)[::-1]
    conv1 = fft_convolve(a1, weights)
    conv2 = fft_convolve(h, weights)
    grad = dx * (conv1[n - 1:2 * n - 1][::-1] - conv2[n - 1:2 * n - 1])
    return c5, grad


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    pad = max(1, n // 10)
    h_tiled = np.concatenate([h[-pad:], h, h[:pad]])
    x_tiled = np.concatenate([
        np.arange(-pad, 0) * (2.0 / n),
        np.arange(n) * (2.0 / n),
        np.arange(n, n + pad) * (2.0 / n)
    ])
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_tiled, h_tiled)


def adam_smooth(h, n, tau, max_iters, time_limit, lr=0.1):
    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * grad ** 2
        m_hat = m / (1 - 0.9 ** t)
        v_hat = v / (1 - 0.999 ** t)
        lr_t = lr / (1.0 + 0.001 * it)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        h_new = project_to_feasible(h_new, n)
        if stall > 150:
            lr *= 0.5
            m *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-9:
                break
        h = h_new
    return best_h, best_obj


def multi_subgrad_opt(h, n, max_iters, time_limit, lr=0.005, tau_w=20.0):
    h = project_to_feasible(h, n).copy()
    best_c5, _ = multi_subgrad_obj(h, tau_w)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        c5_cur, grad = multi_subgrad_obj(h, tau_w)
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = 0.85 * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        if stall > 300:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: moderate resolution smooth optimization
    n1 = 2400
    if remaining() > 100:
        h1 = interp_periodic(best_h, n1)
        h1 = project_to_feasible(h1, n1)
        tl = min(remaining() * 0.22, 180)
        h1, _ = adam_smooth(h1, n1, tau=15.0, max_iters=5000, time_limit=tl, lr=0.2)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: higher resolution
    n2 = 4800
    if remaining() > 100:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(remaining() * 0.25, 220)
        h2, _ = adam_smooth(h2, n2, tau=20.0, max_iters=4000, time_limit=tl * 0.5, lr=0.1)
        if remaining() > 40:
            h2, _ = adam_smooth(h2, n2, tau=50.0, max_iters=3000, time_limit=tl * 0.3, lr=0.04)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: even higher resolution
    n3 = 7200
    if remaining() > 100:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        tl = min(remaining() * 0.28, 250)
        h3, _ = adam_smooth(h3, n3, tau=20.0, max_iters=3000, time_limit=tl * 0.5, lr=0.07)
        if remaining() > 40:
            h3, _ = adam_smooth(h3, n3, tau=60.0, max_iters=2500, time_limit=tl * 0.35, lr=0.025)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: multi-subgradient refinement at current resolution
    if remaining() > 50:
        nf = len(best_h)
        hf = best_h.copy()
        hf, c5f = multi_subgrad_opt(hf, nf, max_iters=30000, time_limit=remaining() * 0.35, lr=0.003, tau_w=25.0)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()

    # Phase 5: perturbation + re-optimize at a new resolution
    if remaining() > 60:
        npert = 4800
        hp = interp_periodic(best_h, npert)
        hp = project_to_feasible(hp, npert)
        noise = np.random.randn(npert) * 0.015
        hp = project_to_feasible(hp + noise, npert)
        hp, _ = adam_smooth(hp, npert, tau=30.0, max_iters=3000, time_limit=remaining() * 0.2, lr=0.06)
        c5p = compute_c5(hp)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 6: final polish
    if remaining() > 20:
        nfin = len(best_h)
        hfin = best_h.copy()
        hfin, c5fin = multi_subgrad_opt(hfin, nfin, max_iters=15000, time_limit=remaining() * 0.4, lr=0.001, tau_w=40.0)
        if c5fin < best_c5:
            best_c5 = c5fin
            best_h = hfin.copy()

    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
