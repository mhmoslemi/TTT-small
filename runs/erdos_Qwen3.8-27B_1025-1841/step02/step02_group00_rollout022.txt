        frac = step / max(n_steps - 1, 1)
        alpha = alpha_init * (alpha_final / alpha_init) ** frac
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        lag = k_star - (n - 1)
        sg = dx * ((1.0 - h[(idx - lag) % n]) - h[(idx + lag) % n])
        h = h - alpha * sg
        h = make_feasible(h, n)
        if step % 200 == 0 or step == n_steps - 1:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def coord_descent(h, n, max_rounds, deadline):
    h = h.copy()
    best_c5 = compute_c5(h)
    deltas = [0.001, -0.001, 0.003, -0.003, 0.01, -0.01, 0.0, 1.0]
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            if time.time() > deadline:
                break
            old = h[j]
            for d in deltas:
                if d == 0.0:
                    cand = 0.0
                elif d == 1.0:
                    cand = 1.0
                else:
                    cand = old + d
                if cand < 0.0 or cand > 1.0:
                    continue
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-15:
                    h = trial
                    best_c5 = c
                    old = cand
                    improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T
    start_time = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 160)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    n = 2048
    h_curr = upsample(best_h, n)
    c5_curr = compute_c5(h_curr)
    if c5_curr < best_c5:
        best_c5 = c5_curr
        best_h = h_curr.copy()

    # Phase 1: SLSQP progressive temperature
    temp_schedule = [
        (10.0, 200),
        (30.0, 200),
        (80.0, 200),
        (200.0, 200),
        (600.0, 150),
        (2000.0, 120),
        (6000.0, 100),
        (20000.0, 80),
    ]
    for T, max_it in temp_schedule:
        if budget_s - (time.time() - start_time) < 30:
            break
        try:
            h_opt = slsqp_optimize(h_curr, n, T, max_it)
            c5_opt = compute_c5(h_opt)
            h_curr = h_opt.copy()
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    # Phase 2: Subgradient descent
    if budget_s - (time.time() - start_time) > 60:
        try:
            h_sg, c5_sg = subgradient_descent(h_curr, n, 5000, 0.3, 0.005)
            if c5_sg < best_c5:
                best_c5 = c5_sg
                best_h = h_sg.copy()
                h_curr = h_sg.copy()
        except Exception:
            pass

    # Phase 3: SLSQP from subgradient result
    if budget_s - (time.time() - start_time) > 30:
        for T in [200.0, 1000.0, 5000.0]:
            if budget_s - (time.time() - start_time) < 25:
                break
            try:
                h_opt = slsqp_optimize(h_curr, n, T, 120)
                c5_opt = compute_c5(h_opt)
                h_curr = h_opt.copy()
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 4: Random restarts
    if budget_s - (time.time() - start_time) > 80:
        for restart in range(4):
            if budget_s - (time.time() - start_time) < 35:
                break
            noise_scale = 0.08 * (restart + 1)
            h_perturb = h_curr + noise_scale * rng.randn(n)
            h_perturb = make_feasible(h_perturb, n)
            try:
                h_opt = slsqp_optimize(h_perturb, n, 100.0, 150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_curr = h_opt.copy()
            except Exception:
                pass

    # Phase 5: Binary projection + SLSQP
    if budget_s - (time.time() - start_time) > 40:
        h_bin = (best_h >= 0.5).astype(float)
        target_ones = n // 2
        n_ones = int(np.sum(h_bin))
        if n_ones > target_ones:
            ones_idx = np.where(h_bin == 1.0)[0]
            rng2 = np.random.RandomState(seed + 1)
            rng2.shuffle(ones_idx)
            for i in range(n_ones - target_ones):
                h_bin[ones_idx[i]] = 0.0
        elif n_ones < target_ones:
            zeros_idx = np.where(h_bin == 0.0)[0]
            rng2 = np.random.RandomState(seed + 1)
            rng2.shuffle(zeros_idx)
            for i in range(target_ones - n_ones):
                h_bin[zeros_idx[i]] = 1.0
        c5_bin = compute_c5(h_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin.copy()
        # SLSQP from binary
        if budget_s - (time.time() - start_time) > 25:
            try:
                h_opt = slsqp_optimize(h_bin, n, 500.0, 150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass

    # Phase 6: Coordinate descent polish
    deadline = start_time + budget_s - 5
    if budget_s - (time.time() - start_time) > 15:
        h_cd, c5_cd = coord_descent(best_h, n, 8, deadline)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 7: Final SLSQP polish
    if budget_s - (time.time() - start_time) > 10:
        try:
            h_opt = slsqp_optimize(best_h, n, 10000.0, 60)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, n)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, n
