```python
import numpy as np
import time
from scipy.optimize import linprog

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_grad_fft(h, tau):
    """Compute gradient of smooth max objective using FFT-based convolution."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    # Softmax weights
    shifted = tau * corr - tau * corr.max()
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()

    one_minus_h = 1.0 - h
    h_rev = h[::-1]

    S = np.convolve(P, h_rev, 'full')[n-1::-1][:n]
    
    grad = (F - S) * dx
    obj = dx * (corr.max() + np.log(exp_c.sum()) / tau)
    return obj, grad, p

def optimize_smooth_fft(h, n, tau_schedule, time_limit, lr=0.3, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam-based optimization with FFT gradient."""
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    m = np.zeros(n)  # Adam first moment
    v = np.zeros(n)  # Adam second moment
    t = 0
    stall = 0
    
    for tau_idx, tau in enumerate(tau_schedule):
        tau_count = 0
        while time.time() - start < time_limit:
            t += 1
            tau_count += 1

            c5_now = None  # We'll compute it less frequently for speed

            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**t)
            v_hat = v / (1 - beta2**t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_to_feasible(h_new, n)
            
            # Check improvement every 50 iterations

                c5_now = compute_c5(h)
                if c5_now < best_c5 - 1e-12:
                    best_c5 = c5_now
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                if stall > 20:
                    break  # Move to next tau
            if tau_count > 30000:
                break
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from scipy import sparse

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_grad_fft(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    shifted = tau * (corr - corr.max())
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    one_minus_h = 1.0 - h
    h_rev = h[::-1]
    F = np.convolve(p, one_minus_h, 'full')[:n]
    S = np.convolve(p, h_rev, 'full')[n-1::-1][:n]
    grad = (F - S) * dx
    obj = dx * (corr.max() + np.log(s) / tau)
    return obj, grad

def optimize_smooth_fft(h, tau_schedule, time_limit, lr=0.5, beta1=0.9, beta2=0.999, eps=1e-8):
    n = len(h)
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    
    for tau in tau_schedule:
        iters = 0
        stall = 0
        while time.time() - start < time_limit and iters < 50000:
            iters += 1
            t += 1
            obj, grad = smooth_grad_fft(h, tau)
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * grad * grad
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
            h = project_to_feasible(h_new, n)
            if iters % 100 == 0:
                c5_now = compute_c5(h)
                if c5_now < best_c5 - 1e-12:
                    best_c5 = c5_now
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
                if stall > 30:
                    break
        # Reset Adam state for new tau
        m = np.zeros(n)
        v = np.zeros(n)
        t = 0
    
    return best_h, best_c5

def build_a_matrix(n):
    """Build sparse matrix A where A[k, i] is the coefficient of h[i] in constraint k.
    Constraint k corresponds to shift s = k - (n-1).
    A[k, i] = 1 if i+s is valid (0 ≤ i+s < n), else 0. The actual coefficient
    will be multiplied by g[i+s] during the LP solve."""
    # For shift s, valid i: max(0, -s) ≤ i ≤ min(n-1, n-1-s)
    rows = []
    cols = []
    for k in range(2 * n - 1):
        s = k - (n - 1)
        i_start = max(0, -s)
        i_end = min(n, n - s)  # exclusive
        if i_start < i_end:
            rows.extend([k] * (i_end - i_start))
            cols.extend(range(i_start, i_end))
    A = sparse.csr_matrix((np.ones(len(rows)), (rows, cols)), shape=(2*n-1, n))
    return A

def alternating_lp(h, n, max_iters=50, time_limit=200):
    start = time.time()
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    A = build_a_matrix(n)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        # h-step: minimize t s.t. A @ (h * g) ≤ t, sum(h) = n/2, 0≤h≤1
        g = 1.0 - h
        # The constraint is: for each k, sum_i A[k,i] * h[i] * g[i+s_k] ≤ t
        # This is NOT linear in h because of the g[i+s_k] factor!
        # Wait... g is FIXED in the h-step. So the coefficient of h[i] in constraint k is g[i+s_k].
        # Let's define: A_eff[k, i] = g[i+s_k] if i+s_k is valid, else 0
        # Then constraint k is: A_eff[k,:] @ h ≤ t
        
        s_shifts = np.arange(2*n-1) - (n-1)  # shifts from -(n-1) to n-1
        # For each k, the coefficients are g[i+s_k] for valid i
        # Build A_eff as sparse
        A_eff = np.zeros((2*n-1, n))
        for k in range(2*n-1):
            s = s_shifts[k]
            i_start = max(0, -s)
            i_end = min(n, n - s)
            if i_start < i_end:
                A_eff[k, i_start:i_end] = g[i_start+s:i_end+s]
        
        # LP: min t s.t. A_eff @ h - t ≤ 0, sum(h) = n/2, 0≤h≤1
        # Variables: [h(0), ..., h(n-1), t]
        # c = [0, ..., 0, 1]
        c = np.zeros(n + 1)
        c[-1] = 1.0
        
        # Inequalities: A_eff @ h - t ≤ 0
        A_ub = np.hstack([A_eff, -np.ones((2*n-1, 1))])
        b_ub = np.zeros(2*n-1)
        
        # Equality: sum(h) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(0, 1)] * n + [(0, None)]
        
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 30})
        if res.success:
            h_new = res.x[:n]
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_feasible(h_new, n)
            c5_now = compute_c5(h_new)
            if c5_now < best_c5 - 1e-10:
                best_c5 = c5_now
                best_h = h_new.copy()
            h = h_new
        
        # g-step (optional, might not help due to symmetry)
        # Actually, since 1-h is also optimal, the g-step is redundant.
        # Skip it.
        
        # Check for convergence
        c5_now = compute_c5(h)
        if abs(c5_now - best_c5) < 1e-8:
            break
    
    return best_h, best_c5
```
