    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def soft_gradient(h, T, top_k=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        shift = k_max - (n - 1)
        return subgrad_step(h, shift) * dx
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-12:
            break
        shift = k_idx - (n - 1)
        grad += w * subgrad_step(h, shift)
    return grad * dx

def adam_optimize(h, n_iter=5000, T_start=60.0, T_end=0.3, lr=0.4, beta1=0.9, beta2=0.999, eps=1e-8, check_every=40, patience=2000):
    n = len(h)
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    best_it = 0
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        grad = soft_gradient(h, T)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h.copy()
                best_it = it + 1
            elif it + 1 - best_it > patience:
                break
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def _obj_func_slsqp(h):
    n = len(h)
    T = _slsqp_state['T']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _cons_fun(h):
    return np.sum(h) - _slsqp_state['n'] / 2.0

def _cons_jac(h):
    return np.ones(_slsqp_state['n'])

_slsqp_state = {'T': 5.0, 'n': 320}

def slsqp_polish(h, T=5.0, maxiter=600):
    n = len(h)
    _slsqp_state['T'] = T
    _slsqp_state['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_obj_func_slsqp, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.arange(old_n) * (2.0 / old_n)
    new_x = np.arange(new_n) * (2.0 / new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def focused_refine(h, n_refine=800, top_k_shifts=10, lr=0.02):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_refine):
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-top_k_shifts:]
        grad = np.zeros(n)
        for k_idx in top_idx:
            shift = k_idx - (n - 1)
            grad += subgrad_step(h, shift)
        grad /= len(top_idx)
        grad *= dx
        m = 0.9 * m + 0.1 * grad
        v = 0.999 * v + 0.001 * (grad * grad)
        m_hat = m / (1.0 - 0.9 ** (it + 1))
        v_hat = v / (1.0 - 0.999 ** (it + 1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = make_feasible(h, n)
        if (it + 1) % 50 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n_init = 480
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        n_ones = n_init // 2
        ones = np.where(h0 == 1)[0]
        zeros = np.where(h0 == 0)[0]
        if len(ones) > n_ones:
            h0[rng.choice(ones, len(ones) - n_ones, replace=False)] = 0.0
        elif len(ones) < n_ones:
            h0[rng.choice(zeros, n_ones - len(ones), replace=False)] = 1.0

    n0 = len(h0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    print(f"Initial: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Adam at initial resolution (high T to explore)
    t_limit = budget_s - 200
    if time.time() - start_time < t_limit:
        h1, c1 = adam_optimize(best_h, n_iter=5000, T_start=60.0, T_end=0.3, lr=0.4, check_every=40, patience=3000)
        if c1 < best_c5:
            best_c5, best_h = c1, h1.copy()
        print(f"Adam1 n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: SLSQP at initial resolution
    if n0 <= 512 and time.time() - start_time < budget_s - 150:
        for T_val in [25.0, 8.0, 3.0]:
            if time.time() - start_time > budget_s - 50:
                break
            hx, cx = slsqp_polish(best_h, T=T_val, maxiter=500)
            if cx < best_c5:
                best_c5, best_h = cx, hx
        print(f"SLSQP n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: Upsample and optimize at higher resolution
    target_ns = []
    if n0 <= 320:
        target_ns = [480, 640]
    elif n0 <= 480:
        target_ns = [640, 800]
    else:
        target_ns = [n0 * 2]

    for new_n in target_ns:
        if time.time() - start_time > budget_s - 150:
            break
        if new_n <= n0:
            continue
        h_up = upsample_h(best_h, new_n)
        c_up = compute_c5(h_up)
        # Adam at new resolution
        if time.time() - start_time < budget_s - 80:
            n_iter_new = 4000 if new_n <= 512 else 2500
            h_g, c_g = adam_optimize(h_up, n_iter=n_iter_new, T_start=50.0, T_end=0.3, lr=0.3, check_every=50, patience=2000)
            if c_g < c_up:
                h_up = h_g
                c_up = c_g
        # SLSQP at new resolution if manageable
        if new_n <= 512 and time.time() - start_time < budget_s - 60:
            hx, cx = slsqp_polish(h_up, T=5.0, maxiter=400)
            if cx < c_up:
                h_up = hx
                c_up = cx
        c_up = compute_c5(h_up)
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up.copy()
        print(f"Res {new_n}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 4: Multi-start exploration
    if time.time() - start_time < budget_s - 120:
        n_pts = len(best_h)
        # Different structured starts
        structs = []
        # 1100
        p1 = np.zeros(n_pts); p1[0:n_pts:4] = 1.0; p1[1:n_pts:4] = 1.0; structs.append(p1)
        # 1001
        p2 = np.zeros(n_pts); p2[0:n_pts:4] = 1.0; p2[3:n_pts:4] = 1.0; structs.append(p2)
        # 111000
        p3 = np.zeros(n_pts); p3[0:n_pts:6] = 1.0; p3[1:n_pts:6] = 1.0; p3[2:n_pts:6] = 1.0; structs.append(p3)
        # 110100
        p4 = np.zeros(n_pts); 
        for i in range(0, n_pts, 6):
            if i < n_pts: p4[i] = 1.0
            if i+1 < n_pts: p4[i+1] = 1.0
            if i+3 < n_pts: p4[i+3] = 1.0
        structs.append(p4)

        for idx, pat in enumerate(structs):
            if time.time() - start_time > budget_s - 40:
                break
            h_p = make_feasible(pat.astype(float), n_pts)
            h_r, c_r = adam_optimize(h_p, n_iter=2500, T_start=40.0, T_end=0.5, lr=0.35, check_every=60, patience=1500)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  Struct{idx}: C5={best_c5:.6f}", flush=True)

        # Perturbed best
        for pert_scale in [0.04, 0.08, 0.15, 0.25]:
            if time.time() - start_time > budget_s - 30:
                break
            h_pert = best_h + rng.randn(n_pts) * pert_scale
            h_pert = make_feasible(h_pert, n_pts)
            h_r, c_r = adam_optimize(h_pert, n_iter=2000, T_start=25.0, T_end=0.3, lr=0.25, check_every=60, patience=1200)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  Perturb{pert_scale}: C5={best_c5:.6f}", flush=True)

        # Random binary start
        if time.time() - start_time < budget_s - 60:
            h_rand = (rng.rand(n_pts) < 0.5).astype(float)
            n_ones = n_pts // 2
            ones_idx = np.where(h_rand == 1)[0]
            zeros_idx = np.where(h_rand == 0)[0]
            if len(ones_idx) > n_ones:
                h_rand[rng.choice(ones_idx, len(ones_idx) - n_ones, replace=False)] = 0.0
            elif len(ones_idx) < n_ones:
                h_rand[rng.choice(zeros_idx, n_ones - len(ones_idx), replace=False)] = 1.0
            h_r, c_r = adam_optimize(h_rand, n_iter=3000, T_start=50.0, T_end=0.5, lr=0.35, check_every=50, patience=2000)
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r.copy()
                print(f"  RandomBinary: C5={best_c5:.6f}", flush=True)

        print(f"MultiStart: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 5: Focused refinement on worst shifts
    if time.time() - start_time < budget_s - 40:
        h_f, c_f = focused_refine(best_h, n_refine=2000, top_k_shifts=12, lr=0.015)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
        print(f"Focused: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 6: Final fine Adam
    if time.time() - start_time < budget_s - 25:
        n_pts = len(best_h)
        h_fin, c_fin = adam_optimize(best_h, n_iter=3000, T_start=4.0, T_end=0.05, lr=0.08, check_every=30, patience=1500)
        if c_fin < best_c5:
            best_c5, best_h = c_fin, h_fin.copy()
        print(f"FinalAdam: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 7: Final SLSQP
    n_pts = len(best_h)
    if n_pts <= 512 and time.time() - start_time < budget_s - 10:
        for T_val in [4.0, 1.5]:
            if time.time() - start_time > budget_s - 5:
                break
            hx, cx = slsqp_polish(best_h, T=T_val, maxiter=300)
            if cx < best_c5:
                best_c5, best_h = cx, hx
        print(f"FinalSLSQP: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
