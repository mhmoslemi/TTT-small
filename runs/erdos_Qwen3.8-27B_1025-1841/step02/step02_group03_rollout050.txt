from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_vec(h):
    return np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        mask = (h < 1.0) if diff > 0 else (h > 0.0)
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    c = np.correlate(x, 1.0 - x, mode='full')

    return np.sum(x) - len(x) / 2.0

def _con_jac(x):
    return np.ones(len(x))

    if candidates is None:
        candidates = [0.0, 1.0, 0.25, 0.5, 0.75]

    for it in range(iters):
        improved = False

            old = h[j]
            best_c_j = best_c
            best_val = old

                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c_j - 1e-14:
                    best_c_j = c
                    best_val = cand
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_c_j
                improved = True
        if not improved:
            break
    return h, best_c

def subgradient_opt(h, n, steps, lr_start, lr_end, T_soft=50.0):
    """Subgradient method with soft weights."""
    h = h.copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    for s in range(steps):
        lr = lr_start * (lr_end / lr_start) ** (s / max(steps-1, 1))
        corr = corr_vec(h)
        kmax = np.max(corr)
        shifted = corr - kmax
        weights = np.exp(T_soft * shifted)
        total_w = np.sum(weights)
        if total_w > 0:
            weights /= total_w
        
        g = np.zeros(n)
        # Only process shifts with significant weight
        active = weights > 1e-6
        for k_idx in np.where(active)[0]:
            w = weights[k_idx]
            delta = k_idx - (n - 1)
            if delta == 0:
                g += w * (1.0 - 2.0 * h)
            elif delta > 0:
                g[delta:] += w * (1.0 - h[:-delta])
                g[:-delta] -= w * h[delta:]
            else:
                d = -delta
                g[:-d] += w * (1.0 - h[d:])
                g[d:] -= w * h[:-d]
        
        h = h - lr * g
        h = make_feasible(h, n)
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c
