import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_at_shift(h, shift):
    """Compute subgradient of correlation at a specific shift."""
    n = len(h)
    s = abs(int(shift))
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def targeted_gd(h, n_iter, top_k, lr_start, lr_end, t_start, t_end):
    """GD focusing on top-k correlations with softmax smoothing."""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = t_start * (t_end / t_start) ** frac
        lr = lr_start * (lr_end / lr_start) ** frac
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        exp_s = np.exp(T * shifted)
        total = np.sum(exp_s)
        
        if total < 1e-300 or not np.isfinite(total):
            k_max = int(np.argmax(corr))
            grad = subgrad_at_shift(h, k_max - (n - 1))
        else:
            weights = exp_s / total
            # Focus on top-k
            idx = np.argsort(weights)[-top_k:]
            grad = np.zeros(n)

                w = weights[k_idx]
                if w < 1e-10:
                    continue
                shift = k_idx - (n - 1)

                best_h = h.copy()

        best_h = h.copy()
    return best_h, best_c5

def binary_sa(h_init, n_iters, T_start, T_min):
    """Simulated annealing for binary patterns."""
    n = len(h_init)
    h = np.round(h_init)
    h = make_feasible(h, n)
    # Ensure exactly n/2 ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    target_ones = n // 2
    rng = np.random.RandomState(0)
    while len(ones) > target_ones:
        idx = ones[rng.randint(len(ones))]
        h[idx] = 0
        ones = np.where(h == 1)[0]
    while len(ones) < target_ones:
        idx = zeros[rng.randint(len(zeros))]
        h[idx] = 1
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    cur_c5 = best_c5
    
    for it in range(n_iters):
        T = T_start * (T_min / T_start) ** (it / n_iters)
        if T < T_min:
            T = T_min
        
        # Random swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i1 = ones[rng.randint(len(ones))]
        i0 = zeros[rng.randint(len(zeros))]
        
        h[i1] = 0
        h[i0] = 1
        new_c5 = compute_c5(h)
        
        if new_c5 < cur_c5 or rng.rand() < np.exp(-(new_c5 - cur_c5) / T):
            cur_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            h[i1] = 1
            h[i0] = 0
    
    return best_h, best_c5

def soft_max_obj(h, T):
    """Softmax approximation to max correlation * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def soft_max_grad(h, T, top_k=50):
    """Gradient of softmax approximation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return subgrad_at_shift(h, k_max - (n - 1)) * dx
    
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-10:
            continue
        shift = k_idx - (n - 1)
        grad += w * subgrad_at_shift(h, shift)
    return grad * dx

def slsqp_optimize(h, T, maxiter=500):
    """SLSQP optimization with softmax objective."""
    n = len(h)
    dx = 2.0 / n
    
    def obj(v):
        return soft_max_obj(v, T)
    
    def grad(v):
        return soft_max_grad(v, T, top_k=30)
    
    def con_fun(v):
        return np.sum(v) - n / 2.0
    
    def con_jac(v):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()

    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_shift(h, shift):
    n = len(h)
    s = abs(int(shift))
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def softmax_grad(h, T, top_k=50):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return subgrad_shift(h, k_max - (n - 1)) * dx
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for k_idx in idx:
        w = weights[k_idx]
        if w < 1e-12:
            break
        grad += w * subgrad_shift(h, k_idx - (n - 1))
    return grad * dx

def prob_gd(h, n_iter, T0, T1, lr0, lr1, top_k=50, use_bb=True):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    prev_h = h.copy()
    prev_grad = None
    h_g = None
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T0 * (T1 / T0) ** frac
        g = softmax_grad(h, T, top_k)
        if use_bb and prev_grad is not None:
            sh = h - prev_h
            denom = np.dot(sh, sh) + 1e-30
            num = np.dot(sh, g - prev_grad)
            bb_lr = num / denom if num > 0 else lr0 * (lr1 / lr0) ** frac
            lr_t = max(0.001, min(0.5, bb_lr))
        else:
            lr_t = lr0 * (lr1 / lr0) ** frac
        h_new = h - lr_t * g
        h_new = make_feasible(h_new, n)
        prev_h = h.copy()
        h = h_new
        prev_grad = g.copy()
        if (it + 1) % 40 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:

    c5 = compute_c5(h)

    return best_h, best_c5

def slsqp_run(h, T, maxiter=500, top_k=30):
    n = len(h)
    dx = 2.0 / n
    def obj(v):
        corr = np.correlate(v, 1.0 - v, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def grad(v):
        return softmax_grad(v, T, top_k)
    def cfun(v):
        return np.sum(v) - n / 2.0
    def cjac(v):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
    try:
        res = minimize(obj, h, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)

    old_n = len(h)
    if new_n <= old_n:
        old_x = np.linspace(0.0, 2.0, old_n)
        new_x = np.linspace(0.0, 2.0, new_n)
        return make_feasible(np.interp(new_x, old_x, h), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)
