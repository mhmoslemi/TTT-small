        improved = False
        for i in range(n):

                if time.time() - start_time > time_limit:
                    return h

            if old_val < 1e-15:

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

        t = 50.0

        soft = mc + np.log(np.sum(np.exp(t * shifted))) / t

    def constr_jac(hv):
        return np.ones(n)

                h_r = rng.rand(n_r)
                h_r = make_feasible(h_r, n_r)

            h_r = coord_descent(h_r, n_r, max_passes=15, time_limit=budget_s - 20, start_time=start_time)
            c_r = compute_c5(h_r)
            update(h_r, c_r)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
