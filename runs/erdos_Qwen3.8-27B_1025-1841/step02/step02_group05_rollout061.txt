```python
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    f = corr * dx  # f[k] for k=0,...,2n-2, shift = k-n+1
    
    if tau is None:
        # Exact max
        max_idx = np.argmax(f)
        max_val = f[max_idx]
        shift = max_idx - n + 1
        grad = np.zeros(n)
        # For each valid j in [max(0,shift), min(n, n+shift)):
        j0 = max(0, shift)
        j1 = min(n, n + shift)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0+shift:j1+shift])
        # For the "other" part:
        j0b = max(0, -shift)
        j1b = min(n, n - shift)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b+shift:j1b+shift]
        grad *= dx
        return max_val, grad
    else:
        # Smooth max
        shifted = tau * (f - f.max())
        shifted = np.clip(shifted, -500, 500)
        exp_f = np.exp(shifted)
        p = exp_f / exp_f.sum()
        obj = dx * (f.max() + np.log(exp_f.sum()) / tau)
        grad = np.zeros(n)

        return obj, grad
```

```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, tau, top_k=20):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    f = corr * dx
    max_f = f.max()
    
    # Softmax with top-k truncation
    active = np.argsort(f)[::-1][:top_k]
    active_f = f[active]
    shifted = tau * (active_f - max_f)
    shifted = np.clip(shifted, -100, 0)
    exp_vals = np.exp(shifted)
    # Also include the "rest" as a single term at 0 (negligible)
    p = exp_vals / exp_vals.sum()
    
    obj = max_f + np.log(np.exp(tau*max_f - tau*max_f).sum() / (1.0)) / tau
    # Actually: log-sum-exp of all f / tau
    # log(exp(tau*f)/tau sum for all k) = max_f + log(sum(exp(tau*(f-max_f))))/tau
    # ≈ max_f + log(sum_top_k(exp(tau*(f_top-max_f))))/tau (if rest is negligible)
    obj = max_f + np.log(exp_vals.sum()) / tau
    
    grad = np.zeros(n)
    for idx, k in enumerate(active):
        pk = p[idx]
        if pk < 1e-12:
            continue
        shift = k - n + 1
        j0 = max(0, shift)
        j1 = min(n, n + shift)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0+shift:j1+shift])
        j0b = max(0, -shift)
        j1b = min(n, n - shift)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b+shift:j1b+shift]
    grad *= dx
    return obj, grad, f
```

```python
        frac = min(1.0, (time.time() - t0) / tlimit)
        tau = tau0 + (tau1 - tau0) * frac  # linear schedule

        if fmax < best:

            bh = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            lr *= 0.5
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        
        m = b1 * m + (1 - b1) * g
        v = b2 * v + (1 - b2) * g * g
        mh = m / (1.0 - b1**t)
        vh = v / (1.0 - b2**t)
        h = project_to_feasible(h - lr * mh / (np.sqrt(vh) + eps), n)
    
    return bh, best

def multi_shift_subgrad(h, n, tlimit, lr0=0.003, topk=5):
    h = project_to_feasible(h.copy(), n)
    best = compute_c5(h)
    bh = h.copy()
    t0 = time.time()
    dx = 2.0 / n
    lr = lr0
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    
    while time.time() - t0 < tlimit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = corr.max() * dx
        if c5 < best:
            best = c5
            bh = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break
        
        idx = np.argpartition(corr, -topk)[-topk:]
        tv = corr[idx]
        w = np.exp(tv - tv[0])
        w /= w.sum()
        
        grad = np.zeros(n)
        for i in range(len(idx)):
            s = idx[i] - n + 1
            a = max(0, s)
            b = min(n, n + s)
            if a < b:
                grad[a:b] += w[i] * (1.0 - h[a+s:b+s])
            a2 = max(0, -s)
            b2 = min(n, n - s)
            if a2 < b2:
                grad[a2:b2] -= w[i] * h[a2+s:b2+s]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return bh, best
```

43. Actually, let me reconsider the Adam optimizer. There's a subtlety: when we project to feasible set after each step, the effective step size is reduced (the projection clips the update). This means Adam's bias correction (dividing by 1-beta^t) might not be appropriate, since the effective updates are smaller than what Adam expects.

Alternative: instead of projecting after each step, use a "barrier" method where the [0,1] constraint is handled via the sigmoid parameterization. Let me try this:

h_j = sigmoid(a_j) = 1/(1+exp(-a_j))
The gradient of f_k w.r.t. a_j is:
∂f_k/∂a_j = ∂f_k/∂h_j * h_j(1-h_j)

And the sum constraint: we need ∑ sigmoid(a_j) = n/2. We can enforce this by periodically adjusting all a_j by a common shift (finding the shift that makes the sum = n/2).

This avoids the projection issue. Let me implement this:

```python
def adam_sigmoid(h_init, n, tlimit, tau0=60.0, tau1=3.0, lr0=0.5, top_k=40):
    # Convert h to a-space: a = logit(h)
    eps_a = 1e-6
    h_safe = np.clip(h_init, eps_a, 1-eps_a)
    a = np.log(h_safe / (1 - h_safe))
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    t0 = time.time()
    b1, b2, eps = 0.9, 0.999, 1e-8
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    lr = lr0
    
    while time.time() - t0 < tlimit:
        t += 1
        frac = min(1.0, (time.time()-t0)/tlimit)
        tau = tau0 + (tau1-tau0)*frac
        
        h = 1.0/(1.0+np.exp(-a))
        # Enforce sum constraint: shift a so that sum(h) = n/2
        # Find shift delta: sum(1/(1+exp(-(a+delta)))) = n/2
        lo, hi = -5.0, 5.0
        for _ in range(40):
            mid = (lo+hi)/2
            s = np.sum(1.0/(1.0+np.exp(-(a+mid))))
            if s > n/2:
                lo = mid
            else:
                hi = mid
        delta = (lo+hi)/2
        a += delta
        h = 1.0/(1.0+np.exp(-a))
        
        # Compute smooth max and gradient w.r.t. a
        dx = 2.0/n
        corr = np.correlate(h, 1.0-h, mode='full')
        f = corr * dx
        fmax = f.max()
        
        if fmax < best_c5:
            best_c5 = fmax
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 500:
            lr *= 0.5
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        
        kk = np.argpartition(f, -top_k)[-top_k:]
        af = f[kk]
        sh = np.clip(tau*(af-fmax), -200, 0)
        ev = np.exp(sh)
        p = ev/ev.sum()
        
        # Gradient of smooth max w.r.t. h
        gh = np.zeros(n)
        for i in range(len(kk)):
            pi = p[i]
            if pi < 1e-13: continue
            k = kk[i]
            s = k - n + 1
            aa = max(0, s)
            bb = min(n, n+s)
            if aa < bb:
                gh[aa:bb] += pi*(1.0-h[aa+s:bb+s])
            aa2 = max(0, -s)
            bb2 = min(n, n-s)
            if aa2 < bb2:
                gh[aa2:bb2] -= pi*h[aa2+s:bb2+s]
        gh *= dx
        
        # Chain rule: grad w.r.t. a = grad w.r.t. h * h*(1-h)
        ga = gh * h * (1.0-h)
        
        m = b1*m + (1-b1)*ga
        v = b2*v + (1-b2)*ga*ga
        mh = m/(1.0-b1**t)
        vh = v/(1.0-b2**t)
        a = a - lr * mh/(np.sqrt(vh)+eps)
    
    return best_h, best_c5
```

45. Let me also experiment with a different optimization strategy: **alternating minimization over blocks**.

Divide h into B blocks of size B_s = n/B. In each iteration, optimize one block (treating others as fixed). The block optimization is a smaller problem (B_s variables) that can be solved more precisely.

For a single block of size B_s, the objective is:
max_k [C_k + (block terms involving the block variables)]

This is still a max of bilinear functions, but with only B_s free variables. We can solve it with a simple 1D grid search (if B_s = 1) or a 2D grid (if B_s = 2), or a small-scale Newton method.

For B_s = 2: we optimize (h_j, h_{j+1}) with constraint h_j + h_{j+1} = c (fixed). This is a 1D problem: set h_j = t, h_{j+1} = c-t, and minimize max_k f_k(t) over t ∈ [max(0,c-1), min(1,c)]. The function max_k f_k(t) is piecewise quadratic in t, and the minimum can be found by checking the breakpoints.

This could be a powerful polishing step. But implementing it efficiently for all blocks is non-trivial. Let me skip it for now.
