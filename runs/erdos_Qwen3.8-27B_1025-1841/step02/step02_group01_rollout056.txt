from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)

    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0, 2, old_n, endpoint=False)
    new_x = np.linspace(0, 2, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _soft_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    return float(m + np.log(np.sum(np.exp(T * (corr - m)))) / T)

def _lag_obj_factory(n, k_star):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        return float(corr[k_star] * dx)
    return obj

def _con_fun_factory(n):
    target = n / 2.0
    def f(h):
        return np.sum(h) - target
    return f

def _con_jac_factory(n):
    def j(h):
        return np.ones(n)
    return j

def run(seed=42, budget_s=1300, **kwargs):
    start = time.time()
    rng = np.random.RandomState(seed)

    def time_left():
        return budget_s - (time.time() - start)

    # Get initial solution
    try:
        best_h = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        best_h = None
    if best_h is None:
        n_init = 320
        best_h = np.zeros(n_init)
        best_h[:n_init // 2] = 1.0
        rng.shuffle(best_h)

    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP with soft-max at multiple resolutions/temperatures
    configs = [
        (512, 5, 400),
        (512, 15, 300),
        (640, 5, 300),
        (384, 3, 400),
        (512, 3, 300),
    ]

    for n_pts, T, maxit in configs:
        if time_left() < 45:
            break
        n = n_pts
        h_start = upsample_h(best_h, n)
        # Small perturbation for diversity
        h_start = h_start + rng.randn(n) * 0.02
        h_start = make_feasible(h_start, n)

        con_fun = _con_fun_factory(n)
        con_jac = _con_jac_factory(n)

        try:
            res = minimize(lambda h: _soft_obj(h, n, T), h_start, method='SLSQP',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                           options={'maxiter': maxit, 'ftol': 1e-14})
            c = compute_c5(res.x)
            if c < best_c5:
                best_c5 = c
                best_h = res.x.copy()
        except Exception:
            pass

        # Second pass: worst-lag focused refinement
        if time_left() < 30:
            break
        n = len(best_h)
        for _wl in range(5):
            if time_left() < 20:
                break
            dx = 2.0 / n
            corr = np.correlate(best_h, 1.0 - best_h, mode='full')
            k_star = int(np.argmax(corr))
            lag_obj = _lag_obj_factory(n, k_star)
            try:
                res2 = minimize(lag_obj, best_h, method='SLSQP',
                                bounds=[(0.0, 1.0)] * n,
                                constraints=[{'type': 'eq', 'fun': con_fun, 'jac': con_jac}],
                                options={'maxiter': 200, 'ftol': 1e-14})
                c2 = compute_c5(res2.x)
                if c2 < best_c5:
                    best_c5 = c2
                    best_h = res2.x.copy()
            except Exception:
                break

    # Phase 2: Coordinate descent with 1D golden-section search on active coordinates
    n = len(best_h)
    h_cd = make_feasible(best_h.copy(), n)
    c_cd = compute_c5(h_cd)
    if c_cd > best_c5:
        h_cd = best_h.copy()
        c_cd = best_c5

    max_sw = 60
    for sweep in range(max_sw):
        if time_left() < 15:
            break
        improved = False
        # Focus on coordinates that appear in the worst few lags
        corr = np.correlate(h_cd, 1.0 - h_cd, mode='full')
        top_k = int(np.max(corr) * (2.0 / n))
        k_star = int(np.argmax(corr))
        # Active indices: those involved in top 5 worst lags
        worst_lags = np.argsort(corr)[-5:]
        active = set()
        for kl in worst_lags:
            lag = kl - (n - 1)
            lo = max(0, -lag)
            hi = min(n, n - abs(lag))
            active.update(range(lo, hi))
        active = sorted(active)
        if len(active) == 0:
            active = list(range(n))
        # Limit to at most 80 coordinates per sweep for speed
        if len(active) > 80:
            rng.shuffle(active)
            active = active[:80]

        for j in active:
            if time_left() < 15:
                break
            old_val = h_cd[j]
            # 1D search: find best value for h[j]
            def obj_1d(v):
                trial = h_cd.copy()
                trial[j] = v
                trial = make_feasible(trial, n)
                return compute_c5(trial)

            try:
                res1d = minimize_scalar(obj_1d, bounds=(0.0, 1.0),
                                        method='bounded',
                                        options={'xatol': 1e-8, 'maxiter': 30})
                if res1d.success and res1d.fun < c_cd - 1e-12:
                    h_cd[j] = res1d.x
                    h_cd = make_feasible(h_cd, n)
                    c_cd = compute_c5(h_cd)
                    improved = True
            except Exception:
                # Fallback: try a few candidate values
                for cv in [0.0, 1.0, 0.5, old_val * 0.5, (old_val + 1.0) * 0.5]:
                    if cv < 0 or cv > 1 or cv == old_val:
                        continue
                    trial = h_cd.copy()
                    trial[j] = cv
                    trial = make_feasible(trial, n)
                    cv_c5 = compute_c5(trial)
                    if cv_c5 < c_cd:
                        h_cd = trial
                        c_cd = cv_c5
                        improved = True

        if c_cd < best_c5:
            best_c5 = c_cd
            best_h = h_cd.copy()
        if not improved:
            break

    # Phase 3: Binary local search (swap-based)
    n_b = len(best_h)
    h_bin = (best_h > 0.5).astype(float)
    n_ones = int(np.sum(h_bin))
    target_ones = n_b // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1)[0]
        h_bin[ones_idx[-(n_ones - target_ones):]] = 0
    elif n_ones < target_ones:
        zeros_idx = np.where(h_bin == 0)[0]
        h_bin[zeros_idx[:target_ones - n_ones]] = 1

    c_bin = compute_c5(h_bin)
    h_ls = best_h.copy()
    c_ls = best_c5
    if c_bin < best_c5:
        c_ls = c_bin
        h_ls = h_bin.copy()
        best_c5 = c_bin
        best_h = h_bin.copy()

    # Swap-based local search on the best solution
    # For each position, try swapping with a few others
    n_s = len(h_ls)
    for ls_sweep in range(30):
        if time_left() < 10:
            break
        improved = False
        order = rng.permutation(n_s)
        for j in order:
            if time_left() < 10:
                break
            # Try a few random swap partners
            for _ in range(3):
                k = rng.randint(n_s)
                if k == j or h_ls[j] == h_ls[k]:
                    continue
                trial = h_ls.copy()
                trial[j], trial[k] = trial[k], trial[j]
                c_tr = compute_c5(trial)
                if c_tr < c_ls - 1e-12:
                    h_ls = trial
                    c_ls = c_tr
                    improved = True
                    if c_tr < best_c5:
                        best_c5 = c_tr
                        best_h = h_ls.copy()
                    break
        if not improved:
            break

    # Phase 4: Final random perturbation search
    n_f = len(best_h)
    h_final = best_h.copy()
    c_final = best_c5
    for it in range(2000):
        if time_left() < 5:
            break
        h_try = h_final.copy()
        n_pert = rng.randint(3, 30)
        idxs = rng.choice(n_f, size=n_pert, replace=False)
        h_try[idxs] += rng.randn(n_pert) * rng.uniform(0.01, 0.08)
        h_try = make_feasible(h_try, n_f)
        c_try = compute_c5(h_try)
        if c_try < c_final:
            h_final = h_try
            c_final = c_try
            if c_final < best_c5:
                best_c5 = c_final
                best_h = h_final.copy()

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
