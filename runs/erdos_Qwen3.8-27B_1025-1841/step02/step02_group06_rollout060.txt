    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5_full(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr


def grad_at_lag(h, d, n):
    g = np.zeros(n)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[d:] += (1.0 - h[:n - d])
        g[:n - d] -= h[d:]
    else:
        dd = -d
        g[:n - dd] += (1.0 - h[dd:])
        g[dd:] -= h[:n - dd]
    return g


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def fw_optimize(h, n, time_limit, topM=3, tau_init=10.0, fw_every=10000,
                check_every=300, lr_init=0.4):
    dx = 2.0 / n
    h = h.copy()
    best_c5, corr = compute_c5_full(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.92
    step = 0
    no_improve = 0
    patience = check_every * 30

    while time.time() - start < time_limit:
        top_idx = np.argpartition(corr, -topM)[-topM:]
        active_d = (top_idx - (n - 1)).astype(int)
        corr_top = corr[top_idx]
        max_c = corr_top.max()
        tau = min(tau_init * (1.0 + step * 0.0005), 200.0)
        p = np.exp(tau * (corr_top - max_c))
        p /= p.sum()

        grad = np.zeros(n)
        for idx in range(topM):
            pk = p[idx]
            if pk < 1e-12:
                continue
            grad += pk * grad_at_lag(h, int(active_d[idx]), n)

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        step += 1
        if step % check_every == 0:
            c5_cur, corr = compute_c5_full(h_new)
            if c5_cur < best_c5 - 1e-16:
                best_c5 = c5_cur
                best_h = h_new.copy()
                no_improve = 0
            else:
                no_improve += check_every

            if no_improve > patience:
                lr *= 0.6
                v *= 0.3
                no_improve = 0
                if lr < 1e-10:
                    break

            if step % fw_every == 0:
                gamma = min(0.3, 2.0 / (step // fw_every + 2))
                sorted_idx = np.argsort(grad)
                h_fw = np.zeros(n)
                h_fw[sorted_idx[:n // 2]] = 1.0
                h_new = (1.0 - gamma) * h_new + gamma * h_fw
                h_new = project_to_feasible(h_new, n)
                c5_fw, corr = compute_c5_full(h_new)
                if c5_fw < best_c5:
                    best_c5 = c5_fw
                    best_h = h_new.copy()

        h = h_new
        if step % 2000 == 0:
            lr = max(lr_init / (1.0 + step * 0.0003), 0.02)

    c5_final, _ = compute_c5_full(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def binary_local_search(h, n, time_limit, n_trials=30000):
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    ones_count = int(h_b.sum())
    if ones_count > target:
        idx = np.where(h_b == 1)[0]
        np.random.shuffle(idx)
        h_b[idx[target:]] = 0
    elif ones_count < target:
        idx = np.where(h_b == 0)[0]
        np.random.shuffle(idx)
        h_b[idx[:target - ones_count]] = 1

    dx = 2.0 / n
    best_c5 = compute_c5_full(h_b)[0]
    best_h = h_b.copy()
    start = time.time()
    trial = 0
    no_improve = 0
    while time.time() - start < time_limit and trial < n_trials:
        if no_improve > 20000:
            break
        i = np.random.randint(n)
        j = np.random.randint(n)
        if h_b[i] == h_b[j]:
            continue
        h_b[i] = 1.0 - h_b[i]
        h_b[j] = 1.0 - h_b[j]
        trial += 1
        c5_new = compute_c5_full(h_b)[0]
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_b.copy()
            no_improve = 0
        else:
            h_b[i] = 1.0 - h_b[i]
            h_b[j] = 1.0 - h_b[j]
            no_improve += 1
    return best_h, best_c5


def directed_swap_search(h, n, time_limit, n_trials=20000):
    dx = 2.0 / n
    h_b = (h >= 0.5).astype(float)
    target = n // 2
    ones_count = int(h_b.sum())
    if ones_count > target:
        idx = np.where(h_b == 1)[0]
        np.random.shuffle(idx)
        h_b[idx[target:]] = 0
    elif ones_count < target:
        idx = np.where(h_b == 0)[0]
        np.random.shuffle(idx)
        h_b[idx[:target - ones_count]] = 1

    best_c5, corr = compute_c5_full(h_b)
    best_h = h_b.copy()
    start = time.time()
    trial = 0
    no_improve = 0
    while time.time() - start < time_limit and trial < n_trials:
        if no_improve > 15000:
            break
        worst_k = corr.argmax()
        d = worst_k - (n - 1)
        # Find positions where h[i]=1, h[i+d]=0 (bad at lag d)
        if d > 0:
            candidates = np.where((h_b[:n - d] == 1) & (h_b[d:] == 0))[0]
        elif d < 0:
            dd = -d
            candidates = np.where((h_b[dd:] == 1) & (h_b[:n - dd] == 0))[0]
        else:
            candidates = np.where(h_b == 0)[0]

        if len(candidates) == 0:
            candidates = np.where(h_b == 1)[0]

        # Try to fill a gap: find i where h[i]=0, h[i+d]=1 and swap h[i] with a random 1
        if d > 0:
            fill_idx = np.where((h_b[:n - d] == 0) & (h_b[d:] == 1))[0]
            if len(fill_idx) > 0:
                i = fill_idx[np.random.randint(len(fill_idx))]
                j_pos = i + d
            else:
                i = candidates[np.random.randint(len(candidates))]
                j_pos = i
        elif d < 0:
            dd = -d
            fill_idx = np.where((h_b[dd:] == 0) & (h_b[:n - dd] == 1))[0]
            if len(fill_idx) > 0:
                i = fill_idx[np.random.randint(len(fill_idx))] + dd
                j_pos = i - dd
            else:
                i = candidates[np.random.randint(len(candidates))] + dd
                j_pos = i
        else:
            i = candidates[0]
            j_pos = i

        ones = np.where(h_b == 1)[0]
        if len(ones) == 0:
            break
        j = ones[np.random.randint(len(ones))]
        if j == i:
            continue

        h_b[i] = 1.0
        h_b[j] = 0.0
        trial += 1
        c5_new, corr = compute_c5_full(h_b)
        if c5_new < best_c5 - 1e-15:
            best_c5 = c5_new
            best_h = h_b.copy()
            no_improve = 0
        else:
            h_b[i] = 0.0
            h_b[j] = 1.0
            no_improve += 1
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5, _ = compute_c5_full(h_init)
    best_h = h_init.copy()

    # Phase 1: Broad optimization at initial resolution
    if remaining() > 350:
        tl = min(remaining() * 0.30, 400.0)
        h1, c5_1 = fw_optimize(best_h, n_init, tl, topM=5, tau_init=8.0,
                               fw_every=8000, check_every=200, lr_init=0.35)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Focused optimization (fewer active lags, higher tau)
    if remaining() > 250:
        tl = min(remaining() * 0.25, 300.0)
        h2, c5_2 = fw_optimize(best_h, n_init, tl, topM=2, tau_init=40.0,
                               fw_every=12000, check_every=300, lr_init=0.25)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Single-lag focus with very high tau
    if remaining() > 150:
        tl = min(remaining() * 0.20, 250.0)
        h3, c5_3 = fw_optimize(best_h, n_init, tl, topM=1, tau_init=100.0,
                               fw_every=15000, check_every=400, lr_init=0.15)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Try higher resolution
    n2 = n_init * 2
    if remaining() > 250:
        h_h = interp_periodic(best_h, n2)
        h_h = project_to_feasible(h_h, n2)
        tl = min(remaining() * 0.30, 350.0)
        h4, c5_4 = fw_optimize(h_h, n2, tl, topM=4, tau_init=10.0,
                               fw_every=6000, check_every=200, lr_init=0.3)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
        else:
            h_back = interp_periodic(h4, n_init)
            h_back = project_to_feasible(h_back, n_init)
            c5_back, _ = compute_c5_full(h_back)
            if c5_back < best_c5:
                best_c5 = c5_back
                best_h = h_back.copy()

    # Phase 5: Binary local search (random swaps)
    if remaining() > 80:
        n_f = len(best_h)
        tl = min(remaining() * 0.15, 100.0)
        h5, c5_5 = binary_local_search(best_h, n_f, tl, n_trials=15000)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()

    # Phase 6: Directed swap search
    if remaining() > 60:
        n_f = len(best_h)
        tl = min(remaining() * 0.12, 80.0)
        h6, c5_6 = directed_swap_search(best_h, n_f, tl, n_trials=10000)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Final FW polish at best resolution
    if remaining() > 50:
        n_f = len(best_h)
        tl = min(remaining() * 0.30, 120.0)
        h7, c5_7 = fw_optimize(best_h, n_f, tl, topM=2, tau_init=50.0,
                               fw_every=20000, check_every=400, lr_init=0.1)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5, _ = compute_c5_full(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
