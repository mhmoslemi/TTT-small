        else:
            stall += 1
        top_idx = np.argsort(corr)[-topk:]
        grad = np.zeros(n)
        for k_star in top_idx:
            sh = k_star - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + sh]
        grad *= dx
        h = project_to_feasible(h - lr * grad, n)
        if stall > 2000:
            lr *= 0.7
            stall = 0
            if lr < 1e-16:
                break
    return best_h, best_c5


def coord_descent_polish(h, n_sweeps=3, time_limit=30.0):
    n = len(h)
    start = time.time()
    best_c5 = compute_c5(h)
    h = h.copy()
    for sweep in range(n_sweeps):
        if time.time() - start > time_limit:
            break
        for i in range(n):
            if time.time() - start > time_limit:
                break
            old_val = h[i]
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                h[i] = np.clip(old_val + delta, 0.0, 1.0)
                h = project_to_feasible(h, n)
                c5_try = compute_c5(h)
                if c5_try < best_c5 - 1e-12:
                    best_c5 = c5_try
                    old_val = h[i]
                else:
                    h[i] = old_val
                    h = project_to_feasible(h, n)
        h[i] = old_val
    return h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Optimize at native resolution with full softmax, moderate tau
    if remaining() > 60:
        h1 = best_h.copy()
        h1, _ = adam_smooth(h1, tau=40.0, max_iters=30000, time_limit=min(remaining() * 0.2, 100.0), lr=0.08, topk=0)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Sharper tau with top-k restriction
    if remaining() > 50:
        h2 = best_h.copy()
        h2, _ = adam_smooth(h2, tau=20.0, max_iters=30000, time_limit=min(remaining() * 0.2, 80.0), lr=0.05, topk=10)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Very sharp, top-5
    if remaining() > 40:
        h3 = best_h.copy()
        h3, _ = adam_smooth(h3, tau=12.0, max_iters=30000, time_limit=min(remaining() * 0.15, 60.0), lr=0.03, topk=5)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Subgradient with top-5, many small steps
    if remaining() > 30:
        h4 = best_h.copy()
        h4, c5_4 = subgrad_topk(h4, max_iters=500000, time_limit=remaining() * 0.25, lr=0.003, topk=5)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Try higher resolution
    n_hi = n_init * 2
    if remaining() > 80 and n_hi <= 3600:
        h_hi = interp_periodic(best_h, n_hi)
        h_hi = project_to_feasible(h_hi, n_hi)
        h_hi, _ = adam_smooth(h_hi, tau=25.0, max_iters=20000, time_limit=min(remaining() * 0.15, 50.0), lr=0.04, topk=8)
        c5_hi = compute_c5(h_hi)
        if c5_hi < best_c5:
            best_c5 = c5_hi
            best_h = h_hi.copy()
        if remaining() > 30:
            h_hi, c5_hi2 = subgrad_topk(h_hi, max_iters=300000, time_limit=remaining() * 0.2, lr=0.002, topk=5)
            if c5_hi2 < best_c5:
                best_c5 = c5_hi2
                best_h = h_hi.copy()

    # Phase 6: Subgradient at very small lr for final polish
    if remaining() > 20:
        h6 = best_h.copy()
        h6, c5_6 = subgrad_topk(h6, max_iters=500000, time_limit=remaining() * 0.3, lr=0.001, topk=3)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()

    # Phase 7: Coordinate descent polish
    if remaining() > 10:
        h7 = best_h.copy()
        tl_cd = min(remaining() * 0.4, 60.0)
        h7, c5_7 = coord_descent_polish(h7, n_sweeps=4, time_limit=tl_cd)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()

    # Phase 8: Final subgradient micro-steps
    if remaining() > 5:
        h8 = best_h.copy()
        h8, c5_8 = subgrad_topk(h8, max_iters=500000, time_limit=remaining() * 0.5, lr=0.0005, topk=3)
        if c5_8 < best_c5:
            best_c5 = c5_8
            best_h = h8.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
