def _grad_smooth(x, n, T, penalty):
    """Gradient of the smoothed objective"""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    M = np.max(corr)
    # Compute weights (log-sum-exp trick)
    shifted = T * (corr - M)
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_shifted = np.exp(shifted)
    S = np.sum(exp_shifted)
    w = exp_shifted / S  # weights, shape (2n-1,)
    
    # Gradient of sum(x) term
    sum_term = 2.0 * penalty * (np.sum(x) - n / 2.0)
    
    # Compute grad = dx * sum_k w[k] * dcorr[k]/dx[j] + sum_term
    # dcorr[k]/dx[j] where lag d = k - (n-1):
    # = (1-x[j-d])*(0<=j-d<n) - x[j+d]*(0<=j+d<n)
    
    grad = np.zeros(n)
    for d in range(-(n-1), n):
        w_d = w[n - 1 + d]
        if w_d < 1e-15:
            continue
        if d >= 0:
            j_max = n - 1 - d
            if j_max < 0:
                continue
            grad[0:j_max+1] += w_d * ((1.0 - x[0:j_max+1]) - x[d:d+j_max+1])
        else:
            d2 = -d
            j_min = d2
            j_max = n - 1 - d2
            if j_min > j_max:
                continue
            grad[j_min:j_max+1] += w_d * ((1.0 - x[j_min+d2:j_max+1+d2]) - x[j_min-d2:j_max+1-d2])
    
    grad *= dx
    grad += sum_term
    return grad

    exp_s = np.exp(shifted)
    S = np.sum(exp_s)
    w = exp_s / S
    grad = np.zeros(n)
    for d in range(-(n - 1), n):
        wd = w[n - 1 + d]
        if wd < 1e-14:
            continue
        if d >= 0:
            jm = n - 1 - d
            if jm < 0:
                continue
            grad[0:jm+1] += wd * ((1.0 - x[0:jm+1]) - x[d:d+jm+1])
        else:
            d2 = -d
            jn = d2
            jm = n - 1 - d2
            if jn > jm:
                continue
            grad[jn:jm+1] += wd * ((1.0 - x[jn+d2:jm+1+d2]) - x[jn-d2:jm+1-d2])
    return grad * dx

def _con_fun(x):
    return np.sum(x) - _G_N / 2.0

def _con_jac(x):
    return np.ones(_G_N)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(80):
        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def project_feasible(h, n):
    target = n / 2.0
    lo = h.min() - 2.0
    hi = h.max() + 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        g = np.clip(h - mid, 0.0, 1.0)
        if g.sum() > target:
            lo = mid
        else:
            hi = mid
    g = np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - g.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = g < 1.0
        else:
            mask = g > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            g[mask] += diff / cnt
            g = np.clip(g, 0.0, 1.0)
    return g

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    global _G_N, _G_T, _G_PEN
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def tleft():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    h_init = make_feasible(h_init, len(h_init))
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)

    # Phase 1: L-BFGS-B at multiple resolutions and temperatures
    resolutions = [960, 1440, 1920, 2400]
    for n_pts in resolutions:
        if tleft() < 180:
            break
        if len(best_h) >= n_pts:
            h_start = best_h[:n_pts].copy()
            h_start = make_feasible(h_start, n_pts)
        else:
            h_start = upsample_h(best_h, n_pts)

        for T in [80.0, 200.0, 500.0, 1000.0]:
            if tleft() < 60:
                break
            for pen in [500.0, 2000.0]:
                if tleft() < 30:
                    break
                _G_N = n_pts
                _G_T = T
                _G_PEN = pen
                bounds = [(0.0, 1.0)] * n_pts
                try:
                    res = minimize(_obj_l, h_start, jac=_grad_l, method='L-BFGS-B',
                                   bounds=bounds,
                                   options={'maxiter': 400, 'ftol': 1e-16, 'gtol': 1e-8,
                                            'maxfun': 10000, 'iprint': -1})
                    h_res = make_feasible(res.x, n_pts)
                    c5_res = compute_c5(h_res)
                    if c5_res < best_c5:
                        best_c5 = c5_res
                        best_h = h_res.copy()
                    h_start = h_res.copy()
                except Exception:
                    pass

    # Phase 2: SLSQP with equality constraint at best resolution
    n_best = len(best_h)
    for T in [300.0, 800.0, 1500.0]:
        if tleft() < 50:
            break
        _G_N = n_best
        _G_T = T
        _G_PEN = 0.0
        bounds = [(0.0, 1.0)] * n_best
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_sl, best_h, jac=_grad_sl, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 500, 'ftol': 1e-15, 'iprint': -1})
            h_res = make_feasible(res.x, n_best)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
        except Exception:
            pass

    # Phase 3: Coordinate descent with fine candidates
    n_pts = len(best_h)
    for rnd in range(50):
        if tleft() < 15:
            break
        improved = False
        for j in range(n_pts):
            if tleft() < 5:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_v = old_val
            step_size = 0.02
            for d in [0.0] + list(np.concatenate([
                np.arange(-step_size, step_size + 1e-9, step_size/2),
                [-0.1, -0.05, 0.05, 0.1]
            ])):
                v = old_val + d
                if v < 0.0 or v > 1.0:
                    continue
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            # Also try 0 and 1
            for v in [0.0, 1.0]:
                if v == old_val:
                    continue
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_v != old_val:
                best_h[j] = best_v
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    # Phase 4: Block coordinate descent (change 2-3 coords)
    if tleft() > 30:
        for rnd in range(10):
            if tleft() < 10:
                break
            improved = False
            for _ in range(200):
                if tleft() < 5:
                    break
                j1 = rng.randint(0, n_pts)
                j2 = rng.randint(0, n_pts)
                if j1 == j2:
                    continue
                for v1 in [0.0, 1.0, best_h[j1]]:
                    for v2 in [0.0, 1.0, best_h[j2]]:
                        if v1 == best_h[j1] and v2 == best_h[j2]:
                            continue
                        trial = best_h.copy()
                        trial[j1] = v1
                        trial[j2] = v2
                        trial = make_feasible(trial, n_pts)
                        c = compute_c5(trial)
                        if c < best_c5 - 1e-14:
                            best_c5 = c
                            best_h = trial.copy()
                            improved = True
                            break
                    if improved:
                        break
            if not improved:
                break

    # Phase 5: Binary snap
    if tleft() > 20:
        for j in range(n_pts):
            if tleft() < 5:
                break
            for v in [0.0, 1.0]:
                if best_h[j] == v:
                    continue
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    best_h = trial.copy()

    # Phase 6: Final L-BFGS-B + SLSQP polish
    if tleft() > 40:
        n_f = len(best_h)
        for T in [1000.0, 2000.0]:
            if tleft() < 20:
                break
            _G_N = n_f
            _G_T = T
            _G_PEN = 5000.0
            bounds = [(0.0, 1.0)] * n_f
            try:
                res = minimize(_obj_l, best_h, jac=_grad_l, method='L-BFGS-B',
                               bounds=bounds,
                               options={'maxiter': 300, 'ftol': 1e-16, 'iprint': -1})
                h_res = make_feasible(res.x, n_f)
                c5_res = compute_c5(h_res)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = h_res.copy()
            except Exception:
                pass

        _G_N = n_f
        _G_T = 2000.0
        _G_PEN = 0.0
        bounds = [(0.0, 1.0)] * n_f
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj_sl, best_h, jac=_grad_sl, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15, 'iprint': -1})
            h_res = make_feasible(res.x, n_f)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res.copy()
        except Exception:
            pass

    # Phase 7: Final fine coordinate descent
    n_pts = len(best_h)
    for rnd in range(20):
        if tleft() < 10:
            break
        improved = False
        for j in range(n_pts):
            if tleft() < 5:
                break
            old_val = best_h[j]
            best_c = best_c5
            best_v = old_val
            for d in [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01]:
                v = old_val + d
                if v < 0.0 or v > 1.0:
                    continue
                trial = best_h.copy()
                trial[j] = v
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_v = v
            if best_v != old_val:
                best_h[j] = best_v
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True
        if not improved:
            break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
