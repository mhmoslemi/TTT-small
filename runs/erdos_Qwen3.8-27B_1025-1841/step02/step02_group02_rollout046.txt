    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

    h = project_to_feasible(h, n).copy()
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2 = 0.8, 0.99
    eps = 1e-8
    stall = 0
    
    for it in range(max_iters):

            break
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        t = it + 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * (grad * grad)
        mh = m / (1 - b1**t)
        vh = v / (1 - b2**t)
        
        h = project_to_feasible(h - lr * mh / (np.sqrt(vh) + eps), n)
        
        if stall > 80:
            lr *= 0.6
            m *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
    
    return best_h, best_obj
