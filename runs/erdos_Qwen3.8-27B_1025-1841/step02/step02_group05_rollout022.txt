import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def grad_single_shift(h, sh):
    """Gradient of correlation at a single shift."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    if sh >= 0:
        j1 = n - sh
        if j1 > 0:
            grad[:j1] += (1.0 - h[sh:sh + j1])
        grad[sh:] -= h[:j1]
    else:
        s = -sh
        j1 = n - s
        if j1 > 0:
            grad[:j1] += (1.0 - h[s:s + j1])
            grad[s:] -= h[:j1]
    grad *= dx
    return grad

        iter_count += 1

        tau = tau_start * (tau_end / tau_start) ** progress

            tau = tau_end
        
        dx = 2.0 / n
        corr = np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

            stall = 0
