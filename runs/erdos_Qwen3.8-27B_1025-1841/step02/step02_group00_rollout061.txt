    dx = 2.0 / n
    corr = np.correlate(h, 1-h, mode='full')
    s = np.argmax(corr)  # index in full mode
    shift = s - (n - 1)  # actual shift

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def project_feasible(h, n):
    """Project h to {0≤h≤1, Σh=n/2} using bisection on offset."""
    h = np.asarray(h, dtype=float).copy()
    target = n / 2.0
    
    def sum_clipped(offset):
        return np.sum(np.clip(h - offset, 0.0, 1.0))
    
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_clipped(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    h_new = np.clip(h - (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    diff = target - np.sum(h_new)
    if abs(diff) > 1e-12:
        mask = (h_new > 1e-10) & (h_new < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h_new[mask] += diff / np.sum(mask)
            h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

    best_c = compute_c5(h)
    for t in range(nsteps):
        step = 0.08 * 2.0 / np.sqrt(t + 1)
        g, _ = subgrad_grad(h, n)
        for mult in (1.0, 0.5, 0.25, 0.1, 0.05):
            h_new = project_feasible(h - step * mult * g, n)
            c = compute_c5(h_new)
            if c < best_c - 1e-15:
                best_c = c
                best_h = h_new.copy()
                h = h_new
                break
        else:
            if t > 80:
                break
    return best_h, best_c

def multi_subgrad_min(h, n, nsteps=300, topk=3):
    """Subgradient using top-k lags for more robust direction."""
    best_h = h.copy()
    best_c = compute_c5(h)
    dx = 2.0 / n
    nfull = 2 * n - 1
    for t in range(nsteps):
        step = 0.05 * 2.0 / np.sqrt(t + 1)
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(corr)[-topk:]
        g = np.zeros(n)
        for si in top_idx:
            d = si - (n - 1)
            gi = np.zeros(n)
            if d >= 0:
                if n - d > 0:
                    gi[:n - d] += 1.0 - h[d:]
                if d < n:
                    gi[d:] -= h[:n - d]
            else:
                s = -d
                if n - s > 0:
                    gi[s:] += 1.0 - h[:n - s]
                    gi[:n - s] -= h[s:]
            g += gi * dx
        g /= topk
        h_new = project_feasible(h - step * g, n)
        c = compute_c5(h_new)
        if c < best_c - 1e-15:
            best_c = c
            best_h = h_new.copy()
            h = h_new
    return best_h, best_c

def coord_desc(h, n, rounds=3, ds=0.008):
    h = h.copy()
    bc = compute_c5(h)
    for _ in range(rounds):
        imp = False
        for j in range(n):
            for nv in (h[j] + ds, h[j] - ds, 0.0, 1.0):
                if nv < -1e-9 or nv > 1 + 1e-9:
                    continue
                if abs(nv - h[j]) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = nv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < bc - 1e-13:
                    h, bc = trial, c
                    imp = True
                    break
        if not imp:
            break
    return h, bc

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 320)

    n = len(h0)
    h0 = make_feasible(h0, n)
    best_c = compute_c5(h0)
    best_h = h0.copy()

    def tl():
        return budget_s - (time.time() - t0)

    # Phase 1: SLSQP with varied temperatures
    T_list = [800, 300, 100, 50, 25, 15, 10, 8, 5, 3, 2]
    for T in T_list:
        if tl() < 20:
            break
        try:
            h = do_slsqp(h0, n, float(T), 200)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h, h0 = c, h.copy(), h.copy()
        except Exception:
            pass

    # Phase 2: Subgradient (single lag)
    if tl() > 20:
        h, c = subgrad_min(h0, n, 500)
        if c < best_c:
            best_c, best_h, h0 = c, h.copy(), h.copy()

    # Phase 3: Multi-lag subgradient
    if tl() > 20:
        h, c = multi_subgrad_min(h0, n, 300, topk=3)
        if c < best_c:
            best_c, best_h, h0 = c, h.copy(), h.copy()

    # Phase 4: SLSQP refinement
    for T in [50, 20, 10, 5]:
        if tl() < 20:
            break
        try:
            h = do_slsqp(h0, n, float(T), 180)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h, h0 = c, h.copy(), h.copy()
        except Exception:
            pass

    # Phase 5: Subgradient round 2
    if tl() > 30:
        h, c = subgrad_min(h0, n, 400)
        if c < best_c:
            best_c, best_h, h0 = c, h.copy(), h.copy()
    if tl() > 20:
        h, c = multi_subgrad_min(h0, n, 200, topk=5)
        if c < best_c:
            best_c, best_h, h0 = c, h.copy(), h0.copy()

    # Phase 6: Random restarts with diverse initializations
    for _ in range(5):
        if tl() < 25:
            break
        r = rng.rand(n)
        h_rand = 0.6 * h0 + 0.4 * r
        h_rand = make_feasible(h_rand, n)
        try:
            h = do_slsqp(h_rand, n, 15.0, 120)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h, h0 = c, h.copy(), h.copy()
        except Exception:
            pass

    # Phase 7: Try structured initializations
    if tl() > 40:
        for p in (3, 4, 5, 6, 7, 8):
            h_struct = np.zeros(n)
            period = n // p
            for i in range(n):
                h_struct[i] = 1.0 if (i % period) < period // 2 else 0.0
            h_struct = 0.5 * h_struct + 0.5 * h0
            h_struct = make_feasible(h_struct, n)
            try:
                h = do_slsqp(h_struct, n, 8.0, 100)
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h, h0 = c, h.copy(), h.copy()
            except Exception:
                pass
            if tl() < 20:
                break

    # Phase 8: Final subgradient + SLSQP
    if tl() > 30:
        h, c = subgrad_min(h0, n, 300)
        if c < best_c:
            best_c, best_h, h0 = c, h.copy(), h.copy()
    if tl() > 20:
        try:
            h = do_slsqp(h0, n, 3.0, 150)
            c = compute_c5(h)
            if c < best_c:
                best_c, best_h = c, h.copy()
        except Exception:
            pass

    # Phase 9: Coordinate descent polish
    if tl() > 10:
        h, c = coord_desc(best_h, n, rounds=2, ds=0.005)
        if c < best_c:
            best_c, best_h = c, h.copy()
    if tl() > 10:
        h, c = coord_desc(best_h, n, rounds=2, ds=0.002)
        if c < best_c:
            best_c, best_h = c, h.copy()

    # Phase 10: Try higher resolution
    if tl() > 80:
        n2 = min(2048, 2 * n)
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c:
            best_c, best_h = c2, h2.copy()
        for T in [30, 10, 5]:
            if tl() < 20:
                break
            try:
                h = do_slsqp(h2, n2, float(T), 120)
                c = compute_c5(h)
                if c < best_c:
                    best_c, best_h, h2 = c, h.copy(), h.copy()
            except Exception:
                pass
        if tl() > 20:
            h, c = subgrad_min(h2, n2, 200)
            if c < best_c:
                best_c, best_h = c, h.copy()

    # Final feasibility and evaluation
    best_h = make_feasible(best_h, len(best_h))
    fc = compute_c5(best_h)
    if fc < best_c:
        best_c = fc

    return best_h, best_c, len(best_h)
