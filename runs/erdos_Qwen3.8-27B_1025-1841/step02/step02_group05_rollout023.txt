        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = np.zeros(n)
    thresh = 1e-14
    for k in range(len(corr)):
        pk = p[k]
        if pk < thresh:
            continue
        sh = k - n + 1
        i0 = max(0, -sh)
        i1 = min(n, n - sh)
        if i0 < i1:
            grad[i0:i1] += pk * (1.0 - h[i0 + sh:i1 + sh])
        i0b = max(0, sh)
        i1b = min(n, n + sh)
        if i0b < i1b:
            grad[i0b:i1b] -= pk * h[i0b - sh:i1b - sh]
    grad *= dx
    return obj, grad

def nesterov_optimize(h, n, time_limit, tau=60.0, lr=0.4, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    h_prev = h.copy()
    stall = 0
    lr_cur = lr

    while time.time() - start < time_limit:
        y = h + momentum * (h - h_prev)
        _, grad = smooth_grad(y, tau)
        h_prev = h.copy()
        h = project_to_feasible(y - lr_cur * grad, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 800:
            lr_cur *= 0.6
            v = np.zeros(n)
            stall = 0
            if lr_cur < 1e-13:
                break
    return best_h, best_c5


def cooling_optimize(h, n, time_limit, tau_start=120.0, tau_end=3.0, lr_base=0.5, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    total = time_limit
    elapsed = 0.0
    momentum_cur = momentum
    h_prev = h.copy()
    stall = 0
    lr_cur = lr_base

    while elapsed < time_limit - 1.0:
        frac = elapsed / total
        tau = tau_start * (tau_end / tau_start) ** frac
        if tau < tau_end:
            tau = tau_end

        y = h + momentum_cur * (h - h_prev)
        _, grad = smooth_grad(y, tau)
        h_prev = h.copy()
        h = project_to_feasible(y - lr_cur * grad, n)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 600:
            lr_cur *= 0.5
            h_prev = h.copy()
            stall = 0
            if lr_cur < 1e-13:
                break
        elapsed = time.time() - start
    return best_h, best_c5


def stochastic_subgrad(h, n, time_limit, lr=0.003, top_k=4, seed=0):
    rng = np.random.RandomState(seed)
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    momentum = 0.85
    stall = 0
    lr_cur = lr
    dx = 2.0 / n

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            lr_cur *= 0.65
            v *= 0.4
            stall = 0
            if lr_cur < 1e-14:
                break
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        k = top_idx[rng.randint(top_k)]
        sh = k - n + 1
        grad = np.zeros(n)
        i0 = max(0, -sh)
        i1 = min(n, n - sh)
        if i0 < i1:
            grad[i0:i1] += (1.0 - h[i0 + sh:i1 + sh])
        i0b = max(0, sh)
        i1b = min(n, n + sh)
        if i0b < i1b:
            grad[i0b:i1b] -= h[i0b - sh:i1b - sh]
        grad *= dx
        v = momentum * v - lr_cur * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def multi_shift_subgrad(h, n, time_limit, lr=0.002, top_k=5):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    momentum = 0.8
    stall = 0
    lr_cur = lr
    dx = 2.0 / n

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr_cur *= 0.7
            v *= 0.4
            stall = 0
            if lr_cur < 1e-14:
                break
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            sh = k - n + 1
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 < i1:
                grad[i0:i1] += w[i] * (1.0 - h[i0 + sh:i1 + sh])
            i0b = max(0, sh)
            i1b = min(n, n + sh)
            if i0b < i1b:
                grad[i0b:i1b] -= w[i] * h[i0b - sh:i1b - sh]
        grad *= dx
        v = momentum * v - lr_cur * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def perturb(h, n, sigma, seed):
    rng = np.random.RandomState(seed)
    noise = rng.randn(n) * sigma
    return project_to_feasible(h + noise, n)


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()

    def remaining():
        return max(0.0, budget_s - (time.time() - t0))

    np.random.seed(seed)

    h_init = initial_h_values.copy()
    n = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: Cooling optimization from initial (use ~25% of time)
    if remaining() > 100:
        tl = min(250, remaining() * 0.25)
        h1, c5_1 = cooling_optimize(best_h, n, tl, tau_start=100, tau_end=5, lr_base=0.4, momentum=0.9)
        if c5_1 < best_c5:
            best_c5, best_h = c5_1, h1.copy()

    # Phase 2: Nesterov with moderate tau
    if remaining() > 80:
        tl = min(150, remaining() * 0.15)
        h2, c5_2 = nesterov_optimize(best_h, n, tl, tau=20.0, lr=0.25, momentum=0.9)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 3: Random restarts with perturbation (use ~30% of time)
    n_restarts = 4
    if remaining() > 120:
        per_restart = remaining() * 0.30 / n_restarts
        for r in range(n_restarts):
            if remaining() < 30:
                break
            sigma = 0.08 * (1.0 + 0.5 * r)
            h_pert = perturb(best_h, n, sigma, seed + r * 100)
            tl = min(per_restart, remaining() * 0.4)
            h_r, c5_r = cooling_optimize(h_pert, n, tl, tau_start=60, tau_end=4, lr_base=0.3, momentum=0.88)
            if c5_r < best_c5:
                best_c5, best_h = c5_r, h_r.copy()

    # Phase 4: Low tau refinement on best
    if remaining() > 60:
        tl = min(120, remaining() * 0.2)
        h4, c5_4 = nesterov_optimize(best_h, n, tl, tau=5.0, lr=0.15, momentum=0.92)
        if c5_4 < best_c5:
            best_c5, best_h = c5_4, h4.copy()

    # Phase 5: Multi-shift subgradient polish
    if remaining() > 50:
        tl = min(100, remaining() * 0.2)
        h5, c5_5 = multi_shift_subgrad(best_h, n, tl, lr=0.002, top_k=5)
        if c5_5 < best_c5:
            best_c5, best_h = c5_5, h5.copy()

    # Phase 6: Stochastic subgradient for fine-tuning
    if remaining() > 40:
        tl = min(80, remaining() * 0.2)
        h6, c5_6 = stochastic_subgrad(best_h, n, tl, lr=0.001, top_k=3, seed=seed + 999)
        if c5_6 < best_c5:
            best_c5, best_h = c5_6, h6.copy()

    # Phase 7: Higher resolution refinement
    if remaining() > 100:
        n2 = 4000
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        c5_2 = compute_c5(h2)
        # Optimize at higher resolution
        if remaining() > 80:
            tl = min(60, remaining() * 0.25)
            h2, c5_2 = nesterov_optimize(h2, n2, tl, tau=30.0, lr=0.3, momentum=0.9)
        if remaining() > 40:
            tl = min(50, remaining() * 0.2)
            h2, c5_2 = nesterov_optimize(h2, n2, tl, tau=8.0, lr=0.1, momentum=0.92)
        if remaining() > 30:
            tl = min(40, remaining() * 0.2)
            h2, c5_2 = stochastic_subgrad(h2, n2, tl, lr=0.001, top_k=3, seed=seed + 777)
        if c5_2 < best_c5:
            best_c5, best_h = c5_2, h2.copy()

    # Phase 8: Final polish with very low tau
    if remaining() > 30:
        n_f = len(best_h)
        tl = min(40, remaining() * 0.5)
        h8, c5_8 = stochastic_subgrad(best_h, n_f, tl, lr=0.0005, top_k=2, seed=seed + 1234)
        if c5_8 < best_c5:
            best_c5, best_h = c5_8, h8.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
