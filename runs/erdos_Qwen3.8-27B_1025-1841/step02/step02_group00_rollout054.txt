            h_cd2, c5_cd2 = cd_polish(best_h, n2, 4, rng, _tl, _tl)
            if c5_cd2 < best_c5:
                best_c5 = c5_cd2
                best_h = h_cd2

    # Phase 6: Upsample to 2560
    if _tl() > 100:
        n3 = 2560
        h_up3 = upsample_h(best_h, n3)
        c5_up3 = compute_c5(h_up3)
        if c5_up3 < best_c5:
            best_c5 = c5_up3
            best_h = h_up3
        for T in [500, 1500]:
            if _tl() < 40:
                break
            h_opt = slsqp_opt(h_up3, n3, T, 100)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

    # Phase 7: Final SLSQP polish at best resolution
    if _tl() > 30:
        nf = len(best_h)
        h_opt = slsqp_opt(best_h, nf, 5000, 80)
        c5 = compute_c5(h_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
