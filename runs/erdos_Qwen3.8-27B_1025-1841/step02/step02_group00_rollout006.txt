import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-13
        else:
            mask = h > 1e-13
        cnt = np.sum(mask)
        if cnt == 0:
            h = np.clip(h + diff / n, 0, 1)
        else:
            h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

# Globals for SLSQP
_GN = 0
_GT = 1.0

def _sm_obj(h):
    n = _GN
    T = _GT
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted)))) * dx

def _cons_eq(h):
    return np.sum(h) - _GN / 2.0

def _cons_jac(h):
    return np.ones(_GN)

    global _GN, _GT
    _GN = n
    _GT = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_eq, 'jac': _cons_jac}]
    res = minimize(_sm_obj, h_init, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def smooth_max_obj_grad(h, T, n, pen_lam=0.0):
    """Objective and gradient for smooth max with optional penalty for sum constraint."""
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    c = r * dx
    c_max = np.max(c)
    log_w = T * (c - c_max)
    w = np.exp(log_w)
    sw = np.sum(w)
    w /= sw
    
    # Objective
    val = (c_max + np.log(sw) / T)
    if pen_lam > 0:
        s = np.sum(h) - n / 2.0
        val += 0.5 * pen_lam * s * s
    
    # Gradient of smooth max
    # grad[j] = dx * (first[j] - second[j])
    nr = 2 * n - 1
    one_minus_h = 1.0 - h
    h_rev = h[::-1].copy()
    
    first = np.array([np.dot(w[n-1-j : 2*n-1-j], one_minus_h) for j in range(n)])
    second = np.array([np.dot(w[j : j+n], h_rev) for j in range(n)])
    grad = dx * (first - second)
    
    if pen_lam > 0:
        grad += pen_lam * (np.sum(h) - n / 2.0)
    
    return val, grad

def lbfgs_optimize(h_init, n, T, max_iter=300, pen_lam=100.0):
