```python
def evaluate_pair_flip(h, i, j, corr):
    """Compute new correlation max after flipping h[i]:0->1, h[j]:1->0"""
    n = len(h)
    s = np.arange(-(n-1), n)  # shifts
    # For position i (0->1):
    # Δ_i at shift s: (1-h[i+s]) if 0<=i+s<n, minus h[i-s] if 0<=i-s<n
    # For position j (1->0):
    # Δ_j at shift s: -(1-h[j+s]) if 0<=j+s<n, plus h[j-s] if 0<=j-s<n
    
    # Vectorized computation
    delta = np.zeros(2*n-1)
    
    # For i:
    valid_ps = (i + s >= 0) & (i + s < n)
    valid_ns = (i - s >= 0) & (i - s < n)
    delta[valid_ps] += 1.0 - h[i + s[valid_ps]]
    delta[valid_ns] -= h[i - s[valid_ns]]
    
    # For j:
    valid_ps = (j + s >= 0) & (j + s < n)
    valid_ns = (j - s >= 0) & (j - s < n)
    delta[valid_ps] -= 1.0 - h[j + s[valid_ps]]
    delta[valid_ns] += h[j - s[valid_ns]]
    
    new_corr = corr + delta
    return np.max(new_corr)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def binarize(h, n):
    """Convert to binary with exactly n/2 ones"""
    h_bin = (h >= 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(np.sum(h_bin))
    target = n // 2
    if n_ones > target:
        # Need to flip some 1s to 0s
        indices = np.where(h_bin == 1)[0]
        # Sort by original h value (lowest first - most likely to be 0)
        order = np.argsort(h[indices])
        for k in range(n_ones - target):
            h_bin[indices[order[k]]] = 0
    elif n_ones < target:
        # Need to flip some 0s to 1s
        indices = np.where(h_bin == 0)[0]
        order = np.argsort(-h[indices])
        for k in range(target - n_ones):
            h_bin[indices[order[k]]] = 1
    return h_bin
```
