import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def smooth_max_and_grad(corr, tau):
    """Compute smooth max of corr and its gradient coefficients."""
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s  # softmax weights
    obj = max_c + np.log(s) / tau
    return obj, p

def overlap_grad_at_shifts(h, p, n):
    """Compute gradient of smooth max using softmax weights p over shifts."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-10:
            continue
        sh = k - n + 1
        # overlap_k = sum_i h[i] * (1 - h[i+sh]) * dx (with appropriate bounds)
        # d(overlap_k)/d(h[i]) = (1 - h[i+sh]) * dx for i in valid range for h[i]
        #                          - h[i-sh] * dx for i in valid range for h[i+sh]
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            indices = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[indices - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            indices = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[indices + sh]
    grad *= dx
    return grad

def optimize_adaptive_tau(h, n, tau_schedule, max_iters, time_limit, initial_lr=0.3):
    """Optimize with adaptive tau schedule (list of (tau, num_iters) or continuous)."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = initial_lr
    v = np.zeros(n)
    beta = 0.9
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Adaptive tau: increase over iterations
        progress = it / max_iters
        tau = tau_schedule(progress)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        obj, p = smooth_max_and_grad(corr, tau)
        
        # Check actual C5
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        grad = overlap_grad_at_shifts(h, p, n)
        
        # Adam-like update
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        # Backtracking: if objective worsened significantly, reduce lr
        corr_new = np.correlate(h_new, 1.0 - h_new, mode='full')
        obj_new, _ = smooth_max_and_grad(corr_new, tau)
        if obj_new > obj + 0.01 / tau:
            lr *= 0.7
            v *= 0.5
            # Don't accept
        else:
            h = h_new
            # Occasionally increase lr if making good progress
            if it % 100 == 0 and lr < initial_lr:
                lr *= 1.05
    
    return best_h, best_c5

    L = 1
    while L < 3 * n:
