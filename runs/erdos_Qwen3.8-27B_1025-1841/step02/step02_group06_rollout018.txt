def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

import numpy as np
import time
from scipy.fft import fft, ifft

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def corr_fft(h):
    """Full linear cross-correlation of h and 1-h via FFT."""
    n = len(h)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    om = 1.0 - h
    A = fft(h, L)
    B = fft(om, L)
    c = ifft(A * np.conj(B)).real
    return np.concatenate([c[n-1:], c[:n-1]])

def grad_top(h, p_top, top_idx, n):
    """Direct gradient for top lags."""
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_top[i]
        if pk < 1e-15:
            continue
        d = top_idx[i] - n + 1
        # (1-h)[j+d] for valid j
        j0 = max(0, -d)
        j1 = min(n, n - d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx + d])
        # -h[j-d] for valid j
        j0b = max(0, d)
        j1b = min(n, n + d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx - d]
    return grad

    best_c5 = corr_fft(h).max() * dx
    best_h = h.copy()
    
    m_state = np.zeros(n)
    v_state = np.zeros(n)
    t_adam = 0
    b1, b2, eps = 0.9, 0.999, 1e-8
    
    start = time.time()
    stall = 0
    patience = max(200, max_iters // 10)
    lr = lr_init
    
    for it in range(max_iters):

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def corr_fft(h):
    n = len(h)
    L = 1
    while L < 2 * n - 1:
        L <<= 1
    om = 1.0 - h
    C = ifft(fft(h, L) * np.conj(fft(om, L))).real
    return np.concatenate([C[L - n + 1:], C[:n]])

def grad_direct(h, p_active, active_idx, n):

            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
            j0b, j1b = 0, n - d
            if j0b < j1b:

            grad[idx2] -= pk * h[idx2 - dd]

            stall += 1

        if topM < 2 * n - 1:
            top_idx = np.argpartition(vals, -topM)[-topM:]
            v_top = vals[top_idx]
            mx = v_top.max()
            ex = np.exp(np.clip(tau * (v_top - mx), -500, 500))
            p_top = ex / ex.sum()
            act = p_top > p_top.max() * 1e-5
            if act.sum() > 0:
                grad = grad_direct(h, p_top[act], top_idx[act], n) * dx
            else:
                grad = grad_direct(h, p_top, top_idx, n) * dx
        else:
            mx = vals.max()
            ex = np.exp(np.clip(tau * (vals - mx), -500, 500))
            p = ex / ex.sum()
            act = p > p.max() * 1e-5
            if act.sum() > 0:
                aa = np.where(act)[0]
                grad = grad_direct(h, p[act], aa, n) * dx
            else:
                grad = grad_direct(h, p, np.arange(len(p)), n) * dx

        lr = lr_init / (1.0 + it * decay_rate)
        t_adam += 1
        m_s = b1 * m_s + (1 - b1) * grad
        v_s = b2 * v_s + (1 - b2) * grad * grad
        m_hat = m_s / (1.0 - b1 ** t_adam)
        v_hat = v_s / (1.0 - b2 ** t_adam)
        h = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + eps), n)

        if stall > patience:
            lr *= 0.5
            m_s *= 0.3
            v_s *= 0.3
            stall = 0
            if lr < 1e-13:
                break
    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def rem():
        return max(0.0, budget_s - (time.time() - t0))

    if 'initial_h_values' in dir() or True:
        try:
            h_init = initial_h_values.copy()
        except NameError:
            n_f = 3600
            x = np.linspace(0, 2, n_f, endpoint=False)
            h_init = np.where(np.sin(np.pi * x * 3) > 0.1, 0.9, 0.1)
            h_init = project_to_feasible(h_init, n_f)

    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: FFT-accelerated optimization at initial res
    if rem() > 60:
        for tau in [12.0, 40.0, 120.0, 400.0]:
            if rem() < 25:
                break
            tl = min(rem() * 0.18, 180.0)
            hh, cc = optimize_fft(best_h, n_init, tau, 60000, tl,
                                   lr_init=0.07, topM=80, decay_rate=0.0015)
            if cc < best_c5:
                best_c5 = cc
                best_h = hh.copy()

    # Phase 2: 2x resolution
    n2 = n_init * 2
    if rem() > 70:
        h2 = project_to_feasible(interp_periodic(best_h, n2), n2)
        for tau in [20.0, 60.0, 200.0]:
            if rem() < 25:
                break
            tl = min(rem() * 0.2, 200.0)
            h2, c2 = optimize_fft(h2, n2, tau, 40000, tl,
                                   lr_init=0.055, topM=50, decay_rate=0.002)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2.copy()

    # Phase 3: 4x resolution
    n3 = n_init * 4
    if rem() > 60:
        h3 = project_to_feasible(interp_periodic(best_h, n3), n3)
        for tau in [30.0, 100.0, 300.0]:
            if rem() < 25:
                break
            tl = min(rem() * 0.2, 200.0)
            h3, c3 = optimize_fft(h3, n3, tau, 25000, tl,
                                   lr_init=0.04, topM=35, decay_rate=0.0025)
            if c3 < best_c5:
                best_c5 = c3
                best_h = h3.copy()

    # Phase 4: 8x resolution (if budget allows)
    nb = len(best_h)
    n4 = nb * 2
    if rem() > 80 and n4 <= 64000:
        h4 = project_to_feasible(interp_periodic(best_h, n4), n4)
        for tau in [50.0, 150.0]:
            if rem() < 30:
                break
            tl = min(rem() * 0.22, 250.0)
            h4, c4 = optimize_fft(h4, n4, tau, 15000, tl,
                                   lr_init=0.03, topM=25, decay_rate=0.003)
            if c4 < best_c5:
                best_c5 = c4
                best_h = h4.copy()

    # Phase 5: Alternative starting points
    if rem() > 50:
        n_alt = len(best_h)
        starts = []
        # Smooth random
        r = np.random.randn(n_alt)
        from scipy.ndimage import gaussian_filter1d
        r = gaussian_filter1d(r, sigma=n_alt // 40)
        starts.append(project_to_feasible(0.5 + 0.3 * r / (r.std() + 1e-10), n_alt))
        # Symmetric
        half = np.random.randn(n_alt // 2)
        h_sym = np.zeros(n_alt)
        h_sym[:n_alt // 2] = half
        h_sym[n_alt // 2:] = half[::-1]
        h_sym = project_to_feasible(0.5 + 0.2 * h_sym / (h_sym.std() + 1e-10), n_alt)
        starts.append(h_sym)
        # Perturbed best
        starts.append(project_to_feasible(best_h + 0.05 * np.random.randn(n_alt), n_alt))

        for s_i, h_s in enumerate(starts):
            if rem() < 20:
                break
            tl = min(rem() * 0.2, 100.0)
            hs, cs = optimize_fft(h_s, n_alt, 60.0, 15000, tl,
                                   lr_init=0.06, topM=50, decay_rate=0.002)
            if cs < best_c5:
                best_c5 = cs
                best_h = hs.copy()

    # Phase 6: Very fine-tune (high tau, low lr)
    if rem() > 20:
        nf = len(best_h)
        hf = best_h.copy()
        tl = rem() * 0.35
        hf, cf = optimize_fft(hf, nf, 2000.0, 30000, tl,
                               lr_init=0.008, topM=5, decay_rate=0.005, patience_factor=5)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Phase 7: Final micro-polish
    if rem() > 10:
        nf = len(best_h)
        hf = best_h.copy()
        tl = rem() * 0.45
        hf, cf = optimize_fft(hf, nf, 5000.0, 20000, tl,
                               lr_init=0.003, topM=3, decay_rate=0.008, patience_factor=5)
        if cf < best_c5:
            best_c5 = cf
            best_h = hf.copy()

    # Final projection and exact evaluation
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    best_c5 = min(best_c5, final_c5)

    return best_h, best_c5, len(best_h)
