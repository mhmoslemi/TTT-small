def make_feasible(h, n):

    n_ones = n // 2

def _soft_obj(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _sum_con(x, n):
    return np.sum(x) - n / 2.0

def _sum_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h_start, n, T=80, maxiter=300, ftol=1e-12):
    cons = [{'type': 'eq', 'fun': lambda x: _sum_con(x, n), 'jac': lambda x: _sum_jac(x, n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return _soft_obj(x, T)
    
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': ftol})
        return make_feasible(res.x, n)

        return h_start.copy()

        if cb < best_c5:
            best_c5, best_h = cb, hbin

    # Phase 3: Multi-resolution refinement (300s)
    for nn in [1280, 1920, 2560, 640, 480]:
        if _rem() < 50:
            break
        ht = _upsample(best_h, nn)
        if _rem() > 70:
            ht = _slsqp(ht, nn, T=120, mi=150)
        ht = _snap(ht)
        ct = _c5(ht)
        if _rem() > 40 and ct <= best_c5 + 0.008:
            ht, ct = _bsearch(ht, min(120, _rem() - 25), rng, use_sa=True)
        if ct < best_c5:
            best_c5, best_h = ct, ht

    # Phase 4: Alternate starting points (150s)
    if _rem() > 80:
        # Perturbed start
        hp = best_h + rng.randn(len(best_h)) * 0.05
        hp = _feasible(hp, len(best_h))
        hp = _slsqp(hp, len(best_h), T=100, mi=150)
        hp = _snap(hp)
        ct = _c5(hp)
        if _rem() > 40 and ct <= best_c5 + 0.01:
            hp, ct = _bsearch(hp, min(100, _rem() - 20), rng, use_sa=True)
        if ct < best_c5:
            best_c5, best_h = ct, hp

    # Phase 5: Final intensive search
    if _rem() > 20:
        hf = _snap(best_h) if best_h.max() > 0.9 else best_h
        cf = _c5(hf)
        if cf > best_c5:
            hf, cf = best_h, best_c5
        hf, cf = _bsearch(hf, _rem() - 15, rng, use_sa=False)
        if cf < best_c5:
            best_c5, best_h = cf, hf

    best_h = _feasible(best_h, len(best_h))
    best_c5 = _c5(best_h)
    return best_h, best_c5, len(best_h)
