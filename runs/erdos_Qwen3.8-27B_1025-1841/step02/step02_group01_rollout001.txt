import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0

def _g_obj(h):
    dx = 2.0 / _G_N
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_G_T * shifted))) / _G_T) * dx

def _g_con(h):
    return np.sum(h) - _G_N / 2.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(40):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def random_binary_h(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = n // 2
    ones_idx = np.where(h == 1)[0]
    zeros_idx = np.where(h == 0)[0]
    if len(ones_idx) > n_ones:
        to_flip = rng.choice(ones_idx, len(ones_idx) - n_ones, replace=False)
        h[to_flip] = 0.0
    elif len(ones_idx) < n_ones:
        to_flip = rng.choice(zeros_idx, n_ones - len(ones_idx), replace=False)
        h[to_flip] = 1.0
    return h

def random_structured_h(n, rng):
    period = rng.randint(3, 12)
    pattern_len = period
    pattern = (rng.rand(pattern_len) < 0.5).astype(float)
    h = np.tile(pattern, n // pattern_len + 1)[:n]
    return make_feasible(h, n)

    corr = np.correlate(h, 1.0 - h, mode='full')
    sorted_c = np.sort(corr)[::-1]
    top_vals = sorted_c[:min(k, len(corr))]
    return (np.sum(top_vals) / k) * dx
