import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p /= np.sum(p)
    # p has length 2n-1
    cent = n - 1
    p_flipped = p[::-1]
    
    # S1[j] = sum_k p[k] * h[j-k+cent]
    h_sh = np.zeros(2 * n)
    h_sh[cent:cent+n] = h
    conv1 = np.convolve(h_sh, p_flipped, mode='full')
    S1 = conv1[2*n-2 : 2*n-2+n]
    
    # S2[j] = sum_k p[k] * h[j+k-cent]  
    conv2 = np.convolve(h, p_flipped, mode='full')
    S2 = conv2[cent : cent+n]
    
    grad = dx * (1.0 - S1 - S2)
    return grad

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def proj_grad_desc(h0, n, T, max_iter=5000, deadline=None):
    h = h0.copy()
    step = 0.1
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break
        g = soft_max_grad(h, n, T)
        # Backtracking line search
        f0 = soft_max_val(h, n, T)
        alpha = step
        for _ in range(30):
            h_new = project(h - alpha * g, n)
            f_new = soft_max_val(h_new, n, T)
            if f_new < f0 - 1e-4 * alpha * np.dot(g, h - h_new):
                break
            alpha *= 0.5
        h = h_new
        if alpha < 1e-8:
            break
    return h

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def soft_max_val(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    p = np.exp(scaled - m)
    p /= np.sum(p)
    s1 = np.convolve(p, h, mode='full')[n - 1:2 * n - 1]
    s2 = np.convolve(p[::-1], h, mode='full')[n - 1:2 * n - 1]
    return dx * (1.0 - s1 - s2)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def proj_grad(h0, n, T, max_iter, deadline):
    h = h0.copy()
    step = 4.0 / (T * n)
    f_best = soft_max_val(h, n, T)
    stall = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = soft_max_grad(h, n, T)
        f0 = f_best
        alpha = step
        h_new = None
        for _ in range(50):
            cand = project(h - alpha * g, n)
            fc = soft_max_val(cand, n, T)
            if fc < f0 - 1e-12:
                h_new = cand
                break
            alpha *= 0.5
        if h_new is None:
            stall += 1
            if stall > 30:
                break
            alpha = 1e-8
            h_new = project(h - alpha * g, n)
            fc = soft_max_val(h_new, n, T)
            if fc >= f_best:
                break
        else:
            stall = 0
            step = min(alpha * 1.8, 2.0 / (T * n) * 10)
        h = h_new
        f_best = soft_max_val(h, n, T)
    return h

def lbfgs_opt(h0, n, T, deadline, maxiter=400):
    def f(x):
        return soft_max_val(x, n, T)
    def g(x):
        return soft_max_grad(x, n, T)
    try:
        res = minimize(f, h0, jac=g, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def binary_local_search(h0, n, deadline, max_flips=50000):
    h = h0.copy()
    h_bin = (h > 0.5).astype(float)
    h_bin = project(h_bin, n)
    c5_cur = evaluate_c5(h_bin)
    best_c5 = c5_cur
    best_h = h_bin.copy()
    rng = np.random.RandomState(42)
    for _ in range(max_flips):
        if time.time() > deadline:
            break
        i = rng.randint(n)
        j = rng.randint(n)
        h_bin[i] = 1.0 - h_bin[i]
        h_bin[j] = 1.0 - h_bin[j]
        h_bin = project(h_bin, n)
        c5_new = evaluate_c5(h_bin)
        if c5_new < c5_cur:
            c5_cur = c5_new
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h_bin.copy()
        else:
            h_bin[i] = 1.0 - h_bin[i]
            h_bin[j] = 1.0 - h_bin[j]
            h_bin = project(h_bin, n)
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float).copy()
    else:
        h0 = np.full(336, 0.5)

    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    n0 = len(h0)

    # Phase 1: Gradient optimization at n0 with increasing T
    T_schedule = [20, 50, 100, 200, 400, 800]
    h_work = best_h.copy()

    for T in T_schedule:
        if time.time() > deadline - 80:
            break
        h_pgd = proj_grad(h_work, n0, T, max_iter=4000, deadline=deadline - 30)
        c5_pgd = evaluate_c5(h_pgd)
        h_work = h_pgd
        if c5_pgd < best_c5:
            best_c5 = c5_pgd
            best_h = h_pgd.copy()

        if time.time() > deadline - 40:
            break
        h_lb = lbfgs_opt(h_work, n0, T, deadline - 5, maxiter=300)
        c5_lb = evaluate_c5(h_lb)
        h_work = h_lb
        if c5_lb < best_c5:
            best_c5 = c5_lb
            best_h = h_lb.copy()

    # Phase 2: Binary local search
    if time.time() < deadline - 100:
        h_bin_start = (best_h > 0.5).astype(float)
        h_bin_start = project(h_bin_start, n0)
        h_bin_best, c5_bin = binary_local_search(h_bin_start, n0, deadline - 60, max_flips=30000)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_best.copy()
        # Refine binary solution
        if time.time() < deadline - 30:
            h_ref = proj_grad(h_bin_best, n0, 200, max_iter=3000, deadline=deadline - 10)
            c5_ref = evaluate_c5(h_ref)
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()

    # Phase 3: Higher resolution
    if time.time() < deadline - 120:
        n_hi = n0 * 2
        h_hi = interp_to(best_h, n0, n_hi)
        h_hi = project(h_hi, n_hi)
        for T in [80, 200, 500]:
            if time.time() > deadline - 60:
                break
            h_pgd = proj_grad(h_hi, n_hi, T, max_iter=2000, deadline=deadline - 20)
            c5_pgd = evaluate_c5(h_pgd)
            h_hi = h_pgd
            if c5_pgd < best_c5:
                best_c5 = c5_pgd
                best_h = h_pgd.copy()
            if time.time() > deadline - 30:
                break
            h_lb = lbfgs_opt(h_hi, n_hi, T, deadline - 5, maxiter=200)
            c5_lb = evaluate_c5(h_lb)
            if c5_lb < best_c5:
                best_c5 = c5_lb
                best_h = h_lb.copy()
            h_hi = h_lb

    # Phase 4: Random restarts at best resolution
    n_best = len(best_h)
    for i in range(30):
        if time.time() > deadline - 15:
            break
        rng_i = np.random.RandomState(seed + 200 + i * 7)
        noise_scale = 0.04 * (1 + 0.3 * i)
        h_pert = best_h + rng_i.randn(n_best) * noise_scale
        h_pert = project(h_pert, n_best)
        h_new = proj_grad(h_pert, n_best, 150, max_iter=1500, deadline=deadline - 10)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        if time.time() < deadline - 20:
            h_new2 = lbfgs_opt(h_new, n_best, 200, deadline - 3, maxiter=200)
            c5_new2 = evaluate_c5(h_new2)
            if c5_new2 < best_c5:
                best_c5 = c5_new2
                best_h = h_new2.copy()

    # Phase 5: Final polish
    if time.time() < deadline - 3:
        h_fin = lbfgs_opt(best_h, len(best_h), 400, deadline, maxiter=200)
        c5_fin = evaluate_c5(h_fin)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
