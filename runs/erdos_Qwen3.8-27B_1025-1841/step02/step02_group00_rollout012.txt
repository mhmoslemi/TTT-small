```python
def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(200):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break

        else:

        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    # = sigmoid_deriv_j * (grad_h_j - sum_i(grad_h_i * sigmoid_deriv_i) / S)
    
    inner = np.dot(grad_h, sigmoid_deriv) / S
    grad_z = sigmoid_deriv * (grad_h - inner)
    
    return obj, grad_z
```
