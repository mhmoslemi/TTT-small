```python
import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    return np.correlate(h, 1.0 - h, mode='full')

    return compute_corr_full(h).max() * (2.0 / n)

    best_c5 = compute_c5(h)
    best_h = h.copy()
```

Let me also think about the "swap" local search more carefully:

For a swap from i to j (decrease h[i] by ε, increase h[j] by ε):
The change in corr[k] is approximately:
Δcorr[k] ≈ ε * [∂corr[k]/∂h[j] - ∂corr[k]/∂h[i]]

We want to minimize the maximum corr[k], so we want to find swaps that decrease the top corr values.

A simple heuristic:
1. Compute corr and find the top M values.
2. For the worst shift k*, compute the gradient.
3. Find the pair (i, j) that most reduces corr[k*] while maintaining feasibility.
4. Apply the swap.

But this is O(n²) per swap, which is too slow for large n. Instead, I can:
1. Compute the gradient for the worst shift.
2. Find the index i where the gradient is most positive (decreasing h[i] helps most).
3. Find the index j where the gradient is most negative (increasing h[j] helps most).
4. Swap a small amount from i to j.

This is O(n) per swap, which is fast.

```python
def local_search_swap(h, n, max_iters, time_limit, start_time):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = compute_corr_full(h)
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the worst shift
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        
        # Compute gradient for this shift
        if sh >= 0:
            g = np.zeros(n)
            if n - sh > 0:
                g[:n-sh] = (1.0 - h[sh:])
                g[sh:] -= h[:n-sh]
        else:
            s = -sh
            g = np.zeros(n)
            if n - s > 0:
                g[s:] = (1.0 - h[:n-s])
                g[:n-s] -= h[s:]
        
        # Find best swap: decrease h at argmax(g), increase h at argmin(g)
        i = np.argmax(g)  # most positive gradient -> decreasing h[i] helps most
        j = np.argmin(g)  # most negative gradient -> increasing h[j] helps most
        
        if h[i] <= 1e-10 or h[j] >= 1 - 1e-10:
            # Can't swap
```

Alternative: use a "direction" approach. At each step:
1. Compute the gradient for the worst shift.
2. Take a step in the negative gradient direction.
3. Project to feasibility.

This is essentially subgradient descent, which is what the previous code did in its "subgrad" phase. The improvement would be in the step size and the targeting.

Let me try a different approach: **mirror descent** or **exponential weighting**.

The idea is to parametrize h as h = exp(θ) / sum(exp(θ)) * N/2 (or something similar), which automatically satisfies 0 < h < ∞. Then project to [0,1] and fix the sum. This gives a smooth parametrization.

Actually, a simpler approach: use a "logistic" parametrization. Let h[i] = sigmoid(θ[i]), which guarantees 0 < h[i] < 1. Then optimize over θ. The sum constraint is handled by adding a Lagrange multiplier.

```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr_full(h).max() * (2.0 / n)

def grad_weighted(h, weights):
    """Gradient of sum_k weights[k] * corr_full[k] w.r.t. h, times dx."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(len(weights)):
        w = weights[k]
        if abs(w) < 1e-14:
            continue
        sh = k - n + 1
        if sh >= 0:
            if n - sh > 0:
                grad[:n-sh] += w * (1.0 - h[sh:])
                grad[sh:] -= w * h[:n-sh]
        else:
            s = -sh
            if n - s > 0:
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] -= w * h[s:]
    return grad * dx

def top_k_weights(corr, k):
    """Return weights that emphasize the top k correlation values."""
    k = min(k, len(corr))
    idx = np.argsort(corr)[-k:]
    w = np.zeros(len(corr))
    w[idx] = 1.0 / k
    return w

def optimize_smooth_adam(h_init, n, tau_list, max_iters, time_limit, start_time):
    dx = 2.0 / n
    h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.1
    
    tau_idx = 0
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        tau = tau_list[min(tau_idx, len(tau_list) - 1)]
        corr = compute_corr_full(h)
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -400, 400)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()
        grad = grad_weighted(h, p)
        
        it1 = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**it1)
        v_hat = v / (1 - beta2**it1)
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h, n)
        
        if stall > 200:
            lr *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        
        if it % 400 == 0 and tau_idx < len(tau_list) - 1:
            tau_idx += 1
    
    return best_h, best_c5

def optimize_targeted(h_init, n, max_iters, time_limit, start_time):
    """Targeted subgradient: focus on top 5 shifts."""
    dx = 2.0 / n
    h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    lr = 0.02
    stall = 0
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        corr = compute_corr_full(h)
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Top 5 shifts
        k = 5
        idx = np.argsort(corr)[-k:]
        w = np.zeros(len(corr))
        w[idx] = 1.0 / k
        grad = grad_weighted(h, w)
        
        h = h - lr * grad
        h = project_to_feasible(h, n)
        
        if stall > 300:
            lr *= 0.5
            stall = 0
            if lr < 1e-12:
                break
    
    return best_h, best_c5

def local_perturb_search(h_init, n, max_iters, time_limit, start_time, rng):
    """Random perturbation local search."""
    dx = 2.0 / n
    h = project_to_feasible(h_init, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(max_iters):
        if time.time() - start_time > time_limit:
            break
        
        # Pick 2 random indices to swap
        i, j = rng.choice(n, size=2, replace=False)
        if i == j:
            continue
        eps = rng.uniform(0.01, 0.1)
        
        h_test = h.copy()
        h_test[i] -= eps
        h_test[j] += eps
        h_test = project_to_feasible(h_test, n)
        
        c5_test = compute_c5(h_test)
        if c5_test < best_c5 - 1e-12:
            h = h_test
            best_c5 = c5_test
            best_h = h_test.copy()
    
    return best_h, best_c5
```
