def make_feasible(h, n):
    h = np.clip(np.array(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break
        h[mask] += diff / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def project_onto_simplex_like(h, n):
    """Project h onto {0<=h<=1, sum(h)=n/2}"""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0

        if abs(diff) < 1e-13:
            break
        if diff > 0:

            mask = room > 1e-10

            total_room = np.sum(room[mask])
            if total_room < 1e-10:
                break

        else:

            total = np.sum(h[mask])
            if total < 1e-10:

    h_new = np.interp(new_x, old_x, h)

def subgradient_step(h, n, alpha):
    """One subgradient step on the max correlation"""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_max = np.argmax(corr)
    # Convert full correlation index to lag
    lag = k_max - (n - 1)
    # Subgradient: sg[j] = (1 - h[j+lag]) - h[j-lag]
    sg = np.zeros(n)
    # (1 - h[j+lag]) term
    if lag >= 0:
        # j+lag in [0, n-1] means j in [0, n-1-lag]
        for j in range(n - lag):
            sg[j] += (1.0 - h[j + lag])
    else:
        # j+lag in [0, n-1] means j in [-lag, n-1]
        for j in range(-lag, n):
            sg[j] += (1.0 - h[j + lag])
    # -h[j-lag] term
    if lag >= 0:
        # j-lag in [0, n-1] means j in [lag, n-1]
        for j in range(lag, n):
            sg[j] -= h[j - lag]
    else:
        # j-lag in [0, n-1] means j in [0, n-1+lag]
        for j in range(n + lag):
            sg[j] -= h[j - lag]
    # Step
    h_new = h - alpha * sg
    return project_onto_simplex_like(h_new, n)
