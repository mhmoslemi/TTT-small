```python
    return best_h, best_c5
```

```python
def swap_delta_correlation(h, i, j, n):
    """Compute delta in all correlations after swapping h[i] (1->0) and h[j] (0->1)."""
    # s ranges from -(n-1) to n-1
    # For each s, compute the change
    # c(s) = sum_m h[m](1-h[m+s]) for 0<=m<n and 0<=m+s<n
    # Only lags where i or j (or i±s or j±s) are in range are affected
    
    delta = np.zeros(2*n-1)  # indexed by p = s + (n-1)
    
    for s in range(-(n-1), n):
        d = 0.0
        # Old terms involving i:
        if 0 <= i+s < n:
            d -= (1 - h[i+s])  # was contributing h[i]*(1-h[i+s]) = 1-h[i+s]
        if 0 <= i-s < n:
            # was contributing h[i-s]*(1-h[i]) = h[i-s]*0 = 0
            # now contributes h[i-s]*(1-0) = h[i-s]
            # BUT if i-s == j, h[i-s] = h[j] = 0, and h'[i-s] = h'[j] = 1
            if i-s == j:
                d += 1  # h'[i-s] = 1
            else:
                d += h[i-s]
        # Old terms involving j:
        if 0 <= j+s < n:
            # was contributing h[j]*(1-h[j+s]) = 0
            # now contributes (1-h'[j+s])
            if j+s == i:
                d += (1 - 0)  # h'[j+s] = h'[i] = 0
            else:
                d += (1 - h[j+s])
        if 0 <= j-s < n:
            # was contributing h[j-s]*(1-h[j]) = h[j-s]*1 = h[j-s]
            # now contributes 0
            d -= h[j-s]
        
        delta[s + (n-1)] = d
    
    return delta
```

```python
def compute_swap_delta(h, i, j, n):
    """Vectorized computation of correlation changes after swapping i(1->0) and j(0->1)."""
    # s ranges from -(n-1) to n-1, stored as p = s + (n-1) in [0, 2n-2]
    s = np.arange(-(n-1), n)  # shape (2n-1,)
    
    # For term h[i+s] (old contribution to subtract): valid when 0 <= i+s < n
    i_plus_s = i + s
    mask1 = (i_plus_s >= 0) & (i_plus_s < n)
    
    # For term h[i-s] (new contribution to add): valid when 0 <= i-s < n
    i_minus_s = i - s
    mask2 = (i_minus_s >= 0) & (i_minus_s < n)
    
    # For term h[j+s] (new contribution to subtract from 1): valid when 0 <= j+s < n
    j_plus_s = j + s
    mask3 = (j_plus_s >= 0) & (j_plus_s < n)
    
    # For term h[j-s] (old contribution to subtract): valid when 0 <= j-s < n
    j_minus_s = j - s
    mask4 = (j_minus_s >= 0) & (j_minus_s < n)
    
    delta = np.zeros(2*n-1)
    
    # -h[i]*(1-h[i+s]) = -(1 - h[i+s]) for valid i+s
    # Special case: if i+s == j, h[i+s] = 0 (old), so term = -(1-0) = -1
    # This is already handled since h[j] = 0
    
    # New term h[i-s]*(1-h'[i]) = h'[i-s] = h[i-s] (or 1 if i-s==j)
    if mask2.any():
        vals = h[np.clip(i_minus_s, 0, n-1)]
        # If i-s == j, the new value is 1 (since j becomes 1)
        special_mask = (i_minus_s == j) & mask2
        vals = np.where(special_mask, 1.0, vals)
        delta[mask2] += vals[mask2]
    
    # New term h'[j]*(1-h'[j+s]) = 1 - h'[j+s]
    if mask3.any():
        vals = h[np.clip(j_plus_s, 0, n-1)]
        special_mask = (j_plus_s == i) & mask3
        vals = np.where(special_mask, 0.0, vals)  # h'[i] = 0
        delta[mask3] += (1.0 - vals)[mask3]
    
    # Old term -h[i]*(1-h[i+s]) = -(1-h[i+s])
    if mask1.any():
        vals = h[np.clip(i_plus_s, 0, n-1)]
        special_mask = (i_plus_s == j) & mask1
        vals = np.where(special_mask, 0.0, vals)  # h[j] = 0 in old
        delta[mask1] -= (1.0 - vals)[mask1]
    
    # Old term -h[j-s]*(1-h[j]) = -h[j-s]*1 = -h[j-s]
    # Wait: old term is h[j-s]*(1-h[j]) = h[j-s]*1 = h[j-s]
    # We subtract old, so: -h[j-s]
    # But if j-s == i, h[j-s] = h[i] = 1
    if mask4.any():
        vals = h[np.clip(j_minus_s, 0, n-1)]
        special_mask = (j_minus_s == i) & mask4
        vals = np.where(special_mask, 1.0, vals)  # h[i] = 1 in old
        delta[mask4] -= vals[mask4]
    
    return delta
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.arange(old_n) * 2.0 / old_n
    new_x = np.arange(new_n) * 2.0 / new_n
    # Periodic interpolation
    pad = new_n
    x_ext = np.concatenate([old_x - 2.0, old_x, old_x + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_new = np.interp(new_x, x_ext, h_ext)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    """Round to binary, ensuring exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    b = (h >= 0.5).astype(float)
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        # Need to turn some 1s to 0s
        ones_mask = b == 1.0
        ones_indices = np.where(ones_mask)[0]
        # Turn to 0 the ones with highest h value (closest to 1)
        # Actually, turn to 0 the ones that are "most 1" to minimize damage
        vals = h[ones_indices]
        order = np.argsort(vals)[::-1]  # most 1 first
        to_flip = ones_indices[order[:n_ones - target]]
        b[to_flip] = 0.0
    elif n_ones < target:
        zeros_mask = b == 0.0
        zeros_indices = np.where(zeros_mask)[0]
        vals = h[zeros_indices]
        order = np.argsort(vals)[::-1]  # closest to 1 first
        to_flip = zeros_indices[order[:target - n_ones]]
        b[to_flip] = 1.0
    return b

def swap_search(h, n, time_limit, rng):
    """Simulated annealing swap search on binary h."""
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    c5_raw = np.max(corr)
    best_c5 = c5_raw
    best_h = h.copy()
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    T0 = 50.0  # initial "temperature" in units of corr values
    T1 = 1.0
    start = time.time()
    it = 0
    
    while time.time() - start < time_limit:
        it += 1
        T = T0 * (T1 / T0) ** (it / (time_limit * 1000))  # rough schedule
        
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        if i == j:
            continue
        
        # Compute swap delta for all lags
        s = np.arange(-(n-1), n)
        
        i_ps = i + s
        m1 = (i_ps >= 0) & (i_ps < n)
        i_ms = i - s
        m2 = (i_ms >= 0) & (i_ms < n)
        j_ps = j + s
        m3 = (j_ps >= 0) & (j_ps < n)
        j_ms = j - s
        m4 = (j_ms >= 0) & (j_ms < n)
        
        delta = np.zeros(2*n-1)
        
        # From i going 1->0: lose h[i]*(1-h[i+s]) = 1-h[i+s], gain h[i-s]*(1-0) = h[i-s]
        if np.any(m1):
            idx = np.clip(i_ps[m1], 0, n-1)
            vals = h[idx]
            vals = np.where(i_ps[m1] == j, 0.0, vals)
            delta[m1] -= (1.0 - vals)
        if np.any(m2):
            idx = np.clip(i_ms[m2], 0, n-1)
            vals = h[idx]
            vals = np.where(i_ms[m2] == j, 1.0, vals)
            delta[m2] += vals
        
        # From j going 0->1: gain (1-h[j+s]), lose h[j-s]*(1-0) = h[j-s]
        if np.any(m3):
            idx = np.clip(j_ps[m3], 0, n-1)
            vals = h[idx]
            vals = np.where(j_ps[m3] == i, 0.0, vals)
            delta[m3] += (1.0 - vals)
        if np.any(m4):
            idx = np.clip(j_ms[m4], 0, n-1)
            vals = h[idx]
            vals = np.where(j_ms[m4] == i, 1.0, vals)
            delta[m4] -= vals
        
        corr_new = corr + delta
        max_new = np.max(corr_new)
        
        if max_new < c5_raw or rng.random() < np.exp(-(max_new - c5_raw) / max(T, 1e-10)):
            corr = corr_new
            c5_raw = max_new
            h[i] = 0.0
            h[j] = 1.0
            # Update ones/zeros
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if max_new < best_c5:
                best_c5 = max_new
                best_h = h.copy()
        # else: reject, don't change anything
    
    return best_h, best_c5 * (2.0 / n)
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 1.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.arange(old_n) * 2.0 / old_n
    new_x = np.arange(new_n) * 2.0 / new_n
    x_ext = np.concatenate([old_x - 2.0, old_x, old_x + 2.0])
    h_ext = np.concatenate([h, h, h])
    h_new = np.interp(new_x, x_ext, h_ext)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    h = np.asarray(h, dtype=float)
    b = (h >= 0.5).astype(float)
    n_ones = int(np.sum(b))
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(b == 1.0)[0]
        vals = h[ones_idx]
        order = np.argsort(vals)[::-1]
        b[ones_idx[order[:n_ones - target]]] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(b == 0.0)[0]
        vals = h[zeros_idx]
        order = np.argsort(vals)[::-1]
        b[zeros_idx[order[:target - n_ones]]] = 1.0
    return b

def grad_at_shift(h, s, n):
    grad = np.zeros(n)
    lo1 = max(0, -s)
    hi1 = min(n - 1, n - 1 - s)
    if lo1 <= hi1:
        grad[lo1:hi1 + 1] += (1.0 - h[lo1 + s:hi1 + s + 1])
    lo2 = max(0, s)
    hi2 = min(n - 1, n - 1 + s)
    if lo2 <= hi2:
        grad[lo2:hi2 + 1] -= h[lo2 - s:hi2 - s + 1]
    return grad

def swap_search(h, n, time_limit, rng):
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
    c5_raw = float(np.max(corr))
    best_c5 = c5_raw
    best_h = h.copy()
    s_arr = np.arange(-(n - 1), n, dtype=np.int64)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    start = time.time()
    n_total = 2 * n - 1
    it = 0
    no_improve = 0
    n_improve_limit = 20000

    while time.time() - start < time_limit:
        it += 1
        T = 30.0 * 0.999998 ** it
        i = int(ones[rng.randint(len(ones))])
        j = int(zeros[rng.randint(len(zeros))])
        if i == j:
            continue
        i_ps = i + s_arr
        m1 = (i_ps >= 0) & (i_ps < n)
        i_ms = i - s_arr
        m2 = (i_ms >= 0) & (i_ms < n)
        j_ps = j + s_arr
        m3 = (j_ps >= 0) & (j_ps < n)
        j_ms = j - s_arr
        m4 = (j_ms >= 0) & (j_ms < n)
        delta = np.zeros(n_total)
        if np.any(m1):
            vals = h[np.clip(i_ps[m1], 0, n - 1)]
            vals = np.where(i_ps[m1] == j, 0.0, vals)
            delta[m1] -= (1.0 - vals)
        if np.any(m2):
            vals = h[np.clip(i_ms[m2], 0, n - 1)]
            vals = np.where(i_ms[m2] == j, 1.0, vals)
            delta[m2] += vals
        if np.any(m3):
            vals = h[np.clip(j_ps[m3], 0, n - 1)]
            vals = np.where(j_ps[m3] == i, 0.0, vals)
            delta[m3] += (1.0 - vals)
        if np.any(m4):
            vals = h[np.clip(j_ms[m4], 0, n - 1)]
            vals = np.where(j_ms[m4] == i, 1.0, vals)
            delta[m4] -= vals
        corr_new = corr + delta
        max_new = float(np.max(corr_new))
        if max_new < c5_raw or rng.random() < np.exp(-(max_new - c5_raw) / max(T, 1e-10)):
            corr = corr_new
            c5_raw = max_new
            h[i] = 0.0
            h[j] = 1.0
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if max_new < best_c5:
                best_c5 = max_new
                best_h = h.copy()
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve > n_improve_limit and time.time() - start < time_limit * 0.7:
            h = best_h.copy()
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
            c5_raw = float(np.max(corr))
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            n_j = len(zeros) // 4

                a = int(ones[rng.randint(len(ones))])
                b = int(zeros[rng.randint(len(zeros))])
                if a != b:
                    z = h[a]
                    h[a] = h[b]
                    h[b] = z
            h = to_binary(h, n)
            corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
            c5_raw = float(np.max(corr))
            if c5_raw > best_c5:
                h = best_h.copy()
                corr = np.correlate(h, 1.0 - h, mode='full').astype(float).copy()
                c5_raw = float(np.max(corr))
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            no_improve = 0

    return best_h, best_c5 * (2.0 / n)

def _slsqp_obj(h):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    log_sum = np.log(np.sum(np.exp(shifted))) + np.max(T * (corr - mc))
    return (mc + log_sum / T) * dx

def _slsqp_grad(h):
    n = _G_N
    T = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = T * (corr - mc)
    shifted -= np.max(shifted)
    w = np.exp(shifted)
    s = np.sum(w)
    if s > 0:
        w /= s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-12:
            continue
        s_val = k - (n - 1)
        grad += w[k] * grad_at_shift(h, s_val, n)
    return grad * dx

def _cons_fun(h):
    return np.sum(h) - _G_N / 2.0

def _cons_jac(h):
    return np.ones(_G_N)

def slsqp_optimize(h0, n, T, max_iter):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    res = minimize(_slsqp_obj, h0, jac=_slsqp_grad, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def coord_descent(h, n, time_limit):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    deltas = [0.02, -0.02, 0.01, -0.01, 0.05, -0.05]
    improved_total = True
    while improved_total and time.time() - start < time_limit:
        improved_total = False
        for j in range(n):
            if time.time() - start > time_limit - 5:
                return best_h, best_c5
            old_val = h[j]
            for d in deltas:
                c_val = old_val + d
                if c_val < 0.0 or c_val > 1.0:
                    continue
                trial = h.copy()
                trial[j] = c_val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = trial
                    old_val = c_val
                    improved_total = True
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = bv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best_c5 = c
                    h = trial
                    improved_total = True
    return best_h, best_c5
```
