```python
    start = time.time()
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    best_h = h.copy()

        if time.time() - start > time_limit:
            break
        for i in range(n):
            if time.time() - start > time_limit:
                    break
            # Try golden section search for h[i]
            # Evaluate objective at a few points
            # Since it's convex in h[i], ternary search works
            old_val = h[i]
            a, b = 0.0, 1.0

            # Otherwise, ternary search
            for _ in range(30):  # 30 iterations of ternary search

                m2 = b - (b-a)/3
                h[i] = m1
                f1 = compute_c5(h)
                h[i] = m2
                f2 = compute_c5(h)
                if f1 < f2:
                    b = m2
                else:
                    a = m1
            h[i] = (a+b)/2
            c5_now = compute_c5(h)
            if c5_now < best_c5:

                best_h = h.copy()

                h[i] = old_val
```

```python
def smooth_obj_grad_fast(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    
    # Only process significant shifts
    threshold = 1e-10 * total
    significant = exp_c > threshold
    idx = np.where(significant)[0]
    
    grad = np.zeros(n)
    for k in idx:
        pk = exp_c[k] / total
        sh = k - n + 1
        if sh >= 0:
            # Part A: i in [sh, n-1]
            grad[sh:] += pk * (1.0 - h[:n-sh])
            # Part B: i in [0, n-1-sh]
            grad[:n-sh] -= pk * h[sh:]
        else:
            s = -sh
            # Part C: i in [0, n-1-s]
            grad[:n-s] += pk * (1.0 - h[s:])
            # Part D: i in [s, n-1]
            grad[s:] -= pk * h[:n-s]
    grad *= dx
    
    obj = dx * (max_c + np.log(total) / tau)
    return obj, grad
```

```python
import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    total = exp_c.sum()
    if total < 1e-300:
        idx = np.argmax(corr)
        p_k = 1.0
        sh = idx - n + 1
        grad = np.zeros(n)
        if sh >= 0:
            grad[sh:] += (1.0 - h[:n - sh])
            grad[:n - sh] -= h[sh:]
        else:
            s = -sh
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] -= h[:n - s]
        grad *= dx
        obj = dx * (max_c + np.log(total + 1e-300) / tau)
        return obj, grad
    p = exp_c / total
    threshold = 1e-12
    significant = p > threshold
    grad = np.zeros(n)
    for k in np.where(significant)[0]:
        pk = p[k]
        sh = k - n + 1
        if sh >= 0:
            grad[sh:] += pk * (1.0 - h[:n - sh])
            grad[:n - sh] -= pk * h[sh:]
        else:
            s = -sh
            grad[:n - s] += pk * (1.0 - h[s:])
            grad[s:] -= pk * h[:n - s]
    grad *= dx
    obj = dx * (max_c + np.log(total) / tau)
    return obj, grad


def optimize_smooth(h, n, tau, time_limit, lr=0.1, max_iter=20000):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    stall = 0
    cur_lr = lr
    for it in range(max_iter):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_grad(h, tau)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            cur_lr *= 0.5
            m *= 0.1
            v *= 0.1
            stall = 0
            if cur_lr < 1e-12:
                break
        t += 1
        m = b1 * m + (1 - b1) * grad
        v = b2 * v + (1 - b2) * grad * grad
        m_hat = m / (1 - b1 ** t)
        v_hat = v / (1 - b2 ** t)
        h = project_to_feasible(h - cur_lr * m_hat / (np.sqrt(v_hat) + eps), n)
    return best_h, best_c5


def optimize_subgrad(h, n, time_limit, lr=0.002, top_k=3):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    cur_lr = lr
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            cur_lr *= 0.6
            v *= 0.3
            stall = 0
            if cur_lr < 1e-14:
                break
        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for ii, k in enumerate(top_idx):
            sh = k - n + 1
            if sh >= 0:
                grad[sh:] += w[ii] * (1.0 - h[:n - sh])
                grad[:n - sh] -= w[ii] * h[sh:]
            else:
                s = -sh
                grad[:n - s] += w[ii] * (1.0 - h[s:])
                grad[s:] -= w[ii] * h[:n - s]
        grad *= dx
        v = beta * v - cur_lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def coordinate_slip(h, n, time_limit, n_coords=20, n_sweeps=50):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    rng = np.random.default_rng(12345)
    for sweep in range(n_sweeps):
        if time.time() - start > time_limit:
            break
        coords = rng.choice(n, size=min(n_coords, n), replace=False)
        for i in coords:
            if time.time() - start > time_limit:
                break
            old = h[i]
            h[i] = 0.0
            f0 = compute_c5(h)
            h[i] = 1.0
            f1 = compute_c5(h)
            h[i] = old
            fmid = compute_c5(h)
            if f0 <= f1 and f0 <= fmid:
                h[i] = 0.0
            elif f1 <= f0 and f1 <= fmid:
                h[i] = 1.0
            else:
                a, b = 0.0, 1.0
                for _ in range(25):
                    m1 = a + (b - a) / 3.0
                    m2 = b - (b - a) / 3.0
                    h[i] = m1
                    f_m1 = compute_c5(h)
                    h[i] = m2
                    f_m2 = compute_c5(h)
                    if f_m1 < f_m2:
                        b = m2
                    else:
                        a = m1
                h[i] = (a + b) / 2.0
            c5_now = compute_c5(h)
            if c5_now < best_c5:
                best_c5 = c5_now
                best_h = h.copy()
    return best_h, best_c5
```
