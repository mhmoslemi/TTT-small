import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_objective(h, n, t):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(t * shifted))) / t) * dx

def slsqp_optimize(h_start, n, t, maxiter):
    def obj(h):
        return soft_objective(h, n, t)
    def con_fun(h):
        return np.sum(h) - n / 2.0
    def con_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    try:
        res = minimize(obj, h_start, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.success or np.sum(res.x**2) > 0:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return make_feasible(h_start, n)

def coord_descent(h, n, max_rounds=30, time_limit=None, start_time=None):
    dx = 2.0 / n
    best_c = compute_c5(h)
    for rnd in range(max_rounds):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit - 15:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit - 15:
                    return h, best_c
            old = h[j]
            candidates = [0.0, 0.2, 0.4, 0.5, 0.6, 0.8, 1.0,
                          old + 0.05, old - 0.05,
                          old + 0.02, old - 0.02,
                          old + 0.01, old - 0.01]
            best_val = old
            best_obj = best_c
            for cv in candidates:
                if cv < -1e-9 or cv > 1 + 1e-9:
                    continue
                cv = np.clip(cv, 0.0, 1.0)
                trial = h.copy()
                trial[j] = cv
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_obj - 1e-12:
                    best_obj = c
                    best_val = cv
            if best_val != old:
                h = h.copy()
                h[j] = best_val
                h = make_feasible(h, n)
                best_c = best_obj
                improved = True
        if not improved:
            break
    return h, best_c

            c_new = compute_c5(h_new)
            if c_new < best_c5 - 1e-8:
                best_c5 = c_new
                best_h = h_new
                h4 = h_new
            else:
                h4 = h_new
        # Coordinate descent on new resolution
        n_cd2 = len(best_h)
        h_cd2, c_cd2 = coord_descent(best_h, n_cd2, max_rounds=10,
                                     time_limit=budget_s, start_time=start_time)
        if c_cd2 < best_c5:
            best_c5 = c_cd2
            best_h = h_cd2

    # Final: binary thresholding attempt
    if time.time() - start_time < budget_s - 30:
        h_bin = (best_h > 0.5).astype(float)
        h_bin = make_feasible(h_bin, len(best_h))
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5 - 1e-8:
            best_c5 = c_bin
            best_h = h_bin
            h_cd3, c_cd3 = coord_descent(best_h, len(best_h), max_rounds=5,
                                         time_limit=budget_s, start_time=start_time)
            if c_cd3 < best_c5:
                best_c5 = c_cd3
                best_h = h_cd3

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
