                break

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_to_feasible(h - step, n)

    return best_h, best_obj


def optimize_momentum(h, n, tau, max_iters, time_limit, lr=0.25, beta=0.9, topk=50):
    h = project_to_feasible(h, n).copy()
    best_obj = np.inf
    best_h = h.copy()
    start = time.time()
    vel = np.zeros(n)
    stall = 0

    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break

        obj, grad = smooth_obj_grad(h, tau, topk=topk)

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 200:
            lr *= 0.5
            vel *= 0.3
            stall = 0
            if lr < 1e-10:
                break

        vel = beta * vel - lr * grad
        h = project_to_feasible(h + vel, n)

    return best_h, best_obj


def optimize_multi_tau(h, n, taus, max_iters, time_limit, lr=0.2, beta=0.9, topk=50):
    """Cycle through different tau values to escape traps."""
    h = project_to_feasible(h, n).copy()
    best_obj = np.inf
    best_h = h.copy()
    start = time.time()
    vel = np.zeros(n)
    stall = 0
    tau_idx = 0

    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break

        tau = taus[tau_idx % len(taus)]
        tau_idx += 1

        obj, grad = smooth_obj_grad(h, tau, topk=topk)

        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 300:
            lr *= 0.5
            vel *= 0.3
            stall = 0
            if lr < 1e-10:
                break

        vel = beta * vel - lr * grad
        h = project_to_feasible(h + vel, n)

    return best_h, best_obj


def optimize_subgrad(h, n, max_iters, time_limit, lr=0.003, beta=0.85):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    vel = np.zeros(n)
    stall = 0
    dx = 2.0 / n

    for it in range(max_iters):
        elapsed = time.time() - start
        if elapsed > time_limit:
            break

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 800:
            lr *= 0.4
            vel *= 0.3
            stall = 0
            if lr < 1e-13:
                break

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            grad[j0a:j1a] -= h[j0a + sh:j1a + sh]
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] += (1.0 - h[j0b - sh:j1b - sh])
        grad *= dx

        vel = beta * vel - lr * grad
        h = project_to_feasible(h + vel, n)

    return best_h, best_c5


def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - start_time))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Adaptive temperature at initial resolution
    if remaining() > 80:
        tl = min(150.0, remaining() * 0.12)
        h1, _ = optimize_adaptive(h_init, n_init, 3.0, 100.0,
                                   max_iters=500000, time_limit=tl,
                                   lr=0.12, topk=80)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()

    # Phase 2: Python multi-tau at initial resolution
    if remaining() > 80:
        tl = min(120.0, remaining() * 0.10)
        h2, _ = optimize_multi_tau(best_h, n_init, [10, 30, 60, 100],
                                    max_iters=500000, time_limit=tl,
                                    lr=0.2, topk=60)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Higher resolution sweeps
    resolutions = []
    for mult in [2, 3, 4, 6]:
        nr = n_init * mult
        if nr <= 8000:
            resolutions.append(nr)

    for n_r in resolutions:
        if remaining() < 100:
            break
        h_r = interp_periodic(best_h, n_r)
        h_r = project_to_feasible(h_r, n_r)
        # Optimize with momentum
        tl = min(100.0, remaining() * 0.10)
        h_r, _ = optimize_momentum(h_r, n_r, tau=40.0,
                                    max_iters=300000, time_limit=tl,
                                    lr=0.15, topk=80)
        # Polish with subgradient
        if remaining() > 40:
            h_r, c5_r = optimize_subgrad(h_r, n_r, max_iters=200000,
                                          time_limit=min(40.0, remaining() * 0.04),
                                          lr=0.002)
        else:
            c5_r = compute_c5(h_r)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 4: Random restarts at best resolution
    n_best = len(best_h)
    n_restarts = 6
    if remaining() > n_restarts * 40:
        scales = [0.05, 0.1, 0.15, 0.08, 0.12, 0.2]
        for r in range(n_restarts):
            if remaining() < 50:
                break
            scale = scales[r % len(scales)] * (2.0 / n_best)
            h_r = best_h + np.random.randn(n_best) * scale
            h_r = project_to_feasible(h_r, n_best)
            tl = min(45.0, remaining() / (n_restarts - r))
            h_r, _ = optimize_momentum(h_r, n_best, tau=50.0,
                                        max_iters=200000, time_limit=tl,
                                        lr=0.1, topk=60)
            c5_r = compute_c5(h_r)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 5: Final subgradient polish
    if remaining() > 40:
        h_f = best_h.copy()
        n_f = len(h_f)
        tl = remaining() * 0.35
        h_f, c5_f = optimize_subgrad(h_f, n_f, max_iters=500000,
                                       time_limit=tl, lr=0.001, beta=0.8)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 6: One more adaptive pass if time remains
    if remaining() > 60:
        h_a = best_h.copy()
        n_a = len(h_a)
        tl = remaining() * 0.4
        h_a, _ = optimize_adaptive(h_a, n_a, 20.0, 200.0,
                                    max_iters=300000, time_limit=tl,
                                    lr=0.05, topk=100)
        c5_a = compute_c5(h_a)
        if c5_a < best_c5:
            best_c5 = c5_a
            best_h = h_a.copy()

    # Ensure feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
