        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-13:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_tiled = np.concatenate([h, h, h])
    x_tiled = np.concatenate([
        np.arange(old_n) * 2.0 / old_n,
        (np.arange(old_n) + old_n) * 2.0 / old_n,
        (np.arange(old_n) + 2 * old_n) * 2.0 / old_n
    ])
    h_new = np.interp(new_x, x_tiled, h_tiled)
    return make_feasible(h_new, new_n)

def smooth_max_value(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    return np.max(corr) + np.log(np.sum(np.exp(T * shifted))) / T

def _slsqp_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    return np.max(corr) + np.log(np.sum(np.exp(T * shifted))) / T

def _slsqp_cons(h, n):
    return np.sum(h) - n / 2.0

def _slsqp_jac(h, n):
    return np.ones(n)

def _slsqp_run(h_start, n, T, max_iter):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _slsqp_cons, 'jac': _slsqp_jac, 'args': (n,)}]
    res = minimize(_slsqp_obj, h_start, args=(n, T), method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def pg_gradient(h, T):
    """Compute gradient of smooth max objective."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    shifted = corr - np.max(corr)
    exp_vals = np.exp(T * shifted)
    weights = exp_vals / np.sum(exp_vals)
    
    center = n - 1
    grad = np.zeros(n)
    
    # Lag 0
    p0 = weights[center]
    grad += p0 * (1.0 - 2.0 * h) * dx
    
    # All other lags
    for s in range(1, n):
        wp = weights[center + s] if (center + s) < len(weights) else 0.0
        wn = weights[center - s] if (center - s) >= 0 else 0.0
        if wp + wn < 1e-14:
            continue
        if wp > 1e-14:
            grad[:n - s] += wp * (1.0 - h[s:]) * dx
            grad[s:] -= wp * h[:n - s] * dx
        if wn > 1e-14:
            grad[s:] += wn * (1.0 - h[:n - s]) * dx
            grad[:n - s] -= wn * h[s:] * dx
    return grad

def projected_gradient_opt(h_init, n, T, num_steps, base_alpha, timeout_check):
    """Projected gradient with backtracking line search."""
    h = h_init.copy()
    f_curr = smooth_max_value(h, T)
    best_h = h.copy()
    best_f = compute_c5(h)
    
    for step in range(num_steps):
        if timeout_check() < 10:
            break
        grad = pg_gradient(h, T)
        # Backtracking line search
        alpha = base_alpha
        for _ in range(20):
            h_trial = h - alpha * grad
            h_trial = make_feasible(h_trial, n)
            f_trial = smooth_max_value(h_trial, T)
            if f_trial < f_curr - 1e-15:
                h = h_trial
                f_curr = f_trial
                c = compute_c5(h)
                if c < best_f:
                    best_f = c
                    best_h = h.copy()
                break
            alpha *= 0.5
        if alpha < 1e-10:
            # Try smaller T or break
            break
    return best_h, best_f

def local_search_groups(h, n, group_sizes, max_swaps, timeout_check):
    """Try modifying small groups of adjacent entries."""
    best_h = h.copy()
    best_c = compute_c5(h)
    rng = np.random.RandomState(12345)
    
    for _ in range(max_swaps):
        if timeout_check() < 5:
            break
        gs = rng.choice(group_sizes)
        start = rng.randint(0, n - gs + 1)
        # Try setting group to various patterns
        for pattern_seed in range(3):
            if pattern_seed == 0:
                group = np.full(gs, 1.0)
            elif pattern_seed == 1:
                group = np.full(gs, 0.0)
            else:
                group = np.full(gs, 0.5)
            trial = best_h.copy()
            trial[start:start + gs] = group
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            if c < best_c - 1e-12:
                best_c = c
                best_h = trial.copy()
    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    
    def _time_left():
        return budget_s - (time.time() - start_time)
    
    # Get parent
    try:
        h_parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_parent = np.tile([1.0, 0.0], 320)
    
    best_h = h_parent.copy()
    best_c5 = compute_c5(h_parent)
    
    # Phase 1: SLSQP at n=640 with high T
    if _time_left() > 60:
        n1 = len(h_parent)
        for T in [200.0, 800.0]:
            if _time_left() < 20:
                break
            try:
                h_opt = _slsqp_run(best_h, n1, T, 200)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()
            except Exception:
                pass
    
    # Phase 2: Upsample to n=1280, SLSQP
    if _time_left() > 120:
        n2 = 1280
        h2 = upsample_h(best_h, n2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
        
        for T in [100.0, 500.0, 2000.0]:
            if _time_left() < 40:
                break
            try:
                h_opt = _slsqp_run(h2, n2, T, 300)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()
                    h2 = h_opt.copy()
                else:
                    h2 = h_opt.copy()
            except Exception:
                pass
    
    # Phase 3: Projected gradient at n=1280
    if _time_left() > 100:
        n3 = 1280
        h3 = upsample_h(best_h, n3)
        c3 = compute_c5(h3)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()
        
        for T_pg in [20.0, 100.0, 400.0]:
            if _time_left() < 30:
                break
            h_pg, c_pg = projected_gradient_opt(h3, n3, T_pg, 3000, 0.05, _time_left)
            if c_pg < best_c5:
                best_c5 = c_pg
                best_h = h_pg.copy()
                h3 = h_pg.copy()
    
    # Phase 4: Upsample to n=2048, projected gradient + SLSQP
    if _time_left() > 150:
        n4 = 2048
        h4 = upsample_h(best_h, n4)
        c4 = compute_c5(h4)
        if c4 < best_c5:
            best_c5 = c4
            best_h = h4.copy()
        
        # Multiple T for projected gradient
        for T_pg in [10.0, 50.0, 200.0]:
            if _time_left() < 30:
                break
            h_pg, c_pg = projected_gradient_opt(h4, n4, T_pg, 5000, 0.03, _time_left)
            if c_pg < best_c5:
                best_c5 = c_pg
                best_h = h_pg.copy()
                h4 = h_pg.copy()
        
        # SLSQP polish
        if _time_left() > 50:
            try:
                h_opt = _slsqp_run(h4, n4, 1000.0, 200)
                c = compute_c5(h_opt)
                if c < best_c5:
                    best_c5 = c
                    best_h = h_opt.copy()
                    h4 = h_opt.copy()
            except Exception:
                pass
    
    # Phase 5: Local search groups on current best
    if _time_left() > 30:
        n5 = len(best_h)
        h_ls, c_ls = local_search_groups(best_h, n5, [1, 2, 3, 4, 5], 500, _time_left)
        if c_ls < best_c5:
            best_c5 = c_ls
            best_h = h_ls.copy()
    
    # Phase 6: Final SLSQP polish
    if _time_left() > 25:
        nf = len(best_h)
        try:
            h_opt = _slsqp_run(best_h, nf, 3000.0, 150)
            c = compute_c5(h_opt)
            if c < best_c5:
                best_c5 = c
                best_h = h_opt.copy()
        except Exception:
            pass
    
    # Phase 7: Another projected gradient pass at very high T
    if _time_left() > 30:
        nr = len(best_h)
        h_rg, c_rg = projected_gradient_opt(best_h, nr, 500.0, 3000, 0.01, _time_left)
        if c_rg < best_c5:
            best_c5 = c_rg
            best_h = h_rg.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
