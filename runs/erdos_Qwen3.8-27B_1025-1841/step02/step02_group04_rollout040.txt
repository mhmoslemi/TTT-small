import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Convert to binary with exactly n/2 ones."""
    h = h.copy()
    h[h > 0.5] = 1.0
    h[h <= 0.5] = 0.0
    n_ones = int(np.sum(h))
    target = n // 2
    rng = np.random.RandomState(42)
    if n_ones > target:
        ones_idx = np.where(h == 1)[0]
        excess = n_ones - target
        # Prefer keeping ones that are "better" (lower local correlation)
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Score each position by how much removing it would help
        # Just use random for simplicity
        flip = rng.choice(ones_idx, excess, replace=False)
        h[flip] = 0.0
    elif n_ones < target:
        zeros_idx = np.where(h == 0)[0]
        deficit = target - n_ones
        flip = rng.choice(zeros_idx, deficit, replace=False)
        h[flip] = 1.0
    return h

def thue_morse_init(n, seed=42):
    rng = np.random.RandomState(seed)
    t = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        t[i] = t[i >> 1] ^ (i & 1)
    # Shift to try different phases
    shift = rng.randint(n)
    t = np.roll(t, shift)
    # Ensure exactly n/2 ones
    n_ones = np.sum(t)
    target = n // 2
    if n_ones > target:
        ones_idx = np.where(t == 1)[0]
        excess = n_ones - target
        flip = rng.choice(ones_idx, excess, replace=False)
        t[flip] = 0
    elif n_ones < target:
        zeros_idx = np.where(t == 0)[0]
        deficit = target - n_ones
        flip = rng.choice(zeros_idx, deficit, replace=False)
        t[flip] = 1
    return t.astype(float)

def greedy_binary_search(h, n, max_rounds=50, time_limit=None, start_time=None):
    """Greedy local search: at each step, find the best pair swap."""
    h = to_binary(h, n)
    rng = np.random.RandomState(42)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    s_range = np.arange(-(n-1), n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    best_h = h.copy()
    
    for r in range(max_rounds):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        
        # Precompute A and B
        # A[p, s_idx] = h[p+s] if valid
        # B[p, s_idx] = 1-h[p-s] if valid
        A = np.zeros((n, n_corr))
        B = np.zeros((n, n_corr))
        for p in range(n):
            valid_A = (p + s_range >= 0) & (p + s_range < n)
            A[p, valid_A] = h[p + s_range[valid_A]]
            valid_B = (p - s_range >= 0) & (p - s_range < n)
            B[p, valid_B] = 1.0 - h[p - s_range[valid_B]]
        
        best_delta = 0
        best_pair = None
        current_max = np.max(corr)
        
        for i in ones_idx:
            delta_i = A[i] - B[i]
            base = corr + delta_i
            
            # For all zeros:
            delta_j = B[zeros_idx] - A[zeros_idx]
            new_maxes = np.max(base[None, :] + delta_j, axis=1)
            
            min_val = np.min(new_maxes)
            if min_val < current_max + best_delta:
                best_delta = min_val - current_max
                best_pair = (i, zeros_idx[np.argmin(new_maxes)])
        
        if best_pair is None:
            break
        
        i, j = best_pair
        h[i] = 0.0
        h[j] = 1.0
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

def sa_binary_search(h, n, max_iters=50000, T_start=0.005, T_end=1e-6, time_limit=None, start_time=None):
    """Simulated annealing on binary sequence."""
    h = to_binary(h, n)
    rng = np.random.RandomState(123)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    s_range = np.arange(-(n-1), n)
    
    corr = np.correlate(h, 1.0 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    best_h = h.copy()
    
    for it in range(max_iters):
        if time_limit is not None:
            if time.time() - start_time > time_limit:
                break
        
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_corr = np.correlate(h, 1.0 - h, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 <= c5 or rng.rand() < np.exp(-(new_c5 - c5) / T):
            c5 = new_c5
            corr = new_corr
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c5

def gradient_refine(h, n, n_iter=3000, T_start=30.0, T_end=1.0, lr=0.1):
    """Soft-max gradient descent."""
    n_params = {'n': n, 'T': T_start}
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        exp_s = np.exp(T * shifted)
        total = np.sum(exp_s)
        weights = exp_s / total
        
        # Top-k weighted subgradient
        top_k = 20
        idx = np.argsort(weights)[-top_k:][::-1]
        grad = np.zeros(n)
        n_corr = 2 * n - 1
        for k_idx in idx:
            w = weights[k_idx]
            if w < 1e-8:
                break
            s = k_idx - (n - 1)
            if s >= 0:
                grad[s:] += w * (1.0 - h[:n-s])
                grad[:n-s] += w * (-h[s:])
            else:
                grad[:n-s] += w * (1.0 - h[s:])
                grad[s:] += w * (-h[:n-s])
        
        h = h - lr_t * grad * (2.0 / n)
        h = make_feasible(h, n)
        
        if (it + 1) % 100 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

    c5 = np.max(corr) * dx
    best_c5 = c5
