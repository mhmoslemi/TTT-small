from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return np.clip(h, 0, 1).copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def project_to_binary(h, n):
    """Project h to binary with exactly n/2 ones."""
    target = n // 2
    h_sorted = np.argsort(-h)  # descending
    binary = np.zeros(n)
    binary[h_sorted[:target]] = 1.0
    return binary

    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)

            break
        if diff > 0:

        else:

            break
        h[mask] = np.clip(h[mask] + diff / cnt, 0.0, 1.0)

        return np.clip(h.copy(), 0, 1)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

    target = n // 2
    idx = np.argsort(-h)[:target]
    b = np.zeros(n)
    b[idx] = 1.0
    return b

    idx = rng.choice(n, size=n_ones, replace=False)
    b = np.zeros(n)
    b[idx] = 1.0
    return b

    """Create a binary pattern with alternating blocks of given size."""
    n_ones = n // 2
    b = np.zeros(n)
    # Create alternating blocks
    pattern = []
    fill = 0
    while fill < n:
        # Add a 1-block
        size = min(block_size, n - fill)
        if size > 0:
            b[fill:fill+size] = 1
            fill += size
        # Add a 0-block
        size = min(block_size, n - fill)
        if size > 0:
            fill += size
    # Adjust to have exactly n_ones ones
    n_current_ones = int(np.sum(b))
    if n_current_ones > n_ones:
        # Remove some ones from the end
        ones_idx = np.where(b == 1)[0]
        for idx in reversed(ones_idx):
            if n_current_ones == n_ones:
                break
            b[idx] = 0
            n_current_ones -= 1
    elif n_current_ones < n_ones:

        for idx in zeros_idx:
            if n_current_ones == n_ones:
                break
            b[idx] = 1
            n_current_ones += 1
    return b

        if delta < 0 or rng.rand() < np.exp(-delta / max(T_current, 1e-10)):
            c5 = new_c5

            ones = np.where(h == 1)[0]
            zeros = np.where(h == 0)[0]
            if c5 < best_c5_c:

                best_h_c = h.copy()
                if c5 < best_c5[0]:
                    best_c5[0] = c5
                    best_h[0] = h.copy()
        else:
            h[i] = 1; h[j] = 0
    
    return best_h_c, best_c5_c
