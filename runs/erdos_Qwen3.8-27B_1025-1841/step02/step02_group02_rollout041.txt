def project_to_feasible(g, n):
    target = n / 2.0
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    one_minus_h = 1.0 - h
    grad_h = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, -sh)
        j1 = min(n, n - sh)

        if j0b < j1b:
            grad_h[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad_h *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad_h
