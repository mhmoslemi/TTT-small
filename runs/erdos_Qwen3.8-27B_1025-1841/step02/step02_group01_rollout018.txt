```python
def smooth_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Softmax weights
    max_c = np.max(corr)
    exp_c = np.exp(T * (corr - max_c))
    p = exp_c / np.sum(exp_c)  # length 2n-1
    
    # Objective
    obj = (max_c / T + np.log(np.sum(exp_c)) / T) * dx  # = (1/T)*log(sum exp(T*corr)) * dx
    
    # Gradient
    grad = np.zeros(n)
    # For each i, sum over k: p[k] * ((1-h[i-m(k)]) - h[i+m(k)])
    # where m(k) = k - (n-1), and we need 0≤i±m(k)<n
    
    # Vectorized: for each lag m, the contribution to grad[i] is:
    # p[m+(n-1)] * (1-h[i-m]) - p[m+(n-1)] * h[i+m]
    # = p[m+(n-1)] * (1-h[i-m]) - p[m+(n-1)] * h[i+m]
    
    # This is: (correlation of p with (1-h)) - (correlation of p with h) at the appropriate shift
    
    # Actually: sum_m p(m) * (1-h[i-m]) = (p convolved with (1-h))[i] (cross-correlation)
    # And: sum_m p(m) * h[i+m] = (p convolved with h)[i] with a different convention
    
    # Let me just do it with a loop for clarity (n is at most ~1000)
    for i in range(n):
        s = 0.0
        for k in range(2*n-1):
            m = k - (n-1)
            if 0 <= i-m < n and 0 <= i+m < n:
                s += p[k] * ((1.0 - h[i-m]) - h[i+m])
            elif 0 <= i-m < n:
                s += p[k] * (1.0 - h[i-m])
            elif 0 <= i+m < n:
                s += p[k] * (-h[i+m])
        grad[i] = s * dx
    
    return obj, grad
```
