```python
def project_to_feasible(g, n):
    """Project g onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Use bisection on the offset
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def compute_grad_for_lag(h, d, n):
    """Gradient of corr[d] with respect to h"""
    # corr[d] = sum_i h[i] * (1-h[i+d]) for valid i
    # d_cd/dh[i] = (1-h[i+d]) if i+d valid, plus contribution from h[i-d] term
    grad = np.zeros(n)
    # corr[d] = sum_{i} h[i] * (1-h[i+d]) where 0<=i<n and 0<=i+d<n
    # = sum_{i: max(0,-d)<= i < min(n, n-d)} h[i] * (1-h[i+d])
    
    # For a fixed d, the valid range of i is:
    # 0 <= i < n and 0 <= i+d < n
    # i.e., max(0, -d) <= i < min(n, n-d)
    
    i_start = max(0, -d)
    i_end = min(n, n - d)
    if i_start >= i_end:
        return grad
    
    idx = np.arange(i_start, i_end)
    # d(corr[d])/dh[i] for i in [i_start, i_end):
    # The term h[i]*(1-h[i+d]) contributes (1-h[i+d]) to d/dh[i]
    # Also, h[i] appears as h[(i-d)+d] in the term for index (i-d):
    #   h[i-d]*(1-h[i]) contributes -h[i-d] to d/dh[i] (if i-d is valid)
    
    # First part: (1-h[i+d]) for i in valid range
    grad[idx] += (1.0 - h[idx + d])
    
    # Second part: -h[i-d] for i where i-d is in valid range
    # i-d is valid means max(0,-d) <= i-d < min(n, n-d)
    # i.e., max(0,-d)+d <= i < min(n, n-d)+d
    i2_start = max(0, -d) + d
    i2_end = min(n, n - d) + d
    i2_start = max(0, i2_start)
    i2_end = min(n, i2_end)
    if i2_start < i2_end:
        idx2 = np.arange(i2_start, i2_end)
        grad[idx2] -= h[idx2 - d]
    
    return grad

def softmax_grad_optimize(h, n, tau, max_iters, time_limit, lr_init=0.2, topM=0):
    """Optimize using softmax-weighted gradient"""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(50, max_iters // 20)
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        g = corr * dx
        cur_max = g.max()
        
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Compute softmax weights
        if topM > 0 and topM < (2 * n - 1):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx]
            max_top = g_top.max()
            shifted = np.clip(tau * (g_top - max_top), -500, 500)
            p_top = np.exp(shifted)
            p_top /= p_top.sum()
        else:
            max_g = g.max()
            shifted = np.clip(tau * (g - max_g), -500, 500)
            p = np.exp(shifted)
            p /= p.sum()
            p_top = p
            top_idx = np.arange(len(g))
        
        # Compute weighted gradient
        grad = np.zeros(n)
        for i in range(len(top_idx)):
            pk = p_top[i]
            if pk < 1e-12:
                continue
            k = top_idx[i]
            d = k - (n - 1)  # convert from correlate index to lag
            grad += pk * compute_grad_for_lag(h, d, n)
        grad *= dx
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
    
    return best_h, best_obj
```

```python
import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def full_corr_fft(h):
    n = len(h)
    N = 1 << (2 * n - 2).bit_length()
    A = np.fft.rfft(h, N)
    B = np.fft.rfft(1.0 - h, N)
    R = np.fft.irfft(A * np.conj(B), N)
    return np.concatenate([R[N - n + 1:], R[:n]])

def compute_c5(h):
    n = len(h)
    if n > 3000:
        corr = full_corr_fft(h)
    else:
        corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_grad_lags(h, lags, n):
    """Compute sum_k p_k * d(c_k)/dh for given lags and weights"""
    # lags: array of lag indices (0 to 2n-2 in numpy correlate convention)
    # We need to convert to actual lag d = lag_idx - (n-1)
    grad = np.zeros(n)
    for i, lag_idx in enumerate(lags):
        d = lag_idx - (n - 1)
        if d == 0:
            grad += (1.0 - 2.0 * h)
        else:
            # c_d = sum_j h[j+d] * (1-h[j]) for valid j
            # d(c_d)/dh[j] = (1-h[j-d]) * 1{0≤j-d<n} - h[j+d] * 1{0≤j+d<n}
            # Part 1: (1-h[j-d]) for j where 0≤j-d<n → d≤j<n+d
            j1_start = max(0, d)
            j1_end = min(n, n + d)
            if j1_start < j1_end:
                grad[j1_start:j1_end] += (1.0 - h[j1_start - d:j1_end - d])
            # Part 2: -h[j+d] for j where 0≤j+d<n → -d≤j<n-d
            j2_start = max(0, -d)
            j2_end = min(n, n - d)
            if j2_start < j2_end:
                grad[j2_start:j2_end] -= h[j2_start + d:j2_end + d]
    return grad

def softmax_grad_step(h, n, tau, topM, use_fft):
    """One step of softmax gradient descent with projection"""
    dx = 2.0 / n
    if use_fft:
        corr = full_corr_fft(h)
    else:
        corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    
    if topM > 0 and topM < (2 * n - 1):
        top_idx = np.argpartition(g, -topM)[-topM:]
        g_top = g[top_idx]
        max_top = g_top.max()
        shifted = np.clip(tau * (g_top - max_top), -500, 500)
        p_top = np.exp(shifted)
        p_top /= p_top.sum()
        # Filter out negligible weights
        mask = p_top > 1e-10
        top_idx = top_idx[mask]
        p_top = p_top[mask]
    else:
        max_g = g.max()
        shifted = np.clip(tau * (g - max_g), -500, 500)
        p = np.exp(shifted)
        p /= p.sum()
        top_idx = np.arange(len(g))
        p_top = p
    
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p_top[i]
        if pk < 1e-12:
            continue
        d = top_idx[i] - (n - 1)
        if d == 0:
            grad += pk * (1.0 - 2.0 * h)
        else:
            j1s = max(0, d)
            j1e = min(n, n + d)
            if j1s < j1e:
                grad[j1s:j1e] += pk * (1.0 - h[j1s - d:j1e - d])
            j2s = max(0, -d)
            j2e = min(n, n - d)
            if j2s < j2e:
                grad[j2s:j2e] -= pk * h[j2s + d:j2e + d]
    
    return grad * dx, g.max()

def optimize_softmax(h, n, time_limit, tau_init=10.0, tau_final=200.0, 
                     lr_init=0.15, topM=None, use_fft=True):
    """Main optimization loop with adaptive temperature"""
    if topM is None:
        topM = min(200, 2 * n - 1)
    
    h = project_to_feasible(h, n).copy()
    best_obj = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = 100
    total_iters = 0
    n_corr_calls = 0
    
    while time.time() - start < time_limit:
        it_frac = min(1.0, total_iters / 2000.0)
        tau = tau_init + (tau_final - tau_init) * it_frac
        
        grad, cur_max = softmax_grad_step(h, n, tau, topM, use_fft)
        n_corr_calls += 1
        
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
        total_iters += 1
    
    return best_h, best_obj, total_iters
```

```python
        h3, c5_3, _ = optimize_softmax(h3, n3, time_limit=min(450.0, remaining() * 0.45),
                                        tau_init=20.0, tau_final=350.0,
                                        lr_init=0.1, topM=250, use_fft=True)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Subgradient polish at current best
    if remaining() > 40:
        n_f = len(best_h)
        h_f, c5_f = subgrad_polish(best_h, n_f, time_limit=min(80.0, remaining() * 0.15), topK=3)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 5: Structured perturbations to escape local minima
    if remaining() > 80:
        n_p = len(best_h)
        perturbations = []
        # Random noise at various scales
        for sc in [0.02, 0.04, 0.06]:
            perturbations.append(sc * np.random.randn(n_p))
        # Sinusoidal at various frequencies
        for freq in [1, 2, 3, 5]:
            perturbations.append(0.04 * np.sin(2 * np.pi * freq * np.arange(n_p) / n_p))
        # Block flips
        for _ in range(3):
            bs = np.random.randint(0, n_p // 3)
            bl = np.random.randint(n_p // 25, n_p // 12)
            p = np.zeros(n_p)
            p[bs:bs + bl] = 0.5
            perturbations.append(p)

        for p in perturbations:
            if remaining() < 30:
                break
            h_test = project_to_feasible(best_h + p, n_p)
            tlim = min(50.0, remaining() * 0.12)
            ht, ct, _ = optimize_softmax(h_test, n_p, time_limit=tlim,
                                          tau_init=30.0, tau_final=300.0,
                                          lr_init=0.1, topM=250, use_fft=True)
            if ct < best_c5:
                best_c5 = ct
                best_h = ht.copy()

    # Phase 6: Final subgradient polish
    if remaining() > 20:
        n_final = len(best_h)
        h_final, c5_final = subgrad_polish(best_h, n_final, time_limit=remaining() * 0.45, topK=3)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Phase 7: One more short softmax pass if time allows
    if remaining() > 30:
        n_s = len(best_h)
        hs, cs, _ = optimize_softmax(best_h, n_s, time_limit=remaining() * 0.3,
                                      tau_init=50.0, tau_final=400.0,
                                      lr_init=0.05, topM=300, use_fft=True)
        if cs < best_c5:
            best_c5 = cs
            best_h = hs.copy()

    # Final feasibility check
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
```
