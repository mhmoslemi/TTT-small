import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def obj_grad(h, n, t, mu):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    tc = t * corr
    tc_max = np.max(tc)
    tc_shifted = tc - tc_max
    exp_vals = np.exp(tc_shifted)
    sum_exp = np.sum(exp_vals)
    soft = (tc_max + np.log(sum_exp)) / t
    obj = soft * dx + mu * (np.sum(h) - n / 2.0) ** 2
    w = exp_vals / sum_exp
    first_terms = np.correlate(1.0 - h, w, 'full')[-n:][::-1]
    second_terms = np.convolve(h, w, 'full')[n - 1:2 * n - 1]
    grad_soft = first_terms - second_terms
    grad = grad_soft * dx + 2.0 * mu * (np.sum(h) - n / 2.0) * np.ones(n)
    return obj, grad

def lse_wrapped(h_flat, n, t, mu):
    h = np.clip(h_flat, 0.0, 1.0)
    return obj_grad(h, n, t, mu)

def coordinate_descent(h, n, max_iter=200, time_limit=None, start_time=None):
    dx = 2.0 / n
    best_c5 = compute_c5(h)
    improved = True
    it = 0
    while improved and it < max_iter:
        improved = False
        it += 1
        if time_limit and start_time and (time.time() - start_time > time_limit):
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_lag = np.argmax(corr)
        for j in range(n):
            if time_limit and start_time and (time.time() - start_time > time_limit):
                break
            old = h[j]
            best_local = best_c5
            best_local_h = None
            for delta in [0.02, -0.02, 0.05, -0.05, 0.01, -0.01, 0.1, -0.1]:
                new_val = old + delta
                if new_val < 0.0 or new_val > 1.0:
                    continue
                trial = h.copy()
                trial[j] = new_val
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_local - 1e-12:
                    best_local = c
                    best_local_h = trial
            if best_local_h is not None:
                h = best_local_h
                best_c5 = best_local
                improved = True
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_h = None
    best_c5 = float('inf')

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is not None:
        h0 = make_feasible(h0, len(h0))
        best_c5 = compute_c5(h0)
        best_h = h0.copy()
    else:
        best_h = make_feasible(0.5 + 0.1 * rng.randn(320), 320)
        best_c5 = compute_c5(best_h)

    resolutions = [320, 400, 480, 640]
    mus = [500.0, 5000.0, 50000.0]

    for n_pts in resolutions:
        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 30
        if remaining < 60:
            break

        h_work = best_h.copy() if len(best_h) == n_pts else upsample_h(best_h, n_pts)
        h_work = make_feasible(h_work, n_pts)

        n_restarts = 4 if n_pts <= 400 else 3
        for r in range(n_restarts):
            elapsed = time.time() - start_time
            remaining = budget_s - elapsed - 30
            if remaining < 40:
                break

            if r == 0:
                h_start = h_work.copy()
            elif r == 1:

                h_start = make_feasible(h_start, n_pts)
            elif r == 2:
                h_start = (h_work + rng.randn(n_pts) * 0.08).copy()
                h_start = make_feasible(h_start, n_pts)
            else:
                h_rand = rng.rand(n_pts)
                h_rand = make_feasible(h_rand, n_pts)
                h_start = 0.5 * h_work + 0.5 * h_rand
                h_start = make_feasible(h_start, n_pts)

                for mu in mus:

                    remaining = budget_s - elapsed - 20
                    if remaining < 20:
                        break

                    def fun_g(h_flat):
                        return lse_wrapped(h_flat, n_pts, t_val, mu)

                    bounds = [(0.0, 1.0)] * n_pts
                    try:
                        res = minimize(fun_g, h_start, method='L-BFGS-B',
                                       bounds=bounds,
                                       jac=True,
                                       options={'maxiter': 200, 'ftol': 1e-12, 'gtol': 1e-8,
                                                'maxls': 50})
                        h_res = make_feasible(res.x, n_pts)
                        c_res = compute_c5(h_res)
                        if c_res < best_c5:
                            best_c5 = c_res
                            best_h = h_res.copy()
                            h_work = h_res.copy()
                    except Exception:
                        pass

        # Coordinate descent polish at this resolution
        elapsed = time.time() - start_time
        if budget_s - elapsed > 60:
            h_cd, c_cd = coordinate_descent(best_h.copy(), n_pts, max_iter=50,
                                            time_limit=budget_s - 30, start_time=start_time)
            if c_cd < best_c5:
                best_c5 = c_cd
                best_h = h_cd.copy()

    # Final high-resolution polish
    elapsed = time.time() - start_time
    if budget_s - elapsed > 80:
        n_final = max(len(best_h), 480)
        h_final = upsample_h(best_h, n_final)
        # Fine coordinate descent
        h_final, c_final = coordinate_descent(h_final, n_final, max_iter=80,
                                              time_limit=budget_s - 20, start_time=start_time)
        if c_final < best_c5:
            best_c5 = c_final
            best_h = h_final.copy()

    # Very fine CD on best
    elapsed = time.time() - start_time
    if budget_s - elapsed > 40:
        h_final2, c_final2 = coordinate_descent(best_h.copy(), len(best_h), max_iter=100,
                                                time_limit=budget_s - 10, start_time=start_time)
        if c_final2 < best_c5:
            best_c5 = c_final2
            best_h = h_final2.copy()

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
