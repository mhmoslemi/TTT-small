import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def symmetrize(h):
    n = len(h)
    idx = (n - np.arange(n)) % n
    return 0.5 * (h + h[idx])


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full').max() * dx


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        j_lo = max(0, -sh)
        j_hi = min(n, n - sh)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += pk * (1.0 - h[j_lo + sh:j_hi + sh])
        j_lo2 = max(0, sh)
        j_hi2 = min(n, n + sh)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= pk * h[j_lo2 - sh:j_hi2 - sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_adam(h, tau, time_limit, lr=0.05):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.8, 0.999, 1e-8
    best_obj, _ = smooth_obj_grad(h, tau)
    best_h = h.copy()
    t = 0
    start = time.time()
    while time.time() - start < time_limit:
        t += 1
        obj, grad = smooth_obj_grad(h, tau)
        if obj < best_obj - 1e-15:
            best_obj = obj
            best_h = h.copy()
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * grad * grad
        m_hat = m / (1.0 - b1 ** t)
        v_hat = v / (1.0 - b2 ** t)
        h_new = symmetrize(h - lr * m_hat / (np.sqrt(v_hat) + eps))
        h = project_to_feasible(h_new, n)
        if t % 1000 == 0:
            lr *= 0.92
    return best_h


def optimize_subgrad(h, time_limit, lr=0.003):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j_lo = max(0, -sh)
        j_hi = min(n, n - sh)
        if j_lo < j_hi:
            grad[j_lo:j_hi] += (1.0 - h[j_lo + sh:j_hi + sh])
        j_lo2 = max(0, sh)
        j_hi2 = min(n, n + sh)
        if j_lo2 < j_hi2:
            grad[j_lo2:j_hi2] -= h[j_lo2 - sh:j_hi2 - sh]
        grad *= dx
        h = project_to_feasible(h - lr * symmetrize(grad), n)
        if it % 3000 == 0:
            lr *= 0.85
    return best_h, best_c5


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)
