                diff -= adj
                if abs(diff) < 1e-13:

        else:

                diff -= adj

    return make_feasible(h_new, new_n)

def slsqp_pass(h_start, n, T, max_iter):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = T * (corr - mc)
        ms = np.max(shifted)
        log_sum = ms + np.log(np.sum(np.exp(shifted - ms)))
        return (mc + log_sum / T) * dx
    def obj_grad(h):
        # Numerical gradient (central differences)
        eps = 1e-7
        grad = np.zeros(n)
        for j in range(n):
            hp = h.copy(); hp[j] += eps
            hm = h.copy(); hm[j] -= eps
            grad[j] = (obj(hp) - obj(hm)) / (2*eps)
        return grad
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0}]
    bounds = [(0.0, 1.0)] * n

                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

    # Phase 7: Final coordinate descent
    if time_left() > 15:
        best_h, best_c5 = coord_descent(best_h, len(best_h), max_passes=3,
                                         time_limit_fn=time_left, rng=rng)
    
    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
