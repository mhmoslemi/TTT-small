                if abs(bv - h_cd[j]) < 1e-8:
                    continue
                trial = h_cd.copy()
                trial[j] = bv
                trial = make_feasible(trial, n_main)
                c = compute_c5(trial)
                if c < c5_cd - 1e-13:
                    c5_cd = c
                    h_cd = trial.copy()
                    improved = True
        if not improved:
            break

    if c5_cd < best_c5:
        best_c5 = c5_cd
        best_h = h_cd.copy()

    # Try higher resolution if time permits
    if _time_left() > 120:
        n_high = 4096
        h_high = upsample(best_h, n_high)
        c5_high = compute_c5(h_high)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_h = h_high.copy()
        for T in [200.0, 1000.0]:
            if _time_left() < 40:
                break
            try:
                h_opt = _slsqp_run(h_high, n_high, T, 100, 1e-13)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_high = h_opt.copy()
            except Exception:
                pass

    # Final SLSQP polish at high T
    if _time_left() > 25:
        n_f = len(best_h)
        try:
            h_opt = _slsqp_run(best_h, n_f, 10000.0, 60, 1e-15)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        except Exception:
            pass

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Validate with provided evaluator if available
    try:
        val, actual_c5 = evaluate_erdos_solution(best_h)
        if val and actual_c5 < best_c5:
            best_c5 = actual_c5
    except Exception:
        pass

    return best_h, best_c5, len(best_h)
