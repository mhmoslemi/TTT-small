    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)

        if j0b < j1b:

            grad[j0b:j1b] -= pk * h[idx + sh]

    obj = dx * (max_c + np.log(s) / tau)

So T2[i] = sum_j p[j]*h[i-j+n-1] (the I condition is automatically satisfied by zero-padding).

This is the convolution (p * h)[i+n-1] where (p*h)[m] = sum_j p[j]*h[m-j].
So T2[i] = conv(p, h)[i+n-1] = conv_ph[i+n-1]. ✓

T3[i] = sum_s q[s]*h[i+s]*I(0≤i+s<n)
= sum_j p[j]*h[i+(j-(n-1))]*I(0≤i+j-(n-1)<n)
= sum_j p[j]*h[i+j-n+1]*I(0≤i+j-n+1<n)

Again, the I condition is automatically satisfied by zero-padding of h.
T3[i] = sum_j p[j]*h[i+j-n+1]

This is the cross-correlation. Let's define: (p ⊛ h)[m] = sum_j p[j]*h[m+j] (cross-correlation)
T3[i] = (p ⊛ h)[i-n+1]

In FFT: cross-correlation can be computed as:
(p ⊛ h)[m] = sum_j p[j]*h[m+j] = conv(p_rev, h)[m + len(p)-1]
where p_rev[j] = p[len(p)-1-j]

Now len(p) = 2n-1, p_rev[j] = p[2n-2-j].
conv(p_rev, h)[m + 2n-2] = sum_j p_rev[j]*h[m+2n-2-j] = sum_j p[2n-2-j]*h[m+2n-2-j]
Let k = 2n-2-j: = sum_k p[k]*h[m+2n-2-(2n-2-k)] = sum_k p[k]*h[m+k]
So (p ⊛ h)[m] = conv(p_rev, h)[m + 2n-2]

T3[i] = (p ⊛ h)[i-n+1] = conv(p_rev, h)[i-n+1+2n-2] = conv(p_rev, h)[i+n-1]

So T3[i] = cross_ph[i+n-1] where cross_ph = IFFT(FFT(p_rev) * FFT(h)). ✓

Great, so the formula is:
grad[i] = (A1[i] - T2[i] - T3[i]) * dx

where:
- A1 = pref[n:2n] - pref[0:n]  (sliding sum of p)
- T2 = conv(p,h)[n-1:2n-1]  (from FFT)
- T3 = conv(p_rev,h)[n-1:2n-1]  (from FFT)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    h = np.clip(g - t, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def fft_smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    size = 2 * n
    size = 1 << (size - 1).bit_length()
    P = np.fft.rfft(p, size)
    H = np.fft.rfft(h, size)
    conv_ph = np.fft.irfft(P * H, size)
    p_rev = p[::-1]
    P_rev = np.fft.rfft(p_rev, size)
    cross_ph = np.fft.irfft(P_rev * H, size)
    pref = np.zeros(2 * n)
    pref[1:] = np.cumsum(p)
    A1 = pref[n:n + n] - pref[0:n]
    T2 = conv_ph[n - 1:2 * n - 1]
    T3 = cross_ph[n - 1:2 * n - 1]
    grad = (A1 - T2 - T3) * dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_fft(h, n, tau, time_limit, lr=None, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 0.3 / tau
    v = np.zeros(n)
    beta = momentum
    stall = 0
    c5_ema = best_c5

    for it in range(500000):
        if time.time() - start > time_limit:
            break
        obj, grad = fft_smooth_grad(h, tau)
        c5_now = compute_c5(h)
        c5_ema = 0.97 * c5_ema + 0.03 * c5_now
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 3000:
            lr *= 0.65
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if time.time() - start > time_limit:
            break

    return best_h, best_c5


def bottleneck_transform(h, sharpness):
    center = h - 0.5
    bt = 0.5 + 0.5 * np.tanh(sharpness * center)
    return project_to_feasible(bt, len(h))


def subgrad_focus(h, n, time_limit, lr=0.001, top_k=3):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.85
    stall = 0

    for it in range(300000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            lr *= 0.7
            v *= 0.4
            stall = 0
            if lr < 1e-15:
                break

        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            sh = k - (n - 1)
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b + sh:j1b + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5
