                # But since we only changed one coordinate, other lags change too
                # So we should just track: did we improve?
                improved = True

            break

    h = make_feasible(h, n)
    return h

But wait, in the coord_minimize function, I'm computing the slopes s[m] for each m, which is O(n) per coordinate. So the full sweep is O(n^2) for computing all s vectors, plus O(n) for each bisection step. Total per sweep: O(n^2 * log(1/eps)) ≈ O(n^2 * 50) for 50 bisection steps.

For n=320: 320^2 * 50 = 5.1M per sweep. With 20 sweeps: 100M. Fast.

But the recomputation of corr after each coordinate update is the bottleneck. Let me avoid that by using a different strategy:

Instead of recomputing corr after each coordinate change, I'll:
1. Compute corr once per sweep
2. For each coordinate, compute the optimal v using the current corr (which is slightly stale after previous changes in the same sweep)
3. Apply all changes
4. Recompute corr for next sweep

This is "Jacobi" style coordinate descent. It's less accurate per iteration but much faster.

Or I can do "Gauss-Seidel" style: update corr incrementally as I go.

Let me think about the incremental update. If I change h[j] from old to new, how does corr change?

For lag m:
corr_new[m] = corr[m] + (new - old) * s_m(j)

where s_m(j) is the slope I computed above for coordinate j at lag m.

So I can update corr in O(n) time (for all m) after each coordinate change. Total per sweep: O(n^2). Much better!

def polish_fast(h, max_sweeps=20, tol=1e-10):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    
    for sweep in range(max_sweeps):
        old_max = corr.max()
        improved = False
        
        for j in range(n):
            old = h[j]
            # Compute slopes for coordinate j
            s = np.zeros(2*n - 1)
            for m in range(2*n - 1):
                idx2 = j + m - n + 1
                if 0 <= idx2 < n:
                    s[m] += (1.0 - h[idx2])
                idx1 = j - m + n - 1
                if 0 <= idx1 < n:
                    s[m] += -h[idx1]
            
            # Find best v
            b = corr - old * s
            # Bisect for optimal v
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2
                vals = b + mid * s
                mx = vals.max()
                active = vals >= mx - 1e-12
                if s[active].min() > 0:
                    hi = mid
                elif s[active].max() < 0:
                    lo = mid
                else:
                    break
            best_v = (lo + hi) / 2
            best_max = np.max(b + best_v * s)
            
            # Check boundaries
            v0max = np.max(b)  # v=0
            v1max = np.max(b + s)  # v=1
            
            if v0max < best_max:
                best_v, best_max = 0.0, v0max
            if v1max < best_max:
                best_v, best_max = 1.0, v1max
            
            if best_max < old_max - tol or best_v != old:
                if best_max <= old_max + tol:  # Accept if not worse
                    h[j] = best_v
                    # Update corr incrementally
                    delta = best_v - old
                    if abs(delta) > 1e-15:
                        corr += delta * s
                    old_max = corr.max()
                    improved = True
        
        if not improved:
            break
    
    h = make_feasible(h, n)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_slopes(h, j):
    """Compute slope vector s[m] for coordinate j."""
    n = len(h)
    s = np.zeros(2 * n - 1)
    # Direct: m in [n-1-j, 2n-2-j]
    m_start = n - 1 - j
    m_end = 2 * n - 2 - j
    if m_start <= m_end:
        length = m_end - m_start + 1
        s[m_start:m_start + length] += (1.0 - h[:length])
    # Indirect: m in [j, j+n-1]
    m_start2 = j
    m_end2 = j + n - 1
    if m_start2 <= m_end2:
        length2 = m_end2 - m_start2 + 1
        s[m_start2:m_start2 + length2] -= h[length2 - 1::-1]
    return s

def find_best_v(b, s, lo=0.0, hi=1.0):
    """Find v in [lo,hi] minimizing max(b + v*s)."""
    # Bisection on subgradient
    for _ in range(60):
        mid = (lo + hi) / 2.0
        vals = b + mid * s
        mx = vals.max()
        active = vals >= mx - 1e-13
        s_act = s[active]
        if len(s_act) == 0:
            break
        if s_act.min() > 1e-14:
            hi = mid
        elif s_act.max() < -1e-14:
            lo = mid
        else:
            break
    v = (lo + hi) / 2.0
    return v, np.max(b + v * s)

def fast_polish(h, max_sweeps=30, time_limit=None, start_time=None):
    """Coordinate descent with exact 1D minimization."""
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = corr.max()
    
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit:
                    break
            
            old = h[j]
            s = coord_slopes(h, j)
            b = corr - old * s
            
            best_v, best_max = find_best_v(b, s)
            # Check boundaries
            v0_max = np.max(b)
            v1_max = np.max(b + s)
            
            if v0_max < best_max:
                best_v, best_max = 0.0, v0_max
            if v1_max < best_max:
                best_v, best_max = 1.0, v1_max
            
            if best_max < current_max - 1e-12:
                h[j] = best_v
                delta = best_v - old
                if abs(delta) > 1e-15:
                    corr += delta * s
                current_max = corr.max()
                improved = True
            elif best_v != old and best_max <= current_max + 1e-12:
                # Accept neutral moves to escape plateaus
                h[j] = best_v
                delta = best_v - old
                if abs(delta) > 1e-15:
                    corr += delta * s
                improved = True
        
        if not improved:
            break
    
    h = make_feasible(h, n)
    return h

def slsqp_optimize(h, n, T=50.0, maxiter=300):
    """SLSQP with log-sum-exp surrogate."""
    dx = 2.0 / n
    
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        soft = mc + np.log(np.sum(np.exp(T * shifted))) / T
        return soft * dx
    
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    
    def constraint_jac(x):
        return np.ones(n)
    
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    
    res = minimize(objective, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

def run(seed=42, budget_s=1300, **kwargs):

    rng = np.random.RandomState(seed)
    
    # Get initial solution
    best_h = None
    try:

    except Exception:

        best_h = (rng.rand(n) < 0.5).astype(float)
        # Fix to have exactly n/2 ones
        n_ones = int(np.sum(best_h))
        target = n // 2

            to_flip = rng.choice(ones, n_ones - target, replace=False)
            best_h[to_flip] = 0
        elif n_ones < target:
            zeros = np.where(best_h == 0)[0]
            to_flip = rng.choice(zeros, target - n_ones, replace=False)
            best_h[to_flip] = 1
    
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP at original resolution
    n0 = len(best_h)
    for T in [30, 80, 200]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 60:
            break
        try:
            h_opt = slsqp_optimize(best_h, n0, T=T, maxiter=200)
            h_opt = make_feasible(h_opt, n0)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        except Exception:
            pass
    
    # Phase 2: Fast coordinate descent polish
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed - 30
    if remaining > 5:
        h_pol = fast_polish(best_h, max_sweeps=50, time_limit=remaining, start_time=start_time)
        c5 = compute_c5(h_pol)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_pol

    for new_n in [480, 640, 960]:
        elapsed = time.time() - start_time
        if budget_s - elapsed < 120:
            break
        
        h_up = upsample_h(best_h, new_n)
        c5_up = compute_c5(h_up)

        remaining = budget_s - (time.time() - start_time) - 60
        if remaining < 30:
            break

            h_opt = make_feasible(h_opt, new_n)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
        except Exception:
            pass

        elapsed = time.time() - start_time
        remaining = budget_s - elapsed - 20
        if remaining > 10 and best_h is not None and len(best_h) == new_n:
            h_pol = fast_polish(best_h, max_sweeps=30, time_limit=min(remaining, 120), start_time=start_time)
            c5 = compute_c5(h_pol)
            if c5 < best_c5:
                best_c5 = c5

    if final_c5 < best_c5:
        best_c5 = final_c5

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_slopes(h, j):
    n = len(h)
    s = np.zeros(2 * n - 1)
    m_start = n - 1 - j
    m_end = 2 * n - 2 - j
    if m_start <= m_end:
        length = m_end - m_start + 1
        s[m_start:m_start + length] += (1.0 - h[:length])
    m_start2 = j
    m_end2 = j + n - 1
    if m_start2 <= m_end2:
        length2 = m_end2 - m_start2 + 1
        s[m_start2:m_start2 + length2] -= h[length2 - 1::-1]
    return s

def find_best_v(b, s):
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        vals = b + mid * s
        mx = vals.max()
        active = vals >= mx - 1e-12
        s_act = s[active]
        if s_act.min() > 1e-12:
            hi = mid
        elif s_act.max() < -1e-12:
            lo = mid
        else:
            break
    v = (lo + hi) / 2.0
    return v, np.max(b + v * s)

def fast_polish(h, max_sweeps=40, time_limit=None, start_time=None):
    n = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full')
    current_max = corr.max()
    for sweep in range(max_sweeps):
        if time_limit is not None and start_time is not None:
            if time.time() - start_time > time_limit:
                break
        improved = False
        for j in range(n):
            if time_limit is not None and start_time is not None:
                if time.time() - start_time > time_limit:
                    break
            old = h[j]
            s = coord_slopes(h, j)
            if np.max(np.abs(s)) < 1e-15:
                continue
            b = corr - old * s
            best_v, best_max = find_best_v(b, s)
            v0_max = np.max(b)
            v1_max = np.max(b + s)
            if v0_max < best_max:
                best_v, best_max = 0.0, v0_max
            if v1_max < best_max:
                best_v, best_max = 1.0, v1_max
            if best_max < current_max - 1e-13:
                h[j] = best_v
                delta = best_v - old
                if abs(delta) > 1e-15:
                    corr += delta * s
                current_max = corr.max()
                improved = True
        if not improved:
            break
    h = make_feasible(h, n)
    return h

def slsqp_optimize(h, n, T=50.0, maxiter=200):
    dx = 2.0 / n
    def objective(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        soft = mc + np.log(np.sum(np.exp(T * shifted))) / T
        return soft * dx
    def constraint_fun(x):
        return np.sum(x) - n / 2.0
    def constraint_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': constraint_fun, 'jac': constraint_jac}]
    res = minimize(objective, h, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
    return res.x

def random_binary_h(n, rng):
    h = (rng.rand(n) < 0.5).astype(float)
    n_ones = n // 2
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > n_ones:
        to_flip = rng.choice(ones, len(ones) - n_ones, replace=False)
        h[to_flip] = 0.0
    elif len(ones) < n_ones:
        to_flip = rng.choice(zeros, n_ones - len(ones), replace=False)
        h[to_flip] = 1.0
    return h
