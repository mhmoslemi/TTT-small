    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return _feasible(h_new, new_n)

    t = _G_T
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    shifted = corr - mc

    shifted = np.clip(shifted, -10.0/t, 0.0)
    soft = mc + np.log(np.sum(np.exp(t * shifted))) / t
    return soft * dx

    return np.sum(h) - _G_N / 2.0

    return np.ones(_G_N)

def _greedy_swap(h, n, rng, max_iters, time_limit):

            h[to_flip] = 1.0
    
    c5 = _c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    dx = 2.0 / n
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        overlaps = corr * dx
        max_lag = np.argmax(overlaps) + (1 - n)  # convert to actual lag
        
        # Find contributing pairs: h[i]=1, h[i+|max_lag|]=0 (or similar)
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try random swap biased toward reducing the max
        # Pick i from ones, j from zeros
        best_delta = 0
        best_pair = None
        
        # Try a small number of candidate swaps
        n_tries = 20
        for _ in range(n_tries):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            corr_t = np.correlate(h, 1.0 - h, mode='full')
            new_c5 = np.max(corr_t) * dx
            h[i] = 1.0
            h[j] = 0.0
            delta = new_c5 - c5
            if delta < best_delta:
                best_delta = delta
                best_pair = (i, j)
        
        if best_pair is not None and best_delta < -1e-12:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            c5 += best_delta
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
        else:
            # No improvement, do a random swap for diversification
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            corr_t = np.correlate(h, 1.0 - h, mode='full')
            c5 = np.max(corr_t) * dx

    if fc5 < best_c5:
        best_c5 = fc5

    return best_h, best_c5, len(best_h)
