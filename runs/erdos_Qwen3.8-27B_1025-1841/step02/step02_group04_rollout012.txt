from scipy.optimize import minimize, minimize_scalar
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def binary_local_search(h, n_iter=5000, rng=None):
    """Local search for binary h: flip pairs to reduce max corr."""
    n = len(h)
    h = np.where(h > 0.5, 1.0, 0.0)
    h = make_feasible(h, n)
    corr = corr_full(h)
    dx = 2.0 / n
    best_c5 = np.max(corr) * dx
    
    ones = set(np.where(h == 1)[0])
    zeros = set(np.where(h == 0)[0])
    
    for it in range(n_iter):
        changed = False
        # Try random flips
        for _ in range(20):
            if rng.randint(10) < 5 and len(ones) > 0 and len(zeros) > 0:
                # Swap: turn one 1 to 0 and one 0 to 1
                i = rng.choice(list(ones))
                j = rng.choice(list(zeros))
                if i == j:
                    continue
                h[i] = 0.0
                h[j] = 1.0
                new_corr = corr_full(h)
                new_c5 = np.max(new_corr) * dx
                if new_c5 < best_c5:
                    corr = new_corr
                    best_c5 = new_c5
                    ones.discard(i)
                    zeros.discard(i)
                    zeros.add(i)
                    ones.discard(j)
                    zeros.discard(j)
                    ones.add(j)
                    changed = True
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        if not changed and it % 1000 == 0:
            # Try perturbations
            for _ in range(5):
                flip_count = rng.randint(3, 8)
                flips_on = rng.choice(list(ones), min(flip_count, len(ones)), replace=False)
                flips_off = rng.choice(list(zeros), min(flip_count, len(zeros)), replace=False)
                for i in flips_on:
                    h[i] = 0.0
                for j in flips_off:
                    h[j] = 1.0
                new_corr = corr_full(h)
                new_c5 = np.max(new_corr) * dx

                    corr = new_corr
                    best_c5 = new_c5
                    for i in flips_on:

                    for j in flips_off:

                else:
                    for i in flips_on:
                        h[i] = 1.0
                    for j in flips_off:
                        h[j] = 0.0
    return h, best_c5

def update_corr(corr, h_before, h_after, i0, n):
    """Update correlation array when h[i0] changes."""
    delta = h_after[i0] - h_before[i0]
    if abs(delta) < 1e-15:
        return corr
    h = h_before.copy()
    h[i0] = h_after[i0]  # for computing (1-h[...])
    
    for k in range(-n+1, n):
        m = n - 1 + k
        change = 0.0
        # First factor: h[i0] * (1-h[i0+k])
        if 0 <= i0 + k < n:
            change += delta * (1.0 - h[i0 + k])
        # Second factor: h[i0-k] * (1-h[i0]) -> change is h[i0-k] * (-delta)
        if 0 <= i0 - k < n:
            change += -delta * h[i0-k]
        corr[m] += change
    return corr

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def _obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Soft-max log-sum-exp
    max_shifted = np.max(shifted)
    val = mc + max_shifted + np.log(np.sum(np.exp(shifted - max_shifted))) / T
    return val * dx

def _grad(h, n, T):
    """Soft-max gradient of the objective."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    max_s = np.max(shifted)
    exp_s = np.exp(shifted - max_s)
    weights = exp_s / np.sum(exp_s)
    
    n_corr = len(corr)
    grad = np.zeros(n)
    # For each correlation index m, the gradient w.r.t. h is:
    # corr[m] = sum_i h[i]*(1-h[i+m-(n-1)]) for valid i
    # d(corr[m])/dh[i] = (1-h[i+m-(n-1)]) if valid - h[i+m-(n-1)] * 0 ... 
    # Actually: d/dh[i] of sum_j h[j]*(1-h[j+k]) where k=m-(n-1):
    # = (1-h[i+k]) if i+k valid [from h[i] in first factor]
    #   + h[i-k]*(-1) if i-k valid [from -h[i-k]*h[i] in second factor]
    # Wait: sum_j h[j]*(1-h[j+k]) = sum_j h[j] - sum_j h[j]*h[j+k]
    # d/dh[i] = 1 - h[i+k] (if i+k valid) - h[i-k] (if i-k valid, from -h[i-k]*h[i])
    #          = 1 - h[i+k] - h[i-k]
    # But also the (1-h[i]) part: when h[i] is the second factor:
    # sum_j h[j]*(1-h[j+k]): h[i] appears as (1-h[i]) when j+k=i, j=i-k
    # So d/dh[i] of that term = h[i-k]*(-1)
    # And h[i] appears as h[i] when j=i: d/dh[i] = (1-h[i+k])
    # Total: (1-h[i+k]) - h[i-k] (for valid indices)
    
    # This is O(n * n_corr) = O(n²) which is fine for n=320
    for m in range(n_corr):
        w = weights[m]
        if w < 1e-8:
            continue
        k = m - (n - 1)
        for i in range(n):
            g = 0.0
            if 0 <= i + k < n:
                g += (1.0 - h[i + k])
            if 0 <= i - k < n:
                g += -h[i - k]
            grad[i] += w * g
    return grad * dx

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        sh = corr - mc
        ms = np.max(sh)
        return (mc + ms + np.log(np.sum(np.exp(sh - ms))) / T) * dx

        mc = np.max(corr)
        sh = corr - mc
        ms = np.max(sh)
        es = np.exp(sh - ms)
        total = np.sum(es)
        i = np.arange(n)

            idx = np.argmax(corr)
            k = idx - (n - 1)
            vf = (i + k >= 0) & (i + k < n)
            grad[vf] += (1.0 - x[i[vf] + k])
            vs = (i - k >= 0) & (i - k < n)

            sig = np.where(w > 1e-7)[0]

                vs = (i - k >= 0) & (i - k < n)
                grad[vs] -= wm * x[i[vs] - k]

    def cons_fun(x):
        return np.sum(x) - n / 2.0
    
    def cons_jac(x):
        return np.ones(n)

import numpy as np
from scipy.optimize import minimize
import time

_OPT = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h += diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0 - 1e-9)[0]
        else:
            idx = np.where(h > 1e-9)[0]
        if len(idx) > 0:
            h[idx] += diff / len(idx)
            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_obj_val(h, T):
    n = _OPT['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = corr - mc
    ms = np.max(sh)
    return (mc + ms + np.log(np.sum(np.exp(sh - ms))) / T) * dx

def smooth_grad(h, T):
    n = _OPT['n']
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    sh = corr - mc
    ms = np.max(sh)
    es = np.exp(sh - ms)
    total = np.sum(es)
    i = np.arange(n)
    grad = np.zeros(n)
    if total < 1e-300:
        idx = np.argmax(corr)
        k = idx - (n - 1)
        vf = (i + k >= 0) & (i + k < n)
        grad[vf] += (1.0 - h[i[vf] + k])
        vs = (i - k >= 0) & (i - k < n)
        grad[vs] -= h[i[vs] - k]
    else:
        w = es / total
        sig = np.where(w > 1e-7)[0]
        for m in sig:
            wm = w[m]
            k = m - (n - 1)
            vf = (i + k >= 0) & (i + k < n)
            grad[vf] += wm * (1.0 - h[i[vf] + k])
            vs = (i - k >= 0) & (i - k < n)
            grad[vs] -= wm * h[i[vs] - k]
    return grad * dx

def _slsqp_grad(x):
    return smooth_grad(x, _OPT['T'])

def _cons_fun(x):
    return np.sum(x) - _OPT['n'] / 2.0

    return np.ones(_OPT['n'])

def gd_optimize(h, n_iter, T_start, T_end, lr, n, check_every=50):
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        _OPT['n'] = n
        g = smooth_grad(h, T)
        h = h - lr_t * g
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:

    c5 = compute_c5(h)
