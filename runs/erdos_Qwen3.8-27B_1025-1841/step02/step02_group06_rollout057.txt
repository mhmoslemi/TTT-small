import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    # np.correlate(h, 1-h, 'full') = convolve(h[::-1], 1-h, 'full')
    H = np.fft.rfft(h[::-1], N)
    Gh = np.fft.rfft(1.0 - h, N)
    corr = np.fft.irfft(H * Gh, N)
    return corr[:2*n-1]

def compute_c5(h):
    n = len(h)
    return corr_fft(h).max() * (2.0 / n)

def topk_gradient(h, k_indices, n, dx, K):
    """Compute gradient for top-K shifts."""
    grad = np.zeros(n)
    for ki in k_indices:
        d = ki - (n - 1)
        # ∂corr[ki]/∂h[i] = (1-h[i+d])*1_{i+d∈[0,n)} - h[i-d]*1_{i-d∈[0,n)}
        # First term: (1-h[i+d]) for i=0,...,n-d-1 (if d>=0) or i=-d,...,n-1 (if d<0)
        if d >= 0:
            # i+d ranges from d to n-1, so i from 0 to n-1-d
            # (1-h[i+d]) contributes to grad[i] for i=0,...,n-1-d
            grad[:n-d] += (1.0 - h[d:])
            # Second term: h[i-d] for i=d,...,n-1
            # -h[i-d] contributes to grad[i] for i=d,...,n-1
            grad[d:] -= h[:n-d]
        else:
            dd = -d  # positive
            # i+d = i-dd, ranges from 0 to n-1-dd, so i from dd to n-1
            # (1-h[i-dd]) contributes to grad[i] for i=dd,...,n-1
            grad[dd:] += (1.0 - h[:n-dd])
            # i-d = i+dd, ranges from dd to n-1+dd... wait
            # i-d = i+dd, must be in [0,n), so i from 0 to n-1-dd
            # -h[i+dd] contributes to grad[i] for i=0,...,n-1-dd
            grad[:n-dd] -= h[dd:]
    return grad * dx / K

def optimize_topk(h, n, topK, max_iters, time_limit, lr_init=0.1):
    dx = 2.0 / n
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = 200

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = corr_fft(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Top-K indices
        k_indices = np.argpartition(corr, -topK)[-topK:]
        grad = topk_gradient(h, k_indices, n, dx, topK)
        
        # Momentum
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.8
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new

    return best_h, best_c5
