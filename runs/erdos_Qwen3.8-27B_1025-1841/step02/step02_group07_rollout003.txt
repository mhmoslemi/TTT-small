import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def softmax_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    W = np.sum(exp_scaled)
    w = exp_scaled / W  # softmax weights
    
    # Gradient of softmax objective
    # dc_j/dh_i = (1-h[i+j-(n-1)]) - h[i-(j-(n-1))] (out of range = 0)
    grad = np.zeros(n)
    k_vals = np.arange(2*n-1) - (n-1)  # from -(n-1) to n-1
    for j in range(2*n-1):
        k = k_vals[j]
        wj = w[j]
        if wj < 1e-15:
            continue
        # term1: (1-h[i+k]) for valid i
        # i+k in [0, n-1] => i in [-k, n-1-k]
        i1_start = max(0, -k)
        i1_end = min(n, n-k)  # exclusive
        if i1_start < i1_end:
            grad[i1_start:i1_end] += wj * (1.0 - h[i1_start+k:i1_end+k])
        # term2: -h[i-k] for valid i
        # i-k in [0, n-1] => i in [k, n-1+k]
        i2_start = max(0, k)
        i2_end = min(n, n+k)  # exclusive
        if i2_start < i2_end:
            grad[i2_start:i2_end] -= wj * h[i2_start-k:i2_end-k]
    
    obj = (m + np.log(W)) * dx / T
    grad *= dx / T
    return obj, grad

def project_sum(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-12:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def penalty_obj_grad(x, n, T, mu):
    h = x  # already bounded by L-BFGS-B
    obj0, grad0 = softmax_obj_and_grad(h, T)
    s = np.sum(h) - n / 2.0
    obj = obj0 + mu * s * s
    grad = grad0 + 2.0 * mu * s
    return obj, grad

def lbfgs_opt(h0, n, T, mu=500, maxiter=200, deadline=None):
    def fun(x):
        o, _ = penalty_obj_grad(x, n, T, mu)
        return o
    def jac(x):
        _, g = penalty_obj_grad(x, n, T, mu)
        return g
    try:
        res = minimize(fun, h0, jac=jac, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10,
                                'maxfun': 5000})
        if res.x is not None and np.all(np.isfinite(res.x)):

    n = len(h)

def softmax_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * dx * corr
    m = np.max(scaled)
    exp_s = np.exp(scaled - m)
    W = np.sum(exp_s)
    w = exp_s / W
    grad = np.zeros(n)
    k_vals = np.arange(2 * n - 1) - (n - 1)
    for j in range(2 * n - 1):
        wj = w[j]
        if wj < 1e-14:
            continue
        k = k_vals[j]
        i1s = max(0, -k)
        i1e = min(n, n - k)
        if i1s < i1e:
            grad[i1s:i1e] += wj * (1.0 - h[i1s + k:i1e + k])
        i2s = max(0, k)
        i2e = min(n, n + k)
        if i2s < i2e:
            grad[i2s:i2e] -= wj * h[i2s - k:i2e - k]
    obj = (m + np.log(W)) * dx / T

    return obj, grad
