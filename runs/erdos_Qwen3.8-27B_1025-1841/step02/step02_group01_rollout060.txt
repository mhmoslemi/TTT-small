from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def project_feasible(h):
    """Project h onto {0≤h≤1, sum(h)=len(h)/2}"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        # Need to increase sum. Add to elements < 1
        mask = h < 1.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    else:
        # Need to decrease sum. Subtract from elements > 0
        mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def grad_c5_active(h, active_shifts):
    """Gradient of sum_{s in active} corr[n-1+s]^2"""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    grad = np.zeros(n)
    for s in active_shifts:
        c = corr[n - 1 + s]
        if abs(c) < 1e-12:
            continue
        for j in range(n):
            val = 0.0
            jm = j - s
            if 0 <= jm < n:
                val += (1.0 - h[jm])
            jp = j + s
            if 0 <= jp < n:
                val -= h[jp]
            grad[j] += 2.0 * c * val
    return grad

    best_c5 = compute_c5(h)
    best_h = h.copy()

def compute_c5(h):

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
    h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

        else:

            jm = j - s

                val += (1.0 - h[jm])
            jp = j + s
            if 0 <= jp < n:
                val -= h[jp]
            grad[j] += 2.0 * c * val

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)

        else:

        cnt = int(np.sum(mask))
        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

            jp = j + s
            if 0 <= jp < n:
                val -= h[jp]
