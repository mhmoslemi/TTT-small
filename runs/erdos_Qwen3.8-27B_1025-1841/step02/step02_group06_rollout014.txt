```python
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Softmax gradient
        max_g = g.max()
        shifted = np.clip(tau * (g - max_g), -500, 500)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-15:
                continue
            d = k - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += pk * (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= pk * h[idx + d]
        grad *= dx
        
        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        h = h_new
        corr = compute_corr(h)
        g = corr * dx

    return best_h, best_obj

                    h[j] = 1

                    break

                        best_pair = (i, j)

                    no_improve = 0
```

```python
def compute_swap_delta(h, i, j, n):
    """Compute delta correlation for swapping h[i]:1→0 and h[j]:0→1"""
    # Full d range: -(n-1) to n-1, corresponding to k = 0 to 2n-2
    d_all = np.arange(-(n-1), n)  # length 2n-1
    
    delta = np.zeros(2*n-1)
    
    # Term 1: -(1-h[i-d]) for valid d (0<=i-d<n)
    d1_range = (i-(n-1), i)  # d values where i-d is in [0, n-1]
    d1 = np.arange(max(0, i-(n-1)), min(n-1, i) + 1)
    idx1 = d1 + (n-1)  # k index
    valid1 = (i - d1 >= 0) & (i - d1 < n)
    delta[idx1[valid1]] -= (1.0 - h[i - d1[valid1]])
    
    # Term 2: +(1-h[j-d]) for valid d (0<=j-d<n)
    d2 = np.arange(max(0, j-(n-1)), min(n-1, j) + 1)
    idx2 = d2 + (n-1)
    valid2 = (j - d2 >= 0) & (j - d2 < n)
    delta[idx2[valid2]] += (1.0 - h[j - d2[valid2]])
    
    # Term 3: +h[i+d] for valid d (0<=i+d<n)
    d3 = np.arange(max(-(n-1), -i), min(n-1, n-1-i) + 1)
    idx3 = d3 + (n-1)
    valid3 = (i + d3 >= 0) & (i + d3 < n)
    delta[idx3[valid3]] += h[i + d3[valid3]]
    
    # Term 4: -h[j+d] for valid d (0<=j+d<n)
    d4 = np.arange(max(-(n-1), -j), min(n-1, n-1-j) + 1)
    idx4 = d4 + (n-1)
    valid4 = (j + d4 >= 0) & (j + d4 < n)
    delta[idx4[valid4]] -= h[j + d4[valid4]]
    
    return delta
```

```python
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full').astype(np.float64)

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def to_binary(h, n):
    h = np.array(h, dtype=np.float64)
    h_bin = (h >= 0.5).astype(np.float64)
    n_ones = int(h_bin.sum())
    target_ones = n // 2
    if n_ones > target_ones:
        ones_idx = np.where(h_bin == 1.0)[0]
        order = np.argsort(h[ones_idx])
        for i in range(n_ones - target_ones):
            h_bin[ones_idx[order[i]]] = 0.0
    elif n_ones < target_ones:

        order = np.argsort(-h[zeros_idx])
        for i in range(target_ones - n_ones):
            h_bin[zeros_idx[order[i]]] = 1.0
    return h_bin

    delta = np.zeros(2 * n - 1)
    # Term 1: -(1-h[i-d]) for d where 0<=i-d<n
    dl = max(-(n-1), i-(n-1))
    dh = min(n-1, i)
    if dl <= dh:
        d = np.arange(dl, dh+1)
        delta[d + n - 1] -= (1.0 - h[i - d])
    # Term 2: +(1-h[j-d]) for d where 0<=j-d<n
    dl = max(-(n-1), j-(n-1))
    dh = min(n-1, j)
    if dl <= dh:
        d = np.arange(dl, dh+1)
        delta[d + n - 1] += (1.0 - h[j - d])
    # Term 3: +h[i+d] for d where 0<=i+d<n

    dh = min(n-1, n-1-i)
    if dl <= dh:

        delta[d + n - 1] += h[i + d]
    # Term 4: -h[j+d] for d where 0<=j+d<n

    dh = min(n-1, n-1-j)
    if dl <= dh:

        delta[d + n - 1] -= h[j + d]

def binary_local_search(h_bin, n, time_limit, topK=10, max_iters=3000000):
    dx = 2.0 / n
    h = h_bin.copy()
    corr = compute_corr(h)
    best_max = corr.max()
    best_h = h.copy()
    start = time.time()
    stall = 0
    stall_limit = 3000
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if stall > stall_limit:
            break
        
        worst_idx = np.argmax(corr)
        d_star = worst_idx - n + 1
        
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        i_check = ones - d_star
        valid_i = (i_check >= 0) & (i_check < n)
        i_score = np.zeros(len(ones))
        i_score[valid_i] = 1.0 - h[i_check[valid_i]]
        
        j_check = zeros + d_star
        valid_j = (j_check >= 0) & (j_check < n)
        j_score = np.zeros(len(zeros))
        j_score[valid_j] = h[j_check[valid_j]]
        
        top_i_idx = np.argpartition(i_score, -topK)[-topK:] if len(i_score) > topK else np.arange(len(ones))
        top_j_idx = np.argpartition(j_score, -topK)[-topK:] if len(j_score) > topK else np.arange(len(zeros))
        
        best_new_max = best_max
        best_pair = None
        best_delta = None
        
        for ti in top_i_idx:
            i = ones[ti]
            for tj in top_j_idx:
                j = zeros[tj]
                delta = swap_delta(h, i, j, n)
                new_max = (corr + delta).max()
                if new_max < best_new_max:
                    best_new_max = new_max
                    best_pair = (i, j)
                    best_delta = delta
        
        if best_pair is not None and best_new_max < best_max - 1e-12:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            corr = corr + best_delta
            best_max = best_new_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            # Random perturbation when stuck
            if stall % 500 == 0 and stall < stall_limit:
                ri = np.random.randint(len(ones))
                rj = np.random.randint(len(zeros))
                i_rand, j_rand = ones[ri], zeros[rj]
                delta_rand = swap_delta(h, i_rand, j_rand, n)
                new_max_rand = (corr + delta_rand).max()
                if new_max_rand < best_max * 1.02:  # accept if not too much worse
                    h[i_rand] = 0.0
                    h[j_rand] = 1.0
                    corr = corr + delta_rand
    
    return best_h, best_max * dx

def continuous_optimize(h, n, tau, max_iters, time_limit, lr_init=0.15):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr(h)
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    patience = max(50, max_iters // 20)

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        cur_max = g.max()
        if cur_max < best_obj - 1e-15:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        max_g = g.max()
        shifted = np.clip(tau * (g - max_g), -500, 500)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-15:
                continue
            d = k - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += pk * (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= pk * h[idx + d]
        grad *= dx
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        
        if stall > patience:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-12:
                break
        
        corr = compute_corr(h)
        g = corr * dx

    return best_h, best_obj
```
