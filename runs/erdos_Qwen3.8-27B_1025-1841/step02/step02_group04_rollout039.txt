   import numpy as np
   from scipy.optimize import minimize
   import time

   def compute_c5(h):
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full')
       return float(np.max(corr) * dx)

   def make_feasible(h, n):
       h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
       target = n / 2.0
       diff = target - np.sum(h)
       if abs(diff) > 1e-10:
           if diff > 0:
               mask = h < 1.0
           else:
               mask = h > 0.0
           cnt = mask.sum()
           if cnt > 0:
               h[mask] += diff / cnt
               h = np.clip(h, 0.0, 1.0)
       return h

   def upsample(h, new_n):
       old_n = len(h)
       if new_n == old_n:
           return h.copy()
       old_x = np.linspace(0.0, 2.0, old_n)
       new_x = np.linspace(0.0, 2.0, new_n)
       h_new = np.interp(new_x, old_x, h)
       return make_feasible(h_new, new_n)

   def softmax_grad(h, T, n_active=50):
       """Compute gradient of softmax-approximated max of correlations."""
       n = len(h)
       dx = 2.0 / n
       corr = np.correlate(h, 1.0 - h, mode='full')
       mc = np.max(corr)
       shifted = corr - mc
       exp_s = np.exp(T * shifted)
       total = exp_s.sum()
       if total < 1e-300:
           # Fallback to hard max
           k_idx = np.argmax(corr)
           shift = k_idx - (n - 1)
           grad = np.zeros(n)
           s = abs(shift)
           if s < n:
               if shift >= 0:
                   grad[s:] += (1.0 - h[:n-s])
                   grad[:n-s] += (-h[s:])
               else:
                   grad[:n-s] += (1.0 - h[s:])
                   grad[s:] += (-h[:n-s])
           return grad * dx
       weights = exp_s / total
       # Only use top n_active weights
       top_idx = np.argpartition(weights, -n_active)[-n_active:]
       top_idx = top_idx[weights[top_idx] > 1e-10]
       grad = np.zeros(n)
       for k_idx in top_idx:
           w = weights[k_idx]
           shift = k_idx - (n - 1)
           s = abs(shift)
           if s < n:
               if shift >= 0:
                   grad[s:] += w * (1.0 - h[:n-s])
                   grad[:n-s] += w * (-h[s:])
               else:
                   grad[:n-s] += w * (1.0 - h[s:])
                   grad[s:] += w * (-h[:n-s])
       return grad * dx

       """Adam optimizer with softmax-approximated gradient."""
       n = len(h)
       h = h.astype(float).copy()
       m = np.zeros(n)  # First moment
       v = np.zeros(n)  # Second moment
       beta1, beta2, eps = 0.8, 0.999, 1e-8
       best_c5 = compute_c5(h)
       best_h = h.copy()
       
       for it in range(n_iter):

               break
           frac = it / max(n_iter - 1, 1)
           T = T_start * (T_end / T_start) ** frac
           grad = softmax_grad(h, T)
           
           t = it + 1
           m = beta1 * m + (1 - beta1) * grad
           v = beta2 * v + (1 - beta2) * grad ** 2
           m_hat = m / (1 - beta1 ** t)
           v_hat = v / (1 - beta2 ** t)
           
           h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
           h = make_feasible(h, n)
           
           if (it + 1) % 50 == 0:

                   best_h = h.copy()
       
       c5 = compute_c5(h)
       if c5 < best_c5:
           best_c5 = c5
           best_h = h.copy()
       return best_h, best_c5

   def coord_descent(h, n_sweeps=500, start_time=None, time_budget=None):

def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _subgrad_step(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

        h0 = make_feasible(h0, n_init)

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    n0 = len(best_h)
    print(f"Initial: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: Adam GD at initial resolution
    if time.time() - start_time < budget_s - 200:
        h1, c1 = adam_gd(best_h, n_iter=4000, T_start=60.0, T_end=1.0, lr=0.06,
                         start_time=start_time, time_budget=budget_s - 180)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        print(f"AdamGD n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: Coordinate descent at initial resolution
    if time.time() - start_time < budget_s - 120:
        h2, c2 = coord_descent(best_h, n_sweeps=300, start_time=start_time,
                               time_budget=budget_s - 100)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        print(f"Coord n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: SLSQP polish at initial resolution
    if n0 <= 512 and time.time() - start_time < budget_s - 60:
        for T_val in [8.0, 3.0]:
            if time.time() - start_time > budget_s - 30:
                break
            h_s, c_s = slsqp_refine(best_h, T=T_val, maxiter=400)
            if c_s < best_c5:
                best_c5, best_h = c_s, h_s
        print(f"SLSQP n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 4: Multi-resolution approach
    target_resolutions = sorted(set([n0 * 2, n0 * 3, n0 * 4]), key=lambda x: x)
    for new_n in target_resolutions:
        if new_n < n0 or time.time() - start_time > budget_s - 200:
            continue
        h_up = upsample(best_h, new_n)
        c_up = compute_c5(h_up)
        print(f"Up to n={new_n}: C5={c_up:.6f}", flush=True)

        # Adam GD at new resolution
        if time.time() - start_time < budget_s - 80:
            h_g, c_g = adam_gd(h_up, n_iter=3000, T_start=40.0, T_end=0.5, lr=0.04,
                               start_time=start_time, time_budget=budget_s - 60)
            if c_g < c_up:
                h_up, c_up = h_g, c_g

        # Coordinate descent
        if time.time() - start_time < budget_s - 40:
            h_cd, c_cd = coord_descent(h_up, n_sweeps=200, start_time=start_time,
                                       time_budget=budget_s - 25)
            if c_cd < c_up:
                h_up, c_up = h_cd, c_cd

        # SLSQP if size allows
        if new_n <= 480 and time.time() - start_time < budget_s - 30:
            h_s, c_s = slsqp_refine(h_up, T=5.0, maxiter=300)
            if c_s < c_up:
                h_up, c_up = h_s, c_s

        if c_up < best_c5:
            best_c5, best_h = c_up, h_up.copy()
        else:
            print(f"  No improvement at n={new_n}, keeping n={len(best_h)}", flush=True)
        print(f"Res {new_n}: best C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 5: Multi-start at best resolution
    n_pts = len(best_h)
    if time.time() - start_time < budget_s - 100 and n_pts <= 640:
        for attempt in range(4):
            if time.time() - start_time > budget_s - 20:
                break
            scale = [0.05, 0.12, 0.2, 0.35][attempt]
            h_p = best_h + rng.randn(n_pts) * scale
            h_p = make_feasible(h_p, n_pts)
            h_r, c_r = adam_gd(h_p, n_iter=2000, T_start=30.0, T_end=1.0, lr=0.05,
                               start_time=start_time, time_budget=budget_s - 40)
            h_cr, c_cr = coord_descent(h_r, n_sweeps=100, start_time=start_time,
                                       time_budget=budget_s - 25)
            if c_cr < c_r:
                h_r, c_r = h_cr, c_cr
            if c_r < best_c5:
                best_c5, best_h = c_r, h_r
                print(f"  Restart {attempt} (scale={scale}): C5={best_c5:.6f}", flush=True)

    # Phase 6: Final fine Adam + coord at current resolution
    if time.time() - start_time < budget_s - 30:
        n_pts = len(best_h)
        h_f, c_f = adam_gd(best_h, n_iter=3000, T_start=5.0, T_end=0.1, lr=0.015,
                           start_time=start_time, time_budget=budget_s - 15)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f
        h_f2, c_f2 = coord_descent(best_h, n_sweeps=200, start_time=start_time,
                                   time_budget=budget_s - 10)
        if c_f2 < best_c5:
            best_c5, best_h = c_f2, h_f2
        print(f"Final refine: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
