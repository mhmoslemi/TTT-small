    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full')
    C_max = np.max(C)
    # Soft-max objective
    logits = T * (C - C_max)
    exp_l = np.exp(logits)
    s = np.sum(exp_l)
    obj = C_max + np.log(s) / T
