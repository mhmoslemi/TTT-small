def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n

        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0

                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def soft_obj(h, n_pts, T):
    dx = 2.0 / n_pts
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -100.0 / T, 0.0)
    soft = mc + np.log1p(np.sum(np.exp(T * shifted) - 1.0)) / T
    return soft * dx

def sum_con_fun(h, n_pts):
    return np.sum(h) - n_pts / 2.0

def sum_con_jac(h, n_pts):
    return np.ones(n_pts)

            v = cur + d
            if 0.0 <= v <= 1.0 and v != cur:
                candidates.append(v)
        candidates.extend([0.0, 1.0])

        if len(candidates) > max_candidates:

        best_val = cur
        best_c = best_c5

            trial[i] = v

                best_val = v
                best_h_trial = trial.copy()
        if best_val != cur:
            h = best_h_trial
            best_c5 = best_c
