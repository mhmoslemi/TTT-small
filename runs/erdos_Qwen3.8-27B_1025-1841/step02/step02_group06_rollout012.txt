def fast_corr(h, n):

    fh = np.fft.rfft(h, size)

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    s0 = np.sum(np.clip(g, 0.0, 1.0))
    if s0 == target:
        return np.clip(g, 0.0, 1.0)
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(40):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.clip(g - t, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - np.sum(result)
    if abs(diff) > 1e-10:
        # Spread correction
        mask = (result > 1e-10) & (result < 1 - 1e-10)
        nmask = mask.sum()
        if nmask > 0:
            result[mask] += diff / nmask
            result = np.clip(result, 0.0, 1.0)
    return result

        # Compute correlation and softmax weights

        if c5_cur < c5_best - 1e-12:

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(45):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def get_fft_size(n):
    size = 1
    while size < 2 * n:
        size <<= 1
    return size


def compute_c5_and_result(h, n, size):
    oth = 1.0 - h
    fh = np.fft.rfft(h, size)
    fo = np.fft.rfft(oth, size)
    result = np.fft.irfft(fh * np.conj(fo), size)
    dx = 2.0 / n
    c5 = max(result[:n].max(), result[size - n + 1:].max()) * dx
    return c5, result


def softmax_circular(result, n, size, tau):
    c_full = np.zeros(size)
    c_full[:n] = result[:n]
    c_full[size - n + 1:] = result[n:]
    max_c = c_full.max()
    if max_c < 1e-15:
        return np.zeros(size)
    shifted = np.clip(tau * (c_full - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    if s < 1e-300:
        w = np.zeros(size)
        w[np.argmax(exp_c)] = 1.0
        return w
    return exp_c / s


def fft_gradient(h, w, n, size, dx):
    fh = np.fft.rfft(h, size)
    fw = np.fft.rfft(w, size)
    s_plus = np.fft.irfft(fh * np.conj(fw), size)[:n]
    s_minus = np.fft.irfft(fh * fw, size)[:n]
    return dx * (1.0 - s_minus - s_plus)


def nesterov_optimize(h, n, size, tau, max_iters, time_limit, lr_init=0.3, momentum=0.8):
    h = project_to_feasible(h, n).copy()
    dx = 2.0 / n
    c5_best, _ = compute_c5_and_result(h, n, size)
    h_best = h.copy()
    v = np.zeros(n)
    start = time.time()
    stall = 0
    patience = 3000
    lr = lr_init

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        c5_cur, result = compute_c5_and_result(h, n, size)
        if c5_cur < c5_best - 1e-12:
            c5_best = c5_cur
            h_best = h.copy()
            stall = 0
        else:
            stall += 1

        w = softmax_circular(result, n, size, tau)
        grad = fft_gradient(h, w, n, size, dx)

        v = momentum * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > patience:
           (lr := lr * 0.7)
            v *= 0.3
            stall = 0
            if lr < 1e-9:
                break
        h = h_new

    return h_best, c5_best


def interp_to_res(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    h_tiled = np.tile(h, 3)
    x_tiled = np.arange(3 * n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n) + 2.0
    return np.interp(x_new, x_tiled, h_tiled)


def make_sinusoidal(n, freq, amp):
    x = np.arange(n) * (2.0 / n)
    h = 0.5 + amp * np.sin(np.pi * freq * x)
    return project_to_feasible(h, n)


def make_tent(n):
    x = np.arange(n) * (2.0 / n)
    h = np.minimum(x * 3.0, (2.0 - x) * 3.0)
    h = np.clip(h, 0.0, 1.0)
    return project_to_feasible(h, n)

    h += 0.05 * rng.randn(n)
