def soft_max_topk(x, n, T, top_k):
    """Soft max of top-k correlations."""
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    if top_k < len(corr):
        # Get top-k values
        idx = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[idx]
    else:
        vals = corr
    m = np.max(vals)
    return float(m + np.log(np.sum(np.exp(vals - m))) / T)

    return soft_max_topk(x, n, T, top_k) + mu * (np.sum(x) - n/2.0)**2

def annealed_lbfgsb(h0, n, T_start, T_end, top_k, mu_start, mu_end, 
                    maxiter_per_phase=200, n_phases=10, deadline=None):

    for phase in range(n_phases):
        if deadline is not None and time.time() > deadline:
            break
        frac = phase / max(n_phases - 1, 1)
        T = T_start + (T_end - T_start) * frac
        mu = mu_start + (mu_end - mu_start) * frac
        
        def f(x):
            return penalty_obj(x, n, T, top_k, mu)

                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter_per_phase, 'ftol': 1e-18, 'gtol': 1e-12})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h = project_sum(res.x, n / 2.0)
        except Exception:
            pass
    return h

def slsqp_refine(h0, n, T=50, maxiter=300, deadline=None):

        return soft_max_full(x, n, T)

    except Exception:
        pass
    return h0

The objective is: f(h) = soft_max_k [corr(k)] where corr(k) = sum_i h(i) * (1-h(i+k)) * dx

The gradient of corr(k) with respect to h(j):

where the sum is over valid i (such that both i and i+k are in [0, n-1]).

Wait, I need to be careful about what "lag k" means in the numpy correlate output.

np.correlate(a, b, mode='full')[k] = sum_i a[i+k-(n-1)] * b[i]

where n = len(a) = len(b), and the sum is over all i such that 0 ≤ i < n and 0 ≤ i+k-(n-1) < n.

So with a = h and b = 1-h:
out[k] = sum_i h[i+k-(n-1)] * (1-h[i])

Let me define s = k - (n-1) (the actual lag). Then:
out[k] = sum_i h[i+s] * (1-h[i])

where the sum is over i such that 0 ≤ i < n and 0 ≤ i+s < n.

d_out[k]/d_h[j] = sum_i [d(h[i+s])/d(h[j]) * (1-h[i]) + h[i+s] * d(1-h[i])/d(h[j])]
= (1-h[j-s]) * I(0≤j-s<n) - h[j+s] * I(0≤j+s<n)

Wait, let me be more careful. out[k] = sum_i h[i+s] * (1-h[i]) where s = k-n+1.

d_out[k]/d_h[j]:
- From the h[i+s] factor: we get (1-h[i]) when i+s = j, i.e., i = j-s. Valid when 0 ≤ j-s < n.
  Contribution: (1-h[j-s])
- From the (1-h[i]) factor: we get -h[i+s] when i = j. Valid when 0 ≤ j < n and 0 ≤ j+s < n.
  Contribution: -h[j+s] (if j+s < n)

So: d_out[k]/d_h[j] = (1-h[j-s]) * I(0 ≤ j-s < n) - h[j+s] * I(0 ≤ j+s < n)

where s = k - n + 1.

This is O(1) per (k, j) pair, so the full gradient computation is O(n * n) = O(n²) (for all k and j). But we only need the gradient for the top-k lags (weighted by the soft-max weights), so it's O(n * top_k).

```python
def analytical_grad_softmax(x, n, T, top_k=50):
    """Compute gradient of soft-max objective w.r.t. x."""
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    
    # Get top-k indices
    if top_k < 2*n - 1:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(2*n - 1)
    
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    weights = np.exp(top_vals - m) / np.sum(np.exp(top_vals - m))
    
    # Gradient
    grad = np.zeros(n)
    for idx_pos, k in enumerate(top_idx):
        w = weights[idx_pos]
        s = k - n + 1  # actual shift
        for j in range(n):
            g = 0.0
            # From h[i+s] factor: i = j-s, need 0 <= j-s < n
            if 0 <= j - s:
                g += (1.0 - x[j - s])
            # From (1-h[i]) factor: i = j, need 0 <= j+s < n
            if 0 <= j + s < n:
                g += -x[j + s]
            grad[j] += w * g * dx
    
    return grad
```

    n = len(h)

def project_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mock)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)

def soft_max_grad_analytic(x, n, T, top_k=30):
    dx = 2.0 / n
    N = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    if top_k < N:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(N)
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    weights = np.exp(top_vals - m) / np.sum(np.exp(top_vals - m))
    
    grad = np.zeros(n)
    for pi, k in enumerate(top_idx):
        w = weights[pi]
        if w < 1e-10:
            continue
        s = int(k) - n + 1
        # For each j, d_corr[k]/d_x[j] = (1-x[j-s]) * I(0<=j-s<n) - x[j+s] * I(0<=j+s<n)

        t2 = np.zeros(n)

            # j - s >= 0 means j >= s. Since s < 0 and j >= 0, this is always true.

            t1[:n - ss] = 1.0 - x[ss:]  # x[j-s] = x[j+ss] for j = 0,...,n-1-ss

        grad += w * dx * (t1 + t2)
    
    return grad

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(x, n, T, top_k=30):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    N = 2 * n - 1
    if top_k < N:
        idx = np.argpartition(corr, -top_k)[-top_k:]
        vals = corr[idx]
    else:
        vals = corr
    m = np.max(vals)
    return float(m + np.log(np.sum(np.exp(vals - m))) / T)

def soft_max_grad(x, n, T, top_k=30):
    dx = 2.0 / n
    N = 2 * n - 1
    corr = np.correlate(x, 1.0 - x, mode='full') * dx
    if top_k < N:
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
    else:
        top_idx = np.arange(N)
    top_vals = corr[top_idx]
    m = np.max(top_vals)
    exps = np.exp(top_vals - m)
    weights = exps / np.sum(exps)
    grad = np.zeros(n)
    for pi in range(len(top_idx)):
        w = weights[pi]
        if w < 1e-12:
            continue
        s = int(top_idx[pi]) - n + 1
        t1 = np.zeros(n)
        t2 = np.zeros(n)
        if s >= 0:
            if n - s > 0:
                t1[s:] = 1.0 - x[:n - s]
                t2[:n - s] = -x[s:]
        else:
            ss = -s
            if n - ss > 0:
                t1[:n - ss] = 1.0 - x[ss:]
                t2[ss:] = -x[:n - ss]
        grad += w * dx * (t1 + t2)
    return grad

def lbfgsb_run(h0, n, T_start, T_end, top_k, mu, maxiter=200, deadline=None):
    h = h0.copy()
    for phase in range(3):
        if deadline is not None and time.time() > deadline:
            break
        frac = phase / 2.0
        T = T_start + (T_end - T_start) * frac
        def fun(x):
            return soft_max_obj(x, n, T, top_k) + mu * (np.sum(x) - n / 2.0) ** 2
        def jac(x):
            return soft_max_grad(x, n, T, top_k) + 2.0 * mu * (np.sum(x) - n / 2.0)
        try:
            res = minimize(fun, h, jac=jac, method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-10, 'maxls': 50})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h = project_sum(res.x, n / 2.0)
        except Exception:
            pass
    return h

def slsqp_polish(h0, n, T=25, maxiter=300, deadline=None):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T, top_k=2 * n - 1)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-14, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_sum(res.x, n / 2.0)
    except Exception:
        pass
    return h0

def interp_to(h, n_from, n_to):
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def binary_round(h):
    n = len(h)
    target = n // 2
    idx = np.argsort(h)[::-1]
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

                    best_stall_limit = int(best_stall_limit * 1.5)
                else:
                    break
                stall = 0
    return h, best_c5
