        h_up4 = project_to_feasible(h_up4, n4)
        tl = min(rem() * 0.25, 300.0)
        h4, c5_4 = softgrad_optimize(h_up4, n4, tl, tau_start=15.0, tau_end=80.0, lr_init=0.10, topM=200)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Subgradient polish
    if rem() > 30:
        nf = len(best_h)
        tl = min(rem() * 0.20, 150.0)
        hp, c5p = subgrad_polish(best_h, nf, tl, lr_init=0.003, topK=3)
        if c5p < best_c5:
            best_c5 = c5p
            best_h = hp.copy()

    # Phase 6: Local search
    if rem() > 40:
        nf = len(best_h)
        tl = min(rem() * 0.18, 120.0)
        hs, c5s = local_search(best_h, nf, tl, n_trials=3000)
        if c5s < best_c5:
            best_c5 = c5s
            best_h = hs.copy()

    # Phase 7: Final subgradient micro-polish
    if rem() > 20:
        nf = len(best_h)
        tl = min(rem() * 0.35, 200.0)
        hf, c5f = subgrad_polish(best_h, nf, tl, lr_init=0.001, topK=2)
        if c5f < best_c5:
            best_c5 = c5f
            best_h = hf.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()

    return best_h, best_c5, len(best_h)
