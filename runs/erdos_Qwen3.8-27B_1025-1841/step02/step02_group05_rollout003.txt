def project_to_feasible(g, n):
    target = n / 2.0
    if abs(g.sum() - target) < 1e-12 and g.min() >= -1e-12 and g.max() <= 1.0 + 1e-12:
        return np.clip(g, 0.0, 1.0)
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
