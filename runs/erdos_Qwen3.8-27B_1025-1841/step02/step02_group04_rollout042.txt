```python
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgrad_exact(h, k_idx):
    n = len(h)
    s = k_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        valid = np.arange(n - s)
        grad[valid] -= h[valid + s]
        grad[valid + s] += (1.0 - h[valid])
    else:
        t = -s
        valid = np.arange(n - t)
        grad[valid] += (1.0 - h[valid + t])
        grad[valid + t] -= h[valid]
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_gradient(h, T, top_k=50):
    corr = corr_full(h)
    mc = np.max(corr)
    shifted = corr - mc
    exponents = T * shifted
    max_exp = np.max(exponents)
    exps = np.exp(exponents - max_exp)
    total = np.sum(exps)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return subgrad_exact(h, k_max)
    weights = exps / total
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(len(h))
    for ki in idx:
        w = weights[ki]
        if w < 1e-12:
            break
        grad += w * subgrad_exact(h, ki)
    return grad

    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.3 * frac)
        grad = soft_gradient(h, T)
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
```
```python
        h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5
```

**Alternative: Block coordinate descent with exact optimization**

For each coordinate h[i], given all other h[j] (j≠i), the objective is:
f(h[i]) = max_k R_k(h) where R_k is linear in h[i] (since R_k = sum_j h[j](1-h[j+k]) and h[i] appears in at most 2 terms for each k).

Actually, R_k as a function of h[i] is:
R_k = (terms not involving h[i]) + h[i] * (coeff of h[i] in R_k)

The coefficient of h[i] in R_k:
- From h[i] * (1-h[i+k]): coefficient is (1-h[i+k])
- From h[i-k] * (1-h[i]): coefficient is -h[i-k]
- (and similar for negative shifts)

So R_k is linear in h[i], and max_k R_k is convex piecewise-linear in h[i]. We can minimize this exactly for each coordinate!

This is a **block coordinate descent** with exact 1D minimization. This could be much more effective than gradient-based methods because it makes the best possible move at each step.

```python
    return h, best_c5
```

One more thought: the ternary search for the 1D problem can be replaced with a more exact method. Since g(t) = max_k (A_k + B_k * t) is a max of linear functions, its minimum over [0,1] occurs at:
- t = 0
- t = 1
- Or at an intersection point of two of the lines

The intersection of lines k and l is at t = (A_l - A_k) / (B_k - B_l) (when B_k ≠ B_l).

So I can collect all intersection points in [0,1], evaluate g at each, and take the minimum. But there are O((2n)^2) pairs, which is O(n^2) per coordinate - too many.

A better approach: use the fact that g(t) is convex and use a bisection-like method on the subgradient. The subgradient of g(t) is the set of B_k for k in argmax. If all active B_k > 0, the minimum is to the left; if all < 0, to the right; if some are + and some -, we're at the minimum.

This is essentially the same as what ternary search does, but potentially faster. Let me stick with ternary search for simplicity.

Actually, for 40 iterations of ternary search on [0,1], the final interval has width (2/3)^40 ≈ 5e-8, which is more than enough precision.

```python
import numpy as np
from scipy.optimize import minimize
import time

_PARAMS = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')

def subgrad_exact(h, k_idx):
    n = len(h)
    s = k_idx - (n - 1)
    grad = np.zeros(n)
    if s >= 0:
        valid = np.arange(n - s)
        grad[valid] -= h[valid + s]
        grad[valid + s] += (1.0 - h[valid])
    else:
        t = -s
        valid = np.arange(n - t)
        grad[valid] += (1.0 - h[valid + t])
        grad[valid + t] -= h[valid]
    return grad

def project_feasible(h, n):
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def soft_gradient(h, T, top_k=50):
    corr = corr_full(h)
    mc = np.max(corr)
    shifted = corr - mc
    exponents = T * (shifted - np.max(shifted))
    exps = np.exp(exponents)
    total = np.sum(exps)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return subgrad_exact(h, k_max)
    weights = exps / total
    idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(len(h))
    for ki in idx:
        w = weights[ki]
        if w < 1e-12:
            break
        grad += w * subgrad_exact(h, ki)
    return grad

def adam_optimize(h, n_iter, T_start, T_end, lr, beta1, beta2, eps):
    n = len(h)
    h = h.astype(float).copy()
    m = np.zeros(n)
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.3 * frac)
        grad = soft_gradient(h, T)
        t = it + 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h = h - lr_t * m_hat / (np.sqrt(v_hat) + eps)
        h = project_feasible(h, n)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return project_feasible(h_new, new_n)

def _obj(hx):
    T = _PARAMS['T']
    n = _PARAMS['n']
    dx = 2.0 / n
    corr = corr_full(hx)
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * (shifted - np.max(shifted)))) / T + 1e-30) * dx

def _grad(hx):
    T = _PARAMS['T']
    n = _PARAMS['n']
    dx = 2.0 / n
    corr = corr_full(hx)
```
