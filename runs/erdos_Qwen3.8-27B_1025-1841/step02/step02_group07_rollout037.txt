import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_weights(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    u = corr * dx
    # Numerically stable softmax
    u_max = np.max(u)
    exp_u = np.exp(T * (u - u_max))
    w = exp_u / np.sum(exp_u)
    return w, u_max

def compute_gradient(h, w):
    """Compute gradient of log-sum-exp objective w.r.t. h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in range(2*n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            # 1-h[p+d] valid for p = 0..n-1-d
            grad[0:n-d] += wk * (1.0 - h[d:n])
            # h[p-d] valid for p = d..n-1
            grad[d:n] -= wk * h[0:n-d]
        else:
            dp = -d  # positive
            # 1-h[p-dp] valid for p = dp..n-1
            grad[dp:n] += wk * (1.0 - h[0:n-dp])
            # h[p+dp] valid for p = 0..n-1-dp
            grad[0:n-dp] -= wk * h[dp:n]
    return dx * grad

def project_feasible(h, n):
    """Project h to satisfy 0<=h<=1 and sum(h)=n/2"""
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def lbfgsb_step(h, n, T, mu, maxiter, deadline):
    """One L-BFGS-B optimization step"""
    def obj(x):
        w, _ = compute_weights(x, T)
        u = np.correlate(x, 1.0-x, mode='full') * (2.0/n)
        u_max = np.max(u)
        val = (u_max + np.log(np.sum(np.exp(T*(u - u_max))))) / T
        val += mu * (np.sum(x) - n/2.0)**2
        return val
    
    def grad(x):
        w, _ = compute_weights(x, T)
        g = compute_gradient(x, w)
        g += 2.0 * mu * (np.sum(x) - n/2.0)
        return g
    
    try:
        res = minimize(obj, h, jac=grad, method='L-BFGS-B',
                       bounds=[(0.0, 1.0)]*n,
                       options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h

def coordinate_descent(h, n, n_rounds, deadline):
    """Local coordinate descent to reduce max correlation"""
    h = h.copy()
    for _ in range(n_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            # Try small perturbations of h[i]
            best_val = evaluate_c5(h)
            for delta in [-0.01, 0.01, -0.005, 0.005, -0.02, 0.02]:
                h_old = h[i]
                h[i] = np.clip(h_old + delta, 0.0, 1.0)
                # Maintain sum constraint approximately
                # (skip for now, will project at end)
                val = evaluate_c5(h)
                if val < best_val - 1e-10:
                    best_val = val
                    improved = True
                else:
                    h[i] = h_old
                    break
            if time.time() > deadline:
                break
        h = project_feasible(h, n)
        if not improved:
            break
    return h

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    
    np.random.seed(seed)
    
    # Initialize

        h = np.array(initial_h_values, dtype=float).copy()
    else:
        n = 336
        h = np.full(n, 0.5)
    
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Phase 1: L-BFGS-B with increasing temperature
    T_schedule = [20, 50, 100, 200, 500, 1000]
    mu = 100.0
    
    for T in T_schedule:
        if time.time() > deadline - 80:
            break
        maxiter = 200
        h = lbfgsb_step(h, n, T, mu, maxiter, deadline)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Phase 2: Coordinate descent refinement
    if time.time() < deadline - 30:
        h_cd = coordinate_descent(best_h, n, 50, deadline)
        c5_cd = evaluate_c5(h_cd)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()
    
    # Phase 3: Evaluate at different resolutions
    # Interpolate best_h to different n values and evaluate
    candidate_ns = [n]
    for mult in [0.5, 1.5, 2, 3]:
        candidate_ns.append(int(n * mult))
    candidate_ns = sorted(set(candidate_ns))
    
    for n_try in candidate_ns:
        if time.time() > deadline - 20:
            break
        if n_try < 16:
            continue
        # Interpolate
        x_orig = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_orig, best_h)
        h_try = project_feasible(h_try, n_try)
        c5_try = evaluate_c5(h_try)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
    
    # Phase 4: More L-BFGS-B at the best resolution
    n_final = len(best_h)

                best_c5 = c5
                best_h = h.copy()
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def lse_value(x, n, T):
    dx = 2.0 / n
    u = np.correlate(x, 1.0 - x, mode='full') * dx
    u_max = np.max(u)
    return float(u_max + np.log(np.sum(np.exp(T * (u - u_max)))) / T)

def lse_grad(x, n, T):
    dx = 2.0 / n
    u = np.correlate(x, 1.0 - x, mode='full') * dx
    u_max = np.max(u)
    exp_u = np.exp(T * (u - u_max))
    w = exp_u / np.sum(exp_u)
    
    # Compute gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        d = k - (n - 1)
        if d >= 0:
            grad[:n - d] += wk * (1.0 - x[d:])
            grad[d:] -= wk * x[:n - d]
        else:
            dp = -d
            grad[dp:] += wk * (1.0 - x[:n - dp])
            grad[:n - dp] -= wk * x[dp:]
    return dx * grad

def project_feasible(x, n):
    target = n / 2.0
    x = np.clip(x, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(x)
        diff = target - s
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = x < 1.0
        else:
            mask = x > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        x[mask] += diff / cnt
        x = np.clip(x, 0.0, 1.0)
    return x

def slsqp_analytical(x, n, T, maxiter, deadline):
    def obj(v):
        return lse_value(v, n, T)
    
    def grad(v):
        return lse_grad(v, n, T)
    
    def cons_fun(v):
        return np.sum(v) - n / 2.0
    
    def cons_jac(v):
        return np.ones(n)
    
    try:
        res = minimize(obj, x, jac=grad, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return x

def gradient_steps(x, n, T, n_steps, lr, deadline):
    x = x.copy()
    for i in range(n_steps):
        if time.time() > deadline:
            break
        g = lse_grad(x, n, T)
        # Add constraint gradient (Lagrange multiplier approximation)
        constraint_violation = np.sum(x) - n / 2.0
        g += 10.0 * constraint_violation  # penalty-based constraint enforcement
        x = x - lr * g
        x = project_feasible(x, n)
    return x

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    np.random.seed(seed)
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float).copy()
    else:
        h = np.full(336, 0.5)
    
    n = len(h)
    h = project_feasible(h, n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Temperature schedule
    T_schedule = [30, 60, 100, 200, 400]
    
    for T in T_schedule:
        if time.time() > deadline - 100:
            break
        h = slsqp_analytical(h, n, T, maxiter=500, deadline=deadline)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient steps for polish
        if time.time() < deadline - 50:
            h = gradient_steps(h, n, T, 100, 0.01, deadline)
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Try different resolutions
    n_best = len(best_h)
    res_to_try = sorted(set([n_best, n_best * 2, n_best // 2, 256, 512, 672]))
    
    for n_try in res_to_try:
        if time.time() > deadline - 30:
            break
        if n_try < 32 or n_try > 1024:
            continue
        x_orig = np.linspace(0, 2, n_best, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_try = np.interp(x_new, x_orig, best_h)
        h_try = project_feasible(h_try, n_try)
        c5_try = evaluate_c5(h_try)
        
        # Do a few optimization steps at this resolution
        if c5_try <= best_c5 + 0.001:
            if time.time() < deadline - 30:
                h_try = slsqp_analytical(h_try, n_try, 100, maxiter=200, deadline=deadline)
                c5_try = evaluate_c5(h_try)
        
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            n_best = n_try
    
    # Final polish
    if time.time() < deadline - 20:
        n_final = len(best_h)
        best_h = slsqp_analytical(best_h, n_final, 200, maxiter=300, deadline=deadline)
        best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
