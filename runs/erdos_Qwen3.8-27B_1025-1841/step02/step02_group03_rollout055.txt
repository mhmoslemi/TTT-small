import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def grad_shift(h, n, s):
    g = np.zeros(n)
    if s == 0:
        g[:] = 1.0 - 2.0 * h
    elif s > 0:
        if s >= n:
            return g
        g[s:] += (1.0 - h[:n - s])
        g[:n - s] -= h[s:]
    else:
        p = -s
        if p >= n:
            return g
        g[:n - p] += (1.0 - h[p:])
        g[p:] -= h[:n - p]
    return g

def _slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = corr.max()
    shifted = T * (corr - mc)
    return (mc + np.log(np.sum(np.exp(shifted))) / T) * dx

def _slsqp_objgrad(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = corr.max()
    shifted = T * (corr - mc)
    log_sum = np.log(np.sum(np.exp(shifted)))
    val = (mc + log_sum / T) * dx
    weights = np.exp(shifted - log_sum)
    grad = np.zeros(n)
    n_active = 0
    for i in range(len(weights)):
        if weights[i] > 1e-14:
            s = int(i) - (n - 1)
            grad += weights[i] * grad_shift(x, n, s)
            n_active += 1
    if n_active > 0:
        grad *= dx
    return val, grad

def coordinate_descent(h, n, n_rounds, deadline):

        for j in range(n):
            if time.time() > deadline:
