def swap_update(h, cc, ka, n, i, j):
    """Apply swap h[i]:1->0, h[j]:0->1. Return new cc and new max."""
    L = 2 * n - 1
    a1 = i - ka + n - 1
    a2 = i + ka - n + 1
    fi = np.zeros(L)
    m1 = (a1 >= 0) & (a1 < n)
    m2 = (a2 >= 0) & (a2 < n)
    fi[m1] += 1.0 - h[a1[m1]]
    fi[m2] -= h[a2[m2]]
    
    b1 = j - ka + n - 1
    b2 = j + ka - n + 1
    fj = np.zeros(L)
    m1j = (b1 >= 0) & (b1 < n)
    m2j = (b2 >= 0) & (b2 < n)
    fj[m1j] += 1.0 - h[b1[m1j]]
    fj[m2j] -= h[b2[m2j]]
    
    new_cc = cc - fi + fj
    h[i] = 0.0
    h[j] = 1.0
    return new_cc, new_cc.max()
