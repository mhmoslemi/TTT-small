def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] += step
        h = np.clip(h, 0.0, 1.0)
    return h

def init_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def apply_change(corr, h, idx, delta, n):
    """Update corr in-place for h[idx] += delta. h is NOT modified by this function."""
    if abs(delta) < 1e-15:
        return
    # First factor: h[idx] * (1-h)[idx + m - (n-1)]
    # m range: max(0, n-1-idx) to min(2n-2, 2n-2-idx)
    m_lo = max(0, n - 1 - idx)
    m_hi = min(2 * n - 2, 2 * n - 2 - idx)
    if m_hi >= m_lo:
        sec = idx + np.arange(m_lo, m_hi + 1) - (n - 1)
        corr[m_lo:m_hi + 1] += delta * (1.0 - h[sec])
    # Second factor: h[idx - m + (n-1)] * (1-h)[idx]
    # m range: max(0, idx) to min(2n-2, idx+n-1)
    m_lo2 = max(0, idx)
    m_hi2 = min(2 * n - 2, idx + n - 1)
    if m_hi2 >= m_lo2:
        fst = idx - (np.arange(m_lo2, m_hi2 + 1) - (n - 1))
        corr[m_lo2:m_hi2 + 1] += -delta * h[fst]
