- For the term where i=p: we remove h[p]*(1-h[p+k]) = 1*(1-h[p+k])
- For the term where i+k=p, i.e., i=p-k: we change h[p-k]*(1-h[p]) from h[p-k]*(1-1)=0 to h[p-k]*(1-0)=h[p-k]
  So we ADD h[p-k] to O(k)

Net change in O(k) from changing h[p]: 1 to 0:
ΔO(k) = -(1-h[p+k]) + h[p-k] (with boundary conditions)

Similarly, changing h[q] from 0 to 1:
ΔO(k) = +(1-h[q+k]) - h[q-k]

For a swap (p: 1→0, q: 0→1):
ΔO(k) = [-(1-h[p+k]) + h[p-k]] + [(1-h[q+k]) - h[q-k]]

We want to find swaps where this is negative for the "hot" lags.

This is a good directed search strategy! Let me implement it.

But there's a subtlety: the swap must maintain the constraint sum(h) = n/2. A swap of one 1→0 and one 0→1 does this automatically.

Let me also think about what "hot lags" means. If the current max is at lag k*, then reducing O(k*) is the priority. But we also need to make sure we don't increase other lags above O(k*).

def soft_gd(h, n_iter=1000, T_start=20.0, T_end=0.5, lr=0.05, max_time=30, seed=0):
    """Soft gradient descent with log-sum-exp smoothing"""
    n = len(h)
    dx = 2.0 / n
    rng = np.random.RandomState(seed)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    for it in range(n_iter):
        if time.time() - start > max_time:
            break
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        lr_t = lr * (1.0 - 0.5 * frac)
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        exp_s = np.exp(np.clip(T * shifted, 0, 500))
        total = np.sum(exp_s)
        
        if total < 1e-300:
            k_max = int(np.argmax(corr))
            shift = k_max - (n - 1)
            grad = np.zeros(n)
            if shift >= 0:
                s = shift
                grad[s:] += (1.0 - h[:n - s])
                grad[:n - s] += (-h[s:])
            else:
                s = -shift
                grad[:n - s] += (1.0 - h[s:])
                grad[s:] += (-h[:n - s])
        else:
            weights = exp_s / total
            top_idx = np.argsort(weights)[-20:][::-1]
            grad = np.zeros(n)
            for k_idx in top_idx:
                w = weights[k_idx]
                if w < 1e-8:
                    break
                shift = k_idx - (n - 1)
                if shift >= 0:
                    s = shift
                    grad[s:] += w * (1.0 - h[:n - s])
                    grad[:n - s] += w * (-h[s:])
                else:
                    s = -shift
                    grad[:n - s] += w * (1.0 - h[s:])
                    grad[s:] += w * (-h[:n - s])
        
        grad *= dx
        h = h - lr_t * grad
        h = make_feasible(h, n)
        
        if (it + 1) % 100 == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5

def _opt_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(np.clip(T * shifted, 0, 500)))) / T) * dx

def _cons_fun(h, n):
    return np.sum(h) - n / 2.0

def slsqp_refine(h, n, T=5.0, max_iter=300, max_time=60, seed=0):
    """SLSQP refinement with soft-max objective"""
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    
    obj = lambda x: _opt_obj(x, T, n)
    
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
        if res.x is not None and time.time() - start < max_time:
            hx = make_feasible(res.x, n)
            cx = compute_c5(hx)
            return hx, cx
    except Exception:
        pass
    
    return h, compute_c5(h)

        if c1 < best_c5:
            best_c5, best_h = c1, h1
        print(f"Phase1 GD n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Phase 2: Binary conversion + intensive iterated local search at n0
    t_remain = budget_s - (time.time() - start_time)
    if t_remain > 100:
        h_bin = to_binary(best_h, n0)
        c_bin = compute_c5(h_bin)
        if c_bin < best_c5:
            best_c5, best_h = c_bin, h_bin.copy()
        
        time_for_ils = min(150, t_remain - 60)
        h_b, c_b = iterated_binary_search(h_bin, max_time=time_for_ils, seed=seed)
        if c_b < best_c5:
            best_c5, best_h = c_b, h_b
        print(f"Phase2 ILS n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Phase 3: SLSQP at n0
    t_remain = budget_s - (time.time() - start_time)
    if t_remain > 80 and n0 <= 400:
        for T_val in [8.0, 3.0]:
            if t_remain < 30:
                break
            h_s, c_s = slsqp_refine(best_h, n0, T=T_val, max_iter=200, max_time=min(30, t_remain - 20))
            if c_s < best_c5:
                best_c5, best_h = c_s, h_s
        print(f"Phase3 SLSQP n={n0}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Phase 4: Upsample to 2*n0 and optimize
    t_remain = budget_s - (time.time() - start_time)
    n2 = min(n0 * 2, 640)
    if t_remain > 200:
        h_up = upsample_h(best_h, n2)
        c_up = compute_c5(h_up)
        
        # Continue GD at new resolution
        if t_remain > 100:
            h_g, c_g = soft_gd(h_up, n_iter=800, T_start=12.0, T_end=0.5, lr=0.03, max_time=min(60, t_remain - 80), seed=seed + 1)
            if c_g < c_up:
                h_up, c_up = h_g, c_g
        
        # Binary refinement at new resolution
        if t_remain > 80:
            h_b2 = to_binary(h_up, n2)
            c_b2 = compute_c5(h_b2)
            if c_b2 < c_up:
                h_up, c_up = h_b2, c_b2
            time_for_ils2 = min(80, t_remain - 60)
            if time_for_ils2 > 20:
                h_b2, c_b2 = iterated_binary_search(h_up, max_time=time_for_ils2, seed=seed + 2)
                if c_b2 < c_up:
                    h_up, c_up = h_b2, c_b2
        
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up
        print(f"Phase4 n={n2}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Phase 5: Another resolution upsample
    t_remain = budget_s - (time.time() - start_time)
    n3 = min(n0 * 3, 960)
    if t_remain > 150 and n3 > n2:
        h_up3 = upsample_h(best_h, n3)
        c_up3 = compute_c5(h_up3)
        
        if t_remain > 60:
            h_g3, c_g3 = soft_gd(h_up3, n_iter=1000, T_start=10.0, T_end=0.3, lr=0.02, max_time=min(50, t_remain - 40), seed=seed + 3)
            if c_g3 < c_up3:
                h_up3, c_up3 = h_g3, c_g3
        
        if t_remain > 40:
            h_b3 = to_binary(h_up3, n3)
            c_b3 = compute_c5(h_b3)
            if c_b3 < c_up3:
                h_up3, c_up3 = h_b3, c_b3
            time_for_ils3 = min(60, t_remain - 30)
            if time_for_ils3 > 15:
                h_b3, c_b3 = iterated_binary_search(h_up3, max_time=time_for_ils3, seed=seed + 4)
                if c_b3 < c_up3:
                    h_up3, c_up3 = h_b3, c_b3
        
        if c_up3 < best_c5:
            best_c5, best_h = c_up3, h_up3
        print(f"Phase5 n={n3}: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    
    # Phase 6: Final refinement on best solution
    t_remain = budget_s - (time.time() - start_time)
    if t_remain > 30:
        n_pts = len(best_h)
        if n_pts <= 500:
            h_f, c_f = slsqp_refine(best_h, n_pts, T=3.0, max_iter=200, max_time=min(30, t_remain - 15))
            if c_f < best_c5:
                best_c5, best_h = c_f, h_f
        h_f2, c_f2 = soft_gd(best_h, n_iter=500, T_start=3.0, T_end=0.1, lr=0.01, max_time=min(20, t_remain - 10), seed=seed + 5)
        if c_f2 < best_c5:
            best_c5, best_h = c_f2, h_f2
    
    # Final: try binary refinement on best
    t_remain = budget_s - (time.time() - start_time)
    if t_remain > 15:
        n_pts = len(best_h)
        h_bin_final = to_binary(best_h, n_pts)
        h_bf, c_bf = iterated_binary_search(h_bin_final, max_time=min(15, t_remain - 5), seed=seed + 6)
        if c_bf < best_c5:
            best_c5, best_h = c_bf, h_bf
    
    # Validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
