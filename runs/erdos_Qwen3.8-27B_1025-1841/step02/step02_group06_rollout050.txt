    u = np.zeros(n)
    best_c5 = compute_c5(z)
    best_h = z.copy()
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # h-update: minimize smooth_max(corr(h)) + (rho/2)||h - z + u||^2
        target = z - u
        def h_obj_grad(hv):
            h_flat = hv.reshape(-1)
            dx = 2.0 / n
            corr = np.correlate(h_flat, 1.0 - h_flat, mode='full') * dx
            max_c = corr.max()
            shifted = np.clip(tau * (corr - max_c), -100, 100)
            exp_s = np.exp(shifted)
            obj = max_c + np.log(exp_s.sum()) / tau + (rho/2) * np.sum((h_flat - target)**2)
            
            p = exp_s / exp_s.sum()
            grad = rho * (h_flat - target)
            active = np.where(p > 1e-10)[0]
            for k in active:

def fista_smooth(h0, n, tau, max_iters, time_limit, lr_init=0.5, topM=0):
    dx = 2.0 / n
    h = project_to_feasible(h0, n).copy()
    y = h.copy()
    t_k = 1.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        # Compute gradient at y
        corr_y = compute_corr_fft(y) * dx
        max_c = corr_y.max()
        shifted = np.clip(tau * (corr_y - max_c), -100, 100)
        exp_s = np.exp(shifted)
        p = exp_s / exp_s.sum()
        
        # Gradient
        grad = np.zeros(n)
        if topM > 0 and topM < (2*n - 1):
            top_idx = np.argpartition(p, -topM)[-topM:]
            p_top = p[top_idx]
            active = top_idx[p_top > 1e-10]
        else:
            active = np.where(p > 1e-10)[0]
        
        for k in active:
            d = k - (n - 1)
            j0, j1 = max(0, d), min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += p[k] * dx * (1.0 - y[idx - d])
            j0b, j1b = max(0, -d), min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= p[k] * dx * y[idx + d]
        
        # Gradient step
        h_new = project_to_feasible(y - lr_init * grad, n)
        
        # Nesterov update
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k**2)) / 2.0
        y = h_new + ((t_k - 1.0) / t_new) * (h_new - h)
        y = np.clip(y, 0.0, 1.0)  # Keep in box (projection handles rest)
        t_k = t_new
        h = h_new
        
        # Check
        if it % 10 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()

            if c5_c < compute_c5(interp_periodic(h_c, n_init)):
                pass  # keep h_c as is
        # Interpolate back
        h_back = interp_periodic(h_c, n_init)
        h_back = project_to_feasible(h_back, n_init)
        c5_back = compute_c5(h_back)
        if c5_back < best_c5:
            best_c5 = c5_back
            best_h = h_back.copy()

    # Phase 2: Fine optimization at n_init with increasing tau
    if remaining() > 60:
        for tau in [10.0, 30.0, 60.0, 120.0]:
            if remaining() < 20:
                break
            tl = min(remaining() * 0.13, 120.0)
            h_f, c5_f = fista_optimize(best_h, n_init, tau, max_iters=4000,
                                        time_limit=tl, lr_init=0.25, topM=150)
            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    # Phase 3: Try higher resolution
    n_hi = n_init * 2
    if remaining() > 120:
        h_hi = interp_periodic(best_h, n_hi)
        h_hi = project_to_feasible(h_hi, n_hi)
        for tau in [20.0, 50.0, 100.0]:
            if remaining() < 30:
                break
            tl = min(remaining() * 0.12, 150.0)
            h_h, c5_h = fista_optimize(h_hi, n_hi, tau, max_iters=2500,
                                        time_limit=tl, lr_init=0.2, topM=120)
            if c5_h < best_c5:
                best_c5 = c5_h
                best_h = h_h.copy()

    # Phase 4: Sharpening attempts
    if remaining() > 30:
        h_sh = affine_sharpen(best_h, len(best_h))
        c5_sh = compute_c5(h_sh)
        if c5_sh < best_c5:
            best_c5 = c5_sh
            best_h = h_sh.copy()

    # Phase 5: Random restarts at current resolution
    if remaining() > 60:
        n_r = n_init
        for attempt in range(4):
            if remaining() < 20:
                break
            noise_scale = 0.01 * (attempt + 1)
            h_r = best_h + noise_scale * np.random.randn(n_r)
            h_r = project_to_feasible(h_r, n_r)
            tl = min(remaining() * 0.1, 60.0)
            h_r, c5_r = fista_optimize(h_r, n_r, tau=50.0, max_iters=2000,
                                        time_limit=tl, lr_init=0.2, topM=150)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Phase 6: Subgradient polish
    if remaining() > 30:
        n_f = len(best_h)
        h_f = best_h.copy()
        h_f, c5_f = subgrad_polish(h_f, max_iters=60000,
                                    time_limit=remaining() * 0.35, topK=3, lr_init=0.003)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Phase 7: Final fine polish with very low temperature
    if remaining() > 20:
        n_f = len(best_h)
        tl = remaining() * 0.3
        h_f, c5_f = fista_optimize(best_h, n_f, tau=300.0, max_iters=3000,
                                    time_limit=tl, lr_init=0.1, topM=50)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_f.copy()

    # Final feasibility and evaluation
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
