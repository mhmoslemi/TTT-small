    """Compute full linear cross-correlation using FFT."""
    n = len(a)
    m = len(b)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)

    return c[:n + m - 1]

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    grad = np.ones(n) * dx  # the "1" part times dx
    
    # Only process significant shifts
    active = p > 1e-12
    for k in np.where(active)[0]:
        pk = p[k] * dx
        sh = k - n + 1
        if sh == 0:
            grad -= pk * (2.0 * h)  # h[i+0] + h[i-0] = 2*h[i]
        elif sh > 0:

**Key insight**: The previous code uses a fixed n=2000 and a schedule of decreasing tau. The issue might be that at n=2000, the discretization is limiting. Or it might be that the optimization landscape at lower tau has many local minima.

Let me try a completely different approach: **Simulated Annealing with the smooth objective**.

In simulated annealing:
- At each step, propose a perturbation (add noise to h, re-project)
- Accept the new state with probability exp(-Δf / T) where Δf is the change in objective
- Gradually decrease T

This can escape local minima!

But the issue is that computing the objective requires a full correlation (O(n^2)), which is expensive for n=2000+.

Alternative: Use the smooth Gaussian approximation with a local move:
- Perturb a small number of random coordinates
- Compute the change in the smooth objective
- Accept/reject

For a single coordinate change h[i] → h[i] + δ:
Δcorr[k] = δ * (1-h[i+sh_k]) - δ * h[i-sh_k] (approximately, for small δ)
Wait, this is for the linear correlation. The change in corr[k] when h[i] changes by δ is:
Δcorr[k] = δ * (1 - h[i+sh_k]) - h[i+sh_k]*0 - δ*h[i-sh_k]... 

Actually, corr[k] = sum_j h[j] - sum_j h[j]*h[j+sh]
∂corr[k]/∂h[i] = 1 - h[i+sh] - h[i-sh] (as derived before)

So for a small change δh[i]:
Δcorr[k] ≈ δh[i] * (1 - h[i+sh_k] - h[i-sh_k])

And the change in the smooth objective:
Δf ≈ (dx/tau) * sum_k p[k] * Δcorr[k] = (dx/tau) * δh[i] * sum_k p[k] * (1-h[i+sh_k]-h[i-sh_k])
= (dx/tau) * δh[i] * (g_raw[i])  (where g_raw is the un-scaled gradient)

This is just the gradient! So local moves are equivalent to gradient steps. SA doesn't help much here.

The real power of SA comes from **non-local moves**. For example:
- Swap two blocks of h
- Add a random "bump" (smooth perturbation)
- Apply a random cosine mode perturbation

import numpy as np
import time
from scipy.fft import dct, idct

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.ones(n) * dx
    active = p > 1e-12
    for k in np.where(active)[0]:
        pk = p[k] * dx
        sh = k - n + 1
        if sh == 0:
            grad -= 2.0 * pk * h
        elif sh > 0:
            grad[:n-sh] -= pk * h[sh:]
            grad[sh:] -= pk * h[:n-sh]
        else:
            a = -sh
            grad[a:] -= pk * h[:n-a]
            grad[:n-a] -= pk * h[a:]
    return dx * (max_c + np.log(s) / tau), grad

def optimize_momentum(h, tau, time_limit, lr=0.3, momentum=0.9, stall_limit=2000):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    curr_lr = lr
    stall = 0
    while time.time() - start < time_limit:
        obj, grad = smooth_grad(h, tau)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > stall_limit:
            break
        v = momentum * v - curr_lr * grad
        h = project_to_feasible(h + v, n)
        curr_lr *= 0.9998
    return best_h, best_c5

def dct_perturb(h, magnitude, n_modes=20, seed=None):
    """Perturb h by adding random DCT modes."""
    rng = np.random.RandomState(seed)
    n = len(h)
    coeffs = dct(h, type=2, norm='ortho')
    # Perturb a random subset of modes
    idx = rng.choice(n, size=min(n_modes, n), replace=False)
    delta = rng.uniform(-magnitude, magnitude, size=len(idx))
    coeffs[idx] += delta
    h_new = idct(coeffs, type=2, norm='ortho')
    return h_new

def subgrad_step(h, lr):
    """Single subgradient step using the maximizing shift."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    k_star = np.argmax(corr)
    sh = k_star - n + 1
    grad = np.ones(n) * dx
    if sh == 0:
        grad -= 2.0 * dx * h
    elif sh > 0:
        grad[:n-sh] -= dx * h[sh:]
        grad[sh:] -= dx * h[:n-sh]
    else:
        a = -sh
        grad[a:] -= dx * h[:n-a]
        grad[:n-a] -= dx * h[a:]
    return h - lr * grad

def optimize_subgrad(h, time_limit, lr=0.002, momentum=0.85):
    n = len(h)
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    curr_lr = lr
    stall = 0
    while time.time() - start < time_limit:
        h = project_to_feasible(h + v, n)
        corr = compute_corr(h)
        c5_now = corr.max() * (2.0/n)
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            curr_lr *= 0.6
            v *= 0.3
            stall = 0
            if curr_lr < 1e-14:
                break
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.ones(n) * (2.0/n)
        if sh == 0:
            grad -= 2.0*(2.0/n)*h
        elif sh > 0:
            grad[:n-sh] -= (2.0/n)*h[sh:]
            grad[sh:] -= (2.0/n)*h[:n-sh]
        else:
            a = -sh
            grad[a:] -= (2.0/n)*h[:n-a]
            grad[:n-a] -= (2.0/n)*h[a:]
        v = momentum * v - curr_lr * grad
    return best_h, best_c5
