def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, np.arange(n), h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    # For numerical stability
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = max_c + np.log(s) / tau
    
    # Gradient
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        # R(k) = sum_{i: valid} h[i]*(1-h[i+sh]) * dx
        # dR(k)/dh[i] = (1-h[i+sh])*dx if i+sh in range, or -h[i+sh]*dx if i-sh in range
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:

            grad[j0:j1] += pk * (1.0 - h[idx - sh])

        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    return obj, grad

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    
    # Estimate total iterations
    # Each iter takes roughly 2*n-1 correlate cost + 2*n-1 loop
    # For n=2000, that's ~12000 ops + python loop over 4000 shifts
    # Let's estimate: ~1ms per iteration for n=2000
    # For 100s budget, ~100000 iterations
    
    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / time_limit)
        
        # Cosine LR schedule
        lr = base_lr * 0.5 * (1 + np.cos(np.pi * progress))
        lr = max(lr, 1e-10)
        
        obj, grad = smooth_obj_grad(h, tau)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:

            best_h = h.copy()
            stall = 0

    """Compute smooth approximation and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = max_c + np.log(s) / tau
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    return obj, grad

def optimize_adam(h, n, tau, time_limit, base_lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
    """Adam optimizer with projection."""
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0

        t += 1
        obj, grad = gradient_c5_smooth(h, tau)

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)

        h = project_to_feasible(h_new, n)
        
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:

        if stall > 5000:
            base_lr *= 0.5
            m *= 0.1
            v *= 0.1
            stall = 0
            if base_lr < 1e-10:
                break

def project_to_feasible(g, n):

    lo = g.min() - 2.0
    hi = g.max() + 2.0

            hi = mid

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def gradient_c5_smooth(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = max_c + np.log(s) / tau

    grad = np.zeros(n)
    thresh = 1e-10
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < thresh:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return obj, grad


def optimize_adam(h, n, tau, time_limit, base_lr=0.01, beta1=0.9, beta2=0.999):
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    stall = 0
    cur_lr = base_lr

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / (time_limit * 0.8))
        lr_t = cur_lr * 0.5 * (1.0 + np.cos(np.pi * progress))
        lr_t = max(lr_t, cur_lr * 0.01)

        obj, grad = gradient_c5_smooth(h, tau)
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        h_new = h - lr_t * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_to_feasible(h_new, n)

        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            cur_lr *= 0.5
            m *= 0.1
            v *= 0.1
            stall = 0
            if cur_lr < 1e-12:
                break

def optimize_subgrad_toph(h, n, time_limit, lr=0.002, top_k=3, momentum=0.85):
    h = project_to_feasible(h.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 5000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        idx_sorted = np.argsort(corr)[::-1]
        top_idx = idx_sorted[:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()

        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b + sh:j1b + sh]
        grad *= dx

        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5


def binary_push(h, n, strength=0.1):
    g = h + strength * (h - 0.5)
