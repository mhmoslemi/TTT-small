def compute_c5(h):

        step = diff / np.sum(mask)
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    return h

def active_set_optimize(h, n, n_active=10, T_smooth=50, max_iter=200):
    """Optimize using active set of lags near the maximum."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)

        corr = np.correlate(best_h, 1.0 - best_h, mode='full')
        max_idx = np.argmax(corr)

        # Find active set: lags within some threshold of max
        threshold = max_val - 0.1 * max_val  # lags that are 90%+ of max
        active_lags = np.where(corr >= threshold)[0]

        if len(active_lags) > n_active:

            # Use log-sum-exp for smoothness
            m = np.max(corr_vals)

        # Optimize

For h(x) = 1/2 + Σ_{k=1}^{K} (a_k cos(πkx) + b_k sin(πkx)):

The LINEAR cross-correlation (not circular!) at shift τ (where |τ| < 2) is:
∫_{max(0,-τ)}^{min(2, 2-τ)} h(x)(1-h(x+τ)) dx

This is more complex than the circular case because of the partial overlap at the boundaries.

≈ ∫₀² h(x)(1-h(x+τ)) dx (periodic)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    # Final correction
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        # Distribute among all non-boundary elements
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
        else:
            idx = np.argmin(h) if diff > 0 else np.argmax(h)
            h[idx] += diff
            h = np.clip(h, 0, 1)
    return h
