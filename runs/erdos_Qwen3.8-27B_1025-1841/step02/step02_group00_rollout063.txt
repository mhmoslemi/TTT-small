def project_onto_set(v, n):
    """Project v onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo, hi = -10, 10
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(v - mid, 0, 1)
        s = np.sum(proj)
        if s > target:
            lo = mid
        else:
            hi = mid
    proj = np.clip(v - (lo + hi) / 2, 0, 1)
    # Fine-tune
    diff = target - np.sum(proj)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = (proj < 1 - 1e-10) & (proj > 0)
        else:
            mask = (proj > 1e-10) & (proj < 1)
        if np.sum(mask) > 0:
            proj[mask] += diff / np.sum(mask)
            proj = np.clip(proj, 0, 1)
    return proj

    lo, hi = -2.0, 2.0

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(corr)
    
    shifted = T * (corr - c_max)
    shifted -= shifted.max()
    exp_vals = np.exp(shifted)
    w = exp_vals / exp_vals.sum()
    
    # grad[j] = dx * (1 - S1[j] - S2[j])
    # S1[j] = full_conv(w, h)[j + n - 1]
    # S2[j] = full_conv(w[::-1], h)[j + n - 1]
    
    N_fft = 1 << (3 * n).bit_length()
    W = np.zeros(N_fft)
    W[:2*n-1] = w
    W_rev = np.zeros(N_fft)
    W_rev[:2*n-1] = w[::-1]
    H = np.zeros(N_fft)
    H[:n] = h
    
    WH = np.fft.rfft(W) * np.fft.rfft(H)
    WRevH = np.fft.rfft(W_rev) * np.fft.rfft(H)
    
    conv_WH = np.fft.irfft(WH, N_fft)
    conv_WRevH = np.fft.irfft(WRevH, N_fft)
    
    S1 = conv_WH[n-1 : 2*n-1]  # indices n-1 to 2n-2
    S2 = conv_WRevH[n-1 : 2*n-1]
    
    smooth_max = c_max + np.log(np.sum(np.exp(T * (corr - c_max)) - (T * (corr - c_max)).max()) + 1) / T
    # Actually: log(sum(exp(T*(c_k - c_max)))) + c_max = c_max + (1/T)*log(sum(exp(T*(c_k - c_max))))
    # = c_max + (1/T)*log(sum(exp(shifted))) where shifted = T*(c_k - c_max)
    # But I already computed exp_vals = exp(shifted - shifted.max())
    # So log(sum(exp(shifted))) = shifted.max() + log(sum(exp_vals)))
    
    obj = dx * (c_max + (shifted.max() + np.log(exp_vals.sum())) / T)
    
    grad = dx * (1.0 - S1 - S2)
    
    return obj, grad

        start_lr = n * 0.25
    
    h = make_feasible(h.copy(), n)
    best = compute_c5(h)
    
    lr = start_lr
    obj, grad = compute_grad(h, T)

            best = c5_trial
            obj, grad = compute_grad(h, T)
            lr = min(lr * 1.2, n)  # increase step size slightly
        else:
            lr *= 0.5
            if lr < 1e-8:
                break
    
    return h, best

            h_cd, c_cd = cd_polish(h_curr, n4, 3, time_left() + start_time, start_time)
            if c_cd < best_c5:
                best_c5 = c_cd
                best_h = h_cd.copy()

    # Phase 5: Final CD polish at best resolution
    n_final = len(best_h)
    cd_n = min(n_final, 2560)
    if cd_n < n_final:
        h_cd_in = upsample_h(best_h, cd_n)
    else:
        h_cd_in = best_h.copy()
    if time_left() > 20:
        h_final, c_final = cd_polish(h_cd_in, cd_n, 6, time_left() + start_time, start_time)
        if c_final < best_c5:
            if cd_n < n_final:
                best_c5 = c_final
                best_h = upsample_h(h_final, n_final)
                # Re-evaluate at full res
                fc = compute_c5(best_h)
                if fc < best_c5:
                    best_c5 = fc
                else:
                    best_c5 = c_final
                    best_h = h_final.copy()
            else:
                best_c5 = c_final
                best_h = h_final.copy()

    # Final feasibility check
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
