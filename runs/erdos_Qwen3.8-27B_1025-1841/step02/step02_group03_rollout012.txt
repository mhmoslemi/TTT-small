import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def to_binary(h, n):
    """Project to binary maintaining sum = n/2"""
    target = n // 2
    # Sort by value descending, take top n/2 as 1s
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

For the Erdős minimum overlap problem, we want a set A ⊂ [0,2] (or its step function indicator) of measure 1 such that for every shift k, the measure of {x : x ∈ A, x+k ∉ A} is minimized.

This is related to "difference sets" and "Sidon sets" in additive combinatorics. The optimal structure in the continuous case is related to the Ruzsa numbers and the structure of sets with small doubling.

For the discrete analog: we want a binary sequence of length n with exactly n/2 ones, minimizing the maximum "cross-correlation" between the sequence and its complement.

This is similar to the problem of minimizing the peak sidelobe level of a sequence's autocorrelation, but for cross-correlation with the complement.

For the optimal solution, the correlation should be as flat as possible (like a "Barker sequence" or "Golay sequence").
