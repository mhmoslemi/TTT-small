def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corr(h):
    n = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx  # actual integral values

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    if remaining() > 60:
        cd_time = min(remaining() - 60, 180)
        best_h, best_c5 = _coord_descent(best_h, N, cd_time, rng)

    # Phase 5: Final SLSQP polish
    if remaining() > 50:
        r_all = corr_values(best_h)
        top = _top_lags(r_all, 5)
        h_new = _slsqp_opt(best_h, N, 2000.0, 200, top)
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new

    # Phase 6: Try higher resolution
    if remaining() > 120:
        N2 = 2560
        h2 = upsample_h(best_h, N2)
        r2 = corr_values(h2)
        top2 = _top_lags(r2, 8)
        h2 = _slsqp_opt(h2, N2, 500.0, 150, top2)
        c2 = compute_c5(h2)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2
        if remaining() > 40:
            h2, c2 = _coord_descent(h2, N2, min(remaining() - 40, 60), rng)
            if c2 < best_c5:
                best_c5 = c2
                best_h = h2

    # Final coordinate descent
    if remaining() > 30:
        best_h, best_c5 = _coord_descent(best_h, len(best_h), min(remaining() - 10, 60), rng)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
