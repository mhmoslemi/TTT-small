def block_lp_step(h_fixed, n):
    """Solve the LP: min t s.t. sum_i h[i]*a_k[i] <= t for all k, 0<=h<=1, sum h = n/2
    where a_k[i] = 1 - h_fixed[i+k-n+1] (the "fixed" part)."""

    # For each k (lag), the constraint is: sum_i h[i] * a_k[i] <= t
    # which is: sum_i a_k[i] * h[i] - t <= 0
    # In standard form: A @ [h; t] <= b

    # For each lag k (0 to 2n-2), d = k - n + 1

    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n + 1))
    b = np.zeros(n_lags)
    
    for k in range(n_lags):
        d = k - n + 1
        if d >= 0:
            # valid i: 0 <= i <= n-1-d
            end = n - d
            A[k, 0:end] = (1.0 - h_fixed[d:n])

            A[k, dd:n] = (1.0 - h_fixed[0:n-dd])
        A[k, n] = -1  # coefficient of t
        b[k] = 0
    
    # Objective: minimize t, so c = [0, 0, ..., 0, 1]
    c = np.zeros(n + 1)
    c[n] = 1
    
    # Bounds: 0 <= h[i] <= 1, t >= 0
    bounds = [(0, 1)] * n + [(0, None)]
    
    # Equality constraint: sum h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([n / 2.0])
    
    result = linprog(c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    
    if result.success:
        h_new = result.x[:n]
        return project(h_new, n)
    return h_fixed

from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(1000):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if not np.any(mask):
            break
        h[mask] += d / np.sum(mask)
        h = np.clip(h, 0.0, 1.0)
    return h

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def gradient_weighted(h, corr, temp=3.0):
    n = len(h)
    n_full = len(corr)
    max_c = np.max(corr)
    exps = np.exp((corr - max_c) / temp)
    total = np.sum(exps)
    if total < 1e-300:
        return np.zeros(n)
    weights = exps / total
    
    g = np.zeros(n)
    for k in range(n_full):
        w = weights[k]
        if w < 1e-8:
            continue
        d = k - (n - 1)
        if d >= 0:
            end = n - d
            if end > 0:
                g[:end] += w * (1.0 - h[d:])
                g[d:] -= w * h[:end]
        else:
            dd = -d
            g[dd:] += w * (1.0 - h[:n-dd])
            g[:n-dd] -= w * h[dd:]
    return g

def subgradient_opt(h0, deadline, lr_init=0.05, top_temp=3.0):
    """Projected subgradient/softmax gradient optimization."""
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = lr_init
    
    for it in range(500000):
        if time.time() > deadline:
            break
        corr = compute_corr(h)
        g = gradient_weighted(h, corr, top_temp)
        
        h_new = project(h - lr * g, n)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            # Slowly increase temperature to explore more

        else:
            lr *= 0.95
            if lr < 1e-7:

                # Perturb

                top_temp = min(top_temp * 1.5, 20.0)
        
        # Decay learning rate
        lr = max(lr * 0.99999, 1e-7)
    
    return best_h, best_c5

    """One block coordinate step: min t s.t. h new correlates with (1-h_fix) <= t for all lags."""
    n_lags = 2 * n - 1
    A = np.zeros((n_lags, n + 1))

            A[k, dd:] = (1.0 - h_fix[:n-dd])

def lp_min_max(a_row_weights, n):
    """Solve: min t s.t. sum_i x[i]*a[k][i] <= t for all k, sum x = n/2, 0<=x<=1
    where a_row_weights is a (n_lags, n) matrix."""
    n_lags = a_row_weights.shape[0]
    A = np.hstack([a_row_weights, -np.ones((n_lags, 1))])
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    bounds = [(0, 1)] * n + [(0, None)]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    res = linprog(c_obj, A_ub=A, b_ub=np.zeros(n_lags), A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

    return None
