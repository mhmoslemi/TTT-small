    corr = np.correlate(h, 1.0 - h, mode='full')

```python
def c5_and_grad_fast(h, tau, top_m=300):
    n = len(h)
    dx = 2.0 / n
    m_total = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = float(corr.max())
    c5 = max_c * dx
    shifted = np.clip(tau * (corr - max_c), -500.0, 500.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    # Select top-m shifts
    if top_m < m_total:
        idxs = np.argpartition(p, -top_m)[-top_m:]
    else:
        idxs = np.arange(m_total)
    grad = np.zeros(n)
    for k in idxs:
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = int(k) - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx / tau
    return c5, grad
```
