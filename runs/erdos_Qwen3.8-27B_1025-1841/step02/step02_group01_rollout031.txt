- For coordinate j, try values in [0, 1] (respecting the sum constraint)
- The sum constraint means: if we change h[j] by δ, we need to adjust other values by -δ/(n-1) (distributed) or adjust one other coordinate.

Let me simplify: I'll use the "adjust one coordinate and compensate in another" approach:
- Change h[j] by δ
- Change h[j'] by -δ (where j' is chosen to minimize impact, e.g., the coordinate with smallest h value if δ > 0)
- This maintains the sum constraint exactly

This is more efficient than the previous approach.

Let me also think about the SLSQP formulation. The objective is:
f(h) = [max(corr) + (1/T) * log(Σ exp(T * (corr - max(corr))))] * dx

The gradient is:
∇f(h) = dx * Σ_j w_j * ∂corr_j/∂h

where w_j are the softmax weights.

The constraints are:
- h[i] ≥ 0 (box)
- h[i] ≤ 1 (box)
- Σ h[i] = n/2 (equality)

SLSQP can handle all of these.

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = int(np.sum(mask))
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def smooth_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    val = mc + np.log(np.sum(np.exp(T * (corr - mc)))) / T
    return val * dx

def smooth_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    w = np.exp(T * (corr - mc))
    w /= np.sum(w)
    grad = np.zeros(n)
    for j in range(2 * n - 1):
        if w[j] < 1e-12:
            continue
        lag = j - (n - 1)
        if lag >= 0:
            grad[:n-lag] += w[j] * (1.0 - h[lag:])
            grad[lag:] += w[j] * (-h[:n-lag])
        else:
            lp = -lag
            grad[lp:] += w[j] * (1.0 - h[:n-lp])
            grad[:n-lp] += w[j] * (-h[lp:])
    return grad * dx

def slsqp_optimize(h0, n, T, maxiter=400, ftol=1e-12):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0,
             'jac': lambda h: np.ones(n)}]
    try:
        res = minimize(smooth_obj, h0, args=(T,), method='SLSQP',
                       jac=lambda h: smooth_grad(h, T),
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        return res.x
    except Exception:
        return h0

def coord_descent(h, n, max_rounds=20, time_limit=None, start_time=None):
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    for rnd in range(max_rounds):
        if time_limit and time.time() - start_time > time_limit - 5:
            break
        improved = False
        for j in range(n):
            if time_limit and time.time() - start_time > time_limit - 5:
                break
            old = h[j]
            # Try candidates: 0, 1, 0.5, and perturbations
            candidates = [0.0, 1.0, 0.5,
                          max(0.0, old - 0.05), max(1.0, old + 0.05),
                          max(0.0, old - 0.02), max(1.0, old + 0.02),
                          max(0.0, old - 0.01), max(1.0, old + 0.01)]
            best_val = old
            best_trial_c5 = best_c5
            for c in candidates:
                if c < -1e-9 or c > 1.0 + 1e-9:
                    continue
                c = float(np.clip(c, 0.0, 1.0))
                if abs(c - old) < 1e-8:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                tc = compute_c5(trial)
                if tc < best_trial_c5 - 1e-12:
                    best_trial_c5 = tc
                    best_val = c
            if best_val != old:
                h[j] = best_val
                h = make_feasible(h, n)
                if best_trial_c5 < best_c5:
                    best_c5 = best_trial_c5
                    improved = True
        if not improved:
            break
    return h, best_c5

def pairwise_cd(h, n, max_rounds=10, time_limit=None, start_time=None):
    """Coordinate descent that adjusts pairs to maintain sum exactly."""
    best_c5 = compute_c5(h)
    rng = np.random.RandomState(0)
    for rnd in range(max_rounds):
        if time_limit and time.time() - start_time > time_limit - 5:
            break
        improved = False
        for _ in range(n * 3):
            if time_limit and time.time() - start_time > time_limit - 5:
                break
            i, j = rng.choice(n, 2, replace=False)
            delta = rng.choice([-0.1, -0.05, -0.02, -0.01, 0.01, 0.02, 0.05, 0.1])
            new_i = h[i] + delta
            new_j = h[j] - delta
            if 0.0 <= new_i <= 1.0 and 0.0 <= new_j <= 1.0:
                trial = h.copy()
                trial[i] = new_i
                trial[j] = new_j
                tc = compute_c5(trial)
                if tc < best_c5 - 1e-12:
                    best_c5 = tc
                    h[i] = new_i
                    h[j] = new_j
                    improved = True
        if not improved:
            break
    return h, best_c5
