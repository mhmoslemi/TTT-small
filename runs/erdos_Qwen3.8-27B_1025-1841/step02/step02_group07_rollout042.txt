```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(h, n, T):
    """Analytical gradient of the softmax approximation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    weights = exps / np.sum(exps)
    
    n = len(h)
    lags = np.arange(-(n-1), n)  # shape (2n-1,)
    
    grad = np.zeros(n)
    for j in range(n):
        # First term: sum over p where both j and j-lag_p are in [0, n-1]
        # lag_p in [j-(n-1), j]
        mask1 = (lags >= j - (n-1)) & (lags <= j)
        idx1 = j - lags[mask1]
        first = np.dot(weights[mask1], 1.0 - h[idx1])
        
        # Second term: sum over p where both j and j+lag_p are in [0, n-1]
        # lag_p in [-j, n-1-j]
        mask2 = (lags >= -j) & (lags <= n-1-j)
        idx2 = j + lags[mask2]
        second = np.dot(weights[mask2], h[idx2])
        
        grad[j] = dx * (first - second)
    
    return grad

def subgrad_step(h, n, lr):
    """One subgradient step for the exact max."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = np.argmax(corr)
    lag = k_star - (n - 1)
    
    # Gradient of c_{k_star} at h
    # dc/dh[j] = (1-h[j-lag]) * 1_{j, j-lag in [0,n-1]} - h[j+lag] * 1_{j, j+lag in [0,n-1]}
    grad = np.zeros(n)
    for j in range(n):
        val = 0.0
        if 0 <= j - lag < n:
            val += (1.0 - h[j - lag])
        if 0 <= j + lag < n:
            val -= h[j + lag]
        grad[j] = val
    
    h_new = h - lr * grad
    h_new = project(h_new, n)
    return h_new, k_star

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def pgd_optimize(h0, n, T, lr_init, n_max_iter, deadline, decrease_T=True):
    """Projected gradient descent with backtracking line search."""
    h = h0.copy()
    best_obj = softmax_obj(h, n, T)
    best_h = h.copy()
    lr = lr_init
    
    for it in range(n_max_iter):
        if time.time() > deadline:
            break
        
        obj = softmax_obj(h, n, T)
        if obj < best_obj:
            best_obj = obj
            best_h = h.copy()
        
        grad = softmax_grad(h, n, T)
        
        # Backtracking line search
        lr_try = lr
        for bt in range(20):
            h_trial = project(h - lr_try * grad, n)
            obj_trial = softmax_obj(h_trial, n, T)
            if obj_trial < obj - 1e-4 * lr_try * np.sum(np.abs(grad) * h - h) :
                # Actually, just use a simple sufficient decrease
                break
            if obj_trial < obj:
                break
            lr_try *= 0.5
        
        h = project(h - lr_try * grad, n)
        
        # Adaptive learning rate
        if obj < best_obj:
            lr = min(lr * 1.05, 0.5)
        else:
            lr *= 0.7
        
        # Decrease temperature
        if decrease_T and it % 500 == 499 and T > 10:
            T = max(10.0, T * 0.95)
    
    return best_h, T

def subgrad_polish(h0, n, n_iter, deadline):
    """Polish with subgradient method."""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    lr = 0.01
    stall_count = 0
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        h_new, k_star = subgrad_step(h, n, lr)
        c5_new = evaluate_c5(h_new)

            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new
            lr = min(lr * 1.02, 0.1)
            stall_count = 0
        else:
            stall_count += 1
            if stall_count > 50:
                lr *= 0.8
                stall_count = 0
    
    return best_h, best_c5

def block_init(n, n_blocks):
    """Create a block pattern initial condition."""
    h = np.zeros(n)
    block_size = n // n_blocks
    for i in range(n_blocks // 2):
        start = i * 2 * block_size
        end = min(start + block_size, n)
        h[start:end] = 1.0
    return project(h, n)

    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:

            if c5_work < best_c5:

            best_c5 = c5_r
            best_h = h_rand.copy()
```

```python
def softmax_grad_fast(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    w = exps / np.sum(exps)  # length 2n-1
    
    # First term: grad_first[j] = np.convolve(w, 1-h, mode='full')[n-1+j]
    conv1 = np.convolve(w, 1.0 - h, mode='full')  # length 3n-2
    grad_first = conv1[n-1 : 2*n-1]  # indices n-1 to 2n-2, length n
    
    # Second term: grad_second[j] = np.convolve(w, h[::-1], mode='full')[2n-2-j]
    conv2 = np.convolve(w, h[::-1], mode='full')  # length 3n-2
    grad_second = conv2[n-1 : 2*n-1][::-1]  # indices n-1 to 2n-2, reversed
    
    grad = dx * (grad_first - grad_second)
    return grad
```

This is O(n log n) using FFT-based convolution (which numpy uses for large inputs). 

Now let me also think about the subgradient step more carefully.

For the subgradient method, at each step:
1. Compute corr = np.correlate(h, 1-h, mode='full')
2. Find k* = argmax(corr)
3. Compute grad of c_{k*}(h):
   lag = k* - (n-1)
   grad[j] = (1-h[j-lag]) * 1_{0<=j-lag<n} - h[j+lag] * 1_{0<=j+lag<n}
4. Update: h = project(h - lr * grad)

For step 3, I can compute this efficiently:
- Create arrays shifted_h_neg[j] = h[j-lag] if 0<=j-lag<n else 0
- Create arrays shifted_h_pos[j] = h[j+lag] if 0<=j+lag<n else 0
- grad = (1 - shifted_h_neg) * mask_neg - shifted_h_pos * mask_pos

where mask_neg[j] = 1 if 0<=j-lag<n else 0, and mask_pos[j] = 1 if 0<=j+lag<n else 0.

This is O(n) per step.

Now, for the overall algorithm, let me think about what will give the best results:

1. **Resolution choice**: Higher resolution allows for more precise representation. But the correlation computation is O(n^2) (for direct computation) or O(n log n) (for FFT). For the softmax gradient, it's O(n log n). For n=672 or n=1000, this should be fast.

2. **Optimization strategy**: 
   - Start with the parent solution at its native resolution
   - Interpolate to higher resolution
   - Run PGD with decreasing temperature
   - Polish with subgradient method
   - Try multiple initial conditions

3. **Key insight**: The previous code used SLSQP which is a general-purpose optimizer but doesn't exploit the structure of the problem. Our analytical gradient approach should be much faster.

```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad_fast(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    w = exps / np.sum(exps)
    
    conv1 = np.convolve(w, 1.0 - h, mode='full')
    grad_first = conv1[n-1 : 2*n-1]
    
    conv2 = np.convolve(w, h[::-1], mode='full')
    grad_second = conv2[n-1 : 2*n-1][::-1]
    
    return dx * (grad_first - grad_second)

def subgrad_update(h, n, lr):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    # (1-h[j-lag]) for valid j
    if lag >= 0:
        grad[lag:n] = 1.0 - h[:n-lag]
        grad[:n-lag] -= h[lag:]
    else:
        l = -lag
        grad[:n-l] = 1.0 - h[l:]
        grad[l:] -= h[:n-l]
    
    h_new = h - lr * grad
    return project(h_new, n), k_star

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def sturmian_init(n, alpha=0.5):
    """Generate a Sturmian (mechanical) word of length n."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.floor((i+1) * alpha) - np.floor(i * alpha)
    return project(h, n)

def pgd_with_analytical_grad(h0, n, T_init, deadline, max_iter=100000):
    """PGD with analytical gradient and adaptive step size."""
    h = h0.copy()
    T = T_init
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = 0.1
    patience = 0
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute objective and gradient
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > 200:
            break
        
        grad = softmax_grad_fast(h, n, T)
        
        # Gradient step with backtracking
        lr_try = lr
        obj_current = softmax_obj(h, n, T)
        found = False
        for bt in range(15):
            h_trial = project(h - lr_try * grad, n)
            obj_trial = softmax_obj(h_trial, n, T)
            if obj_trial < obj_current:
                found = True
                break
            lr_try *= 0.5
        if not found:
            lr = max(lr * 0.5, 1e-6)
            break
        
        h = h_trial
        lr = min(lr * 1.1, 1.0)
        
        # Periodically decrease temperature
        if it % 500 == 499 and T > 5.0:
            T = max(5.0, T * 0.9)
    
    return best_h, best_c5

def subgrad_polish(h0, n, deadline, max_iter=200000):
    """Polish using subgradient method with averaging."""
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = 0.05
    stall = 0
    
    # Polyak averaging
    h_avg = h.copy()
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        h_new, k_star = subgrad_update(h, n, lr)
        c5_new = evaluate_c5(h_new)
        
        h_avg = (it * h_avg + h_new) / (it + 1)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.01, 0.5)
        else:
            stall += 1
            if stall % 100 == 0:
                lr *= 0.9
                if lr < 1e-5:
                    break
        
        h = h_new
    
    # Check averaged solution
    h_avg = project(h_avg, n)
    c5_avg = evaluate_c5(h_avg)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = h_avg
    
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get initial values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = sturmian_init(336)
    
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Work at multiple resolutions
    target_resolutions = [672, 501]
    
    for n_res in target_resolutions:
        if time.time() > deadline - 30:
            break
        
        h_work = interp_to(best_h, len(best_h), n_res)
        h_work = project(h_work, n_res)
        
        # PGD phase
        h_work, c5_work = pgd_with_analytical_grad(h_work, n_res, T_init=50.0, deadline=deadline)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
        
        # Subgradient polish
        if time.time() < deadline - 20:
            h_work, c5_work = subgrad_polish(h_work, n_res, deadline)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()
    
    # Try alternative initial conditions at best resolution
    n_best = len(best_h)
    
    inits = []
    # Sturmian
    inits.append(sturmian_init(n_best, 0.5))
    inits.append(sturmian_init(n_best, 0.618))  # golden ratio
    # Block patterns
    for nb in [4, 8, 12]:
        h_blk = np.zeros(n_best)
        bs = n_best // nb
        for i in range(nb // 2):
            h_blk[i*2*bs:(i*2+1)*bs] = 1.0
        inits.append(project(h_blk, n_best))
    # Random 0-1
    for s in range(3):
        rng = np.random.RandomState(seed + 7777 + s)
        h_r = np.zeros(n_best)
        ones = rng.choice(n_best, size=n_best//2, replace=False)
        h_r[ones] = 1.0
        inits.append(project(h_r, n_best))
    
    for h_init in inits:
        if time.time() > deadline - 20:
            break
        h_work, c5_work = pgd_with_analytical_grad(h_init, n_best, T_init=30.0, deadline=deadline)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
    
    return (best_h, best_c5, len(best_h))
```

```python
def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(100):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-15:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    w = exps / np.sum(exps)
    
    conv1 = np.convolve(w, 1.0 - h, mode='full')
    grad_first = conv1[n-1:2*n-1]
    
    conv2 = np.convolve(w, h[::-1], mode='full')
    grad_second = conv2[n-1:2*n-1][::-1]
    
    return dx * (grad_first - grad_second)

def subgrad_update(h, n, lr):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:n] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
    else:
        l = -lag
        if n - l > 0:
            grad[:n-l] += (1.0 - h[l:])
            grad[l:] -= h[:n-l]
    
    h_new = h - lr * grad
    return project(h_new, n), k_star

    xf = np.linspace(0, 2, n_from, endpoint=False)
```

```python
        else:
            mask = h > 0.0
        cnt = float(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def softmax_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = T * corr
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    w = exps / np.sum(exps)
    conv1 = np.convolve(w, 1.0 - h, mode='full')
    grad_first = conv1[n-1:2*n-1]
    conv2 = np.convolve(w, h[::-1], mode='full')
    grad_second = conv2[n-1:2*n-1][::-1]
    return dx * (grad_first - grad_second)

def subgrad_update(h, n, lr):
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_star = int(np.argmax(corr))
    lag = k_star - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        if n - lag > 0:
            grad[lag:n] += (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
    else:
        l = -lag
        if n - l > 0:
            grad[:n-l] += (1.0 - h[l:])
            grad[l:] -= h[:n-l]
    h_new = h - lr * grad
    return project(h_new, n), k_star

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def sturmian_init(n, alpha):
    h = np.zeros(n)
    for i in range(n):
        h[i] = np.floor((i + 1) * alpha) - np.floor(i * alpha)
    return project(h, n)

def pgd_optimize(h0, n, T_init, deadline, max_iter=500000):
    h = h0.copy()
    T = T_init
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = 5.0
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        grad = softmax_grad(h, n, T)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        obj_cur = softmax_obj(h, n, T)
        lr_try = lr
        h_new = h
        for bt in range(20):
            h_cand = project(h - lr_try * grad, n)
            obj_cand = softmax_obj(h_cand, n, T)
            if obj_cand < obj_cur:
                h_new = h_cand
                break
            lr_try *= 0.5
        else:
            lr = max(lr * 0.5, 0.1)
            if lr < 0.1:
                break
            continue
        h = h_new
        if obj_cand < obj_cur - 1e-8:
            lr = min(lr * 1.1, 50.0)
        else:
            lr = max(lr * 0.8, 0.1)
        if it % 2000 == 1999 and T > 5.0:
            T = max(5.0, T * 0.85)
    return best_h, best_c5

def subgrad_polish(h0, n, deadline, max_iter=500000):
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    lr = 0.05
    stall = 0
    h_avg = h.copy()
    for it in range(max_iter):
        if time.time() > deadline:
            break
        h_new, k_star = subgrad_update(h, n, lr)
        c5_new = evaluate_c5(h_new)
        h_avg = (it * h_avg + h_new) / (it + 1)
        if c5_new < best_c5 - 1e-13:
            best_c5 = c5_new
            best_h = h_new.copy()
            stall = 0
            lr = min(lr * 1.005, 0.3)
        else:
            stall += 1
            if stall > 300:
                lr *= 0.85
                stall = 0
                if lr < 1e-6:
                    break
        h = h_new
    h_avg_p = project(h_avg, n)
    c5_avg = evaluate_c5(h_avg_p)
    if c5_avg < best_c5:
        best_c5 = c5_avg
        best_h = h_avg_p
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = sturmian_init(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)

    # Main optimization at higher resolutions
    for n_res in [672, 504]:
        if time.time() > deadline - 40:
            break
        h_work = interp_to(best_h, len(best_h), n_res)
        h_work = project(h_work, n_res)
        h_work, c5_work = pgd_optimize(h_work, n_res, T_init=50.0, deadline=deadline)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()
        if time.time() < deadline - 30:
            h_work, c5_work = subgrad_polish(h_work, n_res, deadline)
            if c5_work < best_c5:
                best_c5 = c5_work
                best_h = h_work.copy()

    # Alternative initial conditions
    n_best = len(best_h)
    inits = []
    inits.append(sturmian_init(n_best, 0.5))
    inits.append(sturmian_init(n_best, 0.6180339887498949))
    inits.append(sturmian_init(n_best, 0.3819660112501051))
    rng = np.random.RandomState(seed)
    for s in range(5):
        h_r = np.zeros(n_best)
        ones = rng.choice(n_best, size=n_best // 2, replace=False)
        h_r[ones] = 1.0
        inits.append(project(h_r, n_best))
    for nb in [3, 6]:
        h_blk = np.zeros(n_best)
        bs = n_best // nb
        for i in range(nb // 2):
            h_blk[i*2*bs:(i*2+1)*bs] = 1.0
        inits.append(project(h_blk, n_best))

    for h_init in inits:
        if time.time() > deadline - 15:
            break
        h_work, c5_work = pgd_optimize(h_init, n_best, T_init=30.0, deadline=deadline)
        if c5_work < best_c5:
            best_c5 = c5_work
            best_h = h_work.copy()

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
```
