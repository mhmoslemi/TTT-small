    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_OPT_T * shifted))) / _OPT_T) * dx

def _con_fun(x):
    return np.sum(x) - _OPT_N / 2.0

def _con_jac(x):
    return np.ones(_OPT_N)

def get_remaining(start_time, budget_s):
    return budget_s - (time.time() - start_time)

def do_slsqp(h, n, T, maxiter):
    global _OPT_N, _OPT_T
    _OPT_N = n
    _OPT_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
    try:
        res = minimize(_obj, h, method='SLSQP', bounds=bounds,
                      constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        return make_feasible(res.x, n)
    except Exception:
        return h.copy()

def coord_descent(h, n, max_rounds, time_limit):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    deltas = [0.1, 0.05, 0.02, 0.01, 0.005]
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old = best_h[j]
            cands = [0.0, 1.0]
            for d in deltas:
                cands.append(max(0.0, min(1.0, old - d)))
                cands.append(max(0.0, min(1.0, old + d)))
            best_c = best_c5
            best_val = old
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return best_h, best_c5

def binary_anneal(h, n, n_iter, t_start, t_end, time_limit, seed):
    rng = np.random.RandomState(seed)
    b = (h > 0.5).astype(float)
    b = make_feasible(b, n)
    cur_c5 = compute_c5(b)
    best_b = b.copy()
    best_c5 = cur_c5
    ones_idx = np.where(b == 1.0)[0]
    zeros_idx = np.where(b == 0.0)[0]
    sa_start = time.time()
    for it in range(n_iter):
        if time.time() - sa_start > time_limit:
            break
        T = t_start * (t_end / t_start) ** (it / max(1, n_iter))
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        i = ones_idx[rng.randint(len(ones_idx))]
        j = zeros_idx[rng.randint(len(zeros_idx))]
        b[i] = 0.0
        b[j] = 1.0
        new_c5 = compute_c5(b)
        delta = new_c5 - cur_c5
        if delta < 0 or (T > 1e-12 and rng.random() < np.exp(-delta / T)):
            cur_c5 = new_c5
            ones_idx = np.where(b == 1.0)[0]
            zeros_idx = np.where(b == 0.0)[0]
            if new_c5 < best_c5 - 1e-15:
                best_c5 = new_c5
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
    return best_b, best_c5

def sparse_cd(h, n, max_rounds, time_limit, threshold=0.05):
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    start = time.time()
    deltas = [0.03, 0.015, 0.01, 0.005, 0.002]
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        active = np.where((best_h > threshold) & (best_h < 1.0 - threshold))[0]
        if len(active) == 0:
            break
        improved = False
        for j in active:
            if time.time() - start > time_limit:
                break
            old = best_h[j]
            cands = [0.0, 1.0]
            for d in deltas:
                cands.append(max(0.0, min(1.0, old - d)))
                cands.append(max(0.0, min(1.0, old + d)))
            best_c = best_c5
            best_val = old
            for cand in cands:
                if abs(cand - old) < 1e-12:
                    continue
                trial = best_h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n)
                best_c5 = best_c
                improved = True
        if not improved:
            break
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except (NameError, Exception):
        h_init = np.full(960, 0.5)

    n_init = len(h_init)
    h_init = make_feasible(h_init, n_init)
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: SLSQP polish at native resolution
    if get_remaining(start_time, budget_s) > 60:
        h_new = do_slsqp(best_h, n_init, 50.0, 200)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new
    if get_remaining(start_time, budget_s) > 60:
        h_new = do_slsqp(best_h, n_init, 120.0, 150)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new

    # Phase 2: Binary simulated annealing (main exploration)
    if get_remaining(start_time, budget_s) > 130:
        b, c_b = binary_anneal(best_h, n_init, 100000, 0.005, 0.00002,
                               min(120, get_remaining(start_time, budget_s) - 30), seed + 7)
        if c_b < best_c5:
            best_c5 = c_b
            best_h = b.copy()

    # Phase 3: Second SA round with different seed
    if get_remaining(start_time, budget_s) > 80:
        b2, c_b2 = binary_anneal(best_h, n_init, 50000, 0.002, 0.00001,
                                 min(70, get_remaining(start_time, budget_s) - 20), seed + 42)
        if c_b2 < best_c5:
            best_c5 = c_b2
            best_h = b2.copy()

    # Phase 4: Coordinate descent at native resolution
    if get_remaining(start_time, budget_s) > 40:
        best_h, best_c5 = coord_descent(best_h, n_init, max_rounds=5,
                                        time_limit=min(60, get_remaining(start_time, budget_s) - 20))

    # Phase 5: Upsample to 2x and optimize
    n_2x = min(n_init * 2, 2560)
    if get_remaining(start_time, budget_s) > 120:
        h_2x = upsample_h(best_h, n_2x)
        h_new = do_slsqp(h_2x, n_2x, 60.0, 250)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new
        if get_remaining(start_time, budget_s) > 80:
            h_new = do_slsqp(best_h if len(best_h) == n_2x else h_2x, n_2x, 150.0, 150)
            c = compute_c5(h_new)
            if c < best_c5:
                best_c5 = c
                best_h = h_new
        if get_remaining(start_time, budget_s) > 50:
            h_cur = best_h if len(best_h) == n_2x else upsample_h(best_h, n_2x)
            best_h, best_c5 = sparse_cd(h_cur, n_2x, max_rounds=8,
                                         time_limit=min(50, get_remaining(start_time, budget_s) - 20))

    # Phase 6: Upsample to 3x and optimize
    n_3x = min(n_init * 3, 3840)
    if get_remaining(start_time, budget_s) > 150 and n_3x > n_2x:
        h_3x = upsample_h(best_h, n_3x)
        h_new = do_slsqp(h_3x, n_3x, 80.0, 200)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new
        if get_remaining(start_time, budget_s) > 60:
            h_cur = best_h if len(best_h) == n_3x else upsample_h(best_h, n_3x)
            best_h, best_c5 = sparse_cd(h_cur, n_3x, max_rounds=6,
                                         time_limit=min(50, get_remaining(start_time, budget_s) - 20))

    # Phase 7: Intensive coordinate descent at best resolution
    n_pts = len(best_h)
    if get_remaining(start_time, budget_s) > 50:
        best_h, best_c5 = coord_descent(best_h, n_pts, max_rounds=10,
                                        time_limit=min(150, get_remaining(start_time, budget_s) - 20))

    # Phase 8: Final SLSQP with high temperature (focus on max)
    n_pts = len(best_h)
    if get_remaining(start_time, budget_s) > 50:
        h_new = do_slsqp(best_h, n_pts, 400.0, 150)
        c = compute_c5(h_new)
        if c < best_c5:
            best_c5 = c
            best_h = h_new

    # Phase 9: Final sparse CD
    n_pts = len(best_h)
    if get_remaining(start_time, budget_s) > 20:
        h2, c2 = sparse_cd(best_h, n_pts, max_rounds=15,
                           time_limit=min(30, get_remaining(start_time, budget_s) - 5))
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2

    # Final: full CD pass with fine deltas
    n_pts = len(best_h)
    if get_remaining(start_time, budget_s) > 10:
        h3, c3 = coord_descent(best_h, n_pts, max_rounds=5,
                               time_limit=min(15, get_remaining(start_time, budget_s) - 2))
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
