            continue

        j_end = min(n-1, n-1-lag)

            idx = np.arange(j_start, j_end+1)
            g[idx] -= c * x[idx + lag]
    
    return obj, g

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def project_sum(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)

        if d > 0:

        h1_half = optimize_half_LBFGS(h1_half, n1, T, mu=50.0, maxiter=200, deadline=deadline)
        h1_full = to_full(h1_half, n1)
        h1_full = project_sum(h1_full, n1 / 2.0)
        c5 = evaluate_c5(h1_full)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h1_full.copy()
        if time.time() > deadline - 200:
            break

    # Also try SLSQP at n=336 for robustness
    if time.time() < deadline - 200:
        h1_half_s = optimize_half_slsqp(h1_half.copy(), n1, 100.0, maxiter=200, deadline=deadline)
        h1_s = to_full(h1_half_s, n1)
        h1_s = project_sum(h1_s, n1 / 2.0)
        c5 = evaluate_c5(h1_s)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h1_s.copy()

    # Phase 2: Upsample to n=672 and optimize
    n2 = 672
    if time.time() < deadline - 400:
        h2 = interp_to(best_h, len(best_h), n2)
        h2 = project_sum(h2, n2 / 2.0)
        h2_half = to_half(h2, n2)
        h2_half = project_sum_half(h2_half, n2 / 4.0)

        T_schedule_2 = [50, 100, 200, 400, 800]
        mu_schedule_2 = [30.0, 50.0, 100.0, 200.0, 500.0]
        for i, T in enumerate(T_schedule_2):
            if time.time() > deadline - 100:
                break
            mu = mu_schedule_2[i]
            h2_half = optimize_half_LBFGS(h2_half, n2, T, mu=mu, maxiter=150, deadline=deadline)
            h2_full = to_full(h2_half, n2)
            h2_full = project_sum(h2_full, n2 / 2.0)
            c5 = evaluate_c5(h2_full)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h2_full.copy()

    # Phase 3: Try n=1344 if time permits
    n3 = 1344
    if time.time() < deadline - 300:
        h3 = interp_to(best_h, len(best_h), n3)
        h3 = project_sum(h3, n3 / 2.0)
        h3_half = to_half(h3, n3)
        h3_half = project_sum_half(h3_half, n3 / 4.0)

        T_schedule_3 = [100, 200, 400]
        for T in T_schedule_3:
            if time.time() > deadline - 60:
                break
            h3_half = optimize_half_LBFGS(h3_half, n3, T, mu=100.0, maxiter=100, deadline=deadline)
            h3_full = to_full(h3_half, n3)
            h3_full = project_sum(h3_full, n3 / 2.0)
            c5 = evaluate_c5(h3_full)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h3_full.copy()

    # Phase 4: Random restarts at best resolution
    n_best = len(best_h)
    half_n_best = n_best // 2
    best_half = to_half(best_h, n_best)
    best_half = project_sum_half(best_half, n_best / 4.0)

    for attempt in range(15):
        if time.time() > deadline - 20:
            break
        scale = 0.04 + 0.04 * (attempt % 3)
        noise = np.random.randn(half_n_best) * scale
        h_pert = best_half + noise
        h_pert = project_sum_half(h_pert, n_best / 4.0)
        T_r = 150 + 100 * (attempt % 3)
        h_pert = optimize_half_LBFGS(h_pert, n_best, T_r, mu=100.0, maxiter=100, deadline=deadline)
        h_full = to_full(h_pert, n_best)
        h_full = project_sum(h_full, n_best / 2.0)
        c5 = evaluate_c5(h_full)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_full.copy()
            best_half = to_half(best_h, n_best)

    # Phase 5: Final fine-tuning with high T
    if time.time() < deadline - 30:
        best_half = to_half(best_h, n_best)
        best_half = project_sum_half(best_half, n_best / 4.0)
        for T in [500, 1000, 2000]:
            if time.time() > deadline - 10:
                break
            best_half = optimize_half_LBFGS(best_half, n_best, T, mu=500.0, maxiter=80, deadline=deadline)
            h_tf = to_full(best_half, n_best)
            h_tf = project_sum(h_tf, n_best / 2.0)
            c5 = evaluate_c5(h_tf)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_tf.copy()

    best_c5 = evaluate_c5(best_h)
    # Final feasibility check
    best_h = project_sum(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
