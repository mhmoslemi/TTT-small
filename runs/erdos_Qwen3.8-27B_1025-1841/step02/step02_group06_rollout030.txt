    corr = compute_corr(h)
    g = corr * dx
    num_corr = 2 * n - 1
    
    if topM > 0 and topM < num_corr:
        top_idx = np.argpartition(g, -topM)[-topM:]

    else:
        top_idx = np.arange(num_corr)
        g_top = g

import numpy as np
import time
from scipy.signal import fftconvolve


def project_to_feasible(g, n):
    target = n / 2.0
    lo = float(g.min()) - 2.0
    hi = float(g.max()) + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) * 0.5, 0.0, 1.0)


def compute_corr(h):
    n = len(h)
    return fftconvolve(h, (1.0 - h)[::-1], mode='full')


def compute_c5(h):
    n = len(h)
    return float(compute_corr(h).max()) * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_grad(h, n, top_idx, p, dx):
    grad = np.zeros(n)
    nm1 = n - 1
    for i in range(len(top_idx)):
        pk = p[i]
        if pk < 1e-12:
            continue
        d = int(top_idx[i]) - nm1
        if d >= 0:
            if n - d > 0:
                grad[:n - d] += pk * (1.0 - h[d:])
                grad[d:] -= pk * h[:n - d]
        else:
            e = -d
            if n - e > 0:
                grad[e:] += pk * (1.0 - h[:n - e])
                grad[:n - e] -= pk * h[e:]
    return grad * dx


def optimize_softmax(h, n, tau_list, max_iters, time_limit, lr_init, topM, mom=0.9):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    corr = compute_corr(h)
    g = corr * dx
    best_obj = float(g.max())
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    stall = 0
    patience = max(100, max(300, max_iters // 8))
    it = 0
    num_corr = 2 * n - 1

    for tau in tau_list:
        if time.time() - start > time_limit or it >= max_iters:
            break
        lr = lr_init
        while it < max_iters:
            if time.time() - start > time_limit:
                break

            corr = compute_corr(h)
            g = corr * dx
            cur_max = float(g.max())
            if cur_max < best_obj - 1e-14:
                best_obj = cur_max
                best_h = h.copy()
                stall = 0
            else:
                stall += 1

            if topM > 0 and topM < num_corr:
                top_idx = np.argpartition(g, -topM)[-topM:]
                g_sel = g[top_idx]
            else:
                top_idx = np.arange(num_corr)
                g_sel = g

            mg = float(g_sel.max())
            shifted = np.clip(tau * (g_sel - mg), -500, 500)
            exp_c = np.exp(shifted)
            s = float(exp_c.sum())
            p = exp_c / s

            grad = compute_grad(h, n, top_idx, p, dx)
            v = mom * v - lr * grad
            h = project_to_feasible(h + v, n)

            if stall > patience:
                lr *= 0.6
                v *= 0.3
                stall = 0
                if lr < 1e-12:
                    break
            it += 1

    return best_h, best_obj
