import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_upscale(h, target_n):
    n = len(h)
    if n >= target_n:
        return h.copy()
    x_old = np.linspace(0.0, 2.0, n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, target_n, endpoint=False)
    h_ext = np.concatenate([h, h[:1]])
    return np.interp(x_new, np.concatenate([x_old, [2.0]]), h_ext)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr


def grad_corr_smooth(h, p, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(p > 1e-12)[0]
    for k in active:
        pk = p[k]
        sh = k - (n - 1)
        i0 = max(0, -sh)
        i1 = min(n, n - sh)
        if i0 >= i1:
            continue
        grad[i0:i1] -= pk * h[i0 + sh:i1 + sh]
        j0 = i0 + sh
        j1 = i1 + sh
        grad[j0:j1] += pk * (1.0 - h[i0:i1])
    return grad * dx


def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500.0, 0.0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / tau)
    grad = grad_corr_smooth(h, p, n)
    return obj, grad


def optimize_smooth_adaptive(h, n, time_limit, tau_ini, tau_fin, lr0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    max_iters = 200000
    it = 0
    while time.time() - start < time_limit:
        it += 1
        frac = min(it / max_iters, 1.0)
        tau = tau_ini * (tau_fin / tau_ini) ** frac
        obj, grad = smooth_obj_grad(h, tau)
        m = b1 * m + (1.0 - b1) * grad
        v = b2 * v + (1.0 - b2) * (grad * grad)
        m_hat = m / (1.0 - b1 ** it)
        v_hat = v / (1.0 - b2 ** it)
        lr = lr0 * (1.0 - 0.5 * frac)
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h, n)
        if it % 200 == 0:
            c5_cur, _ = compute_c5(h)
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
    c5_cur, _ = compute_c5(h)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h.copy()
    return best_h, best_c5


def optimize_subgrad_topk(h, n, K, time_limit, lr0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    start = time.time()
    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_cur = corr.max() * dx
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        top_k = np.argsort(corr)[-K:]
        grad = np.zeros(n)
        for k in top_k:
            sh = k - (n - 1)
            i0 = max(0, -sh)
            i1 = min(n, n - sh)
            if i0 >= i1:
                continue
            grad[i0:i1] -= h[i0 + sh:i1 + sh]
            j0 = i0 + sh
            j1 = i1 + sh
            grad[j0:j1] += (1.0 - h[i0:i1])
        grad *= dx / K
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            h = h - lr0 * grad / gnorm
        h = project_to_feasible(h, n)
        lr0 *= 0.9995
    c5_cur, _ = compute_c5(h)
    if c5_cur < best_c5:
        best_c5 = c5_cur
        best_h = h.copy()
    return best_h, best_c5

    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    start = time.time()

            best_c5 = c5_cur
            best_h = h.copy()
        k_star = np.argmax(corr)
        sh = k_star - (n - 1)
        i0 = max(0, -sh)

        grad = np.zeros(n)

            j1 = i1 + sh
            grad[j0:j1] += (1.0 - h[i0:i1])
        grad *= dx
        gnorm = np.linalg.norm(grad)
        if gnorm > 1e-15:
            h = h - lr0 * grad / gnorm
        h = project_to_feasible(h, n)

    c5_cur, _ = compute_c5(h)

        best_c5 = c5_cur

def coordinate_polish(h, n, time_limit):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)[0]
    best_h = h.copy()
    start = time.time()
    order = np.random.permutation(n)
    while time.time() - start < time_limit:
        improved = False
        for i in order:
            if time.time() - start > time_limit:
                break
            orig = h[i]
            for trial in [0.0, 1.0, 0.5]:
                if trial == orig:
                    continue
                h[i] = trial
                c5_cur, _ = compute_c5(h)
                if c5_cur < best_c5 - 1e-12:
                    best_c5 = c5_cur
                    best_h = h.copy()
                    improved = True
                    break
                h[i] = orig
    return best_h, best_c5
