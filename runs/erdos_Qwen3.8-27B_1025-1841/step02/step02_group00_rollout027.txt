def frank_wolfe_step(h, k_star, n):
    """Given h and the active shift k_star, compute the FW vertex."""
    # k_star is the index into the full correlation array
    # corr index m corresponds to shift k = m - (n-1)
    k = k_star - (n - 1)
    
    # Compute gradient of corr_k with respect to h
    # corr_k = sum_{i: valid} h[i] * (1 - h[i+k])
    # d/dh[j] = [valid as first] * (1-h[j+k]) - [valid as second] * h[j-k]
    
    g = np.zeros(n)
    
    if k >= 0:
        # Valid first indices: 0 <= i < n-k
        # Valid second indices: 0 <= i-k < n => k <= i < n
        for j in range(n - k):
            g[j] += (1.0 - h[j + k])
        for j in range(k, n):
            g[j] -= h[j - k]
    else:
        k_abs = -k
        # Valid first indices: k_abs <= i < n
        # Valid second indices: 0 <= i < n-k_abs
        for j in range(k_abs, n):
            g[j] += (1.0 - h[j - k_abs])
        for j in range(n - k_abs):
            g[j] -= h[j + k_abs]
    
    # FW linear step: minimize g·s subject to 0<=s<=1, sum(s)=n/2
    # Optimal: s_i = 1 for the n/2 indices with smallest g_i
    n_ones = n // 2
    s = np.zeros(n)
    # Get indices sorted by g
    idx_sorted = np.argsort(g)
    s[idx_sorted[:n_ones]] = 1.0
    
    return s, g

    best_h = h.copy()

        corr = np.correlate(h, 1.0 - h, mode='full')
        k_star = np.argmax(corr)
        current_val = corr[k_star]
        
        s, g = frank_wolfe_step(h, k_star, n)
        
        # Step size
        if step_schedule == 'polyak' and upper_bound is not None:
            gap = upper_bound - current_val
            g_norm_sq = np.sum(g * g) + 1e-10

        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def compute_c5(h):

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0

        diff = target - np.sum(h)
        if abs(diff) < 1e-13:
            break
        if diff > 0:

        else:
            mask = h > 1e-14
        cnt = np.sum(mask)

        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)

                break
    return best_h, best_c5

def coord_descent(h, n, time_limit=60):
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for rnd in range(20):
        if time.time() - t0 > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - t0 > time_limit:
                break
            old = h[j]
            for delta in [0.02, -0.02, 0.01, -0.01, 0.005, -0.005]:
                c = old + delta
                if c < 0.0 or c > 1.0:
                    continue
                trial = h.copy()
                trial[j] = c
                trial = make_feasible(trial, n)
                val = compute_c5(trial)
                if val < best_c5 - 1e-13:
                    best_c5 = val
                    best_h = trial
                    h = trial
                    old = c
                    improved = True
            if improved:
                continue
            for bv in (0.0, 1.0):
                if abs(bv - h[j]) < 1e-9:
                    continue
                trial = h.copy()
                trial[j] = bv
                trial = make_feasible(trial, n)
                val = compute_c5(trial)
                if val < best_c5 - 1e-13:
                    best_c5 = val
                    best_h = trial
                    h = trial
                    improved = True
    return best_h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    global _OPT_N, _OPT_T
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def _tl():
        return budget_s - (time.time() - start_time)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None
    if h0 is None:
        n0 = 640
        h0 = np.tile([1.0, 0.0], n0 // 2)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: FW at moderate resolution
    n1 = 2048
    if _tl() > 120:
        h1 = upsample_h(best_h, n1)
        c5_1 = compute_c5(h1)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
        h1, c5_1 = fw_method(h1, n1, 800, _tl() - 100)
        if c5_1 < best_c5:
            best_c5 = c5_1
            best_h = h1.copy()
    
    # Phase 2: SLSQP polish at n1
    if _tl() > 80:
        h2 = slsqp_opt(best_h, n1, 50.0, 300)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    if _tl() > 60:
        h2 = slsqp_opt(best_h, n1, 200.0, 300)
        c5_2 = compute_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()
    
    # Phase 3: Upsample to higher resolution
    n2 = 4096
    if _tl() > 150:
        h3 = upsample_h(best_h, n2)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        
        # FW at higher res
        if _tl() > 120:
            h3, c5_3 = fw_method(h3, n2, 500, _tl() - 100)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
        
        # SLSQP at higher res
        if _tl() > 60:
            h3 = slsqp_opt(best_h, n2, 50.0, 250)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
        if _tl() > 50:
            h3 = slsqp_opt(best_h, n2, 200.0, 250)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
        if _tl() > 40:
            h3 = slsqp_opt(best_h, n2, 1000.0, 300)
            c5_3 = compute_c5(h3)
            if c5_3 < best_c5:
                best_c5 = c5_3
                best_h = h3.copy()
    
    # Phase 4: Active set equalization
    n_final = len(best_h)
    if _tl() > 30:
        h4, c5_4 = active_set_equality(best_h, n_final, 40, _tl() - 20)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()
    
    # Phase 5: More SLSQP with very high T
    if _tl() > 25:
        h5 = slsqp_opt(best_h, n_final, 5000.0, 400)
        c5_5 = compute_c5(h5)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5.copy()
    
    # Phase 6: Coordinate descent polish
    if _tl() > 15:
        n_cd = min(n_final, 2048)
        if n_cd < n_final:
            h_cd = upsample_h(best_h, n_cd)
        else:
            h_cd = best_h.copy()
        h_cd, c5_cd = coord_descent(h_cd, n_cd, _tl() - 5)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            if n_cd < n_final:
                best_h = upsample_h(h_cd, n_final)
            else:
                best_h = h_cd.copy()
    
    # Phase 7: Final SLSQP squeeze
    if _tl() > 15:
        h6 = slsqp_opt(best_h, len(best_h), 10000.0, 300)
        c5_6 = compute_c5(h6)
        if c5_6 < best_c5:
            best_c5 = c5_6
            best_h = h6.copy()
    
    # Final active set
    if _tl() > 10:
        h7, c5_7 = active_set_equality(best_h, len(best_h), 20, _tl() - 2)
        if c5_7 < best_c5:
            best_c5 = c5_7
            best_h = h7.copy()
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    
    return best_h, final_c5, len(best_h)
