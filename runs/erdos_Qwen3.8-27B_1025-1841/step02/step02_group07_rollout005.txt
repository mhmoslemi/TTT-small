import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_c5_and_grad(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    c5 = max_val * dx
    
    # Find active lags (within tolerance of max)
    tol = 1e-10
    active = np.where(corr >= max_val - tol)[0]
    
    # Compute gradient: average of gradients over active lags
    grad = np.zeros(n)
    for j in active:
        # For output index j:
        # ∂out[j]/∂h[p] = (1-h[p-j+n-1]) * [0<=p-j+n-1<n] - h[p+j-n+1] * [0<=p+j+n-1<n]
        # Let a = -j+n-1, b = j-n+1
        a = -j + n - 1  # offset for first term
        b = j - n + 1   # offset for second term
        # For first term: (1-h[p+a]) where 0<=p+a<n
        # p ranges from max(0,-a) to min(n-1, n-1-a)
        p_start = max(0, -a)
        p_end = min(n, n - a)  # exclusive
        if p_start < p_end:
            idx = np.arange(p_start, p_end)
            grad[idx] += (1.0 - h[idx + a])
        # For second term: -h[p+b] where 0<=p+b<n
        p_start2 = max(0, -b)
        p_end2 = min(n, n - b)
        if p_start2 < p_end2:
            idx2 = np.arange(p_start2, p_end2)
            grad[idx2] -= h[idx2 + b]
    
    grad /= len(active)
    grad *= dx
    return c5, grad

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def subgradient_method(h0, n, deadline, max_iters=5000, tol=1e-12):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Adaptive learning rate
    lr = 0.01
    lr_min = 1e-7
    fail_count = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        c5, grad = compute_c5_and_grad(h)
        
        # Backtracking line search
        accepted = False
        for trial in range(50):
            trial_lr = lr * (0.5 ** trial)
            h_new = project(h - trial_lr * grad, n)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5 - 1e-15:
                h = h_new
                accepted = True
                break
        if not accepted:
            fail_count += 1
            if fail_count > 20:
                # Add random perturbation
                h = project(h + np.random.randn(n) * 0.01, n)
                fail_count = 0
            lr = max(lr * 0.9, lr_min)
        else:
            fail_count = 0
            # Try to increase learning rate
            if lr < 0.1:
                lr = min(lr * 1.05, 0.2)
        
        c5_cur = evaluate_c5(h)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
    
    return best_h, best_c5

        if p2s < p2e:

            grad[idx2] -= w * h[idx2 + b]
    
    grad *= dx
    return grad

                if abs(trial - h[i]) < 1e-6:
                    continue
                h[i] = trial
                h = project_to_feasible(h, n)
                obj = evaluate_c5(h)
                if obj < best_obj:
                    best_obj = obj
                    best_val = trial
            h[i] = best_val
            h = project_to_feasible(h, n)
            if best_obj < best_c5:
                best_c5 = best_obj
                improved = True
        if not improved:
            break
    return h, best_c5

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: Nesterov subgradient at parent resolution
    n1 = len(h_start)
    h1, c5_1 = nesterov_subgrad(h_start, deadline - 500, T0=3.0, T_min=0.02, T_rate=0.996, lr0=0.04)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1.copy()

    # Phase 2: SLSQP polish at parent resolution
    if time.time() < deadline - 300:
        h2 = slsqp_refine(best_h, n1, deadline - 250, maxiter=400)
        c5_2 = evaluate_c5(h2)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2.copy()

    # Phase 3: Coordinate descent polish
    if time.time() < deadline - 100:
        h3, c5_3 = coord_descend_polish(best_h, n1, deadline - 80, n_passes=4)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()

    # Phase 4: Try at 2x resolution
    if time.time() < deadline - 150:
        n2 = n1 * 2
        h4 = interp_to(best_h, n1, n2)
        h4 = project_to_feasible(h4, n2)
        h4, c5_4 = nesterov_subgrad(h4, deadline - 80, T0=1.5, T_min=0.02, T_rate=0.997, lr0=0.03)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4.copy()

    # Phase 5: Multiple random restarts at best resolution
    n_best = len(best_h)
    for restart in range(30):
        if time.time() > deadline - 20:
            break
        scale = 0.04 * (1.0 + 0.5 * np.sin(restart * 0.7))
        h_pert = project_to_feasible(best_h + np.random.randn(n_best) * scale, n_best)
        h_r, c5_r = nesterov_subgrad(h_pert, deadline - 10, T0=0.5, T_min=0.01, T_rate=0.998, lr0=0.02)
        if c5_r < best_c5:
            best_c5 = c5_r
            best_h = h_r.copy()

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        h_final, c5_final = coord_descend_polish(best_h, len(best_h), deadline - 2, n_passes=2)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
