import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def to_binary(h, n):
    idx = np.argsort(-h)
    hb = np.zeros(n)
    hb[idx[:n // 2]] = 1.0
    return hb

def swap_local_search(h, n, rng, time_limit, perturb_every=200):
    start_t = time.time()
    h = h.astype(float).copy()
    cur_val = np.max(np.correlate(h, 1.0 - h, mode='full'))
    best_val = cur_val
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    it = 0
    while time.time() - start_t < time_limit:
        found = False
        for trial in range(40):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            new_val = np.max(np.correlate(h, 1.0 - h, mode='full'))
            if new_val < cur_val - 1e-15:
                cur_val = new_val
                best_val = min(best_val, new_val)
                best_h = h.copy()
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                found = True
                stall = 0
                break
            else:
                h[i] = 1.0
                h[j] = 0.0
        if not found:
            stall += 1
            if stall >= perturb_every:
                for _ in range(2 + rng.randint(3)):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                cur_val = np.max(np.correlate(h, 1.0 - h, mode='full'))
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0
                if cur_val > best_val * 1.02:
                    h = best_h.copy()
                    cur_val = best_val
                    ones = np.where(h == 1)[0].copy()
                    zeros = np.where(h == 0)[0].copy()
        it += 1
    return best_h, best_val * (2.0 / n)

def targeted_swap_search(h, n, rng, time_limit, top_k=3):
    """Focus on reducing the worst shifts specifically."""
    start_t = time.time()
    h = h.astype(float).copy()
    cur_val = np.max(np.correlate(h, 1.0 - h, mode='full'))
    best_val = cur_val
    best_h = h.copy()
    ones = np.where(h == 1)[0].copy()
    zeros = np.where(h == 0)[0].copy()
    stall = 0
    while time.time() - start_t < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        top_idx = np.argsort(-corr)[:top_k]
        found = False
        for trial in range(80):
            # Pick a worst shift
            k = top_idx[rng.randint(top_k)]
            s = k - (n - 1)
            # Find valid (i, i+s) pairs where h[i]=1, h[i+s]=0
            if s >= 0:
                i_lo, i_hi = 0, n - s
            else:
                i_lo, i_hi = -s, n
            candidates_i = []
            for i in range(i_lo, i_hi):
                j = i + s
                if h[i] == 1.0 and h[j] == 0.0:
                    candidates_i.append((i, j))
            if len(candidates_i) == 0:
                continue
            i, j = candidates_i[rng.randint(len(candidates_i))]
            # Swap h[i]=1 with some random zero (not j)
            zeros_no_j = zeros[np.zeros != j] if len(zeros) > 1 else zeros
            if len(zeros_no_j) == 0:
                continue
            p = zeros_no_j[rng.randint(len(zeros_no_j))]
            h[i] = 0.0
            h[p] = 1.0
            new_val = np.max(np.correlate(h, 1.0 - h, mode='full'))
            if new_val < cur_val - 1e-15:
                cur_val = new_val
                best_val = min(best_val, new_val)
                best_h = h.copy()
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                found = True
                stall = 0
                break
            else:
                h[i] = 1.0
                h[p] = 0.0
        if not found:
            stall += 1
            if stall > 300:
                for _ in range(3):
                    i = ones[rng.randint(len(ones))]
                    j = zeros[rng.randint(len(zeros))]
                    h[i] = 0.0
                    h[j] = 1.0
                cur_val = np.max(np.correlate(h, 1.0 - h, mode='full'))
                ones = np.where(h == 1)[0].copy()
                zeros = np.where(h == 0)[0].copy()
                stall = 0
                if cur_val > best_val * 1.03:
                    h = best_h.copy()
                    cur_val = best_val
                    ones = np.where(h == 1)[0].copy()
                    zeros = np.where(h == 0)[0].copy()
    return best_h, best_val * (2.0 / n)

def slsqp_polish(h, n, time_limit):
    """SLSQP with soft max to allow non-binary improvements."""
    dx = 2.0 / n
    start_t = time.time()
    t_param = [60.0]
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        soft = mc + np.log(np.sum(np.exp(t_param[0] * shifted))) / t_param[0]
        return soft * dx

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for temp in [60.0, 150.0, 300.0]:
        if time.time() - start_t > time_limit:
            break
        t_param[0] = temp
        try:
            res = minimize(obj, best_h, method='SLSQP', bounds=bounds,
                           constraints=cons, options={'maxiter': 200, 'ftol': 1e-12})
            c5 = compute_c5(res.x)
            if c5 < best_c5:
                best_c5 = c5
                best_h = res.x.copy()
        except Exception:
            pass
    return best_h, best_c5

def coord_descent_polish(h, n, time_limit):
    start_t = time.time()
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    improved = True
    while improved and time.time() - start_t < time_limit:
        improved = False
        for j in range(n):
            if time.time() - start_t > time_limit:
                break
            old_val = h[j]
            best_trial = old_val
            best_val = best_c5
            for cand in [0.0, 1.0, 0.5, old_val + 0.05, old_val - 0.05,
                         old_val + 0.02, old_val - 0.02,
                         old_val + 0.01, old_val - 0.01]:
                if cand < 0 or cand > 1:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_val - 1e-12:
                    best_val = c
                    best_trial = cand
            if best_trial != old_val:
                h = h.copy()
                h[j] = best_trial
                h = make_feasible(h, n)
                best_c5 = best_val
                improved = True
    return h, best_c5
