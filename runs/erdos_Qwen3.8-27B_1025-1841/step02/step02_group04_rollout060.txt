def softgrad(h, T, top_k=10):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    # Use log-sum-exp for stability
    max_shifted = np.max(shifted)
    if max_shifted > 1e10:  # overflow protection
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        exp_s = np.exp(T * shifted)
        w = exp_s / np.sum(exp_s)
    # Zero out small weights for speed
    idx = np.where(w > 1e-8)[0]
    if len(idx) > top_k:
        idx = np.argsort(w[idx])[-top_k:] + np.where(w > 1e-8)[0]

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softgrad(h, T, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)

    exp_s = np.exp(np.clip(T * shifted, -700, 700))

        w[np.argmax(corr)] = 1.0

    return best_h, best_c5

                grad[s:] += wi * (-h[:n-s])
            elif s == 0:
                grad[:] += wi * (1.0 - h[:])
        else:
            s = -shift
            if s > 0 and s < n:
                grad[s:] += wi * (1.0 - h[:n-s])
                grad[:n-s] += wi * (-h[s:])
    return grad * dx

def gd_nesterov(h, n_iter, T_start, T_end, lr, top_k, beta=0.9, check_every=50):
    n = len(h)
    h = h.copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        h_look = h - beta * v
        g = softgrad(h_look, T, top_k=top_k)
        v = beta * v - lr * g
        h_new = h + v
        h_new = make_feasible(h_new, n)
        h = h_new
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)

def swap_search(h, n_trials, rng, T_anneal=0.01):
    n = len(h)
    h = h.copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

        T = T_anneal * max(0.01, 1.0 - i / n_trials)
        i1, i2 = rng.randint(0, n, 2)
        if i1 == i2:
            continue
        old1, old2 = h[i1], h[i2]
        h[i1], h[i2] = old2, old1
        c = compute_c5(h)
        delta = c - best_c5
        if delta < 0 or rng.random_sample() < np.exp(-delta / max(T, 1e-10)):
            if c < best_c5:
                best_c5 = c
                best_h = h.copy()
        else:
            h[i1], h[i2] = old1, old2
    return best_h, best_c5

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softgrad(h, T, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(np.clip(T * shifted, -700, 700))
    total = np.sum(exp_s)
    if total < 1e-300:
        w = np.zeros(len(corr))
        w[np.argmax(corr)] = 1.0
    else:
        w = exp_s / total
    top_idx = np.argsort(w)[-top_k:][::-1]
    grad = np.zeros(n)
    for k_idx in top_idx:
        wi = w[k_idx]
        if wi < 1e-12:
            continue
        shift = k_idx - (n - 1)
        if shift >= 0:
            s = shift
            if s < n:
                if s > 0:
                    grad[:n-s] += wi * (1.0 - h[s:])
                    grad[s:] += wi * (-h[:n-s])
                else:
                    grad[:] += wi * (1.0 - 2.0 * h[:])

            if time.time() < deadline:
                hx, cx = slsqp_optimize(best_h, 5.0, new_n, maxiter=600)
                if cx < best_c5:
                    best_c5, best_h = cx, hx
        deadline = start_time + budget_s - 30
        if time.time() < deadline:
            h5b, c5b = swap_search(best_h, 30000, rng, T_anneal=0.003, deadline=deadline)
            if c5b < best_c5:
                best_c5, best_h = c5b, h5b
        print(f"P5 Upsample(n={new_n}): C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 6: Multi-start perturbations
    n_pts = len(best_h)
    remaining = budget_s - (time.time() - start_time)
    if remaining > 100:
        for pert in [0.02, 0.05, 0.1]:
            deadline = start_time + budget_s - 30
            if time.time() > deadline:
                break
            h_p = best_h + rng.randn(n_pts) * pert
            h_p = make_feasible(h_p, n_pts)
            hp, cp = gd_nesterov(h_p, 2500, 15.0, 0.3, 0.05, top_k=3, beta=0.85, check_every=40)
            if cp < best_c5:
                best_c5, best_h = cp, hp
                print(f"  Perturb {pert}: C5={best_c5:.6f}", flush=True)
        print(f"P6 MultiStart: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 7: Final fine GD + swap
    remaining = budget_s - (time.time() - start_time)
    if remaining > 20:
        n_pts = len(best_h)
        h7, c7 = gd_nesterov(best_h, 3000, 1.5, 0.01, 0.01, top_k=2, beta=0.95, check_every=20)
        if c7 < best_c5:
            best_c5, best_h = c7, h7
        deadline = start_time + budget_s - 3
        if time.time() < deadline:
            h7b, c7b = swap_search(best_h, 30000, rng, T_anneal=0.001, deadline=deadline)
            if c7b < best_c5:
                best_c5, best_h = c7b, h7b
        print(f"P7 Final: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
