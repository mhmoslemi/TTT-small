```python
def soft_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(temp * shifted)
    p = exp_s / exp_s.sum()  # length 2n-1
    
    # k ranges from -(n-1) to n-1, p[k+(n-1)] is the weight for shift k
    s_vals = np.arange(-(n-1), n)  # actual shifts k
    J = np.arange(n)  # j indices 0 to n-1
    
    # First term: sum_k p[k] * (1-h[j-k]) for 0<=j-k<n
    # = sum_k p[k+(n-1)] * (1-h[j-k]) where 0<=j-k<n
    idx_f = J[:, None] - s_vals[None, :]  # j-k, shape (n, 2n-1)
    mask_f = (idx_f >= 0) & (idx_f < n)
    term1 = np.where(mask_f, 1.0 - h[np.clip(idx_f, 0, n-1)], 0.0)
    term1 = (p[None, :] * term1).sum(axis=1)
    
    # Second term: sum_k p[k] * h[k+j] for 0<=k+j<n
    idx_s = s_vals[None, :] + J[:, None]  # k+j, shape (n, 2n-1)
    mask_s = (idx_s >= 0) & (idx_s < n)
    term2 = np.where(mask_s, h[np.clip(idx_s, 0, n-1)], 0.0)
    term2 = (p[None, :] * term2).sum(axis=1)
    
    return dx * (term1 - term2)
```

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    for _ in range(20):
        diff = target - h.sum()
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return make_feasible(np.interp(new_x, old_x, h), new_n)

def soft_obj(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(temp * shifted))) / temp) * dx

def soft_grad(h, n, temp):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    exp_s = np.exp(temp * shifted)
    p = exp_s / exp_s.sum()
    s_vals = np.arange(-(n - 1), n).astype(np.int64)
    J = np.arange(n, dtype=np.int64)
    idx_f = J[:, None] - s_vals[None, :]
    mask_f = (idx_f >= 0) & (idx_f < n)
    t1_vals = np.where(mask_f, 1.0 - h[np.clip(idx_f, 0, n - 1)], 0.0)
    term1 = (p[None, :] * t1_vals).sum(axis=1)
    idx_s = s_vals[None, :] + J[:, None]
    mask_s = (idx_s >= 0) & (idx_s < n)
    t2_vals = np.where(mask_s, h[np.clip(idx_s, 0, n - 1)], 0.0)
    term2 = (p[None, :] * t2_vals).sum(axis=1)
    return dx * (term1 - term2)

def obj_wrap(h, n, temp):
    return soft_obj(h, n, temp)

def grad_wrap(h, n, temp):
    return soft_grad(h, n, temp)

def project_feasible(h, n):
    h = h.copy()
    for _ in range(60):
        h = np.clip(h, 0.0, 1.0)
        diff = n / 2.0 - h.sum()
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            m = h < 1.0 - 1e-14
        else:
            m = h > 1e-14
        cnt = m.sum()
        if cnt == 0:
            break
        h[m] += diff / cnt
    return np.clip(h, 0.0, 1.0)

def projected_gd(h, n, temp, n_iter, lr, elapsed_fn, budget):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    cur_val = soft_obj(h, n, temp)
    for it in range(n_iter):
        if elapsed_fn() > budget - 60:
            break
        grad = soft_grad(h, n, temp)
        step = lr
        for bt in range(30):
            trial = project_feasible(h - step * grad, n)
            v_trial = soft_obj(trial, n, temp)
            if v_trial < cur_val - 1e-15:
                break
            step *= 0.5
        else:
            break
        h = trial
        cur_val = v_trial
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        lr = min(max(step * 1.2, 1e-7), 0.1)
    return best_h, best_c5

def random_binary_h(n, rng):
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:n // 2]] = 1.0
    return h

def perturb_binary(h, n, rng, n_flip=10):
    h2 = h.copy()
    for _ in range(n_flip):
        i = rng.randint(n)
        h2[i] = 1.0 - h2[i]
    return make_feasible(h2, n)

def perturb_smooth(h, n, rng, sigma=3.0):
    noise = np.zeros(n)
    freqs = np.random.RandomState(rng.randint(10**9)).randint(1, 5)
    for f in freqs:
        amp = rng.uniform(0.02, 0.08)
        phase = rng.uniform(0, 2 * np.pi)
        noise += amp * np.sin(2 * np.pi * f * np.arange(n) / n + phase)
    h2 = h + noise
    return make_feasible(h2, n)

def perturb_block(h, n, rng):
    h2 = h.copy()
    start = rng.randint(0, n - 10)
    length = rng.randint(5, n // 4)
    frac = rng.uniform(0.3, 0.7)
    h2[start:start + length] = frac
    return make_feasible(h2, n)

    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    rng = np.random.RandomState(seed)

        if n_opt not in h_at_res:

        else:

            pass

    # Coordinate descent polish
    if elapsed() < budget_s - 20:
        n_pts = len(best_h)
        improved = True
        for polish_round in range(8):
            if not improved or elapsed() > budget_s - 20:
                break
            improved = False
            for j in range(n_pts):
                if elapsed() > budget_s - 20:

    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
