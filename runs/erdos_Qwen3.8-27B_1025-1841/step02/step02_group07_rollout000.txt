        else:

from scipy.optimize import minimize
import time

    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)

            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_and_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = corr * T
    m = np.max(z)
    exp_z = np.exp(z - m)
    ssum = np.sum(exp_z)
    p = exp_z / ssum
    obj = (m + np.log(ssum)) / T * dx
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        pk = p[k]
        if pk < 1e-15:
            continue
        shift = k - (n - 1)
        if shift >= 0:
            grad[shift:] += pk * dx * (1.0 - h[:n-shift])
            grad[:n-shift] -= pk * dx * h[shift:]
        else:
            s = -shift
            grad[:n-s] += pk * dx * (1.0 - h[s:])
            grad[s:] -= pk * dx * h[:n-s]
    return obj, grad

def slsqp_analytic(h0, n, T, deadline):
    target = n / 2.0
    def obj(x):
        return softmax_obj_and_grad(x, T)[0]
    def grad(x):
        return softmax_obj_and_grad(x, T)[1]
    def eq_con(x):
        return np.sum(x) - target
    def eq_jac(x):
        return np.ones(n)
    try:
        res = minimize(
            fun=obj, x0=h0, jac=grad, method='SLSQP',
            bounds=[(0.0, 1.0)] * n,
            constraints=[{'type': 'eq', 'fun': eq_con, 'jac': eq_jac}],
            options={'ftol': 1e-14, 'maxiter': 300}
        )
        if time.time() < deadline and res.x is not None:
            h_res = project(res.x, n)
            return h_res
    except Exception:
        pass
    return project(h0, n)

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    z = corr * T
    m = np.max(z)
    ez = np.exp(z - m)
    ss = np.sum(ez)
    p = ez / ss
    obj = (m + np.log(ss)) / T * dx
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - (n - 1)
        if sh >= 0:
            grad[sh:] += pk * dx * (1.0 - h[:n - sh])
            grad[:n - sh] -= pk * dx * h[sh:]
        else:
            s = -sh
            grad[:n - s] += pk * dx * (1.0 - h[s:])
            grad[s:] -= pk * dx * h[:n - s]
    return obj, grad

def slsqp_solve(h0, n, T, deadline):
    def f(x):
        return softmax_obj_grad(x, T)[0]
    def g(x):
        return softmax_obj_grad(x, T)[1]
    def ceq(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': ceq, 'jac': cjac}],
                       options={'ftol': 1e-15, 'maxiter': 500})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return project(h0, n)

def proj_grad_solve(h0, n, T, deadline):

    best_h = h.copy()

    lr = 0.3

        gp = grad - np.mean(grad)
        alpha = lr

            if evaluate_c5(h_try) <= obj_cur + 1e-13:

            alpha *= 0.5
        else:
            lr *= 0.5
            stall += 1

        h = h_try
        c5 = evaluate_c5(h)
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        lr = min(lr * 1.02, 3.0)
    return best_h, best_c5

    xf = np.linspace(0, 2, nf, endpoint=False)

def slsqp_solve(h0, n, T, deadline):

        return softmax_obj_grad(x, T)[0]
    def g(x):
        return softmax_obj_grad(x, T)[1]
    def ceq(x):
        return np.sum(x) - n / 2.0
    def cjac(x):
        return np.ones(n)
    try:
        res = minimize(f, h0, jac=g, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': ceq, 'jac': cjac}],
                       options={'ftol': 1e-15, 'maxiter': 600})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return project(h0, n)

def proj_grad_solve(h0, n, T, deadline):
    h = project(h0, n)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = 0.3
    stall = 0
    for step in range(30000):
        if time.time() > deadline:
            break
        _, grad = softmax_obj_grad(h, T)
        gp = grad - np.mean(grad)
        gnorm = np.linalg.norm(gp)
        if gnorm < 1e-12:
            break
        alpha = lr
        obj_cur = evaluate_c5(h)
        accepted = False
        for _ in range(25):
            h_try = project(h - alpha * gp, n)
            if evaluate_c5(h_try) <= obj_cur + 1e-13:
                h = h_try
                accepted = True
                break
            alpha *= 0.5
        if not accepted:
            lr *= 0.5
            stall += 1
            if stall > 100:
                break
            continue
        stall = 0
        lr = min(lr * 1.03, 5.0)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5

def interp_h(h, nf, nt):
    if nf == nt:
        return h.copy()
    xf = np.linspace(0, 2, nf, endpoint=False)
    xt = np.linspace(0, 2, nt, endpoint=False)
    return np.interp(xt, xf, h)

def make_structured(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5
    for freq in [1, 2, 3]:
        h += rng.uniform(-0.15, 0.15) * np.sin(np.pi * freq * x)
    h = project(h, n)
    return h

def make_binary(n, seed):
    rng = np.random.RandomState(seed)
    n_ones = n // 2
    idx = rng.choice(n, n_ones, replace=False)
    h = np.zeros(n)
    h[idx] = 1.0
    return h

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 25

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    n0 = len(h_start)
    best_h = project(h_start, n0)
    best_c5 = evaluate_c5(best_h)

    # Phase 1: SLSQP at original resolution with increasing T
    for T in [60.0, 120.0, 250.0]:
        if time.time() > deadline - 120:
            break
        h_new = slsqp_solve(best_h, n0, T, deadline)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    # Phase 2: Projected gradient for fine-tuning
    t_gd_start = time.time()
    h_gd, c5_gd = proj_grad_solve(best_h, n0, 150.0, min(deadline - 120, t_gd_start + 300))
    if c5_gd < best_c5:
        best_c5 = c5_gd
        best_h = h_gd

    # Phase 3: Higher resolution
    n_hi = 2 * n0
    if time.time() < deadline - 100:
        h_hi = project(interp_h(best_h, n0, n_hi), n_hi)
        for T in [100.0, 200.0]:
            if time.time() > deadline - 50:
                break
            h_new = slsqp_solve(h_hi, n_hi, T, deadline)
            c5 = evaluate_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new
        if time.time() < deadline - 40:
            h_gd2, c5_gd2 = proj_grad_solve(best_h, len(best_h), 200.0, min(deadline - 10, time.time() + 120))
            if c5_gd2 < best_c5:
                best_c5 = c5_gd2
                best_h = h_gd2

    # Phase 4: Diverse restarts at best resolution
    nf = len(best_h)
    restart_pool = []
    # Gaussian perturbations
    for i in range(20):
        h_p = best_h + np.random.randn(nf) * 0.05
        restart_pool.append(project(h_p, nf))
    # Structured starts
    for s in [seed + 7, seed + 13, seed + 42]:
        restart_pool.append(make_structured(nf, s))
    # Binary starts (smoothed)
    for s in [seed + 99, seed + 200]:
        hb = make_binary(nf, s).astype(float)
        # Smooth the binary function
        kernel = np.array([1, 2, 3, 2, 1], dtype=float)
        kernel /= kernel.sum()
        h_smooth = np.convolve(hb, kernel, mode='same')
        restart_pool.append(project(h_smooth, nf))

    for h_p in restart_pool:
        if time.time() > deadline - 15:
            break
        h_new = slsqp_solve(h_p, nf, 150.0, deadline)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
            # Refine the new best with one more SLSQP call
            if time.time() < deadline - 10:
                h_new2 = slsqp_solve(best_h, nf, 300.0, deadline)
                c52 = evaluate_c5(h_new2)
                if c52 < best_c5:
                    best_c5 = c52
                    best_h = h_new2

    # Phase 5: Final polish with high T
    if time.time() < deadline - 10:
        h_new = slsqp_solve(best_h, nf, 400.0, deadline)
        c5 = evaluate_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new
    if time.time() < deadline - 5:
        h_new, c5 = proj_grad_solve(best_h, nf, 400.0, deadline)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new

    best_h = project(best_h, len(best_h))
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
