def project_to_feasible(g, n):

    hi = g.max() + 3.0

        if s > target:
            lo = mid
        else:
            hi = mid

    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        pk = p[k]
        sh = k - n + 1
        if sh >= 0:
            grad[sh:] += pk * (1.0 - h[:n-sh])
            if sh > 0:
                grad[:n-sh] -= pk * h[sh:]
        else:
            s2 = -sh
            grad[:n-s2] += pk * (1.0 - h[s2:])
            grad[s2:] -= pk * h[:n-s2]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad

def optimize_nesterov(h, tau, time_limit, lr_init=0.5, max_iters=100000):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    obj0, grad0 = smooth_grad(h, tau)
    best_obj = obj0
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    beta = 0.9
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        obj, grad = smooth_grad(h, tau)
        if obj < best_obj - 1e-14:
            best_obj = obj
            best_h = h.copy()

        obj_y, grad_y = smooth_grad(h_y, tau)
        
        v_new = beta * v - lr * grad_y
        h_new = project_to_feasible(h + v_new, n)
        
        # Simple backtracking: if obj didn't decrease, reduce lr
        obj_new, _ = smooth_grad(h_new, tau)
        if obj_new > obj + 1e-10:
            lr *= 0.7
            h_new = h
        else:
            if obj_new < obj * 0.999:  # good decrease
                if lr < 2.0:
                    lr *= 1.05
            v = v_new
            h = h_new
    
    return best_h, best_obj

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 3.0
    hi = g.max() + 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def smooth_grad_full(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    active = np.where(p > 1e-10)[0]
    for k in active:
        pk = p[k]
        sh = k - n + 1
        if sh >= 0:
            grad[sh:] += pk * (1.0 - h[:n - sh])
            if sh > 0:
                grad[:n - sh] -= pk * h[sh:]
        else:
            s2 = -sh
            grad[:n - s2] += pk * (1.0 - h[s2:])
            grad[s2:] -= pk * h[:n - s2]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def topk_subgrad(h, top_k=5):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    idx = np.argsort(corr)[-top_k:]
    max_c = corr.max()
    w = np.zeros(2 * n - 1)
    w[idx] = 1.0 / top_k
    grad = np.zeros(n)
    for k in idx:
        pk = w[k]
        sh = k - n + 1
        if sh >= 0:
            grad[sh:] += pk * (1.0 - h[:n - sh])
            if sh > 0:
                grad[:n - sh] -= pk * h[sh:]
        else:
            s2 = -sh
            grad[:n - s2] += pk * (1.0 - h[s2:])
            grad[s2:] -= pk * h[:n - s2]
    grad *= dx
    c5 = max_c * dx
    return c5, grad


def optimize_smooth_phase(h, tau, time_limit, lr_init=0.4, max_iters=500000):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    obj0, _ = smooth_grad_full(h, tau)
    best_obj = obj0
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    beta = 0.9
    stall = 0
    best_stall = 0
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_grad_full(h, tau)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 60 and stall - best_stall > 60:
            lr *= 0.5
            v *= 0.3
            best_stall = stall
            stall = 0
            if lr < 1e-10:
                break

        h = h_new

    return best_h, best_obj


def optimize_topk_phase(h, time_limit, lr_init=0.005, top_k=5, max_iters=500000):
    n = len(h)
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    c5_0, _ = topk_subgrad(h, top_k)
    best_c5 = c5_0
    best_h = h.copy()
    v = np.zeros(n)
    lr = lr_init
    beta = 0.85
    stall = 0
    best_stall = 0
    start = time.time()

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        c5_cur, grad = topk_subgrad(h, top_k)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        v = beta * v - lr * grad
        h_new = project_to_feasible(h + v, n)

        if stall > 100 and stall - best_stall > 100:
            lr *= 0.5
            v *= 0.3
            best_stall = stall
            stall = 0
            if lr < 1e-14:
                break

        h = h_new

    return best_h, best_c5
