    return h

    h = 0.5 * (h + h[::-1].copy())

def smooth_max_val(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def smooth_max_grad_full(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    wsum = np.sum(weights)
    weights /= wsum
    grad = np.zeros(n)
    # s=0: result[n-1] = sum h[i]*(1-h[i]), grad = 1-2*h[i]
    grad += weights[n-1] * (1.0 - 2.0 * h)
    # s>0: result[n-1+s] = sum_{i=0}^{n-1-s} h[i+s]*(1-h[i])
    for s in range(1, n):
        w = weights[n-1+s]
        if w < 1e-15:
            continue
        # d/dh[j]: (1-h[j-s]) for j in [s, n-1]
        grad[s:] += w * (1.0 - h[:n-s])
        # d/dh[j]: -h[j+s] for j in [0, n-1-s]
        grad[:n-s] += -w * h[s:]
    # s<0: result[n-1-s] = sum_{i=s}^{n-1} h[i-s]*(1-h[i])
    for s in range(1, n):

                c5_p = evaluate_c5(pert)
                if c5_p < best_c5:
                    best_c5 = c5_p
                    best_h = pert.copy()

    # Phase 5: Fine-tune with very high T (sharp max)
    if time.time() < deadline - 15:
        h_final = project_feasible(best_h, n)
        h_sym_final = project_symmetric(best_h, n)
        half_final = project_half(h_sym_final[:n_half], n)
        half_final = nesterov_sym(half_final, n, 500.0, max_steps=2000, deadline=deadline - 5)
        full_final = expand_half(half_final)
        c5_final = evaluate_c5(full_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = full_final.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
