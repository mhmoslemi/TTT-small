def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h_init = np.full(960, 0.5)

    n0 = len(h_init)
    h_init = make_feasible(h_init, n0)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: SLSQP at original resolution with multiple T values
    n_pts = n0
    bounds = [(0.0, 1.0)] * n_pts

    for T in [20, 50, 100, 300]:
        if remaining() < 80:
            break
        ns = n_pts
        Tval = T

        def obj_phase1(x):
            return soft_max_obj(x, ns, Tval)

        cons = [{'type': 'eq', 'fun': obj_phase1, 'jac': None}]
        # Use proper constraint
        def con_fun(x):
            return np.sum(x) - ns / 2.0
        def con_jac_fn(x):
            return np.ones(ns)

        cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac_fn}]
        maxiter = 500 if T <= 50 else 350
        try:
            res = minimize(obj_phase1, best_h, method='SLSQP', bounds=bounds,
                          constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 2: Try upsampled resolution
    if remaining() > 250:
        n_hi = n0 * 2
        h_hi = upsample_h(best_h, n_hi)
        bounds_hi = [(0.0, 1.0)] * n_hi
        for T in [50, 150]:
            if remaining() < 80:
                break
            nsh = n_hi
            Th = T

            def obj_hi(x):
                return soft_max_obj(x, nsh, Th)

            def con_fun_hi(x):
                return np.sum(x) - nsh / 2.0
            def con_jac_hi(x):
                return np.ones(nsh)

            cons_hi = [{'type': 'eq', 'fun': con_fun_hi, 'jac': con_jac_hi}]
            try:
                res = minimize(obj_hi, h_hi, method='SLSQP', bounds=bounds_hi,
                              constraints=cons_hi, options={'maxiter': 200, 'ftol': 1e-12})
                c5_res = compute_c5(res.x)
                if c5_res < best_c5:
                    best_c5 = c5_res
                    best_h = make_feasible(res.x, n_hi)
            except Exception:
                pass

    # Phase 3: Aggressive local search
    n_pts = len(best_h)
    for rnd in range(200):
        if remaining() < 8:
            break
        improved = False

        # Single-point optimization with more candidates
        for j in range(n_pts):
            if remaining() < 4:
                break
            old = best_h[j]
            best_c = best_c5
            best_val = old
            cands = [0.0, 1.0, 0.5]
            if old > 0.1:
                cands.append(max(0.0, old - 0.1))
            if old < 0.9:
                cands.append(min(1.0, old + 0.1))
            if abs(old - round(old)) > 0.05:
                cands.append(round(old))
            for cand in cands:
                if cand == old:
                    continue
                c, _ = try_point_change(best_h, j, cand, n_pts, best_c)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != old:
                best_h[j] = best_val
                best_h = make_feasible(best_h, n_pts)
                best_c5 = best_c
                improved = True

        # Block flips (length 2, 3, 4)
        if remaining() > 10:
            for blen in [2, 3, 4]:
                step = max(1, 2 - (blen - 2))
                for j in range(0, n_pts - blen + 1, step):
                    if remaining() < 4:
                        break
                    c, trial = try_block_flip(best_h, j, blen, n_pts, best_c5)
                    if c < best_c5 - 1e-14:
                        best_h = trial
                        best_c5 = c
                        improved = True

        # Block shifts
        if remaining() > 15 and not improved:
            n_shifts = n_pts // 4
            for _ in range(n_shifts):
                if remaining() < 4:
                    break
                blen = rng.randint(2, 6)
                start = rng.randint(0, max(1, n_pts - blen - 1))
                end = start + blen - 1
                direction = rng.choice([-1, 1])
                c, trial = try_block_shift(best_h, start, end, direction, n_pts, best_c5)
                if c < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c
                    improved = True

        # Two-swap
        if remaining() > 15:
            n_swaps = n_pts // 5
            for _ in range(n_swaps):
                if remaining() < 4:
                    break
                j = rng.randint(0, n_pts)
                k = rng.randint(0, n_pts)
                if j == k:
                    continue
                c, trial = try_two_swap(best_h, j, k, n_pts, best_c5)
                if c < best_c5 - 1e-14:
                    best_h = trial
                    best_c5 = c
                    improved = True

        if not improved and rng.rand() > 0.3:
            # Try a small random perturbation to escape local minima
            n_pert = rng.randint(3, 10)
            idxs = rng.choice(n_pts, size=min(n_pert, n_pts), replace=False)
            trial = best_h.copy()
            for idx in idxs:
                trial[idx] = rng.rand()
            trial = make_feasible(trial, n_pts)
            c = compute_c5(trial)
            # Accept with probability (simulated annealing)
            if c < best_c5:
                best_h = trial
                best_c5 = c
                improved = True

        if not improved:
            # One more aggressive pass
            if remaining() > 20:
                for j in range(n_pts):
                    if remaining() < 4:
                        break
                    for cand in [0.0, 1.0]:
                        if abs(best_h[j] - cand) < 0.01:
                            continue
                        c, trial = try_point_change(best_h, j, cand, n_pts, best_c5)
                        if c < best_c5 - 1e-14:
                            best_h = trial
                            best_c5 = c
                            break
                if best_c5 < compute_c5(best_h) + 1e-14:
                    break
            else:
                break

    # Phase 4: Final SLSQP polish
    if remaining() > 50:
        n_pts = len(best_h)
        bounds_fin = [(0.0, 1.0)] * n_pts
        T_fin = 500
        nsf = n_pts

        def obj_fin(x):
            return soft_max_obj(x, nsf, T_fin)

        def con_fun_fin(x):
            return np.sum(x) - nsf / 2.0
        def con_jac_fin(x):
            return np.ones(nsf)

        cons_fin = [{'type': 'eq', 'fun': con_fun_fin, 'jac': con_jac_fin}]
        try:
            res = minimize(obj_fin, best_h, method='SLSQP', bounds=bounds_fin,
                          constraints=cons_fin, options={'maxiter': 200, 'ftol': 1e-14})
            c5_res = compute_c5(res.x)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = make_feasible(res.x, n_pts)
        except Exception:
            pass

    # Phase 5: Final tight coordinate descent
    if remaining() > 10:
        n_pts = len(best_h)
        for rnd in range(50):
            if remaining() < 4:
                break
            improved = False
            for j in range(n_pts):
                if remaining() < 2:
                    break
                old = best_h[j]
                best_c = best_c5
                best_val = old
                for cand in [0.0, 1.0, max(0.0, old - 0.02), min(1.0, old + 0.02)]:
                    if abs(cand - old) < 1e-6:
                        continue
                    c, _ = try_point_change(best_h, j, cand, n_pts, best_c)
                    if c < best_c - 1e-15:
                        best_c = c
                        best_val = cand
                if best_val != old:
                    best_h[j] = best_val
                    best_h = make_feasible(best_h, n_pts)
                    best_c5 = best_c
                    improved = True
            if not improved:
                break

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
