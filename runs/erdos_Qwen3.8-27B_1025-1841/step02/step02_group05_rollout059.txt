def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0

        s = np.sum(np.clip(g - mid, 0.0, 1.0))

        else:

import cvxpy as cp

def solve_linearized(h_current, n, n_shifts_top=20):
    """Solve the linearized convex subproblem."""
    dx = 2.0 / n
    corr = np.correlate(h_current, 1.0 - h_current, mode='full')
    
    # Get top shifts
    idx = np.argsort(corr)[::-1][:n_shifts_top]
    
    h_var = cp.Variable(n)
    t = cp.Variable()
    
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) * dx == 1.0]
    
    for k in idx:
        sh = k - n + 1
        # Linearize f_k around h_current
        # f_k(h) = sum_i h[i] * (1 - h[i+sh])
        # ∇f_k at h_current:
        grad_k = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad_k[j0:j1] += (1.0 - h_current[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad_k[j0b:j1b] -= h_current[j0b + sh:j1b + sh]
        
        f_k_val = corr[k] * dx  # current value
        # Linear approximation: f_k(h) ≈ f_k_val + dx * grad_k · (h - h_current)
        # Simplify: f_k_val + dx * grad_k · h - dx * grad_k · h_current
        # Note: f_k_val = dx * sum h_current[i] * (1-h_current[i+sh])
        #       dx * grad_k · h_current = dx * sum grad_k[i] * h_current[i]
        # These should be related...
        
        # Actually let's just use: f_k(h) ≈ f_k_val * dx + dx * grad_k @ (h - h_current)
        # Wait, corr[k] is already the sum (not yet multiplied by dx)
        # f_k = corr[k] * dx
        # ∂f_k/∂h_i = dx * grad_k[i]
        # Linearization: f_k ≈ corr[k]*dx + dx * grad_k @ (h_var - h_current)
        
        constraints.append(corr[k] * dx + dx * (grad_k @ (h_var - h_current)) <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS)  # or use another solver
    
    return h_var.value, t.value

    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
