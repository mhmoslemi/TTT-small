from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_full(h):
    """Project h onto {0<=h<=1, sum(h)=n/2}."""
    n = len(h)
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0).copy()
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def symmetrize(h):
    """Enforce h[i] = h[n-1-i]."""
    return (h + h[::-1].copy()) / 2.0

def to_sym_vars(h_full):
    """Extract symmetric variables (first half)."""
    n = len(h_full)
    half = n // 2
    return h_full[:half].copy()

def from_sym_vars(h_sym, n):
    """Reconstruct full h from symmetric variables."""
    half = n // 2
    h_full = np.empty(n)
    h_full[:half] = h_sym
    h_full[half:] = h_sym[::-1]
    return h_full

def grad_fk_full(h, k):
    """Gradient of f_k(h) = sum_{valid s} h[s]*(1-h[s+k]) w.r.t. h."""
    n = len(h)
    grad = np.zeros(n)
    if 0 <= k < n:
        grad[:n-k] += (1.0 - h[k:])
        if k > 0:
            grad[k:] -= h[:n-k]
    elif -n < k <= 0:
        kk = -k
        grad[kk:] += (1.0 - h[:n-kk])
        if kk > 0:
            grad[:n-kk] -= h[kk:]
    return grad

def softmax_obj_full(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    val = m + np.log(np.sum(np.exp(scaled - m)))
    return val / T * dx

def slsqp_sym(h_sym, n, T, maxiter=300, deadline=None):
    """Optimize symmetric variables with SLSQP."""
    half = n // 2
    target_sym = half / 2.0  # sum of half should be n/4 = half/2
    
    def f(x):
        hf = from_sym_vars(x, n)
        return softmax_obj_full(hf, T)
    
    def c_eq(x):
        return np.sum(x) - target_sym
    
    try:
        res = minimize(f, h_sym, method='SLSQP',
                       bounds=[(0.0, 1.0)] * half,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-12, 'maxiter': maxiter, 'eps': 1e-8})
        if deadline is not None and time.time() > deadline:
            return h_sym
        if res.x is not None and np.all(np.isfinite(res.x)):
            return np.clip(res.x, 0.0, 1.0)
    except Exception:
        pass
    return h_sym

def subgrad_sym(h_sym, n, lr, n_iters, deadline):
    """Subgradient descent on symmetric variables."""
    half = n // 2
    target_sym = half / 2.0
    dx = 2.0 / n
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        hf = from_sym_vars(h_sym, n)
        corr = np.correlate(hf, 1.0 - hf, mode='full')
        k_star = np.argmax(corr)
        k = k_star - (n - 1)
        grad_full = grad_fk_full(hf, k)
        # Project gradient to symmetric subspace
        grad_sym = grad_full[:half].copy()
        # The constraint is sum(h_sym) = target_sym
        # Use projected gradient
        mean_grad = np.mean(grad_sym)
        grad_sym = grad_sym - mean_grad  # project to sum-zero direction

        # Project back to feasible set
        h_sym_new = np.clip(h_sym_new, 0.0, 1.0)

        else:

                h_sym_new[mask] += d / mask.sum()
        h_sym_new = np.clip(h_sym_new, 0.0, 1.0)

        for i in range(n_polish):
            if time.time() > deadline - 8:
                break
            # Targeted perturbation: identify worst shift and perturb toward reducing it
            hf_cur = np.concatenate([best_hs, best_hs[::-1]])
            g, k_worst = grad_max_shift(hf_cur)
            # Perturb in anti-gradient direction for worst shift + random
            scale = 0.008 * (1.0 + 0.5 * np.random.rand())
            h_try = best_hs - scale * 0.3 * g[:half] + np.random.randn(half) * scale * 0.5
            h_try = np.clip(h_try, 0.0, 1.0)
            h_try = fix_sym_sum(h_try, tgt)
            h_try = slsqp_sym(h_try, n, 200.0, maxiter=80)
            hf_try = np.concatenate([h_try, h_try[::-1]])
            c5_try = evaluate_c5(hf_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_hs = h_try.copy()

    # Phase 3: Try higher resolution if time permits
    if time.time() < deadline - 120:
        n_high = n * 2
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_high, endpoint=False)
        hf_best = np.concatenate([best_hs, best_hs[::-1]])
        hf_high = np.interp(x_new, x_old, hf_best)
        hf_high = symmetrize(hf_high)
        hf_high = project_full(hf_high)
        hs_high = hf_high[:n_high // 2].copy()
        hs_high = fix_sym_sum(hs_high, n_high / 4.0)

        hs_high = slsqp_sym(hs_high, n_high, 100.0, maxiter=300)
        hs_high = slsqp_sym(hs_high, n_high, 300.0, maxiter=300)
        hf_high = np.concatenate([hs_high, hs_high[::-1]])
        c5_high = evaluate_c5(hf_high)
        if c5_high < best_c5:
            best_c5 = c5_high
            best_hs = hs_high
            n = n_high
            half = n // 2

    # Final evaluation
    best_hf = np.concatenate([best_hs, best_hs[::-1]])
    best_hf = project_full(best_hf)
    final_c5 = evaluate_c5(best_hf)

    return (best_hf, final_c5, len(best_hf))
