    xt = np.linspace(0, 2, n_to, endpoint=False)

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_sum(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            return h
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_resample(h, n_to):
    n_from = len(h)
    if n_from == n_to:
        return project_sum(h, n_to)
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    h_new = np.interp(xt, xf, h)
    return project_sum(h_new, n_to)

def smooth_max_val_grad(h, T):
    """Returns (smooth_max_value, gradient) using log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    scaled = T * c
    mx = np.max(scaled)
    exps = np.exp(scaled - mx)
    sum_exps = np.sum(exps)
    val = (mx + np.log(sum_exps)) / T
    p = exps / sum_exps
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-16:
            continue
        offset = k - n + 1
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1+offset:hi1+offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2-offset:hi2-offset]
    grad *= dx
    return val, grad

def topk_max_val_grad(h, k_top):
    """Returns (max of top-k correlations, gradient) for exact-ish max."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    # Get top k indices
    idx = np.argsort(c)[-k_top:]
    val = c[idx].max()
    # Uniform weight on top-k for gradient
    w = c[idx]
    w = np.maximum(w - w.min(), 0) + 1e-10
    w /= w.sum()
    p = np.zeros(2*n-1)
    for i, k in enumerate(idx):
        p[k] = w[i]
    
    grad = np.zeros(n)
    for k in idx:
        pk = p[k]
        if pk < 1e-16:

        offset = k - n + 1
        lo1 = max(0, -offset)
        hi1 = min(n, n - offset)
        if lo1 < hi1:
            grad[lo1:hi1] += pk * (1.0 - h[lo1+offset:hi1+offset])
        lo2 = max(0, offset)
        hi2 = min(n, n + offset)
        if lo2 < hi2:
            grad[lo2:hi2] -= pk * h[lo2-offset:hi2-offset]
    grad *= dx
    return val, grad

def lbbfgs_optimize(h0, T, maxiter, lam=500.0):
    n = len(h0)
    target = n / 2.0

        s = np.sum(h)
        diff = s - target
        val += lam * diff * diff
        grad += 2.0 * lam * diff

    except Exception:

    return h0.copy()

    h = h0.copy()

        c5_new = evaluate_c5(h_new)

            best_local = c5_new
            h = h_new
        else:

def exact_max_step(h, n_top=1, lr=0.002, n_steps=100, deadline=None):

    n = len(h)
    h_best = h.copy()

            if lr_cur < 1e-6:
