def soft_max(h, T):

    n = len(h)

    m = np.max(scaled)

```python
def subgradient_step(h, n, alpha):
    """One subgradient step for min max_k corr[k]*dx."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_val = np.max(corr)
    # Active set: lags within some tolerance of the max
    threshold = max_val - 1.0  # within 1 in corr units (arbitrary)
    active = np.where(corr >= threshold)[0]
    
    # Compute gradient: weighted average of active lag gradients
    grad = np.zeros(n)
    for m in active:
        lag = m - (n - 1)
        if lag == 0:
            grad += (1 - 2*h)
        else:
            # (1-h[j+lag]) for valid j
            if lag > 0:
                # j+lag < n means j < n-lag
                grad[:n-lag] += (1 - h[lag:])
                # -h[j-lag] for valid j: j-lag >= 0 means j >= lag
                grad[lag:] -= h[:n-lag]
            else:
                lag_abs = -lag
                # j+lag < n: always true since lag < 0, j+lag >= 0 means j >= -lag = lag_abs
                grad[lag_abs:] += (1 - h[:n-lag_abs])
                # j-lag < n: j+lag_abs < n, always true; j-lag >= 0: j >= -lag_abs... wait
                # j - lag = j + lag_abs, need 0 <= j+lag_abs < n and 0 <= j < n
                # Since lag_abs > 0, j+lag_abs < n means j < n-lag_abs
                grad[:n-lag_abs] -= h[lag_abs:]
    
    grad /= len(active)
    h_new = h - alpha * grad * dx
    h_new = project(h_new, n)
    return h_new, max_val * dx
```

from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(h, n, T):
    """Analytical gradient of the smoothed max correlation."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    exps = np.exp(scaled - m)
    weights = exps / np.sum(exps)  # softmax weights over lags
    
    # grad = dx * sum_k weights[k] * d(corr[k])/d(h)
    # d(corr[m])/d(h[j]) where corr = correlate(h, 1-h, full)
    # lag = m - (n-1)
    # For lag=0: d/dh[j] = (1-2h[j])
    # For lag>0: d/dh[j] = (1-h[j+lag]) for j < n-lag, plus -h[j-lag] for j >= lag
    # For lag<0: similar with |lag|
    
    grad = np.zeros(n)
    for m_idx in range(2*n - 1):
        w = weights[m_idx]
        if w < 1e-10:
            continue
        lag = m_idx - (n - 1)
        if lag == 0:
            grad += w * (1 - 2*h)
        elif lag > 0:
            grad[:n-lag] += w * (1 - h[lag:])
            grad[lag:] -= w * h[:n-lag]
        else:
            al = -lag
            grad[al:] += w * (1 - h[:n-al])
            grad[:n-al] -= w * h[al:]
    
    return grad * dx

    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def g(x):
        return soft_max_grad(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, method='SLSQP',
                       jac=g,
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': ftol, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def slsqp_opt_nojac(h0, n, T, maxiter=300, ftol=1e-14):

        return np.sum(x) - target

        if res.x is not None and np.all(np.isfinite(res.x)):

    except Exception:
        pass
    return h0

def subgradient_method(h0, n, n_steps, deadline):
    """Subgradient method for min max_k corr[k]*dx."""
    h = h0.copy()
    dx = 2.0 / n
    best_h = h.copy()
    best_val = evaluate_c5(h)
    
    for step in range(n_steps):

            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_val = np.max(corr)
        
        # Active set: lags within 1.0 (in corr units) of max
        threshold = max_val - 2.0
        active = np.where(corr >= threshold)[0]

        grad = np.zeros(n)

            else:

                grad[al:] += (1 - h[:n-al])
                grad[:n-al] -= h[al:]
        grad /= max(1, len(active))

        alpha = 0.5 / np.sqrt(step + 1)

        h_new = project(h_new, n)
        
        val = evaluate_c5(h_new)
        if val < best_val:
            best_val = val
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_val

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = h.sum()
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = mask.sum()
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = scaled.max()
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def soft_max_grad(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = scaled.max()
    exps = np.exp(scaled - m)
    w = exps / exps.sum()
    
    grad = np.zeros(n)
    n_active = np.where(w > 1e-12)[0]
    for mi in n_active:
        lag = mi - (n - 1)
        wi = w[mi]
        if lag == 0:
            grad += wi * (1 - 2*h)
        elif lag > 0:
            grad[:n-lag] += wi * (1 - h[lag:])
            grad[lag:] -= wi * h[:n-lag]
        else:
            al = -lag
            grad[al:] += wi * (1 - h[:n-al])
            grad[:n-al] -= wi * h[al:]
    return grad * dx

def slsqp_solve(h0, n, T, maxiter=400, ftol=1e-14, use_jac=True):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    kwargs = {
        'bounds': [(0.0, 1.0)] * n,
        'constraints': [{'type': 'eq', 'fun': c_eq}],
        'options': {'ftol': ftol, 'maxiter': maxiter}
    }
    if use_jac:
        def g(x):
            return soft_max_grad(x, n, T)
        kwargs['jac'] = g
    try:
        res = minimize(f, h0, method='SLSQP', **kwargs)
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def lbfgs_solve(h0, n, T, penalty=100.0, maxiter=1000):
    target = n / 2.0
    def f(x):
        hc = np.clip(x, 0, 1)
        return soft_max_obj(hc, n, T) + penalty * (np.sum(hc) - target)**2
    def g(x):
        hc = np.clip(x, 0, 1)
        grad = soft_max_grad(hc, n, T)
        grad += 2 * penalty * (np.sum(hc) - target)
        return grad
    try:
        res = minimize(f, h0, method='L-BFGS-B', jac=g,
                       bounds=[(0.0, 1.0)] * n,
                       options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0
