    for pert in [0.05, 0.12, 0.03]:
        if remaining() < 60:
            break
        h_p = best_h + _rng_global.randn(n_pts) * pert
        h_p = make_feasible(h_p, n_pts)
        h_r, c_r = adam_gd(h_p, 2000, T_start=20.0, T_end=0.3, lr=0.03, check_every=50)
        if c_r < best_c5:
            best_c5, best_h = c_r, h_r.copy()
    print(f"MultiStart: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 6: If we have room, try n=640
    if remaining() > 180:
        n_hi = 640
        h_hi = upsample(best_h, n_hi)
        h_h, c_h = adam_gd(h_hi, 3000, T_start=30.0, T_end=0.3, lr=0.03, check_every=50)
        if c_h < compute_c5(h_hi):
            h_hi = h_h
        if remaining() > 60:
            hx, cx = slsqp_optimize(h_hi, n_hi, 5.0, maxiter=500)
            if cx < compute_c5(h_hi):
                h_hi = hx
        c_hi = compute_c5(h_hi)
        if c_hi < best_c5:
            best_c5, best_h = c_hi, h_hi.copy()
        print(f"HiRes n={n_hi}: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Phase 7: Final fine-grained GD
    if remaining() > 40:
        n_pts = len(best_h)
        h_f, c_f = adam_gd(best_h, 3000, T_start=3.0, T_end=0.05, lr=0.015, check_every=20)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
        # One more SLSQP
        if remaining() > 30 and len(best_h) <= 512:
            hx, cx = slsqp_optimize(best_h, len(best_h), 1.5, maxiter=400)
            if cx < best_c5:
                best_c5, best_h = cx, hx
    print(f"Final: C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)

    # Final validation
    n_final = len(best_h)
    best_h = make_feasible(best_h, n_final)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE: n={n_final}, C5={best_c5:.6f}, t={elapsed():.0f}s", flush=True)
    return best_h, best_c5, n_final
