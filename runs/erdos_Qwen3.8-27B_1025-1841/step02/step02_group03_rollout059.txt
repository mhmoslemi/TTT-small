def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def make_binary(h, n):
    idx = np.argsort(-h)
    b = np.zeros(n)
    b[idx[:n//2]] = 1.0
    return b

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def slqp_step(h, n, T, maxiter):
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx
    def con(x):
        return np.sum(x) - n / 2.0
    def con_jac(x):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': con, 'jac': con_jac}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-13})
        h_res = make_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except:
        return h, compute_c5(h)

def binary_sa(h, n, n_iters, rng, t_start=0.01, t_end=0.0003):
    b = make_binary(h, n)
    cur = b.copy()
    cur_c = compute_c5(cur)
    best = b.copy()
    best_c = cur_c
    dx = 2.0 / n
    T = t_start
    cool = (t_end / t_start) ** (1.0 / n_iters)
    for _ in range(n_iters):
        ones = np.nonzero(cur)[0]
        zeros = np.nonzero(1 - cur)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = cur.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        corr = np.correlate(trial, 1.0 - trial, mode='full')
        new_c = float(np.max(corr) * dx)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = trial
            cur_c = new_c
            if new_c < best_c:
                best = trial.copy()
                best_c = new_c
        T *= cool
    return best, best_c

def binary_greedy(h, n, n_iters, rng):
    b = make_binary(h, n)
    cur_c = compute_c5(b)
    dx = 2.0 / n
    for _ in range(n_iters):
        ones = np.nonzero(b)[0]
        zeros = np.nonzero(1 - b)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = b.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        corr = np.correlate(trial, 1.0 - trial, mode='full')
        new_c = float(np.max(corr) * dx)
        if new_c < cur_c - 1e-15:
            b = trial
            cur_c = new_c
    return b, cur_c

def coord_descent(h, n, n_rounds, check_time=None):
    best_c = compute_c5(h)
    for r in range(n_rounds):
        if check_time and check_time() < 10:
            break
        improved = False
        for j in range(n):
            if check_time and check_time() < 5:
                break
            cands = [0.0, 1.0, 0.5,
                     max(0.0, h[j]-0.1), max(0.0, h[j]-0.03),
                     min(1.0, h[j]+0.1), min(1.0, h[j]+0.03)]
            best_cand = h[j]
            for cand in cands:
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_cand = cand
            if best_cand != h[j]:
                h = h.copy()
                h[j] = best_cand
                h = make_feasible(h, n)
                improved = True
        if not improved:
            break
    return h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    rng = np.random.RandomState(seed)
    
    def rem():
        return budget_s - (time.time() - t0)
    
    # Get initial
    try:
        h_init = np.array(initial_h_values, dtype=float).copy()
    except NameError:
        h_init = np.full(960, 0.5)
    n0 = len(h_init)
    h_cur = make_feasible(h_init, n0)
    best_h = h_cur.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: SLSQP at n=480 (coarse)
    if rem() > 400:
        n_c = 480
        h_c = upsample_h(best_h, n_c)
        h_c, c_c = slqp_step(h_c, n_c, T=40.0, maxiter=300)
        if c_c < best_c5:
            best_c5 = c_c
            best_h = make_feasible(h_c, n_c)
    
    # Phase 2: SLSQP at n=960
    if rem() > 300:
        n_s = n0 if n0 >= 480 else 960
        if len(best_h) != n_s:
            best_h = upsample_h(best_h, n_s)
        best_h, best_c5 = slqp_step(best_h, n_s, T=80.0, maxiter=400)
    
    # Phase 3: Binary SA (main search)
    n_sa = len(best_h)
    # Estimate iterations: ~0.8ms per iter for n=960
    n_iters = int(min(600000, rem() * 1100))
    n_iters = max(n_iters, 10000)
    h_bin, c_bin = binary_sa(best_h, n_sa, n_iters, rng,
                              t_start=0.01, t_end=0.0003)
    if c_bin < best_c5:
        best_c5 = c_bin
        best_h = h_bin
    
    # Phase 4: Greedy binary local search

        n_g = int(min(300000, rem() * 1100))
        h_g, c_g = binary_greedy(best_h, n_sa, n_g, rng)
        if c_g < best_c5:
            best_c5 = c_g
            best_h = h_g

    if rem() > 30:
        best_h, best_c5 = coord_descent(best_h, n_sa, n_rounds=3, check_time=rem)
    
    # Phase 6: SLSQP polish
    if rem() > 60:
        n_p = len(best_h)
        best_h, c_p = slqp_step(best_h, n_p, T=150.0, maxiter=200)
        if c_p < best_c5:

    # Phase 7: Final greedy binary search

    best_h = make_feasible(best_h, len(best_h))

    cons = [{'type': 'eq', 'fun': _slsqp_con, 'jac': _slsqp_con_jac}]
    try:
        res = minimize(_slsqp_obj, h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        h_res = make_feasible(res.x, n)
        return h_res, compute_c5(h_res)
    except Exception:
        return h, compute_c5(h)

def binary_sa(h, n, n_iters, rng, t_start=0.01, t_end=0.0002):
    b = make_binary(h, n)
    cur = b.copy()
    cur_c = compute_c5(cur)
    best = b.copy()
    best_c = cur_c
    dx = 2.0 / n
    T = t_start
    cool = (t_end / t_start) ** (1.0 / max(1, n_iters))
    for _ in range(n_iters):
        ones = np.nonzero(cur)[0]
        zeros = np.nonzero(1 - cur)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = cur.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        corr = np.correlate(trial, 1.0 - trial, mode='full')
        new_c = float(np.max(corr) * dx)
        delta = new_c - cur_c
        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-12)):
            cur = trial
            cur_c = new_c
            if new_c < best_c:
                best = trial.copy()
                best_c = new_c
        T *= cool
    return best, best_c

def binary_greedy(h, n, n_iters, rng):
    b = make_binary(h, n)
    cur_c = compute_c5(b)
    dx = 2.0 / n
    for _ in range(n_iters):
        ones = np.nonzero(b)[0]
        zeros = np.nonzero(1 - b)[0]
        i = ones[rng.randint(len(ones))]
        j = zeros[rng.randint(len(zeros))]
        trial = b.copy()
        trial[i] = 0.0
        trial[j] = 1.0
        corr = np.correlate(trial, 1.0 - trial, mode='full')
        new_c = float(np.max(corr) * dx)
        if new_c < cur_c - 1e-15:
            b = trial
            cur_c = new_c
    return b, cur_c

def coord_descent(h, n, n_rounds, check_time):
    best_c = compute_c5(h)
    for r in range(n_rounds):
        if check_time() < 15:
            break
        improved = False
        for j in range(n):
            if check_time() < 8:
                break
            cands = [0.0, 1.0, 0.5,
                     max(0.0, h[j] - 0.1), max(0.0, h[j] - 0.03),
                     min(1.0, h[j] + 0.1), min(1.0, h[j] + 0.03)]
            best_cand = h[j]
            for cand in cands:
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c - 1e-14:
                    best_c = c
                    best_cand = cand
            if best_cand != h[j]:
                h = h.copy()
                h[j] = best_cand
                h = make_feasible(h, n)
                improved = True
        if not improved:
            break
    return h, best_c
