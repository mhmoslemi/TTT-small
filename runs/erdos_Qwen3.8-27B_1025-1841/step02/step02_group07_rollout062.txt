import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_grad(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_vals = np.exp(T * (corr - m))
    w = exp_vals / np.sum(exp_vals)
    
    j_arr = np.arange(n)[:, None]
    q_arr = np.arange(n)[None, :]
    
    idx1 = j_arr + n - 1 - q_arr  # (n,n)
    term1 = w[idx1] @ (1.0 - h)
    
    idx2 = q_arr - j_arr + n - 1  # (n,n)
    term2 = w[idx2] @ h
    
    return dx * (term1 - term2)

def project_h(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def enforce_symmetry(h, n):
    h = (h + h[::-1]) / 2.0
    return h

def projected_gradient(h, n, T, lr, n_steps, deadline, enforce_sym=True):
    h = h.copy()
    best_local = h.copy()
    best_local_c5 = evaluate_c5(h)
    
    for step in range(n_steps):
        if time.time() > deadline:
            break
        
        grad = compute_grad(h, T)
        if enforce_sym:
            grad = (grad + grad[::-1]) / 2.0
        
        # Backtracking line search
        alpha = lr
        f_curr = evaluate_c5(h)
        for _ in range(20):
            h_try = h - alpha * grad
            h_try = project_h(h_try, n)
            if enforce_sym:
                h_try = enforce_symmetry(h_try, n)
                h_try = project_h(h_try, n)
            f_try = evaluate_c5(h_try)
            if f_try < f_curr:
                h = h_try
                f_curr = f_try
                break
            alpha *= 0.5
        
        c5 = evaluate_c5(h)
        if c5 < best_local_c5:
            best_local_c5 = c5
            best_local = h.copy()
    
    return best_local, best_local_c5

            grad = symmetrize(grad)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        alpha = lr
        f_curr = evaluate_c5(h)
        improved = False
        for _ in range(15):
            h_try = h - alpha * grad
            h_try = project_h(h_try, n)
            if sym:
                h_try = symmetrize(h_try)
                h_try = project_h(h_try, n)
            f_try = evaluate_c5(h_try)
            if f_try < f_curr - 1e-15:
                h = h_try
                f_curr = f_try
                improved = True
                break
            alpha *= 0.5
        if improved:
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best = h.copy()
            lr = min(lr * 1.05, 2.0 / max(gnorm, 1e-10))
        else:
            lr *= 0.9
    return best, best_c5

def coord_descent(h0, n, deadline, sym=True, max_iter=200):
    h = h0.copy()
    best = h.copy()
    best_c5 = evaluate_c5(h)
    for it in range(max_iter):
        if time.time() > deadline:
            break
        changed = False
        order = np.random.permutation(n)
        for j in order:
            if time.time() > deadline:
                break
            lo, hi = 0.0, 1.0
            for _ in range(40):
                mid = (lo + hi) / 2.0
                h1 = h.copy()
                h1[j] = mid
                h2 = h.copy()
                h2[j] = (lo + hi) / 2.0
                # Use three-point comparison
                f_mid = evaluate_c5(h1)
                f_lo = evaluate_c5(np.clip(h.copy(), 0, 1))
                # Simple bisection: we want to minimize f(h[j])
                h_a = h.copy()
                h_a[j] = (lo + mid) / 2.0
                h_b = h.copy()
                h_b[j] = (mid + hi) / 2.0
                fa = evaluate_c5(h_a)
                fb = evaluate_c5(h_b)
                if fa < fb:
                    hi = mid
                else:
                    lo = mid
            h[j] = (lo + hi) / 2.0
            h = project_h(h, n)
            if sym:
                h = symmetrize(h)
                h = project_h(h, n)
            changed = True
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best = h.copy()
    return best, best_c5

def bang_bang_refine(h0, n, deadline):
    h = (h0 > 0.5).astype(float)
    h = project_h(h, n)
    best = h.copy()
    best_c5 = evaluate_c5(h)
    idx1 = np.where(h == 1.0)[0]
    idx0 = np.where(h == 0.0)[0]
    improved = True
    while improved and time.time() < deadline:
        improved = False
        for a in idx1.copy():
            if time.time() > deadline + 5:
                break
            for b in idx0.copy():
                h_try = h.copy()
                h_try[a] = 0.0
                h_try[b] = 1.0
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h = h_try
                    best = h.copy()
                    best_c5 = c5_try
                    idx1 = np.where(h == 1.0)[0]
                    idx0 = np.where(h == 0.0)[0]
                    improved = True
                    break
            if time.time() > deadline + 5:
                break
    return best, best_c5

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    n = len(h_start)
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)

    # Symmetrize initial
    h_work = symmetrize(h_start)
    h_work = project_h(h_work, n)
    c5_sym = evaluate_c5(h_work)
    if c5_sym < best_c5:
        best_c5 = c5_sym
        best_h = h_work.copy()

    # Phase 1: Projected gradient with annealing T
    T_schedule = [30.0, 80.0, 200.0, 500.0, 1500.0]
    for T in T_schedule:
        if time.time() > deadline - 60:
            break
        h_new, c5_new = projected_gradient_opt(h_work, n, T, 0.5, 3000, deadline, sym=True)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_work = h_new

    # Phase 2: Without symmetry (try both)
    if time.time() < deadline - 120:
        h_nosym = best_h.copy()
        h_new, c5_new = projected_gradient_opt(h_nosym, n, 200.0, 0.3, 3000, deadline, sym=False)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_work = h_new
        else:
            h_work = best_h.copy()

    # Phase 3: Random perturbations + re-optimize
    n_perturb = 12
    for i in range(n_perturb):
        if time.time() > deadline - 30:
            break
        scale = 0.05 + 0.1 * (i % 3)
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = project_h(h_pert, n)
        h_pert = symmetrize(h_pert)
        h_pert = project_h(h_pert, n)
        T_use = 200.0 + 200.0 * (i % 3)
        h_new, c5_new = projected_gradient_opt(h_pert, n, T_use, 0.3, 1500, deadline, sym=True)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Coordinate descent polish
    if time.time() < deadline - 20:
        h_new, c5_new = coord_descent(best_h, n, deadline, sym=True, max_iter=50)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Try bang-bang refinement
    if time.time() < deadline - 10:
        h_bb, c5_bb = bang_bang_refine(best_h, n, deadline)
        if c5_bb < best_c5:
            best_c5 = c5_bb
            best_h = h_bb.copy()

    # Final: also try non-symmetric gradient on best
    if time.time() < deadline - 5:
        h_new, c5_new = projected_gradient_opt(best_h, n, 500.0, 0.2, 1000, deadline, sym=False)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    best_h = project_h(best_h, n)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n)
