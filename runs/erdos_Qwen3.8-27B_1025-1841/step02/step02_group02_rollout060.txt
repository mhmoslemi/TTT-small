```python
import numpy as np
import time

def corr_fft(h):
    """Compute np.correlate(h, 1-h, 'full') using FFT."""
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    cc = np.fft.irfft(np.conj(H) * G, N)
    return np.concatenate([cc[N-n+1:N], cc[0:n]])

def compute_c5(h):
    return corr_fft(h).max() * (2.0 / len(h))

def project_to_feasible(g, n):
    target = n / 2.0
    lo, hi = g.min() - 2.0, g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(g - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0, 1)

def smooth_grad_topk(h, tau, topk=20):
    """Compute smooth max with top-k shifts, O(n log n + k*n)."""
    n = len(h)
    dx = 2.0 / n
    corr = corr_fft(h)
    max_c = corr.max()
    # softmax over all shifts (for correct p), but only compute grad for top-k
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    
    # Get top-k indices
    top_idx = np.argsort(p)[-topk:]
    
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[np.arange(j0, j1) - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[np.arange(j0b, j1b) + sh]
    
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad, max_c * dx
```

```python
import numpy as np
import time


def corr_fft(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N <<= 1
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    cc = np.fft.irfft(np.conj(H) * G, N)
    return np.concatenate([cc[N - n + 1:N], cc[0:n]])


def compute_c5(h):
    return corr_fft(h).max() * (2.0 / len(h))


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 3.0
    hi = g.max() + 3.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(g - mid, 0, 1)) > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0, 1)


def project_binary(g, n):
    target = n // 2
    if target == 0:
        target = 1
    idx = np.argsort(g)[::-1][:target]
    h = np.zeros(n)
    h[idx] = 1.0
    return h


def smooth_obj_grad(h, tau, topk=40):
    n = len(h)
    dx = 2.0 / n
    corr = corr_fft(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    top_idx = np.argsort(p)[-topk:]
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx + sh]
    grad *= dx
    obj = dx * (max_c + np.log(s) / tau)
    return obj, grad


def optimize_smooth(h_in, tau, topk, max_iters, time_limit, lr_init=0.25):
    n = len(h_in)
    h = project_to_feasible(h_in.copy(), n)
    obj0, _ = smooth_obj_grad(h, tau, topk)
    best_obj = obj0
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, tau, topk)
        if obj < best_obj - 1e-13:
            best_obj = obj
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        if stall > 120:
            lr *= 0.65
            v *= 0.4
            stall = 0
            if lr < 1e-10:
                break
    return best_h, best_obj, compute_c5(best_h)


def optimize_subgrad(h_in, max_iters, time_limit, lr_init=0.008):
    n = len(h_in)
    dx = 2.0 / n
    h = project_to_feasible(h_in.copy(), n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.88
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        corr = corr_fft(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-14:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[j0b:j1b] -= h[idx + sh]

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)
```
