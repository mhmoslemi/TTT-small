    # = sum_k p[k] * h[i+k-n+1]
    
    # In terms of convolution: (h * q)[m] = sum_j h[j] * q[m-j]
    # If q = p: (h * p)[m] = sum_j h[j] * p[m-j]
    # We want sum_k p[k] * h[i+k-n+1] = sum_k h[i+k-n+1] * p[k]
    # Let j = i+k-n+1, so k = j-i+n-1
    # = sum_j h[j] * p[j-i+n-1] for valid k (0<=k<2n-1) and valid j (0<=j<n)
    # = (h * p)[m] where m = j-i+n-1... this doesn't work directly.
    
    # Let me try a different approach. 
    # S1[i] = sum_k p[k] * h[i + k - n + 1]
    # Let's define q[j] = p[j] for j in 0..2n-2
    # S1[i] = sum_k q[k] * h[(i-n+1) + k]
    # = cross_correlation_1d(h, q)[i-n+1] + (n-1)
    # where cross_correlation_1d(h, q)[m] = sum_k q[k] * h[m+k] (with h out of range = 0)
    
    # cross_correlation(h, q) = convolve(h_reversed, q)
    # = np.correlate(q, h[::-1], ...) hmm
    
    # OK I think the cleanest way for n=4000 is just to compute it directly with
    # a strided approach or accept the O(n^2) cost. Let me benchmark:
    # For n=4000, the gradient loop has 2n-1=7999 iterations, each touching up to n=4000 elements.
    # Total: ~32M element operations per gradient. In numpy, this is about 5-10ms.
    # With 1300s budget, that's ~130,000-260,000 gradient evaluations. More than enough!
    
    # Let me just do the direct computation but vectorized where possible.
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        a = abs(sh)
        if a >= n:
            continue
        i0 = a
        i1 = n - a
        # grad[i0:i1] += pk * ((1-h[i0+sh:i1+sh]) - h[i0-sh:i1-sh])
        grad[i0:i1] += pk * ((1.0 - h[i0+sh:i1+sh]) - h[i0-sh:i1-sh])
    
    grad *= dx
    return obj, grad

def project_to_feasible(g, n):
    target = n / 2.0

        return h.copy()

    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    best_h = h.copy()

    v = np.zeros(n)

    stall = 0

    tau_idx = 0
    current_tau = tau_schedule[0]
    current_topk = top_k_schedule[0]

    n_iter = 0
    for it in range(1000000):
        if time.time() - start > time_limit:
            break
        t += 1
        n_iter += 1
        
        # Check if time to reduce tau
        if tau_idx + 1 < n_tauSteps and t > (it+1) * 0.15:
            pass  # handled below
        
        obj, grad = smooth_obj_grad(h, current_tau, current_topk)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            lr *= 0.5
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2

            if current_topk is not None:
                current_topk = min(2*n-1, (current_topk or 20) + 5)
    
    return best_h, best_c5, n_iter
