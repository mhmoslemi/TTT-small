```python
def binary_local_search(h, n, max_rounds=50, time_fn=None):
    """Local search on binary (or near-binary) h."""
    best = h.copy()
    best_c5 = compute_c5(best)
    
    for rnd in range(max_rounds):
        if time_fn is not None and time_fn() < 5:
            break
        improved = False
        
        # Try flipping single bits
        for j in range(n):
            if time_fn is not None and time_fn() < 3:
                break
            
            old_val = best[j]
            # Try the opposite binary value
            flip_to = 0.0 if old_val > 0.5 else 1.0
            
            if abs(old_val - flip_to) < 1e-6:
                continue  # Already at the target binary value
            
            trial = best.copy()
            trial[j] = flip_to
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            
            if c < best_c5 - 1e-14:
                best = trial
                best_c5 = c
                improved = True
        
        # Try flipping pairs of adjacent bits
        for j in range(0, n-1, 1):
            if time_fn is not None and time_fn() < 3:
                break
            
            # Try flipping both j and j+1
            trial = best.copy()
            trial[j] = 1.0 - best[j]
            trial[j+1] = 1.0 - best[j+1]
            # If sum changes by 0, great. Otherwise repair.
            trial = make_feasible(trial, n)
            c = compute_c5(trial)
            
            if c < best_c5 - 1e-14:
                best = trial
                best_c5 = c
                improved = True
        
        # Try single flip (partial)
        for j in range(n):
            if time_fn is not None and time_fn() < 3:
                break
            
            for delta in [-0.5, 0.5, -0.25, 0.25]:
                trial = best.copy()
                trial[j] = np.clip(best[j] + delta, 0.0, 1.0)
                trial = make_feasible(trial, n)
                c = compute_c5(trial)
                if c < best_c5 - 1e-14:
                    best = trial
                    best_c5 = c
                    improved = True
                    break
        
        if not improved:
            break
    
    return best, best_c5
```

```python
    bc = compute_c5(best)
    for _ in range(max_rounds):
        if rem_fn() < 10:
            break
        imp = False
        # Single flips to binary
        for j in range(n):
            tgt = 0.0 if best[j] > 0.5 else 1.0
            if abs(best[j] - tgt) < 0.01:
                continue
            t = best.copy()
            t[j] = tgt
            t = make_feasible(t, n)
            c = compute_c5(t)
            if c < bc - 1e-14:
                best = t
                bc = c
                imp = True
        # Adjacent swaps (preserves sum)
        for j in range(n - 1):
            if rem_fn() < 5:
                break
            a, b = best[j], best[j + 1]
            if abs(a - b) < 0.1:
                continue
            t = best.copy()
            t[j] = b
            t[j + 1] = a
            c = compute_c5(t)
            if c < bc - 1e-14:
                best = t
                bc = c
                imp = True
        # Small perturbations
        for j in range(n):
            if rem_fn() < 5:
                break
            ov = best[j]
            for d in (0.05, -0.05, 0.1, -0.1):
                nv = min(1.0, max(0.0, ov + d))
                if abs(nv - ov) < 1e-8:
                    continue
                t = best.copy()
                t[j] = nv
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < bc - 1e-14:
                    best = t
                    bc = c
                    imp = True
                    break
        if not imp:
            break
    return best, bc

def coord_dsc(h, n, max_rounds, rem_fn, fine=False):
    best = h.copy()
    bc = compute_c5(best)
    if fine:
        dts = (0.005, -0.005, 0.01, -0.01, 0.02, -0.02)
    else:
        dts = (0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01)
    for _ in range(max_rounds):
        if rem_fn() < 5:
            break
        imp = False
        for j in range(n):
            ov = best[j]
            bv, bcost = ov, bc
            for d in dts:
                cv = min(1.0, max(0.0, ov + d))
                if abs(cv - ov) < 1e-9:
                    continue
                t = best.copy()
                t[j] = cv
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < bcost - 1e-14:
                    bcost, bv = c, cv
            for cv in (0.0, 1.0):
                if abs(cv - ov) < 1e-9:
                    continue
                t = best.copy()
                t[j] = cv
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < bcost - 1e-14:
                    bcost, bv = c, cv
            if bv != ov:
                best[j] = bv
                best = make_feasible(best, n)
                bc = bcost
                imp = True
        if not imp:
            break
    return best, bc
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_t = np.arange(old_n, dtype=float)
    new_t = np.arange(new_n, dtype=float) * (old_n / float(new_n))
    h_new = np.interp(new_t, old_t, h)
    return make_feasible(h_new, new_n)

def _sobj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_G_T * shifted))) / _G_T) * dx

def _cfun(x):
    return np.sum(x) - _G_N / 2.0

def _cjac(x):
    return np.ones(len(x))

def do_slsqp(h0, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    h0 = make_feasible(h0, n)
    try:
        res = minimize(_sobj, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        x = make_feasible(np.asarray(res.x, dtype=float), n)
        return x, compute_c5(x)
    except Exception:
        return h0, compute_c5(h0)
```

This is the cross-correlation of (1-x) with w, but with a specific alignment.

Let a = 1-x (length n), v = w (length 2n-1)

= Σ_i (1-x[i]) * w[i + m - 2n + 2]

I want Σ_b (1-x[b]) * w[j-n+1-b]
Set i = b: I need i + m - 2n + 2 = j - n + 1 - i
2i + m - 2n + 2 = j - n + 1
2i = j - n + 1 - m + 2n - 2
2i = j + n - 1 - m

This depends on i, so it's not a standard correlation. The w index is j-n+1-b, which goes in the opposite direction from b.

Let me also add one more trick: after SLSQP, do a "top-correlation targeting" step. Find the shift k* that gives the max correlation, and try to specifically reduce the overlap at that shift by adjusting h.

For a specific shift k*, the correlation is:
c[k*] = Σ_i h[i+k*-n+1] * (1-h[i])

To reduce c[k*], I want to decrease h[i+k*-n+1] where h[i] is large, or increase h[i] where h[i+k*-n+1] is large. But this might increase other correlations.

```python
def targeted_cd(h, n, max_rounds, rem_fn):
    best = h.copy()
    bc = compute_c5(best)
    for _ in range(max_rounds):
        if rem_fn() < 5:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        k_star = np.argmax(corr)
        # Try to reduce correlation at k_star
        # h[j] contributes to c[k_star] when:
        # j = i + k_star - n + 1 (first factor): reduce h[j] if (1-h[i]) > 0
        # j = i (second factor): increase h[j] if h[a] > 0 where a = j + k_star - n + 1
        imp = False
        for j in range(n):
            if rem_fn() < 3:
                break
            # Compute the sign of ∂c[k_star]/∂h[j]
            # First factor: i = j - k_star + n - 1
            i_f = j - k_star + n - 1
            contrib_f = (1 - best[i_f]) if 0 <= i_f < n else 0
            # Second factor: a = j + k_star - n + 1
            a = j + k_star - n + 1
            contrib_s = -best[a] if 0 <= a < n else 0
            grad_k = contrib_f + contrib_s
            
            # Try to move in the direction that reduces c[k_star]
            if grad_k > 0:
                # Increasing h[j] increases c[k_star], so try decreasing
                candidates = [max(0.0, best[j] - 0.05), max(0.0, best[j] - 0.02), 0.0]
            elif grad_k < 0:
                # Increasing h[j] decreases c[k_star], so try increasing
                candidates = [min(1.0, best[j] + 0.05), min(1.0, best[j] + 0.02), 1.0]
            else:
                continue
            
            best_c = bc
            best_val = best[j]
            for cand in candidates:
                if abs(cand - best[j]) < 1e-8:
                    continue
                t = best.copy()
                t[j] = cand
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < best_c - 1e-14:
                    best_c = c
                    best_val = cand
            if best_val != best[j]:
                best[j] = best_val
                best = make_feasible(best, n)
                bc = best_c
                imp = True
        if not imp:
            break
    return best, bc
```

```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 100.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_t = np.arange(old_n, dtype=float)
    new_t = np.arange(new_n, dtype=float) * (old_n / float(new_n))
    h_new = np.interp(new_t, old_t, h)
    return make_feasible(h_new, new_n)

def _sobj(x):
    n = _G_N
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(_G_T * shifted))) / _G_T) * dx

def _cfun(x):
    return np.sum(x) - _G_N / 2.0

def _cjac(x):
    return np.ones(len(x))

def do_slsqp(h0, n, T, maxiter, ftol):
    global _G_N, _G_T
    _G_N = n
    _G_T = T
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cfun, 'jac': _cjac}]
    h0 = make_feasible(h0, n)
    try:
        res = minimize(_sobj, h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol, 'disp': False})
        x = make_feasible(np.asarray(res.x, dtype=float), n)
        return x, compute_c5(x)
    except Exception:
        return h0, compute_c5(h0)

def flip_search(h, n, max_rounds, rem_fn):
    best = h.copy()
    bc = compute_c5(best)
    for _ in range(max_rounds):
        if rem_fn() < 10:
            break
        imp = False
        for j in range(n):
            tgt = 0.0 if best[j] > 0.5 else 1.0
            if abs(best[j] - tgt) < 0.01:
                continue
            t = best.copy()
            t[j] = tgt
            t = make_feasible(t, n)
            c = compute_c5(t)
            if c < bc - 1e-14:
                best = t
                bc = c
                imp = True
        for j in range(n - 1):
            if rem_fn() < 5:
                break
            a, b = best[j], best[j + 1]
            if abs(a - b) < 0.1:
                continue
            t = best.copy()
            t[j], t[j + 1] = b, a
            c = compute_c5(t)
            if c < bc - 1e-14:
                best = t
                bc = c
                imp = True
        for j in range(0, max(1, n - 4)):
            if rem_fn() < 5:
                break
            for blk in (2, 3, 4):
                if j + blk > n:
                    continue
                t = best.copy()
                v = t[j]
                for jj in range(j, j + blk - 1):
                    t[jj] = t[jj + 1]
                t[j + blk - 1] = v
                c = compute_c5(t)
                if c < bc - 1e-14:
                    best = t
                    bc = c
                    imp = True
        for j in range(n):
            if rem_fn() < 5:
                break
            ov = best[j]
            for d in (0.05, -0.05, 0.1, -0.1):
                nv = min(1.0, max(0.0, ov + d))
                if abs(nv - ov) < 1e-8:
                    continue
                t = best.copy()
                t[j] = nv
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < bc - 1e-14:
                    best = t
                    bc = c
                    imp = True
                    break
        if not imp:

    return best, bc

def coord_dsc(h, n, max_rounds, rem_fn, fine=False):
    best = h.copy()
    bc = compute_c5(best)
    dts = ((0.005, -0.005, 0.01, -0.01, 0.02, -0.02) if fine
           else (0.02, -0.02, 0.05, -0.05, 0.1, -0.1, 0.01, -0.01))
    for _ in range(max_rounds):
        if rem_fn() < 5:
            break
        imp = False
        for j in range(n):
            ov = best[j]
            bv, bcost = ov, bc
            for d in dts:
                cv = min(1.0, max(0.0, ov + d))
                if abs(cv - ov) < 1e-9:
                    continue
                t = best.copy()
                t[j] = cv
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < bcost - 1e-14:
                    bcost, bv = c, cv
            for cv in (0.0, 1.0):
                if abs(cv - ov) < 1e-9:

                t[j] = cv

                    bcost, bv = c, cv
            if bv != ov:
                best[j] = bv
                best = make_feasible(best, n)
                bc = bcost
                imp = True
        if not imp:
            break
    return best, bc

def targeted_cd(h, n, max_rounds, rem_fn):

    bc = compute_c5(best)
    for _ in range(max_rounds):
        if rem_fn() < 5:
            break
        corr = np.correlate(best, 1.0 - best, mode='full')
        ks = int(np.argmax(corr))
        imp = False
        for j in range(n):
            if rem_fn() < 3:
                break
            if = j - ks + n - 1
            cf = (1.0 - best[if_]) if 0 <= if_ < n else 0.0
            ia = j + ks - n + 1
            cs = -best[ia] if 0 <= ia < n else 0.0
            gk = cf + cs
            if abs(gk) < 1e-6:
                continue
            cands = []
            if gk > 0:
                cands = [max(0.0, best[j]-0.05), max(0.0, best[j]-0.02), 0.0]
            else:
                cands = [min(1.0, best[j]+0.05), min(1.0, best[j]+0.02), 1.0]
            bv, bcost = best[j], bc
            for cv in cands:
                if abs(cv - best[j]) < 1e-8:
                    continue
                t = best.copy()
                t[j] = cv
                t = make_feasible(t, n)
                c = compute_c5(t)
                if c < bcost - 1e-14:
                    bcost, bv = c, cv
            if bv != best[j]:
                best[j] = bv
                best = make_feasible(best, n)
                bc = bcost
                imp = True
        if not imp:
            break
    return best, bc

def sim_anneal(h0, n, t_start, t_end, n_steps, rng, rem_fn):
    h = h0.copy()
    best = h.copy()
    bc5 = compute_c5(h)
    cc5 = bc5
    alpha = (t_end / max(t_start, 1e-15)) ** (1.0 / max(n_steps, 1))
    T = t_start
    for _ in range(n_steps):
        if rem_fn() < 5:
            break
        T *= alpha
        h_old = h.copy()
        co_old = cc5
        j = rng.randint(0, n)
        h[j] = 1.0 - h[j]
        h = make_feasible(h, n)
        c = compute_c5(h)
        d = c - co_old
        if d < 0 or rng.random() < np.exp(-d / max(T, 1e-15)):
            cc5 = c
            if c < bc5:
                bc5 = c
                best = h.copy()
        else:
            h = h_old
    return best, bc5
```
