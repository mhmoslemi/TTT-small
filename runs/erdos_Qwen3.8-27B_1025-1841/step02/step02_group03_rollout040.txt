def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0

    """Snap h to binary, maintaining sum = n/2."""
    n = len(h)

    ones = np.sum(h_bin)
    target = n / 2

        excess = int(ones - target)
        # Flip the ones with lowest h values (most "borderline")
        one_indices = np.where(h_bin == 1.0)[0]
        # Sort by h value ascending (most borderline first)
        order = np.argsort(h[one_indices])[:excess]
        h_bin[one_indices[order]] = 0.0

        zero_indices = np.where(h_bin == 0.0)[0]
        order = np.argsort(-h[zero_indices])[:deficit]
        h_bin[zero_indices[order]] = 1.0

def _slsqp_obj(x, n, T):
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _slsqp_con(x, n):
    return np.sum(x) - n / 2.0

def _slsqp_jac(x, n):
    return np.ones(n)

def slsqp_refine(h, n, T, maxiter, t_end):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _slsqp_con(x, n), 'jac': lambda x: _slsqp_jac(x, n)}]
    h0 = make_feasible(h, n)
    try:
        res = minimize(lambda x: _slsqp_obj(x, n, T), h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-13})
        if res.success or res.fun < 999:
            return make_feasible(res.x, n)
    except Exception:
        pass
    return h0
