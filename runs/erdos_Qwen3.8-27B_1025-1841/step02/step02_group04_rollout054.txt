def compute_c5(h):
    n = len(h)

    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-15:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        diff = target - np.sum(h)
        if abs(diff) > 1e-12:
            if diff > 0:
                mask = h < 1.0
            else:
                mask = h > 0.0
            cnt = np.sum(mask)
            if cnt > 0:
                h[mask] += diff / cnt
                h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def corr_grad_contribution(h, shift, n):
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    g = np.zeros(n)
    if shift >= 0:
        g[:n-s] += (1.0 - h[s:])
        g[s:] += -h[:n-s]
    else:
        g[s:] += (1.0 - h[:n-s])
        g[:n-s] += -h[s:]
    return g

def soft_obj(h, T, n):

    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc

    corr = np.correlate(h, 1.0 - h, mode='full')

    idx = np.argsort(weights)[-top_k:][::-1]

        if w < 1e-11:
            break
        shift = k_idx - (n - 1)
        grad += w * corr_grad_contribution(h, shift, n)
    return grad * dx

        grad = soft_grad(h, T, n, top_k)

        h = h - lr * vel
        h = make_feasible(h, n)
