        best_h = h_bin.copy()
    h_bin_seed2, c_bin2 = sim_anneal_binary(h_init, n0, 60000, 0.15, 0.0003, seed + 1000)
    if c_bin2 < best_c5:
        best_c5 = c_bin2
        best_h = h_bin_seed2.copy()

    elapsed = time.time() - start_time
    if elapsed > budget_s - 100:
        best_h = make_feasible(best_h, n0)
        return best_h, compute_c5(best_h), n0

    # Phase 2: SLSQP at multiple resolutions and temperatures
    n_start = len(best_h)
    resolutions = [n_start, n_start, n_start, min(n_start + 160, 1200), max(n_start - 80, 500)]
    temp_list = [60.0, 200.0, 500.0, 100.0, 300.0]
    iter_list = [350, 300, 250, 200, 200]

    for idx in range(len(resolutions)):
        if time.time() - start_time > budget_s - 80:
            break
        n_pts = resolutions[idx]
        T_val = temp_list[idx]
        mx_it = iter_list[idx]
        h_start = upsample_h(best_h, n_pts)
        if idx >= 3:
            rng2 = np.random.RandomState(seed + idx * 77)
            h_start = make_feasible(h_start + rng2.randn(n_pts) * 0.015, n_pts)
        _OPT_N = n_pts
        _OPT_T = T_val
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h_start, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': mx_it, 'ftol': 1e-14})
            h_res = make_feasible(res.x, n_pts)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
        except Exception:
            pass

    # Phase 3: Multi-start SLSQP at best resolution
    n_pts = len(best_h)
    rng3 = np.random.RandomState(seed + 500)
    for attempt in range(3):
        if time.time() - start_time > budget_s - 60:
            break
        h_start = make_feasible(best_h + rng3.randn(n_pts) * 0.02, n_pts)
        _OPT_N = n_pts
        _OPT_T = 150.0
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, h_start, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 250, 'ftol': 1e-14})
            h_res = make_feasible(res.x, n_pts)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
        except Exception:
            pass

    # Phase 4: Coordinate descent (broad)
    if time.time() - start_time < budget_s - 50:
        n_pts = len(best_h)
        set_c = [0.0, 1.0, 0.5]
        add_c = [-0.15, -0.08, -0.03, -0.01, 0.01, 0.03, 0.08, 0.15]
        best_h, best_c5 = coord_descent(best_h, n_pts, set_c, add_c, 1e-14, 30)

    # Phase 5: High-T SLSQP polish
    if time.time() - start_time < budget_s - 40:
        n_pts = len(best_h)
        _OPT_N = n_pts
        _OPT_T = 800.0
        bounds = [(0.0, 1.0)] * n_pts
        cons = [{'type': 'eq', 'fun': _con_fun, 'jac': _con_jac}]
        try:
            res = minimize(_obj, best_h, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 200, 'ftol': 1e-15})
            h_res = make_feasible(res.x, n_pts)
            c5_res = compute_c5(h_res)
            if c5_res < best_c5:
                best_c5 = c5_res
                best_h = h_res
        except Exception:
            pass

    # Phase 6: Fine coordinate descent
    if time.time() - start_time < budget_s - 20:
        n_pts = len(best_h)
        set_c = [0.0, 1.0]
        add_c = [-0.01, -0.005, -0.002, 0.002, 0.005, 0.01]
        best_h, best_c5 = coord_descent(best_h, n_pts, set_c, add_c, 1e-15, 25)

    # Phase 7: Binary snap
    if time.time() - start_time < budget_s - 10:
        n_pts = len(best_h)
        best_h, best_c5 = binary_snap(best_h, n_pts)

    # Phase 8: Final micro-descent
    if time.time() - start_time < budget_s - 5:
        n_pts = len(best_h)
        set_c = [0.0, 1.0]
        add_c = [-0.005, -0.002, 0.002, 0.005]
        best_h, best_c5 = coord_descent(best_h, n_pts, set_c, add_c, 1e-15, 10)

    best_h = make_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
