import numpy as np
from scipy.optimize import minimize
import time

_OPT = {}

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(3):
        diff = target - np.sum(h)
        if abs(diff) < 1e-15:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def _subgrad_single(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    else:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    return grad

def softmax_grad(h, T, n_top=80):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    m = np.max(corr)
    shifted = corr - m
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        idx = int(np.argmax(corr))
        shift = idx - (n - 1)
        return _subgrad_single(h, shift) * dx
    weights = exp_s / total
    top_k = min(n_top, len(weights))
    top_idx = np.argsort(weights)[-top_k:][::-1]
    grad = np.zeros(n)
    for ki in top_idx:
        w = weights[ki]
        if w < 1e-12:
            break
        shift = ki - (n - 1)
        grad += w * _subgrad_single(h, shift)
    return grad * dx

def nesterov_gd(h, n_iter, T_start, T_end, lr, beta_start, beta_end,
                check_every=100, time_limit=1000, t_start=0):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    v = np.zeros(n)
    for it in range(n_iter):
        if time.time() - t_start > time_limit:
            break
        frac = it / max(n_iter - 1, 1)
        T = T_start * (T_end / T_start) ** frac
        beta = beta_start + (beta_end - beta_start) * frac
        lr_t = lr * (1.0 - 0.4 * frac)
        h_y = np.clip(h + beta * v, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_y)
        h_y += diff / n
        h_y = np.clip(h_y, 0.0, 1.0)
        g = softmax_grad(h_y, T, n_top=80)
        v = beta * v - lr_t * g
        h = make_feasible(h + v, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def pca_gd(h, n_iter, T_start, T_end, lr_start, time_limit, t_start):
    n = len(h)
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    beta = 0.92

    stale = 0

        if time.time() - t_start > time_limit:
            break

        T = T_start * (T_end / T_start) ** frac

        h_y = np.clip(h + beta * v, 0.0, 1.0)
        diff = n / 2.0 - np.sum(h_y)
        h_y += diff / n
        h_y = np.clip(h_y, 0.0, 1.0)
        g = softmax_grad(h_y, T, n_top=60)
        v = 0.95 * v - lr_t * g

def try_slsqp(h, T, maxiter, ftol, time_limit, t_start):
    n = len(h)
    if n > 500:
        return h, compute_c5(h)
    _OPT['T'] = T
    _OPT['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:

                      bounds=bounds, constraints=cons,

        if res.x is not None:
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
