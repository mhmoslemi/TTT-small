    """Gradient of corr[m] w.r.t. h. m is 0 to 2n-2."""
    t = m - (n - 1)  # lag
    i_min = max(0, -t)
    i_max = min(n - 1, n - 1 - t)
    
    grad = np.zeros(n)
    # First sum: Σ h[i+t], gradient is 1 where j = i+t, i in [i_min, i_max]
    # j = i+t → i = j-t, valid when i_min ≤ j-t ≤ i_max AND 0 ≤ j ≤ n-1
    # So j ∈ [i_min+t, i_max+t] = [max(0,-t)+t, min(n-1,n-1-t)+t]
    # = [max(t, 0), min(n-1+t, n-1)]
    j_start_1 = max(i_min + t, 0)
    j_end_1 = min(i_max + t, n - 1)
    grad[j_start_1:j_end_1+1] += 1.0
    
    # Second sum: Σ h[i+t]*h[i]
    # h[j] appears when j = i+t (i.e., i = j-t) OR j = i (i.e., i = j)
    # Case 1: j = i+t, i = j-t valid: i_min ≤ j-t ≤ i_max, 0 ≤ j-t ≤ n-1, 0 ≤ j ≤ n-1
    #   Contribution: h[i] = h[j-t]
    j_start_2a = max(i_min + t, 0)
    j_end_2a = min(i_max + t, n - 1)
    for j in range(j_start_2a, j_end_2a + 1):
        i = j - t
        if 0 <= i < n:
            grad[j] -= h[i]
    
    # Case 2: j = i, i = j valid: i_min ≤ j ≤ i_max, 0 ≤ j ≤ n-1, 0 ≤ j+t ≤ n-1

    j_start_2b = max(i_min, 0)

        if 0 <= j + t < n:

    return grad

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0).copy()
    target = n / 2.0
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0 - 1e-14
        else:
            mask = h > 1e-14
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def corr_grad_single(h, n, m):
    """Gradient of corr[m] w.r.t. h. m in [0, 2n-2]."""
    t = m - (n - 1)
    i_min = max(0, -t)
    i_max = min(n - 1, n - 1 - t)
    grad = np.zeros(n)
    # First sum gradient: +1 where j = i+t, i in [i_min, i_max]
    j1_lo = max(i_min + t, 0)
    j1_hi = min(i_max + t, n - 1)
    if j1_lo <= j1_hi:
        grad[j1_lo:j1_hi+1] += 1.0
    # Second sum: -h[i] where j=i+t, i in [i_min, i_max], 0<=i<n
    for j in range(j1_lo, j1_hi + 1):
        i = j - t
        if 0 <= i < n:
            grad[j] -= h[i]
    # -h[i+t] where j=i, i in [i_min, i_max], 0<=j+t<n
    j2_lo = max(i_min, 0)
    j2_hi = min(i_max, n - 1)
    for j in range(j2_lo, j2_hi + 1):
        if 0 <= j + t < n:
            grad[j] -= h[j + t]
    return grad

def obj_and_grad_smooth(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    mc = g.max()
    shifted = g - mc
    exps = np.exp(T * shifted)
    Z = exps.sum()
    weights = exps / Z
    obj_val = mc + np.log(Z) / T
    grad = np.zeros(n)
    active = weights > 1e-8
    for k in np.where(active)[0]:
        grad += weights[k] * corr_grad_single(h, n, int(k))
    grad *= dx
    return obj_val, grad

def obj_and_grad_direct(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    g = corr * dx
    k_star = int(np.argmax(g))
    obj_val = g[k_star]
    grad = corr_grad_single(h, n, k_star) * dx
    return obj_val, grad

def slsqp_with_grad(h_start, T, max_iter, use_direct=False):
    n = len(h_start)
    bounds = [(0.0, 1.0)] * n
    
    def obj(h):
        if use_direct:
            return obj_and_grad_direct(h)[0]
        return obj_and_grad_smooth(h, T)[0]
    
    def obj_grad(h):
        if use_direct:
            return obj_and_grad_direct(h)[1]
        return obj_and_grad_smooth(h, T)[1]
    
    cons = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n/2.0, 'jac': lambda h: np.ones(n)}]
    res = minimize(obj, h_start, method='SLSQP', jac=obj_grad, bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(res.x, n)

def slsqp_periodic(h_start, n_half, max_iter, T):
    """Optimize periodic function h[i] = h[i+n_half] with n_half free vars."""
    n = 2 * n_half
    bounds = [(0.0, 1.0)] * n_half
    
    def full_h(hp):
        return np.concatenate([hp, hp])
    
    def obj(hp):
        h = full_h(hp)
        if T == 0:
            return obj_and_grad_direct(h)[0]
        return obj_and_grad_smooth(h, T)[0]
    
    def obj_grad(hp):
        h = full_h(hp)
        if T == 0:
            g = obj_and_grad_direct(h)[1]
        else:
            g = obj_and_grad_smooth(h, T)[1]
        # Peridic: grad[i] = g[i] + g[i+n_half]
        return g[:n_half] + g[n_half:]
    
    cons = [{'type': 'eq', 'fun': lambda hp: np.sum(hp) - n_half/2.0, 'jac': lambda hp: np.ones(n_half)}]
    res = minimize(obj, h_start, method='SLSQP', jac=obj_grad, bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-14})
    return make_feasible(full_h(res.x), n)

def subgradient_cd(h, n, n_iters=200, top_k=5):
    """Subgradient descent on the max correlation."""
    h = h.copy()
    best = compute_c5(h)
    dx = 2.0 / n
    
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        top_indices = np.argsort(corr)[-top_k:]
        # Average gradient of top-k lags
        grad = np.zeros(n)
        for k in top_indices:
            grad += corr_grad_single(h, n, int(k))
        grad *= dx / top_k
        
        # Adaptive step size
        alpha = 0.05 / (1 + it * 0.01)
        
        # Try step
        h_new = h - alpha * grad
        h_new = make_feasible(h_new, n)
        c = compute_c5(h_new)
        if c < best:
            h = h_new
            best = c
        else:
            # Halve step and retry
            h_new = h - alpha * 0.5 * grad
            h_new = make_feasible(h_new, n)
            c = compute_c5(h_new)
            if c < best:
                h = h_new
                best = c
    
    return h, best
