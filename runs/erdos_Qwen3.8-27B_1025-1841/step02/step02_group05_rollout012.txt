def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * (2.0 / n))

    n = len(h)
    if n == target_n:

    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(corr.max() * (2.0 / n))

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    C = corr * dx
    max_C = C.max()
    sh = np.clip(tau * (C - max_C), -500, 500)
    e = np.exp(sh)
    s = e.sum()
    p = e / s
    f = max_C + np.log(s) / tau
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if p[k] < 1e-12:
            continue
        pkd = p[k] * dx
        t = k - n + 1
        if t >= 0:
            grad[t:] += pkd * (1.0 - h[:n - t])
            grad[:n - t] -= pkd * h[t:]
        else:
            grad[:n + t] += pkd * (1.0 - h[-t:])
            grad[-t:] -= pkd * h[:n + t]
    return f, grad

                    if ub.min() < t_lo:
                        feasible = False
                    else:
                        t_hi = min(t_hi, ub.min())
                        feasible = True
                if neg.any():
                    lb = (mid_v - c[neg]) / m[neg]
                    if t_hi < lb.max():
                        feasible = False

                        t_lo = max(t_lo, lb.max())
                if not pos.any() and not neg.any():
                    feasible = (c.max() <= mid_v)
                elif pos.any() and neg.any():
                    feasible = (t_lo <= t_hi)
                else:
                    feasible = (t_lo <= t_hi)

                    t_star = (t_lo + t_hi) / 2
                else:
                    lo_v = mid_v
                    t_star = h[j]

        c = compute_c5(h)
        if c < bc - 1e-14:
            bc = c
            bh = h.copy()
