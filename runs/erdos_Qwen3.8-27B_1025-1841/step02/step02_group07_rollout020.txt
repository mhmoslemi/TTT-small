    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    sum_h = np.sum(h)

            # Determine range of m where h[i] appears
            m_start = n - 1 - i

            M_other = np.max(c[mask]) if np.any(mask) else 0.0

            else:
                M_other = -np.inf
            
            # Minimize f(t) = max(M_other, max(const_m + t*coeff)) + lam*(sum_h - h[i] + t - target)^2
            S = sum_h - h[i]
            
            # Ternary search
            a, b = 0.0, 1.0
            for _ in range(45):
                m1 = (2.0*a + b) / 3.0
                m2 = (a + 2.0*b) / 3.0
                f1 = max(M_other, np.max(const_m + m1 * coeff)) + lam * (S + m1 - target)**2
                f2 = max(M_other, np.max(const_m + m2 * coeff)) + lam * (S + m2 - target)**2
                if f1 <= f2:
                    b = m2
                else:
                    a = m1
            
            t_opt = (a + b) / 2.0
            delta = t_opt - h[i]
            
            if abs(delta) > 1e-15:
                max_change = max(max_change, abs(delta))
                # Update c
                c[m_lo:m_hi+1] += delta * coeff
                sum_h += delta
                h[i] = t_opt
        
        # Periodic projection
        if sweep % 20 == 0:
            h = project_constraint(h, n)
            c = np.correlate(h, 1.0 - h, mode='full')
            sum_h = np.sum(h)
        
        # Check convergence
        if max_change < min_improve:
            no_improve += 1
            if no_improve > 5:
                break
        else:
            no_improve = 0
    
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)
