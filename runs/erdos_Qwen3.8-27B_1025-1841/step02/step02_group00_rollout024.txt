            break
        if diff > 0:
            mask = h < 1.0 - 1e-15
        else:
            mask = h > 1e-15
        cnt = np.sum(mask)
        if cnt == 0:
            break
        step = diff / cnt
        h[mask] = np.clip(h[mask] + step, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        idx = np.argmin(h) if diff > 0 else np.argmax(h)
        h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h, period=2.0)
    return make_feasible(h_new, new_n)

def thue_morse_h(n, seed=0):
    tm = np.zeros(n, dtype=np.int32)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    h = (tm == 1).astype(float)
    return make_feasible(h, n)

def binary_anneal(h_bin, n, time_limit=200, rng=None):
    if rng is None:
        rng = np.random.RandomState(42)
    start = time.time()
    c5 = compute_c5(h_bin)
    best_c5 = c5
    best_h = h_bin.copy()
    ones = list(np.where(h_bin == 1.0)[0])
    zeros = list(np.where(h_bin == 0.0)[0])
    temp = 0.002
    cool = 0.9997
    
    while time.time() - start < time_limit:
        temp = max(temp * cool, 1e-6)
        improved = False
        batch = 80
        for _ in range(batch):
            if time.time() - start > time_limit:
                break
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h_bin[i] = 0.0
            h_bin[j] = 1.0
            new_c5 = compute_c5(h_bin)
            delta = new_c5 - c5
            if delta < 0 or (temp > 1e-6 and rng.rand() < np.exp(-delta / temp)):
                c5 = new_c5
                improved = True
                ones.remove(i)
                zeros.remove(j)
                zeros.append(i)
                ones.append(j)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_bin.copy()
            else:
                h_bin[i] = 1.0
                h_bin[j] = 0.0
        if not improved:
            pass
    
    return best_h, best_c5

def slsqp_smoothed(h_start, n, T, max_iter):
    dx = 2.0 / n
    def obj(h):
        corr = np.correlate(h, 1.0 - h, mode='full')
        mc = np.max(corr)
        shifted = corr - mc
        log_sum = np.log(np.sum(np.exp(T * shifted)))
        return (mc + log_sum / T) * dx
    def cons_fun(h):
        return np.sum(h) - n / 2.0
    def cons_jac(h):
        return np.ones(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': cons_fun, 'jac': cons_jac}]
    res = minimize(obj, h_start, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-15})
    return make_feasible(res.x, n)

def coord_descent(h, n, time_limit=100):
    steps = [0.05, -0.05, 0.1, -0.1, 0.02, -0.02, 0.2, -0.2, 1.0, 0.0]
    start = time.time()
    c5 = compute_c5(h)
    for _ in range(30):
        if time.time() - start > time_limit:
            break
        improved = False
        for j in range(n):
            if time.time() - start > time_limit:
                break
            old = h[j]
            for s in steps:
                new_val = s if s in (1.0, 0.0) else old + s
                if new_val < -1e-9 or new_val > 1 + 1e-9:
                    continue
                trial = h.copy()
                trial[j] = np.clip(new_val, 0.0, 1.0)
                trial = make_feasible(trial, n)
                new_c5 = compute_c5(trial)
                if new_c5 < c5 - 1e-14:
                    h = trial
                    c5 = new_c5
                    improved = True
        if not improved:
            break
    return h, c5

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)
    
    def time_left():
        return budget_s - (time.time() - start_time)
    
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.tile([1.0, 0.0], 160)
    
    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    
    # Phase 1: Binary search at n=1280 with multiple starts
    n1 = 1280
    if time_left() > 500:
        candidates = []
        # From parent
        h1 = upsample_h(best_h, n1)
        candidates.append(h1)
        # Thue-Morse
        candidates.append(thue_morse_h(n1, seed))
        # Random binary blended
        h_rand = (rng.rand(n1) < 0.5).astype(float)
        h_rand = make_feasible(h_rand, n1)
        h_blend = 0.5 * h1 + 0.5 * h_rand
        h_blend = make_feasible(h_blend, n1)
        candidates.append(h_blend)
        
        for h_cand in candidates:
            if time_left() < 200:
                break
            h_thr = (h_cand >= 0.5).astype(float)
            h_thr = make_feasible(h_thr, n1)
            c5_thr = compute_c5(h_thr)
            if c5_thr < best_c5:
                best_c5 = c5_thr
                best_h = h_thr.copy()
            # Binary anneal
            time_alloc = min(150, time_left() * 0.3)
            if time_alloc > 30:
                h_b, c5_b = binary_anneal(h_thr, n1, time_limit=time_alloc, rng=rng)
                if c5_b < best_c5:
                    best_c5 = c5_b
                    best_h = h_b.copy()
    
    # Phase 2: SLSQP at n=1280 with multiple temperatures
    if time_left() > 200:
        n2 = len(best_h)
        # Slightly smooth for SLSQP
        kernel = np.ones(5) / 5.0
        h_sm = np.convolve(best_h, kernel, mode='same')
        h_sm = make_feasible(h_sm, n2)
        
        for T in [30.0, 100.0, 500.0, 2000.0]:
            if time_left() < 40:
                break
            try:
                h_opt = slsqp_smoothed(h_sm, n2, T, max_iter=200)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h_sm = h_opt.copy()
            except Exception:
                pass
    
    # Phase 3: Higher resolution
    if time_left() > 200:
        n3 = 2560
        h3 = upsample_h(best_h, n3)
        c5_3 = compute_c5(h3)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3.copy()
        
        for T in [100.0, 500.0, 2000.0, 8000.0]:
            if time_left() < 30:
                break
            try:
                h_opt = slsqp_smoothed(best_h, n3, T, max_iter=150)
                c5_opt = compute_c5(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
            except Exception:
                pass
    
    # Phase 4: Coordinate descent
    if time_left() > 80:
        n4 = len(best_h)
        cd_n = min(n4, 1280)
        if cd_n < n4:
            h_cd = upsample_h(best_h, cd_n)
        else:
            h_cd = best_h.copy()
        time_cd = min(70, time_left() - 20)
        h_cd, c5_cd = coord_descent(h_cd, cd_n, time_limit=max(time_cd, 10))
        if c5_cd < best_c5:
            if cd_n < n4:
                h_up = upsample_h(h_cd, n4)
                c5_up = compute_c5(h_up)
                if c5_up < best_c5:
                    best_c5 = c5_up
                    best_h = h_up
            else:
                best_c5 = c5_cd
                best_h = h_cd
    
    # Phase 5: Final polish
    if time_left() > 25:
        n5 = len(best_h)
        if n5 > 2000:
            n5 = 1280
            h5 = upsample_h(best_h, n5)
        else:
            h5 = best_h.copy()
        try:
            h_opt = slsqp_smoothed(h5, n5, 5000.0, max_iter=80)
            c5_opt = compute_c5(h_opt)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                if len(h_opt) < len(best_h):
                    best_h = upsample_h(h_opt, len(best_h))
                    best_c5 = compute_c5(best_h)
                else:
                    best_h = h_opt
        except Exception:
            pass
    
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
