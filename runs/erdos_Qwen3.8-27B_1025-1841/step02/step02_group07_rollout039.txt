```python
    # s2 range: [-p, n-1-p] (from p+s valid)
    # Union: [min(p-n+1, -p), max(p, n-1-p)]
    
    # For i:
    si_min = min(i - n + 1, -i)
    si_max = max(i, n - 1 - i)
    # For j:
    sj_min = min(j - n + 1, -j)
    sj_max = max(j, n - 1 - j)
    
    s_min = max(min(si_min, sj_min), -(n-1))
    s_max = min(max(si_max, sj_max), n - 1)
    
    # Now compute delta for each s in [s_min, s_max]
    # But only where the delta is nonzero (i.e., where at least one of the 4 terms is active)
    # For simplicity, compute for all s in the range
    
    s_range = np.arange(s_min, s_max + 1)
    t_range = s_range + (n - 1)  # indices into corr
    
    # Compute delta vectorized
    # term1: (1-h[i-s]) for s where 0<=i-s<n
    # term2: -h[i+s] for s where 0<=i+s<n
    # term3: -(1-h[j-s]) for s where 0<=j-s<n
    # term4: h[j+s] for s where 0<=j+s<n
    
    delta = np.zeros(len(s_range))
    
    # Vectorized computation:
    # i-s is in [0,n-1] when s is in [i-(n-1), i]
    mask1 = (s_range >= i - (n-1)) & (s_range <= i)
    if np.any(mask1):
        delta[mask1] += (1.0 - h[i - s_range[mask1]])
    
    # i+s is in [0,n-1] when s is in [-i, n-1-i]
    mask2 = (s_range >= -i) & (s_range <= n - 1 - i)
    if np.any(mask2):
        delta[mask2] -= h[i + s_range[mask2]]
    
    # j-s is in [0,n-1] when s is in [j-(n-1), j]
    mask3 = (s_range >= j - (n-1)) & (s_range <= j)
    if np.any(mask3):
        delta[mask3] -= (1.0 - h[j - s_range[mask3]])
    
    # j+s is in [0,n-1] when s is in [-j, n-1-j]
    mask4 = (s_range >= -j) & (s_range <= n - 1 - j)
    if np.any(mask4):
        delta[mask4] += h[j + s_range[mask4]]
    
    # New values at affected positions
    new_affected = corr[t_range] + delta
    new_max = np.max(new_affected)
    
    # Max of unchanged portions
    if s_min > -(n-1) or s_max < n-1:
        # There are unchanged portions
        t_all = np.arange(2*n - 1)
        unchanged_mask = np.ones(len(t_all), dtype=bool)
        unchanged_mask[t_range] = False
        if np.any(unchanged_mask):
            unchanged_max = np.max(corr[unchanged_mask])
            new_max = max(new_max, unchanged_max)
    
    return new_max, delta, t_range
```

This is O(n) per swap (vectorized), which for n=336 is very fast. We should be able to do 100K+ swaps per second.

Wait, there's a problem. When I accept a swap, I update h and corr, but the ones/zeros arrays become stale. For n=336, I can just recompute them every 1000 iterations or so.

Also, I need to be careful: after a swap, the delta computation for the NEXT swap must use the UPDATED h. Since I'm updating h in place, this is fine.

Let me also think about the continuous refinement phase. After finding a good binary solution, I want to relax it to continuous values. The idea is:

Starting from a binary h, do a few steps of projected gradient descent on the smooth max objective. The smooth max is:
G(h) = (1/T) * log(sum_s exp(T * c_s(h) * dx))

This is a smooth upper bound on max_s c_s(h) * dx.

The gradient of G(h) can be computed efficiently using the weighted sum formula I derived earlier.

Actually, let me use a simpler approach: just use scipy's L-BFGS-B with the softmax objective, starting from the binary solution (slightly randomized to be in the interior).

```python
    si_max = min(max(i, n - 1 - i), n - 1)
    sj_min = max(min(j - n + 1, -j), -(n - 1))

    delta = np.zeros(len(s_range))
    m1 = (s_range >= i - (n - 1)) & (s_range <= i)

def binary_sa(h_bin, n, deadline, rng, max_iter=300000):
    dx = 2.0 / n
    h = h_bin.copy().astype(float)
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    cur_max = float(np.max(corr) * dx)
    best_max = cur_max
    best_h = h.copy()
    ones = list(np.where(h == 1)[0])
    zeros = list(np.where(h == 0)[0])
    # Greedy phase
    no_improve = 0
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if no_improve > 5000:
            break
        best_delta = 0.0
        best_pair = None
        n_tries = 200
        picked = set()
        for _ in range(n_tries):
            ii = rng.randint(len(zeros))
            jj = rng.randint(len(ones))
            key = (zeros[ii], ones[jj])
            if key in picked:
                continue
            picked.add(key)
            nm, dl, tr = try_swap(corr, h, zeros[ii], ones[jj], n)
            d = cur_max - nm
            if d > best_delta:
                best_delta = d
                best_pair = (zeros[ii], ones[jj], dl, tr, nm)
        if best_pair is not None:
            i, j, dl, tr, nm = best_pair
            h[i] = 1.0
            h[j] = 0.0
            corr[tr] += dl
            ones.remove(j)
            ones.append(i)
            zeros.remove(i)
            zeros.append(j)
            cur_max = nm
            no_improve = 0
            if cur_max < best_max:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
    # SA phase
    if time.time() < deadline:
        T = (cur_max - best_max + 0.005)
        T_end = T * 0.01
        sa_iter = 0
        while time.time() < deadline:
            sa_iter += 1
            T = max(T_end, T * 0.9997)
            i = zeros[rng.randint(len(zeros))]
            j = ones[rng.randint(len(ones))]
            nm, dl, tr = try_swap(corr, h, i, j, n)
            accept = nm < cur_max or (rng.random() < np.exp(-(nm - cur_max) / max(T, 1e-12)))
            if accept:
                h[i] = 1.0
                h[j] = 0.0
                corr[tr] += dl
                ones.remove(j)
                ones.append(i)
                zeros.remove(i)
                zeros.append(j)
                cur_max = nm
                if cur_max < best_max:
                    best_max = cur_max
                    best_h = h.copy()
    return best_h, best_max

def softmax_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def continuous_refine(h0, n, deadline, T_list, maxiter_list):
    h = h0.copy()
    best_h = h.copy()
    best_val = evaluate_c5(h)
    for T, mi in zip(T_list, maxiter_list):
        if time.time() > deadline:
            break
        def f(x):
            return softmax_obj(x, n, T)
        def c_eq(x):
            return np.sum(x) - n / 2.0
        try:
            res = minimize(f, h.copy(), method='L-BFGS-B',
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': c_eq}],
                           options={'ftol': 1e-15, 'maxiter': mi, 'gtol': 1e-10})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_new = project_symmetric(res.x, n)
                v = evaluate_c5(h_new)
                if v < best_val:
                    best_val = v
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_val

def interp_to_n(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    h_i = np.interp(xt, xf, h)
    return project_symmetric(h_i, n_to)
```
