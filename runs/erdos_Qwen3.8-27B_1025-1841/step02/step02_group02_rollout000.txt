def opt_smooth(h, n, tau, tl, lr0=0.2, beta=0.9):

            s_ = ki - n + 1

        if stall > 100:

            v *= 0.3
            stall = 0
            if lr < 1e-10:

    c5r = c5_val(bh)

def binary_local_search(h, n, tl, n_iter=2000):

            zeros = np.where(h == 0.0)[0]

    return bh, bc5

    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    H = np.fft.rfft(h, N)
    G = np.fft.rfft(1.0 - h, N)
    return np.fft.irfft(H * np.conj(G), N)[:2 * n - 1]
