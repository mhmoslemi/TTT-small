def slsqp_optimize(h, T=4.0, maxiter=500):
    n = len(h)
    _OPT['T'] = T
    _OPT['n'] = n
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _cons_fun, 'jac': _cons_jac}]
    try:
        res = minimize(_soft_obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            return hx, compute_c5(hx)
    except Exception:
        pass
    return h, compute_c5(h)

def upsample(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    best_c5 = float('inf')
    best_h = None

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = None

    if h0 is None:
        n_init = 320
        h0 = (rng.rand(n_init) < 0.5).astype(float)
        n_vals = np.sum(h0)
        if n_vals != n_init // 2:
            if n_vals > n_init // 2:
                ones = np.where(h0 == 1.0)[0]
                h0[rng.choice(ones, int(n_vals - n_init//2), replace=False)] = 0.0
            else:
                zeros = np.where(h0 == 0.0)[0]
                h0[rng.choice(zeros, int(n_init//2 - n_vals), replace=False)] = 1.0
        h0 = make_feasible(h0, len(h0))

    best_c5 = compute_c5(h0)
    best_h = h0.copy()
    n0 = len(best_h)
    print(f"Initial: n={n0}, C5={best_c5:.6f}", flush=True)

    # Phase 1: GD with momentum at initial resolution
    remaining = budget_s - (time.time() - start_time)
    if remaining > 30:
        h1, c1 = gd_momentum(best_h, n_iter=3500, T_start=80.0, T_end=0.3,
                             lr=0.12, top_k=80, momentum=0.9, check_every=20, rng=rng)
        if c1 < best_c5:
            best_c5, best_h = c1, h1
        print(f"P1 GD: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 2: SLSQP polishing at multiple temperatures
    n_cur = len(best_h)
    if n_cur <= 512 and (budget_s - (time.time() - start_time)) > 80:
        for T_val in [20.0, 8.0, 3.0, 1.5]:
            if time.time() - start_time > budget_s - 30:
                break
            hx, cx = slsqp_optimize(best_h, T=T_val, maxiter=400)
            if cx < best_c5:
                best_c5, best_h = cx, hx
                print(f"  SLSQP T={T_val}: C5={best_c5:.6f}", flush=True)
        print(f"P2 SLSQP: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 3: Second GD pass with different schedule
    if budget_s - (time.time() - start_time) > 60:
        h2, c2 = gd_momentum(best_h, n_iter=3000, T_start=20.0, T_end=0.1,
                             lr=0.06, top_k=100, momentum=0.85, check_every=30, rng=rng)
        if c2 < best_c5:
            best_c5, best_h = c2, h2
        print(f"P3 GD2: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 4: Upsample to higher resolution
    new_n = min(n0 * 2, 640)
    if new_n > n0 and budget_s - (time.time() - start_time) > 150:
        h_up = upsample(best_h, new_n)
        c_up = compute_c5(h_up)
        print(f"  Upsampled to n={new_n}: C5={c_up:.6f}", flush=True)
        if (budget_s - (time.time() - start_time)) > 80:
            h_up, cu = gd_momentum(h_up, n_iter=3000, T_start=40.0, T_end=0.2,
                                   lr=0.08, top_k=80, momentum=0.9, check_every=30, rng=rng)
            if cu < c_up:
                c_up = cu
        if new_n <= 512 and (budget_s - (time.time() - start_time)) > 60:
            h_up, cu = slsqp_optimize(h_up, T=5.0, maxiter=350)
            if cu < c_up:
                c_up = cu
            h_up, cu = slsqp_optimize(h_up, T=2.0, maxiter=300)
            if cu < c_up:
                c_up = cu
        if c_up < best_c5:
            best_c5, best_h = c_up, h_up
            print(f"  New best at n={new_n}: C5={best_c5:.6f}", flush=True)
        else:
            print(f"  No improvement at n={new_n} (C5={c_up:.6f})", flush=True)
        print(f"P4 upsample: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 5: Multi-start from perturbed best
    n_cur = len(best_h)
    if budget_s - (time.time() - start_time) > 100:
        for scale in [0.02, 0.05, 0.1]:
            if time.time() - start_time > budget_s - 40:
                break
            h_pert = make_feasible(best_h + rng.randn(n_cur) * scale, n_cur)
            hp, cp = gd_momentum(h_pert, n_iter=2000, T_start=30.0, T_end=0.2,
                                 lr=0.08, top_k=80, momentum=0.9, check_every=40, rng=rng)
            if cp < best_c5:
                best_c5, best_h = cp, hp
                print(f"  Perturb s={scale}: C5={best_c5:.6f}", flush=True)
        print(f"P5 multi: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Phase 6: Final SLSQP with very low T
    n_cur = len(best_h)
    if n_cur <= 512 and budget_s - (time.time() - start_time) > 50:
        for T_val in [1.5, 0.8, 0.5]:
            if time.time() - start_time > budget_s - 15:
                break
            hx, cx = slsqp_optimize(best_h, T=T_val, maxiter=500)
            if cx < best_c5:
                best_c5, best_h = cx, hx
                print(f"  Final SLSQP T={T_val}: C5={best_c5:.6f}", flush=True)

    # Phase 7: Fine GD with very low T
    if budget_s - (time.time() - start_time) > 20:
        n_cur = len(best_h)
        hf, cf = gd_momentum(best_h, n_iter=2000, T_start=2.0, T_end=0.05,
                             lr=0.02, top_k=200, momentum=0.7, check_every=30, rng=rng)
        if cf < best_c5:
            best_c5, best_h = cf, hf
        print(f"P7 FineGD: C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
