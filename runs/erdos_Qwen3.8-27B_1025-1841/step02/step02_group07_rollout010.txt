```python
import numpy as np
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def grad_single_lag(h, k, n):
    """Gradient of g_k(h) = sum_i h[i](1-h[i+k]) w.r.t. h[j]
    grad[j] = 1 - h[j+k] - h[j-k] (with zero padding)
    """
    grad = np.ones(n)
    if k > 0:
        grad[:n-k] -= h[k:]  # h[j+k] for j < n-k
        grad[k:] -= h[:n-k]  # h[j-k] for j >= k
    elif k < 0:
        kp = -k
        grad[kp:] -= h[:n-kp]  # h[j+k] = h[j-kp] for j >= kp
        grad[:n-kp] -= h[kp:]  # h[j-k] = h[j+kp] for j < n-kp
    else:
        grad -= 2 * h
    return grad

def grad_top_m(h, m, n):
    """Gradient using top-m active lags (weighted by softmax of correlation values)"""
    c = np.correlate(h, 1.0 - h, mode='full')
    # Softmax over top-m
    idx = np.argsort(c)[::-1][:m]
    vals = c[idx]
    T = 50.0  # temperature for weighting
    weights = np.exp((vals - vals[0]) / T)
    weights /= weights.sum()
    
    grad = np.zeros(n)
    for i, mi in enumerate(idx):
        k = mi - (n - 1)
        grad += weights[i] * grad_single_lag(h, k, n)
    return grad, c

def projected_grad_desc(h, n, deadline, target_c5, max_iter=5000):
    """Projected gradient descent with exact max and momentum."""
    h = h.copy()
    lr = 0.01
    momentum = np.zeros(n)
    beta = 0.9
    best_h = h.copy()
    best_c5 = target_c5
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        
        # Compute current c5
        c = np.correlate(h, 1.0 - h, mode='full')
        max_idx = np.argmax(c)
        max_val = c[max_idx]
        k_star = max_idx - (n - 1)
        
        # Use top-3 active lags
        top3 = np.argsort(c)[::-1][:3]
        w = np.exp((c[top3] - max_val) * 0.5)
        w /= w.sum()
        
        grad = np.zeros(n)
        for i, mi in enumerate(top3):
            k = mi - (n - 1)
            grad += w[i] * grad_single_lag(h, k, n)
        
        # Gradient step with momentum
        momentum = beta * momentum - lr * grad
        h_new = h + momentum
        h_new = project(h_new, n)
        
        # Check if improved
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Adaptive lr: increase slightly
            lr = min(lr * 1.02, 0.05)
        else:
            h = h_new  # Still use it (momentum helps)
            lr = lr * 0.95
        
        if it % 200 == 0:
            # Periodic exact evaluation
            pass
    
    return best_h, best_c5
```

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-13:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def softmax_obj_and_grad(h, n, T, top_m=None):
    """Compute soft-max objective and gradient.
    F_T(h) = (dx/T) * log(sum exp(T * c_k))
    """
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    if top_m is not None and top_m < len(c):
        idx = np.argsort(c)[::-1][:top_m]
        c_use = c[idx]
        shifted = T * (c_use - c_use[0])
        weights = np.exp(shifted)
        weights /= weights.sum()
        # Gradient: sum_k weights[k] * grad_c_k
        grad = np.zeros(n)
        for i, mi in enumerate(idx):
            k = mi - (n - 1)
            # grad_c_k[j] = 1 - h[j+k] - h[j-k]
            g = np.ones(n)
            if k > 0:
                g[:n-k] -= h[k:]
                g[k:] -= h[:n-k]
            elif k < 0:
                kp = -k
                g[kp:] -= h[:n-kp]
                g[:n-kp] -= h[kp:]
            else:
                g -= 2 * h
            grad += weights[i] * g
        # Objective (relative to max, for numerical stability)
        obj = (1.0 / T) * (np.log(np.sum(np.exp(shifted))) + T * c_use[0]) 
        obj *= dx  # Scale to actual integral
        grad *= dx  # Scale gradient
    else:
        shifted = T * (c - c.max())
        weights = np.exp(shifted)
        weights /= weights.sum()
        grad = np.zeros(n)
        for mi in np.nonzero(weights > 1e-15)[0]:
            k = mi - (n - 1)
            g = np.ones(n)
            if k > 0:
                g[:n-k] -= h[k:]
                g[k:] -= h[:n-k]
            elif k < 0:
                kp = -k
                g[kp:] -= h[:n-kp]
                g[:n-kp] -= h[kp:]
            else:
                g -= 2 * h
            grad += weights[mi] * g
        obj = (1.0 / T) * (c.max() + np.log(np.sum(np.exp(shifted))))
        obj *= dx
        grad *= dx
    return obj, grad

def lbfgs_polish(h0, n, T, deadline, maxiter=200):
    """L-BFGS-B polishing with penalty for sum constraint."""

    target = n / 2.0
    penalty = 1000.0

        o, g = softmax_obj_and_grad(x, n, T, top_m=20)

        s = np.sum(x) - target
        o += penalty * s * s
