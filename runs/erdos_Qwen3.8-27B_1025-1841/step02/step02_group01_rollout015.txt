    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    if new_n == old_n:

    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

    """Find optimal h[j] using 1D convex optimization."""
    old = h[j]
    
    # Compute B vector: change in corr per unit change in h[j]
    B = np.zeros(2 * n - 1)
    
    # Phase 1: h[j] appears as h[i+s] where i = j-s, valid if 0<=j-s<n
    # s ranges: j-(n-1) to j (clamped to [-(n-1), n-1])
    lo1 = max(j - (n-1), -(n-1))
    hi1 = min(j, n-1)
    if hi1 >= lo1:
        # B indices: s+(n-1) for s in [lo1, hi1]
        # h indices: j-s for s in [lo1, hi1], i.e., j-hi1 to j-lo1 (reversed)
        B[lo1 + (n-1) : hi1 + (n-1) + 1] += 1.0 - h[j - hi1 : j - lo1 + 1][::-1]
    
    # Phase 2: h[j] appears in (1-h[i]) where i=j, valid if 0<=j+s<n
    # s ranges: -j to n-1-j
    lo2 = -j
    hi2 = n - 1 - j
    if hi2 >= lo2:
        B[lo2 + (n-1) : hi2 + (n-1) + 1] -= h[j + lo2 : j + hi2 + 1]
    
    # f(t) = max_s (corr_full[s] + B[s] * (t - old))
    # = max_s ((corr_full[s] - B[s]*old) + B[s]*t)
    A = corr_full - B * old
    
    # Golden section search on [0, 1]
    gr = (np.sqrt(5.0) - 1.0) / 2.0

    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = np.max(A + B * c_pt)
    fd = np.max(A + B * d_pt)
    
    for _ in range(35):

            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = np.max(A + B * c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = np.max(A + B * d_pt)
    
    t_opt = (a + b) * 0.5
    f_opt = np.max(A + B * t_opt)
    
    f0 = np.max(A)
    f1 = np.max(A + B)
    if f0 < f_opt:
        t_opt = 0.0
    if f1 < f_opt:
        t_opt = 1.0
    
    return t_opt

import numpy as np
from scipy.optimize import minimize
import time

_SLSQP_N = 0
_SLSQP_DX = 0.0

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def coord_opt(h, j, n, corr_full):
    old = h[j]
    B = np.zeros(2 * n - 1)
    B[j:j + n] += 1.0 - h[::-1]
    B[n - 1 - j:2 * n - 1 - j] -= h[0:n - j]
    A = corr_full - B * old
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, 1.0
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = np.max(A + B * c_pt)
    fd = np.max(A + B * d_pt)
    for _ in range(35):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = np.max(A + B * c_pt)
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = np.max(A + B * d_pt)
    t_opt = (a + b) * 0.5
    f_opt = np.max(A + B * t_opt)
    f0 = np.max(A)
    f1 = np.max(A + B)
    if f0 < f_opt:
        t_opt = 0.0
    if f1 < f_opt:
        t_opt = 1.0
    return t_opt

def slsqp_obj(hv):
    corr = np.correlate(hv, 1.0 - hv, mode='full')
    mc = np.max(corr)
    t_param = 100.0
    shifted = corr - mc
    soft = mc + np.log(np.sum(np.exp(t_param * shifted))) / t_param
    return soft * _SLSQP_DX

def slsqp_con(hv):
    return np.sum(hv) - _SLSQP_N / 2.0

def slsqp_con_jac(hv):
    return np.ones(_SLSQP_N)

def coord_descent_pass(h, n, rng, max_passes=10, time_limit=None):
    best = compute_c5(h)
    for p in range(max_passes):
        if time_limit is not None and time.time() > time_limit:
            break
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(n)
        improved = False
        for j in order:
            if time_limit is not None and time.time() > time_limit:
                break
            t_opt = coord_opt(h, j, n, corr_full)
            if abs(t_opt - h[j]) > 1e-8:
                old = h[j]
                h[j] = t_opt
                B = np.zeros(2 * n - 1)
                B[j:j + n] += 1.0 - h[::-1]
                B[n - 1 - j:2 * n - 1 - j] -= h[0:n - j]
                B = np.zeros(2 * n - 1)
                B[j:j + n] += 1.0 - h[::-1].copy()
                # Recompute B with new h for incremental update
                h[j] = t_opt
                B = np.zeros(2 * n - 1)
                B[j:j + n] += 1.0 - h[::-1]
                B[n - 1 - j:2 * n - 1 - j] -= h[0:n - j]
                # Actually the B depends on h at all indices including j
                # Just update corr_full by the delta
                h[j] = old  # temp restore for B computation
                B = np.zeros(2 * n - 1)
                B[j:j + n] += 1.0 - h[::-1]
                B[n - 1 - j:2 * n - 1 - j] -= h[0:n - j]
                h[j] = t_opt
                corr_full += B * (t_opt - old)
                improved = True
        h = make_feasible(h, n)
        c = compute_c5(h)
        if c < best - 1e-12:
            best = c
        elif not improved:
            break
    return h, best
