            # So total coeff of h[i]: (1-h*[i+d])*1_{valid i} + (-h*[i-d])*1_{valid i-d}
            # Constant: sum_j h*[j]*h*[j+d]
            
            # This is what I had. Let me just compute directly.
            
            constraints = []
            h_star = h.copy()
            for k in range(2*n-1):
                d = k - (n-1)
                j0 = max(0, d)
                j1 = min(n, n+d)
                j0b = max(0, -d)
                j1b = min(n, n-d)
                
                # Coefficient vector for this constraint
                c = np.zeros(n)
                const = 0.0
                if j0 < j1:
                    idx = np.arange(j0, j1)
                    c[idx] += (1.0 - h_star[idx - d])
                    const += np.sum(h_star[idx] * (1.0 - h_star[idx - d]))
                if j0b < j1b:
                    idx = np.arange(j0b, j1b)
                    c[idx] -= h_star[idx + d]
                
                constraints.append(dx * cp.sum(c * h_var) + dx * const <= t_var)
            
            obj = cp.Minimize(t_var)
            prob = cp.Problem(obj, constraints + [
                cp.sum(h_var) == n/2.0,
                h_var >= 0, h_var <= 1
            ])
            
            # Solve
            prob.solve(solver=cp.CBC, max_time=60)
            
            if h_var.value is not None:
                h_new = np.clip(h_var.value, 0, 1)
                h_new = project_to_feasible(h_new, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                h = h_new
            else:
                break
        except:
            break
    
    return best_h, best_c5

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_weighted_grad(h, lags, weights, n, dx):
    """Compute weighted sum of gradients for given lags."""
    grad = np.zeros(n)
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-18:
            continue
        k = lags[i]
        d = k - n + 1
        # First part: (1-h[i+d]) for valid i
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += w * (1.0 - h[idx - d])
        # Second part: -h[i-d] for valid i
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= w * h[idx + d]
    return grad * dx
