def compute_delta(h, i, j):
    n = len(h)
    delta = np.zeros(2*n-1)
    # k ranges from -(n-1) to (n-1), index in delta is n-1+k
    
    # h[i-k]-1 for valid i-k: k in [i-(n-1), i]
    k1_lo = max(-(n-1), i-(n-1))
    k1_hi = min(n-1, i)
    if k1_lo <= k1_hi:
        idx = np.arange(k1_lo, k1_hi+1)
        delta[n-1+idx] += h[i-idx] - 1.0
    
    # h[i+k] for valid i+k: k in [-i, n-1-i]
    k2_lo = max(-(n-1), -i)
    k2_hi = min(n-1, n-1-i)
    if k2_lo <= k2_hi:
        idx = np.arange(k2_lo, k2_hi+1)
        delta[n-1+idx] += h[i+idx]
    
    # (1-h[j-k]) for valid j-k: k in [j-(n-1), j]
    k3_lo = max(-(n-1), j-(n-1))
    k3_hi = min(n-1, j)
    if k3_lo <= k3_hi:

    if k4_lo <= k4_hi:
        idx = np.arange(k4_lo, k4_hi+1)
        delta[n-1+idx] -= h[j+idx]
    
    return delta
