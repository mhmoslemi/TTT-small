import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def adam_step_moment(h, m, v, t, lr, beta1, beta2, eps, n):
    m = beta1 * m + (1 - beta1) * h
    v = beta2 * v + (1 - beta2) * (h * h)
    m_hat = m / (1 - beta1 ** t)
    v_hat = v / (1 - beta2 ** t)
    return h - lr * m_hat / (np.sqrt(v_hat) + eps)

def optimize_adam_smooth(h, n, tau_schedule, time_limit, lr=0.01):
    """Optimize using Adam with smooth max objective."""
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    dx = 2.0 / n
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.8, 0.999, 1e-8
    
    # Precompute FFT size
    fft_size = int(2 ** np.ceil(np.log2(3 * n - 1)))
    
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        
        # Get current tau from schedule
        progress = min(1.0, (time.time() - start) / time_limit)
        tau = tau_schedule(progress)
        
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx
        
        # Track best
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 5000:
            lr *= 0.7
            m *= 0.5
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        
        # Compute smooth objective gradient
        shifted = tau * (corr - max_c)
        exp_c = np.exp(np.clip(shifted, -500, 0))
        s_val = exp_c.sum()
        p = exp_c / s_val
        
        # Compute gradient via FFT
        q = 1.0 - 2.0 * h
        q_flipped = q[::-1].copy()
        P_fft = np.fft.rfft(p, fft_size)
        q_fft = np.fft.rfft(q_flipped, fft_size)
        conv = np.fft.irfft(P_fft * q_fft, fft_size)
        grad = dx * conv[n-1 : n-1+n]
        
        # Adam step
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible set
        h = project_to_feasible(h_new, n)

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def fft_gradient(h, p, n, fft_size):
    """Compute gradient of smooth max correlation using FFT."""
    dx = 2.0 / n
    q = 1.0 - 2.0 * h
    q_flipped = q[::-1].copy()
    P_fft = np.fft.rfft(p, fft_size)
    q_fft = np.fft.rfft(q_flipped, fft_size)
    conv = np.fft.irfft(P_fft * q_fft, fft_size)
    return dx * conv[n-1 : 2*n-1]

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    fft_size = int(2 ** np.ceil(np.log2(3 * n - 1)))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    stall = 0
    c5_ema = best_c5
    
    while time.time() - start < time_limit:
        # Adaptive tau
        elapsed = time.time() - start
        progress = min(1.0, elapsed / max(1.0, time_limit))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.7)
        
        # Compute correlation and smooth max
        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx
        c5_ema = 0.98 * c5_ema + 0.02 * c5_now
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 3000:
            lr *= 0.7
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break
        
        # Softmax over correlations
        shifted = tau * (corr - max_c)
        exp_c = np.exp(np.clip(shifted, -500, 0))
        s_val = exp_c.sum()
        p = exp_c / s_val
        
        # Gradient via FFT
        grad = fft_gradient(h, p, n, fft_size)
        
        # Adam update

        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def fft_gradient(h, p, n, fft_size):
    dx = 2.0 / n
    q = 1.0 - 2.0 * h
    flipped_p = p[::-1].copy()
    P_fft = np.fft.rfft(flipped_p, fft_size)
    q_fft = np.fft.rfft(q, fft_size)
    conv = np.fft.irfft(P_fft * q_fft, fft_size)
    return dx * conv[2 * n - 2: 3 * n - 2]


def optimize_adam(h, n, time_limit, tau_start, tau_end, lr=0.01):

    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    fft_size = int(2 ** np.ceil(np.log2(3 * n - 1)))
    m = np.zeros(n)
    v = np.zeros(n)
    t = 0
    beta1, beta2, eps = 0.85, 0.999, 1e-8
    stall = 0

    while time.time() - start < time_limit:
        elapsed = time.time() - start
        progress = min(1.0, elapsed / max(1.0, time_limit))
        tau = tau_start * (tau_end / tau_start) ** (progress ** 0.6)

        corr = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr.max()
        c5_now = max_c * dx

        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            lr *= 0.65
            m *= 0.3
            v *= 0.3
            stall = 0
            if lr < 1e-11:
                break

        shifted = tau * (corr - max_c)
        exp_c = np.exp(np.clip(shifted, -500, 0))
        s_val = exp_c.sum()
        p = exp_c / s_val

        grad = fft_gradient(h, p, n, fft_size)

        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = project_to_feasible(h_new, n)

    return best_h, best_c5


def optimize_subgrad_polish(h, n, time_limit, lr=0.001):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    beta = 0.8
    stall = 0

    while time.time() - start < time_limit:
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-15:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 4000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-14:
                break

        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
        grad *= dx

        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_c5
