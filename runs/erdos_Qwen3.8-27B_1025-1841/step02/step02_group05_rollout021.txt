    max_c = corr.max()
    shifted = tau * (corr - max_c)
    shifted = np.clip(shifted, -500, 500)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()

    top_idx = np.argsort(p)[::-1][:top_k]
    
    grad = np.zeros(n)
    for k in top_idx:
        pk = p[k]
        sh = k - n + 1
        # First term: add pk*(1-h[j+sh]) for valid j

import numpy as np
import time


def fft_correlate(a, v):
    n = len(a)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A = np.fft.rfft(a, N)
    V = np.fft.rfft(v, N)
    R = np.fft.irfft(A * np.conj(V), N)
    out = np.empty(2 * n - 1)
    if n > 1:
        out[:n-1] = R[N-(n-1):N-1:-1]
    out[n-1:] = R[0:n]
    return out


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    corr = fft_correlate(h, 1.0 - h)
    return corr.max() * (2.0 / n)


def smooth_optimize(h, n, time_limit, tau_start=100.0, lr=0.3, top_k=50, 
                    momentum=0.9, tau_min=3.0, seed=42):

    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    v = np.zeros(n)
    tau = tau_start
    stall = 0
    prev_smooth = np.inf
    
    # Estimate iterations and set tau decay
    # Target: go from tau_start to tau_min over the time limit
    # Assume ~2ms per iteration
    est_iters = max(int(time_limit / 0.002), 100)
    tau_decay = (tau_min / tau_start) ** (1.0 / est_iters)
    
    while time.time() - start < time_limit:
        corr = fft_correlate(h, 1.0 - h)
        max_c = corr.max()
        
        shifted = tau * (corr - max_c)
        np.clip(shifted, -400, 400, out=shifted)
        exp_c = np.exp(shifted)
        log_s = np.log(exp_c.sum())
        smooth_val = max_c + log_s / tau
        
        p = exp_c / exp_c.sum()
        
        # Top-K gradient
        top_idx = np.argpartition(corr, -top_k)[-top_k:]
        grad = np.zeros(n)
        for k in top_idx:
            pk = p[k]
            sh = k - n + 1
            j0 = max(0, -sh)
            j1 = min(n, n - sh)
            if j0 < j1:
                grad[j0:j1] += pk * (1.0 - h[j0+sh:j1+sh])
            j0b = max(0, sh)
            j1b = min(n, n + sh)
            if j0b < j1b:
                grad[j0b:j1b] -= pk * h[j0b-sh:j1b-sh]
        grad *= dx
        
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
        
        # Decay tau
        tau = max(tau * tau_decay, tau_min)
        
        # Track smooth progress

            stall = 0
        else:
            stall += 1
        prev_smooth = smooth_val

    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    
    return best_h, best_c5
