import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(30):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return make_feasible(h.copy(), new_n)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def snap_to_binary(h, n):
    target = n // 2
    idx = np.argsort(-h)
    h_bin = np.zeros(n)
    h_bin[idx[:target]] = 1.0
    return h_bin

def binary_sa(h, n, start_time, time_limit, rng, T_init=0.004, T_final=0.00003, max_iter=500000):
    h = h.copy()
    curr_c = compute_c5(h)
    best_h = h.copy()
    best_c = curr_c
    n_half = n // 2
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        progress = min(1.0, it / max_iter)
        T = T_init * (T_final / T_init) ** progress
        if T < 1e-12:
            T = 1e-12
        
        i = ones_list[rng.randint(len(ones_list))]
        j = zeros_list[rng.randint(len(zeros_list))]
        
        h[i] = 0.0
        h[j] = 1.0
        new_c = compute_c5(h)
        delta = new_c - curr_c
        
        if delta < -1e-15 or rng.random() < np.exp(-delta / T):
            curr_c = new_c
            ones_list.remove(i)
            ones_list.append(j)
            zeros_list.remove(j)
            zeros_list.append(i)
            if new_c < best_c - 1e-15:
                best_c = new_c
                best_h = h.copy()
        else:
            h[i] = 1.0
            h[j] = 0.0
    
    return best_h, best_c

def greedy_binary_descent(h, n, start_time, time_limit, rng):
    h = h.copy()
    best_c = compute_c5(h)
    ones_list = np.where(h == 1.0)[0].tolist()
    zeros_list = np.where(h == 0.0)[0].tolist()
    
    no_improve = 0
    max_no_improve = 2000
    while time.time() - start_time < time_limit:
        improved = False
        # Try a batch of random pairs
        n_try = min(200, len(ones_list) * len(zeros_list))
        for _ in range(n_try):
            if time.time() - start_time > time_limit:
                break
            i = ones_list[rng.randint(len(ones_list))]
            j = zeros_list[rng.randint(len(zeros_list))]
            h[i] = 0.0
            h[j] = 1.0
            new_c = compute_c5(h)
            if new_c < best_c - 1e-14:
                best_c = new_c
                ones_list.remove(i)
                ones_list.append(j)
                zeros_list.remove(j)
                zeros_list.append(i)
                no_improve = 0
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        if not improved and no_improve > max_no_improve:
            break
    return h, best_c

def soft_max_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = np.max(corr)
    shifted = corr - mx
    return (mx + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def sum_constraint(x):
    return np.sum(x) - len(x) / 2.0

def sum_constraint_jac(x):
    return np.ones(len(x))

def slsqp_refine(h, n, T, maxiter=200, ftol=1e-14):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac}]
    try:
        res = minimize(soft_max_obj, h, args=(T,), method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': ftol})
        if res.x is not None:
            hx = make_feasible(res.x, n)
            c = compute_c5(hx)
            return hx, c
    except Exception:
        pass
    return h, compute_c5(h)

def coordinate_descent(h, n, start_time, time_limit, step=0.01):
    h = h.copy()
    best_c = compute_c5(h)
    n_pts = len(h)
    for rnd in range(200):
        if time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(n_pts):
            if time.time() - start_time > time_limit:
                break
            old = h[j]
            best_trial = old
            best_c_trial = best_c
            cands = [0.0, 1.0, max(0.0, old - 2*step), max(0.0, old - step),
                     min(1.0, old + step), min(1.0, old + 2*step)]
            for cand in cands:
                if abs(cand - old) < 1e-10:
                    continue
                trial = h.copy()
                trial[j] = cand
                trial = make_feasible(trial, n_pts)
                c = compute_c5(trial)
                if c < best_c_trial - 1e-14:
                    best_c_trial = c
                    best_trial = cand
            if best_trial != old:
                h = h.copy()
                h[j] = best_trial
                h = make_feasible(h, n_pts)
                best_c = best_c_trial
                improved = True
        if not improved:
            break
    return h, best_c
