    for _ in range(45):

        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_full_corr_fft(h):
    n = len(h)
    L = 2 * n
    H = np.fft.rfft(h, L)
    M = np.fft.rfft(1.0 - h, L)
    corr_circ = np.fft.irfft(H * np.conj(M), L)
    idx = (np.arange(2 * n - 1) - n + 1) % L
    return corr_circ[idx]


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return compute_full_corr_fft(h).max() * dx


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def gradient_at_lags(h, lags, weights, n, dx):
    grad = np.zeros(n)
    for i in range(len(lags)):
        w = weights[i]
        if w < 1e-12:
            continue
        d = lags[i] - (n - 1)
        j_lo1 = max(0, d)
        j_hi1 = min(n, n + d)
        if j_lo1 < j_hi1:
            idx1 = np.arange(j_lo1, j_hi1)
            grad[idx1] += w * (1.0 - h[idx1 - d])
        j_lo2 = max(0, -d)
        j_hi2 = min(n, n - d)
        if j_lo2 < j_hi2:
            idx2 = np.arange(j_lo2, j_hi2)
            grad[idx2] -= w * h[idx2 + d]
    return grad * dx


def optimize_adam(h, n, time_limit, tau_start=8.0, tau_end=120.0, topM=80,
                  lr=0.5, momentum=0.9, patience=3000):
    dx = 2.0 / n
    L = 2 * n
    h = project_to_feasible(h, n).copy()
    full_idx = (np.arange(2 * n - 1) - n + 1) % L

    H = np.fft.rfft(h, L)
    M = np.fft.rfft(1.0 - h, L)
    corr_circ = np.fft.irfft(H * np.conj(M), L)
    g = corr_circ[full_idx] * dx
    best_obj = g.max()
    best_h = h.copy()

    start = time.time()
    v = np.zeros(n)
    m = np.zeros(n)
    u = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    stall = 0
    lr_scale = 1.0

    while time.time() - start < time_limit:
        t += 1
        progress = min(1.0, t / 5000.0)
        tau = tau_start + (tau_end - tau_start) * progress

        H = np.fft.rfft(h, L)
        M = np.fft.rfft(1.0 - h, L)
        corr_circ = np.fft.irfft(H * np.conj(M), L)
        g = corr_circ[full_idx] * dx
        cur_max = g.max()

        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > patience:
            lr_scale *= 0.6
            v *= 0.3
            m *= 0.5
            u *= 0.5
            stall = 0
            if lr_scale < 1e-5:
                break

        if topM < len(g):
            top_idx = np.argpartition(g, -topM)[-topM:]
            g_top = g[top_idx].copy()
        else:
            top_idx = np.arange(len(g))
            g_top = g.copy()

        max_g = g_top.max()
        shifted = np.clip(tau * (g_top - max_g), -500, 0)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        if s < 1e-30:
            p = np.ones(len(g_top)) / len(g_top)
        else:
            p = exp_c / s

        grad = gradient_at_lags(h, top_idx, p, n, dx)

        m = beta1 * m + (1.0 - beta1) * grad
        u = beta2 * u + (1.0 - beta2) * grad * grad
        m_hat = m / (1.0 - beta1 ** t)
        u_hat = u / (1.0 - beta2 ** t)
        step = lr * lr_scale * m_hat / (np.sqrt(u_hat) + eps)
        v = momentum * v - step
        h = project_to_feasible(h + v, n)

    return best_h, best_obj


def optimize_focused(h, n, time_limit, topK=10, lr=0.01):
    """Focus on reducing the single worst lag with subgradient."""
    dx = 2.0 / n
    L = 2 * n
    h = project_to_feasible(h, n).copy()
    full_idx = (np.arange(2 * n - 1) - n + 1) % L
    best_obj = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.8
    stall = 0

    while time.time() - start < time_limit:
        H = np.fft.rfft(h, L)
        M = np.fft.rfft(1.0 - h, L)
        corr_circ = np.fft.irfft(H * np.conj(M), L)
        g = corr_circ[full_idx] * dx
        cur_max = g.max()

        if cur_max < best_obj - 1e-12:
            best_obj = cur_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if stall > 2000:
            break

        top_idx = np.argsort(g)[-topK:]
        g_top = g[top_idx]
        max_g = g_top.max()
        shifted = np.clip(50.0 * (g_top - max_g), -500, 0)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()

        grad = gradient_at_lags(h, top_idx, p, n, dx)
        cur_lr = lr / (1.0 + 0.001 * stall)
        v = beta * v - cur_lr * grad
        h = project_to_feasible(h + v, n)

    return best_h, best_obj


def run(seed=42, budget_s=1300, **kwargs):
    t0 = time.time()
    np.random.seed(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - t0))

    h_init = initial_h_values.copy()
    n_init = len(h_init)
    best_c5 = compute_c5(h_init)
    best_h = h_init.copy()

    # Phase 1: Quick optimization at initial resolution
    if remaining() > 120:
        tl = min(remaining() * 0.12, 120.0)
        h1, c1 = optimize_adam(best_h, n_init, tl, tau_start=5.0, tau_end=60.0,
                                topM=50, lr=0.4, patience=2000)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Upsample to 2x and optimize
    n2 = n_init * 2
    if remaining() > 250:
        h2 = interp_periodic(best_h, n2)
        h2 = project_to_feasible(h2, n2)
        tl = min(remaining() * 0.25, 300.0)
        h2, c2 = optimize_adam(h2, n2, tl, tau_start=8.0, tau_end=100.0,
                                topM=80, lr=0.5, patience=3000)
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()

    # Phase 3: Upsample to 4x and main optimization
    n3 = n_init * 4
    if remaining() > 350:
        h3 = interp_periodic(best_h, n3)
        h3 = project_to_feasible(h3, n3)
        tl = min(remaining() * 0.35, 450.0)
        h3, c3 = optimize_adam(h3, n3, tl, tau_start=10.0, tau_end=150.0,
                                topM=100, lr=0.5, patience=4000)
        if c3 < best_c5:
            best_c5 = c3
            best_h = h3.copy()

    # Phase 4: Focused optimization on worst lags
    if remaining() > 200:
        n_f = len(best_h)
        hl = best_h.copy()
        tl = min(remaining() * 0.2, 250.0)
        hl, cf = optimize_focused(hl, n_f, tl, topK=8, lr=0.008)
        if cf < best_c5:
            best_c5 = cf
            best_h = hl.copy()

    # Phase 5: Multiple restarts with perturbation
    if remaining() > 150:
        n_r = len(best_h)
        for attempt in range(4):
            if remaining() < 30:
                break
            noise = 0.015 * (attempt + 1)
            hr = best_h + noise * np.random.randn(n_r)
            hr = project_to_feasible(hr, n_r)
            tl = min(remaining() * 0.15, 60.0)
            hr, cr = optimize_adam(hr, n_r, tl, tau_start=15.0, tau_end=100.0,
                                    topM=60, lr=0.4, patience=1500)
            if cr < best_c5:
                best_c5 = cr
                best_h = hr.copy()

    # Phase 6: Final Adam polish
    if remaining() > 60:
        n_fin = len(best_h)
        hf = best_h.copy()
        tl = min(remaining() * 0.25, 60.0)
        hf, cfin = optimize_adam(hf, n_fin, tl, tau_start=30.0, tau_end=200.0,
                                  topM=40, lr=0.3, patience=2000)
        if cfin < best_c5:
            best_c5 = cfin
            best_h = hf.copy()

    # Phase 7: Final focused polish
    if remaining() > 30:
        n_fin = len(best_h)
        hf = best_h.copy()
        tl = min(remaining() * 0.3, 30.0)
        hf, cfin = optimize_focused(hf, n_fin, tl, topK=5, lr=0.003)
        if cfin < best_c5:
            best_c5 = cfin
            best_h = hf.copy()

    # Final feasibility
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
