import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = max(g.min() - 2.0, -10.0)
    hi = min(g.max() + 2.0, 10.0)
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):

def grad_fk(h, d, n):

    grad = np.zeros(n)
    if d == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        grad[:n-d] += dx
        grad[:n-d] -= dx * h[d:]
        grad[d:] -= dx * h[:n-d]

def lp_optimize(h, n, K=20, solver=None):
    """One LP linearization step."""
    dx = 2.0 / n
    corr = compute_corr(h)
    f = corr * dx
    
    top_idx = np.argpartition(f, -K)[-K:]
    
    # Compute gradients for active lags
    grads = []
    f_vals = []
    for k in top_idx:
        d = k - (n - 1)
        g = grad_fk(h, d, n)
        grads.append(g)
        f_vals.append(f[k])
    
    # LP: min t s.t. f_k(h0) + grad_k^T (h - h0) <= t
    #    = min t s.t. grad_k^T h <= t + (f_k(h0) - grad_k^T h0)
    #    = min t s.t. grad_k^T h - t <= f_k(h0) - grad_k^T h0
    
    # Variable: x = [h; t], size n+1
    # Minimize: [0..0; 1] · x
    # Constraints:
    #   For each k: grad_k^T h - t <= f_k(h0) - grad_k^T h0
    #   -h <= 0 (h >= 0)
    #   h <= 1
    #   sum(h) = n/2
    
    c = np.zeros(n + 1)
    c[n] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    # For each active lag:
    for i in range(K):
        row = np.zeros(n + 1)
        row[:n] = grads[i]
        row[n] = -1.0
        A_ub.append(row)
        b_ub.append(f_vals[i] - np.dot(grads[i], h))
    
    # h <= 1: (this is redundant with sum h = n/2 < n, but include for safety)
    # Actually, with 0<=h<=1 and sum h = n/2, h<=1 is automatically satisfied if we have it.
    # Let's include 0<=h<=1 as bounds instead for efficiency.
    
    # h >= 0 and h <= 1 as bounds:
    bounds = [(0, 1)] * n + [(f.min(), None)]  # t >= min(f) as lower bound
    
    # Equality: sum h = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    A_ub = np.array(A_ub) if A_ub else np.zeros((0, n+1))
    b_ub = np.array(b_ub) if b_ub else np.zeros(0)
    
    from scipy.optimize import linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if not res.success:
        return h.copy(), f.max()
    
    h_new = res.x[:n]
    t_new = res.x[n]
    
    # Verify and project
    h_new = np.clip(h_new, 0, 1)
    # Re-project to get exact sum
    h_new = project_to_feasible(h_new, n)
    
    c5_new = compute_c5(h_new)
    return h_new, c5_new

def bb_subgrad(h, n, max_iters, time_limit, lr_init=None, topK=1):
    """Barzilai-Borwein subgradient descent."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    prev_grad = None
    prev_h = None
    it_count = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        f = corr * dx
        c5_cur = f.max()
        
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Find active lags
        if topK == 1:
            k_star = np.argmax(f)
            grads = [grad_fk(h, k_star - (n-1), n)]
        else:
            top_idx = np.argpartition(f, -topK)[-topK:]
            # Weight by proximity to max
            f_top = f[top_idx]
            w = np.exp(20 * (f_top - f_top.max()))
            w /= w.sum()
            grads = np.zeros((topK, n))
            for i, k in enumerate(top_idx):
                grads[i] = grad_fk(h, k - (n-1), n)
            g = np.zeros(n)
            for i in range(topK):
                g += w[i] * grads[i]
            grads = [g]
        
        g = grads[0]
        
        # BB step size
        if prev_grad is not None:
            s = h - prev_h
            y = g - prev_grad
            sTs = s @ s
            yTy = y @ y
            if sTs > 1e-20 and yTy > 1e-20:
                sy = s @ y
                if sy > 1e-15:
                    lr = sTs / sy
                else:
                    lr = sTs / (yTy + 1e-20)
            else:
                lr = 0.5
            lr = np.clip(lr, 1e-6, 5.0)
        else:
            lr = lr_init if lr_init else 0.1
        
        prev_grad = g.copy()
        prev_h = h.copy()
        
        h_new = project_to_feasible(h - lr * g, n)
        
        # Check if we made progress; if not, reduce step
        c5_new = compute_c5(h_new)
        if c5_new > c5_cur + 1e-6:
            # Step back: try smaller step
            h_new = project_to_feasible(h - lr * 0.5 * g, n)
            c5_new = compute_c5(h_new)
            if c5_new > c5_cur + 1e-6:
                h_new = h.copy()
        
        h = h_new
        it_count += 1
    
    return best_h, best_c5

def fw_step(h, g, n):
    """Frank-Wolfe linear minimization step over {0≤s≤1, sum s = n/2}."""
    # Find the n/2 indices with smallest g values
    # s[i] = 1 if i is in the bottom n/2 of g, else 0
    sorted_idx = np.argsort(g)
    s = np.zeros(n)
    s[sorted_idx[:n//2]] = 1.0
    return s

import numpy as np
import time
from scipy.optimize import minimize_scalar

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    if hi <= lo:
        return np.clip(g, 0.0, 1.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    result = np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)
    return result

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)

def grad_fk(h, d, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if d == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        grad[:n-d] += dx
        grad[:n-d] -= dx * h[d:]
        grad[d:] -= dx * h[:n-d]
    else:
        dd = -d
        grad[dd:] += dx
        grad[dd:] -= dx * h[:n-dd]
        grad[:n-dd] -= dx * h[dd:]
    return grad

def fw_vertex(g, n):
    """Return the FW vertex (0-1 vector with n/2 ones at positions of smallest g)."""
    sorted_idx = np.argsort(g)
    s = np.zeros(n)
    s[sorted_idx[:n//2]] = 1.0
    return s

def fw_optimize(h, n, max_iters, time_limit, topK=1):
    """Frank-Wolfe optimization with acceleration."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    for t in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        f = corr * dx
        c5_cur = f.max()
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Gradient of active lag(s)
        if topK == 1:
            k_star = np.argmax(f)
            g = grad_fk(h, k_star - (n-1), n)
        else:
            top_idx = np.argpartition(f, -topK)[-topK:]
            f_top = f[top_idx]
            w = np.exp(10 * (f_top - f_top.max()))
            w /= w.sum()
            g = np.zeros(n)
            for i, k in enumerate(top_idx):
                g += w[i] * grad_fk(h, k - (n-1), n)
        
        # FW vertex
        s = fw_vertex(g, n)
        
        # Step
        gamma = 2.0 / (t + 2.0)
        h_new = (1 - gamma) * h + gamma * s
        # Project
        h_new = project_to_feasible(h_new, n)
        
        # Check improvement
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
        else:
            # Try smaller step
            h_new = (1 - gamma/2) * h + (gamma/2) * s
            h_new = project_to_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h = h_new
            else:
                # Skip this step
                pass
    
    return best_h, best_c5

def bb_subgrad(h, n, max_iters, time_limit, topK=1, bb_momentum=True):
    """BB (Barzilai-Borwein) subgradient method."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    prev_g = None
    prev_h = None
    lr = 0.1
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        f = corr * dx
        c5_cur = f.max()
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Gradient
        if topK == 1:
            k_star = np.argmax(f)
            g = grad_fk(h, k_star - (n-1), n)
        else:
            top_idx = np.argpartition(f, -topK)[-topK:]
            f_top = f[top_idx]
            w = np.exp(15 * (f_top - f_top.max()))
            w /= w.sum()
            g = np.zeros(n)
            for i, k in enumerate(top_idx):
                g += w[i] * grad_fk(h, k - (n-1), n)
        
        # BB step size
        if prev_g is not None and bb_momentum:
            s = h - prev_h
            y = g - prev_g
            sTs = np.dot(s, s)
            sy = np.dot(s, y)
            if sTs > 1e-20 and abs(sy) > 1e-20:
                lr = np.clip(sTs / sy, 1e-6, 10.0) if sy > 0 else 0.5
            else:
                lr = 0.5
        else:
            lr = np.clip(lr * 0.999, 1e-4, 1.0)
        
        prev_g = g.copy()
        prev_h = h.copy()
        
        # Take step with backtracking
        alpha = lr
        for _ in range(5):
            h_new = project_to_feasible(h - alpha * g, n)
            c5_new = compute_c5(h_new)
            if c5_new <= c5_cur + 1e-12:
                break
            alpha *= 0.5
        
        h = h_new
    
    return best_h, best_c5

def smooth_max_subgrad(h, n, max_iters, time_limit, tau=5.0):
    """Subgradient with smooth max approximation (like previous but tuned)."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    v = np.zeros(n)
    beta = 0.8
    lr = 0.05

        if time.time() - start > time_limit:
            break
        
        corr = compute_corr(h)
        f = corr * dx
        c5_cur = f.max()
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
        
        # Smooth max gradient
        f_max = f.max()
        shifted = np.clip(tau * (f - f_max), -500, 500)
        w = np.exp(shifted)
        w /= w.sum()
        
        # Only keep top portion to save time
        threshold = 0.01
        active_mask = w > threshold
        if active_mask.sum() == 0:
            active_mask[np.argmax(f)] = True
        if active_mask.sum() > 30:
            top_rows = np.where(active_mask)[0]
            keep = np.where(np.isin(np.arange(len(f)), np.argsort(f)[-30:]))[0]
            active_mask = False
            active_mask[keep] = True
        
        g = np.zeros(n)
        for k in np.where(active_mask)[0]:
            d = k - (n - 1)
            g += w[k] * grad_fk(h, d, n)
        
        v = beta * v - lr * g
        h_new = project_to_feasible(h + v, n)
        
        # Adaptive LR
        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            lr = min(lr * 1.05, 1.0)
        else:
            lr *= 0.8
            v *= 0.5
        
        h = h_new
    
    return best_h, best_c5

To minimize f w.r.t. h[8] ∈ [0,1]:
- If coefficient < 0: set h[8] = 1
- If coefficient > 0: set h[8] = 0
- If coefficient = 0: any value works

But we also need to maintain the sum constraint! So coordinate descent doesn't directly work here.

Alternative: **Paired coordinate descent**: swap h[i] and h[j] (maintaining the sum). Or: increase h[i] by δ and decrease h[j] by δ.

This is getting complex. Let me just go with the BB subgradient + FW approach and make it as efficient as possible.

One more idea: **Use the fact that the gradient has a specific banded structure**. The gradient of f_k for lag k has nonzeros only in positions related to the shift d. Specifically, for shift d, the gradient has the form:
grad[j] = dx * (1 - h[j+d]) - dx * h[j-d] (with boundary handling)

This means the gradient is a "shifted" copy of h. I can compute it very efficiently.

For the BB step, I need:
- s = h_new - h_old (the step taken)
- y = g_new - g_old (change in gradient)
- lr = (s^T s) / (s^T y)

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    if hi <= lo:
        return np.clip(g, 0.0, 1.0)
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def compute_c5(h):
    n = len(h)
    return compute_corr(h).max() * (2.0 / n)


def grad_fk(h, d, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    if d == 0:
        grad[:] = dx * (1.0 - 2.0 * h)
    elif d > 0:
        grad[:n - d] += dx
        grad[:n - d] -= dx * h[d:]
        grad[d:] -= dx * h[:n - d]
    else:
        dd = -d
        grad[dd:] += dx
        grad[dd:] -= dx * h[:n - dd]
        grad[:n - dd] -= dx * h[dd:]
    return grad


def fw_vertex(g, n):
    sorted_idx = np.argsort(g)
    s = np.zeros(n)
    s[sorted_idx[:n // 2]] = 1.0
    return s


def bb_subgrad(h, n, max_iters, time_limit, topK=1):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    prev_g = None
    prev_h = None
    lr = 0.1
    stall = 0
    patience = 2000

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        f = corr * dx
        c5_cur = f.max()
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if topK == 1:
            k_star = np.argmax(f)
            g = grad_fk(h, k_star - (n - 1), n)
        else:
            top_idx = np.argpartition(f, -topK)[-topK:]
            f_top = f[top_idx]
            w = np.exp(20.0 * (f_top - f_top.max()))
            w /= w.sum()
            g = np.zeros(n)
            for i in range(topK):
                g += w[i] * grad_fk(h, top_idx[i] - (n - 1), n)

        if prev_g is not None:
            s_vec = h - prev_h
            y_vec = g - prev_g
            sTs = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            if sTs > 1e-20 and sy > 1e-20:
                lr = np.clip(sTs / sy, 1e-5, 5.0)
            elif sTs > 1e-20:
                lr = np.clip(sTs / (np.dot(y_vec, y_vec) + 1e-20), 1e-5, 5.0)
            else:
                lr = 0.5
        else:
            lr = 0.1

        prev_g = g.copy()
        prev_h = h.copy()

        alpha = lr
        h_new = h
        c5_new = c5_cur
        for _ in range(4):
            h_try = project_to_feasible(h - alpha * g, n)
            c5_try = compute_c5(h_try)
            if c5_try <= c5_cur + 1e-13:
                h_new = h_try
                c5_new = c5_try
                break
            alpha *= 0.4

        if stall > patience:
            lr *= 0.5
            stall = 0
            if lr < 1e-8:
                break

        h = h_new

    return best_h, best_c5


def fw_optimize(h, n, max_iters, time_limit):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()

    for t in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        f = corr * dx
        c5_cur = f.max()
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()

        k_star = np.argmax(f)
        g = grad_fk(h, k_star - (n - 1), n)
        s = fw_vertex(g, n)

        gamma = 2.0 / (t + 2.0)
        h_new = (1.0 - gamma) * h + gamma * s
        h_new = project_to_feasible(h_new, n)

        c5_new = compute_c5(h_new)
        if c5_new < c5_cur:
            h = h_new
        else:
            gamma2 = gamma * 0.5
            h_new = (1.0 - gamma2) * h + gamma2 * s
            h_new = project_to_feasible(h_new, n)
            c5_new = compute_c5(h_new)
            if c5_new < c5_cur:
                h = h_new

    return best_h, best_c5


def periodic_optimize(base, p, n_total, max_iters, time_limit):
    """Optimize a p-periodic pattern, tile to n_total."""
    h_pat = project_to_feasible(base[:p].copy(), p)
    best_c5 = np.inf
    best_pat = h_pat.copy()
    start = time.time()
    lx = 2.0 / n_total
    prev_g = None
    prev_h = None
    lr = 0.5

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        h_full = np.tile(h_pat, n_total // p)
        if n_total % p != 0:
            h_full = np.concatenate([h_full, h_pat[:n_total % p]])
        corr = compute_corr(h_full)
        f = corr * lx
        c5_val = f.max()
        if c5_val < best_c5:
            best_c5 = c5_val
            best_pat = h_pat.copy()

        k_star = np.argmax(f)
        d = k_star - (n_total - 1)
        g_full = grad_fk(h_full, d, n_total)
        # Average gradient over periods
        g_pat = np.zeros(p)
        for j in range(p):
            idx = np.arange(j, n_total, p)
            g_pat[j] = g_full[idx].mean()

        if prev_g is not None:
            s_vec = h_pat - prev_h
            y_vec = g_pat - prev_g
            sTs = np.dot(s_vec, s_vec)
            sy = np.dot(s_vec, y_vec)
            if sTs > 1e-20 and sy > 1e-20:
                lr = np.clip(sTs / sy, 1e-4, 10.0)
        prev_g = g_pat.copy()
        prev_h = h_pat.copy()

        h_pat = project_to_feasible(h_pat - lr * g_pat / (n_total // p), p)

    return best_pat, best_c5


def smooth_max_opt(h, n, max_iters, time_limit, tau=30.0):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    v = np.zeros(n)
    beta = 0.85
    lr = 0.08
    stall = 0

    for it in range(max_iters):
        if time.time() - start > time_limit:
            break

        corr = compute_corr(h)
        f = corr * dx
        c5_cur = f.max()
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        f_max = f.max()
        shifted = np.clip(tau * (f - f_max), -500, 500)
        w = np.exp(shifted)
        w /= w.sum()

        # Only process significant lags
        active = w > 0.005
        if active.sum() > 50:
            active = np.zeros_like(w)
            top_rows = np.argsort(w)[-50:]
            active[top_rows] = True
        elif active.sum() == 0:
            active[np.argmax(f)] = True

        g = np.zeros(n)
        for k in np.where(active)[0]:
            if w[k] < 1e-8:
                continue
            d = k - (n - 1)
            g += w[k] * grad_fk(h, d, n)

        v = beta * v - lr * g
        h_new = project_to_feasible(h + v, n)
        c5_new = compute_c5(h_new)

        if c5_new < c5_cur:
            h = h_new
            lr = min(lr * 1.03, 0.5)
        else:
            lr *= 0.7
            v *= 0.5
            h = h_new

        if stall > 3000:
            lr *= 0.6
            v *= 0.3
            stall = 0
            if lr < 1e-10:
                break

    return best_h, best_c5
