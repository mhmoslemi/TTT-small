    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(500):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def compute_lag_gradient(h, n, lag):
    """Compute gradient of L_lag with respect to h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            grad[i] += (1.0 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        l = -lag
        for i in range(l, n):
            grad[i] += (1.0 - h[i - l])
            grad[i - l] += (-h[i])
    return grad * dx

def get_all_lags(h, n):
    """Get all L_k values and the worst lag."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    values = corr * dx
    worst_idx = np.argmax(values)
    worst_lag = worst_idx - (n - 1)
    return values, worst_lag

def slsqp_softmax(h0, n, T, maxiter, deadline):
    """SLSQP with soft-max smooth approximation."""
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        one_minus = 1.0 - x
        corr = np.correlate(x, one_minus, mode='full')
        scaled = corr * T
        m = np.max(scaled)
        return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx
    
    def con_eq(x):
        return np.sum(x) - target
    
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'ftol': 1e-15, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except:
        pass
    return h0

def momentum_subgrad(h0, n, deadline, max_iter=5000):
    """Momentum-based subgradient method."""
    h = h0.copy()
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 1
    base_lr = 1.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        t += 1
        lr = base_lr / (np.sqrt(t) + 1)
        
        values, worst_lag = get_all_lags(h, n)
        c5_current = values[worst_lag + (n-1)]
        
        grad = compute_lag_gradient(h, n, worst_lag)
        
        # Momentum update
        v = beta1 * v + (1 - beta1) * grad
        v_hat = v / (1 - beta1**t)
        
        h_new = h - lr * v_hat
        h_new = project_feasible(h_new, n)
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
    
    return best_h, best_c5

def alternating_lag_opt(h0, n, deadline, max_rounds=100):
    """Alternating: pick worst lag, optimize for it, repeat."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        c5_current = float(np.max(values))
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Optimize to reduce L_{worst_lag}
        # L_lag = dx * (n/2 - sum_i h[i]*h[i+lag])

        # Use a few steps of projected gradient specifically for this lag
        for step in range(50):
            if time.time() > deadline:
                break
            grad = compute_lag_gradient(h, n, worst_lag)
            # Line search
            for alpha in [0.01, 0.005, 0.002, 0.001]:
                h_trial = project_feasible(h - alpha * grad, n)
                val_trial = float(np.dot(h_trial, 1-h_trial))  # quick check
                # Actually check the specific lag
                if lag_ge_0(worst_lag, n):
                    lag = worst_lag
                    val_trial = sum(h_trial[i]*(1-h_trial[i+lag]) for i in range(n-lag)) * (2.0/n)
                else:
                    l = -worst_lag
                    val_trial = sum(h_trial[i]*(1-h_trial[i-l]) for i in range(l, n)) * (2.0/n)
                # Just use the full C5 check occasionally
                h_new = h_trial
                break
            h = h_new
        
        # Check improvement
        c5_check = evaluate_c5(h)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h.copy()
    
    return best_h, best_c5

def lag_ge_0(lag, n):
    return lag >= 0

def fourier_param_opt(M, n, deadline, h_init=None):
    """Optimize using Fourier parameterization h = 0.5 + sum of cos/sin terms."""
    x = np.linspace(0, 2, n, endpoint=False)
    
    # Build basis matrix
    basis = np.zeros((n, 2*M))
    for m in range(1, M+1):
        basis[:, 2*(m-1)] = np.cos(m * np.pi * x)
        basis[:, 2*(m-1)+1] = np.sin(m * np.pi * x)
    
    def build_h(params):
        return 0.5 + basis @ params
    
    def obj(params):
        h = build_h(params)
        if h.min() < -0.01 or h.max() > 1.01:
            return 10.0 + (h.min() < -0.01) * (-h.min()) * 10 + (h.max() > 1.01) * (h.max()-1.01) * 10
        h_clipped = np.clip(h, 0, 1)
        h_proj = project_feasible(h_clipped, n)
        return evaluate_c5(h_proj)
    
    # Initial parameters: project h_init onto Fourier subspace
    if h_init is not None:
        # Least squares: h_init ≈ 0.5 + basis @ params
        target_init = h_init - 0.5
        params0, _, _, _ = np.linalg.lstsq(basis, target_init, rcond=None)
    else:
        params0 = np.zeros(2*M)
    
    # Optimize
    best_params = params0.copy()
    best_val = obj(params0)
    
    # Try multiple random initializations
    rng = np.random.RandomState(0)
    for trial in range(20):
        if time.time() > deadline:
            break
        if trial == 0:
            p = params0.copy()
        else:
            p = params0 + rng.randn(2*M) * 0.05
        
        try:
            res = minimize(obj, p, method='Nelder-Mead',
                          options={'maxiter': 2000, 'xatol': 1e-8, 'fatol': 1e-10})
            val = res.fun
            if val < best_val:
                best_val = val
                best_params = res.x.copy()
        except:
            pass
    
    h_best = build_h(best_params)
    h_best = project_feasible(np.clip(h_best, 0, 1), n)
    return h_best, evaluate_c5(h_best)

import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def grad_exact_lag(h, n, lag):
    """Exact gradient of L_lag w.r.t. h."""
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n-lag] += (1.0 - h[lag:])
        grad[lag:] += (-h[:n-lag])
        grad *= dx
    else:
        l = -lag
        grad[:n-l] += (-h[l:])
        grad[l:] += (1.0 - h[:n-l])
        grad *= dx
    return grad

def get_worst_lag(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    idx = np.argmax(vals)
    return vals[idx], idx - (n-1)

def slsqp_opt(h0, n, T, maxiter, deadline):
    dx = 2.0 / n
    target = n / 2.0
    
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        scaled = corr * T
        m = np.max(scaled)
        return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx

        return np.sum(x) - target

                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'ftol': 1e-15, 'maxiter': maxiter})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except:
        pass
    return h0

def subgrad_momentum(h0, n, deadline, n_iters=20000):
    """Momentum subgradient with exact max."""
    h = h0.copy()
    m = np.zeros(n)  # momentum
    beta = 0.9
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = 0.01
    
    for it in range(n_iters):
        if time.time() > deadline:
            break
        
        c5_curr, lag = get_worst_lag(h, n)
        if c5_curr < best_c5:
            best_c5 = c5_curr
            best_h = h.copy()

        m = beta * m + (1 - beta) * g
        m_norm = np.linalg.norm(m)
        if m_norm > 1e-10:
            m = m / m_norm  # normalize
        
        h_new = h - lr * m
        h_new = project_feasible(h_new, n)
        
        # Adaptive lr: if we made progress, keep or increase; else decrease
        c5_new = evaluate_c5(h_new)
        if c5_new < c5_curr:
            lr = min(lr * 1.05, 0.05)
            h = h_new
        else:
            lr = lr * 0.7
            h = h  # stay put
    
    return best_h, best_c5

Actually, let me think about this more carefully. The subgradient of f(h) = max_k L_k(h) at a point h is:
∂f(h) = conv{∇L_k(h) : k ∈ K*}

where K* is the set of active lags (those achieving the max).

If there's a unique worst lag, the subgradient is just ∇L_{k*}(h).

The subgradient method with decreasing step sizes t_k = c/√k is guaranteed to converge, but slowly. For our purposes, we want to get close to a local optimum in finite time.

A better approach: **Propagatingthe subgradient with restarts.**

- Use a fixed step size
- Every T iterations, check if we've made progress
- If not, restart from the best point with a different step size

Or: **Mirror descent** with the appropriate Bregman divergence (KL divergence for the simplex constraint, squared distance for box constraints).

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(300):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def grad_exact_lag(h, n, lag):
    dx = 2.0 / n
    grad = np.zeros(n)
    if lag == 0:
        grad[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        grad[:n - lag] += (1.0 - h[lag:])
        grad[lag:] += (-h[:n - lag])
        grad *= dx
    else:
        l = -lag
        grad[:n - l] += (-h[l:])
        grad[l:] += (1.0 - h[:n - l])
        grad *= dx
    return grad

def get_worst_lag(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    idx = np.argmax(vals)
    return vals[idx], idx - (n - 1)

def slsqp_opt(h0, n, T, maxiter, deadline):
    target = n / 2.0
    dx = 2.0 / n
    def obj(x):
        corr = np.correlate(x, 1.0 - x, mode='full')
        scaled = corr * T
        m = np.max(scaled)
        return (m + np.log(np.sum(np.exp(scaled - m)))) / T * dx
    def con_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(obj, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': con_eq}],
                       options={'ftol': 1e-15, 'maxiter': maxiter})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return h0

def subgrad_exact(h0, n, deadline, max_iters=30000):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = 0.005
    momentum = np.zeros(n)
    beta = 0.85
    stalled = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c5_curr, lag = get_worst_lag(h, n)
        if c5_curr < best_c5 - 1e-12:
            best_c5 = c5_curr
            best_h = h.copy()
            stalled = 0
        else:
            stalled += 1
            if stalled > 500:
                lr *= 0.5
                momentum *= 0.5
                stalled = 0
                if lr < 1e-5:
                    break
        g = grad_exact_lag(h, n, lag)
        momentum = beta * momentum + (1 - beta) * g
        step = lr * momentum
        step_norm = np.linalg.norm(step)
        if step_norm > 1.0:
            step = step / step_norm
        h_new = h - step
        h_new = project_feasible(h_new, n)
        c5_new = evaluate_c5(h_new)
        if c5_new <= c5_curr + 1e-12:
            h = h_new
        else:
            h = h
            lr *= 0.9
    return best_h, best_c5

def coord_cyclical(h0, n, deadline, n_sweeps=200):
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    lr = 0.02
    for sweep in range(n_sweeps):
        if time.time() > deadline:
            break
        for i in range(n):
            if time.time() > deadline:
                break
            c5_curr, lag = get_worst_lag(h, n)
            if c5_curr < best_c5:
                best_c5 = c5_curr
                best_h = h.copy()
            g = grad_exact_lag(h, n, lag)
            h[i] -= lr * g[i]
            h[i] = np.clip(h[i], 0.0, 1.0)
        h = project_feasible(h, n)
    return best_h, best_c5

def fourier_opt(M, n, deadline, h_init):
    x = np.linspace(0, 2, n, endpoint=False)
    basis = np.zeros((n, 2 * M))
    for m in range(1, M + 1):
        basis[:, 2 * (m - 1)] = np.cos(m * np.pi * x)
        basis[:, 2 * (m - 1) + 1] = np.sin(m * np.pi * x)
    target_init = h_init - 0.5
    params0, _, _, _ = np.linalg.lstsq(basis, target_init, rcond=None)
    
    def build_h(params):
        return 0.5 + basis @ params
    
    def obj(params):
        h = build_h(params)
        pen = 0.0
        if h.min() < 0:
            pen += (h.min() ** 2) * 100
        if h.max() > 1:
            pen += (h.max() - 1) ** 2 * 100
        h_c = np.clip(h, 0, 1)
        h_p = project_feasible(h_c, n)
        return evaluate_c5(h_p) + pen
    
    best_params = params0.copy()
    best_val = obj(params0)
    rng = np.random.RandomState(123)
    for trial in range(15):
        if time.time() > deadline:
            break
        p = params0 + rng.randn(2 * M) * (0.03 if trial > 0 else 0.0)
        p[0] = 0.0  # pin mode-1 amplitude to help
        try:
            res = minimize(obj, p, method='Nelder-Mead',
                          options={'maxiter': 3000, 'xatol': 1e-9, 'fatol': 1e-11})
            if res.fun < best_val:
                best_val = res.fun
                best_params = res.x.copy()
        except Exception:
            pass
    h_best = build_h(best_params)
    h_best = project_feasible(np.clip(h_best, 0, 1), n)
    return h_best, evaluate_c5(h_best)

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)
    
    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)

    n1 = len(h_start)
    p1_deadline = min(t0 + 80, deadline)

    c5_s = evaluate_c5(h_s)
    if c5_s < best_c5:
        best_c5 = c5_s
        best_h = h_s.copy()
    
    # Phase 2: Upsample to n=500, SLSQP
    n2 = 500
    p2_deadline = min(t0 + 300, deadline)
    if time.time() < p2_deadline:
        h_hi = interp_to(best_h, len(best_h), n2)
        h_hi = project_feasible(h_hi, n2)
        for T in [300, 600]:
            if time.time() > p2_deadline:
                break
            h_new = slsqp_opt(h_hi, n2, T=T, maxiter=300, deadline=min(time.time() + 80, p2_deadline))
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
            h_hi = h_new
    
    # Phase 3: Exact subgradient at n=500
    n3 = len(best_h)

    if time.time() < p3_deadline:
        h_sg, c5_sg = subgrad_exact(best_h, n3, deadline=p3_deadline)
        if c5_sg < best_c5:
            best_c5 = c5_sg
            best_h = h_sg.copy()
    
    # Phase 4: SLSQP polish on the subgradient result
    if time.time() < deadline - 100:
        for T in [500, 1000]:
            if time.time() > deadline - 50:

            h_pol = slsqp_opt(best_h, n3, T=T, maxiter=200, deadline=min(time.time() + 60, deadline - 30))
            c5_pol = evaluate_c5(h_pol)
            if c5_pol < best_c5:
                best_c5 = c5_pol
                best_h = h_pol.copy()

    if time.time() < deadline - 60:
        for M in [4, 6, 10]:
            if time.time() > deadline - 30:

            if c5_f < best_c5:
                best_c5 = c5_f
                best_h = h_f.copy()

    n6 = len(best_h)

    # Phase 7: Final subgradient polish

        h_final, c5_final = subgrad_exact(best_h, len(best_h), deadline=deadline)

            best_c5 = c5_final
            best_h = h_final.copy()
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
