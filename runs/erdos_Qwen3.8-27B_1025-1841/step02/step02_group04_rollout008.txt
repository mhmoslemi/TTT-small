import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n, endpoint=False)
    new_x = np.linspace(0.0, 2.0, new_n, endpoint=False)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

def active_subgrad(h, top_k=8):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    k_idx = np.argpartition(corr, -top_k)[-top_k:]
    grad = np.zeros(n)
    for ki in k_idx:
        shift = ki - (n - 1)
        if shift >= 0:
            grad[shift:] += (1.0 - h[:n - shift])
            grad[:n - shift] += (-h[shift:])
        else:
            s = -shift
            grad[:n - s] += (1.0 - h[s:])
            grad[s:] += (-h[:n - s])
    return grad * dx / top_k

def gd_momentum(h, n_iter, lr, momentum, top_k, check_every=20):
    n = len(h)
    h = h.astype(float).copy()
    v = np.zeros(n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    stale = 0
    cur_lr = lr
    for it in range(n_iter):
        grad = active_subgrad(h, top_k)
        v = momentum * v - cur_lr * grad
        h = h + v
        h = make_feasible(h, n)
        if (it + 1) % check_every == 0:
            c5 = compute_c5(h)
            if c5 < best_c5 - 1e-9:
                best_c5 = c5
                best_h = h.copy()
                stale = 0
            else:
                stale += 1
            if stale >= 8:
                v *= 0.3
                cur_lr *= 0.6
                stale = 0
                if cur_lr < 0.002:
                    break
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def binary_local_search(h, max_passes=40, tries_per_pass=2000, kicks=5, kick_size=3, seed=42):
    n = len(h)
    rng = np.random.RandomState(seed)
    h = (h > 0.5).astype(float)
    target = n // 2
    n_ones = int(np.sum(h))
    if n_ones > target:
        idx_ones = np.where(h == 1)[0]
        to_remove = rng.choice(idx_ones, n_ones - target, replace=False)
        h[to_remove] = 0.0
    elif n_ones < target:
        idx_zeros = np.where(h == 0)[0]
        to_add = rng.choice(idx_zeros, target - n_ones, replace=False)
        h[to_add] = 1.0
    h = make_feasible(h, n)
    best_c5 = compute_c5(h)
    kick_count = 0
    for p in range(max_passes):
        improved = False
        n_imp = 0
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        if len(ones_idx) == 0 or len(zeros_idx) == 0:
            break
        for tt in range(tries_per_pass):
            i1 = ones_idx[rng.randint(len(ones_idx))]
            i2 = zeros_idx[rng.randint(len(zeros_idx))]
            if i1 == i2:
                continue
            h[i1] = 0.0
            h[i2] = 1.0
            c5_new = compute_c5(h)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                improved = True
                n_imp += 1
                ones_idx = np.where(h == 1)[0]
                zeros_idx = np.where(h == 0)[0]
            else:
                h[i1] = 1.0
                h[i2] = 0.0

            kick_count += 1
            if kick_count > kicks:
                break

                oi = np.where(h == 1)[0]
                zi = np.where(h == 0)[0]

                    break

                ib = zi[rng.randint(len(zi))]
                h[ia] = 0.0
                h[ib] = 1.0
            h = make_feasible(h, n)

            kick_count = 0
    return h, best_c5

def thue_morse(n):
    tm = np.zeros(n, dtype=int)
    for i in range(1, n):
        tm[i] = tm[i >> 1] ^ (i & 1)
    return (1.0 - tm).astype(float)

def softmax_obj(h, T, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    sm = np.sum(np.exp(T * shifted))
    if sm == 0 or not np.isfinite(sm):
        return mc * dx
    return (mc + np.log(sm) / T) * dx

def slsqp_opt(h, n, T, maxiter=800):
    dx = 2.0 / n
    def obj(x):
        return softmax_obj(x, T, n)
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n / 2.0,
             'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(obj, h, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-14})
        return make_feasible(res.x, n)
    except:
        return make_feasible(h, n)
