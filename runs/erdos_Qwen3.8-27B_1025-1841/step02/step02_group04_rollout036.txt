    h = np.clip(h, 0.0, 1.0)

    t_var = cp.Variable()

    z_vars = {}

        i_end = min(n, n - d)  # exclusive
        valid_i = np.arange(i_start, i_end)

            continue

        z_vars[d] = z_d

            constraints.append(z_d[idx] >= h_var[i] - h_var[j])
            constraints.append(z_d[idx] >= 0)

def do_flip(corr, h, pos, n):
    """Flip h[pos] and update corr in place. Returns new h."""
    N = 2*n - 1
    delta = 1.0 - 2.0 * h[pos]
    
    # Affected lags: d from -min(pos, n-1-pos) to max(n-1-pos, pos)
    # Actually: d where 0≤pos+d<n OR 0≤pos-d<n
    # d range: from min(-pos, pos-n+1) to max(n-1-pos, pos)
    d_start = min(-pos, pos - n + 1)
    d_end = max(n - 1 - pos, pos)
    
    # For each affected d, compute delta_corr
    # delta_corr[d] = delta * (1-h[pos+d]) [if 0≤pos+d<n] - delta * h[pos-d] [if 0≤pos-d<n]
    
    d_arr = np.arange(d_start, d_end + 1)
    k_arr = n - 1 + d_arr  # correlation index
    
    deltas = np.zeros(len(d_arr))
    
    # First term: delta * (1-h[pos+d]) for 0≤pos+d<n
    valid1 = (pos + d_arr >= 0) & (pos + d_arr < n)
    if np.any(valid1):
        deltas[valid1] += delta * (1.0 - h[pos + d_arr[valid1]])
    
    # Second term: -delta * h[pos-d] for 0≤pos-d<n
    valid2 = (pos - d_arr >= 0) & (pos - d_arr < n)
    if np.any(valid2):
        deltas[valid2] -= delta * h[pos - d_arr[valid2]]
    
    corr[k_arr] += deltas
    h[pos] += delta
    return h, corr
```

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float).copy(), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

            constraints.append(z <= 1 - h_var[j])
            constraints.append(z >= h_var[i] - h_var[j])
            constraints.append(z >= 0)
    
    # Group z by d and add overlap constraints
    # For each d: sum of z[d,i] <= t
    from collections import defaultdict
    d_groups = defaultdict(list)
    for idx, d in enumerate(d_list):
        d_groups[d].append(idx)
    
    for d, indices in d_groups.items():
        z_sum = cp.sum([z_list[i] for i in indices])
        constraints.append(z_sum <= t_var)
    
    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    
    for solver in ['CLARABEL', 'SCS', 'GLPK', 'CBC']:
        try:
            prob.solve(solver=solver, **({'max_time': timeout} if solver != 'CLARABEL' else {}))
            if h_var.value is not None:
                h_sol = np.array(h_var.value).flatten()
                return h_sol, float(t_var.value) if t_var.value is not None else None
        except Exception:
            continue
    
    return None, None

def efficient_flip_update(corr, h, pos, n):
    """Update correlation after flipping h[pos]. Returns updated h."""
    N = 2 * n - 1
    delta = 1.0 - 2.0 * h[pos]
    
    d_start = min(-pos, pos - n + 1)
    d_end = max(n - 1 - pos, pos)
    
    d_arr = np.arange(d_start, d_end + 1)
    k_arr = n - 1 + d_arr
    
    # Filter valid k indices
    valid = (k_arr >= 0) & (k_arr < N)
    k_arr = k_arr[valid]
    d_arr = d_arr[valid]
    
    changes = np.zeros(len(d_arr))
    
    valid1 = (pos + d_arr >= 0) & (pos + d_arr < n)
    if np.any(valid1):
        changes[valid1] += delta * (1.0 - h[pos + d_arr[valid1]])
    
    valid2 = (pos - d_arr >= 0) & (pos - d_arr < n)
    if np.any(valid2):
        changes[valid2] -= delta * h[pos - d_arr[valid2]]
    
    corr[k_arr] += changes
    h[pos] += delta
    return h, corr

def binary_local_search(h_bin, max_iter=5000, top_k=5):
    """Local search on binary h: flip bits to reduce max correlation."""
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    N = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    max_corr = np.max(corr)
    best_h = h.copy()
    best_max = max_corr
    
    for it in range(max_iter):
        improved = False
        # Try each position
        for pos in range(n):
            h_try = h.copy()
            h_try, corr_try = efficient_flip_update(corr.copy(), h.copy(), pos, n)
            new_max = np.max(corr_try)
            if new_max < max_corr - 1e-10:
                h, corr = h_try, corr_try
                max_corr = new_max
                improved = True
                if max_corr < best_max:
                    best_max = max_corr
                    best_h = h.copy()
                break
        if not improved:
            # Try random flips (restart-like)
            break
    
    return best_h, best_max * dx
```

Hmm, the binary local search as written copies the entire corr array for each position tried, which is O(n) per position and O(n²) per iteration. For n=320 and 5000 iterations, that's 320² * 5000 = 500M operations. In numpy, each operation is vectorized, so it should be faster, but still might be slow.

Let me optimize: instead of trying all positions in order, let me use a smarter heuristic:
1. Find the lag with max correlation
2. Only try flips at positions that are "1" (since flipping a "1" to "0" at a position that contributes to the max lag is more likely to help)
3. Limit to top-n positions by their contribution to the max lag

Actually, let me think about this differently. For the max lag d*, the correlation is:
corr[d*] = sum_i h[i] * (1-h[i+d*])

To reduce this, we want to either:
- Set h[i] = 0 for some i where h[i]=1 and h[i+d*]=0
- Set h[i+d*] = 1 for some i where h[i]=1 and h[i+d*]=0

But we need to maintain sum(h) = n/2, so we can't just flip. We need to flip one 1→0 and another 0→1.

This is a 2-swap local search. For each pair of positions to swap, the effect on the correlation is complex.

Alternatively, I can allow temporary infeasibility (sum ≠ n/2) and project back. Or I can do single flips but track the sum.

Actually, the simplest approach: do single flips, but after each flip, the sum changes by ±1. We can allow this temporarily and fix it at the end. Or we can do pairs of flips (1→0 and 0→1 simultaneously).

For a 2-swap (flip pos1 from 1→0 and pos2 from 0→1):
The effect on corr is the sum of the individual effects. We can compute this efficiently.

Let me implement a 2-swap local search:

```python
def binary_2swap_search(h_bin, max_iter=10000):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    N = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    max_corr = np.max(corr)
    best_h = h.copy()
    best_max = max_corr
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for it in range(max_iter):
        improved = False
        # For each pair (one_pos, zero_pos), compute the effect
        # This is O(n²/4) per iteration which is too slow for n=320
        # Instead, for each one_pos, find the best zero_pos to swap with
        
        # Strategy: focus on the max lag
        max_lag = np.argmax(corr)
        d_max = max_lag - (n - 1)
        
        # Find positions contributing to the max
        # corr[max_lag] = sum_i h[i] * (1-h[i+d_max])
        # Contributing positions: i where h[i]=1 and h[i+d_max]=0
        
        contrib = []
        for i in range(n):
            j = i + d_max
            if 0 <= j < n and h[i] == 1 and h[j] == 0:
                contrib.append((i, j))
        
        # Try swapping: set h[i]=0, h[j]=1 (removes one contributor)
        # Or set h[i]=0 and h[k]=1 for some other k
        # Simplest: for each (i,j) in contrib, try flipping i (1→0) and j (0→1)
        
        best_improve = 0
        best_pair = None
        
        for i, j in contrib[:20]:  # limit to top 20
            # Flip i from 1→0 and j from 0→1
            # Compute new max
            h[i] = 0
            h[j] = 1
            new_corr = np.correlate(h, 1.0 - h, mode='full')
            new_max = np.max(new_corr)
            h[i] = 1
            h[j] = 0
            
            improve = max_corr - new_max
            if improve > best_improve:
                best_improve = improve
                best_pair = (i, j)
        
        if best_pair is not None:
            i, j = best_pair
            h[i] = 0
            h[j] = 1
            corr = np.correlate(h, 1.0 - h, mode='full')
            max_corr = np.max(corr)
            improved = True
            if max_corr < best_max:
                best_max = max_corr
                best_h = h.copy()
        
        if not improved:
            break
    
    return best_h, best_max * dx
```

This is still O(len(contrib) * n) per iteration (due to np.correlate). For n=320 and ~160 contributors, that's 160 * 320 * 2 = 100K per iteration. With 10000 iterations, that's 10^9. Might be too slow in Python.

Let me use the efficient update instead:

```python
def compute_flip_delta(corr, h, pos, n):
    """Compute the change in corr if we flip h[pos]. Returns (delta_array, valid_indices)."""
    N = 2 * n - 1
    delta_val = 1.0 - 2.0 * h[pos]
    
    d_start = min(-pos, pos - n + 1)
    d_end = max(n - 1 - pos, pos)
    d_arr = np.arange(d_start, d_end + 1)
    k_arr = n - 1 + d_arr
    
    valid = (k_arr >= 0) & (k_arr < N)
    k_arr = k_arr[valid]
    d_arr = d_arr[valid]
    
    changes = np.zeros(len(d_arr))
    valid1 = (pos + d_arr >= 0) & (pos + d_arr < n)
    if np.any(valid1):
        changes[valid1] += delta_val * (1.0 - h[pos + d_arr[valid1]])
    valid2 = (pos - d_arr >= 0) & (pos - d_arr < n)
    if np.any(valid2):
        changes[valid2] -= delta_val * h[pos - d_arr[valid2]]
    
    return k_arr, changes

def binary_2swap_efficient(h_bin, max_iter=20000):
    n = len(h_bin)
    h = h_bin.astype(float).copy()
    dx = 2.0 / n
    N = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode='full').copy()
    max_corr = np.max(corr)
    best_h = h.copy()
    best_max = max_corr
    
    rng = np.random.RandomState(42)
    
    for it in range(max_iter):
        # Focus on top lags
        top_lags = np.argsort(corr)[-10:][::-1]
        
        improved = False
        for max_lag in top_lags[:3]:
            d_max = int(max_lag) - (n - 1)
            # Find contributing positions
            i_start = max(0, -d_max)
            i_end = min(n, n - d_max)
            
            contrib = []
            for i in range(i_start, i_end):
                j = i + d_max
                if h[i] == 1 and h[j] == 0:
                    contrib.append((i, j))
            
            if len(contrib) == 0:
                continue
            
            # Try a few random swaps from contrib
            if len(contrib) > 5:
                sampled = rng.choice(len(contrib), 5, replace=False)
            else:
                sampled = range(len(contrib))
            
            best_improve = 1e-10
            best_swap = None
            
            for idx in sampled:
                i, j = contrib[idx]
                # Compute effect of flipping i (1→0) and j (0→1)
                k1, c1 = compute_flip_delta(corr, h, i, n)
                k2, c2 = compute_flip_delta(corr, h, j, n)
                
                # Apply both changes
                new_corr = corr.copy()
                new_corr[k1] += c1
                # For j, h[j] was 0, now becomes 1 after i flip (no change to h[j] from i flip)
                # Wait, we need to be careful: the delta for j's flip assumes h[j]=0
                # After flipping i, h is unchanged at j (since i≠j), so the delta for j is still valid
                new_corr[k2] += c2
                
                new_max = np.max(new_corr)
                improve = max_corr - new_max
                if improve > best_improve:
                    best_improve = improve
                    best_swap = (i, j)
            
            if best_swap is not None:
                i, j = best_swap
                h[i] = 0
                h[j] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                max_corr = np.max(corr)
                improved = True
                if max_corr < best_max:
                    best_max = max_corr
                    best_h = h.copy()
                break
        
        if not improved:
            # Try random perturbation
            if it % 100 == 0:
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                i = rng.choice(ones)
                j = rng.choice(zeros)
                h[i] = 0
                h[j] = 1
                corr = np.correlate(h, 1.0 - h, mode='full')
                max_corr = np.max(corr)
            else:
                break
    
    return best_h, best_max * dx
```

This should be faster since we only evaluate ~5-15 candidate swaps per top lag, and each evaluation is O(n) for the delta computation + O(n) for the max.

1. LP for structural guidance (small n)
2. Binary local search (efficient 2-swap)
3. Soft gradient descent (improved version)
4. Multi-resolution upsample and optimize
