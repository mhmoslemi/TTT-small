import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n), corr

def optimize_adaptive_smooth(h, n, time_limit, tau_start=5.0, tau_end=200.0, max_iters=100000):
    """Smooth optimization with adaptive tau that increases over time."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5, _ = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    lr = 0.3
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    best_stall = 0
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        if it % 5000 == 0:
            target_c5, corr = compute_c5(h)
            if target_c5 < best_c5 - 1e-14:
                best_c5 = target_c5
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        
        # Adaptive tau
        progress = min(1.0, it / (max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** progress
        
        # Compute smooth objective gradient
        corr_full = np.correlate(h, 1.0 - h, mode='full')
        max_c = corr_full.max()
        shifted = np.clip(tau * (corr_full - max_c), -400, 400)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
        
        grad = np.zeros(n)
        for k in range(2 * n - 1):
            pk = p[k]
            if pk < 1e-10:
                continue
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[j0:j1] += pk * (1.0 - h[idx - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[j0b:j1b] -= pk * h[idx + sh]
        grad *= dx

            if target_c5 < best_c5:
                stall = 0
            else:
                stall += 1
            if stall > 3:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
        
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5

def smooth_grad_fft(h, tau):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation
    corr = np.correlate(h, 1.0 - h, mode='full')  # length 2n-1
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    p = exp_c / exp_c.sum()  # length 2n-1
    
    # FFT-based gradient
    N = 1
    while N < 3 * n:
        N *= 2
    
    A = np.zeros(N, dtype=complex)
    A[:n] = 1.0 - h
    
    B = np.zeros(N, dtype=complex)
    B[:2*n-1] = p
    
    H = np.zeros(N, dtype=complex)
    H[:n] = h
    
    # first_term[j] = sum_s p[s] * (1-h[j+s]) for valid j+s
    # = cross-correlation of B and A at shift j
    # = IFFT(conj(FFT(B)) * FFT(A))[j] for j=0..n-1
    FFT_A = np.fft.fft(A)
    FFT_B = np.fft.fft(B)
    first_term = np.fft.ifft(np.conj(FFT_B) * FFT_A).real[:n]
    
    # second_term[j] = sum_s p[s] * h[j-s] for valid j-s
    # = convolution of B and H at position j
    # = IFFT(FFT(B) * FFT(H))[j] for j=0..n-1
    FFT_H = np.fft.fft(H)
    second_term = np.fft.ifft(FFT_B * FFT_H).real[:n]
    
    grad = dx * (first_term - second_term)
    obj = dx * (max_c + np.log(exp_c.sum()) / tau)
    return obj, grad, p, corr

The best we can do is minimize the maximum deviation from 1/4, which is what we're doing.

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def fft_correlate(a, b):
    """Full cross-correlation via FFT."""
    n = len(a)
    m = len(b)
    N = 1
    while N < n + m - 1:
        N *= 2
    fa = np.fft.rfft(a, N)
    fb = np.fft.rfft(b, N)
    corr = np.fft.irfft(fa * np.conj(fb), N)
    # np.correlate(a, b, 'full')[k] = sum_i a[i] * b[i + k - (m-1)]
    # FFT gives: ifft(FFT(a) * conj(FFT(b)))[k] = sum_i a[i+k] * b[i] (circular, but with zero padding = linear)
    # Actually: ifft(FFT(a) * conj(FFT(b))) gives the cross-correlation with a specific indexing.
    # Let me just use np.correlate for correctness and optimize later if needed.
    return None

def compute_c5_fft(h):
    """Compute C5 using FFT for the correlation."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    N = 1
    while N < 2 * n - 1:
        N *= 2
    fh = np.fft.rfft(h, N)
    f1mh = np.fft.rfft(one_minus_h, N)
    # conj(FFT(b)) * FFT(a) gives correlation of a and b
    # Actually: np.correlate(a, b, 'full')[k] = sum_i a[i] * conj(b[i+k-(len(b)-1)])
    # For real signals: sum_i a[i] * b[i+k-(len(b)-1)]
    # FFT: ifft(FFT(a) * conj(FFT(b))) gives this
    corr = np.fft.irfft(fh * np.conj(f1mh), N)[:2*n-1]
    return corr.max() * dx, corr

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)


def fft_correlation(h):
    n = len(h)
    N = 1
    while N < 2 * n - 1:
        N *= 2
    A_pad = np.zeros(N)
    A_pad[:n] = 1.0 - h
    H_pad = np.zeros(N)
    H_pad[:n] = h
    FA = np.fft.rfft(A_pad)
    FH = np.fft.rfft(H_pad)
    D = np.fft.irfft(np.conj(FH) * FA, N)
    indices = (np.arange(2 * n - 1) - (n - 1)) % N
    return D[indices]


def fft_smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    C_full = fft_correlation(h)
    max_c = C_full.max()
    shifted = np.clip(tau * (C_full - max_c), -400, 400)
    exp_c = np.exp(shifted)
    esum = exp_c.sum()
    p = exp_c / esum

    N = 1
    while N < 3 * n:
        N *= 2

    A_pad = np.zeros(N)
    A_pad[:n] = 1.0 - h
    H_pad = np.zeros(N)
    H_pad[:n] = h
    Q_pad = np.zeros(N)
    Q_pad[:2 * n - 1] = p

    FA = np.fft.rfft(A_pad)
    FH = np.fft.rfft(H_pad)
    FQ = np.fft.rfft(Q_pad)

    E = np.fft.irfft(np.conj(FA) * FQ, N)
    t1_indices = (np.arange(n) - (n - 1)) % N
    T1 = E[t1_indices]

    conv_result = np.fft.irfft(FH * FQ, N)
    T2 = conv_result[n - 1:n - 1 + n]

    grad = dx * (T1 - T2)
    obj = dx * (max_c + np.log(esum) / tau)
    return obj, grad


def optimize_fft_smooth(h, n, tau, time_limit, lr_init=0.4):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.9
    stall = 0
    it = 0
    while time.time() - start < time_limit:
        obj, grad = fft_smooth_grad(h, tau)
        if it % 500 == 0:
            c5_cur = compute_c5(h)
            if c5_cur < best_c5 - 1e-12:
                best_c5 = c5_cur
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
            if stall > 4:
                lr *= 0.7
                v *= 0.5
                stall = 0
                if lr < 1e-10:
                    break
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
        it += 1
    c5_final = compute_c5(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()
    return best_h, best_c5


def optimize_subgrad_fft(h, n, time_limit, lr_init=0.005):
