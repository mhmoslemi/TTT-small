    return np.interp(x_new, x_old, h, period=2.0)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def compute_c5(h):
    n = len(h)
    return float(compute_corr(h).max()) * (2.0 / n)

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -400, 400)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-12:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx2 = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx2 + sh]
    grad *= dx
    obj = dx * (max_c + np.log(max(s, 1e-300)) / tau)
    return obj, grad

def grad_h_smooth(h, tau, topk=None):
    """Compute gradient of smooth max (or top-k average) of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    if topk is not None and topk < 2*n-1:
        # Use top-k average
        idx = np.argpartition(corr, -topk)[-topk:]
        p = np.zeros(2*n-1)
        p[idx] = 1.0/topk
    else:
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -400, 400)
        exp_c = np.exp(shifted)
        s = exp_c.sum()
        p = exp_c / s
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[j0:j1] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx2 = np.arange(j0b, j1b)
            grad[j0b:j1b] -= pk * h[idx2 + sh]
    grad *= dx
    return grad

def optimize_stage(h, n, tau, time_limit, lr_init=0.3, mode='nesterov'):
    """Optimize using nesterov or adam with smooth max."""
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    
    if mode == 'nesterov':
        t_val = 1.0
        h_prev = h.copy()
        y = h.copy()
        lr = lr_init
        for it in range(500000):
            if time.time() - start > time_limit:
                break
            t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_val * t_val)) / 2.0
            y = h + ((t_val - 1.0) / t_new) * (h - h_prev)
            obj, grad = smooth_obj_grad(y, tau)
            h_new = project_to_feasible(y - lr * grad, n)
            h_prev = h.copy()
            h = h_new
            t_val = t_new
            # Decay lr
            lr = lr_init * (1 + it * 0.0001) ** (-0.5)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    else:  # adam
        m = np.zeros(n)
        v = np.zeros(n)
        beta1, beta2 = 0.9, 0.999
        lr = lr_init
        for it in range(5
00000):
            if time.time() - start > time_limit:
                break
            obj, grad = smooth_obj_grad(h, tau)
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(it+1))
            v_hat = v / (1 - beta2**(it+1))
            h_new = project_to_feasible(h - lr * m_hat / (np.sqrt(v_hat) + 1e-8), n)
            h = h_new
            lr = lr_init / (1 + it * 0.001)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return best_h, best_c5

def subgrad_refine(h, n, time_limit, lr_init=0.005, topk=3):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = lr_init
    v = np.zeros(n)
    beta = 0.85
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = compute_corr(h)
        c5_cur = corr.max() * dx
        if c5_cur < best_c5 - 1e-15:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            lr *= 0.7
            v *= 0.5
            stall = 0
            if lr < 1e-10:
                break
        grad = grad_h_smooth(h, tau=1.0, topk=topk)
        v = beta * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5

def binary_polish(h, n, time_limit, max_rounds=30):
    dx = 2.0 / n
    h = h.copy()
    best_c5 = compute_c5(h)
    start = time.time()
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
        improved = False
        perm = np.random.permutation(n)
        for m in perm:
            if time.time() - start > time_limit:
                break
            orig = h[m]
            candidates = []
            if orig > 0.1:
                candidates.append(0.0)
            if orig < 0.9:
                candidates.append(1.0)
            if not candidates:
                continue
            for nv in candidates:
                h[m] = nv
                h_cand = project_to_feasible(h, n)
                c5_cand = compute_c5(h_cand)
                if c5_cand < best_c5 - 1e-14:
                    h = h_cand
                    best_c5 = c5_cand
                    improved = True
                    break
            h[m] = orig
        if not improved:
            break
    return h, best_c5
