import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))

def compute_grad_tops(h, p, top_k):
    """Compute gradient using only top-k shifts (where p[k] is largest)."""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    idx = np.argsort(p)[::-1][:top_k]
    for k in idx:
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        # dc_k/dh[j] = (1-h[j+sh]) * 1_{max(0,-sh)<=j<min(n,n-sh)}
        #           - h[j-sh] * 1_{max(0,sh)<=j<min(n,n+sh)}
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            grad[j0a:j1a] += pk * (1.0 - h[j0a+sh:j1a+sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b-sh:j1b-sh]
    grad *= dx
    return grad

def optimize_adam(h_in, n, tau, time_limit, lr=0.01, top_k=30, bimodal_lambda=0.0):
    """Adam-based optimization of smooth max C5."""
    h = project_to_feasible(h_in, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2 = 0.9, 0.999
    t = 0
    stall = 0
    
    for it in range(50000):
        if time.time() - start > time_limit:
            break
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * (2.0/n)
        
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 2000:
            lr *= 0.7
            stall = 0
            if lr < 1e-13:
                break
        
        # Smooth max gradient
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -500, 0)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()
        
        grad = compute_grad_tops(h, p, top_k)
        
        # Bimodal regularization: minimize -sum h(1-h) => gradient = -(1-2h)
        if bimodal_lambda > 0:
            grad -= bimodal_lambda * (1.0 - 2.0 * h)
        
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        h = project_to_feasible(h_new, n)
    
    return best_h, best_c5

            stall = 0

        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b-sh:j1b-sh]

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n))


def compute_grad_tops(h, p, top_k):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    idx = np.argsort(p)[::-1][:top_k]
    for k in idx:
        pk = p[k]
        if pk < 1e-15:
            continue
        sh = k - n + 1
        j0a = max(0, -sh)
        j1a = min(n, n - sh)
        if j0a < j1a:
            grad[j0a:j1a] += pk * (1.0 - h[j0a + sh:j1a + sh])
        j0b = max(0, sh)
        j1b = min(n, n + sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b - sh:j1b - sh]
    grad *= dx
    return grad


def optimize_adam(h_in, n, tau, time_limit, lr=0.01, top_k=30, bimodal=0.0):
    h = project_to_feasible(h_in, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    m_v = np.zeros(n)
    v_v = np.zeros(n)
    b1, b2 = 0.85, 0.999
    t_step = 0
    stall = 0
    cur_lr = lr

    for it in range(100000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * (2.0 / n)
        if c5_now < best_c5 - 1e-14:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 1500:
            cur_lr *= 0.7
            stall = 0
            if cur_lr < 1e-13:
                break
        max_c = corr.max()
        shifted = np.clip(tau * (corr - max_c), -500.0, 0.0)
        exp_c = np.exp(shifted)
        p = exp_c / exp_c.sum()
        grad = compute_grad_tops(h, p, top_k)
        if bimodal > 0.0:
            grad = grad - bimodal * (1.0 - 2.0 * h)
        t_step += 1
        m_v = b1 * m_v + (1.0 - b1) * grad
        v_v = b2 * v_v + (1.0 - b2) * grad * grad
        mh = m_v / (1.0 - b1 ** t_step)
        vh = v_v / (1.0 - b2 ** t_step)
        h = project_to_feasible(h - cur_lr * mh / (np.sqrt(vh) + 1e-8), n)
    return best_h, best_c5
