def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    h = h + diff / n
    h = np.clip(h, 0.0, 1.0)
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def subgrad_at_shift(h, shift):
    n = len(h)
    s = abs(shift)
    if s >= n:
        return np.zeros(n)
    grad = np.zeros(n)
    if shift >= 0:
        grad[s:] += (1.0 - h[:n - s])
        grad[:n - s] += (-h[s:])
    else:
        grad[:n - s] += (1.0 - h[s:])
        grad[s:] += (-h[:n - s])
    return grad

def active_set_gd(h, n_iter=5000, top_k=8, lr_init=0.1, check_every=50):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_init
    for it in range(n_iter):
        frac = it / max(n_iter - 1, 1)
        lr_t = lr_init * (1.0 - 0.5 * frac)
        # Compute correlations
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Top K worst shifts
        top_idx = np.argsort(corr)[-top_k:]
        # Weight by proximity to max
        mc = corr.max()
        range_c = mc - corr[top_idx[0]] + 1e-10
        weights = 1.0 / range_c  # equal weight (since we take top K)
        # Actually, weight by how close to max
        w = np.exp(-0.5 * (mc - corr[top_idx]) / (range_c + 1e-10))
        w /= w.sum()
        # Gradient

        for ki in range(len(top_idx)):
            shift = top_idx[ki] - (n - 1)
            grad += w[ki] * subgrad_at_shift(h, shift)
        grad *= dx
        # Line search
        improved = False
        for alpha in [lr_t, lr_t * 0.5, lr_t * 0.25, lr_t * 0.1]:
            h_new = make_feasible(h - alpha * grad, n)

                h = h_new

                    best_h = h.copy()
                improved = True
                break
        if not improved:
            lr_t *= 0.5
        if (it + 1) % check_every == 0:
            c_chk = compute_c5(h)
            if c_chk < best_c5:
                best_c5 = c_chk
                best_h = h.copy()
    return best_h, best_c5

def upsample_h(h, new_n):
    old_n = len(h)
    if new_n == old_n:
        return h.copy()
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    h_new = np.interp(new_x, old_x, h)
    return make_feasible(h_new, new_n)

            h_g, c_g = active_set_gd(h_p, n_iter=2000, top_k=10, lr_init=0.08)
            if c_g < best_c5:
                best_c5, best_h = c_g, h_g.copy()
                print(f"  Perturb: C5={best_c5:.6f}", flush=True)
        print(f"P6 MultiStart: C5={best_c5:.6f} t={time.time()-start_time:.0f}s", flush=True)

    # Phase 7: Final fine-tuning (small steps, active set = top 5)
    if tl() > 60:
        h_f, c_f = active_set_gd(best_h, n_iter=4000, top_k=5, lr_init=0.03, check_every=80)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f.copy()
        print(f"P7 FinalGD: C5={best_c5:.6f} t={time.time()-start_time:.0f}s", flush=True)

    # Phase 8: Final SLSQP polish at high T
    n_pts = len(best_h)
    if n_pts <= 512 and tl() > 40:
        hx, cx = slsqp_refine(best_h, n_pts, 100.0, maxiter=300)
        if cx < best_c5:
            best_c5, best_h = cx, hx
        print(f"P8 FinalSLSQP: C5={best_c5:.6f} t={time.time()-start_time:.0f}s", flush=True)

    # Final validation
    best_h = make_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    print(f"DONE: n={len(best_h)}, C5={best_c5:.6f}, t={time.time()-start_time:.0f}s", flush=True)
    return best_h, best_c5, len(best_h)
