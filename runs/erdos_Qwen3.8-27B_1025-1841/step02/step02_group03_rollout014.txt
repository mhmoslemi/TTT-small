        else:
            perm = np.argsort(-h[idx0])
        for i in range(target - ones):
            h_bin[idx0[perm[i]]] = 1.0
    return h_bin

def upsample_h(h, new_n):
    old_n = len(h)
    old_x = np.linspace(0.0, 2.0, old_n)
    new_x = np.linspace(0.0, 2.0, new_n)
    return np.interp(new_x, old_x, h)

def thue_morse_binary(n, rng):
    t = np.array([bin(i).count('1') & 1 for i in range(n)], dtype=np.float64)
    ones = int(np.sum(t))
    target = n // 2
    if ones > target:
        idx1 = np.where(t == 1)[0]
        perm = rng.permutation(len(idx1))
        for i in range(ones - target):
            t[idx1[perm[i]]] = 0.0
    elif ones < target:
        idx0 = np.where(t == 0)[0]
        perm = rng.permutation(len(idx0))
        for i in range(target - ones):
            t[idx0[perm[i]]] = 1.0
    return t

def golden_ratio_binary(n, rng):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    target = n // 2
    positions = [int((k * phi) % n) for k in range(target)]
    h = np.zeros(n, dtype=np.float64)
    for p in positions:
        h[p % n] = 1.0
    if np.sum(h) != target:
        h = thue_morse_binary(n, rng)
    return h

def periodic_binary(n, rng, period):
    h = np.zeros(n, dtype=np.float64)
    pattern_len = period
    ones_per_period = pattern_len // 2
    base = np.zeros(pattern_len)
    base[:ones_per_period] = 1.0
    rng.shuffle(base)
    reps = n // pattern_len
    rem = n % pattern_len
    for r in range(reps):
        h[r*pattern_len:(r+1)*pattern_len] = base
    if rem > 0:
        h[reps*pattern_len:] = base[:rem]
    return to_binary(h, rng)

def binary_local_search(h, n, n_trials, rng, time_check, sa_temp=0.0):
    h = h.copy()
    curr_c = compute_c5(h)
    best_c = curr_c
    best_h = h.copy()
    n_accepted = 0

    for it in range(n_trials):
        if time_check():
            break
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        if len(zeros) == 0 or len(ones) == 0:
            break
        i = zeros[rng.randint(len(zeros))]
        j = ones[rng.randint(len(ones))]

        h[i] = 1.0
        h[j] = 0.0
        c = compute_c5(h)

        if c < curr_c - 1e-15:
            curr_c = c
            n_accepted += 1
            if c < best_c:
                best_c = c
                best_h = h.copy()
        elif sa_temp > 0.0:
            temp = sa_temp * (0.9998 ** it)
            if temp > 1e-15:
                delta = c - curr_c
                accept = rng.random() < np.exp(-delta / temp)
            else:
                accept = False
            if not accept:
                h[i] = 0.0
                h[j] = 1.0
        else:
            h[i] = 0.0
            h[j] = 1.0

    return best_h, best_c

def block_local_search(h, n, n_trials, rng, time_check):
    h = h.copy()
    curr_c = compute_c5(h)
    best_c = curr_c
    best_h = h.copy()

    for it in range(n_trials):
        if time_check():
            break
        blen = rng.randint(2, 7)
        if blen >= n // 2:
            blen = 2
        i = rng.randint(0, n - blen + 1)
        j = rng.randint(0, n - blen + 1)
        if abs(i - j) < blen:
            j = (j + n // 2) % n
            if j + blen > n:
                j = n - blen
        if abs(i - j) < blen:
            continue
        bi = h[i:i+blen].copy()
        bj = h[j:j+blen].copy()
        if not (np.all(bi == 0) and np.all(bj == 1)) and \
           not (np.all(bi == 1) and np.all(bj == 0)):
            continue
        h[i:i+blen] = bj
        h[j:j+blen] = bi
        c = compute_c5(h)
        if c < curr_c - 1e-15:
            curr_c = c
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[i:i+blen] = bi
            h[j:j+blen] = bj

    return best_h, best_c

def _obj_smooth(x, T):
    n = len(x)
    dx = 2.0 / n
    corr = np.correlate(x, 1.0 - x, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def continuous_refine(h_bin, n, rng, time_check):
    _T = 80.0
    h0 = h_bin.astype(np.float64).copy()
    h0 += rng.randn(n) * 0.05
    h0 = make_feasible(h0, n)

    best_c = compute_c5(h_bin)
    best_h = h_bin.copy()

    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: _con_fun(x, n), 'jac': lambda x: np.ones(n)}]

    try:
        res = minimize(lambda x: _obj_smooth(x, _T), h0, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': 300, 'ftol': 1e-12})
        c_res = compute_c5(res.x)
        if c_res < best_c:
            best_c = c_res
            best_h = make_feasible(res.x, n)
    except Exception:
        pass

    try:
        res2 = minimize(lambda x: _obj_smooth(x, 200.0), best_h, method='SLSQP',
                        bounds=bounds, constraints=cons,
                        options={'maxiter': 200, 'ftol': 1e-13})
        c_res2 = compute_c5(res2.x)
        if c_res2 < best_c:
            best_c = c_res2
            best_h = make_feasible(res2.x, n)
    except Exception:
        pass

    return best_h, best_c

def run(seed=42, budget_s=1300, **kwargs):
    start_time = time.time()
    rng = np.random.RandomState(seed)

    def remaining():
        return budget_s - (time.time() - start_time)

    def time_check():
        return (time.time() - start_time) > budget_s - 5.0

    N = 1024

    # Generate multiple starting points
    starts = []

    # Start 1: Parent thresholded
    try:
        h_parent = np.array(initial_h_values, dtype=np.float64).copy()
        h_up = upsample_h(h_parent, N)
        starts.append(('parent', to_binary(h_up, rng)))
    except Exception:
        pass

    # Start 2: Thue-Morse
    starts.append(('thue_morse', thue_morse_binary(N, rng)))

    # Start 3: Golden ratio
    starts.append(('golden', golden_ratio_binary(N, rng)))

    # Start 4: Random
    h_rand = np.zeros(N, dtype=np.float64)
    perm = rng.permutation(N)
    h_rand[perm[:N//2]] = 1.0
    starts.append(('random', h_rand))

    # Start 5: Periodic (period 16)
    starts.append(('periodic16', periodic_binary(N, rng, 16)))

    # Start 6: Periodic (period 10)
    starts.append(('periodic10', periodic_binary(N, rng, 10)))

    # Phase 1: Local search on each start (initial screening)
    best_c5 = np.inf
    best_h = None

    for name, h_start in starts:
        if time_check():
            break
        t_before = time.time()
        h_res, c_res = binary_local_search(h_start, N, 30000, rng, time_check)
        elapsed = time.time() - t_before
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res
            best_name = name

    # Phase 2: Extended local search on best
    if remaining() > 200:
        n_trials_phase2 = min(200000, int(remaining() * 0.4 / 0.002))
        h_res, c_res = binary_local_search(best_h, N, n_trials_phase2, rng, time_check)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res

    # Phase 3: Block moves
    if remaining() > 100:
        n_blocks = min(80000, int(remaining() * 0.2 / 0.003))
        h_res, c_res = block_local_search(best_h, N, n_blocks, rng, time_check)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res

    # Phase 4: Simulated annealing
    if remaining() > 150:
        n_sa = min(80000, int(remaining() * 0.15 / 0.002))
        h_res, c_res = binary_local_search(best_h, N, n_sa, rng, time_check, sa_temp=0.01)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res

    # Phase 5: Another round of local search on best
    if remaining() > 100:
        n_trials_p5 = min(100000, int(remaining() * 0.3 / 0.002))
        h_res, c_res = binary_local_search(best_h, N, n_trials_p5, rng, time_check)
        if c_res < best_c5:
            best_c5 = c_res
            best_h = h_res

    # Phase 6: Continuous refinement (SLSQP)
    if remaining() > 80:
        best_h, best_c5 = continuous_refine(best_h, N, rng, time_check)

    # Phase 7: Final local search if continuous improved
    if remaining() > 60:
        if np.all((best_h == best_h.astype(int).astype(np.float64))):
            pass
        else:
            h_bin_final = to_binary(best_h, rng)
            c_bin = compute_c5(h_bin_final)
            if c_bin >= best_c5:
                # Keep continuous
                pass
            else:
                best_h = h_bin_final
                best_c5 = c_bin
            n_trials_p7 = min(50000, int(remaining() * 0.4 / 0.002))
            h_res, c_res = binary_local_search(to_binary(best_h, rng), N, n_trials_p7, rng, time_check)
            if c_res < best_c5:
                best_c5 = c_res
                best_h = h_res

    # Final validation
    best_h = make_feasible(best_h, N)
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, N
