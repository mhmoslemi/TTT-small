```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_subgrad(h, lag):
    """Subgradient of r(lag) = sum_i h[i+lag]*(1-h[i]) with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        m = lag
        if m >= n:
            return grad
        # r = sum_{i=0}^{n-1-m} h[i+m]*(1-h[i])
        # d/dh[j]: (1-h[j-m]) if m <= j <= n-1 (j is in h[i+m] term)
        #          -h[j+m] if 0 <= j <= n-1-m (j is in (1-h[j]) term)
        if n - 1 - m >= 0:
            grad[0:n-m] += -(h[m:n])  # from -(h[j+m]) for j in [0, n-1-m]
            grad[m:n] += (1.0 - h[0:n-m])  # from (1-h[j-m]) for j in [m, n-1]
    else:
        m = -lag
        if m >= n:
            return grad
        # r = sum_{i=m}^{n-1} h[i+m]*(1-h[i]) = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
        # d/dh[j]: (1-h[j+m]) if 0 <= j <= n-1-m
        #          -h[j-m] if m <= j <= n-1
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += -(h[0:n-m])
    return grad

def top_lag_optimize(h, n_iters=5000, top_k=5, lr_start=0.5, lr_end=0.01):
    """Optimize by cycling through top-k lags."""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    for it in range(n_iters):
        frac = it / max(n_iters - 1, 1)
        lr = lr_start * (lr_end / lr_start) ** frac
        
        corr = np.correlate(h, 1.0 - h, mode='full')
        # Get top-k lags
        k_indices = np.argsort(corr)[-top_k:][::-1]
        
        # Cycle through them
        k_sel = k_indices[it % top_k]
        lag = k_sel - (n - 1)
        
        grad = lag_subgrad(h, lag)
        
        # Projected gradient step
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        
        # Check improvement
        c_new = compute_c5(h_new)
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
        else:
            # Try smaller step
            h_new2 = h - 0.5 * lr * grad
            h_new2 = make_feasible(h_new2, n)
            c_new2 = compute_c5(h_new2)
            if c_new2 < compute_c5(h):
                h = h_new2
                if c_new2 < best_c5:
                    best_c5 = c_new2
                    best_h = h.copy()
    
    return best_h, best_c5
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_subgrad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    m = abs(int(lag))
    if m >= n:
        return grad
    if lag >= 0:
        grad[0:n-m] += (-h[m:n])
        grad[m:n] += (1.0 - h[0:n-m])
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += (-h[0:n-m])
    return grad

def top_lag_descent(h, n_iters=5000, top_k=5, lr_start=0.8, decay=0.999, min_lr=0.01):
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    lr = lr_start
    stalled = 0
    for it in range(n_iters):
        corr = np.correlate(h, 1.0 - h, mode='full')
        k_idx = np.argsort(corr)[-top_k:][::-1]
        k_sel = int(k_idx[it % top_k])
        lag = k_sel - (n - 1)
        grad = lag_subgrad(h, lag)
        
        # Try step
        h_new = h - lr * grad
        h_new = make_feasible(h_new, n)
        c_new = compute_c5(h_new)
        
        if c_new < best_c5:
            best_c5 = c_new
            best_h = h_new.copy()
            h = h_new
            stalled = 0
        else:
            h = h_new  # still move, but track stall
            stalled += 1
            if stalled > 50:
                lr *= 0.95
                stalled = 0
        
        lr = max(lr * decay, min_lr)
    
    return best_h, best_c5

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.clip(shifted, -60.0/T, 0.0)
    logsum = mc + np.log(np.sum(np.exp(T * shifted))) / T
    return logsum * dx

def softmax_grad(h, T, top_k=30):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    shifted = np.clip(shifted, -60.0/T, 0.0)
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        lag = k_max - (n-1)
        return lag_subgrad(h, lag) * dx
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:]
    grad = np.zeros(n)
    for k_i in idx:
        w = weights[k_i]
        if w < 1e-8:
            break
        lag = k_i - (n-1)
```

```python
def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(np.asarray(h, dtype=float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def lag_subgrad(h, lag):
    n = len(h)
    grad = np.zeros(n)
    m = abs(int(lag))
    if m >= n:
        return grad
    if lag >= 0:
        grad[0:n-m] += (-h[m:n])
        grad[m:n] += (1.0 - h[0:n-m])
    else:
        grad[0:n-m] += (1.0 - h[m:n])
        grad[m:n] += (-h[0:n-m])
    return grad

def softmax_obj(h, T):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -60.0 / T, 0.0)
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * dx

def softmax_grad(h, T, top_k=25):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = np.clip(corr - mc, -60.0 / T, 0.0)
    exp_s = np.exp(T * shifted)
    total = np.sum(exp_s)
    if total < 1e-300:
        k_max = int(np.argmax(corr))
        return lag_subgrad(h, k_max - (n - 1)) * dx
    weights = exp_s / total
    idx = np.argsort(weights)[-top_k:]
    grad = np.zeros(n)
    for k_i in idx:
        w = weights[k_i]
        if w < 1e-8:
            continue
        grad += w * lag_subgrad(h, k_i - (n - 1))
    return grad * dx

        k_idx = np.argsort(corr)[-top_k:][::-1]
```
