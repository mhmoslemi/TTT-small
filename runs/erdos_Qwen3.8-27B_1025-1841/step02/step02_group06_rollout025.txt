def project_to_feasible(g, n):

            grad[idx] -= pk * h[idx + d]
    return grad * dx

    corr = np.correlate(h, 1-h, mode='full')
    g = corr * dx
    best_obj = g.max()
    best_h = h.copy()

        g = corr * dx
        cur_max = g.max()
        if cur_max < best_obj - 1e-12:

            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        # Adaptive tau
        progress = min(1.0, it / (max_iters * 0.8))
        tau = tau_start * (tau_end / tau_start) ** progress

import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr.max() * (2.0 / n)

def compute_grad_top(h, g, n, dx, tau, topM):
    top_idx = np.argpartition(g, -topM)[-topM:]
    g_top = g[top_idx]
    mx = g_top.max()
    sh = np.clip(tau * (g_top - mx), -500, 0)
    ec = np.exp(sh)
    s = ec.sum()
    p = ec / s if s > 1e-30 else np.ones(len(top_idx)) / len(top_idx)
    grad = np.zeros(n)
    for i in range(len(top_idx)):
        pk = p[i]
        k = top_idx[i]
        d = k - n + 1
        j0 = max(0, d)
        j1 = min(n, n + d)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - d])
        j0b = max(0, -d)
        j1b = min(n, n - d)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + d]
    return grad * dx

def adam_optimize(h, n, tau, n_iters, time_limit, topM, lr):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_obj = compute_c5(h)
    best_h = h.copy()
    m = np.zeros(n)
    v = np.zeros(n)
    b1, b2, eps = 0.9, 0.999, 1e-8
    t = 0
    start = time.time()
    stall = 0
    cur_lr = lr
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        t += 1
        corr = np.correlate(h, 1-h, mode='full')
        g = corr * dx
        cur = g.max()
        if cur < best_obj - 1e-12:
            best_obj = cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        grad = compute_grad_top(h, g, n, dx, tau, topM)
        
        m = b1 * m + (1-b1) * grad
        v = b2 * v + (1-b2) * grad*grad
        mh = m / (1.0 - b1**t)
        vh = v / (1.0 - b2**t)
        h = project_to_feasible(h - cur_lr * mh / (np.sqrt(vh) + eps), n)
        
        if stall > 400:
            cur_lr *= 0.7
            stall = 0
            if cur_lr < 1e-12:
                break
    
    return best_h, best_obj

def subgrad_polish(h, n, time_limit, topK=3):
    dx = 2.0 / n
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    lr = 0.002
    stall = 0
    
    for it in range(100000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1-h, mode='full')
        c5 = corr.max() * dx
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        top_idx = np.argsort(corr)[-topK:]
        grad = np.zeros(n)
        for ki in top_idx:
            d = ki - n + 1
            j0 = max(0, d)
            j1 = min(n, n + d)
            if j0 < j1:
                idx = np.arange(j0, j1)
                grad[idx] += (1.0 - h[idx - d])
            j0b = max(0, -d)
            j1b = min(n, n - d)
            if j0b < j1b:
                idx = np.arange(j0b, j1b)
                grad[idx] -= h[idx + d]
        grad *= dx / topK
        
        cur_lr = lr / (1.0 + it * 0.01)
        h = project_to_feasible(h - cur_lr * grad, n)
        
        if stall > 200:
            lr *= 0.8
            stall = 0
            if lr < 1e-14:
                break
    
    return best_h, best_c5

Let me also reconsider the learning rates. The gradient entries (before * dx) are of order 1 (bounded by the sum of p_k which is 1). After * dx = 2/n ≈ 0.0005, the gradient entries are about 0.0005.

For Adam, the effective step is lr * m_hat / sqrt(v_hat). Since m_hat ≈ grad and sqrt(v_hat) ≈ |grad|, the step is approximately lr in magnitude. So lr=0.01 means each step changes h by about 0.01.

After the projection, the change might be different, but as a first approximation, lr=0.01 should be fine.

Actually, I realize there might be an issue with the Adam optimizer in this context. The projection can create "plateaus" where many entries are exactly 0 or 1. On these plateaus, the gradient might be zero (if h[i]=0, then 1-h[i+d] is still nonzero, but the product h[i]*(1-h[i+d]) = 0, and the derivative with respect to h[i] is 1-h[i+d] which is nonzero). So the gradient should still be informative even on the plateaus.

Let me also think about whether I should use a different smoothing than softmax. The softmax smoothing is:
F_τ = (1/τ) log Σ exp(τ * f_k)

As τ → ∞, this approaches max_k f_k. But the transition can be sharp, causing the gradient to change abruptly.

An alternative is the "entropic" regularization:
F_τ = max_k f_k - (1/τ) log Σ exp(τ * (f_k - max_j f_j))

But this is the same thing.

Another option: use the "smooth max" based on the power mean:
F_p = (Σ f_k^p)^(1/p)

For p=2 (RMS), this is smooth and easier to differentiate. As p → ∞, it approaches the max.

The gradient of F_2 = sqrt(Σ f_k²) is:
∇F_2 = (1/F_2) * Σ f_k * ∇f_k

This gives more weight to lags with larger f_k, which is what we want. And it's smoother than the softmax for moderate p.

Let me try using p=4 or p=8 instead of the softmax. This might give a smoother landscape for the optimizer.

Actually, for the power mean with p=2:
F_2(h) = sqrt(Σ_k (f_k * dx)²)

The gradient is:
∇F_2[h][j] = (1/F_2) * Σ_k (f_k * dx) * (∂f_k/∂h[j] * dx)
= (dx/F_2) * Σ_k (f_k * dx) * ∂f_k/∂h[j]
= (dx²/F_2) * Σ_k f_k * ∂f_k/∂h[j]
