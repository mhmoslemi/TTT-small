**New idea: Exploit the minimax structure with a "shift game"**

The key insight: C₅(h) = max_k f_k(h). At the optimum, there's a distribution μ* over shifts such that:
- f_{k*}(h*) = C₅ for k in support of μ*
- The gradient of E_μ f_k vanishes at h* (for the inner minimization)

We can alternate:
1. Given h, compute f_k for all k, set μ(k) ∝ exp(β * f_k) (or put mass on top shifts)
2. Given μ, minimize E_μ f_k(h) over h (this is a smooth problem)
3. Repeat

This is essentially the "mirror descent" approach for the outer max, combined with gradient descent for the inner min. The β parameter controls how "peaked" μ is (β→∞ gives the true max).

This is actually what the smooth max gradient does, but we can be more aggressive:
- After each inner optimization, explicitly update μ
- Use a schedule: β starts small, increases over time

            # Use coordinate descent within the block

                            break

import numpy as np

def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(tau * (corr - max_c), -500, 0)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    p = p[p > 1e-14]
    indices = np.where(np.correlate(h, 1.0 - h, mode='full') > 0)[0]
    k_indices = np.arange(2 * n - 1)
    mask = exp_c > 1e-10
    grad = np.zeros(n)
    for k in np.where(mask)[0]:
        pk = p_k = exp_c[k] / s
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            idx = np.arange(j0, j1)
            grad[idx] += pk * (1.0 - h[idx - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            idx = np.arange(j0b, j1b)
            grad[idx] -= pk * h[idx + sh]
    grad *= dx
    return grad

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

    if remaining() > 80:
        h8, c8 = local_search(best_h, n_init, time_limit=min(70, remaining() * 0.4), seed=seed + 100)
        if c8 < best_c5:
            best_c5, best_h = c8, h8

    # Phase 9: Try higher resolution n+500
    n2 = n_init + 500
    if remaining() > 200:
        h9 = interp_periodic(best_h, n2)
        h9 = project_to_feasible(h9, n2)
        c9 = compute_c5(h9)
        if remaining() > 120:
            h9, c9 = optimize_momentum(h9, n2, tau=40.0, time_limit=min(80, remaining() * 0.3), lr=0.1, momentum=0.9)
        if remaining() > 80:
            h9, c9 = optimize_momentum(h9, n2, tau=10.0, time_limit=min(50, remaining() * 0.2), lr=0.04, momentum=0.92)
        if remaining() > 50:
            h9, c9 = local_search(h9, n2, time_limit=min(40, remaining() * 0.4), seed=seed + 200)
        if c9 < best_c5:
            best_c5, best_h = c9, h9

    # Phase 10: Final subgrad polish
    if remaining() > 30:
        n_f = len(best_h)
        h_f, c_f = optimize_subgrad(best_h, n_f, time_limit=min(25, remaining() * 0.5), lr=0.0005)
        if c_f < best_c5:
            best_c5, best_h = c_f, h_f

    # Final projection
    best_h = project_to_feasible(best_h, len(best_h))
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, best_c5, len(best_h)
