    N = 2 * n

    a = np.zeros(N)

    a[:n] = h
    b[:n] = (1.0 - h)[::-1]

    return float(np.max(corr) * np.real(corr[:2*n-1])) * dx

from scipy.optimize import minimize
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def make_feasible(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    for _ in range(20):
        diff = target - np.sum(h)
        if abs(diff) < 1e-12:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def binary_local_search(h, n, rng, max_iters=5000, time_limit_s=60):
    """Random-swap binary local search."""
    t_start = time.time()
    best_h = h.copy()
    best_c = compute_c5(h)
    
    ones = np.where(best_h == 1.0)[0]
    zeros = np.where(best_h == 0.0)[0]
    
    for it in range(max_iters):
        if time.time() - t_start > time_limit_s:
            break
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Try K random swaps
        K = min(200, len(ones) * len(zeros))
        best_swap_c = best_c
        best_i, best_j = -1, -1
        
        for _ in range(K):
            i = rng.choice(ones)
            j = rng.choice(zeros)
            trial = best_h.copy()
            trial[i] = 0.0
            trial[j] = 1.0
            c = compute_c5(trial)
            if c < best_swap_c:
                best_swap_c = c
                best_i, best_j = i, j

            best_h[best_i] = 0.0
            best_h[best_j] = 1.0
            best_c = best_swap_c
            ones = np.where(best_h == 1.0)[0]

def _obj_slsqp(x, n, T):
    h = x
    corr = np.correlate(h, 1.0 - h, mode='full')
    mc = np.max(corr)
    shifted = corr - mc
    return (mc + np.log(np.sum(np.exp(T * shifted))) / T) * (2.0 / n)

def _con_fun(x, n):
    return np.sum(x) - n / 2.0

def _con_jac(x, n):
    return np.ones(len(x))

def slsqp_optimize(h, n, T=100.0, maxiter=300):
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(lambda x: _obj_slsqp(x, n, T), h, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(res.x >= -0.01) and np.all(res.x <= 1.01):
            h_out = make_feasible(np.clip(res.x, 0, 1), n)
            return h_out, compute_c5(h_out)
    except:
        pass
    return h, compute_c5(h)
