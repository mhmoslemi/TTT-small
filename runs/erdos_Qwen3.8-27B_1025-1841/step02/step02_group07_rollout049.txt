def compute_gradient(h, n, T):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode='full')  # length 2n-1
    scaled = T * dx * c
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    weights /= np.sum(weights)
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        offset = k - n + 1  # ranges from -(n-1) to n-1
        w = weights[k]
        if w < 1e-15:
            continue
        # dC_k/dh[j] = (1-h[j-offset]) - h[j+offset] with 0 for OOB
        for j in range(n):
            term1 = 0.0
            term2 = 0.0
            idx1 = j - offset
            if 0 <= idx1 < n:
                term1 = 1.0 - h[idx1]
            idx2 = j + offset
            if 0 <= idx2 < n:
                term2 = h[idx2]
            grad[j] += w * (term1 - term2)
    
    return grad * dx

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project(h, n):
    target = n / 2.0
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    for _ in range(200):
        s = np.sum(h)
        d = target - s
        if abs(d) < 1e-14:
            break
        if d > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = np.sum(mask)
        if cnt == 0:
            break
        h[mask] += d / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def soft_max_obj(h, n, T):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    scaled = corr * T
    m = np.max(scaled)
    return float((m + np.log(np.sum(np.exp(scaled - m)))) / T * dx)

def compute_gradient(h, n, T):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, mode='full')
    scaled = T * dx * c
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    grad = np.zeros(n)
    idx = np.arange(n)
    for k in range(2 * n - 1):
        offset = k - n + 1
        w = weights[k]
        if w < 1e-14:
            continue
        idx1 = idx - offset
        valid1 = (idx1 >= 0) & (idx1 < n)
        term1 = np.zeros(n)
        term1[valid1] = 1.0 - h[idx1[valid1]]
        idx2 = idx + offset
        valid2 = (idx2 >= 0) & (idx2 < n)
        term2 = np.zeros(n)
        term2[valid2] = h[idx2[valid2]]
        grad += w * (term1 - term2)
    return grad * dx

def interp_to(h, n_from, n_to):
    if n_from == n_to:
        return h.copy()
    xf = np.linspace(0, 2, n_from, endpoint=False)
    xt = np.linspace(0, 2, n_to, endpoint=False)
    return np.interp(xt, xf, h)

def projected_gd(h0, n, T, deadline, max_steps=500):
    h = h0.copy()
    f_best = soft_max_obj(h, n, T)
    n_bad = 0
    base_alpha = 5.0 / n
    for step in range(max_steps):
        if time.time() > deadline:
            break
        g = compute_gradient(h, n, T)
        alpha = base_alpha
        improved = False
        for _ in range(40):
            h_trial = h - alpha * g
            h_trial = project(h_trial, n)
            f_trial = soft_max_obj(h_trial, n, T)
            if f_trial <= f_best - 1e-15:
                h = h_trial
                f_best = f_trial
                improved = True
                break
            alpha *= 0.5
        if not improved:
            n_bad += 1
            if n_bad > 30:
                break
        else:
            n_bad = 0
    return h

def slsqp_fine(h0, n, T, deadline):
    target = n / 2.0
    def f(x):
        return soft_max_obj(x, n, T)
    def c_eq(x):
        return np.sum(x) - target
    try:
        res = minimize(f, h0, method='SLSQP',
                       bounds=[(0.0, 1.0)] * n,
                       constraints=[{'type': 'eq', 'fun': c_eq}],
                       options={'ftol': 1e-16, 'maxiter': 200})
        if time.time() > deadline:
            return h0
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project(res.x, n)
    except Exception:
        pass
    return h0

def run(seed=42, budget_s=1300, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 10

    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_start = np.array(initial_h_values, dtype=float).copy()
    else:
        h_start = np.full(336, 0.5)

    best_h = h_start.copy()
    best_c5 = evaluate_c5(best_h)
    n = len(best_h)

    # Phase 1: Analytical gradient descent with temperature schedule
    T_schedule = [15.0, 40.0, 100.0, 300.0, 800.0]
    h_work = best_h.copy()
    for T in T_schedule:
        if time.time() > deadline - 60:
            break
        n_steps = 400 if T < 100 else 250
        h_work = projected_gd(h_work, n, T, deadline, max_steps=n_steps)
        c5 = evaluate_c5(h_work)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_work.copy()

    # Phase 2: SLSQP polish at high T
    if time.time() < deadline - 40:
        h_slsqp = slsqp_fine(best_h, n, 300.0, deadline)
        c5_s = evaluate_c5(h_slsqp)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_slsqp.copy()

    # Phase 3: Random restarts with small structured perturbations
    for attempt in range(60):
        if time.time() > deadline - 20:
            break
        scale = 0.03 + 0.04 * np.random.rand()
        h_pert = best_h + np.random.randn(n) * scale
        h_pert = project(h_pert, n)
        T_r = 200.0 + 200.0 * np.random.rand()
        h_new = projected_gd(h_pert, n, T_r, deadline, max_steps=200)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 4: Try a few larger perturbations for global escape
    for attempt in range(10):
        if time.time() > deadline - 20:
            break
        h_pert = best_h + np.random.randn(n) * 0.12
        h_pert = project(h_pert, n)
        h_new = projected_gd(h_pert, n, 50.0, deadline, max_steps=300)
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()

    # Phase 5: Final polish at very high T
    if time.time() < deadline - 5:
        h_final = projected_gd(best_h, n, 2000.0, deadline, max_steps=300)
        c5_final = evaluate_c5(h_final)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()

    # Final SLSQP squeeze
    if time.time() < deadline - 2:
        h_sq = slsqp_fine(best_h, n, 500.0, deadline)
        c5_sq = evaluate_c5(h_sq)
        if c5_sq < best_c5:
            best_c5 = c5_sq
            best_h = h_sq.copy()

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
