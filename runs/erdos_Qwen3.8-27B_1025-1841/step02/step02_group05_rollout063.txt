import numpy as np
import time

def project_to_feasible(g, n):
    target = n / 2.0
    if g.min() >= 0 and g.max() <= 1:
        s = g.sum()
        if abs(s - target) < 1e-10:
            return g.copy()
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    thresh = (lo + hi) / 2.0
    return np.clip(g - thresh, 0.0, 1.0)

def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)

def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)

def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(beta * (corr - max_c), -600, 600)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / beta)
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return obj, grad

    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 2.0 / beta
    v = np.zeros(n)
    stall = 0

        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, beta)
        c5_now = compute_c5(h)

            best_h = h.copy()
            stall = 0
        else:

        if stall > 3000:
            lr *= 0.6
            v *= 0.2
            stall = 0
            if lr < 1e-15:
                break
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
    
    return best_h, best_c5

def optimize_subgrad_multi(h, n, time_limit, lr=None, top_k=7):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()

    dx = 2.0 / n
    if lr is None:

        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx

            stall = 0
        else:

        if stall > 4000:

            v *= 0.4
            stall = 0
            if lr < 1e-15:

import numpy as np
import time


def project_to_feasible(g, n):
    target = n / 2.0
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(g - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(g - (lo + hi) / 2.0, 0.0, 1.0)


def interp_periodic(h, target_n):
    n = len(h)
    if n == target_n:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(target_n) * (2.0 / target_n)
    return np.interp(x_new, x_old, h, period=2.0)


def compute_c5(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode='full').max() * (2.0 / n)


def smooth_obj_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_c = corr.max()
    shifted = np.clip(beta * (corr - max_c), -500, 500)
    exp_c = np.exp(shifted)
    s = exp_c.sum()
    p = exp_c / s
    obj = dx * (max_c + np.log(s) / beta)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if pk < 1e-14:
            continue
        sh = k - n + 1
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += pk * (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= pk * h[j0b + sh:j1b + sh]
    grad *= dx
    return obj, grad


def optimize_momentum(h, n, beta, time_limit, lr=None, momentum=0.9):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    if lr is None:
        lr = 2.0 / beta
    v = np.zeros(n)
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        obj, grad = smooth_obj_grad(h, beta)
        c5_now = compute_c5(h)
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 3000:
            lr *= 0.6
            v *= 0.2
            stall = 0
            if lr < 1e-15:
                break
        v = momentum * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def optimize_subgrad_single(h, n, time_limit, lr=None):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.008 * dx
    v = np.zeros(n)
    mom = 0.85
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 5000:
            lr *= 0.7
            v *= 0.3
            stall = 0
            if lr < 1e-15:
                break
        k_star = np.argmax(corr)
        sh = k_star - n + 1
        grad = np.zeros(n)
        j0 = max(0, sh)
        j1 = min(n, sh + n)
        if j0 < j1:
            grad[j0:j1] += (1.0 - h[j0 - sh:j1 - sh])
        j0b = max(0, -sh)
        j1b = min(n, n - sh)
        if j0b < j1b:
            grad[j0b:j1b] -= h[j0b + sh:j1b + sh]
        grad *= dx
        v = mom * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def optimize_subgrad_multi(h, n, time_limit, lr=None, top_k=8):
    h = project_to_feasible(h, n).copy()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    start = time.time()
    dx = 2.0 / n
    if lr is None:
        lr = 0.004 * dx
    v = np.zeros(n)
    mom = 0.82
    stall = 0
    for it in range(500000):
        if time.time() - start > time_limit:
            break
        corr = np.correlate(h, 1.0 - h, mode='full')
        c5_now = corr.max() * dx
        if c5_now < best_c5 - 1e-13:
            best_c5 = c5_now
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 4000:
            lr *= 0.7
            v *= 0.4
            stall = 0
            if lr < 1e-15:
                break
        top_idx = np.argsort(corr)[::-1][:top_k]
        top_vals = corr[top_idx]
        w = np.exp(top_vals - top_vals[0])
        w /= w.sum()
        grad = np.zeros(n)
        for i, k in enumerate(top_idx):
            sh = k - n + 1
            j0 = max(0, sh)
            j1 = min(n, sh + n)
            if j0 < j1:
                grad[j0:j1] += w[i] * (1.0 - h[j0 - sh:j1 - sh])
            j0b = max(0, -sh)
            j1b = min(n, n - sh)
            if j0b < j1b:
                grad[j0b:j1b] -= w[i] * h[j0b + sh:j1b + sh]
        grad *= dx
        v = mom * v - lr * grad
        h = project_to_feasible(h + v, n)
    return best_h, best_c5


def bang_bang_project(h, n):
    h_bb = (h > 0.5).astype(float)
    diff = h_bb - h
    order = np.argsort(np.abs(diff))
    need = n / 2.0 - h_bb.sum()
    if need > 1e-9:
        cnt = 0
        for idx in order[:(n // len(np.where(h_bb == 0)[0]))]:
            if h_bb[idx] == 0 and h[idx] < 0.5:
                h_bb[idx] = 1.0
                cnt += 1
    elif need < -1e-9:
        cnt = 0
        for idx in order:
            if h_bb[idx] == 1 and h[idx] > 0.5:
                h_bb[idx] = 0.0
    return h_bb
