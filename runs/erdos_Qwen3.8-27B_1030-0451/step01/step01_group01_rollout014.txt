        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                if it % 4 == 0:
                    radius = min(radius * 1.08, 0.35)
            else:
                no_improve += 1
                if no_improve >= 12:
                    radius = max(radius * 0.45, 0.003)
                    no_improve = 0
                    if radius <= 0.003:
                        break
        else:
            no_improve += 1
            if no_improve >= 15:
                radius = max(radius * 0.45, 0.003)
                no_improve = 0
                if radius <= 0.003:
                    break
    return best_h, best_c

def method_subgrad_polish(h0, deadline, max_iters=20000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.003
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 200 == 0:
            step = max(step * 0.995, 1e-7)
        v = corr_all(h)
        m = v.max()
        beta = 60.0
        expo = np.clip(beta * (v - m), None, 400.0)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 1e-10:
                lag = k - (n - 1)
                g += w[k] * grad_lag_one(h, lag)
        g[0] -= g.mean()
        step_adapt = step / (1.0 + it * 0.0001)
        h_new = h - step_adapt * g
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < best_c - 1e-14:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new.copy()
            stall = 0
        else:
            h = h_new.copy()
            stall += 1
        if stall > 3000:
            break
    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_ext = np.concatenate([h0, h0[:1]])
    h_new = np.interp(x_new * n_old, np.arange(n_old + 1) * 1.0, h_ext, period=n_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum_box(h_new, n_new / 2.0)

def diverse_starts(n, n_starts=4):
    starts = []
    h_base = np.full(n, 0.5)
    starts.append(h_base.copy())
    for i in range(n_starts - 1):
        freq = np.random.randint(2, max(3, n // 10))
        phase = np.random.uniform(0, 2 * np.pi)
        amp = np.random.uniform(0.1, 0.4)
        x = np.arange(n) / n * 2.0 * np.pi
        h = 0.5 + amp * np.sin(freq * x + phase)
        h = np.clip(h, 0.0, 1.0)
        h = project_sum_box(h, n / 2.0)
        starts.append(h)
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 6.0

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n = len(best_h)

    # Phase 1: SLSQP with high beta on original grid
    if time.time() < deadline - 120:
        dl = deadline - 120
        h1, c1 = method_slsqp_refined(best_h, dl,
            betas=[20, 60, 150, 400, 1000, 3000, 8000], max_iter_per_beta=2000)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: LP trust region
    if time.time() < deadline - 60:
        h2, c2 = method_lp_trust(best_h, deadline - 60, max_iters=500)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Another SLSQP round from improved point
    if time.time() < deadline - 40:
        h3, c3 = method_slsqp_refined(best_h, deadline - 40,
            betas=[1000, 3000, 8000, 20000], max_iter_per_beta=3000)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Subgradient polish
    if time.time() < deadline - 20:
        h4, c4 = method_subgrad_polish(best_h, deadline - 20, max_iters=15000)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Multi-start with perturbations on current grid
    if time.time() < deadline - 30:
        for trial in range(6):
            if time.time() > deadline - 30:
                break
            scale = 0.008 + 0.004 * trial
            h_pert = project_sum_box(best_h + np.random.randn(n) * scale, n / 2.0)
            sub_dl = min(deadline - 30, time.time() + 80)
            h_m, c_m = method_slsqp_refined(h_pert, sub_dl,
                betas=[500, 2000, 6000], max_iter_per_beta=800)
            if c_m < best_c:
                best_h, best_c = h_m, c_m

    # Phase 6: Resolution increase
    if time.time() < deadline - 90:
        n_cur = len(best_h)
        for n_t in [int(n_cur * 1.5), int(n_cur * 2)]:
            if n_t > 400:
                break
            if time.time() > deadline - 90:
                break
            h_tr = transfer_resolution(best_h, n_t)
            c_tr = evaluate_c5(h_tr)
            if c_tr < best_c:
                best_h = h_tr.copy()
                best_c = c_tr
            sub_dl = min(deadline - 40, time.time() + 150)
            h_a, c_a = method_lp_trust(best_h, sub_dl, max_iters=300)
            if c_a < best_c:
                best_h, best_c = h_a, c_a
            sub_dl2 = min(deadline - 15, time.time() + 80)
            h_b, c_b = method_slsqp_refined(best_h, sub_dl2,
                betas=[500, 2000, 5000], max_iter_per_beta=1500)
            if c_b < best_c:
                best_h, best_c = h_b, c_b

    # Phase 7: Final polish
    if time.time() < deadline - 5:
        h_f, c_f = method_subgrad_polish(best_h, deadline - 2, max_iters=10000)
        if c_f < best_c:
            best_h, best_c = h_f, c_f

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    best_c = min(best_c, final_c)
    return best_h, float(best_c), len(best_h)
