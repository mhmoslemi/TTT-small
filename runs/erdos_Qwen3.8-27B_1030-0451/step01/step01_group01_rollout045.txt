import numpy as np
import time
from scipy.optimize import linprog

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] = np.minimum(y[idx] + r / len(idx), 1.0)
                r2 = target - y.sum()
                if r2 > 1e-12:
                    idx2 = np.where(y < 1.0 - 1e-8)[0]
                    if len(idx2) > 0:
                        y[idx2] = np.minimum(y[idx2] + r2 / len(idx2), 1.0)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] = np.maximum(y[idx] + r / len(idx), 0.0)
                r2 = target - y.sum()
                if r2 < -1e-12:
                    idx2 = np.where(y > 1e-8)[0]
                    if len(idx2) > 0:
                        y[idx2] = np.maximum(y[idx2] + r2 / len(idx2), 0.0)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        g[lag:] += 1.0 - h[:-lag]
        g[:n - lag] -= h[lag:]
    else:
        l = -lag
        g[:n - l] += 1.0 - h[l:]
        g[l:] -= h[:n - l]
    return g

def lp_adactive(h0, deadline, max_iters=3000):
    """LP trust region with adaptive active set and radius."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.008
    no_improve = 0
    active_cap = 8

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = v.max()

        gap = max(0.0015 * vmax, 0.0003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > active_cap:
            active_idx = np.argsort(v)[-active_cap:]
        if len(active_idx) < 2:
            active_idx = np.argsort(v)[-3:]

        A_ub = np.zeros((len(active_idx), n + 1))
        b_ub = np.zeros(len(active_idx))
        for i, k_idx in enumerate(active_idx):
            lag = int(k_idx) - (n - 1)
            g = grad_lag(h, lag)
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = g @ h - v[k_idx]

        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        lb = np.maximum(h - radius, 0.0)
        ub = np.minimum(h + radius, 1.0)
        bounds = list(zip(lb, ub)) + [(-0.5, 0.5)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')

        if res is not None and res.success and res.x is not None:
            h_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 5e-13:
                best_h = h_new.copy()
                best_c = c_new
                h = h_new.copy()
                no_improve = 0
                if it % 20 == 0:
                    radius = min(radius * 1.15, 0.04)

                no_improve += 1
                if no_improve >= 10:
                    radius = max(radius * 0.5, 0.0005)
                    no_improve = 0

        else:
            no_improve += 1
            if no_improve >= 15:
                radius = max(radius * 0.4, 0.0005)

def pair_exchange(h0, deadline, max_rounds=500):
    """Targeted pair exchange on active lags plus random exploration."""
    n = len(h0)

def subgradient_polish(h0, deadline, max_iters=5000):
    """Subgradient method on the exact max with adaptive step size."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(h)
    step = 0.003
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        m = v.max()
        argmax = np.argmax(v)
        lag = int(argmax) - (n - 1)
        g = grad_lag(h, lag)
        g = g - g.mean()  # project onto sum=0 subspace
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        h_new = h - step * g / g_norm
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new

            stall = 0
            step = min(step * 1.02, 0.02)
        else:
            step *= 0.85
            stall += 1
            if stall > 500:
                break
        h = h_new
    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0, 1, n_old, endpoint=False)

    return h_new
