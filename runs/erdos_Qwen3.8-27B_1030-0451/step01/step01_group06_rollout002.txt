def project_to_sum(x, target_sum):
    """Project x to {y: 0 <= y <= 1, sum(y) = target_sum}"""
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    # Bisection on Lagrange multiplier
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix numerical issues
    diff = h.sum() - target_sum
    if abs(diff) > 1e-12:
        if diff > 0:
            order = np.argsort(h)[::-1]
        else:
            order = np.argsort(h)
        remaining = diff
        for idx in order:

                break
            adj = min(abs(remaining), abs(1.0 - h[idx]) if diff < 0 else h[idx])
            h[idx] -= np.sign(diff) * adj
            remaining -= np.sign(diff) * adj
    return h

import numpy as np
import time
from scipy.optimize import linprog

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_to_sum(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = -2.0
    hi = 3.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    diff = h.sum() - target_sum
    if abs(diff) > 1e-13:
        if diff > 0:
            order = np.argsort(h)[::-1]
        else:
            order = np.argsort(h)
        remaining = diff
        for idx in order:
            if abs(remaining) < 1e-14:
                break
            if remaining > 0:
                adj = min(remaining, h[idx])
                h[idx] -= adj
                remaining -= adj
            else:
                adj = min(-remaining, 1.0 - h[idx])
                h[idx] += adj
                remaining += adj
    return h

def compute_correlation_gradients(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    grads = np.zeros((m, n))
    for k in range(-(n-1), n):
        idx = k + n - 1
        if k == 0:
            grads[idx] = dx * (1.0 - 2.0 * h)
        elif k > 0:
            g = np.zeros(n)
            g[k:] += dx * (1.0 - h[:n-k])
            g[:n-k] += -dx * h[k:]
            grads[idx] = g
        else:
            t = -k
            g = np.zeros(n)
            g[:n-t] += dx * (1.0 - h[t:])
            g[t:] += -dx * h[:n-t]
            grads[idx] = g
    return grads

def lp_direction(h, corr, grads, trust, active_frac=1.0):
    n = len(h)
    m = 2 * n - 1
    k_act = max(1, int(m * active_frac))
    top_idx = np.argpartition(corr, -k_act)[-k_act:]
    corr_a = corr[top_idx]
    grads_a = grads[top_idx]
    nvars = n + 1
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    A_ub = np.zeros((k_act, nvars))
    A_ub[:, :n] = grads_a
    A_ub[:, n] = -1.0
    b_ub = -corr_a
    lower = np.maximum(-h, -trust)
    upper = np.minimum(1.0 - h, trust)
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = list(zip(lower, upper)) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.all(h + d >= -1e-7) and np.all(h + d <= 1 + 1e-7):
                return d
    except Exception:
        pass
    return None

def paired_coord_step(h, corr, grads, i, j):
    """Find optimal x for h[i]=x, h[j]=c-x minimizing max corr."""
    n = len(h)
    c = h[i] + h[j]
    lo = max(0.0, c - 1.0)
    hi = min(1.0, c)
    if hi - lo < 1e-15:
        return h[i]
    alpha = grads[:, i] - grads[:, j]
    beta = corr - alpha * h[i]
    
    def obj(x):
        return np.max(alpha * x + beta)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = lo, hi
    c1 = b - gr * (b - a)
    c2 = a + gr * (b - a)
    f1, f2 = obj(c1), obj(c2)
    for _ in range(60):
        if f1 < f2:
            b = c2
            c2 = c1
            f2 = f1
            c1 = b - gr * (b - a)
            f1 = obj(c1)
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + gr * (b - a)
            f2 = obj(c2)
    x_opt = 0.5 * (a + b)
    return x_opt

def coord_descent_sweep(h, best_val, n_sweeps, rng):
    """Paired coordinate descent with multiple sweeps."""
    n = len(h)
    target = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_v = best_val
    
    for sweep in range(n_sweeps):
        corr = correlation_array(h)
        cur = float(np.max(corr))
        if cur < best_v:
            best_v = cur
            best_h = h.copy()
        grads = compute_correlation_gradients(h)
        improved = False
        pairs = [(rng.integers(n), rng.integers(n))] * 200
        for _ in range(len(pairs)):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            x_new = paired_coord_step(h, corr, grads, i, j)
            if abs(x_new - h[i]) > 1e-12:
                c = h[i] + h[j]
                h[i] = x_new
                h[j] = c - x_new
                h[i] = np.clip(h[i], 0, 1)
                h[j] = np.clip(h[j], 0, 1)
                # Recompute corr and grads periodically
                if _ % 20 == 0:
                    corr = correlation_array(h)
                    grads = compute_correlation_gradients(h)
                improved = True
        if not improved:
            break
    return best_h, best_v

def lp_optimize(h, best_val, deadline, rng, use_active_frac=1.0):
    """LP-based local optimization."""
    n = len(h)
    target = n / 2.0

    stagnant = 0
    best_h = h.copy()

            accepted = False

                    h = h_new.copy()

                    best_v = val
                    best_h = h_new.copy()
                    h = h_new.copy()
                    accepted = True
                    stagnant = 0
                    break
