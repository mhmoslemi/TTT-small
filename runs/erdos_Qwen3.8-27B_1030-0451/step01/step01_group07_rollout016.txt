import time
import math
import numpy as np

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def project_exact(h, target):
    h = np.asarray(h, dtype=float).ravel()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        t = 0.5*(lo+hi)
        s = np.clip(h-t, 0.0, 1.0).sum()
        if s > target: lo = t
        else: hi = t
    out = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    d = out.sum() - target
    if d > 1e-12:
        room = out.sum()
        if room > 1e-15: out -= (d/room)*out
    elif d < -1e-12:
        room = (1.0-out).sum()
        if room > 1e-15: out += (-d/room)*(1.0-out)
    return np.clip(out, 0.0, 1.0)

def objective(h):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    return float(c.max() * dx)

def corr_vec(h):
    return np.correlate(h, 1.0-h, mode='full')

def lse_obj(c, sigma):
    m = c.max()
    if sigma < 1e-6:
        return m
    return m + math.log(np.sum(np.exp(sigma*(c-m))))/sigma

def lse_weights(c, sigma):
    m = c.max()
    e = np.exp(sigma*(c-m))
    return e/e.sum()

def topk_weights(c, k):
    n_c = len(c)
    # Get indices of top-k
    if k >= n_c:
        return np.ones(n_c)/n_c
    idx = np.argpartition(c, -k)[-k:]
    w = np.zeros(n_c)
    vals = c[idx]
    # Softmax within top-k with moderate temperature
    vmax = vals.max()
    e = np.exp(30.0*(vals-vmax))
    w[idx] = e/e.sum()
    return w

def grad_lse(h, sigma):
    n = len(h)
    c = corr_vec(h)
    w = lse_weights(c, sigma)
    return _grad_from_weights(h, w)

def grad_topk(h, k):
    n = len(h)
    c = corr_vec(h)
    w = topk_weights(c, k)
    return _grad_from_weights(h, w)

def _grad_from_weights(h, w):
    n = len(h)
    dx = 2.0/n
    center = n-1
    grad = np.zeros(n)
    # Center lag
    grad += w[center] * (1.0 - 2.0*h)
    for p in range(1, n):
        wp = w[center+p]
        wm = w[center-p]
        if wp < 1e-18 and wm < 1e-18:
            continue
        # Forward lag (k=p): output index center+p
        # d/dh[j] = (1-h[j-p]) for j in [p, n-1], -h[j+p] for j in [0, n-1-p]
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:n-p])
            grad[:n-p] -= wp * h[p:]
        # Backward lag (k=-p): output index center-p
        # d/dh[j] = (1-h[j+p]) for j in [0, n-1-p], -h[j-p] for j in [p, n-1]
        if wm > 1e-18:
            grad[:n-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * h[:n-p]
    grad *= dx
    return grad

def lse_polish(h0, rng, time_limit, sigma, deadline=None):
    n = len(h0)
    target = n/2.0
    h = np.asarray(h0, dtype=float).copy()
    best_h = h.copy()
    best_c5 = objective(h)
    cur_lse = lse_obj(corr_vec(h), sigma)
    lr = 0.05
    t0 = time.monotonic()
    stall = 0
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = grad_lse(h, sigma)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = -g / gn  # normalize for step size control
        # Backtracking line search
        improved = False
        for trial in range(25):
            step = lr * (0.5 ** trial)
            cand = project_exact(h + step * g, target)
            cv = lse_obj(corr_vec(cand), sigma)
            if cv < cur_lse - 1e-14:
                h = cand
                cur_lse = cv
                c5 = objective(h)
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        if not improved:
            # Perturb and continue
            h = project_exact(h + rng.normal(0, 0.02, n), target)
            cur_lse = lse_obj(corr_vec(h), sigma)
            c5 = objective(h)
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            stall += 1
            if stall >= 8:
                lr *= 0.5
                stall = 0

        best_h = h.copy()

def topk_polish(h0, rng, time_limit, k, deadline=None):
    n = len(h0)
    target = n/2.0
    h = np.asarray(h0, dtype=float).copy()
    best_h = h.copy()
    best_c5 = objective(h)
    lr = 0.03
    t0 = time.monotonic()
    stall = 0
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = grad_topk(h, k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = -g / gn
        improved = False
        for trial in range(25):
            step = lr * (0.5 ** trial)
            cand = project_exact(h + step * g, target)
            c5 = objective(cand)
            if c5 < objective(h) - 1e-14:
                h = cand
                if c5 < best_c5 - 1e-12:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
        if not improved:
            h = project_exact(h + rng.normal(0, 0.01, n), target)
            stall += 1
            if stall >= 6:
                lr *= 0.5
                stall = 0
                if lr < 1e-7:
                    break
    c5 = objective(h)
    if c5 < best_c5 - 1e-12:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5
