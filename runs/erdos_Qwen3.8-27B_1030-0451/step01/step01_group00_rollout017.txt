    Returns the delta vector or None if LP fails."""

    M = 2 * n - 1

    current_max = np.max(corr)
    
    # Select active lags
    if M <= max_lags:
        active = np.arange(M)
    else:

        # Take top max_lags, plus any within a small margin of max

        active = active[:max_lags]

    G_act = G[active, :]

    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Subject to:
    #   G_act @ delta <= t - c_act  =>  G_act @ delta - t <= -c_act
    #   sum(delta) = 0
    #   max(-R, -h[j]) <= delta[j] <= min(R, 1-h[j])
    #   t >= 0 (implicit from constraints)

    # A_ub has shape (len(active), n+1)
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = G_act
    A_ub[:, n] = -1.0
    b_ub = -c_act
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds
    bounds = []
    for j in range(n):
        lo = max(-R, -h[j])
        hi = min(R, 1.0 - h[j])
        bounds.append((lo, hi))
    bounds.append((0.0, None))  # t >= 0
    
    try:

            # Check it's meaningful
            if np.max(np.abs(delta)) > 1e-12:
                return delta
    except Exception:
        pass
    return None

                h[j] += s

    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return h_new

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = (np.arange(n) + 0.5) * (2.0 / n)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.05 * np.sin(6.0 * np.pi * x + 1.4)
    h += 0.025 * np.sin(8.0 * np.pi * x + 0.3)
    h += 0.012 * np.sin(12.0 * np.pi * x + 2.1)
    h += rng.randn(n) * 0.01
    return project_constraint(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def update_best(h_new, c5_new):
        nonlocal best_h, best_c5, best_n
        if c5_new < best_c5 - 1e-13:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = len(h_new)
        elif abs(c5_new - best_c5) <= 1e-12 and len(h_new) < best_n:
            best_h = h_new.copy()
            best_c5 = c5_new
            best_n = len(h_new)

    # Phase 0: Initialize
    n_init = 81
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
        h0 = project_constraint(h0, n_init)
    except (NameError, UnboundLocalError):
        h0 = make_start(n_init, seed)

    c5_0 = compute_c5(h0)
    update_best(h0, c5_0)

    # Phase 1: Combined smooth+LP optimization on initial n
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 250
    if remaining > 40:
        h_opt, c5_opt = combined_optimize(h0.copy(), remaining, seed=seed)
        update_best(h_opt, c5_opt)

    # Phase 2: Coordinate descent polish
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 180
    if remaining > 20 and best_h is not None:
        h_opt, c5_opt = coord_descent(best_h.copy(), remaining, seed=seed + 100)
        update_best(h_opt, c5_opt)

    # Phase 3: Try larger n values
    n_candidates = []
    if n_init <= 90:
        n_candidates = [n_init + 15, n_init + 40, n_init + 70, n_init + 110]
    else:
        n_candidates = [n_init + 20, n_init + 50, n_init + 90]

    for idx, n_try in enumerate(n_candidates):
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 60
        if remaining < 35:
            break
        if n_try > 250:
            break

        if best_h is not None and best_n != n_try:
            h_try = resample_h(best_h, n_try)
            h_try = project_constraint(h_try, n_try)
        else:
            h_try = make_start(n_try, seed + n_try * 7)

        alloc = min(remaining - 20, 60)
        if alloc > 15:
            h_opt, c5_opt = combined_optimize(h_try, alloc, seed=seed + n_try)
            update_best(h_opt, c5_opt)

        # Quick coord descent on new n
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 40
        if remaining > 15:
            h_cur = h_opt if c5_opt < compute_c5(h_try) else h_try
            h_opt2, c5_opt2 = coord_descent(h_cur.copy(), min(remaining, 25), seed=seed + n_try + 50)
            update_best(h_opt2, c5_opt2)

    # Phase 4: Final combined optimization on best
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 30
    if remaining > 20 and best_h is not None:
        h_opt, c5_opt = combined_optimize(best_h.copy(), remaining, seed=seed + 900)
        update_best(h_opt, c5_opt)

    # Phase 5: Final coordinate descent
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 10
    if remaining > 10 and best_h is not None:
        h_opt, c5_opt = coord_descent(best_h.copy(), remaining, seed=seed + 1000)
        update_best(h_opt, c5_opt)

    # Final: one more LP attempt
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 3
    if remaining > 5 and best_h is not None:
        h_final = best_h.copy()
        n_f = len(h_final)
        R = 0.02
        for _ in range(5):
            if time.time() - t_start > t_start + budget_s - 2:
                break
            delta = lp_direction(h_final, R)
            if delta is not None:
                for a in [1.0, 0.5, 0.25]:
                    h_try = project_constraint(h_final + a * delta, n_f)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5 - 1e-14:
                        h_final = h_try
                        best_h = h_final.copy()
                        best_c5 = c5_try
                        break
            R *= 0.6

    # Verify and return
    if best_h is not None:
        best_h = project_constraint(best_h, len(best_h))
        best_c5 = compute_c5(best_h)
        assert best_h.shape[0] > 0
        assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
        assert abs(float(np.sum(best_h)) - len(best_h) / 2.0) < 1e-8
    else:
        best_h = np.full(81, 0.5)
        best_c5 = 0.4

    return best_h, best_c5, len(best_h)
