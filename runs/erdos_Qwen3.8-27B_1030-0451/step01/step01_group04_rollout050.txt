        order = np.argsort(h[idx])
        n_remove = int(ones - target_sum)
        hq[idx[order[:n_remove]]] = 0.0
    elif ones < target_sum:
        idx = np.where(hq == 0.0)[0]
        order = np.argsort(h[idx])[::-1]
        n_add = int(target_sum - ones)
        hq[idx[order[:n_add]]] = 1.0

    c5 = compute_c5(hq)
    for _ in range(n_tries):
        ones_list = np.where(hq == 1.0)[0]
        zeros_list = np.where(hq == 0.0)[0]
        if len(ones_list) == 0 or len(zeros_list) == 0:
            break
        a = ones_list[rng.integers(len(ones_list))]
        b = zeros_list[rng.integers(len(zeros_list))]
        hq[a] = 0.0
        hq[b] = 1.0
        c5_new = compute_c5(hq)
        if c5_new < c5:
            c5 = c5_new
        else:
            hq[a] = 1.0
            hq[b] = 0.0
    return hq, c5

def _upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return project_feasible(h.copy(), N_new / 2.0)
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)

def _random_01_start(N, rng):
    h = np.zeros(N)
    idx = rng.permutation(N)[:N // 2]
    h[idx] = 1.0
    return h

def _block_start(N, n_blocks):
    h = np.zeros(N)
    block_size = N // n_blocks
    for i in range(0, N, block_size * 2):
        end = min(i + block_size, N)
        h[i:end] = 1.0
    return project_feasible(h, N / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0

    rng = np.random.default_rng(seed)

    h_init = np.array(initial_h_values, dtype=float)
    N_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    best_N = N_init

    # Phase 1: Optimize at initial resolution
    if time.time() < deadline - 200:
        taus = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002]
        h_opt, c5_opt = _slsqp_optimize(best_h, N_init, taus, 200)
        if c5_opt < best_c5:
            best_h, best_c5, best_N = h_opt, c5_opt, N_init

    # Phase 2: Multi-resolution optimization
    for N_target in [500, 700]:
        if time.time() > deadline - 120:
            break

        target_sum = N_target / 2.0
        h_work = _upscale(best_h, N_target)

        # Main SLSQP optimization
        if time.time() < deadline - 120:
            taus_main = [0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
            h_opt, c5_opt = _slsqp_optimize(h_work, N_target, taus_main, 180)
            if c5_opt < best_c5:
                best_h, best_c5, best_N = h_opt.copy(), c5_opt, N_target

        # L-BFGS-B with penalty (faster per iteration)
        if time.time() < deadline - 120:
            h_start = _upscale(best_h, N_target)
            for tau in [0.005, 0.002, 0.001, 0.0005]:
                if time.time() > deadline - 120:
                    break
                h_lb, c5_lb = _lbfgsb_optimize(h_start, N_target, tau, 500.0, 300)
                if c5_lb < best_c5:
                    best_h, best_c5, best_N = h_lb.copy(), c5_lb, N_target
                h_start = h_lb

        # Multi-start SLSQP
        n_starts = 8
        for s in range(n_starts):
            if time.time() > deadline - 120:
                break
            scale = 0.02 + 0.015 * s
            h_start = _upscale(best_h, N_target)
            h_start = h_start + rng.normal(0, scale, N_target)
            h_start = project_feasible(h_start, target_sum)
            taus_ms = [0.008, 0.004, 0.002, 0.001, 0.0003]
            h_opt, c5_opt = _slsqp_optimize(h_start, N_target, taus_ms, 120)
            if c5_opt < best_c5:
                best_h, best_c5, best_N = h_opt.copy(), c5_opt, N_target

    # Phase 3: Try different structured starts at N=500
    if time.time() < deadline - 100:
        N_try = 500
        for n_blocks in [4, 6, 8, 12]:
            if time.time() > deadline - 100:
                break
            h_blk = _block_start(N_try, n_blocks)
            h_opt, c5_opt = _slsqp_optimize(h_blk, N_try, [0.01, 0.005, 0.002, 0.001], 150)
            if c5_opt < best_c5:
                best_h, best_c5, best_N = h_opt.copy(), c5_opt, N_try

        # Random 0/1 start
        h_rnd = _random_01_start(N_try, rng)
        h_opt, c5_opt = _slsqp_optimize(h_rnd, N_try, [0.01, 0.005, 0.002, 0.001], 150)
        if c5_opt < best_c5:
            best_h, best_c5, best_N = h_opt.copy(), c5_opt, N_try

    # Phase 4: Quantize and search
    if time.time() < deadline - 40:
        hq, c5q = _quantize_search(best_h, rng, n_tries=5000)
        if c5q < best_c5:
            best_h, best_c5, best_N = hq, c5q, len(hq)

        # Re-optimize from quantized
        if time.time() < deadline - 40:
            N_q = best_N
            h_work = project_feasible(best_h.copy(), N_q / 2.0)
            h_opt, c5_opt = _slsqp_optimize(h_work, N_q, [0.003, 0.001, 0.0003, 0.0001], 150)
            if c5_opt < best_c5:
                best_h, best_c5, best_N = h_opt, c5_opt, N_q

    # Phase 5: Final polish with very small tau
    remaining = deadline - time.time()
    if remaining > 20:
        N_work = best_N
        h_work = best_h.copy()
        for tau in [0.00015, 0.00008, 0.00004]:
            if time.time() > deadline - 10:
                break
            h_opt, c5_opt = _slsqp_optimize(h_work, N_work, [tau], 200)
            if c5_opt < best_c5:
                best_h, best_c5 = h_opt, c5_opt
            h_work = h_opt

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
