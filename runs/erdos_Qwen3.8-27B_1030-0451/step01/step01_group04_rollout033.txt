def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_all_lags(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def project_simplex(h, target):
    N = len(h)
    h = np.array(h, dtype=float)
    lo = -float(np.max(h)) - 2.0
    hi = 2.0 - float(np.min(h))

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    res = np.clip(h + t, 0.0, 1.0)
    resid = target - np.sum(res)
    if abs(resid) > 1e-12:
        if resid > 0:
            idx = np.where(res < 1.0)[0]
            if len(idx) > 0:
                res[idx[0]] = min(1.0, res[idx[0]] + resid)
        else:
            idx = np.where(res > 0.0)[0]
            if len(idx) > 0:
                res[idx[0]] = max(0.0, res[idx[0]] + resid)
    return res


def grad_lag(h, r):
    N = len(h)
    s = r - (N - 1)
    g = np.zeros(N)
    if s >= 0:
        n = N - s
        if n > 0:
            g[s:s+n] += (1.0 - h[:n])
            g[:n] -= h[s:s+n]
    else:
        ss = -s
        n = N - ss
        if n > 0:
            g[:n] += (1.0 - h[ss:ss+n])
            g[ss:ss+n] -= h[:n]
    return g


def grad_weighted(h, r_idx, weights):
    g = np.zeros(len(h))
    for r, w in zip(r_idx, weights):
        if abs(w) > 1e-14:
            g += w * grad_lag(h, r)
    return g

    h_new = np.interp(x_new, x_old, h)

                h_bin, target_sum, min(remaining2 - 10.0, 60.0), compute_c5(h_bin))
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin_opt
    
    # Final validation
    final_h = project_simplex(best_h, best_N / 2.0)
    final_c5 = compute_c5(final_h)
    
    if not np.isfinite(final_c5) or final_c5 > 0.5:
        final_h = h_init.copy()
        final_c5 = c5_init
    
    return final_h, final_c5, len(final_h)
