import numpy as np
import time
from scipy.optimize import minimize, linprog

def true_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(x, target):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    x[~np.isfinite(x)] = 0.5
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(20):
        res = float(h.sum()) - target
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def smooth_max_obj(x, eta):
    Q = corr_array(x)
    Qm = float(np.max(Q))
    if Qm * eta > 700:
        return Qm
    vals = np.exp(eta * (Q - Qm))
    return Qm + np.log(np.sum(vals)) / eta

def smooth_max_grad(x, eta):
    n = len(x)
    dx = 2.0 / n
    Q = corr_array(x)
    Qm = float(np.max(Q))
    if Qm * eta > 700:
        # All weight on max
        base = n - 1
        r = int(np.argmax(Q))
        k = r - base
        grad = np.zeros(n)
        if k == 0:
            grad = dx * (1.0 - 2.0 * x)
        elif k > 0:
            grad[k:] += dx * (1.0 - x[:n-k])
            grad[:n-k] += -dx * x[k:]
        else:
            t = -k
            grad[:n-t] += dx * (1.0 - x[t:])
            grad[t:] += -dx * x[:n-t]
        return grad
    exp_vals = np.exp(eta * (Q - Qm))
    p = exp_vals / np.sum(exp_vals)
    base = n - 1
    grad = np.zeros(n)
    pk0 = p[base]
    if pk0 > 1e-15:
        grad += pk0 * dx * (1.0 - 2.0 * x)
    for k in range(1, n):
        pk_pos = p[base + k]
        pk_neg = p[base - k]
        cp = pk_pos
        cn = pk_neg
        if cp > 1e-15:
            grad[k:] += cp * dx * (1.0 - x[:n-k])
            grad[:n-k] += -cp * dx * x[k:]
        if cn > 1e-15:
            grad[:n-k] += cn * dx * (1.0 - x[k:])
            grad[k:] += -cn * dx * x[:n-k]
    return grad

def slsqp_smooth(h, eta, budget_fn, max_iter=300):
    n = len(h)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    try:
        res = minimize(smooth_max_obj, h, jac=lambda x: smooth_max_grad(x, eta),
                       method='SLSQP', bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if budget_fn() < 5:
            return h, true_c5(h)
        if res.success and res.fun < true_c5(h):
            h_new = project_box_sum(res.x, target)
            return h_new, true_c5(h_new)
    except Exception:
        pass
    return h, true_c5(h)

def projected_gradient_step(h, eta, step_size, target):
    n = len(h)
    Q = corr_array(h)
    Qm = float(np.max(Q))
    if Qm * eta > 700:
        base = n - 1
        r = int(np.argmax(Q))
        k = r - base
        grad = np.zeros(n)
        if k == 0:
            grad = dx_factor(n) * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += dx_factor(n) * (1.0 - h[:n-k])
            grad[:n-k] += -dx_factor(n) * h[k:]
        else:
            t = -k
            grad[:n-t] += dx_factor(n) * (1.0 - h[t:])
            grad[t:] += -dx_factor(n) * h[:n-t]
    else:
        exp_vals = np.exp(eta * (Q - Qm))
        p = exp_vals / np.sum(exp_vals)
        base = n - 1
        grad = np.zeros(n)
        pk0 = p[base]
        if pk0 > 1e-15:
            grad += pk0 * dx_factor(n) * (1.0 - 2.0 * h)
        for k in range(1, n):
            cp = p[base + k]
            cn = p[base - k]
            if cp > 1e-15:
                grad[k:] += cp * dx_factor(n) * (1.0 - h[:n-k])
                grad[:n-k] += -cp * dx_factor(n) * h[k:]
            if cn > 1e-15:
                grad[:n-k] += cn * dx_factor(n) * (1.0 - h[k:])
                grad[k:] += -cn * dx_factor(n) * h[:n-k]
    h_new = project_box_sum(h - step_size * grad, target)
    return h_new

def dx_factor(n):
    return 2.0 / n

def lp_polish(h, corr, trust=0.15, budget_fn=lambda: 999):
    n = len(h)
    m = 2 * n - 1
    dx = 2.0 / n
    # Use active constraints (top 40%)
    k_act = max(1, int(m * 0.4))
    top_idx = np.argpartition(corr, -k_act)[-k_act:]
    # Build gradient rows for active constraints
    base = n - 1
    nvars = n + 1
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    A_ub = np.zeros((k_act, nvars))
    b_ub = np.zeros(k_act)
    for ii, r in enumerate(top_idx):
        k = r - base
        g = np.zeros(n)
        if k == 0:
            g[:] = dx * (1.0 - 2.0 * h)
        elif k > 0:
            g[k:] += dx * (1.0 - h[:n-k])
            g[:n-k] += -dx * h[k:]
        else:
            t = -k
            g[:n-t] += dx * (1.0 - h[t:])
            g[t:] += -dx * h[:n-t]
        A_ub[ii, :n] = g
        A_ub[ii, n] = -1.0
        b_ub[ii] = -corr[r]
    lower = np.maximum(-h, -trust)
    upper = np.minimum(1.0 - h, trust)
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = list(zip(lower, upper)) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            h_new = h + d
            if np.all(h_new >= -1e-12) and np.all(h_new <= 1 + 1e-12):
                h_new = np.clip(h_new, 0, 1)
                h_new = project_box_sum(h_new, n / 2.0)
                return h_new
    except Exception:
        pass
    return None

    # Phase 3: Upscale to 2x and optimize
    if remaining() > 60 and len(best_h) <= 1200:
        h_up = np.repeat(best_h, 2)
        h_up = project_box_sum(h_up, len(h_up) / 2.0)
        h_up, v_up = multi_resolve_optimize(h_up, best_val, remaining, rng)
        if v_up < best_val:
            best_val = v_up
            best_h = h_up.copy()
        # Downscale back
        if len(h_up) > len(best_h):
            h_dn = h_up.reshape(len(h_up)//2, 2).mean(axis=1)
            h_dn = project_box_sum(h_dn, len(h_dn) / 2.0)
            v_dn = true_c5(h_dn)
            if v_dn < best_val:
                best_val = v_dn
                best_h = h_dn.copy()

    # Phase 4: Multiple restarts with different initial structures
    if remaining() > 40:
        n_start = len(best_h)
        restarts = [
            ('binary', make_binary_init(n_start, rng)),
        ]
        # Add sinusoidal inits
        for m in [2, 3, 5, 8]:
            restarts.append(('sin_%d' % m, make_sinusoidal_init(n_start, m, 0.3, rng)))
        # Block inits
        for bs in [4, 6, 10]:
            restarts.append(('blk_%d' % bs, make_block_init(n_start, bs, rng)))
        # Perturbed best
        restarts.append(('perturb', project_box_sum(best_h + 0.1 * rng.standard_normal(n_start), n_start / 2.0)))

        for name, h_init in restarts:
            if remaining() < 15:
                break
            h, v = multi_resolve_optimize(h_init, best_val, remaining, rng)
            if v < best_val:
                best_val = v
                best_h = h.copy()

    # Phase 5: Fine polishing with smaller trust
    if remaining() > 15:
        h = best_h.copy()
        n = len(h)
        target = n / 2.0
        trust = 0.05
        for _ in range(100):
            if remaining() < 5:
                break
            corr = corr_array(h)
            cur = float(np.max(corr))
            if cur < best_val:
                best_val = cur
                h = h.copy()
            d = lp_polish(h, corr, trust=trust, budget_fn=remaining)
            if d is not None:
                v = true_c5(d)
                if v < best_val - 1e-14:
                    best_val = v
                    best_h = d.copy()
                    h = d.copy()
                    trust = min(0.15, trust * 1.2)
                else:
                    trust = max(0.01, trust * 0.7)
            else:
                # Try projected gradient
                for eta in [50, 100, 200]:
                    h_try = projected_gradient_step(h, eta, trust, target)
                    v_try = true_c5(h_try)
                    if v_try < best_val - 1e-14:
                        best_val = v_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        break
                else:
                    trust = max(0.01, trust * 0.8)
                    if trust < 0.015:
                        break

    # Phase 6: Try upscaled version for final push
    if remaining() > 80 and len(best_h) * 2 <= 2400:
        h_up = np.repeat(best_h, 2)
        h_up = project_box_sum(h_up, len(h_up) / 2.0)
        # Quick SLSQP at high eta
        h_up, v_up = slsqp_smooth(h_up, 100.0, remaining, max_iter=150)
        if v_up < best_val:
            best_val = v_up
            best_h = h_up.copy()
        # Polishing
        for _ in range(10):
            if remaining() < 5:
                break
            corr = corr_array(best_h)
            d = lp_polish(best_h, corr, trust=0.05, budget_fn=remaining)
            if d is not None:
                v = true_c5(d)
                if v < best_val:
                    best_val = v
                    best_h = d.copy()
            else:
                break
        # Downscale check
        if len(best_h) > n_base:
            factor = len(best_h) // n_base if n_base > 0 else 1
            if factor > 1 and len(best_h) % factor == 0:
                h_dn = best_h.reshape(len(best_h)//factor, factor).mean(axis=1)
                h_dn = project_box_sum(h_dn, len(h_dn) / 2.0)
                v_dn = true_c5(h_dn)
                if v_dn < best_val:
                    best_val = v_dn
                    best_h = h_dn.copy()

    # Final feasibility check
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_val = true_c5(best_h)

    return best_h, float(best_val), len(best_h)
