            lo = mid

            hi = mid

    h = np.interp(xn, xo, h_src)

def _smooth_obj(h, tau, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    return float(m + tau * np.log(e.sum()))

def _smooth_grad(h, tau, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    omh = 1.0 - h
    grad = np.zeros(n)
    active = np.where(alpha > 1e-10)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        if d >= 0:
            hi = n - 1 - d
            if hi >= 0:
                grad[0:hi+1] += a * omh[d:hi+1+d]
            lo2 = d
            if lo2 < n:
                grad[lo2:n] -= a * h[0:n-lo2]
        else:
            dd = -d
            lo = dd
            if lo < n:
                grad[lo:n] += a * omh[0:n-dd]
            hi2 = n - 1 - dd
            if hi2 >= 0:
                grad[0:hi2+1] -= a * h[dd:hi2+1+dd]
    return grad * dx

def _slsqp_optimize(h0, n, tau_sched, deadline):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - target,
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    h_curr = h0.copy()
    stall = 0
    for tau in tau_sched:
        if time.time() > deadline - 3:
            break
        maxiter = 60 if tau > 0.02 else 100
        def obj(h):
            return _smooth_obj(h, tau, n)
        def grad(h):
            return _smooth_grad(h, tau, n)
        try:
            res = minimize(obj, h_curr, method='SLSQP', jac=grad,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        c5 = evaluate_c5(h_curr)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_curr.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                rng = np.random.default_rng(hash((n, tau)) & 0xFFFFFFFF)
                h_curr = project_feasible(h_curr + rng.standard_normal(n) * 0.003, target)
                stall = 0
    return best_h, best_c5

def _projected_gradient_polish(h0, n, max_steps=200, deadline=None):

    target = n / 2.0
    h = h0.copy()

    best_h = h.copy()
    best_c5 = c5
    dx = 2.0 / n
    alpha = 1.0

        if deadline and time.time() > deadline - 2:

        m = c.max()

    # Resolution 720: refine
    n2 = 720
    if time.time() < deadline - 45:
        h_init2 = interp_to_n(best_h, n2)
        c_init2 = compute_all_overlaps(h_init2)
        spread2 = max(1e-4, c_init2.max() - c_init2.mean())
        tau0_2 = max(0.03, 0.4 * spread2)
        tau_sched2 = tau0_2 * (0.55 ** np.arange(12))
        h_o2, c5_o2 = _slsqp_optimize(h_init2, n2, tau_sched2, deadline)
        if c5_o2 < best_c5:
            best_c5 = c5_o2
            best_h = h_o2
            best_n = n2
        if time.time() < deadline - 10:
            h_p2, c5_p2 = _projected_gradient_polish(h_o2 if c5_o2 < best_c5 else best_h, 
                                                       n2, max_steps=120, deadline=deadline)
            if c5_p2 < best_c5:
                best_c5 = c5_p2
                best_h = h_p2
                best_n = n2
    
    # Resolution 960: further refinement if time allows
    n3 = 960
    if time.time() < deadline - 30:
        h_init3 = interp_to_n(best_h, n3)
        c_init3 = compute_all_overlaps(h_init3)
        spread3 = max(1e-4, c_init3.max() - c_init3.mean())
        tau0_3 = max(0.02, 0.3 * spread3)
        tau_sched3 = tau0_3 * (0.55 ** np.arange(10))
        h_o3, c5_o3 = _slsqp_optimize(h_init3, n3, tau_sched3, deadline)
        if c5_o3 < best_c5:
            best_c5 = c5_o3
            best_h = h_o3
            best_n = n3
        if time.time() < deadline - 5:
            h_p3, c5_p3 = _projected_gradient_polish(h_o3 if c5_o3 < best_c5 else best_h,
                                                       n3, max_steps=100, deadline=deadline)
            if c5_p3 < best_c5:
                best_c5 = c5_p3
                best_h = h_p3
                best_n = n3
    
    # Final polish at best resolution
    if time.time() < deadline - 3:
        h_final, c5_final = _projected_gradient_polish(best_h, best_n, max_steps=80, deadline=deadline)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final
    
    # Final feasibility cleanup
    n_final = len(best_h)
    diff = n_final / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] -= diff / c
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = n_final / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] += diff / c
    best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, len(best_h))
