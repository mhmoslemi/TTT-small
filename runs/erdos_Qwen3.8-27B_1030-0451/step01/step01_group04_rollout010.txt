def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h


def upscale(h, N_new):
    N_old = len(h)
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def weighted_gradient(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    exps = np.clip((c - cmax) / tau, -500.0, 500.0)
    w = np.exp(exps)
    w = w / w.sum()
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return g - np.mean(g)

    if np.any(neg_mask):

        lam_max = min(lam_max, float(np.min(-h[neg_mask] / d[neg_mask])))
    return lam_min, lam_max
