def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix sum exactly
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:

                r2 = target - y.sum()
                if r2 > 1e-12:

def upsample(h, n_new):

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.5, 2.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    r2 = target - y.sum()
    if r2 > 1e-9:
        idx = np.where(y < 1.0)[0]
        if len(idx) > 0:

    lb = np.maximum(h - radius, 0.0)
    ub = np.minimum(h + radius, 1.0)
    bounds_list = list(zip(lb, ub)) + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_list, method='highs')
    return res

def optimize_lp(h0, deadline, max_iters=400, init_radius=0.12, max_active=120):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = init_radius
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        c_cur = v.max()
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 40:
            radius *= 0.5
            stall = 0
            if radius < 0.002:
                break
        threshold = c_cur - max(0.004, 0.015 * c_cur)
        active = np.where(v >= threshold)[0]
        if len(active) == 0:
            active = np.argsort(v)[-10:]
        if len(active) > max_active:
            order = np.argsort(v[active])[::-1]
            active = active[order[:max_active]]
        res = lp_subproblem(h, v, active, radius)
        if res is not None and res.success and res.x is not None:
            h_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new <= c_cur + 1e-10:
                h = h_new
                if c_new < c_cur - 1e-13:
                    if it % 4 == 0:
                        radius = min(radius * 1.12, 0.25)
            else:
                radius *= 0.65
        else:
            radius *= 0.6
    return best_h, best_c

def optimize_constraint_gen(h0, deadline, max_iters=300, init_radius=0.1):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    v = corr_all(h)
    active_set = [int(np.argmax(v))]
    radius = init_radius
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_cur = corr_all(h).max()
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 30:
            break
        v = corr_all(h)
        nvar = n + 1
        c_obj = np.zeros(nvar)
        c_obj[n] = 1.0
        A_rows = []
        b_rows = []
        for k in active_set:
            lag = k - (n - 1)
            g = grad_lag(h, lag)
            A_rows.append(np.append(g, -1.0))
            b_rows.append(v[k] - g @ h)
        A_ub = np.array(A_rows)
        b_ub = np.array(b_rows)
        A_eq = np.zeros((1, nvar))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        lb = np.maximum(h - radius, 0.0)
        ub = np.minimum(h + radius, 1.0)
        bounds_list = list(zip(lb, ub)) + [(None, None)]
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds_list, method='highs')
        if res is not None and res.success and res.x is not None:
            h_new = project_sum_box(res.x[:n], n / 2.0)
            v_new = corr_all(h_new)
            c_new = v_new.max()
            k_new = int(np.argmax(v_new))
            if c_new <= v.max() + 1e-9:
                h = h_new
                if k_new not in active_set:
                    if v_new[k_new] > (res.fun if res.fun is not None else 0) + 1e-5:
                        active_set.append(k_new)
                        if len(active_set) > 250:
                            active_set = active_set[-250:]
                if len(active_set) < 30:
                    radius = min(radius * 1.1, 0.2)
                else:
                    radius = max(radius * 0.97, 0.02)
            else:
                radius *= 0.7
        else:
            radius *= 0.6
            if radius < 0.005:
                break
    return best_h, best_c

def binary_push(h, threshold=0.06):
    h = h.copy()
    h[h < threshold] = 0.0
    h[h > 1.0 - threshold] = 1.0
    h = project_sum_box(h, len(h) / 2.0)
    return h

def upsample(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h, h, h])
    x_ext = np.concatenate([x_old, x_old + 2.0, x_old + 4.0])
    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def fine_grad_opt(h0, deadline, max_iters=5000):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    step = 0.008
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        c_cur = v.max()
        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 400:
            break
        top_k = np.argsort(v)[-min(8, len(v)):]
        g = np.zeros(n)
        for k in top_k:
            lag = k - (n - 1)
            g += grad_lag(h, lag)
        g /= len(top_k)
        gn = np.linalg.norm(g)
        if gn < 1e-11:
            break
        h_new = h - step * g / gn
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < c_cur - 1e-14:
            h = h_new
            step = min(step * 1.08, 0.04)
            stall = 0
        else:
            step *= 0.82
            if step < 1e-8:
                break
    return best_h, best_c

def slsqp_smooth(h0, deadline, betas=None):
    n = len(h0)
    if betas is None:
        betas = [6, 12, 24, 48, 96, 192, 384]
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    for beta in betas:
        if time.time() > deadline:
            break
        for attempt in range(2):
            if time.time() > deadline:
                break
            if attempt == 0:
                x0 = best_h.copy()
            else:
                x0 = best_h + np.random.randn(n) * 0.006
                x0 = project_sum_box(x0, n / 2.0)

            def obj_func(x):
                v = corr_all(x)
                m = v.max()
                expo = np.clip(beta * (v - m), None, 400)
                u = np.exp(expo)
                return m - np.log(u.sum()) / beta

            def grad_func(x):
                v = corr_all(x)
                m = v.max()
                expo = np.clip(beta * (v - m), None, 400)
                u = np.exp(expo)
                w = u / u.sum()
                g = np.zeros(n)
                for ki in range(len(v)):
                    if w[ki] > 1e-12:
                        lag = ki - (n - 1)
                        g += w[ki] * grad_lag(x, lag)
                return g

            try:
                res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 150, 'ftol': 1e-15})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-12:
                    best_h = x_new.copy()
                    best_c = c_new
    return best_h, best_c

def multi_start(h0, deadline, n_starts=4):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    remaining = deadline - time.time()
    per_start = remaining / (n_starts + 1)
    for s in range(n_starts):
        if time.time() > deadline - 20:
            break
        sub_dl = time.time() + per_start * 0.9
        h_pert = h0 + np.random.randn(n) * 0.015
        h_pert = project_sum_box(h_pert, n / 2.0)
        h_opt, c_opt = optimize_lp(h_pert, sub_dl, max_iters=80, init_radius=0.04)
        if c_opt < best_c:
            best_h = h_opt
            best_c = c_opt
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(100, 0.5)
        h_init = project_sum_box(h_init, 50.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    # Phase 1: Quick optimization at native resolution
    t1 = deadline - (deadline - start_time) * 0.30
    h_o1, c_o1 = optimize_lp(best_h, t1, max_iters=200, init_radius=0.1, max_active=100)
    if c_o1 < best_c:
        best_h, best_c = h_o1, c_o1

    # Phase 2: Constraint generation at native resolution
    if time.time() < deadline - 60:
        t2 = deadline - (deadline - start_time) * 0.15
        h_o2, c_o2 = optimize_constraint_gen(best_h, t2, max_iters=200, init_radius=0.06)
        if c_o2 < best_c:
            best_h, best_c = h_o2, c_o2

    # Phase 3: Binary push
    if time.time() < deadline - 50:
        for thresh in [0.05, 0.03, 0.02]:
            h_bp = binary_push(best_h, thresh)
            c_bp = evaluate_c5(h_bp)
            if c_bp < best_c:
                best_h, best_c = h_bp, c_bp
        if time.time() < deadline - 30:
            h_o3, c_o3 = optimize_lp(best_h, deadline - 20, max_iters=50, init_radius=0.03, max_active=80)
            if c_o3 < best_c:
                best_h, best_c = h_o3, c_o3

    # Phase 4: Upsample to higher resolution and optimize
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        n_up = min(max(n_cur * 3, 200), 512)
        n_up = (n_up // 2) * 2
        h_up = upsample(best_h, n_up)
        c_up = evaluate_c5(h_up)
        if c_up < best_c:
            best_h, best_c = h_up, c_up
        else:
            best_h = h_up
        t4 = deadline - (deadline - start_time) * 0.08
        h_o4, c_o4 = optimize_lp(best_h, t4, max_iters=200, init_radius=0.06, max_active=120)
        if c_o4 < best_c:
            best_h, best_c = h_o4, c_o4
        # SLSQP polish at higher res
        if time.time() < deadline - 30:
            h_o4b, c_o4b = slsqp_smooth(best_h, deadline - 15, betas=[8, 16, 32, 64])
            if c_o4b < best_c:
                best_h, best_c = h_o4b, c_o4b

    # Phase 5: Try even higher resolution if time permits
    if time.time() < deadline - 120:
        n_cur = len(best_h)
        n_up2 = min(n_cur * 2, 1024)
        n_up2 = (n_up2 // 2) * 2
        h_up2 = upsample(best_h, n_up2)
        c_up2 = evaluate_c5(h_up2)
        if c_up2 < best_c:
            best_h, best_c = h_up2, c_up2
        else:
            best_h = h_up2
        t5 = deadline - 30
        h_o5, c_o5 = optimize_lp(best_h, t5, max_iters=150, init_radius=0.04, max_active=150)
        if c_o5 < best_c:
            best_h, best_c = h_o5, c_o5

    # Phase 6: Multi-start at best resolution
    if time.time() < deadline - 60:
        h_ms, c_ms = multi_start(best_h, deadline - 10, n_starts=3)
        if c_ms < best_c:
            best_h, best_c = h_ms, c_ms

    # Phase 7: Final fine gradient polish
    if time.time() < deadline - 10:
        h_fp, c_fp = fine_grad_opt(best_h, deadline - 1, max_iters=3000)
        if c_fp < best_c:
            best_h, best_c = h_fp, c_fp

    # Final projection and verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
