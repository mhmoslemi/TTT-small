        grad = smooth_max_grad(h_curr, tau)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_curr + (t - 1.0) / t_new * (h_curr - h_prev)
        y_clipped = np.clip(y, 0.0, 1.0)
        
        h_new = project_feasible(y_clipped - alpha * grad, target_sum)
        
        if np.max(np.abs(h_new - h_curr)) < 1e-10:
            break
        
        h_prev = h_curr
        h_curr = h_new
        t = t_new
    
    return h_curr

The problem with FISTA here is that the objective isn't strongly convex, so the convergence rate is only O(1/k^2) for the objective value. Also, the projection onto {0 ≤ h ≤ 1, sum(h) = target} is not the same as the proximal operator of the indicator function in the standard sense (it's a projection onto a non-box set).

Actually, FISTA works for any projection onto a closed convex set, not just the box. So it should work here. The convergence rate is O(L/k^2) where L is the Lipschitz constant of the gradient.

The gradient of the smooth max has Lipschitz constant related to 1/tau (for small tau). So for small tau, we need small steps, which is bad.

For the FISTA phase, I should use a moderate tau (like 0.01-0.05), and then use SLSQP for the final polish with small tau.

Or **harmonic mean of (max - c_i)**: this penalizes being far from max.

Or **top-2 average**: (c_max + c_second_max) / 2. This is simpler and avoids the tau parameter.

Let me try the top-2 approach:
- Identify the top 2 shifts
- Minimize their average (or max)
- This is a 2-term max, which is easier to optimize

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = 0.5 * (lo + hi)
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 1e-12)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h


def upscale_func(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return project_feasible(h.copy(), N_new / 2.0)
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def smooth_max_obj(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = np.clip((c - m) / tau, -500.0, 500.0)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    return m + tau * log_sum_exp


def smooth_max_obj_and_weights(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(shifted)
    total = w.sum()
    w = w / total
    obj = m + tau * np.log(total)
    return obj, w


def smooth_max_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(shifted)
    w = w / w.sum()

    grad = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                grad[d:] += a * (1.0 - h[:n])
                grad[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                grad[:n] += a * (1.0 - h[dd:])
                grad[dd:] -= a * h[:n]
    return grad


def fista_optimize(h_init, target_sum, tau, max_iter=3000, step=None):
    N = len(h_init)
    h_curr = h_init.copy()
    h_prev = h_init.copy()
    t = 1.0
    if step is None:
        step = 0.3 / (2.0 * N)

    for it in range(max_iter):
        grad = smooth_max_grad(h_curr, tau)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h_curr + ((t - 1.0) / t_new) * (h_curr - h_prev)
        y = np.clip(y, 0.0, 1.0)
        h_new = project_feasible(y - step * grad, target_sum)
        diff = np.max(np.abs(h_new - h_curr))
        if diff < 1e-12:
            break
        h_prev = h_curr
        h_curr = h_new
        t = t_new
    return h_curr


def slsqp_f(h, tau):
    return smooth_max_obj(h, tau)


def slsqp_g(h, tau):
    return smooth_max_grad(h, tau)


def sum_eq(h, target):
    return np.sum(h) - target


def sum_eq_jac(h, target):
    return np.ones_like(h)


def slsqp_optimize(h, bounds, constraint, tau, maxiter=500):
    try:
        result = minimize(
            slsqp_f, h.copy(),
            jac=slsqp_g,
            args=(tau,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': maxiter, 'ftol': 1e-15}
        )
        return result.x
    except Exception:
        return h.copy()


def structured_perturb(h, rng, N):
    """Apply a structured perturbation: shift a block, swap, or modify a segment."""
    target_sum = N / 2.0
    method = rng.integers(0, 3)
    h_p = h.copy()

    if method == 0:
        # Shift a contiguous block
        block_len = rng.integers(5, max(6, N // 10))
        start = rng.integers(0, N - block_len)
        shift = rng.integers(-3, 4)
        if shift == 0:
            return h_p
        segment = h_p[start:start + block_len].copy()
        h_p[start:start + block_len] = h_p[start + shift:start + block_len + shift]
        if shift > 0:
            h_p[start + block_len + shift:start + block_len + 2 * shift] = segment
        elif shift < 0:
            h_p[start + 2 * shift + block_len:start + block_len + shift] = segment
    elif method == 1:
        # Swap two segments
        len1 = rng.integers(3, max(4, N // 15))
        len2 = rng.integers(3, max(4, N // 15))
        s1 = rng.integers(0, N - len1)
        s2 = rng.integers(0, N - len2)
        if s2 < s1:
            s1, s2 = s2, s1
        if s1 + len1 > s2:
            return h_p
        tmp1 = h_p[s1:s1 + len1].copy()
        tmp2 = h_p[s2:s2 + len2].copy()
        h_p[s1:s1 + len1] = tmp2[:len1]
        h_p[s2:s2 + len2] = tmp1[:len2]
    else:
        # Gaussian noise on a random segment
        seg_len = rng.integers(10, max(11, N // 5))
        start = rng.integers(0, N - seg_len)
        noise = rng.normal(0, 0.08, seg_len)
        h_p[start:start + seg_len] += noise

    h_p = np.clip(h_p, 0.0, 1.0)
    h_p = project_feasible(h_p, target_sum)
    return h_p
