import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def softmax_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    return m + tau * np.log(e.sum())

def softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    B = np.zeros(n)
    B[0] = alpha[n - 1]
    if n > 1:
        B[1:] = alpha[n:2 * n - 1] + alpha[n - 2::-1]
    p1 = np.convolve(B, 1.0 - h, 'full')
    p2 = np.correlate(h, B, 'full')
    grad = dx * (p1[0:n] - p2[n - 1:n - 1 + n])
    return grad

def power_mean_obj(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    c_pos = np.maximum(c, 1e-30)
    return float(np.mean(c_pos ** p) ** (1.0 / p))

def power_mean_grad(h, p):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    c_pos = np.maximum(c, 1e-30)
    pm = np.mean(c_pos ** p) ** (1.0 / p)
    w = (p / (2 * n - 1)) * (c_pos ** (p - 1)) / pm ** (p - 1)
    alpha = w
    B = np.zeros(n)
    B[0] = alpha[n - 1]
    if n > 1:
        B[1:] = alpha[n:2 * n - 1] + alpha[n - 2::-1]
    p1 = np.convolve(B, 1.0 - h, 'full')
    p2 = np.correlate(h, B, 'full')
    return dx * (p1[0:n] - p2[n - 1:n - 1 + n])

def slsqp_phase(h0, n, tau_list, maxiter_list, deadline):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    constraints = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target),
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    h_curr = h0.copy()
    
    for i, tau in enumerate(tau_list):
        if time.time() > deadline - 5:
            break
        mi = maxiter_list[i] if i < len(maxiter_list) else maxiter_list[-1]
        
        def obj_fn(x, t=tau):
            return softmax_obj(x, t)
        
        def grad_fn(x, t=tau):
            return softmax_grad(x, t)
        
        try:
            res = minimize(obj_fn, h_curr, method='SLSQP', jac=grad_fn,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': mi, 'ftol': 1e-14, 'maxls': 20})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        
        c5e = evaluate_c5(h_curr)
        if c5e < best_c5 - 1e-12:
            best_c5 = c5e
            best_h = h_curr.copy()
    
    return best_h, best_c5

def projected_gradient_polish(h0, n, deadline, max_iter=8000):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h_curr = h0.copy()
    p = 30
    step = 1.0
    n_ks = 2 * n - 1
    
    for it in range(max_iter):
        if time.time() > deadline - 3:
            break
        g = power_mean_grad(h_curr, p)
        step = max(0.01, step * 0.99)
        h_try = project_feasible(h_curr - step * g, target)
        f_try = power_mean_obj(h_try, p)
        f_curr = power_mean_obj(h_curr, p)
        
        if f_try >= f_curr:
            step *= 0.5
            if step < 1e-6:
                step = 0.1
                h_curr = project_feasible(h_curr + np.random.default_rng(it).standard_normal(n) * 0.002, target)
            continue
        
        h_curr = h_try
        c5e = evaluate_c5(h_curr)
        if c5e < best_c5 - 1e-12:
            best_c5 = c5e
            best_h = h_curr.copy()
    
    return best_h, best_c5

def slsqp_finale(h0, n, tau, deadline, maxiter=300):
    target = n / 2.0
    constraints = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target),
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    def obj_fn(x):
        return softmax_obj(x, tau)
    
    def grad_fn(x):
        return softmax_grad(x, tau)
    
    try:
        res = minimize(obj_fn, h0, method='SLSQP', jac=grad_fn,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15, 'maxls': 30})
        h_out = project_feasible(res.x, target)
    except Exception:
        h_out = h0.copy()
    
    return h_out, evaluate_c5(h_out)

def try_diverse_starts(parent, n, deadline, rng):
    """Generate diverse starting points and optimize from each."""
    results = []
    
    # Start 1: direct interpolation
    h1 = interp_h(parent, n)
    results.append(h1)
    
    # Start 2: smooth interpolation (add a bit of smoothing)
    h2 = interp_h(parent, n)
    kernel = np.array([1, 2, 3, 2, 1], dtype=np.float64)
    kernel /= kernel.sum()
    h2 = np.convolve(h2, kernel, mode='same')
    h2 = project_feasible(h2, n / 2.0)
    results.append(h2)
    
    # Start 3: small random perturbation
    h3 = project_feasible(h1 + rng.standard_normal(n) * 0.01, n / 2.0)

    # Start 4: medium random perturbation
    h4 = project_feasible(h1 + rng.standard_normal(n) * 0.03, n / 2.0)

    results.append(h5)
    
    return results
