def compute_line_coeffs(c, h, a, b):
    """Compute Lin(lag) and Q2(lag) for transfer pair (a,b).
    Returns arrays of length 2n-1."""
    n = len(h)
    center = n - 1
    Lc = 2 * n - 1
    
    # Lin(K) = h[a-lag] - h[b-lag] + h[a+lag] - h[b+lag]
    # where lag = K - center, and h[j] = 0 if j not in [0,n-1]
    Lin = np.zeros(Lc)
    for K in range(Lc):
        lag = K - center
        val = 0.0
        if 0 <= a - lag < n:
            val += h[a - lag]
        if 0 <= b - lag < n:
            val -= h[b - lag]
        if 0 <= a + lag < n:
            val += h[a + lag]
        if 0 <= b + lag < n:
            val -= h[b + lag]
        Lin[K] = val
    
    # Q2: 2 at lag=0 (K=center), -1 at lag=a-b and lag=b-a
    Q2 = np.zeros(Lc)
    Q2[center] = 2.0
    if a != b:
        ab = a - b
        if 0 <= center + ab < Lc:
            Q2[center + ab] -= 1.0
        if 0 <= center - ab < Lc:
            Q2[center - ab] -= 1.0
    
    return Lin, Q2

def line_search(c, h, a, b, n_d=100):
    """Find best d for transfer pair (a,b).
    Returns (best_d, best_score) or (None, current_score) if no improvement."""
    n = len(h)
    current_score = float(np.max(c))
    
    d_lo = max(-h[a], h[b] - 1.0)
    d_hi = min(1.0 - h[a], h[b])
    
    if d_hi - d_lo < 1e-12:
        return None, current_score
    
    Lin, Q2 = compute_line_coeffs(c, h, a, b)
    
    # c_new(K) = c[K] - d*Lin[K] - d^2*Q2[K]
    # We want to minimize max_K c_new(K) over d in [d_lo, d_hi]
    
    d_vals = np.linspace(d_lo, d_hi, n_d)
    
    # For each d, compute max_K (c[K] - d*Lin[K] - d^2*Q2[K])

    Lin_row = Lin[np.newaxis, :]  # (1, 2n-1)
    Q2_row = Q2[np.newaxis, :]  # (1, 2n-1)
    
    scores = c_row - d_col * Lin_row - d_col**2 * Q2_row  # (n_d, 2n-1)
