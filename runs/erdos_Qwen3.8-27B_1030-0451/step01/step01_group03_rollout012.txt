import numpy as np
from scipy.optimize import minimize
import time

_G_T = 0.0
_G_N = 0
_G_TAU = 1.0

def _c_eq_fun(h):
    return float(np.sum(h) - _G_T)

def _c_eq_jac(h):
    return np.ones(_G_N)

def _smooth_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _smooth_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        a = alpha[k]
        if a < 1e-10:
            continue
        lo1 = max(0, k)
        hi1 = min(n - 1, k + n - 1)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += a * (1.0 - h[lo1-k:hi1-k+1])
        if k < n:
            lo2 = 0
            hi2 = min(n - 1, n - 1 - k)
            grad[lo2:hi2+1] -= a * h[lo2+k:hi2+k+1]
    return grad * dx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_ = len(hp)
    if nn == np_:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def slsqp_optimize(h0, n, deadline, n_tau=14):
    global _G_T, _G_N, _G_TAU
    _G_T = n / 2.0
    _G_N = n
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    c_vals = corr_vec(h0)
    spread = max(1e-6, c_vals.max() - c_vals.mean())
    tau0 = max(0.012, 0.25 * spread)
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]
    h_curr = h0.copy()
    for i in range(n_tau):
        if time.time() > deadline - 5:
            break
        tau = tau0 * (0.52 ** i)
        if tau < 0.0004:
            tau = 0.0004
        _G_TAU = tau
        maxiter = 60 if tau > 0.01 else 100
        try:

                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        c5 = evaluate_c5(h_curr)
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h_curr.copy()
    return best_h, best_c5

def subgrad_of_max(h, top_k=5, window=0.02):
    """Compute subgradient of max(c) using top-k active lags."""
    n = len(h)
    c = corr_vec(h)
    cmax = c.max()
    active = np.where(c >= cmax - window)[0]
    if len(active) > top_k:
        idx = np.argsort(c[active])[::-1][:top_k]
        active = active[idx]
    grad = np.zeros(n)
    for k in active:
        lo1 = max(0, k)
        hi1 = min(n - 1, k + n - 1)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += (1.0 - h[lo1-k:hi1-k+1])
        if k < n:
            lo2 = 0
            hi2 = min(n - 1, n - 1 - k)
            grad[lo2:hi2+1] -= h[lo2+k:hi2+k+1]
    grad /= len(active)
    return grad, cmax, active

    """Direct projected gradient descent on the true max objective."""

    best_c5 = evaluate_c5(h)

    step = 0.02

    for it in range(max_iters):
        if time.time() > deadline - 5:

        norm = np.linalg.norm(direction)

            step = min(step * 1.05, 0.05)
        else:
            patience += 1
            step *= 0.8

    # All 0.5
    starts.append(np.full(n_par, 0.5))
    return starts
