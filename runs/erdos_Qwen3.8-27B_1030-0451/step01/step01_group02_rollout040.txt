import numpy as np
import time

def make_feasible(x, target):
    n = len(x)
    if n == 0:
        return x
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    lo = -1.0 - x.max()
    hi = 1.0 - x.min()
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-11:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                x[idx] += diff / idx.size
                x = np.clip(x, 0.0, 1.0)
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                x[idx] += diff / idx.size
                x = np.clip(x, 0.0, 1.0)
    return x

def corr_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, "full") * dx

def C5val(h):
    if len(h) == 0:
        return 0.0
    return float(np.max(corr_vals(h)))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = corr_vals(h)
    m = c.max()
    if tau < 1e-14:
        act = c >= m - 1e-10
        cnt = act.sum()
        if cnt == 0:
            act[np.argmax(c)] = True
            cnt = 1
        alpha = np.zeros_like(c)
        alpha[act] = 1.0 / cnt
    else:
        a = np.clip((c - m) / tau, -60, 0)
        e = np.exp(a)
        alpha = e / e.sum()
    G = np.zeros(n)
    ctr = n - 1
    for idx in range(len(c)):
        w = alpha[idx]
        if w < 1e-14:
            continue
        k = idx - ctr
        ws = dx * w
        if k == 0:
            G += ws * (1.0 - 2.0 * h)
        elif k > 0:
            G[:n-k] += ws * (1.0 - h[k:])
            G[k:] -= ws * h[:n-k]
        else:
            kk = -k
            G[kk:] += ws * (1.0 - h[:n-kk])
            G[:n-kk] -= ws * h[kk:]
    return G

def line_search(h, s, target, c0):
    n = len(h)
    pos = s > 1e-14
    neg = s < -1e-14
    lam_hi = np.inf
    lam_lo = 0.0
    if pos.any():
        lam_hi = min(lam_hi, ((1.0 - h[pos]) / s[pos]).min())
    if neg.any():
        lam_lo = max(lam_lo, (-h[neg] / s[neg]).min())
    if lam_hi <= lam_lo + 1e-12:
        return None, c0
    best_lam = 0.0
    best_c = c0
    fracs = [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1, 0.05, 0.03, 0.01]
    for f in fracs:
        lam = f * lam_hi
        if lam <= lam_lo + 1e-12:
            continue
        h_new = make_feasible(h + lam * s, target)
        c_new = C5val(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_lam = lam
    if best_lam <= 0:
        return None, c0
    return best_lam, best_c

def refine_grad(h, target, deadline, tau_init=0.04, tau_min=0.0002, momentum=0.9):
    n = len(h)
    h = h.copy()
    c = C5val(h)
    vel = np.zeros(n)
    tau = tau_init
    stall = 0
    iters = 0
    while time.time() < deadline - 8:
        grad = smooth_grad(h, tau)
        grad -= grad.mean()
        vel = momentum * vel - grad
        s = -vel
        norm = np.linalg.norm(s)
        if norm < 1e-15:
            vel *= 0.5
            continue
        s /= norm
        lam, c_new = line_search(h, s, target, c)
        if lam is not None and c_new < c - 1e-14:
            h = make_feasible(h + lam * s, target)
            c = c_new
            stall = 0
            iters += 1
        else:
            stall += 1
            vel *= 0.5
            if stall >= 8:
                tau *= 0.6
                if tau < tau_min:
                    tau = tau_min
                    stall = 0
            if stall >= 30:
                break
    return h, c, tau

def coord_polish(h, target, deadline, rng):
    n = len(h)
    c = C5val(h)
    for _ in range(5000):
        if time.time() > deadline - 5:
            break
        i = int(rng.integers(n))
        for j in range(n):
            if j == i or abs(j - i) > 200:
                continue
            for delta in (0.02, 0.05, 0.1, -0.02, -0.05, -0.1):
                h_new = h.copy()
                h_new[i] += delta
                h_new[j] -= delta
                if h_new[i] < -0.01 or h_new[i] > 1.01 or h_new[j] < -0.01 or h_new[j] > 1.01:

                h_new = make_feasible(h_new, target)
                c_new = C5val(h_new)
                if c_new < c - 1e-13:
                    h = h_new
                    c = c_new
                    break
    return h, c

def resample_h(h, N):
    n0 = len(h)
    if n0 == N:
        return make_feasible(h.copy(), N / 2.0)
    target = N / 2.0

    x_new = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(x_new, x_old, h)
    return make_feasible(y, target)

def fourier_start(N, rng):
