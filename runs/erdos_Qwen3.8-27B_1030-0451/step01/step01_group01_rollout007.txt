    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    """Gradient of correlation at given lag (unscaled, no dx)."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            m = np.arange(0, n - lag)
            g[m] += (1.0 - h[m + lag])
            g[m + lag] -= h[m]
    else:
        l = -lag
        if l == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            m = np.arange(0, n - l)
            g[m + l] += (1.0 - h[m])
            g[m] -= h[m + l]
    return g

def hessian_lag(n, lag):
    """Hessian of correlation at given lag (unscaled, no dx)."""
    H = np.zeros((n, n))
    if lag == 0:
        H[np.arange(n), np.arange(n)] = -2.0
    elif lag > 0:
        m = np.arange(0, n - lag)
        H[m, m + lag] = -1.0
        H[m + lag, m] = -1.0
    else:
        l = -lag
        m = np.arange(0, n - l)
        H[m + l, m] = -1.0
        H[m, m + l] = -1.0
    return H

def smooth_newton(h0, deadline, max_iters=2000):

    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(h0)
    
    beta = 20.0  # starting smoothing parameter
    beta_growth = 1.05
    max_beta = 500.0
    
    stall = 0

        if time.time() > deadline:
            break
        
        # Compute correlations
        c = corr_all(h)
        c_max = c.max()
        
        # Update best
        if c_max < best_c - 1e-12:
            best_c = c_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 200:
            # Try increasing beta
            beta = min(beta * 2.0, max_beta)
            stall = 0
            if beta >= max_beta:
                break

        expo = np.clip(beta * (c - c_max), None, 500)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z

        g = np.zeros(n)

                lag = k - (n - 1)
                g += w[k] * grad_lag(h, lag)

        # Cov term: sum_k w_k (g_k - g)(g_k - g)^T where g_k = grad_lag(h, lag_k) * dx
        # This is expensive to compute fully. Let's approximate.
        
        # Actually, for the Hessian, the second term is:
        # dx * sum_k w_k * H_lag(k)
        # where H_lag(k) is the Hessian of c_k
        
        # The first term (covariance) is:
        # beta * dx^2 * sum_k w_k (g_k - g)(g_k - g)^T
        
        # For efficiency, let's compute the "average Hessian" part and add a 
        # regularization that approximates the covariance.
        
        # Compute weighted average Hessian
        H_avg = np.zeros((n, n))
        for k in range(len(c)):
            if w[k] > 1e-10:
                lag = k - (n - 1)
                H_avg += w[k] * hessian_lag(n, lag)
        H_avg *= dx
        
        # Add regularization to ensure positive definiteness
        # The covariance term is PSD, so we add a diagonal regularizer
        eig_approx = np.diag(H_avg).mean() if n > 0 else 0
        reg = max(0.01, -eig_approx * 0.5 + 0.05)
        H_total = H_avg + reg * np.eye(n)

        try:
            d = np.linalg.solve(H_total, -g)
        except:
            d = -g / (np.linalg.norm(g) + 1e-10)
        
        # Limit step size
        d_norm = np.linalg.norm(d)
        if d_norm > 0.5:
            d = d * 0.5 / d_norm
        
        # Line search
        h_old = h.copy()
        c_old = c_max
        alpha = 1.0
        for ls in range(10):

            h_new = project_sum_box(h_new, n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new < c_old - 1e-14:

        else:

            g_norm = np.linalg.norm(g)
            if g_norm < 1e-10:

                h = h_new
            else:
                beta = min(beta * 1.5, max_beta)
                if beta >= max_beta:
                    break
                stall += 10
        
        # Gradually increase beta
        if it % 20 == 0:
            beta = min(beta * beta_growth, max_beta)
    
    return best_h, best_c

def project_sum_box(x, target):

    x = np.array(x, dtype=float).copy()

        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:

    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:
        m = np.arange(0, n - lag)
        g[m] += (1.0 - h[m + lag])
        g[m + lag] -= h[m]
    else:
        l = -lag
        m = np.arange(0, n - l)
        g[m + l] += (1.0 - h[m])
        g[m] -= h[m + l]
    return g

def build_weighted_hessian(w, n, dx):
    """Build Hessian of sum_k w[k]*c_k(h). Returns n x n matrix."""
    H = np.zeros((n, n))
    for k in range(len(w)):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        if lag == 0:
            H[np.arange(n), np.arange(n)] += -2.0 * w[k] * dx
        elif lag > 0:
            m = np.arange(0, n - lag)
            H[m, m + lag] += -w[k] * dx
            H[m + lag, m] += -w[k] * dx
        else:
            l = -lag
            m = np.arange(0, n - l)
            H[m + l, m] += -w[k] * dx
            H[m, m + l] += -w[k] * dx
    return H

            pass

    # Phase 2: Newton smoothed on initial grid
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.40)
        try:
            h2, c2 = newton_smoothed(best_h, dl, max_iters=2000)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # Phase 3: Transfer to higher resolution and optimize
    if time_left() > 120:
        n_cur = len(best_h)
        for n_target in [n_cur * 2, n_cur * 2 + 1]:
            if time_left() < 100:
                break
            if n_target > 400:
                break
            try:
                h_tr = transfer_resolution(best_h, n_target)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c + 0.001:
                    best_h = h_tr.copy()
                    best_c = min(best_c, c_tr)
                    sub_dl = min(deadline, time.time() + min(200, time_left() * 0.5))
                    h_a, c_a = lp_trust(best_h, sub_dl, max_iters=300)
                    if c_a < best_c:
                        best_h, best_c = h_a, c_a
                    if time_left() > 60:
                        sub_dl2 = min(deadline, time.time() + min(100, time_left() * 0.4))
                        h_b, c_b = newton_smoothed(best_h, sub_dl2, max_iters=1500)
                        if c_b < best_c:
                            best_h, best_c = h_b, c_b
            except Exception:
                pass
            break  # Only try one higher resolution

    # Phase 4: SLSQP refinement
    if time_left() > 60:
        dl = min(deadline, time.time() + min(150, time_left() * 0.5))
        try:
            h4, c4 = slsqp_smoothed(best_h, dl, max_betas=6)
            if c4 < best_c:
                best_h, best_c = h4, c4
        except Exception:
            pass

    # Phase 5: LP trust region again on best grid
    if time_left() > 40:
        dl = min(deadline, time.time() + min(120, time_left() * 0.5))
        try:
            h5, c5 = lp_trust(best_h, dl, max_iters=300)
            if c5 < best_c:
                best_h, best_c = h5, c5
        except Exception:
            pass

    # Phase 6: Final Newton polish
    if time_left() > 30:
        try:
            h6, c6 = newton_smoothed(best_h, deadline - 2, max_iters=3000)
            if c6 < best_c:
                best_h, best_c = h6, c6
        except Exception:
            pass

    # Phase 7: Final PG polish
    if time_left() > 15:
        try:
            h7, c7 = pg_polish(best_h, deadline - 1, max_iters=5000)
            if c7 < best_c:
                best_h, best_c = h7, c7
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
