import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(h, tau, penalty, target_sum):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    obj = m + tau * np.log(total)
    if penalty > 0:
        obj += penalty * (np.sum(h) - target_sum) ** 2
    return obj


def smooth_grad(h, tau, penalty, target_sum):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    w = w / total
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    if penalty > 0:
        g += 2.0 * penalty * (np.sum(h) - target_sum)
    return g


def project_box_sum(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    return h


def upsample_func(h_old, N_new):
    N_old = len(h_old)
    if N_old == N_new:
        return h_old.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, N_new / 2.0)


def lbfgsb_optimize(h, tau, target_sum, maxiter, penalty):
    N = len(h)
    bounds = [(0.0, 1.0)] * N
    try:
        result = minimize(
            smooth_obj, h.copy(),
            jac=smooth_grad,
            args=(tau, penalty, target_sum),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8, 'maxls': 20}
        )
        return project_box_sum(result.x, target_sum)
    except Exception:
        return h


def projected_gradient(h0, tau, target_sum, n_steps, lr):
    h = h0.copy()
    for _ in range(n_steps):
        g = smooth_grad(h, tau, 0.0, target_sum)
        h_new = h - lr * g
        h_new = project_box_sum(h_new, target_sum)
        if compute_c5(h_new) <= compute_c5(h) + 1e-12:
            h = h_new
        else:
            lr *= 0.5
            h_new = h - lr * g
            h_new = project_box_sum(h_new, target_sum)
            if compute_c5(h_new) <= compute_c5(h) + 1e-12:
                h = h_new
            if lr < 1e-8:
                break
    return h


def make_block_init(N):
    h = np.zeros(N)
    h[:N//4] = 1.0
    h[3*N//4:] = 1.0
    h[N//4:3*N//4] = 0.0
    return project_box_sum(h, N / 2.0)


def make_spread_init(N):
    h = np.ones(N) * 0.5
    return project_box_sum(h, N / 2.0)
