    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def proj_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=2.0, neginf=-2.0)

    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)

            h[idx] = np.clip(h[idx] + diff, 0.0, 1.0)

    return h

    """Gradient of correlation at lag k (k = index - (n-1), so k in [-(n-1), n-1])"""
    g = np.zeros(n)
    if k >= 0:

            g[k:n] += dx * (1.0 - h[0:n-k])

            g[0:n-k] -= dx * h[k:n]

        s = np.clip((c_act - cmax) / tau, -100, 100)

        w = e / e.sum()
        results.append((active, w))
