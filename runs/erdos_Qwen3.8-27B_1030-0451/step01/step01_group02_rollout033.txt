    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):

        if np.sum(np.clip(x+mid, 0, 1)) > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-12:
        if diff > 0:

        for i in idx:

                x[i] += add
                diff -= add
            else:
                dec = min(x[i], diff)
                x[i] -= dec

            if abs(diff) < 1e-12:

def compute_gradient(h, tau):
    """Compute gradient of smooth max of correlation lags."""
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    Cmax = np.max(C)
    exp_C = np.exp((C - Cmax) / tau)
    w = exp_C / np.sum(exp_C)
    
    grad = np.zeros(n)
    center = n - 1
    for k in range(len(w)):
        wk = w[k]
        if wk < 1e-8:
            continue
        d = k - center
        if d == 0:

            grad += wk * dx * (1.0 - 2.0 * h)
        elif d > 0:

            # dC/dh[j] = dx*(1-h[j+d]) for j in [0, n-1-d]
            #           - dx*h[j-d] for j in [d, n-1]
            grad[:n-d] += wk * dx * (1.0 - h[d:])

def score_exact(h):
    n = len(h)
    if n == 0:

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(c))

def make_feasible(x, target):

    # Binary search for shift
    lo, hi = -max(float(np.max(x)), 1.0), 1.0 - min(float(np.min(x)), 0.0)
    for _ in range(60):
        mid = 0.5*(lo+hi)
        if np.sum(np.clip(x+mid, 0, 1)) > target:
            hi = mid
        else:

This is a sum of at most 4 terms, each involving a specific index of h. The key insight is that for a given (p,q), the delta is non-zero only for specific d values:
- d = 0 to n-1-p (p is valid as h[i])

Since d ranges from 1 to n-1:
- p as h[i]: d in [1, min(n-1, n-1-p)]
- p as h[i+d]: d in [max(1, p-n+1), min(n-1, p)]

For typical p in [0, n-1], these ranges cover most of [1, n-1]. So the delta is non-zero for almost all d.
