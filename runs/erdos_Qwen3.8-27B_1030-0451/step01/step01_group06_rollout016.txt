import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    # Water-filling to project onto {0 <= x <= 1, sum(x) = target_sum}
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fine-tune
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def smooth_max_obj(h, beta):
    """Smooth max of correlations using log-sum-exp."""
    n = len(h)
    corr = correlation_array(h)
    # Log-sum-exp trick for stability
    m = np.max(corr)
    if beta * m > 700:
        # Scale to avoid overflow
        val = m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta
    else:
        val = np.log(np.sum(np.exp(beta * corr))) / beta
    return val

def smooth_max_grad(h, beta):
    """Gradient of smooth max of correlations."""
    n = len(h)
    dx = 2.0 / n
    corr = correlation_array(h)
    # Compute softmax weights
    m = np.max(corr)
    log_weights = beta * (corr - m) - np.log(np.sum(np.exp(beta * (corr - m))))
    weights = np.exp(log_weights)
    
    # Now compute gradient as weighted sum of individual correlation gradients
    grad = np.zeros(n)
    base = n - 1
    
    for lag in range(-(n-1), n):
        w = weights[base + lag]
        if w < 1e-15:
            continue
        if lag == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            # Q_lag = dx * sum_{i=0}^{n-lag-1} h[i+lag] * (1-h[i])
            # d/dh[j]: if j >= lag: dx*(1-h[j-lag]); if j < n-lag: -dx*h[j+lag]
            grad[lag:] += w * dx * (1.0 - h[:n-lag])
            grad[:n-lag] += -w * dx * h[lag:]
        else:
            d = -lag
            # Q_{-d} = dx * sum_{i=0}^{n-d-1} h[i] * (1-h[i+d])
            # d/dh[j]: if j < n-d: dx*(1-h[j+d]); if j >= d: -dx*h[j-d]
            grad[:n-d] += w * dx * (1.0 - h[d:])
            grad[d:] += -w * dx * h[:n-d]
    
    return grad

    h = h_flat.copy()  # SLSQP might modify, so copy
    val = smooth_max_obj(h, beta)
    grad = smooth_max_grad(h, beta)
    return val, grad

def opt_smooth_max(h0, beta_start, beta_end, n_iters, deadline, target_sum):
    """Optimize using SLSQP with smooth max objective, increasing beta."""
    h = h0.copy()
    n = len(h)
    best_h = h.copy()

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target_sum,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    betas = np.geomspace(beta_start, beta_end, n_iters)
    for i, beta in enumerate(betas):

            break
        try:

                lambda x: smooth_max_obj(x, beta),
                h,
                jac=lambda x: smooth_max_grad(x, beta),
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 200, 'ftol': 1e-10, 'disp': False}
            )
            if res.success or res.fun < smooth_max_obj(h, beta):
                h = project_feasible(res.x, target_sum)
                val = evaluate_c5(h)
                if val < best_val:
                    best_val = val
                    best_h = h.copy()
        except Exception:
            pass
    
    return best_h, best_val

    return np.repeat(h, factor)

def downscale(h, factor):
    n = len(h)
    new_n = n // factor
    return h[:new_n * factor].reshape(new_n, factor).mean(axis=1)

        best_h = h.copy()
    
    # Phase 2: Try flip

        v_flip = evaluate_c5(h_flip)
        if v_flip < best_val:
            best_val = v_flip
            best_h = h_flip.copy()

import numpy as np

def evaluate_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x

    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)

                h[idx] -= dec
                res -= dec

                inc = min(-res, 1.0 - h[idx])

                res += inc

def smooth_max_val(corr, beta):
    m = np.max(corr)
    shifted = beta * (corr - m)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def smooth_max_grad(h, beta):

    corr = correlation_array(h)
    m = np.max(corr)
    shifted = beta * (corr - m)

        else:
            d = -lag
            grad[:n - d] += w * dx * (1.0 - h[d:])

def pnorm_val(h, p):
    corr = correlation_array(h)
    return float(np.sum(corr ** p) ** (1.0 / p))

def pnorm_grad(h, p):

    dx = 2.0 / n
    corr = correlation_array(h)
    S = np.sum(corr ** p)
    S_qp = S ** (1.0 / p - 1.0)

def downscale(h, factor):
