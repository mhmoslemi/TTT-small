def fista_lse(h0, rng, time_limit, sigma, target, deadline):

    best_c = cur

    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        # Try several random moves
        improved = False
        for _ in range(20):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            d = d_steps[int(rng.integers(len(d_steps)))]
            direction = 1 if rng.random() < 0.5 else -1
            new_val_i = h[i] + direction * d
            new_val_j = h[j] - direction * d
            if new_val_i < -1e-9 or new_val_i > 1 + 1e-9 or new_val_j < -1e-9 or new_val_j > 1 + 1e-9:
                continue
            a = h.copy()
            a[i] = np.clip(new_val_i, 0, 1)
            a[j] = np.clip(new_val_j, 0, 1)
            na = objective(a)
            if na < cur - 1e-14:
                h = a
                cur = na
                improved = True
                if na < best_c - 1e-14:
                    best_c = na
                    best_h = a.copy()
                break
        if not improved:
            # Try swap
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                a = h.copy()
                a[i], a[j] = a[j], a[i]
                na = objective(a)
                if na < cur - 1e-14:
                    h = a
                    cur = na
                    if na < best_c - 1e-14:
                        best_c = na
                        best_h = a.copy()
    
    return best_h, best_c

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)
