def compute_weighted_gradient(h, overlaps, beta=5.0):
    N = len(h)
    dx = 2.0 / N
    
    max_ov = np.max(overlaps)
    exponents = beta * (overlaps - max_ov)
    weights = np.exp(exponents)
    w_sum = weights.sum()
    if w_sum < 1e-30:
        weights = np.zeros_like(overlaps)
        weights[np.argmax(overlaps)] = 1.0
    else:
        weights /= w_sum
    
    g = np.zeros(N)
    center = N - 1
    
    for r in range(2 * N - 1):
        w = weights[r]
        if w < 1e-15:
            continue
        d = abs(r - center)
        if d == 0:
            # θ(0) = dx * sum_i h[i]*(1-h[i])
            # ∂/∂h[k] = dx * (1-2*h[k])
            g += w * dx * (1.0 - 2.0 * h)
        else:
            # θ(d) = dx * sum_{i=0}^{N-1-d} h[i+d]*(1-h[i])
            # ∂/∂h[k] = dx * [(1-h[k-d]) if d<=k - h[k+d] if k+d<N]
            # First part: for k in [d, N-1]

So = sum_{e=0}^{N-1-k} W[e] * h[k+e] = B[k] ✓

So:
A = np.convolve(W, 1-h, 'full')[:N]
h_padded = np.concatenate([h, np.zeros(N-1)])
B_full = np.convolve(W[::-1], h_padded, 'full')
B = B_full[N-1:N-1+N]
g = dx * (A - B)

This is fully vectorized! Just a few convolution operations. Each convolution is O(N log N) with FFT.
