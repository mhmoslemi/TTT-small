```python
    n = len(h)
    L = 2*n - 1
    center = n - 1
    max_raw = np.max(raw)
    exponents = np.clip((raw - max_raw) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.ones(L) / L
    else:
        weights = weights / sw
    
    g = np.zeros(n)
    one_minus_h = 1.0 - h
    for s in range(-(n-1), n):
        K = center + s
        w = weights[K]
        if w < 1e-14:
            continue
        if s >= 0:
            # i-s valid: i in [s, n-1], partner j = i-s in [0, n-1-s]
            # i+s valid: i in [0, n-1-s], partner j = i+s in [s, n-1]
            g[s:] += w * one_minus_h[:n-s]   # (1-h[j]) part
            g[:n-s] += w * (-h[s:])             # -h[j] part

            g[m:] += w * (-h[:n-m])             # -h[j] part
```

```python
def pair_line_search(raw, lin, quad, d_lo, d_hi, dx):
    """Minimize max_K(raw[K] + d*lin[K] + d²*quad[K]) * dx over d in [d_lo, d_hi]"""
    # Grid search
    n_grid = 31
    d_grid = np.linspace(d_lo, d_hi, n_grid)
    best_val = np.inf
    best_d = 0.0
    for d in d_grid:
        val = np.max(raw + d * lin + d*d * quad) * dx
        if val < best_val:
            best_val = val
            best_d = d
    # Local refinement around best_d
    # Try a few local steps
    step = (d_hi - d_lo) / n_grid
    for _ in range(10):
        candidates = [best_d - step, best_d - step/3, best_d + step/3, best_d + step]
        improved = False
        for cd in candidates:
            if cd < d_lo or cd > d_hi:
                continue
            val = np.max(raw + cd * lin + cd*cd * quad) * dx
            if val < best_val - 1e-15:
                best_val = val
                best_d = cd
                improved = True
        if not improved:
            break
        step *= 0.3
    return best_d, best_val
```

```python
def pair_line_search_vec(raw, lin, quad, d_lo, d_hi, dx):
    n_grid = 41
    d_grid = np.linspace(d_lo, d_hi, n_grid)
    # Evaluate all at once: (n_grid, L) matrix
    vals = raw[None, :] + d_grid[:, None] * lin[None, :] + d_grid[:, None]**2 * quad[None, :]
    scores = np.max(vals, axis=1) * dx
    idx = np.argmin(scores)
    best_d = d_grid[idx]
    best_val = scores[idx]
    # Local refinement
    step = (d_hi - d_lo) / n_grid
    for _ in range(8):
        cands = np.array([best_d - step, best_d + step])
        cands = cands[(cands >= d_lo) & (cands <= d_hi)]
        if len(cands) == 0:
            break
        vals_c = raw[None, :] + cands[:, None] * lin[None, :] + cands[:, None]**2 * quad[None, :]
        scores_c = np.max(vals_c, axis=1) * dx
        idx_c = np.argmin(scores_c)
        if scores_c[idx_c] < best_val - 1e-15:
            best_d = cands[idx_c]
            best_val = scores_c[idx_c]
        else:
            break
        step *= 0.5
    return best_d, best_val
```

```python
def project_feasible(z, n):
    S = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
```

```python
def exact_score(h):

def project_feasible(z, n):

    lo = -2.0
    hi = 2.0
    for _ in range(60):

        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_col(h, a):
    n = len(h)
    L = 2 * n - 1
    col = np.zeros(L)
    ks = a
    ke = min(a + n, L)
    if ke > ks:
        j_start = a + n - ke
        j_end = a + n - 1 - ks
        col[ks:ke] += (1.0 - h[j_start:j_end + 1])[::-1]
    ks2 = max(0, n - 1 - a)
    ke2 = min(2 * n - 1 - a, L)
    if ke2 > ks2:
        i_start = ks2 - (n - 1 - a)
        i_end = ke2 - 1 - (n - 1 - a)
        col[ks2:ke2] += -h[i_start:i_end + 1]
    return col

def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old

    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)
```
