    t0 = n / 2.0
    best_h = h0.copy()
    best_c5 = eval_c5(h0)
    c0 = np.correlate(h0, 1.0 - h0, 'full') * (2.0 / n)
    sprd = max(1e-6, c0.max() - c0.mean())
    tl = [_G['tau']] * _G['nsteps']
    ratio = tl[-1] / tl[0]
    for i in range(len(tl)):
        tl[i] = tl[0] * (ratio ** (i / max(1, len(tl) - 1)))
    h_c = h0.copy()
    bnds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _ceq, 'jac': _ceqj}]
    for tau in tl:
        if time.time() > deadline - 5:
            break
        _G['tau'] = tau
        mi = 30 if tau > 0.008 else 50
        try:
            res = minimize(_sobj, h_c, method='SLSQP', jac=_sgrad,
                           bounds=bnds, constraints=cons,
                           options={'maxiter': mi, 'ftol': 1e-14})
            h_c = proj(res.x, t0)
        except Exception:
            pass
        cv = eval_c5(h_c)
        if cv < best_c5 - 1e-12:
            best_c5 = cv
            best_h = h_c.copy()
        elif cv > best_c5 + 1e-3:
            h_c = best_h.copy()
    return best_h, best_c5

def coord_desc(h0, deadline):
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = eval_c5(h0)
    for _ in range(400):
        if time.time() > deadline - 3:
            break
        h = best_h.copy()
        c = np.correlate(h, 1.0 - h, 'full') * dx
        cm = c.max()
        improved = False
        for _round in range(4):
            c_sorted = np.argsort(c)[::-1]
            for tk in c_sorted[:8]:
                d = tk - (n - 1)
                lo = max(0, -d)
                hi = min(n - 1, n - 1 - d)
                if lo > hi:
                    continue
                g = np.zeros(n)
                il = max(0, lo + d)
                ih = min(n - 1, hi + d)
                if il <= ih:
                    g[il:ih+1] += (1.0 - h[il - d:ih - d + 1])
                g[lo:hi+1] -= h[lo + d:hi + d + 1]
                pos = np.where(g > 1e-6)[0]
                neg = np.where(g < -1e-6)[0]
                if len(pos) == 0 or len(neg) == 0:
                    continue
                ps = pos[np.argsort(-g[pos])][:12]
                ns = neg[np.argsort(g[neg])][:12]
                for pi in ps:
                    for ni in ns:
                        if pi == ni:
                            continue
                        bd = 0.0
                        bc = cm
                        for delta in [0.002, 0.008, 0.02, 0.06, 0.12, 0.25]:
                            if h[pi] - delta < 0 or h[ni] + delta > 1:
                                continue
                            h[pi] -= delta
                            h[ni] += delta
                            cn = np.max(np.correlate(h, 1.0 - h, 'full')) * dx
                            if cn < bc:
                                bc = cn
                                bd = delta
                            h[pi] += delta
                            h[ni] -= delta
                        if bd > 0:
                            h[pi] -= bd
                            h[ni] += bd
                            cm = bc
                            improved = True
            if not improved:
                break
        if not improved:
            break
        cv = eval_c5(h)
        if cv < best_c5 - 1e-13:
            best_c5 = cv
            best_h = h.copy()
    return best_h, best_c5

def sa_opt(h0, deadline, max_iter=80000):
    n = len(h0)
    best_h = h0.copy()
    best_c5 = eval_c5(h0)
    h = h0.copy()
    c5 = best_c5
    rng = np.random.default_rng(777)
    T = 0.0006
    cool = 0.9997
    for _ in range(max_iter):
        if time.time() > deadline - 3:
            break
        T *= cool
        if T < 5e-6:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        delta = rng.standard_normal() * 0.04
        if h[i] + delta < 0 or h[i] + delta > 1:
            continue
        if h[j] - delta < 0 or h[j] - delta > 1:
            continue
        oi, oj = h[i], h[j]
        h[i] += delta
        h[j] -= delta
        c5n = eval_c5(h)
        dc = c5n - c5
        if dc < 0 or (T > 1e-8 and rng.random() < np.exp(-dc / T)):
            c5 = c5n
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            h[i] = oi
            h[j] = oj
    return best_h, best_c5

def interp(hp, nn):
    np_ = len(hp)
    if nn == np_:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    return proj(np.interp(xn, xo, hp), nn / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)

    n_par = len(parent)
    if n_par % 2 == 1:
        parent = np.append(parent, 0.5)
        n_par += 1
    parent = np.clip(parent, 0.0, 1.0)
    parent = proj(parent, n_par / 2.0)

    best_h = parent.copy()
    best_c5 = eval_c5(parent)
    best_n = n_par
    rng = np.random.default_rng(seed)

    _G['n'] = n_par
    _G['target'] = n_par / 2.0
    _G['tau'] = 0.05
    _G['nsteps'] = 18

    h_work = parent.copy()
    h_work, c5w = smooth_opt(h_work, deadline - 500)
    if c5w < best_c5:
        best_c5 = c5w
        best_h = h_work

    h_work, c5w = coord_desc(h_work, deadline - 400)
    if c5w < best_c5:
        best_c5 = c5w
        best_h = h_work
        best_n = n_par

    h_work, c5w = sa_opt(h_work, deadline - 200)
    if c5w < best_c5:
        best_c5 = c5w
        best_h = h_work
        best_n = n_par

    if time.time() < deadline - 120:
        h_work = best_h.copy()
        for amp in [0.01, 0.03, 0.06]:
            for tr in range(2):
                if time.time() > deadline - 100:
                    break
                h_try = proj(h_work + rng.standard_normal(n_par) * amp, n_par / 2.0)
                _G['n'] = n_par
                _G['nsteps'] = 12
                ht, cv = smooth_opt(h_try, deadline - 80)
                if cv < best_c5:
                    best_c5 = cv
                    best_h = ht
                    h_work = ht

    if time.time() < deadline - 60:
        n2 = min(480, 2400)
        if n2 != n_par:
            h2 = interp(best_h, n2)
            _G['n'] = n2
            _G['target'] = n2 / 2.0
            _G['nsteps'] = 14
            h2, c5_2 = smooth_opt(h2, deadline - 20)
            h2, c5_2 = coord_desc(h2, deadline - 10)
            if c5_2 < best_c5:
                best_c5 = c5_2
                best_h = h2
                best_n = n2

    n_final = len(best_h)
    tgt = n_final / 2.0
    diff = tgt - best_h.sum()
    if abs(diff) > 1e-12:
        msk = (best_h > 1e-10) & (best_h < 1.0 - 1e-10)
        ctot = msk.sum()
        if ctot > 0:
            best_h[msk] += diff / ctot
    np.clip(best_h, 0.0, 1.0, out=best_h)
    best_c5 = eval_c5(best_h)

    return (best_h, best_c5, best_n)
