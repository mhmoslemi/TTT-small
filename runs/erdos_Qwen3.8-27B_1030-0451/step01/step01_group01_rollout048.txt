    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return None, None

    gr = (np.sqrt(5) - 1) / 2  # golden ratio conjugate

    b = hi
    c = b - gr * (b - a)

            # ∂/∂h[j] = (1-h[j-L])*I(L<=j<=n-1) - h[j+L]*I(0<=j<=n-1-L)
            for j in range(L, n):

            for j in range(0, n - L):
                g[j] -= wk * h[j + L]
        else:
            m_val = -L
            # out[k] = sum_{i=0}^{n-1-m} h[i] * (1-h[i+m])
            # ∂/∂h[j] = (1-h[j+m])*I(0<=j<=n-1-m) - h[j-m]*I(m<=j<=n-1)
            for j in range(0, n - m_val):
                g[j] += wk * (1.0 - h[j + m_val])
            for j in range(m_val, n):
                g[j] -= wk * h[j - m_val]
    g *= dx
    return g
