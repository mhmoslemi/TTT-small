```python
def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_correlations(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def lse_obj(h, tau):
    c = compute_correlations(h)
    m = c.max()
    return m + tau * np.log(np.sum(np.exp((c - m) / tau)))

def lse_grad(h, tau):
    c = compute_correlations(h)
    m = c.max()
    w = np.exp((c - m) / tau)
    w /= w.sum()
    return weighted_grad(h, w)

def weighted_grad(h, w):
    """Compute ∑_d w[d] * ∂f_d/∂h"""
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    for d in range(-(N-1), N):
        r = d + N - 1  # correlation index
        wd = w[r] * dx
        if abs(wd) < 1e-15:
            continue
        if d >= 0:
            # f_d = dx * sum_{n=0}^{N-1-d} h[n]*(1-h[n+d])
            # ∂f_d/∂h[j] = dx*(1-h[j+d]) for j in [0, N-1-d]
            #              - dx*h[j-d] for j in [d, N-1]
            n_vals = N - d
            # First part: j from 0 to n_vals-1, contribution (1-h[j+d])
            grad[:n_vals] += wd * (1.0 - h[d:])
```

```python
import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_correlations(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx


def weighted_grad(h, w):
    N = len(h)
    dx = 2.0 / N
    grad = np.zeros(N)
    for d in range(-(N - 1), N):
        r = d + N - 1
        wd = w[r]
        if abs(wd) < 1e-18:
            continue
        wd *= dx
        if d >= 0:
            n_vals = N - d
            if n_vals <= 0:
                continue
            grad[:n_vals] += wd * (1.0 - h[d:])
            grad[d:] -= wd * h[:n_vals]
        else:
            dd = -d
            n_vals = N - dd
            if n_vals <= 0:
                continue
            grad[dd:] += wd * (1.0 - h[:n_vals])
            grad[:n_vals] -= wd * h[dd:]
    return grad


def lse_weights(c, tau):
    m = c.max()
    exponents = np.clip((c - m) / tau, -600.0, 600.0)
    w = np.exp(exponents)
    s = w.sum()
    if s > 0:
        w /= s
    else:
        w = np.full_like(c, 1.0 / len(c))
    return w, m


def lse_obj(c, tau):
    m = c.max()
    w, _ = lse_weights(c, tau)
    return m + tau * np.log(w.sum() * np.exp(0))  # m + tau*log(sum(exp((c-m)/tau)))
    # Actually: m + tau * log(sum(exp((c-m)/tau)))
    # = m + tau * (log(sum_w_scaled) + 0) where w_scaled = exp((c-m)/tau)


def lse_value(c, tau):
    m = c.max()
    exponents = np.clip((c - m) / tau, -600.0, 600.0)
    return m + tau * np.log(np.sum(np.exp(exponents)))


def lse_gradient(h, tau):
    c = compute_correlations(h)
    w, _ = lse_weights(c, tau)
    return weighted_grad(h, w)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-15)[0]
            if len(idx) > 0:
                for i in range(min(len(idx), int(residual) + 1)):
                    add = min(residual, 1.0 - h[idx[i]])
                    h[idx[i]] += add
                    residual -= add
                    if abs(residual) < 1e-14:
                        break
        else:
            idx = np.where(h > 1e-15)[0]
            if len(idx) > 0:
                for i in range(min(len(idx), int(-residual) + 1)):
                    sub = min(-residual, h[idx[i]])
                    h[idx[i]] -= sub
                    residual += sub
                    if abs(residual) < 1e-14:
                        break
    return h


def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def pgd_optimize(h, N, tau, n_iterations, deadline, momentum=None):
    """Projected gradient descent with LSE smooth max."""
    target_sum = N / 2.0
    h = h.copy()
    best_h_local = h.copy()
    best_c5_local = compute_c5(h)
    
    if momentum is None:
        v = np.zeros(N)
        rho = 0.9
    else:
        v = momentum
        rho = 0.9
    
    alpha = 0.1
    for it in range(n_iterations):
        if time.time() > deadline:
            break
        c = compute_correlations(h)
        obj = lse_value(c, tau)
        w, _ = lse_weights(c, tau)
        g = weighted_grad(h, w)
        
        # Gradient step with momentum
        v = rho * v - alpha * g
        h_new = project_feasible(h + v, target_sum)
        
        # Simple acceptance: check if c5 improved
        c5_new = compute_c5(h_new)
        if c5_new < best_c5_local:
            best_c5_local = c5_new
            best_h_local = h_new.copy()
        h = h_new
        
        # Adaptive step size
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-10:
            break
        
        # Reduce alpha if we're not making progress
        if not (c5_new < obj):
            alpha *= 0.7
        else:
            alpha = min(alpha * 1.05, 1.0 / max(gnorm, 1e-10))
    
    return best_h_local, best_c5_local


def lbfgs_penalty_obj(h, tau, lam):
    N = len(h)
    target = N / 2.0
    c = compute_correlations(h)
    obj = lse_value(c, tau) + lam * (np.sum(h) - target) ** 2
    return obj


def lbfgs_penalty_grad(h, tau, lam):
    N = len(h)
    target = N / 2.0
    c = compute_correlations(h)
    w, _ = lse_weights(c, tau)
    g = weighted_grad(h, w) + 2 * lam * (np.sum(h) - target) * np.ones(N)
    return g


def slsqp_f(h, tau):
    c = compute_correlations(h)
    return lse_value(c, tau)


def slsqp_g(h, tau):
    c = compute_correlations(h)
    w, _ = lse_weights(c, tau)
    return weighted_grad(h, w)


def sum_eq(h, target):
    return np.sum(h) - target


def sum_eq_jac(h, target):
    return np.ones_like(h)


def optimize_at_resolution(h, N, deadline, rng, min_deadline_margin=20.0):
    """Run multiple optimization phases at given resolution."""
    target_sum = N / 2.0
    bounds = [(0.0, 1.0)] * N
    constraint = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac, 'args': (target_sum,)}]
    
    c5_current = compute_c5(h)
    best_h = h.copy()
    
    # Phase 1: Projected gradient descent with decreasing tau
    tau_schedule_pg = [0.05, 0.03, 0.015, 0.008, 0.004]
    for tau in tau_schedule_pg:
        if time.time() > deadline - min_deadline_margin:
            break
        iters = 200
        h_pgd, c5_pgd = pgd_optimize(h, N, tau, iters, deadline - min_deadline_margin, h)
        if c5_pgd < c5_current:
            h = h_pgd
            c5_current = c5_pgd
    
    # Phase 2: L-BFGS-B with penalty
    tau_schedule_lb = [0.02, 0.01, 0.005, 0.002, 0.001]
    lam_schedule = [10.0, 50.0, 200.0, 1000.0, 5000.0]
    for i, (tau, lam) in enumerate(zip(tau_schedule_lb, lam_schedule)):
        if time.time() > deadline - min_deadline_margin:
            break
        try:
            result = minimize(
                lbfgs_penalty_obj, h.copy(),
                jac=lbfgs_penalty_grad,
                args=(tau, lam),
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-12}
            )
            h_candidate = project_feasible(result.x, target_sum)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < c5_current:
                h = h_candidate
                c5_current = c5_candidate
        except Exception:
            pass
    
    # Phase 3: SLSQP with smooth max
    tau_schedule_slsqp = [0.005, 0.002, 0.001, 0.0005]
    for tau in tau_schedule_slsqp:
        if time.time() > deadline - min_deadline_margin:
            break
        try:
            result = minimize(
                slsqp_f, h.copy(),
                jac=slsqp_g,
                args=(tau,),
                method='SLSQP',
                bounds=bounds,
                constraints=constraint,
                options={'maxiter': 200, 'ftol': 1e-15}
            )
            h_candidate = project_feasible(result.x, target_sum)
            c5_candidate = compute_c5(h_candidate)
            if c5_candidate < c5_current:
                h = h_candidate
                c5_current = c5_candidate
        except Exception:
            pass
    
    if c5_current < compute_c5(best_h):
        best_h = h
    
    return best_h, c5_current


def random_perturb(h, rng, scale):
    N = len(h)
    noise = rng.normal(0, scale, N)
    h_pert = h + noise
    return project_feasible(h_pert, N / 2.0)


def cosine_init(N, a):
    """h(x) = 0.5 + a*cos(pi*x/2) ... normalized to have integral 1."""
    x = (np.arange(N) + 0.5) * (2.0 / N)
    h = 0.5 + a * np.cos(np.pi * x)
    # Ensure 0 <= h <= 1
    h = np.clip(h, 0.0, 1.0)
    # Normalize to sum = N/2
    if h.sum() > 0:
        h = project_feasible(h, N / 2.0)
    return h


def two_bump_init(N, center_width):
    """Two Gaussian bumps."""
    x = (np.arange(N) + 0.5) * (2.0 / N)
    h = np.exp(-0.5 * ((x - 0.5) / center_width) ** 2) + np.exp(-0.5 * ((x - 1.5) / center_width) ** 2)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, N / 2.0)
```
