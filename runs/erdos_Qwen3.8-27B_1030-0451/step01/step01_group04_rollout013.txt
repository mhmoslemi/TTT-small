def compute_c5(h):

def compute_correlation(h):
    """Return the full correlation vector c_r = dx * corr_r(h, 1-h)"""
    N = len(h)

def project_feasible(h, target_sum):
    """Project h to feasible set: 0<=h<=1, sum(h)=target_sum"""
    h = np.array(h, dtype=float, copy=True)
