import time
import math
import numpy as np

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    # Final correction
    d = out.sum() - target
    if abs(d) > 1e-15:
        if d > 0:
            # Need to remove mass
            mask = out > 1e-10
            if mask.sum() > 0:
                removable = out[mask]
                total_removable = removable.sum()
                scale = 1.0 - d / total_removable
                out[mask] = removable * scale
                out[mask] = np.clip(out[mask], 0.0, 1.0)
        else:
            # Need to add mass
            mask = out < 1.0 - 1e-10
            if mask.sum() > 0:
                slack = 1.0 - out[mask]
                total_slack = slack.sum()
                add_frac = min(1.0, -d / total_slack)
                out[mask] = out[mask] + slack * add_frac
                out[mask] = np.clip(out[mask], 0.0, 1.0)
    # Final micro-fix
    d = out.sum() - target
    if abs(d) > 1e-14 and len(out) > 0:
        if d > 0:
            idx = np.argmax(out)
        else:
            idx = np.argmin(out)
        out = out.copy()
        out[idx] = float(np.clip(out[idx] - d, 0.0, 1.0))
    return out

def repair(h, target):
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return project_exact(h, target)

def objective(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def lse_value(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    val = m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
    return dx * val

def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp <= 1e-18 and wm <= 1e-18:
            continue
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * (h[p:])
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * (h[:-p])
    grad *= dx
    return grad

def fw_polish(h0, rng, time_limit, sigma, target, deadline=None):
    """Frank-Wolfe polish with LSE objective."""
    n = len(h0)
    m = n // 2  # target sum = n/2, so m = n/2 ones
    h = repair(h0, target).copy()
    cur_val = lse_value(h, sigma)
    best_h = h.copy()
    best_c5 = objective(h)
    
    t_iter = 0
    t0 = time.monotonic()
    stall = 0
    eta = 1.0  # step size multiplier
    
    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        
        t_iter += 1
        g = lse_gradient(h, sigma)
        
        # Linear oracle: pick m entries with smallest gradient values
        idx = np.argpartition(g, m)[:m]
        v = np.zeros(n)
        v[idx] = 1.0
        
        # Determine step size
        gamma_base = 2.0 / (t_iter + 2.0)
        gamma = eta * gamma_base
        gamma = min(gamma, 1.0)
        
        improved = False
        for g_try in [gamma, gamma * 0.5, gamma * 0.25, gamma * 0.1, 0.01, 0.005]:
            if g_try <= 0:
                continue
            h_new = (1.0 - g_try) * h + g_try * v
            new_val = lse_value(h_new, sigma)
            if new_val < cur_val - 1e-15:
                h = h_new
                cur_val = new_val
                c5 = objective(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break
            # Also try a normalized step (to ensure we move toward v)
        
        if not improved:
            # Try move-to-perturbed-start
            h_pert = h + rng.normal(0, 0.01, n)
            h_pert = project_exact(h_pert, target)
            # Or just try the binary solution directly
            idx_top = np.argsort(-h)[:m]
            v_bin = np.zeros(n)
            v_bin[idx_top] = 1.0
            c5_bin = objective(v_bin)
            if c5_bin < best_c5 - 1e-14:
                best_c5 = c5_bin
                best_h = v_bin.copy()
            
            # Add perturbation to escape
            h = project_exact(h + rng.normal(0, 0.02, n), target)
            cur_val = lse_value(h, sigma)
            stall += 1
            if stall >= 10:
                eta *= 0.5
                if eta < 0.01:
                    break
                stall = 0
    
    # Final binary check
    idx_top = np.argsort(-h)[:m]
    v_bin = np.zeros(n)
    v_bin[idx_top] = 1.0
    c5_bin = objective(v_bin)
    if c5_bin < best_c5:
        best_c5 = c5_bin
        best_h = v_bin.copy()
    
    return best_h, best_c5

def pgd_polish(h0, rng, time_limit, sigma, target, deadline=None, momentum=True):
    """Projected gradient descent with optional Nesterov momentum."""
    n = len(h0)
    h = repair(h0, target).copy()
    cur_val = lse_value(h, sigma)
    best_h = h.copy()
    best_c5 = objective(h)
    
    t0 = time.monotonic()
    mu = 0.9  # momentum
    h_prev = h.copy()
    stall = 0
    lr = 0.05
    
    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        
        g = lse_gradient(h, sigma)
        
        if momentum:
            # Nesterov: evaluate gradient at lookahead point
            y = h + mu * (h - h_prev)
            g = lse_gradient(y, sigma)
        
        # Line search
        improved = False
        for step in [lr, lr * 0.5, lr * 0.25, lr * 0.1]:
            h_new = project_exact(h - step * g, target)
            new_val = lse_value(h_new, sigma)
            if new_val < cur_val - 1e-15:
                h_prev = h
                h = h_new
                cur_val = new_val
                c5 = objective(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                break
        
        if not improved:
            h = project_exact(h + rng.normal(0, 0.01, n), target)
            cur_val = lse_value(h, sigma)
            h_prev = h.copy()
            stall += 1
            if stall >= 8:
                lr *= 0.5
                stall = 0
                if lr < 1e-5:
                    break
    
    return best_h, best_c5

def binary_search(h, target):
    """Find the best binary solution with sum = target by selecting top entries."""
    n = len(h)
    m = int(round(target))
    if m <= 0 or m >= n:
        return h.copy()
    idx = np.argsort(-h)[:m]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

def random_binary(n, rng):
    m = n // 2
    idx = rng.permutation(n)[:m]
    v = np.zeros(n)
    v[idx] = 1.0
    return v

def smooth_binary(n, rng, width=3):
    """Create a smooth binary-like function."""
    m = n // 2
    h = np.zeros(n)
    # Create random blocks
    pos = 0
    while pos < n:
        block_len = int(rng.integers(1, max(2, n // 10)))
        if pos + block_len > n:
            block_len = n - pos
        val = 1.0 if rng.random() < 0.5 else 0.0
        h[pos:pos + block_len] = val
        pos += block_len
    # Smooth
    kernel = np.ones(width) / width
    h = np.convolve(h, kernel, mode='same')
    h = repair(h, n / 2.0)
    return h

    topk = np.sort(c)[-K:]

            grad[p:] -= wm * h[:-p]
    return grad


def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = float(c.max())
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = dense_gradient(h, w)
    grad *= dx
    return grad


def topk_value(h, K):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(-c, K - 1)[:K]
    return float(c[idx].sum() * dx / K)


def topk_gradient(h, K):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    idx = np.argpartition(-c, K - 1)[:K]
    w = np.zeros_like(c)
    w[idx] = 1.0 / K
    grad = dense_gradient(h, w)
    grad *= dx
    return grad


def fw_oracle(g, target, n):
    """Solve min g^T v s.t. 0<=v<=1, sum(v)=target. Returns v."""
    k = int(np.floor(target + 1e-12))
    frac = target - k
    idx = np.argsort(g)
    v = np.zeros(n)
    if k > 0:
        v[idx[:k]] = 1.0
    if frac > 1e-12 and k < n:
        v[idx[k]] = frac
    return v


def fw_polish(h0, rng, time_limit, sigma, target, deadline=None, use_topk=False, K=5):
    n = len(h0)
    h = repair(h0, target).copy()
    if use_topk:
        cur_val = topk_value(h, K)
    else:
        cur_val = lse_value(h, sigma)
    best_h = h.copy()
    best_c5 = objective(h)

    t0 = time.monotonic()
    t_iter = 0
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        t_iter += 1

        if use_topk:
            g = topk_gradient(h, K)
        else:
            g = lse_gradient(h, sigma)

        v = fw_oracle(g, target, n)

        improved = False
        base_gamma = 2.0 / (t_iter + 2.0)
        for gm in [min(1.0, base_gamma * 2), base_gamma, base_gamma * 0.5, 0.1, 0.05, 0.02]:
            if gm <= 0:
                break
            h_new = (1.0 - gm) * h + gm * v
            if use_topk:
                nv = topk_value(h_new, K)
            else:
                nv = lse_value(h_new, sigma)
            if nv < cur_val - 1e-15:
                h = h_new
                cur_val = nv
                c5 = objective(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break

        if not improved:
            # Check binary solution
            k = int(np.floor(target + 1e-12))
            frac = target - k
            idx = np.argsort(-h)
            vb = np.zeros(n)
            if k > 0:
                vb[idx[:k]] = 1.0
            if frac > 1e-12 and k < n:
                vb[idx[k]] = frac
            c5b = objective(vb)
            if c5b < best_c5 - 1e-14:
                best_c5 = c5b
                best_h = vb.copy()

            # Perturb
            h = project_exact(h + rng.normal(0, 0.015, n), target)
            if use_topk:
                cur_val = topk_value(h, K)
            else:
                cur_val = lse_value(h, sigma)
            stall += 1
            if stall >= 12:
                break

    # Final binary check
    k = int(np.floor(target + 1e-12))
    frac = target - k
    idx = np.argsort(-h)
    vb = np.zeros(n)
    if k > 0:
        vb[idx[:k]] = 1.0
    if frac > 1e-12 and k < n:
        vb[idx[k]] = frac
    c5b = objective(vb)
    if c5b < best_c5:
        best_c5 = c5b
        best_h = vb.copy()

    return best_h, best_c5


def pgd_polish(h0, rng, time_limit, sigma, target, deadline=None, momentum=True, use_topk=False, K=5):
    n = len(h0)
    h = repair(h0, target).copy()
    if use_topk:
        cur_val = topk_value(h, K)
    else:
        cur_val = lse_value(h, sigma)
    best_h = h.copy()
    best_c5 = objective(h)

    t0 = time.monotonic()
    h_prev = h.copy()
    mu = 0.85
    lr = 0.08
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break

        if momentum:
            y = h + mu * (h - h_prev)
            y = np.clip(y, 0.0, 1.0)
            if use_topk:
                g = topk_gradient(y, K)
            else:
                g = lse_gradient(y, sigma)
        else:
            if use_topk:
                g = topk_gradient(h, K)
            else:
                g = lse_gradient(h, sigma)

        improved = False
        for step in [lr, lr * 0.5, lr * 0.25, lr * 0.1, lr * 0.05]:
            h_new = project_exact(h - step * g, target)
            if use_topk:
                nv = topk_value(h_new, K)
            else:
                nv = lse_value(h_new, sigma)
            if nv < cur_val - 1e-15:
                h_prev = h
                h = h_new
                cur_val = nv
                c5 = objective(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                break

        if not improved:
            h = project_exact(h + rng.normal(0, 0.01, n), target)
            h_prev = h.copy()
            if use_topk:
                cur_val = topk_value(h, K)
            else:
                cur_val = lse_value(h, sigma)
            stall += 1
            if stall >= 10:
                lr *= 0.5
                stall = 0
                if lr < 1e-6:
                    break

    return best_h, best_c5


def binary_local_search(h0, rng, time_limit, deadline=None):
    n = len(h0)
    target = n / 2.0
    h = repair(h0, target).copy()
    # Snap towards binary
    h = np.where(h > 0.6, 1.0, np.where(h > 0.4, 0.5, 0.0))
    h = repair(h, target)
    cur = objective(h)
    best_h = h.copy()
    best_c = cur

    t0 = time.monotonic()
    stall = 0

    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        n_tries = min(n * 5, 2000)
        for _ in range(n_tries):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            h_new = h.copy()
            h_new[i], h_new[j] = h[j], h[i]
            c = objective(h_new)
            if c < cur - 1e-14:
                h = h_new
                cur = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                improved = True
                break
            # Try half-swap
            if abs(h[i] - h[j]) > 0.01:
                h_new2 = h.copy()
                h_new2[i] = 1.0 - h[i]
                h_new2[j] = 1.0 - h[j]
                if abs(h_new2.sum() - target) < 0.6:
                    h_new2 = repair(h_new2, target)
                    c2 = objective(h_new2)
                    if c2 < cur - 1e-14:
                        h = h_new2
                        cur = c2
                        if c2 < best_c:
                            best_c = c2
                            best_h = h.copy()
                        improved = True
                        break

        if not improved:
            stall += 1
            if stall >= 5:
                # Random restart from perturbed best
                h = repair(best_h + rng.normal(0, 0.1, n), target)
                cur = objective(h)
                stall = 0
                if stall > 20:
                    break

    return best_h, best_c


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def random_binary(n, rng):
    m = n // 2
    idx = rng.permutation(n)[:m]
    v = np.zeros(n)
    v[idx] = 1.0
    return v


def smooth_random(n, rng):
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = int(rng.integers(1, max(2, n // 8)))
        if pos + blen > n:
            blen = n - pos
        h[pos:pos + blen] = 1.0 if rng.random() < 0.5 else 0.0
        pos += blen
    w = 3
    kernel = np.ones(w) / w
    h = np.convolve(h, kernel, mode='same')
    return repair(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 12.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        parent = random_binary(131, rng)
    n0 = len(parent)
    h0 = repair(parent, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = objective(best_h)

    def consider(cand_h, cand_c5):
        nonlocal best_h, best_c5
        try:
            cand_h = np.asarray(cand_h, dtype=float).ravel()
            if cand_h.size == 0 or not np.all(np.isfinite(cand_h)):
                return
            cand_h = repair(cand_h, len(cand_h) / 2.0)
            c = objective(cand_h)
            if c < best_c5 - 1e-13:
                best_c5 = c
                best_h = cand_h.copy()
        except Exception:
            pass

    # Phase 1: FW with increasing sigma
    fw_schedule = [(20.0, 0.14), (40.0, 0.10), (80.0, 0.09), (150.0, 0.08), (300.0, 0.07)]
    for sigma, fr in fw_schedule:
        if time.monotonic() > deadline - 1.0:
            break
        alloc = min(fr * effective, deadline - time.monotonic())
        if alloc < 0.5:
            break
        ch, cc = fw_polish(best_h, rng, alloc, sigma, len(best_h) / 2.0, deadline)
        consider(ch, cc)

    # Phase 2: TopK gradient (K=3, 5)
    if time.monotonic() < deadline - 1.0:
        for K in [3, 5]:
            if time.monotonic() > deadline - 0.5:
                break
            alloc = min(0.06 * effective, deadline - time.monotonic())
            if alloc < 0.5:
                break
            ch, cc = fw_polish(best_h, rng, alloc, 0, len(best_h) / 2.0, deadline, use_topk=True, K=K)
            consider(ch, cc)

    # Phase 3: PGD with momentum at high sigma
    if time.monotonic() < deadline - 1.0:
        for sigma in [100.0, 200.0, 400.0]:
            if time.monotonic() > deadline - 0.5:
                break
            alloc = min(0.05 * effective, deadline - time.monotonic())
            if alloc < 0.3:
                break
            ch, cc = pgd_polish(best_h, rng, alloc, sigma, len(best_h) / 2.0, deadline)
            consider(ch, cc)

    # Phase 4: Binary local search
    if time.monotonic() < deadline - 1.0:
        alloc = min(0.12 * effective, deadline - time.monotonic())
        if alloc > 0.5:
            ch, cc = binary_local_search(best_h, rng, alloc, deadline)
            consider(ch, cc)
            # Polish the binary result
            if time.monotonic() < deadline - 0.5:
                alloc2 = min(0.04 * effective, deadline - time.monotonic())
                if alloc2 > 0.3:
                    ch2, cc2 = pgd_polish(ch, rng, alloc2, 200.0, len(ch) / 2.0, deadline)
                    consider(ch2, cc2)

    # Phase 5: Diversification - multiple starts
    if time.monotonic() < deadline - 2.0:
        starts = []
        starts.append(best_h[::-1].copy())
        starts.append(0.5 * (best_h + best_h[::-1]))
        starts.append(random_binary(n0, rng))
        starts.append(smooth_random(n0, rng))
        starts.append(np.roll(best_h, 3).copy())

        remaining = deadline - time.monotonic() - 1.0
        per = min(0.04 * effective, remaining / max(len(starts), 1))
        for s in starts:
            if time.monotonic() > deadline - 1.0:
                break
            if per < 0.3:
                per = 0.3
            s = repair(resample(s, n0), n0 / 2.0)
            # Quick FW then PGD
            ch, cc = fw_polish(s, rng, per * 0.5, 50.0, n0 / 2.0, deadline)
            consider(ch, cc)
            if time.monotonic() < deadline - 0.5 and per * 0.5 > 0.2:
                ch2, cc2 = pgd_polish(ch, rng, per * 0.5, 150.0, n0 / 2.0, deadline)
                consider(ch2, cc2)

    # Phase 6: Resolution exploration
    if time.monotonic() < deadline - 1.0:
        for nn in [n0 - 2, n0 + 2, n0 - 4, n0 + 4, n0 + 6, n0 - 6]:
            if nn < 50 or nn > 400 or time.monotonic() > deadline - 0.5:
                continue
            alloc = min(0.015 * effective, deadline - time.monotonic())
            if alloc < 0.3:
                break
            rs = repair(resample(best_h, nn), nn / 2.0)
            ch, cc = fw_polish(rs, rng, alloc * 0.6, 80.0, nn / 2.0, deadline)
            consider(ch, cc)
            if time.monotonic() < deadline - 0.3 and alloc * 0.4 > 0.2:
                ch2, cc2 = pgd_polish(ch, rng, alloc * 0.4, 200.0, nn / 2.0, deadline)
                consider(ch2, cc2)

    # Phase 7: Final intensive polish
    if time.monotonic() < deadline - 0.3:
        remaining = deadline - time.monotonic() - 0.3
        # FW at very high sigma
        a1 = remaining * 0.35
        if a1 > 0.3:
            ch, cc = fw_polish(best_h, rng, a1, 500.0, len(best_h) / 2.0, deadline)
            consider(ch, cc)
        # PGD at high sigma
        a2 = min(remaining * 0.3, deadline - time.monotonic() - 0.3)
        if a2 > 0.3:
            ch, cc = pgd_polish(best_h, rng, a2, 300.0, len(best_h) / 2.0, deadline)
            consider(ch, cc)
        # TopK K=1 (pure max subgradient via FW)
        a3 = min(remaining * 0.2, deadline - time.monotonic() - 0.3)
        if a3 > 0.3:
            ch, cc = fw_polish(best_h, rng, a3, 0, len(best_h) / 2.0, deadline, use_topk=True, K=1)
            consider(ch, cc)
        # Final tiny PGD
        a4 = min(remaining * 0.15, deadline - time.monotonic() - 0.2)
        if a4 > 0.2:
            ch, cc = pgd_polish(best_h, rng, a4, 500.0, len(best_h) / 2.0, deadline)
            consider(ch, cc)

    # Final: ensure feasibility and get exact score
    best_h = repair(best_h, len(best_h) / 2.0)
    best_c5 = objective(best_h)

    return best_h, float(best_c5), int(len(best_h))
