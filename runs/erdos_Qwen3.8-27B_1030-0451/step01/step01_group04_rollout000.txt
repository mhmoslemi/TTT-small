import numpy as np
import time


def compute_c5_array(h):
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    return corr * dx


def compute_c5(h):
    return float(np.max(compute_c5_array(h)))


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h


def zero_sum(a):
    return a - np.mean(a)


def line_search_exact(h, a, c_values, n_grid=120):
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h

    A = c_values
    L = dx * (np.correlate(a, b, mode='full') - np.correlate(h, a, mode='full'))
    Q = -dx * np.correlate(a, a, mode='full')

    pos_mask = a > 1e-14
    neg_mask = a < -1e-14

    if np.any(pos_mask):
        lo_pos = np.max(-h[pos_mask] / a[pos_mask])
        hi_pos = np.min((1.0 - h[pos_mask]) / a[pos_mask])
    else:
        lo_pos, hi_pos = -np.inf, np.inf

    if np.any(neg_mask):
        lo_neg = np.max((1.0 - h[neg_mask]) / a[neg_mask])
        hi_neg = np.min(-h[neg_mask] / a[neg_mask])
    else:
        lo_neg, hi_neg = -np.inf, np.inf

    delta_min = max(lo_pos, lo_neg) + 1e-12
    delta_max = min(hi_pos, hi_neg) - 1e-12

    if delta_max - delta_min < 1e-10:
        return 0.0, float(np.max(A))

    deltas = np.linspace(delta_min, delta_max, n_grid)
    F_vals = np.max(A[None, :] + deltas[:, None] * L[None, :] + (deltas[:, None])**2 * Q[None, :], axis=1)

    best_i = int(np.argmin(F_vals))
    best_delta = deltas[best_i]
    best_F = float(F_vals[best_i])
    base_F = float(np.max(A))

    if best_i > 0 and best_i < len(deltas) - 1:
        lo = deltas[best_i - 1]
        hi = deltas[best_i + 1]
        for _ in range(25):
            mid1 = lo + (hi - lo) / 3.0
            mid2 = hi - (hi - lo) / 3.0
            f1 = float(np.max(A + mid1 * L + mid1**2 * Q))
            f2 = float(np.max(A + mid2 * L + mid2**2 * Q))
            if f1 > f2:
                lo = mid1
            else:
                hi = mid2
        refined_delta = (lo + hi) / 2.0
        refined_F = float(np.max(A + refined_delta * L + refined_delta**2 * Q))
        if refined_F < best_F:
            best_delta = refined_delta
            best_F = refined_F

    if best_F >= base_F:
        return 0.0, base_F
    return best_delta, best_F


def compute_active_subgradient(c_values, h, tol=0.01):
    N = len(h)
    dx = 2.0 / N
    max_c = float(np.max(c_values))
    active = c_values >= max_c - tol
    n_active = int(np.sum(active))
    if n_active == 0:
        active = c_values >= max_c - 0.05
        n_active = int(np.sum(active))
    if n_active == 0:
        return np.zeros(N)

    g = np.zeros(N)
    active_idx = np.where(active)[0]
    for r in active_idx:
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            g[d:] += (1.0 - h[:n])
            g[:n] -= h[d:]
        else:
            dd = -d
            n = N - dd
            g[:n] += (1.0 - h[dd:])
            g[dd:] -= h[:n]

    g *= dx / n_active
    return zero_sum(g)


def compute_lse_direction(c_values, h, tau):
    N = len(h)
    dx = 2.0 / N
    m = float(np.max(c_values))
    exponents = np.clip((c_values - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w /= w.sum()

    g = np.zeros(N)
    for r in range(2 * N - 1):
        if w[r] < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            g[d:] += w[r] * dx * (1.0 - h[:n])
            g[:n] -= w[r] * dx * h[d:]
        else:
            dd = -d
            n = N - dd
            g[:n] += w[r] * dx * (1.0 - h[dd:])
            g[dd:] -= w[r] * dx * h[:n]

    return zero_sum(g)


def upscale_func(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)
