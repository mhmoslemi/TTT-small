def de_search(h0, pop_size, time_limit, deadline=None):
    n = len(h0)
    target = n / 2.0
    F = 0.8  # mutation factor
    CR = 0.9  # crossover rate
    
    # Initialize population
    pop = [h0.copy()]
    for _ in range(pop_size - 1):
        x = h0 + np.random.normal(0, 0.1, n)
        x = project_exact(x, target)
        pop.append(x)
    
    scores = [objective(p) for p in pop]
    best_idx = np.argmin(scores)
    best_h = pop[best_idx].copy()
    best_c5 = scores[best_idx]
    
    t0 = time.monotonic()
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        
        # For each member (or a random subset), try mutation
        for i in range(pop_size):
            if deadline is not None and time.monotonic() > deadline:
                break
            if i == best_idx:
                continue
            a, b = np.random.choice(pop_size, 2, replace=False)
            if a == i or b == i:
                continue
            mutant = pop[best_idx] + F * (pop[a] - pop[b])
            # Crossover
            cross = np.random.random(n)
            trial = np.where(cross < CR, mutant, pop[i])
            trial = project_exact(trial, target)
            ts = objective(trial)
            if ts < scores[i]:
                pop[i] = trial
                scores[i] = ts
                if ts < best_c5:
                    best_c5 = ts
                    best_h = trial.copy()
                    best_idx = i
    
    return best_h, best_c5

        offspring = []

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']

def project_exact(h, target):
    n = len(h)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-14:
        if d > 0:
            mask = out > 1e-12
        else:
            mask = out < 1.0 - 1e-12
        cnt = mask.sum()
        if cnt > 0 and abs(d / cnt) < 1.0:
            out = out.copy()
            out[mask] -= d / cnt
        out = np.clip(out, 0.0, 1.0)
    return out


def full_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def objective(h):
    n = len(h)
    dx = 2.0 / n
    return float(full_corr(h).max() * dx)


def active_gradient(h, tau, power=1):
    n = len(h)
    dx = 2.0 / n
    c = full_corr(h)
    center = n - 1
    tau_f = tau / dx
    excess = np.maximum(0.0, c - tau_f)
    if power <= 1:
        w = excess.copy()
    else:
        w = excess ** power
    if w.sum() < 1e-30:
        return np.zeros(n)
    g = np.zeros(n)
    w0 = w[center]
    if w0 > 1e-30:
        g += w0 * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp > 1e-30:
            g[p:] += wp * (1.0 - h[:n - p])
            g[:n - p] -= wp * h[p:]
        if wm > 1e-30:
            g[:n - p] += wm * (1.0 - h[p:])
            g[p:] -= wm * h[:n - p]
    return g

    x0 = (np.arange(n0) + 0.5) / n0
