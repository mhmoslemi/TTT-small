import time
from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_jacobian(h):
    """Compute Jacobian of c = dx * correlate(h, 1-h, 'full') w.r.t. h.
    Returns matrix J of shape (2n-1, n) where J[k, i] = dc_k/dh_i."""
    n = len(h)
    dx = 2.0 / n
    num_corr = 2 * n - 1
    J = np.zeros((num_corr, n))
    for k in range(num_corr):
        lag = k - (n - 1)
        # First term: (1 - h[i-lag]) for i where 0 <= i-lag < n
        i_lo = max(0, lag)
        i_hi = min(n - 1, lag + n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            J[k, idx] += (1.0 - h[idx - lag])
        # Second term: -h[i+lag] for i where 0 <= i+lag < n
        j_lo = max(0, -lag)
        j_hi = min(n - 1, -lag + n - 1)
        if j_lo <= j_hi:
            idx = np.arange(j_lo, j_hi + 1)
            J[k, idx] -= h[idx + lag]
    return J * dx

def project_feasible(p, n):
    """Project p onto {x: 0 <= x_i <= 1, sum(x) = n/2}."""
    target = n / 2.0
    lo = min(float(np.min(p)) - 1.0, -1.0)
    hi = max(float(np.max(p)) + 1.0, 1.0)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    # Fix residual
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-14:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            if adj > 1.0 - proj[mask] - 1e-14:
                adj = (1.0 - proj[mask] - 1e-14).min()
            if adj < -(proj[mask] + 1e-14):
                adj = -(proj[mask] + 1e-14).min()
            proj[mask] += adj
    return proj

def resample_h(h_old, n_new):

def solve_lp_direction(h, c, c_max, eps, J):
    """Solve LP to find direction δ that reduces max correlation.
    minimize t
    s.t. J[k,:] @ δ <= t - c[k] for all k
         sum(δ) = 0
         -h[i] <= δ[i] <= 1-h[i]
         -eps <= δ[i] <= eps
    Returns δ or None if infeasible."""
    n = len(h)
    num_corr = 2 * n - 1
    
    # Variables: [t, δ_0, ..., δ_{n-1}]
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[0] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # J[k,:] @ δ <= t - c[k]  =>  -t + J[k,:] @ δ <= -c[k]
    A_ub = np.zeros((num_corr, n + 1))
    A_ub[:, 0] = -1.0
    A_ub[:, 1:] = J
    b_ub = -c.copy()
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 1:] = 1.0
    b_eq = np.zeros(1)
    
    # Variable bounds
    lower = np.zeros(n + 1)
    upper = np.zeros(n + 1)
    lower[0] = -np.inf  # t is unbounded below
    upper[0] = np.inf   # t is unbounded above
    for i in range(n):
        lower[i + 1] = max(-h[i], -eps)
        upper[i + 1] = min(1.0 - h[i], eps)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lower, upper)), method='highs')
        if res.success:
            return res.x[1:]
        elif res.status == 1:  # iteration limit
            if res.x is not None:
                return res.x[1:]
    except Exception:
        pass
    return None

def softmax_gradient(h, c, tau):
    """Compute smooth max gradient."""
    n = len(h)
    c_max = np.max(c)
    cs = c - c_max
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    # Gradient = sum_k w_k * J[k, :]
    J = compute_jacobian(h)
    grad = w @ J
    return grad

            for s in [0.01, 0.005, 0.002]:
                for sign in [1, -1]:
                    if sign > 0:
                        if h[i] - s < 0 or h[j] + s > 1:
                            continue
                    else:
                        if h[j] - s < 0 or h[i] + s > 1:
                            continue
                    h_new = h.copy()
                    h_new[i] -= sign * s
                    h_new[j] += sign * s
                    c_new = compute_c5(h_new)
                    if c_new < c_max - 1e-13:
                        h = h_new
                        c_max = c_new
                        found = True
                        if c_new < best_c - 1e-14:
                            best_c = c_new
                            best_h = h.copy()
                        break
                if found:
                    break
            if found:
                break
        if not found:
            noise = rng.randn(n) * 0.008
            h = project_feasible(h + noise, n)
            if compute_c5(h) > best_c:
                h = best_h.copy()
    return best_h, best_c

def lp_optimize(h0, time_limit, seed):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    eps = 0.03
    stall = 0
    rng = np.random.RandomState(seed)
    iter_count = 0

    while time.time() - t0 < time_limit - 1:
        c_raw = compute_corr(h)
        c = c_raw * (2.0 / n)
        c_max = float(np.max(c))
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h.copy()

        J = compute_jacobian(h)
        delta = solve_lp_direction(h, c, eps, J)

        if delta is None or np.max(np.abs(delta)) < 1e-14:
            stall += 1
            if stall > 8:
                noise = rng.randn(n) * 0.012
                h = project_feasible(h + noise, n)
                stall = 0
                eps = 0.02
                continue
            eps = max(eps * 0.6, 0.003)
            continue

        accepted = False
        for alpha in [1.0, 0.7, 0.4, 0.2, 0.1]:
            h_new = project_feasible(h + alpha * delta, n)
            c_new = compute_c5(h_new)
            if c_new <= c_max + 1e-12:
                h = h_new
                accepted = True
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                break

        if accepted:
            stall = 0
            eps = min(eps * 1.12, 0.08)
        else:
            stall += 1
            eps *= 0.7
            if stall > 10:
                noise = rng.randn(n) * 0.015
                h = project_feasible(h + noise, n)
                eps = 0.025
                stall = 0

        iter_count += 1
    return best_h, best_c

def smooth_optimize(h0, time_limit, seed):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    step = 0.05 / n
    stall = 0

    phases = [(5, 80, 0.3), (80, 300, 0.4), (300, 800, 0.3)]
    for phi in range(len(phases)):
        tau_lo, tau_hi, frac = phases[phi]
        phase_start = t0 + (sum(p[2] for p in phases[:phi])) * time_limit
        phase_end = t0 + (sum(p[2] for p in phases[:phi+1])) * time_limit

        while time.time() < phase_end and time.time() - t0 < time_limit - 2:
            elapsed_phase = time.time() - phase_start
            total_phase = phase_end - phase_start
            if total_phase < 0.1:
                break
            f = min(max(elapsed_phase / total_phase, 0.0), 1.0)
            tau = tau_lo + (tau_hi - tau_lo) * (f ** 0.5)

            c_raw = compute_corr(h)
            c = c_raw * (2.0 / n)
            c_max = float(np.max(c))
            if c_max < best_c - 1e-14:
                best_c = c_max
                best_h = h.copy()

            cs = c - c_max
            w = np.exp(np.minimum(tau * cs, 50.0))
            ws = w.sum()
            if ws < 1e-300:
                h = project_feasible(h + rng.randn(n) * 0.005, n)
                continue
            w = w / ws

            J = compute_jacobian(h)
            grad = w @ J
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-14:
                stall += 1
                if stall > 5:
                    h = project_feasible(h + rng.randn(n) * 0.01, n)
                    stall = 0
                continue

            accepted = False
            for mul in [1.0, 0.5, 0.25, 0.125]:
                alpha = step * mul
                h_new = project_feasible(h - alpha * grad, n)
                c_new = compute_c5(h_new)
                if c_new <= c_max + 1e-12:
                    h = h_new
                    accepted = True
                    if c_new < best_c - 1e-14:
                        best_c = c_new
                        best_h = h.copy()
                    break

            if accepted:
                stall = 0
                step = min(step * 1.04, 2.0)
            else:
                stall += 1
                step *= 0.6
                if stall > 8:
                    h = project_feasible(h + rng.randn(n) * 0.008, n)
                    step = 0.05 / n
                    stall = 0

    return best_h, best_c

def systematic_pair_search(h0, time_limit, seed):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()

    while time.time() - t0 < time_limit - 1:
        c_raw = compute_corr(h)
        c = c_raw * (2.0 / n)
        c_max = float(np.max(c))
        if c_max < best_c - 1e-14:
            best_c = c_max
            best_h = h.copy()

        J = compute_jacobian(h)
        c_max_idx = np.argmax(c)
        active = np.where(c > c_max - 0.005)[0]

        found = False
        for i in range(n):
            for j in range(i + 1, n):
                if time.time() - t0 > time_limit - 2:
                    return best_h, best_c
                dir_change = J[:, i] - J[:, j]
                new_max_approx = c + (-0.01) * dir_change
                if np.max(new_max_approx[c_max_idx:c_max_idx+1]) < c_max - 1e-6:
                    for s in [0.01, 0.005]:
                        if h[i] - s < 0 or h[j] + s > 1:
                            continue
                        h_new = h.copy()
                        h_new[i] -= s
                        h_new[j] += s
                        c_new = compute_c5(h_new)
                        if c_new < c_max - 1e-13:
                            h = h_new
                            found = True
                            if c_new < best_c - 1e-14:
                                best_c = c_new
                                best_h = h.copy()
                            break
                        s2 = 0.005
                        if h[j] - s2 < 0 or h[i] + s2 > 1:
                            continue
                        h_new2 = h.copy()
                        h_new2[i] += s2
                        h_new2[j] -= s2
                        c_new2 = compute_c5(h_new2)
                        if c_new2 < c_max - 1e-13:
                            h = h_new2
                            found = True
                            if c_new2 < best_c - 1e-14:
                                best_c = c_new2
                                best_h = h.copy()
                            break
                    if found:
                        break
            if found:
                break
        if not found:
            rng = np.random.RandomState(int(time.time() * 1000) % 100000)
            noise = rng.randn(n) * 0.006
            h = project_feasible(h + noise, n)
            if compute_c5(h) > best_c + 0.001:
                h = best_h.copy()
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    best_h = None
    best_c = float('inf')
    best_n = 0

    def tl():
        return budget_s - (time.time() - t_start)

    # Phase 1: Initialize
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_init = len(h0)
        h0 = project_feasible(h0, n_init)
        c5_init = compute_c5(h0)
        best_h = h0.copy()
        best_c = c5_init
        best_n = n_init
    except (NameError, UnboundLocalError):
        n_init = 81
        h0 = make_start(n_init, seed)
        best_h = h0.copy()
        best_c = compute_c5(h0)
        best_n = n_init

    # Phase 2: LP optimization (main driver)
    if tl() > 120:
        alloc = min(tl() - 120, 350)
        h_opt, c_opt = lp_optimize(best_h.copy(), alloc, seed)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Phase 3: Systematic pair search
    if tl() > 80:
        alloc = min(tl() - 80, 150)
        h_opt, c_opt = systematic_pair_search(best_h.copy(), alloc, seed + 3)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Phase 4: Another LP pass
    if tl() > 80:
        alloc = min(tl() - 80, 200)
        h_opt, c_opt = lp_optimize(best_h.copy(), alloc, seed + 11)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Phase 5: Smooth gradient
    if tl() > 60:
        alloc = min(tl() - 60, 150)
        h_opt, c_opt = smooth_optimize(best_h.copy(), alloc, seed + 20)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Phase 6: Pair search again
    if tl() > 40:
        alloc = min(tl() - 40, 100)
        h_opt, c_opt = systematic_pair_search(best_h.copy(), alloc, seed + 30)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Phase 7: Try different resolutions
    for n_try in [97, 121]:
        if tl() < 50:
            break
        h_start = resample_h(best_h, n_try)
        alloc = min(tl() - 50, 60)
        if alloc > 15:
            h_opt, c_opt = lp_optimize(h_start, alloc, seed + n_try)
            if c_opt < best_c:
                best_h = h_opt.copy()
                best_c = c_opt
                best_n = n_try

    # Phase 8: Final LP polish
    if tl() > 20 and best_h is not None:
        alloc = min(tl() - 3, 40)
        h_opt, c_opt = lp_optimize(best_h.copy(), alloc, seed + 100)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
            best_n = len(h_opt)

    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    best_c = compute_c5(best_h)

    return best_h, best_c, len(best_h)
