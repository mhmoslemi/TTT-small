def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def corr_all(h):

def grad_lag(h, lag):
    n = len(h)

    if lag >= 0:

            m1 = np.arange(lag, n)
            g[m1] += 1.0 - h[m1 - lag]
            m2 = np.arange(0, n - lag)
            g[m2] -= h[m2 + lag]
    else:

        if l < n:
            m1 = np.arange(0, n - l)
            g[m1] += 1.0 - h[m1 + l]
            m2 = np.arange(l, n)
            g[m2] -= h[m2 - l]
    return g

**Key insight**: The problem with SLP (Sequential Linear Programming) for non-convex problems is that it can cycle or get stuck. The trust region helps, but the choice of active constraints is crucial.

**Alternative: Direct approach using the structure**

Since f_k(h) = sum_{i} h[i](1-h[i+k]) * dx is bilinear, we can write:
f_k(h) = sum_i h[i] * dx - sum_i h[i] * h[i+k] * dx

The first term depends only on the sum of h over a range, and the second term is a bilinear form.

If we let S = sum(h) * dx = 1 (the constraint), then for lag 0:
f_0 = S - sum h[i]^2 * dx = 1 - sum h[i]^2 * dx

To minimize f_0, maximize sum h[i]^2, which means making h as binary as possible.

For lag k > 0:
f_k = (sum_{i=0}^{n-1-k} h[i]) * dx - sum_{i=0}^{n-1-k} h[i]*h[i+k] * dx

The first term is approximately S * (n-k)/n ≈ 1 (for small k). The second term is the cross-correlation.

So for small k, f_k ≈ 1 - R_k where R_k = sum h[i]*h[i+k] * dx is the autocorrelation.

To minimize max_k f_k, we want to maximize the minimum autocorrelation R_k (for k > 0) AND maximize the binary-ness of h (for k = 0).

This is an interesting tension. The optimal h should be "as binary as possible" while also having high autocorrelation at all lags.

This suggests a "block" structure: h looks like a sequence of blocks of 1's and 0's, where the blocks are arranged to maximize autocorrelation.

2. Use the LP to refine it

import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if 0 <= lag < n:
        g[lag:] += 1.0 - h[:-lag] if lag > 0 else 1.0 - h
        if lag > 0:
            g[:-lag] -= h[lag:]
        else:
            g -= h
    elif -n < lag < 0:
        l = -lag
        g[:-l] += 1.0 - h[l:]
        g[l:] -= h[:-l]
    return g

def smooth_max_grad(h, beta):
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 400)
    u = np.exp(expo)
    Z = u.sum()
    f = m - np.log(Z) / beta
    w = u / Z
    n = len(h)
    g = np.zeros(n)
    active = np.where(w > 1e-12)[0]
    for k in active:
        lag = k - (n - 1)
        g += w[k] * grad_lag(h, lag)
    return f, g

def method_slp(h0, deadline, max_iters=800, radius_init=0.25):
    """Sequential LP with adaptive trust region and restarts."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = radius_init
    no_improve = 0
    min_radius = 0.001
    stall_count = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        gap = max(0.004 * vmax, 0.002)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 50:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-50:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-10:]

        A_list = []
        b_list = []
        for k in active_idx:
            lag = k - (n - 1)
            g = grad_lag(h, lag)
            A_list.append(g)
            b_list.append(g @ h - v[k])
        A = np.array(A_list)
        b = np.array(b_list)

        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(A), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.0]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            break

        if res is not None and res.success and res.x is not None:
            h_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 1e-13:
                best_h = h_new.copy()
                best_c = c_new
                h = h_new.copy()
                no_improve = 0
                stall_count = 0
                radius = min(radius * 1.06, 0.35)
            else:
                no_improve += 1
                stall_count += 1
                if no_improve >= 5:
                    radius *= 0.45
                    no_improve = 0
                    if radius < min_radius:
                        # Try a restart with perturbation
                        pert = base_perturbation(h, 0.01)
                        h = project_sum_box(pert, n / 2.0)
                        radius = 0.05
                        no_improve = 0
                        stall_count = 0
                        if stall_count > 20:
                            break
        else:
            no_improve += 1
            if no_improve >= 15:
                break
    return best_h, best_c

def base_perturbation(h, scale):
    """Generate a perturbation that preserves structure."""
    n = len(h)
    # Smooth perturbation using random Fourier modes
    freqs = np.random.randint(1, max(2, n // 4), size=min(6, n // 2))
    phases = np.random.uniform(0, 2 * np.pi, size=len(freqs))
    x = np.arange(n) / n
    pert = np.zeros(n)
    for f, p in zip(freqs, phases):
        pert += np.cos(2 * np.pi * f * x + p)
    pert *= scale / (np.linalg.norm(pert) + 1e-12)
    return h + pert

def method_smooth_descent(h0, deadline, max_iters=10000):
    """Gradient descent with backtracking and adaptive smoothing."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.015
    beta = 8.0
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        f, g = smooth_max_grad(h, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-15:
            break

        accepted = False
        step_try = step
        for _ls in range(12):
            h_new = h - step_try * g / g_norm
            h_new = project_sum_box(h_new, n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new < f - 0.5 * step_try * g_norm:
                h = h_new
                accepted = True
                break
            step_try *= 0.5

        if not accepted:
            step *= 0.6
            if step < 1e-10:
                break
        else:
            step = min(step * 1.03, 0.08)

        c_cur = evaluate_c5(h)
        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 200 == 0:
                beta = min(beta * 1.4, 300.0)

        if stall > 800:
            break
    return best_h, best_c

def method_nesterov(h0, deadline, max_iters=20000):
    """Nesterov accelerated mirror descent with smoothing."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    x = h0.copy()
    y = h0.copy()
    avg = h0.copy()
    beta = 15.0
    step = 0.008

    for k in range(1, max_iters + 1):
        if time.time() > deadline:
            break
        f_y, g_y = smooth_max_grad(y, beta)
        alpha = 2.0 / (k + 2.0)

        x_new = y - step * g_y
        x_new = project_sum_box(x_new, n / 2.0)
        y_new = (1 - alpha) * x_new + alpha * (2.0 * x_new - x)
        y_new = project_sum_box(y_new, n / 2.0)
        x = x_new
        y = y_new

        # Running average for convergence
        avg = (k / (k + 1)) * avg + (1 / (k + 1)) * x

        if k % 10 == 0:
            c_x = evaluate_c5(x)
            c_avg = evaluate_c5(avg)
            c_cur = min(c_x, c_avg)
            if c_cur < best_c - 1e-13:
                best_c = c_cur
                best_h = x.copy() if c_x < c_avg else avg.copy()

        if k % 100 == 0:
            step *= 0.98
            beta = min(beta * 1.1, 200.0)

        if k > 500 and step < 1e-6:
            break
    return best_h, best_c

def method_slsqp_refine(h0, deadline, max_betas=8):
    """SLSQP with increasing beta for final polish."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [20, 40, 80, 160, 320, 640, 1280, 2560][:max_betas]
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return x.sum() - n / 2.0
    def sum_jac(x):
        return np.ones(n)
    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

    for beta in betas:
        if time.time() > deadline:
            break
        for attempt in range(3):
            if time.time() > deadline:
                break
            if attempt == 0:
                x0 = best_h.copy()
            elif attempt == 1:
                x0 = best_h + np.random.randn(n) * 0.005
                x0 = project_sum_box(x0, n / 2.0)
            else:
                x0 = best_h + np.random.randn(n) * 0.01
                x0 = project_sum_box(x0, n / 2.0)

            def obj_func(x):
                return smooth_max_grad(x, beta)[0]
            def grad_func(x):
                return smooth_max_grad(x, beta)[1]

            try:
                res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-13:
                    best_h = x_new.copy()
                    best_c = c_new
    return best_h, best_c

def transfer_resolution(h0, n_new):
    """Interpolate to new grid size."""
    n_old = len(h0)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, np.concatenate([h0, h0[:1]]), period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    return project_sum_box(h_new, n_new / 2.0)
