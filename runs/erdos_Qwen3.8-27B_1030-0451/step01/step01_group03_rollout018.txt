import numpy as np
import time

def project_feasible(y, target):
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    theta = (lo + hi) * 0.5
    h = np.clip(y - theta, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return c

def c5_value(h):
    return float(np.max(corr_values(h)))

def weighted_gradient(h, alpha, dx):
    n = len(h)
    grad = np.zeros(n)
    corlen = 2 * n - 1
    for k in range(corlen):
        a = alpha[k]
        if a < 1e-14:
            continue
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi + 1] += a * (1.0 - h[lo - d:hi + 1 - d])
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= a * h[lo2 + d:hi2 + 1 + d]
    return grad * dx

def smooth_obj_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    esum = e.sum()
    obj = m + tau * np.log(esum)
    alpha = e / esum
    grad = weighted_gradient(h, alpha, dx)
    return obj, grad

def projected_gradient_optimize(h0, n, target, deadline, max_iter_per_tau=200):
    best_h = h0.copy()
    best_c5 = c5_value(h0)
    h_curr = h0.copy()
    
    c0 = corr_values(h0)
    spread = max(1e-6, c0.max() - c0.mean())
    
    tau_start = max(0.02, 0.4 * spread)
    n_tau = 16
    tau_sched = tau_start * (0.6 ** np.arange(n_tau))
    
    for i, tau in enumerate(tau_sched):
        if time.time() > deadline - 2:
            break
        
        alpha_step = max(0.001, 0.05 * tau / (spread + 1e-8))
        n_iter = max(20, int(max_iter_per_tau * (tau / tau_start + 0.3)))
        if tau < 0.02:
            n_iter = max(50, max_iter_per_tau)
        
        stall = 0
        for it in range(n_iter):
            if time.time() > deadline - 2:
                break
            
            obj, grad = smooth_obj_and_grad(h_curr, tau)
            gnorm = np.linalg.norm(grad)
            if gnorm < 1e-10:
                break
            
            # Backtracking line search
            alpha = alpha_step
            y = h_curr - alpha * grad / gnorm * gnorm  # same as h_curr - alpha * grad
            h_new = project_feasible(h_curr - alpha * grad, target)
            new_obj, _ = smooth_obj_and_grad(h_new, tau)
            
            backtracks = 0
            while new_obj > obj + 1e-12 * alpha * gnorm * gnorm and backtracks < 20:
                alpha *= 0.5
                h_new = project_feasible(h_curr - alpha * grad, target)
                new_obj, _ = smooth_obj_and_grad(h_new, tau)
                backtracks += 1
            
            h_curr = h_new
            
            c5_curr = c5_value(h_curr)
            if c5_curr < best_c5 - 1e-10:
                best_c5 = c5_curr
                best_h = h_curr.copy()
                stall = 0
            else:
                stall += 1
                if stall > n_iter // 2:
                    break
        
        # Reheat if stuck
        c5_curr = c5_value(h_curr)
        if c5_curr > best_c5 + 1e-6:
            h_curr = best_h.copy()
    
    return best_h, best_c5

def interp_to(h_src, n_target):
    n_src = len(h_src)
    if n_target == n_src:
        return h_src.copy()
    xo = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_target, endpoint=False)
    h = np.interp(xn, xo, h_src)
    return project_feasible(h, n_target / 2.0)

def make_sinusoidal_start(n, freq, amp, phase):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h, n / 2.0)

def make_multi_freq_start(n, rng):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    for _ in range(rng.integers(2, 5)):
        f = float(rng.integers(1, 8))
        a = float(rng.uniform(0.05, 0.2))
        p = float(rng.uniform(0, 2 * np.pi))
        h += a * np.sin(2.0 * np.pi * f * x / 2.0 + p)
    h = 0.5 + h
    return project_feasible(h, n / 2.0)

def _chained_refinement(parent, n_list, deadline, rng, seed):
    best_h = None
    best_c5 = 1.0
    best_n = 0
    
    for n in n_list:

            break
        
        # Generate multiple starting points
        starts = []
        
        # 1. Interpolated parent
        starts.append(interp_to(parent, n))
        
        # 2. Best from previous resolution (if exists)

            starts.append(interp_to(best_h, n))
        
        # 3. Sinusoidal starts
        for freq in [1, 2, 3, 5]:
            for amp in [0.15, 0.25, 0.35]:
                starts.append(make_sinusoidal_start(n, freq, amp, 0.0))
        
        # 4. Multi-frequency random starts
        for _ in range(4):
            starts.append(make_multi_freq_start(n, rng))
        
        # 5. Random feasible
        for _ in range(3):
            h_rand = rng.uniform(0, 1, n)

        # Deduplicate (by C5 value approximately)
        unique_starts = []
        for s in starts:
            c5s = c5_value(s)
            if c5s < best_c5 - 1e-9:
                unique_starts.append(s)
        
        # Sort by initial C5
        unique_starts.sort(key=lambda x: c5_value(x))
        
        # Try top starts with time budget

        for s in unique_starts:

            if n_starts_tried >= 12:

            n_starts_tried += 1
            
            if c5_opt < best_c5 - 1e-10:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
    
    return best_h, best_c5, best_n

def _local_polish(h_init, n, deadline):
    """Very fine polishing with aggressive cooling."""
    target = n / 2.0
    best_h = h_init.copy()
    best_c5 = c5_value(h_init)
    h_curr = h_init.copy()
    
    # Multiple rounds of decreasing temperature
    c0 = corr_values(h_curr)
    spread = max(1e-6, c0.max() - c0.mean())
    
    tau_seq = [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.002, 0.001]

        for it in range(300):

            h_new = project_feasible(h_curr - alpha * grad, target)
            new_obj, _ = smooth_obj_and_grad(h_new, tau)
            
            bt = 0

                best_c5 = c5_curr
