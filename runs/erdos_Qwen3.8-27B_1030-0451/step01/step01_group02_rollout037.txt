3. Find the "weakest" active d (smallest e_d if we're going left, largest e_d if going right)
4. Find the first breakpoint where this d drops out of the active set
5. Move to that breakpoint
6. Repeat

def pair_swap_descent(h, target, deadline):
    n = len(h)
    dx = 2.0 / n
    c_vals = correlate(h, 1-h, 'full') * dx  # length 2n-1
    current_max = c_vals.max()
    
    while time.time() < deadline - 5:
        improved = False
        # Try many random pairs
        n_pairs = min(500, 2*n)  # try this many pairs
        for _ in range(n_pairs):
            # Pick random i, j with i != j
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue
            
            # Compute e_d = coeff_d(i) - coeff_d(j) for all d
            e = compute_e(h, i, j, dx, n)
            
            # Solve: min_delta max_d (c_vals[d] + delta * e[d])
            # subject to delta in [delta_min, delta_max]
            delta_min = max(-h[i], h[j] - 1)
            delta_max = min(1 - h[i], h[j])
            
            if delta_max <= delta_min:
                continue
            
            # 1D convex minimization
            best_delta, best_val = solve_1d(c_vals, e, delta_min, delta_max)

                h[j] -= best_delta
                h = make_feasible(h, target)  # ensure feasibility
                c_vals = correlate(h, 1-h, 'full') * dx
                current_max = c_vals.max()
                improved = True
        
        if not improved:
            # Try a different approach or break
            break
    
    return h, current_max

The `compute_e` function: for a given (i,j) pair, compute e[d] for all lags d.

For lag d (where the correlation index m corresponds to d = m - (n-1)):
c_d = dx * sum_{m: 0≤m<n, 0≤m+d<n} h[m] * (1 - h[m+d])

The coefficient of h[i] in c_d:
- If i is in [0,n) and i+d is in [0,n): h[i]*(1-h[i+d]) → coeff = (1-h[i+d])*dx
- If i-d is in [0,n) and i is in [0,n): h[i-d]*(1-h[i]) → coeff = -h[i-d]*dx

So coeff_d(i) = dx * ((1-h[i+d]) if 0≤i+d<n else 0) - dx * (h[i-d] if 0≤i-d<n else 0)

And e[d] = coeff_d(i) - coeff_d(j)

In terms of the correlation array index: the array c_vals has index m from 0 to 2n-2, where m = n-1+d, so d = m-(n-1).

So for correlation index m, d = m - (n-1):
e[m] = dx * ((1-h[i+d]) if 0≤i+d<n else 0) - dx * (h[i-d] if 0≤i-d<n else 0)
     - dx * ((1-h[j+d]) if 0≤j+d<n else 0) + dx * (h[j-d] if 0≤j-d<n else 0)

Let me write this more compactly. For a given position p and lag d:
coeff(p, d) = dx * (1 - h[p+d]) * I(0≤p+d<n) - dx * h[p-d] * I(0≤p-d<n)

So e[m] = coeff(i, d) - coeff(j, d) where d = m - (n-1).

```python
def compute_e(h, i, j, dx, n):
    # e has length 2n-1, indexed by m = 0, ..., 2n-2
    # d = m - (n-1)
    m = np.arange(2*n-1)
    d = m - (n-1)
    
    # coeff(i, d) for each d
    ip = i + d  # i+d
    im = i - d  # i-d
    ci = np.zeros(2*n-1)
    mask_p = (ip >= 0) & (ip < n)
    mask_m = (im >= 0) & (im < n)
    ci[mask_p] += dx * (1 - h[ip[mask_p]])
    ci[mask_m] -= dx * h[im[mask_m]]
    
    # coeff(j, d) for each d
    jp = j + d
    jm = j - d
    cj = np.zeros(2*n-1)
    mask_p = (jp >= 0) & (jp < n)
    mask_m = (jm >= 0) & (jm < n)
    cj[mask_p] += dx * (1 - h[jp[mask_p]])
    cj[mask_m] -= dx * h[jm[mask_m]]
    
    return ci - cj
```

import numpy as np

def make_feasible(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.clip(np.full(n, target / n), 0.0, 1.0)
        return x
    lo = -float(np.max(x))
    hi = 1.0 - float(np.min(x))
    if lo > hi:
        lo, hi = 0.0, 0.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(x))
    if abs(r) > 1e-14:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])]
                rem = r
                for i in order:
                    add = min(1.0 - x[i], rem)
                    x[i] += add
                    rem -= add
                    if rem <= 1e-14:
                        break
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                order = idx[np.argsort(-x[idx])]
                rem = -r
                for i in order:
                    sub = min(x[i], rem)
                    x[i] -= sub
                    rem -= sub
                    if rem <= 1e-14:
                        break
    return x

def exact_score(h):
    n = h.size
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h.astype(float), 1.0 - h.astype(float), 'full') * dx

def get_c_vals(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h.astype(float), 1.0 - h.astype(float), 'full') * dx

def compute_e(h, i, j, dx, n):
    m = np.arange(2 * n - 1, dtype=int)
    d = m - (n - 1)
    ip = i + d
    im = i - d
    ci = np.zeros(2 * n - 1)
    mp = (ip >= 0) & (ip < n)
    mm = (im >= 0) & (im < n)
    if np.any(mp):
        ci[mp] += dx * (1.0 - h[ip[mp]])
    if np.any(mm):
        ci[mm] -= dx * h[im[mm]]
    jp = j + d
    jm = j - d
    cj = np.zeros(2 * n - 1)
    mp = (jp >= 0) & (jp < n)
    mm = (jm >= 0) & (jm < n)
    if np.any(mp):
        cj[mp] += dx * (1.0 - h[jp[mp]])
    if np.any(mm):
        cj[mm] -= dx * h[jm[mm]]
    return ci - cj

def solve_1d(c_vals, e, d_lo, d_hi):
    if d_hi - d_lo < 1e-14:
        delta = 0.5 * (d_lo + d_hi)
        return delta, float(np.max(c_vals + delta * e))
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = d_lo
    b = d_hi
    m1 = a + (1.0 - gr) * (b - a)
    m2 = a + gr * (b - a)
    f1 = float(np.max(c_vals + m1 * e))
    f2 = float(np.max(c_vals + m2 * e))
    for _ in range(35):
        if f1 < f2:
            b = m2
            m2 = m1
            f2 = f1
            m1 = a + (1.0 - gr) * (b - a)
            f1 = float(np.max(c_vals + m1 * e))
        else:
            a = m1
            m1 = m2
            f1 = f2
            m2 = a + gr * (b - a)
            f2 = float(np.max(c_vals + m2 * e))
    delta = 0.5 * (a + b)
    val = float(np.max(c_vals + delta * e))
    v_lo = float(np.max(c_vals + d_lo * e))
    v_hi = float(np.max(c_vals + d_hi * e))
    if v_lo < val:
        val = v_lo
        delta = d_lo
    if v_hi < val:
        val = v_hi
        delta = d_hi
    return delta, val

def greedy_pair(h, lag_idx, n, dx, rng):
    d = int(lag_idx) - (n - 1)
    # coeff(p) for lag d: dx*((1-h[p+d]) if valid) - dx*(h[p-d] if valid)
    p = np.arange(n, dtype=int)
    pp = p + d
    pm = p - d
    coeff = np.zeros(n)
    mp = (pp >= 0) & (pp < n)
    mm = (pm >= 0) & (pm < n)
    if np.any(mp):
        coeff[mp] += dx * (1.0 - h[pp[mp]])
    if np.any(mm):
        coeff[mm] -= dx * h[pm[mm]]
    # We want max coeff[j] - coeff[i], i.e., j = argmax, i = argmin
    j = int(np.argmax(coeff))
    i = int(np.argmin(coeff))
    if i == j:
        # pick next best
        mask = np.ones(n, dtype=bool)
        mask[j] = False
        i = int(np.argmin(np.where(mask, coeff, 1e9)))
    return i, j

def subgradient_descent(h, target, deadline, rng):

    dx = 2.0 / float(n)
    h = make_feasible(h, target).copy()
    cur_max = exact_score(h)

    steps = [0.05, 0.02, 0.01, 0.005, 0.002]
    step_i = 0
    stall = 0

    while time.time() < deadline - 5.0:
        c_vals = get_c_vals(h)
        cur_max = float(np.max(c_vals))
        # Find active lags
        threshold = cur_max - 1e-6
        active = np.where(c_vals >= threshold)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(c_vals))])

        # Compute subgradient: sum of gradients of active lags
        g = np.zeros(n)
        n_ctr = n - 1
        for k in active:
            d = int(k) - n_ctr
            coeff = np.zeros(n)
            p = np.arange(n, dtype=int)
            pp = p + d
            pm = p - d
            mp = (pp >= 0) & (pp < n)
            mm = (pm >= 0) & (pm < n)
            if np.any(mp):
                coeff[mp] += dx * (1.0 - h[pp[mp]])
            if np.any(mm):
                coeff[mm] -= dx * h[pm[mm]]
            g += coeff / float(active.size)

        # Remove mean to respect sum constraint
        g -= np.mean(g)
        g_norm = float(np.linalg.norm(g))
        if g_norm < 1e-14:
            break

        alpha = steps[min(step_i, len(steps) - 1)]
        h_new = make_feasible(h - alpha * g, target)
        c_new = exact_score(h_new)

        if c_new < cur_max - 1e-12:
            h = h_new
            cur_max = c_new
            stall = 0
            if c_new < cur_max - 1e-6:
                step_i = max(0, step_i - 1)
        else:
            step_i = min(len(steps) - 1, step_i + 1)
            stall += 1
            if stall > 20:
                break

    return make_feasible(h, target), exact_score(h)

            best = h
    return best

def get_initial():
    val = globals().get('initial_h_values', None)
    if val is None:
        return None
    arr = np.asarray(val, dtype=float).ravel()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    return make_feasible(arr, 0.5 * float(arr.size))

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except (Exception):
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    hard = t0 + total - 4.0

    rng = np.random.default_rng(int(seed) % (2**32))

    h_init = get_initial()
    if h_init is None:
        h_init = fallback_initial(240)

    n0 = h_init.size
    best_h = h_init.copy()
    best_c = exact_score(best_h)
    best_n = n0

    # Phase 1: Optimize at native resolution
    if time.time() < hard - 30:
        stage_end = min(hard - 20, t0 + 0.45 * total)
        h1, c1 = pair_swap_descent(best_h, 0.5 * float(best_n), stage_end, rng)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, h1.size

    # Phase 2: Subgradient refinement
    if time.time() < hard - 15 and best_c > 0.3807:
        end2 = min(hard - 10, time.time() + 0.2 * total)
        h2, c2 = subgradient_descent(best_h, 0.5 * float(best_n), end2, rng)
        if c2 < best_c:
            best_h, best_c, best_n = h2, c2, h2.size
        # Follow up with pair swaps
        if time.time() < hard - 10:
            h3, c3 = pair_swap_descent(best_h, 0.5 * float(best_n), min(hard - 8, time.time() + 0.15 * total), rng)
            if c3 < best_c:
                best_h, best_c, best_n = h3, c3, h3.size

    # Phase 3: Try other resolutions
    if time.time() < hard - 20:
        resolutions = []
        for f in (0.5, 1.5, 2.0):
            N = int(round(best_n * f))
            if 30 <= N <= 2500 and N != best_n:
                resolutions.append(N)
        resolutions = sorted(set(resolutions))
        
        per_res = max(5.0, 0.5 * (hard - time.time()) / max(1, len(resolutions)))
        for N in resolutions[:3]:
            if time.time() > hard - 15:
                break
            h_r = resample_h(best_h, N)
            target_r = 0.5 * float(N)
            deadline_r = min(hard - 10, time.time() + per_res)
            hr, cr = pair_swap_descent(h_r, target_r, deadline_r, rng)
            if cr < best_c - 1e-12:
                best_h, best_c, best_n = hr, cr, hr.size

    # Phase 4: Final refinement at best resolution
    if time.time() < hard - 5:
        final_deadline = hard - 2.0
        hf, cf = pair_swap_descent(best_h, 0.5 * float(best_n), final_deadline, rng)
        if cf < best_c:
            best_h, best_c, best_n = hf, cf, hf.size

    # Ensure feasibility
    best_h = make_feasible(best_h, 0.5 * float(best_n))
    best_c = exact_score(best_h)
    best_n = int(best_h.size)

    if not np.isfinite(best_c):
        best_h = np.full(best_n, 0.5)
        best_c = exact_score(best_h)

    return best_h, float(best_c), best_n
