        hol, col = smooth_descend(h_sr, p5_alloc, seed_val=seed + n_try, tau_max=500)
        if col < best_c5:
            best_h, best_c5 = hol, col
    
    # Phase 6: Final pair search
    p6_budget = min(time_left() * 0.10, 30)
    if p6_budget > 8:
        hol, col = pair_search(best_h.copy(), p6_budget, seed_val=seed + 800)
        if col < best_c5:
            best_h, best_c5 = hol, col
    
    # Phase 7: Final smooth polish
    p7_budget = max(time_left() - 5, 0)
    if p7_budget > 10:
        hol, col = smooth_descend(best_h.copy(), p7_budget, seed_val=seed + 999, tau_max=1000)
        if col < best_c5:
            best_h, best_c5 = hol, col
    
    # Final validation
    best_h = project_feasible(best_h, len(best_h))
    if not np.all(np.isfinite(best_h)):
        best_h = make_start(81, seed)
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
