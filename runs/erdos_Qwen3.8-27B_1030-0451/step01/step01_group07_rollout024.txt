```python
def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    
    grad = np.zeros(n)
    center = n - 1
    one_minus_h = 1.0 - h
    
    # Center term (shift 0)
    grad += w[center] * (1.0 - 2.0 * h)
    
    # Off-center terms
    for p in range(1, n):
        wp = w[center + p] if center + p < 2*n-1 else 0.0
        wm = w[center - p] if center - p >= 0 else 0.0
        wc = wp + wm
        if wc < 1e-18:
            break
        # (1-h[j-p]) for j in [p, n-1]: one_minus_h[0:n-p]
        # -h[j+p] for j in [0, n-p-1]: -h[p:n]
        grad[p:] += wc * one_minus_h[:n-p]
        grad[:n-p] -= wc * h[p:]
    
    grad *= dx
    return grad
```

This is O(n²) in the loop but with early termination when weights become negligible. For large sigma, only the first few p values have significant weight, so this is fast.

```python
    n = len(h)

        return h
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(60):

        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t

    if abs(d) > 1e-12:

        delta = new_c5 - cur_c5
        if delta < -1e-14:
            h = h_new
            if new_c5 < best_c5 - 1e-13:
                best_c5 = new_c5
                best_h = h.copy()
        elif rng.random() < math.exp(-delta / max(T, 1e-12)):
            h = h_new
        elapsed = time.monotonic() - t0
        frac = min(1.0, elapsed / time_limit)
        T = 1e-8 + (5e-5 - 1e-8) * (1 - frac) ** 2
    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 10.0)
    deadline = t0 + effective

    # Get parent
    parent = get_parent()

    # Generate starting candidates at various n
    n_list = [130, 131, 132, 128, 134]
    starts = []  # list of (h, n, score)

    if parent is not None:
        n0 = len(parent)
        h_parent = repair(parent.copy(), n0 / 2.0)
        starts.append((h_parent, n0, objective(h_parent)))
        # Resample to other sizes
        for nn in n_list:
            if nn == n0:
                continue
            rs = repair(resample_h(h_parent, nn), nn / 2.0)
            starts.append((rs, nn, objective(rs)))

    # Thue-Morse starts
    for nn in [130, 132, 128]:
        tm = make_tm_balanced(nn, rng)
        starts.append((tm, nn, objective(tm)))

    # Periodic starts
    for nn in [130, 132]:
        for period, duty in [(8, 0.5), (10, 0.5), (6, 1/3), (12, 0.5)]:
            pp = make_periodic(nn, period, duty, rng)
            sc = objective(pp)
            starts.append((pp, nn, sc))

    # Random balanced starts
    for nn in [130, 132]:
        for _ in range(3):
            rb = make_balanced_01(nn, rng)
            starts.append((rb, nn, objective(rb)))

    # Sort by initial score
    starts.sort(key=lambda x: x[2])

    best_h = starts[0][0].copy()
    best_c5 = starts[0][2]
    best_n = starts[0][1]

    def consider(cand_h):
        nonlocal best_h, best_c5, best_n
        try:
            nn = len(cand_h)
            cand_h = repair(cand_h, nn / 2.0)
            sc = objective(cand_h)
            if np.all(np.isfinite(cand_h)) and 0.0 <= sc and sc < best_c5 - 1e-12:
                best_c5 = sc
                best_h = cand_h.copy()
                best_n = nn
        except Exception:
            pass

    # Phase 1: LSE polish top starts
    top_k = min(6, len(starts))
    for i in range(top_k):
        if time.monotonic() > deadline:
            break
        si, ni, si_score = starts[i]
        alloc = min(0.10 * effective, deadline - time.monotonic())
        if alloc < 1.0:
            break
        # Multi-sigma LSE
        for sigma in [50.0, 25.0, 10.0]:
            rem = deadline - time.monotonic()
            if rem < 0.5:
                break
            a = min(alloc / 3.0, rem)
            ch, cc = lse_polish(si, rng, a, sigma, ni / 2.0, deadline)
            consider(ch)
            si = ch  # chain

    # Phase 2: Active-set minimax on current best
    if time.monotonic() < deadline - 2.0:
        alloc = min(0.18 * effective, deadline - time.monotonic())
        if alloc > 1.0:
            for K in [10, 20, 40]:
                rem = deadline - time.monotonic()
                if rem < 0.5:
                    break
                a = min(alloc / 3.0, rem)
                ch, cc = active_set_minimax(best_h, rng, a, best_n / 2.0, deadline, K=K)
                consider(ch)

    # Phase 3: 0-1 local search
    if time.monotonic() < deadline - 2.0:
        alloc = min(0.15 * effective, deadline - time.monotonic())
        if alloc > 1.0:
            ch, cc = bit_local_search(best_h, rng, alloc, best_n / 2.0, deadline)
            consider(ch)
            # Re-polish after local search
            rem = deadline - time.monotonic()
            if rem > 0.5:
                a = min(3.0, rem)
                ch, cc = lse_polish(best_h, rng, a, 15.0, best_n / 2.0, deadline)
                consider(ch)

    # Phase 4: Diverse annealing
    if time.monotonic() < deadline - 2.0:
        alloc = min(0.20 * effective, deadline - time.monotonic())
        if alloc > 1.0:
            ch, cc = diverse_anneal(best_h, rng, alloc, best_n / 2.0, deadline)
            consider(ch)
            # Polish after anneal
            rem = deadline - time.monotonic()
            if rem > 0.5:
                a = min(4.0, rem)
                ch, cc = lse_polish(best_h, rng, a, 10.0, best_n / 2.0, deadline)
                consider(ch)
                ch, cc = lse_polish(best_h, rng, min(3.0, deadline - time.monotonic()), 5.0, best_n / 2.0, deadline)
                consider(ch)

    # Phase 5: Try other n values with full pipeline
    if time.monotonic() < deadline - 3.0:
        for nn in [130, 132, 128, 134]:
            if time.monotonic() > deadline - 1.0:
                break
            rem = deadline - time.monotonic()
            a = min(0.04 * effective, rem)
            if a < 1.0:
                break
            rs = repair(resample_h(best_h, nn), nn / 2.0)
            ch, cc = lse_polish(rs, rng, a, 20.0, nn / 2.0, deadline)
            consider(ch)
            rem = deadline - time.monotonic()
            if rem > 0.5:
                a2 = min(0.02 * effective, rem)
                ch, cc = active_set_minimax(best_h, rng, a2, nn / 2.0, deadline, K=15)
                consider(ch)

    # Phase 6: Final polish
    if time.monotonic() < deadline - 0.5:
        a = min(5.0, deadline - time.monotonic())
        if a > 0.3:
            ch, cc = lse_polish(best_h, rng, a, 8.0, best_n / 2.0, deadline)
            consider(ch)
        rem = deadline - time.monotonic()
        if rem > 0.3:
            ch, cc = active_set_minimax(best_h, rng, min(3.0, rem), best_n / 2.0, deadline, K=10)
            consider(ch)

    # Final repair and evaluation
    best_h = repair(best_h, best_n / 2.0)
    best_c5 = objective(best_h)
    return best_h, float(best_c5), int(best_n)
```
