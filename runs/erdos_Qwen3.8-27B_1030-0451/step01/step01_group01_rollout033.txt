def project_sum_box(h, target):

    n = len(h)

    lo, hi = -2.0, 2.0

        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)

            hi = mid
        else:
            lo = mid

def optimal_pair_adjustment(h, i, j):
    """Find optimal t = h[i] minimizing max_k c_k, given h[i]+h[j]=s."""
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-10:
        return h[i]
    
    # Precompute alpha and beta for each lag
    # c_k(t) = alpha_k + beta_k * t
    # where we set h[i]=t, h[j]=s-t, others unchanged
    # 
    # For lag m (0 to n-1-1 = 0 to n-2), the correlation at positive lag m:
    #   c_m = sum_{x=0}^{n-1-m} h[x]*(1-h[x+m])
    # For negative lag -m (correlation index n-1-m):
    #   c_{-m} = sum_{x=0}^{n-1-m} h[x+m]*(1-h[x])
    
    # Let's compute alpha and beta arrays
    # alpha_k = c_k when t=0 (h[i]=0, h[j]=s)
    # beta_k = c_k when t=1 (h[i]=1, h[j]=s-1) minus alpha_k
    
    # Actually, let's just compute c_k(t) for a few values of t and use golden section
    
    h_copy = h.copy()
    
    def f(t):
        h_copy[i] = t
        h_copy[j] = s - t
        # Compute max correlation
        dx = 2.0 / n
        res = np.correlate(h_copy, 1.0 - h_copy, mode="full") * dx
        return max(float(np.max(res)), 0.0)
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    for _ in range(40):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
