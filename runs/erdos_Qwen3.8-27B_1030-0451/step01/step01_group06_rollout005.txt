    bounds = list(zip(lower, upper)) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.all(h + d >= -1e-9) and np.all(h + d <= 1 + 1e-9):
                return d
    except Exception:
        pass
    return None

def optimize_lp(h, best_val, deadline, rng, initial_trust=0.2, k_act=30):
    n = len(h)
    target = n / 2.0
    trust = initial_trust
    stagnant = 0
    best_h = h.copy()
    best_v = best_val
    while time.time() < deadline - 3.0 and stagnant < 40:
        corr = correlation_array(h)
        cur = float(np.max(corr))
        if cur < best_v:
            best_v = cur
            best_h = h.copy()
        d = solve_lp_step(h, corr, trust, k_act)
        if d is not None and np.max(np.abs(d)) > 1e-12:
            accepted = False
            for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                h_new = h + alpha * d
                if np.any(h_new < -1e-9) or np.any(h_new > 1 + 1e-9):
                    continue
                val = evaluate_exact(h_new)
                if val < best_v - 1e-14:
                    best_v = val
                    best_h = h_new.copy()
                    h = h_new.copy()
                    accepted = True
                    stagnant = 0
                    break
            if not accepted:
                trust = max(0.01, trust * 0.65)
                stagnant += 1
        else:
            grad = ls_exp_gradient(h, beta=30.0)
            step = trust * 0.4
            for alpha in [1.0, 0.5, 0.25]:
                h_new = project_feasible(h - alpha * step * grad, target)
                val = evaluate_exact(h_new)
                if val < best_v - 1e-14:
                    best_v = val
                    best_h = h_new.copy()
                    h = h_new.copy()
                    stagnant = 0
                    break
            else:
                trust = max(0.01, trust * 0.7)
                stagnant += 1
        if stagnant >= 20:
            h = project_feasible(best_h + 0.08 * rng.standard_normal(n), target)
            trust = 0.15
            stagnant = 0
        if stagnant == 0:
            trust = min(0.4, trust * 1.12)
    return best_h, best_v

def gradient_descent_ls(h, best_val, deadline, rng, beta_start=20.0, beta_end=80.0):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_v = best_val
    n_iter = 0
    max_iter = 2000
    anneal_freq = 200
    while time.time() < deadline - 3.0 and n_iter < max_iter:
        phase = min(1.0, n_iter / max_iter)
        beta = beta_start + (beta_end - beta_start) * phase
        top_frac = max(0.05, 0.2 - 0.1 * phase)
        grad = ls_exp_gradient(h, beta, top_frac)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        grad = grad / gnorm
        # Backtracking line search
        accepted = False
        step0 = 0.1 / beta
        for i in range(20):
            step = step0 * (0.5 ** i)
            h_new = project_feasible(h - step * grad, target)
            val = evaluate_exact(h_new)
            if val < best_v - 1e-14:
                best_v = val
                best_h = h_new.copy()
                h = h_new.copy()
                accepted = True
                break
        if not accepted:
            # Random restart with small perturbation
            h = project_feasible(best_h + 0.03 * rng.standard_normal(n), target)
        if n_iter % anneal_freq == anneal_freq - 1 and n_iter > 0:
            # Annealing: reset beta lower, add perturbation
            h = project_feasible(best_h + 0.05 * rng.standard_normal(n), target)
        n_iter += 1
    return best_h, best_v

def minmax_descent(h, best_val, deadline, rng):
    """Alternate between minimizing max and using subgradient."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_v = best_val
    step = 0.5
    while time.time() < deadline - 3.0:
        corr = correlation_array(h)
        cur = float(np.max(corr))
        if cur < best_v:
            best_v = cur
            best_h = h.copy()
        # Subgradient: average of gradients of top constraints
        top_k = np.where(corr >= corr.max() - 0.01)[0]
        if len(top_k) == 0:
            top_k = np.array([np.argmax(corr)])
        grad = np.zeros(n)
        for idx in top_k:
            k = idx - (n - 1)
            grad += grad_for_k(h, k)
        grad /= len(top_k)
        gn = np.linalg.norm(grad)
        if gn < 1e-12:
            break
        grad /= gn
        accepted = False
        for alpha in [step, step*0.7, step*0.5, step*0.3, step*0.1]:
            h_new = project_feasible(h - alpha * grad, target)
            val = evaluate_exact(h_new)
            if val < best_v - 1e-14:
                best_v = val
                best_h = h_new.copy()
                h = h_new.copy()
                accepted = True
                break
        if accepted:
            step = min(1.0, step * 1.05)
        else:
            step *= 0.7
            if step < 0.01:
                h = project_feasible(best_h + 0.05 * rng.standard_normal(n), target)
                step = 0.3
    return best_h, best_v

def try_different_n(h_best, val_best, deadline, rng):
    """Try optimizing at different resolutions."""
    best_h = h_best.copy()
    best_v = val_best
    best_n = len(h_best)
    for n_try in [100, 80, 150]:
        if time.time() > deadline - 30:
            break
        if n_try < len(h_best):
            # Downscale
            factor = len(h_best) // n_try if len(h_best) % n_try == 0 else max(1, len(h_best) // n_try)
            nn = (len(h_best) // factor)
            if nn != n_try:
                # Use interpolation
                from numpy import interp
                x_old = np.linspace(0, 2, len(h_best), endpoint=False)
                x_new = np.linspace(0, 2, n_try, endpoint=False)
                h_try = interp(x_new, x_old, h_best)
            else:
                h_try = h_best[:nn*factor].reshape(nn, factor).mean(axis=1)
        else:
            # Upscale by repetition
            factor = n_try // len(h_best) if n_try % len(h_best) == 0 else max(1, n_try // len(h_best))
            h_try = np.repeat(h_best, factor)[:n_try]
        h_try = project_feasible(h_try, n_try / 2.0)
        v_try = evaluate_exact(h_try)
        if v_try > best_v + 0.01:
            continue
        # Optimize at this resolution
        sub_deadline = min(deadline - 10, time.time() + 80)
        h_try, v_try = optimize_lp(h_try, min(best_v, v_try), sub_deadline, rng,
                                   initial_trust=0.15, k_act=25)
        if v_try < best_v:
            best_v = v_try
            best_h = h_try.copy()
            best_n = n_try
    return best_h, best_v

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 3.0
    rng = np.random.default_rng(seed)

    # Get initial values
    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        if h0.size > 0:
            h0 = h0.flatten().copy()
            h0 = project_feasible(h0, len(h0) / 2.0)
    except Exception:
        h0 = None

    if h0 is None:
        h0 = np.full(120, 0.5)

    n0 = len(h0)
    best_h = h0.copy()
    best_val = evaluate_exact(best_h)

    # Phase 1: LP optimization at native resolution
    sub_dead = min(deadline - 60, t_start + 300)
    h_try, v_try = optimize_lp(best_h.copy(), best_val, sub_dead, rng,
                               initial_trust=0.25, k_act=35)
    if v_try < best_val:
        best_val = v_try
        best_h = h_try.copy()

    # Phase 2: Log-sum-exp gradient descent
    sub_dead = min(deadline - 60, time.time() + 200)
    h_try, v_try = gradient_descent_ls(best_h.copy(), best_val, sub_dead, rng)
    if v_try < best_val:
        best_val = v_try
        best_h = h_try.copy()

    # Phase 3: LP refinement on best
    sub_dead = min(deadline - 40, time.time() + 150)
    h_try, v_try = optimize_lp(best_h.copy(), best_val, sub_dead, rng,
                               initial_trust=0.08, k_act=40)
    if v_try < best_val:
        best_val = v_try
        best_h = h_try.copy()

    # Phase 4: Min-max subgradient
    sub_dead = min(deadline - 40, time.time() + 120)
    h_try, v_try = minmax_descent(best_h.copy(), best_val, sub_dead, rng)
    if v_try < best_val:
        best_val = v_try
        best_h = h_try.copy()

    # Phase 5: Try different resolutions
    sub_dead = min(deadline - 20, time.time() + 150)
    h_try, v_try = try_different_n(best_h, best_val, sub_dead, rng)
    if v_try < best_val:
        best_val = v_try
        best_h = h_try.copy()

    # Phase 6: Random restarts at best resolution
    n_best = len(best_h)
    for restart in range(8):
        if time.time() > deadline - 15:
            break
        if restart == 0:
            h_try = project_feasible(best_h + 0.1 * rng.standard_normal(n_best), n_best/2.0)
        elif restart == 1:
            h_try = project_feasible(1.0 - best_h + 0.05 * rng.standard_normal(n_best), n_best/2.0)
        elif restart == 2:
            # Smooth then perturb
            kernel = np.ones(5) / 5.0
            h_smooth = np.convolve(best_h, kernel, mode='same')
            h_try = project_feasible(h_smooth + 0.03 * rng.standard_normal(n_best), n_best/2.0)
        elif restart == 3:
            # Binary projection then smooth
            h_bin = (best_h > 0.5).astype(float)
            h_bin = project_feasible(h_bin, n_best / 2.0)
            h_try = project_feasible(0.6 * best_h + 0.4 * h_bin, n_best/2.0)
        else:
            h_try = project_feasible(best_h + 0.15 * rng.standard_normal(n_best), n_best/2.0)
        
        v_try = evaluate_exact(h_try)
        sub_dead = min(deadline - 10, time.time() + 60)
        h_try, v_try = optimize_lp(h_try, min(best_val, v_try), sub_dead, rng,
                                   initial_trust=0.15, k_act=30)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()

    # Phase 7: Final fine-tuning LP on best solution
    if time.time() < deadline - 5:
        sub_dead = deadline - 2
        h_try, v_try = optimize_lp(best_h.copy(), best_val, sub_dead, rng,
                                   initial_trust=0.05, k_act=50)
        if v_try < best_val:
            best_val = v_try
            best_h = h_try.copy()

    # Ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)

    return best_h, float(best_val), len(best_h)
