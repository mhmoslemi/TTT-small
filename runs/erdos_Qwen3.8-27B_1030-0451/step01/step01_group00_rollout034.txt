    target = n / 2.0

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_to_simplex(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def weighted_gradient(h, corr, weights, threshold=1e-10):
    n = len(h)
    grad = np.zeros(n)
    active = np.where(weights > threshold)[0]
    for idx in active:
        wv = weights[idx]
        k = idx - (n - 1)
        if k == 0:
            grad[:] += wv * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wv * (1.0 - h[:n-k])
            grad[:n-k] -= wv * h[k:]
        else:
            L = -k
            grad[:n-L] += wv * (1.0 - h[L:])
            grad[L:] -= wv * h[:n-L]
    return grad

def make_weights(corr, topk, alpha):
    n_corr = len(corr)
    if topk >= n_corr:
        w = np.ones(n_corr) / n_corr
        return w
    # Get the topk threshold
    # np.partition is O(n)
    thresh = np.partition(corr, n_corr - topk)[n_corr - topk]
    diff = corr - thresh
    w = np.exp(alpha * diff)
    s = w.sum()
    if s < 1e-300:
        w = np.zeros(n_corr)
        idx_sorted = np.argsort(corr)[::-1][:topk]
        w[idx_sorted] = 1.0 / topk
        return w
    w /= s
    return w

def compute_c5(h):

    return np.correlate(h, 1.0 - h, mode="full")

def project_to_simplex(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:

def weighted_gradient(h, corr, weights, threshold=1e-10):
    n = len(h)
    grad = np.zeros(n)
    active = np.where(weights > threshold)[0]
    for idx in active:
        wv = weights[idx]
        k = idx - (n - 1)

            grad[:] += wv * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wv * (1.0 - h[:n - k])
            grad[:n - k] -= wv * h[k:]
        else:
            L = -k
            grad[:n - L] += wv * (1.0 - h[L:])
            grad[L:] -= wv * h[:n - L]
    return grad
