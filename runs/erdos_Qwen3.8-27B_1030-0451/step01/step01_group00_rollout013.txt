def lp_direction(h, corr_scores, radius):
    n = len(h)
    m = 2 * n - 1
    f = float(np.max(corr_scores))
    dx = 2.0 / n
    
    # Compute full Jacobian
    jac = np.zeros((m, n))
    for k in range(m):
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                jac[k, i + lag] += dx * (1.0 - h[i])
                jac[k, i] -= dx * h[i + lag]
        else:
            L = -lag
            for i in range(L, n):
                jac[k, i] += dx * (1.0 - h[i - L])
                jac[k, i - L] -= dx * h[i]
    
    # LP: min t
    # vars: [delta_0, ..., delta_{n-1}, t]
    # obj: [0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: jac[k] @ delta - t <= f - c[k]
    # For each k: [jac[k,0], ..., jac[k,n-1], -1] @ [delta, t] <= f - c[k]
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = jac
    A_ub[:, n] = -1.0
    b_ub = f - corr_scores
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds on delta: |delta_i| <= min(radius, h[i], 1-h[i])
    bounds_delta = []
    for i in range(n):
        r_i = min(radius, h[i], 1.0 - h[i])
        bounds_delta.append((-r_i, r_i))
    bounds_delta.append((0.0, None))  # t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                  bounds=bounds_delta, method='highs')
    
    if res.status == 0 and res.x is not None:
        delta = res.x[:n]
        t_val = res.x[n]
        return delta, t_val
    return None, None

    corr = np.correlate(h, 1.0 - h, mode="full")

def project_feasible(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float).copy()
    p[~np.isfinite(p)] = 0.5
    lo = float(np.min(p)) - 2.0
    hi = float(np.max(p)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:

        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-13:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def build_jacobian(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    jac = np.zeros((m, n))
    for k in range(m):
        lag = k - (n - 1)
        if lag >= 0:
            for i in range(n - lag):
                jac[k, i + lag] += dx * (1.0 - h[i])
                jac[k, i] -= dx * h[i + lag]
        else:
            L = -lag
            for i in range(L, n):
                jac[k, i] += dx * (1.0 - h[i - L])
                jac[k, i - L] -= dx * h[i]
    return jac

def lp_solve(h, corr_scores, radius):
    n = len(h)
    m = 2 * n - 1
    f = float(np.max(corr_scores))
    
    jac = build_jacobian(h)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = jac
    A_ub[:, n] = -1.0
    b_ub = f - corr_scores
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    bounds = []
    for i in range(n):
        r_i = min(radius, h[i], 1.0 - h[i]) + 1e-12
        bounds.append((-r_i, r_i))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                      bounds=bounds, method='highs',
                      options={'presolve': True})
        if res.status == 0 and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None

    n = len(h)
    max_c = np.max(corr_scores)
    cs = corr_scores - max_c
    w = np.exp(tau * cs)
    ws = float(np.sum(w))
    if ws < 1e-300:

    w = w / ws
    
    dx = 2.0 / n

def line_search_lp(h, delta, corr_scores, c5_cur, max_tries=8):
    """Line search: try h + alpha*delta for decreasing alpha."""
    n = len(h)
    best_h = h.copy()
    best_c = c5_cur
    
    for alpha in [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05, 0.02]:

        c_new = compute_c5(h_new)

            best_c = c_new
        elif c_new < c5_cur + 1e-12:
            # Accept if not worse (for progress tracking)
            if c_new < best_c:
                best_h = h_new
                best_c = c_new
            break
    
    return best_h, best_c

def pair_swap_polish(h, n_iter, seed_val=0):
    """Coordinate descent via pair swaps."""
    n = len(h)
    h = h.copy()
    c5_cur = compute_c5(h)
    rng = np.random.RandomState(seed_val)
    
    for it in range(n_iter):
        # Pick a random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Amount to transfer: try small amounts
        best_c = c5_cur
        best_amt = 0.0
        max_amt = min(h[i], 1.0 - h[j])
        if max_amt > 0:
            for amt in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                if amt > max_amt:
                    continue
                h_test = h.copy()
                h_test[i] -= amt
                h_test[j] += amt
                c_test = compute_c5(h_test)
                if c_test < best_c:
                    best_c = c_test
                    best_amt = amt
        
        # Also try reverse direction
        max_amt_r = min(h[j], 1.0 - h[i])
        if max_amt_r > 0:
            for amt in [0.01, 0.005, 0.002, 0.001, 0.0005]:
                if amt > max_amt_r:
                    continue
                h_test = h.copy()
                h_test[j] -= amt
                h_test[i] += amt
                c_test = compute_c5(h_test)
                if c_test < best_c:
                    best_c = c_test
                    best_amt = -amt
        
        if best_amt != 0.0:
            if best_amt > 0:
                h[i] -= best_amt
                h[j] += best_amt
            else:
                h[j] -= (-best_amt)
                h[i] += (-best_amt)
            h = project_feasible(h, n)
            c5_cur = compute_c5(h)
    
    return h, c5_cur

                jac[k, i] -= dx * h[i + lag]

            L = -lag
            for i in range(L, n):

        h_init = make_start(81, seed)

    h_init = project_feasible(h_init, n_init)
    c5_init = compute_c5(h_init)
    best_h = h_init.copy()
    best_c5 = c5_init

    # Phase 2: LP optimization (main engine)
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 5
    if rem > 60:
        alloc = int(min(rem * 0.35, 400))
        h_opt, c5_opt = optimize_lp(best_h.copy(), alloc, seed + 1)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 3: Smooth gradient with different schedule
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 5
    if rem > 40:
        alloc = int(min(rem * 0.2, 180))
        h_opt, c5_opt = optimize_smooth(best_h.copy(), alloc, seed + 50)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 4: Pair swap polish
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 5
    if rem > 15:
        n_trials = min(8000, int(rem * 200))
        h_pol, c5_pol = pair_swap_polish(best_h, n_trials, seed + 100,
                                         time_limit=budget_s - elapsed - 3, t_start=t_start)
        if c5_pol < best_c5:
            best_h = h_pol.copy()
            best_c5 = c5_pol

    # Phase 5: Try nearby resolutions
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 10
    if rem > 40:
        n_cur = len(best_h)
        for n_try in [n_cur + 4, n_cur - 4, n_cur + 8, n_cur + 12]:
            if n_try < 16 or n_try > 300:
                continue
            elapsed = time.time() - t_start
            rem = budget_s - elapsed - 10
            if rem < 25:
                break
            h_res = resample(best_h, n_try)
            alloc = int(min(rem * 0.35, 100))
            if alloc > 12:
                h_opt, c5_opt = optimize_lp(h_res, alloc, seed + n_try * 7)
                # Also try smooth
                alloc2 = int(min(rem * 0.2, 50))
                if alloc2 > 8:
                    h_opt2, c5_opt2 = optimize_smooth(h_opt, alloc2, seed + n_try * 13)
                    if c5_opt2 < c5_opt:
                        h_opt, c5_opt = h_opt2, c5_opt2
                if c5_opt < best_c5 - 1e-12:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt

    # Phase 6: Final LP + pair swap on best
    elapsed = time.time() - t_start
    rem = budget_s - elapsed - 3
    if rem > 20:
        alloc = int(rem * 0.5)
        h_opt, c5_opt = optimize_lp(best_h.copy(), alloc, seed + 500)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
        elapsed = time.time() - t_start
        rem = budget_s - elapsed - 3
        if rem > 8:
            n_trials = min(4000, int(rem * 150))
            h_pol, c5_pol = pair_swap_polish(best_h, n_trials, seed + 600,
                                             time_limit=budget_s - elapsed, t_start=t_start)
            if c5_pol < best_c5:
                best_h = h_pol.copy()
                best_c5 = c5_pol

    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
