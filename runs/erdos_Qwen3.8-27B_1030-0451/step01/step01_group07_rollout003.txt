    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    val = m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
    return dx * val


def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * (h[p:])
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * (h[:-p])
    grad *= dx
    return grad


def lse_polish(h, target, time_limit, sigma, deadline=None, seed=0):
    n = len(h)
    h = np.asarray(h, dtype=float).copy()
    rng = np.random.default_rng(seed)
    cur_lse = lse_value(h, sigma)
    best_h = h.copy()
    best_c5 = objective(h)
    eta = 0.05
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = lse_gradient(h, sigma)
        gn = float(np.linalg.norm(g))
        if gn < 1e-13:
            break
        step = g / gn
        improved = False
        e = eta
        for _ in range(25):
            cand = project_exact(h - e * step, target)
            cv = lse_value(cand, sigma)
            if cv < cur_lse - 1e-13:
                h = cand
                cur_lse = cv
                c5 = objective(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                break
            e *= 0.5
        if not improved:
            h = project_exact(h + rng.normal(0.0, 0.02, n), target)
            cur_lse = lse_value(h, sigma)
            c5 = objective(h)
            if c5 < best_c5 - 1e-13:

                best_h = h.copy()
            stall += 1

                stall = 0
                if eta < 1e-7:
                    break
    c5 = objective(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def gen_sturmian(n, target, rng, phi=None):
    if phi is None:
        phi = rng.random()
    rho = target / n
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (math.floor((i + 1) * rho + phi) - math.floor(i * rho + phi)) == 1 else 0.0
    h = project_exact(h, target)
    return h


def gen_random_binary(n, target, rng):
    h = np.zeros(n)
    # Choose target positions to be 1
    n_ones = int(round(target))
    idx = rng.choice(n, size=n_ones, replace=False)
    h[idx] = 1.0
    # Adjust to hit exact target
    h = project_exact(h, target)
    return h


def gen_block(n, target, rng):
    # Alternating blocks with golden-ratio-like lengths
    phi_g = (1.0 + math.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    pos = 0
    bit = 1
    while pos < n:
        # Block length: 1 or 2 (golden ratio企业经营)
        blen = int(round(1.0 + 0.618 * (pos % 3)))
        blen = max(1, min(blen, n - pos))
        h[pos:pos + blen] = bit
        bit = 1 - bit
        pos += blen
    h = project_exact(h, target)
    return h
