def make_feasible(x, target):
    """Project x onto {0<=h<=1, sum(h)=target} using bisection."""
    n = len(x)
    lo, hi = -max(x.max(), 1.0) - 1, 1.0 - min(x.min(), 0.0) + 1
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = np.clip(x + mid, 0, 1).sum()
        if s > target: hi = mid
        else: lo = mid
    x = np.clip(x + 0.5*(lo+hi), 0, 1)
    # Fix any residual
    r = target - x.sum()
    if abs(r) > 1e-12:
        # distribute residual
        if r > 0:
            room = 1.0 - x
            room[room < 1e-15] = 0
            s = room.sum()
            if s > 0:
                x += r * room / s
        else:
            x += r * x / x.sum() if x.sum() > 0 else r/n
    x = np.clip(x, 0, 1)
    return x

def score(h):
    dx = 2.0 / len(h)
    return float(np.max(np.correlate(h, 1.0-h, 'full') * dx))

    m = C.max()
    if tau < 1e-15: return m, C, m
    a = (C - m) / tau
    log_sum = m + tau * np.log(np.exp(a - a.max()).sum() + 1e-300)
    return log_sum, C, m

        alpha = np.zeros_like(C)
        alpha[idx] = 1.0 / len(idx)
    else:
        a = (C - m) / tau
        a -= a.max()
        e = np.exp(a)
        alpha = e / e.sum()
        if top_k and top_k < len(C):
            # Keep only top-k
            top = np.argsort(alpha)[::-1][:top_k]
            alpha_new = np.zeros_like(alpha)
            alpha_new[top] = alpha[top] / alpha[top].sum()
            alpha = alpha_new
    
    grad = np.zeros(n)

        w = alpha[k] * dx
        if d == 0:

        elif d > 0:

            # Second term: i from dd to n-1: h[i-dd]

import numpy as np
import time


def make_feasible(x, target):
    n = len(x)
    if n == 0:
        return x
    x = np.asarray(x, dtype=float).copy()
    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    target = float(target)
    lo = -1.0 - float(np.max(x))
    hi = 1.0 - float(np.min(x))
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(np.sum(x))
    if abs(r) > 1e-13:
        if r > 0.0:
            room = 1.0 - x
            mask = room > 1e-14
            if mask.sum() > 0:
                total_room = room[mask].sum()
                if total_room > 1e-14:
                    x[mask] += r * room[mask] / total_room
                else:
                    x[mask] = np.minimum(1.0, x[mask] + r / mask.sum())
        else:
            mask = x > 1e-14
            total = x[mask].sum()
            if total > 1e-14:
                x[mask] += r * x[mask] / total
            else:
                x = np.maximum(0.0, x + r / n)
    x = np.clip(x, 0.0, 1.0)
    return x


def score_exact(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    C = np.correlate(h, 1.0 - h, "full") * dx
    return float(np.max(C))


def compute_grad(h, C, tau, top_k=50):
    n = len(h)
    dx = 2.0 / float(n)
    m = float(np.max(C))
    if tau < 1e-14:
        active = C >= m - 1e-12
        idx = np.where(active)[0]
        if idx.size == 0:
            idx = np.array([int(np.argmax(C))])
        alpha = np.zeros_like(C)
        alpha[idx] = 1.0 / idx.size
    else:
        a = (C - m) / tau
        a = a - a.max()
        e = np.exp(a)
        s = float(e.sum())
        if s < 1e-300:
            alpha = np.zeros_like(C)
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha = e / s
        if top_k and top_k < C.size:
            top = np.argsort(alpha)[::-1][:top_k]
            alpha2 = np.zeros_like(alpha)
            alpha2[top] = alpha[top]
            ts = alpha2.sum()
            if ts > 1e-300:
                alpha = alpha2 / ts
    grad = np.zeros(n)
    ctr = n - 1
    for k in np.where(alpha > 1e-12)[0]:
        ak = alpha[k] * dx
        d = int(k) - ctr
        if d == 0:
            grad += ak * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += ak * (1.0 - h[:n - d])
            grad[:n - d] -= ak * h[d:]
        else:
            dd = -d
            grad[:n - dd] += ak * (1.0 - h[dd:])
            grad[dd:] -= ak * h[:n - dd]
    return grad


def project_grad_step(h, grad, step, target):
    x = h - step * grad
    return make_feasible(x, target)


def projected_gradient_optimize(h, deadline, max_time=None):
    n = len(h)
    target = 0.5 * float(n)
    h = make_feasible(h, target).copy()
    c = score_exact(h)
    if max_time is None:
        max_time = deadline - time.time()
    taus = [0.1, 0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    tau_idx = 0
    stall = 0
    while time.time() < deadline:
        rem = deadline - time.time()
        if rem < 1.0:
            break
        if tau_idx >= len(taus):
            break
        tau = taus[tau_idx]
        iters_this_tau = 0
        max_iters = max(30, int(min(2000.0, 3 * max(rem, 0.1))))
        step = 0.1
        while iters_this_tau < max_iters:
            if time.time() > deadline - 0.5:
                break
            C = np.correlate(h, 1.0 - h, "full") * (2.0 / n)
            grad = compute_grad(h, C, tau, top_k=60)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-12:
                break
            improved = False
            for trial in range(12):
                step_new = step
                if trial == 0:
                    step_new = min(0.5, max(0.05, 5.0 / (gnorm + 1e-9)))
                else:
                    step_new = step / 2.0
                if step_new < 1e-10:

                h_new = make_feasible(h - step_new * grad, target)
                c_new = score_exact(h_new)
                if c_new < c - 1e-13:
                    h = h_new
                    c = c_new
                    step = step_new
                    improved = True
                    break

                stall = 0
                iters_this_tau += 1
            else:
                stall += 1

def binary_round(h, target):
    n = len(h)
    cnt = int(round(target))
    if cnt <= 0 or cnt >= n:
        return None
    order = np.argsort(h, kind="stable")[::-1][:cnt]
    b = np.zeros(n, dtype=float)
    b[order] = 1.0
    return b


def binary_local_search(h, deadline, max_swaps=500000):
    n = len(h)
    dx = 2.0 / float(n)
    h = make_feasible(h, 0.5 * float(n)).copy()
    c = score_exact(h)
    ones = np.where(h >= 0.5)[0]
    zeros = np.where(h < 0.5)[0]
    if len(ones) == 0 or len(zeros) == 0:
        return h, c
    best_swap_count = 0
    np.random.seed(12345)
    while time.time() < deadline and best_swap_count < max_swaps:
        if len(ones) == 0 or len(zeros) == 0:
            break
        C = np.correlate(h, 1.0 - h, "full") * dx
        max_val = float(np.max(C))
        max_k = int(np.argmax(C))
        d_star = max_k - (n - 1)
        best_delta = 0.0
        best_i = -1
        best_j = -1
        n_ones = len(ones)
        n_zeros = len=zeros
        n_check_i = min(n_ones, 200)
        n_check_j = min(n_zeros, 200)
        idx_i = np.random.choice(n_ones, size=n_check_i, replace=False) if n_ones > n_check_i else np.arange(n_ones)
        idx_j = np.random.choice(n_zeros, size=n_check_j, replace=False) if n_zeros > n_check_j else np.arange(n_zeros)
        for ii in idx_i:
            i = int(ones[ii])
            for jj in idx_j:
                j = int(zeros[jj])
                delta = 0.0
                if 0 <= i - d_star <= n - 1:
                    delta -= h[i - d_star]
                if 0 <= j - d_star <= n - 1:
                    delta += h[j - d_star]
                if 0 <= i + d_star <= n - 1:
                    delta -= h[i + d_star]
                if 0 <= j + d_star <= n - 1:
                    delta += h[j + d_star]
                delta *= dx
                if delta < best_delta - 1e-14:
                    best_delta = delta
                    best_i = i
                    best_j = j
        if best_i >= 0 and best_j >= 0:
            h[best_i] = 0.0
            h[best_j] = 1.0
            ones = np.where(h >= 0.5)[0]
            zeros = np.where(h < 0.5)[0]
            c_new = score_exact(h)
            if c_new < c - 1e-14:
                c = c_new
                best_swap_count += 1
            else:
                h[best_i] = 1.0
                h[best_j] = 0.0
                ones = np.where(h >= 0.5)[0]
                zeros = np.where(h < 0.5)[0]
                break
    return h, c


def resample_to(h, N):
    n0 = len(h)
    target = 0.5 * float(N)
    if n0 == N:
        return make_feasible(h, target)
    xold = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))
    y = np.interp(xnew, xold, h)
    return make_feasible(y, target)

    # Phase 5: Final polish
    if time.time() < hard_deadline - 2.0:
        h5, c5 = active_set_refine(best_h, hard_deadline - 1.0)
        if np.isfinite(c5) and c5 < best_c:
            best_h = h5
            best_c = c5

    # Final feasibility check
    best_h = make_feasible(best_h, 0.5 * float(len(best_h)))
    best_c = score_exact(best_h)
    if not np.isfinite(best_c):
        best_h = make_feasible(np.full(len(best_h), 0.5), 0.5 * float(len(best_h)))
        best_c = score_exact(best_h)

    return best_h, float(best_c), int(len(best_h))
