```python
    if cmax > 0:
```

```python
def line_search_pair(h, i, j, n, max_evals=30):
    """Find the best delta for pair (i,j) using golden section search on exact objective."""
    # Feasible range: h[i]-delta >= 0, h[j]+delta >= 0, h[i]-delta <= 1, h[j]+delta <= 1
    delta_lo = max(-h[j], h[i] - 1.0)
    delta_hi = min(h[i], 1.0 - h[j])
    
    if delta_hi - delta_lo < 1e-12:
        return 0.0, h
    
    # Golden section search
    gr = (np.sqrt(5.0) - 1.0) / 2.0  # golden ratio ~0.618
    
    a = delta_lo
    b = delta_hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    def eval_obj(delta):
        h_new = h.copy()
        h_new[i] -= delta
        h_new[j] += delta
        # Quick bounds check
        return compute_c5(h_new)
    
    fc = eval_obj(c)
    fd = eval_obj(d)
    
    for _ in range(max_evals - 2):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_obj(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_obj(d)
    
    best_delta = 0.5 * (a + b)
    return best_delta, h
```

```python
    """Optimize pair (i,j) using linear approx + exact verification."""
    delta_lo = max(-h[j], h[i] - 1.0)
    delta_hi = min(h[i], 1.0 - h[j])
    
    if delta_hi - delta_lo < 1e-12:
        return h, float(np.max(corr_current) * 2.0/n)
    
    # Compute direction vector a (linear approximation)
    a = np.zeros(2*n-1)
    for corr_idx in range(2*n-1):

        if 0 <= n1 < n:
            val += -(1.0 - h[n1])
        n2 = j - k
        if 0 <= n2 < n:
            val += (1.0 - h[n2])

        if 0 <= n3 < n:
            val += h[n3]
        n4 = j + k
        if 0 <= n4 < n:
            val += -h[n4]
```

```python
def line_search_global(h, direction, n, max_evals=40):
    """Line search along a direction with exact objective."""
    # Find feasible range
    # h + t * direction must stay in [0,1]
    # For direction[i] > 0: t <= (1-h[i])/direction[i]
    # For direction[i] < 0: t <= h[i]/(-direction[i])
    
    pos_mask = direction > 1e-12
    neg_mask = direction < -1e-12
    
    t_max = 10.0  # default large
    if np.any(pos_mask):
        t_max = min(t_max, float(np.min((1.0 - h[pos_mask]) / direction[pos_mask])))
    if np.any(neg_mask):
        t_max = min(t_max, float(np.min(h[neg_mask] / (-direction[neg_mask]))))
    
    t_max = max(t_max, 0.0)
    if t_max < 1e-12:
        return h, compute_c5(h)
    
    # Golden section search on [0, t_max]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a = 0.0
    b = t_max
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    
    def eval_at(t):
        h_trial = h + t * direction
        return compute_c5(h_trial)
    
    fc = eval_at(c)
    fd = eval_at(d)
    
    for _ in range(max_evals - 2):
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = eval_at(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = eval_at(d)
    
    best_t = 0.5 * (a + b)
    best_c5 = min(fc, fd, eval_at(best_t), eval_at(0.0))
    
    # Find the t that gives best_c5
    for t in [0.0, best_t, 0.5*best_t, 0.8*best_t, c, d]:
        t = max(0.0, min(t, t_max))
        c5_t = eval_at(t)
        if c5_t < best_c5:
            best_c5 = c5_t
            best_t = t
    
    h_new = h + best_t * direction
    h_new = np.clip(h_new, 0.0, 1.0)
    # Re-project to maintain sum constraint
    # Actually, if direction is a valid direction (sum = 0), the sum is preserved
    # But projection might be needed for numerical precision
    h_new = project_constraint(h_new, n)
    
    return h_new, best_c5
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    p = np.clip(p, -5.0, 5.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
            diff2 = target - float(np.sum(proj))
            if abs(diff2) > 1e-14:
                mask2 = (proj > 1e-10) & (proj < 1.0 - 1e-10)
                nm2 = int(np.sum(mask2))
                if nm2 > 0:
                    proj[mask2] += diff2 / nm2
    return proj

def compute_grad_proxy(h, corr, n, lam=5.0):
    cmax = float(np.max(corr))
    if cmax < 1e-30:
        return np.zeros(n)
    w = (corr / cmax) ** lam
    wsum = float(w.sum())
    if wsum < 1e-30:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    for ci in range(2 * n - 1):
        wk = w[ci]
        if wk < 1e-15:
            continue
        k = ci - (n - 1)
        if k >= 0:
            grad[k:] += wk * (1.0 - h[:n - k])
            grad[:n - k] -= wk * h[k:]
        else:
            L = -k
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

def pair_direction(h, i, j, n):
    """Compute the linear direction vector a for pair swap h[i]-=d, h[j]+=d."""
    a = np.zeros(2 * n - 1)
    for ci in range(2 * n - 1):
        k = ci - (n - 1)
        val = 0.0
        n1 = i - k
        if 0 <= n1 < n:
            val += -(1.0 - h[n1])
        n2 = j - k
        if 0 <= n2 < n:
            val += (1.0 - h[n2])
        n3 = i + k
        if 0 <= n3 < n:
            val += h[n3]
        n4 = j + k
        if 0 <= n4 < n:
            val += -h[n4]
        a[ci] = val
    return a

def lin_min_delta(corr, a, d_lo, d_hi):
    """Find delta minimizing max(corr + delta*a) in [d_lo, d_hi] using subgradient bisection."""
    lo = d_lo
    hi = d_hi
    best_d = 0.0
    best_f = float(np.max(corr))
    for _ in range(60):
        if hi - lo < 1e-14:
            break
        mid = 0.5 * (lo + hi)
        f_mid = float(np.max(corr + mid * a))
        if f_mid < best_f - 1e-15:
            best_f = f_mid
            best_d = mid
        active = (corr + mid * a) >= f_mid - 1e-10
        if not np.any(active):
            active = np.arange(len(corr)) == int(np.argmax(corr + mid * a))
        min_a = float(np.min(a[active]))
        max_a = float(np.max(a[active]))
        if min_a > 1e-10:
            hi = mid
        elif max_a < -1e-10:
            lo = mid
        else:
            best_d = mid
            break
    return best_d

def eval_pair(h, i, j, delta):
    """Evaluate C5 after pair swap, modifying h in place and restoring."""
    h[i] -= delta
    h[j] += delta
    c5 = compute_c5(h)
    h[i] += delta
    h[j] -= delta
    return c5

def optimize_pair_exact(h, i, j, n, corr, current_c5):
    """Optimize pair (i,j): find best delta using linear approx + exact verification."""
    d_lo = max(-h[j], h[i] - 1.0)
    d_hi = min(h[i], 1.0 - h[j])
    if d_hi - d_lo < 1e-12:
        return h, current_c5, False

    a = pair_direction(h, i, j, n)
    best_d_lin = lin_min_delta(corr, a, d_lo, d_hi)

    # Generate candidate deltas
    candidates = set()
    candidates.add(0.0)
    candidates.add(best_d_lin)
    candidates.add(d_lo)
    candidates.add(d_hi)
    candidates.add(0.5 * best_d_lin)
    candidates.add(0.8 * best_d_lin)
    candidates.add(1.2 * best_d_lin)
    if best_d_lin != 0:
        candidates.add(0.3 * best_d_lin)
        candidates.add(0.7 * best_d_lin)

    dx = 2.0 / n
    best_c5 = current_c5
    best_d = 0.0

    for d in candidates:
        d = float(np.clip(d, d_lo, d_hi))
        c5_val = eval_pair(h, i, j, d)
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_d = d

    if best_d != 0.0:
        h_new = h.copy()
        h_new[i] -= best_d
        h_new[j] += best_d
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new, best_c5, True
    return h, current_c5, False

    h_new = np.interp(x_new, x_old, h_old)

                nh, nc5, imp = optimize_pair_exact(h, i, j, n, corr, current_c5)
                if imp:
                    h = nh
                    current_c5 = nc5
                    corr = compute_corr(h)
                    improved = True
                    if current_c5 < best_c5 - 1e-14:
                        best_c5 = current_c5
                        best_h = h.copy()
                        best_n = n
            # Also try reverse direction (source->sink flipped)
            for score, i, j in pairs[:30]:
                if time.time() - t_final > final_time or remaining() < 10:
                    break
                nh, nc5, imp = optimize_pair_exact(h, j, i, n, corr, current_c5)
                if imp:
                    h = nh
                    current_c5 = nc5
                    corr = compute_corr(h)
                    improved = True
                    if current_c5 < best_c5 - 1e-14:
                        best_c5 = current_c5
                        best_h = h.copy()
                        best_n = n

            if not improved:
                stall += 1
                if stall > 15:
                    break
                rng_f = np.random.RandomState(seed + int(time.time() * 3) % 99999)
                h_p = project_constraint(h + rng_f.randn(n) * 0.008, n)
                c5_p = compute_c5(h_p)
                if c5_p <= current_c5 + 0.003:
                    h = h_p
                    current_c5 = c5_p
                    corr = compute_corr(h)
                stall = 0 if c5_p < current_c5 else stall

    # Final verification
    best_h = project_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    best_n = len(best_h)

    return best_h, best_c5, best_n
```
