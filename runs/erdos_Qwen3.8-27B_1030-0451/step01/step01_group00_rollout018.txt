def compute_corr(h):
    return np.correlate(h.astype(float), 1.0 - h.astype(float), mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_constraint(p, n):
    """Project p onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    p = np.asarray(p, dtype=float)
    p = np.clip(p, -10.0, 10.0)  # handle extreme values
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))

def corr_gradient(h, p):
    """Compute gradient of correlation at position p with respect to h."""
    n = len(h)
    dx = 2.0 / n
    lag = p - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        # c_p = dx * sum_{n=0}^{n-1-lag} h[n+lag] * (1-h[n])
        # d/dh[i] where i = n+lag: += (1-h[n]) = (1-h[i-lag])
        # d/dh[i] where i = n: -= h[n+lag] = h[i+lag]
        for n_idx in range(n - lag):
            grad[n_idx + lag] += dx * (1.0 - h[n_idx])
            grad[n_idx] -= dx * h[n_idx + lag]
    else:
        L = -lag
        # c_p = dx * sum_{n=L}^{n-1} h[n-L] * (1-h[n])
        # Let m = n-L: sum_{m=0}^{n-1-L} h[m] * (1-h[m+L])
        # d/dh[m]: += (1-h[m+L])
        # d/dh[n] where n=m+L: -= h[m]
        for m in range(n - L):
            grad[m] += dx * (1.0 - h[m + L])
            grad[m + L] -= dx * h[m]
    return grad

def all_corr_gradients(h):
    """Compute all 2n-1 correlation gradients."""
    n = len(h)
    grads = np.zeros((2*n-1, n))
    for p in range(2*n-1):
        grads[p] = corr_gradient(h, p)
    return grads

def lp_direction(h, corr_vals, f_val):
    """Solve LP to find direction to reduce max correlation."""
    n = len(h)
    dx = 2.0 / n
    n_corr = 2 * n - 1
    
    # Compute gradients
    grads = all_corr_gradients(h)  # shape (2n-1, n)
    
    # LP: minimize t
    # Variables: [delta_0, ..., delta_{n-1}, t]
    # Constraints:
    #   grads[p] @ delta - t <= f - c[p]  for all p
    #   sum(delta) = 0
    #   -h_i <= delta_i <= 1 - h_i
    #   t >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: grads[p] @ delta - t <= f - c[p]
    A_ub = np.zeros((n_corr, n + 1))
    A_ub[:, :n] = grads  # (2n-1, n)
    A_ub[:, n] = -1.0  # -1 * t
    
    b_ub = f_val - corr_vals  # f - c[p] for each p
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    b_eq = np.zeros(1)
    
    # Bounds
    bounds = [(-h[i], 1.0 - h[i]) for i in range(n)]
    bounds.append((0.0, None))  # t >= 0
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
        if result.success:
            delta = result.x[:n]
            # Clean up tiny values
            delta[np.abs(delta) < 1e-12] = 0.0
            return delta
    except Exception:
        pass
    return None

    n = len(h)
    f_val = float(np.max(corr_vals))
    # Softmax weights
    shifted = corr_vals - f_val
    weights = np.exp(tau * shifted)
    wsum = weights.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    weights = weights / wsum

    grad = np.zeros(n)
    for p in range(len(corr_vals)):
        if weights[p] > 1e-14:
            grad += weights[p] * grads[p]
    return grad

def project_constraint(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float)
    p = np.nan_to_num(p, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)

            lo = mid
        else:
            hi = mid

    diff = target - float(np.sum(proj))

            idx_lo = np.arange(0, cnt)

            J[p, idx_lo] -= dx * h[idx_hi]

            cnt = n - L
            idx_lo = np.arange(0, cnt)

        grad = smooth_gradient(h, corr, 150.0)
        neg_idx = np.where(grad < -1e-6)[0]
        pos_idx = np.where(grad > 1e-6)[0]
        if len(neg_idx) == 0 or len(pos_idx) == 0:
            break
        improved = False
        # Sort by gradient magnitude for efficiency
        neg_idx = neg_idx[np.argsort(grad[neg_idx])[:15]]
        pos_idx = pos_idx[np.argsort(-grad[pos_idx])[:15]]
        for i in neg_idx:
            for j in pos_idx:
                if time.time() - t0 >= time_limit:
                    break
                i, j = int(i), int(j)
                s_max = min(h[i], 1.0 - h[j])
                if s_max < 1e-10:
                    continue
                s_vals = np.linspace(0.05 * s_max, 0.95 * s_max, 8)
                for s in s_vals:
                    h_try = h.copy()
                    h_try[i] = h[i] - s
                    h_try[j] = h[j] + s
                    h_try = project_constraint(h_try, n)
                    c_try = compute_c5(h_try)
                    if c_try < best_c - 1e-13:
                        best_c = c_try
                        best_h = h_try.copy()
                        h = h_try
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if not improved:
            no_improve += 1
            if no_improve > 3:
                break
    return best_h, best_c

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.4)
    h += 0.05 * np.sin(6.0 * np.pi * x + 1.1)
    h += 0.03 * np.sin(9.0 * np.pi * x + 2.0)
    h += rng.randn(n) * 0.015
    return project_constraint(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    def elapsed():
        return time.time() - t_start

    def time_left():
        return budget_s - elapsed()

    best_h = None
    best_c = float('inf')

    # Initialize from parent
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
        h0 = project_constraint(h0, n_init)
        best_h = h0.copy()
        best_c = compute_c5(h0)
    except (NameError, UnboundLocalError):
        n_init = 81
        best_h = make_start(n_init, seed)
        best_c = compute_c5(best_h)

    n = len(best_h)

    # Phase 1: Primary LP optimization
    if time_left() > 80:
        alloc = min(time_left() * 0.38, 400)
        h_opt, c_opt = local_optimize(best_h.copy(), alloc)
        if c_opt < best_c - 1e-12:
            best_h = h_opt
            best_c = c_opt

    # Phase 2: Second LP pass (different perturbation if stuck)
    if time_left() > 60:
        c_before = best_c
        alloc = min(time_left() * 0.25, 250)
        h_opt, c_opt = local_optimize(best_h.copy(), alloc)
        if c_opt < best_c - 1e-12:
            best_h = h_opt
            best_c = c_opt
        # If no improvement, try perturbed start
        elif time_left() > 50:
            rng = np.random.RandomState(seed + 7777)
            h_pert = project_constraint(best_h + rng.randn(n) * 0.012, n)
            h_pert = project_constraint(best_h + rng.randn(n) * 0.008, n)
            alloc2 = min(time_left() * 0.15, 120)
            if alloc2 > 20:
                h_opt, c_opt = local_optimize(h_pert, alloc2)
                if c_opt < best_c - 1e-12:
                    best_h = h_opt
                    best_c = c_opt

    # Phase 3: Coordinate polish
    if time_left() > 30:
        alloc = min(time_left() * 0.15, 100)
        h_opt, c_opt = coordinate_polish(best_h.copy(), alloc)
        if c_opt < best_c - 1e-12:
            best_h = h_opt
            best_c = c_opt

    # Phase 4: Try slightly different resolutions
    for n_try in [101, 141]:
        if time_left() < 40:
            break
        x_old = np.linspace(0, 2, n, endpoint=False)
        x_new = np.linspace(0, 2, n_try, endpoint=False)
        h_interp = np.interp(x_new, x_old, best_h)
        h_interp = project_constraint(h_interp, n_try)
        c_interp = compute_c5(h_interp)
        alloc = min(time_left() * 0.18, 130)
        if alloc > 20:
            h_opt, c_opt = local_optimize(h_interp, alloc)
            if c_opt < best_c - 1e-12:
                best_h = h_opt
                best_c = c_opt

    # Phase 5: Final LP polish on best
    if time_left() > 20:
        alloc = min(time_left() * 0.4, 60)
        h_opt, c_opt = local_optimize(best_h.copy(), alloc)
        if c_opt < best_c - 1e-12:
            best_h = h_opt
            best_c = c_opt

    # Phase 6: Final coordinate polish
    if time_left() > 10:
        h_opt, c_opt = coordinate_polish(best_h.copy(), min(time_left() * 0.5, 40))
        if c_opt < best_c - 1e-12:
            best_h = h_opt
            best_c = c_opt

    # Final verification
    best_h = project_constraint(best_h, len(best_h))
    best_c = compute_c5(best_h)

    return best_h, best_c, len(best_h)
