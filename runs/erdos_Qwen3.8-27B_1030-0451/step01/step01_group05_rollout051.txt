    """Find d in [d_lo, d_hi] minimizing max(raw + d*lin + d^2*quad)."""
    if d_hi - d_lo < 1e-15:
        return 0.0
    
    best_d = 0.0
    best_val = float('inf')
    
    # Grid search
    ds = np.linspace(d_lo, d_hi, n_grid)
    for d in ds:
        vals = raw + d * lin + d*d * quad
        v = np.max(vals)
        if v < best_val:
            best_val = v
            best_d = d

    if best_d > d_lo and best_d < d_hi:
        # Find neighboring grid points
        idx = int((best_d - d_lo) / (d_hi - d_lo) * (n_grid - 1))
        lo = ds[max(0, idx-1)]
        hi = ds[min(n_grid-1, idx+1)]
        # Golden section or just finer grid

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(z, n):
    S = n / 2.0
    lo = -5.0
    hi = 5.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-14:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_lin_quad(h, a, b):
    n = len(h)
    L = 2 * n - 1
    center = n - 1
    lin = np.zeros(L)
    quad = np.zeros(L)
    for k in range(-(n - 1), n):
        K = center + k
        candidates = (a, b, a - k, b - k)
        seen_k = set()
        for i in candidates:
            if i in seen_k:
                continue
            if 0 <= i < n and 0 <= i + k < n:
                seen_k.add(i)
                A = (1.0 if i + k == a else 0.0) - (1.0 if i + k == b else 0.0)
                B = (1.0 if i == a else 0.0) - (1.0 if i == b else 0.0)
                lin[K] += A * (1.0 - h[i]) - B * h[i + k]
                quad[K] -= A * B
    return lin, quad

def compute_gradient_fast(h, weights):
    n = len(h)
    L = 2 * n - 1
    center = n - 1
    g = np.zeros(n)
    for K in range(L):
        w = weights[K]
        if abs(w) < 1e-15:
            continue
        k = K - center
        if k >= 0:
            if k < n:

        else:
            m = -k
            if m < n:
                g[:n - m] += w * (1.0 - h[m:])
                g[m:] += w * (-h[:n - m])
    return g

    best_score = cur_score

    for it in range(max_iters):
        if time.time() > deadline:
            break
        progress = it / max(max_iters, 1)
        T = T0 * (1.0 - progress * 0.9)
        if T < 1e-6:
            T = 1e-6

        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        d_lo = max(-cur_h[a], cur_h[b] - 1.0)
        d_hi = min(1.0 - cur_h[a], cur_h[b])
        if d_hi - d_lo < 1e-14:
            continue
        d = (d_hi - d_lo) * rng.random() - d_lo
        if abs(d) < 1e-14:
            continue

        lin, quad = compute_lin_quad(cur_h, a, b)
        new_raw = cur_raw + d * lin + d * d * quad
        new_score = float(np.max(new_raw)) * dx
        delta = new_score - cur_score

        if delta < -1e-12 or (delta <= 0.02 and rng.random() < np.exp(-delta / max(T, 1e-10))):
            cur_h[a] = np.clip(cur_h[a] + d, 0.0, 1.0)
            cur_h[b] = np.clip(cur_h[b] - d, 0.0, 1.0)
            cur_raw = new_raw
            cur_score = new_score
            if cur_score < best_score:
                best_h = cur_h.copy()
                best_score = cur_score

    return best_h, best_score


def interp_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + max(budget_s - 15, 10)

    try:
        parent = np.array(initial_h_values, dtype=float).copy().flatten()
    except Exception:
        parent = None

    if parent is not None:
        n_p = len(parent)
        if not (n_p >= 2 and np.all(np.isfinite(parent))):
            parent = None

    if parent is not None:
        parent = project_feasible(parent, len(parent))
        score_p = exact_score(parent)
    else:
        n_p = 81
        parent = project_feasible(np.full(n_p, 0.5), n_p)
        score_p = exact_score(parent)

    best_h = parent.copy()
    best_score = score_p
    best_n = len(parent)

    # Phase 1: Aggressive pair descent with high temperature
    if time.time() < deadline - 100:
        d1 = min(deadline - 100, t0 + 250)
        h1, s1 = exact_pair_descent(parent, 40, 0.05, rng, d1)
        if s1 < best_score:
            best_h = h1
            best_score = s1

    # Phase 2: Cooled descent
    if time.time() < deadline - 80:
        d2 = min(deadline - 80, t0 + 450)
        h2, s2 = exact_pair_descent(best_h, 40, 0.02, rng, d2)
        if s2 < best_score:
            best_h = h2
            best_score = s2

    # Phase 3: Simulated annealing for escape
    if time.time() < deadline - 100:
        h3, s3 = sa_exchange(best_h, rng, deadline, 10000, 0.04 * best_score)
        if s3 < best_score:
            best_h = h3
            best_score = s3
        # Follow up with descent after SA
        if time.time() < deadline - 50:
            h3b, s3b = exact_pair_descent(best_h, 30, 0.01, rng, deadline)
            if s3b < best_score:
                best_h = h3b
                best_score = s3b

    # Phase 4: Final cooled descent
    if time.time() < deadline - 30:
        h4, s4 = exact_pair_descent(best_h, 30, 0.005, rng, deadline)
        if s4 < best_score:
            best_h = h4
            best_score = s4

    # Phase 5: Try dimension changes
    if time.time() < deadline - 50:
        for factor in [0.85, 1.15, 0.75]:
            if time.time() > deadline - 30:
                break
            dn = int(round(best_n * factor))
            if dn < 40 or dn > 2500 or dn == best_n:
                continue
            h_d = interp_h(best_h, dn)
            s_d = exact_score(h_d)
            if s_d < best_score + 0.005:
                h_d, s_d = exact_pair_descent(h_d, 20, 0.02, rng, deadline)
                if s_d < best_score:
                    best_h = h_d
                    best_score = s_d
                    best_n = dn

    # Phase 6: One more SA + descent cycle if time
    if time.time() < deadline - 40:
        h6, s6 = sa_exchange(best_h, rng, deadline, 5000, 0.02 * best_score)
        if s6 < best_score:
            best_h = h6
            best_score = s6
            if time.time() < deadline - 15:
                h6b, s6b = exact_pair_descent(best_h, 20, 0.004, rng, deadline)
                if s6b < best_score:
                    best_h = h6b
                    best_score = s6b

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    best_score = final_score

    return best_h, float(best_score), int(best_n)
