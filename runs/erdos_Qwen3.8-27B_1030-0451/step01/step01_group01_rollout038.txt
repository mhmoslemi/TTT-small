        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            mask = y < 1.0
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.minimum(y[mask] + r / cnt, 1.0)
        else:
            mask = y > 0.0
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.maximum(y[mask] + r / cnt, 0.0)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def pair_exchange_optimize(h0, deadline, max_rounds=200000):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    stale = 0
    fracs = (0.5, 0.3, 0.7, 0.2, 0.8, 0.4, 0.6, 0.1, 0.9)

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        cv = corr_values(h)
        c_cur = float(cv.max())

        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 5000:
                break

        thresh = c_cur - 0.003
        active_idx = np.where(cv >= thresh)[0]
        active_lags = np.abs(active_idx - (n - 1))
        active_lags = active_lags[(active_lags > 0) & (active_lags < n)]

        made_move = False

        if len(active_lags) > 0:
            ul = np.unique(active_lags)
            lag_order = []
            for L in ul:
                idxs = np.where(np.abs(np.arange(2 * n - 1) - (n - 1)) == L)[0]
                if len(idxs) > 0:
                    lag_order.append((cv[idxs].max(), int(L)))
            lag_order.sort(reverse=True)

            for _, L in lag_order[:4]:
                if time.time() > deadline:
                    break
                if L >= n:
                    continue
                terms = h[:n - L] * (1.0 - h[L:])
                top_k = min(20, n - L)
                order = np.argsort(terms)[::-1][:top_k]
                for idx in order:
                    if time.time() > deadline:
                        break
                    i = int(idx)
                    j = i + L
                    s = h[i] + h[j]
                    lo_b = max(0.0, s - 1.0)
                    hi_b = min(1.0, s)
                    if hi_b - lo_b < 1e-5:
                        continue
                    old_i, old_j = h[i], h[j]
                    for frac in fracs:
                        t = lo_b + frac * (hi_b - lo_b)
                        if abs(t - old_i) < 1e-7:
                            continue
                        h[i] = t
                        h[j] = s - t
                        c_new = evaluate_c5(h)
                        if c_new < best_c - 1e-13:
                            best_c = c_new
                            best_h = h.copy()
                            stale = 0
                            made_move = True
                            break
                        elif c_new < c_cur - 1e-13:
                            made_move = True
                            break
                        else:
                            h[i] = old_i
                            h[j] = old_j
                    if made_move and c_new < best_c:
                        break

        if not made_move:
            best_rand_val = c_cur
            best_rand_h = None
            n_try = 80 if stale > 500 else 40
            for _ in range(n_try):
                if time.time() > deadline:
                    break
                i = np.random.randint(0, n)
                j = np.random.randint(0, n)
                if i == j:
                    continue
                s = h[i] + h[j]
                lo_b = max(0.0, s - 1.0)
                hi_b = min(1.0, s)
                if hi_b - lo_b < 1e-5:
                    continue
                old_i, old_j = h[i], h[j]
                t = np.random.uniform(lo_b, hi_b)
                h[i] = t
                h[j] = s - t
                c_new = evaluate_c5(h)
                if c_new < best_rand_val:
                    best_rand_val = c_new
                    best_rand_h = h.copy()
                h[i] = old_i
                h[j] = old_j
            if best_rand_h is not None:
                h = best_rand_h
            else:
                break

    return best_h, best_c

def subgradient_optimize(h0, deadline, max_iters=200000):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    momentum = 0.0
    step = 0.03
    stale = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        cv = corr_values(h)
        c_cur = float(cv.max())

        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
            stale = 0
        else:
            stale += 1
            if stale > 30000:
                break

        thresh = c_cur - 0.01 * c_cur
        active_idx = np.where(cv >= thresh)[0]

        g = np.zeros(n)
        for idx in active_idx:
            lag = int(idx) - (n - 1)
            L = abs(lag)
            if L == 0:
                g += (1.0 - 2.0 * h)
            elif lag > 0:
                g[L:] += (1.0 - h[:-L])
                g[:n - L] -= h[L:]
            else:
                g[:n - L] += (1.0 - h[L:])
                g[L:] -= h[:-L]
        g /= len(active_idx)

        momentum = 0.9 * momentum - step * g
        h_new = h + momentum
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)

        if c_new < c_cur - 1e-14:
            h = h_new
            step = min(step * 1.01, 0.06)
        else:
            h = h_new
            step *= 0.95
            if step < 1e-6:
                break

    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def make_binary_seed(n, seed_val):
    rng = np.random.RandomState(seed_val)
    h = np.zeros(n)
    n_ones = n // 2
    positions = rng.choice(n, size=n_ones, replace=False)
    h[positions] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass

    if h_init is None:
        h_init = np.full(84, 0.5)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def time_ok(dl):
        return time.time() < dl

    # Phase 1: Aggressive pair exchange
    dl1 = min(deadline, start_time + budget_s * 0.30)
    try:
        h1, c1 = pair_exchange_optimize(best_h, dl1)
        if c1 < best_c:
            best_h, best_c = h1, c1
    except Exception:
        pass

    # Phase 2: Subgradient refinement
    if time_ok(deadline):
        dl2 = min(deadline, start_time + budget_s * 0.45)
        try:
            h2, c2 = subgradient_optimize(best_h, dl2, max_iters=300000)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # Phase 3: Pair exchange again from improved point
    if time_ok(deadline):
        dl3 = min(deadline, start_time + budget_s * 0.60)
        try:
            h3, c3 = pair_exchange_optimize(best_h, dl3)
            if c3 < best_c:
                best_h, best_c = h3, c3
        except Exception:
            pass

    # Phase 4: Try multiple binary seeds at same resolution
    if time_ok(deadline):
        n_cur = len(best_h)
        dl4 = min(deadline, start_time + budget_s * 0.70)
        remaining = dl4 - time.time()
        if remaining > 30:
            time_per_seed = remaining / 4.0
            for s_val in [123, 456, 789, 101]:
                if time.time() > dl4 - 5:
                    break
                h_bin = make_binary_seed(n_cur, s_val)
                # Scale to continuous
                h_seed = 0.5 + 0.3 * (h_bin - 0.5)
                h_seed = project_sum_box(h_seed, n_cur / 2.0)
                c_seed = evaluate_c5(h_seed)
                sub_dl = min(dl4 - 5, time.time() + time_per_seed * 0.7)
                try:
                    hs, cs = pair_exchange_optimize(h_seed, sub_dl, max_rounds=20000)
                    if cs < best_c:
                        best_h, best_c = hs, cs
                except Exception:
                    pass

    # Phase 5: Resolution transfer
    if time_ok(deadline):
        n_cur = len(best_h)
        target_ns = sorted(set([int(n_cur * f) for f in [1.5, 2.0]]))
        target_ns = [t for t in target_ns if t > n_cur and t <= 250]
        dl5 = min(deadline, start_time + budget_s * 0.88)
        for n_t in target_ns:
            if time.time() > dl5 - 30:
                break
            try:
                h_tr = transfer_resolution(best_h, n_t)
                c_tr = evaluate_c5(h_tr)
                if c_tr > best_c + 0.005:
                    continue
                sub_dl = min(dl5 - 10, time.time() + min(150, (dl5 - time.time()) * 0.5))
                h_a, c_a = pair_exchange_optimize(h_tr, sub_dl, max_rounds=30000)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a
                sub_dl2 = min(dl5 - 5, time.time() + min(60, (dl5 - time.time()) * 0.3))
                try:
                    h_b, c_b = subgradient_optimize(best_h, sub_dl2, max_iters=100000)
                    if c_b < best_c:
                        best_h, best_c = h_b, c_b
                except Exception:
                    pass
            except Exception:
                pass

    # Phase 6: Final polish with pair exchange
    if time.ok(deadline) if False else time.time() < deadline - 3:
        dl6 = deadline - 2.0
        try:
            h6, c6 = pair_exchange_optimize(best_h, dl6, max_rounds=100000)
            if c6 < best_c:
                best_h, best_c = h6, c6
        except Exception:
            pass

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
