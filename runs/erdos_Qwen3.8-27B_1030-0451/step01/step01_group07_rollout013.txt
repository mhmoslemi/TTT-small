def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def project_exact(h, target):
    # Project onto {0<=h<=1, sum(h)=target}
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(70):
        t = 0.5*(lo+hi)
        s = np.clip(h-t, 0.0, 1.0).sum()
        if s > target: lo = t
        else: hi = t
    t = 0.5*(lo+hi)
    out = np.clip(h-t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 0:
        if d < 0:
            room = (1.0-out).sum()
            if room > 0: out += (-d/room)*(1.0-out)
        else:
            room = out.sum()
            if room > 0: out -= (d/room)*out
        out = np.clip(out, 0.0, 1.0)
    return out

def repair(h, target):
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    h = project_exact(h, target)
    d = h.sum() - target
    if abs(d) > 1e-14:
        h = h.copy()
        idx = int(np.argmax(h if d > 0 else (1.0-h)))
        h[idx] = float(np.clip(h[idx]-d, 0.0, 1.0))
    return h

def c5_value(h):
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    return float(c.max() * dx)

def binary_init(n, rng):
    """Create a random binary sequence with exactly n/2 ones."""
    assert n % 2 == 0
    h = np.zeros(n)
    ones = rng.choice(n, size=n//2, replace=False)
    h[ones] = 1.0
    return h

def binary_local_search(h, rng, time_limit, deadline=None):
    """Simulated annealing on binary sequences via pair swaps."""
    n = len(h)
    target = n // 2
    h = h.astype(float).copy()
    # Ensure binary
    h = (h >= 0.5).astype(float)
    # Ensure exactly target ones
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    if len(ones) > target:
        excess = len(ones) - target
        to_flip = rng.choice(ones, excess, replace=False)
        h[to_flip] = 0
    elif len(ones) < target:
        deficit = target - len(ones)
        to_flip = rng.choice(zeros, deficit, replace=False)
        h[to_flip] = 1
    
    cur_c = c5_value(h)
    cur_corr = np.correlate(h, 1.0-h, mode='full')
    best_h = h.copy()
    best_c = cur_c
    T = 0.02
    Tend = 1e-6
    t0 = time.monotonic()
    total = max(time_limit, 1e-9)
    
    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        # Pick a 1 and a 0 to swap
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        i = int(ones[rng.integers(len(ones))])
        j = int(zeros[rng.integers(len(zeros))])
        
        # Try the swap
        h[i], h[j] = 0.0, 1.0
        new_corr = np.correlate(h, 1.0-h, mode='full')
        new_c = float(new_corr.max() * (2.0/n))
        
        delta = new_c - cur_c
        if delta < -1e-14 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur_c = new_c
            cur_corr = new_corr
            if new_c < best_c - 1e-14:
                best_c = new_c
                best_h = h.copy()
        else:
            # Revert
            h[i], h[j] = 1.0, 0.0
        
        frac = min(1.0, (time.monotonic()-t0)/total)
        T = Tend + (0.02 - Tend) * (1-frac)**2
    
    return best_h, best_c

def lse_polish(h0, rng, time_limit, sigma, target, deadline=None):
    """Polish using LSE surrogate."""
    n = len(h0)
    h = np.asarray(h0, dtype=float).copy()
    dx = 2.0/n
    cur_lse = _lse(h, sigma, dx)
    best_h = h.copy()
    best_c = c5_value(h)
    eta = 0.05
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = _lse_grad(h, sigma, dx)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        e = eta
        improved = False
        for _ in range(30):
            cand = project_exact(h - e*g, target)
            cv = _lse(cand, sigma, dx)
            if cv < cur_lse - 1e-13:
                h = cand
                cur_lse = cv
                c = c5_value(h)
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            e *= 0.5
        if not improved:
            # Perturb
            h = project_exact(h + rng.normal(0, 0.01, n), target)
            cur_lse = _lse(h, sigma, dx)
            stall += 1
            if stall >= 8:
                eta *= 0.5
                stall = 0
                if eta < 1e-7:
                    break
    return best_h, best_c

def _lse(h, sigma, dx):
    c = np.correlate(h, 1.0-h, mode='full')
    m = c.max()
    return dx * (m + math.log(np.exp(sigma*(c-m)).sum()) / sigma)
