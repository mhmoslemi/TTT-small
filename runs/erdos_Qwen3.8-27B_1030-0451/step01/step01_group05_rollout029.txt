    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')

    corr_max = np.max(corr)
    weights = np.exp((corr - corr_max) / tau)
    weights /= np.sum(weights)
    
    # Gradient of sum_k w_k * corr_k w.r.t. h[i]
    L = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for s in range(-(n-1), n):

        w = weights[K] * dx
        if w < 1e-10:
            continue
        if s >= 0:

            i_idx = np.arange(n - s)

            grad[i_idx] += w * (-h[i_idx + m])
    
    return grad

        cn = max(21, int(round(pn * factor)))
        if cn != pn and cn <= 300:
            dim_candidates.append(cn)
    # Add some absolute values
    for av in [41, 61, 101, 121, 161, 201]:
        if av != pn and av not in dim_candidates:
            dim_candidates.append(av)
    dim_candidates = sorted(set(dim_candidates))

    for dn in dim_candidates:
        if _remaining() < 40:
            break
        h_d = interpolate_h(best_h, dn)
        s_d = exact_score(h_d)
        if s_d >= best_score + 0.01:
            continue
        # Quick optimization
        for _ in range(3):
            if _remaining() < 35:
                break
            h_d, imp = full_pair_search(h_d, n_evals=12)
            s_d = exact_score(h_d)
            if imp < 1e-12:
                break
        # SLSQP
        h_new = run_slsqp_polish(h_d, dn, 0.005, 100, deadline)
        if h_new is not None:
            s_new = exact_score(h_new)
            if s_new < s_d:
                h_d = h_new
                s_d = s_new
        h_d, imp = full_pair_search(h_d, n_evals=12)
        s_d = exact_score(h_d)
        if s_d < best_score - 1e-14:
            best_h = h_d
            best_score = s_d
            best_n = dn

    # Phase 6: Final polish loop
    for _ in range(3):
        if _remaining() < 20:
            break
        h_new = run_slsqp_polish(best_h, best_n, 0.001, 200, deadline)
        if h_new is not None:
            s_new = exact_score(h_new)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new
        best_h, imp = full_pair_search(best_h, n_evals=20)
        s = exact_score(best_h)
        if s < best_score - 1e-14:
            best_score = s

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-12:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
