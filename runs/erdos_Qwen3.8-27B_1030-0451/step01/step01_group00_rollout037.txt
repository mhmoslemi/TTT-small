def project_constraint(p, n):
    # Project p onto {x: 0 <= x[i] <= 1, sum(x) = n/2}
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)

def compute_gradient(h, corr, tau):
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    # Vectorized gradient computation
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            L = lag
            if L > 0:
                grad[L:] += wk * (1.0 - h[:n - L])
                grad[:n - L] -= wk * h[L:]
            else:
                # lag = 0: c[0] = sum(h[i](1-h[i]))
                # grad = (1-2h) ... actually c[0] = sum h - sum h^2
                # d/dh[j] = 1 - 2h[j]
                grad[:] += wk * (1.0 - 2.0 * h)
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

Hmm, I realize there's an issue with my Nesterov implementation. The standard Nesterov accelerated gradient is:

```
y = x + beta * v  (lookahead)
g = grad(y)
v = beta * v + g  (or v = beta * v + (1-mu) * g for heavy ball)
x = x - lr * v
```

But with projection (for constraints), it's more subtle. The projected Nesterov is:
```
y = x + beta * (x - x_prev)  (extrapolation)
g = grad(project(y))
x_new = project(x - lr * grad + beta * (x - x_prev))
```

Actually, let me use a simpler and well-tested approach: **FISTA (Fast Iterative Shrinkage-Thresholding Algorithm)** style, adapted for our problem.

FISTA:
```
x_0 = initial
y_1 = x_0
t_1 = 1
for k = 1, 2, ...:
    x_{k+1} = proj(y_k - lr * grad(y_k))
    t_{k+1} = (1 + sqrt(1 + 4*t_k^2)) / 2
    y_{k+1} = x_{k+1} + (t_k - 1) / t_{k+1} * (x_{k+1} - x_k)
```

This has O(1/k²) convergence rate for convex problems. Our problem (minimizing the softmax approximation) is convex in h for fixed τ (since the correlation is linear in h... wait, is it?)

c[k] = Σ h[i](1-h[i+k]) = Σ h[i] - Σ h[i]h[i+k]

The first term is linear, the second is quadratic (bilinear). So c[k] is NOT linear in h. The softmax of c[k] is also not convex in general.

So FISTA's guarantees don't directly apply. But in practice, it often works well.

Let me use a simpler approach: **heavy ball method** (momentum gradient descent):
```
v = 0
for each iteration:
    g = grad(h)
    v = beta * v + g
    h_new = proj(h - lr * v)
```

This is simpler and often works as well as Nesterov for non-convex problems.

This should converge to a point where the top lags are balanced.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    s = float(np.sum(p))
    if abs(s - target) < 1e-14 and float(np.min(p)) >= -1e-14 and float(np.max(p)) <= 1.0 + 1e-14:
        return np.clip(p, 0.0, 1.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        sv = float(np.sum(proj))
        if sv > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-13:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def compute_gradient(h, corr, tau):
    n = len(h)
    maxc = float(np.max(corr))
    cs = corr - maxc
    w = np.exp(tau * cs)
    ws = float(w.sum())
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    thr = 1e-14
    for k in range(len(corr)):
        wk = w[k]
        if wk < thr:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            L = lag
            grad[L:] += wk * (1.0 - h[:n - L])
            grad[:n - L] -= wk * h[L:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

def softmax_value(corr, tau):
    maxc = float(np.max(corr))
    cs = corr - maxc
    return float(maxc + np.log(float(np.sum(np.exp(tau * cs)))) / tau)

def make_oscillatory_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.28 * np.sin(2.0 * np.pi * x)
    h += 0.14 * np.sin(4.0 * np.pi * x + 0.6)
    h += 0.07 * np.sin(6.0 * np.pi * x + 1.4)
    h += 0.04 * np.sin(10.0 * np.pi * x + 2.2)
    h += rng.randn(n) * 0.015
    return project_constraint(h, n)

def optimize_momentum(h0, time_limit, tau_start, tau_end, beta=0.85):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    v = np.zeros(n)
    t0 = time.time()
    corr = compute_corr(h)
    best_c5 = float(np.max(corr) * dx)
    best_h = h.copy()
    step = 2.0 / n
    no_improve = 0
    stall_thresh = 400
    while time.time() - t0 < time_limit:
        elapsed = max(time.time() - t0, 1e-6)
        frac = min(elapsed / max(time_limit, 1.0), 1.0)
        tau = tau_start + (tau_end - tau_start) * (frac ** 0.6)
        corr = compute_corr(h)
        c5_curr = float(np.max(corr) * dx)
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        if no_improve > stall_thresh:
            rng = np.random.RandomState()
            xg = np.linspace(0, 2, n, endpoint=False)
            a = rng.randn(4)
            pert = a[0]*np.sin(2*np.pi*xg) + a[1]*np.sin(4*np.pi*xg+1.0) + a[2]*np.sin(6*np.pi*xg+2.0) + a[3]*np.sin(8*np.pi*xg+3.0)
            scale = 0.03 + 0.04 * min(1.0, tau / 300.0)
            h = project_constraint(h + pert * scale + rng.randn(n) * 0.02, n)
            v = np.zeros(n)
            no_improve = 0
            step = 2.0 / n
            continue
        grad = compute_gradient(h, corr, tau)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        v = beta * v + grad
        obj_curr = softmax_value(corr, tau)
        made = False
        for bt in range(15):
            h_new = project_constraint(h - step * v, n)
            corr_new = compute_corr(h_new)
            obj_new = softmax_value(corr_new, tau)
            if obj_new <= obj_curr + 1e-13:
                h = h_new
                made = True
                break
            step *= 0.5
        if made:
            step = min(step * 1.08, 20.0 / n)
        else:
            step = max(step * 0.4, 0.05 / n)
            h = project_constraint(h - step * v, n)
    return best_h, best_c5

def active_set_optimize(h0, time_limit, top_m=8):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    t0 = time.time()
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step = 1.0 / n
    while time.time() - t0 < time_limit:
        corr = compute_corr(h)
        c5_curr = float(np.max(corr) * dx)
        if c5_curr < best_c5 - 1e-14:
            best_c5 = c5_curr
            best_h = h.copy()
        top_lags = np.argpartition(corr, -top_m)[-top_m:]
        max_top = float(np.max(corr[top_lags]))
        top_cs = corr[top_lags] - max_top
        tau = 150.0
        w = np.exp(tau * top_cs)
        w = w / w.sum()
        grad = np.zeros(n)
        for i in range(top_m):
            k_idx = int(top_lags[i])
            wk = w[i]
            lag = k_idx - (n - 1)
            if lag >= 0:
                L = lag
                grad[L:] += wk * (1.0 - h[:n - L])
                grad[:n - L] -= wk * h[L:]
            else:
                L = -lag
                grad[:n - L] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:n - L]
        gn = float(np.linalg.norm(grad))
        if gn < 1e-14:
            break
        made = False
        for bt in range(15):
            h_new = project_constraint(h - step * grad, n)
            c_new = compute_corr(h_new)
            c5_new = float(np.max(c_new) * dx)
            top_new = np.max(c_new[top_lags])
            if top_new <= max_top + 1e-13:
                h = h_new
                made = True
                break
            step *= 0.5
        if made:
            step = min(step * 1.05, 5.0 / n)
        else:
            step = max(step * 0.5, 0.02 / n)
    return best_h, best_c5
