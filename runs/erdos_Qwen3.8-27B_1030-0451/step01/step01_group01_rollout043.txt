import numpy as np

def project_sum_box(h, target):
    """Project h onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(h)
    h = np.array(h, dtype=float).copy()
    # Bisection on shift
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
    y = np.clip(y, 0.0, 1.0)
    # Final fix
    r = target - y.sum()
    if abs(r) > 1e-14:
        idx = np.where((y < 1.0) if r > 0 else (y > 0.0))[0]
        if len(idx) > 0:
            y[idx] += r / len(idx)
    return y

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_full_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def active_lags(h, gap=0.01):
    """Find lags where correlation is within gap of max."""
    v = corr_full_scaled(h)
    vmax = np.max(v)
    return np.where(v >= vmax - gap)[0], v, vmax

def pair_exchange_line_search(h, i, j, n_samples=40):
    """For pair (i,j), find optimal t in feasible range that minimizes C5."""
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-10:
        return None, None
    # Try multiple candidate t values
    best_c = float('inf')
    best_t = None
    t_vals = np.linspace(lo, hi, n_samples)
    # Also include current value
    if h[i] > lo and h[i] < hi:
        t_vals = np.concatenate([t_vals, [h[i]]])
    for t in t_vals:
        h_tr = h.copy()
        h_tr[i] = t
        h_tr[j] = s - t
        c = evaluate_c5(h_tr)
        if c < best_c:
            best_c = c
            best_t = t
    return best_t, best_c

def pair_exchange_targeted(h, target_lag_dist, n_samples=30):

def method_active_lp(h0, deadline, max_iters=300, gap=0.005, max_active=60):
    """LP trust region on active constraints."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.15
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_full_scaled(h)
        vmax = v.max()
        # Active lags
        threshold = vmax - max(gap * vmax, 0.003)
        active_idx = np.where(v >= threshold)[0]
        if len(active_idx) > max_active:
            # Keep only the most active
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-max_active:]]
        
        # Build LP: minimize t s.t. grad_k · (h_new - h) + v_k <= t for active k
        # h_new = h + delta, |delta| <= radius componentwise
        # Actually: minimize t s.t. v_k_new <= t, sum(h_new) = n/2, 0<=h_new<=1
        # Linearization: v_k(h + d) ≈ v_k(h) + grad_k · d
        A = np.zeros((len(active_idx), n))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            if lag >= 0:
                A[i, lag:n] = 1.0 - h[:n-lag]
                A[i, :n-lag] = -h[lag:]
            else:
                l = -lag
                A[i, :n-l] = 1.0 - h[l:]
                A[i, l:] = -h[:n-l]
        
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0  # minimize t
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        b_ub = np.zeros(len(active_idx))
        lb = np.maximum(h - radius, 0.0)
        ub = np.minimum(h + radius, 1.0)
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs',
                          options={'presolve': True})
        except Exception:
            res = None
        
        if res is not None and res.success:
            h_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 1e-11:
                best_h = h_new.copy()
                best_c = c_new
                h = h_new.copy()
                no_improve = 0
                radius = min(radius * 1.05, 0.4)
            else:
                no_improve += 1
                if no_improve >= 5:
                    radius *= 0.6
                    no_improve = 0
                    if radius < 0.005:
                        break
        else:
            no_improve += 1
            if no_improve >= 10:
                radius *= 0.5
                no_improve = 0
                if radius < 0.005:
                    break
    return best_h, best_c

def method_smooth_gradient(h0, deadline, max_iters=3000, beta_start=20, beta_end=500):
    """Smooth gradient descent with increasing beta."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.02
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_full_scaled(h)
        m = v.max()
        # Adaptive beta
        beta = beta_start + (beta_end - beta_start) * min(1.0, it / (max_iters * 0.5))
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        
        # Compute weighted gradient
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] < 1e-12:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                g[lag:n] += w[k] * (1.0 - h[:n-lag])
                g[:n-lag] += w[k] * (-h[lag:])
            else:
                l = -lag
                g[:n-l] += w[k] * (1.0 - h[l:])
                g[l:] += w[k] * (-h[:n-l])
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        h_new = h - step * g / g_norm
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        
        if c_new < best_c - 1e-12:
            if c_new < evaluate_c5(h) - 1e-14:
                h = h_new
                step = min(step * 1.03, 0.1)
            else:
                h = h_new
                step *= 0.9
            stall = 0
            if c_new < best_c:
                best_c = c_new
                best_h = h_new.copy()
        else:
            step *= 0.7
            stall += 1
            if stall > 100:
                break
            if step < 1e-7:
                break
    return best_h, best_c

def method_binary_sa(n, deadline, max_iters=50000):
    """Simulated annealing on binary sequences."""
    # Need even n for balanced binary
    if n % 2 != 0:
        n -= 1
    half = n // 2
    # Start with random balanced binary
    idx = np.random.permutation(n)
    h = np.zeros(n)
    h[idx[:half]] = 1.0
    c = evaluate_c5(h)
    best_h = h.copy()
    best_c = c
    T = 0.5
    cooling = 0.9997
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # Swap a 1 with a 0
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = ones[np.random.randint(len(ones))]
        j = zeros[np.random.randint(len(zeros))]
        if i == j:
            continue
        h_old = h.copy()
        h[i] = 0.0
        h[j] = 1.0
        c_new = evaluate_c5(h)
        delta = c_new - c
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            c = c_new
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h = h_old
        T *= cooling
    return best_h, best_c

def transfer_resolution(h0, n_new):
    """Interpolate to new resolution."""
    n_old = len(h0)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h0, period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def method_coord_pairs(h0, deadline, max_rounds=20):
    """Coordinate pair descent targeting active lags."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        v = corr_full_scaled(best_h)
        vmax = v.max()
        # Get active lag distances
        threshold = vmax - 0.01 * vmax
        active = np.where(v >= threshold)[0]
        lag_dists = np.abs(active - (n - 1))
        # Get unique distances
        unique_dists = np.unique(lag_dists)
        unique_dists = unique_dists[(unique_dists >= 1) & (unique_dists < n)]
        
        improved = False
        for d in unique_dists:
            if time.time() > deadline:
                break
            d_int = int(d)
            for i in range(n - d_int):
                j = i + d_int
                s = best_h[i] + best_h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-8:

                candidates = np.linspace(lo, hi, 8)

    else:
        l = -lag

def binary_pattern(n, period):
    """Create a binary pattern with given period structure."""
    if n % 2 != 0:
        n -= 1
    half = n // 2
    h = np.zeros(n)
    # Fill with 1s at positions that are multiples of period offset
    # Simple: first half ones, second half zeros, then permute
    fill = np.zeros(n, dtype=int)
    fill[:half] = 1
    # Apply permutation based on period
    perm = np.zeros(n, dtype=int)
    for i in range(n):
        perm[i] = (i * period) % n
    h[perm] = fill
    # Ensure exactly half are 1
    # (The permutation might not preserve count if gcd(n,period) > 1)
    # Fix: just ensure sum = n/2
    ones_count = h.sum()
    if ones_count > half:
        # Remove excess
        one_idx = np.where(h == 1)[0]
        np.random.shuffle(one_idx)
        h[one_idx[:int(ones_count - half)]] = 0
    elif ones_count < half:
        zero_idx = np.where(h == 0)[0]
        np.random.shuffle(zero_idx)
        h[zero_idx[:int(half - ones_count)]] = 1
    return h

    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-13:
        if r > 0:
            idx = np.where(y < 1.0 - 1e-14)[0]
        else:
            idx = np.where(y > 1e-14)[0]
        if len(idx) > 0:
            y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[lag:] = (1.0 - h[:n - lag]) * 0 + 1.0 - h[:n - lag] + g[lag:] * 0
            g[lag:] = 1.0 - h[:n - lag]
            g[:n - lag] = -h[lag:]
    else:
        l = -lag
        if l == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            g[:n - l] = 1.0 - h[l:]
            g[l:] = -h[:n - l]
    return g


def method_lp_tight(h0, deadline, max_iters=500):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.25
    gap_frac = 0.003
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_scaled(h)
        vmax = v.max()
        gap = max(gap_frac * vmax, 0.0015)
        active = np.where(v >= vmax - gap)[0]
        max_act = min(120, len(active))
        if len(active) > max_act:
            order = np.argsort(-v[active])
            active = active[order[:max_act]]
        A = np.zeros((len(active), n))
        for i, k in enumerate(active):
            lag = k - (n - 1)
            A[i] = grad_at_lag(h, lag)
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub_arr = np.hstack([A, -np.ones((len(active), 1))])
        b_ub_arr = np.zeros(len(active))
        lb_arr = np.maximum(h - radius, 0.0)
        ub_arr = np.minimum(h + radius, 1.0)
        A_eq_arr = np.zeros((1, n_var))
        A_eq_arr[0, :n] = 1.0
        b_eq_arr = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub_arr, b_ub=b_ub_arr, A_eq=A_eq_arr, b_eq=b_eq_arr,
                          bounds=list(zip(lb_arr, ub_arr)), method='highs',
                          options={'presolve': True})
        except Exception:
            res = None
        if res is not None and res.success:
            h_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 1e-12:
                best_h = h_new.copy()
                best_c = c_new
                h = h_new.copy()
                no_improve = 0
                radius = min(radius * 1.04, 0.6)
            else:
                no_improve += 1
                if no_improve >= 4:
                    radius *= 0.65
                    no_improve = 0
                    if radius < 0.003:
                        break
        else:
            no_improve += 1
            if no_improve >= 8:
                radius *= 0.5
                no_improve = 0
                if radius < 0.003:
                    break
    return best_h, best_c


def method_pair_golden(h0, deadline, max_rounds=30):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        v = corr_scaled(best_h)
        vmax = v.max()
        active = np.where(v >= vmax - 0.006)[0]
        lag_dists = np.abs(active - (n - 1))
        unique_d = np.unique(lag_dists[(lag_dists >= 1) & (lag_dists < n)])
        # Sort by how active they are (most active first)
        scored = []
        for d in unique_d:
            di = int(d)
            mask = np.abs(np.arange(2 * n - 1) - (n - 1)) == di
            if mask.any():
                scored.append((v[mask].max(), d))
        scored.sort(key=lambda x: -x[0])
        improved = False
        for _, d in scored:
            if time.time() > deadline:
                break
            di = int(d)
            for i in range(n - di):
                j = i + di
                s = best_h[i] + best_h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-9:
                    continue
                a, b = lo, hi
                c_pt = b - gr * (b - a)
                d_pt = a + gr * (b - a)
                for _ in range(18):
                    hc = best_h.copy()
                    hc[i], hc[j] = c_pt, s - c_pt
                    fc = evaluate_c5(hc)
                    hd = best_h.copy()
                    hd[i], hd[j] = d_pt, s - d_pt
                    fd = evaluate_c5(hd)
                    if fc < fd:
                        b = d_pt
                        d_pt = c_pt
                        c_pt = b - gr * (b - a)
                    else:
                        a = c_pt
                        c_pt = d_pt
                        d_pt = a + gr * (b - a)
                t_opt = (a + b) / 2.0
                hn = best_h.copy()
                hn[i], hn[j] = t_opt, s - t_opt
                cn = evaluate_c5(hn)
                if cn < best_c - 1e-13:
                    best_c = cn
                    best_h = hn
                    improved = True
        if not improved:
            break
    return best_h, best_c


def method_smooth_descent(h0, deadline, max_iters=5000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.06
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_scaled(h)
        m = v.max()
        beta = 15.0 + 80.0 * min(1.0, it / 2000.0)
        expo = np.clip(beta * (v - m), None, 350)
        w = np.exp(expo)
        Z = w.sum()
        if Z < 1e-300:
            break
        w = w / Z
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] < 1e-14:
                continue
            lag = k - (n - 1)
            g += w[k] * grad_at_lag(h, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        c_cur = evaluate_c5(h)
        alpha = step
        h_try = h
        c_try = c_cur
        for ls in range(12):
            h_try = project_sum_box(h - alpha * g / gn, n / 2.0)
            c_try = evaluate_c5(h_try)
            if c_try < c_cur - 1e-15:
                break
            alpha *= 0.5
        if c_try < c_cur - 1e-14:
            h = h_try
            if c_try < best_c:
                best_c = c_try
                best_h = h.copy()
            stall = 0
            step = min(max(alpha * 1.3, 0.01), 0.2)
        else:
            stall += 1
            step *= 0.7
            if stall > 100:
                break
    return best_h, best_c


def transfer_res(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_ext = np.concatenate([h0, h0[:4]])
    x_ext = np.concatenate([x_old, np.linspace(1.0, 1.0 + 4.0 / n_old, 4, endpoint=False)])
    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new


def downsample_up(h0, factor):
    n = len(h0)
    n_ds = n // factor
    if n_ds % 2 != 0:
        n_ds -= 1
    h_ds = np.zeros(n_ds)
    for i in range(n_ds):
        h_ds[i] = np.mean(h0[i * factor:(i + 1) * factor])
    h_ds = project_sum_box(h_ds, n_ds / 2.0)
    h_back = transfer_res(h_ds, n)
    return h_back


def safe_run(func, *args):
    try:
        return func(*args)
    except Exception:
        h0 = args[0]
        return h0.copy(), evaluate_c5(h0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 5.0

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            if len(h_try) >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, len(h_try) / 2.0)
                if np.isfinite(evaluate_c5(h_try)):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def tl():
        return deadline - time.time()

    # Phase 1: LP trust region (aggressive)
    if tl() > 60:
        h1, c1 = safe_run(method_lp_tight, best_h, min(deadline, t0 + budget_s * 0.20), 600)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Golden section pair search
    if tl() > 40:
        h2, c2 = safe_run(method_pair_golden, best_h, min(deadline, t0 + budget_s * 0.35), 30)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: Smooth descent
    if tl() > 40:
        h3, c3 = safe_run(method_smooth_descent, best_h, min(deadline, t0 + budget_s * 0.50), 6000)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: LP again tighter
    if tl() > 50:
        h4, c4 = safe_run(method_lp_tight, best_h, min(deadline, t0 + budget_s * 0.63), 500)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Multi-start perturbations
    if tl() > 70:
        for att in range(4):
            if tl() < 25:
                break
            rng_scale = 0.015 + 0.01 * att
            pert = np.random.randn(len(best_h)) * rng_scale
            h_try = project_sum_box(best_h + pert, len(best_h) / 2.0)
            c_try = evaluate_c5(h_try)
            start_h = h_try if c_try < best_c else best_h.copy()
            h5, c5 = safe_run(method_lp_tight, start_h, min(deadline - 3, time.time() + tl() * 0.2), 150)
            if c5 < best_c:
                best_h, best_c = h5, c5
            h5b, c5b = safe_run(method_pair_golden, best_h, min(deadline - 3, time.time() + tl() * 0.15), 8)
            if c5b < best_c:
                best_h, best_c = h5b, c5b

    # Phase 6: Down-sample and up (escape local optima)
    if tl() > 50:
        n_cur = len(best_h)
        for fac in [2, 3, 4]:
            if tl() < 30:
                break
            h_ds = downsample_up(best_h, fac)
            c_ds = evaluate_c5(h_ds)
            if c_ds < best_c:
                best_h, best_c = h_ds, c_ds
            else:
                h_a, c_a = safe_run(method_lp_tight, h_ds, min(deadline - 3, time.time() + 30), 80)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a

    # Phase 7: Resolution transfer up
    if tl() > 50:
        n_cur = len(best_h)
        for scale in [1.3, 1.5]:
            if tl() < 35:
                break
            n_t = int(round(n_cur * scale))
            if n_t % 2 != 0:
                n_t += 1
            if n_t <= n_cur or n_t > 280:
                continue
            ht = transfer_res(best_h, n_t)
            ct = evaluate_c5(ht)
            h_a, c_a = safe_run(method_lp_tight, ht, min(deadline - 3, time.time() + 80), 120)
            if c_a < ct:
                ht, ct = h_a, c_a
            h_b, c_b = safe_run(method_pair_golden, ht, min(deadline - 3, time.time() + 50), 10)
            if c_b < ct:
                ht, ct = h_b, c_b
            if ct < best_c:
                best_h, best_c = ht, ct

    # Phase 8: Final smooth polish
    if tl() > 15:
        h8, c8 = safe_run(method_smooth_descent, best_h, min(deadline - 1, time.time() + tl() * 0.7), 5000)
        if c8 < best_c:
            best_h, best_c = h8, c8

    # Phase 9: Final pair polish
    if tl() > 10:
        h9, c9 = safe_run(method_pair_golden, best_h, deadline - 1, 15)
        if c9 < best_c:
            best_h, best_c = h9, c9

    # Final projection and evaluation
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    return best_h, float(final_c), len(best_h)
