def repair_feasible(h, target_sum=None):
    """Repair h to satisfy 0<=h<=1 and sum(h)=target_sum (default N/2)."""
    N = len(h)
    if target_sum is None:
        target_sum = N / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    # Binary search for the threshold
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) / 2.0
    h = np.clip(h + t, 0.0, 1.0)
    # Final correction
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.argsort(h)[::-1]
            for i in idx:
                room = 1.0 - h[i]
                add = min(residual, room)
                h[i] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                room = h[i]
                rem = min(-residual, room)
                h[i] -= rem
                residual += rem
                if abs(residual) < 1e-12:
                    break
    return h

def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)

    N = len(h)
    if lam_min < 0:
        lam_min = max(lam_min, -h.min() / max(d.max(), 1e-15))  # Already handled
    # Sample λ values
    lams = np.linspace(lam_min, lam_max, n_samples)

    best_c5 = float('inf')

import numpy as np
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_corr_vec(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx


def repair_feasible(h, target_sum=None):
    N = len(h)
    if target_sum is None:
        target_sum = N / 2.0
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(h)[::-1]
            for i in idx:
                room = 1.0 - h[i]
                add = min(residual, room)
                h[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                room = h[i]
                rem = min(-residual, room)
                h[i] -= rem
                residual += rem
                if abs(residual) < 1e-14:
                    break
    return h


def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return repair_feasible(h_new)


def get_feasible_interval(h, d):
    pos = d > 1e-14
    neg = d < -1e-14
    if not np.any(pos) and not np.any(neg):
        return None
    lams_lo = []
    lams_hi = []
    if np.any(pos):
        lams_lo.append(np.min(-h[pos] / d[pos]))
        lams_hi.append(np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        lams_lo.append(np.min((1.0 - h[neg]) / d[neg]))
        lams_hi.append(np.min(-h[neg] / d[neg]))
    lam_min = max(lams_lo)
    lam_max = min(lams_hi)
    if lam_min > -1e-15:
        lam_min = 0.0
    if lam_max < 1e-15:
        lam_max = 0.0
    if lam_min >= lam_max - 1e-14:
        return None
    return (lam_min, lam_max)


def exact_line_search(h, d, lam_min, lam_max, n_samples=40):
    N = len(h)
    best_c5 = compute_c5(h)
    best_lam = 0.0
    lams = np.linspace(lam_min, lam_max, n_samples)
    for lam in lams:
        h_new = h + lam * d
        if np.any(h_new < -1e-12) or np.any(h_new > 1 + 1e-12):
            h_new = np.clip(h_new, 0.0, 1.0)
        c5 = compute_c5(h_new)
        if c5 < best_c5:
            best_c5 = c5
            best_lam = lam
    if best_lam != 0.0:
        span = lam_max - lam_min
        half_w = span / n_samples * 1.5
        if best_lam - half_w < lam_min:
            lo = lam_min
            hi = min(best_lam + half_w, lam_max)
        elif best_lam + half_w > lam_max:
            lo = max(best_lam - half_w, lam_min)
            hi = lam_max
        else:
            lo = best_lam - half_w
            hi = best_lam + half_w
        lams_r = np.linspace(lo, hi, 20)
        for lam in lams_r:
            h_new = h + lam * d
            h_new = np.clip(h_new, 0.0, 1.0)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_lam = lam
    return best_lam, best_c5


def compute_subgradient(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    c = corr_raw * dx
    cmax = np.max(c)
    exponents = np.clip((c - cmax) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(c)
        w[np.argmax(c)] = 1.0
    else:
        w = w / w_sum
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r]
        if a < 1e-14:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[dd:] += a * (1.0 - h[:n])
                g[:n] -= a * h[dd:]
    g -= np.mean(g)
    return g


def make_harmonic_directions(N, max_freq=12):
    x = (np.arange(N) + 0.5) * (2.0 / N)
    dirs = []
    for k in range(1, max_freq + 1):
        d1 = np.sin(np.pi * k * x)
        d2 = np.cos(np.pi * k * x)
        d1 -= np.mean(d1)
        d2 -= np.mean(d2)
        n1 = np.linalg.norm(d1)
        n2 = np.linalg.norm(d2)
        if n1 > 1e-10:
            dirs.append(d1 / n1)
        if n2 > 1e-10:
            dirs.append(d2 / n2)
    return dirs


def local_search_ewea(h, rng, deadline, max_iters=500):
    N = len(h)
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    tau = 0.03
    stagnation = 0
    harmonics = make_harmonic_directions(N, max_freq=8)

    for it in range(max_iters):
        if time.time() > deadline - 5:
            break

        tau = max(0.003, 0.03 * (0.97 ** it))
        g = compute_subgradient(h, tau)
        g_norm = np.linalg.norm(g)
        if g_norm > 1e-12:
            g = g / g_norm

        directions = []
        directions.append(g)
        directions.append(-g)

        if len(harmonics) > 0:
            n_harm = min(4, len(harmonics))
            idx = rng.integers(0, len(harmonics), size=n_harm)
            for i in idx:
                directions.append(harmonics[i].copy())

        for _ in range(3):
            d_rand = rng.standard_normal(N)
            d_rand -= np.mean(d_rand)
            nr = np.linalg.norm(d_rand)
            if nr > 1e-12:
                directions.append(d_rand / nr)

        # Pair transfers from high to low gradient
        top_k = min(20, N // 5)
        g_rank = np.argsort(-np.abs(g))
        positives = g_rank[g[g_rank] > 0][:top_k]
        negatives = g_rank[g[g_rank] < 0][:top_k]
        for _ in range(min(8, len(positives), len(negatives))):
            pi = rng.integers(0, len(positives))
            ni = rng.integers(0, len(negatives))
            i_idx = positives[pi]
            j_idx = negatives[ni]
            if i_idx == j_idx:
                continue
            d_pair = np.zeros(N)
            d_pair[i_idx] = 1.0
            d_pair[j_idx] = -1.0
            directions.append(d_pair)

        improved = False
        for d in directions:
            if time.time() > deadline - 5:
                break
            d = np.array(d, dtype=float)
            d -= np.mean(d)
            dn = np.linalg.norm(d)
            if dn < 1e-14:
                continue
            d = d / dn
            interval = get_feasible_interval(h, d)
            if interval is None:
                continue
            lam_min, lam_max = interval
            best_lam, new_c5 = exact_line_search(h, d, lam_min, lam_max, n_samples=35)
            if new_c5 < current_c5 - 1e-12:
                h = h + best_lam * d
                h = np.clip(h, 0.0, 1.0)
                h = repair_feasible(h)
                current_c5 = compute_c5(h)
                improved = True
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    stagnation = 0
                else:
                    stagnation += 1

        if not improved:
            stagnation += 1
            if stagnation > 15:
                # Restart with perturbation
                scale = 0.05
                h = h + rng.normal(0, scale, N)
                h = repair_feasible(h)
                current_c5 = compute_c5(h)
                if current_c5 > best_c5:
                    h = best_h.copy()
                    current_c5 = best_c5
                # Try subgradient again with larger tau
                tau = max(tau * 2, 0.01)
                stagnation = 0
                continue
            if stagnation > 40:
                break

    return best_h, best_c5
