- If s ≥ 0: i_min = 0, i_max = n-1-s. So j ∈ [0, n-1-s], j+s ∈ [s, n-1]. Valid.
- If s < 0: i_min = -s, i_max = n-1. So j ∈ [-s, n-1], j+s ∈ [0, n-1+s]. Valid (since n-1+s ≥ 0 when s ≥ -(n-1)).

So the condition j ∈ [i_min, i_max] automatically ensures j+s is valid. Good.

Similarly for the first term: j-s ∈ [i_min, i_max] means j ∈ [i_min+s, i_max+s].
- If s ≥ 0: j ∈ [s, n-1]. And j-s ∈ [0, n-1-s], valid.
- If s < 0: j ∈ [0, n-1+s]. And j-s ∈ [-s, n-1], valid.

For s < 0: dc_k/dh[j] = dx * [(1-h[j-s]) * I(j ≤ n-1+s) - h[j+s] * I(j ≥ -s)]

def project_feasible(z):
    n = len(z)
    S = n / 2.0
    lo, hi = -5.0, 5.0

    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)

def grad_for_active(h, active_lags):
    """Compute gradient of active lag correlations w.r.t. h.
    Returns (m, n) matrix where m = len(active_lags)."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    m = len(active_lags)
    J = np.zeros((m, n))
    for idx, k in enumerate(active_lags):
        s = k - center
        if s >= 0:
            for j in range(s, n):
                J[idx, j] += dx * (1.0 - h[j - s])
            for j in range(0, n - s):
                J[idx, j] -= dx * h[j + s]
        else:
            m_ = -s
            for j in range(0, n - m_):
                J[idx, j] += dx * (1.0 - h[j + m_])
            for j in range(m_, n):
                J[idx, j] -= dx * h[j - m_]
    return J

        A_rows.append(row)
        b_rows.append(0.0)
    
    row = np.zeros(nvars)
    row[n:2*n] = 1.0
    A_rows.append(row)
    b_rows.append(rho)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = []
    for i in range(n):
        bounds.append((-h[i], 1.0 - h[i]))
    for i in range(n):
        bounds.append((0.0, None))
    bounds.append((0.0, None))
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            return res.x[:n]
    except Exception:
        pass
    return None
