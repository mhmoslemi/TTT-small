def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-12:

            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:

                y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            m1 = np.arange(lag, n)
            g[m1] += 1.0 - h[m1 - lag]
            m2 = np.arange(0, n - lag)
            g[m2] -= h[m2 + lag]
    else:

        if l < n:

            g[m1] += 1.0 - h[m1 + l]
            m2 = np.arange(l, n)
            g[m2] -= h[m2 - l]
    return g

    h_new = np.interp(x_new, x_old, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def multi_start_lp(h0, deadline, n_starts=5, radius_base=0.08):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    for s in range(n_starts):
        if time.time() > deadline:
            break
        if s == 0:
            h_s = h0.copy()
        else:
            scale = 0.02 * s
            h_s = project_sum_box(h0 + np.random.randn(n) * scale, n / 2.0)
        dl_s = min(deadline, time.time() + (deadline - time.time()) / (n_starts - s))
        h_r, c_r = cutting_plane_lp(h_s, dl_s, max_iters=400, init_radius=radius_base)
        if c_r < best_c:
            best_h = h_r
            best_c = c_r
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 6.0
    deadline = start_time + budget_s - safety

    # Get initial values
    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)
        h_init = project_sum_box(h_init, 42.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n_best = len(best_h)

    # Phase 1: Cutting-plane LP on initial resolution (most impactful)
    if time.time() < deadline - 60:
        dl1 = min(deadline, start_time + budget_s * 0.25)
        h1, c1 = cutting_plane_lp(best_h, dl1, max_iters=1000, init_radius=0.06)
        if c1 < best_c:
            best_h, best_c = h1, c1

    # Phase 2: Nesterov smoothed gradient
    if time.time() < deadline - 40:
        dl2 = min(deadline, start_time + budget_s * 0.40)
        h2, c2 = nesterov_smoothed(best_h, dl2, max_iters=8000)
        if c2 < best_c:
            best_h, best_c = h2, c2

    # Phase 3: SLSQP refinement with increasing beta
    if time.time() < deadline - 60:
        dl3 = min(deadline, start_time + budget_s * 0.60)
        h3, c3 = slsqp_refine(best_h, dl3, max_betas=8)
        if c3 < best_c:
            best_h, best_c = h3, c3

    # Phase 4: Multi-start LP to escape local optima
    if time.time() < deadline - 60:
        dl4 = min(deadline, start_time + budget_s * 0.75)
        h4, c4 = multi_start_lp(best_h, dl4, n_starts=4, radius_base=0.05)
        if c4 < best_c:
            best_h, best_c = h4, c4

    # Phase 5: Resolution transfer - try higher resolution
    if time.time() < deadline - 80:
        n_cur = len(best_h)
        for n_t in [int(n_cur * 1.5), int(n_cur * 2.0)]:
            if n_t > 300 or time.time() > deadline - 60:
                break
            try:
                h_tr = transfer_resolution(best_h, n_t)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c:
                    best_h, best_c = h_tr, c_tr
                dl_sub = min(deadline - 20, time.time() + 120)
                h_r, c_r = cutting_plane_lp(best_h, dl_sub, max_iters=300, init_radius=0.04)
                if c_r < best_c:
                    best_h, best_c = h_r, c_r
                if time.time() < deadline - 40:
                    h_r2, c_r2 = nesterov_smoothed(best_h, min(deadline - 20, time.time() + 60), max_iters=3000)
                    if c_r2 < best_c:
                        best_h, best_c = h_r2, c_r2
            except Exception:
                pass

    # Phase 6: Final polish
    if time.time() < deadline - 5:
        h6, c6 = nesterov_smoothed(best_h, deadline - 1, max_iters=10000)
        if c6 < best_c:
            best_h, best_c = h6, c6

    # Final projection and evaluation
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
