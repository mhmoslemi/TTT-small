import numpy as np

def _fix_sum(x, target):
    x = np.asarray(x, dtype=float).copy()
    n = x.size
    if n == 0:
        return x
    lo, hi = -np.max(x) - 0.5, 1.0 - np.min(x) + 0.5
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            cand = np.where(x < 1.0)[0]
            if cand.size:
                i = cand[np.argmin(x[cand])]
                x[i] = min(x[i] + diff, 1.0)
        else:
            cand = np.where(x > 0.0)[0]
            if cand.size:
                i = cand[np.argmax(x[cand])]
                x[i] = max(x[i] + diff, 0.0)
    return np.clip(x, 0.0, 1.0)


def _score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))


def _corr_scaled(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, 'full') * dx


def _compute_grad(h, alpha, dx):
    n = len(h)
    g = np.zeros(n)
    ctr = n - 1
    for k in range(len(alpha)):
        ak = alpha[k]
        if ak <= 1e-15:
            continue
        d = k - ctr
        if d == 0:
            g += ak * dx * (1.0 - 2.0 * h)
        elif d > 0:
            g[:n - d] += ak * dx * (1.0 - h[d:])
            g[d:] += -ak * dx * h[:n - d]
        else:
            dd = -d
            g[dd:] += ak * dx * (1.0 - h[:n - dd])
            g[:n - dd] += -ak * dx * h[dd:]
    return g


def _optimize_lse(h0, deadline, target, rng, tau_init=0.08, tau_min=0.002, momentum=0.85):
    n = len(h0)
    dx = 2.0 / float(n)
    h = h0.copy()
    best_h = h.copy()
    best_c = _score(h)
    vel = np.zeros(n)
    tau = tau_init
    step = 0.4
    stall = 0
    it = 0

    while time.time() < deadline - 2.0:
        it += 1
        C = _corr_scaled(h)
        Cmax = C.max()
        shifted = (C - Cmax) / tau
        shifted -= shifted.max()
        exp_v = np.exp(shifted)
        alpha = exp_v / exp_v.sum()

        g = _compute_grad(h, alpha, dx)
        g = g - g.mean()
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        vel = momentum * vel - step * (g / gnorm)
        h_new = _fix_sum(h + vel, target)
        c_new = _score(h_new)

        if c_new < best_c - 1e-12:
            h = h_new
            vel = vel * 0.5
            best_c = c_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall % 5 == 0:
                step *= 0.7
            if step < 0.005:
                h = best_h.copy()
                vel = np.zeros(n)

                best_c = _score(h)
                stall = 0

def _binary_init(h, target):
    n = len(h)
    cnt = int(round(target))
    if cnt <= 0 or cnt >= n:
        return None
    order = np.argsort(h, kind='stable')[::-1][:cnt]
    b = np.zeros(n, dtype=float)
    b[order] = 1.0
    return b


def _binary_swap_eval(h, corr, i, j, n):
    """Evaluate change in max when swapping h[j]=1->0, h[i]=0->1."""
    dx = 2.0 / float(n)
    ctr = n - 1
    best_old = float(corr.max()) * dx
    diff = np.zeros(2 * n - 1, dtype=float)
    for pos in (i, j):
        sign = 1.0 if pos == i else -1.0
        for d in range(-n + 1, n):
            if d == 0:
                k = ctr
                if 0 <= k < 2 * n - 1:
                    diff[k] += sign * (1.0 - 2.0 * h[pos])
            elif d > 0:
                if 0 <= pos + d < n:
                    k = ctr + d
                    if 0 <= k < 2 * n - 1:
                        diff[k] += sign * (1.0 - h[pos + d])
                if 0 <= pos - d < n:
                    k = ctr - d
                    if 0 <= k < 2 * n - 1:
                        diff[k] += -sign * h[pos - d]
            else:
                dd = -d
                if 0 <= pos - dd < n:
                    k = ctr + d
                    if 0 <= k < 2 * n - 1:
                        diff[k] += sign * (1.0 - h[pos - dd])
                if 0 <= pos + dd < n:
                    k = ctr - d
                    if 0 <= k < 2 * n - 1:
                        diff[k] += -sign * h[pos + dd]
    new_max = float((corr + diff * dx).max())
    return new_max, best_old


def _binary_local_search(h_init, deadline, target, rng, max_iter=50000):
    n = len(h_init)
    dx = 2.0 / float(n)
    b = np.zeros(n, dtype=float)
    order = np.argsort(h_init, kind='stable')[::-1]
    b[order[:int(round(target))]] = 1.0
    corr = np.correlate(b, 1.0 - b, 'full') * dx
    best_c = float(corr.max())
    best_b = b.copy()

    ones = set(np.where(b == 1.0)[0].tolist())
    zeros = set(np.where(b == 0.0)[0].tolist())
    T = 0.01
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline - 3.0:
            break
        if no_improve > 8000:
            T *= 0.9
            if T < 1e-5:
                break
            no_improve = 0

        i = rng.choice(list(zeros))
        j = rng.choice(list(ones))
        i, j = int(i), int(j)

        b[i] = 1.0
        b[j] = 0.0
        new_corr = np.correlate(b, 1.0 - b, 'full') * dx
        new_c = float(new_corr.max())
        delta = new_c - best_c

        if delta < -1e-12 or (rng.random() < np.exp(-max(delta, 0) / T)):
            corr = new_corr
            best_c_local = new_c
            if new_c < best_c - 1e-12:
                best_c = new_c
                best_b = b.copy()
                no_improve = 0
            else:
                no_improve += 1
            ones.discard(j)
            zeros.discard(i)
            ones.add(i)
            zeros.add(j)
        else:
            b[i] = 0.0
            b[j] = 1.0
            no_improve += 1

    return best_b, best_c


def _antipodal_project(h):
    n = len(h)
    if n % 2 != 0:
        return h.copy()
    m = n // 2
    u = (1.0 + h[:m] - h[m:]) / 2.0
    return np.concatenate([u, 1.0 - u[::-1]])


def _resample(h, N):
    n0 = len(h)
    target = 0.5 * float(N)
    if n0 == N:
        return _fix_sum(h, target)
    xold = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))
    y = np.interp(xnew, xold, h)
    return _fix_sum(y, target)

                best_h = h_pol
                best_c = c_pol

    # Phase 4: Try smaller resolutions for potentially better discrete solutions
    if time.time() < hard - 30.0:
        for Nt in (best_n // 2, best_n // 3, 200, 240):
            if Nt < 40 or Nt % 2 != 0:
                continue
            if time.time() > hard - 20.0:
                break
            h_small = _resample(best_h, Nt)
            h_small_ant = _antipodal_project(h_small)
            h_small_ant = _fix_sum(h_small_ant, 0.5 * float(Nt))
            if _score(h_small_ant) < _score(h_small):
                h_small = h_small_ant
            # Quick binary search at this resolution
            h_sb, c_sb = _binary_local_search(h_small, min(hard - 15.0, time.time() + 40.0), 0.5 * float(Nt), rng, max_iter=30000)
            if np.isfinite(c_sb) and c_sb < best_c - 1e-10:
                best_h = h_sb
                best_c = c_sb
                best_n = Nt
                # Polish
                if time.time() < hard - 15.0:
                    h_pol2, c_pol2 = _optimize_lse(best_h, min(hard - 8.0, time.time() + 60.0), 0.5 * float(Nt), rng, tau_init=0.02, tau_min=0.002)
                    if np.isfinite(c_pol2) and c_pol2 < best_c - 1e-10:
                        best_h = h_pol2
                        best_c = c_pol2

    # Final polish
    if time.time() < hard - 5.0:
        h_final, c_final = _optimize_lse(best_h, hard - 3.0, 0.5 * float(best_n), rng, tau_init=0.01, tau_min=0.001)
        if np.isfinite(c_final) and c_final < best_c - 1e-10:
            best_h = h_final
            best_c = c_final

    best_n = int(best_h.size)
    best_h = _fix_sum(best_h, 0.5 * float(best_n))
    best_c = _score(best_h)

    if not np.isfinite(best_c):
        best_h = _fix_sum(np.full(best_n, 0.5, dtype=float), 0.5 * float(best_n))
        best_c = _score(best_h)

    return best_h, float(best_c), best_n
