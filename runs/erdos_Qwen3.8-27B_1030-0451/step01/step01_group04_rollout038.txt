    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            for i in idx:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-14:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            for i in idx:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if abs(residual) < 1e-14:
                    break
    return h


def slsqp_f(h, tau):
    obj, _ = smooth_obj_grad(h, tau)
    return obj


def slsqp_g(h, tau):
    _, g = smooth_obj_grad(h, tau)
    return g


def sum_eq_fun(h, target_sum):
    return np.sum(h) - target_sum


def sum_eq_jac(h, target_sum):
    return np.ones_like(h)


def nesterov_pg(h_init, tau, target_sum, max_iter, lr_init, deadline):
    N = len(h_init)
    h = h_init.copy()
    y = h_init.copy()
    v = np.zeros(N)
    best_h = h.copy()
    best_exact = compute_c5(h_init)
    curr_surr = smooth_obj_grad(h, tau)[0]
    lr = lr_init
    mu = 0.5
    stall = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        y = y + mu * (y - h)
        _, g = smooth_obj_grad(np.clip(y, 0.0, 1.0), tau)
        v = 0.85 * v + lr * g
        h_new = project_feasible(y - v, target_sum)
        obj_new, _ = smooth_obj_grad(h_new, tau)
        if obj_new > curr_surr + 1e-8:
            lr *= 0.5
            h_new = project_feasible(y - 0.5 * v, target_sum)
            obj_new, _ = smooth_obj_grad(h_new, tau)
        if obj_new < curr_surr:
            h = h_new
            curr_surr = obj_new
            lr = min(lr * 1.05, lr_init)
            stall = 0
        else:
            stall += 1
            if stall > 20:
                break
        exact = compute_c5(h)
        if exact < best_exact:
            best_exact = exact
            best_h = h.copy()
        mu = min(0.95, 0.5 + 0.45 * it / max(10, max_iter))
    return best_h, best_exact


def exact_line_search(h, d, target_sum, deadline):
    N = len(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lo, hi = -5.0, 5.0
    d_norm = np.sqrt(np.sum(d * d))
    if d_norm < 1e-12:
        return best_h, best_c5
    d = d / d_norm
    for _ in range(40):
        if time.time() > deadline:
            break
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        h1 = project_feasible(h + m1 * d, target_sum)
        h2 = project_feasible(h + m2 * d, target_sum)
        c1 = compute_c5(h1)
        c2 = compute_c5(h2)
        if c1 < best_c5:
            best_c5 = c1
            best_h = h1.copy()
        if c2 < best_c5:
            best_c5 = c2
            best_h = h2.copy()
        if c1 < c2:
            hi = m2
        else:
            lo = m1
    return best_h, best_c5


def fourier_perturb(h, rng, n_modes, amp):
    N = len(h)
    h_p = h.copy()
    for k in range(1, n_modes + 1):
        freq = 2.0 * np.pi * k / N
        a = rng.normal(0, amp)
        b = rng.normal(0, amp)
        h_p += a * np.sin(freq * np.arange(N)) + b * np.cos(freq * np.arange(N))
    return project_feasible(h_p, N / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 12.0

    h_init = np.array(initial_h_values, dtype=float)
    N_init = len(h_init)
    c5_init = compute_c5(h_init)

    best_h = h_init.copy()
    best_c5 = c5_init
    best_N = N_init

    rng = np.random.default_rng(seed)
    target_sum = N_init / 2.0

    bounds = [(0.0, 1.0)] * N_init
    constraint = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac,
                   'args': (target_sum,)}]

    # Phase 1: SLSQP at initial resolution with decreasing tau
    h = h_init.copy()
    c5_cur = c5_init
    tau_sched = [0.04, 0.02, 0.01, 0.005, 0.0025, 0.0012, 0.0006, 0.0003]

    for i, tau in enumerate(tau_sched):
        if time.time() > deadline - 150:
            break
        mx = 250 if N_init <= 400 else 120
        try:
            res = minimize(slsqp_f, h.copy(), jac=slsqp_g, args=(tau,),
                           method='SLSQP', bounds=bounds, constraints=constraint,
                           options={'maxiter': mx, 'ftol': 1e-15})
            hc = project_feasible(res.x, target_sum)
            c5c = compute_c5(hc)
            if c5c < c5_cur:
                h = hc
                c5_cur = c5c
                if c5_cur < best_c5:
                    best_c5 = c5_cur
                    best_h = h.copy()
                    best_N = N_init
        except Exception:
            pass

    # Phase 2: Nesterov projected gradient polishing at N_init
    if time.time() < deadline - 80:
        h_work = h.copy()
        for tau in [0.0004, 0.0002, 0.0001]:
            if time.time() > deadline - 40:
                break
            for lr0 in [0.008, 0.004, 0.002]:
                if time.time() > deadline - 25:
                    break
                h_pg, c5_pg = nesterov_pg(h_work, tau, target_sum,
                                          max_iter=300, lr_init=lr0,
                                          deadline=deadline - 20)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_N = N_init
                    h_work = h_pg.copy()

    # Phase 3: Exact line search from best solution
    if time.time() < deadline - 40:
        ts = best_N / 2.0
        theta = np.correlate(best_h, 1.0 - best_h, mode='full') * (2.0 / best_N)
        r_star = int(np.argmax(theta))
        d_shift = r_star - (best_N - 1)
        d = np.zeros(best_N)
        if d_shift >= 0:
            n = best_N - d_shift
            if n > 0:
                d[d_shift:] += (1.0 - best_h[:n])
                d[:n] -= best_h[d_shift:]
        else:
            dd = -d_shift
            n = best_N - dd
            if n > 0:
                d[:n] += (1.0 - best_h[dd:])
                d[dd:] -= best_h[:n]
        d -= np.mean(d)
        h_ls, c5_ls = exact_line_search(best_h, d, ts, deadline - 15)
        if c5_ls < best_c5:
            best_c5 = c5_ls
            best_h = h_ls.copy()
            best_N = best_N

    # Phase 4: Upscale to higher resolution and optimize
    for N_new in [500, 700]:
        if time.time() > deadline - 60:
            break
        x_old = (np.arange(best_N) + 0.5) * (2.0 / best_N)
        x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
        h_up = np.interp(x_new, x_old, best_h)
        h_up = project_feasible(h_up, N_new / 2.0)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            best_N = N_new

        ts_new = N_new / 2.0
        if N_new <= 500 and time.time() < deadline - 50:
            bounds_new = [(0.0, 1.0)] * N_new
            cons_new = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac,
                         'args': (ts_new,)}]
            for tau in [0.008, 0.004, 0.002, 0.001]:
                if time.time() > deadline - 30:
                    break
                try:
                    res = minimize(slsqp_f, h_up.copy(), jac=slsqp_g, args=(tau,),
                                   method='SLSQP', bounds=bounds_new, constraints=cons_new,
                                   options={'maxiter': 150, 'ftol': 1e-15})
                    hc = project_feasible(res.x, ts_new)
                    c5c = compute_c5(hc)
                    if c5c < c5_up:
                        h_up = hc
                        c5_up = c5c
                    if c5_up < best_c5:
                        best_c5 = c5_up
                        best_h = h_up.copy()
                        best_N = N_new
                except Exception:
                    pass
        else:
            # Projected gradient for larger N
            for tau in [0.005, 0.002, 0.001]:
                if time.time() > deadline - 25:
                    break
                h_pg, c5_pg = nesterov_pg(h_up, tau, ts_new,
                                          max_iter=400, lr_init=0.004,
                                          deadline=deadline - 15)
                if c5_pg < c5_up:
                    h_up = h_pg
                    c5_up = c5_pg
                if c5_up < best_c5:
                    best_c5 = c5_up
                    best_h = h_up.copy()
                    best_N = N_new

    # Phase 5: Multi-start with diverse perturbations
    if time.time() < deadline - 25:
        N_p = best_N
        ts_p = N_p / 2.0
        for attempt in range(8):
            if time.time() > deadline - 12:
                break
            amp = 0.04 * (1.0 + 0.5 * attempt)
            h_pert = fourier_perturb(best_h, rng, 3 + attempt, amp)
            c5_pert = compute_c5(h_pert)

            if N_p <= 500:
                bounds_p = [(0.0, 1.0)] * N_p
                cons_p = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac,
                           'args': (ts_p,)}]
                try:
                    res = minimize(slsqp_f, h_pert.copy(), jac=slsqp_g, args=(0.004,),
                                   method='SLSQP', bounds=bounds_p, constraints=cons_p,
                                   options={'maxiter': 120, 'ftol': 1e-15})
                    hc = project_feasible(res.x, ts_p)
                    c5c = compute_c5(hc)
                    if c5c < best_c5:
                        best_c5 = c5c
                        best_h = hc.copy()
                        best_N = N_p
                except Exception:
                    pass
            else:
                h_pg, c5_pg = nesterov_pg(h_pert, 0.003, ts_p,
                                          max_iter=200, lr_init=0.003,
                                          deadline=deadline - 10)
                if c5_pg < best_c5:
                    best_c5 = c5_pg
                    best_h = h_pg.copy()
                    best_N = N_p

    # Final: one more SLSQP polish on the best
    remaining = deadline - time.time()
    if remaining > 20:
        N_fin = len(best_h)
        ts_fin = N_fin / 2.0
        bounds_fin = [(0.0, 1.0)] * N_fin
        cons_fin = [{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac,
                     'args': (ts_fin,)}]
        try:
            res = minimize(slsqp_f, best_h.copy(), jac=slsqp_g, args=(0.0002,),
                           method='SLSQP', bounds=bounds_fin, constraints=cons_fin,
                           options={'maxiter': 200, 'ftol': 1e-15})
            hc = project_feasible(res.x, ts_fin)
            c5c = compute_c5(hc)
            if c5c < best_c5:
                best_c5 = c5c
                best_h = hc
                best_N = N_fin
        except Exception:
            pass

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
