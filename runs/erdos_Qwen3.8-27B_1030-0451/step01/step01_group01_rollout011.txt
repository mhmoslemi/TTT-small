def grad_lag(h, lag):
    """Gradient of correlation at given lag."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m1 = np.arange(lag, n)
        g[m1] += 1.0 - h[m1 - lag]
        m2 = np.arange(0, n - lag)
        g[m2] -= h[m2 + lag]
    else:
        l = -lag
        m1 = np.arange(0, n - l)
        g[m1] += 1.0 - h[m1 + l]
        m2 = np.arange(l, n)
        g[m2] -= h[m2 - l]
    return g

        b_eq = np.array([n / 2.0])

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-13:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                if it % 10 == 0:
                    radius = min(radius * 1.1, 0.15)
            else:
                no_improve += 1
                if no_improve >= 6:
                    radius = max(radius * 0.65, 0.0005)
                    no_improve = 0
                    if radius <= 0.001:
                        break
        else:
            no_improve += 1
            if no_improve >= 8:
                radius = max(radius * 0.5, 0.0005)
                no_improve = 0
                if radius <= 0.001:
                    break
    return best_h, best_c

def method_subgrad_avg(h0, deadline, max_iters=5000, target=None):
    """Subgradient with Polyak/averaging for the non-smooth max."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    if target is None:
        target = best_c * 0.999
    h_avg = h0.copy()
    t = 0
    step_scale = 0.5
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t += 1
        v = corr_all(h_cur) if 'h_cur' in dir() else corr_all(h_avg)
        h_cur = h_avg if it == 0 else h_cur
        k_star = np.argmax(v)
        lag = k_star - (n - 1)
        g = grad_lag(h_cur, lag)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        # Polyak-like step
        gap = max(v[k_star] - target, 1e-6)
        step = step_scale * gap / (g_norm * g_norm)
        step = min(step, 0.1)
        h_new = h_cur - step * g
        h_new = project_sum_box(h_new, n / 2.0)
        h_avg = (t * h_avg + h_new) / (t + 1)
        c_avg = evaluate_c5(h_avg)
        c_new = evaluate_c5(h_new)
        if min(c_avg, c_new) < best_c - 1e-13:
            if c_avg <= c_new:
                best_h = h_avg.copy()
                best_c = c_avg
            else:
                best_h = h_new.copy()
                best_c = c_new
    return best_h, best_c

def method_slsqp_smooth(h0, deadline, max_betas=10):
    """SLSQP with log-sum-exp, carefully scheduled betas."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096][:max_betas]
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

    for beta in betas:
        if time.time() > deadline:
            break
        for attempt in range(3):
            if time.time() > deadline:
                break
            if attempt == 0:
                x0 = best_h.copy()
            else:
                noise_scale = 0.003 / np.sqrt(attempt)
                x0 = best_h + np.random.randn(n) * noise_scale
                x0 = project_sum_box(x0, n / 2.0)

            def obj_func(x):
                return smooth_obj_grad(x, beta)[0]

            def grad_func(x):
                return smooth_obj_grad(x, beta)[1]

            try:
                res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-13:
                    best_h = x_new.copy()
                    best_c = c_new
    return best_h, best_c

def method_projo_grad_ada(h0, deadline, max_iters=10000):
    """Adaptive projected gradient on smoothed objective with backtracking."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    beta = 32.0
    step = 0.01
    stall = 0
    momentum = 0.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        f_cur, g = smooth_obj_grad(h, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        # Backtracking line search
        found = False
        s = step
        for bs in range(10):
            h_trial = h - s * g / g_norm
            h_trial = project_sum_box(h_trial, n / 2.0)
            f_trial, _ = smooth_obj_grad(h_trial, beta)
            if f_trial < f_cur - 1e-10:
                found = True
                break
            s *= 0.5
        if not found:
            break
        momentum = 0.9 * momentum + 0.1 * (h_trial - h)
        h = project_sum_box(h_trial + momentum, n / 2.0)
        c_cur = evaluate_c5(h)
        if c_cur < best_c - 1e-13:
            best_h = h.copy()
            best_c = c_cur
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
    return best_h, best_c

def transfer_resolution(h0, n_new):
    """Interpolate h0 to a grid of n_new points using FFT-based resampling."""
    n_old = len(h0)
    if n_new == n_old:
        return h0.copy()
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h0, period=n_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0
    deadline = start_time + budget_s - safety

    # Get initial values
    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)
        h_init = project_sum_box(h_init, 42.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n_best = len(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: LP trust region with small radius (main workhorse)
    if time_left() > 60:
        dl1 = min(deadline - 30, start_time + budget_s * 0.25)
        try:
            h1, c1 = method_lp_trust(best_h, dl1, max_iters=2000, init_radius=0.03)
            if c1 < best_c:
                best_h, best_c = h1, c1
        except Exception:
            pass

    # Phase 2: SLSQP smooth refinement
    if time_left() > 40:
        dl2 = min(deadline - 30, start_time + budget_s * 0.40)
        try:
            h2, c2 = method_slsqp_smooth(best_h, dl2, max_betas=10)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # Phase 3: LP trust region again (from improved point, even smaller radius)
    if time_left() > 40:
        dl3 = min(deadline - 20, start_time + budget_s * 0.55)
        try:
            h3, c3 = method_lp_trust(best_h, dl3, max_iters=2000, init_radius=0.015)
            if c3 < best_c:
                best_h, best_c = h3, c3
        except Exception:
            pass

    # Phase 4: Subgradient with averaging (directly targets max lag)
    if time_left() > 30:
        dl4 = min(deadline - 15, start_time + budget_s * 0.65)
        try:
            h4, c4 = method_subgrad_avg(best_h, dl4, max_iters=5000, target=best_c * 0.9995)
            if c4 < best_c:
                best_h, best_c = h4, c4
        except Exception:
            pass

    # Phase 5: Another LP round
    if time_left() > 30:
        dl5 = min(deadline - 15, start_time + budget_s * 0.75)
        try:
            h5, c5 = method_lp_trust(best_h, dl5, max_iters=2000, init_radius=0.01)
            if c5 < best_c:
                best_h, best_c = h5, c5
        except Exception:
            pass

    # Phase 6: Try higher resolution
    if time_left() > 80:
        n_cur = len(best_h)
        candidates = []
        for f in [1.4, 1.8, 2.2]:
            nt = int(round(n_cur * f))
            if nt > n_cur and nt <= 300:
                candidates.append(nt)
        candidates = sorted(set(candidates))
        for n_t in candidates:
            if time_left() < 50:
                break
            try:
                h_tr = transfer_resolution(best_h, n_t)
                c_tr = evaluate_c5(h_tr)
                work_h = h_tr.copy() if c_tr < best_c else best_h.copy()
                # Optimize on new resolution
                sub_dl = min(deadline - 10, time.time() + min(200, time_left() * 0.5))
                h_a, c_a = method_lp_trust(work_h, sub_dl, max_iters=800, init_radius=0.02)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a
                sub_dl2 = min(deadline - 10, time.time() + min(60, time_left() * 0.3))
                h_b, c_b = method_slsqp_smooth(best_h, sub_dl2, max_betas=4)
                if c_b < best_c:
                    best_h, best_c = h_b, c_b
            except Exception:
                pass

    # Phase 7: Final polish with LP (very small radius)
    if time_left() > 15:
        try:
            h7, c7 = method_lp_trust(best_h, deadline - 2, max_iters=1500, init_radius=0.005)
            if c7 < best_c:
                best_h, best_c = h7, c7
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    n_best = len(best_h)
    if n_best > 2500:
        # Downsample if too large
        best_h = transfer_resolution(best_h, 2500)
        best_c = evaluate_c5(best_h)
        n_best = 2500

    return best_h, float(best_c), n_best
