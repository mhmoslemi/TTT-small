def make_binary(n, rng):
    """Create a random binary solution with sum = n/2 (or as close as possible)."""
    # For even n: n/2 ones
    # For odd n: (n-1)/2 ones and one 0.5, rest zeros

        ones = rng.choice(n, n // 2, replace=False)

        choose = rng.choice(n, (n + 1) // 2, replace=False)
