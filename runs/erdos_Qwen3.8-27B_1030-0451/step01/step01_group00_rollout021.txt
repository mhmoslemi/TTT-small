import time
from scipy.optimize import linprog
from scipy.optimize import minimize as sci_minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def corr_gradient(h):
    """Compute gradient matrix G[k, i] = d(corr[k])/d(h[i])"""
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        lag = k - (n - 1)
        if lag >= 0:
            for j in range(lag, n):
                G[k, j] += 1.0 - h[j - lag]
                G[k, j - lag] += -h[j]
        else:
            L = -lag
            for j in range(0, n - L):
                G[k, j] += 1.0 - h[j + L]
                G[k, j + L] += -h[j]
    return G

def smooth_gradient(h, corr, tau):
    n = len(h)
    cmax = np.max(corr)
    cs = corr - cmax
    # Clip to avoid overflow
    cs = np.clip(cs, -100.0/tau, 0.0)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    # Weighted sum of gradients
    G = corr_gradient(h)
    grad = G.T @ w  # (n,)
    return grad

def lp_step(h, R, n_active=40):
    """Try one LP-based step. Returns (new_h, improved)"""
    n = len(h)
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    
    # Select active set
    active_idx = np.argsort(-corr)[:n_active]
    
    # Build gradient matrix for active rows
    G = corr_gradient(h)
    G_act = G[active_idx, :]  # (n_active, n)
    c_act = corr[active_idx]  # (n_active,)

                        best_c5 = c5t
                        best_h = h.copy()
                        any_improved = True
                    else:
                        h[i] -= s
                        h[j] += s
                if s_neg > 1e-7 and not any_improved:
                    s = min(step, s_neg * 0.7)
                    h[i] -= s
                    h[j] += s
                    c5t = compute_c5(h)
                    if c5t < best_c5 - 1e-14:
                        best_c5 = c5t
                        best_h = h.copy()
                        any_improved = True
                    else:
                        h[i] += s
                        h[j] -= s
        if any_improved:
            step = min(step * 1.08, 0.07)
        else:
            step = max(step * 0.65, 0.004)
            if step <= 0.0045:
                break
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return np.interp(x_new, x_old, h_old)

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.20 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.05 * np.sin(6.0 * np.pi * x + 1.3)
    h += 0.03 * np.sin(10.0 * np.pi * x + 2.5)
    h += rng.randn(n) * 0.012
    return project(h, n)

def coord_descent_focused(h0, time_limit, deadline):
    """Coordinate descent that focuses on pairs most likely to help."""
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    step = 0.02
    while time.time() < deadline - 5:
        corr = compute_corr(h)
        cmax = np.max(corr)
        G = corr_gradient(h)
        active = np.where(corr >= cmax - 2.0 / n * 0.05)[0]
        any_improved = False
        # Try moving mass away from values that increase active correlations
        grad_stuff = G[active].sum(axis=0) if len(active) > 0 else np.zeros(n)
        # Sort indices by "pressure" - positive grad stuff means increasing h[i] helps
        order = np.argsort(-grad_stuff)
        for idx in range(n):
            i = int(order[idx])
            if time.time() > deadline - 8:
                break
            if grad_stuff[i] > 0:
                # Want to increase h[i], so decrease some h[j]
                j_candidates = np.argsort(-h)
            else:
                j_candidates = np.argsort(h)
            for jj in range(min(20, n - 1)):
                j = int(j_candidates[jj])
                if j == i:
                    continue
                if grad_stuff[i] > 0:
                    s_pos = min(1.0 - h[i], h[j])
                    if s_pos > 1e-7:
                        s = min(step, s_pos * 0.6)
                        h[i] += s
                        h[j] -= s
                        c5t = compute_c5(h)
                        if c5t < best_c5 - 1e-14:
                            best_c5 = c5t
                            best_h = h.copy()
                            any_improved = True
                        else:
                            h[i] -= s
                            h[j] += s
                else:
                    s_neg = min(1.0 - h[j], h[i])
                    if s_neg > 1e-7:
                        s = min(step, s_neg * 0.6)
                        h[i] -= s
                        h[j] += s
                        c5t = compute_c5(h)
                        if c5t < best_c5 - 1e-14:
                            best_c5 = c5t
                            best_h = h.copy()
                            any_improved = True
                        else:
                            h[i] += s
                            h[j] -= s
        if any_improved:
            step = min(step * 1.1, 0.06)
        else:
            step = max(step * 0.6, 0.005)
            if step <= 0.0055:
                # Fall back to random coordinates
                rng = np.random.RandomState(int(time.time() * 999) % 1000000)
                for _ in range(200):
                    if time.time() > deadline - 8:
                        break
                    i, j = rng.choice(n, 2, replace=False)
                    s_pos = min(1.0 - h[i], h[j])
                    if s_pos > 1e-7:
                        s = min(0.01, s_pos * 0.5)
                        h[i] += s
                        h[j] -= s
                        c5t = compute_c5(h)
                        if c5t < best_c5 - 1e-14:
                            best_c5 = c5t
                            best_h = h.copy()
                            any_improved = True
                        else:
                            h[i] -= s
                            h[j] += s
                if not any_improved:
                    break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s
    np.random.seed(seed)

    n0 = 81
    try:
        h = np.array(initial_h_values, dtype=float)
        n0 = len(h)
    except (NameError, UnboundLocalError):
        h = make_start(n0, seed)

    h = project(h, n0)
    best_h = h.copy()
    best_c5 = compute_c5(h)

    # Phase 1: Smooth gradient descent
    alloc = min(140, max(30, budget_s * 0.14))
    if time.time() < deadline - alloc - 30:
        ho, co = smooth_descent(best_h.copy(), alloc, deadline)
        if co < best_c5:
            best_h, best_c5 = ho, co

    # Phase 2: LP refinement
    alloc = min(80, max(20, budget_s * 0.07))
    if time.time() < deadline - alloc - 20:
        ho, co = lp_refine(best_h.copy(), alloc, deadline)
        if co < best_c5:
            best_h, best_c5 = ho, co

    # Phase 3: Coordinate descent (full)
    alloc = min(200, max(30, budget_s * 0.20))
    if time.time() < deadline - alloc - 20:
        ho, co = coord_descent(best_h.copy(), alloc, deadline)
        if co < best_c5:
            best_h, best_c5 = ho, co

    # Phase 4: Focused coordinate descent
    alloc = min(150, max(20, budget_s * 0.14))
    if time.time() < deadline - alloc - 20:
        ho, co = coord_descent_focused(best_h.copy(), alloc, deadline)
        if co < best_c5:
            best_h, best_c5 = ho, co

    # Phase 5: Multi-resolution
    for n_try in [121, 161]:
        remaining = deadline - time.time() - 100
        if remaining < 60:
            break
        h_new = resample_h(best_h, n_try)
        h_new = project(h_new, n_try)
        c5_new = compute_c5(h_new)
        # Smooth at new resolution
        a_s = min(remaining * 0.3, 45)
        if a_s > 15:
            h_new, c5_new = smooth_descent(h_new, a_s, deadline)
        remaining = deadline - time.time() - 80
        a_c = min(remaining * 0.5, 40)
        if a_c > 10:
            h_new, c5_new = coord_descent(h_new, a_c, deadline)
        if c5_new < best_c5:
            best_h, best_c5 = h_new, c5_new

    # Phase 6: Final polish
    remaining = deadline - time.time() - 5
    if remaining > 10:
        ho, co = coord_descent(best_h.copy(), remaining, deadline)
        if co < best_c5:
            best_h, best_c5 = ho, co

    # Final verification
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
