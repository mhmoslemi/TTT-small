    """
    Optimize h[i] and h[j] (with h[i]+h[j]=const) to minimize C.
    Returns new h[i], new h[j], and improvement.
    """
    s = h[i] + h[j]
    t_lo = max(0.0, s - 1.0)
    t_hi = min(1.0, s)
    
    if t_hi - t_lo < 1e-12:
        return h[i], h[j]
    
    def c_at_t(t):
        delta = compute_delta(i, j, t, h, n)
        c_vals = corr_base + delta * dx
        return float(np.max(c_vals))
    
    # Ternary search
    for _ in range(max_evals // 3):
        m1 = t_lo + (t_hi - t_lo) / 3.0
        m2 = t_hi - (t_hi - t_lo) / 3.0
        c1 = c_at_t(m1)
        c2 = c_at_t(m2)
        if c1 < c2:
            t_hi = m2
        else:
            t_lo = m1
    
    t_opt = 0.5 * (t_lo + t_hi)
    return t_opt, s - t_opt

def make_feasible(x, target):
    """Project x onto {0<=h_i<=1, sum(h)=target}."""
    n = len(x)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    # Binary search for offset
    lo = -1.0 - x.max()
    hi = 1.0 - x.min()
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(x+mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5*(lo+hi), 0, 1)
    # Fix remaining error
    err = target - x.sum()
    if abs(err) > 1e-13:
        if err > 0:
            idx = np.where(x < 1.0)[0]
            if len(idx) > 0:
                order = idx[np.argsort(x[idx])]
                rem = err
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
                    rem -= inc
                    if rem <= 1e-14:
                        break
        else:
            idx = np.where(x > 0.0)[0]
            if len(idx) > 0:
                order = idx[np.argsort(x[idx])[::-1]]
                rem = -err
                for i in order:
                    dec = min(x[i], rem)
                    x[i] -= dec
                    rem -= dec
                    if rem <= 1e-14:
                        break
    x = np.clip(x, 0, 1)
    return x
