def evaluate_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=1.0, neginf=0.0)
    
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = mask.sum()
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return c

def compute_active_gradient(h, active_lags, weights):

    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for idx, w in zip(active_lags, weights):
        if abs(w) < 1e-15:
            continue
        r = idx - (n - 1)
        if r >= 0:
            # h[i] * (1-h[i+r]) for i = 0..n-1-r

                #   if j-r >= 0: contribution from h[j-r]*(1-h[j]): -h[j-r]

                grad[r:] += w * dx * (-h[:n-r])  # -h[j-r] for j=r..n-1

def optimize_active_lags(h, n, deadline, max_iters=200):
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    target = n / 2.0
    np_lags = 2 * n - 1
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        
        c = np.correlate(h, 1.0 - h, 'full') * dx
        cmax = c.max()
        cmin = c.min()
        spread = max(1e-10, cmax - c.mean())
        
        # Find active lags
        tol = max(1e-4, 0.1 * spread)
        active = np.where(c >= cmax - tol)[0]
        
        # Limit to top K for efficiency
        K = min(15, len(active))
        if len(active) > K:
            # Sort by correlation value, take top K
            active = active[np.argsort(c[active])][-K:]
        
        # Try several weightings
        for weight_scheme in ['uniform', 'softmax', 'top_biased']:
            if time.time() > deadline - 5:
                break
            
            # Compute weights
            if weight_scheme == 'uniform':
                w = np.ones(len(active)) / len(active)
            elif weight_scheme == 'softmax':
                tau = max(1e-6, spread * 0.1)
                w = np.exp((c[active] - cmax) / tau)
                w = w / w.sum()
            else:  # top_biased
                w = np.linspace(0.5, 2.0, len(active))
                w = w / w.sum()
            
            # Compute weighted gradient

            # Actually, the scaling doesn't matter for the direction, just use w directly
            grad = compute_active_gradient(h, active, w)
            
            # Project to tangent space
            d = -(grad - np.mean(grad))
            
            # Skip if direction is too small
            if np.max(np.abs(d)) < 1e-12:
                continue
            
            # Normalize
            d = d / np.max(np.abs(d))
            
            # Line search: try several step sizes
            # First find max feasible step
            pos_mask = d > 1e-10
            neg_mask = d < -1e-10
            t_max = 1.0  # since d is normalized to max abs 1

            if neg_mask.any():

            t_max = max(t_max, 1e-8)
            
            # Try geometric sequence of step sizes
            best_t = 0
            best_t_c5 = best_c5
            for alpha in [1.0, 0.5, 0.25, 0.1, 0.04, 0.01, 0.005]:

            elif best_t > 0 and best_t_c5 < best_c5:
                h = h_try_best
                # small improvement, accept but don't update best

    """Simulated annealing on binary assignments."""
    rng = np.random.default_rng(seed)
