import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}"""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -max(x) - 1, 1 - min(x) + 1
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    # Final fix
    r2 = target - y.sum()
    if r2 > 0:
        idx = np.where(y < 1.0 - 1e-12)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    elif r2 < 0:
        idx = np.where(y > 1e-12)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m1 = np.arange(lag, n)
        g[m1] += 1.0 - h[m1 - lag]
        m2 = np.arange(0, n - lag)
        g[m2] -= h[m2 + lag]
    else:
        l = -lag
        m1 = np.arange(0, n - l)
        g[m1] += 1.0 - h[m1 + l]
        m2 = np.arange(l, n)
        g[m2] -= h[m2 - l]
    return g

import cvxpy as cp

def method_qp_trust(h0, deadline, max_iters=200):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.1
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        gap = max(0.005 * vmax, 0.003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 100:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-100:]]
        
        # Build QP
        h_var = cp.Variable(n)
        t_var = cp.Variable()
        
        # Quadratic regularization
        obj = cp.Minimize(t_var + (1.0/(2*radius)) * cp.sum_squares(h_var - h))
        
        # Linearized correlation constraints
        constraints = [h_var == ... ]  # sum constraint
        constraints.append(cp.sum(h_var) == n/2)
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        for k in active_idx:
            lag = k - (n-1)
            g_k = grad_lag(h, lag)
            # Linearized: g_k @ h_new - v[k] <= t_var
            constraints.append(g_k @ h_var - v[k] <= t_var)
        
        prob = cp.Problem(obj, constraints)
        prob.solve(solver=cp.SCS, max_iters=1000)
        
        if h_var.value is not None:
            h_new = project_sum_box(h_var.value, n/2)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 1e-10:

                best_c = c_new
                h = h_new.copy()

        if h_var.value is not None:
            step_size = np.linalg.norm(h_var.value - h)
            if step_size < radius * 0.5:
                radius = min(radius * 1.5, 0.5)
            elif step_size > radius * 0.9:
                radius = max(radius * 0.7, 0.01)
    
    return best_h, best_c

def method_grad_polish(h0, deadline, max_iters=10000):
    """Adaptive gradient descent with Barzilai-Borwein step size."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    h_prev = h0.copy()
    g_prev = None
    step = 0.01
    stall = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        # Compute smoothed gradient
        v = corr_all(h)
        m = v.max()
        # Use adaptive beta
        beta = max(20.0, 20.0 * (1 + it * 0.01))
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        if Z < 1e-30:
            break
        w /= Z
        
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 1e-10:
                lag = k - (n - 1)
                g += w[k] * grad_lag(h, lag)
        
        # Barzilai-Borwein step size
        if g_prev is not None:
            s = h - h_prev
            y = g - g_prev
            s_norm = np.dot(s, s)
            y_norm = np.dot(y, y)
            if s_norm > 1e-14 and y_norm > 1e-14:
                sy = np.dot(s, y)
                if sy > 1e-14:
                    step_bb = s_norm / sy
                    step = 0.5 * step + 0.5 * np.clip(step_bb, 1e-6, 0.1)
        
        h_prev = h.copy()
        g_prev = g.copy()
        
        # Gradient step
        h_new = h - step * g
        h_new = project_sum_box(h_new, n/2.0)
        
        c_new = evaluate_c5(h_new)
        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new

import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -max(x) - 1.0, 1.0 - min(x) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    r2 = target - y.sum()
    if r2 > 0:
        idx = np.where(y < 1.0 - 1e-12)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    elif r2 < 0:
        idx = np.where(y > 1e-12)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m1 = np.arange(lag, n)
        g[m1] += 1.0 - h[m1 - lag]
        m2 = np.arange(0, n - lag)
        g[m2] -= h[m2 + lag]
    else:
        l = -lag
        m1 = np.arange(0, n - l)
        g[m1] += 1.0 - h[m1 + l]
        m2 = np.arange(l, n)
        g[m2] -= h[m2 - l]
    return g

def method_lp_trust(h0, deadline, max_iters=3000):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.04
    no_improve = 0
    big_kick = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = v.max()

        gap = max(0.004 * vmax, 0.002)
        active_idx = np.where(v >= vmax - gap)[0]
        max_act = 130
        if len(active_idx) > max_act:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-max_act:]]

        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)
            b_lin[i] = A[i] @ h - v[k]

        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.0]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq,
                         b_eq=[n / 2.0], bounds=list(zip(lb, ub)),
                         method='highs', options={'presolve': True})
        except Exception:
            res = None

        if res is None or not res.success:
            no_improve += 1
            if no_improve > 20:
                h = h + np.random.randn(n) * 0.01
                h = project_sum_box(h, n / 2.0)
                radius = 0.06
                no_improve = 0
                big_kick += 1
                if big_kick > 15:
                    break
            continue

        x_lp = res.x[:n]
        direction = x_lp - h
        dir_norm = np.linalg.norm(direction)
        if dir_norm < 1e-10:
            no_improve += 1
            continue

        best_c_step = evaluate_c5(h)
        best_alpha = 1.0
        for alpha in [0.4, 0.6, 0.8, 1.0, 1.3, 1.6]:
            x_try = h + alpha * direction
            x_try = project_sum_box(x_try, n / 2.0)
            c_try = evaluate_c5(x_try)
            if c_try < best_c_step - 1e-12:
                best_c_step = c_try
                best_alpha = alpha

        h_new = h + best_alpha * direction
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)

        if c_new < best_c - 1e-10:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new.copy()
            no_improve = 0
            if no_improve == 0 and it % 8 == 0:
                radius = min(radius * 1.12, 0.3)
        else:
            no_improve += 1

        if no_improve >= 10:
            radius = max(radius * 0.55, 0.002)
            no_improve = 0
            if radius <= 0.002:
                h = best_h + np.random.randn(n) * 0.015
                h = project_sum_box(h, n / 2.0)
                radius = 0.08
                no_improve = 0
                big_kick += 1
                if big_kick > 20:
                    break
    return best_h, best_c

def method_grad_bb(h0, deadline, max_iters=20000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    h_prev = h.copy()
    g_prev = None
    step = 0.008
    stall = 0
    beta_base = 25.0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        m = v.max()
        beta = beta_base * (1.0 + 0.005 * it)
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        if Z < 1e-30:
            break
        w /= Z

        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 5e-9:
                lag = k - (n - 1)
                g += w[k] * grad_lag(h, lag)

        if g_prev is not None:
            s = h - h_prev
            y = g - g_prev
            sy = np.dot(s, y)
            ss = np.dot(s, s)
            if sy > 1e-14 and ss > 1e-14:
                step_bb = ss / sy
                step = 0.4 * step + 0.6 * np.clip(step_bb, 1e-6, 0.05)

        h_prev = h.copy()
        g_prev = g.copy()
        h_new = h - step * g
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)

        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new
            stall = 0
        h = h_new.copy()
        stall += 1
        if stall > 800:
            h = best_h + np.random.randn(n) * 0.008
            h = project_sum_box(h, n / 2.0)
            step = 0.008
            stall = 0
    return best_h, best_c

def method_slsqp_refine(h0, deadline, max_betas=10):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [5, 10, 20, 40, 80, 160, 320, 640, 1280, 2560][:max_betas]
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

    for beta in betas:
        if time.time() > deadline:
            break
        for attempt in range(2):
            if time.time() > deadline:
                break
            x0 = best_h.copy() if attempt == 0 else project_sum_box(
                best_h + np.random.randn(n) * 0.006, n / 2.0)

            def obj_func(x, b=beta):
                vv = corr_all(x)
                mm = np.max(vv)
                expo = np.clip(b * (vv - mm), None, 500)
                u = np.exp(expo)
                return mm - np.log(u.sum()) / b

            def grad_func(x, b=beta):
                vv = corr_all(x)
                mm = np.max(vv)
                expo = np.clip(b * (vv - mm), None, 500)
                u = np.exp(expo)
                w = u / u.sum()
                gg = np.zeros(len(x))
                for k in range(len(vv)):
                    if w[k] > 1e-10:
                        lag = k - (len(x) - 1)
                        gg += w[k] * grad_lag(x, lag)
                return gg

            try:
                res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                              bounds=bounds, constraints=cons,
                              options={'maxiter': 100, 'ftol': 1e-15})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-11:
                    best_h = x_new.copy()
                    best_c = c_new
    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_ext = np.concatenate([h0, h0[:1]])
    h_new = np.interp(x_new * (n_old - 1) + 0.0, x_old * (n_old - 1) + 0.0, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def method_discrete_refine(h0, deadline, max_rounds=200):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()

    # Partial binarization: push values very close to 0 or 1
    for _ in range(50):
        changed = False
        for i in range(n):
            if h[i] < 0.03:
                old = h[i]
                h[i] = 0.0
                h = project_sum_box(h, n / 2.0)
                c = evaluate_c5(h)
                if c < best_c - 1e-11:
                    best_h = h.copy()
                    best_c = c
                    changed = True
                else:
                    h[i] = old
                    h = project_sum_box(h, n / 2.0)
            elif h[i] > 0.97:
                old = h[i]
                h[i] = 1.0
                h = project_sum_box(h, n / 2.0)
                c = evaluate_c5(h)
                if c < best_c - 1e-11:
                    best_h = h.copy()
                    best_c = c
                    changed = True
                else:
                    h[i] = old
                    h = project_sum_box(h, n / 2.0)
        if not changed:
            break
        if time.time() > deadline:
            break

    h = best_h.copy()

    # Swap-based local search on near-binary values
    temp = 0.002
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        ones = np.where(h > 0.85)[0]
        zeros = np.where(h < 0.15)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break

        c_cur = evaluate_c5(h)
        best_imp = 1e-9
        best_pair = None
        n_tries = min(len(ones) * len(zeros), 400)

        for _ in range(n_tries):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c_new = evaluate_c5(h)
            h[i] += 1.0
            h[j] -= 1.0
            h = project_sum_box(h, n / 2.0)
            imp = c_cur - c_new
            if imp > best_imp:
                best_imp = imp
                best_pair = (i, j)

        if best_pair is not None:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            h = project_sum_box(h, n / 2.0)
            c_new = evaluate_c5(h)
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        temp *= 0.97
    return best_h, best_c
