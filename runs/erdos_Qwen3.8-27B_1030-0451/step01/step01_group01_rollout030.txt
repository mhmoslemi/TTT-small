    n = len(h)
    d = abs(lag)
    if d == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        g[:n-d] += (1.0 - h[d:])
        g[d:] -= h[:n-d]
    return g

def pair_line_search(h, i, j, n_samples=30):
    """Find best alpha for pair (i,j) exchange. Returns (best_alpha, best_c5)."""
    n = len(h)
    a_min = max(-h[i], h[j] - 1.0)
    a_max = min(1.0 - h[i], h[j])
    if a_max - a_min < 1e-10:
        return 0.0, eval_c5(h)
    
    c_cur = eval_c5(h)
    best_alpha = 0.0
    best_c = c_cur
    
    # Sample the interval
    alphas = np.linspace(a_min, a_max, n_samples)
    for a in alphas:
        h_new = h.copy()
        h_new[i] += a
        h_new[j] -= a
        c = eval_c5(h_new)
        if c < best_c:
            best_c = c
            best_alpha = a
    
    # Fine search around best
    if best_alpha != 0.0:
        eps = (a_max - a_min) / n_samples * 0.5
        lo_b = max(a_min, best_alpha - eps)
        hi_b = min(a_max, best_alpha + eps)
        for _ in range(30):
            m1 = lo_b + (hi_b - lo_b) / 3.0
            m2 = hi_b - (hi_b - lo_b) / 3.0
            h1 = h.copy(); h1[i] += m1; h1[j] -= m1

                hi_b = m2
            else:
                lo_b = m1

        c = eval_c5(h_new)
        if c < best_c:
            best_c = c
            best_alpha = a_best
    
    return best_alpha, best_c

def binarize_push(h, strength):
    """Push h toward binary while preserving sum."""
    n = len(h)
    d = h * (1.0 - h) * (2.0 * h - 1.0)  # pushes away from 0.5
    h_new = h + strength * d
    h_new = project_sum_box(h_new, n / 2.0)
    return h_new

def project_sum_box(x, target):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-15:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-15:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-11:
        idx = np.where((y > 1e-15) & (y < 1.0 - 1e-15))[0]
        if len(idx) > 0:
            y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)

def eval_c5(h):
    n = len(h)

def corr_vec(h):

def grad_lag(h, lag):
    n = len(h)
    d = abs(lag)
    if d == 0:
        return 1.0 - 2.0 * h
    g = np.zeros(n)
    if lag > 0:
        g[lag:] += (1.0 - h[:n-lag])
        g[:n-lag] -= h[lag:]
    else:
        g[:n-d] += (1.0 - h[d:])
        g[d:] -= h[:n-d]
    return g
