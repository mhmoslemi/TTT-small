where slopes_j is the slope array for coordinate j (computed before the update).

This is O(n) per update, which is already cheaper than recomputing c from scratch (O(n²)).

So the full algorithm per sweep:
1. For each j = 0, 1, ..., n-1:
   a. Compute slopes for j: O(n)
   b. Golden section: O(40n)
   c. Update h[j] and c: O(n)
   Total per j: O(41n)
2. Project h onto constraint set: O(n log n) (bisection)
3. Recompute c from scratch: O(n²) (or trust the incremental updates)

Full sweep: O(41n² + n²) ≈ O(42n²).

This is efficient enough. Let me code this up.

One more consideration: the golden section search finds the minimum of max_k (a_k*t + b_k) on [0,1]. Since this is a convex piecewise-linear function, the minimum is at either t=0, t=1, or at a point where two lines cross. The golden section will find it in O(log(1/ε)) iterations, where ε is the desired precision. 40 iterations give ε ≈ (0.618)^40 ≈ 10^(-10), which is more than enough.

But actually, for a piecewise-linear function, we can do better. The minimum is at a vertex, which is either t=0, t=1, or at the intersection of two "active" lines. We can find the minimum by:
1. Evaluating at t=0 and t=1
2. If neither is the minimum, the minimum is at an interior point where the "upper envelope" changes slope from negative to positive.

Finding this exactly requires sorting the lines by slope and finding where the maximum changes. This is O(n log n) per coordinate, which is worse than the golden section's O(40n) for small n. So golden section is fine.

Actually, for n=240 (so 2n-1=479 lines), the golden section with 40 iterations does 40*479 ≈ 19K operations per coordinate. 240 coordinates gives 4.7M per sweep. 100 sweeps = 470M. That's about 1 second in NumPy (vectorized).

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def cyclic_cd(h_init, n, max_sweeps=200, deadline=None):
    h = h_init.copy()
    dx = 2.0 / n
    n2 = 2 * n - 1
    c = np.correlate(h, 1.0 - h, 'full') * dx
    best_c5 = c.max()
    best_h = h.copy()
    gr = (np.sqrt(5) - 1) / 2
    target = n / 2.0

    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline - 5:
            break
        for j in range(n):
            slopes = np.zeros(n2)
            # First term: (1-h[j-d]) for valid j-d
            lo1 = max(0, j)
            hi1 = min(n2 - 1, j + n - 1)
            if lo1 <= hi1:
                idx = j + n - 1 - np.arange(lo1, hi1 + 1)
                slopes[lo1:hi1 + 1] = 1.0 - h[idx]
            # Second term: -h[j+d] for valid j+d
            lo2 = max(0, n - 1 - j)
            hi2 = min(n2 - 1, 2 * n - 2 - j)
            if lo2 <= hi2:
                idx = j + np.arange(lo2, hi2 + 1) - (n - 1)
                slopes[lo2:hi2 + 1] -= h[idx]

            a = slopes * dx
            b = c - a * h[j]

            # Golden section to minimize max(a*t + b) on [0,1]
            lo, hi = 0.0, 1.0
            cv = hi - gr * (hi - lo)
            dv = lo + gr * (hi - lo)
            fc = (a * cv + b).max()
            fd = (a * dv + b).max()
            for _ in range(35):
                if fc < fd:
                    hi = dv
                    dv = cv
                    fd = fc
                    cv = hi - gr * (hi - lo)
                    fc = (a * cv + b).max()
                else:
                    lo = cv
                    cv = dv
                    fc = fd
                    dv = lo + gr * (hi - lo)
                    fd = (a * dv + b).max()
            t_opt = (lo + hi) * 0.5

            delta = t_opt - h[j]
            if abs(delta) > 1e-9:
                h[j] = t_opt
                c += a * delta
                c5 = c.max()
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()

        # Project to maintain sum constraint
        h = project_feasible(h, target)
        c = np.correlate(h, 1.0 - h, 'full') * dx
        c5 = c.max()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def smooth_slsqp(h0, n, deadline):

    G_N = n
    G_TAU = 1.0

    def c_eq_fun(h):
        return float(np.sum(h) - G_T)

    def c_eq_jac(h):
        return np.ones(G_N)

    def c_obj(h):
        dx = 2.0 / n
        cv = np.correlate(h, 1.0 - h, 'full') * dx
        m = cv.max()
        s = np.clip((cv - m) / G_TAU, -400.0, 400.0)
        e = np.exp(s)
        return float(m + G_TAU * np.log(e.sum()))

    def c_grad(h):
        dx = 2.0 / n
        n2 = 2 * n - 1
        cv = np.correlate(h, 1.0 - h, 'full') * dx
        m = cv.max()
        s = np.clip((cv - m) / G_TAU, -400.0, 400.0)
        e = np.exp(s)
        alpha = e / e.sum()
        grad = np.zeros(n)
        active = np.where(alpha > 1e-10)[0]
        for k in active:
            a_k = alpha[k]
            d = k - (n - 1)
            lo1 = max(0, k)
            hi1 = min(n2 - 1, k + n - 1)
            if lo1 <= hi1:
                idx = G_N - 1 + lo1 - k
                # Use the slope formula: (1-h[j-d]) part
                # For coordinate j, first term valid when j in [max(0,d), min(n-1,n-1+d)]
                # Let's use the standard approach
                pass
            # Direct computation per coordinate is too slow; use vectorized
            # slope for all j at once
            s_arr = np.zeros(n)
            lo1 = max(0, d)
            hi1 = min(n - 1, n - 1 + d)
            if lo1 <= hi1:
                s_arr[lo1:hi1 + 1] += (1.0 - h[lo1 - d:hi1 + 1 - d])
            lo2 = max(0, -d)
            hi2 = min(n - 1, n - 1 - d)
            if lo2 <= hi2:
                s_arr[lo2:hi2 + 1] -= h[lo2 + d:hi2 + 1 + d]

    target = n / 2.0
    c5_init = evaluate_c5(h0)

        if n > 2400 or n <= best_n:
            continue
        if time.time() > deadline - 40:
            break
        h_interp = interp_h(best_h, n)
        h_cd, c5_cd = cyclic_cd(h_interp, n, max_sweeps=60, deadline=deadline)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            best_n = n

    # Phase 5: SLSQP polish on best solution if small enough
    if time.time() < deadline - 40 and best_n <= 500:
        h_slsqp, c5_slsqp = smooth_slsqp(best_h, best_n, deadline)
        if c5_slsqp < best_c5:
            best_c5 = c5_slsqp
            best_h = h_slsqp
            best_n = best_n

    # Phase 6: Additional CD sweeps on best if time remains
    if time.time() < deadline - 20:
        remaining = deadline - time.time()
        sweeps = min(200, int(remaining / max(0.05, best_n**2 * 0.001)))
        h_cd, c5_cd = cyclic_cd(best_h, best_n, max_sweeps=max(10, sweeps), deadline=deadline)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd
            best_n = best_n

    # Final feasibility and evaluation
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n / 2.0)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
