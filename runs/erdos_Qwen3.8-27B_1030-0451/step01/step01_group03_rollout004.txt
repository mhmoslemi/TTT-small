    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _ls_grad(h):
    n = _G_N
    dx = _G_DXX
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    active = np.where(alpha > 1e-10)[0]
    for k in active:
        a = alpha[k] * dx
        d = int(k) - (n - 1)
        if d >= 0:
            hi1 = n - d
            grad[:hi1] += a * (1.0 - h[d:hi1+d])
            grad[d:] -= a * h[:hi1]
        else:
            dd = -d
            hi1 = n - dd
            grad[:hi1] += a * (1.0 - h[dd:hi1+dd])
            grad[dd:] -= a * h[:hi1]
    return grad

def _c_eq(h):
    return float(np.sum(h) - _G_T)

def _c_eq_jac(h):
    return np.ones(_G_N)

def slsqp_optimize(h0, n, deadline, tau0=None, n_tau=16):
    global _G_T, _G_N, _G_DXX, _G_TAU
    _G_T = n / 2.0
    _G_N = n
    _G_DXX = 2.0 / n
    
    target = n / 2.0
    best_h = h0.copy()

    tau = tau0
    for i in range(n_tau):
        tau_sched.append(tau)
        if i < n_tau // 3:

            tau *= 0.5
        else:
            tau *= 0.35
    
    # Ensure minimum tau
    tau_sched = [max(t, 1e-5) for t in tau_sched]
    
    constraints = [{'type': 'eq', 'fun': _c_eq, 'jac': _c_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    h_curr = h0.copy()
    stall = 0
    
    for i, tau in enumerate(tau_sched):

def active_set_opt(h0, n, deadline, top_K=10):
    """Active-set projected gradient focusing on top lags."""
    dx = 2.0 / n
    target = n / 2.0
    h_curr = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    max_iter = 5000
    stagnant = 0
    
    for it in range(max_iter):
        if time.time() > deadline - 3:
            break
        
        c = np.correlate(h_curr, 1.0 - h_curr, 'full') * dx
        cmax = float(c.max())
        
        # Identify active lags (within margin of max)
        margin = max(0.01 * cmax, 0.005)
        active_mask = c >= cmax - margin
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > top_K:
            # Keep only top K
            order = np.argsort(c[active_idx])[::-1][:top_K]
            active_idx = active_idx[order]
        if len(active_idx) == 0:
            active_idx = np.array([int(np.argmax(c))])
        
        # Compute weighted gradient (weight by proximity to max)
        weights = np.ones(len(active_idx))
        c_act = c[active_idx]
        # Sharper weights for lags closer to max
        w = 1.0 / (cmax - c_act + 1e-8)
        weights = w / w.sum()
        
        grad = np.zeros(n)
        for idx, wt in zip(active_idx, weights):
            a = wt * dx
            d = int(idx) - (n - 1)
            if d >= 0:
                hi1 = n - d
                grad[:hi1] += a * (1.0 - h_curr[d:hi1+d])
                grad[d:] -= a * h_curr[:hi1]
            else:
                dd = -d
                hi1 = n - dd
                grad[:hi1] += a * (1.0 - h_curr[dd:hi1+dd])
                grad[dd:] -= a * h_curr[:hi1]
        
        # Project onto tangent (sum constraint)
        gt = grad - grad.mean()
        gnorm = np.linalg.norm(gt)
        if gnorm < 1e-14:
            break
        
        # Line search
        improved = False
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            y = h_curr - alpha * gt
            h_try = project_feasible(y, target)
            c5_try = evaluate_c5(h_try)
            if c5_try < cmax - 1e-13:
                h_curr = h_try
                cmax = c5_try
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_curr.copy()
                improved = True
                stagnant = 0
                break
        
        if not improved:
            stagnant += 1
            if stagnant > 20:
                break
    
    return best_h, best_c5

def paired_descend(h0, n, deadline, max_rounds=100):
    """Paired coordinate descent for final polish."""
    dx = 2.0 / n
    target = n / 2.0
    h_curr = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    n_round = 0
    while n_round < max_rounds and time.time() < deadline - 3:
        n_round += 1
        c = np.correlate(h_curr, 1.0 - h_curr, 'full') * dx
        cmax = float(c.max())
        
        # Get gradient of the max correlation
        k_max = int(np.argmax(c))
        d_max = k_max - (n - 1)
        grad = np.zeros(n)
        if d_max >= 0:
            hi1 = n - d_max
            grad[:hi1] += (1.0 - h_curr[d_max:hi1+d_max])
            grad[d_max:] -= h_curr[:hi1]
        else:
            dd = -d_max
            hi1 = n - dd
            grad[:hi1] += (1.0 - h_curr[dd:hi1+dd])
            grad[dd:] -= h_curr[:hi1]
        
        gt = grad - grad.mean()
        
        # Top coordinates by |gradient|
        top = min(20, n // 15)
        top_idx = np.argsort(np.abs(gt))[-top:][::-1]
        
        improved = False
        for i_pos in range(len(top_idx)):
            if time.time() > deadline - 3:
                break
            i = int(top_idx[i_pos])
            for j_pos in range(len(top_idx)):
                if j_pos == i_pos:
                    continue
                j = int(top_idx[j_pos])
                # Direction: increase h[i] if gt[i] < 0, decrease if gt[i] > 0
                # Transfer from j to i or vice versa
                for sign in [1, -1]:
                    # h[i] += delta * sign, h[j] -= delta * sign
                    # Check bounds
                    hi_i = 1.0 - h_curr[i] if sign > 0 else h_curr[i]
                    hi_j = 1.0 - h_curr[j] if sign < 0 else h_curr[j]
                    max_delta = min(hi_i, hi_j)
                    if max_delta < 1e-8:
                        continue
                    for frac in [0.3, 0.6, 1.0]:
                        delta = frac * max_delta
                        h_try = h_curr.copy()
                        h_try[i] += delta * sign
                        h_try[j] -= delta * sign
                        c5_try = evaluate_c5(h_try)
                        if c5_try < best_c5 - 1e-13:
                            best_c5 = c5_try
                            best_h = h_try.copy()
                            h_curr = h_try
                            improved = True
                            break
        
        if not improved:
            break
    
    return best_h, best_c5

        a = alpha[k] * dx
        d = int(k) - (n - 1)
        if d >= 0:
            hi1 = n - d
            grad[:hi1] += a * (1.0 - h[d:hi1+d])
            grad[d:] -= a * h[:hi1]
        else:
            dd = -d
            hi1 = n - dd
            grad[:hi1] += a * (1.0 - h[dd:hi1+dd])
            grad[dd:] -= a * h[:hi1]
    return grad

def _c_eq(h):
    return float(np.sum(h)
