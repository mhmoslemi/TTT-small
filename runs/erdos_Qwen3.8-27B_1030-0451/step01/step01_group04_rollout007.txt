import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_alphas(h, j, N):
    alphas = np.zeros(2 * N - 1)
    for r in range(2 * N - 1):
        d = r - (N - 1)
        if d == 0:
            continue
        elif d > 0:
            alpha = 0.0
            if j <= N - 1 - d:
                alpha += (1.0 - h[j + d])
            if j >= d:
                alpha -= h[j - d]
            alphas[r] = alpha
        else:
            p = -d
            alpha = 0.0
            if j >= p:
                alpha += (1.0 - h[j - p])
            if j <= N - 1 - p:
                alpha -= h[j + p]
            alphas[r] = alpha
    return alphas

def coordinate_descent(h, n_grid=100, max_passes=20, time_limit=None, start_time=None):
    N = len(h)
    h = h.copy()
    corr = np.correlate(h, 1.0 - h, mode='full').astype(float)
    
    for pass_num in range(max_passes):
        if time_limit and time.time() - start_time > time_limit:
            break
        improved = False
        for j in range(N):
            if time_limit and time.time() - start_time > time_limit:
                break
            
            old_val = h[j]
            alphas = compute_alphas(h, j, N)
            base_0 = corr[N - 1] - old_val * (1.0 - old_val)
            
            t_grid = np.linspace(0.0, 1.0, n_grid)
            diff = (t_grid - old_val)[:, np.newaxis]  # (n_grid, 1)
            linear_vals = corr[np.newaxis, :] + alphas[np.newaxis, :] * diff  # (n_grid, 2N-1)
            linear_vals[:, N - 1] = base_0 + t_grid * (1.0 - t_grid)
            
            f_vals = linear_vals.max(axis=1)
            best_idx = np.argmin(f_vals)
            new_val = t_grid[best_idx]
            
            if abs(new_val - old_val) > 1e-10:
                delta = new_val - old_val
                # Update correlations incrementally
                for r in range(2 * N - 1):
                    if r == N - 1:
                        continue
                    corr[r] += delta * alphas[r]
                corr[N - 1] = base_0 + new_val * (1.0 - new_val)
                h[j] = new_val
                improved = True
        
        if not improved:
            break
    
    return h

Let me verify the update formula once more. For d > 0:
corr_d = ∑_{i=0}^{N-1-d} h[i]*(1-h[i+d])

Change h[j] by δ:
- If j is in [0, N-1-d]: the term h[j]*(1-h[j+d]) changes by δ*(1-h[j+d])
- If j is in [d, N-1]: the term h[j-d]*(1-h[j]) changes by -h[j-d]*δ

So Δcorr_d = δ * [(1-h[j+d])*I(j≤N-1-d) - h[j-d]*I(j≥d)]

This is exactly δ * alpha_d as I defined it. ✓

Actually, for a given j, the alphas can be computed more efficiently:
