def compute_corr(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):

    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_to_feasible(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 2.0
    hi = float(np.max(p)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-13:
        mask = (proj > 1e-12) & (proj < 1.0 - 1e-12)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
            diff2 = target - float(np.sum(proj))

                if nm2 > 0:

                    proj = np.clip(proj, 0.0, 1.0)
    return proj

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    # lag 0 (index n-1)
    G[n-1] = dx * (1.0 - 2.0 * h)
    for L in range(1, n):
        # Positive lag L (index n-1+L): c = dx * sum_{i=0}^{n-1-L} h[i+L]*(1-h[i])
        g_pos = np.zeros(n)
        g_pos[L:] += dx * (1.0 - h[:n-L])
        g_pos[:n-L] += -dx * h[L:]
        G[n-1+L] = g_pos
        # Negative lag -L (index n-1-L): c = dx * sum_{i=L}^{n-1} h[i-L]*(1-h[i])
        g_neg = np.zeros(n)
        g_neg[:n-L] += dx * (1.0 - h[L:])
        g_neg[L:] += -dx * h[:n-L]
        G[n-1-L] = g_neg
    return G

def soft_max_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    c = corr * dx
    f = float(np.max(c))

    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n), f, c
    w = w / ws
    G = build_grad_matrix(h)
    grad = G.T @ w
    return grad, f, c

def lp_direction(h, c, f, radius, n_active=60):
    n = len(h)
    n_lags = 2 * n - 1
    G = build_grad_matrix(h)
    # Select active lags
    thresh = f - 0.08 * f
    active_idx = np.where(c >= thresh)[0]
    if len(active_idx) > n_active:

    m_rows = len(rows)
    A_ub = np.zeros((m_rows, n + 1))
    A_ub[:, :n] = rows
    A_ub[:, n] = -1.0
    b_ub = rhs
    
    # Add sum constraint as equality
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds: -radius <= delta_i <= radius, t unbounded below (but >= -f to keep meaningful)

    except Exception:

    return None, None

def coordinate_pair_minimize(h, i, j, c, G):
    """Minimize max_k (c_k + t*(G[k,i]-G[k,j])) over feasible t."""
    n = len(h)
    slopes = G[:, i] - G[:, j]
    t_lo = max(-h[i], h[j] - 1.0)
    t_hi = min(1.0 - h[i], h[j])
    if t_hi - t_lo < 1e-12:
        return 0.0, None
    # Evaluate on fine grid
    n_pts = 300
    t_vals = np.linspace(t_lo, t_hi, n_pts)
    new_c = c[np.newaxis, :] + t_vals[:, np.newaxis] * slopes[np.newaxis, :]
    obj = np.max(new_c, axis=1)
    idx = np.argmin(obj)
    t_best = t_vals[idx]
    # Refine
    if 0 < idx < n_pts - 1:
        t_left = t_vals[idx-1]
        t_right = t_vals[idx+1]
        for _ in range(40):
            tm1 = t_left + (t_right - t_left) / 3.0
            tm2 = t_right - (t_right - t_left) / 3.0
            o1 = np.max(c + tm1 * slopes)
            o2 = np.max(c + tm2 * slopes)
            if o1 < o2:
                t_right = tm2
            else:
                t_left = tm1
        t_best = (t_left + t_right) / 2.0
    return t_best, None

        sanity_h = project_to_feasible(np.full(final_n, 0.5), final_n)
        final_c5 = compute_c5(sanity_h)
    
    return sanity_h, final_c5, final_n
