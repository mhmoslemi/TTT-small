    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
    return proj

def gradient_full(h, corr, tau):
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        L = abs(lag)
        m = n - L
        grad[:m] += wk * (1.0 - h[L:])
        grad[L:] -= wk * h[:m]
    return grad

def gradient_topk(h, corr, k_top, inner_tau=50.0):
    n = len(h)
    total = len(corr)
    k = min(k_top, total)
    top_idx = np.argpartition(-corr, k)[:k]
    max_val = float(np.max(corr))
    top_corr = corr[top_idx]
    cs = top_corr - max_val
    w = np.exp(inner_tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for i in range(k):
        wk = w[i]
        if wk < 1e-14:
            continue
        k_idx = top_idx[i]
        lag = k_idx - (n - 1)
        L = abs(lag)
        m = n - L
        grad[:m] += wk * (1.0 - h[L:])
        grad[L:] -= wk * h[:m]
    return grad

def upsample_h(h, n_new):
    n_old = len(h)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    return np.interp(x_new, x_old, h)

def smooth_optimize(h0, time_limit, tau_max=500, use_topk=False, k_top=10, inner_tau=50.0):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    t0 = time.time()
    step = 2.0 / n

    phases = [
        (0.3 * tau_max, 0.30),
        (tau_max * 0.6, 0.30),
        (tau_max * 0.85, 0.20),
        (tau_max, 0.20),
    ]

    for tau_target, frac in phases:
        if time.time() - t0 > time_limit - 3:
            break
        phase_time = time_limit * frac
        phase_t0 = time.time()

        while time.time() - phase_t0 < phase_time and time.time() - t0 < time_limit - 2:
            elapsed_p = max(time.time() - phase_t0, 1e-6)
            frac_e = min(elapsed_p / max(phase_time, 0.1), 1.0)
            tau = 5.0 + max(tau_target - 5.0, 0.0) * (frac_e ** 0.6)

            corr = compute_corr(h)
            c5_val = float(np.max(corr) * dx)

            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()

            if use_topk:
                grad = gradient_topk(h, corr, k_top, inner_tau)
            else:
                grad = gradient_full(h, corr, tau)

            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-14:
                h = project_constraint(h + np.random.randn(n) * 0.05 / np.sqrt(n), n)
                step = 2.0 / n
                continue

            made_progress = False
            for _bt in range(12):
                h_new = project_constraint(h - step * grad, n)
                c5_new = compute_c5(h_new)
                if c5_new <= c5_val + 1e-12:
                    h = h_new
                    made_progress = (c5_new < c5_val - 1e-11)
                    break
                step *= 0.5
            else:
                step = max(step, 0.5 / n)
                h = project_constraint(h - step * grad, n)

            if made_progress:
                step = min(step * 1.10, 8.0 / n)
            elif step < 0.005 / n:
                step = 0.005 / n
            else:
                step *= 0.92

    return best_h, best_c5

def block_perturb(h, n_blocks, seed_val):
    n = len(h)
    rng = np.random.RandomState(seed_val)
    h_new = h.copy()
    block_size = n // n_blocks
    for b in range(n_blocks):
        start = b * block_size
        end = min(start + block_size, n)
        if end - start < 2:
            continue
        offset = rng.randn(end - start) * 0.08
        h_new[start:end] += offset
    return project_constraint(h_new, n)

def make_start_harmonic(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    freqs = [1, 2, 3, 4, 5, 7]
    amps = [0.28, 0.18, 0.12, 0.08, 0.05, 0.03]
    phases = [0.0, 0.7, 1.4, 2.1, 0.3, 2.8]
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(np.pi * f * x + p)
    h = 0.5 + 0.4 * h
    h = h + rng.randn(n) * 0.02
    return project_constraint(h, n)

def make_start_pulsed(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.zeros(n)
    num_pulses = rng.randint(2, 5)
    for j in range(num_pulses):
        center = rng.uniform(0.1, 1.9)
        width = rng.uniform(0.15, 0.5)
        amp = rng.uniform(0.5, 1.0)
        h += amp * np.exp(-0.5 * ((x - center) / width) ** 2)
    h = h / (1.0 + h)
    h = h + rng.randn(n) * 0.03
    return project_constraint(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')

    def rem():
        return budget_s - (time.time() - t_start)

    # Phase 1: Start from initial_h_values
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except (NameError, UnboundLocalError):
        h0 = make_start_harmonic(81, seed)

    n0 = len(h0)
    h0 = project_constraint(h0, n0)
    best_c5 = compute_c5(h0)
    best_h = h0.copy()

    # Phase 2: Quick optimization at initial n
    alloc = min(rem() - 400, 60)
    if alloc > 10:
        h_opt, c5_opt = smooth_optimize(best_h.copy(), alloc, tau_max=200)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 3: Upsample to n=250 and optimize
    if rem() > 200:
        h_up = upsample_h(best_h, 250)
        h_up = project_constraint(h_up, 250)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_h = h_up.copy()
            best_c5 = c5_up
        alloc = min(rem() - 150, 150)
        if alloc > 15:
            h_opt, c5_opt = smooth_optimize(h_up, alloc, tau_max=500)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Phase 4: Upsample to n=500 with top-k focus
    if rem() > 150:
        h_up = upsample_h(best_h, 500)
        h_up = project_constraint(h_up, 500)
        c5_up = compute_c5(h_up)
        if c5_up < best_c5:
            best_h = h_up.copy()
            best_c5 = c5_up
        alloc = min(rem() - 100, 200)
        if alloc > 15:
            h_opt, c5_opt = smooth_optimize(h_up, alloc, tau_max=400, use_topk=True, k_top=8, inner_tau=80.0)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
        # Second pass with higher tau
        if rem() > 80:
            alloc2 = min(rem() - 50, 100)
            if alloc2 > 10:
                h_opt, c5_opt = smooth_optimize(best_h.copy(), alloc2, tau_max=1500, use_topk=True, k_top=6, inner_tau=120.0)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt

    # Phase 5: Try independent starts at n=200-400
    for n_try in [200, 300, 400]:
        if rem() < 80:
            break
        for init_fn in [make_start_harmonic, make_start_pulsed]:
            if rem() < 50:
                break
            h_try = init_fn(n_try, seed + n_try * 11)
            alloc = min(rem() - 35, 60)
            if alloc > 10:
                h_opt, c5_opt = smooth_optimize(h_try, alloc, tau_max=600, use_topk=True, k_top=10)
                if c5_opt < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_opt

    # Phase 6: Perturbation restarts on best solution
    for restart in range(4):
        if rem() < 50:
            break
        h_pert = block_perturb(best_h, 3 + restart, seed + 500 + restart * 17)
        alloc = min(rem() - 30, 40)
        if alloc > 10:
            h_opt, c5_opt = smooth_optimize(h_pert, alloc, tau_max=800, use_topk=True, k_top=8, inner_tau=100.0)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Phase 7: Final high-precision polish
    if rem() > 20 and best_h is not None:
        n_cur = len(best_h)
        # If we have a decent n, do a final very sharp optimization
        alloc = rem() - 5
        if alloc > 10:
            h_opt, c5_opt = smooth_optimize(best_h.copy(), alloc, tau_max=3000, use_topk=True, k_top=5, inner_tau=200.0)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt

    # Final projection and verification
    best_h = project_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
