def build_grad(h):

    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project(h_new, n_new)

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.32 * np.sin(2.0 * np.pi * x)
    h += 0.16 * np.sin(4.0 * np.pi * x + 0.6)
    h += 0.09 * np.sin(6.0 * np.pi * x + 1.4)
    h += 0.05 * np.sin(9.0 * np.pi * x + 2.0)
    h += 0.025 * np.sin(13.0 * np.pi * x + 0.8)
    h += rng.randn(n) * 0.025
    return project(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    
    best_h = None
    best_c5 = float('inf')
    
    n0 = 81
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
        h0 = project(h0, n0)
    except (NameError, UnboundLocalError):
        h0 = make_start(n0, seed)
    
    c5_0 = compute_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_0
    
    # Main optimization on parent
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 120
    if remaining > 80:
        h_opt, c5_opt = optimize_smoothing(h0.copy(), remaining, seed)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Coordinate descent on best
    elapsed = time.time() - t_start
    if budget_s - elapsed > 30:
        h_pol, c5_pol = coord_polish(best_h.copy(), min(budget_s - elapsed - 20, 30), seed + 11)
        if c5_pol < best_c5:
            best_h = h_pol
            best_c5 = c5_pol
    
    # Try different resolutions
    n_candidates = [n0 + 20, n0 + 50, n0 + 100]
    for n_try in n_candidates:
        if n_try > 250:
            break
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 50
        if remaining < 40:
            break
        h_start = resample_h(best_h, n_try)
        alloc = min(remaining // 2, 70)
        if alloc > 25:
            h_opt, c5_opt = optimize_smoothing(h_start, alloc, seed + n_try)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
            # Quick coord polish on the result
            if time.time() - t_start < budget_s - 15:
                h_pol, c5_pol = coord_polish(h_opt.copy(), min(10, budget_s - (time.time() - t_start) - 10), seed + n_try + 22)
                if c5_pol < best_c5:
                    best_h = h_pol
                    best_c5 = c5_pol
    
    # If parent n was small, also try slightly smaller
    if n0 >= 90:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 30
        if remaining > 40:
            n_small = max(61, n0 - 20)
            h_small = resample_h(best_h, n_small)
            alloc = min(remaining // 2, 50)
            if alloc > 20:
                h_opt, c5_opt = optimize_smoothing(h_small, alloc, seed + 555)
                if c5_opt < best_c5:
                    best_h = h_opt
                    best_c5 = c5_opt
    
    # Final optimization pass on the best found
    elapsed = time.time() - t_start
    if best_h is not None and budget_s - elapsed > 30:
        h_final, c5_final = optimize_smoothing(best_h.copy(), budget_s - elapsed - 5, seed + 9999)
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    
    # Final coordinate polish
    elapsed = time.time() - t_start
    if best_h is not None and budget_s - elapsed > 10:
        h_cp, c5_cp = coord_polish(best_h.copy(), budget_s - elapsed - 3, seed + 777)
        if c5_cp < best_c5:
            best_h = h_cp
            best_c5 = c5_cp
    
    # Verify and return
    best_h = project(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
