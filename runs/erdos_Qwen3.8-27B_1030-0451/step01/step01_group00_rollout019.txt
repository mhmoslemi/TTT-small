                        best_h = h_t.copy()
                        break
    return best_h, best_c5

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.12 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.07 * np.sin(6.0 * np.pi * x + 1.3)
    h += 0.04 * np.sin(9.0 * np.pi * x + 2.0)
    h += 0.02 * np.sin(13.0 * np.pi * x + 0.4)
    h += rng.randn(n) * 0.015
    return project_feasible(h, n)

def resample_profile(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_feasible(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def elapsed():
        return time.time() - t_start

    def remaining():
        return budget_s - elapsed()

    def update_state(h_cand, c_cand, n_cand):
        nonlocal best_h, best_c5, best_n
        if c_cand < best_c5 - 1e-13:
            best_h = h_cand.copy()
            best_c5 = c_cand
            best_n = n_cand

    # Phase 1: Initialize
    n_init = 81
    try:
        h0 = np.asarray(initial_h_values, dtype=float)
        n_init = len(h0)
        h0 = project_feasible(h0, n_init)
        c5_0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_0
        best_n = n_init
    except (NameError, UnboundLocalError):
        h0 = make_start(n_init, seed)
        best_h = h0.copy()
        best_c5 = compute_c5(h0)
        best_n = n_init

    # Phase 2: Main LP + smooth optimization on parent n
    rem = remaining() - 30
    if rem > 60:
        h_opt, c5_opt = local_optimize(best_h.copy(), rem, seed)
        update_state(h_opt, c5_opt, len(h_opt))

    # Phase 3: Pair polish on best
    rem = remaining() - 25
    if rem > 20 and best_h is not None:
        h_pol, c5_pol = pair_polish(best_h.copy(), min(rem, 120), seed + 1)
        update_state(h_pol, c5_pol, len(h_pol))

    # Phase 4: More LP optimization rounds
    rem = remaining() - 25
    if rem > 40 and best_h is not None:
        rng = np.random.RandomState(seed + 500)
        h_pert = project_feasible(best_h + rng.randn(len(best_h)) * 0.025, len(best_h))
        alloc = min(rem, 150)
        h_opt, c5_opt = local_optimize(h_pert, alloc, seed + 100)
        update_state(h_opt, c5_opt, len(h_opt))
        h_opt2, c5_opt2 = local_optimize(best_h.copy(), min(rem - alloc, 120), seed + 200)
        update_state(h_opt2, c5_opt2, len(h_opt2))

    # Phase 5: Try different resolutions
    for n_try in [101, 131, 161, 201]:
        rem = remaining() - 20
        if rem < 30:
            break
        h_start = resample_profile(best_h, n_try)
        alloc = min(rem // 3, 80)
        if alloc < 15:
            alloc = min(rem - 20, 60)
        if alloc < 12:
            break
        h_opt, c5_opt = local_optimize(h_start, alloc, seed + n_try)
        if c5_opt < best_c5 - 1e-6:
            update_state(h_opt, c5_opt, n_try)
            # Polish at this resolution
            rem2 = remaining() - 20
            if rem2 > 15:
                h_p2, c5_p2 = pair_polish(h_opt, min(rem2, 40), seed + n_try + 1)
                update_state(h_p2, c5_p2, n_try)

    # Phase 6: Final polish at best resolution
    rem = remaining() - 5
    if rem > 15 and best_h is not None:
        h_fin, c5_fin = local_optimize(best_h.copy(), min(rem, 80), seed + 999)
        update_state(h_fin, c5_fin, len(h_fin))

    # Phase 7: Final pair polish
    rem = remaining() - 3
    if rem > 10 and best_h is not None:
        h_fp, c5_fp = pair_polish(best_h.copy(), min(rem, 30), seed + 5555)
        update_state(h_fp, c5_fp, len(h_fp))

    # Final validation
    if best_h is None:
        n_fb = 81
        best_h = make_start(n_fb, seed)
        best_c5 = compute_c5(best_h)
        best_n = n_fb
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return best_h, best_c5, len(best_h)
