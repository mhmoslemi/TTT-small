import numpy as np
import time
from scipy.optimize import minimize, minimize_scalar


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = 0.5 * (lo + hi)
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            for idx in np.where(h < 1.0)[0]:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
                if residual < 1e-12:
                    break
        else:
            for idx in np.where(h > 0.0)[0]:
                rem = max(residual, -h[idx])
                h[idx] += rem
                residual -= rem
                if residual > -1e-12:
                    break
    return h


def smooth_obj(h, tau):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    return m + tau * np.log(total)


def smooth_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w = w / w.sum()
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r]
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * dx * (1.0 - h[:n])
                g[:n] -= a * dx * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * dx * (1.0 - h[dd:])
                g[dd:] -= a * dx * h[:n]
    return g


def slsqp_f(h, tau):
    return smooth_obj(h, tau)


def slsqp_g(h, tau):
    return smooth_grad(h, tau)


def sum_eq(h, target):
    return np.sum(h) - target


def sum_eq_jac(h, target):
    return np.ones_like(h)


def pair_descent(h, target_sum, max_sweeps=200, start_time=None, deadline=None):
    N = len(h)
    h = h.copy()
    current_c5 = compute_c5(h)
    pairings = [
        [(j, (j + 1) % N) for j in range(0, N - 1, 1) if j != (j+1) % N],
        [(j, (j + N // 3) % N) for j in range(N) if j != (j + N // 3) % N],
        [(j, (j + N // 2) % N) for j in range(N // 2) if j != (j + N // 2) % N],
    ]
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        improved = False
        for pairing in pairings:
            for (j, k) in pairing:
                if j == k:
                    continue
                old_j, old_k = h[j], h[k]
                t_lo = max(-old_j, old_k - 1.0)
                t_hi = min(1.0 - old_j, old_k)
                if t_lo >= t_hi - 1e-15:
                    continue

                def obj_1d(t):
                    h[j] = old_j + t
                    h[k] = old_k - t
                    return compute_c5(h)

                try:
                    res = minimize_scalar(obj_1d, bounds=(t_lo, t_hi), method='bounded',
                                         options={'maxiter': 40})
                    if res.fun < current_c5 - 1e-13:
                        h[j] = old_j + res.x
                        h[k] = old_k - res.x
                        current_c5 = res.fun
                        improved = True
                    else:
                        h[j] = old_j
                        h[k] = old_k
                except Exception:
                    h[j] = old_j
                    h[k] = old_k
        if not improved:
            break
    return h, current_c5


def proj_subgradient(h, tau, target_sum, max_iters=300, deadline=None):
    N = len(h)
    current_c5 = compute_c5(h)
    step_size = 0.05
    for it in range(max_iters):
        if deadline is not None and time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full') * (2.0 / N)
        m = np.max(c)
        exponents = np.clip((c - m) / tau, -500.0, 500.0)
        w = np.exp(exponents)
        w = w / w.sum()
        g = np.zeros(N)
        for r in range(2 * N - 1):
            a = w[r]
            if a < 1e-12:
                continue
            d = r - (N - 1)
            if d >= 0:
                n = N - d
                if n > 0:
                    g[d:] += a * (2.0 / N) * (1.0 - h[:n])
                    g[:n] -= a * (2.0 / N) * h[d:]
            else:
                dd = -d
                n = N - dd
                if n > 0:
                    g[:n] += a * (2.0 / N) * (1.0 - h[dd:])
                    g[dd:] -= a * (2.0 / N) * h[:n]
        alpha = step_size
        found = False
        for _ in range(30):
            h_trial = project_feasible(h - alpha * g, target_sum)
            c5_trial = compute_c5(h_trial)
            if c5_trial < current_c5 - 1e-14:
                h = h_trial
                current_c5 = c5_trial
                found = True
                break
            alpha *= 0.5
        if not found:
            step_size *= 0.5
            if step_size < 1e-8:
                break
    return h, current_c5
