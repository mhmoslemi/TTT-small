import numpy as np

_G_TAU = 1.0
_G_LAM = 0.0

def evaluate_c5(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(mask.sum())
        if c > 0:
            step = diff / c
            if abs(step) < 1.0:
                h[mask] += step
            else:
                # Need to distribute more carefully
                # Just clip and hope it's close
                h[mask] += diff / c
        np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def _compute_obj_grad(h, n, tau, lam):
    dx = 2.0 / n
    target = n / 2.0
    c = np.correlate(h, 1.0 - h, 'full') * dx
    c_max = c.max()
    s = np.clip((c - c_max) / tau, -500.0, 500.0)
    e = np.exp(s)
    e_sum = e.sum()
    alpha = e / e_sum
    obj = c_max + tau * np.log(e_sum) + lam * (h.sum() - target) ** 2
    
    omh = 1.0 - h
    grad_s = np.zeros(n)
    active = np.where(alpha > 1e-12)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)

            grad_s[lo:hi + 1] += a * omh[lo - d:hi + 1 - d]

def _subgradient_polish(h0, n, deadline, max_steps=300):
    """Polish using subgradient of the true (non-smooth) max objective."""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    target = n / 2.0
    dx = 2.0 / n
    
    alpha = 0.003
    c = best_c5  # current objective value (used for step size adaptation)
    
    for step in range(max_steps):
        if time.time() > deadline - 2:
            break
        
        h = best_h
        cvec = np.correlate(h, 1.0 - h, 'full') * dx
        c_max = cvec.max()
        
        # Active set: k values within epsilon of max
        eps = 0.02  # relative to the value
        active = np.where(cvec >= c_max - eps)[0]
        M = len(active)
        if M == 0:
            break
        
        # Subgradient: average of gradients over active set
        omh = 1.0 - h
        grad = np.zeros(n)
        for k in active:
            d = k - (n - 1)
            lo = max(0, d)
            hi = min(n - 1, n - 1 + d)
            if lo <= hi:
                grad[lo:hi + 1] += omh[lo - d:hi + 1 - d]
            lo2 = max(0, -d)
            hi2 = min(n - 1, n - 1 - d)
            if lo2 <= hi2:
                grad[lo2:hi2 + 1] -= h[lo2 + d:hi2 + 1 + d]
        grad = grad * dx / M
        
        # Projected step
        h_new = project_feasible(h - alpha * grad, target)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Slowly decrease alpha
            alpha *= 0.98
        else:
            alpha *= 0.5
            if alpha < 1e-6:
                break
    
    return best_h, best_c5

def evaluate_c5(h):

def project_feasible(y, target):
    lo, hi = -2.0, 3.0

    target = n / 2.0
    c = np.correlate(h, 1.0 - h, 'full') * dx
    c_max = c.max()
    s = np.clip((c - c_max) / tau, -500.0, 500.0)
    e = np.exp(s)
    e_sum = e.sum()
    alpha = e / e_sum
    obj = c_max + tau * np.log(e_sum) + lam * (h.sum() - target) ** 2

    omh = 1.0 - h
    grad_s = np.zeros(n)
    active = np.where(alpha > 1e-12)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:

        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad_s[lo2:hi2 + 1] -= a * h[lo2 + d:hi2 + 1 + d]

    grad = grad_s * dx + 2.0 * lam * (h.sum() - target)
    return obj, grad
