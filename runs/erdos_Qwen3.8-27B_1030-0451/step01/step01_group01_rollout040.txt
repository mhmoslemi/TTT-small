import numpy as np
import time
from scipy.optimize import linprog

def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -max(np.abs(x)).max() - 1, max(np.abs(x)).max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            mask = y < 1.0
            if mask.sum() > 0:
                add = r / mask.sum()
                y[mask] = np.clip(y[mask] + add, 0.0, 1.0)
        else:
            mask = y > 0.0
            if mask.sum() > 0:
                sub = r / mask.sum()
                y[mask] = np.clip(y[mask] + sub, 0.0, 1.0)
    # Final fix
    r = target - y.sum()
    if abs(r) > 1e-15:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)

    l = abs(lag)

    g = np.zeros(n)

import numpy as np

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            mask = y < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.clip(y[mask] + r / cnt, 0.0, 1.0)
        else:
            mask = y > 1e-15
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.clip(y[mask] + r / cnt, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-14:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))


def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_lag(h, lag):
    n = len(h)
    l = abs(lag)
    g = np.zeros(n)
    if l >= n:
        return g
    if l == 0:
        g[:] = 1.0 - 2.0 * h
    else:
        g[l:] += 1.0 - h[:n - l]
        g[:n - l] -= h[l:]
    return g


def lp_trust(h, deadline, max_iters=500, init_radius=0.06, gap_frac=0.008):
    n = len(h)
    h_cur = h.copy()
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    radius = init_radius
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h_cur)
        vmax = v.max()
        gap = max(gap_frac * vmax, 0.002)
        active = np.where(v >= vmax - gap)[0]
        if len(active) > 120:
            order = np.argsort(v[active])
            active = active[order[-120:]]
        if len(active) == 0:
            active = np.array([np.argmax(v)])

        n_ac = len(active)
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0

        A_ub = np.zeros((n_ac, n_var))
        b_ub = np.zeros(n_ac)
        for i, k in enumerate(active):
            lag = k - (n - 1)
            A_ub[i, :n] = grad_lag(h_cur, lag)
            A_ub[i, n] = -1.0
            b_ub[i] = -v[k]

        lb = np.maximum(h_cur - radius, 0.0)
        ub = np.minimum(h_cur + radius, 1.0)

        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])

            break

                break
            continue

def pair_refine(h, deadline, max_rounds=30, n_pairs_per_lag=12):

            break

        improved = False
        for k in active:
            if time.time() > deadline:
                break
            lag = abs(k - (n - 1))
            if lag == 0 or lag >= n:
                continue
            # Get pairs at this lag, sorted by |h[i]-h[i+lag]| descending
            diffs = np.abs(h_cur[:n - lag] - h_cur[lag:])
            if n_pairs_per_lag < len(diffs):
                top_idx = np.argsort(diffs)[-n_pairs_per_lag:]
            else:
                top_idx = np.arange(len(diffs))
            for i in top_idx:
                j = i + lag
                d_val = h_cur[i] - h_cur[j]
                if abs(d_val) < 1e-10:
                    continue
                # Try moving towards equalizing + some overshoot

                t_max_neg = min(h_cur[i], 1.0 - h_cur[j])

                        t = -frac * t_max_neg  # move i down, j up

                            continue
                    if h_cur[i] + t < -1e-12 or h_cur[i] + t > 1 + 1e-12:
                        continue
                    if h_cur[j] - t < -1e-12 or h_cur[j] - t > 1 + 1e-12:
                        continue
                    h_trial = h_cur.copy()
                    h_trial[i] += t
                    h_trial[j] -= t
                    c_trial = evaluate_c5(h_trial)
                    if c_trial < best_c - 1e-13:

                        best_h = h_trial.copy()
                        h_cur = h_trial.copy()
