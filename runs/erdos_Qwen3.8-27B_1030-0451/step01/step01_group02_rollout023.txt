def make_feasible(y, target):
    """Project y onto {0<=x<=1, sum(x)=target} using bisection on shift."""
    y = np.asarray(y, dtype=float).ravel()
    n = y.size
    target = float(target)
    
    lo = -1.0 - y.max()
    hi = 1.0 - y.min()
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fine adjustment
    r = target - x.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size > 0:
                order = idx[np.argsort(x[idx])]
                rem = r
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
