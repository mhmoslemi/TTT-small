def project_feasible(h, target_sum):

    m = np.max(c)

def smooth_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)

def slp_step(h, c, target_sum):
    """One SLP iteration: linearize the max and solve LP."""
    N = len(h)
    dx = 2.0 / N
    n_corr = 2 * N - 1
    
    # Compute gradients of each c_k w.r.t. h
    # c_k = sum_n h[n]*(1-h[n+m]) * dx where m = k-(N-1)
    # dc_k/dh[i] = dx * [(1-h[i+m]) * valid(i as n) - h[i-m] * valid(i-m as n)]
    
    # Build the LP:
    # minimize t
    # subject to: c_k + sum_i (dc_k/dh[i]) * (h_new[i] - h[i]) <= t for all k
    #             0 <= h_new[i] <= 1
    #             sum h_new[i] = target_sum
    
    # Variables: [h_new[0], ..., h_new[N-1], t]  (N+1 variables)
    # The objective is to minimize t: c_obj = [0, 0, ..., 0, 1]
    
    c_obj = np.zeros(N + 1)
    c_obj[N] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each k: sum_i grad_k_i * h_new[i] - t <= c_k - sum_i grad_k_i * h[i]
    # which is: sum_i grad_k_i * h_new[i] + (-1)*t <= c_k - sum_i grad_k_i * h[i]
    
    grads = np.zeros((n_corr, N))
    for k in range(n_corr):
        m = k - (N - 1)
        for i in range(N):
            g = 0.0
            # Term from h[n] factor: (1-h[i+m]) if 0<=i+m<N
            if 0 <= i + m < N:
                g += (1.0 - h[i + m])
            # Term from (1-h[n+m]) factor: -h[i-m] if 0<=i-m<N
            if 0 <= i - m < N:
                g -= h[i - m]
            grads[k, i] = g * dx
    
    # A_ub has n_corr rows, N+1 columns
    A_ub = np.zeros((n_corr, N + 1))
    A_ub[:, :N] = grads
    A_ub[:, N] = -1.0  # coefficient of t
    
    b_ub = c.copy() - grads @ h
    
    # Bounds: 0 <= h_new[i] <= 1, t is free (or t >= 0)
    bounds_lb = np.zeros(N + 1)
    bounds_ub = np.ones(N + 1)
    bounds_ub[N] = np.inf  # t is unbounded above
    bounds = list(zip(bounds_lb, bounds_ub))
    
    # Equality: sum h_new[i] = target_sum
    A_eq = np.zeros((1, N + 1))
    A_eq[0, :N] = 1.0
    b_eq = np.array([target_sum])
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success:
        return result.x[:N]
    return None

        if 0 <= i - m < N:

        gi *= dx

        if 0 <= j + m < N:
            gj += (1.0 - h[j + m])
