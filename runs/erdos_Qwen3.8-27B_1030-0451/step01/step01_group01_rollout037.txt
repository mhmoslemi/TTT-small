```python
import time
from scipy.optimize import minimize, linprog

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def project_sum_box(h, target):
    # Project onto {x: 0<=x<=1, sum(x)=target}
    n = len(h)
    h = np.array(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-15:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-15:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    # Final fix
    r2 = target - y.sum()
    if abs(r2) > 1e-14:
        if r2 > 0:
            idx = np.where(y < 1.0 - 1e-12)[0]
        else:
            idx = np.where(y > 1e-12)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def pair_exchange_search(h0, deadline, max_iters=100000):

    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    
    # Precompute lag for each correlation index
    lags = np.arange(2*n-1) - (n-1)

        v = corr_values(h)

        if c_cur < best_c - 1e-13:

            best_h = h.copy()

        threshold = c_cur - max(0.002, 0.01 * c_cur)
        active_idx = np.where(v >= threshold)[0]
        
        if len(active_idx) == 0:
            active_idx = np.argmax(v)
            active_idx = np.array([active_idx])
        
        # For each active lag, try pair exchanges
        improved = False
        for k_idx in active_idx:

            d = abs(lag)
            if d == 0:

            # Pairs: (i, i+d) for valid i
```

```python
    corr_index is 0-indexed in the full correlation output (length 2n-1).
    """

    d = abs(lag)

    if d == 0:
        g[:] = dx * (1.0 - 2.0 * h)
```

```python
def solve_lp_step(h, active_indices, v, radius, n):
    """Solve LP for trust region step."""
    m = len(active_indices)
    if m == 0:
        return None
    
    # Build A_ub: m rows, (n+1) columns
    A_ub = np.zeros((m, n + 1))
    b_ub = np.zeros(m)
    for i, k in enumerate(active_indices):
        g = corr_gradient(h, k)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = -v[k]
    
    # Objective: minimize t
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Equality: sum(z) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    
    # Bounds
    lb = np.maximum(h - radius, 0.0)
    ub = np.minimum(h + radius, 1.0)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, None))  # t >= 0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0.0],
                  bounds=bounds, method='highs')
    
    if res is not None and res.success and res.x is not None:
        return res.x[:n]
    return None
```
