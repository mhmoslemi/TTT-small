    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def make_direction(h, N, rng, mode, corr_vals, tau_scale=1.0):
    d = np.zeros(N)
    if mode == 0:
        cmax = np.max(corr_vals)
        tau = 0.008 * tau_scale
        weights = np.exp(np.clip((corr_vals - cmax) / tau, -300, 0))
        total = weights.sum()
        if total < 1e-15:
            weights = np.zeros_like(corr_vals)
            weights[np.argmax(corr_vals)] = 1.0
        else:
            weights /= total
        d = -gradient_weighted(h, weights)
    elif mode == 1:
        k = max(3, N // 20)
        top_idx = np.argsort(corr_vals)[-k:]
        weights = np.zeros(2 * N - 1)
        weights[top_idx] = 1.0 / k
        d = -gradient_weighted(h, weights)
    elif mode == 2:
        cmax = np.max(corr_vals)
        tau = 0.003
        weights = np.exp(np.clip((corr_vals - cmax) / tau, -300, 0))
        total = weights.sum()
        if total < 1e-15:
            weights = np.zeros_like(corr_vals)
            weights[np.argmax(corr_vals)] = 1.0
        else:
            weights /= total
        d = -gradient_weighted(h, weights) + rng.normal(0, 0.3, N)
    elif mode == 3:
        freq = rng.integers(1, 5)
        phase = rng.uniform(0, 2 * np.pi)
        x = (np.arange(N) + 0.5) * (2.0 / N)
        d = np.sin(2 * np.pi * freq * x + phase)
    elif mode == 4:
        freq = rng.integers(4, 18)
        phase = rng.uniform(0, 2 * np.pi)
        x = (np.arange(N) + 0.5) * (2.0 / N)
        d = np.sin(2 * np.pi * freq * x + phase)
    elif mode == 5:
        d = rng.normal(0, 1, N)
        k = max(3, N // 40)
        kernel = np.ones(k) / k
        d = np.convolve(d, kernel, mode='same')
    elif mode == 6:
        d = rng.normal(0, 1, N)
    elif mode == 7:
        # Cosine-weighted gradient
        cmax = np.max(corr_vals)
        tau = 0.015
        weights = np.exp(np.clip((corr_vals - cmax) / tau, -300, 0))
        total = weights.sum()
        if total < 1e-15:
            weights = np.zeros_like(corr_vals)
            weights[np.argmax(corr_vals)] = 1.0
        else:
            weights /= total
        d = -gradient_weighted(h, weights)
    else:
        d = rng.normal(0, 1, N)

    d = d - np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        d = rng.normal(0, 1, N)
        d = d - np.mean(d)
        norm = np.linalg.norm(d)
    d = d / norm
    return d


def exact_line_search(h, d, n_coarse=15, n_fine=15):
    N = len(h)
    t_max = np.inf
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    if np.any(pos_mask):
        t_max = min(t_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    if np.any(neg_mask):
        t_max = min(t_max, np.min(-h[neg_mask] / d[neg_mask]))
    if not np.isfinite(t_max) or t_max <= 1e-15:
        return None, None

    t_coarse = np.linspace(0, t_max, n_coarse)
    best_t = 0.0
    best_obj = compute_c5(h)

    for t in t_coarse:
        h_t = h + t * d
        if h_t.min() < -1e-10 or h_t.max() > 1.0 + 1e-10:
            continue
        obj = compute_c5(h_t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    if best_t <= 0:
        return None, None

    t_width = t_max / n_coarse
    t_low = max(0, best_t - t_width)
    t_high = min(t_max, best_t + t_width)
    t_fine = np.linspace(t_low, t_high, n_fine)

    for t in t_fine:
        h_t = h + t * d
        if h_t.min() < -1e-10 or h_t.max() > 1.0 + 1e-10:
            continue
        obj = compute_c5(h_t)
        if obj < best_obj:
            best_obj = obj
            best_t = t

    if best_t <= 0:
        return None, None

    h_new = h + best_t * d
    h_new = project_feasible(h_new, N / 2.0)
    final_obj = compute_c5(h_new)
    return h_new, final_obj


def do_pair_transfer(h, N, rng, c5_work, n_tries=50):
    best_h = h
    best_c5 = c5_work
    for _ in range(n_tries):
        i = rng.integers(0, N)
        j = rng.integers(0, N)
        if i == j:
            continue
        amt = rng.uniform(0.002, 0.05)
        actual = min(amt, h[i], 1.0 - h[j])
        if actual <= 1e-10:
            continue
        h_new = h.copy()
        h_new[i] -= actual
        h_new[j] += actual
        obj = compute_c5(h_new)
        if obj < best_c5 - 1e-12:
            best_h = h_new
            best_c5 = obj
    return best_h, best_c5


def descent_phase(h, N, rng, n_iters, c5_work, deadline, label=""):
    best_h = h.copy()
    best_c5 = c5_work
    stag = 0
    tau_scale = 1.0

    for it in range(n_iters):
        if time.time() > deadline:
            break

        # Adapt tau scale: decrease over iterations for finer active set
        tau_scale = max(0.3, 1.0 - it / (2.0 * n_iters))

        mode = it % 8
        if it % 40 == 35:
            mode = 6  # occasional pure random
        elif it % 40 == 30:
            mode = 7  # wide softmax

        corr_vals = compute_corr_vals(h)
        d = make_direction(h, N, rng, mode, corr_vals, tau_scale)
        h_new, obj_new = exact_line_search(h, d, n_coarse=12, n_fine=12)

        if h_new is not None and obj_new < c5_work - 1e-11:
            h = h_new
            c5_work = obj_new
            stag = 0
            if c5_work < best_c5 - 1e-13:
                best_c5 = c5_work
                best_h = h.copy()
        else:
            stag += 1

        # Periodic pair transfers when stagnant
        if stag > 15 and stag % 15 == 0:
            h, c5_work = do_pair_transfer(h, N, rng, c5_work, n_tries=30)
            if c5_work < best_c5 - 1e-13:
                best_c5 = c5_work
                best_h = h.copy()
                stag = 0

        if stag > 80:
            # Reset to best and try different directions
            h = best_h.copy()
            c5_work = best_c5
            stag = 0

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 30.0

    h_init = np.array(initial_h_values, dtype=float)
    N_init = len(h_init)
    h_init = project_feasible(h_init, N_init / 2.0)
    c5_init = compute_c5(h_init)

    best_h = h_init.copy()
    best_c5 = c5_init
    best_N = N_init

    rng = np.random.default_rng(seed)

    # Phase 1: Descent at initial resolution
    h1, c5_1 = descent_phase(h_init, N_init, rng, 800, c5_init, deadline - 300, "P1")
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
        best_N = N_init

    # Phase 2: Try N=500
    if time.time() < deadline - 120:
        N2 = 500
        h2_start = upscale(best_h, N2)
        c5_2_start = compute_c5(h2_start)
        if c5_2_start < best_c5:
            best_c5 = c5_2_start
            best_h = h2_start.copy()
            best_N = N2
        h2, c5_2 = descent_phase(h2_start, N2, rng, 400, c5_2_start, deadline - 80, "P2")
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            best_N = N2

    # Phase 3: Try N=400
    if time.time() < deadline - 60:
        N3 = 400
        h3_start = upscale(best_h, N3)
        c5_3_start = compute_c5(h3_start)
        if c5_3_start < best_c5:
            best_c5 = c5_3_start
            best_h = h3_start.copy()
            best_N = N3
        h3, c5_3 = descent_phase(h3_start, N3, rng, 300, c5_3_start, deadline - 40, "P3")
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
            best_N = N3

    # Phase 4: Re-descend at best found resolution with fresh random directions
    if time.time() < deadline - 20:
        rng2 = np.random.default_rng(seed + 1000)
        h4, c5_4 = descent_phase(best_h.copy(), best_N, rng2, 300, best_c5, deadline - 10, "P4")
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
            best_N = best_N

    # Phase 5: Final fine pair-transfer polish
    if time.time() < deadline - 5:
        h5, c5_5 = do_pair_transfer(best_h.copy(), best_N, rng, best_c5, n_tries=200)
        if c5_5 < best_c5:
            best_c5 = c5_5
            best_h = h5

    # Final verification
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    final_c5 = compute_c5(best_h)

    return best_h, final_c5, len(best_h)
