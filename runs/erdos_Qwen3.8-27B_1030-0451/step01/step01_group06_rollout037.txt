import numpy as np
import time
from scipy.optimize import linprog, minimize
from scipy.ndimage import uniform_filter1d

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def smooth_max(corr, beta):
    """Log-sum-exp approximation of max."""
    m = np.max(corr)
    return m + np.log(np.sum(np.exp(beta * (corr - m)))) / beta

def smooth_max_grad(corr, beta):
    """Gradient of log-sum-exp w.r.t. corr."""
    m = np.max(corr)
    weights = np.exp(beta * (corr - m))
    weights /= weights.sum()
    return weights

def compute_corr_and_grads(h):

    # corr_k = sum_{i: 0<=i<n-k} h[i]*(1-h[i+k]) * dx (for k>=0)

def smooth_objective_and_grad(h, beta):
    """Compute smooth objective and its gradient."""
    corr, grads = compute_corr_and_grads(h)
    obj = smooth_max(corr, beta)
    weights = smooth_max_grad(corr, beta)
    grad = weights @ grads  # shape (n,)
    return obj, grad

def optimize_with_lbfgsb(h, beta_start, beta_end, budget_remaining, rng):
    """Optimize using L-BFGS-B with smooth max objective."""
    n = len(h)
    target = n / 2.0
    current_h = h.copy()
    current_val = evaluate_exact(current_h)
    
    # Anneal beta from beta_start to beta_end
    n_anneal = 20
    betas = np.geomspace(beta_start, beta_end, n_anneal)
    
    for beta in betas:
        if budget_remaining() < 5:
            break
        # Define objective and gradient
        def obj_func(x):
            # Project to feasible set
            x_proj = project_feasible(x, target)
            val, g = smooth_objective_and_grad(x_proj, beta)
            return val
        def grad_func(x):
            x_proj = project_feasible(x, target)
            val, g = smooth_objective_and_grad(x_proj, beta)
            return g
        
        result = minimize(obj_func, current_h, jac=grad_func, method='L-BFGS-B',
                         bounds=[(0, 1)] * n,
                         options={'maxiter': 200, 'maxfun': 5000, 'ftol': 1e-12})
        if result.success or result.fun < obj_func(current_h):
            candidate = project_feasible(result.x, target)
            actual_val = evaluate_exact(candidate)
            if actual_val < current_val:
                current_val = actual_val
                current_h = candidate

def binary_round(h, target):
    """Round h to binary and adjust to meet the sum constraint."""

    # Sort by value, set top n/2 to 1, rest to 0
    idx = np.argsort(h)[::-1]
    binary = np.zeros(n)
    binary[idx[:int(target)]] = 1.0
    return binary

def optimize_binary(h, budget_remaining, rng):

    n = len(h_start)
    target = int(n / 2)

    best_energy = current_energy

    T_end = 0.001

        if budget_remaining() < 2:

        ones.add(j)

            b[j] = 0.0
            ones.add(i)
            ones.discard(j)

Actually, let me just precompute the correlation and update it incrementally.

When we change h[i] from 1 to 0 and h[j] from 0 to 1:
- For corr at lag k (full correlation index m = k + n - 1):
  - The term h[i]*(1-h[i+k]) changes: if i+k is valid, it goes from 1*(1-h[i+k]) to 0*(1-h[i+k]) = 0, so change = -(1-h[i+k])
  - The term h[j]*(1-h[j+k]) changes: if j+k is valid, it goes from 0*(1-h[j+k]) to 1*(1-h[j+k]), so change = +(1-h[j+k])

Wait, I need to be more careful. The full correlation is:
corr[m] = sum_i h[i] * (1 - h[i + m - (n-1)])

Let me define lag l = m - (n-1). Then:
corr[l] = sum_{i: valid} h[i] * (1 - h[i + l])

where "valid" means both i and i+l are in [0, n-1].

When h changes at positions i (1→0) and j (0→1):
For corr at lag l:
- If i is in the "first" position (i is valid and i+l is valid): 
  - Before: h[i]*(1-h[i+l]) = 1*(1-h[i+l])
  - After: 0*(1-h[i+l]) = 0
  - Change: -(1-h[i+l]) = h[i+l] - 1
- If i is in the "second" position (i-l is valid and i is the second):
  - Before: (1-h[i-l])*h[i] = (1-h[i-l])*1
  - After: (1-h[i-l])*0 = 0
  - Change: -(1-h[i-l]) = h[i-l] - 1
- Similarly for j (0→1):
  - If j is in the "first" position: change = +(1-h[j+l])
  - If j is in the "second" position: change = +(1-h[j-l])

So the total change in corr[l] is the sum of these contributions.

This is O(1) per lag, so O(n) per swap total (for all lags). With n=120, that's 120 operations per swap, which is very fast.

import numpy as np
import time
from scipy.optimize import linprog

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def compute_corr_grads(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m_total = 2 * n - 1
    grads = np.zeros((m_total, n))
    for m in range(m_total):
        k = m - (n - 1)
        if k == 0:
            grads[m] = dx * (1.0 - 2.0 * h)
        elif k > 0:
            grads[m][:n-k] += dx * (1.0 - h[k:])
            grads[m][k:] += -dx * h[:n-k]
        else:
            t = -k
            grads[m][t:] += dx * (1.0 - h[:n-t])
            grads[m][:n-t] += -dx * h[t:]
    return corr, grads

def optimize_lp(h, budget_remaining, rng, n_rounds=200):
    n = len(h)
    target = n / 2.0
    trust = 0.3
    stagnant = 0
    max_stag = 12
    alphas = [1.0, 0.7, 0.5, 0.3, 0.15, 0.07, 0.03]
    best_h = h.copy()
    best_v = evaluate_exact(h)
    
    for _ in range(n_rounds):
        if budget_remaining() < 2:
            break
        corr = correlation_array(h)
        cur_max = float(np.max(corr))
        if cur_max < best_v:
            best_v = cur_max
            best_h = h.copy()
        
        m = 2 * n - 1
        k_active = max(1, int(m * 0.5))
        top_idx = np.argpartition(corr, -k_active)[-k_active:]
        corr_a = corr[top_idx]
        _, grads = compute_corr_grads(h)
        grads_a = grads[top_idx]
        
        nvars = n + 1
        c_obj = np.zeros(nvars)
        c_obj[n] = 1.0
        A_ub = np.zeros((k_active, nvars))
        A_ub[:, :n] = grads_a
        A_ub[:, n] = -1.0
        b_ub = -corr_a - 1e-7
        lower = np.maximum(-h, -trust)
        upper = np.minimum(1.0 - h, trust)
        A_eq = np.zeros((1, nvars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        bounds = list(zip(lower, upper)) + [(None, None)]
        
        d = None
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                d_c = res.x[:n]
                if np.all(h + d_c >= -1e-9) and np.all(h + d_c <= 1 + 1e-9):
                    d = d_c
        except Exception:
            pass
        
        accepted = False
        if d is not None and np.max(np.abs(d)) > 1e-14:
            for alpha in alphas:
                h_new = h + alpha * d
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_feasible(h_new, target)
                val = evaluate_exact(h_new)
                if val < best_v - 1e-15:
                    best_v = val
                    best_h = h_new.copy()
                    h = h_new.copy()
                    accepted = True
                    stagnant = 0
                    break
        if not accepted:
            trust = max(0.01, trust * 0.6)
            stagnant += 1
        else:
            trust = min(0.4, trust * 1.1)
        
        if stagnant >= max_stag:
            h = project_feasible(best_h + 0.1 * rng.standard_normal(n), target)
            trust = 0.25
            stagnant = 0
    
    return best_h, best_v

import numpy as np

from scipy.optimize import linprog

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def compute_corr_grads(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m_total = 2 * n - 1
    grads = np.zeros((m_total, n))
    for m in range(m_total):
        k = m - (n - 1)
        if k == 0:
            grads[m] = dx * (1.0 - 2.0 * h)
        elif k > 0:
            grads[m][:n-k] += dx * (1.0 - h[k:])
            grads[m][k:] += -dx * h[:n-k]
        else:
            t = -k
            grads[m][t:] += dx * (1.0 - h[:n-t])
            grads[m][:n-t] += -dx * h[t:]
    return corr, grads

    target = n / 2.0
    trust = 0.25
    stagnant = 0
    max_stag = 12

    best_h = h.copy()
    best_v = evaluate_exact(h)
    for _ in range(n_rounds):
        if budget_remaining() < 2:

        corr = correlation_array(h)
        cur_max = float(np.max(corr))
        if cur_max < best_v:
            best_v = cur_max
            best_h = h.copy()
        m = 2 * n - 1
        k_active = max(1, int(m * 0.5))
        top_idx = np.argpartition(corr, -k_active)[-k_active:]

                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        T = max(T * cooling, T_min)
    return best_b, best

def binary_sa_4swap(h_start, budget_remaining, rng, T_init=0.07, T_min=0.004, max_iter=50000):
    n = len(h_start)
    target = int(n / 2.0)
    b = np.zeros(n)
    idx = np.argsort(h_start)[::-1][:target]
    b[idx] = 1.0
    ones_mask = b.astype(bool)
    corr = correlation_array(b)
    max_corr = float(np.max(corr))
    best = max_corr
    best_b = b.copy()
    T = T_init
    cooling = (T_min / T_init) ** (1.0 / max(max_iter, 1))
    ones_idx = np.where(ones_mask)[0]
    zeros_idx = np.where(~ones_mask)[0]
    for it in range(max_iter):
        if budget_remaining() < 1:
            break
        if len(ones_idx) < 2 or len(zeros_idx) < 2:
            break
        i1, i2 = rng.choice(ones_idx, size=2, replace=False)
        j1, j2 = rng.choice(zeros_idx, size=2, replace=False)
        b[i1] = 0.0
        b[i2] = 0.0
        b[j1] = 1.0
        b[j2] = 1.0
        new_corr = correlation_array(b)
        new_max = float(np.max(new_corr))
        delta = new_max - max_corr
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            max_corr = new_max
            corr = new_corr
            ones_mask[i1] = False
            ones_mask[i2] = False
            ones_mask[j1] = True
            ones_mask[j2] = True
            ones_idx = np.where(ones_mask)[0]
            zeros_idx = np.where(~ones_mask)[0]
            if new_max < best:
                best = new_max
                best_b = b.copy()
        else:
            b[i1] = 1.0
            b[i2] = 1.0
            b[j1] = 0.0
            b[j2] = 0.0
        T = max(T * cooling, T_min)
    return best_b, best

def binary_sa_informed(h_start, budget_remaining, rng, T_init=0.04, T_min=0.002, max_iter=80000):
    n = len(h_start)
    target = int(n / 2.0)
    b = np.zeros(n)
    idx = np.argsort(h_start)[::-1][:target]
    b[idx] = 1.0
    ones_mask = b.astype(bool)
    corr = correlation_array(b)
    max_corr = float(np.max(corr))
    best = max_corr
    best_b = b.copy()
    T = T_init
    cooling = (T_min / T_init) ** (1.0 / max(max_iter, 1))
    ones_idx = np.where(ones_mask)[0]
    zeros_idx = np.where(~ones_mask)[0]
    for it in range(max_iter):
        if budget_remaining() < 1:
            break
        # Informed move: pick the worst lag, try to fix it
        worst_m = int(np.argmax(corr))
        worst_lag = worst_m - (n - 1)
        # Find a position i where b[i]=1 and b[i+lag]=0 (if lag>0) or similar
        found = False
        if worst_lag > 0:
            candidates_i = [i for i in ones_idx if i + worst_lag < n and b[i + worst_lag] == 0]
        elif worst_lag < 0:
            t = -worst_lag
            candidates_i = [i for i in ones_idx if i - t >= 0 and b[i - t] == 0]
        else:
            candidates_i = list(ones_idx)
        if candidates_i:
            i = candidates_i[int(rng.integers(len(candidates_i)))]
        else:
            i = int(rng.choice(ones_idx))
        # Pick a zero to swap with (prefer one that doesn't worsen other lags)
        if rng.random() < 0.7:
            # Informed: pick zero near the worst lag region
            if worst_lag > 0 and i + worst_lag < n:
                j_candidates = [x for x in zeros_idx if x != i]
            else:
                j_candidates = list(zeros_idx)
        else:
            j_candidates = list(zeros_idx)
        if not j_candidates:
            j_candidates = list(zeros_idx)
        j = j_candidates[int(rng.integers(len(j_candidates)))]
        if j == i:
            j = int(rng.choice(zeros_idx))
            if j == i:
                continue
        b[i] = 0.0
        b[j] = 1.0
        new_corr = correlation_array(b)
        new_max = float(np.max(new_corr))
        delta = new_max - max_corr
        if delta < 0 or (T > T_min and rng.random() < np.exp(-delta / T)):
            max_corr = new_max
            corr = new_corr
            ones_mask[i] = False
            ones_mask[j] = True
            ones_idx = np.where(ones_mask)[0]
            zeros_idx = np.where(~ones_mask)[0]
            if new_max < best:
                best = new_max
                best_b = b.copy()
        else:
            b[i] = 1.0
            b[j] = 0.0
        T = max(T * cooling, T_min)
    return best_b, best

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 3.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64)
        if h0.size > 0:
            h0 = h0.flatten()
            h0 = project_feasible(h0, len(h0) / 2.0)
    except Exception:
        h0 = None

    if h0 is None:
        h0 = np.full(120, 0.5)

    n = len(h0)
    best_h = h0.copy()
    best_val = evaluate_exact(best_h)

    # Phase 1: Quick LP (40s)
    if remaining() > 40:
        h, v = optimize_lp(h0.copy(), remaining, rng, n_rounds=60)
        if v < best_val:
            best_val = v
            best_h = h.copy()

    # Phase 2: SA from initial (100s)
    if remaining() > 100:
        h_b, v_b = binary_sa(best_h.copy(), remaining, rng, T_init=0.06, max_iter=80000)
        if v_b < best_val:
            best_val = v_b
            best_h = h_b.copy()

    # Phase 3: Informed SA (100s)
    if remaining() > 100:
        h_b, v_b = binary_sa_informed(best_h.copy(), remaining, rng, T_init=0.05, max_iter=60000)
        if v_b < best_val:
            best_val = v_b
            best_h = h_b.copy()

    # Phase 4: 4-swap SA (80s)
    if remaining() > 80:
        h_b, v_b = binary_sa_4swap(best_h.copy(), remaining, rng, T_init=0.08, max_iter=30000)
        if v_b < best_val:
            best_val = v_b
            best_h = h_b.copy()

    # Phase 5: SA from random starts (120s)
    if remaining() > 120:
        for r in range(4):
            if remaining() < 30:
                break
            b_rand = np.zeros(n)
            idx_r = rng.choice(n, size=n // 2, replace=False)
            b_rand[idx_r] = 1.0
            h_b, v_b = binary_sa(b_rand, remaining, rng, T_init=0.08, max_iter=25000)
            if v_b < best_val:
                best_val = v_b
                best_h = h_b.copy()

    # Phase 6: More SA from best (100s)
    if remaining() > 100:
        h_b, v_b = binary_sa(best_h.copy(), remaining, rng, T_init=0.04, T_min=0.001, max_iter=60000)
        if v_b < best_val:
            best_val = v_b
            best_h = h_b.copy()

    # Phase 7: Continuous LP refinement (80s)
    if remaining() > 80:
        h, v = optimize_lp(best_h.copy(), remaining, rng, n_rounds=100)
        if v < best_val:
            best_val = v
            best_h = h.copy()

    # Phase 8: Upscale to 2x (120s)
    if remaining() > 120:
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, len(h_up) / 2.0)
        h, v = optimize_lp(h_up, remaining, rng, n_rounds=60)
        if v < best_val:
            h_dn = h[:n * 2].reshape(n, 2).mean(axis=1)
            h_dn = project_feasible(h_dn, n / 2.0)
            v_dn = evaluate_exact(h_dn)
            if v_dn < best_val:
                best_val = v_dn
                best_h = h_dn.copy()
            else:
                best_val = v
                best_h = h.copy()

    # Phase 9: Final LP polish (40s)
    if remaining() > 40:
        h, v = optimize_lp(best_h.copy(), remaining, rng, n_rounds=80)
        if v < best_val:
            best_val = v
            best_h = h.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)
    return best_h, float(best_val), len(best_h)
