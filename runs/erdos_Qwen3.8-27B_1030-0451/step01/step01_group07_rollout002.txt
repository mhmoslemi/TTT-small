import time
import math
import numpy as np
from scipy.optimize import minimize_scalar

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size > 0 and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def evaluate(h):
    n = len(h)
    return float(np.correlate(h, 1.0 - h, mode='full').max() * 2.0 / n)

def project(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    # Fix residual
    d = out.sum() - target
    if abs(d) > 1e-10:
        if d > 0:
            mask = out > 1e-12
            if mask.sum() > 0:
                out[mask] -= d / mask.sum()
        else:
            mask = out < 1.0 - 1e-12
            if mask.sum() > 0:
                out[mask] += (-d) / mask.sum()
    out = np.clip(out, 0.0, 1.0)
    return out

def make_initial(n, rng):
    """Create a random initial feasible vector."""
    h = np.zeros(n)
    # Start with alternating pattern
    h[0::2] = 1.0
    h = project(h, n / 2.0)
    return h

def active_gradient(h, k=5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    # Top-k lags
    top = np.argsort(c)[::-1][:k]
    grad = np.zeros(n)
    for lag_idx in top:
        w = c[lag_idx]
        if w <= 1e-12:
            continue
        p = lag_idx - center
        if p == 0:
            grad += w * (1.0 - 2.0 * h)
        elif p > 0:
            grad[p:] += w * (1.0 - h[:-p])
            grad[:-p] -= w * h[p:]
        else:
            p = -p
            grad[:-p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:-p]
    return grad * dx, c

def polish_pairs(h, rng, t_limit, deadline, top_k=5, n_candidates=30):
    """Iterative pair-transfer optimization."""
    n = len(h)
    target = n / 2.0
    cur = evaluate(h)
    best_h = h.copy()
    best = cur
    no_improve = 0
    
    while time.monotonic() < t_limit and time.monotonic() < deadline:
        grad, cvec = active_gradient(h, top_k)
        
        # Candidate pairs: positions with large |grad|
        gidx = np.argsort(np.abs(grad))[::-1][:min(50, n)]
        
        improved_this_round = False
        for _ in range(n_candidates):
            if time.monotonic() > t_limit or time.monotonic() > deadline:
                break
            a = gidx[rng.integers(len(gidx))]
            b = gidx[rng.integers(len(gidx))]
            i, j = int(a), int(b)
            if i == j:
                continue
            
            # Direction: reduce max by moving from high-grad to low-grad
            if grad[i] >= grad[j]:
                donor, recv = i, j
            else:
                donor, recv = j, i
            
            # Feasible t range: h[donor]-t in [0,1], h[recv]+t in [0,1]
            t_lo = max(h[donor] - 1.0, -h[recv])
            t_hi = min(h[donor], 1.0 - h[recv])
            if t_hi - t_lo < 1e-12:
                continue
            
            # Define objective along the line
            h_base = h.copy()
            def obj(t):
                h_base[donor] = h[donor] - t
                h_base[recv] = h[recv] + t
                return float(np.correlate(h_base, 1.0 - h_base, mode='full').max())
            
            # Quick scan
            best_t = 0.0
            best_v = obj(0.0)
            for t in [t_lo, t_hi, 0.5*(t_lo+t_hi), 0.25*(t_lo+t_hi), 0.75*(t_lo+t_hi)]:
                v = obj(t)
                if v < best_v:
                    best_v = v
                    best_t = t
            
            # Refine
            if time.monotonic() < deadline - 0.05:
                try:
                    res = minimize_scalar(obj, bounds=(t_lo, t_hi), method='bounded',
                                   options={'maxiter': 20, 'xatol': 1e-8})
                    if res.fun < best_v:
                        best_v = res.fun
                        best_t = res.x
                except Exception:
                    pass
            
            # Accept
            cur_v = obj(0.0)
            if best_v < cur_v - 1e-10:
                h[donor] = h[donor] - best_t
                h[recv] = h[recv] + best_t
                h = np.clip(h, 0.0, 1.0)
                h = project(h, target)
                new_cur = evaluate(h)
                if new_cur < best - 1e-12:
                    best = new_cur
                    best_h = h.copy()
                cur = new_cur
                improved_this_round = True
                no_improve = 0
        
        if not improved_this_round:
            no_improve += 1
            if no_improve >= 8:
                # Perturb
                h = h + rng.normal(0, 0.03, n)
                h = project(h, target)
                cur = evaluate(h)
                no_improve = 0
                top_k = min(15, top_k + 3)
            elif no_improve >= 3:
                top_k = min(15, top_k + 1)
    
    # Final check
    final = evaluate(h)
    if final < best:
        best = final
        best_h = h.copy()
    
    return best_h, best

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)
