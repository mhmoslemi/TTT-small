```python
def minimax_1d_envelope(slopes, intercepts, t_lo, t_hi):
    """Find t in [t_lo, t_hi] minimizing max_k (slopes[k]*t + intercepts[k])."""
    m = len(slopes)
    
    # Evaluate at boundaries
    val_lo = np.max(slopes * t_lo + intercepts)
    val_hi = np.max(slopes * t_hi + intercepts)
    
    best_val = min(val_lo, val_hi)
    best_t = t_lo if val_lo <= val_hi else t_hi
    
    # Build upper envelope
    order = np.argsort(slopes)
    # Remove dominated (same slope, lower intercept)
    lines = []
    for idx in order:
        if lines and slopes[idx] == slopes[lines[-1]]:
            if intercepts[idx] > intercepts[lines[-1]]:
                lines[-1] = idx
        else:
            lines.append(idx)
    
    # Build envelope: list of (x_start, idx)
    env = []
    for idx in lines:
        while env:
            last_idx = env[-1][1]
            last_start = env[-1][0]
            if slopes[idx] == slopes[last_idx]:
                break
            x_int = (intercepts[last_idx] - intercepts[idx]) / (slopes[idx] - slopes[last_idx])
            if x_int <= last_start + 1e-15:
                env.pop()
            else:
                break
        if not env:
            env.append((-np.inf, idx))
        else:
            last_idx = env[-1][1]
            x_int = (intercepts[last_idx] - intercepts[idx]) / (slopes[idx] - slopes[last_idx])
            if x_int > env[-1][0]:
                env.append((x_int, idx))
    
    # Evaluate at envelope vertices within [t_lo, t_hi]
    for k in range(len(env)):
        x_start = env[k][0]
        idx = env[k][1]
        x_end = env[k+1][0] if k+1 < len(env) else np.inf
        
        # Check x_start if in range
        if x_start >= t_lo - 1e-15 and x_start <= t_hi + 1e-15:
            t_cand = np.clip(x_start, t_lo, t_hi)
            v = np.max(slopes * t_cand + intercepts)
            if v < best_val - 1e-12:
                best_val = v
                best_t = t_cand
        
        # Check x_end if in range and finite
        if np.isfinite(x_end) and x_end >= t_lo - 1e-15 and x_end <= t_hi + 1e-15:
            t_cand = np.clip(x_end, t_lo, t_hi)
            v = np.max(slopes * t_cand + intercepts)
            if v < best_val - 1e-12:
                best_val = v
                best_t = t_cand
    
    return best_t, best_val
```

```python
def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    x = np.array(x, dtype=float).copy()
```
