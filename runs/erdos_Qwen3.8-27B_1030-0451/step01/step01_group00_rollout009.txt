import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(p, n):
    """Project p onto {0<=x<=1, sum(x)=n/2}"""
    target = n / 2.0
    lo = float(np.min(p)) - 1.5
    hi = float(np.max(p)) + 1.5
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
            # Second pass for any that hit bounds
            diff2 = target - float(np.sum(proj))
            if abs(diff2) > 1e-12:
                mask2 = (proj > 1e-9) & (proj < 1.0 - 1e-9)
                nm2 = int(np.sum(mask2))
                if nm2 > 0:
                    proj[mask2] = np.clip(proj[mask2] + diff2 / nm2, 0.0, 1.0)
    return proj

def smooth_gradient(h, tau):
    """Compute smoothed gradient of max correlation."""
    n = len(h)
    corr = compute_corr(h)
    cmax = np.max(corr)
    cs = corr - cmax
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n), corr
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad, corr

def line_search_1d(h, d, n, dx):
    """Exact line search along direction d (sum d = 0).
    Returns (best_alpha, best_h, best_c5)."""
    # Find feasible range
    pos_mask = d > 1e-12
    neg_mask = d < -1e-12
    
    alpha_max_pos = float('inf')
    if np.any(pos_mask):
        alpha_max_pos = np.min((1.0 - h[pos_mask]) / d[pos_mask])
    alpha_max_neg = float('inf')
    if np.any(neg_mask):
        alpha_max_neg = np.min(h[neg_mask] / (-d[neg_mask]))
    
    alpha_max_pos = max(alpha_max_pos, 0.0)
    alpha_max_neg = max(alpha_max_neg, 0.0)
    
    if alpha_max_pos < 1e-14 and alpha_max_neg < 1e-14:
        return 0.0, h.copy(), compute_c5(h)
    
    # Coarse grid
    candidates = [0.0]
    if alpha_max_pos > 1e-10:
        for t in [0.1, 0.25, 0.4, 0.6, 0.8, 1.0]:
            candidates.append(alpha_max_pos * t)
    if alpha_max_neg > 1e-10:
        for t in [0.1, 0.25, 0.4, 0.6, 0.8, 1.0]:
            candidates.append(-alpha_max_neg * t)
    
    best_alpha = 0.0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for a in candidates:
        h_try = h + a * d
        h_try = project_feasible(h_try, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_c5 = c5_try
            best_alpha = a
            best_h = h_try.copy()
    
    # Refine around best
    if best_alpha != 0.0 and (alpha_max_pos > 1e-10 or alpha_max_neg > 1e-10):
        span = max(alpha_max_pos, alpha_max_neg)
        step = span / 30.0
        for offset in [-2, -1, 0, 1, 2]:
            a = best_alpha + offset * step
            if a <= -alpha_max_neg - 1e-12 or a >= alpha_max_pos + 1e-12:
                continue
            h_try = h + a * d
            h_try = project_feasible(h_try, n)
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                best_c5 = c5_try
                best_alpha = a
                best_h = h_try.copy()
    
    return best_alpha, best_h, best_c5

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.05 * np.sin(6.0 * np.pi * x + 1.3)
    h += 0.03 * np.sin(10.0 * np.pi * x + 2.5)
    h += rng.randn(n) * 0.015
    return project_feasible(h, n)

def local_search(h, time_limit, tau_schedule=None):
    """Local search with exact line search along multiple directions."""
    n = len(h)
    dx = 2.0 / n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    
    if tau_schedule is None:
        tau_schedule = [10, 30, 80, 200, 500, 1000]
    
    stall_count = 0
    max_stall = 30
    
    for tau in tau_schedule:
        while time.time() - t0 < time_limit - 2:
            # Compute gradient
            grad, corr = smooth_gradient(h, tau)
            gnorm = float(np.linalg.norm(grad))
            
            if gnorm < 1e-14:
                # Perturb
                rng = np.random.RandomState(int(time.time() * 1000) % 100000)
                pert = rng.randn(n) * 0.02
                pert -= pert.mean()
                h_new = project_feasible(h + pert, n)
                c5_new = compute_c5(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                stall_count += 1
                if stall_count > max_stall:
                    break
                continue
            
            # Direction 1: Full gradient (negative)
            d1 = -grad
            d1 -= d1.mean()  # Ensure sum zero
            d1 = d1 / (np.max(np.abs(d1)) + 1e-30)
            
            a, h_new, c5_new = line_search_1d(h, d1, n, dx)
            if c5_new < best_c5 - 1e-14:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                stall_count = 0
            elif c5_new < best_c5 + 1e-12:
                # Accept to move (non-worsening)
                h = h_new.copy()
            
            # Direction 2: Top pair transfers
            # Sort by gradient (cost)
            order = np.argsort(grad)
            n_pairs = min(40, n // 4)
            # Sources: high grad (expensive to increase), Sinks: low grad
            sources = order[-n_pairs:]  # highest grad
            sinks = order[:n_pairs]      # lowest grad
            
            made_progress = False
            for si in range(len(sources)):
                for tj in range(len(sinks)):
                    i = sources[si]
                    j = sinks[tj]
                    if i == j:
                        continue
                    # Transfer from i to j: d[i] = -1, d[j] = 1
                    d2 = np.zeros(n)
                    d2[i] = -1.0
                    d2[j] = 1.0
                    a, h_try, c5_try = line_search_1d(h, d2, n, dx)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        made_progress = True
                        stall_count = 0
                        break
                if made_progress:
                    break
            
            if not made_progress and c5_new >= best_c5 + 1e-10:
                stall_count += 1
            
            if stall_count > max_stall:
                # Perturb and reset
                rng = np.random.RandomState(int(time.time() * 1000) % 100000 + 1)
                noise_scale = 0.01 + 0.02 * stall_count / max_stall
                pert = rng.randn(n) * noise_scale
                pert -= pert.mean()
                h = project_feasible(h + pert, n)
                stall_count = 0
                max_stall = max(max_stall // 2, 10)
    
    return best_h, best_c5

def resample(h, n_old, n_new):
    """Resample h from n_old points to n_new points."""
    x_old = 2.0 * (np.arange(n_old) + 0.5) / n_old
    x_new = 2.0 * (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    h_new = project_feasible(h_new, n_new)
    return h_new

    np.random.seed(seed)
    
    best_h = None
    best_c5 = float('inf')

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.5
    hi = float(np.max(p)) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] = np.clip(proj[mask] + diff / nm, 0.0, 1.0)
            diff2 = target - float(np.sum(proj))
            if abs(diff2) > 1e-13:
                mask2 = (proj > 1e-9) & (proj < 1.0 - 1e-9)
                nm2 = int(np.sum(mask2))
                if nm2 > 0:
                    proj[mask2] = np.clip(proj[mask2] + diff2 / nm2, 0.0, 1.0)
    return proj

def smooth_gradient(h, tau):
    n = len(h)
    corr = compute_corr(h)
    cmax = float(np.max(corr))
    cs = corr - cmax
    w = np.exp(tau * cs)
    ws = float(w.sum())
    if ws < 1e-300:
        return np.zeros(n), corr
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad, corr

def line_search(h, d, n):
    """Exact line search along direction d (assuming sum(d)=0). Returns (best_h, best_c5)."""
    pos = d > 1e-12
    neg = d < -1e-12
    a_max = 1e30
    if np.any(pos):
        a_max = min(a_max, float(np.min((1.0 - h[pos]) / d[pos])))
    a_min = -1e30
    if np.any(neg):
        a_min = max(a_min, float(-np.min(h[neg] / (-d[neg]))))
    a_max = max(a_max, 0.0)
    a_min = min(a_min, 0.0)
    if a_max - a_min < 1e-14:
        return h.copy(), compute_c5(h)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    fracs = [0.0, 0.05, 0.15, 0.3, 0.5, 0.7, 0.85, 1.0]
    for f in fracs:
        a = a_min + f * (a_max - a_min)
        a = max(a, a_min)
        a = min(a, a_max)
        ht = h + a * d
        ht = project_feasible(ht, n)
        c5t = compute_c5(ht)
        if c5t < best_c5 - 1e-15:
            best_c5 = c5t
            best_h = ht.copy()

            if c5t < best_c5 - 1e-15:
                best_c5 = c5t
                best_h = ht.copy()
                best_a = a
    return best_h, best_c5

def resample_to(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    x_old = 2.0 * (np.arange(n_old) + 0.5) / n_old

    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, n_new)

    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    dead = time_limit - 1.0

    for tau in tau_list:
        stuck = 0
        while time.time() - t0 < dead:

            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-13:
                stuck += 1
                if stuck > 20:
                    break
                rng = np.random.RandomState(int(time.time() * 1e6) % 999999)
                pert = rng.randn(n) * 0.015
                pert -= pert.mean()
                h = project_feasible(h + pert, n)
                continue

            improved = False
            # Full gradient direction
            d = -grad - (-grad).mean()
            dmax = float(np.max(np.abs(d)))
            if dmax > 1e-14:
                d = d / dmax
                h_try, c5_try = line_search(h, d, n)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                elif c5_try <= best_c5 + 1e-12:
                    h = h_try.copy()
            # Opposite direction (sometimes helps with smoothing artifacts)
            d2 = grad - grad.mean()
            d2max = float(np.max(np.abs(d2)))
            if d2max > 1e-14:
                d2 = d2 / d2max
                h_try, c5_try = line_search(h, d2, n)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True

            # Pair transfers
            if time.time() - t0 >= dead:
                break
            order = np.argsort(grad)
            nk = min(n_pairs, n // 5)
            sources = order[-nk:]
            sinks = order[:nk]
            # Score pairs by gradient difference, try top ones first
            pair_scores = []
            for si in range(len(sources)):
                for tj in range(len(sinks)):
                    i = sources[si]
                    j = sinks[tj]
                    if i == j:
                        continue
                    pair_scores.append((grad[i] - grad[j], i, j))
            pair_scores.sort(key=lambda x: -x[0])
            for score, i, j in pair_scores[:min(60, len(pair_scores))]:
                if time.time() - t0 >= dead:
                    break
                if score < 0:
                    break
                dp = np.zeros(n)
                dp[i] = -1.0
                dp[j] = 1.0
                h_try, c5_try = line_search(h, dp, n)
                if c5_try < best_c5 - 1e-14:
                    best_c5 = c5_try
                    best_h = h_try.copy()
                    h = h_try.copy()
                    improved = True
                    break

            # Random direction
            if not improved and time.time() - t0 < dead:
                rng = np.random.RandomState(int(time.time() * 1e6) % 999999 + 7)
                dr = rng.randn(n)
                dr -= dr.mean()
                drmax = float(np.max(np.abs(dr)))
                if drmax > 1e-14:
                    dr = dr / drmax
                    h_try, c5_try = line_search(h, dr, n)
                    if c5_try < best_c5 - 1e-14:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        improved = True

            if not improved:
                stuck += 1
                if stuck > 25:
                    rng = np.random.RandomState(int(time.time() * 1e6) % 999999 + 13)
                    sc = 0.01 + 0.03 * (stuck - 25) / 10.0
                    pert = rng.randn(n) * sc
                    pert -= pert.mean()

                    stuck = 0
