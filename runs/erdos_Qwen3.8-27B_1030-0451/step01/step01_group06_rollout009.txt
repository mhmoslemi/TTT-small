import numpy as np

from scipy.optimize import linprog

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    res = h.sum() - target_sum
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def compute_gradients(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for k in range(-(n-1), n):
        idx = k + n - 1
        if k == 0:
            grads[idx] = dx * (1.0 - 2.0*h)
        elif k > 0:
            g = np.zeros(n)
            g[k:] += dx * (1.0 - h[:n-k])
            g[:n-k] += -dx * h[k:]
            grads[idx] = g
        else:
            t = -k
            g = np.zeros(n)
            g[:n-t] += dx * (1.0 - h[t:])
            g[t:] += -dx * h[:n-t]
            grads[idx] = g
    return grads

def solve_lp(h, corr, grads, trust=None):
    n = len(h)
    m = 2*n - 1
    nvars = n + 1
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    A_ub = np.zeros((m, nvars))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -corr
    if trust is not None:
        lower = np.maximum(-h, -trust)
        upper = np.minimum(1.0 - h, trust)
    else:
        lower = -h.copy()
        upper = (1.0 - h).copy()
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    bounds = list(zip(lower, upper)) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.all(h + d >= -1e-9) and np.all(h + d <= 1 + 1e-9):
                return d
    except Exception:
        pass
    return None

def optimize_lp(h, best_val, deadline, max_iters=200):
    n = len(h)
    h_curr = h.copy()
    best_h = h.copy()
    best_v = best_val
    trust = 0.4
    stagnant = 0
    it = 0
    while time.time() < deadline and it < max_iters and stagnant < 60:
        it += 1
        corr = correlation_array(h_curr)
        cur_max = float(np.max(corr))
        if cur_max < best_v:
            best_v = cur_max
            best_h = h_curr.copy()
        grads = compute_gradients(h_curr)
        d = solve_lp(h_curr, corr, grads, trust)
        if d is None or np.max(np.abs(d)) < 1e-10:
            d = solve_lp(h_curr, corr, grads, None)
            if d is None or np.max(np.abs(d)) < 1e-10:
                stagnant += 1
                trust = max(0.03, trust * 0.85)
                continue
        accepted = False
        for alpha in [1.0, 0.6, 0.4, 0.25, 0.1]:
            h_try = h_curr + alpha * d
            if np.any(h_try < -1e-9) or np.any(h_try > 1 + 1e-9):
                h_try = project_feasible(h_try, n/2.0)
            val = evaluate_exact(h_try)
            if val < best_v - 1e-12:
                best_v = val
                best_h = h_try.copy()
                h_curr = h_try.copy()
                accepted = True
                stagnant = 0
                break
        if not accepted:
            trust = max(0.02, trust * 0.7)
            stagnant += 1
        else:
            trust = min(0.6, trust * 1.25)
    return best_h, best_v

def smooth_lse_value(h, eps):
    corr = correlation_array(h)
    s = corr / eps
    s_max = np.max(s)
    return eps * (s_max + np.log(np.sum(np.exp(s - s_max))))

def smooth_lse_grad(h, eps):
    n = len(h)
    corr = correlation_array(h)
    s = corr / eps
    s_max = np.max(s)
    w = np.exp(s - s_max)
    w /= w.sum()
    grads = compute_gradients(h)
    return grads.T @ w

def optimize_momentum(h, deadline, max_total_iters=600):
    n = len(h)
    target = n / 2.0
    h_curr = h.copy()
    velocity = np.zeros(n)
    best_h = h.copy()
    best_val = evaluate_exact(h_curr)
    mu = 0.85
    eps_stages = [0.08, 0.04, 0.02, 0.01, 0.005]
    iters_per_stage = max_total_iters // len(eps_stages)
    for eps in eps_stages:
        if time.time() > deadline:
            break
        lr = 0.003
        for it in range(iters_per_stage):
            if time.time() > deadline:
                break
            g = smooth_lse_grad(h_curr, eps)
            velocity = mu * velocity - lr * g
            h_trial = project_feasible(h_curr + velocity, target)
            val = evaluate_exact(h_trial)
            if val < best_val - 5e-13:
                best_val = val
                best_h = h_trial.copy()
                h_curr = h_trial.copy()
                lr = min(lr * 1.03, 0.02)
            else:
                velocity *= 0.4
                lr = max(1e-6, lr * 0.65)
                if lr < 1e-6:
                    break
    return best_h, best_val

def optimize_combined(h, best_val, deadline):
    n = len(h)
    target = n / 2.0
    # Phase A: LP
    h_a, v_a = optimize_lp(h, best_val, deadline, max_iters=150)
    best_h, best_v = h_a, v_a
    # Phase B: momentum on smooth
    if time.time() < deadline - 10:
        h_b, v_b = optimize_momentum(best_h, deadline, max_total_iters=300)
        if v_b < best_v:
            best_h, best_v = h_b, v_b
    # Phase C: LP again on result
    if time.time() < deadline - 10:
        h_c, v_c = optimize_lp(best_h, best_v, deadline, max_iters=150)
        if v_c < best_v:
            best_h, best_v = h_c, v_c
    return best_h, best_v

def upscale(h, factor):
    return np.repeat(h, factor)

def downscale(h, factor):
    n = len(h)
    new_n = n // factor
    return h[:new_n * factor].reshape(new_n, factor).mean(axis=1)

            best_h = h.copy()

    v_comp = evaluate_exact(h_comp)
    if v_comp < best_val:
        best_val = v_comp
        best_h = h_comp.copy()
        if time.time() < deadline - 30:
            h, v = optimize_combined(best_h, best_val, deadline)
            if v < best_val:
                best_val = v
                best_h = h.copy()

    # Phase 3: Upscale to 2x
    if time.time() < deadline - 80:
        n_orig = len(best_h)
        h_up = upscale(best_h, 2)
        h_up = project_feasible(h_up, len(h_up) / 2.0)
        h, v = optimize_lp(h_up, best_val, deadline, max_iters=200)
        if v < best_val:
            best_val = v
            best_h = h.copy()
        # Downscale back
        h_dn = downscale(h_up if not (v < best_val) else best_h, 2)
        h_dn = project_feasible(h_dn, n_orig / 2.0)
        v_dn = evaluate_exact(h_dn)
        if v_dn < best_val:
            best_val = v_dn
            best_h = h_dn.copy()
            if time.time() < deadline - 20:
                h, v = optimize_lp(best_h, best_val, deadline, max_iters=100)
                if v < best_val:
                    best_val = v
                    best_h = h.copy()

    # Phase 4: Random restarts at native resolution
    n = len(best_h)
    target = n / 2.0
    n_restarts = 0
    while time.time() < deadline - 20 and n_restarts < 10:
        n_restarts += 1
        if n_restarts == 1:
            h_try = project_feasible(best_h + 0.06 * rng.standard_normal(n), target)
        elif n_restarts == 2:
            h_try = project_feasible(1.0 - best_h + 0.04 * rng.standard_normal(n), target)
        elif n_restarts == 3:
            h_try = project_feasible(best_h + 0.12 * rng.standard_normal(n), target)

            smoothed = np.convolve(best_h, np.ones(9)/9.0, mode='same')
            h_try = project_feasible(smoothed + 0.04 * rng.standard_normal(n), target)
        elif n_restarts == 5:
            binary = (best_h > 0.5).astype(float)
            h_try = project_feasible(0.6*best_h + 0.4*binary, target)
        elif n_restarts == 6:
            h_try = project_feasible(best_h + 0.2 * rng.standard_normal(n), target)
        elif n_restarts == 7:
            h_try = project_feasible(np.clip(best_h + 0.1, 0, 1), target)
        elif n_restarts == 8:
            h_try = project_feasible(rng.random(n), target)
        else:
            h_try = project_feasible(best_h + 0.03 * rng.standard_normal(n), target)

        h, v = optimize_combined(h_try, best_val, deadline)
        if v < best_val:
            best_val = v
            best_h = h.copy()

    # Phase 5: Final fine-tuning
    if time.time() < deadline - 5:
        h, v = optimize_lp(best_h, best_val, deadline, max_iters=100)
        if v < best_val:
            best_val = v
            best_h = h.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)
    return best_h, float(best_val), len(best_h)
