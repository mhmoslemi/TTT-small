    m = float(np.max(C))
    a = (C - m) / tau
    e = np.exp(a - np.max(a))
    s = float(np.sum(e))
    if s < 1e-300:
        alpha = np.zeros_like(C)
        alpha[int(np.argmax(C))] = 1.0
    else:
        alpha = e / s
    dirf = _get_dir()
    ctr = n - 1
    G = np.zeros(n)
    for k in range(len(C)):
        ak = float(alpha[k])
        if ak < 1e-13:
            continue
        d = dirf * (k - ctr)
        coeff = ak * dx
        if d == 0:
            G += coeff * (1.0 - 2.0 * h)
        elif d > 0:
            G[d:] += coeff * (1.0 - h[:n - d])
            G[:n - d] -= coeff * h[d:]
        else:
            dd = -d
            G[:n - dd] += coeff * (1.0 - h[dd:])
            G[dd:] -= coeff * h[:n - dd]
    return G


def _gradient_step(h, c, target, deadline, tau=0.01):
    n = len(h)
    improved = False
    for tau_try in (tau, 0.005, 0.002, 0.001):
        if time.time() > deadline - 5:
            break
        G = _softmax_gradient(h, tau_try)
        if not np.all(np.isfinite(G)):
            continue
        G = G - float(np.mean(G))
        gn = float(np.linalg.norm(G))
        if gn < 1e-15:
            break
        G = G / gn
        for alpha in (0.05, 0.03, 0.02, 0.01, 0.005, 0.002, 0.001):
            if time.time() > deadline - 3:
                break
            h2 = _make_feasible(h - alpha * G, target)
            c2 = _C5(h2)
            if np.isfinite(c2) and c2 < c - 1e-13:
                return h2, c2
    return h, c


def _lp_optimize(h, target, deadline, max_iters=200, max_lags=None):
    n = len(h)
    c = _C5(h)
    if not np.isfinite(c):
        return h, c
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline - 3.0:
            break
        C = _compute_C(h)
        h_new, t = _lp_step(h, C, max_lags)
        if h_new is None:
            stall += 1
            if stall > 10:
                break
            continue
        lam, c_new = _line_search(h, h_new, c, target)
        if lam is not None and c_new < c - 1e-14:
            h = h + lam * (h_new - h)
            h = _make_feasible(h, target)
            c = c_new
            if c > t + 0.02:
                pass
            stall = 0
        else:
            stall += 1
            if stall > 10:
                break
    return h, c


def _resample(h, N):
    n0 = len(h)
    target = 0.5 * float(N)
    if n0 == N:
        return _make_feasible(h, target)
    xold = (np.arange(n0) + 0.5) * (2.0 / n0)
    xnew = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(xnew, xold, h)
    return _make_feasible(y, target)


def _get_initial():
    val = globals().get("initial_h_values", None)
    if val is None:
        return None
    arr = np.asarray(val, dtype=float).ravel()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    return _make_feasible(arr, 0.5 * float(arr.size))


def _fallback_initial(N=360):
    x = (np.arange(N) + 0.5) * (2.0 / N)
    best, bestc = None, np.inf
    for f in (1, 2, 3, 4):
        for ph in (0.0, 0.5 * np.pi, np.pi):
            y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph)
            h = _make_feasible(y, 0.5 * N)
            c = _C5(h)
            if c < bestc:
                bestc = c
                best = h
    return best


def _try_size(h_best, c_best, N, deadline, rng):
    n = len(h_best)
    if N == n:
        h_cand = h_best.copy()
    else:
        h_cand = _resample(h_best, N)
    target = 0.5 * float(N)
    c_cand = _C5(h_cand)

    # LP phase 1: all lags
    time_left = deadline - time.time()
    if time_left > 15:
        alloc = min(time_left - 5, 0.6 * time_left)
        h_cand, c_cand = _lp_optimize(h_cand, target, time.time() + alloc,
                                      max_iters=150, max_lags=None)

    # Gradient refinement
    if time.time() < deadline - 5:
        for _ in range(5):
            if time.time() > deadline - 3:
                break
            h_cand, c_cand = _gradient_step(h_cand, c_cand, target, deadline)

    # LP phase 2: top lags only (faster convergence)
    if time.time() < deadline - 10:
        remaining = deadline - time.time() - 5
        if remaining > 8:
            h_cand, c_cand = _lp_optimize(h_cand, target, time.time() + remaining,
                                          max_iters=80, max_lags=400)

    # Final gradient polish
    if time.time() < deadline - 3:
        h_cand, c_cand = _gradient_step(h_cand, c_cand, target, deadline)

    h_cand = _make_feasible(h_cand, target)
    c_cand = _C5(h_cand)
    if not np.isfinite(c_cand):
        return h_best, c_best, n
    return h_cand, c_cand, N


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    hard = t0 + total - 4.0

    rng = np.random.default_rng(int(seed) % (1 << 32))

    h_init = _get_initial()
    if h_init is None:
        h_init = _fallback_initial(360)

    n0 = int(h_init.size)
    best_h = h_init.copy()
    best_c = _C5(best_h)
    best_n = n0
    target0 = 0.5 * float(n0)

    # Try the initial size first with full LP+gradient
    if time.time() < hard - 20:
        alloc1 = min(hard - time.time() - 10, 0.45 * total)
        h1, c1 = _try_size(best_h, best_c, n0, time.time() + alloc1, rng)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, int(len(h1))

    # Try other sizes
    sizes_to_try = []
    for f in (1, 2, 3, 4, 6, 0.75, 0.5):
        N = int(round(n0 * f))
        if 80 <= N <= 2400 and N != best_n and N not in sizes_to_try:
            sizes_to_try.append(N)
    sizes_to_try.sort(key=lambda x: abs(x - n0))
    sizes_to_try = sizes_to_try[:4]

    for Nt in sizes_to_try:
        if time.time() > hard - 25:
            break
        if best_c <= 0.3808 - 1e-8:
            break
        remaining = hard - time.time() - 10
        if remaining < 15:
            break
        alloc = min(remaining * 0.7, max(20, 0.25 * total))
        stage_deadline = time.time() + alloc
        h2, c2, n2 = _try_size(best_h, best_c, Nt, stage_deadline, rng)
        if c2 < best_c - 1e-14:
            best_h, best_c, best_n = h2, c2, n2

    # Final polish on best solution
    if time.time() < hard - 8:
        target_best = 0.5 * float(best_n)
        remaining = hard - time.time() - 4
        if remaining > 10:
            h3, c3 = _lp_optimize(best_h, target_best, time.time() + remaining * 0.6,
                                  max_iters=100, max_lags=None)
            if c3 < best_c:
                best_h, best_c = h3, c3
            h3, c3 = _gradient_step(best_h, best_c, target_best, time.time() + remaining * 0.3)
            if c3 < best_c:
                best_h, best_c = h3, c3

    best_h = _make_feasible(best_h, 0.5 * float(best_n))
    best_c = _C5(best_h)
    if not np.isfinite(best_c):
        best_h = _make_feasible(np.full(best_n, 0.5), 0.5 * best_n)
        best_c = _C5(best_h)
    return best_h, float(best_c), int(best_n)
