```
def find_min_t(c, A, B, t_lo, t_hi, n_lags):
    # c, A, B are arrays of length n_lags (only affected lags)
    # The function is f(t) = max_p (c[p] + A[p]*t + B[p]*t^2)
    # We want to find t that minimizes f(t)
    # Using subgradient bisection
    
    lo = t_lo
    hi = t_hi
    for _ in range(100):  # sufficient for double precision
        mid = 0.5 * (lo + hi)
        # Compute f(mid) and subgradient
        vals = c + A * mid + B * mid * mid
        fval = vals.max()
        active = vals >= fval - 1e-12
        # Subgradient: A[p] + 2*B[p]*mid for active p
        subgrad = (A[active] + 2 * B[active] * mid)
        sg_min = subgrad.min()
        sg_max = subgrad.max()
        if sg_min >= -1e-12:
            # Minimum is at or to the left of mid
            hi = mid
        elif sg_max <= 1e-12:
            # Minimum is at or to the right of mid
            lo = mid
        else:
            # 0 is in subgradient, mid is the minimum
            break
    t_opt = 0.5 * (lo + hi)
    return t_opt
```

```python
                b += 1.0

            # Change: h[ip]*t

                # j+p = i means p = i-j, and ip = i-p = i-(i-j) = j. Yes!
                # So this is the same as "Term x=j with j+p=i" case.
                # We already added the quadratic there. Don't add again.
```
