def compute_grad(h, raw, tau):
    # Temperature-weighted gradient
    n = len(h)
    L = 2*n - 1
    center = n - 1
    raw_max = np.max(raw)
    exponents = np.clip((raw - raw_max) / tau, -500, 500)
    w = np.exp(exponents)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.ones(L) / L
    else:
        w /= w_sum

        if wk < 1e-12:
            continue
        shift = K - center
        if shift >= 0:
            g[:n-shift] += wk * (1.0 - h[shift:])
            g[shift:] += wk * (-h[:n-shift])

            m = -shift
            g[m:] += wk * (1.0 - h[:n-m])
            g[:n-m] += wk * (-h[m:])

def pair_line_minimize(h, raw, a, b, n_grid=24, refine=2):
