        return x
    lo = float(np.min(x)) - 2.0

        else:
            hi = mid

                res -= dec

            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-13:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:

                res -= dec

                res += inc

    weights = exp_s / total
    f = M + tau * np.log(total)

def compute_lse_gradient(h, weights):

    dx = 2.0 / n

                t2 = np.zeros(n)
                t2[k:] = h[:n - k]
                grad += w_neg * dx * (t1 - t2)
    return grad

def lse_fun_and_grad(h, tau):
    f, weights = compute_lse_values(h, tau)
    grad = compute_lse_gradient(h, weights)
    return f, grad

def solve_smooth_stage(h_init, tau, target, maxiter, time_limit):
    n = len(h_init)
    t0 = time.time()
    timeout_flag = [False]

    def fun_and_grad(h):
        if time.time() - t0 > time_limit:
            timeout_flag[0] = True
            f, _ = compute_lse_values(h, tau)
            g = np.zeros(n)
            return f, g
        return lse_fun_and_grad(h, tau)

    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target}]
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(fun_and_grad, h_init.copy(), jac=True, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        if timeout_flag[0]:
            return None
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, target)
    except Exception:
        pass
    return None

                best_h = h_new.copy()
