To do better than 1/2, we need h to be non-constant but still "self-similar." This is a tension: non-constancy helps at k=0 (reduces ∫h(1-h)) but hurts at other k (increases the mismatch).

At k=0: ∫h(1-h) = 1 - ∫h². To minimize this, we want ∫h² large, i.e., h close to 0 or 1 (binary).
At k=1: ∫h(x)(1-h(x+1)) (over [0,1]). To minimize this, we want h(x) ≈ h(x+1), i.e., h is periodic with period 1.

These two goals are in tension! Binary + period-1 would mean h = 1 on some set S in [0,1] and h = 1 on S+1 in [1,2]. Then ∫h = 2*|S| = 1, so |S| = 1/2. And ∫h(1-h) at k=0 is 0 (binary), while at k=1 it's ∫h(x)(1-h(x+1)) = ∫_S (1-1) + ∫_{[0,1]\S} (1-0) = |[0,1]\S| = 1/2.

So C₅ ≥ 1/2 for period-1 binary functions. This matches what I found before.

The optimal must balance: make h mostly binary (to reduce k=0 overlap) while making h(x) and h(x+1) as similar as possible (to reduce k=1 overlap). The compromise is a "graded" transition.

This suggests the optimal h has a structure like: high on [0, a], transition on [a, b], low on [b, 1], then similar pattern shifted by 1 but with some blending.

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_correlation(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx


def gradient_c_r_vec(h, r):
    N = len(h)
    dx = 2.0 / N
    d = r - (N - 1)
    g = np.zeros(N)
    if d >= 0:
        g[d:] += dx * (1.0 - h[:N - d])
        g[:N - d] -= dx * h[d:]
    else:
        dp = -d
        g[:N - dp] += dx * (1.0 - h[dp:])
        g[dp:] -= dx * h[:N - dp]
    return g


def compute_LQ(h, direction):
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h
    L = (np.correlate(direction, b, mode='full') - np.correlate(h, direction, mode='full')) * dx
    Q = -np.correlate(direction, direction, mode='full') * dx
    return L, Q


def line_search_exact(h, direction, A, L, Q):
    """Find best delta minimizing max_r(A[r] + delta*L[r] + delta^2*Q[r])."""
    pos = direction > 1e-14
    neg = direction < -1e-14

    d_lo = 1e-10
    d_hi = -d_lo

    if np.any(pos):
        d_lo = max(d_lo, np.min(-h[pos] / direction[pos]) + 1e-10)
    if np.any(neg):
        d_hi = min(d_hi, np.min((1.0 - h[neg]) / direction[neg]) - 1e-10)

    if d_hi <= d_lo:
        return 0.0, float(np.max(A))

    current_max = float(np.max(A))
    best_delta = 0.0
    best_val = current_max

    # Coarse grid
    n_coarse = 100
    deltas = np.linspace(d_lo, d_hi, n_coarse)
    vals = A[None, :] + deltas[:, None] * L[None, :] + (deltas[:, None] ** 2) * Q[None, :]
    J = np.max(vals, axis=1)
    idx = np.argmin(J)
    if J[idx] < best_val:
        best_val = J[idx]
        best_delta = deltas[idx]

    # Refine
    if best_delta != 0.0:
        step = (d_hi - d_lo) / n_coarse
        lo = max(d_lo, best_delta - step)
        hi = min(d_hi, best_delta + step)
        n_fine = 60
        fine_d = np.linspace(lo, hi, n_fine)
        fine_v = A[None, :] + fine_d[:, None] * L[None, :] + (fine_d[:, None] ** 2) * Q[None, :]

        fidx = np.argmin(fine_J)
        if fine_J[fidx] < best_val:
            best_val = fine_J[fidx]
            best_delta = fine_d[fidx]

    return best_delta, best_val


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)

            best_h = h3
            best_N = N3

    # Phase 4: Final polish on best
    remaining = deadline - time.time()
    if remaining > 40:
        Nf = best_N
        hf = upscale(best_h, Nf)
        rng4 = np.random.default_rng(seed + 200)
        dl4 = time.time() + remaining - 5
        hf, c5_f = optimize_single_resolution(hf, Nf / 2.0, dl4, rng4)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = hf
            best_N = Nf

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
