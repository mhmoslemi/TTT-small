def project_feasible(h, target_sum):

            m, grad = subgradient_of_max(h_final, top_k=5)
            direction = -(grad - np.mean(grad))
            d_norm = np.linalg.norm(direction)
            if d_norm < 1e-14:
                break
            direction /= d_norm
            h_trial = project_feasible(h_final + alpha_p * direction, target_final)
            val = compute_c5(h_trial)
            if val < c5_final - 1e-15:
                h_final = h_trial
                c5_final = val
                alpha_p = min(alpha_p * 1.03, 0.01)
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final.copy()
            else:
                alpha_p *= 0.7
                if alpha_p < 1e-8:
                    break

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
