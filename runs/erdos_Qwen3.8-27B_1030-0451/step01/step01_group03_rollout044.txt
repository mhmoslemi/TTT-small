def _direct_subgrad_step(h, n, step_size=None):
    """One step of projected subgradient descent on the max function."""
    dx = 2.0 / n
    c = _compute_c(h)
    m = c.max()
    
    # Subgradient: uniform over active constraints
    active = np.where(c >= m - 1e-10)[0]
    if len(active) == 0:
        return h, m
    w = np.zeros(len(c))
    w[active] = 1.0 / len(active)
    
    # Compute gradient as weighted sum
    grad = np.zeros(n)
    for k in active:
        a = w[k]
        d = k - (n - 1)
        if d >= 0:
            lo1, hi1 = 0, n - 1 - d
            grad[lo1:hi1+1] += a * (1.0 - h[lo1+d:hi1+1+d])
            lo2, hi2 = d, n - 1

            grad[lo1:hi1+1] += a * (1.0 - h[lo1-p:hi1+1-p])

from scipy.optimize import minimize
import time

# Globals for SLSQP
_G_T = 0.0
_G_N = 0
_G_TAU = 1.0

# === Objective and gradient ===

def _compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def _smooth_obj(h):
    c = _compute_c(h)
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _smooth_grad(h):
    c = _compute_c(h)
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    active = np.where(alpha > 1e-14)[0]
    
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        if d >= 0:
            hi = n - d
            grad[:hi] += a * (1.0 - h[d:d+hi])
            grad[d:] -= a * h[:n-d]
        else:
            p = -d
            grad[p:] += a * (1.0 - h[:n-p])
            grad[:n-p] -= a * h[p:]
    
    return grad * dx

# === Constraints ===

def _c_eq_fun(h):
    return float(np.sum(h) - _G_T)

def _c_eq_jac(h):
    return np.ones(_G_N)

# === Evaluation ===

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

# === Utility ===

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def make_palindrome(h):
    """Make h palindromic: h[i] = h[n-1-i], maintaining sum constraint."""
    n = len(h)
    target = n / 2.0
    h_pal = 0.5 * (h + h[::-1].copy())
    return project_feasible(h_pal, target)

# === Subgradient polish ===

def _subgrad_polish(h, n, max_steps=20, deadline=None):
    """Direct subgradient descent on the exact max function."""
    target = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for step in range(max_steps):
        if deadline is not None and time.time() > deadline - 2:
            break
            
        c = _compute_c(h)
        m = c.max()
        if m <= best_c5 + 1e-12:
            # Use current as best
            pass
        
        # Find active constraints (within tolerance of max)
        tol = 1e-4
        active = np.where(c >= m - tol)[0]
        
        # Compute subgradient (uniform over active)
        a = 1.0 / len(active)
        dx = 2.0 / n
        grad = np.zeros(n)
        
        for k in active:
            d = k - (n - 1)
            if d >= 0:
                hi = n - d
                grad[:hi] += a * (1.0 - h[d:d+hi])
                grad[d:] -= a * h[:n-d]
            else:
                p = -d
                grad[p:] += a * (1.0 - h[:n-p])
                grad[:n-p] -= a * h[p:]
        
        grad *= dx
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        best_step_h = h
        best_step_c5 = m
        for alpha_try in [0.05/gnorm, 0.02/gnorm, 0.01/gnorm, 0.005/gnorm, 0.001/gnorm]:
            h_try = project_feasible(h - alpha_try * grad, target)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_step_c5:
                best_step_c5 = c5_try
                best_step_h = h_try
        
        if best_step_c5 < best_c5 - 1e-12:
            best_h = best_step_h.copy()
            best_c5 = best_step_c5
            h = best_step_h
        else:
            # No improvement, try smaller steps or break
            h = project_feasible(h - 0.001 * grad / gnorm, target)
    
    return best_h, best_c5

# === Main optimization for a given resolution ===

def _optimize_at_resolution(h0, n, deadline, rng):
    """Run multi-phase optimization at resolution n."""
    global _G_T, _G_N, _G_TAU
    _G_T = n / 2.0
    _G_N = n

    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)

    # Phase 2: medium (5 runs)

    tau_phase3 = tau_phase2[-1] * (0.4 ** np.arange(5))

    bounds = [(0.0, 1.0)] * n

    stall = 0
