Actually, the cleanest approach for the gradient of a single lag:

For offset `off` (can be negative), the correlation is:
c_off = sum over valid (i,j) where j = i + off of h[j] * (1-h[i])

Equivalently: c_off = sum_i h[i+off] * (1-h[i]) where both indices are in [0, n).

The gradient dc_off/dh[m]:

- If m = i for some valid i (i.e., i = m): contributes -h[i + off] = -h[m + off]

So: dc_off/dh[m] = (1 - h[m-off]) * I[m-off in [0,n)) + (-h[m+off]) * I[m+off in [0,n))

import time

import numpy as np


def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def project_exact(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d < 0:
            slack = 1.0 - out
            room = slack.sum()
            if room > 0:
                out = out + (d / room) * slack
        else:
            room = out.sum()
            if room > 0:
                out = out + (d / room) * out
    return np.clip(out, 0.0, 1.0)


def repair(h, target):

    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)

def lse_value(c, sigma):
    m = c.max()
    e = np.exp(sigma * (c - m))
    return m + math.log(e.sum()) / sigma

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def lse_polish_nesterov(h0, rng, time_limit, sigma_start, sigma_end, target, deadline):
    """LSE optimization with Nesterov momentum and adaptive sigma."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = objective(h)
    best_lse = lse_value(corr_full(h), sigma_start)

    v = np.zeros(n)
    beta = 0.85
    eta = 0.05
    eta_min = 1e-7
    t0 = time.monotonic()
    total_time = max(time_limit, 0.01)
    stall_count = 0
    iters = 0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit or (deadline is not None and time.monotonic() > deadline):
            break
        frac = min(1.0, elapsed / total_time)
        sigma = sigma_start + (sigma_end - sigma_start) * frac * frac

        c = corr_full(h)
        cur_lse = lse_value(c, sigma)
        g = lse_grad_from_corr(c, h, sigma) * dx
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break

        g_norm = g / (gn + 1e-30)
        step_dir = beta * v - g_norm
        v = step_dir

        improved = False
        e = eta
        for _ in range(30):
            cand = project_exact(h - e * step_dir, target)
            cc = corr_full(cand)
            cv = lse_value(cc, sigma)
            if cv < cur_lse - 1e-14:
                h = cand
                cur_lse = cv
                c5 = float(cc.max() * dx)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True
                stall_count = 0
                break
            e *= 0.5

        if not improved:
            h = project_exact(h + rng.normal(0.0, 0.02, n), target)
            cur_lse = lse_value(corr_full(h), sigma)
            c5 = objective(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
            stall_count += 1
            if stall_count >= 8:
                eta *= 0.6
                v *= 0.5
                stall_count = 0
                if eta < eta_min:
                    break
        elif eta > eta_min:
            eta = min(eta * 1.1, 0.15)

        iters += 1

    c5 = objective(h)
    if c5 < best_c5 - 1e-13:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5, iters


def subgrad_polish(h0, time_limit, target, deadline):
    """Direct subgradient method on the max correlation."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = objective(h)
    t0 = time.monotonic()
    total = max(time_limit, 0.01)
    eta = 0.02
    iters = 0

    while True:
        elapsed = time.monotonic() - t0
        if elapsed > time_limit or (deadline is not None and time.monotonic() > deadline):
            break
        c = corr_full(h)
        cur_c5 = float(c.max() * dx)
        g = topk_subgrad(c, h, k=2)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        g = g / gn * dx

        eta_use = eta * (1.0 - elapsed / total)
        if eta_use < 1e-8:
            break

        improved = False
        e = eta_use
        for _ in range(20):
            cand = project_exact(h - e * g, target)
            cc = corr_full(cand)
            cv = float(cc.max() * dx)
            if cv < cur_c5 - 1e-14:
                h = cand
                cur_c5 = cv
                if cv < best_c5 - 1e-13:
                    best_c5 = cv
                    best_h = h.copy()
                improved = True
                break
            e *= 0.5

        if not improved:
            h = project_exact(h + np.random.default_rng(iters * 7 + 3).normal(0, 0.01, n), target)
            c = corr_full(h)
            cur_c5 = float(c.max() * dx)
            if cur_c5 < best_c5 - 1e-13:
                best_c5 = cur_c5
                best_h = h.copy()
            iters += 1

        if elapsed > total * 0.7:
            eta *= 0.85

    c5 = objective(h)
    if c5 < best_c5 - 1e-13:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def correlated_perturb(h, c, rng, n_perturb=10):
    """Perturb h in a direction that reduces the top correlations."""
    n = len(h)
    target = n / 2.0
    center = n - 1
    top_indices = np.argsort(c)[::-1][:5]
    direction = np.zeros(n)
    for ki in top_indices:
        off = int(ki) - center
        wl = abs(c[ki]) / (c[top_indices].sum() + 1e-30)
        a1 = max(0, off)
        b1 = min(n, n + off)
        if b1 > a1:
            direction[a1:b1] += wl * (1.0 - h[a1 - off:b1 - off])
        a2 = max(0, -off)
        b2 = min(n, n - off)
        if b2 > a2:
            direction[a2:b2] -= wl * h[a2 + off:b2 + off]
    norm = float(np.linalg.norm(direction))
    if norm < 1e-15:
        return h.copy()
    direction = direction / norm
    scale = 0.03 * rng.random()
    cand = project_exact(h + scale * direction, target)
    return cand


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 6.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        n0 = 131
        h0 = make_seed(n0, rng)
    else:
        n0 = len(parent)
        h0 = repair(parent, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = objective(best_h)

    def consider(cand_h):
        nonlocal best_h, best_c5
        try:
            ch = repair(np.asarray(cand_h, dtype=float), len(cand_h) / 2.0)
            c = objective(ch)
            if np.isfinite(c) and c < best_c5 - 1e-12:
                best_c5 = c
                best_h = ch.copy()
        except Exception:
            pass

    # Phase 1: LSE with Nesterov momentum, sigma 25 -> 80
    if time.monotonic() < deadline:
        alloc = min(0.25 * effective, deadline - time.monotonic())
        alloc = max(alloc, 1.0)
        ch, cc, _ = lse_polish_nesterov(best_h, rng, alloc, 25.0, 80.0,
                                        len(best_h) / 2.0, deadline)
        consider(ch)

    # Phase 2: Higher sigma LSE 80 -> 200
    if time.monotonic() < deadline:
        alloc = min(0.22 * effective, deadline - time.monotonic())
        alloc = max(alloc, 1.0)
        ch, cc, _ = lse_polish_nesterov(best_h, rng, alloc, 80.0, 200.0,
                                        len(best_h) / 2.0, deadline)
        consider(ch)

    # Phase 3: Very high sigma 200 -> 500
    if time.monotonic() < deadline:
        alloc = min(0.18 * effective, deadline - time.monotonic())
        alloc = max(alloc, 0.5)
        ch, cc, _ = lse_polish_nesterov(best_h, rng, alloc, 200.0, 500.0,
                                        len(best_h) / 2.0, deadline)
        consider(ch)

    # Phase 4: Direct subgradient polishing
    if time.monotonic() < deadline:
        alloc = min(0.15 * effective, deadline - time.monotonic())
        alloc = max(alloc, 0.5)
        ch, cc = subgrad_polish(best_h, alloc, len(best_h) / 2.0, deadline)
        consider(ch)

    # Phase 5: Correlated perturbations + short LSE polish
    if time.monotonic() < deadline:
        alloc_total = min(0.12 * effective, deadline - time.monotonic())
        if alloc_total > 2.0:
            n_pert = 5
            per_alloc = alloc_total / n_pert
            for _ in range(n_pert):
                if time.monotonic() > deadline:
                    break
                c = corr_full(best_h)
                cand = correlated_perturb(best_h, c, rng)
                if time.monotonic() < deadline - 1.0:
                    a = min(per_alloc, deadline - time.monotonic())
                    if a > 0.3:
                        ch, cc, _ = lse_polish_nesterov(cand, rng, a, 50.0, 150.0,
                                                        len(cand) / 2.0, deadline)
                        consider(ch)

    # Phase 6: Try nearby resolutions
    if time.monotonic() < deadline - 2.0:
        for dn in (-2, 2, -4, 4):
            if time.monotonic() > deadline - 1.0:
                break
            nn = n0 + dn
            if nn < 80 or nn > 200:
                continue
            alloc = min(0.03 * effective, deadline - time.monotonic())
            if alloc < 0.5:
                break
            rs = repair(resample(best_h, nn), nn / 2.0)
            ch, cc, _ = lse_polish_nesterov(rs, rng, alloc, 40.0, 120.0, nn / 2.0, deadline)
            consider(ch)

    # Phase 7: Final subgradient + LSE polish cycle
    if time.monotonic() < deadline - 0.5:
        alloc = min(0.05 * effective, deadline - time.monotonic())
        if alloc > 0.5:
            half = alloc / 2.0
            ch, cc = subgrad_polish(best_h, half, len(best_h) / 2.0, deadline)
            consider(ch)
            if time.monotonic() < deadline - 0.3:
                a2 = min(half, deadline - time.monotonic())
                if a2 > 0.2:
                    ch, cc, _ = lse_polish_nesterov(best_h, rng, a2, 100.0, 300.0,
                                                    len(best_h) / 2.0, deadline)
                    consider(ch)

    # Final repair and verification
    best_h = repair(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_c5 = objective(best_h)
    n_pts = len(best_h)
    return best_h, float(best_c5), int(n_pts)
