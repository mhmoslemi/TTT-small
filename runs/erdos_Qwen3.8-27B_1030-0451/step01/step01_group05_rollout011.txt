    valid_b_s = (b_s >= q_lo) & (b_s <= q_hi) & (b_s >= 0) & (b_s < n)
    valid_a = (a >= q_lo) & (a <= q_hi)
    valid_b = (b >= q_lo) & (b <= q_hi)
    
    t1 = np.where(valid_a_s, 1.0 - h[np.clip(a_s, 0, n - 1)], 0.0)
    t2 = np.where(valid_b_s, 1.0 - h[np.clip(b_s, 0, n - 1)], 0.0)
    t3 = np.where(valid_a, h[np.clip(a_sp, 0, n - 1)], 0.0)
    t4 = np.where(valid_b, h[np.clip(b_sp, 0, n - 1)], 0.0)
    
    lr = t1 - t2 - t3 + t4
    
    quad = np.zeros(L)
    quad[s == 0] -= 2.0
    if a != b:
        quad[s == (a - b)] += 1.0
        quad[s == (b - a)] += 1.0
    
    return lr, quad
