import numpy as np
import time
from scipy.optimize import minimize, linprog

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag == 0:
            g[:] = 1.0 - 2.0 * h
        else:
            m1 = np.arange(lag, n)
            g[m1] += 1.0 - h[m1 - lag]
            m2 = np.arange(0, n - lag)
            g[m2] -= h[m2 + lag]
    else:
        l = -lag
        m1 = np.arange(0, n - l)
        g[m1] += 1.0 - h[m1 + l]
        m2 = np.arange(l, n)
        g[m2] -= h[m2 - l]
    return g

def smooth_obj_val(h, beta):
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 500)
    u = np.exp(expo)
    Z = u.sum()
    return m - np.log(max(Z, 1e-15)) / beta

def smooth_grad(h, beta):
    n = len(h)
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 500)
    u = np.exp(expo)
    Z = u.sum()
    w = u / Z
    g = np.zeros(n)
    for k in range(len(v)):
        if w[k] > 1e-12:
            lag = k - (n - 1)
            g += w[k] * grad_lag(h, lag)
    return g

def nesterov_smooth(h0, deadline, beta_init=8, beta_max=2000, max_iters=3000):
    n = len(h0)
    x = h0.copy()
    x_prev = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    mu = 0.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = min(beta_init * (1.05 ** it), beta_max)
        y = x + mu * (x - x_prev)
        y = np.clip(y, 0.0, 1.0)
        g = smooth_grad(y, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        step = 0.02
        backtracks = 0
        f_y = smooth_obj_val(y, beta)
        while backtracks < 15:
            x_new = y - step * g
            x_new = project_sum_box(x_new, n / 2.0)
            f_new = smooth_obj_val(x_new, beta)
            if f_new <= f_y + 0.01 * step * gn * (x_new - y):
                break
            step *= 0.5
            backtracks += 1
        if backtracks >= 15:
            step *= 0.5
            x_new = project_sum_box(y - step * g, n / 2.0)
        c_new = evaluate_c5(x_new)
        if c_new < best_c - 1e-12:
            best_c = c_new
            best_h = x_new.copy()
        mu_new = (1.0 + np.sqrt(1.0 + 4.0 * mu * mu)) / 2.0
        x_prev = x
        x = x_new
        mu = min(mu_new, 2.0)
    return best_h, best_c

def lp_trust(h0, deadline, max_iters=400, init_radius=0.10):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = init_radius
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        gap = max(0.008 * vmax, 0.003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 60:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-60:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-15:]
        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)
            b_lin[i] = A[i] @ h - v[k]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [0.95]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        took = 0
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-10:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                if it % 8 == 0:
                    radius = min(radius * 1.05, 0.20)
            else:
                no_improve += 1
        else:
            no_improve += 1
        if no_improve >= 10:
            radius = max(radius * 0.6, 0.002)
            no_improve = 0
            if radius <= 0.002:
                break
    return best_h, best_c

def slsqp_high(h0, deadline, max_betas=12):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [8, 16, 32, 64, 128, 256, 512, 1024, 2048, 4096, 8192, 16384][:max_betas]
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

    for beta in betas:
        if time.time() > deadline:
            break
        for attempt in range(3):
            if time.time() > deadline:
                break
            if attempt == 0:
                x0 = best_h.copy()
            elif attempt == 1:
                x0 = best_h + np.random.randn(n) * 0.003
                x0 = project_sum_box(x0, n / 2.0)
            else:
                x0 = best_h + np.random.randn(n) * 0.008
                x0 = project_sum_box(x0, n / 2.0)

            def obj_func(x, b=beta):
                return smooth_obj_val(x, b)

            def grad_func(x, b=beta):
                return smooth_grad(x, b)

            try:
                res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-12:
                    best_h = x_new.copy()
                    best_c = c_new
    return best_h, best_c

def interp_to_size(h0, n_new):
    n_old = len(h0)
    if n_new == n_old:
        return h0.copy()
    x_old = np.arange(n_old) * 1.0 / n_old
    x_new = np.arange(n_new) * 1.0 / n_new
    h_ext = np.concatenate([h0, h0[:1]])
    h_new = np.interp(x_new, x_old, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def projo_grad_backtrack(h0, deadline, max_iters=5000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    x = h0.copy()
    step = 0.01
    beta = 30.0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_cur = evaluate_c5(x)
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = x.copy()
            stall = 0
        else:
            stall += 1
        if stall > 500:
            break
        g = smooth_grad(x, beta)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        f_cur = smooth_obj_val(x, beta)
        x_new = project_sum_box(x - step * g, n / 2.0)
        f_new = smooth_obj_val(x_new, beta)
        if f_new <= f_cur + 1e-4 * step * gn * (x_new - x):
            x = x_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.7
            if step < 1e-8:
                break
    return best_h, best_c

def random_perturb(h0, sigma=0.02, n_perturbs=5, deadline=None):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    for i in range(n_perturbs):
        if deadline is not None and time.time() > deadline:
            break
        h_p = h0 + np.random.randn(n) * sigma
        h_p = project_sum_box(h_p, n / 2.0)
        c_p = evaluate_c5(h_p)
        if c_p < best_c:
            best_h = h_p.copy()
            best_c = c_p
    return best_h, best_c
