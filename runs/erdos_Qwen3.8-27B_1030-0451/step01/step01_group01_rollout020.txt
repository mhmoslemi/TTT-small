def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y += r * slack / ts
        else:
            ty = y.sum()
            if ty > 1e-12:
                y += r * y / ty
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_full(h):

    n = len(h)

def corr_at_lag(h, lag):

    if lag >= 0:
        if lag >= n:
            return 0.0
        return dx * np.sum(h[:n-lag] * (1.0 - h[lag:]))
    else:
        l = -lag
        if l >= n:
            return 0.0
        return dx * np.sum(h[l:] * (1.0 - h[:-l]))

def grad_lag(h, lag):
    """Gradient of correlation at specific lag w.r.t. h."""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[:n-lag] += (1.0 - h[lag:])
            g[lag:] -= h[:n-lag]
    else:
        l = -lag
        if l < n:
            g[l:] += (1.0 - h[:n-l])
            g[:n-l] -= h[l:]
    return g

def method_subgrad(h0, deadline, max_iters=5000, top_k=20):
    """Projected subgradient with Polyak step size."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    c_opt = 0.37  # lower bound estimate for Polyak step
    gamma = 0.5
    it = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_full(h)
        c_cur = float(np.max(v))
        if c_cur < best_c:
            best_c = c_cur
            best_h = h.copy()
        
        # Subgradient: weighted combination of gradients of active constraints
        idx = np.argsort(v)[::-1][:top_k]
        active_lags = idx - (n - 1)
        
        # Exponential weighting
        m = v.max()
        beta = 50.0
        expo = np.exp(beta * (v[idx] - m))
        w = expo / expo.sum()
        
        g = np.zeros(n)
        for i, lag in enumerate(active_lags):
            g += w[i] * grad_lag(h, lag)
        
        # Polyak step size
        gap = c_cur - c_opt
        if gap < 1e-10:
            break
        step = gamma * gap / (np.linalg.norm(g) ** 2 + 1e-10)
        step = min(step, 0.1)
        
        # Update
        h_new = h - step * g
        h_new = project_sum_box(h_new, n / 2.0)
        
        c_new = evaluate_c5(h_new)
        if c_new <= c_cur:
            h = h_new
            gamma = min(gamma * 1.01, 5.0)
        else:
            gamma = max(gamma * 0.95, 0.1)
    
    return best_h, best_c
