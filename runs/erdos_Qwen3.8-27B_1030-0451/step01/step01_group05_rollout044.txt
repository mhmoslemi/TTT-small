    n = len(h)
    s = np.arange(-(n-1), n)
    col = np.zeros(2*n-1)
    valid1 = (i + s >= 0) & (i + s < n)
    col[valid1] += (1.0 - h[i + s[valid1]])
    valid2 = (i - s >= 0) & (i - s < n)
    col[valid2] -= h[i - s[valid2]]
    return col

def compute_lin_quad(h, a, b):
    n = len(h)
    center = n - 1
    col_a = compute_col(h, a)
    col_b = compute_col(h, b)
    lin = col_a - col_b
    quad = np.zeros(2*n-1)
    quad[center] = -2.0
    quad[center + (b - a)] += 2.0
    quad[center + (a - b)] += 2.0
    return lin, quad

def line_minimize(raw, lin, quad, d_lo, d_hi, n_grid=64):
    if d_hi - d_lo < 1e-14:
        return 0.0, raw.max()
    
    d_vals = np.linspace(d_lo, d_hi, n_grid)
    # Vectorized: raw[None,:] + d_vals[:,None]*lin[None,:] + d_vals[:,None]**2*quad[None,:]
    # But this creates a (n_grid, L) array which for n_grid=64, L=161 is 10304 floats - fine
    vals = raw[None, :] + d_vals[:, None] * lin[None, :] + (d_vals[:, None] ** 2) * quad[None, :]
    scores = vals.max(axis=1)
    
    best_idx = np.argmin(scores)
    best_d = d_vals[best_idx]
    best_score = scores[best_idx]
    
    # Refine around best
    if n_grid > 2:
        step = (d_hi - d_lo) / (n_grid - 1)
        lo = max(d_lo, best_d - step)
        hi = min(d_hi, best_d + step)
        for _ in range(20):
            if hi - lo < 1e-15:
                break
            d1 = lo + (hi - lo) / 3.0
            d2 = hi - (hi - lo) / 3.0
            f1 = (raw + d1 * lin + d1*d1 * quad).max()
            f2 = (raw + d2 * lin + d2*d2 * quad).max()
            if f1 < f2:
                hi = d2
            else:
                lo = d1
        best_d = (lo + hi) / 2.0
        best_score = (raw + best_d * lin + best_d*best_d * quad).max()
    
    return best_d, best_score

def weighted_gradient(h, raw, tau):
    """Compute the weighted gradient (subgradient of smooth max)."""
    n = len(h)
    L = 2*n - 1
    center = n - 1
    tau_eff = max(1e-10, tau * raw.max())
    exps = np.exp(np.clip((raw - raw.max()) / tau_eff, -500, 500))
    weights = exps / exps.sum()
    
    grad = np.zeros(n)
    s_range = np.arange(-(n-1), n)
    for idx in range(L):
        w = weights[idx]
        if w < 1e-10:
            continue
        s = s_range[idx]
        i_start = max(0, -s)
        i_end = min(n, n - s)
        if i_end <= i_start:
            continue
        grad[i_start:i_end] += w * (1.0 - h[i_start+s:i_end+s])
        grad[i_start+s:i_end+s] -= w * h[i_start:i_end]
    
    return grad

def pair_search(h, raw, tau, rng, n_iter=500, n_candidates=6, deadline=None):
    """Local search via exact pair exchanges."""
    n = len(h)
    dx = 2.0 / n
    cur_score = raw.max() * dx
    best_h = h.copy()
    best_raw = raw.copy()
    best_score = cur_score
    stall = 0
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline:
            break
        
        grad = weighted_gradient(h, raw, tau)
        
        # Get candidate coordinates
        sorted_idx = np.argsort(grad)
        a_candidates = sorted_idx[:n_candidates]
        b_candidates = sorted_idx[::-1][:n_candidates]
        
        moved = False
        for a in a_candidates:
            for b in b_candidates:
                a, b = int(a), int(b)
                if a == b:
                    continue
                d_lo = max(-h[a], h[b] - 1.0)
                d_hi = min(1.0 - h[a], h[b])
                if d_hi <= d_lo + 1e-13:
                    continue
                
                lin, quad = compute_lin_quad(h, a, b)
                best_d, line_score = line_minimize(raw, lin, quad, d_lo, d_hi)
                
                if line_score * dx < cur_score - max(1e-12, 1e-10 * cur_score):
                    h[a] += best_d
                    h[b] -= best_d
                    raw = raw + best_d * lin + best_d * best_d * quad
                    raw = np.maximum(raw, 0.0)
                    cur_score = raw.max() * dx
                    moved = True
                    if cur_score < best_score:
                        best_score = cur_score
                        best_h = h.copy()
                        best_raw = raw.copy()
                    break
            if moved:
                break
        
        if moved:
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
            # Try random pairs
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                d_lo = max(-h[a], h[b] - 1.0)
                d_hi = min(1.0 - h[a], h[b])
                if d_hi > d_lo + 1e-13:
                    lin, quad = compute_lin_quad(h, a, b)
                    best_d, line_score = line_minimize(raw, lin, quad, d_lo, d_hi)
                    if line_score * dx < cur_score - max(1e-12, 1e-10 * cur_score):
                        h[a] += best_d
                        h[b] -= best_d
                        raw = raw + best_d * lin + best_d * best_d * quad
                        raw = np.maximum(raw, 0.0)
                        cur_score = raw.max() * dx
                        moved = True
                        if cur_score < best_score:
                            best_score = cur_score
                            best_h = h.copy()
                            best_raw = raw.copy()
    
    # Final verification
    if best_score < cur_score + 1e-12:
        exact_s = exact_score(best_h)
        if exact_s < best_score + 1e-10:
            best_score = exact_s
        else:
            best_score = exact_s
            best_raw = np.correlate(best_h, 1.0 - best_h, mode='full')
    
    return best_h, best_raw, best_score
