    # quadratic coefficient:

    #   K=center+(a-b): +1 (if a≠b, which it always is)
    #   K=center+(b-a): +1
    #   All other K: 0

import numpy as np
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def compute_raw(h):
    return np.correlate(h, 1.0 - h, mode='full')


def project_feasible(z, n):
    S = n / 2.0
    z = np.clip(z, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def weighted_grad(h, raw, tau):
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    raw_max = np.max(raw)
    exponents = np.clip((raw - raw_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.ones(2 * n - 1) / (2 * n - 1)
    else:
        weights /= sw
    grad = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        s = K - center
        if s >= 0:
            grad[s:] += w * (1.0 - h[:n - s])
            grad[:n - s] += w * (-h[s:])
        else:
            m = -s
            grad[:n - m] += w * (1.0 - h[m:])
            grad[m:] += w * (-h[:n - m])
    return grad


def compute_lin(h, a, b):
    n = len(h)
    center = n - 1
    lin = np.zeros(2 * n - 1)
    rev = 1.0 - h[center::-1]
    lin[a:a + n] += rev
    lin[center - a:center - a + n] -= h[:]
    lin[b:b + n] -= rev
    lin[center - b:center - b + n] += h[:]
    return lin


def line_search(raw, lin, quad, d_lo, d_hi, n_grid=30):
    if d_hi - d_lo < 1e-14:
        return 0.0, np.max(raw)
    d_vals = np.linspace(d_lo, d_hi, n_grid)
    best_d = 0.0
    best_s = np.inf
    for d in d_vals:
        vals = raw + d * lin + d * d * quad
        s = np.max(vals)
        if s < best_s:
            best_s = s
            best_d = d
    # Local refinement around best
    d_left = d_vals[max(0, np.argmin(np.array([np.max(raw + d * lin + d * d * quad) for d in d_vals])) - 1)]
    d_right = d_vals[min(len(d_vals) - 1, np.argmax(np.array([np.max(raw + d * lin + d * d * quad) for d in d_vals])) + 1)]
    # Just do a finer grid around best
    span = max(0.1 * (d_hi - d_lo), abs(best_d) * 0.5)
    d_fine = np.linspace(max(d_lo, best_d - span), min(d_hi, best_d + span), 20)
    for d in d_fine:
        vals = raw + d * lin + d * d * quad
        s = np.max(vals)
        if s < best_s:
            best_s = s
            best_d = d
    return best_d, best_s


def pair_descent(h, raw, rng, deadline, tau_start=0.06, n_pairs=8, max_passes=30):
    n = len(h)
    center = n - 1
    cur_score = float(np.max(raw)) * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    no_improve = 0
    tau = tau_start

    while time.time() < deadline:
        improved = False
        grad = weighted_grad(h, raw, tau)
        # Select candidate pairs
        pairs = []
        order = np.argsort(grad)
        for k in range(min(4, n // 2)):
            pairs.append((int(order[k]), int(order[n - 1 - k])))
        for _ in range(4):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                pairs.append((a, b))

        best_move = None
        best_move_score = cur_score

        for (a, b) in pairs:
            if a == b:
                continue
            d_lo = max(-h[a], h[b] - 1.0)
            d_hi = min(1.0 - h[a], h[b])
            if d_hi - d_lo < 1e-12:
                continue
            lin = compute_lin(h, a, b)
            quad = np.zeros(2 * n - 1)
            quad[center] = -2.0
            Ka = center + (a - b)
            Kb = center + (b - a)
            if 0 <= Ka < 2 * n - 1:
                quad[Ka] += 1.0
            if 0 <= Kb < 2 * n - 1:
                quad[Kb] += 1.0
            d_opt, s_opt = line_search(raw, lin, quad, d_lo, d_hi, 25)
            if s_opt * (2.0 / n) < best_move_score - 1e-13:
                best_move = (a, b, d_opt)
                best_move_score = s_opt * (2.0 / n)

        if best_move is not None:
            a, b, d = best_move
            h[a] += d
            h[b] -= d
            h = np.clip(h, 0.0, 1.0)
            raw = compute_raw(h)
            cur_score = float(np.max(raw)) * (2.0 / n)
            improved = True
            no_improve = 0
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            no_improve += 1
            tau *= 0.7
            if no_improve > 15:
                # Try random perturbation to escape
                if time.time() < deadline - 5:
                    for _ in range(5):
                        a = int(rng.integers(0, n))
                        b = int(rng.intensors(0, n)) if False else int(rng.integers(0, n))
                        if a == b:
                            continue
                        d_lo = max(-h[a], h[b] - 1.0)
                        d_hi = min(1.0 - h[a], h[b])
                        if d_hi - d_lo < 1e-8:
                            continue
                        d = (d_lo + d_hi) * 0.5 * (0.5 + rng.random())
                        if d <= d_lo or d >= d_hi:
                            d = d_lo + rng.random() * (d_hi - d_lo)
                        h[a] += d
                        h[b] -= d
                        h = np.clip(h, 0.0, 1.0)
                        # Project back to maintain sum
                        h = project_feasible(h, n)
                        raw = compute_raw(h)
                        new_s = float(np.max(raw)) * (2.0 / n)
                        if new_s < best_score * 1.05:
                            cur_score = new_s
                            break
                        else:
                            h = np.clip(h, 0.0, 1.0)
                            h = project_feasible(h, n)

        if improved:
            tau = max(0.001, tau * 0.95)
        if no_improve > 40:
            break

    return best_h, best_score


def simulated_anneal(h, raw, rng, deadline, t_start=0.005, t_end=0.0001):
    n = len(h)
    center = n - 1
    cur_score = float(np.max(raw)) * (2.0 / n)
    best_h = h.copy()
    best_score = cur_score
    t0 = time.time()
    total_time = max(1.0, deadline - t0)

    while time.time() < deadline:
        elapsed_frac = (time.time() - t0) / total_time
        T = t_start * (t_end / t_start) ** elapsed_frac
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        d_lo = max(-h[a], h[b] - 1.0)
        d_hi = min(1.0 - h[a], h[b])
        if d_hi - d_lo < 1e-10:
            continue
        d = d_lo + rng.random() * (d_hi - d_lo)
        h[a] += d
        h[b] -= d
        h = np.clip(h, 0.0, 1.0)
        new_raw = compute_raw(h)
        new_score = float(np.max(new_raw)) * (2.0 / n)
        delta = new_score - cur_score
        if delta < 0 or rng.random() < np.exp(-delta / T):
            raw = new_raw
            cur_score = new_score
            if cur_score < best_score:
                best_score = cur_score
                best_h = h.copy()
        else:
            h[a] -= d
            h[b] += d

    return best_h, best_score


def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)

        # Follow up with descent after SA
        if time.time() < deadline - 20:
            h4b, s4b = pair_descent(h4.copy(), compute_raw(h4), rng, min(deadline, time.time() + 100), tau_start=0.02)
            if s4b < best_score:
                _try_update(h4b, s4b, best_n)

    # Phase 5: Multi-start with diverse initializations
    if time.time() < deadline - 60:
        # Periodic patterns
        for period in [4, 6, 8, 10, 12]:
            if time.time() > deadline - 30:
                break
            x = np.linspace(0, 2.0, best_n, endpoint=False)
            t = np.mod(x, 2.0 / period)
            h_p = (t < 1.0 / period).astype(float)
            h_p = project_feasible(h_p, best_n)
            s_p = exact_score(h_p)
            if s_p < best_score + 0.01:
                h_p, s_p = pair_descent(h_p, compute_raw(h_p), rng, min(deadline, time.time() + 60), tau_start=0.04)
                if s_p < best_score:
                    _try_update(h_p, s_p, best_n)

        # Random restarts
        for _ in range(3):
            if time.time() > deadline - 30:
                break
            h_r = np.full(best_n, 0.5 + 0.3 * (rng.random(best_n) - 0.5))
            h_r = project_feasible(h_r, best_n)
            h_r, s_r = pair_descent(h_r, compute_raw(h_r), rng, min(deadline, time.time() + 80), tau_start=0.05)
            if s_r < best_score:
                _try_update(h_r, s_r, best_n)

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        h6, s6 = pair_descent(best_h.copy(), compute_raw(best_h), rng, min(deadline, time.time() + 60), tau_start=0.01, n_pairs=10)
        if s6 < best_score:
            _try_update(h6, s6, best_n)

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-12:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
