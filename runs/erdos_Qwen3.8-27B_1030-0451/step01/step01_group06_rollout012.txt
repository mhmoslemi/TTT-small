    n_native = len(h0)
    best_h = h0.copy()
    best_val = evaluate_exact(best_h)
    
    def remaining():
        return deadline - time.time()
    
    # Phase 1: Smooth optimization at native resolution
    if remaining() > 80:
        h_smooth = smooth_phase(h0, [5, 20, 50, 100, 200], t_start + remaining() * 0.25, rng)
        v_smooth = evaluate_exact(h_smooth)
        if v_smooth < best_val:
            best_val = v_smooth
            best_h = h_smooth.copy()
    
    # Phase 2: LP optimization at native resolution
    if remaining() > 60:
        h_lp = best_h.copy()
        h_lp, v_lp = lp_phase(h_lp, best_val, t_start + remaining() * 0.45, rng)
        if v_lp < best_val:
            best_val = v_lp
            best_h = h_lp.copy()
    
    # Phase 3: Multi-start at native resolution
    if remaining() > 50:
        target = n_native / 2.0
        starts = []
        starts.append(project_feasible(1.0 - best_h, target))
        starts.append(project_feasible(best_h + 0.1 * rng.standard_normal(n_native), target))
        starts.append(project_feasible(best_h + 0.2 * rng.standard_normal(n_native), target))
        # Binary-like
        binary = (best_h > 0.5).astype(float)
        binary = project_feasible(binary, target)
        starts.append(project_feasible(0.6 * best_h + 0.4 * binary, target))
        # Low-pass
        from scipy.ndimage import uniform_filter1d
        sm = uniform_filter1d(best_h, size=7)
        starts.append(project_feasible(sm + 0.05 * rng.standard_normal(n_native), target))
        
        for s in starts:
            if remaining() < 15:
                break
            v_s = evaluate_exact(s)
            if v_s < best_val:
                best_val = v_s
                best_h = s.copy()
            h_trial = s.copy()
            time_per = min(remaining() * 0.5, 120)
            h_out, v_out = lp_phase(h_trial, best_val, t_start + time_per, rng)
            if v_out < best_val:
                best_val = v_out
                best_h = h_out.copy()
    
    # Phase 4: Upscale to 2x and optimize
    if remaining() > 100 and n_native <= 400:
        h_up = upscale(best_h, 2)
        h_up = project_feasible(h_up, len(h_up) / 2.0)
        v_up = evaluate_exact(h_up)
        h_work = h_up.copy()
        time_up = min(remaining() * 0.5, 300)
        h_up_opt, v_up_opt = lp_phase(h_work, min(best_val, v_up), t_start + time_up, rng)
        if v_up_opt < best_val:
            best_val = v_up_opt
            best_h = h_up_opt.copy()
        else:
            # Try downscaling the optimized version back
            if len(h_up_opt) > n_native and len(h_up_opt) % n_native == 0:
                h_dn = downscale(h_up_opt, len(h_up_opt) // n_native)
                h_dn = project_feasible(h_dn, n_native / 2.0)
                v_dn = evaluate_exact(h_dn)
                if v_dn < best_val:
                    best_val = v_dn
                    best_h = h_dn.copy()
    
    # Phase 5: If we have a high-res solution, also try 3x upscale of native
    if remaining() > 80 and n_native <= 300:
        h_up3 = upscale(best_h, 3) if len(best_h) == n_native else upscale(best_h, 2)
        h_up3 = project_feasible(h_up3, len(h_up3) / 2.0)
        h_tmp, v_tmp = lp_phase(h_up3, best_val, t_start + min(remaining() * 0.4, 180), rng)
        if v_tmp < best_val:
            best_val = v_tmp
            best_h = h_tmp.copy()
    
    # Phase 6: Final fine-tuning on the best solution
    if remaining() > 15:
        h_fin = best_h.copy()
        n_fin = len(h_fin)
        trust_f = 0.03
        for _ in range(200):
            if remaining() < 5:
                break
            corr = correlation_array(h_fin)
            d = lp_solve(h_fin, corr, trust_f, max(5, n_fin // 3))
            if d is not None:
                for alpha in [1.0, 0.5, 0.25, 0.1]:
                    h_new = h_fin + alpha * d
                    if np.any(h_new < -1e-9) or np.any(h_new > 1 + 1e-9):
                        continue
                    val = evaluate_exact(h_new)
                    if val < best_val - 1e-15:
                        best_val = val
                        best_h = h_new.copy()
                        h_fin = h_new.copy()
                        break
            trust_f = max(0.002, trust_f * 0.9)
    
    # Final: ensure feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)
    
    return best_h, float(best_val), len(best_h)
