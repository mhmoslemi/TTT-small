import numpy as np
import time

def make_feasible(x, target):
    """Project x to [0,1]^n with sum = target using bisection + adjustment."""
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    x = np.clip(x, 0.0, 1.0)
    
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5)
    
    # Bisection on offset
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fine adjustment
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            # Need to add more mass
            order = np.argsort(x)  # most room first
            rem = diff
            for i in order:
                add = min(1.0 - x[i], rem)
                x[i] += add
                rem -= add
                if rem < 1e-15:
                    break
        else:
            # Need to remove mass
            order = np.argsort(x)[::-1]  # most mass first
            rem = -diff
            for i in order:
                rem_i = min(x[i], rem)
                x[i] -= rem_i
                rem -= rem_i
                if rem < 1e-15:
                    break
    
    x = np.clip(x, 0.0, 1.0)
    return x

def corr_values(h):
    """Compute all 2n-1 correlation values (not yet scaled by dx)."""
    n = len(h)
    return np.correlate(h, 1.0 - h, "full")

def exact_C5(h):
    """Compute C5 = max of correlation values * dx."""
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, "full")
    return float(np.max(c) * dx)

def softmax_weights(c, tau):
    """Compute softmax weights alpha[k] = exp((c[k]-max)/tau) / sum(...)"""
    cmax = float(np.max(c))
    if tau < 1e-15:
        # Hard max: uniform over argmax
        alpha = np.zeros_like(c)
        amax = c >= cmax - 1e-12
        if np.any(amax):
            alpha[amax] = 1.0 / float(np.sum(amax))
        else:
            alpha[int(np.argmax(c))] = 1.0
        return alpha
    a = (c - cmax) / tau
    e = np.exp(a)
    s = float(np.sum(e))
    if s < 1e-300:
        alpha = np.zeros_like(c)
        alpha[int(np.argmax(c))] = 1.0
        return alpha
    return e / s

def smoothed_gradient(h, alpha):
    """Compute gradient of smoothed objective.
    alpha[k] are the softmax weights for each lag.
    Returns gradient has same length as h.
    """
    n = len(h)
    dx = 2.0 / float(n)
    grad = np.zeros(n)
    # corr[k] corresponds to lag d = k - (n-1)
    ctr = n - 1
    for k in range(len(alpha)):
        a = alpha[k]
        if a < 1e-15:
            continue
        d = k - ctr  # lag
        coeff = a * dx
        if d == 0:
            grad += coeff * (1.0 - 2.0 * h)
        elif d > 0:
            # grad[m] += coeff * (1-h[m+d]) for m where m+d < n
            # grad[m] += -coeff * h[m-d] for m where m-d >= 0
            grad[:n-d] += coeff * (1.0 - h[d:])
            grad[d:] += -coeff * h[:n-d]
        else:
            dd = -d
            grad[dd:] += coeff * (1.0 - h[:n-dd])
            grad[:n-dd] += -coeff * h[dd:]
    return grad

def projected_direction(h, grad, target):
    """Compute feasible descent direction and max step size."""
    # Project gradient to sum-zero subspace
    s = grad - np.mean(grad)
    # Find max step for h - alpha*s to stay in [0,1]
    # h[i] - alpha*s[i] in [0,1]
    # If s[i] > 0: alpha < h[i]... wait, h[i] - alpha*s[i] >= 0 → alpha <= h[i]/s[i]
    #              alpha < (1-h[i])/s[i] ... no
    # h[i] - alpha*s[i] >= 0 and h[i] - alpha*s[i] <= 1
    # If s[i] > 0: h[i] - alpha*s[i] >= 0 → alpha <= h[i]/s[i]
    #              h[i] - alpha*s[i] <= 1 → alpha >= (h[i]-1)/s[i] (always true if h[i]<=1)
    # If s[i] < 0: h[i] - alpha*s[i] >= 0 → alpha >= -h[i]/s[i] (since s[i]<0, -h[i]/s[i] = h[i]/|s[i]|)
    #              h[i] - alpha*s[i] <= 1 → alpha <= (1-h[i])/|s[i]|
    
    n = len(h)
    max_step = np.inf
    
    pos = s > 1e-15
    if np.any(pos):
        max_step = min(max_step, float(np.min(h[pos] / s[pos])))
    
    neg = s < -1e-15
    if np.any(neg):
        max_step = min(max_step, float(np.min((1.0 - h[neg]) / (-s[neg]))))
    
    if max_step <= 1e-15:
        return None, 0.0
    
    max_step = min(max_step, 1.0) * 0.99  # safety margin
    return s, max_step

def line_search_backtrack(h, s, max_step, c0, target, cands=None):

    if cands is None:
        alphas = [max_step, 0.7*max_step, 0.5*max_step, 0.3*max_step,
                  0.2*max_step, 0.1*max_step, 0.05*max_step, 0.02*max_step,
                  0.01*max_step, 0.005*max_step]
    else:
        alphas = cands
    
    best_h = None
    best_c = c0
    
    for a in alphas:
        if a <= 0 or a > max_step:
            continue
        h_new = h - a * s
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = make_feasible(h_new, target)
        c_new = exact_C5(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
    
    return best_h, best_c

def pair_move_eval(h, c_vals, dx, i, j, delta):

def fourier_start(N, rng):
    """Generate a Fourier-based initial point."""
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    # Sum of a few cosine terms
    n_terms = int(rng.integers(2, 6))
    y = np.zeros(N)
    for idx in range(n_terms):
        f = int(rng.integers(1, N // 3 + 1))
        ph = float(rng.uniform(0, 2 * np.pi))
        amp = float(rng.uniform(0.05, 0.3))
        y += amp * np.cos(np.pi * f * x + ph)
    y = 0.5 + y
    return make_feasible(y, 0.5 * float(N))

def resample_h(h, N):

    n0 = len(h)
    target = 0.5 * float(N)
    if n0 == N:
        return make_feasible(h, target)
    
    x0 = (np.arange(n0) + 0.5) * (2.0 / float(n0))

        y = np.zeros(N)
        for t in range(n_terms):
            f = int(rng.integers(1, max(2, N // 4)))
            ph = float(rng.uniform(0, 2 * np.pi))
            amp = float(rng.uniform(0.03, 0.25))
            y += amp * np.cos(np.pi * f * x + ph)
        y = 0.5 + y + rng.normal(0, 0.02, N)
        h_cand = make_feasible(y, 0.5 * float(N))
        c_cand = exact_C5(h_cand)
        if c_cand < best_c:
            best_c = c_cand
            best_h = h_cand
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    hard_dl = t0 + total - 5.0
    rng = np.random.default_rng(int(seed) % (1 << 32))

    hinit = globals().get("initial_h_values", None)
    if hinit is not None:
        hinit = np.asarray(hinit, dtype=float).ravel()
        if hinit.size > 0 and np.all(np.isfinite(hinit)):
            n0 = hinit.size
            best_h = make_feasible(hinit, 0.5 * float(n0))
        else:
            n0 = 360
            best_h = make_feasible(np.full(n0, 0.5), 0.5 * float(n0))
    else:
        n0 = 360
        best_h = make_feasible(np.full(n0, 0.5), 0.5 * float(n0))

    best_n = len(best_h)
    best_c = exact_C5(best_h)

    # Phase 1: Gradient descent (broad)
    dl1 = min(hard_dl - 80.0, t0 + 0.30 * total)
    if time.time() < dl1:
        h_opt, c_opt = gradient_step(best_h, 0.5 * best_n, 0.012, dl1)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, len(h_opt)

    # Phase 2: Pair golden-section search (fine)
    dl2 = min(hard_dl - 60.0, t0 + 0.50 * total)
    if time.time() < dl2:
        h_opt, c_opt = pair_refine(best_h, 0.5 * best_n, 80, rng, dl2, max_pairs_per_round=15)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, len(h_opt)

    # Phase 3: Gradient descent (fine, small tau)
    dl3 = min(hard_dl - 50.0, t0 + 0.65 * total)
    if time.time() < dl3:
        h_opt, c_opt = gradient_step(best_h, 0.5 * best_n, 0.003, dl3)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, len(h_opt)

    # Phase 4: Pair refinement again
    dl4 = min(hard_dl - 40.0, t0 + 0.78 * total)
    if time.time() < dl4:
        h_opt, c_opt = pair_refine(best_h, 0.5 * best_n, 60, rng, dl4, max_pairs_per_round=12)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, len(h_opt)

    # Phase 5: Try different resolutions
    if time.time() < hard_dl - 80.0:
        res_list = []
        for mult in (0.75, 1.25, 1.5, 0.67):
            N = int(round(best_n * mult))
            N = max(200, min(1200, N))
            if N != best_n and N not in res_list:
                res_list.append(N)
        for N in res_list[:2]:
            if time.time() > hard_dl - 60.0 or best_c <= 0.3808:
                break
            h_res = resample_h(best_h, N)
            stage_dl = min(hard_dl - 50.0, time.time() + 50.0)
            h_opt, c_opt = gradient_step(h_res, 0.5 * N, 0.008, stage_dl)
            stage_dl2 = min(hard_dl - 40.0, time.time() + 30.0)
            h_opt2, c_opt2 = pair_refine(h_opt, 0.5 * N, 30, rng, stage_dl2, max_pairs_per_round=10)
            c_final = min(c_opt, c_opt2)
            h_final = h_opt if c_opt <= c_opt2 else h_opt2
            if c_final < best_c:
                best_h, best_c, best_n = h_final, c_final, len(h_final)

    # Phase 6: Final intensive pair + gradient polish
    dl6 = hard_dl - 8.0
    if time.time() < dl6:
        h_opt, c_opt = pair_refine(best_h, 0.5 * best_n, 40, rng, dl6, max_pairs_per_round=10)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, len(h_opt)
        if time.time() < hard_dl - 5.0:
            h_opt, c_opt = gradient_step(best_h, 0.5 * best_n, 0.002, hard_dl - 5.0)
            if c_opt < best_c:
                best_h, best_c, best_n = h_opt, c_opt, len(h_opt)

    # Final feasibility check
    best_h = make_feasible(best_h, 0.5 * float(len(best_h)))
    best_c = exact_C5(best_h)
    best_n = len(best_h)

    return best_h, float(best_c), int(best_n)
