import time

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def project_feasible(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        t = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - t, 0.0, 1.0))
        if s > target:
            lo = t
        else:
            hi = t
    out = np.clip(h - 0.5*(lo+hi), 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-10:
        if d > 0:
            mask = out > 1e-15
            cnt = mask.sum()
            if cnt > 0:
                out[mask] -= d / cnt
                out = np.clip(out, 0.0, 1.0)
        else:
            mask = out < 1.0 - 1e-15
            cnt = mask.sum()
            if cnt > 0:
                out[mask] += d / cnt  # d is negative, so this subtracts
                out = np.clip(out, 0.0, 1.0)
    # Final fix
    d = out.sum() - target
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argmax(out)
            out[idx] = max(0.0, out[idx] - d)
        else:
            idx = np.argmin(out)
            out[idx] = min(1.0, out[idx] - d)
    return out

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def get_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def lse_objective(h, sigma):
    c = get_corr(h)
    m = c.max()
    return m + math.log(np.sum(np.exp(sigma * (c - m)))) / sigma

def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-16:
            continue
        lag = k - center
        if lag == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += wk * (1.0 - h[:-lag])
            grad[:-lag] -= wk * h[lag:]
        else:
            p = -lag
            grad[:-p] += wk * (1.0 - h[p:])
            grad[p:] -= wk * h[:-p]
    return grad * dx

def topk_gradient(h, K):
    """Gradient of the sum of top-K correlations (times dx)."""
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    # Get top K indices
    if K >= len(c):
        idx = np.argsort(c)[::-1]
    else:
        # Partial sort for efficiency
        part = np.argpartition(c, -K)[-K:]
        idx = part[np.argsort(c[part])[::-1]]
    center = n - 1
    grad = np.zeros(n)
    for k in idx:
        lag = k - center
        if lag == 0:
            grad += (1.0 - 2.0 * h)
        elif lag > 0:
            grad[lag:] += (1.0 - h[:-lag])
            grad[:-lag] -= h[lag:]
        else:
            p = -lag
            grad[:-p] += (1.0 - h[p:])
            grad[p:] -= h[:-p]
    return grad * dx / K  # average of top-K gradients

def lse_optimize(h0, target, sigma_schedule, time_limit, rng, use_momentum=True):
    """LSE optimization with optional momentum."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    n = len(h)
    t0 = time.monotonic()
    
    momentum = np.zeros(n) if use_momentum else None
    beta = 0.9
    alpha = 0.1
    
    for sigma in sigma_schedule:
        if time.monotonic() - t0 > time_limit:
            break
        # Inner loop for this sigma
        min_obj = lse_objective(h, sigma)
        best_obj_for_sigma = min_obj
        inner_time = min(5.0, (time_limit - (time.monotonic() - t0)) * 0.3)
        inner_t0 = time.monotonic()
        stall = 0
        
        while time.monotonic() - inner_t0 < inner_time:
            g = lse_gradient(h, sigma)
            if use_momentum:
                momentum = beta * momentum + (1 - beta) * g
                step = momentum
            else:
                step = g
            
            gn = np.linalg.norm(step)
            if gn < 1e-14:
                break
            
            # Line search
            improved = False
            a = alpha
            for _ in range(30):
                cand = project_feasible(h - a * step, target)
                obj = lse_objective(cand, sigma)
                if obj < min_obj - 1e-14:
                    h = cand
                    min_obj = obj
                    c5 = evaluate_c5(h)
                    if c5 < best_c5 - 1e-13:
                        best_c5 = c5
                        best_h = h.copy()
                    if obj < best_obj_for_sigma - 1e-12:
                        best_obj_for_sigma = obj
                        stall = 0
                    else:
                        stall += 1
                    improved = True
                    break
                a *= 0.5
            
            if not improved:
                # Try perturbing
                h = project_feasible(h + rng.normal(0, 0.02, n), target)
                min_obj = lse_objective(h, sigma)
                stall += 1
                if stall > 10:
                    alpha *= 0.5
                    stall = 0
                    if alpha < 1e-8:
                        break
        
        # Update alpha for next sigma
        alpha = max(0.01, min(alpha * 1.5, 0.3))
        momentum = np.zeros(n)  # reset momentum between sigmas
    
    return best_h, best_c5

def targeted_moves(h0, target, time_limit, rng):
    """Pair moves targeting active lags."""
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    n = len(h)
    t0 = time.monotonic()
    center = n - 1
    
    while time.monotonic() - t0 < time_limit:
        c = get_corr(h)
        # Find top 5 lags
        top_idx = np.argsort(c)[-5:]
        active_lags = [int(top_idx[i]) - center for i in range(len(top_idx))]
        
        # Generate candidate moves targeting active lags
        moved = False
        for lag in active_lags:
            if abs(lag) == 0 or abs(lag) >= n:
                continue
            lag = int(lag)
            # For this lag, find the position m where h[m+lag]*(1-h[m]) is largest
            if lag > 0:
                valid_m = np.arange(0, n - lag)
            else:
                valid_m = np.arange(-lag, n)
            
            if len(valid_m) == 0:
                continue
            
            # Compute the products
            if lag > 0:
                prods = h[lag:] * (1.0 - h[:-lag])
            else:
                p = -lag
                prods = h[:-p] * (1.0 - h[p:])
            
            im = int(np.argmax(prods))
            
            # For lag > 0: the high product is at positions im and im+lag
            # To reduce: move mass from im+lag to im (decrease h[im+lag], increase h[im])
            # For lag < 0: positions are im and im+p
            # To reduce: move mass from im to im+p
            
            if lag > 0:
                i_src = im + lag  # has high h, contributes to product
                i_dst = im        # has low h (high 1-h), we want to increase it
            else:
                p = -lag
                i_src = im + p
                i_dst = im
            
            if i_src == i_dst:
                continue
            
            # Try moving a small amount from src to dst
            for frac in [0.3, 0.15, 0.05, 0.02]:
                delta = frac * min(h[i_src], 1.0 - h[i_dst])
                if delta < 1e-6:
                    continue
                cand = h.copy()
                cand[i_src] -= delta
                cand[i_dst] += delta
                cand = project_feasible(cand, target)
                c5 = evaluate_c5(cand)
                if c5 < best_c5 - 1e-14:
                    h = cand
                    best_c5 = c5
                    best_h = h.copy()
                    moved = True
                    break
        
        if not moved:
            # Random pair move
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                delta = rng.random() * 0.1
                if rng.random() < 0.5:
                    if h[i] >= delta and h[j] + delta <= 1:
                        cand = h.copy()
                        cand[i] -= delta
                        cand[j] += delta
                        c5 = evaluate_c5(cand)
                        if c5 < best_c5 - 1e-14:
                            h = cand
                            best_c5 = c5
                            best_h = h.copy()
    
    return best_h, best_c5

def anneal_targeted(h0, target, time_limit, rng):

            im = int(rng.choice(valid))
            if lag > 0:

            else:
                i_src, i_dst = im + (-lag), im
            # Sometimes use the "wrong" direction for exploration

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 12.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        parent = make_binary(131, rng)
    n0 = len(parent)
    target0 = n0 / 2.0
    h0 = project_feasible(parent.copy(), target0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)

    # Phase 1: LSE main optimization
    alloc1 = 0.42 * effective
    ch, cc = lse_optimize(best_h, target0,
                          [400, 200, 100, 60, 40, 25, 15, 10, 7, 5, 4, 3, 2, 1.5, 1, 0.7, 0.5],
                          alloc1, rng, deadline)
    if cc < best_c5:
        best_c5 = cc
        best_h = ch

    # Phase 2: Subgradient (top-K) optimization
    if time.monotonic() < deadline:
        alloc2 = min(0.18 * effective, deadline - time.monotonic())
        ch, cc = subgrad_optimize(best_h, target0, [1, 2, 3, 5, 8, 12, 20],
                                  alloc2, rng, deadline)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch

    # Phase 3: Targeted local search
    if time.monotonic() < deadline:
        alloc3 = min(0.14 * effective, deadline - time.monotonic())
        ch, cc = targeted_local(best_h, target0, alloc3, rng, deadline)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch

    # Phase 4: Biased annealing (global escape)
    if time.monotonic() < deadline:
        alloc4 = min(0.12 * effective, deadline - time.monotonic())
        ch, cc = anneal_biased(best_h, target0, alloc4, rng, deadline)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch

    # Phase 5: Post-anneal LSE polish
    if time.monotonic() < deadline:
        alloc5 = min(0.06 * effective, deadline - time.monotonic())
        ch, cc = lse_optimize(best_h, target0, [8, 4, 2, 1, 0.5, 0.3],
                              alloc5, rng, deadline)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch

    # Phase 6: Resolution sweep
    if time.monotonic() < deadline:
        for dn in [2, -2, 4, -4, 6, -6]:
            if time.monotonic() > deadline - 0.5:
                break
            nn = n0 + dn
            if nn < 80 or nn > 200:
                continue
            alloc = min(0.025 * effective, deadline - time.monotonic())
            if alloc < 1.0:
                continue
            rs = project_feasible(resample_h(best_h, nn), nn / 2.0)
            ch, cc = lse_optimize(rs, nn / 2.0, [20, 10, 5, 2, 1, 0.5],
                                  alloc, rng, deadline)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch

    # Phase 7: Final targeted polish
    if time.monotonic() < deadline:
        alloc7 = min(0.04 * effective, deadline - time.monotonic())
        if alloc7 > 0.5:
            ch, cc = targeted_local(best_h, target0, alloc7, rng, deadline)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch

    # Final: ensure feasibility
    best_h = project_feasible(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    n_pts = len(best_h)
    return best_h, float(best_c5), int(n_pts)
