    """Golden section search for min of f(t) on [lo, hi]."""
    gr = (math.sqrt(5) - 1) / 2
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = _line_f(h, i, j, c)
    fd = _line_f(h, i, j, d)
    for _ in range(max_iter):
        if fc < fd:
            b = d; d = c; fd = fc
            c = b - gr * (b - a)
            fc = _line_f(h, i, j, c)
        else:
            a = c; c = d; fc = fd
            d = a + gr * (b - a)
            fd = _line_f(h, i, j, d)
    t = 0.5 * (a + b)
    return t, _line_f(h, i, j, t)

So when i and j appear in the same correlation term (i.e., j = i+ℓ or i = j+ℓ), there's a cross term. This makes the analysis more complex.

In general, for a given lag ℓ:
- If j = i+ℓ (i.e., j-i = ℓ): the term m=j (or m=i depending on convention) has both h[i] and h[j]. Change involves t².
- If i = j+ℓ: same thing.
- Otherwise: the changes are independent and linear.

Specifically, for lag ℓ:
- Position i appears as h[m+ℓ] when m = i-ℓ (if valid), and as (1-h[m]) when m = i (if valid).
- Position j appears as h[m+ℓ] when m = j-ℓ (if valid), and as (1-h[m]) when m = j (if valid).

If i-ℓ = j (i.e., i = j+ℓ): then the term at m=j has h[j+ℓ]*(1-h[j]) = h[i]*(1-h[j]). Both change.
If j-ℓ = i (i.e., j = i+ℓ): then the term at m=i has h[i+ℓ]*(1-h[i]) = h[j]*(1-h[i]). Both change.

So the special cases are when j-i = ℓ or i-j = ℓ.

def obj_val(h):
    n = len(h)
    return float(np.correlate(h, 1.0-h, mode='full').max() * 2.0/n)

def grad_active(h, tau_frac):
    """Gradient of active-excess weighted correlation sum."""
    n = len(h)
    dx = 2.0/n
    c = np.correlate(h, 1.0-h, mode='full')
    cm = c.max()
    tau = cm * (1.0 - tau_frac)
    w = np.maximum(0.0, c - tau) * dx
    center = n - 1
    g = np.zeros(n)
    if w[center] > 0:
        g += w[center] * (1.0 - 2.0*h)
    for p in range(1, n):
        wp = w[center+p]
        wm = w[center-p]
        if wp > 0:
            g[p:] += wp * (1.0 - h[:-p])
            g[:-p] -= wp * h[p:]
        if wm > 0:
            g[:-p] += wm * (1.0 - h[p:])
            g[p:] -= wm * h[:-p]
    return g, cm * dx

def line_eval(h, i, j, t):
    """Evaluate objective after transferring t from i to j."""
    h2 = h.copy()
    h2[i] -= t
    h2[j] += t
    return obj_val(h2)

def pair_search(h, i, j, deadline):
    """Line search for optimal t in pair (i,j)."""
    t_lo = -min(h[j], 1.0 - h[i])
    t_hi = min(h[i], 1.0 - h[j])
    if t_hi - t_lo < 1e-14:
        return 0.0, obj_val(h)
    
    # Grid of 10 points
    n_grid = 10
    pts = np.linspace(t_lo, t_hi, n_grid)
    best_t = 0.0
    best_v = obj_val(h)
    for t in pts:
        if abs(t) < 1e-14:
            continue
        if time.monotonic() > deadline:
            break
        v = line_eval(h, i, j, t)
        if v < best_v:
            best_v = v
            best_t = t
    
    # Golden section refinement around best
    span = (t_hi - t_lo) / n_grid
    lo = max(t_lo, best_t - 2*span)
    hi = min(t_hi, best_t + 2*span)
    if hi - lo > 1e-10 and time.monotonic() < deadline - 0.1:
        gr = (math.sqrt(5) - 1) / 2.0
        a, b = lo, hi
        c1 = b - gr*(b-a)
        c2 = a + gr*(b-a)
        f1 = line_eval(h, i, j, c1)
        f2 = line_eval(h, i, j, c2)
        for _ in range(20):
            if time.monotonic() > deadline:
                break
            if f1 < f2:
                b = c2; c2 = c1; f2 = f1
                c1 = b - gr*(b-a)
                f1 = line_eval(h, i, j, c1)
            else:
                a = c1; c1 = c2; f1 = f2
                c2 = a + gr*(b-a)
                f2 = line_eval(h, i, j, c2)
        t_mid = 0.5*(a+b)
        v_mid = line_eval(h, i, j, t_mid)
        if v_mid < best_v:
            best_v = v_mid
            best_t = t_mid
    
    return best_t, best_v

def pair_polish(h0, rng, time_limit, deadline):
    """Main pair-transfer polishing."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    cur_v = obj_val(h)
    best_h = h.copy()
    best_v = cur_v
    t0 = time.monotonic()
    stall = 0
    
    while time.monotonic() < deadline and (time.monotonic() - t0) < time_limit:
        # Compute gradient
        g, cmv = grad_active(h, 0.15)
        
        # Generate pairs: top gradient positions
        K = 12
        gidx = np.argsort(-np.abs(g))[:K]
        pairs = []
        for a_i in range(len(gidx)):
            for b_i in range(a_i+1, len(gidx)):
                pairs.append((int(gidx[a_i]), int(gidx[b_i])))
        # Add random pairs
        for _ in range(40):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                pairs.append((i, j))
        
        # Try each pair, keep best
        iter_best_t = 0.0
        iter_best_v = cur_v
        iter_best_pair = None
        for (i, j) in pairs:
            if time.monotonic() > deadline:
                break
            t, v = pair_search(h, i, j, deadline)
            if v < iter_best_v - 1e-12:
                iter_best_v = v
                iter_best_t = t
                iter_best_pair = (i, j)
        
        if iter_best_pair is not None:
            h[iter_best_pair[0]] -= iter_best_t
            h[iter_best_pair[1]] += iter_best_t
            h = np.clip(h, 0.0, 1.0)
            cur_v = obj_val(h)
            stall = 0
            if cur_v < best_v - 1e-12:
                best_v = cur_v
                best_h = h.copy()
        else:
            stall += 1
            if stall >= 20:
                # Perturb
                h = repair(h + rng.normal(0, 0.02, n), target)
                cur_v = obj_val(h)
                if cur_v > best_v:  # perturbation made it worse, but that's OK for exploration
                    pass  # continue from perturbed state
                elif cur_v < best_v:
                    best_v = cur_v
                    best_h = h.copy()
                stall = 0
    
    return best_h, best_v
