    w = np.exp(exponents)
    total = w.sum()
    w = w / total
    obj = m + tau * np.log(total)
    return obj, w

Let me now think more carefully about the active-set line search and make it efficient.

For the line search, the key observation is:
- We move in direction d (with sum(d) = 0)
- H(λ) = h + λ*d must stay in [0,1]^N
- C5(H(λ)) = max_r θ_r(H(λ))
- Each θ_r(H(λ)) is a quadratic in λ: θ_r + λ*(d·∇θ_r) + λ²/2*(d^T H_r d)
  where H_r is the Hessian of θ_r

Wait, let me compute the Hessian. For θ_r with shift d_r:
θ_r = dx * sum_i h[i] * (1 - h[i-d_r])  (for d_r >= 0)

∂θ_r/∂h[i] = dx * (1 - h[i-d_r]) for i >= d_r
∂θ_r/∂h[i] = -dx * h[i+d_r] for i < N - d_r

∂²θ_r/∂h[i]∂h[j] = -dx * δ_{j, i-d_r} for i >= d_r (i.e., ∂/∂h[j] of (1-h[i-d_r]) is -δ_{j,i-d_r})
And: ∂²θ_r/∂h[i]∂h[j] = -dx * δ_{j, i+d_r} for i < N-d_r (i.e., ∂/∂h[j] of -h[i+d_r] is -δ_{j,i+d_r})

So the Hessian is a (sparse) antisymmetric matrix! The diagonal is 0, and H_r[i,j] = -dx*(δ_{j,i-d_r} + δ_{j,i+d_r}) for valid indices.

This means θ_r(H(λ)) = θ_r(h) + λ*(d·∇θ_r) + λ²/2 * (d^T H_r d)

And d^T H_r d = -dx * sum_i (d[i]*(d[i-d_r] + d[i+d_r])) (for valid indices)

This is a scalar that we can compute efficiently for a given direction d.

So for the line search, we need to find λ that minimizes:
max_r [a_r + b_r * λ + c_r * λ²]

where a_r = θ_r(h), b_r = d·∇θ_r * dx, c_r = d^T H_r d * dx / 2

This is the minimax of quadratics, which is non-trivial in general. But we can do a grid search + refinement.

Actually, for the line search, a practical approach is:
1. Evaluate C5 at λ = 0, λ_max/2, λ_max (and maybe a few more points)
2. Find the minimum by sampling
3. Refine with golden section if needed

The cost of each evaluation is O(N) for the correlation. With N=350 and maybe 50 evaluations per line search, and maybe 100 line searches, that's 50*100*350 ≈ 1.75M operations, which is fast.

    total = w.sum()
    w = w / total
    obj = m + tau * np.log(total)
    return obj, w


def smooth_grad(h, weights):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = weights[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return g


def slsqp_f(h, tau):
    obj, _ = smooth_obj_and_weights(h, tau)
    return obj


def slsqp_g(h, tau):
    _, w = smooth_obj_and_weights(h, tau)
    return smooth_grad(h, w)


def sum_eq(h, target):
    return np.sum(h) - target


def sum_eq_jac(h, target):
    return np.ones_like(h)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h


def upscale_func(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def random_perturb(h, rng, scale):
    N = len(h)
    target_sum = N / 2.0
    noise = rng.normal(0, scale, N)
    h_pert = h + noise
    h_pert = np.clip(h_pert, 0.0, 1.0)
    h_pert = project_feasible(h_pert, target_sum)
    return h_pert


def gradient_single_lag(r, h, N):
    g = np.zeros(N)
    d = r - (N - 1)
    if d >= 0:
        n = N - d
        if n > 0:
            g[d:] += (1.0 - h[:n])
            g[:n] -= h[d:]
    else:
        dd = -d
        n = N - dd
        if n > 0:
            g[:n] += (1.0 - h[dd:])
            g[dd:] -= h[:n]
    return g


def active_set_optimize(h, N, deadline, max_iters=5000):
    target_sum = N / 2.0
    h = h.copy()
    current_c5 = compute_c5(h)
    no_improve = 0
    min_improve = 1e-11

    for it in range(max_iters):
        if time.time() > deadline:
            break

        lags = compute_lags(h)
        max_val = np.max(lags)

        if max_val < current_c5 - 1e-13:
            current_c5 = max_val
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 200:
            break

        sorted_idx = np.argsort(lags)[::-1]
        top_k = min(3, len(sorted_idx))
        top_vals = lags[sorted_idx[:top_k]]
        spread = top_vals[0] - top_vals[-1] + 1e-15
        w = np.exp(-(top_vals - top_vals[-1]) / (0.02 * max_val + 1e-15))
        w /= w.sum()

        g = np.zeros(N)
        for k in range(top_k):
            g += w[k] * gradient_single_lag(sorted_idx[k], h, N)

        g -= np.mean(g)
        dir_norm = np.linalg.norm(g)
        if dir_norm < 1e-12:
            break

        direction = -g / dir_norm

        lam_max = np.inf
        for i in range(N):
            if direction[i] > 1e-15:
                lam_max = min(lam_max, (1.0 - h[i]) / direction[i])
            elif direction[i] < -1e-15:
                lam_max = min(lam_max, -h[i] / direction[i])

        if lam_max < 1e-14:
            break
        lam_max = min(lam_max, 0.3)

        best_lam = 0.0
        best_val = current_c5

        n_samples = 15
        lams = np.linspace(0, lam_max, n_samples + 1)
        for lam in lams[1:]:
            h_test = h + lam * direction
            h_test = np.clip(h_test, 0.0, 1.0)
            h_test = project_feasible(h_test, target_sum)
            v = compute_c5(h_test)
            if v < best_val:
                best_val = v
                best_lam = lam

        if best_lam == 0.0 or best_val >= current_c5 - min_improve:
            no_improve += 3
            continue

        lo = lams[max(0, int(np.argmin([compute_c5(np.clip(h + l * direction, 0, 1)) if l > 0 else current_c5 for l in lams])) - 1)]
        hi = min(lams[-1], lam_max)
        for _ in range(8):
            m1 = lo + (hi - lo) * 0.382
            m2 = lo + (hi - lo) * 0.618
            h1 = project_feasible(np.clip(h + m1 * direction, 0, 1), target_sum)
            h2 = project_feasible(np.clip(h + m2 * direction, 0, 1), target_sum)
            if compute_c5(h1) < compute_c5(h2):
                hi = m2
            else:
                lo = m1
        lam_star = (lo + hi) * 0.5
        h_star = project_feasible(np.clip(h + lam_star * direction, 0, 1), target_sum)
        v_star = compute_c5(h_star)
        if v_star < current_c5 - min_improve:
            h = h_star
            current_c5 = v_star
            no_improve = 0

    return h, current_c5


def pairwise_refine(h, N, deadline, max_threshold=200):
    target_sum = N / 2.0
    h = h.copy()
    current_c5 = compute_c5(h)

    lags = compute_lags(h)
    max_val = np.max(lags)
    active = np.where(lags >= max_val - 1e-9)[0]
    if len(active) == 0:
        active = np.array([np.argmax(lags)])

    g = np.zeros(N)
    for r in active[:3]:
        g += gradient_single_lag(r, h, N)
    g -= np.mean(g)

    order = np.argsort(g)
    n_try = min(N, max_threshold)

    for frac in np.linspace(0.0, 1.0, 10):
        if time.time() > deadline:
            break
        src_start = N - max(2, int(frac * n_try))
        dst_end = min(N, max(2, int(frac * n_try)))
        srcs = order[src_start:]
        dsts = order[:dst_end]

        for i in srcs[:30]:
            if time.time() > deadline:
                break
            for j in dsts[:30]:
                if i == j:
                    continue
                delta_max = min(h[i], 1.0 - h[j])
                if delta_max < 1e-12:
                    continue
                best_d = 0.0
                best_v = current_c5
                for d_frac in [0.2, 0.5, 0.8]:
                    delta = d_frac * delta_max
                    h_test = h.copy()
                    h_test[i] -= delta
                    h_test[j] += delta
                    v = compute_c5(h_test)
                    if v < best_v:
                        best_v = v
                        best_d = delta
                if best_d > 0 and best_v < current_c5 - 1e-12:
                    h[i] -= best_d
                    h[j] += best_d
                    current_c5 = best_v

    return h, current_c5


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 25.0

    h_init = np.array(initial_h_values, dtype=float)
    N_init = len(h_init)

    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    best_N = N_init

    rng = np.random.default_rng(seed)

    n_values = [350, 300, 400, 250]

    for N_target in n_values:
        if time.time() > deadline - 120:
            break

        h = upscale_func(h_init, N_target)
        target_sum = N_target / 2.0
        h = project_feasible(h, target_sum)
        c5_current = compute_c5(h)

        bounds = [(0.0, 1.0)] * N_target
        constraint = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac,
                       'args': (target_sum,)}]

        tau_schedule = [0.04, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005, 0.0002, 0.0001]

        for tau in tau_schedule:
            if time.time() > deadline - 200:
                break
            maxiter = min(500, max(80, int(3000 * tau)))
            try:
                result = minimize(
                    slsqp_f, h.copy(),
                    jac=slsqp_g,
                    args=(tau,),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': maxiter, 'ftol': 1e-15}
                )
                h_candidate = project_feasible(result.x, target_sum)
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < c5_current:
                    h = h_candidate
                    c5_current = c5_candidate
            except Exception:
                pass

        if time.time() < deadline - 80:
            h, c5_current = active_set_optimize(h, N_target, deadline - 80)

        if time.time() < deadline - 40:
            h, c5_current = pairwise_refine(h, N_target, deadline - 40)

        for attempt in range(5):
            if time.time() > deadline - 20:
                break
            scale = 0.015 * (1.0 - attempt / 5.0)
            h_pert = random_perturb(h, rng, max(scale, 0.003))
            try:
                result = minimize(
                    slsqp_f, h_pert.copy(),
                    jac=slsqp_g,
                    args=(0.0003,),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 200, 'ftol': 1e-15}
                )
                h_candidate = project_feasible(result.x, target_sum)
                c5_candidate = compute_c5(h_candidate)
                if c5_candidate < c5_current:
                    h = h_candidate
                    c5_current = c5_candidate
            except Exception:
                pass

        if time.time() < deadline - 10:
            h, c5_current = active_set_optimize(h, N_target, deadline - 10)

        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            best_N = N_target

    if time.time() < deadline - 5:
        final_h, final_c5 = active_set_optimize(best_h, best_N, deadline - 5)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    final_N = len(best_h)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, final_N
