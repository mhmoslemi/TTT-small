def project_sum_box(x, target):

def grad_single_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            g[lag:n] += 1.0 - h[:n - lag]
            g[:n - lag] -= h[lag:]
    else:
        l = -lag
        if l < n:
            g[:n - l] += 1.0 - h[l:]
            g[l:] -= h[:n - l]
    return g

            cross_sector[:] = cross_full[:]
            cw = np.zeros(N_used + 1)
            cw[1:] = np.cumsum(w)
            cumw[:] = cw
            return grad_smoothed(x, w, N, hF, wF, conv_sector, cross_sector, cumw, N_used)

        for attempt in range(3):
            if time.time() > deadline:
                break
            if attempt == 0:
                x0 = best_h.copy()
            elif attempt == 1:
                x0 = best_h + np.random.randn(n) * 0.01
                x0 = project_sum_box(x0, n / 2.0)
            else:
                x0 = best_h + np.random.randn(n) * 0.03
                x0 = project_sum_box(x0, n / 2.0)
            try:
                res = minimize(obj_fn, x0, jac=grad_fn, method='L-BFGS-B',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-16, 'gtol': 1e-8})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-12:
                    best_h = x_new.copy()
                    best_c = c_new
    return best_h, best_c


def method_lp_trust(h0, deadline, max_iters=400):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.15
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        gap = max(0.005 * vmax, 0.003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 60:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-60:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-15:]
        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i] = grad_single_lag(h, lag)
            b_lin[i] = A[i] @ h - v[k]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-0.5]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [2.0]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-13:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                if it % 4 == 0:
                    radius = min(radius * 1.15, 0.3)
            else:
                no_improve += 1
                if no_improve >= 6:
                    radius = max(radius * 0.5, 0.002)
                    no_improve = 0
                    if radius <= 0.002:
                        break
        else:
            no_improve += 1
            if no_improve >= 10:
                radius = max(radius * 0.5, 0.002)
                no_improve = 0
                if radius <= 0.002:
                    break
    return best_h, best_c


def method_projo_grad(h0, deadline, max_iters=3000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.003
    stall = 0
    N = 1
    while N < 3 * n + 2:
        N *= 2
    h_padded = np.zeros(N)
    h_padded[:n] = h
    hF = np.fft.fft(h_padded)
    w_padded = np.zeros(N)
    wF = np.fft.fft(w_padded)
    conv_sector = np.zeros(N)
    cross_sector = np.zeros(N)
    cumw = np.zeros(N + 1)
    for it in range(max_iters):
        if time.time() > deadline:
            break
        c_cur = evaluate_c5(h)
        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        v = corr_all(h)
        m = v.max()
        beta = 50.0
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        w_padded[:] = 0.0
        w_padded[:2 * n - 1] = w
        wF[:] = np.fft.fft(w_padded)
        sub_padded = np.zeros(N)
        sub_padded[:n] = 1 - h
        subF = np.fft.fft(sub_padded)
        cross_full = np.real(np.fft.ifft(wF * np.conj(subF)))
        cross_sector[:] = cross_full[:]
        conv_full = np.real(np.fft.ifft(wF * subF))
        conv_sector[:] = conv_full[:]
        cw = np.zeros(N + 1)
        cw[1:] = np.cumsum(w_padded)
        cumw[:] = cw
        g = grad_smoothed(h, w, N, hF, wF, conv_sector, cross_sector, cumw, N)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        h_new = h - step * g / g_norm
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < c_cur - 1e-14:
            h = h_new
            step = min(step * 1.04, 0.02)
        else:
            step *= 0.85
            if step < 1e-7:
                break
    return best_h, best_c


def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new


def initial_fallback(n=100):
    h = np.full(n, 0.5)
    return project_sum_box(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0
    deadline = start_time + budget_s - safety

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = initial_fallback(100)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: LP trust region on initial grid
    if time_left() > 60:
        lp_dl = min(deadline, start_time + budget_s * 0.20)
        try:
            h1, c1 = method_lp_trust(best_h, lp_dl, max_iters=400)
            if c1 < best_c:
                best_h = h1
                best_c = c1
        except Exception:
            pass

    # Phase 2: Projected gradient polish on initial grid
    if time_left() > 40:
        pg_dl = min(deadline, start_time + budget_s * 0.30)
        try:
            h2, c2 = method_projo_grad(best_h, pg_dl, max_iters=2000)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    # Phase 3: Transfer to larger grid and use FFT L-BFGS
    if time_left() > 120:
        n_cur = len(best_h)
        target_n = min(800, max(400, n_cur * 5))
        if target_n > n_cur:
            h_tr = transfer_resolution(best_h, target_n)
            c_tr = evaluate_c5(h_tr)
            if c_tr <= best_c + 0.01:
                try:
                    lb_dl = min(deadline, start_time + budget_s * 0.60)
                    h3, c3 = method_fft_lbfgs(h_tr, lb_dl, max_betas=10)
                    if c3 < best_c:
                        best_h = h3
                        best_c = c3
                except Exception:
                    pass

    # Phase 4: Second transfer to even larger grid
    if time_left() > 100:
        n_cur = len(best_h)
        if n_cur < 2000:
            target_n2 = min(2000, max(n_cur + 200, int(n_cur * 1.5)))
            if target_n2 > n_cur:
                h_tr2 = transfer_resolution(best_h, target_n2)
                c_tr2 = evaluate_c5(h_tr2)
                if c_tr2 <= best_c + 0.02:
                    try:
                        lb_dl2 = min(deadline, start_time + budget_s * 0.80)
                        h4, c4 = method_fft_lbfgs(h_tr2, lb_dl2, max_betas=8)
                        if c4 < best_c:
                            best_h = h4
                            best_c = c4
                    except Exception:
                        pass

    # Phase 5: Final LP on a smaller grid for precision
    if time_left() > 50:
        n_final = len(best_h)
        if n_final > 150:
            h_small = transfer_resolution(best_h, 150)
            c_small = evaluate_c5(h_small)
            try:
                lp_dl2 = min(deadline, start_time + budget_s * 0.90)
                h5, c5 = method_lp_trust(h_small, lp_dl2, max_iters=300)
                if c5 < best_c:
                    best_h = h5
                    best_c = c5
            except Exception:
                pass
        else:
            try:
                lp_dl2 = min(deadline, start_time + budget_s * 0.90)
                h5, c5 = method_lp_trust(best_h, lp_dl2, max_iters=300)
                if c5 < best_c:
                    best_h = h5
                    best_c = c5
            except Exception:
                pass

    # Phase 6: Final projected gradient
    if time_left() > 15:
        try:
            h6, c6 = method_projo_grad(best_h, deadline - 1, max_iters=2000)
            if c6 < best_c:
                best_h = h6
                best_c = c6
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
