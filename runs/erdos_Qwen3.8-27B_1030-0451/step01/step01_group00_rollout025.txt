def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_feasible(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float)
    lo = float(np.min(p)) - 1.5
    hi = float(np.max(p)) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(p - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lamb = 0.5 * (lo + hi)
    result = np.clip(p - lamb, 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-14:
        interior = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            result[interior] += residual / cnt
    return np.clip(result, 0.0, 1.0)

def smooth_grad(h, corr, tau):
    n = len(h)
    cm = np.max(corr)
    cs = corr - cm
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:m]
        else:
            L = -lag
            m = n - L
            grad[:m] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:m]
    return grad

    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old  # normalized to [0,1]
    x_new = (np.arange(n_new) + 0.5) / n_new - 0.5  # hmm this doesn't work simply
    
    # Actually, the samples are at midpoints of equal intervals
    # x_old[i] = (i + 0.5) / n_old * 2  (in [0,2])
    # x_new[j] = (j + 0.5) / n_new * 2
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

    # If grad[i] > 0, we want to decrease h[i] (move mass away from i)
    # If grad[j] < 0, we want to increase h[j] (move mass to j)
    # So a good pair is (i, j) where grad[i] > 0 and grad[j] < 0
    # Score: grad[i] - grad[j] (larger is better)
    
    # Get top sources and sinks
    sources = np.argsort(grad)[-top_k//2:]  # highest grad
    sinks = np.argsort(grad)[:top_k//2]  # lowest grad
    
    pairs = []
    for i in sources:
        for j in sinks:
            if i != j:
                pairs.append((int(i), int(j)))
    
    return pairs

import numpy as np
import time


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def project_feasible(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float)
    lo = float(np.min(p)) - 1.5
    hi = float(np.max(p)) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(p - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lamb = 0.5 * (lo + hi)
    result = np.clip(p - lamb, 0.0, 1.0)
    residual = target - float(np.sum(result))
    if abs(residual) > 1e-14:
        interior = (result > 1e-12) & (result < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            result[interior] += residual / cnt
    return np.clip(result, 0.0, 1.0)


def smooth_grad(h, corr, tau):
    n = len(h)
    cm = float(np.max(corr))
    cs = corr - cm
    w = np.exp(tau * cs)
    ws = float(np.sum(w))
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            m = n - lag
            grad[:m] += wk * (1.0 - h[lag:])
            grad[lag:] -= wk * h[:m]
        else:
            L = -lag
            m = n - L
            grad[:m] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:m]
    return grad


def pair_line_search_fast(h, i, j):
    hi = h[i]
    hj = h[j]
    a_upper = min(hi, 1.0 - hj)
    a_lower = -min(hj, 1.0 - hi)

    if a_upper - a_lower < 1e-10:
        return compute_c5(h)

    best_alpha = 0.0
    best_c = compute_c5(h)

    # Quick endpoint + midpoint check
    qm = 0.5 * (a_lower + a_upper)
    for a in (a_lower, qm, a_upper):
        if abs(a) < 5e-3:
            continue
        h[i] = hi - a
        h[j] = hj + a
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_alpha = a

    # Full grid if quick check found improvement
    if abs(best_alpha) > 5e-3:
        n_grid = 20
        alphas = np.linspace(a_lower, a_upper, n_grid)
        for a in alphas:
            if abs(a) < 1e-12:
                continue
            h[i] = hi - a
            h[j] = hj + a
            c = compute_c5(h)
            if c < best_c:
                best_c = c
                best_alpha = a

        # Local refinement
        if abs(best_alpha) > 1e-10:
            margin = (a_upper - a_lower) / n_grid * 0.6
            lo_r = max(best_alpha - margin, a_lower)
            hi_r = min(best_alpha + margin, a_upper)
            for a in np.linspace(lo_r, hi_r, 14):
                if abs(a) < 1e-12:
                    continue
                h[i] = hi - a
                h[j] = hj + a
                c = compute_c5(h)
                if c < best_c:
                    best_c = c
                    best_alpha = a

    h[i] = hi - best_alpha
    h[j] = hj + best_alpha
    return best_c


def gradient_phase(h, time_limit, seed):

    h = h.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)

    step = 1.0
    while time.time() - t0 < time_limit - 2:
        frac = min((time.time() - t0) / max(time_limit, 1.0), 1.0)
        tau = 5.0 + 300.0 * frac

    max_stall = 12
    amp_schedule = [0.01, 0.02, 0.035, 0.05, 0.07]
