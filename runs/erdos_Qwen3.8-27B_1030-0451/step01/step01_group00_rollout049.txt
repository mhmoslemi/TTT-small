    return jac

def lp_direction(h, corr, f, radius, eta=0.01):
    """Find direction delta using LP minimax approach.
    Returns delta (np array of length n) or None if LP fails.
    """
    n = len(h)
    jac = corr_jacobian(h)  # shape (2n-1, n), each row is d/dh (c_k)
    
    # Variables: [delta_0, ..., delta_{n-1}, s_0, ..., s_{n-1}, t]
    # Total: 2n + 1 variables
    nvar = 2 * n + 1
    
    # Objective: eta * sum(s_i) + t
    c_obj = np.zeros(nvar)
    for i in range(n):
        c_obj[n + i] = eta  # s_i coefficients
    c_obj[2 * n] = 1.0     # t coefficient
    
    # Inequality constraints: A_ub @ x <= b_ub
    rows_ub = []
    rows_ube = []
    
    # delta_i <= s_i  =>  delta_i - s_i <= 0
    for i in range(n):
        row = np.zeros(nvar)
        row[i] = 1.0
        row[n + i] = -1.0
        rows_ub.append(row)
        rows_ube.append(0.0)
    
    # -delta_i <= s_i  =>  -delta_i - s_i <= 0
    for i in range(n):
        row = np.zeros(nvar)
        row[i] = -1.0
        row[n + i] = -1.0
        rows_ub.append(row)
        rows_ube.append(0.0)
    
    # delta_i <= radius  =>  delta_i <= radius
    for i in range(n):
        row = np.zeros(nvar)
        row[i] = 1.0
        rows_ub.append(row)
        rows_ube.append(radius)
    
    # -delta_i <= radius  =>  -delta_i <= radius
    for i in range(n):
        row = np.zeros(nvar)
        row[i] = -1.0
        rows_ub.append(row)
        rows_ube.append(radius)
    
    # s_i >= 0 handled by bounds
    
    # Correlation constraints: jac[k] @ delta - t <= -c_k
    # i.e., jac[k] @ delta - t <= -(c_k)
    for k in range(2 * n - 1):
        row = np.zeros(nvar)
        row[:n] = jac[k]
        row[2 * n] = -1.0
        rows_ub.append(row)
        rows_ube.append(-corr[k] * (2.0 / n))  # -c_k where c_k = dx * corr[k]
    
    A_ub = np.array(rows_ub)
    b_ub = np.array(rows_ube)
    
    # Equality: sum(delta_i) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = []
    for i in range(n):
        bounds.append((-radius, radius))  # delta_i
    for i in range(n):
        bounds.append((0.0, None))  # s_i >= 0
    bounds.append((None, None))  # t free
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            delta = res.x[:n].copy()
            t_val = res.x[2 * n]
            if np.all(np.isfinite(delta)) and np.linalg.norm(delta) > 1e-12:
                return delta, t_val
    except Exception:
        pass
    
    return None, None

import numpy as np
import time
from scipy.optimize import linprog


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)


def project_constraint(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float)
    p = np.where(np.isfinite(p), p, 0.5)
    lo = float(np.min(p)) - 2.0
    hi = float(np.max(p)) + 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
    return proj


def corr_gradient_matrix(h):
    n = len(h)
    dx = 2.0 / n
    jac = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        lag = k - (n - 1)
        if lag == 0:
            jac[k, :] = 1.0 - 2.0 * h
        elif lag > 0:
            jac[k, lag:] += (1.0 - h[:n - lag])
            jac[k, :n - lag] -= h[lag:]
        else:
            l = -lag
            jac[k, :n - l] += (1.0 - h[l:])
            jac[k, l:] -= h[:n - l]
    jac *= dx
    return jac


def softmax_weighted_grad(h, jac, tau):
    corr = np.correlate(h, 1.0 - h, mode="full")
    dx = 2.0 / len(h)
    c = corr * dx
    f = np.max(c)
    exponents = tau * (c - f)
    exponents -= np.max(exponents)
    w = np.exp(exponents)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(len(h))
    w /= ws
    return w @ jac


def solve_lp_dir(h, radius, eta, top_k):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx

    if top_k < 2 * n - 1:
        sorted_c = np.sort(c)
        thresh = sorted_c[-top_k]
        active = np.where(c >= thresh - 1e-15)[0]
    else:
        active = np.arange(2 * n - 1)

    jac = corr_gradient_matrix(h)
    nvar = 2 * n + 1

    c_obj = np.zeros(nvar)
    c_obj[n:2 * n] = eta
    c_obj[2 * n] = 1.0

    n_ineq = 4 * n + len(active)
    A = np.zeros((n_ineq, nvar))
    b = np.zeros(n_ineq)

    idx = 0
    for i in range(n):
        A[idx, i] = 1.0
        A[idx, n + i] = -1.0
        idx += 1
    for i in range(n):
        A[idx, i] = -1.0
        A[idx, n + i] = -1.0
        idx += 1
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = radius
        idx += 1
    for i in range(n):
        A[idx, i] = -1.0
        b[idx] = radius
        idx += 1
    for k in active:
        A[idx, :n] = jac[k]
        A[idx, 2 * n] = -1.0
        b[idx] = -c[k]
        idx += 1

    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)

    bounds = [(-radius, radius)] * n + [(0.0, None)] * n + [(None, None)]

    try:
        res = linprog(c_obj, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[:n].copy()
            if np.all(np.isfinite(delta)) and np.max(np.abs(delta)) > 1e-13:
                return delta
    except Exception:
        pass
    return None


def line_search_dir(h, direction, c_cur, max_steps=14):
    n = len(h)
    dnorm = np.linalg.norm(direction)
    if dnorm < 1e-14:
        return None
    direction = direction / dnorm
    step = 0.3
    for _ in range(max_steps):
        h_new = project_constraint(h + step * direction, n)
        c_new = compute_c5(h_new)
        if c_new < c_cur - 1e-14:
            return h_new, c_new
        step *= 0.55
    return None


def line_search_grad(h, grad, c_cur, max_steps=14):
    n = len(h)
    gnorm = np.linalg.norm(grad)
    if gnorm < 1e-14:
        return None
    direction = grad / gnorm
    step = 0.06
    for _ in range(max_steps):
        h_new = project_constraint(h - step * direction, n)
        c_new = compute_c5(h_new)
        if c_new < c_cur - 1e-14:
            return h_new, c_new
        step *= 0.5
    return None


def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.05 * np.sin(6.0 * np.pi * x + 1.3)
    h += 0.02 * np.sin(12.0 * np.pi * x + 2.1)
    h += 0.01 * np.sin(18.0 * np.pi * x + 0.3)
    h += rng.randn(n) * 0.025
    return project_constraint(h, n)


def local_optimize(h0, time_limit, seed_val=42):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()

    radius = 0.03
    stall = 0
    rng = np.random.RandomState(seed_val)
    eta_base = 0.004
    use_lp = (n <= 130)
    top_k = min(2 * n - 1, max(10, (2 * n - 1) // 3))

    while time.time() - t0 < time_limit - 1.0:
        c_cur = compute_c5(h)
        if c_cur < best_c5 - 1e-13:
            best_c5 = c_cur
            best_h = h.copy()

        accepted = False
        eta = eta_base * (1.0 + 0.3 * min(stall, 20) / 10.0)

        if use_lp:
            delta = solve_lp_dir(h, radius, eta, top_k)
            if delta is not None:
                res = line_search_dir(h, delta, c_cur)
                if res is not None:
                    h, c_cur = res
                    accepted = True
                    stall = 0
                    radius = min(0.12, radius * 1.06)

        if not accepted:
            tau = 15.0 + 12.0 * min(stall, 25)
            jac = corr_gradient_matrix(h)
            grad = softmax_weighted_grad(h, jac, tau)
            res = line_search_grad(h, grad, c_cur)
            if res is not None:
                h, c_cur = res
                accepted = True
                stall = 0
                radius = min(0.12, radius * 1.04)

        if not accepted:
            stall += 1
            radius *= 0.72
            if radius < 0.004:
                scale = 0.012 + 0.004 * min(stall, 12)
                h = project_constraint(h + rng.randn(n) * scale, n)
                stall = 0
                radius = 0.03

        c_check = compute_c5(h)
        if c_check < best_c5 - 1e-13:
            best_c5 = c_check
            best_h = h.copy()

    return best_h, best_c5


def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)
