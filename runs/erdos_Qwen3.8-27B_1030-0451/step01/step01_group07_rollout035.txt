def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    res = out.sum() - target
    if abs(res) > 1e-14:
        free = (out > 1e-8) & (out < 1.0 - 1e-8)
        if free.sum() > 0:
            out[free] = out[free] - res / free.sum()
        else:
            idx = int(np.argmax(out if res > 0 else 1.0 - out))
            out[idx] = out[idx] - res
        out = np.clip(out, 0.0, 1.0)
    return out

def compute_corr(h):
    """Compute full correlation of h with 1-h."""
    return np.correlate(h, 1.0 - h, mode='full')

def objective_from_corr(corr, n):
    """Compute C5 from correlation array."""
    dx = 2.0 / n
    return float(corr.max() * dx)

def objective(h):
    n = len(h)
    corr = compute_corr(h)
    return objective_from_corr(corr, n)

def round_to_binary(h, n):
    """Round h to 0-1 valued with exactly n/2 ones."""
    h = np.asarray(h, dtype=float)
    # Sort by value descending, take top n/2
    k = n // 2
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)
    out[idx[:k]] = 1.0
    return out
