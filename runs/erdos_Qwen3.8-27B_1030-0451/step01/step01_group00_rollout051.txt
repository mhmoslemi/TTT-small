    cs = corr - f
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    jac = build_jacobian(h)
    grad = w @ jac  # weighted sum of Jacobian rows

    c_obj[n] = 1.0  # t

        lo_bound = min(radius, h[i])

def pair_optimize(h, n_pairs=200):

    for _ in range(n_pairs):
        i = rng.randint(n)
        j = rng.randint(n)

        max_shift = min(h[i], 1.0 - h[j])

            continue

        for frac in [0.1, 0.3, 0.5, 0.7, 0.9]:

            h_new[j] += s
            c5_new = compute_c5(h_new)
            if c5_new < current_c5 - 1e-12:

                current_c5 = c5_new
                break
    
    return h, current_c5
