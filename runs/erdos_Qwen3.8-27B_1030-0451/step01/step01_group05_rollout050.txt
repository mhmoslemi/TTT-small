```python
def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def project_feasible(z, n):
    S = n / 2.0
    z = np.array(z, dtype=float)
    lo = -5.0
    hi = 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)

        cnt = int(np.sum(interior))
        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

def compute_lin_quad(a, b, h, n):
    """Compute lin and quad arrays for pair transfer (a increases, b decreases)."""
    center = n - 1
```

```python
def compute_lin(a, b, h, n):
    L = 2 * n - 1
    # lin = P_a - P_b - Q_a + Q_b
    # P_a[a:a+n] = 1 - h[::-1], P_b[b:b+n] = 1 - h[::-1]
    # So P_a - P_b has (1-h[::-1]) in positions a..a+n-1 and -(1-h[::-1]) in positions b..b+n-1
    # With overlap.
    
    lin = np.zeros(L)
    # Add P_a contribution
    lin[a:a+n] += 1.0 - h[::-1]
    # Subtract P_b contribution
    lin[b:b+n] -= 1.0 - h[::-1]
    # Subtract Q_a contribution
    lin[n-1-a:2*(n-1)-a+1] -= h
    # Add Q_b contribution
    lin[n-1-b:2*(n-1)-b+1] += h
    return lin
```

Actually, let me reconsider the line search. Instead of a grid + ternary, I can use a smarter approach:

The function f(d) = max_K (raw[K] + d*lin[K] + d²*quad[K]) is the upper envelope of 2n-1 quadratic functions. The minimum of f(d) on [d_lo, d_hi] occurs at either:
1. A vertex of one of the quadratics (where d = -lin[K]/(2*quad[K]) for some K with quad[K] ≠ 0)
2. An intersection point of two of the quadratics
3. An endpoint d_lo or d_hi

Since only 3 quadratics have nonzero quad (the rest are linear), the intersections to check are:
- Intersections of the 3 quadratics with each other: O(1)
- Intersections of each quadratic with the piecewise-linear envelope: O(n)
- Intersections of the linear functions with each other: O(n²) in worst case, but in practice we only need the ones near the minimum

A practical approach:
1. Evaluate f(d) on a coarse grid (say 30 points)
2. Find the best point and its neighbors
3. Do a local ternary search in that neighborhood
4. Also check the vertices of the 3 special quadratics (if they're in [d_lo, d_hi])

```python
import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def project_feasible(z, n):
    S = n / 2.0
    z = np.array(z, dtype=float).copy()
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(z - mid, 0.0, 1.0)) > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(mask))
        if cnt > 0:
            h[mask] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_lin(a, b, h, n):
    L = 2 * n - 1
    lin = np.zeros(L)
    lin[a:a+n] += 1.0 - h[::-1]
    lin[b:b+n] -= 1.0 - h[::-1]
    lin[n-1-a:(n-1-a)+n] -= h
    lin[n-1-b:(n-1-b)+n] += h
    return lin

def make_quad(a, b, n):
    L = 2 * n - 1
    quad = np.zeros(L)
    quad[n-1] = -2.0
    quad[n-1 + (b-a)] = 1.0
    quad[n-1 + (a-b)] = 1.0
    return quad

def line_search(raw, lin, quad, d_lo, d_hi, n_grid=40):
    if d_hi - d_lo < 1e-14:
        return 0.0, float(np.max(raw))
    d_vals = np.linspace(d_lo, d_hi, n_grid)
    # Vectorized evaluation: shape (n_grid, 2n-1)
    vals = raw[np.newaxis, :] + d_vals[:, np.newaxis] * lin[np.newaxis, :] + (d_vals[:, np.newaxis]**2) * quad[np.newaxis, :]
    scores = np.max(vals, axis=1)
    idx = int(np.argmin(scores))
    best_grid_d = d_vals[idx]
    best_grid_s = scores[idx]
    
    # Local refinement with ternary search
    lo = d_vals[max(0, idx-1)]
    hi = d_vals[min(n_grid-1, idx+1)]
    for _ in range(30):
        m1 = lo + (hi - lo) / 3.0
        m2 = hi - (hi - lo) / 3.0
        s1 = np.max(raw + m1 * lin + m1*m1 * quad)
        s2 = np.max(raw + m2 * lin + m2*m2 * quad)
        if s1 <= s2:
            hi = m2
        else:
            lo = m1
        if hi - lo < 1e-12:
            break
    d_opt = (lo + hi) / 2.0
    s_opt = float(np.max(raw + d_opt * lin + d_opt*d_opt * quad))
    
    # Also check endpoints
    s_lo = float(np.max(raw + d_lo * lin + d_lo*d_lo * quad))
    s_hi = float(np.max(raw + d_hi * lin + d_hi*d_hi * quad))
    if s_lo < s_opt:
        d_opt, s_opt = d_lo, s_lo
    if s_hi < s_opt:
        d_opt, s_opt = d_hi, s_hi
    
    return d_opt, s_opt

def smooth_gradient(h, raw, tau):
    n = len(h)
    L = 2 * n - 1
    c_max = np.max(raw)
    exponents = np.clip((raw - c_max) / tau, -500.0, 500.0)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.ones(L) / L
    else:
        weights /= sw
    g = np.zeros(n)
    for K in range(L):
        w = weights[K]
        if w < 1e-15:
            continue
        k = K - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - h[:n-k])
            g[:n-k] += w * (-h[k:])
        else:
            m = -k
            g[:n-m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n-m])
    return g
```
