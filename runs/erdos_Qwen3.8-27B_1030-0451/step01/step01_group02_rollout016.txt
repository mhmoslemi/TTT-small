The method:

The "solve_qp" step: the objective is sum_k w_k * dx * sum_{i in O_k} h[i](1-h[i+k])
= dx * [h^T (sum_k w_k D_k) h + (sum_k w_k e_k)^T h]

where D_k has -1 at (i, i+k) and (i+k, i) for i in O_k, and e_k has 1 at positions in O_k.

This is a QP of the form: min h^T A h + b^T h s.t. 0≤h≤1, sum(h)=target

Where A = -dx * sum_k w_k D_k (which should be PSD for the problem to be convex... let me check).

Actually, h[i](1-h[i+k]) = h[i] - h[i]h[i+k]. So:
sum_k w_k * sum_{i in O_k} h[i](1-h[i+k]) = sum_k w_k * sum_{i in O_k} h[i] - sum_k w_k * sum_{i in O_k} h[i]h[i+k]

The first part is linear: (sum_k w_k e_k)^T h
The second part: sum_k w_k * (h^T P_k h) where P_k has 1/2 at (i,i+k) and (i+k,i) for i in O_k.

So the total is: h^T (-sum_k w_k P_k) h + (sum_k w_k e_k)^T h + const

The Hessian is -2*sum_k w_k P_k. For this to be a minimization (convex), we need the Hessian to be PSD, i.e., sum_k w_k P_k to be NSD (negative semidefinite). But P_k has positive diagonal entries (when i and i+k are both in range, the (i,i) entry gets contributions from P_k and P_{-k})...

def _make_feasible(y, target):

    n = y.size

        return y

def _sl_jac(x, tau, dx):

    else:

        amax = float(np.max(a))

        x = do_slsqp(best_h, tau, maxit, deadline)
        if x is not None:
            c = exact_C5(x)
            if np.isfinite(c) and c < best_c - 1e-14:
                best_h, best_c = x, c
            elif np.isfinite(c) and c < best_c:
                best_h, best_c = x, c

    # Nesterov subgradient phases
    ns_taus = (0.01, 0.005, 0.003, 0.002, 0.001, 0.0005, 0.0003, 0.0002)
    for round_idx in range(3):
        if time.time() > deadline - 20:
            break
        ns_taus_r = ns_taus[min(round_idx, len(ns_taus) - 3):]
        xi, ci = nesterov_subgrad(best_h, target, 200, deadline - 20, ns_taus_r)
        if ci < best_c - 1e-14:
            best_h, best_c = xi, ci
            # Polish with SLSQP
            if time.time() < deadline - 15:
                x = do_slsqp(best_h, 0.001, 40, deadline)
                if x is not None:
                    c = exact_C5(x)
                    if np.isfinite(c) and c < best_c:
                        best_h, best_c = x, c

    # Stochastic local
    if time.time() < deadline - 10:
        ntries = int(max(50, min(300, 5000.0 / max(n, 1))))
        xi, ci = stochastic_local(best_h, target, ntries, deadline, rng)
        if ci < best_c - 1e-14:
            best_h, best_c = xi, ci
            # Polish
            if time.time() < deadline - 10:
                x = do_slsqp(best_h, 0.0005, 30, deadline)
                if x is not None:
                    c = exact_C5(x)
                    if np.isfinite(c) and c < best_c:
                        best_h, best_c = x, c

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    global _sl_target
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    hard = t0 + max(10.0, total) - 8.0
    rng = np.random.default_rng(int(seed) % (1 << 32))

    hinit = get_initial()
    if hinit is None:
        hinit = fallback_init(240)
    n0 = int(hinit.size)
    best_h = hinit.copy()
    best_c = exact_C5(best_h)

    # Phase 1: Optimize at initial resolution
    stage1_deadline = min(hard - 30.0, t0 + 0.4 * total)
    best_h, best_c = optimize_multi_tau(best_h, stage1_deadline, rng)

    # Phase 2: Try other resolutions
    candidates_n = []
    for f in (0.5, 1.0, 1.5, 2.0, 3.0, 4.0):
        N = int(round(n0 * f))
        if 80 <= N <= 2000 and N != n0:
            candidates_n.append(N)
    candidates_n = sorted(set(candidates_n))
    # Also try some fixed sizes
    for N in (240, 360, 480, 600, 720, 960, 1200):
        if 80 <= N <= 2000 and N not in candidates_n:
            candidates_n.append(N)
    candidates_n = sorted(set(candidates_n))

    # Limit to top candidates based on time
    max_res = 4 if total < 500 else 6
    if len(candidates_n) > max_res:
        # Prioritize sizes close to n0 and standard sizes
        candidates_n = candidates_n[:max_res]

    for Nt in candidates_n:
        if time.time() > hard - 40:
            break
        if best_c <= 0.3807:
            break
        remaining = hard - time.time()
        if remaining < 30:
            break

        # Allocate time proportionally
        alloc = min(remaining * 0.4, max(20.0, 0.25 * total / len(candidates_n)))
        stage_dl = min(hard - 20.0, time.time() + alloc)

        h_cand = resample(best_h, Nt)
        # Add perturbation for diversity
        h_cand2 = _make_feasible(h_cand + rng.normal(0.0, 0.03, size=Nt), 0.5 * float(Nt))
        h_to_opt = h_cand if exact_C5(h_cand) <= exact_C5(h_cand2) else h_cand2

        h_opt, c_opt = optimize_multi_tau(h_to_opt, stage_dl, rng)
        if np.isfinite(c_opt) and c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt

    # Phase 3: Final polish at best resolution
    if time.time() < hard - 15:
        n_best = len(best_h)
        final_dl = hard - 3.0
        best_h, best_c = optimize_multi_tau(best_h, final_dl, rng)

    # Ensure feasibility
    n_final = len(best_h)
    best_h = _make_feasible(best_h, 0.5 * float(n_final))
    best_c = exact_C5(best_h)
    if not np.isfinite(best_c):
        best_h = _make_feasible(np.full(n_final, 0.5), 0.5 * float(n_final))
        best_c = exact_C5(best_h)

    return best_h, float(best_c), int(n_final)
