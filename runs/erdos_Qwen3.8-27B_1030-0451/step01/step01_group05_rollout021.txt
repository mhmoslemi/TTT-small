            break
        grad, cmax = subgrad_direction(cur, n)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        improved = False
        for a in [alpha, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            h_new = project_feasible(cur - a * grad, n)
            s_new = exact_score(h_new)
            if s_new < cur_score - 1e-14:
                cur = h_new
                cur_score = s_new
                improved = True
                break
        if improved:
            if cur_score < best_score - 1e-13:
                best = cur.copy()
                best_score = cur_score
        else:
            alpha *= 0.6
            if alpha < 1e-6:
                break
    return best, best_score


def pair_search(h, n, deadline, max_iters=3000, tol_frac=0.1):
    cur = h.copy()
    cur_score = exact_score(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng(12345)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        overlaps = compute_all_overlaps(cur)
        c_max = np.max(overlaps)
        active = np.where(overlaps >= c_max - tol_frac * c_max)[0]
        dx = 2.0 / n
        grad = np.zeros(n)
        for p in active[:30]:
            s = int(p) - (n - 1)
            if s >= 0 and s < n:
                m = n - s
                grad[0:m] += (1.0 - cur[s:s + m])
                grad[s:s + m] -= cur[0:m]
            elif s < 0:
                m = -s
                if m < n:
                    mm = n - m
                    grad[m:n] += (1.0 - cur[0:mm])
                    grad[0:mm] -= cur[m:n]
        grad *= dx
        imin = int(np.argmin(grad))
        imax = int(np.argmax(grad))
        improved = False
        n_try = 4
        for _ in range(n_try):
            if no_improve > 20:
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a == b:
                    continue
            else:
                if _ == 0:
                    a, b = imin, imax
                else:
                    a = imin if rng.random() < 0.5 else int(rng.integers(0, n))
                    b = imax if rng.random() < 0.5 else int(rng.integers(0, n))
                if a == b:
                    continue
            for (ai, bi) in [(a, b), (b, a)]:
                max_d = min(1.0 - cur[ai], cur[bi])
                if max_d < 1e-12:
                    continue
                for frac in [1.0, 0.5, 0.25, 0.1, 0.05]:
                    d = max_d * frac
                    if d < 1e-12:
                        continue
                    cur[ai] += d
                    cur[bi] -= d
                    s = exact_score(cur)
                    cur[ai] -= d
                    cur[bi] += d
                    if s < cur_score - 1e-13:
                        cur[ai] += d
                        cur[bi] -= d
                        cur_score = s
                        improved = True
                        break
                if improved:
                    break
            if improved:
                break
        if improved:
            no_improve = 0
            if cur_score < best_score - 1e-13:
                best = cur.copy()
                best_score = cur_score
        else:
            no_improve += 1
            if no_improve > 300:
                break
    return best, best_score


def block_swap_search(h, n, deadline, max_iters=2000):
    cur = h.copy()
    cur_score = exact_score(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng(99)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        blen = int(rng.integers(2, min(8, n // 3)))
        a = int(rng.integers(0, n - blen))
        b = int(rng.integers(0, n - blen))
        if b <= a + blen and b + blen > a:
            continue
        cur[a:a + blen], cur[b:b + blen] = cur[b:b + blen].copy(), cur[a:a + blen].copy()
        cur = project_feasible(cur, n)
        s = exact_score(cur)
        if s < cur_score - 1e-13:
            cur_score = s
            no_improve = 0
            if cur_score < best_score - 1e-13:
                best = cur.copy()
                best_score = cur_score
        else:
            cur[a:a + blen], cur[b:b + blen] = cur[b:b + blen].copy(), cur[a:a + blen].copy()
            cur = project_feasible(cur, n)
            no_improve += 1
            if no_improve > 200:
                break
    return best, best_score


def reversal_search(h, n, deadline, max_iters=1000):
    cur = h.copy()
    cur_score = exact_score(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng(77)
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        a = int(rng.integers(0, n - 1))
        b = int(rng.integers(a + 1, n))
        cur[a:b] = cur[a:b][::-1].copy()
        s = exact_score(cur)
        if s < cur_score - 1e-13:
            cur_score = s
            no_improve = 0
            if cur_score < best_score - 1e-13:
                best = cur.copy()
                best_score = cur_score
        else:
            cur[a:b] = cur[a:b][::-1].copy()
            no_improve += 1
            if no_improve > 150:
                break
    return best, best_score


def sim_anneal_pair(h, n, deadline, max_iters=5000):
    cur = h.copy()
    cur_score = exact_score(cur)
    best = cur.copy()
    best_score = cur_score
    rng = np.random.default_rng(31337)
    T0 = 0.02
    T_min = 0.001
    n_sa = max_iters
    no_improve = 0
    for it in range(n_sa):
        if time.time() > deadline:
            break
        T = T0 * (T_min / T0) ** (it / max(1, n_sa))
        overlaps = compute_all_overlaps(cur)
        c_max = np.max(overlaps)
        active = np.where(overlaps >= c_max - 0.1 * c_max)[0]
        dx = 2.0 / n
        grad = np.zeros(n)
        for p in active[:20]:
            s = int(p) - (n - 1)
            if s >= 0 and s < n:
                m = n - s
                grad[0:m] += (1.0 - cur[s:s + m])
                grad[s:s + m] -= cur[0:m]
            elif s < 0:
                m = -s
                if m < n:
                    mm = n - m
                    grad[m:n] += (1.0 - cur[0:mm])
                    grad[0:mm] -= cur[m:n]
        grad *= dx
        imin = int(np.argmin(grad))
        imax = int(np.argmax(grad))
        if it % 5 == 0:
            a, b = imin, imax
        else:
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
        if a == b:
            continue
        max_d = min(1.0 - cur[a], cur[b])
        if max_d < 1e-12:
            max_d = min(cur[a], 1.0 - cur[b])
            if max_d < 1e-12:
                continue
            tmp_a, tmp_b = a, b
            a, b = tmp_b, tmp_a
            max_d = min(1.0 - cur[a], cur[b])
            if max_d < 1e-12:
                continue
        d = max_d * (0.3 + 0.7 * rng.random())
        if d < 1e-12:
            continue
        cur[a] += d
        cur[b] -= d
        s_new = exact_score(cur)
        delta = s_new - cur_score
        if delta < 0 or (T > 0 and rng.random() < math.exp(-delta / T)):
            cur_score = s_new
            no_improve = 0
            if cur_score < best_score - 1e-13:
                best = cur.copy()
                best_score = cur_score
        else:
            cur[a] -= d
            cur[b] += d
            no_improve += 1
        if no_improve > 500:
            break
    return best, best_score


def make_binary_balanced(n, rng):
    perm = rng.permutation(n)
    h = np.zeros(n)
    half = n // 2
    h[perm[:half]] = 1.0
    if n % 2 == 1:
        pass
    return h


def make_sturmian(n, slope, phase, rng):
    h = np.zeros(n)
    for i in range(n):
        val = (i + 1) * slope + phase
        h[i] = 1.0 if val - math.floor(val) > 0.5 else 0.0
    return project_feasible(h, n)


def make_spectral_flat(n, rng):
    h = np.full(n, 0.5 + 0.3 * (rng.random(n) - 0.5))
    h = project_feasible(h, n)
    for _ in range(50):
        freqs = np.fft.rfftfreq(n)
        H = np.fft.rfft(h)
        power = np.abs(H) ** 2
        if np.max(power) > 0:
            flat_target = np.sum(power) / len(power)
            grad = np.zeros(n)
            for k in range(len(power)):
                if power[k] > 1.5 * flat_target:
                    w = (power[k] - flat_target) / flat_target
                    if k == 0:
                        grad += w
                    else:
                        grad += 2.0 * w * np.cos(2.0 * np.pi * k * np.arange(n) / n)
            h_new = project_feasible(h - 0.5 * grad / (np.max(np.abs(grad)) + 1e-10), n)
            if exact_score(h_new) < exact_score(h):
                h = h_new
            else:
                break
    return h


def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)


def slsqip_solve(h0, tau, n, maxiter, deadline):
    L = 2 * n - 1
    dx = 2.0 / n

    def sl_obj(x):
        raw = np.correlate(x, 1.0 - x, mode='full')
        c = raw * dx
        c_max = np.max(c)
        exponents = np.clip((c - c_max) / tau, -400, 400)
        return c_max + tau * np.log(np.sum(np.exp(exponents)))

    def sl_grad(x):
        raw = np.correlate(x, 1.0 - x, mode='full')
        c = raw * dx
        c_max = np.max(c)
        exponents = np.clip((c - c_max) / tau, -400, 400)
        weights = np.exp(exponents)
        sw = np.sum(weights)
        if sw < 1e-300:
            return np.zeros(n)
        weights /= sw
        g = np.zeros(n)
        for K in range(2 * n - 1):
            w = weights[K] * dx
            if w < 1e-14:
                continue
            k = K - (n - 1)
            if k >= 0:
                g[k:] += w * (1.0 - x[:n - k])
                g[:n - k] += w * (-x[k:])
            else:
                m = -k
                g[:n - m] += w * (1.0 - x[m:])
                g[m:] += w * (-x[:n - m])
        return g

    def con_fun(x):
        return np.sum(x) - n / 2.0

    def con_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun=sl_obj, x0=h0, jac=sl_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + max(30.0, budget_s - 25.0)

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None:
        n_p = len(parent)
        if not (n_p >= 2 and np.all(np.isfinite(parent))):
            parent = None
    if parent is not None:
        parent = project_feasible(parent, n_p)
    else:
        n_p = 81
        parent = project_feasible(np.full(n_p, 0.5), n_p)

    score_p = exact_score(parent)
    best_h = parent.copy()
    best_score = score_p
    best_n = n_p

    def try_update(h, score, n):
        nonlocal best_h, best_score, best_n
        if score < best_score - 1e-13:
            best_h = h.copy()
            best_score = float(score)
            best_n = int(n)

    # Phase 1: SLSQP on parent
    if time.time() < deadline - 30:
        h_work = parent.copy()
        n = n_p
        for tau in [0.05, 0.03, 0.015, 0.008, 0.004]:
            if time.time() > deadline - 25:
                break
            maxiter = min(150, max(30, int((deadline - time.time()) * 2)))
            h_new = slsqip_solve(h_work, tau, n, maxiter, deadline)
            if h_new is not None:
                s = exact_score(h_new)
                if s < score_p - 1e-10:
                    h_work = h_new
                    score_p = s
                    try_update(h_work, s, n)

    # Phase 2: Subgradient search
    if time.time() < deadline - 60:
        sub_dl = min(deadline - 60, time.time() + 120)
        h_sg, s_sg = subgrad_search(h_work, n_p, sub_dl, max_iters=300)
        if s_sg < best_score - 1e-13:
            try_update(h_sg, s_sg, n_p)
            h_work = h_sg
            score_p = s_sg

    # Phase 3: Pair search
    if time.time() < deadline - 80:
        h_ps, s_ps = pair_search(h_work, n_p, deadline - 80, max_iters=4000)
        if s_ps < best_score - 1e-13:
            try_update(h_ps, s_ps, n_p)
            h_work = h_ps
            score_p = s_ps

    # Phase 4: Simulated annealing
    if time.time() < deadline - 100:
        sa_dl = min(deadline - 100, time.time() + 150)
        h_sa, s_sa = sim_anneal_pair(h_work, n_p, sa_dl, max_iters=6000)
        if s_sa < best_score - 1e-13:
            try_update(h_sa, s_sa, n_p)
            h_work = h_sa
            score_p = s_sa

    # Phase 5: Block and reversal moves
    if time.time() < deadline - 50:
        h_bl, s_bl = block_swap_search(h_work, n_p, deadline - 30, max_iters=1000)
        if s_bl < best_score - 1e-13:
            try_update(h_bl, s_bl, n_p)
            h_work = h_bl
            score_p = s_bl
    if time.time() < deadline - 30:
        h_rv, s_rv = reversal_search(h_work, n_p, deadline - 20, max_iters=800)
        if s_rv < best_score - 1e-13:
            try_update(h_rv, s_rv, n_p)
            h_work = h_rv

    # Phase 6: SLSQP polish
    for tau in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > deadline - 20:
            break
        maxiter = min(200, max(40, int((deadline - time.time()) * 3)))
        h_new = slsqip_solve(h_work, tau, n_p, maxiter, deadline)
        if h_new is not None:
            s = exact_score(h_new)
            if s < best_score - 1e-13:
                h_work = h_new
                best_score = float(s)
                best_h = h_work.copy()
                score_p = s

    # Phase 7: Pair polish
    if time.time() < deadline - 15:
        h_ps2, s_ps2 = pair_search(h_work, n_p, deadline - 10, max_iters=3000, tol_frac=0.05)
        if s_ps2 < best_score - 1e-13:
            try_update(h_ps2, s_ps2, n_p)
            h_work = h_ps2

    # Phase 8: Multi-start with different initializations
    if time.time() < deadline - 80:
        starts = []
        # Shifted binary
        for shift in [0, 1, 2, 3, 5, 8]:
            if time.time() > deadline - 30:
                break
            h_bin = make_binary_balanced(n_p, np.random.default_rng(seed + shift * 100))
            starts.append(h_bin)
        # Sturmian
        for slope_num in [1, 2, 3]:
            slope = slope_num / (2 * slope_num + 1)
            h_st = make_sturmian(n_p, slope, rng.random(), rng)
            starts.append(h_st)
        # Spectral
        h_sp = make_spectral_flat(n_p, np.random.default_rng(seed + 4242))
        starts.append(h_sp)

        for h_start in starts:
            if time.time() > deadline - 40:
                break
            s_st = exact_score(h_start)
            if s_st >= best_score + 0.02:
                continue
            # Quick SLSQP
            h_t = slsqip_solve(h_start, 0.03, n_p, 80, deadline)
            if h_t is not None:
                s_t = exact_score(h_t)
                if s_t < s_st:
                    h_start = h_t
                    s_st = s_t
            # Quick pair search
            if time.time() < deadline - 20:
                h_t, s_t = pair_search(h_start, n_p, deadline - 15, max_iters=2000)
                if s_t < s_st:
                    h_start = h_t
                    s_st = s_t
            if s_st < best_score - 1e-13:
                try_update(h_start, s_st, n_p)
                h_work = h_start

    # Phase 9: Dimension exploration
    dim_options = []
    for d in range(-4, 5):
        cand = n_p + d
        if 50 <= cand <= 120 and cand != n_p:
            dim_options.append(cand)
    # Also try some ratio-based
    for factor in [0.85, 1.15, 1.3]:
        cand = int(round(n_p * factor))
        if 50 <= cand <= 120 and cand not in dim_options:
            dim_options.append(cand)

    for dn in dim_options:
        if time.time() > deadline - 20:
            break
        h_d = interpolate_h(best_h, dn)
        s_d = exact_score(h_d)
        if s_d >= best_score + 0.01:
            continue
        h_d = slsqip_solve(h_d, 0.02, dn, 80, deadline)
        if h_d is not None:
            s_d = exact_score(h_d)
        if time.time() < deadline - 15:
            h_d, s_d = pair_search(h_d, dn, deadline - 10, max_iters=1500)
        if s_d < best_score - 1e-13 or (abs(s_d - best_score) < 1e-7 and dn < best_n):
            try_update(h_d, s_d, dn)
            h_work = h_d

    # Phase 10: Final fine-tuning
    if time.time() < deadline - 5:
        h_final, s_final = pair_search(h_work, n_p, deadline - 3, max_iters=2000, tol_frac=0.03)
        if s_final < best_score - 1e-13:
            try_update(h_final, s_final, n_p)

    # Final validation
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-12:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
