    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        n = 84
        h_init = np.full(n, 0.5)
        h_init = project_sum_box(h_init, n / 2.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n_best = len(best_h)

    def time_ok():
        return time.time() < deadline

    # Phase 1: LP trust region for initial progress
    if time_ok():
        dl = min(deadline, start_time + budget_s * 0.18)
        try:
            h1, c1 = method_lp_trust(best_h, dl, max_iters=300)
            if c1 < best_c:
                best_h = h1
                best_c = c1
        except Exception:
            pass

    # Phase 2: Subgradient with momentum (main refinement)
    if time_ok():
        dl = min(deadline, start_time + budget_s * 0.40)
        try:
            h2, c2 = method_subgrad_momentum(best_h, dl, beta_start=15, beta_end=300,
                                             max_iters=30000, mom=0.82, step0=0.006)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    # Phase 3: Multi-start perturbation
    if time_ok():
        perts = multi_perturb(best_h, n_pert=7)
        for p in perts:
            if time.time() > deadline - 30:
                break
            c_p = evaluate_c5(p)
            if c_p >= best_c:
                continue
            # Optimize from this perturbation
            dl_sub = min(deadline - 20, time.time() + 60)
            try:
                hp, cp = method_subgrad_momentum(p, dl_sub, beta_start=20, beta_end=200,
                                                 max_iters=5000, mom=0.80, step0=0.004)
                if cp < best_c:
                    best_h = hp
                    best_c = cp
            except Exception:
                pass

    # Phase 4: SLSQP refinement with high beta
    if time_ok():
        dl = min(deadline, start_time + budget_s * 0.72)
        try:
            h4, c4 = method_slsqp_refine(best_h, dl, betas=[40, 80, 150, 300])
            if c4 < best_c:
                best_h = h4
                best_c = c4
        except Exception:
            pass

    # Phase 5: Targeted coordinate descent on active lags
    if time_ok():
        dl = min(deadline, start_time + budget_s * 0.85)
        try:
            h5, c5 = method_coord_active(best_h, dl, max_rounds=100)
            if c5 < best_c:

                best_c = c5
        except Exception:

    # Phase 6: Final subgradient polish
    if time_ok():
        try:
            h6, c6 = method_subgrad_momentum(best_h, deadline - 2, beta_start=50, beta_end=500,
                                             max_iters=20000, mom=0.90, step0=0.003, top_m=10)
            if c6 < best_c:
                best_h = h6
                best_c = c6
        except Exception:
            pass

    # Phase 7: Try nearby resolutions
    if time.time() < deadline - 40:
        n_cur = len(best_h)
        for n_try in [n_cur - 4, n_cur + 4, n_cur - 8, n_cur + 8]:
            if time.time() > deadline - 25 or n_try < 40:
                break
            try:
                h_tr = transfer_resolution(best_h, n_try)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c - 0.0001:
                    # Worth optimizing at this resolution
                    dl_sub = min(deadline - 15, time.time() + 50)
                    h_opt, c_opt = method_subgrad_momentum(h_tr, dl_sub, beta_start=20,
                                                           beta_end=200, max_iters=5000)
                    if c_opt < best_c:
                        best_h = h_opt
                        best_c = c_opt
            except Exception:
                pass

    # Final verification and cleanup
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
