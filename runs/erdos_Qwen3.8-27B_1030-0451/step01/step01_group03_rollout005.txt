import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def subgradient_c5(h, top_k=8):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    if top_k is not None and top_k < len(c):
        thresh = np.partition(c, -top_k)[-top_k]
        active = np.where(c >= thresh - 1e-12)[0]
    else:
        active = np.where(c >= m - 1e-10)[0]
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in active:
        d = int(k) - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi+1] += omh[lo-d:hi+1-d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= h[lo2+d:hi2+1+d]
    grad *= dx / max(1, len(active))
    return grad, m

def steepest_descent_exact(h, max_iter=200, deadline=None):
    """Steepest descent with exact line search on C5."""
    n = len(h)
    target = n / 2.0
    best_c5 = evaluate_c5(h)
    h_curr = h.copy()
    
    for it in range(max_iter):
        if deadline is not None and time.time() > deadline - 2:
            break
        grad, m = subgradient_c5(h_curr, top_k=10)
        # Project to sum-zero direction
        grad_proj = grad - grad.mean()
        norm = np.sqrt(np.sum(grad_proj**2))
        if norm < 1e-14:
            break
        direction = -grad_proj / norm
        
        # Find max feasible step
        pos_mask = direction > 1e-15
        neg_mask = direction < -1e-15
        t_max = 1e10
        if pos_mask.any():
            t_max = min(t_max, (1.0 - h_curr[pos_mask]).min() / direction[pos_mask].max())
        if neg_mask.any():
            t_max = min(t_max, h_curr[neg_mask].min() / (-direction[neg_mask].max()))
        t_max = max(t_max, 1e-10)
        
        # Backtracking line search
        t = min(t_max * 0.5, 0.05)
        improved = False
        for ls in range(40):
            h_try = h_curr + t * direction
            if h_try.min() < -1e-12 or h_try.max() > 1.0 + 1e-12:
                t *= 0.5
                continue
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5 - 1e-14:
                h_curr = h_try
                best_c5 = c5_try
                improved = True
                break
            t *= 0.5
        if not improved:
            # Try smaller and smaller steps
            t = t_max * 0.01
            for ls in range(20):
                h_try = h_curr + t * direction
                if h_try.min() < -1e-12 or h_try.max() > 1.0 + 1e-12:
                    t *= 0.5
                    continue
                c5_try = evaluate_c5(h_try)
                if c5_try < best_c5 - 1e-15:
                    h_curr = h_try
                    best_c5 = c5_try
                    improved = True
                    break
                t *= 0.5
            if not improved:
                break
    
    return h_curr, best_c5

def pair_moves_descent(h, n_rounds=50, deadline=None):
    """Coordinate pair moves with exact line search."""
    n = len(h)
    target = n / 2.0
    best_c5 = evaluate_c5(h)
    h_curr = h.copy()
    
    for rnd in range(n_rounds):
        if deadline is not None and time.time() > deadline - 2:
            break
        grad, m = subgradient_c5(h_curr, top_k=6)
        grad_proj = grad - grad.mean()
        
        # Find best pair: decrease h at most positive, increase at most negative
        i_idx = int(np.argmax(grad_proj))
        j_idx = int(np.argmin(grad_proj))
        
        if grad_proj[i_idx] - grad_proj[j_idx] < 1e-12:
            break
        
        # Direction: h[i] -= t, h[j] += t
        t_min = max(h_curr[i_idx] - 1.0, -h_curr[j_idx])
        t_max = min(h_curr[i_idx], 1.0 - h_curr[j_idx])
        if t_min >= t_max - 1e-14:
            break
        
        # Golden section
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a, b = t_min + 1e-10, t_max - 1e-10
        c_pt = b - gr * (b - a)
        d_pt = a + gr * (b - a)
        
        h_curr[i_idx] -= c_pt
        h_curr[j_idx] += c_pt
        fc = evaluate_c5(h_curr)
        h_curr[i_idx] += d_pt - c_pt
        h_curr[j_idx] -= d_pt - c_pt
        fd = evaluate_c5(h_curr)
        
        for _ in range(50):
            if fc < fd:
                b = d_pt; d_pt = c_pt; fd = fc
                c_pt = b - gr * (b - a)
                h_curr[i_idx] += c_pt - d_pt
                h_curr[j_idx] -= c_pt - d_pt
                fc = evaluate_c5(h_curr)
            else:
                a = c_pt; c_pt = d_pt; fc = fd
                d_pt = a + gr * (b - a)
                h_curr[i_idx] += d_pt - c_pt
                h_curr[j_idx] -= d_pt - c_pt
                fd = evaluate_c5(h_curr)
        
        c5_now = evaluate_c5(h_curr)
        if c5_now >= best_c5:
            # Revert to best
            break
        best_c5 = c5_now
    
    return h_curr, best_c5

def smooth_slsqp(h0, n, deadline):
    """Smooth SLSQP with log-sum-exp surrogate."""
    global _G_T, _G_N, _G_TAU
    _G_T = n / 2.0
    _G_N = n
    target = n / 2.0
    best_c5 = evaluate_c5(h0)
    best_h = h0.copy()
    
    dx = 2.0 / n
    c_vals = np.correlate(h0, 1.0 - h0, 'full') * dx
    spread = max(1e-6, c_vals.max() - c_vals.mean())
    
    taus = [max(0.05, 0.4 * spread) * (0.5 ** i) for i in range(18)]
    tau_sched = [t for t in taus if t > 3e-4]
    
    h_curr = h0.copy()
    
    for tau in tau_sched:
        if time.time() > deadline - 5:
            break
        _G_TAU = tau
        maxiter = 30 if tau > 0.01 else 60
        try:
            res = minimize(_slsqp_obj, h_curr, method='SLSQP', jac=_slsqp_grad,
                           bounds=[(0.0, 1.0)] * n,
                           constraints=[{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}],
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        c5e = evaluate_c5(h_curr)
        if c5e < best_c5:
            best_c5 = c5e
            best_h = h_curr.copy()
    
    return best_h, best_c5

_G_T = 0.0
_G_N = 0
_G_TAU = 1.0

def _slsqp_eq(h):
    return float(np.sum(h) - _G_T)

def _slsqp_eq_jac(h):
    return np.ones(_G_N)

def _slsqp_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _slsqp_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    active = np.where(alpha > 1e-13)[0]
    for k in active:
        a = alpha[k]
        d = int(k) - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi+1] += a * omh[lo-d:hi+1-d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= a * h[lo2+d:hi2+1+d]
    return grad * dx

def perturb_search(h0, n, deadline, rng, n_starts=8):
    """Try multiple small perturbations, optimize each, keep best."""
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    target = n / 2.0
    
    amps = [0.001, 0.002, 0.004, 0.007, 0.012, 0.02, 0.035, 0.05]
    for amp in amps:

        h_opt, c5_opt = steepest_descent_exact(h_try, max_iter=100, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            h0 = h_opt  # continue from improved point
    
    return best_h, best_c5

def multi_topk_descent(h0, deadline):
    """Steepest descent with varying top_k for the active set."""
    n = len(h0)
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    for top_k in [3, 5, 8, 12, 20, 40]:

        if norm < 1e-14:

            else:

                if t < 1e-8:

def cyclic_transforms(h):
    """Generate cyclic shifts and reversals of h."""
    n = len(h)
    results = []
    for shift in range(-3, 4):
        results.append(np.roll(h, shift))
    results.append(h[::-1].copy())
    results.append(np.roll(h[::-1].copy(), n // 2))
    return results
