                best_h = ch

    # Phase 4: Symmetrization + re-polish
    if time.monotonic() < deadline - 2.0:
        sh, sc = try_symmetrize(best_h)
        if sc < best_c5 - 1e-12:
            best_c5 = sc
            best_h = sh
        # Quick LSE polish after symmetrization
        alloc = min(0.04 * effective, deadline - time.monotonic())
        if alloc > 0.5:
            ch, cc = lse_polish(best_h, 8.0, alloc, len(best_h) / 2.0, deadline)
            if cc < best_c5 - 1e-12:
                best_c5 = cc
                best_h = ch

    # Phase 5: Very fine final polish
    if time.monotonic() < deadline - 0.5:
        alloc = min(0.03 * effective, deadline - time.monotonic())
        if alloc > 0.3:
            ch, cc = lse_polish(best_h, 5.0, alloc, len(best_h) / 2.0, deadline)
            if cc < best_c5 - 1e-12:
                best_c5 = cc
                best_h = ch

    # Final feasibility check
    best_h = project_exact(best_h, len(best_h) / 2.0)
    best_c5 = objective(best_h)
    return best_h, float(best_c5), int(len(best_h))
