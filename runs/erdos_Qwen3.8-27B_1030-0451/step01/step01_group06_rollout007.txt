```python
def softmax_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    exp_tau_corr = np.exp(tau * (corr - np.max(corr)))  # for numerical stability
    weights = exp_tau_corr / np.sum(exp_tau_corr)
    # Gradient
    grad = np.zeros(n)
    for k in range(-(n-1), n):
        if weights[k + n - 1] < 1e-10:
            continue
        w = weights[k + n - 1]
        if k == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif k > 0:
            g = np.zeros(n)
            g[k:] += dx * (1.0 - h[:n-k])
            g[:n-k] += -dx * h[k:]
            grad += w * g
        else:
            t = -k
            g = np.zeros(n)
            g[:n-t] += dx * (1.0 - h[t:])
            g[t:] += -dx * h[:n-t]
            grad += w * g
    return grad, np.max(corr)
```

```python
def solve_bcd_lp(h_fixed, other_h, n):
    """Solve the LP: min t s.t. corr_k(h_fixed * (1-other_h)) <= t for all k
       subject to 0 <= h_fixed <= 1, sum(h_fixed) = n/2"""
    dx = 2.0 / n
    m = 2 * n - 1
    nvars = n + 1  # h[0..n-1] and t
    
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = np.zeros((m, nvars))
    b_ub = np.zeros(m)
    
    for k in range(-(n-1), n):
        row = k + n - 1
        if k >= 0:
            # sum_{i=0}^{n-1-k} h[i] * (1-other_h[i+k]) * dx <= t
            coeffs = (1.0 - other_h[k:]) * dx
            A_ub[row, :len(coeffs)] += coeffs
        else:
            t_shift = -k
            # sum_{i=t_shift}^{n-1} h[i] * (1-other_h[i-t_shift]) * dx <= t
            coeffs = (1.0 - other_h[:n-t_shift]) * dx
            A_ub[row, t_shift:] += coeffs
        A_ub[row, n] = -1.0  # -t
        b_ub[row] = 0.0
    
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None
```

```python
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def solve_bcd_lp(h_fixed, h_other, n):
    """Solve: min t s.t. for all k: sum_i h[i]*(1-h_other[i+k])*dx <= t
       subject to 0<=h<=1, sum(h)=n/2
       h_fixed is the current iterate (for potential warm start or reference)"""
    dx = 2.0 / n
    m = 2 * n - 1
    nvars = n + 1
    
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((m, nvars))
    b_ub = np.zeros(m)
    
    for k in range(-(n-1), n):
        row = k + n - 1
        A_ub[row, n] = -1.0
        if k >= 0:
            # Sum over i=0 to n-1-k: h[i] * (1-h_other[i+k]) * dx
            length = n - k
            A_ub[row, :length] += (1.0 - h_other[k:k+length]) * dx
        else:
            t_shift = -k
            # Sum over i=t_shift to n-1: h[i] * (1-h_other[i-t_shift]) * dx
            length = n - t_shift
            A_ub[row, t_shift:t_shift+length] += (1.0 - h_other[:length]) * dx
    
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return res.x[:n]
    return None

    n = len(h)
    best_h = h.copy()
    best_val = evaluate_exact(h)
    h_curr = h.copy()
    
    # Adaptive damping
    if damping is None:
        damping = [1.0, 1.0, 1.0, 0.8, 0.7, 0.5, 0.4, 0.3, 0.2, 0.15]
    
    stagnant = 0
    max_stag = 50
    
    for it in range(max_iters):
        if budget_remaining() < 5.0:
            break
        
        # Solve LP for new h given current h
        h_new = solve_bcd_lp(h_curr, h_curr, n)
        if h_new is None:
            break
        
        # Apply damping
        alpha = damping[min(it, len(damping)-1)]
        h_candidate = (1 - alpha) * h_curr + alpha * h_new
        
        # Evaluate
        val = evaluate_exact(h_candidate)
        if val < best_val - 1e-14:
            best_val = val
            best_h = h_candidate.copy()
            h_curr = h_candidate.copy()
            stagnant = 0
        else:
            # Try full step
            val_full = evaluate_exact(h_new)
            if val_full < best_val - 1e-14:
                best_val = val_full
                best_h = h_new.copy()
                h_curr = h_new.copy()
                stagnant = 0
            else:
                stagnant += 1
                if stagnant > max_stag:
                    # Perturb and restart
                    h_curr = project_feasible(best_h + 0.05 * rng.standard_normal(n), n/2.0)
                    stagnant = 0
    
    return best_h, best_val
```

```python
def solve_lp_trust(h, corr, trust):
    """Standard LP trust region step."""
    n = len(h)
    m = 2 * n - 1
    nvars = n + 1

        h0 = np.full(120, 0.5)
    
    best_h = h0.copy()
    best_val = evaluate_exact(best_h)
    
    # Phase 1: BCD at native resolution (main workhorse)
    if remaining() > 30:
        budget1 = remaining() * 0.5
        orig_deadline = deadline
        deadline = t_start + budget1
        
        def rem1():
            return deadline - time.time()
        
        h, v = bcd_optimize(best_h, rem1, rng, max_iters=50000)
        deadline = orig_deadline
        if v < best_val:
            best_val = v
            best_h = h.copy()
    
    # Phase 2: LP trust region refine
    if remaining() > 20:
        h, v = refine_lp(best_h, best_val, remaining)
        if v < best_val:
            best_val = v
            best_h = h.copy()
    
    # Phase 3: Try flip symmetry
    h_flip = project_feasible(1.0 - best_h, len(best_h) / 2.0)
    v_flip = evaluate_exact(h_flip)
    if v_flip < best_val:
        best_val = v_flip
        best_h = h_flip.copy()
    
    # Phase 4: BCD again on potentially improved solution
    if remaining() > 30:
        budget4 = remaining() * 0.3
        orig_deadline = deadline
        deadline = t_start + budget4
        
        def rem4():
            return deadline - time.time()
        
        h, v = bcd_optimize(best_h, rem4, rng, max_iters=30000)
        deadline = orig_deadline
        if v < best_val:
            best_val = v
            best_h = h.copy()
    
    # Phase 5: Upscale to 2x
    if remaining() > 80:
        n_orig = len(best_h)
        factor = 2
        h_up = np.repeat(best_h, factor)
        h_up = project_feasible(h_up, len(h_up) / 2.0)
        
        budget5 = remaining() * 0.3
        orig_deadline = deadline
        deadline = t_start + budget5
        
        def rem5():
            return deadline - time.time()
        
        h, v = bcd_optimize(h_up, rem5, rng, max_iters=20000)
        deadline = orig_deadline
        
        v_check = evaluate_exact(h)
        if v_check < best_val:
            best_val = v_check
            best_h = h.copy()
        
        # Downscale back
        h_dn = h[:len(h)//factor*factor].reshape(-1, factor).mean(axis=1)
        h_dn = project_feasible(h_dn, len(h_dn) / 2.0)
        v_dn = evaluate_exact(h_dn)
        if v_dn < best_val:
            best_val = v_dn
            best_h = h_dn.copy()
        
        # Also try unique upscaled
        v_up = evaluate_exact(h)
        if v_up < best_val:
            best_val = v_up
            best_h = h.copy()
    
    # Phase 6: Random restarts with BCD
    if remaining() > 40:
        n = len(best_h)
        n_restarts = min(6, int(remaining() // 15))
        for restart in range(n_restarts):
            if remaining() < 15:
                break
            if restart == 0:
                h_try = project_feasible(best_h + 0.08 * rng.standard_normal(n), n/2.0)
            elif restart == 1:
                h_try = project_feasible(1.0 - best_h + 0.04 * rng.standard_normal(n), n/2.0)
            elif restart == 2:
                # Smooth perturbation
                from scipy.ndimage import uniform_filter1d
                sm = uniform_filter1d(best_h, size=7)
                h_try = project_feasible(sm + 0.06 * rng.standard_normal(n), n/2.0)
            elif restart == 3:
                # Binary-like
                binary = (best_h > 0.5).astype(float)
                binary = project_feasible(binary, n/2.0)
                h_try = project_feasible(0.6 * best_h + 0.4 * binary + 0.03*rng.standard_normal(n), n/2.0)
            else:
                h_try = project_feasible(best_h + 0.12 * rng.standard_normal(n), n/2.0)
            
            budget_r = min(remaining() * 0.15, 60.0)
            orig_deadline = deadline
            deadline = t_start + budget_r
            
            def rem_r():
                return deadline - time.time()
            
            h, v = bcd_optimize(h_try, rem_r, rng, max_iters=10000)
            deadline = orig_deadline
            if v < best_val:
                best_val = v
                best_h = h.copy()
    
    # Phase 7: Final LP refinement
    if remaining() > 15:
        h, v = refine_lp(best_h, best_val, remaining)
        if v < best_val:
            best_val = v
            best_h = h.copy()
    
    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)
    
    return best_h, float(best_val), len(best_h)
```
