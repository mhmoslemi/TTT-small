import numpy as np
from scipy.optimize import minimize
import time
import math


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(z, n):
    S = n / 2.0
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def smoothed_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = float(np.max(c))
    exps = np.exp((c - c_max) / tau)
    sw = exps.sum()
    if sw < 1e-300:
        return np.zeros(n)
    weights = exps / sw
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K]
        if w < 1e-10:
            continue
        lag = K - (n - 1)
        if lag == 0:
            g += w * dx * (1.0 - 2.0 * h)
        elif lag > 0:
            L = lag
            if n > L:
                grad_L = np.zeros(n)
                grad_L[:n - L] = (1.0 - h[L:n]) - h[:n - L]
                grad_L[n - L:] = -h[:L]
                g += w * dx * grad_L
        else:
            m = -lag
            if n > m:
                grad_L = np.zeros(n)
                grad_L[:n - m] -= h[m:]
                grad_L[m:] += 1.0 - h[:n - m]
                g += w * dx * grad_L
    return g


def fw_vertex(g, n):
    S = n / 2.0
    order = np.argsort(g)
    u = np.zeros(n)
    S_int = int(np.floor(S))
    S_frac = S - S_int
    for i in range(min(S_int, n)):
        u[order[i]] = 1.0
    if S_frac > 1e-12 and S_int < n:
        u[order[S_int]] = S_frac
    return u


def line_search_pair(h, a, b, n):
    cur = h.copy()
    cur_score = exact_score(cur)
    d_max_pos = min(1.0 - cur[a], cur[b])
    d_max_neg = min(cur[a], 1.0 - cur[b])
    best_d = 0.0
    best_s = cur_score
    if d_max_pos > 1e-12:
        for frac in np.linspace(0.05, 1.0, 25):
            d = d_max_pos * frac
            cur[a] += d
            cur[b] -= d
            s = exact_score(cur)
            if s < best_s - 1e-15:
                best_s = s
                best_d = d
            cur[a] -= d
            cur[b] += d
    if d_max_neg > 1e-12:
        for frac in np.linspace(0.05, 1.0, 25):
            d = -d_max_neg * frac
            cur[a] += d
            cur[b] -= d
            s = exact_score(cur)
            if s < best_s - 1e-15:
                best_s = s
                best_d = d
            cur[a] -= d
            cur[b] += d
    if best_d != 0.0:
        cur[a] += best_d
        cur[b] -= best_d
    return cur, best_s, best_d


def line_search_fw(h, u, n):
    cur = h.copy()
    cur_score = exact_score(cur)
    best_gamma = 0.0
    best_s = cur_score
    for gamma in np.linspace(0.05, 1.0, 20):
        h_new = (1.0 - gamma) * cur + gamma * u
        h_new = project_feasible(h_new, n)

            best_s = s
            best_gamma = gamma
            best_h = h_new.copy()
    if best_gamma > 0:

    return cur, cur_score, 0.0

def freq_to_h(coeffs, n, n_freq, x):
    h = np.full(n, 0.5)
    for k in range(1, n_freq + 1):
        h += coeffs[2 * (k - 1)] * np.cos(2.0 * np.pi * k * x)
        h += coeffs[2 * (k - 1) + 1] * np.sin(2.0 * np.pi * k * x)
    return project_feasible(h, n)


def global_freq_search(n, n_freq, pop_size, n_gen, rng, deadline, best_score_so_far):
    x = (np.arange(n) + 0.5) / n
    scale = 1.0 / np.sqrt(np.arange(1, n_freq + 1, dtype=float))
    max_coeff = 1.5
    pop = np.zeros((pop_size, 2 * n_freq))
    for i in range(pop_size):
        pop[i] = rng.uniform(-max_coeff, max_coeff, 2 * n_freq) * scale
    scores = np.zeros(pop_size)
    for i in range(pop_size):
        h = freq_to_h(pop[i], n, n_freq, x)
        scores[i] = exact_score(h)

    best_i = int(np.argmin(scores))
    best_h = freq_to_h(pop[best_i], n, n_freq, x)
    best_s = scores[best_i]
    best_coeffs = pop[best_i].copy()

    for gen in range(n_gen):
        if time.time() > deadline:
            break
        for i in range(pop_size):
            if time.time() > deadline:
                break
            a = int(rng.integers(0, pop_size))
            b = int(rng.integers(0, pop_size))
            c = int(rng.integers(0, pop_size))
            while a == b or a == c or b == c:
                a = int(rng.integers(0, pop_size))
                b = int(rng.integers(0, pop_size))
                c = int(rng.integers(0, pop_size))
            F = 0.5 + 0.5 * rng.random()
            CR = 0.7
            trial = pop[a].copy()
            for j in range(2 * n_freq):
                if rng.random() < CR:
                    trial[j] = pop[a][j] + F * (pop[b][j] - pop[c][j])
                trial[j] = np.clip(trial[j], -3.0 * scale[j % n_freq], 3.0 * scale[j % n_freq])
            h_trial = freq_to_h(trial, n, n_freq, x)
            s_trial = exact_score(h_trial)
            if s_trial <= scores[i]:
                pop[i] = trial
                scores[i] = s_trial
                if s_trial < best_s:
                    best_s = s_trial
                    best_h = h_trial.copy()
                    best_coeffs = trial.copy()

    return best_h, best_s

        if s_new < best_score:
            best = h_new
            best_score = s_new

    return best, best_score


def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15.0

    # Get initial construction
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None:
        n_p = len(parent)
        if not (n_p >= 2 and np.all(np.isfinite(parent))):
            parent = None

    if parent is not None:
        parent = project_feasible(parent, n_p)
    else:
        n_p = 81
        parent = project_feasible(np.full(n_p, 0.5), n_p)

    n = n_p
    best_h = parent.copy()
    best_score = exact_score(parent)

    def check_time():
        return time.time() < deadline

    # Phase 1: Intensive gradient local search from parent
    if check_time():
        tau_init = 0.03
        max_iters = 3000
        h1, s1 = gradient_local_search(parent, n, tau_init, max_iters, deadline, rng)
        if s1 < best_score - 1e-14:
            best_h = h1
            best_score = s1

    # Phase 2: Second local search with smaller tau
    if check_time():
        tau2 = 0.005
        max_iters2 = 5000
        h2, s2 = gradient_local_search(best_h, n, tau2, max_iters2, deadline, rng)
        if s2 < best_score - 1e-14:
            best_h = h2
            best_score = s2

    # Phase 3: Frequency basis global search
    if check_time():
        for n_freq in [4, 6, 8]:
            if time.time() > deadline - 30:
                break
            remaining = deadline - time.time()
            n_gen = min(200, int(remaining / (0.02 * (4 + n_freq))))
            n_gen = max(n_gen, 30)
            h_g, s_g = global_freq_search(n, n_freq, 25, n_gen, rng, deadline, best_score)
            if s_g < best_score:
                # Refine with local search
                h_g, s_g = gradient_local_search(h_g, n, 0.01, 1000, deadline, rng)
            if s_g < best_score - 1e-14:
                best_h = h_g
                best_score = s_g

    # Phase 4: Random perturbation search
    if check_time():
        h_r, s_r = random_perturb_search(best_h, n, 500, deadline, rng, best_score)
        if s_r < best_score:
            h_r, s_r = gradient_local_search(h_r, n, 0.01, 2000, deadline, rng)
        if s_r < best_score - 1e-14:
            best_h = h_r
            best_score = s_r

    # Phase 5: Try different dimensions
    if check_time():
        dim_options = [n]
        pn = n
        for factor in [0.9, 0.95, 1.05, 1.1, 1.2, 1.3, 0.8, 1.5]:
            cand_n = int(round(pn * factor))
            if 40 <= cand_n <= 300 and cand_n not in dim_options:
                dim_options.append(cand_n)

        for dn in dim_options:
            if time.time() > deadline - 20:
                break
            if dn == n:
                continue
            h_d = interpolate_h(best_h, dn)
            s_d = exact_score(h_d)
            if s_d < best_score + 0.01:
                h_d, s_d = gradient_local_search(h_d, dn, 0.02, 1500, deadline, rng)
                if s_d < best_score - 1e-14 or (abs(s_d - best_score) < 1e-7 and dn < n):
                    best_h = h_d
                    best_score = s_d
                    n = dn

    # Phase 6: Final intensive polish
    if check_time():
        remaining = deadline - time.time()
        if remaining > 30:
            tau_final = 0.002
            max_final = min(10000, int(remaining * 20))
            h_f, s_f = gradient_local_search(best_h, n, tau_final, max_final, deadline, rng)
            if s_f < best_score - 1e-14:
                best_h = h_f
                best_score = s_f

    # Phase 7: Ultra-fine pair search
    if check_time():
        cur = best_h.copy()
        cur_score = best_score
        g = smoothed_gradient(cur, 0.001)
        for _ in range(200):
            if time.time() > deadline - 5:
                break
            sorted_idx = np.argsort(g)
            neg_idx = sorted_idx[:4]
            pos_idx = sorted_idx[-4:]
            improved = False
            for a in neg_idx:
                if improved:
                    break
                for b in pos_idx:
                    if a == b:
                        continue
                    h_new, s_new, d = line_search_pair(cur, int(a), int(b), n)
                    if d != 0.0 and s_new < cur_score - 1e-15:
                        cur = h_new
                        cur_score = s_new
                        improved = True
                        g = smoothed_gradient(cur, 0.001)
                        break
            if not improved:
                break
        if cur_score < best_score - 1e-14:
            best_h = cur
            best_score = cur_score

    # Final validation and correction
    best_h = project_feasible(best_h, n)
    final_score = exact_score(best_h)
    if final_score < best_score - 1e-12:
        best_score = final_score
    else:
        best_score = final_score

    # Ensure valid
    if not np.all(np.isfinite(best_h)) or len(best_h) < 2:
        best_h = parent.copy()
        best_score = exact_score(parent)
        n = len(parent)

    return best_h, float(best_score), int(n)
