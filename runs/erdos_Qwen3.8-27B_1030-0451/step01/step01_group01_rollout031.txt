import numpy as np
import time
from scipy.optimize import linprog, minimize

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
        r2 = target - y.sum()
        if abs(r2) > 1e-10:
            idx = np.where(y < 1.0 if r2 > 0 else y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif lag > 0:
        j1 = np.arange(0, n - lag)
        g[j1] += (1.0 - h[j1 + lag]) * dx
        j2 = np.arange(lag, n)
        g[j2] -= h[j2 - lag] * dx
    else:
        l = -lag
        j1 = np.arange(0, n - l)
        g[j1] -= h[j1 + l] * dx
        j2 = np.arange(l, n)
        g[j2] += (1.0 - h[j2 - l]) * dx
    return g

def weighted_grad(h, top_idx, weights):
    n = len(h)
    g = np.zeros(n)
    for i, k in enumerate(top_idx):
        lag = k - (n - 1)
        g += weights[i] * grad_lag(h, lag)
    return g

            i = rng.randint(n)

            if c4a < best_c:
                best_h, best_c = h4a, c4a
        except Exception:
            pass
        if tl() > 10:
            try:
                h4b, c4b = method_subgrad(best_h, deadline - 10, M=20, init_step=0.02,
                                          decay=0.94, max_step=0.1, max_iters=80000)
                if c4b < best_c:
                    best_h, best_c = h4b, c4b
            except Exception:
                pass

    # Phase 5: LP trust region again from improved point
    if tl() > 30:
        dl5 = min(deadline, start_time + budget_s * 0.88)
        try:
            h5, c5 = method_lp_trust(best_h, dl5, max_iters=300)
            if c5 < best_c:
                best_h, best_c = h5, c5
        except Exception:
            pass

    # Phase 6: Pair swap local search
    if tl() > 20:
        try:
            h6, c6 = method_pair_swap(best_h, min(deadline, time.time() + min(30, tl())))
            if c6 < best_c:
                best_h, best_c = h6, c6
        except Exception:
            pass

    # Phase 7: Final subgradient polish
    if tl() > 5:
        try:
            h7, c7 = method_subgrad(best_h, deadline - 1, M=10, init_step=0.005,
                                    decay=0.92, max_step=0.05, min_step=1e-9,
                                    max_iters=200000)
            if c7 < best_c:
                best_h, best_c = h7, c7
        except Exception:
            pass

    # Phase 8: Try resolution increase if time allows
    if tl() > 80:
        n_cur = len(best_h)
        if n_cur < 200:
            n_new = min(int(n_cur * 1.5), 250)
            try:
                h_tr = transfer_resolution(best_h, n_new)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c:
                    best_h = h_tr
                    best_c = c_tr
                sub_dl = min(deadline, time.time() + min(60, tl() * 0.5))
                h_a, c_a = method_subgrad(best_h, sub_dl, M=10, init_step=0.02,
                                          decay=0.95, max_iters=50000)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a
            except Exception:
                pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
