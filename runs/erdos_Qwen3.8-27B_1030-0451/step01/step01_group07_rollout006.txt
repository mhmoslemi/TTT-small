def project_exact(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0

        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)

        else:

            if tot > 0:

def soft_weighted_grad(h, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    mx = corr.max()
    w = np.exp(beta * (corr - mx))
    w /= w.sum()
    grad = np.zeros(n)

            grad[k:] += wm * (1.0 - h[:n - k])
            grad[:n - k] -= wm * h[k:]
        else:
            p = -k
            grad[:n - p] += wm * (1.0 - h[p:])

def resample_h(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

            # Try moving mass from idx to a random other coordinate
            # and from a random coordinate to idx
            for direction in range(2):
                if direction == 0 and h[idx] <= 1e-8:
                    continue
                if direction == 1 and (1.0 - h[idx]) <= 1e-8:
                    continue
                # Pick partner
                j = int(rng.integers(n))
                if j == idx:
                    continue
                if direction == 0:
                    # h[idx] -= t, h[j] += t
                    t_max = min(h[idx], 1.0 - h[j])
                    if t_max < 1e-10:
                        continue
                    t_opt, v_opt = pair_1d_opt(h, idx, j, t_max, n_grid=15)
                    if v_opt < cur - 1e-10:
                        h[idx] -= t_opt
                        h[j] += t_opt
                        h = np.clip(h, 0.0, 1.0)
                        h = repair(h, target)
                        cur = objective(h)
                        improved = True
                else:
                    # h[j] -= t, h[idx] += t
                    t_max = min(h[j], 1.0 - h[idx])
                    if t_max < 1e-10:
                        continue
                    t_opt, v_opt = pair_1d_opt(h, j, idx, t_max, n_grid=15)
                    if v_opt < cur - 1e-10:
                        h[j] -= t_opt
                        h[idx] += t_opt
                        h = np.clip(h, 0.0, 1.0)
                        h = repair(h, target)
                        cur = objective(h)
                        improved = True
            if improved and cur < best_c - 1e-12:
                best_c = cur
                best_h = h.copy()
        sweep += 1
        if not improved and sweep > 10:
            # Stuck, add perturbation
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            if a != b:
                t = rng.random() * 0.05 * min(h[a], 1.0 - h[b])
                if t > 1e-9:
                    h2 = h.copy()
                    h2[a] -= t
                    h2[b] += t
                    h2 = repair(h2, target)
                    if objective(h2) < cur + 0.003:
                        h = h2
                        cur = objective(h)
    c_final = objective(h)
    if c_final < best_c:
        best_c = c_final
        best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    deadline = t0 + max(10.0, float(budget_s) - 5.0)

    parent = get_parent()
    if parent is None:
        n_init = 131
        h_init = (rng.random(n_init) >= 0.5).astype(float)
        parent = repair(h_init, n_init / 2.0)

    n0 = len(parent)
    h_start = repair(parent, n0 / 2.0)
    best_h = h_start.copy()
    best_c5 = objective(best_h)

    def consider(cand):
        nonlocal best_h, best_c5
        try:
            cand = repair(np.asarray(cand, dtype=float), len(cand) / 2.0)
            c = objective(cand)
            if np.isfinite(c) and c < best_c5 - 1e-10:
                best_c5 = c
                best_h = cand.copy()
        except Exception:
            pass

    # Phase 1: Pair polish on parent (most of the time)
    alloc1 = 0.45 * (deadline - t0)
    d1 = t0 + alloc1
    if time.monotonic() < d1:
        ch, cc = pair_polish(h_start, rng, d1, K=60, stagnation_limit=6)
        consider(ch)

    # Phase 2: Transform and polish
    if time.monotonic() < deadline - 5.0:
        transforms = make_transforms(best_h)
        per = 0.04 * (deadline - t0)
        for tf in transforms:
            if time.monotonic() > deadline - 5.0:
                break
            alloc = min(per, deadline - time.monotonic() - 5.0)
            if alloc < 2.0:
                break
            cand = repair(resample_h(tf, n0), n0 / 2.0)
            ch, cc = pair_polish(cand, rng, time.monotonic() + alloc, K=40, stagnation_limit=5)
            consider(ch)

    # Phase 3: Coordinate descent on best
    if time.monotonic() < deadline - 5.0:
        alloc = min(0.15 * (deadline - t0), deadline - time.monotonic() - 5.0)
        if alloc > 3.0:
            ch, cc = coordinate_descent(best_h, rng, time.monotonic() + alloc)
            consider(ch)

    # Phase 4: Try different resolutions
    if time.monotonic() < deadline - 3.0:
        for dn in (-3, 3, -5, 5, -2, 2):
            nn = n0 + dn
            if nn < 80 or nn > 200 or time.monotonic() > deadline - 3.0:
                continue
            alloc = min(0.03 * (deadline - t0), deadline - time.monotonic() - 3.0)
            if alloc < 2.0:
                break
            rs = repair(resample_h(best_h, nn), nn / 2.0)
            ch, cc = pair_polish(rs, rng, time.monotonic() + alloc, K=30, stagnation_limit=4)
            consider(ch)

    # Phase 5: Final refinement - intensive pair polish with smaller gap
    if time.monotonic() < deadline - 1.0:
        alloc = min(0.12 * (deadline - t0), deadline - time.monotonic() - 1.0)
        if alloc > 2.0:
            # Restart from best with tighter parameters
            h_refine = best_h.copy()
            n_ref = len(h_refine)
            target_ref = n_ref / 2.0
            cur = objective(h_refine)
            best_local = cur
            best_local_h = h_refine.copy()
            gap = 0.002

            order = list(range(n_ref))
            while time.monotonic() < deadline - 1.0:
                rng.shuffle(order)
                grad = subgradient(h_refine, gap)
                g_soft = soft_weighted_grad(h_refine, 80.0)
                g = grad + g_soft

                # Try top pairs from gradient
                cands = []
                for a in range(n_ref):
                    if h_refine[a] <= 1e-9:
                        continue
                    for b in range(n_ref):
                        if a == b or (1.0 - h_refine[b]) <= 1e-9:
                            continue
                        sc = (g[a] - g[b]) * min(h_refine[a], 1.0 - h_refine[b])
                        if sc > 0.005:
                            cands.append((sc, a, b))
                cands.sort(key=lambda x: -x[0])
                top = cands[:50]

                any_improved = False
                for (_, a, b) in top:
                    if time.monotonic() > deadline - 1.0:
                        break
                    t_max = min(h_refine[a], 1.0 - h_refine[b])
                    if t_max < 1e-10:
                        continue
                    t_opt, v_opt = pair_1d_opt(h_refine, a, b, t_max, n_grid=30)
                    if v_opt < cur - 1e-11:
                        h_refine[a] -= t_opt
                        h_refine[b] += t_opt
                        h_refine = np.clip(h_refine, 0.0, 1.0)
                        h_refine = repair(h_refine, target_ref)
                        cur = objective(h_refine)
                        any_improved = True
                        if cur < best_local - 1e-11:
                            best_local = cur
                            best_local_h = h_refine.copy()

                if not any_improved:
                    gap = min(0.02, gap * 1.2)
                    # Random kick
                    a = int(rng.integers(n_ref))
                    b = int(rng.integers(n_ref))
                    if a != b:
                        t = rng.random() * 0.03 * min(h_refine[a], 1.0 - h_refine[b])
                        if t > 1e-10:
                            h2 = h_refine.copy()
                            h2[a] -= t
                            h2[b] += t
                            h2 = repair(h2, target_ref)
                            if objective(h2) < cur + 0.001:
                                h_refine = h2
                                cur = objective(h_refine)

            consider(best_local_h)

    # Final: ensure feasibility and return
    best_h = repair(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_c5 = objective(best_h)
    n_pts = len(best_h)
    if not np.isfinite(best_c5) or best_c5 > 0.5:
        best_h = repair(h_start, n0 / 2.0)
        best_c5 = objective(best_h)
        n_pts = n0
    return best_h, float(best_c5), int(n_pts)
