            h_up = aug_lagrangian_minimize(h_up, 0.005, ts, maxiter=200)
            h_up = aug_lagrangian_minimize(h_up, 0.001, ts, maxiter=200)
            c5_up = compute_c5(h_up)
            if c5_up < best_c5:
                best_c5 = c5_up
                best_h = h_up.copy()
                best_N = N_up

    best_h = project_onto_feasible(best_h, best_N / 2.0)
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, best_N
