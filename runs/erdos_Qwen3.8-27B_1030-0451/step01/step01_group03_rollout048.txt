import numpy as np
import time
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))

def grad_f_d(h, d):
    """Gradient of f_d(h) where f_d = dx * sum_{i: valid} h[i]*(1-h[i+d])"""
    n = len(h)
    dx = 2.0 / n
    h_pad = np.zeros(2*n)
    h_pad[n:2*n] = h  # h_pad[i] = h[i-n] for i in [n, 2n)
    
    # Actually, let me just compute directly
    g = np.zeros(n)
    
    # For each k, g[k] = dx * [(1-h[k+d]) * 1_{0<=k+d<n} - h[k-d] * 1_{0<=k-d<n}]
    
    # Term 1: (1-h[k+d]) for k where 0<=k+d<n
    # i.e., k from max(0,-d) to min(n-1, n-1-d)
    lo1 = max(0, -d)
    hi1 = min(n-1, n-1-d)
    if lo1 <= hi1:
        g[lo1:hi1+1] += (1.0 - h[lo1+d:hi1+1+d])
    
    # Term 2: -h[k-d] for k where 0<=k-d<n
    # i.e., k from max(0,d) to min(n-1, n-1+d)
    lo2 = max(0, d)
    hi2 = min(n-1, n-1+d)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2-d:hi2+1-d]
    
    return g * dx

def project_feasible(y, target):
    """Project y onto {h: 0<=h<=1, sum(h)=target}"""
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo+hi)*0.5, 0.0, 1.0)
    # Fix sum
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    h = np.interp(xn, xo, h_old)
    return project_feasible(h, n_new / 2.0)

def solve_lp(h_start, cuts, target, n):
    """Solve the LP: min t s.t. g_k·h + c_k <= t, sum h = target, 0<=h<=1"""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    
    for g, c in cuts:
        constraints.append(g @ h_var + c <= t_var)
    
    objective = cp.Minimize(t_var)
    prob = cp.Problem(objective, constraints)
    
    try:
        # Try GLOP first, then PDLP, then ECOS
        for solver in [cp.GLOP, cp.PDLP, cp.ECOS]:
            try:
                prob.solve(solver=solver)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    return h_var.value, t_var.value
            except:
                continue
        # Fallback to CLP or default
        prob.solve()
        if prob.status in ['optimal', 'optimal_inaccurate']:
            return h_var.value, t_var.value
    except:
        pass
    
    return h_start, None  # Fallback

def cutting_plane_optimize(h0, n, deadline, max_iter=200):
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_C = evaluate_c5(h)
    cuts = []
    
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        
        # Solve LP
        if cuts:
            h_lp, t_lp = solve_lp(h, cuts, target, n)
        else:
            h_lp = h.copy()
            t_lp = best_C
        
        if h_lp is None:
            break
        
        h_lp = np.clip(h_lp, 0.0, 1.0)
        h_lp = project_feasible(h_lp, target)
        
        # Evaluate true objective
        C_lp = evaluate_c5(h_lp)
        
        if C_lp < best_C:
            best_C = C_lp
            best_h = h_lp.copy()
        
        # Convergence check
        if t_lp is not None and C_lp - t_lp < 1e-5:
            break
        
        # Find most violated constraint
        c_vals = np.correlate(h_lp, 1.0 - h_lp, 'full') * (2.0 / n)
        d_star_full = int(np.argmax(c_vals))
        d_star = d_star_full - (n - 1)
        
        # Create cut
        g = grad_f_d(h_lp, d_star)
        c = c_vals[d_star_full] - g @ h_lp
        cuts.append((g, c))
        
        h = h_lp
        
        # Limit number of cuts
        if len(cuts) > 300:
            # Remove oldest cuts (or least useful ones)
            cuts = cuts[-200:]
    
    return best_h, best_C

def projected_gradient_step(h, n, alpha):
    """One step of projected gradient with exact subgradient"""
    target = n / 2.0
    dx = 2.0 / n
    c_vals = np.correlate(h, 1.0 - h, 'full') * dx
    d_star_full = int(np.argmax(c_vals))
    d_star = d_star_full - (n - 1)
    g = grad_f_d(h, d_star)
    h_new = project_feasible(h - alpha * g, target)
    return h_new

        else:
            no_improve += 1

import numpy as np
import time

try:
    import cvxpy as cp
    HAS_CP = True
except Exception:
    HAS_CP = False


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))


def grad_corr(h, j):
    """Gradient of corr_val[j] = np.correlate(h,1-h,'full')[j]*dx w.r.t. h."""
    n = len(h)
    dx = 2.0 / n
    d = j - (n - 1)
    g = np.zeros(n)
    # Term: (1-h[k-d]) for k where 0<=k-d<n
    lo1 = max(0, d)
    hi1 = min(n - 1, n - 1 + d)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += (1.0 - h[lo1 - d:hi1 + 1 - d])
    # Term: -h[k+d] for k where 0<=k+d<n
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 + d:hi2 + 1 + d]
    return g * dx


def project_feasible(y, target):
    lo_b, hi_b = -3.0, 4.0
    for _ in range(70):
        mid = (lo_b + hi_b) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo_b = mid
        else:
            hi_b = mid
    h = np.clip(y - (lo_b + hi_b) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            adj = diff / cnt
            if adj > 0:
                h[mask] = np.minimum(h[mask] + adj, 1.0)
            else:
                h[mask] = np.maximum(h[mask] + adj, 0.0)
    return h


def interp_to_n(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    xo = np.linspace(0, 2, n_old, endpoint=False)
    xn = np.linspace(0, 2, n_new, endpoint=False)
    h = np.interp(xn, xo, h_old)
    return project_feasible(h, n_new / 2.0)


def _solve_lp(cuts, target, n):
    """Solve min t s.t. g_k.h + c_k <= t, sum h = target, 0<=h<=1. Returns (h, t) or (None,None)."""
    h_var = cp.Variable(n)
    t_var = cp.Variable()
    constraints = [
        cp.sum(h_var) == target,
        h_var >= 0,
        h_var <= 1
    ]
    for g, c in cuts:
        constraints.append(cp.dot(g, h_var) + c <= t_var)
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    for solver in [cp.GLOP, cp.PDLP, cp.CLARABEL, cp.ECOS, cp.SCS]:
        try:
            prob.solve(solver=solver)
            if prob.status in ('optimal', 'optimal_inaccurate'):
                return np.asarray(h_var.value).flatten(), float(t_var.value)
        except Exception:
            continue
    try:
        prob.solve()
        if prob.status in ('optimal', 'optimal_inaccurate'):
            return np.asarray(h_var.value).flatten(), float(t_var.value)
    except Exception:
        pass
    return None, None


def cutting_plane_optimize(h0, n, deadline, max_iter=400):
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_C = evaluate_c5(h)
    cuts = []
    max_cuts = 250
    gap_tol = 5e-5
    stagnant = 0

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break

        if cuts:
            h_lp, t_lp = _solve_lp(cuts, target, n)
            if h_lp is None:
                break
            h_lp = np.clip(h_lp, 0.0, 1.0)
            h_lp = project_feasible(h_lp, target)
        else:
            h_lp = h.copy()
            t_lp = best_C

        C_lp = evaluate_c5(h_lp)
        if C_lp < best_C:
            best_C = C_lp
            best_h = h_lp.copy()
            stagnant = 0
        else:
            stagnant += 1

        if t_lp is not None and (C_lp - t_lp) < gap_tol:
            break

        if stagnant > 15:
            cuts = []
            stagnant = 0
            continue

        c_vals = np.correlate(h_lp, 1.0 - h_lp, 'full') * (2.0 / n)
        j_star = int(np.argmax(c_vals))
        g = grad_corr(h_lp, j_star)
        c_const = c_vals[j_star] - g @ h_lp
        cuts.append((g, c_const))

        if len(cuts) > max_cuts:
            # Remove least binding cuts
            slacks = []
            for idx, (gg, cc) in enumerate(cuts):
                sl = t_lp - (gg @ h_lp + cc) if t_lp is not None else 0.0
                slacks.append(sl)
            slacks = np.array(slacks)
            keep = np.argsort(slacks)[:max_cuts]
            cuts = [cuts[i] for i in sorted(keep)]

        h = h_lp

    return best_h, best_C


def subgradient_optimize(h0, n, deadline, max_iter=20000):
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_C = evaluate_c5(h)
    alpha = 0.005
    no_improve = 0
    best_no_improve_alpha = alpha

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break

        c_vals = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
        j_star = int(np.argmax(c_vals))
        g = grad_corr(h, j_star)

        h_new = project_feasible(h - alpha * g, target)
        C_new = evaluate_c5(h_new)

        if C_new < best_C - 1e-13:
            best_C = C_new
            best_h = h_new.copy()
            no_improve = 0
            alpha = min(alpha * 1.03, 0.05)
        else:
            no_improve += 1
            alpha *= 0.85

        h = h_new

        if no_improve > 200:
            break

    return best_h, best_C


def block_gradient_optimize(h0, n, deadline, max_iter=5000):
    """Use average of top-k active gradients for faster convergence."""
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_C = evaluate_c5(h)
    alpha = 0.008
    no_improve = 0
    top_k = 5

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break

        c_vals = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
        # Get top-k indices
        k = min(top_k, len(c_vals))
        top_idx = np.argsort(c_vals)[-k:]
        # Weight by softmax of top values
        top_vals = c_vals[top_idx]
        w = np.exp((top_vals - top_vals.max()) / 0.01)
        w /= w.sum()

        g = np.zeros(n)
        for idx, wi in zip(top_idx, w):
            g += wi * grad_corr(h, int(idx))

        h_new = project_feasible(h - alpha * g, target)
        C_new = evaluate_c5(h_new)

        if C_new < best_C - 1e-13:
            best_C = C_new
            best_h = h_new.copy()
            no_improve = 0
            alpha = min(alpha * 1.02, 0.04)
        else:
            no_improve += 1
            alpha *= 0.9

        h = h_new
        if no_improve > 300:
            break

    return best_h, best_C

    # Phase 1: Cutting plane at increasing resolutions
    n_list = []
    for mult in [1, 2, 3, 4, 6, 8, 10]:
        n = n_parent * mult
        if n <= 2400 and n not in n_list:
            n_list.append(n)
    n_list.sort()

    for n in n_list:

            break
        remaining = deadline - time.time()
        if remaining < 30:
            break
        h0 = interp_to_n(h_parent, n)

        if C_opt < best_C:

            best_h = h_opt.copy()

    # Phase 3: Cutting plane again on polished solution

        n_best = len(best_h)
        h_opt, C_opt = cutting_plane_optimize(best_h, n_best, deadline, max_iter=200)
        if C_opt < best_C:
            best_C = C_opt
            best_h = h_opt.copy()

    # Phase 4: Try perturbations from best

        n_best = len(best_h)
        for amp in [0.003, 0.008, 0.02]:
            if time.time() > deadline - 15:
                break
            h_pert = project_feasible(best_h + rng.standard_normal(n_best) * amp, n_best / 2.0)
            h_opt, C_opt = cutting_plane_optimize(h_pert, n_best, deadline, max_iter=150)
            if C_opt < best_C:
                best_C = C_opt
                best_h = h_opt.copy()

    # Phase 5: Try refinement at higher resolution
    if time.time() < deadline - 60:
        n_cur = len(best_h)
        n_next = min(n_cur * 2, 2400)
        if n_next > n_cur:
            h0 = interp_to_n(best_h, n_next)
            h_opt, C_opt = cutting_plane_optimize(h0, n_next, deadline, max_iter=200)
            if C_opt < best_C:

    # Final feasibility enforcement
