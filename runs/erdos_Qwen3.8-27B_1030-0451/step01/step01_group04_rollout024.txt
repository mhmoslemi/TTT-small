def fast_grad(h, w):
    N = len(h)
    dx = 2.0 / N
    A_arr = w[N:2*N-1]  # length N-1: A[1], A[2], ..., A[N-1]
    B_arr = w[N-2:-1:-1]  # length N-1: B[1], B[2], ..., B[N-1] = w[N-2], w[N-3], ..., w[0]
    w0 = w[N-1]
    g = 1.0 - h
    
    N_fft = next_fast_len(2*N)
    
    A_p = np.zeros(N_fft); A_p[:N-1] = A_arr
    B_p = np.zeros(N_fft); B_p[:N-1] = B_arr
    Af = A_arr[::-1]
    Bf = B_arr[::-1]
    Af_p = np.zeros(N_fft); Af_p[:N-1] = Af
    Bf_p = np.zeros(N_fft); Bf_p[:N-1] = Bf
    h_p = np.zeros(N_fft); h_p[:N] = h
    g_p = np.zeros(N_fft); g_p[:N] = g
    
    # Linear convolutions
    conv_Bg = np.real(ifft(fft(B_p) * fft(g_p)))
    conv_Ah = np.real(ifft(fft(A_p) * fft(h_p)))
    conv_Afg = np.real(ifft(fft(Af_p) * fft(g_p)))
    conv_Bfh = np.real(ifft(fft(Bf_p) * fft(h_p)))
    
    # S1: length N, S1[0]=0, S1[j]=conv_Bg[j-1] for j>=1
    S1 = np.zeros(N)
    S1[1:] = conv_Bg[:N-1]
    
    # S3: length N, S3[0]=0, S3[j]=conv_Ah[j-1] for j>=1
    S3 = np.zeros(N)
    S3[1:] = conv_Ah[:N-1]
    
    # S2: S2[j]=conv_Afg[j+N-1]
    S2 = conv_Afg[N-1:N-1+N]
    
    # S4: S4[j]=conv_Bfh[j+N-1]
    S4 = conv_Bfh[N-1:N-1+N]
    
    return dx * (w0 * (1.0 - 2.0*h) + S2 + S1 - S3 - S4)
