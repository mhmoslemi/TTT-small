def solve_lp_direction(h, corr, G, dx, active_tol=0.05, radius=0.1):
    n = len(h)
    c = corr * dx
    f = np.max(c)
    # Active set: correlations within active_tol of f
    active = np.where(c > f - active_tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # LP: minimize t
    # Variables: delta (n) + t (1) = n+1
    # Constraints for each active k:
    #   G[k, :] @ delta - t <= f - c[k]
    #   i.e., G[k, :] @ delta + (-1)*t <= f - c[k]
    # Sum constraint: sum(delta) = 0
    # Bounds: -h[i] <= delta[i] <= 1-h[i] (and also |delta[i]| <= radius)
    #         t >= 0
    
    m = len(active)
    # A_ub: m + 0 rows (only inequalities are the correlation constraints)
    A_ub = np.zeros((m, n+1))
    A_ub[:, :n] = G[active, :]  # G is gradient/dx, but we need gradient = dx*G
    # Wait, c_k + grad_k . delta <= f + t
    # c_k + dx * G[k,:] . delta <= f + t
    # dx * G[k,:] . delta - t <= f - c_k
    A_ub[:, :n] = dx * G[active, :]
    A_ub[:, n:] = -1.0
    b_ub = f - c[active]
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, n+1))
    A_eq[:, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    lb = np.maximum(-h, -radius)
    ub = np.minimum(1.0 - h, radius)
    bounds_delta = list(zip(lb, ub))
    bounds_t = [(0.0, None)]
    bounds_all = bounds_delta + [bounds_t]
    
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # minimize t
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds_all, method='highs')
    
    if res.success:
        return res.x[:n]
    return None

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_constraint(p, n):
    target = n / 2.0
    p = np.clip(p, -10.0, 10.0)  # safety
    lo = float(np.min(p)) - 2.0
    hi = float(np.max(p)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)

    diff = target - float(np.sum(proj))

        nm = int(np.sum(mask))

    """Solve LP to find a direction that reduces the max correlation.
    Returns delta (direction) or None if LP fails."""
    n = len(h)
    f = np.max(c)
    # Active set: top fraction of correlations
    thresh = f - active_frac * (f - np.min(c))
    if thresh < f * 0.5:
        thresh = f * 0.8
    active = np.where(c >= thresh)[0]
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Variables: delta (n) + t (1)
    # Minimize t
    # Constraints: dx * G[k,:] @ delta - t <= f - c[k] for k in active
    # Equality: sum(delta) = 0
    # Bounds: max(-h[i], -radius) <= delta[i] <= min(1-h[i], radius), t free
    
    m = len(active)
    A_ub = np.zeros((m, n + 1))

    A_ub[:, n] = -1.0
