import numpy as np

def project_sum_box(h, target):
    """Robust projection onto {0<=x<=1, sum(x)=target}."""
    h = np.array(h, dtype=float).copy()
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        y = np.clip(h + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            mask = y < 1.0
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.clip(y[mask] + r / cnt, 0, 1)
        else:
            mask = y > 0.0
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.clip(y[mask] + r / cnt, 0, 1)
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx[0]] = np.clip(y[idx[0]] + r, 0, 1)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx[0]] = np.clip(y[idx[0]] + r, 0, 1)
    return np.clip(y, 0.0, 1.0)

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    v = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(v))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def grad_lag(h, k):
    """Gradient of correlation at lag k (k can be negative)."""
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        idx1 = np.arange(0, n - k)
        g[idx1] += (1.0 - h[idx1 + k])
        idx2 = np.arange(k, n)
        g[idx2] -= h[idx2 - k]
    else:
        d = -k
        idx1 = np.arange(d, n)
        g[idx1] += (1.0 - h[idx1 - d])
        idx2 = np.arange(0, n - d)
        g[idx2] -= h[idx2 + d]
    return g

def pair_exchange_descent(h0, deadline, max_rounds=200, gap_frac=0.08, max_pairs_per_lag=40):
    """Exact pair-exchange descent targeting active lags."""
    n = len(h0)
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    h = h0.copy()
    stall = 0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = v.max()
        gap = max(gap_frac * vmax, 0.002)
        active_idx = np.where(v >= vmax - gap)[0]
        # Sort by correlation value descending
        active_idx = active_idx[np.argsort(v[active_idx])[::-1]]
        improved = False
        
        for k_idx in active_idx:
            if time.time() > deadline:
                break
            k = k_idx - (n - 1)
            if k == 0:
                continue
            d = abs(k)
            # Pairs at distance d
            if k > 0:
                pair_ranges = [(i, i + d) for i in range(n - d)]
            else:
                pair_ranges = [(i, i + d) for i in range(n - d)]
            
            # Shuffle for diversity
            np.random.shuffle(pair_ranges)
            tested = 0
            for (i, j) in pair_ranges[:max_pairs_per_lag]:
                if time.time() > deadline:
                    break
                tested += 1
                s = h[i] + h[j]

def subgrad_accelerated(h0, deadline, max_iters=30000):
    """Subgradient method with Nesterov acceleration and Polyak step."""
    n = len(h0)
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    h = h0.copy()
    y = h0.copy()  # Nesterov momentum
    c_best_global = best_c
    mu = 1.0
    t = 1.0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        v = corr_all(y)
        c_y = v.max()
        
        if c_y < best_c:
            best_c = c_y
            best_h = y.copy()
            c_best_global = best_c
        
        # Active set subgradient
        gap = max(0.1 * c_y, 0.005)
        active = np.where(v >= c_y - gap)[0]
        g = np.zeros(n)
        for idx in active:
            k = idx - (n - 1)
            g += grad_lag(h, k)
        g /= max(len(active), 1)
        
        # Project onto zero-sum directions
        g = g - np.mean(g)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        
        # Polyak step size
        alpha = (c_y - c_best_global) / (g_norm * g_norm + 1e-12)
        alpha = np.clip(alpha, 0.001, 1.0)
        
        h_new = project_sum_box(y - alpha * g, n / 2.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        beta = (t - 1.0) / t_new
        y_new = h_new + beta * (h_new - y)
        y_new = project_sum_box(y_new, n / 2.0)
        t = t_new
        h = h_new
        y = y_new
    
    return best_h, best_c

def slsqp_exact(h0, deadline, max_iter=300):
    """SLSQP with exact non-smooth objective."""
    n = len(h0)
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    bounds = [(0.0, 1.0)] * n
    
    def obj(x):
        return eval_c5(x)
    
    def con_fun(x):
        return x.sum() - n / 2.0
    
    def con_jac(x):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    
    for attempt in range(3):
        if time.time() > deadline:
            break
        if attempt == 0:
            x0 = best_h.copy()
        else:
            x0 = best_h + np.random.randn(n) * 0.01
            x0 = project_sum_box(x0, n / 2.0)
        try:
            res = minimize(obj, x0, method='SLSQP', bounds=bounds,
                          constraints=cons,
                          options={'maxiter': max_iter, 'ftol': 1e-12, 'iprint': 0})
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = eval_c5(x_new)
                if c_new < best_c - 1e-12:
                    best_h = x_new.copy()
                    best_c = c_new
        except Exception:
            pass
    
    return best_h, best_c

def binary_seed(n, rng, alpha=1.0):
    """Generate a balanced binary seed."""
    if n % 2 != 0:
        n -= 1
    half = n // 2
    h_bin = np.zeros(n)
    h_bin[:half] = 1.0
    rng.shuffle(h_bin)
    h_cont = (1.0 - alpha) * 0.5 + alpha * h_bin
    return project_sum_box(h_cont, n / 2.0)

    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

                    h_dn, c_dn = pair_exchange_descent(best_h, sub_dl2, max_rounds=80, gap_frac=0.04)
                    if c_dn < best_c:
                        best_h, best_c = h_dn, c_dn
                except Exception:
                    pass

    # Phase 7: Final SLSQP polish
    if time_left() > 30:
        try:
            h7, c7 = slsqp_exact(best_h, deadline - 3, max_iter=150)
            if c7 < best_c:
                best_h, best_c = h7, c7
        except Exception:
            pass

    # Phase 8: Final local polish
    if time_left() > 15:
        try:
            h8, c8 = local_polish(best_h, max(deadline - 2, time.time() + 5), n_trials=200)
            if c8 < best_c:
                best_h, best_c = h8, c8
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = eval_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
