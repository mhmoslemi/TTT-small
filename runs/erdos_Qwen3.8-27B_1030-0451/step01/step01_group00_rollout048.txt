import time
from scipy.optimize import linprog

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_constraint(p, n):
    target = n / 2.0
    p = np.clip(p, -2.0, 2.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0

        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm

    """Get indices of top M correlations, or those within gap of max."""
    c = corr.copy()
    f = c.max()
    mask = c > f - gap
    idx = np.where(mask)[0]
    if len(idx) > M:
        # Take top M
        top_idx = np.argsort(c)[-M:]
        return top_idx
    return idx

def compute_grad_for_lags(h, k_list):
    """Compute gradients of c_k for k in k_list. k_list are indices into full correlation.
    Returns array of shape (len(k_list), n) where row i is grad of c_{k_list[i]}.
    """
    n = len(h)
    grads = np.zeros((len(k_list), n))
    for idx, k in enumerate(k_list):
        lag = k - (n - 1)
        if lag >= 0:
            # Term: sum_{i=lag}^{n-1} h[i]*(1-h[i-lag])
            # d/dh[i] = (1-h[i-lag]) for i >= lag
            # d/dh[i-lag] = -h[i] for i >= lag, i.e., d/dh[j] = -h[j+lag] for j < n-lag
            i_range = np.arange(lag, n)
            j_range = i_range - lag
            grads[idx, i_range] += (1.0 - h[j_range])
            grads[idx, j_range] -= h[i_range]
        else:
            L = -lag
            # Term: sum_{i=0}^{n-1-L} h[i]*(1-h[i+L])
            # d/dh[i] = (1-h[i+L])
            # d/dh[i+L] = -h[i]
            i_range = np.arange(0, n - L)
            j_range = i_range + L
            grads[idx, i_range] += (1.0 - h[j_range])
            grads[idx, j_range] -= h[i_range]
    return grads

def solve_lp_direction(h, corr, radius, M_active=12):
    """Solve LP to find direction d that reduces max correlation.
    Returns d (or None if LP fails).
    """
    n = len(h)
    dx = 2.0 / n
    f = float(np.max(corr))
    
    # Select active constraints: top M correlations
    sorted_idx = np.argsort(corr)[::-1][:M_active]
    
    # Compute gradients for these correlations
    grads = compute_grad_for_lags(h, sorted_idx)  # shape (M, n)
    
    # LP: min t
    # s.t. grads[k] @ d - t <= f - corr[k]  for each k
    #      sum(d) = 0
    #      -min(r, h[i]) <= d[i] <= min(r, 1-h[i])
    #
    # Variables: d[0..n-1], t
    # Objective: [0, ..., 0, 1] (minimize t)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints A_ub @ x <= b_ub
    # For each active k: grads[k] @ d - t <= f - corr[k]
    # [grads[k], -1] @ [d, t] <= f - corr[k]
    A_ub = np.zeros((M_active, n + 1))
    b_ub = np.zeros(M_active)
    for i, k_idx in enumerate(sorted_idx):
        A_ub[i, :n] = grads[i] * dx  # scale by dx to work with normalized correlations
        A_ub[i, n] = -1.0
        b_ub[i] = f - corr[k_idx] * dx
    
    # Box constraints on d: lo_d[i] <= d[i] <= hi_d[i]
    lo_d = (-np.minimum(radius, h)).astype(float)
    hi_d = np.minimum(radius, 1.0 - h).astype(float)
    
    bounds = list(zip(lo_d, hi_d)) + [(None, None)]  # t is free
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            # Ensure sum is exactly 0
            d -= np.mean(d)
            return d
    except Exception:
        pass
    return None

def softmax_gradient(h, corr, tau):
    """Compute softmax-weighted gradient of max correlation."""
    n = len(h)
    dx = 2.0 / n
    c_norm = corr * dx
    f = np.max(c_norm)
    cs = c_norm - f
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    
    # Get the lags
    k_list = np.arange(len(corr))
    grads = compute_grad_for_lags(h, k_list)  # shape (2n-1, n)
    
    # Weighted sum
    grad = (w[:, None] * grads).sum(axis=0) * dx
    return grad

     d_i ≤ 1 - h_i (from h_i + d_i ≤ 1)

def project_constraint(p, n):
    target = n / 2.0
    p = np.clip(p, -2.0, 2.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(70):

        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)

def compute_grads_subset(h, k_list):
