    best_c = evaluate_c5(best_h)
    dx = 2.0 / n
    cuts = []
    no_improve = 0
    radius = 0.15

    for it in range(max_iters):
        if time.time() > deadline - 1:
            break
        v = corr_all(h)
        vmax = np.max(v)
        
        # Select active lags for new cuts
        gap = max(0.003 * vmax, 0.001)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(v)])
        
        # Add new cuts (up to max 5 new per iteration)
        new_count = 0
        for k in active_idx:
            if new_count >= 5:
                break
            g, b = make_cut(h, k, h, v)
            # Check if this cut is already close to existing ones
            is_dup = False
            for existing in cuts:
                if np.abs(existing[0] - g).max() < 1e-10:
                    is_dup = True
                    break
            if not is_dup:
                cuts.append((g, b))
                new_count += 1
        
        # Trim cuts if too many - keep those most "active" (tightest)
        if len(cuts) > max_cuts:
            violations = np.array([c[0] @ h - c[1] for c in cuts])
            keep = np.argsort(violations)[-max_cuts:]
            cuts = [cuts[i] for i in keep]
        
        # Solve LP
        h_new, t_lp = solve_lp_with_cuts(cuts, h, target, radius, c_lower=0.0)
        
        if h_new is not None:
            h_new = project_sum_box(h_new, target)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 1e-14:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
                no_improve = 0
                # Slowly increase radius as we improve
                radius = min(radius * 1.02, 0.3)
            else:
                no_improve += 1
                if no_improve >= 5:
                    radius = max(radius * 0.7, 0.02)
                    no_improve = 0
                h = project_sum_box(0.5 * h + 0.5 * best_h, target)
        else:
            no_improve += 1
            if no_improve >= 10:
                radius = max(radius * 0.5, 0.01)
                no_improve = 0
    
    return best_h, best_c

def smooth_obj_grad(h, beta):
    """Log-sum-exp smoothed objective and gradient."""
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 500)
    u = np.exp(expo)
    Z = u.sum()
    f = m + np.log(Z) / beta
    w = u / Z
    n = len(h)
    g = np.zeros(n)
    for k in range(len(v)):
        if w[k] > 1e-14:
            lag = k - (n - 1)
            g += w[k] * grad_lag_vec(h, lag)
    return f, g

def method_nesterov(h0, beta, deadline, max_iters=5000, step0=0.02):
    """Nesterov-accelerated projected subgradient."""
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h_t = h0.copy()
    h_prev = h0.copy()
    
    for it in range(max_iters):
        if time.time() > deadline - 1:
            break
        if it > 0:
            mu = (float(it - 1) / (it + 2)) ** 2
            y = h_t + mu * (h_t - h_prev)
        else:
            y = h_t.copy()
        y = np.clip(y, 0.0, 1.0)
        y = project_sum_box(y, target)
        
        f, g = smooth_obj_grad(y, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        step = step0 / (1.0 + 0.3 * it)
        h_new = y - step * g
        h_new = project_sum_box(h_new, target)
        
        c_new = evaluate_c5(h_new)
        if c_new < best_c - 1e-14:
            best_c = c_new
            best_h = h_new.copy()
        
        h_prev = h_t
        h_t = h_new
    
    return best_h, best_c

def method_coord_refine(h0, deadline, max_rounds=30):
    """Pairwise coordinate descent with sum-preserving moves."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        improved = False
        n_sub = min(n, 200)
        indices = np.random.permutation(n)[:n_sub]
        for i in indices:
            if time.time() > deadline - 5:
                break
            for j in indices:
                if j <= i:
                    continue
                if h[j] - h[i] < 0.01:
                    continue
                for eps in [0.02, 0.05, 0.1, 0.15]:
                    if h[j] - h[i] < eps:
                        continue
                    if h[i] + eps > 1.0:
                        continue
                    h_tr = h.copy()
                    h_tr[i] += eps
                    h_tr[j] -= eps
                    c_tr = evaluate_c5(h_tr)
                    if c_tr < best_c - 1e-13:
                        best_c = c_tr
                        best_h = h_tr.copy()
                        h = h_tr
                        improved = True
                        break
        if not improved and rnd > 3:
            break
    
    return best_h, best_c

def transfer_resolution(h0, n_new):
    """Interpolate to new grid size."""
    n_old = len(h0)
    x_old = np.arange(n_old)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def binary_local_search(h0, deadline, max_flips=5000):
    """Try flipping bits on a binary projection."""
    n = len(h0)
    target = n / 2.0
    # Project to binary
    h_bin = (h0 > 0.5).astype(float)
    # Fix sum
    deficit = target - h_bin.sum()
    if deficit > 0:
        order = np.argsort(1.0 - h0)[::-1][:int(deficit)]
        h_bin[order] = 1.0
    elif deficit < 0:
        order = np.argsort(h0)[:int(-deficit)]
        h_bin[order] = 0.0
    h_bin = np.clip(h_bin, 0.0, 1.0)
    
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    c_bin = evaluate_c5(h_bin)
    if c_bin < best_c:
        best_c = c_bin
        best_h = h_bin.copy()
    
    # Try single-bit flips (swap a 1 for a 0)
    ones = np.where(h_bin == 1.0)[0]
    zeros = np.where(h_bin == 0.0)[0]
    flips = 0
    rng = np.random.default_rng(42)
    for _ in range(max_flips):
        if time.time() > deadline - 5:
            break
        flips += 1
        i = ones[rng.integers(len(ones))]
        j = zeros[rng.integers(len(zeros))]
        h_tr = h_bin.copy()
        h_tr[i] = 0.0
        h_tr[j] = 1.0
        c_tr = evaluate_c5(h_tr)
        if c_tr < best_c - 1e-13:
            best_c = c_tr
            best_h = h_tr.copy()
            h_bin = h_tr.copy()
            ones = np.where(h_bin == 1.0)[0]
            zeros = np.where(h_bin == 0.0)[0]
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0
    deadline = start_time + budget_s - safety

    # Get initial values
    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        n = 100
        h_init = np.full(n, 0.5)
        h_init = project_sum_box(h_init, n / 2.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n_best = len(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: LP with accumulated cuts (primary method)
    if time_left() > 60:
        phase1_dl = start_time + budget_s * 0.35
        try:
            h1, c1 = method_lp_accumulated(best_h, min(deadline, phase1_dl),
                                           max_iters=5000, max_cuts=250)
            if c1 < best_c:
                best_h = h1
                best_c = c1
        except Exception:
            pass

    # Phase 2: Nesterov accelerated subgradient
    if time_left() > 40:
        phase2_dl = min(deadline, start_time + budget_s * 0.50)
        for beta in [50, 100, 200]:
            if time_left() < 15:
                break
            try:
                h2, c2 = method_nesterov(best_h, beta, phase2_dl,
                                         max_iters=3000, step0=0.015)
                if c2 < best_c:
                    best_h = h2
                    best_c = c2
            except Exception:
                pass

    # Phase 3: Another LP round from improved point
    if time_left() > 40:
        phase3_dl = min(deadline, start_time + budget_s * 0.65)
        try:
            h3, c3 = method_lp_accumulated(best_h, phase3_dl,
                                           max_iters=4000, max_cuts=300)
            if c3 < best_c:
                best_h = h3
                best_c = c3
        except Exception:
            pass

    # Phase 4: Coordinate refinement
    if time_left() > 30:
        phase4_dl = min(deadline, start_time + budget_s * 0.78)
        try:
            h4, c4 = method_coord_refine(best_h, phase4_dl, max_rounds=20)
            if c4 < best_c:
                best_h = h4
                best_c = c4
        except Exception:
            pass

    # Phase 5: Binary local search
    if time_left() > 30:
        phase5_dl = min(deadline, start_time + budget_s * 0.85)
        try:
            h5, c5 = binary_local_search(best_h, phase5_dl, max_flips=3000)
            if c5 < best_c:
                best_h = h5
                best_c = c5
        except Exception:
            pass

    # Phase 6: Resolution increase and re-optimize
    if time_left() > 80:
        n_cur = len(best_h)
        target_n = max(n_cur + 10, min(n_cur + 20, 120))
        if target_n > n_cur:
            try:
                h_tr = transfer_resolution(best_h, target_n)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c:
                    best_h = h_tr.copy()
                    best_c = c_tr
                # Optimize on new grid
                sub_dl = min(deadline - 10, time.time() + min(120, time_left() * 0.6))
                h_a, c_a = method_lp_accumulated(best_h, sub_dl,
                                                 max_iters=2000, max_cuts=200)
                if c_a < best_c:
                    best_h = h_a
                    best_c = c_a
            except Exception:
                pass

    # Phase 7: Final LP polish
    if time_left() > 20:
        try:
            h7, c7 = method_lp_accumulated(best_h, deadline - 2,
                                           max_iters=3000, max_cuts=300)
            if c7 < best_c:
                best_h = h7
                best_c = c7
        except Exception:
            pass

    # Phase 8: Final Nesterov polish
    if time_left() > 10:
        try:
            h8, c8 = method_nesterov(best_h, 200, deadline - 2,
                                     max_iters=2000, step0=0.01)
            if c8 < best_c:
                best_h = h8
                best_c = c8
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
