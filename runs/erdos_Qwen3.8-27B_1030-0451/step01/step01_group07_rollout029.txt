            g[p:] += wp * (1.0 - h[:-p])
            g[:-p] -= wp * h[p:]
        wm = weights[center - p]
        if abs(wm) > 1e-18:
            g[:-p] += wm * (1.0 - h[p:])
            g[p:] -= wm * h[:-p]
    g *= dx
    return g

def active_set_grad(h, c_values, tau, topk=20):
    """
    Compute gradient of sum_{r in A} (C_r - tau)^2
    where A = {r : C_r > tau}
    """
    n = len(h)
    excess = np.maximum(0.0, c_values - tau)
    # Use top-k for efficiency
    k = min(topk, int((excess > 1e-12).sum()))
    if k == 0:
        return np.zeros(n), 0.0
    # Get the top-k indices
    top_idx = np.argsort(-excess)[:k]
    weights = np.zeros(len(c_values))
    weights[top_idx] = 2.0 * excess[top_idx]
    grad = corr_gradient(h, weights)
    return grad, float(excess.sum())

def lse_smooth(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    val = m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
    return dx * val

def lse_grad(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    return corr_gradient(h, w * sigma)

def de_evolve(pop, pop_c5, F, CR, n, rng, max_generations, deadline):
    """Differential evolution on feasible population."""
    N = len(pop)
    best_h = pop[int(np.argmin(pop_c5))].copy()
    best_c5 = min(pop_c5)
    
    for gen in range(max_generations):
        if time.monotonic() > deadline:
            break
        for i in range(N):
            if time.monotonic() > deadline:
                break
            # Pick 3 distinct indices
            while True:
                a, b, c = rng.integers(N, size=3)
                if len(set([a, b, c, i])) == 4:
                    break
            # Mutation
            mutant = pop[a] + F * (pop[b] - pop[c])
            # Crossover
            crosser = rng.random(n) < CR
            if not crosser.any():
                crosser[0] = True
            trial = np.where(crosser, mutant, pop[i])
            # Project

            # Evaluate

                pop_c5[i] = tc5
                if tc5 < best_c5:
                    best_c5 = tc5
                    best_h = trial.copy()
    return best_h, best_c5

        best_h = h.copy()
    return best_h, best_c5

def sturmian_sequence(n, alpha, phi):
    """Generate a Sturmian (mechanical) word of length n."""
    h = np.zeros(n)
    for i in range(n):
        h[i] = math.floor((i + 1) * alpha + phi) - math.floor(i * alpha + phi)
    return h

def resample(h, n_new):

    return np.interp(x1, x0, h)

def de_evolve(pop, pop_c5, F, CR, n, rng, max_gens, deadline):
    N = len(pop)
    best_h = pop[int(np.argmin(pop_c5))].copy()
    best_c5 = float(min(pop_c5))
    for gen in range(max_gens):
        if time.monotonic() > deadline:
            break
        for i in range(N):
            if time.monotonic() > deadline:
                break
            picks = []
            while len(picks) < 3:
                p = int(rng.integers(N))
                if p != i and p not in picks:
                    picks.append(p)
            mutant = pop[picks[0]] + F * (pop[picks[1]] - pop[picks[2]])
            mask = rng.random(n) < CR
            if not mask.any():
                mask[(rng.integers(n),)] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_feasible(trial, n)
            tc5 = compute_c5(trial)
            if tc5 < pop_c5[i]:
                pop[i] = trial
                pop_c5[i] = tc5
                if tc5 < best_c5:
                    best_c5 = tc5
                    best_h = trial.copy()
    return best_h, best_c5


def lse_local(h, n, sigma, time_limit, deadline, eta0=0.1):
    h = np.asarray(h, dtype=float).copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur
    t0 = time.monotonic()
    eta = eta0
    stall = 0
    while time.monotonic() - t0 < time_limit and time.monotonic() < deadline:
        g = lse_grad(h, sigma)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        improved = False
        e = eta
        for _ in range(32):
            cand = project_feasible(h - e * g, n)
            cv = compute_c5(cand)
            if cv < cur - 1e-15:
                h = cand
                cur = cv
                if cv < best_c5 - 1e-15:
                    best_c5 = cv
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            e *= 0.5
        if not improved:
            nz = rng_local_normal(n)
            h = project_feasible(h + 0.008 * nz, n)
            cur = compute_c5(h)
            if cur < best_c5 - 1e-15:
                best_c5 = cur
                best_h = h.copy()
            stall += 1
            if stall >= 10:
                eta *= 0.4
                stall = 0
                if eta < 1e-9:
                    break
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


_rng_local = np.random.default_rng(12345)


def rng_local_normal(n):
    return _rng_local.normal(0.0, 1.0, n)


def active_local(h, n, time_limit, deadline):
    h = np.asarray(h, dtype=float).copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur
    t0 = time.monotonic()
    eta = 0.05
    stall = 0
    delta = 0.005
    while time.monotonic() - t0 < time_limit and time.monotonic() < deadline:
        c_vals = compute_c5_all(h)
        c_max = c_vals.max()
        tau = c_max - delta
        g = active_set_grad(h, c_vals, tau, topk=40)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            delta *= 0.5
            continue
        improved = False
        e = eta
        for _ in range(32):
            cand = project_feasible(h - e * g, n)
            cv = compute_c5(cand)
            if cv < cur - 1e-15:
                h = cand
                cur = cv
                if cv < best_c5 - 1e-15:
                    best_c5 = cv
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            e *= 0.5
        if not improved:
            nz = rng_local_normal(n)
            h = project_feasible(h + 0.005 * nz, n)
            cur = compute_c5(h)
            if cur < best_c5 - 1e-15:
                best_c5 = cur
                best_h = h.copy()
            stall += 1
            if stall >= 12:
                eta *= 0.4
                delta *= 0.5
                stall = 0
                if eta < 1e-9:
                    break
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def hill_climb(h, n, time_limit, deadline, step=0.02):
    h = np.asarray(h, dtype=float).copy()
    cur = compute_c5(h)
    best_h = h.copy()
    best_c5 = cur
    t0 = time.monotonic()
    rng_hc = np.random.default_rng(99999)
    while time.monotonic() - t0 < time_limit and time.monotonic() < deadline:
        j = int(rng_hc.integers(n))
        delta = (rng_hc.random() * 2 - 1) * step
        if h[j] + delta < -0.01 or h[j] + delta > 1.01:
            continue
        trial = h.copy()
        trial[j] += delta
        trial = project_feasible(trial, n)
        tc5 = compute_c5(trial)
        if tc5 < cur - 1e-15:
            h = trial
            cur = tc5
            if tc5 < best_c5 - 1e-15:
                best_c5 = tc5
                best_h = h.copy()
        else:
            step *= 0.95
            if step < 1e-6:
                break
    return best_h, best_c5


def resample_to(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def sturmian(n, alpha, phi):
    h = np.zeros(n)
    for i in range(n):
        h[i] = math.floor((i + 1) * alpha + phi) - math.floor(i * alpha + phi)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    global _rng_local
    _rng_local = np.random.default_rng(777)
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 8.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is not None:
        h0 = parent.copy()
    else:
        h0 = sturmian(131, 0.5, 0.0)
    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Phase 1: DE global search
    pop_size = min(50, max(20, int(0.30 * effective / 0.002)))
    pop_size = min(pop_size, 60)
    pop = [best_h.copy()]
    pop_c5 = [best_c5]

    phases = [0.0, 0.11, 0.23, 0.37, 0.44, 0.55, 0.61, 0.73, 0.83, 0.05]
    for ph in phases:
        if len(pop) >= pop_size:
            break
        s = sturmian(n0, 0.5, ph)
        s = project_feasible(s, n0)
        sc5 = compute_c5(s)
        pop.append(s)
        pop_c5.append(sc5)
        if sc5 < best_c5:
            best_c5 = sc5
            best_h = s.copy()

    # Add perturbed copies of best
    while len(pop) < pop_size:
        p = pop[rng.integers(len(pop))].copy()
        p = p + rng.normal(0, 0.05, n0)
        p = project_feasible(p, n0)
        pop.append(p)
        pop_c5.append(compute_c5(p))

    de_deadline = t0 + 0.32 * effective
    bh, bc = de_evolve(pop, np.array(pop_c5), 0.65, 0.7, n0, rng, 500, de_deadline)
    if bc < best_c5:
        best_c5 = bc
        best_h = bh.copy()

    # Phase 2: LSE local optimization with decreasing sigma
    sigma_list = [150.0, 80.0, 40.0, 20.0]
    for sg in sigma_list:
        if time.monotonic() > deadline - 3.0:
            break
        alloc = min(0.08 * effective, deadline - time.monotonic() - 1.0)
        if alloc < 0.5:
            break
        ch, cc = lse_local(best_h, n0, sg, alloc, deadline, eta0=0.08)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch.copy()

    # Phase 3: Active-set local optimization
    if time.monotonic() < deadline - 3.0:
        alloc = min(0.12 * effective, deadline - time.monotonic() - 1.0)
        if alloc > 1.0:
            ch, cc = active_local(best_h, n0, alloc, deadline)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 4: Multi-resolution
    if time.monotonic() < deadline - 3.0:
        try_ns = [n0, n0 + 2, n0 - 2, n0 + 4, n0 - 4]
        seen = set()
        for nn in try_ns:
            if nn in seen or nn < 60 or nn > 300:
                continue
            seen.add(nn)
            if time.monotonic() > deadline - 2.0:
                break
            alloc = min(0.05 * effective, deadline - time.monotonic() - 1.0)
            if alloc < 0.5:
                break
            rs = project_feasible(resample_to(best_h, nn), nn)
            ch, cc = lse_local(rs, nn, 100.0, min(alloc * 0.6, 60.0), deadline, eta0=0.06)
            if cc < best_c5:
                alloc2 = min(alloc * 0.4, deadline - time.monotonic() - 0.5)
                if alloc2 > 0.5:
                    ch2, cc2 = active_local(ch, nn, alloc2, deadline)
                    if cc2 < cc:
                        cc = cc2
                        ch = ch2
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 5: Hill climbing
    if time.monotonic() < deadline - 1.0:
        alloc = min(0.06 * effective, deadline - time.monotonic() - 0.5)
        if alloc > 0.5:
            ch, cc = hill_climb(best_h, len(best_h), alloc, deadline, step=0.015)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 6: Final LSE polish
    if time.monotonic() < deadline - 0.5:
        alloc = min(0.04 * effective, deadline - time.monotonic() - 0.3)
        if alloc > 0.3:
            ch, cc = lse_local(best_h, len(best_h), 200.0, alloc, deadline, eta0=0.03)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Final feasibility lock
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    n_pts = len(best_h)
    return best_h, float(best_c5), int(n_pts)
