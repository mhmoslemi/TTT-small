import numpy as np
import time

_CORR_DIR = 1
_CORR_DIR_SET = False


def _calibrate_corr_direction():
    global _CORR_DIR, _CORR_DIR_SET
    if _CORR_DIR_SET:
        return _CORR_DIR
    n = 13
    rng = np.random.default_rng(7)
    h = rng.uniform(0.2, 0.8, n)
    target = n / 2
    for _ in range(200):
        diff = target - h.sum()
        if abs(diff) < 1e-12:
            break
        idx = np.where((diff > 0) == (h < 1))
        h[idx[0]] += diff / len(idx[0])
        h = np.clip(h, 0, 1)
    cact = np.correlate(h, 1.0 - h, "full")
    ctr = n - 1
    raw1 = np.empty(2 * n - 1)
    raw2 = np.empty(2 * n - 1)
    for k in range(2 * n - 1):
        d = k - ctr
        if d == 0:
            raw1[k] = raw2[k] = np.sum(h * (1 - h))
        elif d > 0:
            raw1[k] = np.sum(h[d:] * (1 - h[:-d]))
            raw2[k] = np.sum(h[:-d] * (1 - h[d:]))
        else:
            dd = -d
            raw1[k] = np.sum(h[:-dd] * (1 - h[dd:]))
            raw2[k] = np.sum(h[dd:] * (1 - h[:-dd]))
    _CORR_DIR = 1 if np.max(np.abs(raw1 - cact)) < np.max(np.abs(raw2 - cact)) else -1
    _CORR_DIR_SET = True
    return _CORR_DIR


def project_feasible(v, target):
    n = len(v)
    target = float(target)
    lo = -1.0 - v.max()
    hi = 1.0 - v.min()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(v + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(v + 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-11:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if len(idx):
                order = idx[np.argsort(x[idx])]
                rem = diff
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
                    rem -= inc
                    if rem <= 1e-13:
                        break
        else:
            diff = -diff
            idx = np.where(x > 0.0)[0]
            if len(idx):
                order = idx[np.argsort(-x[idx])]
                rem = diff
                for i in order:
                    dec = min(x[i], rem)
                    x[i] -= dec
                    rem -= dec
                    if rem <= 1e-13:
                        break
    return np.clip(x, 0.0, 1.0)


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, "full") * dx


def exact_C5(h):
    C = corr_scaled(h)
    return float(np.max(C))


def compute_grad(h, tau, dx):
    n = len(h)
    C = np.correlate(h, 1.0 - h, "full") * dx
    m = float(np.max(C))
    a = (C - m) / tau
    e = np.exp(a)
    s = float(e.sum())
    if s < 1e-300:
        alpha = np.zeros_like(C)
        alpha[int(np.argmax(C))] = 1.0
    else:
        alpha = e / s
    G = np.zeros(n)
    dirf = _CORR_DIR
    ctr = n - 1
    for k in range(2 * n - 1):
        ak = alpha[k]
        if ak < 1e-12:
            continue
        coeff = ak * dx
        d = dirf * (k - ctr)
        if d == 0:
            G += coeff * (1.0 - 2.0 * h)
        elif d > 0:
            G[d:] += coeff * (1.0 - h[:-d])
            G[:-d] -= coeff * h[d:]
        else:
            dd = -d
            G[:n - dd] += coeff * (1.0 - h[dd:])
            G[dd:] -= coeff * h[:-dd]
    obj = m + tau * np.log(max(s, 1e-300))
    return G, obj


def momentum_pgd(h, deadline, target, tau_init=0.03, tau_min=0.0005, max_iter=20000):
    n = len(h)
    dx = 2.0 / float(n)
    h = h.copy()
    c = exact_C5(h)
    best_h = h.copy()
    best_c = c
    tau = tau_init
    v = np.zeros(n)
    beta = 0.85
    base_step = 0.8
    it = 0
    stall = 0
    restart_count = 0

    while time.time() < deadline - 2 and it < max_iter:
        it += 1
        G, _ = compute_grad(h, tau, dx)
        G = G - G.mean()
        gnrm = np.linalg.norm(G)
        if gnrm < 1e-12:
            break
        G = G / gnrm * 0.5

        improved = False
        for alpha in (base_step, base_step * 0.5, base_step * 0.25, base_step * 0.1, base_step * 0.05, base_step * 0.01):
            v = beta * v - alpha * G
            h_try = project_feasible(h + v, target)
            c_try = exact_C5(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                c = c_try
                improved = True
                break
        if not improved:
            v = -base_step * 0.01 * G
            h_try = project_feasible(h + v, target)
            c_try = exact_C5(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                c = c_try
                improved = True

        if improved:
            if c < best_c:
                best_c = c
                best_h = h.copy()
            stall = 0
        else:
            stall += 1
            v = np.zeros(n)
            restart_count += 1
            if restart_count > 3:
                restart_count = 0
            base_step *= 0.5
            tau = max(tau * 0.7, tau_min)
            if stall > 40:
                break

        if it % 10 == 0:
            tau = max(tau * 0.97, tau_min)
        if it % 50 == 0:
            beta = min(beta + 0.005, 0.95)

    return best_h, best_c


def active_grad(h, deadline, target, n_active=200):
    n = len(h)
    dx = 2.0 / float(n)
    h = h.copy()
    c = exact_C5(h)
    best_h = h.copy()
    best_c = c
    dirf = _CORR_DIR
    ctr = n - 1
    it = 0
    stall = 0

    while time.time() < deadline - 2 and it < 5000:
        it += 1
        C = corr_scaled(h)
        maxC = float(np.max(C))
        thresh = maxC - max(0.01, 0.05 * maxC * 0.1)
        active = np.where(C >= thresh)[0]
        if len(active) == 0:
            active = np.argsort(C)[-n_active:]
        elif len(active) > n_active:
            active = active[np.argsort(C[active])][-n_active:]

        G = np.zeros(n)
        for k in active:
            coeff = dx / len(active)
            d = dirf * (int(k) - ctr)
            if d == 0:
                G += coeff * (1.0 - 2.0 * h)
            elif d > 0:
                G[d:] += coeff * (1.0 - h[:-d])
                G[:-d] -= coeff * h[d:]
            else:
                dd = -d
                G[:n - dd] += coeff * (1.0 - h[dd:])
                G[dd:] -= coeff * h[:-dd]
        G = G - G.mean()

        improved = False
        for alpha in (0.3, 0.15, 0.08, 0.04, 0.02, 0.01, 0.005):
            h_try = project_feasible(h - alpha * G, target)
            c_try = exact_C5(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                c = c_try
                best_h = h.copy()
                best_c = c
                improved = True
                break

        if not improved:
            stall += 1
            if stall > 15:
                break
        else:
            stall = 0

    return best_h, best_c


def swap_local_search(h, deadline, target, max_swaps=20000):
    n = len(h)
    h = h.copy()
    c = exact_C5(h)
    best_h = h.copy()
    best_c = c
    attempts = 0
    stall = 0

    while time.time() < deadline - 2 and attempts < max_swaps:
        attempts += 1
        ones = np.where(h > 0.3)[0]
        zeros = np.where(h < 0.7)[0]
        if len(ones) == 0 or len(zeros) == 0:
            break
        i = int(ones[np.random.randint(len(ones))])
        j = int(zeros[np.random.randint(len(zeros))])
        if i == j:
            continue
        delta = min(h[i], 1.0 - h[j], 0.5)
        if delta <= 1e-6:
            continue
        h[i] -= delta
        h[j] += delta
        h[np.where(h < 0)[0]] = 0.0
        h[np.where(h > 1)[0]] = 1.0
        h = project_feasible(h, target)
        c_try = exact_C5(h)
        if c_try < best_c - 1e-14:
            best_c = c_try
            best_h = h.copy()
            c = c_try
            stall = 0
        else:
            h[i] += delta
            h[j] -= delta
            h = project_feasible(h, target)
            stall += 1
            if stall > 500:
                break

    return best_h, best_c


def resample_to(h, N):
    n0 = len(h)
    target = N / 2.0
    if n0 == N:
        return project_feasible(h.copy(), target)
    x_old = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x_new = (np.arange(N) + 0.5) * (2.0 / float(N))
    y = np.interp(x_new, x_old, h)
    return project_feasible(y, target)


def fourier_init(N, seed_val):
    rng = np.random.default_rng(seed_val)
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    best = None
    bestc = np.inf
    for f in range(1, 7):
        for ph in np.linspace(0, 2 * np.pi, 6, endpoint=False):
            y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph)
            h = project_feasible(y, N / 2.0)
            c = exact_C5(h)
            if c < bestc:
                bestc = c
                best = h.copy()
    for _ in range(6):
        f = int(1 + rng.integers(0, 5))
        ph = float(rng.uniform(0, 2 * np.pi))
        y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph) + 0.08 * rng.normal(0, 1, N)
        h = project_feasible(y, N / 2.0)
        c = exact_C5(h)
        if c < bestc:
            bestc = c
            best = h.copy()
    return best, bestc


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    hard = t0 + total - 10.0

    rng = np.random.default_rng(int(seed) % (1 << 32))
    _calibrate_corr_direction()

    hinit = globals().get("initial_h_values", None)
    if hinit is not None:
        hinit = np.asarray(hinit, dtype=float).ravel()
        n0 = len(hinit)
        target0 = n0 / 2.0
        best_h = project_feasible(hinit.copy(), target0)
        best_c = exact_C5(best_h)
    else:
        n0 = 480
        target0 = n0 / 2.0
        best_h, best_c = fourier_init(n0, 42)

    # Multi-resolution strategy
    Ns = []
    for f in (1, 2):
        N = int(round(n0 * f))
        if 40 <= N <= 1200:
            Ns.append(N)
    n_half = max(120, n0 // 2)
    if n_half not in Ns:
        Ns.append(n_half)
    n_third = max(120, n0 // 3)
    if n_third not in Ns:
        Ns.append(n_third)
    Ns = sorted(set(Ns))

    # Sort by priority: prefer sizes close to n0 and n0//2
    Ns.sort(key=lambda N: -min(abs(N - n0), abs(N - n_half)))

    max_stages = 4 if total < 500 else 5
    if len(Ns) > max_stages:
        Ns = Ns[:max_stages]

    for idx, Nt in enumerate(Ns):
        if time.time() > hard - 30:
            break
        if best_c <= 0.3808:
            break

        remaining = hard - time.time()
        if remaining < 30:
            break

        if idx == 0:
            stage_time = min(remaining * 0.4, max(60, 0.35 * total))
        else:
            stage_time = min(remaining * 0.35, max(40, 0.20 * total))

        stage_dl = min(hard - 15, time.time() + stage_time)
        tgt = Nt / 2.0

        if Nt == n0:
            h_cand = best_h.copy()
        else:
            h_cand = resample_to(best_h, Nt)

        # Phase A: Momentum PGD
        h_a, c_a = momentum_pgd(h_cand, stage_dl * 0.5 + t0, tgt, tau_init=0.03)
        if c_a < exact_C5(h_cand):
            h_cand = h_a

        # Phase B: Active set gradient
        h_b, c_b = active_grad(h_cand, min(stage_dl, time.time() + (stage_dl - time.time()) * 0.5), tgt)
        if c_b < exact_C5(h_cand):
            h_cand = h_b

        # Phase C: Swap local search
        h_c, c_c = swap_local_search(h_cand, stage_dl, tgt, max_swaps=5000)
        if c_c < exact_C5(h_cand):
            h_cand = h_c

        # Phase D: Final momentum pass
        if time.time() < stage_dl - 5:
            h_d, c_d = momentum_pgd(h_cand, stage_dl, tgt, tau_init=0.01, tau_min=0.0003)
            if c_d < exact_C5(h_cand):
                h_cand = h_d

        # Evaluate at original resolution
        if Nt != n0:
            h_check = resample_to(h_cand, n0)
            c_check = exact_C5(h_check)
            if c_check < best_c:
                best_h = h_check
                best_c = c_check
        else:
            c_check = exact_C5(h_cand)
            if c_check < best_c:
                best_h = h_cand
                best_c = c_check

    # Final polishing at n0
    if time.time() < hard - 10:
        tgt0 = n0 / 2.0
        h_final, c_final = momentum_pgd(best_h, hard - 5, tgt0, tau_init=0.008, tau_min=0.0002)

    if not np.isfinite(best_c):
        best_h = project_feasible(np.full(n0, 0.5), n0 / 2.0)
        best_c = exact_C5(best_h)

    return best_h, float(best_c), int(len(best_h))
