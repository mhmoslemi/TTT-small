C5 = max_j ∫₀² h(x)(1-h(x+j)) dx

import numpy as np

def score_exact(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    return float(np.max(np.correlate(h, 1.0 - h, "full") * dx))

def make_feasible(x, target):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5)
    target = float(target)
    lo = -1.0 - float(np.max(x))
    hi = 1.0 - float(np.min(x))
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-13:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])]
                rem = r
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
                    rem -= inc
                    if rem <= 0:
                        break
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                order = idx[np.argsort(-x[idx])]
                rem = -r
                for i in order:
                    dec = min(x[i], rem)
                    x[i] -= dec
                    rem -= dec
                    if rem <= 0:
                        break
    return np.clip(x, 0.0, 1.0)

def antipodal_full(x):
    m = len(x)
    return np.concatenate([x, 1.0 - x[::-1]])

def antipodal_half(h):
    n = len(h)
    m = n // 2
    return h[:m].copy()

def compute_R_fft(h, dx):
    n = len(h)
    L = 1
    while L < 2 * n - 1:
        L *= 2
    F = np.fft.rfft(h, L)
    P = F * np.conj(F)
    r = np.fft.irfft(P, L)
    return r[:n] * dx

def smooth_min_R(R, tau):
    mR = float(R.min())
    if tau < 1e-15:
        return mR
    a = np.clip(-(R - mR) / tau, -500, 500)
    e = np.exp(a)
    s = float(e.sum())
    if s < 1e-300:
        return mR
    return mR - tau * float(np.log(s))

def get_alpha(R, tau):
    mR = float(R.min())
    if tau < 1e-15:
        act = R <= mR + 1e-10
        alpha = np.zeros(len(R))
        cnt = float(act.sum())
        if cnt > 0:
            alpha[act] = 1.0 / cnt
        else:
            alpha[int(np.argmin(R))] = 1.0
    else:
        a = np.clip(-(R - mR) / tau, -500, 500)
        e = np.exp(a)
        s = float(e.sum())
        if s < 1e-300:
            alpha = np.zeros(len(R))
            alpha[int(np.argmin(R))] = 1.0
        else:
            alpha = e / s
    return alpha

def grad_full_h(h, alpha, dx):
    """Gradient of -tau*log(sum(exp(-R_j/tau))) w.r.t. h (full space).
    This equals dx * sum_j alpha[j] * (h[m+j] + h[m-j]) for valid indices."""
    n = len(h)
    G = np.zeros(n)
    for j in range(n):
        aj = float(alpha[j])
        if aj < 1e-15:
            continue
        coeff = aj * dx
        if j == 0:
            G += 2.0 * coeff * h
        else:
            G[:n - j] += coeff * h[j:]
            G[j:] += coeff * h[:n - j]
    return G

def conj_obj(x, tau):
    h = antipodal_full(x)
    n = len(h)
    dx = 2.0 / n
    R = compute_R_fft(h, dx)
    return -smooth_min_R(R, tau)

def conj_grad(x, tau):
    h = antipodal_full(x)
    n = len(h)
    m = n // 2
    dx = 2.0 / n
    R = compute_R_fft(h, dx)
    alpha = get_alpha(R, tau)
    G_full = grad_full_h(h, alpha, dx)
    G_x = G_full[:m] - G_full[n - 1 - np.arange(m)][::-1]
    return G_x

    s = float(np.sum(h)) - 0.5 * n
    return base + lam * s * s

    alpha = get_alpha(R, tau)

    G += 2.0 * lam * s
    return G

def pair_displace_search(h, deadline, rng=None):
    """Move epsilon from one position to another (preserves sum)."""
    n = len(h)
    if rng is None:
        rng = np.random.default_rng(99999)

        j = int(rng.integers(0, n))

                h[j] += eps
