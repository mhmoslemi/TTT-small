        A = np.zeros((len(active_idx), n))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)

        n_lags = len(A)
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0

        A_ub = np.hstack([A, -np.full((n_lags, 1), n / 2.0)])
        b_ub = np.zeros(n_lags)

        lb = np.maximum(h - radius, 0.0)
        ub = np.minimum(h + radius, 1.0)
        bounds = list(zip(lb, ub)) + [(0.0, None)]

        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
        except Exception:
            continue

        if res is None or not res.success:
            stall += 1
            if stall > 15:
                radius *= 0.6
                if radius < 0.003:
                    break
            continue

        h_lp = res.x[:n]
        h_lp = np.clip(h_lp, 0.0, 1.0)

        improved = False
        for alpha in [1.0, 0.8, 0.6, 0.4, 0.2]:
            h_try = (1.0 - alpha) * h + alpha * h_lp
            h_try = project_sum_box(h_try, n / 2.0)
            c_try = evaluate_c5(h_try)
            if c_try < best_c - 1e-12:
                h = h_try
                best_h = h_try.copy()
                best_c = c_try
                improved = True
                stall = 0
                break
            elif c_try < evaluate_c5(h) + 1e-14:
                h = h_try
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 8 == 0:
                radius = max(radius * 0.6, 0.003)
            if radius <= 0.003 and stall > 25:
                break

    return best_h, best_c

def lse_objective(x, beta):
    v = corr_all(x)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 300)
    u = np.exp(expo)
    return m - np.log(u.sum()) / beta

def lse_gradient(x, beta):
    n = len(x)
    v = corr_all(x)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 300)
    u = np.exp(expo)
    Z = u.sum()
    w = u / Z
    g = np.zeros(n)
    for k in range(len(v)):
        if w[k] > 1e-12:
            lag = k - (n - 1)
            g += w[k] * grad_lag(x, lag)
    return g

def sum_eq_fun(x, n):
    return x.sum() - n / 2.0

def sum_eq_jac(x, n):
    return np.ones(n)

def method_slsqp_refine(h0, deadline):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [8, 16, 32, 64, 128, 256]
    bounds = [(0.0, 1.0)] * n

    for beta in betas:
        if time.time() > deadline:
            break
        x0 = best_h.copy()
        try:
            res = minimize(lse_objective, x0, jac=lse_gradient, args=(beta,),
                           method='SLSQP', bounds=bounds,
                           constraints=[{'type': 'eq', 'fun': sum_eq_fun, 'jac': sum_eq_jac, 'args': (n,)}],
                           options={'maxiter': 100, 'ftol': 1e-14})
        except Exception:
            continue
        if res is not None and res.x is not None:
            x_new = project_sum_box(res.x, n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new

    return best_h, best_c

def method_subgrad_bb(h0, deadline, max_iters=3000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    h_prev = h.copy()
    g_prev = None
    step = 0.02
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        v = corr_all(h)
        idx_max = np.argmax(v)
        lag = idx_max - (n - 1)
        g = grad_lag(h, lag)

        if g_prev is not None:
            s = h - h_prev
            y = g - g_prev
            sy = s @ y
            yy = y @ y
            if yy > 1e-14 and sy > 1e-14:
                step = min(max(sy / yy, 0.001), 0.05)

        h_prev = h.copy()
        g_prev = g.copy()

        h_new = h - step * g
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)

        if c_new < best_c - 1e-12:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            stall = 0
        elif c_new < evaluate_c5(h):
            h = h_new
            stall = 0
        else:
            h = h
            stall += 1
            step *= 0.9
            if step < 0.001:
                break
            if stall > 200:
                break

    return best_h, best_c

def method_perturb_restart(h0, deadline, n_perturb=5, scale=0.03):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)

    for i in range(n_perturb):
        if time.time() > deadline:
            break
        h_pert = h0 + np.random.randn(n) * scale
        h_pert = project_sum_box(h_pert, n / 2.0)
        c_pert = evaluate_c5(h_pert)
        dl = min(deadline, time.time() + max(10, (deadline - time.time()) / (n_perturb - i)))
        h_opt, c_opt = method_sca_active(h_pert, dl, max_iters=100, radius=0.2, n_active=40)
        if c_opt < best_c:
            best_h = h_opt
            best_c = c_opt

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0
    deadline = start_time + budget_s - safety

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def check_time():
        return time.time() < deadline

    # Phase 1: SCA all lags at initial resolution
    if check_time():
        dl1 = min(deadline, start_time + budget_s * 0.20)
        h1, c1 = method_sca_all_lags(best_h, dl1, max_iters=400, init_radius=0.4)
        if c1 < best_c:
            best_h = h1
            best_c = c1

    # Phase 2: SCA active set refinement at initial resolution
    if check_time():
        dl2 = min(deadline, start_time + budget_s * 0.30)
        h2, c2 = method_sca_active(best_h, dl2, max_iters=300, radius=0.2, n_active=50)
        if c2 < best_c:
            best_h = h2
            best_c = c2

    # Phase 3: Upsample and optimize at higher resolution
    n_cur = len(best_h)
    target_n = min(max(n_cur * 2, 150), 250)
    if check_time() and target_n > n_cur:
        h_up = upsample_h(best_h, target_n)
        c_up = evaluate_c5(h_up)
        if c_up < best_c:
            best_h = h_up
            best_c = c_up
        else:
            best_h = h_up  # still use higher res as starting point

        dl3 = min(deadline, start_time + budget_s * 0.50)
        h3, c3 = method_sca_all_lags(best_h, dl3, max_iters=300, init_radius=0.35)
        if c3 < best_c:
            best_h = h3
            best_c = c3

        if check_time():
            dl3b = min(deadline, start_time + budget_s * 0.60)
            h3b, c3b = method_sca_active(best_h, dl3b, max_iters=200, radius=0.15, n_active=60)
            if c3b < best_c:
                best_h = h3b
                best_c = c3b

    # Phase 4: Even higher resolution if time permits
    n_cur2 = len(best_h)
    target_n2 = min(n_cur2 + 80, 300)
    if check_time() and (deadline - time.time()) > 120 and target_n2 > n_cur2:
        h_up2 = upsample_h(best_h, target_n2)
        c_up2 = evaluate_c5(h_up2)
        working = h_up2 if c_up2 < best_c else h_up2
        if c_up2 < best_c:
            best_h = h_up2
            best_c = c_up2

        dl4 = min(deadline - 30, time.time() + max(30, (deadline - 30 - time.time())))
        h4, c4 = method_sca_all_lags(working, dl4, max_iters=200, init_radius=0.3)
        if c4 < best_c:
            best_h = h4
            best_c = c4

    # Phase 5: SLSQP smoothing refinement
    if check_time() and (deadline - time.time()) > 30:
        dl5 = min(deadline - 10, time.time() + 60)
        h5, c5 = method_slsqp_refine(best_h, dl5)
        if c5 < best_c:
            best_h = h5
            best_c = c5

    # Phase 6: Subgradient BB polish
    if check_time() and (deadline - time.time()) > 20:
        dl6 = min(deadline - 5, time.time() + min(80, (deadline - 5 - time.time())))
        h6, c6 = method_subgrad_bb(best_h, dl6, max_iters=2000)
        if c6 < best_c:
            best_h = h6
            best_c = c6

    # Phase 7: Perturbation restarts
    if check_time() and (deadline - time.time()) > 30:
        dl7 = deadline - 5
        h7, c7 = method_perturb_restart(best_h, dl7, n_perturb=4, scale=0.02)
        if c7 < best_c:
            best_h = h7
            best_c = c7

    # Final: one more SCA round at whatever resolution we have
    if check_time() and (deadline - time.time()) > 15:
        dl8 = deadline - 2
        h8, c8 = method_sca_all_lags(best_h, dl8, max_iters=100, init_radius=0.15)
        if c8 < best_c:
            best_h = h8
            best_c = c8

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
