def run_lp_step(h, rho, active_thresh):
    n = len(h)
    dx = 2.0 / n
    L = 2*n - 1
    center = n - 1
    
    # Compute current lags
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    
    # Select active lags
    active = np.where(c >= c_max - active_thresh)[0]
    
    # Build Jacobian for active lags
    # J[k, i] = dc[k]/dh[i]
    J = np.zeros((len(active), n))
    for idx, k in enumerate(active):
        s = k - center
        q0 = max(0, -s)
        q1 = min(n - 1, n - 1 - s)
        # For each q in [q0, q1]:
        # dc/dh[q] += (1 - h[q+s]) * dx
        # dc/dh[q+s] += -h[q] * dx
        for q in range(q0, q1 + 1):
            J[idx, q] += (1.0 - h[q + s]) * dx
            J[idx, q + s] += -h[q] * dx
    
    # LP: min t
    # s.t. J @ x - t <= -c[active]  (i.e., c[active] + J@x <= t)
    #      sum(x) = 0
    #      max(-rho, -h[i]) <= x[i] <= min(rho, 1-h[i])
    
    # Variables: [x_0, ..., x_{n-1}, t]
    # Objective: [0, ..., 0, 1]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ var <= b_ub
    # For each active lag k: J[k,:] @ x - 1*t <= -c[k]
    A_ub = np.zeros((len(active), n + 1))
    A_ub[:, :n] = J
    A_ub[:, n] = -1.0
    b_ub = -c[active]
    
    # Equality: sum(x) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    # Bounds
    lb = np.maximum(-rho, -h)
    ub = np.minimum(rho, 1.0 - h)
    bounds = list(zip(lb, ub))
    bounds.append((0.0, c_max + 2.0))  # t >= 0, t bounded above

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def build_jacobian(h, active_k, center):
    n = len(h)
    dx = 2.0 / n
    m = len(active_k)
    J = np.zeros((m, n))
    for idx, k in enumerate(active_k):
        s = k - center
        q0 = max(0, -s)
        q1 = min(n - 1, n - 1 - s)
        for q in range(q0, q1 + 1):
            J[idx, q] += (1.0 - h[q + s]) * dx
            J[idx, q + s] += -h[q] * dx
    return J

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -5.0, 5.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def build_jacobian(h, active_k, center):
    n = len(h)
    dx = 2.0 / n
    m = len(active_k)
    J = np.zeros((m, n))
    for idx in range(m):
        k = active_k[idx]
        s = k - center
        q0 = max(0, -s)
        q1 = min(n - 1, n - 1 - s)
        for q in range(q0, q1 + 1):
            J[idx, q] += (1.0 - h[q + s]) * dx
            J[idx, q + s] += -h[q] * dx
    return J
