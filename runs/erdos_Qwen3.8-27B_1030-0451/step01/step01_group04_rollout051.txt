    return float(np.max(corr) * dx), corr * dx

def compute_all_correlations(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

    h: current point
    c_vals: array of all correlation values (length 2N-1)
    tau: smoothing parameter
    step_size: step size for gradient

        w = np.zeros_like(c_vals)
        w[np.argmax(c_vals)] = 1.0

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def _smooth_obj(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -300.0, 0.0)
    w = np.exp(exponents)
    total = w.sum()
    return m + tau * np.log(total + 1e-30)

def _smooth_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -300.0, 0.0)
    w = np.exp(exponents)
    total = w.sum()
    w = w / (total + 1e-30)
    grad = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                grad[d:] += a * (1.0 - h[:n])
                grad[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                grad[:n] += a * (1.0 - h[dd:])
                grad[dd:] -= a * h[:n]
    return grad

def _sum_fun(x, target):
    return np.sum(x) - target

def _sum_jac(x, target):
    return np.ones_like(x)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = 0.5 * (lo + hi)
    h = np.clip(h + t, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(1.0 - h)
        else:
            idx = np.argsort(h)
        i = 0
        while abs(diff) > 1e-12 and i < len(idx):
            j = idx[i]
            if diff > 0:
                amt = min(diff, 1.0 - h[j])
            else:
                amt = max(diff, -h[j])
            h[j] += amt
            diff -= amt
            i += 1
    return h
