        # For positive lag d: Q(+d) = sum_{i=0}^{n-d-1} h[i+d]*(1-h[i])
        # d/dh[j] for j in [0, n-d-1]: -dx * h[j+d] (from 1-h[j] term)
        # d/dh[j] for j in [d, n-1]: dx * (1-h[j-d]) (from h[j] term)
        grad[:n_overlap] += wd_pos * dx * (-h[d:])
        grad[d:] += wd_pos * dx * (1.0 - h[:n_overlap])
        # For negative lag: Q(-d) = sum_{i=0}^{n-d-1} h[i]*(1-h[i+d])
        # d/dh[j] for j in [0, n-d-1]: dx * (1-h[j+d])
        # d/dh[j] for j in [d, n-1]: -dx * h[j-d]
        grad[:n_overlap] += wd_neg * dx * (1.0 - h[d:])
        grad[d:] += wd_neg * dx * (-h[:n_overlap])
    
    return w, grad

def smooth_optimize(h, max_time, rng, tau_start=0.1, tau_end=0.01):
    """SLSQP optimization on log-sum-exp surrogate with continuation."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = evaluate_exact(best_h)
    t_start = time.time()
    
    taus = np.geomspace(tau_start, tau_end, 8)
    
    for tau in taus:
        if time.time() - t_start > max_time:
            break
        
        def obj_and_grad(x):
            x = np.clip(x, 0, 1)
            corr = correlation_array(x)
            M = np.max(corr)
            u = (corr - M) / tau
            u = np.clip(u, -745, 745)
            exp_u = np.exp(u)
            Z = np.sum(exp_u)
            if Z < 1e-300:
                obj = M
                w = np.zeros(2 * n - 1)
                idx_max = np.where(corr == M)[0]
                w[idx_max] = 1.0 / len(idx_max)
            else:
                obj = M + tau * np.log(Z)
                w = exp_u / Z
            
            center = n - 1
            grad = np.zeros(n)
            dx = 2.0 / n
            grad += w[center] * dx * (1.0 - 2.0 * x)
            for d in range(1, n):
                wp = w[center + d]
                wn = w[center - d]
                no = n - d
                grad[:no] += wp * dx * (-x[d:])
                grad[d:] += wp * dx * (1.0 - x[:no])
                grad[:no] += wn * dx * (1.0 - x[d:])
                grad[d:] += wn * dx * (-x[:no])
            return obj, grad
        
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - target, 'jac': lambda x: np.ones(n)}
        ]
        bounds = [(0, 1)] * n
        
        x0 = best_h.copy()
        for attempt in range(3):
            if time.time() - t_start > max_time:
                break
            try:
                res = minimize(obj_and_grad, x0, jac=True, method='SLSQP',
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 200, 'ftol': 1e-10})
                if res.success and res.x is not None:
                    x_new = project_box_sum(res.x, target)
                    val = evaluate_exact(x_new)
                    if val < best_val - 1e-14:
                        best_val = val
                        best_h = x_new.copy()
                    x0 = x_new.copy()
                else:
                    break
            except Exception:
                break
    
    return best_h, best_val

def gradient_descent(h, max_time, rng):
    """Plain gradient descent with backtracking on the exact max (subgradient)."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_val = evaluate_exact(best_h)
    t_start = time.time()
    lr = 0.3
    
    while time.time() - t_start < max_time:
        corr = correlation_array(h)
        k_star = int(np.argmax(corr))
        lag = k_star - (n - 1)
        
        # Subgradient of the max
        grad = np.zeros(n)
        dx = 2.0 / n
        if lag == 0:
            grad = dx * (1.0 - 2.0 * h)
        elif lag > 0:
            no = n - lag
            grad[:no] += dx * (-h[lag:])
            grad[lag:] += dx * (1.0 - h[:no])
        else:
            d = -lag
            no = n - d
            grad[:no] += dx * (1.0 - h[d:])
            grad[d:] += dx * (-h[:no])
        
        # Average over top lags for smoother direction
        top_k = 5
        idx_top = np.argsort(corr)[-top_k:]
        grad_avg = np.zeros(n)
        for idx in idx_top:
            l = idx - (n - 1)
            g = np.zeros(n)
            if l == 0:
                g = dx * (1.0 - 2.0 * h)
            elif l > 0:
                no = n - l
                g[:no] += dx * (-h[l:])
                g[l:] += dx * (1.0 - h[:no])
            else:
                d = -l
                no = n - d
                g[:no] += dx * (1.0 - h[d:])
                g[d:] += dx * (-h[:no])
            grad_avg += g
        grad_avg /= top_k
        
        for alpha_mult in [1.0, 0.5, 0.25, 0.1]:
            alpha = lr * alpha_mult
            h_new = project_box_sum(h - alpha * grad_avg, target)
            val = evaluate_exact(h_new)
            if val < best_val - 1e-14:
                best_val = val
                best_h = h_new.copy()
                h = h_new.copy()
                lr = min(1.0, lr * 1.1)
                break
        else:
            lr = max(0.01, lr * 0.7)
    
    return best_h, best_val

def resize_h(h, new_n):
    """Resize h to new_n points using interpolation."""
    old_n = len(h)
    if old_n == new_n:
        return h.copy()
    old_x = np.linspace(0, 1, old_n + 1)[:-1] + 0.5 / old_n
    new_x = np.linspace(0, 1, new_n + 1)[:-1] + 0.5 / new_n
    h_interp = np.interp(new_x, old_x, h)
    return project_box_sum(h_interp, new_n / 2.0)

def get_initial(seed):
    """Get initial h values from parent or create default."""
    rng = np.random.default_rng(seed)
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).flatten()
        if h0.size > 0 and np.all(np.isfinite(h0)):
            n = len(h0)
            h0 = project_box_sum(h0, n / 2.0)
            return h0
    except Exception:
        pass
    # Default: 120 points, half 1s half 0s in a pattern
    n = 120
    h = np.zeros(n)
    h[::2] = 1.0
    return project_box_sum(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    def time_left():
        return deadline - time.time()
    
    h0 = get_initial(seed)
    n0 = len(h0)
    best_h = h0.copy()
    best_val = evaluate_exact(best_h)
    
    # Phase 1: Binary local search at native resolution
    if time_left() > 80:
        h_bin = to_binary(h0, n0)
        h_bin, v_bin = binary_local_search(h_bin, min(60, time_left() - 10), rng)
        if v_bin < best_val:
            best_val = v_bin
            best_h = h_bin.copy()
    
    # Phase 2: Smooth optimization from best
    if time_left() > 80:
        h_smooth, v_smooth = smooth_optimize(best_h, min(50, time_left() - 10), rng)
        if v_smooth < best_val:
            best_val = v_smooth
            best_h = h_smooth.copy()
    
    # Phase 3: Gradient descent
    if time_left() > 40:
        h_gd, v_gd = gradient_descent(best_h, min(30, time_left() - 10), rng)
        if v_gd < best_val:
            best_val = v_gd
            best_h = h_gd.copy()
    
    # Phase 4: Try different resolutions
    resolutions = [80, 100, 140, 160, 200]
    for n_try in resolutions:
        if time_left() < 40:
            break
        h_try = resize_h(best_h, n_try)
        # Binary search
        if time_left() > 50:
            h_b = to_binary(h_try, n_try)
            h_b, v_b = binary_local_search(h_b, min(25, time_left() - 15), rng)
            if v_b < best_val:
                best_val = v_b
                best_h = h_b.copy()
        # Smooth
        if time_left() > 30:
            h_s, v_s = smooth_optimize(h_try, min(20, time_left() - 10), rng)
            if v_s < best_val:
                best_val = v_s
                best_h = h_s.copy()
    
    # Phase 5: Sequence of from best smooth solution
    if time_left() > 40:
        h_seq, v_seq = smooth_optimize(best_h, min(30, time_left() - 10), rng, tau_start=0.05, tau_end=0.005)
        if v_seq < best_val:
            best_val = v_seq
            best_h = h_seq.copy()
    
    # Phase 6: More binary search restarts
    if time_left() > 60:
        for restart in range(4):
            if time_left() < 20:
                break
            if restart == 0:
                h_r = to_binary(best_h + 0.2 * rng.standard_normal(len(best_h)), len(best_h))
            elif restart == 1:
                h_r = to_binary(1.0 - best_h, len(best_h))
            elif restart == 2:
                h_r = to_binary(np.roll(best_h, int(rng.integers(1, len(best_h) // 2))), len(best_h))
            else:
                h_r = to_binary(best_h + 0.3 * rng.standard_normal(len(best_h)), len(best_h))
            h_r, v_r = binary_local_search(h_r, min(15, time_left() - 10), rng)
            if v_r < best_val:
                best_val = v_r
                best_h = h_r.copy()
    
    # Phase 7: Final smooth polish
    if time_left() > 20:
        h_final, v_final = smooth_optimize(best_h, min(15, time_left() - 5), rng, tau_start=0.03, tau_end=0.005)
        if v_final < best_val:
            best_val = v_final
            best_h = h_final.copy()
    
    # Final: ensure feasibility
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)
    
    return best_h, float(best_val), len(best_h)
