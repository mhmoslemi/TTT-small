def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        k = K - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] += w * (-h[k:])
        else:
            m = -k
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n - m])
    # Make zero-sum
    g -= np.mean(g)
    return g

            h = np.clip(h, 0.0, 1.0)
    return h

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        k = K - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] += w * (-h[k:])
        else:
            m = -k
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n - m])
    g -= np.mean(g)
    return g

def compute_lin_quad(h, d):
    """Compute Lin and Quad coefficients for line h + t*d."""
    n = len(h)
    L = 2 * n - 1
    Lin = np.zeros(L)
    Quad = np.zeros(L)
    for K in range(L):
        s = K - (n - 1)
        if s >= 0:
            idx = np.arange(n - s)
        else:
            idx = np.arange(-s, n)
        i_idx = idx
        j_idx = idx + s
        Lin[K] = np.sum(d[i_idx] * (1.0 - h[j_idx]) - h[i_idx] * d[j_idx])
        Quad[K] = -np.sum(d[i_idx] * d[j_idx])
    return Lin, Quad

def exact_line_search(h, d):
    """Find best step along direction d (zero-sum) from h.
    Returns (best_h, best_score)."""
    n = len(h)
    dx = 2.0 / n
    cur_score = exact_score(h)
    
    # Check if d is effectively zero
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-14:
        return h.copy(), cur_score
    
    # Normalize d to unit norm for consistency (doesn't change the line, just parameterization)
    # Actually, don't normalize - the feasible interval depends on the scale
    
    # Compute Lin, Quad
    Lin, Quad = compute_lin_quad(h, d)
    R0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Feasible interval
    t_lo = -np.inf
    t_hi = np.inf
    for i in range(n):
        di = d[i]
        if abs(di) < 1e-14:
            continue
        if di > 0:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        else:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)
    
    # Cap to reasonable range
    t_lo = max(t_lo, -10.0)
    t_hi = min(t_hi, 10.0)
    if t_hi <= t_lo + 1e-10:
        return h.copy(), cur_score
    
    safety = 1e-10
    t_lo += safety
    t_hi -= safety
    if t_hi <= t_lo:
        return h.copy(), cur_score
    
    # Find active lags
    R0_max = np.max(R0)
    active_thresh = R0_max - 0.05 / dx  # in unscaled units
    active = np.where(R0 >= active_thresh)[0]
    
    # Generate candidate t values
    candidates = [0.0, t_lo, t_hi]
    
    # Vertices of active quadratics
    for K in active:
        if abs(Quad[K]) > 1e-14:
            tv = -Lin[K] / (2.0 * Quad[K])
            if t_lo < tv < t_hi:
                candidates.append(tv)
    
    # Pairwise intersections of top active lags
    n_active = min(len(active), 20)  # Limit to top 20
    active_top = active[:n_active]
    for i in range(len(active_top)):
        for j in range(i + 1, len(active_top)):
            K1, K2 = active_top[i], active_top[j]
            a = Quad[K1] - Quad[K2]
            b = Lin[K1] - Lin[K2]
            c = R0[K1] - R0[K2]
            if abs(a) < 1e-14:
                if abs(b) > 1e-14:
                    tv = -c / b
                    if t_lo < tv < t_hi:
                        candidates.append(tv)
            else:
                disc = b * b - 4.0 * a * c
                if disc >= 0:
                    sq = np.sqrt(disc)
                    tv1 = (-b + sq) / (2.0 * a)
                    tv2 = (-b - sq) / (2.0 * a)
                    if t_lo < tv1 < t_hi:
                        candidates.append(tv1)
                    if t_lo < tv2 < t_hi:
                        candidates.append(tv2)
    
    # Uniform samples
    candidates.extend(np.linspace(t_lo, t_hi, 80).tolist())
    
    # Evaluate
    best_s = cur_score
    best_t = 0.0
    
    for t in candidates:
        if t < t_lo - 1e-10 or t > t_hi + 1e-10:
            continue
        R_t = R0 + t * Lin + t * t * Quad
        s = float(np.max(R_t) * dx)
        if s < best_s - 1e-14:
            best_s = s
            best_t = t
    
    if best_t == 0.0:
        return h.copy(), cur_score
    
    h_new = np.clip(h + best_t * d, 0.0, 1.0)
    actual_s = exact_score(h_new)
    
    return h_new, actual_s

def compute_lin_quad(h, d):
    n = len(h)
    L = 2 * n - 1
    Lin = np.zeros(L)
    Quad = np.zeros(L)
    for K in range(L):
        s = K - (n - 1)  # shift
        if s >= 0:
            # pairs: (h[i+s], (1-h)[i]) for i = 0, ..., n-1-s
            i = np.arange(n - s)
            ia = i + s  # index into h for first factor
            ib = i      # index into (1-h) for second factor
        else:
            m = -s
            # pairs: (h[i], (1-h)[i+m]) for i = 0, ..., n-1-m
            i = np.arange(n - m)
            ia = i      # index into h
            ib = i + m  # index into (1-h)
        
        h_ia = h[ia]
        h_ib = h[ib]
        d_ia = d[ia]
        d_ib = d[ib]
        
        Lin[K] = np.sum(d_ia * (1.0 - h_ib) - h_ia * d_ib)
        Quad[K] = -np.sum(d_ia * d_ib)
    
    return Lin, Quad
```

            t_hi = min(t_hi, (1.0 - h[i]) / di)
        else:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)
    
    t_lo = max(t_lo, -5.0 / d_norm)
    t_hi = min(t_hi, 5.0 / d_norm)
    if t_hi <= t_lo + 1e-10:
        return h.copy(), cur_score
    
    safety = 1e-10 * max(1.0, t_hi - t_lo)
    t_lo += safety
    t_hi -= safety
    if t_hi <= t_lo:
        return h.copy(), cur_score
    
    # Add strategic candidate points
    t_uniform = np.linspace(t_lo, t_hi, n_samples)
    
    # Add vertices of quadratics
    extra_pts = []
    for K in range(len(R0)):
        if abs(Quad[K]) > 1e-14:
            tv = -Lin[K] / (2.0 * Quad[K])
            if t_lo < tv < t_hi:
                extra_pts.append(tv)
    
    # Add pairwise intersections of top active lags
    R0_max = np.max(R0)
    active = np.where(R0 >= R0_max - 0.05/dx)[0][:15]
    for i in range(len(active)):
        for j in range(i+1, len(active)):
            K1, K2 = active[i], active[j]
            a = Quad[K1] - Quad[K2]
            b = Lin[K1] - Lin[K2]
            c = R0[K1] - R0[K2]
            if abs(a) < 1e-14:
                if abs(b) > 1e-14:
                    tv = -c / b
                    if t_lo < tv < t_hi:
                        extra_pts.append(tv)
            else:
                disc = b*b - 4*a*c
                if disc >= 0:
                    sq = np.sqrt(disc)
                    tv1 = (-b + sq) / (2.0 * a)
                    tv2 = (-b - sq) / (2.0 * a)
                    if t_lo < tv1 < t_hi:
                        extra_pts.append(tv1)
                    if t_lo < tv2 < t_hi:
                        extra_pts.append(tv2)
    
    all_t = np.unique(np.concatenate([t_uniform, np.array(extra_pts), np.array([0.0])]))
    all_t = all_t[(all_t >= t_lo) & (all_t <= t_hi)]
    
    # Vectorized evaluation
    t_col = all_t[:, np.newaxis]  # (n_t, 1)
    R_all = R0[np.newaxis, :] + t_col * Lin[np.newaxis, :] + t_col**2 * Quad[np.newaxis, :]
    scores = np.max(R_all, axis=1) * dx
    
    best_idx = np.argmin(scores)
    best_s = scores[best_idx]
    best_t = all_t[best_idx]
    
    if best_s >= cur_score - 1e-14:
        return h.copy(), cur_score
    
    h_new = np.clip(h + best_t * d, 0.0, 1.0)
    actual_s = exact_score(h_new)
    return h_new, actual_s
```

This should be very fast. The vectorized evaluation creates a (n_t, L) array and does element-wise operations, which numpy handles efficiently.

import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = np.sum(interior)
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw

    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        k = K - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] += w * (-h[k:])
        else:
            m = -k
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n - m])
    g -= np.mean(g)
    return g


def compute_lin_quad(h, d):
    n = len(h)
    L = 2 * n - 1
    Lin = np.zeros(L)
    Quad = np.zeros(L)
    for K in range(L):
        s = K - (n - 1)
        if s >= 0:
            i = np.arange(n - s)
            ia = i + s
            ib = i
        else:
            m = -s
            i = np.arange(n - m)
            ia = i
            ib = i + m
        Lin[K] = np.sum(d[ia] * (1.0 - h[ib]) - h[ia] * d[ib])
        Quad[K] = -np.sum(d[ia] * d[ib])
    return Lin, Quad


def line_search(h, d, n_samples=150):
    n = len(h)
    dx = 2.0 / n
    cur_score = exact_score(h)
    d_norm = np.linalg.norm(d)
    if d_norm < 1e-14:
        return h.copy(), cur_score

    R0 = np.correlate(h, 1.0 - h, mode='full')
    Lin, Quad = compute_lin_quad(h, d)

    t_lo = -np.inf
    t_hi = np.inf
    for i in range(n):
        di = d[i]
        if abs(di) < 1e-14:
            continue
        if di > 0:
            t_lo = max(t_lo, -h[i] / di)
            t_hi = min(t_hi, (1.0 - h[i]) / di)
        else:
            t_lo = max(t_lo, (1.0 - h[i]) / di)
            t_hi = min(t_hi, -h[i] / di)

    t_lo = max(t_lo, -5.0 / d_norm)
    t_hi = min(t_hi, 5.0 / d_norm)
    if t_hi <= t_lo + 1e-10:
        return h.copy(), cur_score

    safety = 1e-10 * max(1.0, t_hi - t_lo)
    t_lo += safety
    t_hi -= safety
    if t_hi <= t_lo:
        return h.copy(), cur_score

    t_uniform = np.linspace(t_lo, t_hi, n_samples)
    extra_pts = [0.0]

    for K in range(len(R0)):
        if abs(Quad[K]) > 1e-14:
            tv = -Lin[K] / (2.0 * Quad[K])
            if t_lo < tv < t_hi:
                extra_pts.append(tv)

    R0_max = np.max(R0)
    active = np.where(R0 >= R0_max - 0.05 / dx)[0]
    active = active[:min(len(active), 12)]
    for i in range(len(active)):
        for j in range(i + 1, len(active)):
            K1, K2 = active[i], active[j]
            a = Quad[K1] - Quad[K2]
            b = Lin[K1] - Lin[K2]
            c = R0[K1] - R0[K2]
            if abs(a) < 1e-14:
                if abs(b) > 1e-14:
                    tv = -c / b
                    if t_lo < tv < t_hi:
                        extra_pts.append(tv)
            else:
                disc = b * b - 4.0 * a * c
                if disc >= 0:
                    sq = np.sqrt(disc)
                    tv1 = (-b + sq) / (2.0 * a)
                    tv2 = (-b - sq) / (2.0 * a)
                    if t_lo < tv1 < t_hi:
                        extra_pts.append(tv1)
                    if t_lo < tv2 < t_hi:
                        extra_pts.append(tv2)

    all_t = np.unique(np.concatenate([t_uniform, np.array(extra_pts)]))
    all_t = all_t[(all_t >= t_lo) & (all_t <= t_hi)]

    t_col = all_t[:, np.newaxis]
    R_all = R0[np.newaxis, :] + t_col * Lin[np.newaxis, :] + t_col ** 2 * Quad[np.newaxis, :]
    scores = np.max(R_all, axis=1) * dx

    best_idx = np.argmin(scores)
    best_s = scores[best_idx]
    best_t = all_t[best_idx]

    if best_s >= cur_score - 1e-14:
        return h.copy(), cur_score

    h_new = np.clip(h + best_t * d, 0.0, 1.0)
    actual_s = exact_score(h_new)
    return h_new, actual_s


def generate_fourier_dir(n, k, phase, rng):
    x = (np.arange(n) + 0.5) / n * 2.0
    if rng.random() < 0.5:
        d = np.sin(2.0 * np.pi * k * x / 2.0 + phase)
    else:
        d = np.cos(2.0 * np.pi * k * x / 2.0 + phase)
    d -= np.mean(d)
    return d


def generate_block_dir(n, rng):
    a = int(rng.integers(0, n // 2))
    b = int(rng.integers(n // 2, n))
    la = int(rng.integers(2, max(3, n // 4)))
    lb = int(rng.integers(2, max(3, n // 4)))
    a = a % (n - la)
    b = b % (n - lb)
    d = np.zeros(n)
    d[a:a + la] = 1.0 / la
    d[b:b + lb] = -1.0 / lb
    if a + la > b or b + lb > a:
        pass
    d -= np.mean(d)
    return d


def make_fourier_initial(n, n_harmonics, rng):
    x = (np.arange(n) + 0.5) / n * 2.0
    h = np.full(n, 0.5)
    for k in range(1, n_harmonics + 1):
        amp = rng.uniform(0.05, 0.3) / min(k, 2)
        phase = rng.uniform(0, 2 * np.pi)
        if rng.random() < 0.5:
            h += amp * np.sin(2.0 * np.pi * k * x / 2.0 + phase)
        else:
            h += amp * np.cos(2.0 * np.pi * k * x / 2.0 + phase)
    return project_feasible(h, n)


def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)


class SlObj:
    def __init__(self, tau):
        self.tau = tau

    def __call__(self, h):
        n = len(h)
        dx = 2.0 / n
        raw = np.correlate(h, 1.0 - h, mode='full')
        c = raw * dx
        c_max = np.max(c)
        exponents = np.clip((c - c_max) / self.tau, -500, 500)
        return c_max + self.tau * np.log(np.sum(np.exp(exponents)))


class SlGrad:
    def __init__(self, tau):
        self.tau = tau

    def __call__(self, h):
        return smooth_grad(h, self.tau)


class SumCon:
    def __init__(self, n):
        self.n = n

    def fun(self, x):
        return np.sum(x) - self.n / 2.0

    def jac(self, x):
        return np.ones(self.n)


def run_slsqp(h0, tau, n, maxiter, deadline):
    sc = SlObj(tau)
    sj = SlGrad(tau)
    conc = SumCon(n)
    cons = [{'type': 'eq', 'fun': conc.fun, 'jac': conc.jac}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            fun=sc, x0=h0, jac=sj, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None
