import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_correlations(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr


def smooth_objective(c, tau):
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    max_exp = np.max(exponents)
    log_sum = max_exp + np.log(np.exp(exponents - max_exp).sum())
    return m + tau * log_sum


def smooth_gradient(h, c, tau):
    N = len(h)
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    max_exp = np.max(exponents)
    weights = np.exp(exponents - max_exp)
    total = weights.sum()
    w = weights / total

    grad = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r]
        if a < 1e-14:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                grad[d:] += a * (1.0 - h[:n])
                grad[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                grad[:n] += a * (1.0 - h[dd:])
                grad[dd:] -= a * h[:n]
    return grad


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h


def projected_gradient_optimize(h, target_sum, tau, max_iter, init_alpha, deadline):
    h_work = h.copy()
    alpha = init_alpha
    prev_obj = smooth_objective(compute_correlations(h_work), tau)

    for it in range(max_iter):
        if time.time() > deadline:
            break
        c = compute_correlations(h_work)
        grad = smooth_gradient(h_work, c, tau)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-10:
            break

        # Backtracking line search (Armijo)
        accepted = False
        a = alpha
        for _ in range(40):
            h_trial = project_feasible(h_work - a * grad, target_sum)
            c_trial = compute_correlations(h_trial)
            obj_trial = smooth_objective(c_trial, tau)
            if obj_trial <= prev_obj - 1e-4 * a * gnorm:
                accepted = True
                break
            a *= 0.5
        if not accepted:
            alpha = max(a, 1e-6)
            break
        h_work = h_trial
        prev_obj = obj_trial
        alpha = min(a * 1.5, 1.0)

    return h_work, compute_c5(h_work)


def slsqp_obj(h, tau):
    return smooth_objective(compute_correlations(h), tau)


def slsqp_grad(h, tau):
    return smooth_gradient(h, compute_correlations(h), tau)


def slsqp_sum(h, target):
    return np.sum(h) - target


def slsqp_sum_jac(h, target):
    return np.ones_like(h)


def upsample(h_old, N_new):
    N_old = len(h_old)
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_feasible(h_new, N_new / 2.0)
