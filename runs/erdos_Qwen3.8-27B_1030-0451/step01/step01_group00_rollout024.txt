import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def pair_search(h, i, j):
    n = len(h)
    hi = h[i]
    hj = h[j]
    a_lo = max(-hi, hj - 1.0)
    a_hi = min(1.0 - hi, hj)
    if a_hi - a_lo < 1e-10:
        return h, compute_c5(h)
    
    n_grid = 9
    alphas = np.linspace(a_lo, a_hi, n_grid)
    best_c = compute_c5(h)
    best_a = 0.0
    for a in alphas:
        h[i] = hi + a
        h[j] = hj - a
        c = compute_c5(h)
        if c < best_c:
            best_c = c
            best_a = a
    h[i] = hi
    h[j] = hj
    
    if abs(best_a) < 1e-12:
        return h, best_c
    
    idx = int(np.argmin(np.abs(alphas - best_a)))
    lo_a = alphas[max(0, idx - 1)]
    hi_a = alphas[min(n_grid - 1, idx + 1)]
    for _ in range(12):
        m1 = lo_a + 0.382 * (hi_a - lo_a)
        m2 = hi_a - 0.382 * (hi_a - lo_a)
        h[i] = hi + m1
        h[j] = hj - m1
        c1 = compute_c5(h)
        h[i] = hi + m2
        h[j] = hj - m2
        c2 = compute_c5(h)
        if c1 <= c2:
            hi_a = m2
        else:
            lo_a = m1
    h[i] = hi
    h[j] = hj
    
    best_a = (lo_a + hi_a) * 0.5
    h[i] = hi + best_a
    h[j] = hj - best_a
    c_new = compute_c5(h)
    if c_new >= best_c:
        h[i] = hi
        h[j] = hj
        return h, best_c
    return h, c_new

def composite_grad(h, c_vals, n):
    max_c = float(np.max(c_vals))
    if max_c < 1e-12:
        return np.zeros(n)
    threshold = max_c - 0.12 * max_c
    active_mask = c_vals >= threshold
    w = np.zeros_like(c_vals)
    w[active_mask] = c_vals[active_mask]
    w_sum = w.sum()
    if w_sum < 1e-30:
        return np.zeros(n)
    w /= w_sum
    center = n - 1
    g = np.zeros(n)
    for k in np.where(w > 1e-14)[0]:
        wk = w[k]
        lag = k - center
        L = abs(lag)
        if lag >= 0:
            for i in range(n):
                t1 = 1.0 - h[min(i + L, n - 1)] if i + L < n else 0.0
                t2 = h[i - L] if i - L >= 0 else 0.0
                g[i] += wk * (t1 - t2)
        else:
            for i in range(n):
                t1 = 1.0 - h[i - L] if i - L >= 0 else 0.0
                t2 = h[i + L] if i + L < n else 0.0
                g[i] += wk * (t1 - t2)
    return g

        n_top = min(200, n * n - n)

def resample_h(h_old, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)
