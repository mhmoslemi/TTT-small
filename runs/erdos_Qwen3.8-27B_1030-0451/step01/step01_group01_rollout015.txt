            active_idx = np.argsort(v)[-10:]
        
        # Build A, b for LP: A @ x <= b
        # For each active lag k: grad_k @ h_new <= v[k] - (v[k] - vmax) * 0.1
        # i.e., we want grad_k @ h_new <= v[k] + small_slack
        A_rows = []
        b_rows = []
        for k in active_idx:
            lag = k - (n - 1)
            gr = grad_lag(h, lag)
            A_rows.append(gr)
            # Linearization: v[k] + gr @ (x - h) <= v[k] + slack
            # i.e., gr @ x <= v[k] + slack - gr @ h + gr @ h = v[k] + slack... 
            # Actually: v[k] + gr @ (x - h) <= v[k] means gr @ x <= gr @ h
            # But that's the tangent plane. We want: linearized corr <= target
            # Linearized corr at x = v[k] + gr @ (x - h)
            # We want: v[k] + gr @ (x - h) <= current_max_target
            # Let target = max(vmax * 0.999, v[k])
            A_rows[-1] = gr
            b_rows.append(gr @ h + 0.0)  # tangent: gr@(x-h) <= 0 means gr@x <= gr@h
            # Actually we want to REDUCE, so: gr @ x <= gr @ h - epsilon
            # But that's too tight. Let's use: gr @ x <= gr @ h (just don't increase)
        
        A = np.array(A_rows) if A_rows else np.zeros((0, n))
        b = np.array(b_rows) if b_rows else np.zeros(0)
        
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        
        A_ub = np.hstack([A, -np.ones((len(A_rows), 1))]) if len(A_rows) > 0 else np.zeros((0, n_var))
        b_ub = b.copy()
        
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-2.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [vmax + 0.1]])
        
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n_half])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        
        if res is not None and res.success and res.x is not None:
            x_new = np.clip(res.x[:n], 0.0, 1.0)
            x_new = project_sum_box(x_new, n_half)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                # Add cuts from the new point
                v_new = corr_all(h)
                for k in active_idx:
                    lag = k - (n - 1)
                    gr = grad_lag(h, lag)
                    A_rows.append(gr)
                    b_rows.append(gr @ h)
                    cut_lag.append(lag)
                if it % 3 == 0:
                    radius = min(radius * 1.08, 0.2)
            else:
                radius = max(radius * 0.92, 0.002)
        else:
            radius = max(radius * 0.9, 0.001)
            if radius <= 0.001:
                break
        
        # Prune cuts to keep LP manageable
        if len(cut_lag) > 200:
            cut_lag = cut_lag[-100:]
            # (we just keep the most recent ones)
        
        if it % 20 == 0 and it > 0:
            # Reset cuts periodically to avoid accumulation issues
            cuts_i = []
            cuts_b = []
            cut_lag = []
    
    return best_h, best_c

def method_tau_slsqp(h0, deadline, max_tau_iters=20):
    n = len(h0)
    n_half = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    tau = best_c * 1.001
    bounds = [(0.0, 1.0)] * n
    cons = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac, 'args': (n_half,)}]
    
    for tau_it in range(max_tau_iters):
        if time.time() > deadline:
            break
        
        x0 = best_h.copy()
        try:
            res = minimize(_tau_obj, x0, jac=_tau_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           args=(tau,),
                           options={'maxiter': 200, 'ftol': 1e-15})
        except Exception:
            break
        
        if res is not None and res.x is not None:
            x_new = project_sum_box(res.x, n_half)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
            
            v = corr_all(x_new)
            if v.max() <= tau + 1e-9:
                tau = max(c_new * 0.998, 0.3)
            else:
                if c_new < tau:
                    tau = c_new * 0.997
                else:
                    tau = c_new * 1.001
        else:
            break
    
    return best_h, best_c

def method_sim_anneal(h0, deadline, max_iters=50000):
    n = len(h0)
    n_half = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    T = 0.001
    T_min = 1e-7
    cool = 0.9997
    
    rng = np.random.RandomState(12345)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if it % 50 == 0:
            T *= cool
            if T < T_min:
                T = T_min
        
        # Pick random pair
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        # Decide direction: move from higher to lower (or random)
        if rng.random() < 0.7:
            if h[i] >= h[j]:
                src, dst = i, j
            else:
                src, dst = j, i
        else:
            src, dst = i, j
        
        eps_pool = [0.005, 0.01, 0.02, 0.05, 0.1]
        eps = eps_pool[rng.randint(len(eps_pool))]
        if h[src] < eps:
            continue
        
        h_tr = h.copy()
        h_tr[src] -= eps
        h_tr[dst] += eps
        if h_tr[dst] > 1.0:
            h_tr[dst] = 1.0
            h_tr[src] -= (1.0 - h[dst])
            if h_tr[src] < 0:
                continue
        
        c_tr = evaluate_c5(h_tr)
        delta = c_tr - best_c
        if delta < 0:
            h = h_tr
            best_h = h_tr.copy()
            best_c = c_tr
        elif rng.random() < np.exp(-delta / max(T, 1e-10)):
            h = h_tr
    
    return best_h, best_c

def method_subgrad_averaged(h0, deadline, max_iters=3000):
    n = len(h0)
    n_half = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    
    h_avg = h0.copy()
    h = h0.copy()
    step0 = 0.02
    beta = 30.0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        t = it + 1
        step = step0 / np.sqrt(t)
        
        v = corr_all(h)
        m = v.max()
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 5e-5:
                lag = k - (n - 1)
                g += w[k] * grad_lag(h, lag)
        
        h_new = h - step * g
        h_new = project_sum_box(h_new, n_half)
        
        h_avg = (1.0 - 1.0/t) * h_avg + (1.0/t) * h_new
        h_avg = project_sum_box(h_avg, n_half)
        
        c_new = evaluate_c5(h_new)
        c_avg = evaluate_c5(h_avg)
        
        if c_new < best_c:
            best_h = h_new.copy()
            best_c = c_new
        if c_avg < best_c:
            best_h = h_avg.copy()
            best_c = c_avg
        
        h = h_new
    
    return best_h, best_c

def method_grad_descent_refined(h0, deadline, max_iters=10000):
    n = len(h0)
    n_half = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.01
    beta = 50.0
    stall = 0
    best_step = step
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        m = v.max()
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 1e-6:
                lag = k - (n - 1)
                g += w[k] * grad_lag(h, lag)
        
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        
        h_new = h - step * g
        h_new = project_sum_box(h_new, n_half)
        
        c_new = evaluate_c5(h_new)
        if c_new < best_c - 1e-13:
            best_h = h_new.copy()
            best_c = c_new
            h = h_new
            stall = 0
            if c_new < best_c:
                best_step = step
                step = min(step * 1.03, 0.05)
        else:
            stall += 1
            if c_new < evaluate_c5(h) - 1e-14:
                h = h_new
            step *= 0.95
            if step < 1e-7:
                break
        if stall > 500:
            # Restart with larger step
            step = best_step
            stall = 0
            h = best_h.copy()
            if step < 0.001:
                break
    
    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 1.0, n_old + 1)
    x_new = np.linspace(0.0, 1.0, n_new + 1)
    h_ext = np.concatenate([h0, h0[:1]])
    h_new = np.interp(x_new, x_old, h_ext)[:-1]
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def method_pairwise_gradient(h0, deadline, max_rounds=100):
    """For each coordinate, find best direction using gradient info."""
    n = len(h0)
    n_half = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    beta = 40.0
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        
        v = corr_all(h)
        m = v.max()
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 1e-5:
                lag = k - (n - 1)
                g += w[k] * grad_lag(h, lag)
        
        improved = False
        # Process coordinates in order of |gradient|
        order = np.argsort(-np.abs(g))
        
        for idx in order:
            if time.time() > deadline:
                break
            if abs(g[idx]) < 1e-10:
                continue
            # Try to adjust h[idx] and compensate with another coordinate
            direction = -1.0 if g[idx] > 0 else 1.0
            # Find best partner to compensate
            for try_eps in [0.05, 0.02, 0.01, 0.005]:
                h_tr = h.copy()
                h_tr[idx] += direction * try_eps
                if h_tr[idx] < 0 or h_tr[idx] > 1:
                    continue
                # Compensate on the coordinate with opposite gradient and most slack
                comp_budget = direction * try_eps
                candidates = np.argsort(np.sign(g) * (-g))
                for c_idx in candidates[:5]:
                    if c_idx == idx:
                        continue
                    h_tr[c_idx] -= comp_budget
                    if h_tr[c_idx] < 0 or h_tr[c_idx] > 1:
                        continue
                    c_tr = evaluate_c5(h_tr)
                    if c_tr < best_c - 1e-13:
                        best_h = h_tr.copy()
                        best_c = c_tr
                        h = h_tr.copy()
                        improved = True
                        break
        if not improved:
            break
    
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 3.0
    deadline = start_time + budget_s - safety

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)
        h_init = project_sum_box(h_init, 42.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: Gradient descent refined (fast, many small steps)
    if time_left() > 60:
        dl1 = start_time + budget_s * 0.12
        try:
            h1, c1 = method_grad_descent_refined(best_h, dl1, max_iters=8000)
            if c1 < best_c:
                best_h, best_c = h1, c1
        except Exception:
            pass

    # Phase 2: Cutting-plane LP
    if time_left() > 60:
        dl2 = start_time + budget_s * 0.30
        try:
            h2, c2 = method_cutting_plane_lp(best_h, dl2, max_iters=500)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # Phase 3: Tau-penalty SLSQP
    if time_left() > 60:
        dl3 = start_time + budget_s * 0.48
        try:
            h3, c3 = method_tau_slsqp(best_h, dl3, max_tau_iters=15)
            if c3 < best_c:
                best_h, best_c = h3, c3
        except Exception:
            pass

    # Phase 4: More LP from improved point
    if time_left() > 40:
        dl4 = start_time + budget_s * 0.60
        try:
            h4, c4 = method_cutting_plane_lp(best_h, dl4, max_iters=300)
            if c4 < best_c:
                best_h, best_c = h4, c4
        except Exception:
            pass

    # Phase 5: Pairwise gradient refinement
    if time_left() > 30:
        dl5 = start_time + budget_s * 0.70
        try:
            h5, c5v = method_pairwise_gradient(best_h, dl5, max_rounds=80)
            if c5v < best_c:
                best_h, best_c = h5, c5v
        except Exception:
            pass

    # Phase 6: Resolution transfer + optimize
    if time_left() > 80:
        n_cur = len(best_h)
        for scale in [1.33, 1.5, 2.0]:
            if time_left() < 60:
                break
            n_new = int(round(n_cur * scale))
            if n_new > 250 or n_new <= n_cur:
                continue
            try:
                h_tr = transfer_resolution(best_h, n_new)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c:
                    best_h, best_c = h_tr.copy(), c_tr
                # Optimize at new resolution
                sub_dl = min(deadline - 5, time.time() + min(120, time_left() * 0.4))
                h_a, c_a = method_grad_descent_refined(best_h, sub_dl, max_iters=3000)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a
                sub_dl2 = min(deadline - 5, time.time() + min(60, time_left() * 0.3))
                h_b, c_b = method_cutting_plane_lp(best_h, sub_dl2, max_iters=100)
                if c_b < best_c:
                    best_h, best_c = h_b, c_b
            except Exception:
                pass

    # Phase 7: Final gradient polish
    if time_left() > 15:
        try:
            h7, c7 = method_grad_descent_refined(best_h, deadline - 1, max_iters=5000)
            if c7 < best_c:
                best_h, best_c = h7, c7
        except Exception:
            pass

    # Phase 8: Subgradient averaging (very short, for last-mile improvement)
    if time_left() > 10:
        try:
            h8, c8 = method_subgrad_averaged(best_h, deadline - 1, max_iters=2000)
            if c8 < best_c:
                best_h, best_c = h8, c8
        except Exception:
            pass

    # Final: verify and project
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
