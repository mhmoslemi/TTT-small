def project_feasible(x, n):
    """Project x onto {0<=h<=1, sum(h)=n/2}."""
    x = np.asarray(x, dtype=np.float64).copy()
    target = n / 2.0
    # Water-filling to find lambda such that sum(clip(x-lambda, 0, 1)) = target
    lo = np.min(x) - 2.0
    hi = np.max(x) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix residual
    for _ in range(10):
        res = h.sum() - target
        if abs(res) < 1e-12:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            rem = res
            for idx in order:
                dec = min(rem, h[idx])
                h[idx] -= dec
                rem -= dec
                if rem <= 0:

        else:

            rem = -res
            for idx in order:
                inc = min(rem, 1.0 - h[idx])

                if rem <= 0:
