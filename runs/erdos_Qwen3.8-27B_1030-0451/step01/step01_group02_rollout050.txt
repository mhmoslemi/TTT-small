```python
def project_feasible(x, target):
    """Project x onto {h: 0<=h<=1, sum(h)=target}"""
    n = len(x)
    # Bisection on lambda: h_i = clip(x_i - lambda, 0, 1), find lambda so sum = target
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(100):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(x - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5*(lo+hi), 0, 1)
```
