```python
def project_box_sum(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(x.min()) - 2.0
    hi = float(x.max()) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)

    res = h.sum() - target_sum
    if abs(res) > 1e-12:
```

```python
def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    base = n - 1
    Q = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Softmax weights
    Q_shifted = alpha * (Q - Q.max())
    exp_Q = np.exp(Q_shifted)
    w = exp_Q / exp_Q.sum()
    
    # Compute gradient as weighted sum
    grad = np.zeros(n)
    
    # Lag 0
    c0 = w[base]
    if c0 > 1e-15:
        grad += c0 * dx * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        # Positive lag
        wp = w[base + d]
        if wp > 1e-15:
            grad[:n-d] += wp * dx * (1.0 - h[d:])
            grad[d:] += -wp * dx * h[:n-d]
        
        # Negative lag
        wn = w[base - d]
        if wn > 1e-15:
            grad[:n-d] += -wn * dx * h[d:]
            grad[d:] += wn * dx * (1.0 - h[:n-d])
    
    lse_val = Q.max() + np.log(exp_Q.sum()) / alpha
    return lse_val, grad
```

```python
import numpy as np

from scipy.optimize import linprog

def project_box_sum(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    res = h.sum() - target_sum
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 1e-14:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= -1e-14:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def true_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_rows(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lse_gradient(h, alpha):
    n = len(h)
    dx = 2.0 / n
    base = n - 1
    Q = np.correlate(h, 1.0 - h, mode='full') * dx
    Q_shifted = alpha * (Q - Q.max())
    exp_Q = np.exp(Q_shifted)
    w = exp_Q / exp_Q.sum()
    grad = np.zeros(n)
    c0 = w[base]
    if c0 > 1e-15:
        grad += c0 * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[base + d]
        if wp > 1e-15:
            grad[:n - d] += wp * dx * (1.0 - h[d:])
            grad[d:] += -wp * dx * h[:n - d]
        wn = w[base - d]
        if wn > 1e-15:
            grad[:n - d] += -wn * dx * h[d:]
            grad[d:] += wn * dx * (1.0 - h[:n - d])
    lse_val = Q.max() + np.log(exp_Q.sum()) / alpha
    return lse_val, grad

def gradient_row(h, d):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    g[:n - d] += dx * (1.0 - h[d:])
    g[d:] += -dx * h[:n - d]
    return g

def gradient_row_neg(h, d):

    dx = 2.0 / n
    g = np.zeros(n)
    g[:n - d] += -dx * h[d:]
    g[d:] += dx * (1.0 - h[:n - d])
    return g

def solve_lp_correct(h, corr, trust, active_frac=1.0):
    n = len(h)
    base = n - 1
    m = 2 * n - 1
    k_active = max(1, int(m * active_frac))
    top_idx = np.argpartition(corr, -k_active)[-k_active:]
    
    nvars = n + 1
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    A_ub = np.zeros((k_active, nvars))
    b_ub = np.zeros(k_active)
    
    for i, idx in enumerate(top_idx):
        k = idx - base
        if k == 0:
            A_ub[i, :n] = gradient_row_zero(h)
        elif k > 0:
            A_ub[i, :n] = gradient_row(h, k)
        else:
            A_ub[i, :n] = gradient_row_neg(h, -k)
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[idx]
    
    lower = np.maximum(-h, -trust)

    A_eq = np.zeros((1, nvars))

    b_eq = np.array([0.0])
    bounds = list(zip(lower, upper)) + [(None, None)]
```
