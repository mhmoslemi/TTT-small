    n = len(h)

        for p in range(n):

                val += (1.0 - h[p + s])
            # p as second factor: h[p-s]*(1-h[p])
            if 0 <= p - s <= n - 1:
                val += (-h[p - s])
            J[k, p] = val
    return J

    n = len(h)

    L = 2 * n - 1
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx

    # lin_k = J_raw[k,a] - J_raw[k,b]

        # quad
        if (a + s == b) or (b + s == a):

    new_c = c + dx * d * lin + dx * d * d * quad
    return float(np.max(new_c))

            f1 = f2
            c2 = lo + gr * (hi - lo)
            f2 = pair_quadratic_score(h, a, b, c2)
    
    d_opt = (lo + hi) / 2.0
    f_opt = pair_quadratic_score(h, a, b, d_opt)

def interp_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n

def fw_linear_oracle(h, g, n):
    """Find u = argmin g^T x s.t. 0<=x<=1, sum(x)=n/2"""
    S = n / 2.0
    order = np.argsort(g)
    S_int = int(np.floor(S))
    S_frac = S - S_int
    u = np.zeros(n)
    for i in range(S_int):
        u[order[i]] = 1.0
    if S_frac > 1e-12:
        u[order[S_int]] = S_frac
    return u

def fw_step(h, n, tau=0.01):
    """One Frank-Wolfe step with exact subgradient."""
    dx = 2.0 / n
    L = 2 * n - 1
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    
    # Compute weights for active lags
    exponents = np.clip((c - c_max) / tau, -100, 0)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return h
    weights /= sw
    
    # Compute subgradient
    g = np.zeros(n)
    for k in range(L):
        w = weights[k] * dx
        if w < 1e-15:
            continue
        s = k - (n - 1)
        for p in range(n):
            if 0 <= p + s <= n - 1:
                g[p] += w * (1.0 - h[p + s])
            if 0 <= p - s <= n - 1:
                g[p] += w * (-h[p - s])
    
    # Linear oracle
    u = fw_linear_oracle(h, g, n)
    
    # Backtracking
    best_h = h
    best_s = c_max
    gammas = [0.5, 0.3, 0.2, 0.1, 0.05]
    for gamma in gammas:
        h_new = (1.0 - gamma) * h + gamma * u
        h_new = project_feasible(h_new, n)
        s = exact_score(h_new)
        if s < best_s - 1e-14:
            best_s = s
            best_h = h_new
    return best_h
