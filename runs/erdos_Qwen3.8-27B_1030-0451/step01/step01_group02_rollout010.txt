    """Solve min_t max_d(a[d]*t + b[d]) for t in [L, U]."""
    def f(t):
        return np.max(a * t + b)
    
    # Golden section search
    gr = (np.sqrt(5) - 1) / 2
    a1, b1 = L, U
    c1 = b1 - gr * (b1 - a1)
    d1 = a1 + gr * (b1 - a1)
    fc = f(c1)
    fd = f(d1)
    for _ in range(40):
        if fc < fd:
            b1 = d1
            d1 = c1
            fd = fc
            c1 = b1 - gr * (b1 - a1)
            fc = f(c1)
        else:
            a1 = c1
            c1 = d1
            fc = fd
            d1 = a1 + gr * (b1 - a1)
            fd = f(d1)
    t_opt = 0.5 * (a1 + b1)
    # Also check endpoints
    for t in (L, U, t_opt):
        pass
    t_best = min([L, U, t_opt], key=f)
    return t_best

def make_feasible(y, target):
