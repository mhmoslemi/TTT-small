    cs = corr - np.max(corr)
    w = np.exp(tau * cs)

def compute_c5(h):
    n = len(h)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_gradient(h, corr, tau):

    cs = corr - np.max(corr)
    w = np.exp(tau * cs)

    w = w / ws

        else:

            if m < n:
