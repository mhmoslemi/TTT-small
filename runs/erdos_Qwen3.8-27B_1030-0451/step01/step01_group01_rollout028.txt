Better: If we change h[i] by δ and h[j] by -δ, the problem is:
minimize max_k (f_k + δ * (a_k(i) - a_k(j)))
subject to: 0 ≤ h[i]+δ ≤ 1, 0 ≤ h[j]-δ ≤ 1

This is a 1D convex minimization. The optimal δ can be found by:
- Computing the upper envelope of the lines {f_k + δ*c_k}
- Finding the minimum of this envelope on [δ_min, δ_max]

For a convex piecewise-linear function, the minimum is at a "kink" point (intersection of two lines) or at a boundary.

The kink points are the O(n²) intersections of pairs of lines. But we can find the minimum more efficiently:

The function g(δ) = max_k (b_k + c_k*δ) is the upper envelope of n lines.
The minimum of g on [0, D] occurs at:
- δ = 0 (if g is decreasing at 0, which means all c_k for active k are negative)
- δ = D (if g is increasing at D)
- Or at a point where two lines with c_k > 0 and c_k < 0 (relative to the max at 0) cross

Actually, the simplest correct approach: since g is convex, use bisection on the subgradient:
- g'(δ) (in the subgradient sense) is monotonically non-decreasing
- Find δ where 0 ∈ ∂g(δ)

To compute the subgradient at δ:
- Active set: A(δ) = {k : b_k + c_k*δ = g(δ)}
- ∂g(δ) = [min_{k∈A} c_k, max_{k∈A} c_k]
- If 0 ∈ ∂g(δ), we're at the optimum
- If min c_k > 0, we need to decrease δ (g is increasing)
- If max c_k < 0, we need to increase δ (g is decreasing)

This gives a bisection that converges in O(log(D/ε)) iterations, each O(n).

import numpy as np
import time
from scipy.optimize import minimize, linprog
from math import log, sqrt

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
        r2 = target - y.sum()
        if r2 > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
        elif r2 < 0:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def coeff_matrix(h):
    """Compute the coefficient matrix: coeff[k, m] = partial overlap_k / partial h[m] (without dx)."""
    n = len(h)
    C = np.zeros((n, n))
    for k in range(n):
        for m in range(n):
            c = 0.0
            if m <= n - 1 - k:
                c += 1.0 - h[m + k]
            if m >= k:
                c -= h[m - k]
            C[k, m] = c
    return C

def smooth_obj_and_grad(h, beta):
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 500)
    u = np.exp(expo)
    Z = u.sum()
    f = m - np.log(Z) / beta
    w = u / Z
    n = len(h)
    C = coeff_matrix(h)
    dx = 2.0 / n
    g = (C.T @ w) * dx
    return f, g

def method_polyak_subgrad(h0, deadline, max_iters=5000):
    """Polyak subgradient method for min max_k f_k(h)."""
    n = len(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    # Polyak step: alpha = (f(x) - f*) / ||g||^2 where f* is lower bound
    # We use best_c as upper bound estimate and 0 as lower bound
    # Actually use: alpha = (c_cur - c_best_prev) / (||g||^2 + eps)
    alpha = 0.05
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        c_cur = np.max(v)
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        gap = max(0.02 * c_cur, 0.003)
        active = np.where(v >= c_cur - gap)[0]
        if len(active) == 0:
            active = np.array([np.argmax(v)])
        C = coeff_matrix(h)
        g = (C[active, :].mean(axis=0)) * dx
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        # Polyak-like step size
        step = alpha / (g_norm + 1e-10)
        step = min(step, 0.15)
        h_new = h - step * g
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < c_cur - 1e-14:
            h = h_new
            alpha = min(alpha * 1.02, 0.3)
        else:
            alpha *= 0.7
            if alpha < 1e-6:
                break
    return best_h, best_c

def method_pairwise_exact(h0, deadline, max_rounds=30, n_max=250):
    """Pairwise mass transfer with exact 1D bisection on the minimax."""
    n = len(h0)
    if n > n_max:
        return h0.copy(), evaluate_c5(h0)
    dx = 2.0 / n
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    C_full = np.zeros((n, n))

    def recompute_C():
        for k in range(n):
            for m in range(n):
                c = 0.0
                if m <= n - 1 - k:
                    c += 1.0 - h[m + k]
                if m >= k:
                    c -= h[m - k]
                C_full[k, m] = c

    recompute_C()

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        any_improve = False
        v = corr_all(h)
        c_cur = np.max(v)
        # Only consider pairs where h[j] > h[i] + threshold
        for i in range(n):
            for j in range(i + 1, n):
                if h[j] - h[i] < 0.005:
                    continue
                D = min(h[j], 1.0 - h[i])
                if D < 1e-8:
                    continue
                # c_k = C_full[k,i] - C_full[k,j]
                ck = C_full[:, i] - C_full[:, j]
                # We want to minimize max_k (v[k] + dx*delta*ck[k]) over delta in [0, D]
                # This is convex in delta. Use bisection on subgradient.
                lo_d, hi_d = 0.0, D
                for _ in range(20):
                    mid_d = (lo_d + hi_d) / 2.0
                    vals = v + dx * mid_d * ck
                    mx = np.max(vals)
                    active_idx = np.where(np.abs(vals - mx) < 1e-12)[0]
                    if len(active_idx) == 0:
                        active_idx = np.where(vals == mx)[0]
                    slopes = ck[active_idx]
                    s_max = np.max(slopes)
                    s_min = np.min(slopes)
                    if s_min <= 0.0 <= s_max:
                        lo_d = mid_d
                        hi_d = mid_d
                        break
                    elif s_max < 0.0:
                        lo_d = mid_d  # decreasing, go right
                    else:
                        hi_d = mid_d  # increasing, go left
                delta_star = (lo_d + hi_d) / 2.0
                if delta_star < 1e-10:
                    continue
                # Evaluate improvement
                h_tr = h.copy()
                h_tr[i] += delta_star
                h_tr[j] -= delta_star
                h_tr = np.clip(h_tr, 0.0, 1.0)
                h_tr = project_sum_box(h_tr, n / 2.0)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c - 1e-12:
                    best_c = c_tr
                    best_h = h_tr.copy()
                    h = h_tr.copy()
                    any_improve = True
                    recompute_C()
        if not any_improve:
            break
    return best_h, best_c

def method_lp_trust_v2(h0, deadline, max_iters=400):
    """Improved LP trust region with adaptive active set."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.15
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        # Adaptive gap: start tight, loosen if stuck
        gap = max(0.008 * vmax, 0.002 + 0.003 * min(no_improve, 5))
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 60:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-60:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-10:]

        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for ii, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[ii] = grad_at_lag(h, lag)
            b_lin[ii] = A[ii] @ h - v[k]

        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.0]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')

            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-11:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                radius = min(radius * 1.08, 0.3)
            else:
                no_improve += 1
                if no_improve >= 6:
                    radius = max(radius * 0.6, 0.005)
                    no_improve = 0
                    if radius <= 0.005:
                        break
        else:
            no_improve += 1
            if no_improve >= 10:
                radius = max(radius * 0.5, 0.005)
                no_improve = 0
                if radius <= 0.005:
                    break
    return best_h, best_c

def grad_at_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            m1 = np.arange(lag, n)
            g[m1] += 1.0 - h[m1 - lag]
            m2 = np.arange(0, n - lag)
            g[m2] -= h[m2 + lag]
    else:
        l = -lag
        if l < n:
            m1 = np.arange(0, n - l)
            g[m1] += 1.0 - h[m1 + l]
            m2 = np.arange(l, n)
            g[m2] -= h[m2 - l]
    return g

def method_lbfgs(h0, deadline, max_betas=4):
    """L-BFGS-B on log-sum-exp smoothed objective."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [8, 16, 32, 64, 128, 256, 512, 1024]

    def obj_func(x, beta):
        return smooth_obj_and_grad(x, beta)[0]

    def grad_func(x, beta):
        return smooth_obj_and_grad(x, beta)[1]

    for bi in range(min(max_betas, len(betas))):
        if time.time() > deadline:
            break
        beta = betas[bi]
        bounds = [(0.0, 1.0)] * n

        def obj_wrap(x):
            return obj_func(x, beta)

        def grad_wrap(x):
            return grad_func(x, beta)

        try:
            res = minimize(obj_wrap, best_h.copy(), jac=grad_wrap,
                           method='L-BFGS-B', bounds=bounds,
                           options={'maxiter': 200, 'ftol': 1e-15})
        except Exception:
            continue
        if res is not None and res.x is not None:
            x_new = project_sum_box(res.x, n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h0, period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def sinusoidal_perturb(h0, freq, amp):
    n = len(h0)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    pert = h0.copy() + amp * np.sin(2.0 * np.pi * freq * x / 2.0)
    pert = project_sum_box(pert, n / 2.0)
    return pert

def initial_fallback(n=100):
    h = np.full(n, 0.5)
    return project_sum_box(h, n / 2.0)
