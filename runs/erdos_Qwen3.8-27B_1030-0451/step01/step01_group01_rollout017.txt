import numpy as np

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -3.0, 3.0
    for _ in range(90):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
        r2 = target - y.sum()
        if r2 > 1e-12:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
        elif r2 < -1e-12:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))


def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m1 = np.arange(lag, n)
        g[m1] += 1.0 - h[m1 - lag]
        m2 = np.arange(0, n - lag)
        g[m2] -= h[m2 + lag]
    else:
        l = -lag
        m1 = np.arange(0, n - l)
        g[m1] += 1.0 - h[m1 + l]
        m2 = np.arange(l, n)
        g[m2] -= h[m2 - l]
    return g


def weighted_grad(h, weights, n_terms=200):
    n = len(h)
    g = np.zeros(n)
    idx = np.argsort(weights)[::-1][:n_terms]
    for k in idx:
        if weights[k] > 1e-12:
            lag = k - (n - 1)
            g += weights[k] * grad_lag(h, lag)
    return g


def fista_lse(h0, deadline, beta_start=10, beta_end=2000, max_iters=5000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    y = h0.copy()
    t = 1.0
    step = 1.0 / (beta_start * n)
    n_betas = 12
    for b_idx in range(n_betas):
        beta = beta_start * (beta_end / beta_start) ** (b_idx / max(n_betas - 1, 1))
        step = 4.0 / (beta * n)
        h = best_h.copy()
        y = h.copy()
        t = 1.0
        for it in range(max_iters // n_betas):
            if time.time() > deadline:
                break
            v = corr_all(y)
            m = v.max()
            expo = np.clip(beta * (v - m), None, 500)
            u = np.exp(expo)
            Z = u.sum()
            w = u / Z
            g = weighted_grad(y, w)
            y_new = y - step * g
            h_new = project_sum_box(y_new, n / 2.0)
            t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0 * t * t))
            y = h_new + (t - 1.0) / t_new * (h_new - h)
            y = np.clip(y, -0.5, 1.5)
            h = h_new
            t = t_new
            c_cur = evaluate_c5(h)
            if c_cur < best_c - 1e-11:
                best_c = c_cur
                best_h = h.copy()
    return best_h, best_c


def lp_trust_region(h0, deadline, max_iters=400):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.10
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = v.max()
        gap = max(0.02 * vmax, 0.008)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 60:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-60:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-15:]
        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)
            b_lin[i] = A[i] @ h - v[k]
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.0]])
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-11:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                if it % 8 == 0:
                    radius = min(radius * 1.08, 0.20)
            else:
                no_improve += 1
                if no_improve >= 6:
                    radius = max(radius * 0.6, 0.005)
                    no_improve = 0
                    if radius <= 0.005:
                        break
        else:
            no_improve += 1
            if no_improve >= 10:
                radius = max(radius * 0.5, 0.003)
                no_improve = 0
                if radius <= 0.003:
                    break
    return best_h, best_c


def slsqp_polish(h0, deadline, beta=64, max_iters=200):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    bounds = [(0.0, 1.0)] * n

    def obj(x):
        v = corr_all(x)
        m = v.max()
        expo = np.clip(beta * (v - m), None, 500)
        u = np.exp(expo)
        return m + np.log(u.sum()) / beta

    def grad(x):
        v = corr_all(x)
        m = v.max()
        expo = np.clip(beta * (v - m), None, 500)
        u = np.exp(expo)
        Z = u.sum()
        w = u / Z
        return weighted_grad(x, w)

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]
    try:
        res = minimize(obj, h0.copy(), jac=grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iters, 'ftol': 1e-15})
        if res is not None and res.x is not None:
            x_new = project_sum_box(res.x, n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c:
                best_h = x_new
                best_c = c_new
    except Exception:
        pass
    return best_h, best_c


def projected_subgrad(h0, deadline, max_iters=3000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.01
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        m = v.max()
        beta = 50.0
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        g = weighted_grad(h, w)
        gn = np.linalg.norm(g)
        if gn < 1e-13:
            break
        h_new = project_sum_box(h - step * g / gn, n / 2.0)

def interpolate_h(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_interp = np.interp(x_new, x_old, np.concatenate([h0, h0[:1]]))
    h_interp = np.clip(h_interp, 0.0, 1.0)
    return project_sum_box(h_interp, n_new / 2.0)


def random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = rng.rand(n)
    return project_sum_box(h, n / 2.0)


def block_init(n, seed):
    rng = np.random.RandomState(seed)
    n_blocks = rng.randint(3, 8)
    edges = np.sort(rng.choice(range(1, n), n_blocks - 1, replace=False))
    edges = np.concatenate([[0], edges, [n]])
    h = np.zeros(n)
    for i in range(n_blocks):
        h[edges[i]:edges[i + 1]] = 1.0
    h = project_sum_box(h, n / 2.0)
    return h


def sinusoid_init(n, freq, phase, seed):
    x = np.linspace(0, 1, n, endpoint=False)
    h = 0.5 + 0.4 * np.sin(2 * np.pi * freq * x + phase)
    h = np.clip(h, 0, 1)
    return project_sum_box(h, n / 2.0)

            h10, c10 = lp_trust_region(best_h, deadline - 0.5, max_iters=50)
            if c10 < best_c:
                best_h, best_c = h10, c10
        except Exception:
            pass

    # Final projection and evaluation
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    best_c = min(best_c, final_c)

    return best_h, float(best_c), len(best_h)
