def resample_h(h_old, n_new):
    """Resample h from n_old points to n_new points using linear interpolation."""
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)

def coordinate_polish(h, time_limit, seed=42):

def build_grad_matrix(h):
    n = len(h)
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        j = k - (n - 1)
        if j > 0:
            G[k, j:] += dx * (1.0 - h[:n - j])
            G[k, :n - j] -= dx * h[j:]
        elif j < 0:
            l = -j
            G[k, :n - l] += dx * (1.0 - h[l:])
            G[k, l:] -= dx * h[:n - l]
        else:
            G[k] = dx * (1.0 - 2.0 * h)
    return G

def lp_direction(h, G, c_norm, f, step_size, top_m=None):
    n = len(h)
    total_k = 2 * n - 1
    if top_m is None:
        top_m = total_k
    m_actual = min(top_m, total_k)
    active_idx = np.argsort(c_norm)[-m_actual:]
    active_idx = active_idx[::-1]
    M = len(active_idx)
    G_a = G[active_idx]
    c_a = c_norm[active_idx]
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0
    A_ub = np.zeros((M, nv))
    A_ub[:, 0:n] = G_a
    A_ub[:, n] = -1.0
    b_ub = f - c_a
    A_eq = np.zeros((1, nv))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        lo = -min(h[i], step_size)
        hi = min(1.0 - h[i], step_size)
        bounds.append((lo, hi))
    bounds.append((-np.inf, np.inf))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 3})
        if res.status == 0:
            delta = res.x[0:n]
            if np.all(np.isfinite(delta)) and np.linalg.norm(delta) > 1e-14:
                return delta
    except Exception:
        pass
    return None

def softmax_gradient(h, G, c_norm, f, tau):
    cs = c_norm - f
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(len(h))
    w = w / ws
    return G.T @ w

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.30 * np.sin(2.0 * np.pi * x)
    h += 0.15 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.08 * np.sin(6.0 * np.pi * x + 1.5)
    h += 0.04 * np.sin(10.0 * np.pi * x + 2.3)
    h += 0.02 * np.sin(14.0 * np.pi * x + 3.1)
    h += rng.randn(n) * 0.025
    return project_constraint(h, n)

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return project_constraint(h_new, n_new)

def local_optimize(h0, time_limit, seed_val=42):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed_val)
    step_size = 0.04
    stall = 0
    tau = 40.0
    consecutive_fail = 0

    while time.time() - t0 < time_limit - 1:
        corr_raw = compute_corr(h)
        dx = 2.0 / n
        c_norm = corr_raw * dx
        f = float(np.max(c_norm))
        c5_val = float(np.max(corr_raw) * dx)

        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            stall = 0
            consecutive_fail = 0

        G = build_grad_matrix(h)
        improved = False

        # Primary: LP-based direction
        delta = lp_direction(h, G, c_norm, f, step_size, top_m=None)
        if delta is not None:
            multis = [1.0, 0.8, 0.6, 0.4, 0.25, 0.12]
            for mult in multis:
                h_try = project_constraint(h + mult * delta, n)
                c_try = compute_c5(h_try)
                if c_try <= c5_val + 1e-13:
                    h = h_try
                    if c_try < c5_val - 1e-12:
                        improved = True
                        step_size = min(step_size * 1.15, 0.12)
                    break
            if not improved:
                consecutive_fail += 1
                step_size *= 0.8
        else:
            consecutive_fail += 1

        # Secondary: smooth gradient if LP failed or no improvement
        if not improved:
            grad = softmax_gradient(h, G, c_norm, f, tau)
            gnorm = float(np.linalg.norm(grad))
            if gnorm > 1e-12:
                step_g = step_size * 0.7 / gnorm
                for mult in [1.0, 0.5, 0.25, 0.1]:
                    h_try = project_constraint(h - mult * step_g * grad, n)
                    c_try = compute_c5(h_try)
                    if c_try <= c5_val + 1e-13:
                        h = h_try
                        if c_try < c5_val - 1e-12:
                            improved = True
                        break
            tau = min(tau * 1.03, 300.0)

        if improved:
            stall = 0
        else:
            stall += 1

        # Escape mechanisms
        if stall > 12:
            scale = 0.015 + 0.01 * (stall // 10)
            scale = min(scale, 0.06)
            h = project_constraint(h + rng.randn(n) * scale, n)
            stall = 0
            tau = 25.0
            step_size = 0.04

        # Binary snap attempt
        if stall > 6 and rng.random() < 0.08:
            h_snap = np.where(h > 0.5, 1.0, 0.0)
            ones = int(np.sum(h_snap))
            needed = n // 2
            if ones > needed:
                idx = np.where(h_snap == 1.0)[0]
                rng.shuffle(idx)
                for ii in range(ones - needed):
                    h_snap[ii if False else idx[ii]] = 0.0
            elif ones < needed:
                idx = np.where(h_snap == 0.0)[0]
                rng.shuffle(idx)
                for ii in range(needed - ones):
                    h_snap[idx[ii]] = 1.0
            h_blend = project_constraint(0.6 * h + 0.4 * h_snap, n)
            if compute_c5(h_blend) < c5_val:

        # Recover from degradation
        if consecutive_fail > 30 and compute_c5(h) > best_c5 + 0.005:
            h = best_h.copy()
            consecutive_fail = 0
            stall = 0

    return best_h, best_c5

def coordinate_polish(h, time_limit, seed_val=42):
    n = len(h)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed_val)
    amounts = [0.01, 0.03, 0.05, 0.08, 0.12, 0.18]

    while time.time() - t0 < time_limit - 0.5:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for amt in amounts:
            if h[i] >= amt and h[j] + amt <= 1.0 + 1e-12:
                h_try = h.copy()
                h_try[i] -= amt
                h_try[j] += min(amt, 1.0 - h[j])

                if c_try < best_c5 - 1e-13:
                    h = h_try
                    best_h = h.copy()
                    best_c5 = c_try

            if h[j] >= amt and h[i] + amt <= 1.0 + 1e-12:

                h_try[j] -= amt
                h_try[i] += min(amt, 1.0 - h[i])
