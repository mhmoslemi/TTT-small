def fista_project(h, S):
    """Project h onto {0 <= h <= 1, sum(h) = S} using binary search."""
    N = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        proj = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > S:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) * 0.5
    proj = np.clip(h - t, 0.0, 1.0)
    # Fix any residual
    residual = S - np.sum(proj)
    if abs(residual) > 1e-12:
        if residual > 0:
            free = np.where(proj < 1.0)[0]
            if len(free) > 0:
                for i in free:
                    add = min(residual, 1.0 - proj[i])
                    proj[i] += add
                    residual -= add
                    if abs(residual) < 1e-12:
                        break
        else:
            free = np.where(proj > 0.0)[0]
            if len(free) > 0:
                for i in free:
                    sub = min(-residual, proj[i])
                    proj[i] -= sub
                    residual += sub
                    if abs(residual) < 1e-12:
                        break
    return proj

def fista_step(h, y, grad, lr, S):
    """One FISTA step."""
    z = y - lr * grad
    return fista_project(z, S)

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_overlaps(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def grad_lse(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(exponents)
    s = np.sum(e)
    w = e / s
    grad = np.zeros(N)
    for r in range(2 * N - 1):
        if w[r] < 1e-12:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                grad[:n] += w[r] * (1.0 - h[d:])
                grad[d:] -= w[r] * h[:n]
        else:
            dp = -d
            n = N - dp
            if n > 0:
                grad[dp:] += w[r] * (1.0 - h[:n])
                grad[:n] -= w[r] * h[dp:]
    return grad * dx, m + tau * np.log(s)


def project_box_sum(h, S):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s_val = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s_val > S:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    proj = np.clip(h - t, 0.0, 1.0)
    residual = S - np.sum(proj)
    if abs(residual) > 1e-10:
        if residual > 0:
            free = np.where(proj < 1.0)[0]
            for idx in free:
                add = min(residual, 1.0 - proj[idx])
                proj[idx] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
        else:
            free = np.where(proj > 0.0)[0]
            for idx in free:
                sub = min(-residual, proj[idx])
                proj[idx] -= sub
                residual += sub
                if abs(residual) < 1e-12:
                    break
    return proj


def upscale(h, N_new):
    N_old = len(h)
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_box_sum(h_new, N_new / 2.0)


def lse_func(h, tau):
    return grad_lse(h, tau)[1]


def lse_grad(h, tau):
    return grad_lse(h, tau)[0]


def make_sum_constraint(S):
    def fun(x):
        return np.sum(x) - S
    def jac(x):
        return np.ones_like(x)
    return {'type': 'eq', 'fun': fun, 'jac': jac}


def slsqp_optimize(h, tau, maxiter, deadline):
    N = len(h)
    S = N / 2.0
    bounds = [(0.0, 1.0)] * N
    constraint = make_sum_constraint(S)
    try:
        result = minimize(
            lse_func, h.copy(),
            jac=lse_grad,
            args=(tau,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_box_sum(result.x, S)
    except Exception:
        return h.copy()


def subgradient_polish(h, S, n_iters, deadline, rng):
    N = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    step_scale = 1.0

    for it in range(n_iters):
        if time.time() > deadline:
            break

        over = compute_overlaps(h)
        M = np.max(over)

        if M < best_c5 - 1e-14:
            best_c5 = M
            best_h = h.copy()

        # Find top lags for smoothed subgradient
        top_k = min(20, 2 * N - 1)
        sorted_idx = np.argsort(over)[::-1]
        top_vals = over[sorted_idx[:top_k]]
        top_shifts = sorted_idx[:top_k] - (N - 1)

        # Weighted gradient (soft max weights)
        temp = 0.01
        w = np.exp((top_vals - top_vals[0]) / temp)
        w = w / np.sum(w)

        grad = np.zeros(N)
        dx = 2.0 / N
        for k in range(top_k):
            if w[k] < 1e-10:
                continue
            d = int(top_shifts[k])
            if d >= 0:
                n = N - d
                if n > 0:
                    grad[:n] += w[k] * (1.0 - h[d:])
                    grad[d:] -= w[k] * h[:n]
            else:
                dp = -d
                n = N - dp
                if n > 0:
                    grad[dp:] += w[k] * (1.0 - h[:n])
                    grad[:n] -= w[k] * h[dp:]
        grad *= dx

        # Make zero-mean
        grad = grad - np.mean(grad)

        # Try decreasing step sizes
        improved = False
        for trial in range(12):
            alpha = step_scale * (0.5 ** trial)
            if alpha < 1e-8:
                break
            h_new = project_box_sum(h - alpha * grad, S)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5 - 1e-12:
                h = h_new
                best_c5 = c5_new
                best_h = h.copy()
                step_scale = alpha * 1.5
                improved = True
                break
        if not improved:
            step_scale *= 0.7

    return best_h, best_c5


def pair_polish(h, S, n_rounds, deadline, rng):
    N = len(h)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    dx = 2.0 / N

    for rnd in range(n_rounds):
        if time.time() > deadline:
            break

        over = compute_overlaps(h)
        M = np.max(over)
        if M < best_c5 - 1e-14:
            best_c5 = M
            best_h = h.copy()

        # Active set
        K = 15
        top_vals, top_idx = np.partition(over, -K)[-K:], np.argsort(over)[-K:]
        top_shifts = top_idx - (N - 1)

        # Try random pairs
        n_pairs = 50
        for _ in range(n_pairs):
            if time.time() > deadline:
                break
            i = rng.integers(0, N)
            j = rng.integers(0, N)
            if i == j:
                continue

            # Feasible delta range
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi - 1e-10:
                continue

            # Compute 1D function: theta_d(delta) for active d
            # theta_d = sum h[n]*(1-h[n+d])*dx
            # After change: h[i]+delta, h[j]-delta
            # Compute linear and quadratic coefficients for each active lag
            a_coeffs = np.zeros(K)
            p_coeffs = np.zeros(K)
            q_coeffs = np.zeros(K)

            for k in range(K):
                d = int(top_shifts[k])
                a_coeffs[k] = top_vals[k]
                # Linear: compute grad_d[i] - grad_d[j]
                gi = 0.0
                gj = 0.0
                if d >= 0:
                    n_terms = N - d
                    if 0 <= i < n_terms:
                        gi += (1.0 - h[i + d])
                    if d <= i < N:
                        gi -= h[i - d]
                    if 0 <= j < n_terms:
                        gj += (1.0 - h[j + d])
                    if d <= j < N:
                        gj -= h[j - d]
                else:
                    dp = -d
                    n_terms = N - dp
                    if dp <= i < N:
                        gi += (1.0 - h[i - dp])
                    if 0 <= i < n_terms:
                        gj_term = h[i + dp] if i + dp < N else 0
                        gi -= gj_term
                    if dp <= j < N:
                        gj += (1.0 - h[j - dp])
                    if 0 <= j < n_terms:
                        gj -= h[j + dp]
                p_coeffs[k] = dx * (gi - gj)

                # Quadratic from cross terms
                q = 0.0
                if d >= 0:
                    if i + d == j and i + d < N:
                        q += 1.0
                    if j + d == i and j + d < N:
                        q += 1.0
                else:
                    dp = -d
                    if i - dp == j and i - dp >= 0:
                        q += 1.0
                    if j - dp == i and j - dp >= 0:
                        q += 1.0
                q_coeffs[k] = dx * q

            # Minimize max_k (a + p*delta + q*delta^2) over [d_lo, d_hi]
            # Sample and refine
            n_samples = 40
            deltas = np.linspace(d_lo, d_hi, n_samples)
            f_vals = np.zeros(n_samples)
            for si in range(n_samples):
                delta = deltas[si]
                f_vals[si] = np.max(a_coeffs + p_coeffs * delta + q_coeffs * delta * delta)

            best_si = np.argmin(f_vals)
            if f_vals[best_si] >= M - 1e-12:
                continue

            # Refine around best sample
            lo = deltas[max(0, best_si - 1)]
            hi = deltas[min(n_samples - 1, best_si + 1)]
            for _ in range(30):
                mid1 = lo + (hi - lo) / 3.0
                mid2 = hi - (hi - lo) / 3.0
                f1 = np.max(a_coeffs + p_coeffs * mid1 + q_coeffs * mid1 * mid1)
                f2 = np.max(a_coeffs + p_coeffs * mid2 + q_coeffs * mid2 * mid2)
                if f1 < f2:
                    hi = mid2
                else:
                    lo = mid1
            delta_opt = (lo + hi) / 2.0
            f_opt = np.max(a_coeffs + p_coeffs * delta_opt + q_coeffs * delta_opt * delta_opt)

            if f_opt < M - 1e-12:
                h[i] += delta_opt
                h[j] -= delta_opt
                h = project_box_sum(h, S)
                M = compute_c5(h)
                if M < best_c5:
                    best_c5 = M
                    best_h = h.copy()

    return best_h, best_c5

            h = slsqp_optimize(h, tau, maxiter, deadline)
            h = project_box_sum(h, S3)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Subgradient polish at N=700
        if time.time() < deadline - 30:
            h, c5 = subgradient_polish(h, S3, 15000, deadline - 30, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Pair polish at N=700
        if time.time() < deadline - 15:
            h, c5 = pair_polish(h, S3, 100, deadline - 15, rng)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Phase 4: Final SLSQP polish on best solution
    if time.time() < deadline - 10:
        N_best = len(best_h)
        S_best = N_best / 2.0
        h = best_h.copy()
        for tau in [0.0003, 0.0002, 0.0001]:
            if time.time() > deadline - 5:
                break
            h = slsqp_optimize(h, tau, 200, deadline)
            h = project_box_sum(h, S_best)
            c5 = compute_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    # Final subgradient attempt
    if time.time() < deadline - 2:
        N_best = len(best_h)
        S_best = N_best / 2.0
        _, c5 = subgradient_polish(best_h, S_best, 5000, deadline, rng)
        if c5 < best_c5:
            best_c5 = c5
            best_h = project_box_sum(best_h, S_best)

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
