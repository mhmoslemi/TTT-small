            np.clip(proj, 0, 1, out=proj)
    return proj

def gradient_of_weighted_corr(h, p):
    """Gradient of sum_k p[k] * corr[k] w.r.t. h.
    corr = np.correlate(h, 1-h, mode='full')
    """
    n = len(h)
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        pk = p[k]
        if abs(pk) < 1e-16:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if lag == 0:
                grad += pk * (1.0 - 2.0 * h)
            else:
                grad[:n - lag] += pk * (1.0 - h[lag:])
                grad[lag:] -= pk * h[:n - lag]
        else:
            L = -lag
            grad[:n - L] -= pk * h[L:]
            grad[L:] += pk * (1.0 - h[:n - L])
    return grad

def softmax_weights(corr, tau):
    """Compute p_k = exp(tau*corr_k) / sum_j exp(tau*corr_j)"""
    tcorr = tau * corr
    mx = np.max(tcorr)
    exp_vals = np.exp(tcorr - mx)
    s = np.sum(exp_vals)
    if s < 1e-300:
        idx = np.argmax(corr)
        p = np.zeros_like(corr)
        p[idx] = 1.0
        return p
    return exp_vals / s

    """FISTA optimization with adaptive tau."""
    n = len(h0)
    h = h0.astype(float).copy()
    h_prev = h.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    
    t0 = time.time()
    t_fista = 1.0
    step = 0.5 / n  # initial step size
    
    # Adaptive tau: start moderate, increase over time
    tau = 20.0
    last_improve = 0
    stall_count = 0
    
    iters = 0
    while time.time() - t0 < time_limit:
        iters += 1
        # Adaptive tau
        elapsed_frac = (time.time() - t0) / max(time_limit, 1e-6)
        # Gradually increase tau
        tau = 20.0 * (1.0 + 19.0 * elapsed_frac ** 0.7)
        
        corr = compute_corr(h)
        c5_val = float(np.max(corr) * dx)
        
        if c5_val < best_c5 - 1e-14:
            best_c5 = c5_val
            best_h = h.copy()
            last_improve = iters
            stall_count = 0
        elif iters - last_improve > 50:
            stall_count += 1
            if stall_count > 5:
                # Perturb and reset momentum
                h_prev = h.copy()
                h = project_constraint(h + np.random.randn(n) * 0.02, n)
                stall_count = 0
                last_improve = iters
        
        # Compute gradient
        p = softmax_weights(corr, tau)
        grad = gradient_of_weighted_corr(h, p)
        
        # FISTA momentum
        t_prev = t_fista
        t_fista = (1.0 + np.sqrt(1.0 + 4.0 * t_fista * t_fista)) / 2.0
        momentum = (t_prev - 1.0) / t_fista
        
        z = h + momentum * (h - h_prev)
        z_clipped = project_constraint(z, n)
        
        # Compute gradient at z
        corr_z = compute_corr(z_clipped)
        p_z = softmax_weights(corr_z, tau)
        grad_z = gradient_of_weighted_corr(z_clipped, p_z)
        
        # Backtracking line search
        y = z_clipped - step * grad_z
        made_progress = False
        for _ in range(20):
            h_new = project_constraint(y, n)
            c_new = compute_corr(h_new)
            c5_new = float(np.max(c_new) * dx)
            if c5_new <= c5_val + 1e-12:
                h_prev = h.copy()
                h = h_new
                made_progress = True
                break
            step *= 0.5
            y = z_clipped - step * grad_z
        
        if not made_progress:
            step = max(step, 0.01 / n)
            h = project_constraint(z_clipped - step * grad_z, n)
            h_prev = h_prev  # keep old
        else:
            step = min(step * 1.5, 2.0 / n)
    
    return best_h, best_c5

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(p - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-13:
        interior = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            proj[interior] += diff / cnt
            np.clip(proj, 0.0, 1.0, out=proj)
    return proj

def gradient_weighted(h, p):
    n = len(h)
    grad = np.zeros(n)

            continue

            if lag == 0:

            else:
                grad[:n - lag] += pk * (1.0 - h[lag:])
                grad[lag:] -= pk * h[:n - lag]
        else:
            L = -lag
            grad[:n - L] += pk * (1.0 - h[L:])
            grad[L:] -= pk * h[:n - L]
    return grad
