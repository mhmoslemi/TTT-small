def make_feasible(x, target):
    """Project x onto [0,1]^n with sum = target."""
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    x = np.clip(x, 0.0, 1.0)
    # Bisection to find offset
    lo, hi = -1.0 - np.max(x), 1.0 - np.min(x)
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    r = target - x.sum()
    if abs(r) > 1e-13:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                # Add to smallest elements first
                order = idx[np.argsort(x[idx])]
                rem = r
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
                    rem -= inc
                    if rem <= 1e-14:

        else:
            idx = np.where(x > 0.0)[0]

                for i in order:
                    dec = min(x[i], rem)
                    x[i] -= dec
                    rem -= dec
                    if rem <= 1e-14:
                        break
    x = np.clip(x, 0.0, 1.0)
    return x

    return np.correlate(h, 1.0 - h, "full") * dx
