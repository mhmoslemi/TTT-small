```python
import numpy as np
import time
from scipy.optimize import linprog, minimize

def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
                y = np.clip(y, 0, 1)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
                y = np.clip(y, 0, 1)
    return y

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def coord_descent(h, deadline, max_sweeps=200, grid_size=150):
    """Exact coordinate descent for the Erdős overlap problem."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    h = h.copy()
    a_grid = np.linspace(0, 1, grid_size)
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = np.random.permutation(n)
        improved = False
        for i in order:
            if time.time() > deadline:
                break
            # Compute A[lag] and B[lag] for all lags
            c = corr_all(h)  # O(n^2) - could be optimized but n is small
            # For each lag, compute A and B
            A = np.zeros(2*n-1)  # indexed by 0..2n-2, corresponding to lags -(n-1)..(n-1)
            B = np.zeros(2*n-1)
            
            for k in range(2*n-1):
                lag = k - (n-1)
                if lag == 0:
                    B[k] = c[k] - h[i]*(1-h[i])*dx
                    A[k] = dx  # linear coefficient (quadratic term is -dx*a^2)
                elif lag > 0:
                    # A = (1-h[i+lag])*dx*I(i+lag<n) - h[i-lag]*dx*I(i>=lag)
                    a_coeff = 0.0
                    if i + lag < n:
                        a_coeff += (1 - h[i+lag]) * dx
                    if i >= lag:
                        a_coeff -= h[i-lag] * dx
                    A[k] = a_coeff
                    B[k] = c[k] - h[i] * A[k]
                else:
                    m = -lag
                    a_coeff = 0.0
                    if i >= m:
                        a_coeff += (1 - h[i-m]) * dx
                    if i + m < n:
                        a_coeff -= h[i+m] * dx
                    A[k] = a_coeff
                    B[k] = c[k] - h[i] * A[k]
            
            # Evaluate g(a) = max(max_{lag!=0}(A*lag*a + B[lag]), B[0] + dx*a - dx*a^2)
            # For lag=0: B[0] + dx*a - dx*a^2 (quadratic)
            # For lag!=0: A[lag]*a + B[lag] (linear)
            
            k0 = n-1  # index for lag 0
            # Linear part (all lags except 0)
            linear_vals = A * a_grid[None, :] + B[None, :]  # shape (2n-1, grid_size)
            linear_vals[k0, :] = -np.inf  # exclude lag 0 from linear max
            lin_max = np.max(linear_vals, axis=0)  # max over lags of linear part
            
            # Quadratic part (lag 0)
            quad_vals = B[k0] + dx * a_grid - dx * a_grid**2
            
            # Overall max
            g = np.maximum(lin_max, quad_vals)
            
            # Find minimum
            idx_best = np.argmin(g)
            a_new = a_grid[idx_best]
            
            # Refine with finer grid around the best point
            if 0 < idx_best < grid_size - 1:
                a_lo = a_grid[idx_best - 1]
                a_hi = a_grid[idx_best + 1]
                fine_grid = np.linspace(a_lo, a_hi, 50)
                fine_lin = A * fine_grid[None, :] + B[None, :]
                fine_lin[k0, :] = -np.inf
                fine_lin_max = np.max(fine_lin, axis=0)
                fine_quad = B[k0] + dx * fine_grid - dx * fine_grid**2
                fine_g = np.maximum(fine_lin_max, fine_quad)
                idx_fine = np.argmin(fine_g)
                a_new = fine_grid[idx_fine]
            
            if a_new != h[i]:
                h[i] = a_new
                improved = True
        
        # Project to sum constraint
        h = project_sum_box(h, n / 2.0)
        c_cur = evaluate_c5(h)
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
        elif not improved:
            break
    
    return best_h, best_c
```

```python
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_c5(best_h)

        if time.time() > deadline:
            break
        order = np.random.permutation(n)

            A = np.zeros(2*n-1)
            B = np.zeros(2*n-1)

                    B[k] = c[k] - h[i]*(1-h[i])*dx
                    A[k] = dx
                elif lag > 0:
                    a_coeff = 0.0
                    if i + lag < n:
                        a_coeff += (1 - h[i+lag]) * dx
                    if i >= lag:
                        a_coeff -= h[i-lag] * dx
                    A[k] = a_coeff
                    B[k] = c[k] - h[i] * A[k]

                        a_coeff -= h[i+m] * dx

                    B[k] = c[k] - h[i] * A[k]

            quad = B[k0] + dx*a_grid - dx*a_grid**2

            a_new = a_grid[idx_best]

                h[i] = a_new
                # Update correlation
```

```python
import numpy as np
import time
from scipy.optimize import linprog, minimize

def project_sum_box(x, target):
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            m1 = np.arange(lag, n)
            g[m1] += 1.0 - h[m1 - lag]
            m2 = np.arange(0, n - lag)
            g[m2] -= h[m2 + lag]
    else:
        l = -lag
        if l < n:
            m1 = np.arange(0, n - l)
            g[m1] += 1.0 - h[m1 + l]
            m2 = np.arange(l, n)
            g[m2] -= h[m2 - l]
    return g

def coord_descent(h, deadline, max_sweeps=2000, grid_size=200):
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    h = h.copy()
    a_grid = np.linspace(0.0, 1.0, grid_size)
    k0 = n - 1
    n_lags = 2 * n - 1
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        order = np.random.permutation(n)
        any_change = False
        for i in order:
            if time.time() > deadline:
                break
            c = corr_all(h)
            A = np.zeros(n_lags)
            B = np.zeros(n_lags)
            hi = h[i]
            for k in range(n_lags):
                lag = k - k0
                if lag == 0:
                    A[k] = dx
                    B[k] = c[k] - hi * (1.0 - hi) * dx
                elif lag > 0:
                    a_val = 0.0
                    if i + lag < n:
                        a_val += (1.0 - h[i + lag]) * dx
                    if i >= lag:
                        a_val -= h[i - lag] * dx
                    A[k] = a_val
                    B[k] = c[k] - hi * a_val
                else:
                    m = -lag
                    a_val = 0.0
                    if i >= m:
                        a_val += (1.0 - h[i - m]) * dx
                    if i + m < n:
                        a_val -= h[i + m] * dx
                    A[k] = a_val
                    B[k] = c[k] - hi * a_val
            lin = A[None, :] * a_grid[:, None] + B[None, :]
            lin[:, k0] = -np.inf
            lin_max = np.max(lin, axis=1)
            quad = B[k0] + dx * a_grid - dx * a_grid ** 2
            g = np.maximum(lin_max, quad)
            idx_b = np.argmin(g)
            a_new = a_grid[idx_b]
            if 0 < idx_b < grid_size - 1:
                lo_a = a_grid[idx_b - 1]
                hi_a = a_grid[idx_b + 1]
                fine = np.linspace(lo_a, hi_a, 40)
                f_lin = A[None, :] * fine[:, None] + B[None, :]
                f_lin[:, k0] = -np.inf
                f_quad = B[k0] + dx * fine - dx * fine ** 2
                f_g = np.maximum(np.max(f_lin, axis=1), f_quad)
                a_new = fine[np.argmin(f_g)]
            if abs(a_new - hi) > 1e-12:
                h[i] = a_new
                any_change = True
        if not any_change:
            break
        h = project_sum_box(h, n / 2.0)
        c_cur = evaluate_c5(h)
        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
    return best_h, best_c

def smooth_obj_grad(h, beta):
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 400)
    u = np.exp(expo)
    Z = u.sum()
    f = m - np.log(Z) / beta
    w = u / Z
    n = len(h)
    g = np.zeros(n)
    for k in range(len(v)):
        if w[k] > 1e-12:
            lag = k - (n - 1)
            g += w[k] * grad_lag(h, lag)
    return f, g

def method_lp_trust(h0, deadline, max_iters=300):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = 0.15
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        gap = max(0.02 * vmax, 0.003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 100:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-100:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-20:]
        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i_ai, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i_ai] = grad_lag(h, lag)
            b_lin[i_ai] = A[i_ai] @ h - v[k]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.5]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                if it % 4 == 0:
                    radius = min(radius * 1.15, 0.3)
            else:
                no_improve += 1
                if no_improve >= 6:
                    radius = max(radius * 0.5, 0.002)
                    no_improve = 0
                    if radius <= 0.002:
                        break
        else:
            no_improve += 1
            if no_improve >= 10:
                radius = max(radius * 0.5, 0.002)
                no_improve = 0
                if radius <= 0.002:
                    break
    return best_h, best_c

def basin_hopping_coord(h0, deadline, n_hops=30, max_sweeps_per_hop=300):

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_cat = np.concatenate([h0, h0[:1]])
    h_new = np.interp(x_new, x_old, h_cat, period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def binary_random_init(n, seed):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0

def fourier_init(n, seed):
    rng = np.random.RandomState(seed)
    freqs = [1, 2, 3]
    phases = rng.uniform(0, 2 * np.pi, size=len(freqs))
    x = np.linspace(0, 2, n, endpoint=False)
    sig = 0.5
    for f, p in zip(freqs, phases):
        sig += 0.15 * np.sin(np.pi * f * x + p)
    h = np.clip(sig, 0, 1)
    h = project_sum_box(h, n / 2.0)
    return h

            h3, c3 = basin_hopping_coord(best_h, dl, n_hops=25, max_sweeps_per_hop=400)
            if c3 < best_c:
                best_h, best_c = h3, c3
        except Exception:
            pass

    # Phase 4: LP trust region refinement
    if time_left() > 80:
        dl = min(deadline, start_time + budget_s * 0.65)
        try:
            h4, c4 = method_lp_trust(best_h, dl, max_iters=400)
            if c4 < best_c:
                best_h, best_c = h4, c4
        except Exception:
            pass

    # Phase 5: Cool-down coordinate descent
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.75)
        try:
            h5, c5 = coord_descent(best_h, dl, max_sweeps=5000, grid_size=400)
            if c5 < best_c:
                best_h, best_c = h5, c5
        except Exception:
            pass

    # Phase 6: Resolution transfer
    if time_left() > 100:
        n_cur = len(best_h)
        targets = []
        for f in [1.4, 1.75, 2.0, 2.5]:
            t = int(round(n_cur * f))
            if t > n_cur and t <= 300:
                targets.append(t)
        targets = sorted(set(targets))
        for n_t in targets:
            if time_left() < 80:
                break
            try:
                h_tr = transfer_resolution(best_h, n_t)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c:
                    best_h, best_c = h_tr, c_tr
                sub_dl = min(deadline - 10, time.time() + min(200, time_left() * 0.5))
                h_a, c_a = coord_descent(best_h, sub_dl, max_sweeps=2000, grid_size=200)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a
                if time_left() > 40:
                    sub_dl2 = min(deadline - 5, time.time() + min(60, time_left() * 0.3))
                    h_b, c_b = method_lp_trust(best_h, sub_dl2, max_iters=200)
                    if c_b < best_c:
                        best_h, best_c = h_b, c_b
            except Exception:
                pass

    # Phase 7: Try alternative initializations at current best resolution
    if time_left() > 80:
        n_alt = len(best_h)
        for alt_seed in [123, 456, 789, 1000, 2024]:
            if time_left() < 40:
                break
            try:
                h_alt = binary_random_init(n_alt, alt_seed)
                sub_dl = min(deadline - 5, time.time() + min(60, time_left() * 0.5))
                h_a, c_a = coord_descent(h_alt, sub_dl, max_sweeps=1000, grid_size=150)
                if c_a < best_c:
                    best_h, best_c = h_a, c_a
            except Exception:
                pass

    # Phase 8: Final polish
    if time_left() > 20:
        try:
            h6, c6 = coord_descent(best_h, deadline - 1, max_sweeps=10000, grid_size=500)
            if c6 < best_c:
                best_h, best_c = h6, c6
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
```
