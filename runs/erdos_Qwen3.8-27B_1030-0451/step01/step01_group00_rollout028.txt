                total_moves += 1

                    best_h = h.copy()

                stall = 0
        else:
            stall = 0

def smooth_gradient_descent(h, time_limit, seed=42):
    """Gradient descent with log-sum-exp smoothing."""
    n = len(h)
    h = h.astype(float).copy()
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    rng = np.random.RandomState(seed)
    t0 = time.time()
    dx = 2.0 / n
    
    tau = 50.0  # smoothing parameter
    step = 0.01
    
    while time.time() - t0 < time_limit - 5:
        # Compute correlation
        corr = np.correlate(h, 1.0 - h, mode="full")
        cmax = float(np.max(corr))
        
        # Log-sum-exp for smooth max
        shifted = corr - cmax
        logsumexp = cmax + np.log(np.sum(np.exp(tau * shifted))) / tau
        obj = logsumexp * dx
        
        # Gradient of logsumexp objective
        # d/dh_i of logsumexp = (1/tau) * sum_k exp(tau*(corr_k - cmax)) * d(corr_k)/dh_i
        weights = np.exp(tau * shifted) / np.sum(np.exp(tau * shifted))
        
        # Compute gradient
        grad = np.zeros(n)
        for k in range(len(corr)):
            wk = weights[k]
            if wk < 1e-15:
                continue
            lag = k - (n - 1)
            if lag >= 0:
                # corr[k] = sum_{i=0}^{n-1-lag} h[i+lag] * (1-h[i])
                # d/dh_j: if j in [lag, n-1]: (1-h[j-lag])
                #          if j in [0, n-1-lag]: -h[j+lag]
                n1 = n - lag
                grad[lag:n] += wk * (1.0 - h[:n1])
                grad[:n1] -= wk * h[lag:]
            else:
                L = -lag
                # corr[k] = sum_{i=L}^{n-1} h[i] * (1-h[i-L])
                n1 = n - L
                grad[:n1] += wk * (1.0 - h[L:])
                grad[L:] -= wk * h[:n1]
        
        # Project gradient to sum-zero subspace
        grad -= np.mean(grad)
        
        # Take step
        h_new = h - step * grad
        h_new = project_constraint(h_new, n)
        c5_new = compute_c5(h_new)
        
        if c5_new < c5 - 1e-14:
            h = h_new
            c5 = c5_new
            if c5 < best_c5 - 1e-12:
                best_c5 = c5
                best_h = h.copy()
            step = min(step * 1.1, 0.1)
        else:
            step *= 0.7
            if step < 0.001:
                # Stuck, perturb
                h += rng.randn(n) * 0.01
                h = project_constraint(h, n)
                c5 = compute_c5(h)
                step = 0.01
        
        # Increase tau periodically
        if (int((time.time() - t0) * 2) % 100) == 0:
            tau = min(tau * 1.05, 500.0)
    
    return best_h, best_c5

def project_constraint(p, n):
