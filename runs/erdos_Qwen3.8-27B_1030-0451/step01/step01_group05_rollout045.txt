import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

def project_feasible(z, n):
    S = n / 2.0
    z = np.clip(z, -10.0, 10.0)
    lo, hi = float(np.min(z)) - 1.0, float(np.max(z)) + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-10:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_lin_quad(h, a, b):
    n = len(h)
    L = 2 * n - 1
    lin = np.zeros(L)
    omh = 1.0 - h
    hr = h[::-1]
    lin[n-1-a : 2*n-1-a] += omh
    lin[a : a+n] -= hr
    lin[n-1-b : 2*n-1-b] -= omh
    lin[b : b+n] += hr
    quad = np.zeros(L)
    quad[n-1] = 2.0
    kab = (n-1) + (b - a)
    if 0 <= kab < L and 0 <= 2*b - a < n:
        quad[kab] = -1.0
    kba = (n-1) + (a - b)
    if 0 <= kba < L and 0 <= 2*a - b < n:
        quad[kba] = -1.0
    return lin, quad

def find_best_d(raw, lin, quad, d_lo, d_hi, n):
    if d_hi - d_lo < 1e-12:
        return 0.0
    dx = 2.0 / n
    # Coarse grid
    ng = 31
    ds = np.linspace(d_lo, d_hi, ng)
    best_s = np.max(raw) * dx
    best_d = 0.0
    for d in ds:
        nr = raw + d * lin + d * d * quad
        s = np.max(nr) * dx
        if s < best_s:
            best_s = s
            best_d = d
    # Refine
    step = (d_hi - d_lo) / (ng - 1)
    for _ in range(3):
        lo2 = max(d_lo, best_d - step)
        hi2 = min(d_hi, best_d + step)
        if hi2 - lo2 < 1e-13:

            nr = raw + d * lin + d * d * quad
            s = np.max(nr) * dx
            if s < best_s:

                best_d = d
    return best_d

def compute_grad(h, raw, tau):
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    cmax = np.max(raw) * dx
    ex = np.clip((raw * dx - cmax) / tau, -400.0, 400.0)
    w = np.exp(ex)
    sw = np.sum(w)
    if sw < 1e-300:
        w = np.full(L, 1.0 / L)
    else:
        w /= sw
    omh = 1.0 - h
    hr = h[::-1]
    g = np.zeros(n)
    for i in range(n):
        g[i] = np.dot(w[n-1-i : 2*n-1-i], omh) - np.dot(w[i : n+i], hr)
    return g

    cur_raw = np.correlate(cur, 1.0 - cur, mode='full').copy()
    cur_score = float(np.max(cur_raw) * (2.0 / n))
    best = cur.copy()
    best_score = cur_score
    
    tau_list = [0.06, 0.04, 0.025, 0.015, 0.008, 0.004, 0.002, 0.001]
    total_moves = 0
    
    for tau in tau_list:
        if time.time() > deadline:
            break
        
        no_improve = 0
        max_no_improve = 100
        
        while no_improve < max_no_improve:
            if time.time() > deadline:
                break
            
            # Compute gradient
            g = compute_grad(cur, cur_raw, tau)
            
            # Generate candidate pairs
            pairs = []
            gmin = int(np.argmin(g))
            gmax = int(np.argmax(g))
            pairs.append((gmin, gmax))

            pairs.append((int(sidx[0]), int(sidx[-1])))
            pairs.append((int(sidx[1]), int(sidx[-2])))
            
            # Sorted pairs

            pairs.append((int(sidx[n//8]), int(sidx[7*n//8])))
            
            # Random pairs
            for _ in range(6):
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a != b:
                    pairs.append((a, b))
            
            # Try all pairs, find best
            best_move = None
            best_move_score = cur_score
            
            for (a, b) in pairs:
                if a == b:
                    continue
                ha, hb = cur[a], cur[b]
                d_lo = max(-ha, hb - 1.0)
                d_hi = min(1.0 - ha, hb)
                if d_hi - d_lo < 1e-12:
                    continue
                
                lin, quad = compute_lin_quad(cur, a, b)
                bd = find_best_d(cur_raw, lin, quad, d_lo, d_hi, n)
                if abs(bd) < 1e-13:
                    continue
                nr = cur_raw + bd * lin + bd * bd * quad
                ns = float(np.max(nr) * (2.0 / n))
                
                if ns < best_move_score - 1e-12:
                    best_move_score = ns
                    best_move = (a, b, bd, lin, quad)
            
            if best_move is not None:
                a, b, bd, lin, quad = best_move
                cur[a] += bd
                cur[b] -= bd
                cur_raw = cur_raw + bd * lin + bd * bd * quad
                cur_score = best_move_score
                no_improve = 0
                total_moves += 1
                
                if cur_score < best_score - 1e-13:

                    best_score = cur_score
                
                # Periodic recompute to prevent drift
                if total_moves % 20 == 0:
                    cur_raw = np.correlate(cur, 1.0 - cur, mode='full')
                    cur_score = float(np.max(cur_raw) * (2.0 / n))
            else:
                no_improve += 1
    
    return best, best_score

def sa_search(h0, rng, deadline):
    n = len(h0)
    cur = h0.copy()

    cur_score = float(np.max(cur_raw) * (2.0 / n))
    best = cur.copy()
    best_score = cur_score
    
    T = 0.02
    T_min = 0.0005
    cooling = 0.97
    
    iters = 0
    while time.time() < deadline:
        iters += 1
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        
        ha, hb = cur[a], cur[b]
        d_lo = max(-ha, hb - 1.0)
        d_hi = min(1.0 - ha, hb)
        if d_hi - d_lo < 1e-12:
            continue
        
        lin, quad = compute_lin_quad(cur, a, b)
        bd = find_best_d(cur_raw, lin, quad, d_lo, d_hi, n)
        if abs(bd) < 1e-13:
            continue
        
        nr = cur_raw + bd * lin + bd * bd * quad
        ns = float(np.max(nr) * (2.0 / n))
        
        delta = ns - cur_score
        if delta < -1e-13 or (T > T_min and rng.random() < np.exp(-delta / T)):
            cur[a] += bd
            cur[b] -= bd
            cur_raw = nr
            cur_score = ns
            if cur_score < best_score - 1e-13:
                best = cur.copy()
                best_score = cur_score
        
        T *= cooling
        if T < T_min:
            T = T_min
            # Reset to best and cool more
            cur = best.copy()
            cur_raw = np.correlate(cur, 1.0 - cur, mode='full')
            cur_score = best_score
    
    return best, best_score
