```python
def line_search_quadratic(h, d, c0, n):
    """Find best t along h + t*d using quadratic model of correlations."""
    dx = 2.0 / n
    g = (np.correlate(d, 1-h, 'full') - np.correlate(h, d, 'full')) * dx
    q = np.correlate(d, d, 'full') * dx
    # c(t) = c0 + t*g - t^2*q
    # Find t_max for feasibility
    pos = d > 1e-12
    neg = d < -1e-12
    t_max = np.inf
    if np.any(pos):
        t_max = min(t_max, np.min((1-h[pos]) / d[pos]))
    if np.any(neg):
        t_max = min(t_max, np.min(h[neg] / (-d[neg])))
    if not np.isfinite(t_max) or t_max <= 0:
        return 0.0, c0.max()
    t_max = min(t_max, 1.0)  # cap step size
    
    # Grid search
    n_grid = 80
    ts = np.linspace(0, t_max, n_grid + 1)
    c5_vals = np.zeros(n_grid + 1)
    for idx, t in enumerate(ts):
        c_t = c0 + t*g - t*t*q
        c5_vals[idx] = c_t.max()
    
    idx_best = np.argmin(c5_vals)
    t_best = ts[idx_best]
    c5_best = c5_vals[idx_best]
    
    # Refine around best
    if 0 < idx_best < n_grid:
        t_lo = ts[idx_best - 1]
        t_hi = ts[idx_best + 1]
        ts_fine = np.linspace(t_lo, t_hi, 40)
        for idx, t in enumerate(ts_fine):
            c_t = c0 + t*g - t*t*q
            if c_t.max() < c5_best:
                c5_best = c_t.max()
                t_best = t
    
    return t_best, c5_best
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    lo, hi = -3.0, 4.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def line_search_quad(h, d, c0, n):
    """Quadratic line search: c(t) = c0 + t*g - t^2*q"""
    dx = 2.0 / n
    g = (np.correlate(d, 1.0 - h, 'full') - np.correlate(h, d, 'full')) * dx
    q = np.correlate(d, d, 'full') * dx
    
    # Feasibility bound
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    t_max = 1.0  # default cap
    if np.any(pos_mask):
        t_max = min(t_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        t_max = min(t_max, float(np.min(h[neg_mask] / (-d[neg_mask]))))
    if t_max <= 0:
        return 0.0, c0.max()
    
    # Grid search
    n_grid = 100
    ts = np.linspace(0.0, t_max, n_grid + 1)
    best_t = 0.0
    best_val = c0.max()
    
    # Vectorized: for each t, compute max(c0 + t*g - t^2*q)
    # This is n_grid+1 evaluations of max of (2n-1)-vectors
    for idx in range(n_grid + 1):
        t = ts[idx]
        c_t = c0 + t * g - t * t * q
        val = c_t.max()
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine
    if 0 < best_t < t_max:
        # Find neighbors
        idx_best = int(round(best_t / t_max * n_grid))
        t_lo = ts[max(0, idx_best - 1)]
        t_hi = ts[min(n_grid, idx_best + 1)]
        ts_fine = np.linspace(t_lo, t_hi, 50)
        for t in ts_fine:
            c_t = c0 + t * g - t * t * q
            val = c_t.max()
            if val < best_val:
                best_val = val
                best_t = t
    
    return best_t, best_val

def compute_weights(c0, active_idx, w_type, rng):
    """Compute weights for active lags based on type."""
    c_act = c0[active_idx]
    c_max = c_act.max()
    k = len(active_idx)
    
    if w_type == 0:  # Uniform
        return np.ones(k) / k
    elif w_type == 1:  # Softmax tight
        tau = max(1e-4, (c_max - c_act.min()) * 0.1)
        e = np.exp(np.clip((c_act - c_max) / tau, -500, 0))
        return e / e.sum()
    elif w_type == 2:  # Softmax loose
        tau = max(1e-3, (c_max - c_act.min()) * 1.0)
        e = np.exp(np.clip((c_act - c_max) / tau, -500, 0))
        return e / e.sum()
    elif w_type == 3:  # Top-2 only
        if k >= 2:
            top2 = np.argsort(c_act)[-2:]
            w = np.zeros(k)
            w[top2] = 0.5
            return w
        return np.ones(k) / k
    elif w_type == 4:  # Random
        w = rng.exponential(1.0, size=k)
        return w / w.sum()
    else:  # Inverse uniform (penalize non-active)
        return np.ones(k) / k

def grad_weighted(h, active_idx, weights, n):
    """Compute gradient of weighted sum of active correlations."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, ell in enumerate(active_idx):
        w = weights[idx]
        if w < 1e-14:
            continue
        r = (n - 1) - ell
        if r >= 0:
            # Valid i: 0 to n-1-r, pairs (i, i+r)
            hi = n - 1 - r
            if hi >= 0:
                grad[0:hi+1] += w * dx * (1.0 - h[r:hi+1+r])  # d[i] contributes (1-h[i+r])
                grad[r:hi+1+r] -= w * dx * h[0:hi+1]  # h[i+r] contributes -h[i]
        else:
            a = -r
            hi = n - 1 - a
            if hi >= 0:
                grad[a:hi+1+a] += w * dx * (1.0 - h[0:hi+1])  # h[i-a] pair
                grad[0:hi+1] -= w * dx * h[a:hi+1+a]
    return grad

def optimize_active(h, n, deadline, rng):
    """Main optimization loop using active-lag directions."""
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    stall = 0
    
    while time.time() < deadline - 5:
        c0 = np.correlate(h, 1.0 - h, 'full') * dx
        cmax = c0.max()
        
        # Active set
        tol = max(1e-5, 0.03 * (cmax - c0.mean()))
        active = np.where(c0 >= cmax - tol)[0]
        if len(active) > 30:
            # Keep top 30
            active = active[np.argsort(c0[active])[-30:]]
        
        improved = False
        for w_type in range(5):
            if time.time() > deadline - 3:
                break
            weights = compute_weights(c0, active, w_type, rng)
            grad = grad_weighted(h, active, weights, n)
            
            # Project to tangent (sum zero)
            d = -(grad - grad.mean())
            d_norm = np.max(np.abs(d))
            if d_norm < 1e-14:
                continue
            d /= d_norm
            
            # Line search
            t_best, c5_try = line_search_quad(h, d, c0, n)
            
            if t_best > 1e-12 and c5_try < best_c5 - 1e-10:
                h_new = h + t_best * d
                h_new = project_feasible(h_new, n / 2.0)
                c5_exact = evaluate_c5(h_new)
                if c5_exact < best_c5 - 1e-10:
                    best_c5 = c5_exact
                    best_h = h_new.copy()
                    h = h_new
                    improved = True
        
        if not improved:
            stall += 1
            if stall >= 3:
                # Perturb
                amp = 0.005 * min(stall, 5)
                h = project_feasible(h + rng.standard_normal(n) * amp, n / 2.0)
                stall = 0
    
    return best_h, best_c5
```

```python
        t_max = min(t_max, float(np.min((1.0 - h[pos_mask]) / d[pos_mask])))
    if np.any(neg_mask):
        t_max = min(t_max, float(np.min(h[neg_mask] / (-d[neg_mask]))))
    if t_max <= 1e-14:
        return 0.0, c0.max()
    t_max = min(t_max, 1.5)

    n_grid = 80
    ts = np.linspace(0.0, t_max, n_grid + 1)
    best_t = 0.0
    best_val = c0.max()
    for idx in range(n_grid + 1):
        t = ts[idx]
        c_t = c0 + t * g - t * t * q
        val = c_t.max()
        if val < best_val - 1e-14:
            best_val = val
            best_t = t

    if best_t > 0 and best_t < t_max:
        idx_best = int(round(best_t / t_max * n_grid))
        t_lo = ts[max(0, idx_best - 1)]
        t_hi = ts[min(n_grid, idx_best + 1)]
        ts_fine = np.linspace(t_lo, t_hi, 40)
        for t in ts_fine:
            c_t = c0 + t * g - t * t * q
            val = c_t.max()
            if val < best_val - 1e-14:
                best_val = val
                best_t = t

    return best_t, best_val

def grad_active(h, active_idx, weights, n):
    dx = 2.0 / n
    grad = np.zeros(n)
    for idx, ell in enumerate(active_idx):
        w = weights[idx]
        if w < 1e-14:
            continue
        r = (n - 1) - ell
        if r >= 0:
            hi = n - 1 - r
            if hi >= 0:
                grad[r:r + hi + 1] += w * dx * (1.0 - h[0:hi + 1])
                grad[0:hi + 1] -= w * dx * h[r:r + hi + 1]
        else:
            a = -r
            hi = n - 1 - a
            if hi >= 0:
                grad[0:hi + 1] += w * dx * (1.0 - h[a:a + hi + 1])
                grad[a:a + hi + 1] -= w * dx * h[0:hi + 1]
    return grad

def make_weights(c0, active, w_type, rng):
    c_act = c0[active]
    c_max = c_act.max()
    k = len(active)
    if k == 0:
        return np.array([])
    spread = max(1e-6, c_max - c_act.min())
    if w_type == 0:
        return np.ones(k) / k
    elif w_type == 1:
        tau = max(1e-5, spread * 0.05)
        e = np.exp(np.clip((c_act - c_max) / tau, -500.0, 0.0))
        s = e.sum()
        return e / s if s > 0 else np.ones(k) / k
    elif w_type == 2:
        tau = max(1e-4, spread * 0.5)
        e = np.exp(np.clip((c_act - c_max) / tau, -500.0, 0.0))
        s = e.sum()
        return e / s if s > 0 else np.ones(k) / k
    elif w_type == 3:
        if k >= 2:
            top2 = np.argsort(c_act)[-2:]
            w = np.zeros(k)
            w[top2] = 0.5
            return w
        return np.ones(k) / k
    elif w_type == 4:
        w = rng.exponential(1.0, size=k)
        s = w.sum()
        return w / s if s > 0 else np.ones(k) / k
    elif w_type == 5:
        if k >= 3:
            top3 = np.argsort(c_act)[-3:]
            w = np.zeros(k)
            w[top3[0]] = 0.5
            w[top3[1]] = 0.3
            w[top3[2]] = 0.2
            return w
        return np.ones(k) / k
    elif w_type == 6:
        half = k // 2
        order = np.argsort(c_act)
        w = np.zeros(k)
        w[order[half:]] = 0.7 / (k - half)
        w[order[:half]] = 0.3 / max(1, half)
        return w
    else:
        return np.ones(k) / k

def optimize_at_n(h, n, deadline, rng, max_stall=50):
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    stall = 0
    tol_factor = 0.02
    n_outer = 0

    while time.time() < deadline - 3:
        c0 = np.correlate(h, 1.0 - h, 'full') * dx
        cmax = c0.max()
        cmean = c0.mean()

        tol = max(1e-5, tol_factor * (cmax - cmean))
        active = np.where(c0 >= cmax - tol)[0]
        if len(active) > 40:
            active = active[np.argsort(c0[active])[-40:]]
        if len(active) == 0:
            active = np.array([int(np.argmax(c0))])

        improved = False
        n_wtypes = min(7, 3 + (n_outer // 20) % 4)
        for w_type in range(n_wtypes):
            if time.time() > deadline - 2:
                break
            weights = make_weights(c0, active, w_type, rng)
            grad = grad_active(h, active, weights, n)
            d = -(grad - grad.mean())
            d_norm = np.max(np.abs(d))
            if d_norm < 1e-14:
                continue
            d /= d_norm

            t_best, c5_try = line_search_quad(h, d, c0, n)
            if t_best > 1e-10 and c5_try < best_c5 - 1e-11:
                h_new = h + t_best * d
                h_new = project_feasible(h_new, n / 2.0)
                c5_exact = evaluate_c5(h_new)
                if c5_exact < best_c5 - 1e-11:
                    best_c5 = c5_exact
                    best_h = h_new.copy()
                    h = h_new
                    improved = True

        n_outer += 1
        if not improved:
            stall += 1
            if stall >= max_stall:
                max_stall = int(max_stall * 0.7)
                amp = 0.003 * min(stall * 0.02, 3.0)
                h = project_feasible(h + rng.standard_normal(n) * amp, n / 2.0)
                tol_factor = max(0.005, tol_factor * 0.8)
                stall = 0
            elif stall >= 5:
                amp = 0.002 * min(stall * 0.1, 2.0)
                h = project_feasible(h + rng.standard_normal(n) * amp, n / 2.0)
                stall = 0
            elif stall >= 2:
                tol_factor = max(0.005, tol_factor * 0.7)

        if n_outer > 500:
            break

    return best_h, best_c5

def try_alternate_starts(h_parent, n_parent, deadline, rng):
    """Try a few alternate starting points."""
    results = []

    # Step function: h=1 on [0, 0.5] U [1.5, 2], 0 elsewhere
    n = n_parent
    h_step = np.zeros(n)
    q = n // 8
    h_step[:q] = 1.0
    h_step[7*q:8*q] = 1.0
    # Adjust to have sum = n/2
    current_sum = h_step.sum()
    target = n / 2.0
    if current_sum != target:
        diff = target - current_sum
        n_zero = int((h_step < 0.5).sum())
        if n_zero > 0:
            add_each = diff / n_zero
            zeros = np.where(h_step < 0.5)[0]
            h_step[zeros] = np.clip(h_step[zeros] + add_each, 0, 1)
        h_step = project_feasible(h_step, target)
    results.append(h_step)

    # Cosine: h(x) = 0.5 + 0.5*cos(2*pi*x)
    x = np.linspace(0, 2, n, endpoint=False)
    h_cos = 0.5 + 0.5 * np.cos(2 * np.pi * x)
    h_cos = project_feasible(h_cos, target)
    results.append(h_cos)

    # Triangular
    h_tri = np.ones(n) * 0.5
    half = n // 2
    ramp = np.linspace(0.5, 1.0, half)
    h_tri[:half] = ramp
    h_tri[half:] = ramp[::-1]
    h_tri = project_feasible(h_tri, target)
    results.append(h_tri)

    # Blended with parent
    for alpha in [0.3, 0.5, 0.7]:
        h_blend = (1 - alpha) * h_parent + alpha * results[0]
        h_blend = project_feasible(h_blend, target)
        results.append(h_blend)

    return results

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    rng = np.random.default_rng(seed)

    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=np.float64)
    else:
        n_fallback = 240
        x_f = np.linspace(0, 2, n_fallback, endpoint=False)
        h_parent = 0.5 + 0.3 * np.cos(2 * np.pi * x_f)
        h_parent = project_feasible(h_parent, n_fallback / 2.0)

    n_par = len(h_parent)
    h_parent = project_feasible(h_parent, n_par / 2.0)
    par_c5 = evaluate_c5(h_parent)

    best_h = h_parent.copy()
    best_c5 = par_c5
    best_n = n_par

    # Phase 0: Try alternate starts at parent resolution
    if time.time() < deadline - 100:
        alternates = try_alternate_starts(h_parent, n_par, deadline, rng)
        for alt in alternates:
            if time.time() > deadline - 80:
                break
            alt_c5 = evaluate_c5(alt)
            if alt_c5 < best_c5:
                best_c5 = alt_c5
                best_h = alt.copy()
                best_n = len(alt)

    # Phase 1: Optimize best start at parent resolution
    phase1_deadline = min(deadline, t0 + budget_s * 0.25)
    if time.time() < phase1_deadline:
        h_opt, c5_opt = optimize_at_n(best_h.copy(), best_n, phase1_deadline, rng, max_stall=30)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Resolution ladder
    n_candidates = []
    for mult in [2, 3, 4, 5, 6, 8, 10, 12]:
        n = n_par * mult
        if n <= 2400:
            n_candidates.append(n)
    n_candidates = sorted(set(n_candidates))

    for n_target in n_candidates:
        if time.time() > deadline - 30:
            break
        remaining = deadline - time.time()
        if remaining < 25:
            break

        h0 = interp_h(best_h, n_target)
        c5_0 = evaluate_c5(h0)

        # Interpolation might slightly worsen; allow small margin
        if c5_0 > best_c5 + 0.008:
            continue

        # Allocate time proportional to resolution
        phase_time = min(remaining * 0.55, max(30, 80 * n_target / 2400))
        phase_deadline = min(deadline - 20, time.time() + phase_time)

        h_opt, c5_opt = optimize_at_n(h0, n_target, phase_deadline, rng, max_stall=25)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_target

    # Phase 3: Final polish at best resolution
    n_best = len(best_h)
    if time.time() < deadline - 15:
        h_opt, c5_opt = optimize_at_n(best_h.copy(), n_best, deadline, rng, max_stall=15)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Try one more level up if time remains
    if time.time() < deadline - 20:
        n_next = n_best + n_par
        if n_next <= 2400:
            h0 = interp_h(best_h, n_next)
            c5_0 = evaluate_c5(h0)
            if c5_0 < best_c5 + 0.003:
                h_opt, c5_opt = optimize_at_n(h0, n_next, deadline, rng, max_stall=10)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                    n_next_actual = n_next

    # Final feasibility repair
    n_final = len(best_h)
    best_h = project_feasible(best_h, n_final / 2.0)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = n_final / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] += diff / cnt
            np.clip(best_h, 0.0, 1.0, out=best_h)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, n_final)
```
