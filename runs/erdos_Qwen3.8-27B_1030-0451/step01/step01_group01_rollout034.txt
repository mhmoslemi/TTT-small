    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    x = np.array(x, dtype=float).copy()

    elif n_ones < target:
        idx = np.where(h == 0.0)[0]
        rng.shuffle(idx)
        h[idx[:target - n_ones]] = 1.0
    return h

def binary_sa(h_start, deadline, max_iters=2000000, seed=42, T_init=1.0):
    rng = np.random.RandomState(seed)
    n = len(h_start)
    dx = 2.0 / n
    h = make_binary(h_start, n, seed=seed)
    best_h = h.copy()
    best_val = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
    curr_val = best_val
    T = T_init
    T_min = 0.0005
    cool_rate = 0.9997
    it = 0
    stall = 0
    stall_limit = 8000
    while time.time() < deadline and it < max_iters:
        batch = 200
        for _ in range(batch):
            if time.time() > deadline:
                break
            it += 1
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            v = np.max(np.correlate(h, 1.0 - h, mode="full"))
            new_val = v * dx
            if new_val < curr_val:
                curr_val = new_val
                if new_val < best_val - 1e-15:
                    best_val = new_val
                    best_h = h.copy()
                    stall = 0
                else:
                    stall += 1
            else:
                delta = new_val - curr_val
                if delta < 10.0 * T and rng.random() < np.exp(-delta / T):
                    curr_val = new_val
                else:
                    h[i] = 1.0
                    h[j] = 0.0
        T = max(T * cool_rate, T_min)
        if stall > stall_limit:
            if T > T_min * 5:
                T = min(T * 2.5, 0.8)
            else:
                h = best_h + (rng.rand(n) - 0.5) * 0.3
                h = make_binary(h, n, seed=it)
                curr_val = float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx
                stall = 0
    return best_h, best_val

def lp_trust_region(h0, deadline, max_iters=400, init_radius=0.10):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = init_radius
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        gap = max(0.015 * vmax, 0.003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 100:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-100:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-30:]
        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)
            b_lin[i] = A[i] @ h - v[k]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [2.0]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        improved = False
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                improved = True
                if it % 10 == 0:
                    radius = min(radius * 1.15, 0.3)
            else:
                no_improve += 1
        else:
            no_improve += 1
        if not improved:
            if no_improve >= 10:
                radius = max(radius * 0.5, 0.002)
                no_improve = 0
                if radius <= 0.002:
                    break
    return best_h, best_c

def projected_grad_opt(h0, deadline, max_iters=5000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    step = 0.003
    beta = 30.0
    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        m = v.max()
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 1e-12:
                lag = k - (n - 1)
                g += w[k] * grad_lag(h, lag)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-14:
            break
        c_cur = evaluate_c5(h)
        if c_cur < best_c - 1e-13:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        h_new = h - step * g
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < c_cur:
            h = h_new
            step = min(step * 1.03, 0.02)
        else:
            step *= 0.85
            if step < 1e-7:
                break
    return best_h, best_c

def interpolate_h(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_ext = np.concatenate([h0, [h0[0]]])
    x_ext = np.concatenate([x_old, [x_old[0] + 1.0]])
    h_new = np.interp(x_new, x_old, h0, left=h0[-1], right=h0[0])
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)

    n_init = len(h_init)
    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: Binary SA from initial (n_init)
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.18)
        try:
            hb, cb = binary_sa(h_init, dl, max_iters=1500000, seed=seed, T_init=0.8)
            if cb < best_c:
                best_h = hb
                best_c = cb
        except Exception:
            pass

    # Phase 2: Binary SA from Thue-Morse (n_init)
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.30)
        try:
            tm = thue_morse_binary(n_init)
            hb, cb = binary_sa(tm, dl, max_iters=1000000, seed=seed + 1, T_init=1.2)
            if cb < best_c:
                best_h = hb
                best_c = cb
        except Exception:
            pass

    # Phase 3: LP trust region on best solution
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.48)
        try:
            h_sm = project_sum_box(best_h + 0.02, n_init / 2.0)
            h_sm = np.clip(h_sm, 0.05, 0.95)
            h_sm = project_sum_box(h_sm, n_init / 2.0)
            hc, cc = lp_trust_region(h_sm, dl, max_iters=300, init_radius=0.12)
            if cc < best_c:
                best_h = hc
                best_c = cc
        except Exception:
            pass

    # Phase 4: Projected gradient polish
    if time_left() > 40:
        dl = min(deadline, start_time + budget_s * 0.58)
        try:
            hp, cp = projected_grad_opt(best_h, dl, max_iters=3000)
            if cp < best_c:
                best_h = hp
                best_c = cp
        except Exception:
            pass

    # Phase 5: Try larger resolution n=168
    if time_left() > 120:
        n_big = 168
        h_big = interpolate_h(best_h, n_big)
        c_big = evaluate_c5(h_big)
        if c_big < best_c:
            best_h = h_big
            best_c = c_big
        # Binary SA on big grid
        if time_left() > 80:
            dl = min(deadline, start_time + budget_s * 0.78)
            try:
                hb, cb = binary_sa(h_big, dl, max_iters=800000, seed=seed + 2, T_init=1.0)
                if cb < best_c:
                    best_h = hb
                    best_c = cb
            except Exception:
                pass
        # LP on big grid
        if time_left() > 40:
            dl2 = min(deadline, start_time + budget_s * 0.90)
            try:
                h_sm2 = project_sum_box(best_h, n_big / 2.0)
                hc, cc = lp_trust_region(h_sm2, dl2, max_iters=200, init_radius=0.08)
                if cc < best_c:
                    best_h = hc
                    best_c = cc
            except Exception:
                pass

    # Phase 6: Final LP + projected gradient on best
    if time_left() > 30:
        n_cur = len(best_h)
        dl = min(deadline - 2, start_time + budget_s * 0.97)
        try:
            hf, cf = lp_trust_region(best_h, dl, max_iters=200, init_radius=0.06)
            if cf < best_c:
                best_h = hf
                best_c = cf
        except Exception:
            pass

    if time_left() > 10:
        n_cur = len(best_h)
        try:
            hf, cf = projected_grad_opt(best_h, deadline - 1, max_iters=2000)
            if cf < best_c:
                best_h = hf
                best_c = cf
        except Exception:
            pass

    # Final: also try binary at current resolution
    if time_left() > 5:
        n_cur = len(best_h)
        hb = make_binary(best_h, n_cur, seed=seed + 99)
        cb = float(np.max(np.correlate(hb, 1.0 - hb, mode="full"))) * (2.0 / n_cur)
        if cb < best_c:
            best_h = hb
            best_c = cb

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
