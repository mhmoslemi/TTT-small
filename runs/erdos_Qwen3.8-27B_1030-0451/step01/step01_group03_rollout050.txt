    """Compute gradient of softmax objective.
    h: array of length n
    alpha: array of length 2n-1 (weights for each correlation lag)
    Returns: gradient array of length n
    """

    grad = np.zeros(n)

    # Contribution: alpha[k] * [(1-h[j+d]) for valid j+d, -h[j-d] for valid j-d]

from scipy.optimize import minimize
from scipy.special import logsumexp
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -3.0, 4.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
    np.clip(h, 0.0, 1.0, out=h)
    # Final fix
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return project_feasible(hp.copy(), nn * 0.5)
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn * 0.5)

def c5_obj_and_grad(h, tau, n, dx):
    """Softmax objective and its gradient."""
    c = np.correlate(h, 1.0 - h, 'full')
    c_scaled = c * dx
    m = c_scaled.max()
    shifted = (c_scaled - m) / tau
    shifted = np.clip(shifted, -40.0, 40.0)
    log_val = logsumexp(shifted)
    F = m + tau * log_val
    
    alpha = np.exp(shifted - log_val)
    
    # Gradient
    grad = np.zeros(n)
    active = alpha > 1e-10
    for k in np.where(active)[0]:
        a = alpha[k]
        d = int(k) - (n - 1)
        if d >= 0:
            if n - d > 0:
                grad[0:n-d] += a * (1.0 - h[d:n])
                grad[d:n] -= a * h[0:n-d]
        else:
            dd = -d
            if n - dd > 0:
                grad[dd:n] += a * (1.0 - h[0:n-dd])
                grad[0:n-dd] -= a * h[dd:n]
    grad *= dx
    return F, grad

def c5_obj(h):
    tau = c5_obj._tau
    n = c5_obj._n
    dx = c5_obj._dx
    F, _ = c5_obj_and_grad(h, tau, n, dx)
    return F

def c5_grad(h):
    tau = c5_obj._tau
    n = c5_obj._n
    dx = c5_obj._dx
    _, g = c5_obj_and_grad(h, tau, n, dx)
    return g

def c5_eq_fun(h):
    return float(h.sum() - c5_obj._target)

def c5_eq_jac(h):

def solve_at_resolution(h0, n, deadline, min_time_remaining=5):

    global_best_h = h0.copy()
    global_best_c5 = evaluate_c5(h0)
    target = n * 0.5
    dx = 2.0 / n
    
    # Set up SLSQP
    constraints = [{'type': 'eq', 'fun': c5_eq_fun, 'jac': c5_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    # Temperature schedule

import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp
import time

# Module-level parameters for objective/gradient callbacks
_P_TAU = 0.01
_P_N = 0
_P_DX = 0.0
_P_TARGET = 0.0
_P_JAC = None

def _obj_cb(h):
    n = _P_N
    dx = _P_DX
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    sh = np.clip((c - m) / _P_TAU, -40.0, 40.0)
    lv = logsumexp(sh)
    return float(m + _P_TAU * lv)

def _grad_cb(h):
    n = _P_N
    dx = _P_DX
    tau = _P_TAU
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    sh = np.clip((c - m) / tau, -40.0, 40.0)
    lv = logsumexp(sh)
    alpha = np.exp(sh - lv)
    grad = np.zeros(n)
    active_idx = np.where(alpha > 1e-11)[0]
    for k in active_idx:
        a = alpha[int(k)]
        d = int(k) - (n - 1)
        if d >= 0:
            if n - d > 0:
                grad[0:n-d] += a * (1.0 - h[d:n])
                grad[d:n] -= a * h[0:n-d]
        else:
            dd = -d
            if n - dd > 0:
                grad[dd:n] += a * (1.0 - h[0:n-dd])
                grad[0:n-dd] -= a * h[dd:n]
    return grad * dx

def _eq_fun(h):
    return float(h.sum() - _P_TARGET)

def _eq_jac(h):
    return _P_JAC

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -3.0, 4.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    diff = target - h.sum()
    if abs(diff) > 1e-15:
        h += diff / len(h)
        np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return project_feasible(hp.copy(), nn * 0.5)
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn * 0.5)

def _setup_params(n, tau):
    global _P_TAU, _P_N, _P_DX, _P_TARGET, _P_JAC
    _P_TAU = tau
    _P_N = n
    _P_DX = 2.0 / n
    _P_TARGET = n * 0.5
    _P_JAC = np.ones(n)

def optimize_slsqp(h0, n, deadline, min_remain=8):
    target = n * 0.5
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    c0 = np.correlate(h0, 1.0 - h0, 'full') * (2.0 / n)
    spread = max(1e-5, c0.max() - c0.min())

    # Two-phase temperature schedule
    tau_start = max(0.03, 0.5 * spread)
    phase1_ratio = 0.7
    n1 = 12
    phase2_start = tau_start * (phase1_ratio ** n1)
    tau_end = max(1e-5, phase2_start * 0.02)
    n2 = 15
    if phase2_start > tau_end:
        phase2_ratio = (tau_end / phase2_start) ** (1.0 / n2)
    else:
        phase2_ratio = 0.7
    taus_p1 = tau_start * (phase1_ratio ** np.arange(n1))
    if phase2_start > tau_end:
        taus_p2 = phase2_start * (phase2_ratio ** np.arange(n2))
    else:
        taus_p2 = np.full(n2, phase2_start * 0.9)

    h_curr = h0.copy()
    all_taus = np.concatenate([taus_p1, taus_p2])

    for i, tau in enumerate(all_taus):
        if time.time() > deadline - min_remain:
            break
        _setup_params(n, tau)
        if tau > 0.005:
            maxiter = 50
        elif tau > 0.001:
            maxiter = 100
        else:
            maxiter = 150
        try:
            res = minimize(_obj_cb, h_curr, method='SLSQP', jac=_grad_cb,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': maxiter, 'ftol': 1e-14})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        c5v = evaluate_c5(h_curr)
        if c5v < best_c5 - 1e-13:
            best_c5 = c5v
            best_h = h_curr.copy()

    return best_h, best_c5

def pgd_refine(h0, n, deadline, min_remain=5):
    """Projected gradient descent with Nesterov momentum as a polish step."""
    target = n * 0.5
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h_curr = h0.copy()
    z = h0.copy()
    momentum = 0.0
    lr = 1.0 / (10.0 * dx)
    iters = min(300, int((deadline - time.time() - min_remain) * 8))
    iters = max(iters, 30)

    for it in range(iters):
        if time.time() > deadline - min_remain:
            break
        yy = z + momentum * (z - h_curr)
        tau = 0.002
        _setup_params(n, tau)
        g = _grad_cb(yy)
        # Gradient step
        new_h = project_feasible(yy - lr * g, target)
        # Check improvement
        c5_new = evaluate_c5(new_h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
        # Update momentum (Nesterov)
        momentum = (1.0 + it + 1) / (it + 3) * 0.9
        z = h_curr
        h_curr = new_h

    return best_h, best_c5

def make_perturbed_start(parent, n, rng, amp):
    h0 = interp_h(parent, n)
    pert = rng.standard_normal(n) * amp
    return project_feasible(h0 + pert, n * 0.5)

def make_sine_start(n, freq):
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5 + (1.0 / freq) * np.sin(np.pi * freq * x / 2.0)
    h = np.clip(h, 0.0, 1.0)
    return project_feasible(h, n * 0.5)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)

    parent = project_feasible(parent, len(parent) * 0.5)
    rng = np.random.default_rng(seed)

    best_h = parent.copy()
    best_c5 = evaluate_c5(parent)
    best_n = len(parent)

    # Phase A: optimize at n=480 (fast, multiple starts)
    n_a = 480
    if time.time() < deadline - 60:
        h0 = interp_h(parent, n_a)
        h_opt, c5_opt = optimize_slsqp(h0, n_a, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_a

    # Phase B: try perturbation at same resolution
    if time.time() < deadline - 50:
        n_a = best_n if best_n != len(parent) else 480
        for amp in [0.008, 0.015]:
            if time.time() > deadline - 30:
                break
            h0 = make_perturbed_start(parent, n_a, rng, amp)
            h_opt, c5_opt = optimize_slsqp(h0, n_a, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_a

    # Phase C: try higher resolution n=720
    n_c = 720
    if time.time() < deadline - 80:
        h0 = interp_h(best_h, n_c)
        h_opt, c5_opt = optimize_slsqp(h0, n_c, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_c

    # Phase D: PGD polish on best solution
    if time.time() < deadline - 10:
        n_d = best_n
        h_opt, c5_opt = pgd_refine(best_h, n_d, deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final feasibility cleanup
    best_n = len(best_h)
    target = best_n * 0.5
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-7) & (best_h < 1.0 - 1e-7)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] -= diff / cnt
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = target - best_h.sum()
    if abs(diff) > 1e-14:
        best_h += diff / best_n
        np.clip(best_h, 0.0, 1.0, out=best_h)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
