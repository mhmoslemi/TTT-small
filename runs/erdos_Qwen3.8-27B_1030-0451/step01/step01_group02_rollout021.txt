def project_feasible(y, target):

                err += rem
                if err >= 0: break

def _project(y, target):

    n = y.size
    if n == 0:
        return y
    target = float(target)
    target = max(0.0, min(float(n), target))
    lo = -5.0
    hi = 5.0
    for _ in range(100):

        if s > target:

    x = np.clip(y + 0.5*(lo+hi), 0.0, 1.0)
    err = target - float(x.sum())
    if abs(err) > 1e-12:

            order = np.argsort(room, kind='stable')[::-1]
            for idx in order:
                add = min(float(room[idx]), err)
                x[idx] += add
                err -= add
                if err <= 0:
                    break
        else:
            order = np.argsort(x, kind='stable')
            for idx in order:
                rem = min(float(x[idx]), -err)
                x[idx] -= rem
                err += rem
                if err >= 0:

def _corr_dx(h):
    n = len(h)
    dx = 2.0/float(n)
    return np.correlate(h, 1.0-h, 'full')*dx

    c = _corr_dx(h)
    return float(np.max(c))

    n = len(h)
    dx = 2.0/float(n)
    c = _corr_dx(h)
    # softmax weights
    cm = float(np.max(c))
    a = (c - cm)/tau
    e = np.exp(a)
    s = float(np.sum(e))
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w = e/s
    # w indexed 0..2n-2, corresponding to d = k-(n-1)
    # grad[p] = dx * sum_d w[d+ n-1] * [1_{0<=p-d<=n-1}(1-h[p-d]) - 1_{0<=p+d<=n-1}(h[p+d])]
    # A[p] = sum_q w[n-1+p-q] * (1-h[q])
    # B[p] = sum_q w[n-1+(q-p)] * h[q]

    # B = convolve(h, w[::-1], 'full') indexed at [n-1+p]
    A_full = np.convolve(1.0-h, w, 'full')
    B_full = np.convolve(h, w[::-1], 'full')
    A = A_full[n-1:n-1+n]
    B = B_full[n-1:n-1+n]
    return dx*(A - B)
