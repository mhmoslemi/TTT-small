    return float(np.sum(h) - _G_T)

def _c_eq_jac(h):
    return np.ones(_G_N)

def _c_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _c_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    active = np.where(alpha > 1e-10)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi + 1] += a * omh[lo - d:hi + 1 - d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= a * h[lo2 + d:hi2 + 1 + d]
    return grad * dx

    return float(np.max(c) * dx)

def repair(h, target):

    lo, hi = -2.0, 3.0
