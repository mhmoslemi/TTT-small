    tau = min(0.015, max(0.005, 0.2 * (c_vals.max() - c_vals.mean())))
    
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n
    stall = 0
    
    for rnd in range(15):
        if time.time() > deadline - 5:
            break
        if tau < min_tau:
            break
        
        _G_TAU = tau
        maxiter = int(max(50, min(150, 5000.0 / n * 30)))
        try:
            res = minimize(_obj_func, h_curr, method='SLSQP', jac=_grad_func,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-15})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        
        c5_now = evaluate_c5(h_curr)
        if c5_now < best_c5 - 1e-11:
            best_c5 = c5_now
            best_h = h_curr.copy()
            stall = 0
            tau *= 0.7
        else:
            stall += 1
            tau *= 0.85
            if stall >= 3:
                rng = np.random.default_rng((n * 31337 + rnd * 6961) & 0xFFFFFFFF)
                h_curr = project_feasible(h_curr + rng.standard_normal(n) * 0.004, target)
                stall = 0
    
    return best_h, best_c5

def make_starts(parent_n_points, n, rng):
    """Generate diverse starting points at resolution n."""
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    base = np.interp(x, np.linspace(0.0, 2.0, parent_n_points, endpoint=False),
                     np.full(parent_n_points, 0.5))
    base = project_feasible(base, n / 2.0)
    
    starts = [base]
    
    # Smoothed step-like functions with different transition widths
    for width in [0.05, 0.1, 0.15]:
        t = (x - 1.0) / width
        h = 1.0 / (1.0 + np.exp(t))
        h = project_feasible(h, n / 2.0)
        starts.append(h)
    
    # Multi-block
    for nblocks in [2, 3]:
        block_len = 2.0 / nblocks
        h = np.zeros(n)
        for i in range(nblocks):
            center = (i + 0.5) * block_len
            h[(x > center - block_len * 0.4) & (x < center + block_len * 0.4)] = 1.0
        h = project_feasible(h, n / 2.0)
        starts.append(h)
    
    return starts

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 10
    
    # Get parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)
    parent = project_feasible(parent, len(parent) / 2.0)
    
    par_c5 = evaluate_c5(parent)
    best_h = parent.copy()
    best_c5 = par_c5
    best_n = len(parent)
    
    rng = np.random.default_rng(seed)
    
    # Phase 1: Optimize parent at its native resolution
    if time.time() < deadline - 60:
        h_opt, c5_opt = optimize_slsqp(parent.copy(), len(parent), deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 2: Resolution sweep - interpolate and optimize at higher resolutions
    n_list = [360, 480, 600, 720, 960, 1200]
    
    for n in n_list:
        if time.time() > deadline - 45:
            break
        remaining = deadline - time.time()
        if remaining < 25:
            break
        
        h_interp = interp_h(best_h, n)
        c5_interp = evaluate_c5(h_interp)
        
        # Try the interpolated parent
        if time.time() < deadline - 20:
            h_opt, c5_opt = optimize_slsqp(h_interp, n, deadline)
            # Subgradient polish
            if time.time() < deadline - 10:
                h_sub, c5_sub = subgradient_refine(h_opt, n, deadline, n_steps=400)
                if c5_sub < c5_opt:
                    h_opt = h_sub
                    c5_opt = c5_sub
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
        
        # Try alternative starting points at this resolution
        if time.time() < deadline - 50:
            starts = make_starts(n, n, rng)
            for h0 in starts:
                if time.time() > deadline - 30:
                    break
                c5_s = evaluate_c5(h0)
                # Only pursue if potentially competitive
                if c5_s < best_c5 + 0.05:
                    h_opt, c5_opt = optimize_slsqp(h0, n, deadline)
                    if time.time() < deadline - 10:
                        h_sub, c5_sub = subgradient_refine(h_opt, n, deadline, n_steps=300)
                        if c5_sub < c5_opt:
                            h_opt = h_sub
                            c5_opt = c5_sub
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt
                        best_n = n
    
    # Phase 3: Sharper refinement on the best solution
    if time.time() < deadline - 15:
        h_ref, c5_ref = weighted_slsqp(best_h, best_n, deadline)
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref
    
    # Phase 4: Final subgradient polish
    if time.time() < deadline - 8:
        h_sf, c5_sf = subgradient_refine(best_h, best_n, deadline, n_steps=600, top_k=5)
        if c5_sf < best_c5:
            best_c5 = c5_sf
            best_h = h_sf
    
    # Feasibility cleanup
    n_final = len(best_h)
    diff = n_final / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] -= diff / c
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = n_final / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] += diff / c
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_final)
