import time
import math
import numpy as np

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def repair(h, target):
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    d = h.sum() - target
    if abs(d) < 1e-13:
        return h
    lo = -1.0
    hi = 1.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h + t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    h = np.clip(h + t, 0.0, 1.0)
    d = h.sum() - target
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.where(h > 0.01)[0]

                h[idx2] = float(np.clip(h[idx2] - d, 0.0, 1.0))
        else:
            idx = np.where(h < 0.99)[0]
            if len(idx) > 0:
                idx2 = idx[int(np.argmin(h[idx]))]
                h[idx2] = float(np.clip(h[idx2] - d, 0.0, 1.0))
    return h

def corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def c5_of(h):
    n = len(h)
    return float(corr(h).max() * (2.0 / n))

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return repair(np.interp(x1, x0, h), n_new / 2.0)

def lse_h(h, sigma):
    n = len(h)
    c = corr(h)
    m = c.max()
    return float((m + math.log(np.exp(sigma * (c - m)).sum()) / sigma) * (2.0/n))

def lse_grad(h, sigma):
    """Compute gradient of LSE objective w.r.t. h using the correlation structure."""
    n = len(h)
    dx = 2.0 / n
    c = corr(h)
    center = n - 1
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()  # weights for each lag
    
    grad = np.zeros(n)
    # Lag 0 (center): c = sum h[i]*(1-h[i]), grad = (1-2h) * w[center]
    grad += w[center] * (1.0 - 2.0 * h)
    
    # For lag m > 0 (index center + m in the correlation array):
    # c_{center+m} = sum_{i=m}^{n-1} h[i]*(1-h[i-m])
    # d/dh[j] = [j>=m]*(1-h[j-m]) - [j<n-m]*(h[j+m])
    for lag in range(1, n):
        wp = w[center + lag]
        wm = w[center - lag] if center - lag >= 0 else 0.0
        if wp < 1e-15 and wm < 1e-15:
            continue
        # Positive lag contribution
        if wp > 1e-15:
            # h[j] as first factor (j >= lag): contributes (1-h[j-lag])
            grad[lag:] += wp * (1.0 - h[:-lag])
            # h[j] in second factor (j < n-lag): contributes -h[j+lag]
            grad[:-lag] -= wp * h[lag:]
        # Negative lag contribution
        if wm > 1e-15:
            # c_{center-lag} = sum_{i=0}^{n-1-lag} h[i]*(1-h[i+lag])
            # d/dh[j] = [j < n-lag]*(1-h[j+lag]) - [j >= lag]*(h[j-lag])
            grad[:-lag] += wm * (1.0 - h[lag:])
            grad[lag:] -= wm * h[:-lag]
    
    grad *= dx
    return grad

def lse_polish(h, rng, time_limit, sigma, target, deadline=None):
    h = np.asarray(h, dtype=float).copy()
    best_h = h.copy()
    best_c5 = c5_of(h)
    cur_lse = lse_h(h, sigma)
    eta = 0.05
    t0 = time.monotonic()
    stall = 0
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = lse_grad(h, sigma)
        gn = float(np.linalg.norm(g))
        if gn < 1e-12:
            break
        improved = False
        e = eta
        for _ in range(25):
            cand = repair(h - e * g, target)
            cv = lse_h(cand, sigma)
            if cv < cur_lse - 1e-14:
                h = cand
                cur_lse = cv
                c5 = c5_of(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            e *= 0.5
        if not improved:
            h = repair(h + rng.normal(0.0, 0.01, len(h)), target)
            cur_lse = lse_h(h, sigma)
            c5 = c5_of(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
            stall += 1
            if stall >= 8:
                eta *= 0.5
                stall = 0
                if eta < 1e-7:
                    break
    c5 = c5_of(h)
    if c5 < best_c5 - 1e-13:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5

def greedy_local_search(h, rng, time_limit, deadline=None, batch_size=50):
    """Local search using pair swaps and small transfers."""
    h = np.asarray(h, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    c = corr(h)
    cur_max = float(c.max())
    best_h = h.copy()
    best_max = cur_max
    t0 = time.monotonic()
    no_improve = 0
    
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        best_delta = 0
        best_pair = None
        best_type = None
        
        # Generate random candidate pairs
        for _ in range(batch_size):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            di = h[j] - h[i]
            if abs(di) < 1e-9:
                continue
            
            # Try swap
            swap_max = 0
            for trial in range(2):
                if trial == 0:
                    # Full swap
                    new_h = h.copy()
                    new_h[i], new_h[j] = h[j], h[i]
                    new_c = corr(new_h)
                    trial_max = float(new_c.max())
                else:
                    # Partial: move half the difference
                    new_h = h.copy()
                    new_h[i] += 0.5 * di
                    new_h[j] -= 0.5 * di
                    new_h = np.clip(new_h, 0.0, 1.0)
                    # Fix sum (should be fine since we moved equal amounts)
                    trial_max = float(corr(new_h).max())
                
                if trial_max < cur_max - 1e-12:
                    if trial_max < swap_max or swap_max == 0:
                        swap_max = trial_max
                        if best_delta == 0 or trial_max < cur_max + best_delta:
                            best_delta = trial_max - cur_max
                            best_pair = (i, j)
                            best_type = trial
        
        if best_pair is not None:
            i, j = best_pair
            h_copy = h.copy()
            if best_type == 0:
                h_copy[i], h_copy[j] = h[j], h[i]
            else:
                di = h[j] - h[i]
                h_copy[i] += 0.5 * di
                h_copy[j] -= 0.5 * di
            h = np.clip(h_copy, 0.0, 1.0)
            h = repair(h, target)
            cur_max = float(corr(h).max())
            no_improve = 0
            if cur_max < best_max - 1e-13:
                best_max = cur_max
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve >= 20:
                # Add noise to escape
                h = repair(h + rng.normal(0, 0.05, n), target)
                cur_max = float(corr(h).max())
                no_improve = 0
                if cur_max < best_max - 1e-13:
                    best_max = cur_max
                    best_h = h.copy()
                if no_improve > 100:
                    break
    
    c5 = c5_of(best_h)
    return best_h, c5
