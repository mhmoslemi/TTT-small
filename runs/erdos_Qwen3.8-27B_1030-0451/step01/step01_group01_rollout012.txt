def project_sum_box(x, target):
    """Robustly project x onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    # Bisection on shift
    lo, hi = -1.5, 1.5
    for _ in range(60):
        mid = (lo + hi) / 2
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2, 0.0, 1.0)
    # Fix residual
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-15:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-15:

    # Final fix
