    return float(np.max(corr) * dx)

    N = len(h)

    corr = np.correlate(h, 1.0 - h, mode='full')

def smooth_obj_and_weights(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    w_norm = w / total
    obj = m + tau * np.log(total)
    return obj, w_norm

def smooth_grad(h, weights):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = weights[r] * dx
        if a < 1e-15:

        d = r - (N - 1)
        if d >= 0:

            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return g

import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, tau, penalty=10.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    obj = m + tau * np.log(total)
    s = np.sum(h) - N / 2.0
    obj += penalty * s * s
    return obj

def smooth_grad(h, tau, penalty=10.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w = w / w.sum()
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    s = np.sum(h) - N / 2.0
    g += 2.0 * penalty * s
    return g

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h

def solve_1d_max(A, B):
    lo, hi = 0.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        slope = A[np.argmax(A * mid + B)]
        if slope > 0:
            hi = mid
        elif slope < 0:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5

def coord_descent(h, n_sweeps, s_all):
    N = len(h)
    h = h.copy()
    c_full = np.correlate(h, 1.0 - h, mode='full')
    for _ in range(n_sweeps):
        for i in range(N):
            idx1 = i + s_all
            mask1 = (idx1 >= 0) & (idx1 < N)
            idx2 = i - s_all
            mask2 = (idx2 >= 0) & (idx2 < N)
            A = np.zeros(2 * N - 1)
            A[mask1] = 1.0 - h[idx1[mask1]]
            A[mask2] -= h[idx2[mask2]]
            B = c_full - A * h[i]
            x_opt = solve_1d_max(A, B)
            delta = x_opt - h[i]
            if abs(delta) > 1e-12:
                h[i] = x_opt
                c_full += A * delta
    return h

        bounds = [(0.0, 1.0)] * N
        res = minimize(

            options={'maxiter': 200, 'ftol': 1e-16, 'gtol': 1e-8}

        h_new = project_feasible(res.x, N / 2.0)
        c5 = compute_c5(h_new)

Actually, for a convex piecewise linear function:
- If f is differentiable at mid (unique argmax) and f'(mid) > 0, then the minimum is at or to the left of mid.
- If f is differentiable at mid and f'(mid) < 0, then the minimum is at or to the right of mid.
- If f is not differentiable at mid (multiple argmax) and 0 is in the subgradient, then mid is a minimizer.

- The invariant is: there exists a minimizer in [lo, hi]
- Initially [0, 1] contains a minimizer

- If slope < 0 at mid: f is decreasing at mid, so any minimizer is ≥ mid → [mid, hi] contains a minimizer ✓
- If slope = 0: mid is a minimizer (or in the set of minimizers) ✓

def compute_c5(h):

def smooth_obj(h, tau, penalty=10.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)

    total = w.sum()
    obj = m + tau * np.log(total)
    s = np.sum(h) - N / 2.0
    obj += penalty * s * s
    return obj


def smooth_grad(h, tau, penalty=10.0):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx

    for _ in range(50):
        mid = (lo + hi) * 0.5
        slope = A[np.argmax(A * mid + B)]
        if slope > 0:
            hi = mid
        elif slope < 0:
            lo = mid
        else:
            break
    return (lo + hi) * 0.5


def solve_1d_subproblem(A, B, quad_B):
    """Solve the 1D subproblem including the quadratic zero-lag term."""
    # First solve the linear-only problem
    x_lin = solve_1d_bisection(A, B)
    
    # Check if the quadratic term is active
    f_lin = np.max(A * x_lin + B)
    f_quad = x_lin - x_lin * x_lin + quad_B
    
    if f_quad > f_lin + 1e-10:
        # Quadratic term dominates; do grid search
        x_grid = np.linspace(0, 1, 200)
        f_grid = np.max(A[None, :] * x_grid[:, None] + B[None, :], axis=1)
        f_grid = np.maximum(f_grid, x_grid - x_grid ** 2 + quad_B)
        idx = np.argmin(f_grid)
        x_lo = x_grid[max(0, idx - 1)]
        x_hi = x_grid[min(199, idx + 1)]
        x_fine = np.linspace(x_lo, x_hi, 40)
        f_fine = np.max(A[None, :] * x_fine[:, None] + B[None, :], axis=1)
        f_fine = np.maximum(f_fine, x_fine - x_fine ** 2 + quad_B)
        return x_fine[np.argmin(f_fine)]
    
    return x_lin
