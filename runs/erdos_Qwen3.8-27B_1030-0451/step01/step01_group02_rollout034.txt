def eval_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))

def make_feasible(x, target):
    """Make x feasible: 0 <= x[i] <= 1, sum(x) = target."""
    x = np.asarray(x, dtype=float).ravel().copy()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if not np.all(np.isfinite(x)):
        x[:] = 0.5
    x = np.clip(x, 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) < 1e-12:
        return x
    if r > 0:
        # Need to add mass; add to positions with most room
        room = 1.0 - x
        order = np.argsort(-room)  # most room first
        rem = r
        for i in order:
            if rem <= 1e-14:
                break
            add = min(room[i], rem)
            x[i] += add
            rem -= add
    else:
        # Need to remove mass; remove from positions with most value
        order = np.argsort(-x)  # most value first
        rem = -r
        for i in order:
            if rem <= 1e-14:
                break
            rem_i = min(x[i], rem)
            x[i] -= rem_i
            rem -= rem_i
    x = np.clip(x, 0.0, 1.0)
    # Final adjustment for floating point
    r = target - float(x.sum())
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                x[idx[0]] += min(r, 1.0 - x[idx[0]])
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                x[idx[0]] += max(r, -x[idx[0]])
    return x

def marginal_costs(h, active_lags):
    """Compute dC[k]/dh[i] for all i and given active lags.
    Returns array of shape (n, len(active_lags))."""
    n = len(h)
    ctr = n - 1
    mc = np.zeros((n, len(active_lags)))
    for jj, k in enumerate(active_lags):
        d = k - ctr  # shift
        for i in range(n):
            val = 0.0
            # i as first factor: h[i] * (1-h[i+d])
            ip = i + d
            if 0 <= ip < n:
                val += (1.0 - h[ip])
            # i as second factor: h[i-d] * (1-h[i]) -> derivative w.r.t. h[i] is -h[i-d]
            im = i - d
            if 0 <= im < n:
                val -= h[im]
            mc[i, jj] = val
    return mc

def compute_all_mc(h, active_lags):
    """Compute dC[k]/dh[i] for all i and given active lags. Vectorized."""
    n = len(h)
    ctr = n - 1
    mc = np.zeros((n, len(active_lags)))
    for jj, k in enumerate(active_lags):
        d = int(k) - ctr
        if d == 0:
            mc[:, jj] = 1.0 - 2.0 * h
        elif d > 0:
            mc[:n-d, jj] += (1.0 - h[d:])
            mc[d:, jj] += -h[:n-d]
        else:
            dd = -d
            mc[dd:, jj] += (1.0 - h[:n-dd])
            mc[:n-dd, jj] += -h[dd:]
    return mc

def logistic_h(params, n):
    """Generate h from logistic parameters.
    params = [T, a1, b1, a2, b2, ..., aJ, bJ]
    g(x) = T * sum_j (a_j * cos(j*pi*x) + b_j * sin(j*pi*x)) + bias
    bias is determined by the constraint sum(h) = n/2.
    """
    n = int(n)
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    T = params[0]
    J = (len(params) - 1) // 2
    g = np.zeros(n)
    for j in range(1, J + 1):
        a = params[2 * j - 1]
        b = params[2 * j]
        g += a * np.cos(j * np.pi * x) + b * np.sin(j * np.pi * x)
    g *= T
    
    # Find bias so that sum(sigmoid(g + bias)) = n/2
    target = 0.5 * n
    lo = -20.0
    hi = 20.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        h = 1.0 / (1.0 + np.exp(-(g + mid)))
        s = float(h.sum())
        if s > target:
            hi = mid
        else:
            lo = mid
    h = 1.0 / (1.0 + np.exp(-(g + 0.5 * (lo + hi))))
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, target)

def logistic_eval(params, n):
    h = logistic_h(params, n)
    return eval_c5(h), h

def global_logistic_search(n, deadline, rng, max_evals=200):
    """Search over logistic parameters."""
    J = 3  # number of Fourier modes
    n_params = 1 + 2 * J  # T + J amplitudes + J phases (or real/imag)
    
    best_c = float('inf')
    best_params = None
    best_h = None
    
    # Random restarts
    n_restarts = max(5, max_evals // 20)
    evals_per_restart = max_evals // n_restarts
    
    for r in range(n_restarts):
        if time.time() > deadline - 5.0:
            break
        # Random initial params
        params = [float(rng.uniform(1.0, 5.0))]  # T
        for j in range(J):
            params.append(float(rng.uniform(-1.0, 1.0)))
            params.append(float(rng.uniform(-1.0, 1.0)))
        params = np.array(params)
        
        # Nelder-Mead on the logistic parameters
        from scipy.optimize import minimize
        def obj(p):
            return logistic_eval(p, n)[0]
        
        try:
            res = minimize(obj, params, method='Nelder-Mead',
                          options={'maxiter': evals_per_restart, 'xatol': 0.01, 'fatol': 1e-5})
            if res.fun < best_c:
                best_c = res.fun
                best_params = res.x.copy()
                best_h = logistic_h(res.x, n)
        except Exception:
            pass
    
    if best_h is None:
        return None, float('inf')
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import minimize

_LOG_N = 200


def eval_c5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))


def make_feasible(x, target):
    x = np.asarray(x, dtype=float).ravel().copy()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    if not np.all(np.isfinite(x)):
        x[:] = 0.5
    x = np.clip(x, 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) < 1e-12:
        return x
    if r > 0:
        room = 1.0 - x
        order = np.argsort(-room)
        rem = r
        for i in order:
            if rem <= 1e-14:
                break
            add = min(room[i], rem)
            x[i] += add
            rem -= add
    else:
        order = np.argsort(-x)
        rem = -r
        for i in order:
            if rem <= 1e-14:
                break
            dec = min(x[i], rem)
            x[i] -= dec
            rem -= dec
    x = np.clip(x, 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                x[idx[0]] += min(r, 1.0 - x[idx[0]])
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                x[idx[0]] += max(r, -x[idx[0]])
    return x


def compute_all_mc(h, active_lags):
    n = len(h)
    ctr = n - 1
    mc = np.zeros((n, len(active_lags)))
    for jj in range(len(active_lags)):
        d = int(active_lags[jj]) - ctr
        if d == 0:
            mc[:, jj] = 1.0 - 2.0 * h
        elif d > 0:
            mc[:n - d, jj] = (1.0 - h[d:])
            mc[d:, jj] += -h[:n - d]
        else:
            dd = -d
            mc[dd:, jj] = (1.0 - h[:n - dd])
            mc[:n - dd, jj] += -h[dd:]
    return mc


def pair_descent(h, target, deadline, max_stall=20, eps_init=0.008):
    n = len(h)
    h = h.copy()
    dx = 2.0 / float(n)
    c = eval_c5(h)
    C = np.correlate(h, 1.0 - h, 'full') * dx
    stall = 0
    eps = eps_init
    max_pairs = max(5, min(20, 3000 // max(n, 1)))
    max_evol = 8

    while time.time() < deadline - 2.0 and stall < max_stall:
        C_max = float(np.max(C))
        if C_max < c - 1e-12:
            c = C_max
        active = np.where(C >= C_max - eps)[0]
        if active.size == 0:
            active = np.array([int(np.argmax(C))])
        if active.size > 120:
            order = C[active].argsort()[::-1][:120]
            active = active[order]

        mc = compute_all_mc(h, active)
        score = np.max(mc, axis=1)

        cands_i = np.where(h < 0.999)[0]
        cands_j = np.where(h > 0.001)[0]
        if cands_i.size == 0 or cands_j.size == 0:
            break

        si = score[cands_i]
        sj = score[cands_j]
        top_i = cands_i[np.argsort(si)[:max_pairs]]
        top_j = cands_j[np.argsort(sj)[::-1][:max_pairs]]

        best_h = None
        best_c = c
        improved = False

        for i in top_i:
            for j in top_j:
                if i == j:
                    continue
                ml = min(1.0 - h[i], h[j])
                if ml < 1e-8:
                    continue
                for lam in (0.4 * ml, 0.15 * ml, 0.05 * ml):
                    h[i] += lam
                    h[j] -= lam
                    c_new = eval_c5(h)
                    if c_new < best_c - 1e-13:
                        best_c = c_new
                        best_h = h.copy()
                        improved = True
                    h[i] -= lam
                    h[j] += lam
                if time.time() > deadline - 3.0:
                    break
            if time.time() > deadline - 3.0:
                break

        if improved and best_h is not None:
            h = best_h.copy()
            c = best_c
            C = np.correlate(h, 1.0 - h, 'full') * dx
            stall = 0
        else:
            if eps > 0.002:
                eps *= 0.5
            else:
                break
    return h, c


def logistic_h(params, n):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    T = float(params[0])
    J = (len(params) - 1) // 2
    g = np.zeros(n)
    for j in range(1, J + 1):
        a = float(params[2 * j - 1])
        b = float(params[2 * j])
        cos_j = np.cos(j * np.pi * x)
        sin_j = np.sin(j * np.pi * x)
        g += a * cos_j + b * sin_j
    g *= T
    target = 0.5 * n
    if abs(T) < 1e-6:
        return make_feasible(np.full(n, 0.5), target)
    lo = -25.0
    hi = 25.0
    for _ in range(40):
        mid = 0.5 * (lo + hi)
        h_tmp = 1.0 / (1.0 + np.exp(-np.clip(g + mid, -30, 30)))
        s = float(h_tmp.sum())
        if s > target:
            hi = mid
        else:
            lo = mid
    bias = 0.5 * (lo + hi)
    h = 1.0 / (1.0 + np.exp(-np.clip(g + bias, -30, 30)))
    h = np.clip(h, 0.0, 1.0)
    return make_feasible(h, target)


def _logistic_obj(p):
    h = logistic_h(p, _LOG_N)
    return eval_c5(h)


def logistic_search(n, deadline, rng, max_evals=300):
    global _LOG_N
    _LOG_N = n
    J = 3
    n_params = 1 + 2 * J
    best_c = float('inf')
    best_params = None

    n_restarts = max(4, max_evals // 25)
    evals_per = max(8, max_evals // n_restarts)

    for r in range(n_restarts):
        if time.time() > deadline - 8.0:
            break
        params = np.zeros(n_params)
        params[0] = float(rng.uniform(1.5, 6.0))
        for j in range(1, n_params):
            params[j] = float(rng.uniform(-1.5, 1.5))

        c0 = _logistic_obj(params)
        if not np.isfinite(c0):
            continue

        try:
            res = minimize(_logistic_obj, params, method='Nelder-Mead',
                           options={'maxiter': evals_per, 'xatol': 0.05, 'fatol': 1e-6})
            if res.fun < best_c:
                best_c = res.fun
                best_params = res.x.copy()
        except Exception:
            pass

        if c0 < best_c:
            best_c = c0
            best_params = params.copy()

    if best_params is None:
        return None, float('inf')
    best_h = logistic_h(best_params, n)
    bc = eval_c5(best_h)
    return best_h, bc


def resample_h(h_old, n_new):
    n0 = len(h_old)
    target = 0.5 * float(n_new)
    if n0 == n_new:
        return make_feasible(h_old, target)
    x_old = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    x_new = (np.arange(n_new) + 0.5) * (2.0 / float(n_new))
    y = np.interp(x_new, x_old, h_old)
    return make_feasible(y, target)


def _get_initial():
    val = globals().get('initial_h_values', None)
    if val is None:
        return None
    arr = np.asarray(val, dtype=float).ravel()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    return make_feasible(arr, 0.5 * float(arr.size))


def fallback_initial(n=240):
    x = (np.arange(n) + 0.5) * (2.0 / float(n))
    best = None
    bestc = float('inf')
    for f in (1, 2, 3, 4):
        for ph in (0.0, np.pi / 3, 2.0 * np.pi / 3, np.pi, 4.0 * np.pi / 3):
            y = 0.5 + 0.4 * np.cos(f * np.pi * x + ph)
            h = make_feasible(y, 0.5 * float(n))
            c = eval_c5(h)
            if c < bestc:
                bestc = c
                best = h
    return best


def mutate_and_descent(h, target, n, deadline, rng, pair_time):
    sigmas = (0.03, 0.02, 0.01, 0.005)
    best_h = h.copy()
    best_c = eval_c5(h)
    for sig in sigmas:
        if time.time() > deadline - 5.0:
            break
        for _ in range(3):
            if time.time() > deadline - 5.0:
                break
            h2 = make_feasible(h + rng.normal(0.0, sig, size=n), target)
            c2 = eval_c5(h2)
            if c2 < best_c + 0.005:
                hd, cd = pair_descent(h2, target, min(deadline, time.time() + pair_time))
                if cd < best_c - 1e-12:
                    best_h = hd
                    best_c = cd
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0.0:
        total = 1100.0
    hard = t0 + max(10.0, total) - 8.0

    rng = np.random.default_rng(int(seed) % (1 << 32))

    hinit = _get_initial()
    if hinit is None:
        hinit = fallback_initial(240)

    n0 = int(hinit.size)
    best_h = hinit.copy()
    best_c = eval_c5(best_h)
    best_n = n0

    # Phase A: Pair descent on initial
    if time.time() < hard - 60.0:
        pd_time = min(50.0, 0.3 * total)
        h_pa, c_pa = pair_descent(best_h, 0.5 * n0, time.time() + pd_time)
        if c_pa < best_c - 1e-14:
            best_h, best_c, best_n = h_pa, c_pa, n0

    # Phase B: Try multiple resolutions
    res_list = []
    for fac in (0.25, 0.35, 0.5, 0.65, 0.85):
        N = int(round(n0 * fac))
        if 80 <= N <= 800 and N not in res_list:
            res_list.append(N)
    for N in (160, 200, 240, 320, 400, 480):
        if N not in res_list and 80 <= N <= 800:
            res_list.append(N)
    res_list = sorted(set(res_list))

    # Prioritize smaller sizes (faster to optimize)
    res_list.sort(key=lambda x: x)

    max_res_tries = 5 if total < 600 else 6
    if len(res_list) > max_res_tries:
        # Pick a spread of sizes
        idxs = np.linspace(0, len(res_list) - 1, max_res_tries, dtype=int)
        res_list = [res_list[i] for i in sorted(set(idxs.tolist()))]
        if len(res_list) > max_res_tries:
            res_list = res_list[:max_res_tries]

    budget_per_res = (hard - time.time() - 120.0) / max(len(res_list), 1)
    for N in res_list:
        if time.time() > hard - 80.0:
            break
        if N == best_n:
            h_cand = best_h.copy()
        else:
            h_cand = resample_h(best_h, N)
        target = 0.5 * float(N)
        pair_time = max(8.0, min(budget_per_res * 0.7, 40.0))
        h_pd, c_pd = pair_descent(h_cand, target, time.time() + pair_time)
        if c_pd < best_c - 1e-14:
            best_h, best_c, best_n = h_pd, c_pd, N

        # Try mutation from this point
        if time.time() < hard - 60.0 and N <= 400:
            mut_time = max(5.0, min(budget_per_res * 0.3, 20.0))
            h_mut, c_mut = mutate_and_descent(h_pd, target, N, time.time() + mut_time, rng, mut_time * 0.4)
            if c_mut < best_c - 1e-14:
                best_h, best_c, best_n = h_mut, c_mut, N

    # Phase C: Logistic global search
    for N_log in (240, 360):
        if time.time() > hard - 50.0:
            break
        log_time = max(10.0, min(50.0, (hard - time.time() - 30.0) * 0.4))
        h_log, c_log = logistic_search(N_log, time.time() + log_time, rng, max_evals=200)
        if h_log is not None and c_log < best_c + 0.01:
            target = 0.5 * float(N_log)
            h_pol, c_pol = pair_descent(h_log, target, min(hard - 10.0, time.time() + 15.0))
            if c_pol < best_c - 1e-14:
                best_h, best_c, best_n = h_pol, c_pol, N_log

    # Phase D: Final polish
    if time.time() < hard - 5.0:
        target = 0.5 * float(best_n)
        final_time = max(5.0, hard - time.time() - 3.0)
        h_fin, c_fin = pair_descent(best_h, target, time.time() + final_time, max_stall=30, eps_init=0.003)
        if c_fin < best_c - 1e-14:
            best_h, best_c = h_fin, c_fin

    # Final validation
    best_n = int(best_h.size)
    best_h = make_feasible(best_h, 0.5 * float(best_n))
    best_c = eval_c5(best_h)
    if not np.isfinite(best_c):
        best_h = np.full(best_n, 0.5, dtype=float)
        best_c = eval_c5(best_h)

    return best_h, float(best_c), best_n
