import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -1.0, 2.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _corr_and_alpha(h, n, dx, tau):
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    return c, alpha

def _grad_from_alpha(h, n, dx, alpha):
    grad = np.zeros(n)
    omh = 1.0 - h
    active = np.where(alpha > 1e-10)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi + 1] += a * omh[lo - d:hi + 1 - d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= a * h[lo2 + d:hi2 + 1 + d]
    grad *= dx
    return grad

def _smooth_obj(h, n, dx, tau):
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    return tau * (s.max() + np.log(np.exp(s).sum()))

def _smooth_grad(h, n, dx, tau):
    _, alpha = _corr_and_alpha(h, n, dx, tau)
    return _grad_from_alpha(h, n, dx, alpha)

def _slsqp_stage(h0, n, deadline, tau, maxiter):
    target = n / 2.0
    dx = 2.0 / n
    constraints = [{'type': 'eq', 'fun': lambda h: float(h.sum() - target),
                    'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_smooth_obj, h0, args=(n, dx, tau),
                       jac=_smooth_grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-15})
        return project_feasible(res.x, target)
    except Exception:
        return h0.copy()

def _optimize_schedule(h0, n, deadline, n_tau=25, decay=0.62, tau_start=None):
    target = n / 2.0
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    c_init = np.correlate(h0, 1.0 - h0, 'full') * dx
    spread = max(1e-6, float(c_init.max() - c_init.mean()))
    if tau_start is None:
        tau_start = max(0.025, 0.25 * spread)

    h_curr = h0.copy()
    stalls = 0

    for i in range(n_tau):
        if time.time() > deadline - 8:
            break
        tau = tau_start * (decay ** i)
        if tau < 5e-5:
            break
        maxiter = 60 if tau > 0.015 else (120 if tau > 0.005 else 200)
        h_curr = _slsqp_stage(h_curr, n, deadline, tau, maxiter)
        c5 = evaluate_c5(h_curr)
        if c5 < best_c5 - 5e-7:
            best_c5 = c5
            best_h = h_curr.copy()
            stalls = 0
        else:
            stalls += 1
            if stalls >= 3 and tau > 0.003:
                rng_esc = np.random.default_rng(hash((n, i, int(tau*1e6))) & 0xFFFFFFFF)
                h_curr = project_feasible(h_curr + rng_esc.standard_normal(n) * 0.004, target)
                stalls = 0

    return best_h, best_c5

def _interp(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h = np.interp(xn, xo, h_old)
    return project_feasible(h, n_new / 2.0)

def _sine_init(n, freq, amp, phase):
    x = np.arange(n) / n * 2.0
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + phase)

def _fourier_init(n, rng, n_modes=4):
    coeffs = rng.standard_normal((n, n_modes)) * 0.15
    x = np.arange(n, dtype=np.float64)
    h = np.zeros(n)
    for m in range(n_modes):
        freq = m + 1
        phase = rng.uniform(0, 2 * np.pi)
        h += coeffs[:, m] * np.sin(2.0 * np.pi * freq * x / n + phase)
    h = 0.5 + h
    return project_feasible(h, n / 2.0)

def _local_search(h, n, deadline, max_iter=5000):
    """Component-wise local search guided by subgradient of the max correlation."""
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    h_curr = h.copy()
    c = np.correlate(h_curr, 1.0 - h_curr, 'full') * dx
    step = 0.01

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        if step < 1e-5:
            break

        c = np.correlate(h_curr, 1.0 - h_curr, 'full') * dx
        k_max = int(np.argmax(c))
        d = k_max - (n - 1)
        # Gradient of the worst correlation
        grad_k = np.zeros(n)
        omh = 1.0 - h_curr
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad_k[lo:hi + 1] = omh[lo - d:hi + 1 - d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad_k[lo2:hi2 + 1] -= h_curr[lo2 + d:hi2 + 1 + d]
        grad_k *= dx

        # Try moving the most improvable component
        # Negative gradient means decreasing h reduces the objective
        candidates = np.where(np.abs(grad_k) > 1e-6)[0]
        if len(candidates) == 0:
            break

        # Pick the component with largest |gradient|
        i_star = candidates[np.argmax(np.abs(grad_k[candidates]))]
        direction = -np.sign(grad_k[i_star])

        h_try = h_curr.copy()
        h_try[i_star] += direction * step

            best_c5 = c5_try
            best_h = h_try.copy()
            h_curr = h_try
            step = min(step * 1.05, 0.02)
        else:
            step *= 0.9

    return best_h, best_c5
