import numpy as np
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj_and_weights(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = np.clip((c - m) / tau, -500.0, 500.0)
    exps = np.exp(shifted)
    w_sum = exps.sum()
    w = exps / w_sum
    obj = m + tau * np.log(w_sum)
    return obj, w


def compute_gradient(h, weights, dx):
    N = len(h)
    grad = np.zeros(N)
    for d in range(N):
        w_val = weights[N - 1 + d] * dx
        if w_val < 1e-18:
            continue
        n = N - d
        grad[:n] += w_val * (1.0 - h[d:])
        grad[d:] -= w_val * h[:n]
    for d in range(1, N):
        w_val = weights[N - 1 - d] * dx
        if w_val < 1e-18:
            continue
        m = d if d < N - d else N - d
        grad[d:d + m] += w_val * (1.0 - h[:m])
        grad[:m] -= w_val * h[d:d + m]
    return grad


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    return h


def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def projected_gradient(h, tau, target_sum, deadline, max_iter=100000):
    N = len(h)
    dx = 2.0 / N
    h = project_feasible(h, target_sum).copy()
    alpha = 1.0
    prev_obj = smooth_obj_and_weights(h, tau)[0]
    stagnation = 0

    for it in range(max_iter):
        if time.time() > deadline:
            break
        obj, w = smooth_obj_and_weights(h, tau)
        if obj < prev_obj - 1e-8:
            alpha = min(alpha * 1.02, 10.0)
            prev_obj = obj
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 100:
                alpha *= 0.5
                stagnation = 0

        grad = compute_gradient(h, w, dx)
        gnorm = np.dot(grad, grad)
        if gnorm < 1e-20:
            break

        step = alpha
        h_new = h.copy()
        for _ in range(30):
            h_try = project_feasible(h - step * grad, target_sum)
            new_obj, _ = smooth_obj_and_weights(h_try, tau)
            if new_obj <= obj - 1e-4 * step * gnorm:
                h_new = h_try
                break
            step *= 0.5
        else:
            h_new = project_feasible(h - step * grad, target_sum)

        h = h_new

    return h


def directed_refine(h, target_sum, deadline, max_iter=50000):
    """Targeted refinement: focus on reducing the worst lags."""
    N = len(h)
    dx = 2.0 / N
    h = project_feasible(h, target_sum).copy()
    alpha = 0.1

    for it in range(max_iter):
        if time.time() > deadline:
            break
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = np.max(corr)
        # Focus on lags within 2% of max
        threshold = c_max - 0.02 * c_max
        active = corr >= threshold
        if not np.any(active):
            break

        # Build a "frozen" weight vector: only active lags have weight
        w_active = np.zeros_like(corr)
        w_active[active] = 1.0
        w_active /= w_active.sum()

        grad = compute_gradient(h, w_active, dx)
        gnorm = np.dot(grad, grad)
        if gnorm < 1e-20:
            break

        step = alpha
        for _ in range(30):
            h_try = project_feasible(h - step * grad, target_sum)
            new_cmax = np.max(np.correlate(h_try, 1.0 - h_try, mode='full') * dx)
            if new_cmax < c_max:
                h = h_try
                break
            step *= 0.5
        else:
            h = project_feasible(h - step * grad, target_sum)

    return h


def random_restart(h, rng, scale, target_sum):
    noise = rng.normal(0, scale, len(h))
    return project_feasible(h + noise, target_sum)
