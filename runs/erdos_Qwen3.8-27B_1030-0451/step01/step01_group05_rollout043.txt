    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):

    lo = float(np.min(z)) - 2.0
    hi = float(np.max(z)) + 2.0

        s = np.sum(np.clip(z - mid, 0.0, 1.0))

        else:
            hi = mid

    h = np.clip(z - mu, 0.0, 1.0)

    if abs(diff) > 1e-14:

        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_all_cols(h):
    n = len(h)
    L = 2 * n - 1
    cols = np.zeros((n, L))
    for i in range(n):
        s_lo = -i
        s_hi = n - 1 - i
        K_lo = n - 1 + s_lo
        K_hi = n - 1 + s_hi
        cols[i, K_lo:K_hi + 1] += 1.0 - h
        s_lo2 = i - (n - 1)
        s_hi2 = i
        K_lo2 = n - 1 + s_lo2
        K_hi2 = n - 1 + s_hi2
        cols[i, K_lo2:K_hi2 + 1] -= h[n - 1::-1]
    return cols

def line_search_max_quadratic(raw, lin, quad, d_lo, d_hi, dx, n_grid):
    """Minimize max_K(raw[K] + d*lin[K] + d^2*quad[K]) * dx over d in [d_lo, d_hi]."""
    best_d = 0.0
    best_val = np.inf

    # Evaluate on grid
    if d_hi - d_lo < 1e-14:
        val = np.max(raw + d_lo * lin + d_lo * d_lo * quad) * dx
        return d_lo, val

    grid = np.linspace(d_lo, d_hi, n_grid)
    for d in grid:
        f = np.max(raw + d * lin + d * d * quad) * dx
        if f < best_val:
            best_val = f
            best_d = d

    # Golden section refinement around best grid point
    idx = int(np.argmin([np.max(raw + d * lin + d * d * quad) for d in grid]))
    lo_k = grid[max(0, idx - 1)]
    hi_k = grid[min(len(grid) - 1, idx + 1)]

    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a1 = hi_k - gr * (hi_k - lo_k)
    b1 = lo_k + gr * (hi_k - lo_k)
    f_a1 = np.max(raw + a1 * lin + a1 * a1 * quad) * dx
    f_b1 = np.max(raw + b1 * lin + b1 * b1 * quad) * dx

    for _ in range(30):
        if hi_k - lo_k < 1e-14:
            break
        if f_a1 < f_b1:
            hi_k = b1
            b1 = a1
            f_b1 = f_a1
            a1 = hi_k - gr * (hi_k - lo_k)
            f_a1 = np.max(raw + a1 * lin + a1 * a1 * quad) * dx
        else:
            lo_k = a1
            a1 = b1
            f_a1 = f_b1
            b1 = lo_k + gr * (hi_k - lo_k)
            f_b1 = np.max(raw + b1 * lin + b1 * b1 * quad) * dx

    mid = (lo_k + hi_k) / 2.0
    f_mid = np.max(raw + mid * lin + mid * mid * quad) * dx
    if f_mid < best_val:
        best_val = f_mid
        best_d = mid

    # Also check endpoints
    f_lo = np.max(raw + d_lo * lin + d_lo * d_lo * quad) * dx
    f_hi = np.max(raw + d_hi * lin + d_hi * d_hi * quad) * dx
    if f_lo < best_val:
        best_val = f_lo
        best_d = d_lo
    if f_hi < best_val:
        best_val = f_hi
        best_d = d_hi

    return best_d, best_val

def compute_gradient_weights(raw, tau):
    n = len(raw)
    max_raw = np.max(raw)
    exponents = np.clip((raw - max_raw) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.ones(n) / n
    else:
        weights /= sw
    return weights

    best_score = cur_score

    for it in range(max_iters):
        if time.time() > deadline:
            break

        w = compute_gradient_weights(raw, tau)
        cols = compute_all_cols(h)
        g = cols @ w

        high_idx = order[-5:][::-1]

                S_int = int(np.floor(n / 2.0))
                S_frac = n / 2.0 - S_int
                top = np.argsort(g)[:S_int]
                u[top] = 1.0
                if S_frac > 1e-10:
                    u[int(order[S_int])] = S_frac
                for gamma in [0.3, 0.15, 0.08, 0.04, 0.02]:
                    h_try = (1.0 - gamma) * h + gamma * u
                    h_try = project_feasible(h_try, n)
                    s_try = exact_score(h_try)
                    if s_try < cur_score - 1e-12:
                        h = h_try
                        raw = np.correlate(h, 1.0 - h, mode='full').astype(float)
                        cur_score = float(np.max(raw) * dx)
                        if cur_score < best_score:
                            best = h.copy()
                            best_score = cur_score
                        break
                # If still stuck, break
                if time.time() > deadline - 30:
                    break

    return best, best_score

def sa_escape(h0, raw0, score0, rng, deadline, max_iters=20000):
    h = h0.copy()
    n = len(h)
    dx = 2.0 / n
    raw = raw0.copy()
    cur_score = score0
    best = h.copy()
    best_score = score0

    T_start = 0.05 * score0
    T_end = 1e-5 * score0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        T = T_start * (T_end / T_start) ** (it / max(float(max_iters), 1.0))
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue

        d_lo = max(-h[a], h[b] - 1.0)
        d_hi = min(1.0 - h[a], h[b])
        if d_hi - d_lo < 1e-14:
            continue

        lin = np.zeros(2 * n - 1)
        cols_a = compute_all_cols(h)[a]
        cols_b = compute_all_cols(h)[b]
        lin = cols_a - cols_b
        quad = np.zeros(2 * n - 1)
        quad[n - 1 + (b - a)] += 1.0
        quad[n - 1 + (a - b)] += 1.0

        # Quick grid search for best d
        d_mid = (d_lo + d_hi) * 0.5
        d_span = (d_hi - d_lo) * 0.3
        d_try = np.clip(d_mid + d_span * (2.0 * rng.random() - 1.0), d_lo, d_hi)

        f_try = float(np.max(raw + d_try * lin + d_try * d_try * quad) * dx)
        delta = f_try - cur_score

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-15)):
            h[a] += d_try
            h[b] -= d_try
            h = np.clip(h, 0.0, 1.0)
            raw = np.correlate(h, 1.0 - h, mode='full').astype(float)
            cur_score = float(np.max(raw) * dx)
            if cur_score < best_score:
                best = h.copy()
                best_score = cur_score

    return best, best_score

def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)

def make_periodic(n, periods, duties, phases, rng):
    x = (np.arange(n) + 0.5) / n * 2.0
    h = np.zeros(n)
    for p, d, ph in zip(periods, duties, phases):
        t = ((x + 2.0 * ph / p) % (2.0 / p)) * p / 2.0
        h += (t < d).astype(float)
    h /= len(periods)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + max(budget_s - 15.0, 5.0)

    # Get initial
    try:
        parent = np.array(initial_h_values, dtype=float).flatten().copy()
    except Exception:
        parent = None

    if parent is not None and len(parent) >= 10 and np.all(np.isfinite(parent)):
        n_p = len(parent)
        parent = project_feasible(parent, n_p)
    else:
        n_p = 128
        parent = project_feasible(np.full(n_p, 0.5), n_p)

    best_h = parent.copy()
    best_score = exact_score(best_h)
    best_n = len(best_h)

    def _try_update(h, score, n):
        nonlocal best_h, best_score, best_n
        if score < best_score - 1e-13:
            best_h = h.copy()
            best_score = score
            best_n = n

    # Phase 1: Exact pair descent on parent
    if time.time() < deadline - 60:
        tau_sched = [0.03, 0.015, 0.008, 0.004, 0.002]
        h1, s1 = exact_pair_descent(best_h.copy(), tau_sched, 50000, rng, deadline, n_grid=35)
        _try_update(h1, s1, best_n)

    # Phase 2: Try different n values
    dim_candidates = []
    for factor in [1.3, 1.5, 1.75, 2.0, 0.8, 0.7]:
        cn = int(round(best_n * factor))
        if 40 <= cn <= 2500 and cn != best_n:
            cn = cn if cn % 2 == 0 else cn + 1
            dim_candidates.append(cn)
    dim_candidates = sorted(set(dim_candidates))

    for dn in dim_candidates:
        if time.time() > deadline - 40:
            break
        h_d = interpolate_h(best_h, dn)
        s_d = exact_score(h_d)
        # Quick optimization
        tau_sched_d = [0.02, 0.01, 0.005, 0.002]
        h_d, s_d = exact_pair_descent(h_d, tau_sched_d, 20000, rng, deadline, n_grid=30)
        if s_d < best_score - 1e-13 or (abs(s_d - best_score) < 1e-7 and dn < best_n):
            _try_update(h_d, s_d, dn)

    # Phase 3: More refinement on best
    if time.time() < deadline - 50:
        tau_sched2 = [0.01, 0.005, 0.002, 0.001]
        h3, s3 = exact_pair_descent(best_h.copy(), tau_sched2, 50000, rng, deadline, n_grid=50)
        _try_update(h3, s3, best_n)

    # Phase 4: SA escape
    if time.time() < deadline - 80:
        raw_best = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
        h_sa, s_sa = sa_escape(best_h.copy(), raw_best, best_score, rng, deadline, max_iters=15000)
        if s_sa < best_score - 1e-13:
            _try_update(h_sa, s_sa, best_n)
        elif s_sa < best_score + 0.01:
            # Refine the SA result
            if time.time() < deadline - 30:
                h_sa2, s_sa2 = exact_pair_descent(h_sa, [0.01, 0.005, 0.002], 20000, rng, deadline, n_grid=40)
                _try_update(h_sa2, s_sa2, best_n)

    # Phase 5: Periodic multi-start
    if time.time() < deadline - 100:
        for trial in range(6):
            if time.time() > deadline - 30:
                break
            num_periods = 2 + trial % 3
            periods = [2.0 / (1 + rng.integers(1, 4)) for _ in range(num_periods)]
            duties = [0.3 + 0.3 * rng.random() for _ in range(num_periods)]
            phases = [rng.random() for _ in range(num_periods)]
            h_try = make_periodic(best_n, periods, duties, phases, rng)
            s_try = exact_score(h_try)
            if s_try < 0.45:  # Only proceed if reasonable
                h_try, s_try = exact_pair_descent(h_try, [0.02, 0.01, 0.005], 15000, rng, deadline, n_grid=30)
                _try_update(h_try, s_try, best_n)

    # Phase 6: Final polish with very fine line search
    if time.time() < deadline - 15:
        tau_final = [0.003, 0.001, 0.0005]
        h_f, s_f = exact_pair_descent(best_h.copy(), tau_final, 30000, rng, deadline, n_grid=60)
        _try_update(h_f, s_f, best_n)

    # Final verification
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-10:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
