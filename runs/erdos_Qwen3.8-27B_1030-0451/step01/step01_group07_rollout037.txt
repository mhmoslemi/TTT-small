import time
import math
import numpy as np
from scipy.optimize import minimize_scalar


def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size > 0 and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -0.5, 1.5
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d < 0:
            slack = 1.0 - out
        else:
            slack = out
        total = slack.sum()
        if total > 1e-12:
            out = out + (-d / total) * slack
    return np.clip(out, 0.0, 1.0)


def objective(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')


def active_grad(h, corr, tau, dx):
    n = len(h)
    center = n - 1
    grad = np.zeros(n)
    for c in range(2 * n - 1):
        w = corr[c] - tau
        if w <= 1e-13:
            continue
        p = c - center
        if p == 0:
            grad += w * (1.0 - 2.0 * h)
        elif p > 0:
            grad[p:] += w * (1.0 - h[:n - p])
            grad[:n - p] -= w * h[p:]
        else:
            q = -p
            grad[:n - q] += w * (1.0 - h[q:])
            grad[q:] -= w * h[:n - q]
    grad *= dx
    return grad


def line_objective(t, h, i, j):
    h_t = h.copy()
    h_t[i] -= t
    h_t[j] += t
    return objective(h_t)


def make_random_seed(n, rng):
    h = (rng.random(n) >= 0.5).astype(float)
    h = project_exact(h, n / 2.0)
    return h


def binary_round(h, target):
    hr = (h > 0.5).astype(float)
    return project_exact(hr, target)


def build_pool(h0, n, target, rng):
    pool = [h0.copy()]
    pool.append(binary_round(h0, target))
    for _ in range(3):
        pert = h0 + rng.normal(0, 0.15, n)
        pool.append(project_exact(pert, target))
    # One more: flip a random subset
    flip_mask = rng.random(n) < 0.3
    flipped = h0.copy()
    flipped[flip_mask] = 1.0 - flipped[flip_mask]
    pool.append(project_exact(flipped, target))
    return pool


def pair_polish(h_init, target, time_limit, deadline, rng, gap_init=0.02):
    n = len(h_init)
    h = h_init.copy()
    cur_c5 = objective(h)
    best_h = h.copy()
    best_c5 = cur_c5

    t0 = time.monotonic()
    stall = 0
    gap = gap_init
    gr = (math.sqrt(5.0) - 1.0) / 2.0

    while (time.monotonic() - t0) < time_limit:
        if time.monotonic() > deadline:
            break

        corr = compute_corr(h) * (2.0 / n)
        c5_raw = float(corr.max())
        tau = c5_raw - gap * c5_raw

        grad = active_grad(h, corr, tau, 2.0 / n)

        # Sort indices by gradient
        order = np.argsort(grad)
        half = n // 2

        # Build candidate pairs
        candidates = []
        # High-grad donors to low-grad receivers
        for k in range(min(12, half)):
            donor = int(order[n - 1 - k])
            receiver = int(order[k])
            if donor != receiver:
                candidates.append((donor, receiver))
        # Medium-range pairs
        for k in range(min(4, half - 1)):
            a = int(order[half + k])
            b = int(order[half - 1 - k])
            if a != b:
                candidates.append((a, b))
        # Random pairs
        for _ in range(4):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                candidates.append((i, j))

        improved = False
        for (i, j) in candidates:
            if time.monotonic() > deadline:
                break

            lo = max(h[i] - 1.0, -h[j])
            hi = min(h[i], 1.0 - h[j])
            if hi - lo < 1e-11:
                continue

            # Quick endpoint check
            vals_to_try = []
            mid = 0.5 * (lo + hi)
            vals_to_try.append(mid)
            if lo > 1e-11:
                vals_to_try.append(lo * 0.99)
            if hi < 1.0 - 1e-11:
                vals_to_try.append(hi * 0.99)

            best_quick = cur_c5
            best_quick_t = 0.0
            for tv in vals_to_try:
                h_t = h.copy()
                h_t[i] -= tv
                h_t[j] += tv
                cv = objective(h_t)
                if cv < best_quick:
                    best_quick = cv
                    best_quick_t = tv

            # If quick search found improvement, refine
            if best_quick < cur_c5 - 1e-12:
                try:
                    res = minimize_scalar(line_objective, args=(h, i, j),
                                          bounds=(lo, hi), method='bounded',
                                          options={'maxiter': 25, 'xatol': 1e-8})
                    if res.fun < best_quick:
                        best_quick = float(res.fun)
                        best_quick_t = float(res.x)
                except Exception:
                    pass

            if cur_c5 - best_quick > 1e-12:
                h[i] -= best_quick_t
                h[j] += best_quick_t
                h = np.clip(h, 0.0, 1.0)
                d = h.sum() - target
                if abs(d) > 1e-13:
                    h = project_exact(h, target)
                cur_c5 = objective(h)
                if cur_c5 < best_c5 - 1e-14:
                    best_c5 = cur_c5
                    best_h = h.copy()
                improved = True
                stall = 0
                break

        if not improved:
            stall += 1
            if stall % 15 == 0:
                gap = min(gap * 1.4, 0.25)
            if stall >= 50:
                # Perturbation: a few random transfers
                for _ in range(4):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    if h[i] > h[j] and h[i] > 0.05 and 1.0 - h[j] > 0.05:
                        t = rng.random() * 0.2 * min(h[i], 1.0 - h[j])
                        h[i] -= t
                        h[j] += t
                    elif h[j] > h[i] and h[j] > 0.05 and 1.0 - h[i] > 0.05:
                        t = rng.random() * 0.2 * min(h[j], 1.0 - h[i])
                        h[j] -= t
                        h[i] += t
                h = np.clip(h, 0.0, 1.0)
                h = project_exact(h, target)
                cur_c5 = objective(h)
                if cur_c5 < best_c5:
                    best_c5 = cur_c5
                    best_h = h.copy()
                stall = 0

    c5_final = objective(h)
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h.copy()

    return best_h, best_c5


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(10.0, float(budget_s) - 6.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        parent = make_random_seed(131, rng)
    n = len(parent)
    target = n / 2.0
    h0 = project_exact(parent, target)

    best_h = h0.copy()
    best_c5 = objective(best_h)

    def consider(cand_h):
        nonlocal best_h, best_c5
        try:
            ch = project_exact(np.asarray(cand_h, dtype=float), len(cand_h) / 2.0)
            c = objective(ch)
            if c < best_c5 - 1e-13:
                best_c5 = c
                best_h = ch
        except Exception:
            pass

    # Build starting pool
    pool = build_pool(h0, n, target, rng)

    # Phase 1: Polish each pool member
    n_pool = len(pool)
    per_alloc = effective * 0.55 / n_pool
    for idx, start_h in enumerate(pool):
        if time.monotonic() > deadline - 10.0:
            break
        alloc = min(per_alloc, deadline - time.monotonic() - 5.0)
        if alloc < 3.0:
            break
        ch, cc = pair_polish(start_h, target, alloc, deadline, rng, gap_init=0.015)
        consider(ch)

    # Phase 2: Polish the global best more intensively
    if time.monotonic() < deadline - 15.0:
        remaining = deadline - time.monotonic() - 5.0
        alloc = min(remaining * 0.4, 200.0)
        if alloc > 5.0:
            ch, cc = pair_polish(best_h, target, alloc, deadline, rng, gap_init=0.008)
            consider(ch)

    # Phase 3: Try nearby resolutions
    if time.monotonic() < deadline - 10.0:
        for dn in [2, -2, 4, -4, 6]:
            nn = n + dn
            if nn < 80 or nn > 200:
                continue
            if time.monotonic() > deadline - 8.0:
                break
            alloc = min(0.05 * effective, deadline - time.monotonic() - 5.0)
            if alloc < 3.0:
                break
            rs = project_exact(resample(best_h, nn), nn / 2.0)
            ch, cc = pair_polish(rs, nn / 2.0, alloc, deadline, rng, gap_init=0.02)
            consider(ch)

    # Phase 4: Final intensive polish on best
    if time.monotonic() < deadline - 3.0:
        remaining = deadline - time.monotonic() - 2.0
        alloc = min(remaining * 0.5, 80.0)
        if alloc > 2.0:
            ch, cc = pair_polish(best_h, target, alloc, deadline, rng, gap_init=0.005)
            consider(ch)

    # Final feasibility check
    best_h = project_exact(best_h, len(best_h) / 2.0)
    best_c5 = objective(best_h)

    return best_h, float(best_c5), int(len(best_h))
