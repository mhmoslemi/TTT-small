def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
        r2 = target - y.sum()
        if r2 > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
        elif r2 < 0:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):

def nesterov_pg(h0, deadline, max_iters=10000):

    h = h0.copy()
    y = h0.copy()
    prev_y = h0.copy()
    t_k = 1.0
    step = 0.01
    beta_smooth = 30.0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(y)
        m = v.max()
        expo = np.clip(beta_smooth * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 1e-12:
                lag = k - (n - 1)
                g += w[k] * grad_lag(y, lag)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        momentum = (t_k - 1.0) / t_new
        y_new = y + momentum * (y - prev_y)
        y_new = y_new - step * g
        y_new = project_sum_box(y_new, n / 2.0)
        c_new = evaluate_c5(y_new)
        if c_new < best_c:
            best_h = y_new.copy()
            best_c = c_new
        step = min(step * 1.005, 0.05)
        if c_new > m + 0.01:
            step *= 0.7
        prev_y = y
        y = y_new
        t_k = t_new
    return best_h, best_c

def smooth_slsqp(h0, deadline, max_rounds=5):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [8, 16, 32, 64, 128, 256]
    for beta in betas[:max_rounds]:
        if time.time() > deadline:
            break
        bounds = [(0.0, 1.0)] * n

        def obj_fn(x):
            v = corr_all(x)
            m = np.max(v)
            expo = np.clip(beta * (v - m), None, 500)
            u = np.exp(expo)
            return m - np.log(u.sum()) / beta

        def obj_grad(x):
            v = corr_all(x)
            m = np.max(v)
            expo = np.clip(beta * (v - m), None, 500)
            u = np.exp(expo)
            w = u / u.sum()
            g = np.zeros(n)
            for k in range(len(v)):
                if w[k] > 1e-13:
                    lag = k - (n - 1)
                    g += w[k] * grad_lag(x, lag)
            return g

        def sum_fn(x):
            return x.sum() - n / 2.0

        def sum_jac(x):
            return np.ones(n)

        cons = [{'type': 'eq', 'fun': sum_fn, 'jac': sum_jac}]
        try:
            res = minimize(obj_fn, best_h, jac=obj_grad, method='SLSQP',
                           bounds=bounds, constraints=cons,
                           options={'maxiter': 300, 'ftol': 1e-15})
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-12:
                    best_h = x_new
                    best_c = c_new
        except Exception:
            pass
    return best_h, best_c

                best_c = c5
        except Exception:
            pass

    # Phase 6: Nesterov PG polish
    if time.time() < deadline - 10:
        dl6 = min(deadline, start_time + budget_s * 0.88)
        try:
            h6, c6 = nesterov_pg(best_h, dl6, max_iters=5000)
            if c6 < best_c:
                best_h = h6
                best_c = c6
        except Exception:
            pass

    # Phase 7: Random restarts
    if time.time() < deadline - 30:
        dl7 = min(deadline, start_time + budget_s * 0.93)
        try:
            h7, c7 = random_restarts(best_h, dl7, n_restarts=3, pert_scale=0.015)
            if c7 < best_c:
                best_h = h7
                best_c = c7
        except Exception:
            pass

    # Phase 8: Final cutting plane polish
    if time.time() < deadline - 5:
        try:
            h8, c8 = cutting_plane_method(best_h, deadline - 1, max_iters=300)
            if c8 < best_c:
                best_h = h8
                best_c = c8
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
