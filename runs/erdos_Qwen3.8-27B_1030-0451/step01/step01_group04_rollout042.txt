    v_grid = np.linspace(0, 1, n_grid)
    
    for p in range(n_passes):
        improved = False
        for i in range(N):
            r_min = max(0, i)
            r_max = min(2*N-2, i+N-1)
            r_vals = np.arange(r_min, r_max+1)
            other = r_vals - i  # other index, in [0, N-1]
            h_other = h[other]

        current_c5 = float(np.max(corr))

        for p in range(800):
            if time.time() > deadline - 60:
                break
            order = rng.permutation(N) if p % 3 == 0 else None
            h, corr, current_c5, improved = cd_pass(h, corr, v_grid, N, order)
            if (p + 1) % 15 == 0:
                h = project_feasible(h, target_sum)
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                current_c5 = float(np.max(corr))
            if not improved:
                break

        h = project_feasible(h, target_sum)
        current_c5 = compute_c5(h)

        # Phase 2: SLSQP polish
        bounds = [(0.0, 1.0)] * N
        constraint = [{'type': 'eq', 'fun': sum_eq, 'jac': sum_eq_jac,
                       'args': (target_sum,)}]

        tau_list = [0.01, 0.005, 0.002, 0.001, 0.0005]
        for tau in tau_list:
            if time.time() > deadline - 40:
                break
            try:
                result = minimize(
                    slsqq_f, h.copy(),
                    jac=slsqq_g,
                    args=(tau,),
                    method='SLSQP',
                    bounds=bounds,
                    constraints=constraint,
                    options={'maxiter': 300, 'ftol': 1e-14}
                )
                h_cand = project_feasible(result.x, target_sum)
                c5_cand = compute_c5(h_cand)
                if c5_cand < current_c5:
                    h = h_cand
                    current_c5 = c5_cand
            except Exception:
                pass

        # Phase 3: More coordinate descent after SLSQP
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        current_c5 = float(np.max(corr))
        for p in range(400):
            if time.time() > deadline - 20:
                break
            h, corr, current_c5, improved = cd_pass(h, corr, v_grid, N)
            if (p + 1) % 10 == 0:
                h = project_feasible(h, target_sum)
                corr = np.correlate(h, 1.0 - h, mode='full') * dx
                current_c5 = float(np.max(corr))
            if not improved:
                break

        h = project_feasible(h, target_sum)
        current_c5 = compute_c5(h)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            best_N = N

    # Try random restarts at smaller N
    for N_rand in [200, 300]:
        if time.time() > deadline - 30:
            break
        for restart in range(3):
            if time.time() > deadline - 20:
                break
            h = rng.random(N_rand)
            h = project_feasible(h, N_rand / 2.0)
            v_grid = np.linspace(0.0, 1.0, 200)
            dx = 2.0 / N_rand
            corr = np.correlate(h, 1.0 - h, mode='full') * dx

            for p in range(600):
                if time.time() > deadline - 15:
                    break
                h, corr, _, improved = cd_pass(h, corr, v_grid, N_rand)
                if (p + 1) % 10 == 0:
                    h = project_feasible(h, N_rand / 2.0)
                    corr = np.correlate(h, 1.0 - h, mode='full') * dx
                if not improved:
                    break

            h = project_feasible(h, N_rand / 2.0)
            c5_rand = compute_c5(h)
            if c5_rand < best_c5:
                best_c5 = c5_rand
                best_h = h.copy()
                best_N = N_rand

    # Final: return best
    final_c5 = compute_c5(best_h)
    if final_c5 > best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
