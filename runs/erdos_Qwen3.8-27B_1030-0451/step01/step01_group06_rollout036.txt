import numpy as np
import time
from scipy.optimize import minimize, linprog

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    # Binary search for the projection
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix rounding
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                dec = min(res, h[idx] - 1e-15)
                if dec <= 0:
                    continue
                h[idx] -= dec
                res -= dec
                if abs(res) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                inc = min(-res, 1.0 - h[idx] - 1e-15)
                if inc <= 0:
                    continue
                h[idx] += inc
                res += inc
                if abs(res) < 1e-14:
                    break
    return h

def compute_smooth_objective_and_grad(h, T):
    """Compute smooth max of correlations and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    # Log-sum-exp for numerical stability
    M = np.max(corr)
    logsum = M + np.log(np.sum(np.exp(corr - M)))
    obj = logsum / T
    # Weights for each correlation
    weights = T * np.exp(corr - M) / np.sum(np.exp(corr - M))
    # Gradient of each correlation with respect to h
    # corr[j] for j in [0, 2n-2]
    # corr[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) * dx
    # d(corr[n-1+k])/d(h[m]) = dx * (delta(m, i+k) * (1-h[i]) - delta(m, i) * h[i+k]) summed over i
    # = dx * [(1-h[m-k]) if m >= k] - [h[m+k] if m <= n-1-k]
    # More carefully:
    # d/d(h[m]) sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) * dx
    # = dx * sum_{i=0}^{n-1-k} [delta(m, i+k)*(1-h[i]) + h[i+k]*(-delta(m, i))]
    # = dx * [(1-h[m-k]) * 1(m>=k) - h[m+k] * 1(m <= n-1-k)]
    # Wait, i goes from 0 to n-1-k, so i+k goes from k to n-1.
    # delta(m, i+k) = 1 when i+k = m, i.e., i = m-k. Valid when m-k >= 0 and m-k <= n-1-k, i.e., k <= m <= n-1.
    # delta(m, i) = 1 when i = m. Valid when 0 <= m <= n-1-k.
    
    # So for corr[n-1+k] (k >= 1):
    # d/d(h[m]) = dx * [(1-h[m-k]) if k <= m <= n-1 else 0] - dx * [h[m+k] if 0 <= m <= n-1-k else 0]
    
    # Similarly for corr[n-1-k] (k >= 1):
    # corr[n-1-k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
    # d/d(h[m]) = dx * [(1-h[m+k]) if 0 <= m <= n-1-k else 0] - dx * [h[m-k] if k <= m <= n-1 else 0]
    
    # For corr[n-1] (k=0):
    # corr[n-1] = sum_i h[i]*(1-h[i]) * dx
    # d/d(h[m]) = dx * (1-2*h[m])
    
    grad = np.zeros(n)
    # k=0
    grad += weights[n-1] * dx * (1 - 2*h)
    
    # k > 0: corr at index n-1+k
    for k in range(1, n):
        w = weights[n-1+k]
        if w < 1e-15:
            continue
        # d/d(h[m]) for m = k to n-1: (1-h[m-k])
        if k < n:
            grad[k:] += w * dx * (1 - h[:n-k])
        # d/d(h[m]) for m = 0 to n-1-k: -h[m+k]
        if n-k > 0:
            grad[:n-k] += -w * dx * h[k:]
    
    # k > 0: corr at index n-1-k
    for k in range(1, n):
        w = weights[n-1-k]
        if w < 1e-15:
            continue
        # corr[n-1-k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
        # d/d(h[m]) = dx * [(1-h[m+k]) if 0 <= m <= n-1-k] - dx * [h[m-k] if k <= m <= n-1]
        if n-k > 0:
            grad[:n-k] += w * dx * (1 - h[k:])
        if k < n:
            grad[k:] += -w * dx * h[:n-k]

import time
from scipy.optimize import linprog

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec

            for idx in order:
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
                if abs(res) < 1e-14:
                    break
    return h

def compute_linearized_corrs(h, delta):
    """Compute linearized correlations: corr_k + grad_k · delta"""
    n = len(h)
    dx = 2.0 / n
    corr = correlation_array(h)
    # Linearization: h[i]*h[i+k] ≈ h[i]*h[i+k] + h[i]*delta[i+k] + delta[i]*h[i+k]
    # So: new_corr_k ≈ corr_k + sum (h[i]*delta[i+k] + delta[i]*h[i+k]) * dx (with signs)
    # Actually: corr_k = sum h[i]*(1-h[i+k]) * dx (for the appropriate direction)
    # d(corr_k)/d(delta[i]) = dx * (1 - h[i+k]) for the i in h[i] term
    #                      - dx * h[i] for the i+k in h[i+k] term
    
    # Let me compute the gradient of each corr with respect to h, then apply delta.
    # Actually, for the LP, I need: corr_k(h + delta) ≈ corr_k(h) + sum_m (d corr_k / d h[m]) * delta[m]
    
    # Let me compute this efficiently.
    # corr[n-1+k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i]) * dx
    # = sum_{i=0}^{n-1-k} [h[i+k] - h[i+k]*h[i]] * dx
    # d/d(h[m]): 
    #   from h[i+k]: dx if m = i+k for some i in [0, n-1-k], i.e., m in [k, n-1]
    #   from -h[i+k]*h[i]: -dx*(h[i]) if m = i+k, -dx*(h[i+k]) if m = i
    #   = dx*(1-h[i]) if m = i+k (m in [k, n-1])
    #   - dx*h[i+k] if m = i (m in [0, n-1-k])
    
    # So: d(corr[n-1+k])/d(h[m]) = dx * [(1-h[m-k]) * 1(k≤m≤n-1) - h[m+k] * 1(0≤m≤n-1-k)]
    
    # For the LP, I need the matrix G where G[k, m] = d(corr[k])/d(h[m])
    # This is a (2n-1) × n matrix.
    
    G = np.zeros((2*n-1, n))
    # k=0: G[n-1, :] = dx * (1 - 2*h)
    G[n-1] = dx * (1 - 2*h)
    
    for k in range(1, n):
        # corr at index n-1+k

        # corr[n-1-k] = sum_{i=0}^{n-1-k} h[i]*(1-h[i+k]) * dx
        # d/d(h[m]) for m in [0, n-1-k]: dx*(1-h[m+k])
        G[n-1-k, :n-k] = dx * (1 - h[k:])
        # d/d(h[m]) for m in [k, n-1]: -dx*h[m-k]
        G[n-1-k, k:] += -dx * h[:n-k]
    
    linearized = corr + G.dot(delta)
    return linearized, G

def solve_lp_full(h, corr, G, trust):
    """Solve LP: min t s.t. G@delta - t ≤ -corr, |delta| ≤ trust, sum(delta) = 0"""
    n = len(h)
    m = 2*n - 1
    nvars = n + 1
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    A_ub = np.zeros((m, nvars))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = -corr
    lower = np.maximum(-h, -trust)
    upper = np.minimum(1.0 - h, trust)
    A_eq = np.zeros((1, nvars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = list(zip(lower, upper)) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.all(h + d >= -1e-9) and np.all(h + d <= 1 + 1e-9):
                return d
    except Exception:
        pass
    return None
