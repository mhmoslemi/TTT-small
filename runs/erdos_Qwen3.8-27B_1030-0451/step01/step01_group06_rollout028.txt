def _compute_weights(Q, tau):
    M = np.max(Q)
    scaled = (Q - M) / tau
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    s = np.sum(exp_s)

    weights = exp_s / s
    F = M + tau * (max_s + np.log(s))

import numpy as np
import time
from scipy.optimize import minimize

_SLSQP_STATE = {'tau': 0.1, 'n': 120}

def _compute_Q(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def _slsqp_obj(h_flat):
    tau = _SLSQP_STATE['tau']
    n = _SLSQP_STATE['n']
    dx = 2.0 / n
    Q = _compute_Q(h_flat)
    M = np.max(Q)
    scaled = (Q - M) / tau
    max_s = np.max(scaled)
    exp_s = np.exp(scaled - max_s)
    s = np.sum(exp_s)
    P = 2 * n - 1
    if s < 1e-300:
        top_k = min(3, P)
        idx = np.argsort(Q)[-top_k:]
        F = M
        w = np.zeros(P)
        w[idx] = 1.0 / top_k
    else:
        F = M + tau * (max_s + np.log(s))
        w = exp_s / s
    grad = np.zeros(n)
    omh = 1.0 - h_flat
    for r in range(P):
        wr = w[r]
        if wr < 1e-14:
            continue
        k = r - (n - 1)
        if k == 0:
            grad += wr * dx * (1.0 - 2.0 * h_flat)
        elif k > 0:
            grad[k:] += wr * dx * omh[:n-k]
            grad[:n-k] -= wr * dx * h_flat[k:]
        else:
            m = -k
            grad[:n-m] += wr * dx * omh[m:]
            grad[m:] -= wr * dx * h_flat[:n-m]
    return F, grad

def _slsqp_eq(h_flat):
    n = _SLSQP_STATE['n']
    return np.sum(h_flat) - n / 2.0

def _slsqp_eq_grad(h_flat):
    n = _SLSQP_STATE['n']

    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    res = h.sum() - target_sum
    for _ in range(10):
        if abs(res) < 1e-12:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def smooth_continuation(h, deadline, tau_list=None, max_iter=200):
    n = len(h)
    _SLSQP_STATE['n'] = n
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_grad}]
    if tau_list is None:
        tau_list = [0.3, 0.2, 0.14, 0.1, 0.07, 0.05, 0.035, 0.025, 0.018, 0.012, 0.008, 0.005]
    best_h = h.copy()
    best_val = evaluate_exact(h)

        if time.time() > deadline:
            break
        _SLSQP_STATE['tau'] = tau

            res = minimize(_slsqp_obj, h, jac=True,

                          constraints=constraints,

            if res.x is not None and res.x.size == n:

                v = evaluate_exact(h_new)
                if v < best_val:
                    best_val = v
                    best_h = h_new.copy()
                h = h_new
        except Exception:
            pass
    return best_h, best_val

        is_high = (block_idx % (period_blocks * 2)) < period_blocks
