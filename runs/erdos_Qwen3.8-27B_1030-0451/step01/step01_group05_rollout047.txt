For all other s:
- 4 distinct terms, all linear:
  (i=a, j=a+s): if valid, lin += (1-h[a+s])

def compute_lin_quad(h, a, b):
    n = len(h)
    center = n - 1
    L = 2 * n - 1
    lin = np.zeros(L)
    quad = np.zeros(L)
    
    ha = h[a]
    hb = h[b]
    
    for K in range(L):
        s = K - center
        
        if s == b - a:
            # Special case: a+s = b, so (i=a,j=b) is a shared term
            # Term (i=a, j=b): both change
            lin[K] = (1.0 - hb) + ha
            quad[K] = -1.0
            # Term (i=a-s, j=a) = (i=2a-b, j=a)
            i2 = a - s  # = 2a - b
            if 0 <= i2 < n and i2 != b:
                lin[K] += -h[i2]
            # Term (i=b, j=b+s) = (i=b, j=2b-a)
            j2 = b + s  # = 2b - a
            if 0 <= j2 < n and j2 != a:
                lin[K] += -(1.0 - h[j2])
            # Note: (i=b-s, j=b) = (i=a, j=b) already counted
            
        elif s == a - b:
            # Special case: b+s = a, so (i=b,j=a) is a shared term
            # Term (i=b, j=a): both change
            lin[K] = -(1.0 - ha) - hb
            quad[K] = 1.0
            # Term (i=a, j=a+s) = (i=a, j=2a-b)
            j3 = a + s  # = 2a - b
            if 0 <= j3 < n and j3 != b:
                lin[K] += (1.0 - h[j3])
            # Term (i=b-s, j=b) = (i=2b-a, j=b)
            i3 = b - s  # = 2b - a
            if 0 <= i3 < n and i3 != a:
                lin[K] += h[i3]
            # Note: (i=a-s, j=a) = (i=b, j=a) already counted
            
        else:
            # General case: 4 distinct terms, all linear
            coeff = 0.0
            j1 = a + s
            if 0 <= j1 < n:
                coeff += (1.0 - h[j1])
            i1 = a - s
            if 0 <= i1 < n:
                coeff += -h[i1]
            j2 = b + s
            if 0 <= j2 < n:
                coeff += -(1.0 - h[j2])
            i2 = b - s
            if 0 <= i2 < n:
                coeff += h[i2]
            lin[K] = coeff
            # quad[K] = 0
    
    return lin, quad

Now for the line search. Given raw, lin, quad, and the valid interval [d_lo, d_hi]:

score(d) = dx * max_K (raw[K] + d*lin[K] + d²*quad[K])

I want to find d* = argmin over d in [d_lo, d_hi].

Since this might not be convex (due to the quad = -1 term), I'll use a grid + local refinement approach:

```python
def line_search(raw, lin, quad, d_lo, d_hi, dx, n_grid=30):
    """Find d in [d_lo, d_hi] that minimizes max_K(raw + d*lin + d*quad)."""
    if d_hi - d_lo < 1e-14:
        return 0.0
    
    best_d = 0.0
    best_val = np.max(raw)  # d=0 gives current score
    
    # Grid search
    ds = np.linspace(d_lo, d_hi, n_grid)
    for d in ds:
        vals = raw + d * lin + d * d * quad
        score = np.max(vals)
        if score < best_val:
            best_val = score
            best_d = d
    
    # Local refinement around best_d
    # Use a few iterations of golden-section-like refinement
    # But since it might not be convex, just do a finer grid locally
    if best_d > d_lo and best_d < d_hi:
        span = (d_hi - d_lo) / n_grid
        for _ in range(3):
            ds_fine = np.linspace(best_d - span, best_d + span, n_grid)
            ds_fine = np.clip(ds_fine, d_lo, d_hi)
            for d in ds_fine:
                vals = raw + d * lin + d * d * quad
                score = np.max(vals)
                if score < best_val:
                    best_val = score
                    best_d = d
            span *= 0.5
    
    return best_d
```

This is O(n_grid * (2n-1)) per line search, which is very fast for n=81.

def pair_descent(h, deadline, n_candidates=8, n_grid=30):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_score = np.max(raw) * dx
    best_score = current_score
    best_h = h.copy()
    
    while time.time() < deadline:
        # Compute sensitivity (subgradient approximation)
        # For the max, use lags within some tolerance of the max
        raw_max = np.max(raw)
        # Use all lags with weight proportional to how close they are to max
        weights = np.exp((raw - raw_max) / (0.01 * raw_max + 1e-10))
        weights /= np.sum(weights)
        
        # Compute per-coordinate sensitivity
        g = np.zeros(n)
        for K in range(2 * n - 1):
            w = weights[K]
            if w < 1e-10:
                continue
            s = K - (n - 1)
            for i in range(n):
                # Contribution of h[i] to raw[K]
                # h[i] appears as first factor when i+s is valid
                # h[i] appears as second factor (as 1-h[i-s]) when i-s is valid
                contrib = 0.0
                if 0 <= i + s < n:
                    contrib += (1.0 - h[i + s])
                if 0 <= i - s < n:
                    contrib += -h[i - s]
                g[i] += w * contrib
        
        # Sort by sensitivity
        order = np.argsort(g)
        
        # Generate candidate pairs
        pairs = []
        # Extreme pairs
        for i in range(min(n_candidates, n // 2)):
            a = int(order[n - 1 - i])  # high sensitivity (decrease)
            b = int(order[i])           # low sensitivity (increase)
            pairs.append((a, b))
        # Random pairs
        for _ in range(n_candidates // 2):
            a = np.random.randint(n)
            b = np.random.randint(n)
            if a != b:
                pairs.append((int(a), int(b)))
        
        improved = False
        for (a, b) in pairs:
            if a == b:
                continue
            
            # Try both directions: h[a]+=d,h[b]-=d and h[b]+=d,h[a]-=d
            for (ia, ib) in [(a, b), (b, a)]:
                lin, quad = compute_lin_quad(h, ia, ib)
                d_lo = max(-h[ia], h[ib] - 1.0)
                d_hi = min(1.0 - h[ia], h[ib])
                if d_hi <= d_lo + 1e-14:
                    continue
                
                d_star = line_search(raw, lin, quad, d_lo, d_hi, dx, n_grid)
                if abs(d_star) < 1e-14:
                    continue
                
                # Check if this improves
                new_raw = raw + d_star * lin + d_star * d_star * quad
                new_score = np.max(new_raw) * dx
                
                if new_score < current_score - 1e-12:
                    # Accept
                    h[ia] += d_star
                    h[ib] -= d_star
                    raw = new_raw
                    current_score = new_score
                    improved = True
                    if current_score < best_score:
                        best_score = current_score
                        best_h = h.copy()
        
        if not improved:
            # Try random pairs with aggressive search
            any_improve = False
            for _ in range(50):
                if time.time() > deadline:
                    break
                a = np.random.randint(n)
                b = np.random.randint(n)
                if a == b:
                    continue
                (ia, ib) = (a, b) if np.random.random() < 0.5 else (b, a)
                lin, quad = compute_lin_quad(h, ia, ib)
                d_lo = max(-h[ia], h[ib] - 1.0)
                d_hi = min(1.0 - h[ia], h[ib])
                if d_hi <= d_lo + 1e-14:
                    continue
                d_star = line_search(raw, lin, quad, d_lo, d_hi, dx, n_grid)
                if abs(d_star) < 1e-14:
                    continue
                new_raw = raw + d_star * lin + d_star * d_star * quad
                new_score = np.max(new_raw) * dx
                if new_score < current_score - 1e-12:
                    h[ia] += d_star
                    h[ib] -= d_star
                    raw = new_raw
                    current_score = new_score
                    any_improve = True
                    if current_score < best_score:
                        best_score = current_score
                        best_h = h.copy()
            if not any_improve:
                break
    
    return best_h, best_score

where active = {K : raw(K) = max(raw)}.

where Z is the normalization.

The partial derivative ∂raw(K)/∂h[i] for shift s = K - center is:

- minus h[i-s] if i-s is in [0, n-1]

```python
def compute_gradient(h, weights):
    """Compute weighted sum of per-coordinate sensitivities."""
    n = len(h)
    center = n - 1
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K]
        if abs(w) < 1e-12:
            continue
        s = K - center
        if s >= 0:
            # h[i] appears as first factor when i+s < n, i.e., i < n-s
            # h[i] appears as second factor (as 1-h) when i-s >= 0, i.e., i >= s
            # Contribution to g[i] from first factor: w * (1-h[i+s]) for i in [0, n-s-1]
            # Contribution to g[i] from second factor: w * (-h[i-s]) for i in [s, n-1]
            # g[i] += w * (1-h[i+s]) for i < n-s
            # g[i] += w * (-h[i-s]) for i >= s
            if s == 0:
                g[:] += w * (1.0 - h[:])
                g[:] += w * (-h[:])
            else:
                g[:n-s] += w * (1.0 - h[s:])
                g[s:] += w * (-h[:n-s])
        else:
            m = -s  # m > 0
            # h[i] appears as first factor when i+s < n, i.e., i < n-m (since s=-m)
            # Actually s = -m, so i+s = i-m, valid when i >= m
            # h[i] appears as first factor: partner j = i+s = i-m, valid when 0 <= i-m < n, i.e., m <= i < n
            # Contribution: w * (1-h[i-m])
            # h[i] appears as second factor: partner j = i-s = i+m, valid when 0 <= i+m < n, i.e., 0 <= i < n-m
            # Contribution: w * (-h[i+m])
            g[m:] += w * (1.0 - h[:n-m])
            g[:n-m] += w * (-h[m:])
    return g
```

import numpy as np
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)


def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_lin_quad(h, a, b):
    n = len(h)
    center = n - 1
    L = 2 * n - 1
    lin = np.zeros(L)
    quad = np.zeros(L)
    ha, hb = h[a], h[b]
    for K in range(L):
        s = K - center
        if s == b - a:
            lin[K] = (1.0 - hb) + ha
            quad[K] = -1.0
            i2 = a - s
            if 0 <= i2 < n and i2 != b:
                lin[K] += -h[i2]
            j2 = b + s
            if 0 <= j2 < n and j2 != a:
                lin[K] += -(1.0 - h[j2])
        elif s == a - b:
            lin[K] = -(1.0 - ha) - hb
            quad[K] = 1.0
            j3 = a + s
            if 0 <= j3 < n and j3 != b:
                lin[K] += (1.0 - h[j3])
            i3 = b - s
            if 0 <= i3 < n and i3 != a:
                lin[K] += h[i3]
        else:
            c = 0.0
            j1 = a + s
            if 0 <= j1 < n:
                c += (1.0 - h[j1])
            i1 = a - s
            if 0 <= i1 < n:
                c += -h[i1]
            j2 = b + s
            if 0 <= j2 < n:
                c += -(1.0 - h[j2])
            i2 = b - s
            if 0 <= i2 < n:
                c += h[i2]
            lin[K] = c
    return lin, quad


def line_search(raw, lin, quad, d_lo, d_hi, n_grid=50):
    if d_hi - d_lo < 1e-14:
        return 0.0
    best_d = 0.0
    best_val = float(np.max(raw))
    q_idx = np.where(quad != 0.0)[0]
    has_quad = len(q_idx) > 0
    ds = np.linspace(d_lo, d_hi, n_grid)
    for d in ds:
        v = raw + d * lin
        if has_quad:
            v = v.copy()
            v[q_idx] += d * d * quad[q_idx]
        sc = float(np.max(v))
        if sc < best_val:
            best_val = sc
            best_d = float(d)
    if best_d != 0.0 and best_d > d_lo and best_d < d_hi:
        span = max((d_hi - d_lo) / n_grid, 1e-12)
        for _ in range(5):
            lo_r = max(d_lo, best_d - span)
            hi_r = min(d_hi, best_d + span)
            if hi_r - lo_r < 1e-14:
                break
            ds_f = np.linspace(lo_r, hi_r, n_grid)
            for d in ds_f:
                v = raw + d * lin
                if has_quad:
                    v = v.copy()
                    v[q_idx] += d * d * quad[q_idx]
                sc = float(np.max(v))
                if sc < best_val:
                    best_val = sc
                    best_d = float(d)
            span *= 0.35
    return best_d


def compute_gradient(h, weights):
    n = len(h)
    center = n - 1
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K]
        if abs(w) < 1e-12:
            continue
        s = K - center
        if s == 0:
            g[:] += w * (1.0 - 2.0 * h[:])
        elif s > 0:
            g[:n - s] += w * (1.0 - h[s:])
            g[s:] += w * (-h[:n - s])
        else:
            m = -s
            g[m:] += w * (1.0 - h[:n - m])
            g[:n - m] += w * (-h[m:])
    return g

    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full').astype(float)
    current_score = float(np.max(raw) * dx)
    best_score = current_score
    best_h = h.copy()
    no_improve = 0
    tau = 0.04
    iter_count = 0

    while time.time() < deadline:
        raw_max = float(np.max(raw))
        exponents = np.clip((raw - raw_max) / tau, -500.0, 500.0)
        weights = np.exp(exponents)
        sw = float(np.sum(weights))
        if sw < 1e-300:
            weights = np.ones(2 * n - 1) / (2 * n - 1)
        else:
            weights /= sw

        g = compute_gradient(h, weights)
        order = np.argsort(g)

        pairs = []
        half = min(n_cand, n // 3)
        for i in range(half):
            a = int(order[n - 1 - i])
            b = int(order[i])
            if a != b:

        half = min(n_cand - 2 * half, n // 4)
        for i in range(max(1, half)):

            b = int(order[min(n - 1, n // 2 + i)])

        for _ in range(max(2, n_cand // 3)):

            best_score = s_ps2

    # Phase 4: Try different dimensions
    if time.time() < deadline - 40:
        dim_options = []
        for factor in [0.9, 0.95, 1.05, 1.1, 1.2, 0.85, 1.15, 1.3, 0.75]:
            cand_n = int(round(best_n * factor))
            if 40 <= cand_n <= 200 and cand_n != best_n:
                dim_options.append(cand_n)
        dim_options = sorted(set(dim_options))

        for dn in dim_options:
            if time.time() > deadline - 20:
                break
            h_d = interpolate_to_n(best_h, dn)
            s_d = exact_score(h_d)
            if s_d < best_score + 0.003:
                h_d, s_d = pair_descent(h_d, min(deadline - 10, t0 + (deadline - t0) * (0.7 + 0.3 * len(dim_options) / max(1, len(dim_options)))), rng,
                                        max_no_improve=100, n_cand=10, n_grid=40)
                if s_d < best_score:
                    best_h = h_d
                    best_score = s_d
                    best_n = dn

    # Phase 5: Another SA round if time remains
    if time.time() < deadline - 40:
        raw_cur = np.correlate(best_h, 1.0 - best_h, mode='full').astype(float)
        h_sa2, s_sa2 = sa_exploration(best_h.copy(), raw_cur, deadline - 20, rng,
                                      n_iter=15000, init_temp=0.002 * best_score)
        if s_sa2 < best_score:
            best_h = h_sa2
            best_score = s_sa2

    # Phase 6: Final polish
    if time.time() < deadline - 10:
        h_work = best_h.copy()
        h_fin, s_fin = pair_descent(h_work, deadline, rng, max_no_improve=200, n_cand=20, n_grid=80)
        if s_fin < best_score:
            best_h = h_fin
            best_score = s_fin

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-11:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
