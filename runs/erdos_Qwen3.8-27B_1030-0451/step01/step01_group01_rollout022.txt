def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target} via bisection."""
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-13:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-14:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-14:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    r = target - y.sum()
    if r > 1e-12:
        idx = np.where(y < 1.0 - 1e-14)[0]
        if len(idx) > 0:
            y[idx] += r / len(idx)
    elif r < -1e-12:

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        m = n - lag
        if m > 0:
            g[:m] += (1.0 - h[lag:lag + m])
            g[lag:lag + m] -= h[:m]
    else:
        l = -lag
        m = n - l
        if m > 0:
            g[:m] += (1.0 - h[l:l + m])
            g[l:l + m] -= h[:m]
    return g

    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    betas = [6, 12, 24, 48, 96, 192, 384, 768][:max_betas]
    bounds = [(0.0, 1.0)] * n

    def sum_fun(x):
        return x.sum() - n / 2.0

    def sum_jac(x):
        return np.ones(n)

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': sum_jac}]

    for beta in betas:
        if time.time() > deadline:
            break
        for attempt in range(3):
            if time.time() > deadline:
                break
            if attempt == 0:
                x0 = best_h.copy()
            elif attempt == 1:
                x0 = best_h + np.random.randn(n) * 0.004
                x0 = project_sum_box(x0, n / 2.0)
            else:
                x0 = best_h + np.random.randn(n) * 0.012
                x0 = project_sum_box(x0, n / 2.0)

            try:
                res = minimize(
                    lambda x: smooth_obj_and_grad(x, beta)[0],
                    x0,
                    jac=lambda x: smooth_obj_and_grad(x, beta)[1],
                    method='SLSQP', bounds=bounds, constraints=cons,
                    options={'maxiter': 250, 'ftol': 1e-15})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-13:
                    best_h = x_new.copy()
                    best_c = c_new

    return best_h, best_c

def method_nesterov_refine(h0, deadline, max_iters=15000):
    """Nesterov accelerated projected gradient with LSE smoothing."""
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    y = h0.copy()
    t_k = 1.0
    base_step = 0.015
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        f_y, g_y = smooth_obj_and_grad(y, 25.0)
        g_norm = np.linalg.norm(g_y)
        if g_norm < 1e-14:
            break
        step = base_step / (1.0 + 0.005 * it)
        h = project_sum_box(y - step * g_y, n / 2.0)
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t_k * t_k)) / 2.0
        alpha = (t_k - 1.0) / t_new
        y = project_sum_box((1.0 + alpha) * h - alpha * y, n / 2.0)
        t_k = t_new

        if it % 50 == 0:
            c_y = evaluate_c5(y)
            if c_y < best_c - 1e-13:
                best_c = c_y
                best_h = y.copy()
                stall = 0
            else:
                stall += 1
                if stall > 60:
                    break

    c_final = evaluate_c5(y)
    if c_final < best_c:
        best_c = c_final
        best_h = y.copy()
    return best_h, best_c

def method_direct_descent(h0, deadline, max_iters=500):
    """Targeted descent on the maximizer with feasibility check."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    step = 0.01

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        k_star = int(np.argmax(v))
        lag_star = k_star - (n - 1)
        g_star = grad_lag(h, lag_star)

        # Try several step sizes
        improved = False
        for s in [step, step * 0.5, step * 0.25, step * 2.0]:
            h_try = h - s * g_star
            h_try = project_sum_box(h_try, n / 2.0)
            c_try = evaluate_c5(h_try)
            if c_try < best_c - 1e-13:
                best_c = c_try
                best_h = h_try.copy()
                h = h_try.copy()
                step = min(step * 1.1, 0.03)
                improved = True
                break
        if not improved:
            step *= 0.7
            if step < 1e-6:
                break

    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    t_old = np.linspace(0, 1, n_old)
    t_new = np.linspace(0, 1, n_new)
    h_new = np.interp(t_new, t_old, h0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0
    deadline = start_time + budget_s - safety

    def time_left():
        return deadline - time.time()

    # Get initial values
    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        n = 84
        h_init = np.full(n, 0.5)
        h_init = project_sum_box(h_init, n / 2.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    # Phase 1: LP trust region (primary driver)
    if time_left() > 100:
        dl1 = min(deadline, start_time + budget_s * 0.30)
        try:
            h1, c1 = method_lp_trust(best_h, dl1, max_iters=500, init_radius=0.06)
            if c1 < best_c:
                best_h = h1
                best_c = c1
        except Exception:
            pass

    # Phase 2: SLSQP refinement
    if time_left() > 60:
        dl2 = min(deadline, start_time + budget_s * 0.50)
        try:
            h2, c2 = method_slsqp_refine(best_h, dl2, max_betas=8)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    # Phase 3: Second LP round (tighter)
    if time_left() > 50:
        dl3 = min(deadline, start_time + budget_s * 0.68)
        try:
            h3, c3 = method_lp_trust(best_h, dl3, max_iters=400, init_radius=0.03)
            if c3 < best_c:
                best_h = h3
                best_c = c3
        except Exception:
            pass

    # Phase 4: Nesterov + direct descent polish
    if time_left() > 30:
        dl4 = min(deadline, start_time + budget_s * 0.78)
        try:
            h4, c4 = method_nesterov_refine(best_h, dl4, max_iters=12000)
            if c4 < best_c:
                best_h = h4
                best_c = c4
        except Exception:
            pass
    if time_left() > 20:
        dl5 = min(deadline, start_time + budget_s * 0.85)
        try:
            h5, c5 = method_direct_descent(best_h, dl5, max_iters=800)
            if c5 < best_c:
                best_h = h5
                best_c = c5
        except Exception:
            pass

    # Phase 5: Resolution transfer and re-optimize
    if time_left() > 80:
        n_cur = len(best_h)
        targets = sorted(set([
            int(round(n_cur * 0.85)),
            int(round(n_cur * 1.15)),
            int(round(n_cur * 1.3)),
            int(round(n_cur * 1.5)),
        ]))
        targets = [t for t in targets if 20 <= t <= 300 and t != n_cur]
        for n_t in targets:
            if time_left() < 45:
                break
            try:
                h_tr = transfer_resolution(best_h, n_t)
                c_tr = evaluate_c5(h_tr)
                if c_tr < best_c:
                    best_h = h_tr.copy()
                    best_c = c_tr
                sub_dl = min(deadline - 5, time.time() + min(100, time_left() * 0.35))
                h_a, c_a = method_lp_trust(best_h, sub_dl, max_iters=120, init_radius=0.04)
                if c_a < best_c:
                    best_h = h_a
                    best_c = c_a
                sub_dl2 = min(deadline - 5, time.time() + min(40, time_left() * 0.2))
                h_b, c_b = method_direct_descent(best_h, sub_dl2, max_iters=300)
                if c_b < best_c:
                    best_h = h_b
                    best_c = c_b
            except Exception:
                pass

    # Phase 6: Final ultra-tight LP polish
    if time_left() > 15:
        try:
            h6, c6 = method_lp_trust(best_h, deadline - 2, max_iters=200, init_radius=0.015)
            if c6 < best_c:
                best_h = h6
                best_c = c6
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
