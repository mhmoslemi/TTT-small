```python
def compute_delta_off(h):
    n = len(h)
    d = 2*n - 1
    delta = np.zeros((n, d))
    for j in range(n):
        for m in range(d):
            lag = m - (n-1)
            i1 = j - lag  # where h[j] appears as h[i]
            i2 = j + lag  # where h[j] appears as h[i-lag]
            val = 0.0
            if 0 <= i1 < n:
                val += -(1 - h[i1])
            if 0 <= i2 < n:
                val += h[i2]
            delta[j, m] = val
    return delta
```

For the continuous case, the integral at shift k is:

For k = 0: C(0) = ∫h(x)(1-h(x))dx. If h ∈ [0,1], this is ≤ ∫h(x)dx = 1 (achieved when h is binary). For smooth h, it's less.

For large |k| (|k| > 2): C(k) = 0 (no overlap).

For k near 1: C(1) = ∫₀¹ h(x)(1-h(x+1))dx + ∫₁² h(x)(1-h(x+1))dx
The second integral is 0 since x+1 > 2 for x > 1.

If h has the property h(x+1) = 1-h(x) for x ∈ [0,1] (anti-periodicity), then C(1) = ∫₀¹ h(x)*h(x)dx = ∫₀¹ h²(x)dx.

Actually, here's another idea: **coordinate descent**. Instead of moving all variables at once, update one variable at a time (or a small group).

For a single variable h[j], given all other h[i] fixed, the objective is:
f(h[j]) = max_k (sum_{i≠j} contribution_i + contribution_j(h[j]))

where contribution_j(h[j]) is the part of corr_k that involves h[j].

For a specific k, the contribution of h[j] to corr_k is:
- If j appears as h[i] (i=j): h[j]*(1-h[j-lag_k])
- If j appears as h[i-lag_k] (i=j+lag_k): h[j+lag_k]*(1-h[j])

So f_j(h[j]) = max_k (A_k + h[j]*(1-h[j-lag_k])*[j-lag_k valid] + h[j+lag_k]*(1-h[j])*[j+lag_k valid])

where A_k is the sum of all other contributions to corr_k.

This is a piecewise linear function of h[j] (since each term is linear in h[j]). The minimum of the max of piecewise linear functions can be found by checking the breakpoints.

The breakpoints occur where h[j] = h[j-lag_k] or h[j] = 1-h[j+lag_k], etc. There are O(n) such breakpoints.

So for each j, the optimal h[j] is either 0, 1, or one of O(n) breakpoint values. We can find the minimum by evaluating f_j at all breakpoints.

This gives O(n) per coordinate, O(n²) per full sweep. Very fast!

```python
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):

                grad[lag:] += wk * (1.0 - h[:n - lag])

            if L < n:

            grad = smooth_gradient(h, corr, tau)
            gnorm = float(np.linalg.norm(grad))
            
            if gnorm < 1e-14:
                h = project_constraint(h + rng.randn(n) * 0.01, n)
                continue
            
            made = False
            for _ in range(15):
                h_new = project_constraint(h - step * grad, n)
                c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode="full")) * dx
                if c_new <= c5_val + 1e-13:
                    h = h_new
                    made = c_new < c5_val - 1e-12
                    break
                step *= 0.5
            
            if made:
                step = min(step * 1.08, 6.0 / n)
            elif step < 0.05 / n:
                step = 0.05 / n
            else:
                step *= 0.9
        
        if p < n_passes - 1 and time.time() - t0 < time_limit - 10:
            h = project_constraint(h + rng.randn(n) * 0.03, n)
    
    return best_h, best_c5

def coordinate_descent(h0, time_limit, seed=42):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)
    order = np.arange(n)
    stall = 0

    while time.time() - t0 < time_limit:
        np.random.seed(seed + int((time.time() - t0) * 7))
        np.random.shuffle(order)
        improved = False
        for j in order:
            if time.time() - t0 > time_limit - 2:
                break
            corr = np.correlate(h, 1.0 - h, mode="full")
            c5_cur = float(np.max(corr) * dx)
            
            # Try moving h[j] to 0 and 1, and a few intermediate values
            candidates = [0.0, 1.0]
            # Add breakpoints: values where h[j] = h[j-lag] or 1-h[j+lag]
            for m in range(2 * n - 1):
                lag = m - (n - 1)
                i1 = j - lag
                i2 = j + lag
                if 0 <= i1 < n:
                    v = 1.0 - h[i1]
                    if 0.0 < v < 1.0:
                        candidates.append(v)
                if 0 <= i2 < n:
                    v = h[i2]
                    if 0.0 < v < 1.0:
                        candidates.append(v)
            
            best_val = h[j]
            best_score = c5_cur
            
            for v in set(candidates):
                h[j] = v
                h_proj = project_constraint(h, n)
                c_new = float(np.max(np.correlate(h_proj, 1.0 - h_proj, mode="full")) * dx)
                if c_new < best_score - 1e-14:
                    best_score = c_new
                    best_val = v
                    best_h_proj = h_proj.copy()
            
            if best_val != h[j]:
                h = project_constraint(h, n)
                h[j] = project_constraint(np.concatenate([h[:j], [best_val], h[j+1:]]), n)
                c5_new = compute_c5(h)
                if c5_new < best_c5 - 1e-14:
                    best_c5 = c5_new
                    best_h = h.copy()
                    improved = True
                else:
                    h = project_constraint(h, n)
        
        if improved:
            stall = 0
        else:
            stall += 1
            if stall > 3:
                h = project_constraint(h + rng.randn(n) * 0.02, n)
                stall = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    
    best_h = None
    best_c5 = float('inf')
    
    def update(h_cand, c5_cand):
        nonlocal best_h, best_c5
        if c5_cand < best_c5:
            best_c5 = c5_cand
            best_h = h_cand.copy()
    
    # Phase 1: Start from initial_h_values if available
    n_base = 81
    try:
        h0 = np.array(initial_h_values, dtype=float)
        n_base = len(h0)
        h0 = project_constraint(h0, n_base)
        c5_0 = compute_c5(h0)
        update(h0, c5_0)
    except (NameError, UnboundLocalError):
        h0 = make_structured_start(n_base, seed)
        update(h0, compute_c5(h0))
    
    # Phase 2: Smooth refine from initial
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 300
    if remaining > 30:
        h_r, c5_r = smooth_refine(h0.copy(), remaining, seed)
        update(h_r, c5_r)
    
    # Phase 3: Binary local search from multiple starts
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 200
    
    if remaining > 40:
        # Binary from initial
        h_bin = np.zeros(n_base)
        sorted_idx = np.argsort(h0)
        h_bin[sorted_idx[n_base//2:]] = 1.0
        c5_bin = compute_c5(h_bin)
        update(h_bin, c5_bin)
        
        alloc = min(remaining * 0.4, 200)
        if alloc > 20:
            h_b, c5_b = binary_local_search(h_bin, alloc, seed)
            update(h_b, c5_b)
    
    # Phase 4: Binary local search with different n and starts
    for n_try in [121, 161, 201]:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 80
        if remaining < 25:
            break
        
        h_start = make_binary_start(n_try, seed + n_try * 3)
        alloc = min(remaining * 0.35, 120)
        if alloc > 15:
            h_b, c5_b = binary_local_search(h_start, alloc, seed + n_try)
            update(h_b, c5_b)
    
    # Phase 5: Smooth refine best binary solution
    if best_h is not None:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 40
        if remaining > 20:
            h_r, c5_r = smooth_refine(best_h.copy(), remaining, seed + 999)
            update(h_r, c5_r)
    
    # Phase 6: Coordinate descent on best solution
    if best_h is not None:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 20
        if remaining > 15:
            h_cd, c5_cd = coordinate_descent(best_h.copy(), remaining, seed + 777)
            update(h_cd, c5_cd)
    
    # Phase 7: Final smooth refine
    if best_h is not None:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 5
        if remaining > 10:
            h_r, c5_r = smooth_refine(best_h.copy(), remaining, seed + 1234)
            update(h_r, c5_r)
    
    if best_h is None:
        best_h = h0
        best_c5 = compute_c5(h0)
    
    best_h = project_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
```
