def project_constraint(p, n):
    target = n / 2.0
    p = np.clip(p, -10, 10)  # handle any extreme values
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0

        proj = np.clip(p - mid, 0.0, 1.0)

def corr_gradient(h, lag):
    """Gradient of correlation at given lag index (0 to 2n-2)."""
    n = len(h)
    lag_val = lag - (n - 1)
    grad = np.zeros(n)
    for j in range(n):
        # First term: h[j] appears as first factor when j-lag_val is valid
        j2 = j - lag_val
        if 0 <= j2 < n:
            grad[j] += (1.0 - h[j2])
        # Second term: h[j] appears in (1-h[i-lag_val]) when i-lag_val = j, i = j+lag_val
        j3 = j + lag_val
        if 0 <= j3 < n:
            grad[j] -= h[j3]
    return grad

def lp_direction(h, c_vals, f, radius, eta):
    """Solve LP trust-region subproblem to find descent direction."""
    n = len(h)
    m = 2 * n - 1  # number of lags
    
    # Variables: delta[0..n-1], s[0..n-1], t
    # n + n + 1 = 2n+1 variables
    nv = 2 * n + 1
    
    # Objective: minimize t + eta * sum(s)
    c_obj = np.zeros(nv)
    c_obj[n] = eta  # s variables
    c_obj[n + n] = 1.0  # t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag k: grad_k @ delta <= t - c_k + f  =>  grad_k @ delta - t <= f - c_k
    A_rows = []
    b_rows = []
    for k in range(m):

        row = np.zeros(nv)
        row[:n] = gk
        row[n + n] = -1.0  # -t
        A_rows.append(row)
        b_rows.append(f - c_vals[k])
    
    # Trust region: |delta_i| <= radius
    # delta_i <= radius, delta_i >= -radius
    for i in range(n):
        row = np.zeros(nv)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(radius)
        row2 = np.zeros(nv)
        row2[i] = -1.0
        A_rows.append(row2)
        b_rows.append(radius)
    
    # s_i >= delta_i => -delta_i + s_i >= 0 => delta_i - s_i <= 0
    # s_i >= -delta_i => delta_i + s_i >= 0 => -delta_i - s_i <= 0
    for i in range(n):
        row = np.zeros(nv)
        row[i] = 1.0
        row[n + i] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
        row2 = np.zeros(nv)
        row2[i] = -1.0
        row2[n + i] = -1.0
        A_rows.append(row2)
        b_rows.append(0.0)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(delta) = 0
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Bounds
    bounds = [(0, None)] * n  # delta is free... wait, we have trust region bounds
    # Actually delta is bounded by [-radius, radius]
    bounds = [(-radius, radius)] * n + [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            delta = res.x[:n]
            # Check sum
            if abs(np.sum(delta)) < 1e-6:
                return delta
            else:
                delta -= np.sum(delta) / n  # project to zero sum
                return delta
    except Exception:
        pass
    return None

    n = len(h)
    f = np.max(c_vals)
    shifted = c_vals - f
    w = np.exp(tau * shifted)
    ws = np.sum(w)
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    G = all_corr_grads(h)
    grad = G.T @ w
    return grad

def lp_direction(h, c_vals, f, radius, eta):
    n = len(h)
    m = 2 * n - 1
    G = all_corr_grads(h)
    nv = 2 * n + 1
    c_obj = np.zeros(nv)
    c_obj[n:2*n] = eta
    c_obj[2*n] = 1.0
    A_rows = []
    b_rows = []
    for k in range(m):
        row = np.zeros(nv)
        row[:n] = G[k]
        row[2*n] = -1.0
        A_rows.append(row)
        b_rows.append(f - c_vals[k])
    for i in range(n):
        row = np.zeros(nv)
        row[i] = 1.0
        A_rows.append(row)
        b_rows.append(radius)
    for i in range(n):
        row = np.zeros(nv)
        row[i] = -1.0
        A_rows.append(row)
        b_rows.append(radius)
    for i in range(n):
        row = np.zeros(nv)
        row[i] = 1.0
        row[n+i] = -1.0
        A_rows.append(row)
        b_rows.append(0.0)
        row2 = np.zeros(nv)
        row2[i] = -1.0
        row2[n+i] = -1.0
        A_rows.append(row2)
        b_rows.append(0.0)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = [(-radius, radius)] * n + [(0, None)] * n + [(0, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success:
            delta = res.x[:n].copy()
            delta -= np.sum(delta) / n
            if np.abs(delta).max() > radius * 1.01:
                scale = radius / (np.abs(delta).max() + 1e-30)
                delta *= scale
                delta -= np.sum(delta) / n
            return delta
    except Exception:
        pass
    return None

def line_search(h, direction, c_vals, f, n_iter=8):
    n = len(h)
    best_step = 0.0
    best_h = h.copy()
    best_f = f
    step = 1.0
    for _ in range(n_iter):
        h_try = project_constraint(h + step * direction, n)
        c_try = compute_corr_vals(h_try)
        f_try = float(np.max(c_try))
        if f_try < best_f - 1e-13:
            best_f = f_try
            best_h = h_try.copy()
            best_step = step
        step *= 0.5
        if step < 1e-6:
            break
    return best_h, best_step, best_f

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.05 * np.sin(6.0 * np.pi * x + 1.3)
    h += 0.03 * np.sin(10.0 * np.pi * x + 2.5)
    h += rng.randn(n) * 0.01
    return project_constraint(h, n)

def local_optimize_lp(h0, time_limit, seed_val=42):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed_val)
    radius = 0.03
    eta = 0.001
    stall_count = 0
    iter_count = 0

    while time.time() - t0 < time_limit - 2:
        c_vals = compute_corr_vals(h)
        f = float(np.max(c_vals))
        if f < best_c5 - 1e-13:
            best_c5 = f
            best_h = h.copy()

        direction = lp_direction(h, c_vals, f, radius, eta)
        made_progress = False

        if direction is not None and np.abs(direction).max() > 1e-10:
            h_new, step_used, f_new = line_search(h, direction, c_vals, f)
            if f_new < f - 1e-13:
                h = h_new
                radius = min(radius * 1.08, 0.15)
                stall_count = 0
                made_progress = True
        elif direction is not None:
            grad = smooth_gradient(h, c_vals, 200.0)
            gnorm = float(np.linalg.norm(grad))
            if gnorm > 1e-10:
                grad_dir = -grad / gnorm
                h_new, step_used, f_new = line_search(h, grad_dir, c_vals, f)
                if f_new < f - 1e-13:
                    h = h_new
                    radius = max(radius * 0.9, 0.01)
                    stall_count = 0
                    made_progress = True

        if not made_progress:
            stall_count += 1
            radius *= 0.85
            if radius < 0.005:
                radius = 0.04
                noise_scale = 0.02 + 0.01 * stall_count * 0.1
                h = project_constraint(h + rng.randn(n) * noise_scale, n)
                stall_count = 0
                eta = 0.001 * (1.0 + 0.5 * (iter_count % 5))

        iter_count += 1
        if iter_count % 50 == 0 and stall_count > 15:
            h = best_h.copy() + rng.randn(n) * 0.015
            h = project_constraint(h, n)
            radius = 0.05
            stall_count = 0

    return best_h, best_c5

def coordinate_pair_search(h, time_limit, seed_val=99):
    n = len(h)
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed_val)

    while time.time() - t0 < time_limit - 2:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        max_transfer = min(h[i], 1.0 - h[j])
        if max_transfer < 1e-10:
            continue
        fractions = [0.1, 0.25, 0.4, 0.5, 0.6, 0.75, 0.9]
        best_f = best_c5
        best_pair_h = None
        for fr in fractions:
            transfer = fr * max_transfer
            h_try = h.copy()
            h_try[i] -= transfer
            h_try[j] += transfer
            c_try = compute_c5(h_try)
            if c_try < best_f - 1e-12:
                best_f = c_try
                best_pair_h = h_try.copy()
        if best_pair_h is not None:
            h = best_pair_h
            best_h = h.copy()
            best_c5 = best_f

    return best_h, best_c5

def resample_to_n(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 1.0 / n_old
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
    h_interp = np.interp(x_new, x_old, h_old)
    return project_constraint(h_interp, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def update_best(h_cand, c5_cand):
        nonlocal best_h, best_c5, best_n
        if c5_cand < best_c5 - 1e-12:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            best_n = len(h_cand)

    # Phase 0: Get initial from parent or generate
    n_base = 81
    try:
        h_parent = np.asarray(initial_h_values, dtype=float)
        n_base = len(h_parent)
        h_parent = project_constraint(h_parent, n_base)
        c5_parent = compute_c5(h_parent)
        best_h = h_parent.copy()
        best_c5 = c5_parent
        best_n = n_base
    except (NameError, UnboundLocalError):
        h_parent = make_start(n_base, seed)
        best_h = h_parent.copy()
        best_c5 = compute_c5(h_parent)
        best_n = n_base

    # Phase 1: LP optimization on parent
    elapsed = time.time() - t_start
    alloc1 = min(budget_s * 0.35, budget_s - elapsed - 60)
    if alloc1 > 20:
        h_opt, c5_opt = local_optimize_lp(best_h.copy(), alloc1, seed)
        if c5_opt < best_c5:
            update_best(h_opt, c5_opt)

    # Phase 2: Multi-start with same n
    elapsed = time.time() - t_start
    remaining_multi = budget_s - elapsed - 100
    if remaining_multi > 40:
        n_starts = 4
        per_start = max(10, remaining_multi / n_starts - 5)
        for s in range(n_starts):
            elapsed = time.time() - t_start
            if budget_s - elapsed < 30:
                break
            rng_s = np.random.RandomState(seed + 1000 + s * 7)
            h_start = project_constraint(best_h + rng_s.randn(best_n) * 0.04, best_n)
            h_opt, c5_opt = local_optimize_lp(h_start, min(per_start, budget_s - elapsed - 30), seed + s)
            if c5_opt < best_c5:
                update_best(h_opt, c5_opt)

    # Phase 3: Try different n values
    elapsed = time.time() - t_start
    n_targets = []
    if best_n <= 90:
        n_targets = [best_n + 10, best_n + 20, best_n + 40, best_n + 70]
    else:
        n_targets = [best_n - 10, best_n - 30, int(best_n * 1.2), int(best_n * 1.5)]
    n_targets = [nt for nt in n_targets if 40 <= nt <= 250]
    remaining_res = budget_s - elapsed - 80
    if remaining_res > 30 and len(n_targets) > 0:
        per_res = max(15, remaining_res / len(n_targets) - 5)
        for nt in n_targets:
            elapsed = time.time() - t_start
            if budget_s - elapsed < 25:
                break
            h_resamped = resample_to_n(best_h, nt)
            h_opt, c5_opt = local_optimize_lp(h_resamped, min(per_res, budget_s - elapsed - 25), seed + nt)
            if c5_opt < best_c5:
                update_best(h_opt, c5_opt)

    # Phase 4: Coordinate pair search on best
    elapsed = time.time() - t_start
    remaining_pair = budget_s - elapsed - 15
    if remaining_pair > 10 and best_h is not None:
        h_opt, c5_opt = coordinate_pair_search(best_h.copy(), remaining_pair, seed + 500)
        if c5_opt < best_c5:
            update_best(h_opt, c5_opt)

    # Phase 5: Final LP polish + pair search
    elapsed = time.time() - t_start
    remaining_final = budget_s - elapsed - 3
    if remaining_final > 8 and best_h is not None:
        half = remaining_final // 2
        if half > 5:
            h_opt, c5_opt = local_optimize_lp(best_h.copy(), half, seed + 777)
            if c5_opt < best_c5:
                update_best(h_opt, c5_opt)
        remaining2 = budget_s - time.time() - 2
        if remaining2 > 8:
            h_opt, c5_opt = coordinate_pair_search(best_h.copy(), remaining2, seed + 888)
            if c5_opt < best_c5:
                update_best(h_opt, c5_opt)

    # Final verification
    best_h = project_constraint(best_h, best_n)
    best_c5 = compute_c5(best_h)

    # Fallback if something went wrong
    if best_h is None or not np.all(np.isfinite(best_h)) or best_c5 > 0.5:
        best_h = np.full(n_base, 0.5)
        best_c5 = compute_c5(best_h)
        best_n = n_base

    return best_h, best_c5, best_n
