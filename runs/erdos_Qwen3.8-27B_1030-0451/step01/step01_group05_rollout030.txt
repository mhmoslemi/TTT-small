import numpy as np
import time

def exact_score(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):
    # Bisection to find mu such that sum(clip(z-mu, 0, 1)) = n/2
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    # Fix exact sum
    diff = S - np.sum(h)
    if abs(diff) > 1e-15:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            adj = diff / cnt
            h[interior] += adj
            h = np.clip(h, 0.0, 1.0)
    return h

def line_coeffs(h, d):
    R0 = np.correlate(h, 1.0 - h, mode='full')
    R1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    R2 = -np.correlate(d, d, mode='full')
    return R0, R1, R2

def feas_interval(h, d):
    n = len(h)
    t_lo = -1e15
    t_hi = 1e15
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, np.max(-h[pos] / d[pos]))
        t_hi = min(t_hi, np.max((1.0 - h[pos]) / d[pos]))  # wait, should be min
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = max(t_lo, np.min((1.0 - h[neg]) / d[neg]))
        t_hi = min(t_hi, np.max(-h[neg] / d[neg]))
    # Clamp
    t_lo = max(t_lo, -5.0)
    t_hi = min(t_hi, 5.0)
    return t_lo, t_hi

def minmax_line(R0, R1, R2, t_lo, t_hi):
    """Minimize max(R0 + t*R1 + t^2*R2) over t in [t_lo, t_hi]"""
    if t_hi <= t_lo:
        return 0.0, float(np.max(R0))
    
    # Build candidate set
    cands = [t_lo, t_hi]
    if t_lo < 0 < t_hi:
        cands.append(0.0)
    
    # Vertices
    neg_R2 = R2 < -1e-15
    if np.any(neg_R2):
        tverts = -R1[neg_R2] / (2.0 * R2[neg_R2])
        valid = (tverts >= t_lo) & (tverts <= t_hi)
        cands.extend(tverts[valid].tolist())
    
    # Grid
    span = t_hi - t_lo
    if span > 1e-10:
        ngrid = min(80, max(20, int(span * 20)))
        grid = np.linspace(t_lo, t_hi, ngrid)
        cands.extend(grid.tolist())
    
    # Evaluate
    best_t = cands[0]
    best_v = float('inf')
    for tc in cands:
        if tc < t_lo - 1e-14 or tc > t_hi + 1e-14:
            continue
        tc = max(t_lo, min(t_hi, tc))
        v = np.max(R0 + tc * R1 + tc * tc * R2)
        if v < best_v:
            best_v = v
            best_t = tc
    
    # Golden section refinement
    # Work in a window around best_t
    win = span * 0.1
    a = max(t_lo, best_t - win)
    b = min(t_hi, best_t + win)
    if b - a > 1e-14:
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        c = b - gr * (b - a)
        d_pt = a + gr * (b - a)
        fc = np.max(R0 + c * R1 + c * c * R2)
        fd = np.max(R0 + d_pt * R1 + d_pt * d_pt * R2)

def exact_score(h):
    dx = 2.0 / len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-15:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def line_coeffs(h, d):
    R0 = np.correlate(h, 1.0 - h, mode='full')
    R1 = np.correlate(d, 1.0 - h, mode='full') - np.correlate(h, d, mode='full')
    R2 = -np.correlate(d, d, mode='full')
    return R0, R1, R2


def feas_interval(h, d):

    t_hi = np.inf
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, float(np.max(-h[pos] / d[pos])))
        t_hi = min(t_hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        t_lo = max(t_lo, float(np.max((1.0 - h[neg]) / d[neg])))
        t_hi = min(t_hi, float(np.min(-h[neg] / d[neg])))
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)

            no_improve += 1

        if no_improve > 300 and iter_count % 150 == 0:
            pert = 0.02 * rng.standard_normal(n)
            pert -= np.mean(pert)
            h_pert = project_feasible(h + pert, n)
            s_pert = exact_score(h_pert)
            if s_pert < best_score + 0.02:
                h = h_pert
                cur_score = s_pert
                no_improve = 0

        if no_improve > 500:
            break

    return best_h, best_score


def interpolate_to(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)


def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 30.0

    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        h0 = np.full(81, 0.5)

    n0 = len(h0)
    h0 = project_feasible(h0, n0)
    score0 = exact_score(h0)

    best_h = h0.copy()
    best_score = score0

    # Phase 1: Optimize on base dimension
    h_work = h0.copy()
    t_phase1_deadline = min(deadline - 120, t0 + int(budget_s * 0.45))
    h_opt, s_opt = optimize_h(h_work, n0, rng, t_phase1_deadline, max_iters=100000)
    if s_opt < best_score:
        best_h = h_opt.copy()
        best_score = s_opt

    # Phase 2: Multi-n exploration
    n_base = len(best_h)
    n_candidates = sorted(set([
        int(round(n_base * f))
        for f in [0.85, 0.9, 0.95, 1.0, 1.05, 1.1, 1.15, 1.25, 1.4, 0.75, 1.5]
        if 50 <= int(round(n_base * f)) <= 250
    ]))

    for dn in n_candidates:
        if time.time() > deadline - 40:
            break
        if dn == n_base:
            continue
        h_d = interpolate_to(best_h, dn)
        s_d = exact_score(h_d)
        if s_d > best_score + 0.02:
            continue
        # Quick optimization
        sub_deadline = min(deadline - 30, time.time() + 120)
        h_d, s_d = optimize_h(h_d, dn, rng, sub_deadline, max_iters=30000)
        if s_d < best_score:
            best_h = h_d.copy()
            best_score = s_d

    # Phase 3: Focused gradient polish on best
    if time.time() < deadline - 15:
        n_best = len(best_h)
        h_pol = best_h.copy()
        for _ in range(2000):
            if time.time() > deadline - 8:
                break
            g = smooth_grad(h_pol, 0.003)
            d = -g + np.mean(g)
            nd = np.linalg.norm(d)
            if nd < 1e-14:
                break
            d /= nd
            t_lo, t_hi = feas_interval(h_pol, d)
            if t_hi <= t_lo + 1e-14:
                # Try pair direction
                i = int(np.argmin(g))
                j = int(np.argmax(g))
                if i == j:
                    break
                d = np.zeros(n_best)
                d[i] = 1.0
                d[j] = -1.0
                nd = np.sqrt(2.0)
                d /= nd
                t_lo, t_hi = feas_interval(h_pol, d)
                if t_hi <= t_lo + 1e-14:
                    break
            R0, R1, R2 = line_coeffs(h_pol, d)
            t_star, _ = minmax_line(R0, R1, R2, t_lo, t_hi)
            if abs(t_star) < 1e-14:
                break
            h_new = h_pol + t_star * d
            h_new = project_feasible(h_new, n_best)
            s_new = exact_score(h_new)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new
            elif s_new < h_pol and True:
                h_pol = h_new
            else:
                # Try smaller step
                t_sm = t_star * 0.3
                if abs(t_sm) < 1e-14:
                    break
                h_new = h_pol + t_sm * d
                h_new = project_feasible(h_new, n_best)
                s_new2 = exact_score(h_new)
                if s_new2 < best_score - 1e-14:
                    best_h = h_new
                    best_score = s_new2
                    h_pol = h_new
                else:
                    break

    # Phase 4: Final pair refinement
    if time.time() < deadline - 8:
        n_best = len(best_h)
        h_final = best_h.copy()
        s_final = best_score
        improved = True
        max_rounds = 5000
        for rnd in range(max_rounds):
            if time.time() > deadline - 5:
                break
            if not improved:
                break
            improved = False
            g = smooth_grad(h_final, 0.01)
            for _ in range(3):
                i = int(np.argmin(g))
                j = int(np.argmax(g))
                if i == j:
                    break
                d = np.zeros(n_best)
                d[i] = 1.0
                d[j] = -1.0
                h_new, moved = do_line_search(h_final, d, n_best)
                if moved:
                    s_new = exact_score(h_new)
                    if s_new < s_final - 1e-14:
                        h_final = h_new
                        s_final = s_new
                        improved = True
                        if s_final < best_score:
                            best_h = h_final.copy()
                            best_score = s_final
                        g = smooth_grad(h_final, 0.01)
                        break
            # Also try random pairs
            if time.time() > deadline - 5:
                break
            for _ in range(5):
                i = int(rng.integers(0, n_best))
                j = int(rng.integers(0, n_best))
                if i == j:
                    continue
                d = np.zeros(n_best)
                d[i] = 1.0
                d[j] = -1.0
                h_new, moved = do_line_search(h_final, d, n_best)
                if moved:
                    s_new = exact_score(h_new)
                    if s_new < s_final - 1e-14:
                        h_final = h_new
                        s_final = s_new
                        improved = True
                        if s_final < best_score:
                            best_h = h_final.copy()
                            best_score = s_final

    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    final_score = exact_score(best_h)
    best_score = min(best_score, final_score)

    return best_h, float(best_score), int(len(best_h))
