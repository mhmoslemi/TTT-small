import time
import math
import numpy as np
from scipy.optimize import linprog

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    # Bisection on threshold to achieve sum = target
    lo, hi = -1.0, 1.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    # Fix residual
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            room = out.copy()
            idx = np.argsort(-room)[:max(1, int(abs(d)))]
            for i in idx:
                adj = min(d, out[i])
                out[i] -= adj
                d -= adj
                if abs(d) < 1e-12:
                    break
        else:
            room = (1.0 - out).copy()
            idx = np.argsort(-room)[:max(1, int(-d))]
            for i in idx:
                adj = min(-d, 1.0 - out[i])
                out[i] += adj
                d += adj
                if abs(d) < 1e-12:
                    break
    return out

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx), c

def compute_gradients(h):
    """Compute gradients of all correlation values w.r.t. h."""
    n = len(h)
    center = n - 1
    # We'll store gradients as a list of arrays, one per lag
    # grad[p] has shape (n,) where p ranges from -(n-1) to n-1
    # But we'll index them by offset from center: 0 to 2n-2
    grads = np.zeros((2 * n - 1, n))
    
    # p = 0: c[center] = sum h[i]*(1-h[i]), grad = 1 - 2h
    grads[center] = 1.0 - 2.0 * h
    
    # p > 0: c[center+p] = sum_{i=0}^{n-1-p} h[i+p]*(1-h[i])
    for p in range(1, n):
        # Contribution from h[i+p] term: for j in [p, n-1], grad[j] += (1-h[j-p])
        # Contribution from (1-h[i]) term: for j in [0, n-1-p], grad[j] -= h[j+p]
        j_range1 = np.arange(p, n)
        grads[center + p, j_range1] += (1.0 - h[j_range1 - p])
        j_range2 = np.arange(0, n - p)
        grads[center + p, j_range2] -= h[j_range2 + p]
        
        # c[center-p] = sum_{i=0}^{n-1-p} h[i]*(1-h[i+p])
        # grad[j] += (1-h[j+p]) for j in [0, n-1-p]
        # grad[j] -= h[j-p] for j in [p, n-1]
        j_range1 = np.arange(0, n - p)
        grads[center - p, j_range1] += (1.0 - h[j_range1 + p])
        j_range2 = np.arange(p, n)
        grads[center - p, j_range2] -= h[j_range2 - p]
    
    return grads

def solve_lp_direction(h, c, grads, top_k=None):
    """Solve LP to find descent direction for min max c_p."""
    n = len(h)
    num_lags = 2 * n - 1
    
    if top_k is not None:
        # Only use top_k lags
        order = np.argsort(c)[::-1]
        active = order[:top_k]
    else:
        active = np.arange(num_lags)
    
    m = len(active)
    
    # Variables: d[0..n-1], t
    # Minimize t
    # c_p + g_p . d <= t for p in active
    # -h[i] <= d[i] <= 1-h[i]
    # sum(d) = 0
    
    # c_min = -c_vec.A + G d <= 0 (rearranged: G d + c_vec.A <= t, i.e., G d - 1*t + c_vec.A <= 0)
    # Actually: c_p + g_p . d - t <= 0
    
    # A_ub rows: [g_p, -1] for each active p (m rows)
    # Plus box: d[i] <= 1-h[i] -> [e_i, 0] dot [d,t] <= 1-h[i]
    #         -d[i] <= h[i] -> [-e_i, 0] dot [d,t] <= h[i]
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0  # minimize t
    
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    
    # Lag constraints: g_p . d - t <= -c_p
    for idx, p in enumerate(active):
        A_ub[idx, :n] = grads[p]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -c[p]
    
    # Box constraints
    offset = m
    for i in range(n):
        A_ub[offset + i, i] = 1.0
        b_ub[offset + i] = 1.0 - h[i]
        A_ub[offset + n + i, i] = -1.0
        b_ub[offset + n + i] = h[i]
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True})
    
    if not res.success:
        return None
    
    d = res.x[:n]
    # Check if this gives improvement
    t_val = res.x[n]
    if t_val >= c.max() - 1e-10:
        return None  # No improvement possible
    
    return d

def lse_polish(h0, rng, time_limit, sigma, target, deadline=None):
    """LSE-smoothed gradient descent as fallback/refinement."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    
    cur_lse = None
    best_h = h.copy()
    best_c5 = exact_score(h)[0]
    eta = 0.05
    t0 = time.monotonic()
    stall = 0
    
    def lse_val(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full')
        m = c.max()
        return m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
    
    def lse_grad(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full')
        m = c.max()
        e = np.exp(sigma * (c - m))
        w = e / e.sum()
        grad = np.zeros(n)
        grad += w[center] * (1.0 - 2.0 * hh)
        for p in range(1, n):
            wp = w[center + p]
            wm = w[center - p]
            if wp > 1e-18:
                grad[p:] += wp * (1.0 - hh[:-p])
                grad[:-p] -= wp * (hh[p:])
            if wm > 1e-18:
                grad[:-p] += wm * (1.0 - hh[p:])
                grad[p:] -= wm * (hh[:-p])
        return grad * dx
    
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = lse_grad(h)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        improved = False
        e = eta
        for _ in range(25):
            cand = project_exact(h - e * g, target)
            cv = lse_val(cand)
            if cur_lse is None or cv < cur_lse - 1e-14:
                h = cand
                cur_lse = cv
                c5 = exact_score(h)[0]
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            e *= 0.5
        if not improved:
            h = project_exact(h + rng.normal(0.0, 0.02, n), target)
            cur_lse = lse_val(h)
            c5 = exact_score(h)[0]
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
            stall += 1
            if stall >= 8:
                eta *= 0.5
                stall = 0
                if eta < 1e-7:
                    break
    
    c5 = exact_score(h)[0]
    if c5 < best_c5 - 1e-13:
        best_c5 = c5
        best_h = h.copy()

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

def make_binary_seed(n, rng):

    h = (rng.random(n) >= 0.5).astype(float)
    h = project_exact(h, n / 2.0)

def cosine_seed(n, rng):
    """Create a cosine-based seed."""
    x = np.arange(n) / n
    phase = rng.random() * 2 * math.pi
    freq = rng.integers(2, 8)
    h = 0.5 + 0.5 * np.cos(2 * math.pi * freq * x + phase)
    h = project_exact(h, n / 2.0)
    return h

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']

            a = np.asarray(val, dtype=float).ravel()

        except Exception:

def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            idx = np.argsort(-out)
            for i in idx:
                adj = min(d, out[i])
                out[i] -= adj
                d -= adj
                if abs(d) < 1e-12:
                    break
        else:
            idx = np.argsort(-(1.0 - out))
            for i in idx:
                adj = min(-d, 1.0 - out[i])
                out[i] += adj
                d += adj
                if abs(d) < 1e-12:
                    break
    return out


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx), c


def compute_gradients(h):
    n = len(h)
    center = n - 1
    grads = np.zeros((2 * n - 1, n))
    grads[center] = 1.0 - 2.0 * h
    for p in range(1, n):
        j1 = np.arange(p, n)
        grads[center + p, j1] += (1.0 - h[j1 - p])
        j2 = np.arange(0, n - p)
        grads[center + p, j2] -= h[j2 + p]
        j3 = np.arange(0, n - p)
        grads[center - p, j3] += (1.0 - h[j3 + p])
        j4 = np.arange(p, n)
        grads[center - p, j4] -= h[j4 - p]
    return grads


def solve_lp_direction(h, c, grads, top_k):
    n = len(h)
    num_lags = 2 * n - 1
    if top_k >= num_lags:
        active = np.arange(num_lags)
    else:
        order = np.argsort(c)[::-1]
        active = order[:top_k]
    m = len(active)
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    A_ub = np.zeros((m + 2 * n, n + 1))
    b_ub = np.zeros(m + 2 * n)
    for idx, p in enumerate(active):
        A_ub[idx, :n] = grads[p]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -c[p]
    offset = m
    for i in range(n):
        A_ub[offset + i, i] = 1.0
        b_ub[offset + i] = 1.0 - h[i]
        A_ub[offset + n + i, i] = -1.0
        b_ub[offset + n + i] = h[i]
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  method='highs', options={'presolve': True})
    if not res.success:
        return None
    d = res.x[:n]
    t_val = res.x[n]
    if t_val >= c.max() - 1e-10:
        return None
    return d

def lse_polish(h0, rng, time_limit, sigma, target, deadline):

    center = n - 1

    stall = 0
    cur_lse = None

    def lse_val(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full')
        m = c.max()
        return m + math.log(np.exp(sigma * (c - m)).sum()) / sigma

    def lse_grad(hh):
        c = np.correlate(hh, 1.0 - hh, mode='full')
        m = c.max()
        e = np.exp(sigma * (c - m))
        w = e / e.sum()
        grad = np.zeros(n)
        grad += w[center] * (1.0 - 2.0 * hh)
        for p in range(1, n):

            wm = w[center - p]
            if wp > 1e-18:
                grad[p:] += wp * (1.0 - hh[:-p])
                grad[:-p] -= wp * (hh[p:])
            if wm > 1e-18:
                grad[:-p] += wm * (1.0 - hh[p:])
                grad[p:] -= wm * (hh[:-p])
        return grad * dx

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def make_binary_seed(n, rng):
    h = (rng.random(n) >= 0.5).astype(float)
    return project_exact(h, n / 2.0)

    phase = rng.random() * 2 * math.pi
    freq = rng.integers(2, 7)
    h = 0.5 + 0.5 * np.cos(2 * math.pi * freq * x + phase)
    return project_exact(h, n / 2.0)

            move = h[ii] - 0.0
            h[ii] = 1.0

            h[j] = min(1.0, h[j] + 0)

                        best_c5 = cc2
                        best_h = ch2.copy()

    # Phase 4: Transform-based starts
    if time.monotonic() < deadline:
        transforms = [best_h[::-1].copy(), 0.5 * (best_h + best_h[::-1]), np.roll(best_h, 1).copy(), np.roll(best_h, -1).copy()]
        per_tf = 0.02 * effective
        for tf in transforms:
            if time.monotonic() > deadline:
                break
            alloc = min(per_tf, deadline - time.monotonic())
            if alloc < 0.5:
                break
            tf = project_exact(resample(tf, n0), n0 / 2.0)
            ch, cc = lp_optimize(tf, rng, alloc, deadline, 50)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 5: Random and structured restarts
    if time.monotonic() < deadline:
        remaining = deadline - time.monotonic()
        n_restarts = min(5, int(remaining / max(1.0, 0.04 * effective)))
        per_restart = min(0.03 * effective, remaining / max(1, n_restarts))
        for _ in range(n_restarts):
            if time.monotonic() > deadline:
                break
            if per_restart < 1.0:
                break
            r = rng.random()
            if r < 0.4:
                hs = make_binary_seed(n0, rng)
            elif r < 0.7:
                hs = cosine_seed(n0, rng)
            else:
                hs = project_exact(resample(best_h, n0) + rng.normal(0, 0.12, n0), n0 / 2.0)
            ch, cc = lp_optimize(hs, rng, per_restart, deadline, 50)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 6: Final intensive LP with tighter top_k
    if time.monotonic() < deadline:
        alloc = min(0.08 * effective, deadline - time.monotonic())
        if alloc > 1.0:
            ch, cc = lp_optimize(best_h, rng, alloc, deadline, 30)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 7: Final LSE polish at high sigma
    if time.monotonic() < deadline - 0.5:
        alloc = min(0.04 * effective, deadline - time.monotonic())
        if alloc > 0.5:
            ch, cc = lse_polish(best_h, rng, alloc, 80.0, len(best_h) / 2.0, deadline)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch.copy()

    # Final cleanup
    best_h = project_exact(best_h, len(best_h) / 2.0)
    best_c5 = exact_score(best_h)[0]
    return best_h, float(best_c5), int(len(best_h))
