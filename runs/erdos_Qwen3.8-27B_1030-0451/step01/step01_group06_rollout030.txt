from scipy.optimize import minimize

def evaluate_exact(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix residual
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            for idx in np.argsort(h)[::-1]:
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
                if res <= 0:
                    break
        else:
            for idx in np.argsort(h):
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
                if res >= 0:
                    break
    return h

def compute_smooth_objective_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1  # number of lags
    
    # Compute all Q values
    Q = np.correlate(h, 1.0 - h, mode='full') * dx
    
    # Log-sum-exp for stability
    M = np.max(Q)
    exp_terms = np.exp((Q - M) / tau)
    log_sum = np.log(np.sum(exp_terms))
    F = M + tau * log_sum
    
    # Weights
    w = exp_terms / np.sum(exp_terms)
    
    # Compute gradient: sum_r w_r * dQ_r/dh
    grad = np.zeros(n)
    
    # For each lag k, add w_r * gradient of Q_r
    for r in range(m):
        if w[r] < 1e-15:
            continue
        k = r - (n - 1)
        if k == 0:
            grad += w[r] * dx * (1.0 - 2.0 * h)
        elif k > 0:
            # dQ_k/dh[j] = dx*(1-h[j-k]) for j in [k, n-1] - dx*h[j+k] for j in [0, n-1-k]
            g = np.zeros(n)
            g[k:] += dx * (1.0 - h[:n-k])  # (1-h[j-k]) for j>=k
            g[:n-k] += -dx * h[k:]  # -h[j+k] for j<=n-1-k
            grad += w[r] * g
        else:
            m_lag = -k  # positive
            # dQ_{-m}/dh[j] = dx*(1-h[j+m]) for j in [0, n-1-m] - dx*h[j-m] for j in [m, n-1]
            g = np.zeros(n)
            g[:n-m_lag] += dx * (1.0 - h[m_lag:])  # (1-h[j+m]) for j <= n-1-m
            g[m_lag:] += -dx * h[:n-m_lag]  # -h[j-m] for j >= m
            grad += w[r] * g
    
    return F, grad

def evaluate_exact(h):

    if n == 0:

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(x, target_sum):

    n = len(x)
    if n == 0:

        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(20):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            for idx in np.argsort(h)[::-1]:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            for idx in np.argsort(h):
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def smooth_obj_grad(h, tau):
    """Compute log-sum-exp objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    
    Q = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(Q)
    shifted = (Q - M) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -500, 500)
    exp_terms = np.exp(shifted)
    sum_exp = np.sum(exp_terms)
    if sum_exp <= 0:
        sum_exp = 1e-300
    F = M + tau * np.log(sum_exp)
    w = exp_terms / sum_exp
    
    # Gradient
    grad = np.zeros(n)
    for r in range(m):
        wr = w[r]
        if wr < 1e-12:
            continue
        k = r - (n - 1)
        if k == 0:
            grad += wr * dx * (1.0 - 2.0 * h)
        elif k > 0:
            # dQ_k/dh[j]: for j in [k, n-1]: dx*(1-h[j-k]); for j in [0, n-1-k]: -dx*h[j+k]
            g = np.zeros(n)
            g[k:n] += dx * (1.0 - h[0:n-k])
            g[0:n-k] += -dx * h[k:n]
            grad += wr * g
        else:
            ml = -k

            g = np.zeros(n)
            g[0:n-ml] += dx * (1.0 - h[ml:n])
            g[ml:n] += -dx * h[0:n-ml]
            grad += wr * g
    
    return F, grad

def smooth_solve(h_init, tau, max_iter=200, ftol=1e-10, deadline=None):
    """Solve the smooth optimization problem for given tau."""
    n = len(h_init)
    target = n / 2.0
    
    # Define objective
    def obj(x):
        x = np.clip(x, 0.0, 1.0)  # SLSQP should respect bounds, but be safe
        val, _ = smooth_obj_grad(x, tau)
        return val
    
    def grad(x):
        x = np.clip(x, 0.0, 1.0)
        _, g = smooth_obj_grad(x, tau)
        return g
    
    def constraint_eq(x):
        return x.sum() - target
    
    def constraint_eq_grad(x):
        return np.ones(n)
    
    constraints = [{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_eq_grad}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(obj, h_init, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': ftol})
        if res.success or res.x is not None:
            h_new = project_feasible(res.x, target)
            return h_new
    except Exception:
        pass
    return None

    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    # Bisection for lambda
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix residual
    for _ in range(20):
        res = h.sum() - target_sum
        if abs(res) < 1e-15:
            break
        if res > 0:

                    break

                h[idx] -= d
                res -= d

                    break

                res += d
