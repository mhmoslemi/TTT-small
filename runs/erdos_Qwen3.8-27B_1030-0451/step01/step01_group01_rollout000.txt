import numpy as np
import time

def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            mask = y < 1.0
            m = mask.sum()
            if m > 0:
                y[mask] = np.minimum(y[mask] + r / m, 1.0)
        else:
            mask = y > 0.0
            m = mask.sum()
            if m > 0:
                y[mask] = np.maximum(y[mask] + r / m, 0.0)
    return np.clip(y, 0.0, 1.0)

def correlate_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def eval_c5(h):
    return float(np.max(correlate_vals(h)))

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        i1 = np.arange(0, n - k)
        g[i1] += (1.0 - h[i1 + k])
        i2 = np.arange(k, n)
        g[i2] -= h[i2 - k]
    else:
        l = -k
        i1 = np.arange(l, n)
        g[i1] += (1.0 - h[i1 - l])
        i2 = np.arange(0, n - l)
        g[i2] -= h[i2 + l]
    return g

def subgrad_momentum(h0, deadline, max_iters=5000, mom=0.7, step0=0.01):
    """Subgradient with momentum and adaptive step size."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    v = np.zeros(n)  # momentum velocity
    step = step0
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        cv = correlate_vals(h)
        c_cur = cv.max()
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 500:
                break
        
        # Compute weighted gradient (top correlations)
        idx = np.argsort(cv)[-min(n, 50):]
        w = np.exp(20 * (cv[idx] - cv.max()))
        w /= w.sum()
        
        g = np.zeros(n)
        for ii, k_idx in enumerate(idx):
            lag = k_idx - (n - 1)
            g += w[ii] * grad_lag(h, lag)
        
        # Momentum
        v = mom * v - step * g
        h_new = project_sum_box(h + v, n / 2.0)
        
        # Check improvement
        c_new = eval_c5(h_new)
        if c_new < c_cur:
            h = h_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.9
            v *= 0.5
            if step < 1e-7:
                break
    
    return best_h, best_c

def pairwise_cd(h0, deadline, max_rounds=20):
    """Pairwise coordinate descent with 1D line search."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if abs(h[i] - h[j]) < 1e-4:
                    continue
                # Determine direction: move from higher to lower
                if h[i] > h[j]:
                    t_min, t_max = 0.0, min(1.0 - h[i], h[j])
                else:
                    t_min, t_max = -min(1.0 - h[j], h[i]), 0.0
                
                if t_max - t_min < 1e-5:
                    continue
                
                # Evaluate at a few points
                t_vals = np.linspace(t_min, t_max, 8)
                best_t = 0.0
                best_tval = best_c
                
                for t in t_vals:
                    if t == 0.0:
                        continue
                    h_tr = h.copy()
                    if h[i] > h[j]:
                        h_tr[i] += t
                        h_tr[j] -= t
                    else:
                        h_tr[i] -= t
                        h_tr[j] += t
                    h_tr = np.clip(h_tr, 0, 1)
                    h_tr = project_sum_box(h_tr, n / 2.0)
                    c_tr = eval_c5(h_tr)
                    if c_tr < best_tval - 1e-12:
                        best_tval = c_tr
                        best_t = t
                
                if best_t != 0.0:
                    if h[i] > h[j]:
                        h[i] += best_t
                        h[j] -= best_t
                    else:
                        h[i] -= best_t
                        h[j] += best_t
                    h = np.clip(h, 0, 1)
                    h = project_sum_box(h, n / 2.0)
                    if best_tval < best_c:
                        best_c = best_tval
                        best_h = h.copy()
                    improved = True
        
        if not improved:
            break
    
    return best_h, best_c

import numpy as np
import time
from scipy.optimize import linprog

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            mask = y < 1.0
            m = mask.sum()
            if m > 0:
                y[mask] = np.minimum(y[mask] + r / m, 1.0)
        else:
            mask = y > 0.0
            m = mask.sum()
            if m > 0:
                y[mask] = np.maximum(y[mask] + r / m, 0.0)
    return np.clip(y, 0.0, 1.0)

def correlate_vals(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def eval_c5(h):
    return float(np.max(correlate_vals(h)))

def grad_lag(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        i1 = np.arange(0, n - k)
        g[i1] += (1.0 - h[i1 + k])
        i2 = np.arange(k, n)
        g[i2] -= h[i2 - k]
    else:
        l = -k
        i1 = np.arange(l, n)
        g[i1] += (1.0 - h[i1 - l])
        i2 = np.arange(0, n - l)
        g[i2] -= h[i2 + l]
    return g

def method_momentum_subgrad(h0, deadline, max_iters=8000):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    v = np.zeros(n)
    step = 0.008
    mom = 0.6
    no_improve = 0
    plateaus = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cv = correlate_vals(h)
        c_cur = cv.max()
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
            no_improve = 0
            plateaus = 0
        else:
            no_improve += 1
            if no_improve > 30:
                plateaus += 1
            if plateaus > 60:
                break
        
        top_n = min(n, 40)
        idx = np.argsort(cv)[-top_n:]
        w = np.exp(25.0 * (cv[idx] - cv.max()))
        sw = w.sum()
        if sw < 1e-30:
            w = np.ones(top_n) / top_n
        else:
            w /= sw
        
        g = np.zeros(n)
        for ii, k_idx in enumerate(idx):
            lag = k_idx - (n - 1)
            g += w[ii] * grad_lag(h, lag)
        
        v = mom * v - step * g
        v_norm = np.linalg.norm(v)
        if v_norm < 1e-14:
            break
        h_new = project_sum_box(h + v, n / 2.0)
        c_new = eval_c5(h_new)
        
        if c_new < c_cur - 1e-14:
            h = h_new
            if it % 100 == 0:
                step = min(step * 1.03, 0.04)
                mom = min(mom + 0.002, 0.85)
        else:
            step *= 0.92
            v *= 0.3
            h = project_sum_box(h - 0.5 * v, n / 2.0)
            if step < 1e-8:
                break
    return best_h, best_c

def method_pairwise_cd(h0, deadline, max_rounds=30):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                diff = h[i] - h[j]
                if abs(diff) < 1e-4:
                    continue
                if diff > 0:
                    t_max = min(1.0 - h[i], h[j])
                else:
                    t_max = min(h[i], 1.0 - h[j])
                if t_max < 1e-5:
                    continue
                best_t = 0.0
                best_val = best_c
                for frac in [0.25, 0.5, 0.75]:
                    t = t_max * frac
                    h_tr = h.copy()
                    if diff > 0:
                        h_tr[i] += t
                        h_tr[j] -= t
                    else:
                        h_tr[i] -= t
                        h_tr[j] += t
                    h_tr = np.clip(h_tr, 0.0, 1.0)
                    s = h_tr.sum()
                    if abs(s - n / 2.0) > 1e-10:
                        h_tr = project_sum_box(h_tr, n / 2.0)
                    c_tr = eval_c5(h_tr)
                    if c_tr < best_val - 1e-12:
                        best_val = c_tr
                        best_t = t
                if best_t > 0:
                    if diff > 0:
                        h[i] += best_t
                        h[j] -= best_t
                    else:
                        h[i] -= best_t
                        h[j] += best_t
                    h = np.clip(h, 0.0, 1.0)
                    if best_val < best_c:
                        best_c = best_val
                        best_h = h.copy()
                    improved = True
        if not improved:
            break
    return best_h, best_c

def method_lp_tight(h0, deadline, max_iters=400):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = eval_c5(best_h)
    radius = 0.08
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        cv = correlate_vals(h)
        vmax = cv.max()
        gap = max(0.005 * vmax, 0.003)
        active = np.where(cv >= vmax - gap)[0]
        if len(active) > 60:
            order = np.argsort(cv[active])
            active = active[order[-60:]]
        if len(active) == 0:
            active = np.argsort(cv)[-15:]
        
        A = np.zeros((len(active), n))
        b_lin = np.zeros(len(active))
        for i, k in enumerate(active):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)
            b_lin[i] = A[i] @ h - cv[k]
        
        nv = n + 1
        c_obj = np.zeros(nv)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [2.0]])
        A_eq = np.zeros((1, nv))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = eval_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                radius = min(radius * 1.05, 0.2)
            else:
                no_improve += 1
                if no_improve >= 6:
                    radius = max(radius * 0.6, 0.005)
                    no_improve = 0
                    if radius <= 0.005:
                        break
        else:
            no_improve += 1
            if no_improve >= 10:
                radius = max(radius * 0.5, 0.003)
                no_improve = 0
                if radius <= 0.003:
                    break
    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_ext = np.concatenate([h0, h0[:1]])
    x_ext = np.concatenate([x_old, [x_old[-1] + 2.0 / n_old]])
    h_new = np.interp(x_new, x_ext, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def make_block_pattern(n, seed=0):
    rng = np.random.RandomState(seed)
    h = np.zeros(n)
    remaining = n // 2
    pos = list(range(n))
    rng.shuffle(pos)
    for p in pos:
        if remaining <= 0:
            break
        h[p] = 1.0
        remaining -= 1
    return project_sum_box(h, n / 2.0)

def make_sinusoidal(n, n_waves=1, phase=0.0):
    x = np.linspace(0, 2 * n_waves, n, endpoint=False)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * x / (2 * n_waves) + phase)
    h = np.clip(h, 0.0, 1.0)
    return project_sum_box(h, n / 2.0)

def make_random_smooth(n, seed=0, smooth=3):
    rng = np.random.RandomState(seed)
    h = rng.random(n)
    kernel = np.ones(smooth) / smooth
    h = np.convolve(h, kernel, mode='same')
    h = np.clip(h, 0.0, 1.0)
    return project_sum_box(h, n / 2.0)
