def binary_flip_search(h, max_iters, time_limit):
    n = len(h)
    dx = 2.0 / n
    h = (h >= 0.5).astype(float)  # Threshold to binary
    # Ensure exactly n/2 ones
    # (adjust if needed)
    best_val = true_c5(h)
    for it in range(max_iters):
        if time.time() > time_limit:
            break
        corr = corr_array(h)
        cur_max = np.max(corr)
        if cur_max < best_val:
            best_val = cur_max
        
        # Try each flip
        best_gain = 0
        best_pos = -1
        for p in range(n):
            # Compute new max if we flip position p
            delta = 1.0 - 2.0 * h[p]  # +1 if 0→1, -1 if 1→0
            # Update correlations
            new_corr = corr.copy()
            # Zero lag: no change for binary
            # Positive lags
            for k in range(1, n):
                # Lag +k: Q_{base+k} = dx * sum h_{i+k}(1-h_i)
                # Position p as h_{i+k}: i = p-k, valid if 0<=p-k<=n-k-1 i.e., k<=p<=n-1
                if k <= p <= n-1:
                    new_corr[base+k] += delta * dx * (1 - h[p-k])
                # Position p as (1-h_i): i = p, valid if 0<=p<=n-k-1 i.e., p<=n-k-1
                if p <= n-k-1:
                    new_corr[base+k] -= delta * dx * h[p+k]
                # Lag -k: Q_{base-k} = dx * sum h_i(1-h_{i+k})
                # Position p as h_i: i=p, valid if 0<=p<=n-k-1
                if p <= n-k-1:
                    new_corr[base-k] += delta * dx * (1 - h[p+k])
                # Position p as (1-h_{i+k}): i+k=p, i=p-k, valid if k<=p
                if k <= p:
                    new_corr[base-k] -= delta * dx * h[p-k]
            
            new_max = np.max(new_corr)
            gain = cur_max - new_max
            if gain > best_gain:
                best_gain = gain
                best_pos = p
        
        if best_pos >= 0:
            h[best_pos] = 1.0 - h[best_pos]
        else:
            break  # No improving flip found
    return h, best_val

import numpy as np
import time
from scipy.optimize import minimize

def true_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-13:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def ensure_binary_count(h, n_ones):
    h = h.copy()
    cur = int(np.sum(h))
    if cur > n_ones:
        idx = np.where(h == 1)[0]
        for i in idx[n_ones:]:
            h[i] = 0.0
    elif cur < n_ones:
        idx = np.where(h == 0)[0]
        for i in idx[:n_ones - cur]:
            h[i] = 1.0
    return h

def compute_sensitivity(h):
    n = len(h)
    dx = 2.0 / n
    base = n - 1
    S = np.zeros((2 * n - 1, n))
    S[base, :] = dx * (1.0 - 2.0 * h)
    for k in range(1, n):
        S[base + k, k:] = dx * (1.0 - h[:n - k])
        S[base + k, :n - k] += -dx * h[k:]
        S[base - k, :n - k] = dx * (1.0 - h[k:])
        S[base - k, k:] += -dx * h[:n - k]
    return S

def binary_local_search(h, deadline, rng=None, use_sa=False, sa_temp=0.01):
    n = len(h)
    if rng is None:
        rng = np.random.default_rng(0)
    h = ensure_binary_count((h >= 0.5).astype(float), n // 2)
    best_val = true_c5(h)
    best_h = h.copy()
    stagnant = 0
    
    while time.time() < deadline:
        Q = corr_array(h)
        cur_max = float(np.max(Q))
        if cur_max < best_val:
            best_val = cur_max
            best_h = h.copy()
            stagnant = 0
        else:
            stagnant += 1
        
        S = compute_sensitivity(h)
        zeros = np.where(h == 0)[0]
        ones = np.where(h == 1)[0]
        
        if len(zeros) == 0 or len(ones) == 0:
            break
        
        best_gain = 1e-12
        best_pz = -1
        best_po = -1
        
        for pz in zeros:
            spz = S[:, pz]
            for po in ones:
                new_max = np.max(Q + spz - S[:, po])
                gain = cur_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_pz = pz
                    best_po = po
        
        if best_pz >= 0:
            h[best_pz] = 1.0
            h[best_po] = 0.0
        else:

                for _ in range(30):
                    pz = zeros[rng.integers(len(zeros))]
                    po = ones[rng.integers(len(ones))]
                    new_max = np.max(Q + S[:, pz] - S[:, po])
                    delta = new_max - cur_max
                    if delta < 0 or rng.random() < np.exp(-delta / sa_temp):
                        h[pz] = 1.0
                        h[po] = 0.0
                        break
                sa_temp *= 0.95
            else:
                break
        
        if stagnant > 500:
            break
    
    return best_h, best_val

def logsumexp_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    m = np.max(corr)
    exp_shifted = np.exp((corr - m) / tau)
    sum_exp = exp_shifted.sum()
    val = m + tau * np.log(sum_exp)
    w = exp_shifted / sum_exp
    grad = np.zeros(n)
    base = n - 1
    for idx in range(2 * n - 1):
        wr = w[idx]
        if wr < 1e-14:
            continue
        k = idx - base
        if k == 0:
            grad += wr * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wr * dx * (1.0 - h[:n - k])
            grad[:n - k] -= wr * dx * h[k:]
        else:
            t = -k
            grad[:n - t] += wr * dx * (1.0 - h[t:])
            grad[t:] -= wr * dx * h[:n - t]
    return val, grad

def slsqp_optimize(h, target_sum, tau, deadline, max_iter=200):
    n = len(h)
    h = project_feasible(h, target_sum)
    
    def obj(x):
        x = np.clip(x, 0.0, 1.0)
        v, g = logsumexp_obj_grad(x, tau)
        penalty = 100.0 * (np.sum(x) - target_sum) ** 2
        g = g + 200.0 * (np.sum(x) - target_sum)
        return v + penalty, g
    
    res = minimize(obj, h, jac=True, method='L-BFGS-B',
                   bounds=[(0, 1)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol': 1e-8})
    h_out = project_feasible(res.x, target_sum)
    return h_out, true_c5(h_out)

    idx = rng.permutation(n)
    h[idx[:n // 2]] = 1.0
    return h
