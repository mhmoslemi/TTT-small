def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):

def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-13:
        if d < 0:
            slack = 1.0 - out
            room = slack.sum()
            if room > 1e-15:
                out += (-d / room) * slack
        else:
            room = out.sum()
            if room > 1e-15:
                out -= (d / room) * out
    out = np.clip(out, 0.0, 1.0)
    d2 = out.sum() - target
    if abs(d2) > 1e-14 and len(out) > 0:
        if d2 > 0:
            idx = int(np.argmax(out))
        else:
            idx = int(np.argmin(out))
        out = out.copy()
        out[idx] = float(np.clip(out[idx] - d2, 0.0, 1.0))
    return out


def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx), c


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def lse_grad(h, sigma):
    """Compute gradient of LSE approximation to max correlation."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    e = np.exp(sigma * (c - cmax))
    w = e / e.sum()
    n_active = np.sum(w > 1e-18)
    if n_active == 0:
        return np.zeros(n)
    grad = np.zeros(n)
    center = n - 1
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-18:
            continue
        k = m - center
        if k == 0:
            grad += wm * (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += wm * (1.0 - h[k:])
            grad[k:] -= wm * h[:-k]
        else:
            k2 = -k
            grad[k2:] += wm * (1.0 - h[:n-k2])
            grad[:n-k2] -= wm * h[k2:]
    grad *= dx
    return grad


def exact_subgrad(h, c, active_frac):
    """Compute exact subgradient of max correlation using active set."""
    n = len(h)
    center = n - 1
    cmax = c.max()
    span = cmax - c.min()
    if span < 1e-15:
        return np.zeros(n)
    threshold = cmax - active_frac * span
    active = c >= threshold
    n_active = max(1, int(active.sum()))
    grad = np.zeros(n)
    idxs = np.where(active)[0]
    for m in idxs:
        k = m - center
        if k == 0:
            grad += (1.0 - 2.0 * h)
        elif k > 0:
            grad[:n-k] += (1.0 - h[k:])
            grad[k:] -= h[:-k]
        else:
            k2 = -k
            grad[k2:] += (1.0 - h[:n-k2])
            grad[:n-k2] -= h[k2:]
    return grad / n_active
