def _f_grad(f, m, tau):
    """Gradient of log-sum-exp objective in f-space."""
    n = 2 * m
    dx = 2.0 / n
    h = make_anti_periodic(f)
    c = np.correlate(h, 1.0 - h, 'full') * dx
    cmax = c.max()
    s = np.clip((c - cmax) / tau, -500, 500)
    e = np.exp(s)
    alpha = e / e.sum()
    
    # Gradient with respect to h
    gh = np.zeros(n)
    for k in range(len(c)):
        if alpha[k] < 1e-10:
            continue
        a = alpha[k]
        r = (n - 1) - k  # lag
        if r >= 0:
            hi = n - 1 - r
            if hi >= 0:
                gh[0:hi+1] += a * (1.0 - h[r:hi+1+r])
                gh[r:hi+1+r] -= a * h[0:hi+1]
        else:

            hi = n - 1 - aa

                gh[0:hi+1] -= a * h[aa:hi+1+aa]

    gf = gh[:m] - gh[m:]
    return gf

- gf[j] = gh[j] - gh[j+m] for j = 0, ..., m-1

- d(h[j])/d(f[j]) = 1, so contribution is +gh[j]
- d(h[j+m])/d(f[j]) = -1, so contribution is -gh[j+m]

def project_feasible(y, target=None):

    if abs(residual) > 1e-14:

def active_lag_polish_f(f0, m, deadline, rng):

    n = 2 * m
    dx = 2.0 / n
    f_curr = f0.copy()
    best_f = f0.copy()

        cmean = c.mean()
        tol = max(1e-4, 0.05 * (cmax - cmean))
        active = np.where(c >= cmax - tol)[0]

            active = active[np.argsort(c[active])[::-1][:10]]
        elif len(active) < 3:
