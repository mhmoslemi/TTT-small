import numpy as np
import time

def project_sum_box(h, target):
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) / 2.0
        trial = np.clip(h + mid, 0.0, 1.0)
        if trial.sum() > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - result.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - result
            ts = slack.sum()
            if ts > 1e-12:
                result += r * slack / ts
        else:
            tm = result.sum()
            if tm > 1e-12:
                result += r * result / tm
    result = np.clip(result, 0.0, 1.0)
    r2 = target - result.sum()
    if abs(r2) > 1e-13:
        if r2 > 0:
            idx = np.where(result < 1.0 - 1e-12)[0]
        else:
            idx = np.where(result > 1e-12)[0]
        if len(idx) > 0:
            result[idx] += r2 / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def get_corrs(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag_idx):
    n = len(h)
    g = np.zeros(n)
    k = lag_idx - (n - 1)
    if k >= 0:
        if k < n:
            g[k:] += (1.0 - h[:-k]) if k > 0 else 1.0 - h
        if k > 0:
            g[:n-k] -= h[k:]
        else:
            g -= h
    else:
        L = -k
        if L < n:
            g[:n-L] += (1.0 - h[L:])
        if L > 0:
            g[L:] -= h[:-L]
        else:
            g -= h
    return g

def smooth_obj_grad(h, beta):
    corrs = get_corrs(h)
    m = np.max(corrs)
    expo = np.clip(beta * (corrs - m), None, 500)
    u = np.exp(expo)
    Z = u.sum()
    f = m + np.log(Z) / beta
    w = u / Z
    n = len(h)
    g = np.zeros(n)
    for k_idx in range(len(w)):
        if w[k_idx] > 1e-10:
            g += w[k_idx] * grad_lag(h, k_idx)
    return f, g

def pair_exchange_search(h, deadline, rng, max_rounds=50):

    best_c = evaluate_c5(best_h)
    h = h.copy()
    eps_base = 0.15
    eps_decay = 0.998
    eps = eps_base
    no_improve = 0
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        if no_improve > 2000:
            eps = min(eps * 2.0, eps_base)
            no_improve = 0
        for _ in range(200):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            for direction in (1, -1):
                di = direction * eps
                dj = -di
                if h[i] + di < 0 or h[i] + di > 1:
                    continue
                if h[j] + dj < 0 or h[j] + dj > 1:

                h[j] += dj

                    best_h = h.copy()
                    no_improve = 0
                    break
                else:
                    h[i] -= di
                    h[j] -= dj
            else:
                no_improve += 1

def binary_local_search(h, deadline, rng):

def active_lag_descent(h, deadline, rng):
