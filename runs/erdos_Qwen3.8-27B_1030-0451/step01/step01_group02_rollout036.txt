The objective f(h) = max_k C_k(h) where C_k(h) = dx * ∑ h[i]*(1-h[i+d_k]).

h_{new} = project(h - α * g)

1. Compute y = h - α*g
2. Find t such that ∑ clip(y+t, 0, 1) = n/2 (bisection on t)

def make_feasible(y, target_sum):
    """Project y to [0,1]^n with sum = target_sum using bisection on offset."""
    n = len(y)
    if n == 0:
        return y
    target_sum = float(target_sum)
    y = np.asarray(y, dtype=float)
    
    # Bisection on offset t: find t such that sum(clip(y+t, 0, 1)) = target_sum
    lo = -float(y.max()) - 1.0
    hi = 1.0 - float(y.min()) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    x = np.clip(y + 0.5*(lo+hi), 0.0, 1.0)
    
    # Fix small residual
    residual = target_sum - float(x.sum())
    if abs(residual) > 1e-10:
        if residual > 0:
            # Add to positions with most room
            room = 1.0 - x
            order = np.argsort(room)[::-1]
            rem = residual
            for i in order:
                add = min(room[i], rem)
                x[i] += add
                rem -= add
                if rem <= 1e-14:
                    break
        else:
            # Remove from positions with most value
            order = np.argsort(x)[::-1]
            rem = -residual
            for i in order:
                remv = min(x[i], rem)
                x[i] -= remv
                rem -= remv
                if rem <= 1e-14:

def subgradient_step(h, C, alpha, target_sum, n_active=5):
    """One subgradient step on the continuous relaxation."""
    n = len(h)
    dx = 2.0 / float(n)
    
    # Select active lags (top n_active)
    top_idx = np.argsort(C)[::-1][:n_active]
    weights = np.ones(len(top_idx)) / len(top_idx)
    
    # Compute gradient
    g = np.zeros(n)
    ctr = n - 1
    for ii, k in enumerate(top_idx):
        wk = float(weights[ii]) * dx
        d = int(k) - ctr
        if d == 0:
            g += wk * (1.0 - 2.0 * h)
        elif d > 0:
            g[d:] += wk * (1.0 - h[:-d])
            g[:-d] += -wk * h[d:]
        else:
            dd = -d
            g[:n-dd] += wk * (1.0 - h[dd:])
            g[dd:] += -wk * h[:-dd]
    
    # Gradient step with projection
    y = h - alpha * g
    x = make_feasible(y, target_sum)
    return x

                best_c = maxC
                best_h = h.copy()
            alpha = min(alpha * 1.05, 0.15)
            no_improve = 0
        else:
            alpha *= 0.6
            no_improve += 1
            if no_improve > 30:
                # Try random perturbation
                h_new = make_feasible(h + rng.normal(0, 0.01, size=n), target)
                c_new = exact_C5(h_new)
                if c_new < maxC - 1e-12:
                    h = h_new
                    C = corr_scaled(h)
                    maxC = c_new
                    if maxC < best_c:
                        best_c = maxC
                        best_h = h.copy()
                    no_improve = 0
                    alpha = 0.02
                else:
                    break
    
    return best_h, best_c

def gen_structured_binary(n, rng):
    """Generate structured binary initial sequences."""
    x = np.arange(n, dtype=float)
    best = None
    best_c = float('inf')
    for f in range(1, 9):
        for f2 in [0, f // 2 + 1, f + 1]:
            for ph in [0.0, np.pi / 2, np.pi]:
                y = 0.5 + 0.4 * np.cos(2 * np.pi * f * x / n + ph)
                if f2 > 0:
                    y += 0.2 * np.cos(2 * np.pi * f2 * x / n)
                b = binarize(y, n)
                c = exact_C5(b)
                if c < best_c:
                    best_c = c
                    best = b
    # Also try random
    for _ in range(5):
        r = 0.5 + rng.normal(0, 0.3, size=n)
        b = binarize(r, n)
        c = exact_C5(b)
        if c < best_c:
            best_c = c
            best = b
    return best, best_c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    total = float(budget_s)
    hard = t0 + total - 3.0
    
    rng = np.random.default_rng(int(seed) % (1 << 32))
    
    # Get initial
    val = globals().get("initial_h_values", None)
    if val is not None:
        arr = np.asarray(val, dtype=float).ravel()
        n0 = int(arr.size)
    else:
        n0 = 720
        arr = make_feasible(0.5 + 0.3 * np.cos(2 * np.pi * 3 * np.arange(n0) / n0), 0.5 * n0)
    
    best_h = None
    best_c = float('inf')
    best_n = n0
    
    # Phase 1: Binary search from binarized initial
    h_bin = binarize(arr, n0)
    c_start = exact_C5(h_bin)
    stage1_end = t0 + 0.25 * total
    h_opt, c_opt = binary_local_search(h_bin.copy(), min(hard, stage1_end), rng, k_active=4)
    if c_opt < best_c:
        best_c = c_opt
        best_h = h_opt.copy()
        best_n = n0
    
    # Phase 2: Binary search from structured starts
    stage2_end = t0 + 0.45 * total
    h_struct, c_struct = gen_structured_binary(n0, rng)
    if c_struct < best_c:
        h_opt, c_opt = binary_local_search(h_struct.copy(), min(hard, stage2_end), rng, k_active=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    else:
        # Continue refining best
        h_opt, c_opt = binary_local_search(best_h.copy(), min(hard, stage2_end), rng, k_active=3)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    # Phase 3: More binary search with different k_active
    stage3_end = t0 + 0.60 * total
    for ka in [2, 6]:
        if time.time() > stage3_end:
            break
        h_opt, c_opt = binary_local_search(best_h.copy(), min(hard, time.time() + 0.1 * total), rng, k_active=ka)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    # Phase 4: Continuous refinement
    if time.time() < hard - 30 and best_h is not None:
        h_cont = best_h.copy()
        stage4_end = min(hard - 15, time.time() + 0.2 * total)
        h_opt, c_opt = continuous_subgradient(h_cont, stage4_end, rng, k_active=4)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = len(h_opt)
    
    # Phase 5: Binarize continuous result and do more binary search
    if time.time() < hard - 20 and best_h is not None:
        h_bin2 = binarize(best_h, len(best_h))
        c_bin2 = exact_C5(h_bin2)
        cur_best = min(best_c, c_bin2)
        if c_bin2 < best_c:
            best_c = c_bin2
            best_h = h_bin2.copy()
        stage5_end = min(hard - 5, time.time() + 0.15 * total)
        h_opt, c_opt = binary_local_search(h_bin2.copy(), stage5_end, rng, k_active=4)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
    
    # Phase 6: Try higher resolution
    if time.time() < hard - 25 and best_h is not None:
        n_cur = len(best_h)
        for factor in [2, 3]:
            if time.time() > hard - 10:
                break
            N_new = n_cur * factor
            if N_new > 2500 or N_new % 2 != 0:
                continue
            x_old = (np.arange(n_cur) + 0.5) * (2.0 / n_cur)
            x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
            h_up = np.interp(x_new, x_old, best_h)
            h_upb = binarize(h_up, N_new)
            stage_end = min(hard - 5, time.time() + (hard - time.time()) * 0.4)
            h_opt, c_opt = binary_local_search(h_upb, stage_end, rng, k_active=4)
            if c_opt < best_c:
                best_c = c_opt
                best_h = h_opt.copy()
                best_n = N_new
    
    # Final polish: a few more targeted swaps
    if time.time() < hard - 3 and best_h is not None and best_h.sum() == 0.5 * len(best_h):
        h_opt, c_opt = binary_local_search(best_h.copy(), hard - 1, rng, k_active=5, max_rounds=500)
        if c_opt < best_c:
            best_c = c_opt
            best_h = h_opt.copy()
            best_n = len(h_opt)
    
    # Ensure feasibility
    if best_h is None:
        best_h = make_feasible(np.full(n0, 0.5), 0.5 * n0)
        best_c = exact_C5(best_h)
        best_n = n0
    
    n_final = len(best_h)
    best_h = make_feasible(best_h, 0.5 * float(n_final))
    best_c = exact_C5(best_h)
    
    return best_h, float(best_c), int(n_final)
