import numpy as np

from scipy.optimize import minimize

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_all_corrs(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_obj(h, tau):
    c = compute_all_corrs(h)
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    obj = m + tau * np.log(total)
    return obj, w / total

def smooth_grad(h, weights):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = weights[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return g

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h

def fista_smooth(h0, tau, n_iters, target_sum, lr=None):
    """FISTA with smooth max approximation."""
    N = len(h0)
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    
    # Estimate Lipschitz constant
    # The Hessian of the smooth max is bounded by 4/(tau) * (sum of squared weights) * (something)
    # Practically, use a safe upper bound
    if lr is None:
        lr = tau / (4.0 * (2*N-1))  # conservative
    
    for k in range(n_iters):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        obj, w = smooth_obj(y, tau)
        g = smooth_grad(y, w)
        
        h_new = project_feasible(y - lr * g, target_sum)
        h_prev = h
        h = h_new
        t = t_new
    
    return h

def subgradient_fista(h0, target_sum, n_iters, top_k=5, lr_scale=0.1):
    """FISTA-style with exact subgradient (top-k averaged)."""
    N = len(h0)
    dx = 2.0 / N
    h = h0.copy()
    h_prev = h0.copy()
    t = 1.0
    lr0 = lr_scale / np.sqrt(N)
    
    for k in range(n_iters):
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = h + (t - 1.0) / t_new * (h - h_prev)
        y = np.clip(y, 0.0, 1.0)
        
        c = compute_all_corrs(y)
        # Pick top k lags
        top_indices = np.argsort(c)[-top_k:]
        
        # Average subgradient over top k
        g = np.zeros(N)
        for r in top_indices:
            d = r - (N - 1)
            if d >= 0:
                n = N - d
                if n > 0:
                    g[d:] += (1.0 - y[:n])
                    g[:n] -= y[d:]
            else:
                dd = -d
                n = N - dd
                if n > 0:
                    g[:n] += (1.0 - y[dd:])
                    g[dd:] -= y[:n]
        g *= dx / len(top_indices)
        
        lr = lr0 / (np.sqrt(k + 1.0))
        h_new = project_feasible(y - lr * g, target_sum)
        h_prev = h
        h = h_new
        t = t_new
    
    return h
