import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def get_corr_dx(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx


def lse_weights(c, tau):
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    w = w / total
    return m + tau * np.log(total), w


def lse_obj_grad(h, tau):
    c = get_corr_dx(h)
    obj, w = lse_weights(c, tau)
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return obj, g


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h


def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def _slsqp_fun(h, tau):
    obj, _ = lse_obj_grad(h, tau)
    return obj


def _slsqp_jac(h, tau):
    _, g = lse_obj_grad(h, tau)
    return g


def _sum_eq(h, target):
    return np.sum(h) - target


def _sum_eq_jac(h, target):
    return np.ones_like(h)


def run_slsqp(h0, N, tau, maxiter):
    target_sum = N / 2.0
    bounds = [(0.0, 1.0)] * N
    constraint = [{'type': 'eq', 'fun': _sum_eq, 'jac': _sum_eq_jac,
                   'args': (target_sum,)}]
    try:
        result = minimize(
            _slsqp_fun, h0.copy(),
            jac=_slsqp_jac,
            args=(tau,),
            method='SLSQP',
            bounds=bounds,
            constraints=constraint,
            options={'maxiter': maxiter, 'ftol': 1e-14}
        )
        return project_feasible(result.x, target_sum)
    except Exception:
        return h0.copy()


def _lbfgs_fun(x, tau, mu, target_sum):
    obj, _ = lse_obj_grad(x, tau)
    penalty = mu * (np.sum(x) - target_sum) ** 2
    return obj + penalty


def _lbfgs_jac(x, tau, mu, target_sum):
    _, g = lse_obj_grad(x, tau)
    g = g + 2.0 * mu * (np.sum(x) - target_sum)
    return g


def run_lbfgs(h0, N, tau, mu, maxiter):
    target_sum = N / 2.0
    bounds = [(0.0, 1.0)] * N
    try:
        result = minimize(
            _lbfgs_fun, h0.copy(),
            jac=_lbfgs_jac,
            args=(tau, mu, target_sum),
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': maxiter, 'gtol': 1e-12, 'maxfun': maxiter * 5}
        )
        return project_feasible(result.x, target_sum)
    except Exception:
        return h0.copy()


def run_momentum(h0, N, tau, lr, momentum, maxiter):
    target_sum = N / 2.0
    x = h0.copy()
    v = np.zeros_like(x)
    best_x = x.copy()
    best_obj = lse_obj_grad(x, tau)[0]

    for k in range(maxiter):
        obj, g = lse_obj_grad(x, tau)
        v = momentum * v - lr * g
        x_new = project_feasible(x + v, target_sum)
        obj_new = lse_obj_grad(x_new, tau)[0]
        if obj_new < best_obj:
            best_obj = obj_new
            best_x = x_new.copy()
        x = x_new
    return best_x


def make_periodic_h(N, period):
    """Create a periodic square-wave pattern with the given period (in number of points)."""
    h = np.zeros(N)
    half = period // 2
    for i in range(N):
        if (i % period) < half:
            h[i] = 1.0
    return project_feasible(h, N / 2.0)
