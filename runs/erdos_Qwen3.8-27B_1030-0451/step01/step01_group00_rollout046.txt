from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_constraint(p, n):
    target = n / 2.0
    p = np.nan_to_num(p, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def build_corr_gradient_matrix(h):
    """Build the 161 x 81 matrix where row k is the gradient of c_k at h.
    Returns matrix G where G[k, i] = d c_k / d h_i (without the dx factor)."""
    n = len(h)
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for k in range(n_lags):
        lag = k - (n - 1)
        if lag >= 0:
            # grad[i] = (1-h[i-lag]) for i >= lag, minus h[i+lag] for i < n-lag
            G[k, lag:] += (1.0 - h[:n - lag])
            G[k, :n - lag] -= h[lag:]
        else:
            L = -lag
            G[k, :n - L] += (1.0 - h[L:])
            G[k, L:] -= h[:n - L]
    return G

def lp_direction(h, corr, n_lags, radius):
    """Find LP direction to minimize linearized max correlation.
    Returns delta (n-vector) or None if LP fails."""
    n = len(h)
    dx = 2.0 / n
    c = corr * dx  # normalized correlations
    
    G = build_corr_gradient_matrix(h)  # G[k, i] = d corr_k / d h_i (raw, without dx)
    # The derivative of c_k = dx * corr_k is dx * G[k, :]
    # Linearization: c_k(h) + dx * (G[k] @ delta) <= s
    # => dx * G[k] @ delta - s <= -c_k(h)
    
    # LP variables: [delta (n), s (1)]
    # Objective: minimize s => c_obj = [0...0, 1]
    # Constraints (inequality, A_ub @ x <= b_ub):
    #   dx * G[k] @ delta - s <= -c[k]  for each k
    #   i.e., [dx * G[k], -1] @ [delta, s] <= -c[k]
    # Equality: sum(delta) = 0 => [ones(1,n), 0] @ [delta, s] = 0
    # Bounds: -radius <= delta_i <= radius, s unbounded (or s >= 0)
    
    m = n_lags
    A_ub = np.zeros((m, n + 1))
    A_ub[:, :n] = dx * G
    A_ub[:, n] = -1.0
    b_ub = -c
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(-radius, radius)] * n + [(None, None)]
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    try:

            c_new = compute_c5(h)
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h.copy()
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    
    best_h = None
    best_c5 = float('inf')
    
    # Initialize from parent or fallback
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
        h0 = project_constraint(h0, n_init)
        c5_0 = compute_c5(h0)
        best_h = h0.copy()
        best_c5 = c5_0
    except (NameError, UnboundLocalError, Exception):
        n_init = 81
        h0 = make_start(n_init, seed)
        best_h = h0.copy()
        best_c5 = compute_c5(h0)
    
    # Main optimization from parent
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 100
    if remaining > 60:
        h_opt, c5_opt = optimize_phased(best_h.copy(), remaining, seed)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
    
    # Perturbation restart
    elapsed = time.time() - t_start
    if budget_s - elapsed > 80:
        rng = np.random.RandomState(seed + 300)
        h_pert = best_h + rng.randn(len(best_h)) * 0.05
        h_pert = project_constraint(h_pert, len(best_h))
        alloc = min(budget_s - elapsed - 50, 80)
        if alloc > 20:
            h_opt, c5_opt = optimize_phased(h_pert, alloc, seed + 400)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
    
    # Try other resolutions
    for n_try in [97, 121, 151]:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - 40
        if remaining < 30:
            break
        h_start = make_start(n_try, seed + n_try * 11)
        alloc = min(remaining // 2, 70)
        if alloc > 15:
            h_opt, c5_opt = optimize_phased(h_start, alloc, seed + n_try * 3)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
    
    # Final polish on best
    elapsed = time.time() - t_start
    if best_h is not None and budget_s - elapsed > 20:
        alloc = min(budget_s - elapsed - 3, 50)
        if alloc > 10:
            h_opt, c5_opt = optimize_phased(best_h.copy(), alloc, seed + 999)
            if c5_opt < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5_opt
    
    # Final verification
    if best_h is not None:
        best_h = project_constraint(best_h, len(best_h))
        best_c5 = compute_c5(best_h)
    else:
        n_fb = 81
        best_h = np.full(n_fb, 0.5)
        best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
