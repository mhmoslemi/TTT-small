    """Compute gradient of squared-excess objective focused on active lags."""
    n = len(h)
    dx = 2.0 / n

            grad[p:] -= w * h[:n-p]
        else:
            # p < 0, let q = -p
            q = -p

            if n - q > 0:
                grad[q:] += w * (1.0 - h[:n-q])
            grad[:n-q] -= w * h[q:]
    grad *= dx
    return grad, c, cmax

import time
import math
import numpy as np
from scipy.optimize import minimize_scalar


def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def exact_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def project_exact(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d < 0:
            room = 1.0 - out
            tr = room.sum()
            if tr > 1e-15:
                out = out + (-d / tr) * room
        else:
            room = out
            tr = room.sum()
            if tr > 1e-15:
                out = out - (d / tr) * room
    out = np.clip(out, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-14 and len(out) > 0:
        if d > 0:
            idx = int(np.argmax(out))
        else:
            idx = int(np.argmax(1.0 - out))
        out = out.copy()
        out[idx] = float(np.clip(out[idx] - d, 0.0, 1.0))
    return out


def repair(h, target):
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    return project_exact(h, target)


def active_grad(h, tau_frac):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cmax = c.max()
    tau = cmax - tau_frac
    v = np.maximum(0.0, c - tau)
    grad = np.zeros(n)
    center = n - 1
    for k in range(2 * n - 1):
        w = v[k]
        if w < 1e-15:
            continue
        p = k - center
        if p == 0:
            grad += w * (1.0 - 2.0 * h)
        elif p > 0:
            if n - p > 0:
                grad[:n - p] += w * (1.0 - h[p:])
            grad[p:] -= w * h[:n - p]
        else:
            q = -p
            if n - q > 0:
                grad[q:] += w * (1.0 - h[:n - q])
            grad[:n - q] -= w * h[q:]
    grad *= dx
    return grad, c, cmax


def pair_line_val(t, h_base, i, j):
    hc = h_base.copy()
    hc[i] = h_base[i] - t
    hc[j] = h_base[j] + t
    n = len(hc)
    dx = 2.0 / n
    c = np.correlate(hc, 1.0 - hc, mode='full')
    return float(c.max() * dx)


def pair_search(h, i, j, n_grid=16):
    n = len(h)
    lo = max(h[i] - 1.0, -h[j]) + 1e-10
    hi = min(h[i], 1.0 - h[j]) - 1e-10
    if hi <= lo:
        return 0.0, exact_obj(h)
    best_t = 0.0
    best_val = exact_obj(h)
    ts = np.linspace(lo, hi, n_grid + 1)
    for t in ts:
        hc = h.copy()
        hc[i] = h[i] - t
        hc[j] = h[j] + t
        val = exact_obj(hc)
        if val < best_val:
            best_val = val
            best_t = float(t)
    span = (hi - lo) / n_grid
    res_lo = max(lo, best_t - span)
    res_hi = min(hi, best_t + span)
    if res_hi - res_lo > 1e-12:
        try:
            res = minimize_scalar(pair_line_val, bounds=(res_lo, res_hi),
                                  args=(h, i, j), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 40})
            if res.fun < best_val - 1e-14:
                best_t = float(res.x)
                best_val = float(res.fun)
        except Exception:
            pass
    return best_t, best_val


def apply_transfer(h, i, j, t):
    hc = h.copy()
    hc[i] = float(np.clip(h[i] - t, 0.0, 1.0))
    hc[j] = float(np.clip(h[j] + t, 0.0, 1.0))
    return hc


def select_pairs(h, grad, c_arr, cmax, deadline_delta, rng, n_pairs=25):
    n = len(h)
    center = n - 1
    active_lags = [k for k in range(2 * n - 1) if c_arr[k] >= cmax - deadline_delta]
    pairs = []
    seen = set()
    # Lag-aware: for each active lag, pick pairs at that offset
    for k in active_lags[:8]:
        p = k - center
        if p == 0:
            continue
        if p > 0 and p < n:
            for i in range(0, n - p, max(1, (n - p) // 5)):
                pair = (i, i + p)
                if pair not in seen and (i, i + p) not in seen and (i + p, i) not in seen:
                    seen.add(pair)
                    seen.add((i + p, i))
                    pairs.append(pair)
        elif p < 0 and -p < n:
            q = -p
            for i in range(q, n, max(1, (n - q) // 5)):
                pair = (i, i - q)
                if pair not in seen and (i - q, i) not in seen:
                    seen.add(pair)
                    seen.add((i - q, i))
                    pairs.append(pair)
    # Gradient-based
    abs_g = np.abs(grad)
    top_idx = np.argsort(abs_g)[-min(n_pairs, n):]
    for a in range(len(top_idx)):
        for b in range(a + 1, min(a + 6, len(top_idx))):
            i, j = int(top_idx[a]), int(top_idx[b])
            if (i, j) not in seen and (j, i) not in seen:
                seen.add((i, j))
                seen.add((j, i))
                pairs.append((i, j))
    # Random
    while len(pairs) < n_pairs:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j and (i, j) not in seen:
            seen.add((i, j))
            seen.add((j, i))
            pairs.append((i, j))
    return pairs[:n_pairs]


def pair_polish(h, budget_s, deadline, rng, tau_frac=0.02, max_pairs=25, n_grid=16):
    n = len(h)
    target = n / 2.0
    cur_h = h.copy()
    cur_c = exact_obj(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    t0 = time.monotonic()
    stall = 0
    tau = tau_frac

    while time.monotonic() < deadline and (time.monotonic() - t0) < budget_s:
        try:
            grad, c_arr, cmax = active_grad(cur_h, tau)
        except Exception:
            break
        pairs = select_pairs(cur_h, grad, c_arr, cmax, max(0.005, tau * 0.5), rng, max_pairs)
        improved = False
        for (i, j) in pairs:
            if time.monotonic() > deadline:
                break
            t_opt, val = pair_search(cur_h, i, j, n_grid=n_grid)
            if val < cur_c - 1e-13:
                cur_h = apply_transfer(cur_h, i, j, t_opt)
                # Fix sum drift
                d = cur_h.sum() - target
                if abs(d) > 1e-10:
                    if d > 0:
                        idx = int(np.argmax(cur_h))
                        cur_h[idx] = float(np.clip(cur_h[idx] - d, 0.0, 1.0))
                    else:
                        idx = int(np.argmax(1.0 - cur_h))
                        cur_h[idx] = float(np.clip(cur_h[idx] + (-d), 0.0, 1.0))
                cur_c = exact_obj(cur_h)
                improved = True
                if cur_c < best_c - 1e-14:
                    best_c = cur_c
                    best_h = cur_h.copy()
        if not improved:
            stall += 1
            if stall <= 8:
                cur_h = cur_h + rng.normal(0.0, 0.008, n)
                cur_h = repair(cur_h, target)
                cur_c = exact_obj(cur_h)
            elif stall <= 16:
                cur_h = cur_h + rng.normal(0.0, 0.03, n)
                cur_h = repair(cur_h, target)
                cur_c = exact_obj(cur_h)
                tau = max(0.003, tau * 0.7)
            else:
                # Larger escape
                k = int(rng.integers(n))
                cur_h[k] = rng.random()
                cur_h = repair(cur_h, target)
                cur_c = exact_obj(cur_h)
                stall = 0
                tau = max(0.003, tau * 0.6)
            if cur_c < best_c - 1e-14:
                best_c = cur_c
                best_h = cur_h.copy()
        if stall >= 40:
            break

    final_c = exact_obj(cur_h)
    if final_c < best_c - 1e-14:
        best_c = final_c
        best_h = cur_h.copy()
    return best_h, best_c


def pair_anneal(h, budget_s, deadline, rng):
    n = len(h)
    target = n / 2.0
    cur_h = h.copy()
    cur_c = exact_obj(cur_h)
    best_h = cur_h.copy()
    best_c = cur_c
    T = 3e-5
    Tend = 1e-9
    t0 = time.monotonic()
    total = max(budget_s, 1e-9)

    while time.monotonic() < deadline and (time.monotonic() - t0) < budget_s:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        # Random transfer amount
        lo = max(cur_h[i] - 1.0, -cur_h[j]) + 1e-12
        hi = min(cur_h[i], 1.0 - cur_h[j]) - 1e-12
        if hi <= lo:
            continue
        t = lo + rng.random() * (hi - lo)
        # With some probability, also try swap-like
        if rng.random() < 0.3:
            t = 0.5 * (hi - lo) * (2.0 * rng.random() - 1.0)
            t = max(lo, min(hi, t))
        hc = apply_transfer(cur_h, i, j, t)
        d = hc.sum() - target
        if abs(d) > 1e-10:
            if d > 0:
                idx = int(np.argmax(hc))
                hc[idx] = float(np.clip(hc[idx] - d, 0.0, 1.0))
            else:
                idx = int(np.argmax(1.0 - hc))
                hc[idx] = float(np.clip(hc[idx] - d, 0.0, 1.0))
        nc = exact_obj(hc)
        delta = nc - cur_c
        if delta < -1e-14:
            cur_h = hc
            cur_c = nc
            if nc < best_c - 1e-14:
                best_c = nc
                best_h = hc.copy()
        else:
            frac = min(1.0, (time.monotonic() - t0) / total)
            T = Tend + (3e-5 - Tend) * (1.0 - frac) * (1.0 - frac)
            if rng.random() < math.exp(-delta / max(T, 1e-15)):
                cur_h = hc
                cur_c = nc
    return best_h, best_c


def lse_polish(h0, rng, time_limit, sigma, target, deadline):
    n = len(h0)
    dx = 2.0 / n

    def lse_val(hc):
        c = np.correlate(hc, 1.0 - hc, mode='full')
        m = c.max()
        val = m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
        return dx * val

    def lse_grad(hc):
        c = np.correlate(hc, 1.0 - hc, mode='full')
        m = c.max()
        e = np.exp(sigma * (c - m))
        w = e / e.sum()
        grad = np.zeros(n)
        center = n - 1
        grad += w[center] * (1.0 - 2.0 * hc)
        for p in range(1, n):
            wp = w[center + p]
            wm = w[center - p]
            if wp > 1e-18:
                grad[p:] += wp * (1.0 - hc[:-p])
                grad[:-p] -= wp * hc[p:]
            if wm > 1e-18:
                grad[:-p] += wm * (1.0 - hc[p:])
                grad[p:] -= wm * hc[:-p]
        grad *= dx
        return grad

    h = h0.copy()
    cur_lse = lse_val(h)
    best_h = h.copy()
    best_c5 = exact_obj(h)
    eta = 0.04
    t0 = time.monotonic()
    stall = 0

    while time.monotonic() < deadline and (time.monotonic() - t0) < time_limit:
        g = lse_grad(h)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        improved = False
        e = eta
        for _ in range(25):
            cand = project_exact(h - e * g, target)
            cv = lse_val(cand)
            if cv < cur_lse - 1e-14:
                h = cand
                cur_lse = cv
                c5 = exact_obj(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            e *= 0.5
        if not improved:
            h = project_exact(h + rng.normal(0.0, 0.008, n), target)
            cur_lse = lse_val(h)
            c5 = exact_obj(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
            stall += 1
            if stall >= 5:
                eta *= 0.5
                stall = 0
                if eta < 1e-8:
                    break
    c5 = exact_obj(h)
    if c5 < best_c5 - 1e-14:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def make_transforms(h):
    out = []
    out.append(h[::-1].copy())
    out.append(0.5 * (h + h[::-1]))
    out.append(np.roll(h, 1).copy())
    out.append(np.roll(h, -1).copy())
    out.append(np.roll(h, 2).copy())
    out.append(np.roll(h, -3).copy())
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(10.0, float(budget_s) - 12.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is not None:
        n0 = len(parent)
        h0 = repair(parent, n0 / 2.0)
    else:
        n0 = 131
        h0 = (rng.random(n0) >= 0.5).astype(float)
        h0 = repair(h0, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = exact_obj(best_h)

    def consider(cand):
        nonlocal best_h, best_c5
        try:
            cc = exact_obj(cand)
            if np.all(np.isfinite(cand)) and 0.0 <= cc < 1.0 and cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = cand.copy()
        except Exception:
            pass

    # Phase 1: LSE polish for baseline
    remaining = deadline - time.monotonic()
    if remaining > 15:
        for sigma in [60.0, 25.0, 10.0]:
            if time.monotonic() > deadline - 10:
                break
            a = min(0.08 * effective, deadline - time.monotonic() - 5)
            if a < 2:
                break
            ch, cc = lse_polish(best_h, rng, a, sigma, len(best_h) / 2.0, deadline - 3)
            consider(ch)

    # Phase 2: Pair-transfer polishing with decreasing tau
    remaining = deadline - time.monotonic()
    if remaining > 30:
        tau_schedule = [0.04, 0.02, 0.01, 0.005, 0.003]
        for tau in tau_schedule:
            if time.monotonic() > deadline - 15:
                break
            a = min(0.14 * effective, deadline - time.monotonic() - 10)
            if a < 4:
                break
            ch, cc = pair_polish(best_h, a, deadline - 5, rng,
                                 tau_frac=tau, max_pairs=30, n_grid=14)
            consider(ch)

    # Phase 3: Pair annealing for diversification
    remaining = deadline - time.monotonic()
    if remaining > 25:
        a = min(0.18 * effective, deadline - time.monotonic() - 10)
        if a > 5:
            ch, cc = pair_anneal(best_h, a, deadline - 5, rng)
            consider(ch)
            # Follow up with pair polish
            if time.monotonic() < deadline - 5:
                a2 = min(0.06 * effective, deadline - time.monotonic() - 5)
                if a2 > 2:
                    ch, cc = pair_polish(best_h, a2, deadline - 3, rng,
                                         tau_frac=0.008, max_pairs=25, n_grid=14)
                    consider(ch)

    # Phase 4: Transforms
    remaining = deadline - time.monotonic()
    if remaining > 30:
        n = len(best_h)
        tf_list = make_transforms(best_h)
        per = min(0.035 * effective, max(2.0, (deadline - time.monotonic() - 10) / max(1, len(tf_list))))
        for cand in tf_list:
            if time.monotonic() > deadline - 10:
                break
            if per < 1.5:
                break
            cand_rs = repair(resample(cand, n), n / 2.0)
            cc = exact_obj(cand_rs)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = cand_rs.copy()
            a = min(per, deadline - time.monotonic() - 5)
            if a < 1:
                break
            ch, cc2 = pair_polish(cand_rs, a, deadline - 3, rng,
                                  tau_frac=0.015, max_pairs=20, n_grid=12)
            consider(ch)

    # Phase 5: Multi-resolution
    remaining = deadline - time.monotonic()
    if remaining > 25:
        n = len(best_h)
        for dn in [-4, 4, -8, 8, -12]:
            nn = n + dn
            if nn < 80 or nn > 200 or time.monotonic() > deadline - 15:
                continue
            a = min(0.035 * effective, deadline - time.monotonic() - 10)
            if a < 3:
                break
            rs = repair(resample(best_h, nn), nn / 2.0)
            ch, cc = pair_polish(rs, a, deadline - 5, rng,
                                 tau_frac=0.01, max_pairs=20, n_grid=12)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 6: Final combined polish
    remaining = deadline - time.monotonic()
    if remaining > 8:
        a = min(0.04 * effective, remaining - 3)
        if a > 2:
            ch, cc = lse_polish(best_h, rng, a * 0.4, 8.0, len(best_h) / 2.0, deadline - 3)
            consider(ch)
            if time.monotonic() < deadline - 3:
                a2 = min(0.03 * effective, deadline - time.monotonic() - 2)
                if a2 > 1:
                    ch, cc = pair_polish(best_h, a2, deadline - 2, rng,
                                         tau_frac=0.004, max_pairs=25, n_grid=16)
                    consider(ch)

    # Final: one more very tight pair polish
    if time.monotonic() < deadline - 1:
        a = min(3.0, deadline - time.monotonic() - 0.5)
        if a > 0.5:
            ch, cc = pair_polish(best_h, a, deadline - 0.5, rng,
                                 tau_frac=0.003, max_pairs=20, n_grid=18)
            consider(ch)

    best_h = repair(best_h, len(best_h) / 2.0)
    best_c5 = exact_obj(best_h)
    return best_h, float(best_c5), int(len(best_h))
