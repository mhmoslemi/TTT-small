def single_lag_descent(h0, time_limit, target, deadline=None):
    """Focus on reducing the single worst lag via exact gradient."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    cur = objective(h)
    best_h = h.copy()
    best_c5 = cur
    t0 = time.monotonic()
    step = 0.05
    stall = 0
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        center = n - 1
        worst_idx = int(np.argmax(c))
        k = worst_idx - center
        g = grad_lag(h, k) * (2.0 / n)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        improved = False
        alpha = step
        for _ in range(20):
            cand = project_exact(h - alpha * (g / gn), target)
            cv = objective(cand)
            if cv < cur - 1e-14:
                h = cand
                cur = cv
                if cv < best_c5 - 1e-14:
                    best_c5 = cv
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            alpha *= 0.5
        if not improved:
            stall += 1
            if stall > 15:
                break
    c5 = objective(h)
    if c5 < best_c5 - 1e-14:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def binary_push(h, target):
    n = len(h)
    b = (h >= 0.5).astype(float)
    s = b.sum()
    half = n / 2.0
    if s > half:
        excess = int(s - half)
        ones = np.where(b == 1)[0]
        order = np.argsort(h[ones])[:excess]
        b[ones[order]] = 0
    elif s < half:
        deficit = int(half - s)
        zeros = np.where(b == 0)[0]
        order = np.argsort(-h[zeros])[:deficit]
        b[zeros[order]] = 1
    return b


def informed_local_search(h0, rng, time_limit, target, deadline=None):
    """Local search that targets active-lag positions."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    cur = objective(h)
    best_h = h.copy()
    best_c5 = cur
    T = 5e-5
    Tend = 1e-9
    t0 = time.monotonic()
    total = max(time_limit, 1e-9)

    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        m = c.max()
        center = n - 1
        active = np.where(c >= m * 0.95 - 0.001)[0]
        lag = active[int(rng.integers(len(active)))]
        k = lag - center
        if abs(k) >= n:
            continue
        ak = abs(k)
        if ak == 0:
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
        elif k > 0:
            i = int(rng.integers(n - k))
            j = i + k
        else:
            j = int(rng.integers(n - ak))
            i = j + ak
        a = h.copy()
        if rng.random() < 0.6:
            d = rng.random() * 0.3
            if d <= min(h[i], 1.0 - h[j]):
                a[i] -= d
                a[j] += d
            elif d <= min(h[j], 1.0 - h[i]):
                a[j] -= d
                a[i] += d
            else:
                continue
        else:
            a[i], a[j] = a[j], a[i]
        a = repair(a, target)
        na = objective(a)
        delta = na - cur
        if delta < -1e-14:
            h = a
            cur = na
            if na < best_c5 - 1e-14:
                best_c5 = na
                best_h = a.copy()
        else:
            frac = min(1.0, (time.monotonic() - t0) / total)
            T = Tend + (5e-5 - Tend) * (1.0 - frac) * (1.0 - frac)
            if rng.random() < math.exp(-max(delta, 0.0) / max(T, 1e-12)):
                h = a
                cur = na
    c5 = objective(h)
    if c5 < best_c5 - 1e-14:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def make_structured_seed(n, rng):
    """Create a structured binary seed with approximate symmetry."""
    h = np.zeros(n)
    half = n // 2
    # Alternate blocks with some randomness
    block_size = rng.integers(3, 8)
    val = rng.random()
    for i in range(n):
        if (i // block_size) % 2 == 0:
            h[i] = 1.0 if val + rng.random() * 0.3 > 0.5 else 0.0
        else:
            h[i] = 0.0 if val + rng.random() * 0.3 > 0.5 else 1.0
    h = repair(h, n / 2.0)
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 12.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        n0 = 131
        r = rng.random(n0)
        parent = (r >= 0.5).astype(float)
    n0 = len(parent)
    h0 = repair(parent, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = objective(best_h)

    # Phase 1: Top-K LSE with decreasing sigma (focused -> broad)
    # This is the main optimization phase
    sigma_K_pairs = [(300, 5), (150, 5), (80, 8), (50, 10), (30, 12), (20, 15)]
    phase1_total = 0.30 * effective
    n_p1 = len(sigma_K_pairs)
    for si in range(n_p1):
        if time.monotonic() > deadline:
            break
        alloc = phase1_total / n_p1
        alloc = min(alloc, deadline - time.monotonic())
        if alloc < 0.5:
            break
        sig = sigma_K_pairs[si][0]
        K = sigma_K_pairs[si][1]
        ch, cc = topk_lse_optimize(best_h, rng, alloc, len(best_h) / 2.0,
                                   K, sig, True, deadline)
        if cc < best_c5 - 1e-13:
            best_c5 = cc
            best_h = ch.copy()

    # Phase 2: Binary push and re-optimize
    if time.monotonic() < deadline - 2.0:
        b = binary_push(best_h, len(best_h) / 2.0)
        c_b = objective(b)
        if c_b < best_c5 - 1e-13:
            best_c5 = c_b
            best_h = b.copy()
        # Polish from binary
        alloc = min(0.06 * effective, deadline - time.monotonic())
        if alloc > 0.5:
            ch, cc = topk_lse_optimize(b, rng, alloc, len(b) / 2.0,
                                       8, 50, True, deadline)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 3: Single-lag descent (attack the worst lag directly)
    if time.monotonic() < deadline - 2.0:
        alloc = min(0.10 * effective, deadline - time.monotonic())
        if alloc > 0.5:
            ch, cc = single_lag_descent(best_h, alloc, len(best_h) / 2.0, deadline)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 4: Informed local search targeting active lags
    if time.monotonic() < deadline - 2.0:
        alloc = min(0.18 * effective, deadline - time.monotonic())
        if alloc > 1.0:
            ch, cc = informed_local_search(best_h, rng, alloc, len(best_h) / 2.0, deadline)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 5: Final top-K polish with high sigma
    if time.monotonic() < deadline - 1.0:
        alloc = min(0.10 * effective, deadline - time.monotonic())
        if alloc > 0.5:
            ch, cc = topk_lse_optimize(best_h, rng, alloc, len(best_h) / 2.0,
                                       5, 200, False, deadline)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 6: Try nearby resolutions
    if time.monotonic() < deadline - 2.0:
        for dn in [2, -2, 3, -3, 4, -4]:
            nn = n0 + dn
            if nn < 80 or nn > 250 or time.monotonic() > deadline:
                continue
            alloc = min(0.025 * effective, deadline - time.monotonic())
            if alloc < 0.3:
                break
            rs = repair(resample(best_h, nn), nn / 2.0)
            ch, cc = topk_lse_optimize(rs, rng, alloc, nn / 2.0,
                                       8, 50, True, deadline)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = ch.copy()

    # Phase 7: One more binary push on the global best
    if time.monotonic() < deadline - 1.0:
        b = binary_push(best_h, len(best_h) / 2.0)
        c_b = objective(b)
        if c_b < best_c5 - 1e-13:
            best_c5 = c_b
            best_h = b.copy()
        alloc = min(0.04 * effective, deadline - time.monotonic())
        if alloc > 0.3:
            ch, cc = topk_lse_optimize(best_h, rng, alloc, len(best_h) / 2.0,
                                       5, 100, True, deadline)
            if cc < best_c5 - 1e-13:
                best_c5 = cc
                best_h = ch.copy()

    # Final feasibility lock
    best_h = repair(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_c5 = objective(best_h)

    return best_h, float(best_c5), int(len(best_h))
