import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def project_to_zero_sum(d):
    """Project a direction to be zero-sum."""
    return d - np.mean(d)

def feasible_lambda_range(h, d):
    """Find the range of lambda such that 0 <= h + lambda*d <= 1."""
    pos = d > 1e-15
    neg = d < -1e-15
    
    lam_min = 0.0
    lam_max = 0.0
    
    if np.any(pos):
        lam_min = max(lam_min, -np.min(h[pos] / d[pos]))
        lam_max = min(lam_max, (1.0 - h[pos]) / d[pos]).min() if not np.isinf(lam_max) else lam_max
        # Need to be more careful
        lam_min_candidates = -h[pos] / d[pos]
        lam_max_candidates = (1.0 - h[pos]) / d[pos]
        lam_min = max(lam_min, np.max(lam_min_candidates))
        lam_max = min(lam_max, np.min(lam_max_candidates))
    
    if np.any(neg):
        lam_min_candidates = (1.0 - h[neg]) / d[neg]
        lam_max_candidates = -h[neg] / d[neg]
        lam_min = max(lam_min, np.max(lam_min_candidates))
        lam_max = min(lam_max, np.min(lam_max_candidates))
    
    return lam_min, lam_max

def line_search(h, d, n_samples=50):
    """Find the best lambda along direction d."""
    lam_min, lam_max = feasible_lambda_range(h, d)
    if lam_max <= lam_min:
        return 0.0, compute_c5(h)
    
    # Sample the interval
    lams = np.linspace(lam_min, lam_max, n_samples)
    best_lambda = 0.0
    best_val = compute_c5(h)
    
    for lam in lams:
        h_new = h + lam * d
        # Numerical safety
        h_new = np.clip(h_new, 0.0, 1.0)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_lambda = lam
    
    # Local refinement around best
    span = lam_max - lam_min
    lo = max(lam_min, best_lambda - span * 0.04)

        h_new = np.clip(h + lam * d, 0.0, 1.0)
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_lambda = lam
    
    return best_lambda, best_val

    g = np.zeros(N)

def subgradient_direction(h, n_active=10, temperature=1.0):
    """Compute a subgradient direction for the max of correlations."""
    N = len(h)
    corr = compute_corr(h)
    c_max = np.max(corr)
    
    # Weight by exponential of (c - c_max) / temperature
    exponents = np.clip((corr - c_max) / temperature, -500, 0)
    weights = np.exp(exponents)
    weights /= np.sum(weights)
    
    # Only use top n_active
    threshold = np.sort(weights)[-n_active] if n_active < len(weights) else weights.min()
    weights[weights < threshold] = 0
    weights /= np.sum(weights)
    
    # Compute weighted gradient
    g = np.zeros(N)
    center = N - 1
    for r in range(2 * N - 1):
        if weights[r] < 1e-15:
            continue
        d = r - center
        g += weights[r] * compute_gradient_for_lag(h, d)
    
    # Project to zero-sum
    g = g - np.mean(g)
    
    return g, c_max

def pair_transfer_direction(h, i, j):
    """Create a direction that transfers mass from i to j."""
    d = np.zeros(len(h))
    d[i] = -1.0
    d[j] = 1.0
    return d

def smooth_direction(h, k, phase=0.0):
    """Create a smooth sinusoidal direction."""
    N = len(h)
    x = np.arange(N) / N
    d = np.sin(2 * np.pi * k * x + phase)
    d = d - np.mean(d)
    return d

def repair_feasible(h, target):
    """Project h to satisfy sum(h) = target and 0 <= h <= 1."""
    h = np.clip(h, 0.0, 1.0)
    # Binary search for the shift
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) / 2
    h = np.clip(h + t, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argsort(1.0 - h)
            for i in idx:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if residual < 1e-12:
                    break
        else:
            idx = np.argsort(h)
            for i in idx:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if abs(residual) < 1e-12:
                    break
    return h

import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    N = len(h)
    return np.correlate(h, 1.0 - h, mode='full')

def grad_lag(h, d):
    """Gradient of correlation at lag offset d (d=r-(N-1))."""
    N = len(h)
    g = np.zeros(N)
    if d == 0:
        g[:] = 1.0 - 2.0 * h
    elif d > 0:
        g[:N-d] += 1.0
        g[:N-d] -= h[d:]
        g[d:] -= h[:N-d]
    else:
        dd = -d
        g[dd:] += 1.0
        g[dd:] -= h[:N-dd]
        g[:N-dd] -= h[dd:]
    return g

def subgrad_dir(h, n_top, temp):
    N = len(h)
    corr = compute_corr(h)
    cmax = np.max(corr)
    ew = np.exp(np.clip((corr - cmax) / temp, -200, 0))
    ew /= ew.sum()
    # Keep top n_top
    if n_top < len(ew):
        thresh = np.sort(ew)[-n_top]
        ew[ew < thresh] = 0
        ew /= ew.max() if ew.max() > 0 else 1
        ew /= ew.sum()
    
    center = N - 1
    g = np.zeros(N)
    for r in range(2 * N - 1):
        if ew[r] < 1e-12:
            continue
        g += ew[r] * grad_lag(h, r - center)
    g -= np.mean(g)
    return g, cmax

def feasible_range(h, d):
    pos = d > 1e-14
    neg = d < -1e-14
    lo = 0.0
    hi = 0.0
    if np.any(pos):
        lo = max(lo, float(-np.min(h[pos] / d[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / d[pos])))
    if np.any(neg):
        lo = max(lo, float(np.min((1.0 - h[neg]) / d[neg])))
        hi = min(hi, float(-np.min(h[neg] / d[neg])))
    if hi <= lo:
        return 0.0, 0.0
    return lo, hi

def line_search(h, d, n_coarse=30, n_fine=20):
    lo, hi = feasible_range(h, d)
    if hi <= lo + 1e-15:
        return 0.0, compute_c5(h)
    
    best_lam = 0.0
    best_val = compute_c5(h)
    
    # Coarse search
    for lam in np.linspace(lo, hi, n_coarse):
        h_new = h + lam * d
        if h_new.min() < -1e-10 or h_new.max() > 1.0 + 1e-10:
            continue
        val = compute_c5(h_new)
        if val < best_val:
            best_val = val
            best_lam = lam
    
    # Fine search around best
    span = hi - lo
    lo_f = max(lo, best_lam - 0.05 * span)
    hi_f = min(hi, best_lam + 0.05 * span)
    if hi_f > lo_f:
        for lam in np.linspace(lo_f, hi_f, n_fine):
            h_new = h + lam * d
            if h_new.min() < -1e-10 or h_new.max() > 1.0 + 1e-10:
                continue
            val = compute_c5(h_new)
            if val < best_val:
                best_val = val
                best_lam = lam
    
    return best_lam, best_val

def repair(h, target):
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    # Fix residual
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for i in order:
            if res > 0:
                add = min(res, 1.0 - h[i])
                h[i] += add
                res -= add
            else:
                sub = min(-res, h[i])
                h[i] -= sub
                res += sub
            if abs(res) < 1e-12:
                break
    return h

def upscale_parent(h, N_new):
    N_old = len(h)
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return repair(h_new, N_new / 2.0)
