```python
def compute_gradient_fast(h, w):
    """
    h: array of length n
    w: array of length 2n-1 (weights for each correlation)
    Returns gradient of sum_k w[k] * c[k]
    """
    n = len(h)
    grad = np.zeros(n)
    
    # Lag 0
    w0 = w[n-1]
    grad += w0 * (1.0 - 2.0 * h)
    
    # Positive lags
    for ell in range(1, n):
        wk = w[n-1+ell]
        if abs(wk) < 1e-14:
            continue
        # (1-h[j-ell]) for j >= ell
        grad[ell:] += wk * (1.0 - h[:-ell])
        # -h[j+ell] for j <= n-1-ell
        grad[:n-ell] -= wk * h[ell:]
    
    # Negative lags
    for L in range(1, n):
        wk = w[n-1-L]
        if abs(wk) < 1e-14:
            continue
        # (1-h[j+L]) for j <= n-1-L
        grad[:n-L] += wk * (1.0 - h[L:])
        # -h[j-L] for j >= L
        grad[L:] -= wk * h[:n-L]
    
    return grad
```

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def compute_gradient_fast(h, w):
    n = len(h)
    grad = np.zeros(n)
    # Lag 0
    w0 = w[n-1]
    if abs(w0) > 1e-14:
        grad += w0 * (1.0 - 2.0 * h)
    # Positive lags
    for ell in range(1, n):
        wk = w[n-1+ell]
        if abs(wk) < 1e-14:
            continue
        grad[ell:] += wk * (1.0 - h[:-ell])
        grad[:n-ell] -= wk * h[ell:]
    # Negative lags
    for L in range(1, n):
        wk = w[n-1-L]
        if abs(wk) < 1e-14:
            continue
        grad[:n-L] += wk * (1.0 - h[L:])
        grad[L:] -= wk * h[:n-L]
    return grad

def softmax_weights(corr, tau):
    cmax = np.max(corr)
    cs = corr - cmax
    if tau < 1:
        return np.ones(len(corr)) / len(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        w = np.zeros_like(corr)
        w[np.argmax(corr)] = 1.0
    else:
        w /= ws
    return w

def pair_optimize(h, s, t, dx):
    """Try to improve by transferring mass between s and t."""
    n = len(h)
    best_c = compute_c5(h)
    best_alpha = 0.0
    
    # Direction: h[s] -= alpha, h[t] += alpha
    a_max = min(h[s], 1.0 - h[t])
    if a_max > 1e-10:
        # Try several values
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 0.99]:
            a = frac * a_max
            h[s] -= a
            h[t] += a
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c = c
                best_alpha = a
            h[s] += a
            h[t] -= a
    
    # Direction: h[t] -= alpha, h[s] += alpha
    a_max2 = min(h[t], 1.0 - h[s])
    if a_max2 > 1e-10:
        for frac in [0.1, 0.3, 0.5, 0.7, 0.9, 0.99]:
            a = frac * a_max2
            h[t] -= a
            h[s] += a
            c = compute_c5(h)
            if c < best_c - 1e-12:
                best_c = c
                best_alpha = -a  # negative means t to s
            h[t] += a
            h[s] -= a
    
    if best_alpha != 0:
        if best_alpha > 0:
            h[s] -= best_alpha
            h[t] += best_alpha
        else:
            h[t] -= (-best_alpha)
            h[s] += (-best_alpha)
    
    return h, best_c

def make_start(n, seed_val):

    x = np.linspace(0.0, 2.0, n, endpoint=False)

    h += rng.randn(n) * 0.02
    return project_constraint(h, n)

def gradient_descent(h, n, time_limit, tau_schedule):

    dx = 2.0 / n
    
    for pass_i, tau in enumerate(tau_schedule):

                best_c = c5_val
                best_h = h.copy()
            
            w = softmax_weights(corr, tau)
            grad = compute_gradient_fast(h, w)
            
            # Project gradient to tangent space (sum = 0)
            grad_tangent = grad - np.mean(grad)
            gnorm = np.linalg.norm(grad_tangent)
            
            if gnorm < 1e-12:
                break
            
            # Line search
            made_progress = False
            for _ in range(15):

                if c_new <= c5_val + 1e-13:
                    h = h_new
                    if c_new < c5_val - 1e-12:
                        made_progress = True
                    break
                step *= 0.5
            
            if not made_progress:
                stall_count += 1
                if stall_count > 50:

            rng = np.random.RandomState(42 + pass_i * 7)

    return best_h, best_c

    best_h = h.copy()

    no_improve_sweeps = 0

        sweep_start = time.time()

        cmax = np.max(corr)
```
