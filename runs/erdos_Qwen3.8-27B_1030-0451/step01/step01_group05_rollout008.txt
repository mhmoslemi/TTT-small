def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    # Fix sum
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    # Final sum fix
    diff2 = S - np.sum(h)
    if abs(diff2) > 1e-14:
        # Find best coordinate to adjust
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if np.any(interior):
            idx = int(np.argmax(interior * (h - 0.5) ** 2 + interior * 0.5))
            # Use the coordinate closest to 0.5
            dists = np.abs(h - 0.5)
            dists[~interior] = 10.0
            idx = int(np.argmin(dists))
            h[idx] += diff2
            h = np.clip(h, 0.0, 1.0)
    return h

def build_basis(n, max_k):
    """Build frequency basis matrix. Returns (B, freq_list) where B is n x (2*max_k+1)"""
    x = (np.arange(n) + 0.5) / n
    cols = []
    # Linear mode
    lin = x - x.mean()
    lin = lin / max(1e-12, np.ptp(lin))
    cols.append(lin)
    # Cosine and sine modes
    for k in range(1, max_k + 1):
        c = np.cos(2.0 * np.pi * k * x)
        c = c - c.mean()
        c = c / max(1e-12, np.ptp(c))
        cols.append(c)
        s = np.sin(2.0 * np.pi * k * x)
        s = s - s.mean()
        s = s / max(1e-12, np.ptp(s))
        cols.append(s)
    B = np.column_stack(cols)
    return B

def map_coeffs_to_h(coeffs, B, n, scale):

    z = 0.5 + scale * (B @ coeffs)

def global_basis_search(n, B, scale, pop_size, max_gens, rng, deadline, initial_h=None):
    """Differential evolution in coefficient space"""
    p = B.shape[1]
    # Initialize population
    X = np.zeros((pop_size, p))
    F = np.zeros(pop_size)
    
    for i in range(pop_size):
        X[i] = rng.uniform(-1.5, 1.5, size=p) / np.sqrt(np.arange(1, p+1))
        h_i = map_coeffs_to_h(X[i], B, n, scale)
        F[i] = exact_score(h_i)
    
    # Track best
    idx_best = int(np.argmin(F))
    best_f = F[idx_best]
    best_h = map_coeffs_to_h(X[idx_best], B, n, scale)
    
    # Add initial_h as a population member if provided
    if initial_h is not None:
        F[-1] = exact_score(initial_h)  # Actually we can't easily invert the mapping
        # Skip this - just use random init
    
    C = 0.8  # mutation factor
    CR = 0.8  # crossover rate
    
    for gen in range(max_gens):
        if time.time() > deadline:
            break
        
        for i in range(pop_size):
            # Select 3 distinct individuals
            r1, r2, r3 = rng.choice(pop_size, 3, replace=False)
            # Mutation
            v = X[r1] + C * (X[r2] - X[r3])
            # Crossover
            trial = X[i].copy()
            for j in range(p):
                if rng.random() < CR or j == int(rng.integers(0, p)):
                    trial[j] = v[j]
            # Bound
            trial = np.clip(trial, -3.0, 3.0)
            
            # Evaluate
            h_trial = map_coeffs_to_h(trial, B, n, scale)
            f_trial = exact_score(h_trial)
            
            if f_trial <= F[i]:
                X[i] = trial
                F[i] = f_trial
                if f_trial < best_f:
                    best_f = f_trial
                    best_h = h_trial.copy()
        
        # Occasionally reseed worst individuals
        if gen % 10 == 5:
            for _ in range(3):
                idx_worst = int(np.argmax(F))
                X[idx_worst] = rng.uniform(-1.5, 1.5, size=p) / np.sqrt(np.arange(1, p+1))
                F[idx_worst] = exact_score(map_coeffs_to_h(X[idx_worst], B, n, scale))
    
    return best_h, best_f

def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)

def run_slsqp(h0, tau, n, maxiter, deadline):
    """SLSQP optimization with smooth surrogate"""
    dx = 2.0 / n
    
    def obj(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = c.max()
        exponents = np.clip((c - c_max) / tau, -500, 500)
        return c_max + tau * math.log(np.sum(np.exp(exponents)))
    
    def obj_grad(h):
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = c.max()
        exponents = np.clip((c - c_max) / tau, -500, 500)
        weights = np.exp(exponents)
        sw = weights.sum()
        weights /= sw
        
        g = np.zeros(n)
        center = n - 1
        for K in range(2 * n - 1):
            w = weights[K] * dx
            if w < 1e-14:
                continue
            m = K - center
            if m >= 0:
                g[:n-m] += w * (1.0 - h[m:])
                g[m:] += w * (-h[:n-m])
            else:
                m2 = -m
                g[m2:] += w * (1.0 - h[:n-m2])
                g[:n-m2] += w * (-h[m2:])
        return g
    
    def con_fun(h):
        return np.sum(h) - n / 2.0
    
    def con_jac(h):
        return np.ones(n)
    
    cons = [{'type': 'eq', 'fun': con_fun, 'jac': con_jac}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(
            fun=obj, x0=h0, jac=obj_grad, method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

def local_polish(h, n, deadline, rng):
    """Multi-stage local polish"""
    cur = h.copy()
    cur_score = exact_score(cur)
    
    # Stage 1: SLSQP with decreasing tau
    for tau in [0.02, 0.01, 0.005, 0.002, 0.001]:
        if time.time() > deadline - 5:
            break
        h_new = run_slsqp(cur, tau, n, 150, deadline)
        if h_new is not None:
            s = exact_score(h_new)
            if s < cur_score - 1e-13:
                cur = h_new
                cur_score = s
    
    # Stage 2: Pair search
    if time.time() < deadline - 10:
        cur, cur_score = pair_search_exact(cur, 3000, rng, deadline)
    
    # Stage 3: More SLSQP
    for tau in [0.001, 0.0005]:
        if time.time() > deadline - 3:
            break
        h_new = run_slsqp(cur, tau, n, 100, deadline)
        if h_new is not None:
            s = exact_score(h_new)
            if s < cur_score - 1e-13:
                cur = h_new
                cur_score = s
    
    # Stage 4: Final pair search
    if time.time() < deadline - 5:
        cur, cur_score = pair_search_exact(cur, 2000, rng, deadline)
    
    return cur, cur_score
