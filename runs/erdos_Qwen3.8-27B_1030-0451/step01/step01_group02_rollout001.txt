import numpy as np
import time
import math

def corr_scaled(h):
    n = len(h)
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, "full") * dx

def exact_C5(h):
    if len(h) == 0:
        return 0.0
    return float(np.max(corr_scaled(h)))

def make_feasible(y, target):
    y = np.asarray(y, dtype=float).ravel()
    n = y.size
    if n == 0:
        return y
    target = float(target)
    if not np.all(np.isfinite(y)):
        y = np.full(n, 0.5)
    y = np.clip(y, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(y + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-14:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])]
                rem = r
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
                    rem -= inc
                    if rem <= 1e-14:
                        break
        else:
            idx = np.where(x > 0.0)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])[::-1]]
                rem = -r
                for i in order:
                    dec = min(x[i], rem)
                    x[i] -= dec
                    rem -= dec
                    if rem <= 1e-14:
                        break
    return np.clip(x, 0.0, 1.0)

def compute_grad(h, tau, dx):
    n = len(h)
    C = np.correlate(h, 1.0 - h, "full") * dx
    M = float(np.max(C))
    ctr = n - 1
    if tau <= 1e-12:
        act = C >= M - 1e-9
        cnt = float(np.count_nonzero(act))
        alpha = np.zeros_like(C)
        if cnt > 0:
            alpha[act] = 1.0 / cnt
        else:
            alpha[int(np.argmax(C))] = 1.0
    else:
        a = (C - M) / tau
        e = np.exp(a)
        s = float(e.sum())
        if s < 1e-300:
            alpha = np.zeros_like(C)
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha = e / s
    G = np.zeros(n)
    active = np.where(alpha > 1e-6)[0]
    for k in active:
        w = alpha[k] * dx
        d = int(k) - ctr
        if d == 0:
            G += w * (1.0 - 2.0 * h)
        elif d > 0:
            G[d:] += w * (1.0 - h[:-d])
            G[:-d] -= w * h[d:]
        else:
            dd = -d
            G[:n - dd] += w * (1.0 - h[dd:])
            G[dd:] -= w * h[:-dd]
    return G

def smooth_grad_line(h, target, deadline, max_iter=500):
    n = len(h)
    dx = 2.0 / float(n)
    h = make_feasible(h, target).copy()
    c = exact_C5(h)
    best_h = h.copy()
    best_c = c

    tau = 0.04
    min_tau = 0.0005
    tau_decay = 0.97
    momentum = 0.4
    v = np.zeros(n)
    stall = 0

    it = 0
    while time.time() < deadline and it < max_iter:
        it += 1
        G = compute_grad(h, tau, dx)
        G = G - float(np.mean(G))
        gnorm = float(np.linalg.norm(G))
        if gnorm < 1e-14:
            break

        v = momentum * v + G / (gnorm + 1e-16)
        direction = -v / (float(np.linalg.norm(v)) + 1e-16)

        improved = False
        for scale in (0.08, 0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001):
            h_new = h + scale * direction
            if np.any(h_new < -0.05) or np.any(h_new > 1.05):
                continue
            h_new = make_feasible(h_new, target)
            c_new = exact_C5(h_new)
            if c_new < c - 1e-11:
                h = h_new
                c = c_new
                improved = True
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                stall = 0
                break
        if not improved:
            stall += 1
            if stall >= 5:
                tau = max(min_tau, tau * tau_decay)
            if stall >= 30:
                h = make_feasible(h + np.random.default_rng(it).normal(0, 0.02, n), target)
                c = exact_C5(h)
                if c > best_c + 0.01:
                    h = best_h.copy()
                    c = best_c
                stall = 0
                tau = 0.02
                v = np.zeros(n)
        else:
            if stall == 0 and tau > min_tau and it % 10 == 0:
                tau = max(min_tau, tau * 0.95)
    return best_h, best_c

def pair_moves(h, c, target, rng, n_trials=200, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c = c
    for t in range(n_trials):
        if deadline is not None and time.time() > deadline:
            break
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        for sign in (1, -1):
            for delta in (0.04, 0.02, 0.01, 0.005):
                hi = best_h[i] + sign * delta
                hj = best_h[j] - sign * delta
                if hi < 0 or hi > 1 or hj < 0 or hj > 1:
                    continue
                h_new = best_h.copy()
                h_new[i] = hi
                h_new[j] = hj
                c_new = exact_C5(h_new)
                if c_new < best_c - 1e-11:
                    best_c = c_new
                    best_h = h_new
                    break
    return best_h, best_c

def coord_descent(h, c, target, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_c = c
    rng = np.random.default_rng(99)
    order = rng.permutation(n)
    for idx in order:
        if deadline is not None and time.time() > deadline:
            break
        i = int(idx)
        vi = best_h[i]
        # Try to flip coordinate i toward 0 or 1
        for target_val in (0.0, 1.0):
            delta = target_val - vi
            if abs(delta) < 1e-6:
                continue
            # Need another coordinate to absorb
            if delta > 0:
                # increase i, need to decrease some j
                candidates = np.where(best_h > 0.05)[0]
            else:
                # decrease i, need to increase some j
                candidates = np.where(best_h < 0.95)[0]
            if len(candidates) == 0:
                continue
            for j in candidates[:8]:
                if j == i:
                    continue
                h_new = best_h.copy()
                h_new[i] = target_val
                h_new[j] = best_h[j] - delta
                if 0 <= h_new[j] <= 1:
                    h_new = make_feasible(h_new, target)
                    c_new = exact_C5(h_new)
                    if c_new < best_c - 1e-11:
                        best_c = c_new
                        best_h = h_new
                        break
    return best_h, best_c

def resample_h(h, N):
    n0 = len(h)
    target = 0.5 * float(N)
    if n0 == N:
        return make_feasible(h, target)
    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))
    xold = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    y = np.interp(xnew, xold, h)
    return make_feasible(y, target)

def fourier_start(N, rng, n_harm=6):
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    a = rng.uniform(-0.15, 0.15, size=n_harm + 1)
    b = rng.uniform(-0.15, 0.15, size=n_harm + 1)
    y = 0.5
    for k in range(1, n_harm + 1):
        y += a[k] * np.cos(np.pi * k * x) + b[k] * np.sin(np.pi * k * x)
    return make_feasible(y, 0.5 * float(N))

def binarize(h, target):
    n = len(h)
    if abs(target - round(target)) > 1e-8:
        return None
    cnt = int(round(target))
    if cnt <= 0 or cnt >= n:
        return None
    order = np.argsort(h, kind="stable")[::-1][:cnt]
    b = np.zeros(n)
    b[order] = 1.0
    return b

def refine(h, deadline, rng):
    n = len(h)
    target = 0.5 * float(n)
    h = make_feasible(h, target).copy()
    c = exact_C5(h)
    best_h = h.copy()
    best_c = c

    phases = [
        ("subgrad", 0.5),
        ("pair", 0.15),
        ("coord", 0.1),
        ("subgrad", 0.35),
        ("pair", 0.05),
    ]

    phase_idx = 0
    stall_total = 0
    while time.time() < deadline - 5 and phase_idx < len(phases):
        ptype, pfrac = phases[phase_idx]
        phase_deadline = min(deadline - 5, time.time() + pfrac * (deadline - time.time()) * 3.3)
        if phase_deadline > deadline - 5:
            phase_deadline = deadline - 5

        if ptype == "subgrad":
            h2, c2 = smooth_grad_line(h, target, phase_deadline, max_iter=300)
        elif ptype == "pair":
            h2, c2 = pair_moves(h, c, target, rng, n_trials=500, deadline=phase_deadline)
        else:
            h2, c2 = coord_descent(h, c, target, deadline=phase_deadline)

        if c2 < c - 1e-12:
            h = h2
            c = c2
            stall_total = 0
        else:
            stall_total += 1

        if c < best_c:
            best_c = c
            best_h = h.copy()

        phase_idx += 1

    # Try binarization
    if n % 2 == 0 and time.time() < deadline - 5:
        hb = binarize(h, target)
        if hb is not None:
            cb = exact_C5(hb)
            if cb < best_c:
                best_h = hb
                best_c = cb
            h2, c2 = pair_moves(hb, cb, target, rng, n_trials=300, deadline=deadline - 5)
            if c2 < best_c:
                best_h = h2
                best_c = c2

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):

    best_c = exact_C5(best_h)
    best_n = n0

    # Determine resolutions to try
    Ns = []
    for f in (1, 2):

    if N_half not in Ns:

    Ns = sorted(set(Ns))
