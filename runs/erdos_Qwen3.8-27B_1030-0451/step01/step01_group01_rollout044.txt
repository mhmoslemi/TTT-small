    if n == 0:

    for _ in range(80):

        y = np.clip(h + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)

        if r > 0:
            idx = np.where(y < 1.0)[0]
