        affected_j = set()

        if 0 <= a < n and 0 <= a+sh < n:
            affected_j.add(a)

        j1 = a - sh
        if 0 <= j1 < n and 0 <= a < n:
            affected_j.add(j1)
        # j = b: h[j] = h[b] is affected
        if 0 <= b < n and 0 <= b+sh < n:
            affected_j.add(b)
        # j = b-sh: h[j+sh] = h[b] is affected
        j2 = b - sh
        if 0 <= j2 < n and 0 <= b < n:
            affected_j.add(j2)

            partner = j + sh
            # Compute dp (d coefficient of h[partner]) and dj (d coefficient of h[j])
            dp = 0.0
            if partner == a:
                dp = 1.0
            elif partner == b:
                dp = -1.0
            dj = 0.0
            if j == a:
                dj = 1.0
            elif j == b:
                dj = -1.0
            
            # (h[partner] + dp*d) * (1 - h[j] - dj*d)
            # = h[partner](1-h[j]) + d*[dp*(1-h[j]) - dj*h[partner]] + d²*(-dp*dj)

            quad[k] += -dp * dj

Now for the line search: given lin, quad, raw, and the domain [d_lo, d_hi], find d that minimizes:
f(d) = max(raw + d*lin + d²*quad)  (elementwise max)

Actually, f(d) = max_k (raw[k] + d*lin[k] + d²*quad[k])

The unaffected lags (where lin[k]=0 and quad[k]=0) just contribute raw[k], so:
f(d) = max( max_{k: lin[k]!=0 or quad[k]!=0} (raw[k] + d*lin[k] + d²*quad[k]), max_{k: lin[k]==0 and quad[k]==0} raw[k] )

Let C = max of raw over unaffected lags.
Let the affected lags be indexed by i_1, ..., i_m.
f(d) = max(C, q_1(d), ..., q_m(d)) where q_i(d) = raw[i_i] + d*lin[i_i] + d²*quad[i_i]

To minimize f(d) over [d_lo, d_hi]:
1. Evaluate at endpoints: f(d_lo), f(d_hi)
2. For each affected lag i: find d where q_i(d) = C (crossing with the constant). These are candidates.
   Also find the vertex of q_i if it's a minimum (quad[i] > 0) and it's in range.
3. For each pair of affected lags (i, j): find d where q_i(d) = q_j(d). These are candidates.
4. Evaluate f(d) at all candidate points, pick the minimum.

The number of candidates is O(m + m²) which is fine for small m.

```python
def line_search(raw, lin, quad, d_lo, d_hi):
    """Find d in [d_lo, d_hi] minimizing max(raw + d*lin + d²*quad)."""
    L = len(raw)
    # Find affected lags
    affected = np.where((lin != 0) | (quad != 0))[0]
    m = len(affected)
    
    if m == 0:
        return 0.0  # No change possible
    
    # Constant = max of raw over unaffected
    C = np.max(raw[~np.isin(np.arange(L), affected)])
    
    # Collect candidate d values
    candidates = [d_lo, d_hi]
    
    # For each affected lag, find crossing with C and vertex
    for i in affected:
        A = quad[i]
        B = lin[i]
        Cq = raw[i] - C  # q_i(d) - C = A*d² + B*d + Cq
        # q_i(d) = C when A*d² + B*d + Cq = 0
        if abs(A) < 1e-15:
            # Linear: B*d + Cq = 0
            if abs(B) > 1e-15:
                d = -Cq / B
                if d_lo <= d <= d_hi:
                    candidates.append(d)
        else:
            disc = B*B - 4*A*Cq
            if disc >= 0:
                sq = np.sqrt(disc)
                for d in [(-B+sq)/(2*A), (-B-sq)/(2*A)]:
                    if d_lo <= d <= d_hi:
                        candidates.append(d)
            # Vertex (minimum if A > 0)
            if A > 0:
                d_vertex = -B / (2*A)
                if d_lo <= d_vertex <= d_hi:
                    candidates.append(d_vertex)
    
    # Pairwise crossings
    for i in range(len(affected)):
        for j in range(i+1, len(affected)):
            ai, bi = affected[i], affected[j]
            dA = quad[ai] - quad[bi]
            dB = lin[ai] - lin[bi]
            dC = raw[ai] - raw[bi]
            if abs(dA) < 1e-15:
                if abs(dB) > 1e-15:
                    d = -dC / dB
                    if d_lo <= d <= d_hi:
                        candidates.append(d)
            else:
                disc = dB*dB - 4*dA*dC
                if disc >= 0:
                    sq = np.sqrt(disc)
                    for d in [(-dB+sq)/(2*dA), (-dB-sq)/(2*dA)]:
                        if d_lo <= d <= d_hi:
                            candidates.append(d)
    
    # Also add some uniform grid points for safety
    n_grid = min(20, max(5, int((d_hi - d_lo) * 100)))
    if n_grid > 0:
        grid = np.linspace(d_lo, d_hi, n_grid + 1)
        candidates.extend(grid.tolist())
    
    # Evaluate f(d) at all candidates
    best_d = 0.0
    best_val = np.max(raw)  # current value (d=0)
    
    for d in candidates:
        if d < d_lo - 1e-15 or d > d_hi + 1e-15:
            continue
        d = np.clip(d, d_lo, d_hi)
        val = np.max(raw + d*lin + d*d*quad)
        if val < best_val - 1e-14:
            best_val = val
            best_d = d
    
    return best_d
```

For pair selection: I want to find pairs (a, b) where increasing h[a] and decreasing h[b] reduces the max overlap. A good heuristic is:
- Compute for each coordinate i, a "descent direction score" based on the weighted sum of col_i(k) over active lags.
- Coordinates with low score should be increased (they're less "active"), and coordinates with high score should be decreased.

The score for coordinate i is:
score_i = sum_k w_k * col_i(k)

where w_k are weights (e.g., based on how close raw[k] is to the max).

col_i(k) can be computed as:
- For shift sh = k - (n-1):
  - If 0 <= i+sh < n: contribution from (1-h[i+sh]) [first factor]
  - If 0 <= i-sh < n: contribution from -h[i-sh] [second factor]

So: score_i = sum_k w_k * [(1-h[i+sh_k])*valid1 - h[i-sh_k]*valid2]

This can be computed efficiently with slicing.

Let me write a more complete implementation:

```python
def coord_scores(h, raw, tau):
    """Compute per-coordinate descent scores based on active lags."""
    n = len(h)
    center = n - 1
    L = 2 * n - 1
    
    # Weights based on proximity to max
    raw_max = np.max(raw)
    exponents = np.clip((raw - raw_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.ones(L) / L
    else:
        weights /= sw
    
    scores = np.zeros(n)
    
    for k in range(L):
        w = weights[k]
        if w < 1e-15:
            continue
        sh = k - center
        # For coordinate i: col_i(k) = (1-h[i+sh])*[valid] - h[i-sh]*[valid]
        # First term: i+sh in [0,n) means i in [-sh, n-1-sh]
        # Second term: i-sh in [0,n) means i in [sh, n-1+sh]
        
        if sh >= 0:
            # First term: i in [0, n-1-sh], partner = i+sh in [sh, n-1]
            idx1 = np.arange(0, n - sh)
            scores[idx1] += w * (1.0 - h[sh:n])
            # Second term: i in [sh, n-1], partner = i-sh in [0, n-1-sh]
            idx2 = np.arange(sh, n)
            scores[idx2] += w * (-h[0:n-sh])
        else:
            m = -sh
            # First term: i in [m, n-1], partner = i+sh = i-m in [0, n-1-m]
            idx1 = np.arange(m, n)
            scores[idx1] += w * (1.0 - h[0:n-m])
            # Second term: i in [0, n-1-m], partner = i-sh = i+m in [m, n-1]
            idx2 = np.arange(0, n - m)
            scores[idx2] += w * (-h[m:n])
    
    return scores
```

def pair_optimize(h, raw, n_passes, deadline, rng):
    """Local optimization via exact pair exchanges."""
    n = len(h)
    cur_h = h.copy()
    cur_raw = raw.copy()
    cur_score = float(np.max(cur_raw))
    best_h = cur_h.copy()
    best_score = cur_score
    
    for p in range(n_passes):
        if time.time() > deadline:
            break
        
        scores = coord_scores(cur_h, cur_raw, 0.3)
        
        # Select candidate pairs: low score (increase) paired with high score (decrease)
        order = np.argsort(scores)
        n_low = max(5, n // 10)
        n_high = max(5, n // 10)
        
        candidates = []
        for i in range(n_low):
            for j in range(n_high):
                a = int(order[i])  # low score, increase
                b = int(order[-1-j])  # high score, decrease
                if a != b:
                    candidates.append((a, b))
        
        # Add some random pairs
        for _ in range(n_low):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                candidates.append((a, b))
        
        best_improvement = 0.0
        best_move = None
        
        for (a, b) in candidates:
            # Compute line coefficients
            lin, quad = compute_pair_line(cur_h, a, b, cur_raw)
            
            # Domain
            d_lo = max(-cur_h[a], cur_h[b] - 1.0)
            d_hi = min(1.0 - cur_h[a], cur_h[b])
            if d_hi < d_lo:
                continue
            
            # Line search
            d = line_search(cur_raw, lin, quad, d_lo, d_hi)
            if d == 0.0:
                continue
            
            new_score = float(np.max(cur_raw + d*lin + d*d*quad))
            improvement = cur_score - new_score
            
            if improvement > best_improvement + 1e-12:
                best_improvement = improvement
                best_move = (a, b, d)
        
        if best_move is not None:
            a, b, d = best_move
            cur_h[a] += d
            cur_h[b] -= d
            # Recompute raw exactly (to avoid accumulation error)
            cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_raw = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            cur_score = float(np.max(cur_raw))
            if cur_score < best_score - 1e-14:
                best_score = cur_score
                best_h = cur_h.copy()
        else:
            # No improvement this pass; try with different tau or break
            break
    
    return best_h, float(best_score * (2.0/n))

    for k in range(L):
        w = weights[k]
        if w < 1e-15:
            continue
        sh = k - center
        if sh >= 0:
            if sh < n:
                scores[0:n-sh] += w * (1.0 - h[sh:n])
                scores[sh:n] += w * (-h[0:n-sh])
        else:
            m = -sh
            if m < n:
                scores[m:n] += w * (1.0 - h[0:n-m])
                scores[0:n-m] += w * (-h[m:n])
    
    return scores

def local_pair_descent(h, deadline, rng, max_passes=200):
    n = len(h)
    dx = 2.0 / n
    cur_h = h.copy()
    cur_raw = np.correlate(cur_h, 1.0 - cur_h, mode='full')
    cur_max = float(np.max(cur_raw))
    best_h = cur_h.copy()
    best_max = cur_max
    
    stagnation = 0
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        
        # Adaptive tau
        raw_range = float(np.max(cur_raw) - np.percentile(cur_raw, 50))
        tau = max(0.05, 0.3 * raw_range)
        
        scores = coord_scores(cur_h, cur_raw, tau)
        order = np.argsort(scores)
        
        # Candidate pairs
        n_cand = max(8, n // 8)
        candidates = []
        for i in range(n_cand):
            for j in range(n_cand):
                a = int(order[i])
                b = int(order[-1 - j])
                if a != b:
                    candidates.append((a, b))
        
        # Random pairs
        for _ in range(n_cand // 2):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                candidates.append((a, b))
        
        best_improvement = 0.0
        best_move = None
        
        for (a, b) in candidates:
            lin, quad = compute_pair_line(cur_h, a, b, cur_raw)
            d_lo = max(-cur_h[a], cur_h[b] - 1.0)
            d_hi = min(1.0 - cur_h[a], cur_h[b])
            if d_hi - d_lo < 1e-14:
                continue
            
            d = line_search(cur_raw, lin, quad, d_lo, d_hi)
            if abs(d) < 1e-14:
                continue
            
            new_max = float(np.max(cur_raw + d * lin + d * d * quad))
            improvement = cur_max - new_max
            if improvement > best_improvement + 1e-12:
                best_improvement = improvement
                best_move = (a, b, d)
        
        if best_move is not None:
            a, b, d = best_move
            cur_h[a] += d
            cur_h[b] -= d
            cur_h = np.clip(cur_h, 0.0, 1.0)
            # Fix sum if needed
            target_sum = n / 2.0
            cur_sum = float(np.sum(cur_h))
            if abs(cur_sum - target_sum) > 1e-10:
                # The exchange preserves sum by construction, but clipping might break it
                # Redistribute
                interior = (cur_h > 1e-10) & (cur_h < 1.0 - 1e-10)
                cnt = int(np.sum(interior))
                if cnt > 0:
                    cur_h[interior] += (target_sum - cur_sum) / cnt
                    cur_h = np.clip(cur_h, 0.0, 1.0)
            
            cur_raw = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            cur_max = float(np.max(cur_raw))
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_h = cur_h.copy()
            stagnation = 0
        else:
            stagnation += 1
            if stagnation > 10:
                break
    
    return best_h, float(best_max * dx)

def sa_escape(h, raw, deadline, rng, max_iters=5000, T_start=None, T_end=None):
    n = len(h)
    dx = 2.0 / n
    cur_h = h.copy()
    cur_raw = raw.copy()
    cur_max = float(np.max(cur_raw))
    best_h = cur_h.copy()
    best_max = cur_max
    
    if T_start is None:
        T_start = 0.3 * cur_max
    if T_end is None:
        T_end = 0.01 * T_start
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        T = T_start * (T_end / T_start) ** (it / max_iters)
        
        a = int(rng.integers(0, n))
        b = int(rng.integers(0, n))
        if a == b:
            continue
        
        lin, quad = compute_pair_line(cur_h, a, b, cur_raw)
        d_lo = max(-cur_h[a], cur_h[b] - 1.0)
        d_hi = min(1.0 - cur_h[a], cur_h[b])
        if d_hi - d_lo < 1e-14:
            continue
        
        d = line_search(cur_raw, lin, quad, d_lo, d_hi)
        if abs(d) < 1e-14:
            # Try a small random move instead
            d = rng.uniform(d_lo, d_hi)
            if abs(d) < 1e-14:
                continue
        
        new_max = float(np.max(cur_raw + d * lin + d * d * quad))
        delta = new_max - cur_max
        
        if delta < 0 or (T > 1e-10 and rng.random() < np.exp(-delta / T)):
            cur_h[a] += d
            cur_h[b] -= d
            cur_h = np.clip(cur_h, 0.0, 1.0)
            target_sum = n / 2.0
            cur_sum = float(np.sum(cur_h))
            if abs(cur_sum - target_sum) > 1e-10:
                interior = (cur_h > 1e-10) & (cur_h < 1.0 - 1e-10)
                cnt = int(np.sum(interior))
                if cnt > 0:
                    cur_h[interior] += (target_sum - cur_sum) / cnt
                    cur_h = np.clip(cur_h, 0.0, 1.0)
            cur_raw = np.correlate(cur_h, 1.0 - cur_h, mode='full')
            cur_max = float(np.max(cur_raw))
            if cur_max < best_max - 1e-12:
                best_max = cur_max
                best_h = cur_h.copy()
    
    return best_h, float(best_max * dx)

def interpolate_h(h_old, new_n):

    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)

def make_random_h(n, rng):

def make_periodic_h(n, rng):
    # Try a few periodic patterns
    period = int(rng.integers(max(2, n // 6), max(3, n // 3)))
    duty = rng.uniform(0.3, 0.7)

    t = ((x - 2.0 * phase) % (2.0 / period)) * period / 2.0
