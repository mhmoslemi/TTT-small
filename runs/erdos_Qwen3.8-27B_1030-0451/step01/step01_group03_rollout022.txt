import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def smooth_max_and_grad(h, tau):
    """Return (value, gradient) of smooth max of correlation."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    L = m + tau * np.log(e.sum())
    alpha = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    active = np.where(alpha > 1e-10)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi+1] += a * omh[lo-d:hi+1-d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= a * h[lo2+d:hi2+1+d]
    return L, grad * dx

def _c_eq_fun(h):
    return float(np.sum(h) - _global_target)

def _c_eq_jac(h):
    return np.ones(len(h))

_global_target = 0.0

def slsqp_smooth(h0, n, deadline, tau0, n_tau=12, maxiter_per_tau=60):
    """SLSQP optimization with decreasing tau schedule."""
    global _global_target
    _global_target = n / 2.0
    target = n / 2.0
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h_curr = h0.copy()
    
    c_vals = np.correlate(h0, 1-h0, 'full') * (2.0/n)
    spread = max(1e-4, c_vals.max() - c_vals.mean())
    tau_sched = tau0 * (0.55 ** np.arange(n_tau))
    
    constraints = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    for i, tau in enumerate(tau_sched):
        if time.time() > deadline - 2:
            break
        
        def obj(h):
            return smooth_max_and_grad(h, tau)[0]
        
        def grad(h):
            return smooth_max_and_grad(h, tau)[1]
        
        mi = max(30, maxiter_per_tau - 10*i) if i < 6 else maxiter_per_tau
        try:
            res = minimize(obj, h_curr, method='SLSQP', jac=grad,
                          bounds=bounds, constraints=constraints,
                          options={'maxiter': mi, 'ftol': 1e-12})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        
        c5e = evaluate_c5(h_curr)
        if c5e < best_c5 - 1e-12:

            grad[lo2:hi2] -= a * h[lo2-d:hi2-d]
    return grad

    c_vals = np.correlate(h0, 1 - h0, 'full') * (2.0 / n)
    spread = max(1e-4, c_vals.max() - c_vals.mean())
    if tau_init is None:
        tau_init = 0.5 * spread
    tau = tau_init
    alpha = 0.02
    stall = 0

    for i in range(n_iter):
        if time.time() > deadline - 2:
            break
        L = smooth_max_val(h, tau)
        g = smooth_max_grad(h, tau)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break

        h_new = project_feasible(h - alpha * g, target)
        L_new = smooth_max_val(h_new, tau)

        if L_new > L + 1e-10:
            alpha *= 0.5
            if alpha < 1e-9:
                alpha = 0.01
            continue

        h = h_new
        c5e = evaluate_c5(h)
        if c5e < best_c5 - 1e-12:
            best_c5 = c5e
            best_h = h.copy()
            stall = 0
            if alpha < 0.1:
                alpha *= 1.05
        else:
            stall += 1
            if stall > 300:
                alpha *= 0.7
                stall = 0

        if i % 500 == 499 and tau > 0.008:
            tau *= 0.6

    return best_h, best_c5

def fixed_point_optimize(h0, n, deadline, n_iter=8000):
    target = n / 2.0
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h = h0.copy()

    c_vals = np.correlate(h0, 1 - h0, 'full') * dx
    spread = max(1e-4, c_vals.max() - c_vals.mean())
    tau_p = 0.4 * spread
    alpha = 0.03
    stall = 0

    for i in range(n_iter):
        if time.time() > deadline - 2:
            break
        c = np.correlate(h, 1 - h, 'full') * dx
        m = c.max()
        s = np.clip((c - m) / tau_p, -500.0, 500.0)
        e = np.exp(s)
        p = e / e.sum()

        omh = 1.0 - h
        grad = np.zeros(n)
        active = np.where(p > 1e-10)[0]
        for k in active:
            a = p[k]
            d = k - (n - 1)
            lo1 = max(0, -d)
            hi1 = min(n, n - d)
            if lo1 < hi1:
                grad[lo1:hi1] += a * omh[lo1+d:hi1+d]
            lo2 = max(0, d)
            hi2 = min(n, n + d)
            if lo2 < hi2:
                grad[lo2:hi2] -= a * h[lo2-d:hi2+d]
        grad *= dx

        h_new = project_feasible(h - alpha * grad, target)
        c5_new = evaluate_c5(h_new)

        if c5_new < best_c5 - 1e-13:
            h = h_new
            best_c5 = c5_new
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 400:
                alpha *= 0.6
                stall = 0

        if i % 400 == 399 and tau_p > 0.005:
            tau_p *= 0.5

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(120, 0.5)

    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)
    par_c5 = evaluate_c5(parent)

    best_h = parent.copy()
    best_c5 = par_c5
    best_n = n_par

    # Phase 1: Optimize at parent resolution
    if time.time() < deadline - 60:
        sub_dl = time.time() + min(25, deadline - time.time() - 60)
        h_opt, c5_opt = slsqp_optimize(parent, n_par, sub_dl, tau0=0.02, n_tau=10, maxiter=80)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Resolution ladder
    n_list = []
    for m in [2, 3, 4, 5, 6, 8, 10]:
        n = n_par * m
        if n <= 2400:
            n_list.append(n)
    n_list.sort()

    for n in n_list:
        if time.time() > deadline - 50:
            break
        remaining = deadline - time.time()
        if remaining < 35:
            break
        h0 = interp_to(best_h, n)
        time_per = min(remaining * 0.3, 50.0)
        sub_dl = time.time() + time_per

        if n > 600:
            h_opt, c5_opt = proj_grad_optimize(h0, n, sub_dl, n_iter=4000)
        else:
            h_opt, c5_opt = slsqp_optimize(h0, n, sub_dl, n_tau=10, maxiter=40)

        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    # Phase 3: Fixed-point refinement
    if time.time() < deadline - 40:
        n = best_n
        time_left = deadline - time.time() - 10
        sub_dl = time.time() + min(time_left * 0.45, 50.0)
        h_opt, c5_opt = fixed_point_optimize(best_h, n, sub_dl, n_iter=8000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n

    # Phase 4: SLSQP polish if n is small
    if time.time() < deadline - 15 and best_n <= 600:
        sub_dl = time.time() + min(30, deadline - time.time() - 5)
        h_opt, c5_opt = slsqp_optimize(best_h, best_n, sub_dl, tau0=0.008, n_tau=8, maxiter=100)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = best_n

    # Phase 5: Random restarts
    if time.time() < deadline - 12:
        rng = np.random.default_rng(seed)
        n = best_n
        remaining = deadline - time.time() - 8
        n_restarts = max(1, min(5, int(remaining / 20)))

        for r in range(n_restarts):
            if time.time() > deadline - 8:
                break
            amp = 0.008 + 0.01 * r
            pert = rng.standard_normal(n) * amp
            h_try = project_feasible(best_h + pert, n / 2.0)
            sub_dl = time.time() + min(18, deadline - time.time() - 5)
            if n > 600:
                h_opt, c5_opt = proj_grad_optimize(h_try, n, sub_dl, n_iter=2500)
            else:
                h_opt, c5_opt = slsqp_optimize(h_try, n, sub_dl, tau0=0.008, n_tau=8, maxiter=60)
            if c5_opt < best_c5 - 1e-12:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n

    # Phase 6: One more proj_grad pass on final best
    if time.time() < deadline - 10:
        sub_dl = time.time() + min(20, deadline - time.time() - 3)
        h_opt, c5_opt = proj_grad_optimize(best_h, best_n, sub_dl, n_iter=3000)
        if c5_opt < best_c5 - 1e-12:
            best_c5 = c5_opt
            best_h = h_opt

    # Final cleanup
    final_n = len(best_h)
    diff = final_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] -= diff / cnt
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = final_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] += diff / cnt
            np.clip(best_h, 0.0, 1.0, out=best_h)

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
