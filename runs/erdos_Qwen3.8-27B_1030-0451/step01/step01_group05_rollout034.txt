        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_grads(h):
    n = len(h)
    grad = np.zeros((2 * n - 1, n))
    center = n - 1
    for k in range(n):
        nk = n - k
        grad[center + k, :nk] += (1.0 - h[k:])
        grad[center + k, k:] += (-h[:nk])
    for m in range(1, n):
        nm = n - m
        grad[center - m, m:] += (1.0 - h[:nm])
        grad[center - m, :nm] += (-h[m:])
    return grad

def slp_optimize(h0, n, max_iters, top_m, deadline):
    S = n / 2.0
    h = h0.copy()
    cur_score = exact_score(h)
    n_lags = 2 * n - 1
    if top_m is None or top_m > n_lags:
        top_m = n_lags
    alphas = [1.0, 0.8, 0.6, 0.4, 0.25, 0.1, 0.05]
    stall_count = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        raw = np.correlate(h, 1.0 - h, mode='full')
        grad = compute_grads(h)
        if top_m < n_lags:
            idx = np.argsort(raw)[::-1][:top_m]
        else:
            idx = np.arange(n_lags)
        m = len(idx)
        A_ub = np.zeros((m, n + 1))
        b_ub = np.zeros(m)
        for i, K in enumerate(idx):
            A_ub[i, :n] = grad[K]
            A_ub[i, n] = -1.0
            b_ub[i] = raw[K] - grad[K] @ h
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([S])
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        bounds = [(0.0, 1.0)] * n + [(0.0, 5.0)]
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'disp': False})
        except Exception:
            break
        if res.status != 0:
            break
        h_lp = np.clip(res.x[:n], 0.0, 1.0)
        improved = False
        for alpha in alphas:
            h_try = h + alpha * (h_lp - h)
            h_try = project_feasible(h_try, n)
            s_try = exact_score(h_try)
            if s_try < cur_score - 1e-14:
                h = h_try
                cur_score = s_try
                improved = True
                stall_count = 0
                break
        if not improved:
            stall_count += 1
            if stall_count > 5:
                break
    return h, cur_score

def pair_search(h0, max_iters, deadline, rng, use_grad):
    n = len(h0)
    h = h0.copy()
    cur_score = exact_score(h)
    best_h = h.copy()
    best_score = cur_score
    no_improve = 0
    fracs = [1.0, 0.6, 0.4, 0.2, 0.08, 0.03]
    for it in range(max_iters):
        if time.time() > deadline:
            break
        if use_grad and no_improve % 20 == 0:
            raw = np.correlate(h, 1.0 - h, mode='full')
            tau = 0.005 * max(cur_score, 0.1)
            scores = raw * (2.0 / n)
            c_max = np.max(scores)
            weights = np.exp(np.clip((scores - c_max) / tau, -300, 0))
            sw = np.sum(weights)
            if sw < 1e-300:
                weights = np.ones(len(raw)) / len(raw)
            else:
                weights /= sw
            grad = compute_grads(h)
            g = (weights[:, None] * grad).sum(axis=0)
            imin = int(np.argmin(g))
            imax = int(np.argmax(g))
            pairs = [(imin, imax)]
            for _ in range(2):
                a = int(rng.integers(n))
                b = int(rng.integers(n))
                if a != b:
                    pairs.append((a, b))
        else:
            a = int(rng.integers(n))
            b = int(rng.integers(n))
            pairs = [(a, b)] if a != b else []
        improved = False
        for (a, b) in pairs:
            if a == b:
                continue
            for direction in [0, 1]:
                if direction == 0:
                    ia, ib = a, b
                    max_d = min(1.0 - h[ia], h[ib])
                else:
                    ia, ib = b, a
                    max_d = min(1.0 - h[ia], h[ib])
                if max_d < 1e-12:
                    continue
                for frac in fracs:
                    d = max_d * frac
                    if d < 1e-12:
                        continue
                    h[ia] += d
                    h[ib] -= d
                    s = exact_score(h)
                    h[ia] -= d
                    h[ib] += d
                    if s < cur_score - 1e-14:
                        h[ia] += d
                        h[ib] -= d
                        cur_score = s
                        improved = True
                        break
                if improved:
                    break
        if improved:
            no_improve = 0
            if cur_score < best_score - 1e-14:
                best_h = h.copy()
                best_score = cur_score
        else:
            no_improve += 1
            if no_improve > 400:
                break
    return best_h, best_score

def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)

def make_structured(n, rng):
    x = np.arange(n) / n * 2.0
    h = 0.5 + 0.4 * np.sin(2 * np.pi * x * (0.5 + 0.3 * rng.random()))
    h = h + 0.3 * np.sin(2 * np.pi * x * (1.5 + 0.5 * rng.random()) + rng.random() * 2 * np.pi)
    return project_feasible(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 12.0

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None
    if parent is None or not np.all(np.isfinite(parent)) or len(parent) < 10:
        n_init = 81
        parent = project_feasible(np.full(n_init, 0.5), n_init)
    n_p = len(parent)
    parent = project_feasible(parent, n_p)

    best_h = parent.copy()
    best_score = exact_score(best_h)
    best_n = n_p

    # Phase 1: SLP with all lags
    if time.time() < deadline - 150:
        h_s, s_s = slp_optimize(best_h, best_n, 150, None, deadline - 150)
        if s_s < best_score - 1e-13:
            best_h = h_s
            best_score = s_s

    # Phase 2: Pair search polish
    if time.time() < deadline - 100:
        h_p, s_p = pair_search(best_h, 6000, deadline - 80, rng, True)
        if s_p < best_score - 1e-13:
            best_h = h_p
            best_score = s_p

    # Phase 3: SLP with focused lags (different top_m)
    for tm in [30, 60, 15, 100]:
        if time.time() > deadline - 80:
            break
        h_s, s_s = slp_optimize(best_h, best_n, 80, tm, deadline - 70)
        if s_s < best_score - 1e-13:
            best_h = h_s
            best_score = s_s
            if time.time() < deadline - 60:
                h_p, s_p = pair_search(best_h, 3000, deadline - 50, rng, True)
                if s_p < best_score - 1e-13:
                    best_h = h_p
                    best_score = s_p

    # Phase 4: Try different dimensions
    dim_cands = []
    for f in [0.85, 0.9, 1.05, 1.1, 1.15, 1.25, 0.75, 1.5, 1.4]:
        cn = int(round(best_n * f))
        if 30 <= cn <= 500 and cn != best_n:
            dim_cands.append(cn)
    # Also add some fixed sizes
    for fixed in [64, 80, 96, 100, 120, 150, 200]:
        if 30 <= fixed <= 500 and fixed != best_n:
            dim_cands.append(fixed)
    dim_cands = sorted(set(dim_cands))

    for dn in dim_cands:
        if time.time() > deadline - 35:
            break
        h_d = interpolate_h(best_h, dn)
        s_d = exact_score(h_d)
        if s_d < best_score + 0.008:
            h_d, s_d = slp_optimize(h_d, dn, 80, 40, deadline - 25)
            if time.time() < deadline - 20:
                h_d, s_d = pair_search(h_d, 2000, deadline - 15, rng, True)
            if s_d < best_score - 1e-13:
                best_h = h_d
                best_score = s_d
                best_n = dn

    # Phase 5: Random restarts
    if time.time() < deadline - 60:
        for ri in range(4):
            if time.time() > deadline - 30:
                break
            h_r = make_structured(best_n, rng)
            h_r, s_r = slp_optimize(h_r, best_n, 60, 50, deadline - 25)
            if time.time() < deadline - 20:
                h_r, s_r = pair_search(h_r, 2000, deadline - 15, rng, True)
            if s_r < best_score - 1e-13:
                best_h = h_r
                best_score = s_r

    # Phase 6: Final intensive polish
    if time.time() < deadline - 8:
        h_f, s_f = slp_optimize(best_h, best_n, 50, 20, deadline - 4)
        if s_f < best_score - 1e-13:
            best_h = h_f
            best_score = s_f
        h_f, s_f = pair_search(best_h, 5000, deadline, rng, True)
        if s_f < best_score - 1e-13:
            best_h = h_f
            best_score = s_f

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-11:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
