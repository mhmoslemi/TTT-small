        if abs(cn) > 1e-15:
            grad[:valid] += cn * dx * (1.0 - h[d:])
            grad[d:] += -cn * dx * h[:valid]

def evaluate_exact(h):

    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(x, target):

    n = len(x)
    if n == 0:
        return x
    x[~np.isfinite(x)] = 0.5

    hi = float(np.max(x)) + 2.0

        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

def smooth_positive_part(u, eta):
    z = np.clip(u / eta, -50.0, 50.0)
    s = 1.0 / (1.0 + np.exp(-z))
    return u * s

def smooth_obj(h, tau, eta, beta):
    n = len(h)
    corr = corr_array(h)
    u = corr - tau
    pos = np.maximum(smooth_positive_part(u, eta), 0.0)
    if beta == 2:
        return np.sum(pos * pos)
    else:
        return np.sum(pos ** beta)

def smooth_grad(h, tau, eta, beta):
    n = len(h)
    dx = 2.0 / n
    corr = corr_array(h)
    u = corr - tau
    z = np.clip(u / eta, -50.0, 50.0)
    s = 1.0 / (1.0 + np.exp(-z))
    pos = np.maximum(u * s, 0.0)
    dsp = s + u * s * (1.0 - s) / eta
    
    if beta == 2:
        w = 2.0 * pos * dsp
    else:
        w = beta * (pos ** (beta - 1)) * dsp
    
    grad = np.zeros(n)
    base = n - 1
    
    c0 = w[base]
    if abs(c0) > 1e-15:
        grad += c0 * dx * (1.0 - 2.0 * h)

        cn = w[base - d]
        if abs(cp) < 1e-15 and abs(cn) < 1e-15:
            continue
        valid = n - d
        if abs(cp) > 1e-15:

            grad[:valid] += -cp * dx * h[d:]
        if abs(cn) > 1e-15:
            grad[:valid] += cn * dx * (1.0 - h[d:])
            grad[d:] += -cn * dx * h[:valid]

    """SLSQP optimization of smooth positive-part objective."""
    n = len(h)

        return smooth_obj(x, tau, eta, beta)

            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(50):
        res = h.sum() - target
        if abs(res) < 1e-13:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def smooth_obj_grad(h, tau, eta, beta):
    n = len(h)
    dx = 2.0 / n
    corr = corr_array(h)
    u = corr - tau
    z = np.clip(u / eta, -50.0, 50.0)
    s = 1.0 / (1.0 + np.exp(-z))
    pos = np.maximum(u * s, 0.0)
    dsp = s + u * s * (1.0 - s) / eta
    if beta == 2:
        obj = np.sum(pos * pos)
        w = 2.0 * pos * dsp
    else:
        obj = np.sum(pos ** beta)
        w = beta * (pos ** (beta - 1)) * dsp
    grad = np.zeros(n)
    base = n - 1
    c0 = w[base]
    if abs(c0) > 1e-15:
        grad += c0 * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        cp = w[base + d]
        cn = w[base - d]
        if abs(cp) < 1e-15 and abs(cn) < 1e-15:
            continue
        valid = n - d
        if abs(cp) > 1e-15:
            grad[d:] += cp * dx * (1.0 - h[:valid])
            grad[:valid] += -cp * dx * h[d:]
        if abs(cn) > 1e-15:
            grad[:valid] += cn * dx * (1.0 - h[d:])
            grad[d:] += -cn * dx * h[:valid]
    return obj, grad

def slsqp_solve(h, tau, eta, beta, target, max_iter, timeout_s):
    n = len(h)
    t0 = time.time()
    def obj_f(x):
        if time.time() - t0 > timeout_s:
            raise RuntimeError("timeout")
        v, _ = smooth_obj_grad(x, tau, eta, beta)
        return v
    def grad_f(x):
        if time.time() - t0 > timeout_s:
            raise RuntimeError("timeout")
        _, g = smooth_obj_grad(x, tau, eta, beta)
        return g
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_f, h, jac=grad_f, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': max_iter, 'ftol': 1e-13})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_box_sum(res.x, target)
    except Exception:
        pass
    return h

def proj_grad_descent(h, tau, eta, beta, target, max_iter, timeout_s):
    n = len(h)
    t0 = time.time()
    best = h.copy()
    best_exact = evaluate_exact(h)
    _, cur_smooth = smooth_obj_grad(h, tau, eta, beta)
    step = 0.03
    for it in range(max_iter):
        if time.time() - t0 > timeout_s:
            break
        grad = smooth_obj_grad(h, tau, eta, beta)[1]
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        grad_dir = grad / gn
        accepted = False
        for _ in range(35):
            h_new = project_box_sum(h - step * grad_dir, target)
            s_new = smooth_obj_grad(h_new, tau, eta, beta)[0]
            if s_new < cur_smooth * (1.0 - 1e-5):
                h = h_new
                cur_smooth = s_new
                exact = evaluate_exact(h)
                if exact < best_exact:
                    best = h_new.copy()
                    best_exact = exact
                accepted = True
                step = min(0.1, step * 1.2)
                break
            step *= 0.5
            if step < 1e-12:
                break
        if not accepted:
            step *= 0.7
            if step < 1e-13:
                break
    return best, best_exact

def upscale_rep(h, factor):
    return np.repeat(h, factor)

def downscale_avg(h, factor):
    n = len(h)
    new_n = n // factor
    return h[:new_n * factor].reshape(new_n, factor).mean(axis=1)
