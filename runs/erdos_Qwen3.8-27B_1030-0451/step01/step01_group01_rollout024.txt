Let me also think about the pair swap more carefully. For a pair (i, j), we want to find the optimal ε that minimizes C₅ when we transfer ε from j to i (h[i] += ε, h[j] -= ε).

The change in f_k for each k is:
Δf_k = ε * ∂f_k/∂h[i] - ε * ∂f_k/∂h[j] = ε * (g_k[i] - g_k[j])

So the new f_k = f_k + ε * (g_k[i] - g_k[j])

We want max_k (f_k + ε * (g_k[i] - g_k[j])) to be minimized over ε ∈ [max(0, h[i]-h[j]+...), ...]

Wait, we need h[i] + ε ≤ 1 and h[j] - ε ≤ 1, h[i] + ε ≥ 0, h[j] - ε ≥ 0.
So ε ∈ [max(-h[i], -(1-h[j])), min(1-h[i], h[j])]

The new C₅ = max_k (f_k + ε * d_k) where d_k = g_k[i] - g_k[j]

This is a 1D piecewise-linear minimization! We can solve it exactly by finding the minimum of a max of linear functions of ε.

The minimum of max_k (a_k + b_k * ε) over ε is found at a point where at least two of the lines intersect at the maximum.

This is the "chebyshev approximation" problem in 1D, which can be solved by:
1. For each pair of lines (a_i + b_i*ε) and (a_j + b_j*ε), find their intersection
2. Evaluate the max at each intersection point (and at the boundary of the feasible interval)
3. Take the minimum

        if y.sum() > target:
            hi = mid
        else:

                if d_sign < 0:
                    eps_max = min(h[i], 1.0 - h[j])
                best_eps = 0.0
                best_val = vmax
                eps_lo = 0.0
                eps_hi = eps_max

                for _ in range(30):
                    eps_mid = (eps_lo + eps_hi) / 2.0
                    new_f = f_act + eps_mid * d_k
                    new_max = new_f.max()
                    other_max = v.max()
                    candidate = max(new_max, other_max)
                    if candidate < best_val:
                        best_val = candidate
                        best_eps = eps_mid
                    if d_k[np.argmax(new_f)] > 0:
                        eps_lo = eps_mid
                    else:
                        eps_hi = eps_mid

                if best_eps > 0.001:
                    if d_sign > 0:
                        h_tr = h.copy()
                        h_tr[i] += best_eps
                        h_tr[j] -= best_eps
                    else:
                        h_tr = h.copy()
                        h_tr[j] += best_eps
                        h_tr[i] -= best_eps
                    c_tr = evaluate_c5(h_tr)
                    if c_tr < best_c - 1e-12:
                        best_c = c_tr
                        best_h = h_tr.copy()
                        h = h_tr.copy()
                        pair_improved = True

        if not pair_improved:
            break

    return best_h, best_c

def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 1.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_new, endpoint=False)
    h_ext = np.concatenate([h0, h0[:1]])
    h_new = np.interp(x_new, x_old, h_ext, period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def multi_start_smooth(h0, deadline, n_starts=4, max_iters_each=300):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    per_start_time = max(5.0, (deadline - time.time()) / n_starts - 2.0)

    for s in range(n_starts):
        if time.time() > deadline - 5:
            break
        if s == 0:
            x0 = h0.copy()
        else:
            scale = 0.02 * (s + 1)
            x0 = h0 + np.random.randn(n) * scale
            x0 = project_sum_box(x0, n / 2.0)
            c0 = evaluate_c5(x0)
            if c0 > best_c * 1.15:
                continue
        sub_dl = min(deadline - 5, time.time() + per_start_time)
        try:
            h_res, c_res = smooth_subgrad(x0, sub_dl, max_iters=max_iters_each)
            if c_res < best_c:
                best_h = h_res
                best_c = c_res
        except Exception:
            pass

    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0
    deadline = start_time + budget_s - safety

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        n0 = 100
        h_init = np.full(n0, 0.5)
        h_init = project_sum_box(h_init, n0 / 2.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: Interpolate to n=120 and optimize
    n1 = 120
    if time_left() > 120:
        h_cur = transfer_resolution(best_h, n1)
        c_cur = evaluate_c5(h_cur)
        if c_cur <= best_c:
            best_h = h_cur
            best_c = c_cur
        dl1 = min(deadline, time.time() + budget_s * 0.18)
        try:
            h_r, c_r = smooth_subgrad(best_h, dl1, max_iters=800, beta_init=12.0)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
        except Exception:
            pass
        if time_left() > 60:
            dl1b = min(deadline, time.time() + budget_s * 0.08)
            try:
                h_r2, c_r2 = lp_trust_region(best_h, dl1b, max_iters=200)
                if c_r2 < best_c:
                    best_h, best_c = h_r2, c_r2
            except Exception:
                pass

    # Phase 2: Multi-start at current resolution
    if time_left() > 80:
        dl2 = min(deadline, time.time() + budget_s * 0.10)
        try:
            h_m, c_m = multi_start_smooth(best_h, dl2, n_starts=3, max_iters_each=200)
            if c_m < best_c:
                best_h, best_c = h_m, c_m
        except Exception:
            pass

    # Phase 3: Interpolate to n=160
    n2 = 160
    if time_left() > 150:
        h_cur = transfer_resolution(best_h, n2)
        c_cur = evaluate_c5(h_cur)
        if c_cur <= best_c:
            best_h = h_cur
            best_c = c_cur
        else:
            best_h = h_cur
            best_c = min(best_c, c_cur)
        dl3 = min(deadline, time.time() + budget_s * 0.15)
        try:
            h_r, c_r = smooth_subgrad(best_h, dl3, max_iters=1000, beta_init=15.0)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
        except Exception:
            pass
        if time_left() > 60:
            dl3b = min(deadline, time.time() + budget_s * 0.08)
            try:
                h_r2, c_r2 = lp_trust_region(best_h, dl3b, max_iters=200)
                if c_r2 < best_c:
                    best_h, best_c = h_r2, c_r2
            except Exception:
                pass

    # Phase 4: Interpolate to n=200
    n3 = 200
    if time_left() > 120:
        h_cur = transfer_resolution(best_h, n3)
        c_cur = evaluate_c5(h_cur)
        if c_cur <= best_c:
            best_h = h_cur
            best_c = c_cur
        else:
            best_h = h_cur
            best_c = min(best_c, c_cur)
        dl4 = min(deadline, time.time() + budget_s * 0.12)
        try:
            h_r, c_r = smooth_subgrad(best_h, dl4, max_iters=1200, beta_init=18.0)
            if c_r < best_c:
                best_h, best_c = h_r, c_r
        except Exception:
            pass
        if time_left() > 50:
            dl4b = min(deadline, time.time() + budget_s * 0.06)
            try:
                h_r2, c_r2 = lp_trust_region(best_h, dl4b, max_iters=150, active_max=30)
                if c_r2 < best_c:
                    best_h, best_c = h_r2, c_r2
            except Exception:
            pass

    # Phase 5: Pair local search
    if time_left() > 60:
        dl5 = min(deadline, time.time() + budget_s * 0.08)
        try:
            h_p, c_p = pair_local_search(best_h, dl5, max_rounds=4, window=10)
            if c_p < best_c:
                best_h, best_c = h_p, c_p
        except Exception:
            pass

    # Phase 6: Final smooth polish
    if time_left() > 30:
        dl6 = min(deadline - 2, time.time() + budget_s * 0.05)
        try:
            h_f, c_f = smooth_subgrad(best_h, dl6, max_iters=2000, beta_init=25.0, beta_max=300.0)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
        except Exception:
            pass

    # Phase 7: One more LP pass on best resolution
    if time_left() > 25:
        dl7 = min(deadline - 2, time.time() + 20)
        try:
            h_l, c_l = lp_trust_region(best_h, dl7, max_iters=100, active_max=25)
            if c_l < best_c:
                best_h, best_c = h_l, c_l
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    best_c = min(best_c, final_c)

    return best_h, float(best_c), len(best_h)
