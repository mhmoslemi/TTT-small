import numpy as np

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[lag:] += (1.0 - h[:n - lag])
        g[:n - lag] -= h[lag:]
    else:
        l = -lag
        g[:n - l] += (1.0 - h[l:])
        g[l:] -= h[:n - l]
    return g

def smooth_obj_grad(h, beta):
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 500)
    u = np.exp(expo)
    Z = u.sum()
    f = m - np.log(Z) / beta
    w = u / Z
    n = len(h)
    g = np.zeros(n)
    for k in range(len(v)):
        if w[k] > 1e-12:
            lag = k - (n - 1)
            g += w[k] * grad_lag(h, lag)
    return f, g

    if time_left() > 40:
        dl = min(deadline, time.time() + min(80, time_left() * 0.5))
        for attempt in range(3):
            if time.time() > dl:
                break
            try:
                if attempt == 0:
                    h_start = best_h.copy()
                elif attempt == 1:
                    h_start = best_h + np.random.randn(len(best_h)) * 0.05
                    h_start = project_sum_box(h_start, len(best_h) / 2.0)
                else:
                    h_start = best_h + np.random.randn(len(best_h)) * 0.1
                    h_start = project_sum_box(h_start, len(best_h) / 2.0)
                h_r, c_r = method_lbfgsb(h_start, min(deadline, time.time() + 25), max_betas=5)
                if c_r < best_c:
                    best_h = h_r
                    best_c = c_r
            except Exception:
                pass

    # Phase 7: Final LP trust region polish
    if time_left() > 30:
        dl = min(deadline, time.time() + min(100, time_left() * 0.5))
        try:
            h7, c7 = method_active_set_lp(best_h, dl, max_iters=200)
            if c7 < best_c:
                best_h = h7
                best_c = c7
        except Exception:
            pass

    # Phase 8: Projected gradient final polish
    if time_left() > 15:
        try:
            h8, c8 = method_proj_grad(best_h, deadline - 2, max_iters=3000)
            if c8 < best_c:
                best_h = h8
                best_c = c8
        except Exception:
            pass

    # Phase 9: One more L-BFGS-B round with high beta
    if time_left() > 20:
        n_cur = len(best_h)
        bounds = [(0.0, 1.0)] * n_cur
        beta = 512
        rho = 200.0
        def obj_final(x):
            f, _ = smooth_obj_grad(x, beta)
            s = x.sum() - n_cur / 2.0
            return f + rho * s * s
        def grad_final(x):
            _, g = smooth_obj_grad(x, beta)
            s = x.sum() - n_cur / 2.0
            return g + 2.0 * rho * s * np.ones(n_cur)
        try:
            res = minimize(obj_final, best_h, jac=grad_final, method='L-BFGS-B',
                           bounds=bounds, options={'maxiter': 300, 'ftol': 1e-18, 'gtol': 1e-10})
            if res.x is not None:
                x_new = project_sum_box(res.x, n_cur / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c:
                    best_h = x_new
                    best_c = c_new
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
