        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi + 1] += a * omh[lo - d:hi + 1 - d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= a * h[lo2 + d:hi2 + 1 + d]
    return grad * dx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def smooth_objective(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -300.0, 300.0)
    return m + tau * np.log(np.exp(s).sum())

def smooth_gradient(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -300.0, 300.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    active = np.where(alpha > 1e-10)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        lo = max(0, d)
        hi = min(n - 1, n - 1 + d)
        if lo <= hi:
            grad[lo:hi + 1] += a * omh[lo - d:hi + 1 - d]
        lo2 = max(0, -d)
        hi2 = min(n - 1, n - 1 - d)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= a * h[lo2 + d:hi2 + 1 + d]
    return grad * dx

def proj_grad_optimize(h0, n, tau, max_iter, deadline):
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_f = smooth_objective(h, tau)
    step = 1.0 / (2.0 * n)
    
    h_prev = None
    g_prev = None
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        g = smooth_gradient(h, tau)
        y = h - step * g
        h_new = project_feasible(y, target)
        f_new = smooth_objective(h_new, tau)
        
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        
        # Barzilai-Borwein step size update
        if it > 0 and h_prev is not None:
            s_vec = h_new - h_prev
            y_vec = g - g_prev
            sBs = np.dot(s_vec, s_vec)
            sBy = np.dot(s_vec, y_vec)
            yBy = np.dot(y_vec, y_vec)
            if sBs > 1e-14 and yBy > 1e-14:
                if it % 2 == 0:
                    step = max(1e-6, min(10.0, sBs / max(sBy, 1e-14)))
                else:
                    step = max(1e-6, min(10.0, sBy / max(yBy, 1e-14)))
                
                # Backtracking if step too large
                if np.abs(step * g).max() > 0.3:
                    step = 0.3 / (np.abs(g).max() + 1e-14)
        
        h_prev = h.copy()
        h = h_new
        g_prev = g
    
    return best_h, best_f

def fw_optimize(h0, n, max_iter, deadline):
    target = n / 2.0
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_f = smooth_objective(h, 0.05)
    
    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break
        tau = max(0.01, 0.2 / (1.0 + 0.01 * t))
        g = smooth_gradient(h, tau)
        
        # Frank-Wolfe: solve min g^T x s.t. 0<=x<=1, sum(x)=target
        sorted_idx = np.argsort(g)
        x = np.zeros(n)
        x[sorted_idx[:int(target)]] = 1.0
        if target != int(target):
            frac = target - int(target)
            x[sorted_idx[int(target)]] = frac
        
        gamma = 2.0 / (t + 2.0)
        h_new = (1.0 - gamma) * h + gamma * x
        h_new = project_feasible(h_new, target)
        
        f_new = smooth_objective(h_new, tau)
        if f_new < best_f:
            best_f = f_new
            best_h = h_new.copy()
        h = h_new
    
    return best_h, best_f

def slsqp_polish(h0, n, deadline):
    global _G_T, _G_N, _G_TAU
    _G_T = n / 2.0
    _G_N = n
    target = n / 2.0
    
    c5_init = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_init
    
    c_vals = np.correlate(h0, 1.0 - h0, 'full') * (2.0 / n)
    spread = max(1e-6, c_vals.max() - c_vals.mean())
    tau0 = max(0.01, 0.3 * spread)
    
    n_tau = 8
    tau_sched = tau0 * (0.6 ** np.arange(n_tau))
    
    constraints = [{'type': 'eq', 'fun': _c_eq_fun, 'jac': _c_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    h_curr = h0.copy()
    for i, tau in enumerate(tau_sched):
        if time.time() > deadline - 1:
            break
        _G_TAU = tau
        maxiter = 30 if tau > 0.01 else 50
        try:
            res = minimize(_c_obj, h_curr, method='SLSQP', jac=_c_grad,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-12})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        c5e = evaluate_c5(h_curr)
        if c5e < best_c5:
            best_c5 = c5e
            best_h = h_curr.copy()
    
    return best_h, best_c5

def make_random_start(n, rng):
    y = rng.standard_normal(n)
    return project_feasible(y, n / 2.0)

def make_sinusoidal_start(n, freq=2, amp=0.3, rng=None):
    x = np.linspace(0, 2, n, endpoint=False)
    h = 0.5 + amp * np.sin(np.pi * freq * x)
    return project_feasible(h, n / 2.0)

def make_block_start(n, frac=0.5):
    h = np.zeros(n)
    h[:int(n * frac)] = 1.0
    return project_feasible(h, n / 2.0)

def full_pipeline(h0, n, deadline):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    # Stage 1: Frank-Wolfe (fast, many iterations)
    if time.time() < deadline - 5:
        h_fw, _ = fw_optimize(h0, n, 500, deadline)
        c5_fw = evaluate_c5(h_fw)
        if c5_fw < best_c5:
            best_c5 = c5_fw
            best_h = h_fw
    
    # Stage 2: Projected gradient with smooth objective
    tau_sched = [0.1, 0.05, 0.02, 0.01, 0.005]
    current_h = best_h.copy()
    for tau in tau_sched:
        if time.time() > deadline - 3:
            break
        cur_c5 = evaluate_c5(current_h)
        # Use c5 as proxy for iterations count
        max_it = 200 if n <= 960 else 100
        h_pg, _ = proj_grad_optimize(current_h, n, tau, max_it, deadline)
        c5_pg = evaluate_c5(h_pg)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg
        current_h = h_pg
    
    # Stage 3: SLSQP polish
    if time.time() < deadline - 5:
        h_sp, c5_sp = slsqp_polish(best_h, n, deadline)
        if c5_sp < best_c5:
            best_c5 = c5_sp
            best_h = h_sp
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    
    # Get parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)
    
    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)
    par_c5 = evaluate_c5(parent)
    
    best_h = parent.copy()
    best_c5 = par_c5
    best_n = n_par
    
    rng = np.random.default_rng(seed)
    
    # Resolutions to try (sorted, small to large)
    n_list = []
    for mult in [2, 3, 4, 6, 8]:
        n_candidate = n_par * mult
        if n_candidate % 2 == 0 and n_candidate <= 2400:
            n_list.append(n_candidate)
    n_list = sorted(set(n_list))
    
    # Allocate time: spend more on larger resolutions
    # Give each resolution a time budget proportional to its index
    total_weight = sum(i + 1 for i in range(len(n_list)))
    
    for idx, n in enumerate(n_list):
        if time.time() > deadline - 30:
            break
        
        # Time budget for this resolution
        remaining_res = (len(n_list) - idx)
        remaining_time = deadline - time.time()
        time_for_this = min(remaining_time * 0.7, remaining_time / remaining_res * 1.5)
        stage_deadline = min(time.time() + time_for_this, deadline - 10)
        
        if stage_deadline < time.time() + 10:
            stage_deadline = deadline - 5
        
        # Multiple starting points for this resolution
        starts = []
        
        # 1. Interpolated parent
        starts.append(interp_h(parent, n))
        
        # 2. Perturbed parent
        starts.append(project_feasible(interp_h(parent, n) + rng.standard_normal(n) * 0.02, n / 2.0))
        
        # 3. Sinusoidal
        starts.append(make_sinusoidal_start(n, freq=3, amp=0.25, rng=rng))
        
        # 4. Random
        starts.append(make_random_start(n, rng))
        
        # 5. Another sinusoidal with different params
        starts.append(make_sinusoidal_start(n, freq=5, amp=0.2, rng=rng))
        
        # Evaluate all starts, keep top 3 by C5
        scored = []
        for s in starts:
            c5_s = evaluate_c5(s)
            scored.append((c5_s, s))
        scored.sort(key=lambda x: x[0])
        
        # Optimize top starts
        top_k = 3 if n <= 960 else 2
        for rank in range(min(top_k, len(scored))):
            if time.time() > stage_deadline:
                break
            h_start = scored[rank][1]
            h_opt, c5_opt = full_pipeline(h_start, n, stage_deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n
    
    # Final refinement on best solution: try slightly higher resolution
    if time.time() < deadline - 30:
        n_next = best_n + n_par
        if n_next <= 2400 and n_next % 2 == 0:
            h0 = interp_h(best_h, n_next)
            h_opt, c5_opt = full_pipeline(h0, n_next, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_next
    
    # Final shake: perturb best and re-optimize
    if time.time() < deadline - 15:
        n = best_n
        for amp in [0.005, 0.01]:
            if time.time() > deadline - 5:
                break
            h_shake = project_feasible(best_h + rng.standard_normal(n) * amp, n / 2.0)
            h_opt, c5_opt = full_pipeline(h_shake, n, deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
    
    # Final feasibility cleanup
    best_n = len(best_h)
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] -= diff / c
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = best_n / 2.0 - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] += diff / c
    best_c5 = evaluate_c5(best_h)
    
    return (best_h, best_c5, best_n)
