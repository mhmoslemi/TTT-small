def min_of_max_lines(c_arr, g_arr, d_min, d_max):
    """Find d in [d_min, d_max] that minimizes max_k (c_arr[k] + d*g_arr[k])"""
    n = len(c_arr)
    # Create sorted by slope
    order = np.argsort(g_arr)
    slopes = g_arr[order]
    intercepts = c_arr[order]
    
    # Build upper envelope (convex hull of lines)
    # Each line: y = slopes[i]*x + intercepts[i]
    # Upper envelope: for each x, the max value
    # Stack of lines that are on the upper envelope
    hull = []  # list of indices into the sorted arrays
    for i in range(n):
        while len(hull) >= 2:
            # Check if hull[-1] is redundant
            # The intersection of hull[-2] and hull[-1] is to the right of
            # the intersection of hull[-1] and i, then hull[-1] is redundant
            i1, i2 = hull[-2], hull[-1]
            # x12 = (intercepts[i1] - intercepts[i2]) / (slopes[i2] - slopes[i1])
            # x23 = (intercepts[i2] - intercepts[i]) / (slopes[i] - slopes[i2])
            # If x12 >= x23, then i2 is redundant
            denom1 = slopes[i2] - slopes[i1]
            denom2 = slopes[i] - slopes[i2]
            if abs(denom1) < 1e-15:
                # Same slope, keep the one with higher intercept
                if intercepts[i] > intercepts[i2]:
                    hull.pop()
                continue
            if abs(denom2) < 1e-15:
                # Same slope
                if intercepts[i] > intercepts[i2]:
                    hull.pop()
                continue
            x12 = (intercepts[i1] - intercepts[i2]) / denom1
            x23 = (intercepts[i2] - intercepts[i]) / denom2
            if x12 >= x23:
                hull.pop()
            else:
                break
        hull.append(i)
    
    # Find the minimum of the upper envelope over [d_min, d_max]
    # The upper envelope is piecewise linear, with breakpoints at the x-coordinates
    # where adjacent hull lines cross.
    
    if len(hull) == 1:
        # Single line, minimum at one of the boundaries
        d_opt = d_min if slopes[hull[0]] >= 0 else d_max
        best_val = c_arr[hull[0]] + d_opt * g_arr[hull[0]]
        return d_opt, best_val
    
    # Compute breakpoints
    breakpoints = [d_min]
    for i in range(len(hull) - 1):
        i1, i2 = hull[i], hull[i+1]
        denom = slopes[i2] - slopes[i1]
        if abs(denom) > 1e-15:
            x = (intercepts[i1] - intercepts[i2]) / denom
            if d_min < x < d_max:
                breakpoints.append(x)
    breakpoints.append(d_max)
    breakpoints = sorted(set(breakpoints))
    
    # Evaluate at all breakpoints
    best_d = d_min
    best_val = float('inf')
    for d in breakpoints:
        # Find the active line at d (or just compute max)
        vals = c_arr + d * g_arr
        val = np.max(vals)
        if val < best_val:
            best_val = val
            best_d = d
    
    return best_d, best_val

def min_max_plus_linear(c_arr, g_arr, lam, d_min, d_max):
    """Minimize max_k(c_arr[k] + d*g_arr[k]) + lam*d over d in [d_min, d_max]"""
    n_lines = len(c_arr)
    
    # Sort by slope
    order = np.argsort(g_arr)
    slopes = g_arr[order]
    intercepts = c_arr[order]
    
    # Build upper envelope
    hull = []
    for i in range(n_lines):
        while len(hull) >= 2:
            i2, i1 = hull[-1], hull[-2]
            denom1 = slopes[i2] - slopes[i1]
            if abs(denom1) < 1e-15:
                if intercepts[i] > intercepts[i2]:
                    hull.pop()
                continue
            denom2 = slopes[i] - slopes[i2]
            if abs(denom2) < 1e-15:
                if intercepts[i] > intercepts[i2]:
                    hull.pop()
                continue
            x12 = (intercepts[i1] - intercepts[i2]) / denom1
            x23 = (intercepts[i2] - intercepts[i]) / denom2
            if x12 >= x23:
                hull.pop()
            else:
                break
        hull.append(i)
    
    # Find where the slope of the envelope + lam crosses 0
    # The envelope's slope at any point is the slope of the active line
    # We want: active_slope + lam = 0, i.e., active_slope = -lam
    
    if len(hull) == 1:
        # Single line active everywhere
        active_slope = slopes[hull[0]]
        if active_slope + lam > 0:
            d_opt = d_min  # function increasing, minimize at left
        elif active_slope + lam < 0:
            d_opt = d_max  # function decreasing, minimize at right
        else:
            d_opt = d_min  # constant, either endpoint
        best_val = c_arr[hull[0]] + d_opt * g_arr[hull[0]] + lam * d_opt
        return d_opt, best_val
    
    # Compute breakpoints of the envelope
    # Between hull[i] and hull[i+1], the intersection is at:
    x_limits = [d_min]
    for i in range(len(hull)-1):
        i1, i2 = hull[i], hull[i+1]
        denom = slopes[i2] - slopes[i1]
        if abs(denom) > 1e-15:
            x = (intercepts[i1] - intercepts[i2]) / denom
            if d_min ≤ x ≤ d_max:
                x_limits.append(x)
    x_limits.append(d_max)
    x_limits = sorted(set(x_limits))
    
    # In each interval [x_limits[i], x_limits[i+1]], the active line is hull[i]
    # (or whichever line is on top)
    # The function in that interval is: c[hull[i]] + d*slopes[hull[i]] + lam*d
    # = c[hull[i]] + d*(slopes[hull[i]] + lam)
    # Minimum in the interval is at one of the endpoints.
    
    best_d = x_limits[0]
    best_val = float('inf')
    for i in range(len(x_limits)-1):
        active = hull[i]
        slope_total = slopes[active] + lam
        if slope_total > 1e-15:
            d_cand = x_limits[i]  # increasing, min at left
        elif slope_total < -1e-15:
            d_cand = x_limits[i+1]  # decreasing, min at right
        else:
            d_cand = x_limits[i]  # constant
        val = c_arr[active] + d_cand * g_arr[active] + lam * d_cand
        if val < best_val:
            best_val = val
            best_d = d_cand
    
    # Also check the endpoints explicitly (they should be covered above)
    for d_cand in [d_min, d_max]:
        val = np.max(c_arr + d_cand * g_arr) + lam * d_cand
        if val < best_val:
            best_val = val
            best_d = d_cand
    
    return best_d, best_val

def coordinate_descent(h, lam, max_sweeps, deadline):
    n = len(h)
    S = n / 2.0
    dx = 2.0 / n
    L = 2 * n - 1
    center = n - 1
    
    for sweep in range(max_sweeps):
        if time.time() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        improved = False
        for j in range(n):
            # Compute gradient of each c[k] w.r.t. h[j]
            g = np.zeros(L)
            s_range = np.arange(-(n-1), n)
            # ∂c(s)/∂h[j] = (1-h[j+s]) * I[0≤j+s<n] - h[j-s] * I[0≤j-s<n]
            # where s = s_range[k]
            valid_plus = (j + s_range) >= 0 & (j + s_range) < n
            valid_minus = (j - s_range) >= 0 & (j - s_range) < n
            g[valid_plus] = 1.0 - h[j + s_range[valid_plus]]
            g[valid_minus] += -h[j - s_range[valid_minus]]
            
            # Find optimal d
            d_min = -h[j]
            d_max = 1.0 - h[j]
            d_opt, _ = min_max_plus_linear(c, g, lam, d_min, d_max)
            
            if abs(d_opt) > 1e-10:
                h[j] += d_opt
                h = np.clip(h, 0, 1)
                # Project to satisfy sum constraint
                diff = S - np.sum(h)
                if abs(diff) > 1e-10:
                    # Distribute among interior points
                    interior = (h > 1e-8) & (h < 1.0 - 1e-8)
                    cnt = np.sum(interior)
                    if cnt > 0:
                        h[interior] += diff / cnt
                        h = np.clip(h, 0.0, 1.0)
                    else:
                        # All at boundary, adjust the closest to boundary
                        if diff > 0:
                            # Need to increase sum, find points closest to 1
                            dist_to_1 = 1.0 - h
                            idx = np.argmin(dist_to_1)
                            h[idx] += min(diff, 1.0 - h[idx])
                        else:
                            h[idx] += max(diff, -h[idx])
                improved = True
        if not improved:
            break
    
    return h
