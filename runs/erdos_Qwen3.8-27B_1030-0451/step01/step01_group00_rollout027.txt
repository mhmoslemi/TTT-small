```python
def gradient_at_lag(h, lag, n):

    # corr[lag] = sum_i h[i] * (1-h)[i + lag - (n-1)]
    # = sum_i h[i] * (1 - h[i + lag - n + 1])
    # Let offset = lag - (n-1)
    # corr[lag] = sum_i h[i] * (1 - h[i + offset])
    # where both i and i+offset are in [0, n-1]
    
    offset = lag - (n - 1)
    grad = np.zeros(n)
    
    if offset >= 0:
        # i ranges from 0 to n-1-offset
        # d/dh[i] h[i]*(1-h[i+offset]) = (1-h[i+offset]) for i in [0, n-1-offset]
        # d/dh[j] h[i]*(1-h[j]) = -h[i] for j = i+offset, i in [0, n-1-offset]
        # So j ranges from offset to n-1
        valid_i = range(n - offset)
        for i in valid_i:
            grad[i] += (1.0 - h[i + offset])
            grad[i + offset] -= h[i]
    else:
        # offset < 0, let L = -offset > 0
        # i+offset = i-L must be >= 0, so i >= L
        # i ranges from L to n-1
        L = -offset
        for i in range(L, n):
            grad[i] += (1.0 - h[i - L])
            grad[i - L] -= h[i]
    
    return grad
```

- autocorr(g)[k] to be positive for k=1,2,... (to reduce corr[k])

The maximum of sum g[i]² subject to |g[i]| ≤ 1/2 and sum g[i] = 0 is achieved when g[i] = ±1/2 (i.e., h is binary). In that case, sum g[i]² = n * 1/4 = n/4, and corr[0] = 0.

```python
import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 2.0
    hi = float(np.max(p)) + 2.0
    for _ in range(120):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-14:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_grad(h, corr, tau):
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-15:
            continue
        offset = k - (n - 1)
        if offset >= 0:
            L = n - offset
            idx = np.arange(L)
            grad[idx] += wk * (1.0 - h[idx + offset])
            grad[idx + offset] -= wk * h[idx]
        else:
            L = -offset
            idx = np.arange(L, n)
            grad[idx] += wk * (1.0 - h[idx - L])
            grad[idx - L] -= wk * h[idx]
    return grad

def subgrad_max(h, corr, tol):
    n = len(h)
    mx = np.max(corr)
    active = np.where(corr >= mx - tol)[0]
    if len(active) == 0:
        active = np.array([np.argmax(corr)])
    grad = np.zeros(n)
    for k in active:
        offset = k - (n - 1)
        if offset >= 0:
            L = n - offset
            idx = np.arange(L)
            grad[idx] += (1.0 - h[idx + offset])
            grad[idx + offset] -= h[idx]
        else:
            L = -offset
            idx = np.arange(L, n)
            grad[idx] += (1.0 - h[idx - L])
            grad[idx - L] -= h[idx]
    grad /= len(active)
    return grad

def opt_phase1(h0, time_limit, beta=0.85):
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / n
    v = np.zeros(n)
    t0 = time.time()
    step = 0.8 / n
    stall = 0
    max_stall = 500

    while time.time() - t0 < time_limit - 3:
        elapsed = time.time() - t0
        frac = min(elapsed / max(time_limit, 1.0), 1.0)
        tau = 8.0 + 400.0 * (frac ** 0.6)

        corr = compute_corr(h)
        c5_val = float(np.max(corr) * dx)
        improved = False
        if c5_val < best_c5 - 1e-13:
            best_c5 = c5_val
            best_h = h.copy()
            improved = True
            stall = 0
        else:
            stall += 1

        if stall > max_stall:
            rng_s = np.random.RandomState(int(time.time()) % 10000)
            h = project_constraint(h + rng_s.randn(n) * 0.03, n)
            v *= 0.3
            stall = 0
            max_stall = int(max_stall * 1.3)
            continue

        grad = smooth_grad(h, corr, tau)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break

        v = beta * v - step * grad
        h_new = project_constraint(h + v, n)

        if c_new <= c5_val + 1e-11:

        else:

                h = h_new2
                v *= 0.5
            else:

def opt_phase2(h0, time_limit):

    best_c5 = compute_c5(h)
```
