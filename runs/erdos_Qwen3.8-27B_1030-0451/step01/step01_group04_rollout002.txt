    """Find valid range of delta such that 0 <= h + delta*a <= 1"""
    pos = a > 1e-15
    neg = a < -1e-15
    delta_min = 0.0

    b = 1.0 - h

    # Shape: (n_grid, 2N-1)
    vals = C0[None, :] + grid[:, None] * L[None, :] - (grid[:, None] ** 2) * Q[None, :]

import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_correlations(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_to_box_simplex(h, target_sum):
    """Project h onto {0 <= x <= 1, sum(x) = target_sum}"""
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = 0.5 * (lo + hi)
    h = np.clip(h + t, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] = min(1.0, h[idx[0]] + residual)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] = max(0.0, h[idx[0]] + residual)
    return h

def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_box_simplex(h_new, N_new / 2.0)

def get_delta_bounds(h, a):
    """Find [delta_min, delta_max] such that 0 <= h + delta*a <= 1 for all components."""
    pos = a > 1e-14
    neg = a < -1e-14
    d_min = 0.0
    d_max = 0.0

        d_min = max(d_min, np.min(-h[pos] / a[pos]))

        d_max = min(d_max, np.min(-h[neg] / a[neg]))
    return d_min, d_max

def exact_line_search(h, a, n_coarse=300, n_fine=300):
    """Minimize max_r c_r(h + delta*a) over valid delta."""
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h
    
    d_min, d_max = get_delta_bounds(h, a)
    if d_max - d_min < 1e-13:
        return 0.0, None
    
    # Margin
    m = 1e-10 * max(1.0, abs(d_min), abs(d_max))
    d_lo = d_min + m
    d_hi = d_max - m
    if d_hi - d_lo < 1e-13:
        return 0.0, None
    
    # Precompute
    C0 = np.correlate(h, b, mode='full') * dx
    L = (np.correlate(a, b, mode='full') - np.correlate(h, a, mode='full')) * dx
    Q = np.correlate(a, a, mode='full') * dx
    
    # Coarse grid
    grid = np.linspace(d_lo, d_hi, n_coarse)
    vals = C0[None, :] + grid[:, None] * L[None, :] - (grid ** 2)[:, None] * Q[None, :]
    F = vals.max(axis=1)
    best_idx = np.argmin(F)
    best_val = F[best_idx]
    best_delta = grid[best_idx]
    
    # Fine grid around best
    span = (d_hi - d_lo) / n_coarse
    lo2 = max(d_lo, best_delta - 3 * span)
    hi2 = min(d_hi, best_delta + 3 * span)
    if hi2 - lo2 < 1e-13:
        return best_delta, best_val
    grid2 = np.linspace(lo2, hi2, n_fine)
    vals2 = C0[None, :] + grid2[:, None] * L[None, :] - (grid2 ** 2)[:, None] * Q[None, :]
    F2 = vals2.max(axis=1)
    best_idx2 = np.argmin(F2)
    best_val = F2[best_idx2]
    best_delta = grid2[best_idx2]
    
    return best_delta, best_val

def compute_gradient_max(c_active, h, N):
    """Compute subgradient of max using active correlations.
    For active r with shift d = r-(N-1):
    ∂c_r/∂h[m] = dx * [(1-h[m-d])*I - h[m+d]*I]
    """
    dx = 2.0 / N
    grad = np.zeros(N)
    for r in c_active:
        d = r - (N - 1)
        # h[m+d] term: m in [max(0,-d), min(N-1, N-1-d)]
        #   contributes -h[m+d] * dx to grad[m]
        lo1 = max(0, -d)
        hi1 = min(N - 1, N - 1 - d)
        if lo1 <= hi1:
            m_indices = np.arange(lo1, hi1 + 1)
            grad[m_indices] -= dx * h[m_indices + d]
        
        # (1-h[m-d]) term: m in [max(0,d), min(N-1, N-1+d)]
        #   contributes (1-h[m-d]) * dx to grad[m]
        lo2 = max(0, d)
        hi2 = min(N - 1, N - 1 + d)
        if lo2 <= hi2:
            m_indices = np.arange(lo2, hi2 + 1)
            grad[m_indices] += dx * (1.0 - h[m_indices - d])
    return grad

def make_zero_sum(a):
    return a - np.mean(a)

def normalize(a):
    n = np.linalg.norm(a)
    if n < 1e-14:

import numpy as np
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def compute_correlations(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx


def project_to_box_simplex(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = 0.5 * (lo + hi)
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] = min(1.0, h[idx[0]] + residual)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] = max(0.0, h[idx[0]] + residual)
    return h


def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_to_box_simplex(h_new, N_new / 2.0)


def get_delta_bounds(h, a):
    pos = a > 1e-14
    neg = a < -1e-14
    d_min = 0.0
    d_max = 0.0
    if np.any(pos):
        d_min = max(d_min, float(np.min(-h[pos] / a[pos])))
        d_max = min(d_max, float(np.min((1.0 - h[pos]) / a[pos])))
    if np.any(neg):
        d_min = max(d_min, float(np.min((1.0 - h[neg]) / a[neg])))
        d_max = min(d_max, float(np.min(-h[neg] / a[neg])))
    return d_min, d_max


def exact_line_search(h, a, N, n_coarse=400, n_fine=400):
    b = 1.0 - h
    d_min, d_max = get_delta_bounds(h, a)
    if d_max - d_min < 1e-13:
        return 0.0, None
    m = 1e-10 * max(1.0, abs(d_min), abs(d_max))
    d_lo = d_min + m
    d_hi = d_max - m
    if d_hi - d_lo < 1e-13:
        return 0.0, None
    dx = 2.0 / N
    C0 = np.correlate(h, b, mode='full') * dx
    L = (np.correlate(a, b, mode='full') - np.correlate(h, a, mode='full')) * dx
    Q = np.correlate(a, a, mode='full') * dx
    grid = np.linspace(d_lo, d_hi, n_coarse)
    g2 = grid ** 2
    F = (C0[None, :] + grid[:, None] * L[None, :] - g2[:, None] * Q[None, :]).max(axis=1)
    best_idx = int(np.argmin(F))
    best_val = F[best_idx]
    best_delta = grid[best_idx]
    span = (d_hi - d_lo) / n_coarse
    lo2 = max(d_lo, best_delta - 3 * span)
    hi2 = min(d_hi, best_delta + 3 * span)
    if hi2 - lo2 > 1e-13:
        grid2 = np.linspace(lo2, hi2, n_fine)
        g2_2 = grid2 ** 2
        F2 = (C0[None, :] + grid2[:, None] * L[None, :] - g2_2[:, None] * Q[None, :]).max(axis=1)
        best_idx2 = int(np.argmin(F2))
        if F2[best_idx2] < best_val:
            best_val = F2[best_idx2]
            best_delta = grid2[best_idx2]
    return best_delta, best_val


def single_gradient(r, h, N):
    dx = 2.0 / N
    d = r - (N - 1)
    grad = np.zeros(N)
    lo1 = max(0, -d)
    hi1 = min(N - 1, N - 1 - d)
    if lo1 <= hi1:
        idx = np.arange(lo1, hi1 + 1)
        grad[idx] -= dx * h[idx + d]
    lo2 = max(0, d)
    hi2 = min(N - 1, N - 1 + d)
    if lo2 <= hi2:
        idx = np.arange(lo2, hi2 + 1)
        grad[idx] += dx * (1.0 - h[idx - d])
    return grad


def make_zero_sum(a):
    return a - np.mean(a)


def normalize_dir(a):
    n = np.linalg.norm(a)
    if n < 1e-14:
        return None
    return a / n


def optimize_at_resolution(h, N, rng, deadline, max_iter=None):
    target_sum = N / 2.0
    best_c5 = compute_c5(h)
    best_h = h.copy()
    h = h.copy()
    stagnation = 0
    iter_count = 0
    if max_iter is None:
        max_iter = 100000

    while time.time() < deadline - 5 and iter_count < max_iter:
        iter_count += 1
        c = compute_correlations(h)
        c_max = float(np.max(c))

        if c_max < best_c5:
            best_c5 = c_max
            best_h = h.copy()

        k = min(12, len(c))
        active = np.argsort(c)[-k:]
        active_sorted = active[np.argsort(c[active])[::-1]]

        directions = []

        # Combined active gradient
        grad_combo = np.zeros(N)
        for r in active_sorted:
            grad_combo += single_gradient(int(r), h, N)
        a = make_zero_sum(-grad_combo)
        nd = normalize_dir(a)
        if nd is not None:
            directions.append(nd)

        # Individual active gradients (top 4)
        for r in active_sorted[:4]:
            g = single_gradient(int(r), h, N)
            a = make_zero_sum(-g)
            nd = normalize_dir(a)
            if nd is not None:
                directions.append(nd)

        # Anti-gradient: move mass from high-h to low-h in a structured way
        sorted_idx = np.argsort(h)
        low_half = sorted_idx[:N // 4]
        high_half = sorted_idx[-N // 4:]
        a_struct = np.zeros(N)
        a_struct[low_half] = 1.0
        a_struct[high_half] = -1.0
        a_struct = make_zero_sum(a_struct)
        nd = normalize_dir(a_struct)
        if nd is not None:
            directions.append(nd)

        # Random directions
        for _ in range(4):
            a = rng.standard_normal(N)
            a = make_zero_sum(a)
            nd = normalize_dir(a)
            if nd is not None:
                directions.append(nd)

        # Smooth random (low frequency)
        f = np.fft.rfft(rng.standard_normal(N))
        freqs = np.arange(len(f))
        mask = freqs < N // 8
        f[~mask] *= 0.1
        a = np.fft.irfft(f, N)
        a = make_zero_sum(a)
        nd = normalize_dir(a)
        if nd is not None:
            directions.append(nd)

        improved = False
        for a in directions:
            delta, val = exact_line_search(h, a, N)
            if val is not None and val < c_max - 1e-14:
                h_new = h + delta * a
                h_new = np.clip(h_new, 0.0, 1.0)
                h_new = project_to_box_simplex(h_new, target_sum)
                c5_new = compute_c5(h_new)
                if c5_new < c_max:
                    h = h_new
                    improved = True
                    break

        if not improved:
            stagnation += 1
            if stagnation % 30 == 0:
                scale = min(0.05, 0.01 * (1 + stagnation // 60))
                h_pert = h + rng.standard_normal(N) * scale
                h_pert = project_to_box_simplex(h_pert, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5 + 0.01:
                    h = h_pert
                stagnation = 0
            elif stagnation > 200:
                break
        else:
            stagnation = 0

    return best_h, best_c5
