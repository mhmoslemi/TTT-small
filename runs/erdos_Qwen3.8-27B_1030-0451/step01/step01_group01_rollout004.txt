        c_arr = np.correlate(h, 1.0 - h, mode="full")
        c_cur = int(c_arr.max())
        phase_dur = total_time / max(num_restarts - restart, 1)
        T0 = T_start * (0.6 ** restart)
        t0 = time.time()
        no_improve = 0
        max_stall = 5000 + 3000 * restart

        while time.time() < deadline:
            if no_improve > max_stall:
                break
            frac = min((time.time() - t0) / phase_dur, 1.0)
            T = T0 * (T_end / T_start) ** frac

            oi = np.where(h == 1)[0]
            zi = np.where(h == 0)[0]
            if len(oi) < 2 or len(zi) < 2:
                break
            i1, i2 = np.random.choice(oi, 2, replace=False)
            j1, j2 = np.random.choice(zi, 2, replace=False)
            h[i1] = 0
            h[i2] = 0
            h[j1] = 1
            h[j2] = 1
            ca = np.correlate(h, 1.0 - h, mode="full")
            cn = int(ca.max())
            if cn < c_cur:
                c_cur = cn
                if cn < best_c:
                    best_c = cn
                    best_h = h.copy()
                    no_improve = 0
                else:
                    no_improve += 1
            else:
                if T > 0.1 and np.random.random() < np.exp(-(cn - c_cur) / T):
                    c_cur = cn
                    no_improve += 1
                else:
                    h[i1] = 1
                    h[i2] = 1
                    h[j1] = 0
                    h[j2] = 0
                    no_improve += 1
    return best_h, best_c * 2.0 / n

def grad_correlation(h, k):
    n = len(h)
    g = np.zeros(n)
    if k >= 0:
        g[:n - k] += (1.0 - h[k:])
        g[k:] -= h[:n - k]
    else:
        l = -k
        g[:n - l] += (1.0 - h[l:])
        g[l:] -= h[:n - l]
    return g

def continuous_refine(h0, deadline, max_iters=3000):
    n = len(h0)
    h = project_sum_box(h0, n / 2.0)
    best_c = evaluate_c5(h)
    best_h = h.copy()
    step = 0.008
    patience = 0

    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        if patience > 500:
            break
        c_arr = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        c_max = c_arr.max()
        thr = c_max * 0.97
        active = np.where(c_arr >= thr)[0]
        g = np.zeros(n)
        for k_idx in active:
            k = k_idx - (n - 1)
            g += grad_correlation(h, k)
        if len(active) > 0:
            g /= len(active)
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        h_new = h - step * g
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < best_c - 1e-13:
            h = h_new
            best_c = c_new
            best_h = h_new.copy()
            step = min(step * 1.03, 0.04)
            patience = 0
        else:
            step *= 0.92
            patience += 1
            if step < 1e-7:
                break
    return best_h, best_c

def lp_trust_region(h0, deadline, max_iters=200):
    from scipy.optimize import linprog
    n = len(h0)
    h = project_sum_box(h0, n / 2.0)
    best_h = h.copy()
    best_c = evaluate_c5(h)
    radius = 0.1
    no_improve = 0

    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        v = np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)
        vmax = v.max()
        gap = max(0.02 * vmax, 0.003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 60:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-60:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-10:]
        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i, k_idx in enumerate(active_idx):
            k = k_idx - (n - 1)
            A[i] = grad_correlation(h, k)
            b_lin[i] = A[i] @ h - v[k_idx] * n / 2.0
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-0.5]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.0]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                radius = min(radius * 1.08, 0.2)
            else:
                no_improve += 1
                if no_improve >= 6:
                    radius = max(radius * 0.6, 0.005)
                    no_improve = 0
                    if radius <= 0.005:
                        break
        else:
            no_improve += 1
            if no_improve >= 10:
                radius = max(radius * 0.6, 0.005)
                no_improve = 0
                if radius <= 0.005:
                    break
    return best_h, best_c

def tabu_search(h_init, start_time, deadline, tabu_len=30, max_iters=50000):
    n = len(h_init)
    num_ones = n // 2
    h = to_binary(h_init, num_ones)
    c_arr = np.correlate(h, 1.0 - h, mode="full")
    c_cur = int(c_arr.max())
    best_c = c_cur
    best_h = h.copy()
    tabu = {}
    it = 0

    while it < max_iters and time.time() < deadline:
        oi = np.where(h == 1)[0]
        zi = np.where(h == 0)[0]
        i = oi[np.random.randint(len(oi))]
        j = zi[np.random.randint(len(zi))]
        key = (min(i, j), max(i, j))
        if tabu.get(key, -1000) >= it - tabu_len:
            it += 1
            continue
        h[i] = 0
        h[j] = 1
        ca = np.correlate(h, 1.0 - h, mode="full")
        cn = int(ca.max())
        if cn < c_cur or cn < best_c:
            c_cur = cn
            if cn < best_c:
                best_c = cn
                best_h = h.copy()
            tabu[key] = it + tabu_len
        else:
            h[i] = 1
            h[j] = 0
        it += 1
        if len(tabu) > 2000:
            tabu = {k: v for k, v in tabu.items() if v > it - tabu_len}
    return best_h, best_c * 2.0 / n

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 6.0
    deadline = start_time + budget_s - safety

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n_best = len(best_h)

    def tl():
        return deadline - time.time()

    # Phase 1: Binary SA (35% of time)
    if tl() > 30:
        dl1 = min(deadline, start_time + budget_s * 0.35)
        h_b, c_b = sa_binary(best_h, start_time, dl1, num_restarts=10)
        if c_b < best_c:
            best_h = h_b
            best_c = c_b

    # Phase 2: Binary 2-swap SA (20% of time)
    if tl() > 20:
        dl2 = min(deadline, start_time + budget_s * 0.55)
        h_b2, c_b2 = sa_binary_2swap(best_h, start_time, dl2, num_restarts=5)
        if c_b2 < best_c:
            best_h = h_b2
            best_c = c_b2

    # Phase 3: Tabu search (15% of time)
    if tl() > 15:
        dl3 = min(deadline, start_time + budget_s * 0.70)
        h_t, c_t = tabu_search(best_h, start_time, dl3, tabu_len=40, max_iters=100000)
        if c_t < best_c:
            best_h = h_t
            best_c = c_t

    # Phase 4: LP trust region on continuous (15% of time)
    if tl() > 15:
        dl4 = min(deadline, start_time + budget_s * 0.85)
        h_lp, c_lp = lp_trust_region(best_h, dl4, max_iters=300)
        if c_lp < best_c:
            best_h = h_lp
            best_c = c_lp

    # Phase 5: Continuous gradient polish (10% of time)
    if tl() > 8:
        dl5 = min(deadline, start_time + budget_s * 0.95)
        h_g, c_g = continuous_refine(best_h, dl5, max_iters=5000)
        if c_g < best_c:
            best_h = h_g
            best_c = c_g

    # Phase 6: Final binary SA polish from best
    if tl() > 5:
        dl6 = deadline - 1
        h_f, c_f = sa_binary(best_h, time.time(), dl6, num_restarts=3)
        if c_f < best_c:
            best_h = h_f
            best_c = c_f

    # Final: also try continuous from best binary
    if tl() > 3:
        h_final = project_sum_box(best_h, len(best_h) / 2.0)
        c_final = evaluate_c5(h_final)
        if c_final < best_c:
            best_h = h_final
            best_c = c_final

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
