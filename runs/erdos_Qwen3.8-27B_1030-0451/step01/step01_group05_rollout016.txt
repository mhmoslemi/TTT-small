def project_feasible(z, n):
    """Project z onto {x: 0<=x_i<=1, sum(x)=n/2}."""
    target = n / 2.0
    lo = np.min(z) - 1.0
    hi = np.max(z) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    h = np.clip(z - mu, 0.0, 1.0)

    diff = target - np.sum(h)
    if abs(diff) > 1e-13:

        if cnt > 0:

            h = np.clip(h, 0.0, 1.0)

    """Gradient of lag k (signed) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        g[:n-k] += (1.0 - h[k:])
        g[k:] -= h[:n-k]
    else:
        m = -k
        g[m:] += (1.0 - h[:n-m])
        g[:n-m] -= h[m:]

    g = np.zeros(n)
    for K in sorted_idx:
        k = K - (n - 1)
        g += grad_lag(h, k)
    g /= top_k
    return g, c_max

                best_h = h.copy()
            else:
                h = best_h.copy()
                break
    
    return best_h, best_s

def make_structured(n, rng, style=0):
    """Create structured initial patterns."""
    if style == 0:
        # Sinusoidal
        freq = rng.integers(2, 10)
        amp = 0.25 + 0.25 * rng.random()
        x = (np.arange(n) + 0.5) / n
        h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x + rng.random() * 2 * np.pi)
    elif style == 1:
        # Block pattern
        n_blocks = int(rng.integers(3, 12))
        h = np.zeros(n)
        bs = n // n_blocks
        for i in range(n_blocks):
            if rng.random() < 0.5:
                h[i * bs:min((i + 1) * bs, n)] = 1.0
    elif style == 2:
        # Two-frequency sinusoidal
        f1, f2 = int(rng.integers(2, 6)), int(rng.integers(6, 14))
        x = (np.arange(n) + 0.5) / n
        h = 0.5 + 0.25 * np.sin(2 * np.pi * f1 * x) + 0.15 * np.sin(2 * np.pi * f2 * x)
    elif style == 3:
        # Random smooth
        coarse = rng.random(max(4, n // 8))
        h = np.interp(np.arange(n), np.arange(len(coarse)), coarse)
    elif style == 4:
        # Alternating with noise
        h = 0.5 + 0.4 * np.sign(np.arange(n) % 2 - 0.5)
        h += 0.15 * (rng.random(n) - 0.5)
    else:
        h = 0.5 + 0.4 * (rng.random(n) - 0.5)
    
    return project_feasible(h, n)

def optimize_at_dim(h0, n, deadline, rng):
    """Full optimization pipeline at a given dimension."""
    h = h0.copy()
    cur_s = score(h)
    
    if time.time() > deadline:
        return h, cur_s
    
    # Phase A: Gradient direction steps
    tau = 0.05
    for round_i in range(8):
        if time.time() > deadline - 20:
            break
        h_new, s_new = gradient_direction_step(h, tau, n, min(deadline, time.time() + 60))
        if s_new < cur_s - 1e-12:
            h = h_new
            cur_s = s_new
        tau *= 0.6
        if tau < 0.003:
            tau = 0.003
    
    # Phase B: Pair refinement
    if time.time() < deadline - 30:
        max_it = min(50000, int((deadline - time.time()) * 20))
        h, cur_s = pair_refine(h, n, min(deadline, time.time() + max_it / 8), rng, max_it)
    
    # Phase C: Block moves
    if time.time() < deadline - 20:
        for nb in [4, 6, 8, 12]:
            if time.time() > deadline:
                break
            h_new, s_new = block_move(h, n, rng, nb)
            if s_new < cur_s - 1e-12:
                h = h_new
                cur_s = s_new
    
    # Phase D: More pair refinement
    if time.time() < deadline - 15:
        max_it = min(30000, int((deadline - time.time()) * 25))
        h, cur_s = pair_refine(h, n, deadline, rng, max_it)
    
    # Phase E: Final gradient polish
    if time.time() < deadline - 5:
        h_new, s_new = gradient_direction_step(h, 0.005, n, deadline)
        if s_new < cur_s - 1e-12:
            h = h_new
            cur_s = s_new
    
    return h, cur_s

def interpolate_to_dim(h_src, n_new):
    n_src = len(h_src)
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_src, h_src)
    return project_feasible(h_new, n_new)

# Global deadline for block_move (set in run)
deadline_global = 999999.0

def run(seed=42, budget_s=1100, **kwargs):
    global deadline_global
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline_global = t0 + budget_s - 25.0
    deadline = deadline_global
    
    # Get initial construction
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None
    
    if parent is not None and len(parent) > 1 and np.all(np.isfinite(parent)):
        parent = project_feasible(parent, len(parent))
        n_p = len(parent)
    else:
        n_p = 81
        parent = project_feasible(np.full(n_p, 0.5), n_p)
    
    best_h = parent.copy()
    best_s = score(parent)
    best_n = n_p
    
    # Track multi-dimensional results
    results = [(parent.copy(), best_s, n_p)]
    
    # Phase 1: Optimize at parent dimension
    h_opt, s_opt = optimize_at_dim(parent.copy(), n_p, min(deadline, t0 + 300), rng)
    if s_opt < best_s:
        best_h, best_s, best_n = h_opt.copy(), s_opt, n_p
        results.append((h_opt.copy(), s_opt, n_p))
    else:
        results.append((parent.copy(), best_s, n_p))
    
    # Phase 2: Dimension sweep
    dim_candidates = []
    for factor in [0.6, 0.7, 0.8, 0.9, 1.1, 1.2, 1.3, 1.5, 1.75, 2.0]:
        dn = int(round(n_p * factor))
        if 32 <= dn <= 500 and dn != n_p:
            dim_candidates.append(dn)
    # Add some absolute values
    for dn in [48, 64, 80, 96, 112, 128, 160, 192, 240]:
        if 32 <= dn <= 500 and dn != n_p:
            dim_candidates.append(dn)
    dim_candidates = sorted(set(dim_candidates))
    
    # Sort by proximity to best_n
    dim_candidates.sort(key=lambda x: abs(x - best_n))
    
    time_per_dim = max(30, min(120, (deadline - time.time()) / max(1, len(dim_candidates))))
    
    for dn in dim_candidates:
        if time.time() > deadline - 40:
            break
        dim_deadline = min(deadline - 20, time.time() + time_per_dim)
        
        h_interp = interpolate_to_dim(best_h, dn)
        s_interp = score(h_interp)
        
        # Only optimize if promising
        if s_interp > best_s + 0.02:
            continue
        
        h_opt, s_opt = optimize_at_dim(h_interp, dn, dim_deadline, rng)
        if s_opt < best_s:
            best_h, best_s, best_n = h_opt.copy(), s_opt, dn
            results.append((h_opt.copy(), s_opt, dn))
    
    # Phase 3: Multi-start at best dimension
    if time.time() < deadline - 60:
        for start_i in range(6):
            if time.time() > deadline - 30:
                break
            style = start_i % 5
            h_init = make_structured(best_n, rng, style)
            s_init = score(h_init)
            if s_init > best_s + 0.08:
                continue
            sub_deadline = min(deadline - 20, time.time() + max(40, (deadline - time.time()) / 3))
            h_opt, s_opt = optimize_at_dim(h_init, best_n, sub_deadline, rng)
            if s_opt < best_s:
                best_h, best_s, best_n = h_opt.copy(), s_opt, best_n
                results.append((h_opt.copy(), s_opt, best_n))
    
    # Phase 4: Final intensive polish at best
    if time.time() < deadline - 15:
        h_final, s_final = pair_refine(best_h.copy(), best_n, deadline, rng, 30000)
        if s_final < best_s:
            best_h, best_s = h_final, s_final
    
    # If we found a better solution at a different dimension, re-check
    # by interpolating back and forth
    if time.time() < deadline - 10 and len(results) > 1:
        # Try going to slightly different dimensions of the best
        for factor in [0.95, 1.05, 1.1, 0.9]:
            if time.time() > deadline - 5:
                break
            dn = int(round(best_n * factor))
            if dn == best_n or dn < 32 or dn > 500:
                continue
            h_try = interpolate_to_dim(best_h, dn)
            s_try = score(h_try)
            if s_try < best_s + 0.005:
                h_try, s_try = pair_refine(h_try, dn, deadline, rng, 5000)
                if s_try < best_s:
                    best_h, best_s, best_n = h_try, s_try, dn
    
    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_s = score(best_h)
    
    return best_h, float(final_s), int(best_n)
