def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))

def project_feasible(y, target=None):
    n = len(y)
    if target is None:
        target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=1.0, neginf=0.0)
    lo = -2.0
    hi = 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    npar = len(hp)
    if nn == npar:
        return project_feasible(hp)
    xo = np.linspace(0.0, 2.0, npar, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def compute_grad_active(h, active_idx, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for idx, w in zip(active_idx, weights):
        if abs(w) < 1e-15:
            continue
        r = idx - (n - 1)
        n_lo = max(0, -r)
        n_hi = min(n - 1, n - 1 - r)
        # Term 1: (1-h)[i-r], i in [max(0, r, n_lo+r), min(n-1, n-1+r, n_hi+r)]
        lo1 = max(0, r, n_lo + r)
        hi1 = min(n - 1, n - 1 + r, n_hi + r)
        if lo1 <= hi1:
            grad[lo1:hi1+1] += w * dx * (1.0 - h[lo1 - r:hi1 - r + 1])
        # Term 2: -h[i+r], i in [max(n_lo, -r), min(n_hi, n-1-r)]
        lo2 = max(n_lo, -r)
        hi2 = min(n_hi, n - 1 - r)
        if lo2 <= hi2:
            grad[lo2:hi2+1] -= w * dx * h[lo2 + r:hi2 + r + 1]
    return grad

def line_search(h0, d, dx, t_max):

        h_try = project_feasible(h0 + t * d, target)

        if f < best_f:

            best_t = t

        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    npar = len(hp)
    if nn == npar:
        return project_feasible(hp)
    xo = np.linspace(0.0, 2.0, npar, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def compute_grad_active(h, active_idx, weights, dx):
    n = len(h)
    grad = np.zeros(n)
    for idx, w in zip(active_idx, weights):
        if abs(w) < 1e-15:
            continue
        r = idx - (n - 1)
        n_lo = max(0, -r)
        n_hi = min(n - 1, n - 1 - r)
        lo1 = max(0, r, n_lo + r)
        hi1 = min(n - 1, n - 1 + r, n_hi + r)
        if lo1 <= hi1:
            grad[lo1:hi1 + 1] += w * dx * (1.0 - h[lo1 - r:hi1 - r + 1])
        lo2 = max(n_lo, -r)
        hi2 = min(n_hi, n - 1 - r)
        if lo2 <= hi2:
            grad[lo2:hi2 + 1] -= w * dx * h[lo2 + r:hi2 + r + 1]
    return grad

def do_line_search(h0, d, target):
    n = len(h0)
    pos = d > 1e-10
    neg = d < -1e-10
    t_cap = 1.0
    if pos.any():
        t_cap = min(t_cap, (1.0 - h0[pos]).min() / d[pos].max())
    if neg.any():
        t_cap = min(t_cap, (-h0[neg]).min() / (-d[neg]).max())
    t_cap = max(1e-8, t_cap * 1.05)
    t_cap = min(t_cap, 0.8)
    f0 = evaluate_c5(h0)
    n_pts = 14
    t_grid = np.linspace(0.0, t_cap, n_pts)
    best_t = 0.0
    best_f = f0
    for t in t_grid[1:]:
        h_try = project_feasible(h0 + t * d, target)
        f = evaluate_c5(h_try)
        if f < best_f:
            best_f = f
            best_t = t
        if f > best_f + 0.02:
            break
    if best_t > 1e-10 and best_t < t_cap * 0.9:
        lo = max(0.0, best_t - t_cap / n_pts)
        hi = min(t_cap, best_t + t_cap / n_pts)
        gr = (np.sqrt(5.0) - 1.0) / 2.0
        a1 = lo + (1.0 - gr) * (hi - lo)
        b1 = lo + gr * (hi - lo)
        fa = evaluate_c5(project_feasible(h0 + a1 * d, target))
        fb = evaluate_c5(project_feasible(h0 + b1 * d, target))
        for _ in range(16):
            if fa < fb:
                hi = b1
                b1 = a1
                fb = fa
                a1 = lo + (1.0 - gr) * (hi - lo)
                fa = evaluate_c5(project_feasible(h0 + a1 * d, target))
            else:
                lo = a1
                a1 = b1
                fa = fb
                b1 = lo + gr * (hi - lo)
                fb = evaluate_c5(project_feasible(h0 + b1 * d, target))
        t_ref = (lo + hi) * 0.5
        f_ref = evaluate_c5(project_feasible(h0 + t_ref * d, target))
        if f_ref < best_f:
            best_f = f_ref
            best_t = t_ref
    return best_t, best_f

def active_lag_descent(h0, n, deadline, max_iter=400):
    dx = 2.0 / n
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h_curr = h0.copy()
    c5_curr = best_c5
    stall = 0
    consecutive_worst = 0

    for it in range(max_iter):
        if time.time() > deadline - 3:
            break

        c = np.correlate(h_curr, 1.0 - h_curr, 'full') * dx
        cmax = c.max()
        cmean = c.mean()
        spread = max(1e-8, cmax - cmean)

        tol = max(1e-5, 0.04 * spread)
        active_mask = c >= cmax - tol
        active_idx = np.where(active_mask)[0]
        if len(active_idx) > 20:
            order = np.argsort(c[active_idx])[::-1]
            active_idx = active_idx[order[:20]]
        if len(active_idx) == 0:
            active_idx = np.array([np.argmax(c)])

        c_act = c[active_idx]
        improved = False

        # Direction 1: uniform
        w1 = np.ones(len(active_idx)) / len(active_idx)
        # Direction 2: aggressive softmax
        tau2 = max(1e-6, 0.05 * spread)
        e2 = np.exp(np.clip((c_act - c_act.max()) / tau2, -500, 500))
        w2 = e2 / e2.sum()
        # Direction 3: single max lag
        w3 = np.zeros(len(active_idx))
        w3[np.argmax(c_act)] = 1.0
        # Direction 4: top half heavy
        half = max(1, len(active_idx) // 2)
        order4 = np.argsort(c_act)[::-1]
        w4 = np.zeros(len(active_idx))
        w4[order4[:half]] = 0.08 / half
        w4[order4[half:]] = 0.02 / max(1, len(active_idx) - half)
        # Direction 5: bottom half (balancing)
        order5 = np.argsort(c_act)
        w5 = np.zeros(len(active_idx))
        w5[order5[:half]] = 0.10 / half
        w5[order5[half:]] = 0.05 / max(1, len(active_idx) - half)
        # Direction 6: smooth softmax
        tau6 = max(1e-5, 0.5 * spread)
        e6 = np.exp((c_act - c_act.max()) / tau6)
        w6 = e6 / e6.sum()

        directions = [w1, w2, w3, w4, w5, w6]

        for w in directions:
            if time.time() > deadline - 3:
                break
            g = compute_grad_active(h_curr, active_idx, w, dx)
            d = -(g - g.mean())
            d_norm = np.max(np.abs(d))
            if d_norm < 1e-13:
                continue
            d = d / d_norm

            t_best, f_best = do_line_search(h_curr, d, target)

            if t_best > 1e-10 and f_best < c5_curr - 1e-11:
                h_curr = project_feasible(h_curr + t_best * d, target)
                c5_curr = f_best
                improved = True
                if c5_curr < best_c5 - 1e-12:
                    best_c5 = c5_curr
                    best_h = h_curr.copy()
                    stall = 0
                    consecutive_worst = 0
                else:
                    stall += 1
            else:
                stall += 1
                consecutive_worst += 1

        if improved:
            stall = max(0, stall - 2)

        if stall > 10:
            rstall = int(stall // 5)
            rng_esc = np.random.default_rng((n * 1000 + it + rstall) & 0xFFFFFFFF)
            amp = 0.004 + 0.003 * min(stall // 5, 15)
            h_curr = project_feasible(h_curr + rng_esc.standard_normal(n) * amp, target)
            c5_curr = evaluate_c5(h_curr)
            stall = 0
            consecutive_worst = 0

        if stall > 60:
            break

    return best_h, best_c5

def fourier_seed(n, rng):
    h = np.full(n, 0.5)
    n_modes = 5
    for k in range(1, n_modes + 1):
        amp = rng.uniform(0.02, 0.08) / k
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * k * np.arange(n) / n + phase)
    return project_feasible(h, n / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)

    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)

    best_h = parent.copy()
    best_c5 = evaluate_c5(parent)
    best_n = n_par

    rng = np.random.default_rng(seed)

    # Phase 1: Optimize at parent resolution
    t_phase1 = time.time()
    budget_p1 = min(budget_s * 0.35, 380)
    dl1 = t0 + budget_p1

    h_opt, c5_opt = active_lag_descent(parent, n_par, dl1, max_iter=600)
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt
        best_n = n_par

    # Try a few perturbed starts at parent resolution
    if time.time() < dl1 - 20:
        for trial in range(4):
            if time.time() > dl1 - 5:
                break
            amp = 0.015 * (trial + 1)
            h_start = project_feasible(parent + rng.standard_normal(n_par) * amp, n_par / 2.0)
            h_opt, c5_opt = active_lag_descent(h_start, n_par, dl1, max_iter=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_par

    # Phase 2: Resolution ladder
    n_list = []
    for mult in [2, 3, 4, 5, 6, 8, 10]:
        n = n_par * mult
        if n <= 2400:
            n_list.append(n)

    for n in n_list:
        if time.time() > deadline - 20:
            break
        remaining = deadline - time.time()
        if remaining < 20:
            break

        # Time allocation: smaller n gets more time
        frac = (n_par * n_par) / (n * n)
        budget_this = remaining * min(frac * 2.5, 0.4)
        budget_this = max(budget_this, 15.0)
        dl_this = time.time() + budget_this

        h0 = interp_h(best_h, n)
        h_best_n = h0.copy()
        c5_best_n = evaluate_c5(h0)

        # Main optimization
        h_opt, c5_opt = active_lag_descent(h0, n, dl_this, max_iter=150)
        if c5_opt < c5_best_n:
            c5_best_n = c5_opt
            h_best_n = h_opt

        # One perturbed start if time allows
        if time.time() < dl_this - 10:
            h_start = project_feasible(h0 + rng.standard_normal(n) * 0.02, n / 2.0)
            h_opt, c5_opt = active_lag_descent(h_start, n, dl_this, max_iter=80)
            if c5_opt < c5_best_n:
                c5_best_n = c5_opt
                h_best_n = h_opt

        if c5_best_n < best_c5:
            best_c5 = c5_best_n
            best_h = h_best_n
            best_n = n

    # Phase 3: Polish at best resolution with more iterations
    if time.time() < deadline - 30:
        n_pol = best_n
        h_start = best_h.copy()
        budget_pol = deadline - time.time() - 10
        dl_pol = time.time() + budget_pol
        h_opt, c5_opt = active_lag_descent(h_start, n_pol, dl_pol, max_iter=300)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Try Fourier-based seeds at a couple resolutions
    if time.time() < deadline - 40:
        for n_try in [n_par, n_par * 2]:
            if time.time() > deadline - 15:
                break
            h_four = fourier_seed(n_try, rng)
            dl_try = time.time() + min(40, deadline - time.time() - 5)
            h_opt, c5_opt = active_lag_descent(h_four, n_try, dl_try, max_iter=200)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Final: ensure feasibility
    best_n = len(best_h)
    target = best_n / 2.0
    best_h = project_feasible(best_h, target)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] += diff / c
            np.clip(best_h, 0.0, 1.0, out=best_h)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
