    Qm = np.max(Qs)
    ev = np.exp(Qs - Qm)
    Z = np.sum(ev)
    return np.max(Q) + np.log(Z) / eta

def smooth_max_grad(h, eta):
    n = len(h)
    dx = 2.0 / n
    Q = corr_array(h)
    base = n - 1
    Qs = eta * Q
    Qm = np.max(Qs)
    ev = np.exp(Qs - Qm)
    Z = np.sum(ev)
    w = ev / Z
    grad = np.zeros(n)
    if w[base] > 1e-15:
        grad += w[base] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[base + d]
        wn = w[base - d]
        if wp > 1e-15:
            grad[d:] += wp * dx * (1.0 - h[:n - d])
            grad[:n - d] -= wp * dx * h[d:]
        if wn > 1e-15:
            grad[:n - d] += wn * dx * (1.0 - h[d:])
            grad[d:] -= wn * dx * h[:n - d]
    return grad

def slsqp_optimize(h0, eta, target, max_iter, timeout):
    n = len(h0)
    t0 = time.time()
    
    def obj_and_grad(x):
        if time.time() - t0 > timeout:
            return smooth_max_obj(x, eta), np.zeros_like(x)
        return smooth_max_obj(x, eta), smooth_max_grad(x, eta)
    
    def obj_and_grad_wrapper(x):
        return smooth_max_obj(x, eta), smooth_max_grad(x, eta)
    
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target,
             'jac': lambda x: np.ones_like(x)}]
    bounds = [(0.0, 1.0)] * n
    
    res = minimize(obj_and_grad_wrapper, h0, jac=True, method='SLSQP',
                   bounds=bounds, constraints=cons,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    
    if res.success and np.all(res.x >= -1e-12) and np.all(res.x <= 1 + 1e-12):
        h_new = project_box_sum(res.x, target)
        return h_new
    return None

def lp_polish(h, trust, target, time_limit):
    n = len(h)
    dx = 2.0 / n
    base = n - 1
    Q = corr_array(h)
    m = 2 * n - 1
    t0 = time.time()
    
    for iteration in range(50):
        if time.time() - t0 > time_limit:
            break
        cur_max = np.max(Q)
        k_act = max(5, int(m * 0.4))
        top_idx = np.argpartition(Q, -k_act)[-k_act:]
        
        # Build gradient matrix for active constraints
        A = np.zeros((k_act, n))
        for ii, idx in enumerate(top_idx):
            k = idx - base
            if k == 0:
                A[ii] = dx * (1.0 - 2.0 * h)
            elif k > 0:
                A[ii, k:] += dx * (1.0 - h[:n - k])
                A[ii, :n - k] -= dx * h[k:]
            else:
                d = -k
                A[ii, :n - d] += dx * (1.0 - h[d:])
                A[ii, d:] -= dx * h[:n - d]
        
        b = -Q[top_idx]
        
        nvars = n
        c_obj = np.zeros(nvars)
        # Minimize sum of absolute changes (L1 proxy via two variables is complex, use L2)
        # Actually just minimize t where A*d <= b, ||d|| <= trust
        # Use: minimize 0 subject to A*d <= b, -trust <= d_i <= trust, sum(d)=0
        A_ub = np.vstack([A, np.eye(n), -np.eye(n)])
        b_ub = np.concatenate([b, trust * np.ones(n), trust * np.ones(n)])
        A_eq = np.array([[1.0] * n])
        b_eq = np.array([0.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          method='highs', options={'time_limit': 5.0})
            if res.status == 0:
                d = res.x
                if np.max(np.abs(d)) > 1e-10:
                    h_new = project_box_sum(h + d, target)
                    Q_new = corr_array(h_new)
                    if np.max(Q_new) < cur_max - 1e-15:
                        h = h_new
                        Q = Q_new
                        continue
        except Exception:
            pass
        break
    return h

def subgrad_step(h, eta, target):
    n = len(h)
    Q = corr_array(h)
    base = n - 1
    Qs = eta * Q
    Qm = np.max(Qs)
    ev = np.exp(Qs - Qm)
    w = ev / np.sum(ev)
    grad = np.zeros(n)
    dx = 2.0 / n
    if w[base] > 1e-15:
        grad += w[base] * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = w[base + d]
        wn = w[base - d]
        if wp > 1e-15:
            grad[d:] += wp * dx * (1.0 - h[:n - d])
            grad[:n - d] -= wp * dx * h[d:]
        if wn > 1e-15:
            grad[:n - d] += wn * dx * (1.0 - h[d:])
            grad[d:] -= wn * dx * h[:n - d]
    return -grad

def make_candidates(h, rng, target):
    """Generate diverse candidates from h."""
    n = len(h)
    cands = [h.copy()]
    cands.append(project_box_sum(h[::-1].copy(), target))
    cands.append(project_box_sum(1.0 - h, target))
    cands.append(project_box_sum(h + 0.05 * rng.standard_normal(n), target))
    cands.append(project_box_sum(h + 0.1 * rng.standard_normal(n), target))
    cands.append(project_box_sum(0.7 * h + 0.3 * (h > np.median(h)).astype(float), target))
    # Shifted
    cands.append(project_box_sum(np.roll(h, 1), target))
    cands.append(project_box_sum(np.roll(h, -1), target))
    return cands

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 5.0
    rng = np.random.default_rng(seed)
    
    def remaining():
        return deadline - time.time()
    
    # Initialize from parent
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).flatten()
    except Exception:
        h0 = None
    
    if h0 is None or len(h0) == 0:
        h0 = np.full(120, 0.5)
    
    n0 = len(h0)
    best_h = project_box_sum(h0, n0 / 2.0)
    best_val = true_c5(best_h)
    
    # Phase 1: SLSQP at native resolution with multiple eta values
    eta_list = [30, 80, 150, 300, 600]
    for eta in eta_list:
        if remaining() < 30:
            break
        # Start from best
        h_start = best_h.copy()
        h_res = slsqp_optimize(h_start, eta, n0 / 2.0, max_iter=300, timeout=remaining() - 10)
        if h_res is not None:
            v = true_c5(h_res)
            if v < best_val:
                best_val = v
                best_h = h_res.copy()
        
        # Random restart
        if remaining() < 20:
            break
        h_rando = project_box_sum(best_h + 0.08 * rng.standard_normal(n0), n0 / 2.0)
        h_res2 = slsqp_optimize(h_rando, eta, n0 / 2.0, max_iter=200, timeout=remaining() - 10)
        if h_res2 is not None:
            v2 = true_c5(h_res2)
            if v2 < best_val:
                best_val = v2
                best_h = h_res2.copy()
    
    # Phase 2: Try diverse candidates at native resolution
    if remaining() > 60:
        cands = make_candidates(best_h, rng, n0 / 2.0)
        for cand in cands:
            if remaining() < 20:
                break
            for eta in [100, 300]:
                if remaining() < 15:
                    break
                h_res = slsqp_optimize(cand, eta, n0 / 2.0, max_iter=150, timeout=max(5, remaining() - 10))
                if h_res is not None:
                    v = true_c5(h_res)
                    if v < best_val:
                        best_val = v
                        best_h = h_res.copy()
    
    # Phase 3: Upscale to 2x
    if remaining() > 60:
        h_up = np.repeat(best_h, 2)
        n_up = len(h_up)
        h_up = project_box_sum(h_up, n_up / 2.0)
        v_up = true_c5(h_up)
        
        for eta in [50, 150, 400]:
            if remaining() < 30:
                break
            h_res = slsqp_optimize(h_up, eta, n_up / 2.0, max_iter=200, timeout=remaining() - 10)
            if h_res is not None:
                v = true_c5(h_res)
                if v < best_val:
                    best_val = v
                    best_h = h_res.copy()
        
        # Also try direct optimization from repeated version with perturbation
        if remaining() > 30:
            h_pert = project_box_sum(h_up + 0.03 * rng.standard_normal(n_up), n_up / 2.0)
            h_res = slsqp_optimize(h_pert, 200, n_up / 2.0, max_iter=200, timeout=remaining() - 10)
            if h_res is not None:
                v = true_c5(h_res)
                if v < best_val:
                    best_val = v
                    best_h = h_res.copy()
        
        # Downscale back and check
        if len(best_h) == n_up:
            h_dn = best_h.reshape(n0, 2).mean(axis=1)
            h_dn = project_box_sum(h_dn, n0 / 2.0)
            v_dn = true_c5(h_dn)
            if v_dn < best_val:
                best_val = v_dn
                best_h = h_dn.copy()
    
    # Phase 4: LP-based polish on best solution
    if remaining() > 30:
        n_best = len(best_h)
        h_lp = lp_polish(best_h, 0.15, n_best / 2.0, time_limit=remaining() - 10)
        v_lp = true_c5(h_lp)
        if v_lp < best_val:
            best_val = v_lp
            best_h = h_lp.copy()
    
    # Phase 5: More SLSQP at 2x with different starts
    if remaining() > 40:
        h_up2 = np.repeat(best_h, 2)
        n_up2 = len(h_up2)
        h_up2 = project_box_sum(h_up2, n_up2 / 2.0)
        
        for eta in [200, 500]:
            if remaining() < 25:
                break
            h_res = slsqp_optimize(h_up2, eta, n_up2 / 2.0, max_iter=250, timeout=remaining() - 10)
            if h_res is not None:
                v = true_c5(h_res)
                if v < best_val:
                    best_val = v
                    best_h = h_res.copy()
        
        # Try downscale
        if len(best_h) == n_up2:
            factor = n_up2 // len(h0)
            if factor > 1 and n_up2 % factor == 0:
                n_orig = n_up2 // factor
                h_dn = best_h.reshape(n_orig, factor).mean(axis=1)
                h_dn = project_box_sum(h_dn, n_orig / 2.0)
                v_dn = true_c5(h_dn)
                if v_dn < best_val:
                    best_val = v_dn
                    best_h = h_dn.copy()
    
    # Phase 6: Final LP polish
    if remaining() > 15:
        n_best = len(best_h)
        h_lp2 = lp_polish(best_h, 0.08, n_best / 2.0, time_limit=remaining() - 5)
        v_lp2 = true_c5(h_lp2)
        if v_lp2 < best_val:
            best_val = v_lp2
            best_h = h_lp2.copy()
    
    # Phase 7: Try 4x upscale if time allows
    if remaining() > 50 and len(best_h) <= 240:
        factor = max(1, min(4, 2400 // len(best_h)))
        if factor > 1:
            h_4x = np.repeat(best_h, factor)
            n_4x = len(h_4x)
            h_4x = project_box_sum(h_4x, n_4x / 2.0)
            for eta in [100, 300]:
                if remaining() < 25:
                    break
                h_res = slsqp_optimize(h_4x, eta, n_4x / 2.0, max_iter=150, timeout=remaining() - 10)
                if h_res is not None:
                    v = true_c5(h_res)
                    if v < best_val:
                        best_val = v
                        best_h = h_res.copy()
            # Downscale back
            if len(best_h) == n_4x and n_4x % factor == 0:
                n_orig = n_4x // factor
                h_dn = best_h.reshape(n_orig, factor).mean(axis=1)
                h_dn = project_box_sum(h_dn, n_orig / 2.0)
                v_dn = true_c5(h_dn)
                if v_dn < best_val:
                    best_val = v_dn
                    best_h = h_dn.copy()
    
    # Final feasibility check
    best_h = project_box_sum(best_h, len(best_h) / 2.0)
    best_val = true_c5(best_h)
    
    # Try external evaluator if available
    try:
        ext_val = evaluate_erdos_solution(best_h)
        if ext_val is not None and np.isfinite(ext_val):
            if ext_val < best_val:
                best_val = ext_val
    except Exception:
        pass
    
    return best_h, float(best_val), len(best_h)
