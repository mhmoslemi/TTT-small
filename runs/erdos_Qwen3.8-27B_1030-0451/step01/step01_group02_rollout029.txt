import numpy as np
import time

def score_exact(h):
    """Compute C5 = max of correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    C = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(C))

def subgradient_at(h, C, active_indices, dx):
    """Compute subgradient of max at active lags"""
    n = len(h)
    G = np.zeros(n)
    ctr = n - 1
    for k in active_indices:
        d = int(k) - ctr  # actual lag
        if d == 0:
            G += (1.0 - 2.0 * h) * dx
        elif d > 0:
            G[d:] += (1.0 - h[:-d]) * dx
            G[:-d] -= h[d:] * dx
        else:
            dd = -d
            G[:-dd] += (1.0 - h[dd:]) * dx
            G[dd:] -= h[:-dd] * dx
    return G

def project_feasible(x, target):
    """Project x onto {0 <= x[i] <= 1, sum(x) = target}"""
    x = np.clip(x, 0.0, 1.0)
    r = target - x.sum()
    if abs(r) < 1e-12:
        return x
    if r > 0:
        idx = np.where(x < 1.0)[0]
        if idx.size == 0:
            return x
        order = idx[np.argsort(x[idx])]
        rem = r
        for i in order:
            inc = min(1.0 - x[i], rem)
            x[i] += inc
            rem -= inc
            if rem <= 1e-14:
                break
    else:
        idx = np.where(x > 0.0)[0]
        if idx.size == 0:
            return x
        order = idx[np.argsort(-x[idx])]
        rem = -r
        for i in order:
            dec = min(x[i], rem)
            x[i] -= dec
            rem -= dec
            if rem <= 1e-14:
                break
    return np.clip(x, 0.0, 1.0)

def softmax_subgradient(h, tau):
    """Compute smooth subgradient using softmax over lags"""
    n = len(h)
    dx = 2.0 / float(n)
    C = np.correlate(h, 1.0 - h, 'full') * dx
    C_shifted = C - np.max(C)
    weights = np.exp(C_shifted / tau)
    weights /= weights.sum()
    G = np.zeros(n)
    ctr = n - 1
    for k in range(len(C)):
        w = weights[k]
        if w < 1e-12:
            continue
        d = k - ctr
        if d == 0:
            G += w * (1.0 - 2.0 * h) * dx
        elif d > 0:
            G[d:] += w * (1.0 - h[:-d]) * dx
            G[:-d] -= w * h[d:] * dx
        else:
            dd = -d
            G[:-dd] += w * (1.0 - h[dd:]) * dx
            G[dd:] -= w * h[:-dd] * dx
    return G, float(np.max(C))

def _make_feasible(x, target):
    x = np.clip(np.asarray(x, dtype=float).copy(), 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) < 1e-13:
        return x
    if r > 0:
        idx = np.where(x < 1.0)[0]
        if idx.size:
            for i in idx[np.argsort(x[idx])]:
                d = min(1.0 - x[i], r)
                x[i] += d
                r -= d
                if r <= 1e-14:
                    break
    else:
        idx = np.where(x > 0.0)[0]
        if idx.size:
            for i in idx[np.argsort(-x[idx])]:
                d = min(x[i], -r)
                x[i] -= d
                r += d
                if r >= -1e-14:
                    break
    return np.clip(x, 0.0, 1.0)

def _grad_at(h, m, d, n):
    """Gradient of C_d at coordinate m"""
    dx = 2.0 / float(n)
    if d == 0:
        return (1.0 - 2.0 * h[m]) * dx
    elif d > 0:
        g = 0.0
        if m >= d:
            g += (1.0 - h[m - d]) * dx
        if m <= n - 1 - d:
            g -= h[m + d] * dx
        return g
    else:
        p = -d
        g = 0.0
        if m <= n - 1 - p:
            g += (1.0 - h[m + p]) * dx
        if m >= p:
            g -= h[m - p] * dx
        return g

def _pair_solve(h, C, i, j, active):
    """Find best delta for h[i]+=delta, h[j]-=delta. Returns (delta, new_max) or None."""
    n = len(h)
    ctr = n - 1
    m = len(active)
    if m == 0:
        return None
    
    a_vals = np.empty(m)
    for idx, k in enumerate(active):
        d = k - ctr
        gi = _grad_at(h, i, d, n)
        gj = _grad_at(h, j, d, n)
        a_vals[idx] = gi - gj
    
    c_vals = C[active]
    dmin = max(-h[i], h[j] - 1.0) + 1e-12
    dmax = min(1.0 - h[i], h[j]) - 1e-12
    if dmin >= dmax:
        return None
    
    # Solve min_delta max_k(c_vals[k] + a_vals[k]*delta) for delta in [dmin, dmax]
    candidates = [dmin, dmax]
    for a in range(m):
        for b in range(a + 1, m):
            da = a_vals[a] - a_vals[b]
            if abs(da) < 1e-14:
                continue
            d = (c_vals[b] - c_vals[a]) / da
            if dmin <= d <= dmax:
                candidates.append(d)
    
    best_d = dmin
    best_v = float(np.max(c_vals + a_vals * dmin))
    for d in candidates:
        v = float(np.max(c_vals + a_vals * d))
        if v < best_v:
            best_v = v
            best_d = d
    
    if best_v >= float(np.max(C)) - 1e-12:
        return None
    
    return best_d, best_v
