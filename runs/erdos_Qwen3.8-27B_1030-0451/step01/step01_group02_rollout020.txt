def exact_C5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full") * dx
    return float(np.max(c))

def make_feasible(y, target):
    """Project y onto {x: 0<=x<=1, sum(x)=target}"""
    y = np.asarray(y, float).ravel()
    n = y.size
    target = float(target)
    
    # Bisection on the shift
    lo, hi = -max(np.abs(y).max(), 0), max(abs(y.max()-1), abs(1-y.min()), 0) + 1
    # More carefully:
    lo = -1.0 - float(np.max(np.abs(y)))
    hi = 1.0 + float(np.max(np.abs(y)))
    
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(y+mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    
    x = np.clip(y + 0.5*(lo+hi), 0, 1)
    # Fix residual
    r = target - x.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            # Add to positions with most room
            order = idx[np.argsort(x[idx])]
        else:
            idx = np.where(x > 0.0)[0]
            order = idx[np.argsort(x[idx])[::-1]]
        rem = abs(r)
        for i in order:
            if r > 0:
                add = min(1.0 - x[i], rem)
                x[i] += add
                rem -= add
            else:
                sub = min(x[i], rem)
                x[i] -= sub
                rem -= sub
            if rem < 1e-14:

    return np.clip(x, 0, 1)

    """Build the linearization of C[k] at h. Returns (row, offset) such that

    n = len(h)
    target = 0.5 * n
    d = _corr_dir
    maxC = float(np.max(C))

    active = np.where(C >= maxC - slack)[0]
    if active.size > max_constr:

        active = active[np.argsort(C[active])[::-1][:max_constr]]
    
    if active.size == 0:
        return None, None
    
    m = active.size
    # LP: min t
    # s.t. offset_i + row_i @ (x - h) <= t  for i in active
    #      0 <= x <= 1
    #      sum(x) = target
    # Variables: x (n) + t (1) = n+1
    # Constraint: row_i @ x - t <= offset_i - row_i @ h
    
    A_ub = np.zeros((m, n+1))
    b_ub = np.zeros(m)
    
    for ii, k in enumerate(active):
        row, offset = build_lp_constraint_row(k, h, d, n)
        A_ub[ii, :n] = row
        A_ub[ii, n] = -1.0
        b_ub[ii] = offset - float(row @ h)
