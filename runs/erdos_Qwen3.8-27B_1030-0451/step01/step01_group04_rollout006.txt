def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def line_search(h, a, N, dx, active_mask):
    """Find optimal step along direction a (zero-sum) for h."""
    # Compute A, L, Q
    A = np.correlate(h, 1.0 - h, 'full') * dx
    b = 1.0 - h
    L = np.correlate(a, b, 'full') * dx - np.correlate(h, a, 'full') * dx
    Q = -np.correlate(a, a, 'full') * dx
    
    # Feasible interval for delta
    delta_min = 0.0
    delta_max = 0.0
    for i in range(N):
        if a[i] > 1e-15:
            delta_min = max(delta_min, -h[i] / a[i])
            delta_max = min(delta_max, (1.0 - h[i]) / a[i])
        elif a[i] < -1e-15:
            delta_min = max(delta_min, (1.0 - h[i]) / a[i])
            delta_max = min(delta_max, -h[i] / a[i])
    
    if delta_max - delta_min < 1e-12:
        return 0.0, None
    
    delta_min += 1e-12
    delta_max -= 1e-12
    
    # Evaluate F(delta) = max(A + delta*L + delta^2*Q)
    # Sample at many points
    n_samples = 100
    deltas = np.linspace(delta_min, delta_max, n_samples)
    # F at each delta
    vals = A[None, :] + deltas[:, None] * L[None, :] + deltas[None, :]**2 * Q[None, :]
    F = vals.max(axis=1)
    
    best_idx = np.argmin(F)
    best_F = F[best_idx]
    best_delta = deltas[best_idx]
    
    # Refine with golden section around best region
    if best_idx > 0 and best_idx < n_samples - 1:
        lo = deltas[max(0, best_idx - 2)]
        hi = deltas[min(n_samples-1, best_idx + 2)]
        # Golden section
        gr = (np.sqrt(5) - 1) / 2
        a_gold = hi - gr * (hi - lo)
        b_gold = lo + gr * (hi - lo)
        
        def F_at(d):
            return float(np.max(A + d * L + d*d * Q))
        
        f_a = F_at(a_gold)
        f_b = F_at(b_gold)
        for _ in range(50):

                hi = b_gold
                b_gold = a_gold
                f_b = f_a

                f_a = F_at(a_gold)

                b_gold = lo + gr * (hi - lo)
                f_b = F_at(b_gold)

        # t1: (1-h[k+m]) for k in [0, N-1-m]
        if N - m > 0:
            g[:N-m] += 1.0 - h[m:]
        # t2: -h[k-m] for k in [m, N-1]
        if m < N:
            g[m:] -= h[:N-m]
    return g

Then the total gradient (subgradient) is:
G = sum_{r active} w_r * compute_gradient_r(h, d_r, N) * dx

And the zero-sum projection is: G_zero = G - mean(G)

The steepest descent direction in the feasible set would be -G_zero projected to the box constraints. But for simplicity, I can just use -G_zero as a direction (after zero-sum projection).

However, using the full gradient direction means all N components are nonzero, which makes the line search interval very tight (limited by the most constrained coordinate).

The pairwise direction (e_i - e_j) gives a wider feasible interval and is easier to optimize. So let me stick with the score-based pairwise approach.

                no_improve = 0

            # Occasionally try a larger perturbation
            if no_improve % 60 == 0 and no_improve > 0:
                scale = 0.05
                h_pert = h + rng.normal(0, scale, N)
                h_pert = np.clip(h_pert, 0.0, 1.0)
                h_pert = fix_sum(h_pert, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < best_c5 + 0.01:
                    h = h_pert
                else:
                    h = best_h.copy()
                no_improve = 0

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, N
