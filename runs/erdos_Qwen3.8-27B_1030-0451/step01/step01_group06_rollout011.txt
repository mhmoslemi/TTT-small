import numpy as np
import time
from scipy.optimize import minimize, linprog

def true_c5(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def corr_rows(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(x, target):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    x[~np.isfinite(x)] = 0.5
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target
        if abs(res) < 1e-12:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def lse_compute(h, T):
    n = len(h)
    corr = corr_rows(h)
    c_max = np.max(corr)
    shifted = T * (corr - c_max)
    shifted = np.clip(shifted, -500, 500)
    exp_s = np.exp(shifted)
    log_sum = np.log(np.sum(exp_s))
    obj = c_max + log_sum / T
    weights = exp_s / np.sum(exp_s)
    return obj, weights

def lse_objective(h, T):
    obj, _ = lse_compute(h, T)
    return obj

def lse_gradient(h, T):
    n = len(h)
    dx = 2.0 / n
    _, weights = lse_compute(h, T)
    base = n - 1
    grad = np.zeros(n)
    w0 = weights[base]
    if w0 > 1e-15:
        grad += w0 * dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        wp = weights[base + d]
        wn = weights[base - d]
        if wp < 1e-15 and wn < 1e-15:
            continue
        Lvec = h[:n - d]
        Rvec = h[d:]
        if wp > 1e-15:
            grad[d:] += wp * dx * (1.0 - Lvec)
            grad[:n - d] += -wp * dx * Rvec
        if wn > 1e-15:
            grad[:n - d] += wn * dx * (1.0 - Rvec)
            grad[d:] += -wn * dx * Lvec
    return grad

def sum_con_fun(h):
    return h.sum() - len(h) / 2.0

def sum_con_jac(h):
    return np.ones(len(h))

def run_slsqo(h0, T, maxiter):
    n = len(h0)
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(lse_objective, h0, args=(T,),
                       jac=lse_gradient,
                       method='SLSQP', bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': sum_con_fun, 'jac': sum_con_jac}],
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return res
    except Exception:
        return None

def compute_grads_array(h):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    grads = np.zeros((m, n))
    base = n - 1
    grads[base] = dx * (1.0 - 2.0 * h)
    for d in range(1, n):
        grads[base + d, d:] += dx * (1.0 - h[:n - d])
        grads[base + d, :n - d] += -dx * h[d:]
        grads[base - d, :n - d] += dx * (1.0 - h[d:])
        grads[base - d, d:] += -dx * h[:n - d]
    return grads

    n = len(h)
    m = 2 * n - 1
    k_act = max(1, int(m * active_frac))
    top_idx = np.argpartition(corr, -k_act)[-k_act:]
    corr_a = corr[top_idx]
    grads_a = grads[top_idx]
    nvars = n + 1
    c_obj = np.zeros(nvars)
    c_obj[n] = 1.0
    A_ub = np.zeros((k_act, nvars))
    A_ub[:, :n] = grads_a
    A_ub[:, n] = -1.0
    b_ub = -corr_a
    lower = np.maximum(-h, -trust)
    upper = np.minimum(1.0 - h, trust)
    A_eq = np.ones((1, n))

    bounds = list(zip(lower, upper)) + [(None, None)]
    try:

            d = res.x[:n]
            if np.all(h + d >= -1e-9) and np.all(h + d <= 1 + 1e-9):

    except Exception:
        pass
    return None
