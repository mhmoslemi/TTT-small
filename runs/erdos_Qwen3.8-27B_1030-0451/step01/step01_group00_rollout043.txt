import numpy as np
import time
from scipy.optimize import linprog

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def project_feasible(p, n):
    """Project p onto {x : 0<=x_i<=1, sum(x)=n/2}"""
    target = n / 2.0
    # Bisection on lambda for clip(p - lambda, 0, 1) to sum to target
    lo = float(np.min(p)) - 1.5
    hi = float(np.max(p)) + 1.5
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(p - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    # Fix residual
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-14:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
    return proj

def corr_gradient(h, dx):
    """Compute gradient of all correlations w.r.t. h. Returns array of shape (2n-1, n)."""
    n = len(h)
    G = np.zeros((2*n-1, n))
    for k in range(2*n-1):
        lag = k - (n-1)
        for j in range(n):
            g = 0.0
            # j appears as i in sum: need 0 <= j-lag < n
            if 0 <= j - lag < n:
                g += (1.0 - h[j-lag])
            # j appears as i-lag in sum: need 0 <= j+lag < n
            if 0 <= j + lag < n:
                g -= h[j+lag]
            G[k, j] = dx * g
    return G

def lp_direction(h, corr, radius, eta):
    """Solve LP to find descent direction."""
    n = len(h)
    dx = 2.0 / n
    f = float(np.max(corr) * dx)
    c_norm = corr * dx  # normalized correlations
    
    G = corr_gradient(h, dx)  # shape (2n-1, n)
    
    # LP: minimize eta*sum(|delta|) + t
    # Using split: delta = delta_pos - delta_neg
    # Variables: [delta_pos(n), delta_neg(n), t] = 2n+1 vars
    # Objective: eta*(sum(delta_pos) + sum(delta_neg)) + t
    # Constraints:
    #   G @ (delta_pos - delta_neg) - t <= f - c_norm  (2n-1 constraints)
    #   sum(delta_pos) - sum(delta_neg) = 0  (1 equality)
    #   0 <= delta_pos[i] <= ub[i]  where ub[i] = min(h[i]+radius, ... ) wait
    #   Actually: -lb[i] <= delta[i] <= ub[i]
    #   With delta = dp - dn: dp >= 0, dn >= 0
    #   dp[i] - dn[i] <= ub[i] => dp[i] <= ub[i] + dn[i]
    #   This is getting complicated. Let me use bounded variables directly.
    
    # Alternative: use bounded variables for delta directly
    # Variables: [delta(0..n-1), t] = n+1 vars
    # Objective: [eta, eta, ..., eta, 1] @ x
    # A_ub: G - [1,1,...,1,0] ... wait
    # Constraint: G[k,:] @ delta - t <= f - c_norm[k]
    # In matrix form: [G, -e] @ [delta; t] <= f - c_norm
    # where e = [0,...,0,1]
    
    c_obj = np.concatenate([eta * np.ones(n), [1.0]])
    
    A_ub = np.zeros((2*n-1, n+1))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0
    b_ub = f - c_norm
    
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = [0.0]
    
    lb = np.minimum(h, radius)
    ub = np.minimum(1.0 - h, radius)
    bounds = list(zip(-lb, ub)) + [(0, None)]
    
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.success and result.x is not None:
        delta = result.x[:n]
        t_val = result.x[n]
        return delta, t_val
    return None, None

def pair_exchange_improve(h, n_iter_per_pair=10, max_pairs=200, seed=0):
    """Try random pair exchanges to improve objective."""
    n = len(h)
    h = h.copy()
    c5 = compute_c5(h)
    rng = np.random.RandomState(seed)
    
    for it in range(max_pairs):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        # Transfer mass from i to j (if h[i] > h[j]) or j to i
        if h[i] >= h[j]:
            src, dst = i, j
        else:
            src, dst = j, i
        max_s = min(h[src], 1.0 - h[dst])
        if max_s < 1e-6:

        # Try several transfer amounts

            h_try[src] -= s
            h_try[dst] += s
            c5_try = compute_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_c5 = c5_try
                best_s = s
        if best_s > 0:
            h[src] -= best_s
            h[dst] += best_s
            c5 = best_c5
    
    return h, c5

def make_start(n, seed_val):
    """Create a diverse starting point."""
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    # Use a mix of frequencies
    h = np.zeros(n)
    freqs = np.array([1, 2, 3, 5, 8, 13])
    amplitudes = np.array([0.3, 0.2, 0.15, 0.1, 0.06, 0.04])

        h += a * np.sin(2.0 * np.pi * f * x + p)
    h = 0.5 + h
    h += rng.randn(n) * 0.03
    return project_feasible(h, n)

def resample(h_old, n_new):
    """Resample h from n_old to n_new using linear interpolation."""
    n_old = len(h_old)
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0/n_old  # midpoints
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0/n_new
    h_new = np.interp(x_new, x_old, h_old, period=2.0)
    return project_feasible(h_new, n_new)

        else:
            hi = mid

    if alloc > 15 and best_h is not None:
        h_opt, c5_opt = lp_optimize(best_h.copy(), alloc, seed + 999)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Phase 7: Final pair search
    alloc = min(remaining() - 15, 30)
    if alloc > 5 and best_h is not None:
        h_opt, c5_opt = pair_search(best_h.copy(), alloc, seed + 2000)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Final verification
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    
    return best_h, best_c5, len(best_h)
