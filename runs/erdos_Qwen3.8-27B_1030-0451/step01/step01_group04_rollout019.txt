    m = c.max()
    exps = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(exps)
    s = e.sum()
    w = e / s
    obj = m + tau * np.log(s)
    # Gradient
    g = np.zeros(N)
    for r in range(2*N-1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N-1)
        if d >= 0:
            n = N - d
            g[d:d+n] += a * (1.0 - h[:n])
            g[:n] -= a * h[d:d+n]
        else:
            dd = -d
            n = N - dd
            g[:n] += a * (1.0 - h[dd:dd+n])
            g[dd:dd+n] -= a * h[:n]
    return obj, g

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.5, 1.5
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-10:
        if residual > 0:
            idx = np.where(h < 1.0 - 1e-12)[0]
            for i in idx:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-10:
                    break
        else:
            idx = np.where(h > 1e-12)[0]
            for i in idx:
                sub = max(residual, -h[i])
                h[i] += sub
                residual -= sub
                if abs(residual) < 1e-10:
                    break
    return h

def fista(h0, target_sum, tau, max_iter=5000, step0=None):
    N = len(h0)
    if step0 is None:
        step0 = 100.0 / N
    x_prev = h0.copy()
    y = h0.copy()
    t_k = 1.0
    step = step0
    
    for it in range(max_iter):
        obj, grad = get_obj_grad(y, tau)
        
        step_try = step
        x_new = None
        for _ in range(30):
            cand = y - grad * step_try
            cand = project_feasible(cand, target_sum)
            obj_c, _ = get_obj_grad(cand, tau)
            rhs = obj + np.dot(grad, cand - y) + (0.5 / step_try) * np.sum((cand-y)**2)
            if obj_c <= rhs + 1e-9:
                x_new = cand
                break
            step_try *= 0.5
        if x_new is None:
            x_new = project_feasible(y - grad * step_try, target_sum)
        step = max(step_try * 1.3, 1e-12)
        
        t_new = 0.5 * (1.0 + np.sqrt(1.0 + 4.0*t_k*t_k))
        y = x_new + ((t_k - 1.0) / t_new) * (x_new - x_prev)
        y = np.clip(y, 0.0, 1.0)  # keep in box (loose projection)
        t_k = t_new
        x_prev = x_new.copy()
    
    return x_prev

def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)
