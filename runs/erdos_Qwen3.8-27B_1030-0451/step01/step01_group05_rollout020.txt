def golden_line_search(h, d, n, max_evals=200):
    """Find optimal t along h + t*d using golden section search on exact score."""
    # Find feasible interval
    t_lo, t_hi = 0.0, 0.0
    for i in range(n):
        di = d[i]
        if abs(di) < 1e-14:
            continue
        hi_i = (1.0 - h[i]) / di
        lo_i = -h[i] / di
        t_lo = max(t_lo, min(lo_i, hi_i))
        t_hi = min(t_hi, max(lo_i, hi_i))
    
    if t_hi - t_lo < 1e-12:
        return h, exact_score(h)
    
    # Add small safety margin
    margin = (t_hi - t_lo) * 1e-8
    t_lo += margin
    t_hi -= margin
    
    # Golden section
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = t_lo, t_hi
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    fc = exact_score(np.clip(h + c_pt * d, 0, 1))
    fd = exact_score(np.clip(h + d_pt * d, 0, 1))
    evals = 2
    
    for _ in range(max_evals - 2):
        if fc < fd:
            b = d_pt
            d_pt = c_pt
            fd = fc
            c_pt = b - gr * (b - a)
            fc = exact_score(np.clip(h + c_pt * d, 0, 1))
        else:
            a = c_pt
            c_pt = d_pt
            fc = fd
            d_pt = a + gr * (b - a)
            fd = exact_score(np.clip(h + d_pt * d, 0, 1))
        evals += 1
        if b - a < 1e-10:
            break
    
    t_best = (a + b) / 2.0
    h_new = np.clip(h + t_best * d, 0, 1)
    h_new = project_feasible(h_new, n)
    s = exact_score(h_new)
    return h_new, s

def project_feasible(z, n):

    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)

def fw_vertex(h, g, n):
    """Find Frank-Wolfe vertex: min <g, v> s.t. sum(v)=n/2, 0<=v<=1"""
    order = np.argsort(g)
    S = n / 2.0
    v = np.zeros(n)
    S_int = int(np.floor(S))
    S_frac = S - S_int
    for i in range(S_int):
        v[order[i]] = 1.0
    if S_frac > 1e-12:
        v[order[S_int]] = S_frac
    return v

    """Project d to zero-sum space"""
    d = d - np.mean(d)

import numpy as np

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        k = K - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] += w * (-h[k:])
        else:
            m = -k
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n - m])
    return g


def fw_vertex(g, n):

    v = np.zeros(n)

    S_frac = S - S_int
    for i in range(S_int):

    if S_frac > 1e-12:
        v[order[S_int]] = S_frac
    return v

def run_slsqp_polish(h, n, tau, maxiter, deadline):
    class SlObj:
        def __init__(self, tau_val):
            self.tau = tau_val
        def __call__(self, x):
            nn = len(x)
            ddx = 2.0 / nn
            raw = np.correlate(x, 1.0 - x, mode='full')
            c = raw * ddx
            c_max = np.max(c)
            exponents = np.clip((c - c_max) / self.tau, -500, 500)
            return c_max + self.tau * np.log(np.sum(np.exp(exponents)))

    def obj_func(x):
        return SlObj(tau)(x)

    def sum_fun(x):
        return np.sum(x) - n / 2.0

    cons = [{'type': 'eq', 'fun': sum_fun, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(
            fun=obj_func, x0=h.copy(), method='SLSQP',
            bounds=bounds, constraints=cons,
            options={'maxiter': maxiter, 'ftol': 1e-14, 'disp': False}
        )
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

        dim_factors = [0.85, 0.9, 0.95, 1.05, 1.1, 1.2, 1.4, 0.75, 1.6]
        tried_dims = set([best_n])
        for factor in dim_factors:
            if time.time() > deadline - 25:
                break
            cand_n = int(round(best_n * factor))
            if 25 <= cand_n <= 300 and cand_n not in tried_dims:
                tried_dims.add(cand_n)
                h_d = interpolate_h(best_h, cand_n)
                s_d = exact_score(h_d)
                if s_d < best_score + 0.01:
                    sub_dl = deadline - 20.0
                    time_alloc = max(10.0, (deadline - time.time()) * 0.15)
                    sub_dl = min(sub_dl, time.time() + time_alloc)
                    h_d, s_d = optimize_h(h_d, cand_n, rng, sub_dl, num_iters=30000)
                    # Quick SLSQP
                    if time.time() < deadline - 15:
                        h_new = run_slsqp_polish(h_d, cand_n, 0.01, 50, deadline)
                        if h_new is not None:
                            s_new = exact_score(h_new)
                            if s_new < s_d:
                                h_d = h_new
                                s_d = s_new
                    if s_d < best_score - 1e-12:
                        best_h = h_d.copy()
                        best_score = s_d
                        best_n = cand_n

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        remaining = deadline - time.time() - 3
        if remaining > 5:
            h_opt, s_opt = optimize_h(best_h, best_n, rng, deadline - 3.0,
                                       num_iters=int(remaining * 40))
            if s_opt < best_score - 1e-13:
                best_h = h_opt
                best_score = s_opt

    # Final SLSQP
    if time.time() < deadline - 3:
        h_new = run_slsqp_polish(best_h, best_n, 0.003, 50, deadline)
        if h_new is not None:
            s_new = exact_score(h_new)
            if s_new < best_score - 1e-12:
                best_h = h_new
                best_score = s_new

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-12:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
