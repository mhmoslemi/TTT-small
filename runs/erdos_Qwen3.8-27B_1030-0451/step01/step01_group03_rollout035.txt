def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target=None):
    y = np.asarray(y, dtype=np.float64)
    n = len(y)
    if target is None:
        target = n / 2.0
    # Bisection for shift
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def compute_grad_lags(h, lags, dx):
    """Compute gradients for specific correlation lags.
    lags: indices into the full correlation vector (0 to 2n-2)
    Returns list of gradient arrays.
    """
    n = len(h)
    grads = []
    for ell in lags:
        r = (n - 1) - ell
        g = np.zeros(n)
        # Term 1: j in [max(0,-r), min(n-1, n-1-r)]
        lo1 = max(0, -r)
        hi1 = min(n - 1, n - 1 - r)
        if lo1 <= hi1:
            g[lo1:hi1+1] += (1.0 - h[lo1+r:hi1+1+r])
        # Term 2: j in [max(0,r), min(n-1, n-1+r)]
        lo2 = max(0, r)
        hi2 = min(n - 1, n - 1 + r)
        if lo2 <= hi2:
            g[lo2:hi2+1] -= h[lo2-r:hi2+1-r]
        grads.append(g * dx)
    return grads

def active_lag_directions(h, dx, K=15, tolerance_frac=0.15):

    c = np.correlate(h, 1.0 - h, 'full') * dx

def line_search_exact(h, d, c5_current, max_iter=50):
    """Exact line search for C5(h + t*d) with feasibility projection.
    Returns (best_t, best_h, best_c5)."""
    n = len(h)
    target = n / 2.0
    
    # Find max safe t
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    t_max = 1.0
    if pos_mask.any():
        t_max = min(t_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    if neg_mask.any():
        t_max = min(t_max, np.min(h[neg_mask] / (-d[neg_mask])))
    t_max = max(0.0, t_max) * 1.1  # Allow slight overshoot
    
    # Golden section search on [0, t_max]
    gr = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, t_max
    
    if b <= a:
        return (0.0, h.copy(), c5_current)
    
    c_pt = b - gr * (b - a)
    d_pt = a + gr * (b - a)
    
    h_c = project_feasible(h + c_pt * d, target)
    h_d = project_feasible(h + d_pt * d, target)
    f_c = evaluate_c5(h_c)
    f_d = evaluate_c5(h_d)
    
    best_t = 0.0
    best_h = h.copy()
    best_c5 = c5_current
    
    for _ in range(max_iter):
        if f_c < f_d:
            b = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = b - gr * (b - a)
            h_c = project_feasible(h + c_pt * d, target)
            f_c = evaluate_c5(h_c)
        else:
            a = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a + gr * (b - a)
            h_d = project_feasible(h + d_pt * d, target)
            f_d = evaluate_c5(h_d)
        
        if f_c < best_c5:
            best_t = c_pt
            best_c5 = f_c
            best_h = h_c.copy()
        if f_d < best_c5:
            best_t = d_pt
            best_c5 = f_d
            best_h = h_d.copy()
        
        if abs(b - a) < 1e-6 * max(1.0, t_max):
            break
    
    return (best_t, best_h, best_c5)

    n = len(h)
    dx = 2.0 / n
    c5 = evaluate_c5(h)
    best_h = h.copy()

        else:

    return best_h, best_c5

def make_cyclic_shifts(h, n_shifts=5):

def make_fourier_variant(h, rng):
    """Create a Fourier-based perturbation of h."""
    n = len(h)
    hft = np.fft.rfft(h)
    # Perturb high frequencies less
    freqs = np.arange(len(hft))

    hft_pert = hft + noise

        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def compute_lag_grad(h, ell, dx):
    n = len(h)
    r = (n - 1) - ell
    g = np.zeros(n)
    lo1 = max(0, -r)
    hi1 = min(n - 1, n - 1 - r)
    if lo1 <= hi1:
        g[lo1:hi1 + 1] += (1.0 - h[lo1 + r:hi1 + 1 + r])
    lo2 = max(0, r)
    hi2 = min(n - 1, n - 1 + r)
    if lo2 <= hi2:
        g[lo2:hi2 + 1] -= h[lo2 - r:hi2 + 1 - r]
    return g * dx

def line_search_exact(h, d, c5_cur):
    n = len(h)
    target = n / 2.0
    pos_mask = d > 1e-14
    neg_mask = d < -1e-14
    t_max = 2.0
    if pos_mask.any():
        t_max = min(t_max, np.min((1.0 - h[pos_mask]) / d[pos_mask]))
    if neg_mask.any():
        t_max = min(t_max, np.min(h[neg_mask] / (-d[neg_mask])))
    t_max = max(0.0, t_max) * 1.05
    if t_max < 1e-10:
        return (0.0, h.copy(), c5_cur)
    gr = (np.sqrt(5.0) - 1.0) / (2.0 * np.sqrt(5.0) + 1.0)
    gr2 = (np.sqrt(5.0) - 1.0) / 2.0
    a, b = 0.0, t_max
    c_pt = b - gr2 * (b - a)
    d_pt = a + gr2 * (b - a)
    h_c = project_feasible(h + c_pt * d, target)
    h_d = project_feasible(h + d_pt * d, target)
    f_c = evaluate_c5(h_c)
    f_d = evaluate_c5(h_d)
    best_c5 = min(c5_cur, f_c, f_d)
    best_h = h.copy() if best_c5 == c5_cur else (h_c if f_c == best_c5 else h_d).copy()
    for _ in range(45):
        if f_c < f_d:
            b = d_pt
            d_pt = c_pt
            f_d = f_c
            c_pt = b - gr2 * (b - a)
            h_c = project_feasible(h + c_pt * d, target)
            f_c = evaluate_c5(h_c)
        else:
            a = c_pt
            c_pt = d_pt
            f_c = f_d
            d_pt = a + gr2 * (b - a)
            h_d = project_feasible(h + d_pt * d, target)
            f_d = evaluate_c5(h_d)
        if f_c < best_c5:
            best_c5 = f_c
            best_h = h_c.copy()
        if f_d < best_c5:
            best_c5 = f_d
            best_h = h_d.copy()
        if abs(b - a) < 1e-7 * max(1.0, t_max):
            break
    return (0.0, best_h, best_c5)

def get_active_directions(h, dx, K=12, tol_frac=0.12):
    n = len(h)
    c = np.correlate(h, 1.0 - h, 'full') * dx
    cmax = c.max()
    cmean = c.mean()
    tol = max(1e-6, tol_frac * max(1e-8, cmax - cmean))
    active = np.where(c >= cmax - tol)[0]
    if len(active) == 0:
        active = np.array([int(np.argmax(c))])
    if len(active) > K:
        order = np.argsort(c[active])[::-1][:K]
        active = active[order]
    active = np.sort(active)
    grads = []
    for ell in active:
        grads.append(compute_lag_grad(h, ell, dx))
    directions = []
    c_act = c[active]
    # Uniform
    g = sum(grads) / len(grads)
    d = -(g - g.mean())
    nm = np.max(np.abs(d))
    if nm > 1e-12:
        directions.append(d / nm)
    # Tight softmax
    tau = max(1e-6, 0.04 * max(1e-8, c_act.max() - c_act.min()))
    exps = np.exp(np.clip((c_act - c_act.max()) / tau, -400, 400))
    w = exps / exps.sum()
    g = np.zeros(n)
    for i in range(len(grads)):
        if w[i] > 1e-10:
            g += w[i] * grads[i]
    d = -(g - g.mean())
    nm = np.max(np.abs(d))
    if nm > 1e-12:
        directions.append(d / nm)
    # Loose softmax
    tau2 = max(1e-6, 0.2 * max(1e-8, c_act.max() - c_act.min()))
    exps2 = np.exp(np.clip((c_act - c_act.max()) / tau2, -400, 400))
    w2 = exps2 / exps2.sum()
    g = np.zeros(n)
    for i in range(len(grads)):
        if w2[i] > 1e-10:
            g += w2[i] * grads[i]
    d = -(g - g.mean())
    nm = np.max(np.abs(d))
    if nm > 1e-12:
        directions.append(d / nm)
    # Single max lag
    g = grads[0]
    d = -(g - g.mean())
    nm = np.max(np.abs(d))
    if nm > 1e-12:
        directions.append(d / nm)
    # Top-2 split
    if len(grads) >= 2:
        w3 = np.zeros(len(grads))
        w3[0] = 0.6
        w3[1] = 0.4
        g = np.zeros(n)
        for i in range(len(grads)):
            if w3[i] > 0:
                g += w3[i] * grads[i]
        d = -(g - g.mean())
        nm = np.max(np.abs(d))
        if nm > 1e-12:
            directions.append(d / nm)
    return directions

# Global state for coordinate descent
_CD_N = 0
_CD_DX = 0.0

def _cd_eval_t(h_other_sum_plus, h_j_old, t, A, B):
    return float(np.max(B + t * A))

def coord_descend_c5(h, deadline, max_sweeps=500):
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    c5 = evaluate_c5(h)
    best_c5 = c5
    best_h = h.copy()
    r_all = (n - 1) - np.arange(2 * n - 1)
    for sweep in range(max_sweeps):
        if time.time() > deadline - 2:
            break
        c = np.correlate(h, 1.0 - h, 'full') * dx
        c5_now = float(c.max())
        improved = False
        for j in range(n):
            valid1 = (j + r_all >= 0) & (j + r_all < n)
            valid2 = (j - r_all >= 0) & (j - r_all < n)
            A = np.zeros(2 * n - 1)
            A[valid1] = (1.0 - h[j + r_all[valid1]]) * dx
            A[valid2] += (-h[j - r_all[valid2]]) * dx
            B = c - h[j] * A
            # Pool adjustment: h'[j]=t, h'[i]=h[i]+(h[j]-t)/(n-1) for i!=j
            # c'[k] = B[k] + t*A[k] + (h[j]-t)/(n-1) * sum_{i!=j} A_i_k
            # where A_i_k is coefficient of h[i] in c[k]
            # This is complex; use simpler approach:
            # Just optimize t ignoring the pool effect (it's a small correction)
            # f(t) = max_k (B[k] + t * A[k])
            # Find min of this convex PL function on [0,1]
            # Use golden section
            gr = (np.sqrt(5.0) - 1.0) / 2.0
            lo_t, hi_t = 0.0, 1.0
            f1_t = lo_t + (1.0 - gr) * (hi_t - lo_t)
            f2_t = lo_t + gr * (hi_t - lo_t)
            f1 = float(np.max(B + f1_t * A))
            f2 = float(np.max(B + f2_t * A))
            for _ in range(30):
                if f1 < f2:
                    hi_t = f2_t
                    f2_t = f1_t
                    f2 = f1
                    f1_t = lo_t + (1.0 - gr) * (hi_t - lo_t)
                    f1 = float(np.max(B + f1_t * A))
                else:
                    lo_t = f1_t
                    f1_t = f2_t
                    f1 = f2
                    f2_t = lo_t + gr * (hi_t - lo_t)
                    f2 = float(np.max(B + f2_t * A))
                if hi_t - lo_t < 1e-8:
                    break
            t_opt = (lo_t + hi_t) * 0.5
            # Apply with pool adjustment
            delta = t_opt - h[j]
            if abs(delta) < 1e-10:
                continue
            h_new = h.copy()
            h_new[j] = t_opt
            if n > 1:
                h_new[:j] += -delta / (n - 1)
                h_new[j + 1:] += -delta / (n - 1)
            h_new = project_feasible(h_new, target)
            c5_new = evaluate_c5(h_new)
            if c5_new < c5_now - 1e-12:
                h = h_new
                c5_now = c5_new
                improved = True
                if c5_now < best_c5:
                    best_c5 = c5_now
                    best_h = h.copy()
        if not improved:
            break
    return best_h, best_c5

def active_optimize(h, deadline, max_rounds=300, K=12):
    n = len(h)
    dx = 2.0 / n
    c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = c5
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline - 2:
            break
        dirs = get_active_directions(h, dx, K=K, tol_frac=0.12)
        if not dirs:
            dirs = get_active_directions(h, dx, K=K + 6, tol_frac=0.4)
        improved = False
        for d in dirs:
            if time.time() > deadline - 2:
                break
            _, h_new, c5_new = line_search_exact(h, d, c5)
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()
                c5 = c5_new
                improved = True
            elif c5_new < c5 - 1e-12:
                h = h_new.copy()
                c5 = c5_new
                improved = True
        if improved:
            stall = 0
        else:
            stall += 1
            if stall >= 6:
                break
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.default_rng(seed)

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)

    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)

    # Build diverse candidates
    candidates = []
    candidates.append((parent.copy(), evaluate_c5(parent)))

    # Complement
    h_comp = project_feasible(1.0 - parent, n_par / 2.0)
    candidates.append((h_comp.copy(), evaluate_c5(h_comp)))

    # Cyclic shifts
    for s in [1, 2, n_par // 4, n_par // 3]:
        h_s = project_feasible(np.roll(parent, s), n_par / 2.0)
        candidates.append((h_s.copy(), evaluate_c5(h_s)))

    # Interpolated to larger n
    for mult in [2, 3, 4]:
        n_new = n_par * mult
        if n_new <= 2400:
            h_i = interp_h(parent, n_new)
            candidates.append((h_i.copy(), evaluate_c5(h_i)))

    # Sort by C5
    candidates.sort(key=lambda x: x[1])

    # Phase 1: Optimize best candidates with active-lag method
    best_h = candidates[0][0].copy()
    best_c5 = candidates[0][1]

    for idx in range(min(4, len(candidates))):
        if time.time() > deadline - 60:
            break
        h_try = candidates[idx][0].copy()
        if time.time() > deadline - 30:
            break
        h_opt, c5_opt = active_optimize(h_try, deadline, max_rounds=200, K=12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Phase 2: Coordinate descent polish
    if time.time() < deadline - 30:
        h_cd, c5_cd = coord_descend_c5(best_h, deadline, max_sweeps=300)
        if c5_cd < best_c5:
            best_c5 = c5_cd
            best_h = h_cd.copy()

    # Phase 3: Resolution refinement
    if time.time() < deadline - 50:
        best_n = len(best_h)
        n_next = min(best_n * 2, 2400)
        if n_next > best_n:
            h_fine = interp_h(best_h, n_next)
            c5_fine = evaluate_c5(h_fine)
            if c5_fine < best_c5 + 0.005:
                h_opt, c5_opt = active_optimize(h_fine, deadline, max_rounds=150, K=12)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                if time.time() < deadline - 20:
                    h_cd2, c5_cd2 = coord_descend_c5(best_h, deadline, max_sweeps=200)
                    if c5_cd2 < best_c5:
                        best_c5 = c5_cd2
                        best_h = h_cd2.copy()

    # Phase 4: Random restarts if time allows
    if time.time() < deadline - 60:
        for trial in range(6):
            if time.time() > deadline - 30:
                break
            amp = 0.01 + 0.02 * trial
            h_pert = project_feasible(best_h + rng.standard_normal(len(best_h)) * amp, len(best_h) / 2.0)
            h_opt, c5_opt = active_optimize(h_pert, deadline, max_rounds=80, K=10)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()

    # Phase 5: Final polish with narrow tolerance
    if time.time() < deadline - 10:
        h_opt, c5_opt = active_optimize(best_h, deadline, max_rounds=100, K=20)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()

    # Final feasibility
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
