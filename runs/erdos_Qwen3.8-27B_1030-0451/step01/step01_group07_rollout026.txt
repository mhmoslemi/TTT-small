def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']

def repair(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            room = out
            tr = room.sum()
            if tr > 0:
                out -= (d / tr) * room
        else:
            room = 1.0 - out
            tr = room.sum()
            if tr > 0:
                out += (-d / tr) * room
        out = np.clip(out, 0.0, 1.0)
    return out

    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new

def active_gradient(h, tol=0.0):
    """Compute gradient of C5 using active lag weighting."""
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    cmax = c.max()
    # Weight lags by excess over threshold
    thr = cmax - tol
    w = np.maximum(c - thr, 0.0)

    else:
        w = w / w.sum()
    
    grad = np.zeros(n)
    center = n - 1
    # lag 0: c[center] = sum h[i]*(1-h[i])
    # d/dh[i] = (1-h[i]) - h[i] = 1-2h[i]
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp > 1e-15:
            # lag +p: c[center+p] = sum_{i=0}^{n-1-p} h[i]*(1-h[i+p])
            # d/dh[i] for i < n-p: (1-h[i+p])
            # d/dh[i] for i >= p: -h[i-p]
            if n - p > 0:
                grad[:n-p] += wp * (1.0 - h[p:])
                grad[p:] -= wp * h[:n-p]
        if wm > 1e-15:
            # lag -p: c[center-p] = sum_{i=p}^{n-1} h[i]*(1-h[i-p])
            # d/dh[i] for i >= p: (1-h[i-p])
            # d/dh[i] for i < n-p: -h[i+p]
            if n - p > 0:
                grad[p:] += wm * (1.0 - h[:n-p])
                grad[:n-p] -= wm * h[p:]
    grad *= dx
    return grad

    """Evaluate C5(t) = max_m (c0[m] + a[m]*t^2 + b[m]*t)"""
    vals = c0 + a * t * t + b * t
    return vals.max()

def optimize_pair(h, i, j, c0, dx):
    """Find optimal t for pair (i,j). Returns (t_opt, c5_opt)."""
    n = len(h)
    # Compute a, b coefficients
    idx = np.arange(-(n-1), n)
    b = np.zeros(2*n-1)
    a = np.zeros(2*n-1)

    if mask1.any():
        b[mask1] -= (1.0 - h[i + idx[mask1]])
    
    mask2 = (i - idx >= 0) & (i - idx < n)
    if mask2.any():
        b[mask2] += h[i - idx[mask2]]
    
    mask3 = (j + idx >= 0) & (j + idx < n)
    if mask3.any():
        b[mask3] += (1.0 - h[j + idx[mask3]])
    
    mask4 = (j - idx >= 0) & (j - idx < n)
    if mask4.any():
        b[mask4] -= h[j - idx[mask4]]
    
    if i != j:
        a[j - i + (n-1)] += 1.0
        a[i - j + (n-1)] += 1.0
    
    # Feasible range for t
    t_min = max(h[i] - 1.0, -h[j])
    t_max = min(h[i], 1.0 - h[j])
    
    if t_max - t_min < 1e-15:
        return 0.0, c0.max()
    
    # Evaluate at endpoints and use golden section
    c_at_min = pair_line_eval(t_min, c0, a, b)
    c_at_max = pair_line_eval(t_max, c0, a, b)
    c_at_zero = c0.max()  # t=0
    
    # Quick check: if endpoint is already best, done
    best_c = min(c_at_min, c_at_max, c_at_zero)
    if best_c == c_at_zero:
        return 0.0, c0.max()
    
    # Golden section search
    gr = (math.sqrt(5.0) - 1.0) / 2.0  # golden ratio conjugate
    lo = t_min
    hi = t_max
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = pair_line_eval(x1, c0, a, b)
    f2 = pair_line_eval(x2, c0, a, b)

            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = pair_line_eval(x1, c0, a, b)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = pair_line_eval(x2, c0, a, b)
    
    t_opt = 0.5 * (lo + hi)
    c_opt = pair_line_eval(t_opt, c0, a, b)
    
    # Check against endpoints
    if c_at_min < c_opt:
        t_opt, c_opt = t_min, c_at_min
    if c_at_max < c_opt:

    return t_opt, c_opt * dx

import time
import math
import numpy as np


def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def repair(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            tr = out.sum()
            if tr > 0:
                out -= (d / tr) * out
        else:
            tr = (1.0 - out).sum()
            if tr > 0:
                out += (-d / tr) * (1.0 - out)
        out = np.clip(out, 0.0, 1.0)
    return out


def objective(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def make_initial(n, rng):
    h = (rng.random(n) >= 0.5).astype(float)
    return repair(h, n / 2.0)


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def active_gradient(h, tol_frac=0.02):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    cmax = c.max()
    thr = cmax - tol_frac * cmax
    w = np.maximum(c - thr, 0.0)
    wsum = w.sum()
    if wsum < 1e-18:
        w = np.ones(len(c)) / len(c)
    else:
        w /= wsum
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        np_ = w[center + p]
        nm = w[center - p]
        if np_ > 1e-15:
            grad[:n - p] += np_ * (1.0 - h[p:])
            grad[p:] -= np_ * h[:n - p]
        if nm > 1e-15:
            grad[p:] += nm * (1.0 - h[:n - p])
            grad[:n - p] -= nm * h[p:]
    grad *= dx
    return grad


def pair_line_eval(t, c0, a, b):
    return (c0 + a * t * t + b * t).max()


def optimize_pair(h, i, j, c0):
    n = len(h)
    idx = np.arange(-(n - 1), n)
    b = np.zeros(2 * n - 1)
    a = np.zeros(2 * n - 1)
    m1 = (i + idx >= 0) & (i + idx < n)
    if m1.any():
        b[m1] -= (1.0 - h[i + idx[m1]])
    m2 = (i - idx >= 0) & (i - idx < n)
    if m2.any():
        b[m2] += h[i - idx[m2]]
    m3 = (j + idx >= 0) & (j + idx < n)
    if m3.any():
        b[m3] += (1.0 - h[j + idx[m3]])
    m4 = (j - idx >= 0) & (j - idx < n)
    if m4.any():
        b[m4] -= h[j - idx[m4]]
    if i != j:
        a[j - i + (n - 1)] += 1.0
        a[i - j + (n - 1)] += 1.0
    t_min = max(h[i] - 1.0, -h[j])
    t_max = min(h[i], 1.0 - h[j])
    if t_max - t_min < 1e-14:
        return 0.0, c0.max()
    c_min = pair_line_eval(t_min, c0, a, b)
    c_max = pair_line_eval(t_max, c0, a, b)
    c_zero = c0.max()
    best_c = min(c_min, c_max, c_zero)
    if best_c >= c_zero:
        return 0.0, c_zero
    gr = (math.sqrt(5.0) - 1.0) / 2.0
    lo, hi = t_min, t_max
    x1 = hi - gr * (hi - lo)
    x2 = lo + gr * (hi - lo)
    f1 = pair_line_eval(x1, c0, a, b)
    f2 = pair_line_eval(x2, c0, a, b)
    for _ in range(45):
        if f1 < f2:
            hi = x2
            x2 = x1
            f2 = f1
            x1 = hi - gr * (hi - lo)
            f1 = pair_line_eval(x1, c0, a, b)
        else:
            lo = x1
            x1 = x2
            f1 = f2
            x2 = lo + gr * (hi - lo)
            f2 = pair_line_eval(x2, c0, a, b)
    t_opt = 0.5 * (lo + hi)
    c_opt = pair_line_eval(t_opt, c0, a, b)
    if c_min < c_opt:
        t_opt, c_opt = t_min, c_min
    if c_max < c_opt:
        t_opt, c_opt = t_max, c_max
    return t_opt, c_opt


def pair_polish(h, deadline, rng, n_pairs=400, n_random=80, stall_limit=5, perturb_scale=0.06):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    c0 = corr_full(h)
    cmax = c0.max() * dx
    best_h = h.copy()
    best_c5 = cmax
    stall = 0
    while time.monotonic() < deadline:
        grad = active_gradient(h, tol_frac=0.015)
        pairs = []
        gd = np.abs(grad[:, None] - grad[None, :])
        np.fill_diagonal(gd, 0.0)
        flat = gd.ravel()
        top_k = min(n_pairs, flat.size)
        top_idx = np.argpartition(flat, -top_k)[-top_k:]
        top_idx = top_idx[np.argsort(flat[top_idx])[::-1]]
        for fi in top_idx:
            ii = int(fi // n)
            jj = int(fi % n)
            if ii != jj:
                pairs.append((ii, jj))
        for _ in range(n_random):
            ii = int(rng.integers(n))
            jj = int(rng.integers(n))
            if ii != jj:
                pairs.append((ii, jj))
        improved = False
        for (ii, jj) in pairs:
            if time.monotonic() > deadline:
                break
            t_opt, c_new_raw = optimize_pair(h, ii, jj, c0)
            c_new = c_new_raw * dx
            if c_new < cmax - 1e-11:
                h[ii] -= t_opt
                h[jj] += t_opt
                h[ii] = max(0.0, min(1.0, h[ii]))
                h[jj] = max(0.0, min(1.0, h[jj]))
                c0 = corr_full(h)
                cmax = c0.max() * dx
                if cmax < best_c5 - 1e-12:
                    best_c5 = cmax
                    best_h = h.copy()
                improved = True
        if not improved:
            stall += 1
            if stall >= stall_limit:
                h = repair(h + rng.normal(0.0, perturb_scale, n), target)
                c0 = corr_full(h)
                cmax = c0.max() * dx
                stall = 0
                perturb_scale *= 0.7
                if perturb_scale < 0.001:
                    break
    return best_h, best_c5
