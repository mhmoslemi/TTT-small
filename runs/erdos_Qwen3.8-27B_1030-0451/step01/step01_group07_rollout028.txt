                a[j] -= d
                a[i] += d
        else:
            tmp = a[i]
            a[i] = a[j]
            a[j] = tmp

        na = objective(a)
        delta = na - cur
        frac = min(1.0, (time.monotonic() - t0) / total)
        T = temp_end + (temp_start - temp_end) * (1.0 - frac) ** 2

        if delta < -1e-14 or rng.random() < math.exp(-delta / max(T, 1e-14)):
            h = a
            cur = na
            if na < best_c - 1e-13:
                best_c = na
                best_h = a.copy()

    return best_h, best_c


def make_perturbed_templates(h, rng, n):
    """Generate diverse perturbed versions of h."""
    templates = []
    x = np.arange(n)
    # Fourier mode perturbations
    for k in range(1, 6):
        for phase in [0, math.pi / 2]:
            amp = 0.08 * math.exp(-0.3 * k)
            pert = amp * np.cos(2.0 * math.pi * k * x / n + phase)
            templates.append(project_exact(h + pert, n / 2.0))
    # Random sparse swaps
    for _ in range(4):
        h2 = h.copy()
        nswap = rng.integers(3, 8)
        for _ in range(nswap):
            i, j = rng.integers(n, 2)
            h2[i], h2[j] = h2[j], h2[i]
        templates.append(repair(h2, n / 2.0))
    # Random transfer
    for _ in range(3):
        h2 = h.copy()
        for _ in range(rng.integers(5, 15)):
            i, j = rng.integers(n, 2)
            d = rng.random() * 0.15
            if d <= min(h2[i], 1.0 - h2[j]):
                h2[i] -= d
                h2[j] += d
        templates.append(repair(h2, n / 2.0))
    return templates


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 6.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        n = 131
        r = rng.random(n)
        parent = (r >= 0.5).astype(float)
        parent = repair(parent, n / 2.0)

    n0 = len(parent)
    h0 = repair(parent, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = objective(best_h)

    def consider(cand):
        nonlocal best_h, best_c5
        try:
            c = objective(cand)
            if np.isfinite(c) and c < best_c5 - 1e-12:
                best_c5 = c
                best_h = cand.copy()
        except Exception:
            pass

    # Phase 1: Quick LSE polish at native resolution
    alloc = min(0.06 * effective, deadline - time.monotonic())
    if alloc > 1.0:
        ch, cc = momentum_lse(h0, 250.0, t0 + alloc, mu=0.7, step_init=0.03, patience=40, rng=rng)
        consider(ch)

    # Phase 2: Multi-sigma polish at native resolution
    sigma_schedule = [(150.0, 0.08), (60.0, 0.08), (25.0, 0.06)]
    for sigma, fr in sigma_schedule:
        alloc = min(fr * effective, deadline - time.monotonic())
        if alloc < 1.0:
            break
        ch, cc = momentum_lse(best_h, sigma, t0 + alloc, mu=0.8, step_init=0.04, patience=50, rng=rng)
        consider(ch)

    # Phase 3: Multi-resolution exploration
    n_candidates = [n0 - 6, n0 - 3, n0 + 3, n0 + 6, n0 + 10, n0 + 14]
    n_candidates = [n for n in n_candidates if 80 <= n <= 250]
    # Deduplicate
    seen = set()
    n_unique = []
    for n in n_candidates:
        if n not in seen:
            seen.add(n)
            n_unique.append(n)

    per_res = 0.08 * effective
    for nn in n_unique:
        if time.monotonic() > deadline:
            break
        alloc = min(per_res, deadline - time.monotonic())
        if alloc < 1.0:
            break
        rs = repair(resample(best_h, nn), nn / 2.0)
        ch1, cc1 = momentum_lse(rs, 200.0, t0 + alloc * 0.4, mu=0.7, step_init=0.03, patience=30, rng=rng)
        consider(ch1)
        alloc2 = min(alloc * 0.4, deadline - time.monotonic())
        if alloc2 > 0.5:
            ch2, cc2 = momentum_lse(ch1, 40.0, t0 + alloc2, mu=0.85, step_init=0.04, patience=50, rng=rng)
            consider(ch2)

    # Phase 4: Diverse template exploration at best resolution
    n_best = len(best_h)
    templates = make_perturbed_templates(best_h, rng, n_best)
    per_tmpl = 0.03 * effective
    for tmpl in templates:
        if time.monotonic() > deadline:
            break
        alloc = min(per_tmpl, deadline - time.monotonic())
        if alloc < 0.5:
            break
        ch, cc = momentum_lse(tmpl, 100.0, t0 + alloc, mu=0.8, step_init=0.05, patience=40, rng=rng)
        consider(ch)

    # Phase 5: Targeted annealing at best resolution
    if time.monotonic() < deadline - 5.0:
        alloc = min(0.20 * effective, deadline - time.monotonic())
        if alloc > 3.0:
            ch, cc = targeted_anneal(best_h, rng, t0 + alloc, temp_start=3e-5, temp_end=5e-10)
            consider(ch)
            # Polish after anneal
            if time.monotonic() < deadline - 2.0:
                alloc2 = min(0.04 * effective, deadline - time.monotonic())
                if alloc2 > 1.0:
                    ch2, cc2 = momentum_lse(best_h, 50.0, t0 + alloc2, mu=0.85, step_init=0.04, patience=50, rng=rng)
                    consider(ch2)

    # Phase 6: Refine around best - try nearby resolutions of the best
    if time.monotonic() < deadline - 3.0:
        n_ref = len(best_h)
        for dn in [-2, 2, 4, -4, 6]:
            nn = n_ref + dn
            if nn < 80 or nn > 250 or time.monotonic() > deadline:
                continue
            alloc = min(0.025 * effective, deadline - time.monotonic())
            if alloc < 0.5:
                break
            rs = repair(resample(best_h, nn), nn / 2.0)
            ch, cc = momentum_lse(rs, 200.0, t0 + alloc * 0.5, mu=0.7, step_init=0.03, patience=30, rng=rng)
            consider(ch)
            alloc2 = min(alloc * 0.4, deadline - time.monotonic())
            if alloc2 > 0.3:
                ch2, cc2 = momentum_lse(ch, 30.0, t0 + alloc2, mu=0.85, step_init=0.04, patience=50, rng=rng)
                consider(ch2)

    # Phase 7: Final fine polish
    if time.monotonic() < deadline - 1.0:
        alloc = min(0.03 * effective, deadline - time.monotonic())
        if alloc > 1.0:
            ch, cc = momentum_lse(best_h, 500.0, t0 + alloc, mu=0.9, step_init=0.02, patience=60, rng=rng)
            consider(ch)

    # Final feasibility guarantee
    n_final = len(best_h)
    best_h = repair(np.asarray(best_h, dtype=float), n_final / 2.0)
    best_c5 = objective(best_h)
    return best_h, float(best_c5), int(n_final)
