def active_set_direction(h, n_active=5):
    """Find a direction that most evenly reduces the top n_active correlation values."""
    c = get_corr_values(h)
    N = len(h)
    dx = 2.0 / N
    
    # Get top n_active lags
    top_indices = np.argsort(c)[-n_active:]
    
    # Compute gradients for each active lag
    grads = []
    for r in top_indices:
        g = np.zeros(N)
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += dx * (1.0 - h[:n])
                g[:n] -= dx * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += dx * (1.0 - h[dd:])
                g[dd:] -= dx * h[:n]
        grads.append(g)
    
    grads = np.array(grads)  # shape: (n_active, N)
    
    # We want to find direction d such that <grads[i], d> is as negative as possible
    # for all i, subject to ||d|| = 1 (or some normalization)
    # This is the Chebyshev problem: min_t max_i <grads[i], d>
    # Equivalently: find d that minimizes the maximum directional derivative
    
    # Simple approach: use the average gradient (which minimizes the average, not the max)
    # Better: use the "worst-case" direction
    # For a quick approximation: normalize the sum of gradients
    direction = grads.sum(axis=0)
    norm = np.linalg.norm(direction)
    if norm > 1e-10:
        direction /= norm
    return -direction

def grad_from_weights(h, w):

    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if abs(a) < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return g

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h

def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def get_corr_values(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx


def lse_obj(x, tau):
    c = get_corr_values(x)
    m = np.max(c)
    s = (c - m) / tau
    s = np.clip(s, -200.0, 200.0)
    return m + tau * np.log(np.sum(np.exp(s)))


def lse_weights_fn(x, tau):
    c = get_corr_values(x)
    m = np.max(c)
    s = (c - m) / tau
    s = np.clip(s, -200.0, 200.0)
    w = np.exp(s)
    return w / w.sum()


def grad_from_weights(h, w):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if abs(a) < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return g


def lse_grad(x, tau):
    w = lse_weights_fn(x, tau)
    return grad_from_weights(x, w)


def topk_obj(x, k):
    c = get_corr_values(x)
    if k >= len(c):
        return np.mean(c)
    idx = np.argpartition(c, -k)[-k:]
    return np.mean(c[idx])


def topk_grad(x, k):
    c = get_corr_values(x)
    if k >= len(c):
        w = np.ones(len(c)) / len(c)
    else:
        idx = np.argpartition(c, -k)[-k:]
        w = np.zeros(len(c))
        w[idx] = 1.0 / k
    return grad_from_weights(x, w)


def sum_constraint(x, target_sum):
    return np.sum(x) - target_sum


def sum_constraint_jac(x, target_sum):
    return np.ones_like(x)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                add = min(residual, 1.0 - h[idx[0]])
                h[idx[0]] += add
                residual -= add
                if abs(residual) > 1e-12 and len(idx) > 1:
                    h[idx[1]] += min(residual, 1.0 - h[idx[1]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                sub = max(residual, -h[idx[0]])
                h[idx[0]] += sub
    return h


def upscale_to(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, float(N_new // 2))


def run_slsqp(h, N_target, tau, maxiter, deadline):
    target_sum = float(N_target // 2)
    bounds = [(0.0, 1.0)] * N_target
    constraint = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac,
                   'args': (target_sum,)}]
    try:
        result = minimize(lse_obj, h.copy(), jac=lse_grad, args=(tau,),
                          method='SLSQP', bounds=bounds, constraints=constraint,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(result.x, target_sum)
    except Exception:
        return h


def run_slsqp_topk(h, N_target, k, maxiter, deadline):
    target_sum = float(N_target // 2)
    bounds = [(0.0, 1.0)] * N_target
    constraint = [{'type': 'eq', 'fun': sum_constraint, 'jac': sum_constraint_jac,
                   'args': (target_sum,)}]
    try:
        result = minimize(topk_obj, h.copy(), jac=topk_grad, args=(k,),
                          method='SLSQP', bounds=bounds, constraints=constraint,
                          options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(result.x, target_sum)
    except Exception:
        return h


def evolution_phase(h, N_target, rng, n_steps, deadline):
    target_sum = float(N_target // 2)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    for step in range(n_steps):
        if time.time() > deadline:
            break
        progress = step / max(n_steps - 1, 1)
        scale = 0.04 * (1.0 - progress) ** 1.5 + 0.005
        noise = rng.normal(0, scale, N_target)
        h_new = project_feasible(best_h + noise, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    return best_h, best_c5
