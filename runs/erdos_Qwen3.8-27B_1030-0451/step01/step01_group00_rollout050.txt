def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    p = np.nan_to_num(p, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def corr_grad_row(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(lag, n):
            j = i - lag
            grad[i] += dx * (1.0 - h[j])
            grad[j] -= dx * h[i]
    else:
        L = -lag
        for i in range(0, n - L):
            grad[i] += dx * (1.0 - h[i + L])
            grad[i + L] -= dx * h[i]
    return grad

def get_active(h, dx, n_active):
    corr = compute_corr(h)
    n = len(h)
    n_lags = len(corr)
    if n_lags <= n_active:
        idx = np.arange(n_lags)
    else:
        idx = np.argsort(corr)[-n_active:]
    grads = np.zeros((len(idx), n))
    for pos, q in enumerate(idx):
        lag = q - (n - 1)
        grads[pos] = corr_grad_row(h, lag, dx)
    return corr, grads, idx

    c_obj = np.zeros(nvar)

    b_list = []

        b_list.append(0.0)

        row[i] = -1.0
        row[n+i] = -1.0

        b_list.append(0.0)
    A_ub = np.array(A_list)
    b_ub = np.array(b_list)
    A_eq = np.zeros((1, nvar))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        bounds.append((-min(radius, h[i]+1e-12), min(radius, 1.0-h[i]+1e-12)))

def softmax_grad(h, tau):

import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    p = np.nan_to_num(p, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def corr_grad_row(h, lag, dx):
    n = len(h)
    grad = np.zeros(n)
    if lag >= 0:
        for i in range(lag, n):
            j = i - lag
            grad[i] += dx * (1.0 - h[j])
            grad[j] -= dx * h[i]
    else:
        L = -lag
        for i in range(0, n - L):
            grad[i] += dx * (1.0 - h[i + L])
            grad[i + L] -= dx * h[i]
    return grad

def get_active(h, dx, n_active):
    corr = compute_corr(h)
    n = len(h)
    n_lags = len(corr)
    if n_lags <= n_active:
        idx = np.arange(n_lags)
    else:
        idx = np.argsort(corr)[-n_active:]
    grads = np.zeros((len(idx), n))
    for pos, q in enumerate(idx):
        lag = q - (n - 1)
        grads[pos] = corr_grad_row(h, lag, dx)
    return corr, grads, idx

def lp_direction(h, corr, grads, idx, f, radius, eta):
    n = len(h)
    M = len(idx)
    nvar = 2 * n + 1
    c_obj = np.zeros(nvar)
    c_obj[n:2 * n] = eta
    c_obj[2 * n] = 1.0
    A_rows = []
    b_rows = []
    for m in range(M):
        row = np.zeros(nvar)
        row[0:n] = grads[m]
        row[2 * n] = -1.0
        A_rows.append(row)
        b_rows.append(f - corr[idx[m]])
    for i in range(n):
        r1 = np.zeros(nvar)
        r1[i] = 1.0
        r1[n + i] = -1.0
        A_rows.append(r1)
        b_rows.append(0.0)
        r2 = np.zeros(nvar)
        r2[i] = -1.0
        r2[n + i] = -1.0
        A_rows.append(r2)
        b_rows.append(0.0)
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    A_eq = np.zeros((1, nvar))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        bounds.append((-min(radius, h[i] + 1e-12), min(radius, 1.0 - h[i] + 1e-12)))
    bounds.extend([(0.0, radius)] * n)
    bounds.append((0.0, 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0 and res.x is not None:
            d = res.x[0:n]
            if np.all(np.isfinite(d)):
                return d
    except Exception:
        pass
    return None

def softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        if w[k] < 1e-14:
            continue
        lag = k - (n - 1)
        grad += w[k] * corr_grad_row(h, lag, dx)
    return grad

def lp_optimize(h0, tlimit, seed_val=42):

    best_h = h.copy()

    t0 = time.time()
    rng = np.random.RandomState(seed_val)

            best_h = h.copy()
            last_improve = time.time()
        
        _, grads, idx = get_active(h, dx, n_act)
        f = c5_val
        delta = lp_direction(h, corr, grads, idx, f, radius, eta)
        
        progress_this_iter = False
        if delta is not None and np.linalg.norm(delta) > 1e-14:

                    if c_new < c5_val - 1e-12:

                        radius = min(radius * 1.04, 0.12)
                        progress_this_iter = True
                    else:
                        stall += 1
                    break
            if not progress_this_iter and stall % 3 == 0:

            tau = 30.0 + 80.0 * min(stall / 8.0, 5.0)

            gnorm = np.linalg.norm(grad)
            if gnorm > 1e-12:
                step = max(0.003, radius * 0.6)

                c_new = compute_c5(h_new)
                if c_new <= c5_val + 1e-13:

                    if c_new < c5_val - 1e-12:

def coord_search(h, tlimit, seed_val=99):
    n = len(h)
    rng = np.random.RandomState(seed_val)
    t0 = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    c5_curr = best_c5
    
    while time.time() - t0 < tlimit:
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        for direction in (1, -1):
            if direction > 0:
                max_amt = min(h[i], 1.0 - h[j])
            else:
                max_amt = min(1.0 - h[i], h[j])
            if max_amt < 1e-8:
                continue
            for frac in (0.02, 0.05, 0.1, 0.2):
                amt = frac * max_amt * direction
                h_new = h.copy()
                h_new[i] += amt
                h_new[j] -= amt
                h_new = project_constraint(h_new, n)
                c_new = compute_c5(h_new)
                if c_new < best_c5 - 1e-13:
                    best_c5 = c_new
                    best_h = h_new.copy()
                    h = h_new
                    c5_curr = c_new
                    break
    
    return best_h, best_c5
