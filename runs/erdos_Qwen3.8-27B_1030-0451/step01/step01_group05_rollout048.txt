def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def line_search_pair(h, raw, a, b, n):
    """Find optimal d for h[a]+=d, h[b]-=d. Returns (best_d, best_score)."""
    # Bounds on d
    d_lo = max(-h[a], h[b] - 1.0)
    d_hi = min(1.0 - h[a], h[b])
    if d_hi - d_lo < 1e-14:
        return 0.0, float(np.max(raw))
    
    # Compute lin and quad for each lag
    L = 2 * n - 1
    center = n - 1
    ha = h[a]
    hb = h[b]
    
    lin = np.zeros(L)
    quad = np.zeros(L)
    
    # For each lag K, shift = K - center
    for K in range(L):
        s = K - center
        if s == 0:
            lin[K] = 2.0 * (hb - ha)
            quad[K] = -2.0
        elif s > 0:
            # c(s) = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
            if s <= a <= n-1:
                lin[K] += (1.0 - h[a-s])
            if 0 <= a <= n-1-s:
                lin[K] += (-h[a+s])
            if s <= b <= n-1:
                lin[K] += (-(1.0 - h[b-s]))
            if 0 <= b <= n-1-s:
                lin[K] += (h[b+s])
            if abs(a-b) == s:
                quad[K] = 1.0
        else:
            # shift = -|s|, c(-|s|) = sum_{j=0}^{n-1-|s|} h[j]*(1-h[j+|s|])
            s_abs = -s
            if 0 <= a <= n-1-s_abs:
                lin[K] += (1.0 - h[a+s_abs])
            if s_abs <= a <= n-1:
                lin[K] += (-h[a-s_abs])
            if 0 <= b <= n-1-s_abs:
                lin[K] += (-(1.0 - h[b+s_abs]))
            if s_abs <= b <= n-1:
                lin[K] += (h[b-s_abs])
            if abs(a-b) == s_abs:
                quad[K] = 1.0
    
    # Grid search for d
    best_d = 0.0
    best_val = float(np.max(raw))  # current value at d=0
    
    # Sample points
    n_grid = 30
    ds = np.linspace(d_lo, d_hi, n_grid)
    # Add some non-uniform points near 0
    if d_lo < 0 < d_hi:
        extra = np.linspace(min(d_lo, 0), max(d_hi, 0), 10)
        ds = np.unique(np.concatenate([ds, extra]))
    
    for d in ds:
        vals = raw + d * lin + d * d * quad
        mx = np.max(vals)
        if mx < best_val:
            best_val = mx
            best_d = d
    
    # Local refinement around best_d
    step = (d_hi - d_lo) / n_grid
    for _ in range(3):
        lo_r = max(d_lo, best_d - step)
        hi_r = min(d_hi, best_d + step)
        if hi_r - lo_r < 1e-14:
            break
        ds_fine = np.linspace(lo_r, hi_r, 15)
        for d in ds_fine:
            vals = raw + d * lin + d * d * quad
            mx = np.max(vals)
            if mx < best_val:
                best_val = mx
                best_d = d
        step *= 0.3
    
    return best_d, best_val

def compute_sensitivity(h, raw, tau, n):
    """Compute weighted sensitivity of each coordinate to the objective."""
    L = 2 * n - 1
    center = n - 1
    
    # Active lag weights
    raw_max = np.max(raw)
    exponents = np.clip((raw - raw_max) / max(tau, 1e-10), -500, 500)
    weights = np.exp(exponents)
    wsum = np.sum(weights)
    if wsum < 1e-300:
        weights = np.ones(L) / L
    else:
        weights /= wsum
    
    # Compute sensitivity for each coordinate
    sens = np.zeros(n)
    
    # For each lag, add the contribution to each coordinate's sensitivity
    for K in range(L):
        w = weights[K]
        if w < 1e-10:
            continue
        s = K - center
        if s == 0:
            # c(0) = sum h[i](1-h[i]), dc(0)/dh[i] = 1-2h[i]
            sens += w * (1.0 - 2.0 * h)
        elif s > 0:
            # c(s) = sum_{j=0}^{n-1-s} h[j+s]*(1-h[j])
            # dc/dh[i]:
            #   h[i] as h[j+s] (j=i-s): valid if s<=i<=n-1, contribution (1-h[i-s])
            #   h[i] as h[j] (j=i): valid if 0<=i<=n-1-s, contribution h[i+s] (with minus sign for 1-h[j])
            # Wait: c(s) = sum h[j+s]*(1-h[j])
            # ∂c/∂h[i]: h[i] appears as h[j+s] when j=something, or as h[j] when j=something
            # If i = j+s for some j: j = i-s, term is h[i]*(1-h[i-s]), derivative w.r.t. h[i] is (1-h[i-s])
            #   valid if 0<=i-s<=n-1-s, i.e., s<=i<=n-1
            # If i = j for some j: term is h[i+s]*(1-h[i]), derivative w.r.t. h[i] is -h[i+s]
            #   valid if 0<=i<=n-1-s
            if s <= n-1:
                idx = np.arange(s, n)
                sens[idx] += w * (1.0 - h[idx - s])
            if n-1-s >= 0:
                idx = np.arange(0, n-s)
                sens[idx] += w * (-h[idx + s])
        else:
            s_abs = -s
            # c(-s_abs) = sum_{j=0}^{n-1-s_abs} h[j]*(1-h[j+s_abs])
            # ∂c/∂h[i]:
            # If i = j: term h[i]*(1-h[i+s_abs]), derivative (1-h[i+s_abs]), valid if 0<=i<=n-1-s_abs
            # If i = j+s_abs: j = i-s_abs, term h[i-s_abs]*(1-h[i]), derivative -h[i-s_abs], valid if s_abs<=i<=n-1
            if n-1-s_abs >= 0:
                idx = np.arange(0, n-s_abs)
                sens[idx] += w * (1.0 - h[idx + s_abs])
            if s_abs <= n-1:
                idx = np.arange(s_abs, n)
                sens[idx] += w * (-h[idx - s_abs])
    
    return sens

    """Iterative pair exchange optimization."""
    best_h = h.copy()
    best_score = exact_score(h)
    cur = h.copy()
    cur_score = exact_score(cur)
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        raw = np.correlate(cur, 1.0 - cur, mode='full')
        raw_max = np.max(raw)
        
        # Compute sensitivity
        sens = compute_sensitivity(cur, raw, tau, n)
        
        # Generate candidate pairs
        order = np.argsort(sens)
        candidates = []
        
        # Best pair: min sens (increase) and max sens (decrease)
        candidates.append((int(order[0]), int(order[-1])))
        # Second best
        candidates.append((int(order[1]), int(order[-2])))
        # Third
        candidates.append((int(order[2]), int(order[-3])))
        
        # Random pairs
        for _ in range(5):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a != b:
                candidates.append((a, b))
        
        # Try each candidate
        best_move = None
        best_move_score = cur_score
        
        for (a, b) in candidates:
            if a == b:
                continue
            d_opt, score_raw = line_search_pair(cur, raw, a, b, n)
            score = score_raw * (2.0 / n)
            if score < best_move_score - 1e-12:
                best_move_score = score
                best_move = (a, b, d_opt)
        
        if best_move is not None:
            a, b, d = best_move
            cur[a] += d
