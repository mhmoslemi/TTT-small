    n = len(h)
    dx = 2.0 / n
    m = np.max(C)
    if tau < 1e-14:

        act = C >= m - 1e-10

        alpha[act] = 1.0 / np.sum(act)

        e = np.exp(a)
        alpha = e / e.sum()

import numpy as np
import time

def corr_scaled(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, "full") * (2.0 / float(n))

def exact_C5(h):
    if len(h) == 0:
        return 0.0
    c = corr_scaled(h)
    return float(np.max(c))

def make_feasible(x, target):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5)
    x = np.clip(x, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    r = target - float(x.sum())
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])]
                for i in order:
                    inc = min(1.0 - x[i], r)
                    x[i] += inc
                    r -= inc
                    if r <= 1e-14:
                        break
        else:
            r = -r
            idx = np.where(x > 0.0)[0]
            if idx.size:
                order = idx[np.argsort(-x[idx])]
                for i in order:
                    dec = min(x[i], r)
                    x[i] -= dec
                    r -= dec
                    if r <= 1e-14:
                        break
    return np.clip(x, 0.0, 1.0)

def active_gradient(h, C, tau):
    n = len(h)
    dx = 2.0 / float(n)
    m = float(np.max(C))
    if tau < 1e-14:
        act = C >= m - 1e-10
        cnt = float(np.sum(act))
        if cnt < 1.0:
            alpha = np.zeros_like(C)
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha = np.zeros_like(C)
            alpha[act] = 1.0 / cnt
    else:
        a = (C - m) / tau
        e = np.exp(a)
        s = float(np.sum(e))
        if s < 1e-300:
            alpha = np.zeros_like(C)
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha = e / s
    G = np.zeros(n)
    n1 = n - 1
    active_idx = np.where(alpha > 1e-12)[0]
    for idx in active_idx:
        ak = float(alpha[idx])
        k_val = int(idx) - n1
        coeff = ak * dx
        if k_val == 0:
            G += coeff * (1.0 - 2.0 * h)
        elif k_val > 0:
            G[k_val:] += coeff * (1.0 - h[:n - k_val])
            G[:n - k_val] += -coeff * h[k_val:]
        else:
            dd = -k_val
            G[:n - dd] += coeff * (1.0 - h[dd:])
            G[dd:] += -coeff * h[:n - dd]
    return G

def binary_from_h(h, target):

    cnt = int(round(target))
    if cnt <= 0 or cnt >= n:
        return None
    order = np.argsort(h, kind="stable")[::-1][:cnt]
    b = np.zeros(n, dtype=float)
    b[order] = 1.0
    return b

def resample_h(h, N):

        return make_feasible(h, target)

    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))

        G = active_gradient(h, C, tau)

        gn = float(np.linalg.norm(G))
        if gn < 1e-12:

        g = G / gn

def binary_sa(h_bin, deadline, rng, max_iter=500000):

    target = 0.5 * float(n)
    h = h_bin.copy()
    C = corr_scaled(h)
    c = float(np.max(C))
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    T = 0.005
    cool = 0.9997
    best_c = c
    best_h = h.copy()
    iters = 0

    while time.time() < deadline and iters < max_iter:
        if len(ones) == 0 or len(zeros) == 0:
            break
        p = ones[rng.integers(len(ones))]
        q = zeros[rng.integers(len(zeros))]
        if p == q:
            iters += 1
            continue
        h[p] = 0.0
        h[q] = 1.0
        C_new = corr_scaled(h)
        c_new = float(np.max(C_new))
        delta = c_new - c
        if delta < 0 or rng.random() < np.exp(-max(delta, 0) / max(T, 1e-12)):
            c = c_new
            C = C_new
            ones = np.where(h == 1.0)[0]
            zeros = np.where(h == 0.0)[0]
            if c < best_c:
                best_c = c
                best_h = h.copy()
        else:
            h[p] = 1.0
            h[q] = 0.0
        T *= cool
        iters += 1

    return best_h, best_c

        h_bin = binary_from_h(h_res, 0.5 * float(N))
        if h_bin is None:
            continue
        c_bin_init = exact_C5(h_bin)
        sa_deadline = min(hard - 30.0, time.time() + 0.15 * total)
        h_sa, c_sa = binary_sa(h_bin, sa_deadline, rng)
        if np.isfinite(c_sa) and c_sa < best_c:
            best_h, best_c, best_n = h_sa, c_sa, int(len(h_sa))
        # Try continuous optimization from the binary solution
        if time.time() < hard - 20.0:
            h_cont = make_feasible(h_sa + rng.normal(0, 0.02, N), 0.5 * float(N))
            cont_deadline = min(hard - 15.0, time.time() + 0.1 * total)
            h_cont_opt, c_cont = gradient_optimize(h_cont, cont_deadline, rng)
            if np.isfinite(c_cont) and c_cont < best_c:
                best_h, best_c, best_n = h_cont_opt, c_cont, int(len(h_cont_opt))

    # Phase 4: More continuous optimization from best
    if time.time() < hard - 10.0:
        final_deadline = hard - 3.0
        h_fin, c_fin = gradient_optimize(best_h.copy(), final_deadline, rng)
        if np.isfinite(c_fin) and c_fin < best_c:
            best_h, best_c, best_n = h_fin, c_fin, int(len(h_fin))

    # Phase 5: Try different smooth starts if time remains
    if time.time() < hard - 30.0 and best_c > 0.3808:
        for N in [240, 360, 480]:
            if time.time() > hard - 15.0:
                break
            freqs = list(range(1, 5))
            phases = [float(rng.uniform(0, 2 * np.pi)) for _ in freqs]
            h_smooth = smooth_candidate(N, freqs, phases, rng)
            c_s = exact_C5(h_smooth)
            if c_s < best_c + 0.02:
                dl = min(hard - 10.0, time.time() + 20.0)
                h_so, c_so = gradient_optimize(h_smooth, dl, rng)
                if np.isfinite(c_so) and c_so < best_c:
                    best_h, best_c, best_n = h_so, c_so, int(len(h_so))

    # Final check
    best_n = int(best_h.size)
    best_h = make_feasible(best_h, 0.5 * float(best_n))
    best_c = exact_C5(best_h)
    if not np.isfinite(best_c):
        best_h = make_feasible(np.full(best_n, 0.5), 0.5 * float(best_n))
        best_c = exact_C5(best_h)

    return best_h, float(best_c), best_n
