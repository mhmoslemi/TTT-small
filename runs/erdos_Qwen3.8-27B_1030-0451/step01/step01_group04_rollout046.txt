```python
def sca_lp_step_scipy(h0, N):
    """One SCA iteration using scipy linprog."""
    n_constraints = 2 * N - 1
    n_vars = N + 1  # N for h, 1 for t
    
    # Objective: minimize t
    c_obj = np.zeros(n_vars)
    c_obj[N] = 1.0  # t is the last variable
    
    # Build A_ub and b_ub
    A_ub = np.zeros((n_constraints, n_vars))
    b_ub = np.zeros(n_constraints)
    
    for r in range(n_constraints):
        d = r - (N - 1)
        A = np.zeros(N)
        if d >= 0:
            n = N - d
            const_r = np.dot(h0[:n], h0[d:d+n])
            A[:n] += (1.0 - h0[d:d+n])
            A[d:N] += -h0[:n]
        else:
            d_abs = -d
            n = N - d_abs
            const_r = np.dot(h0[d_abs:n+d_abs], h0[:n])
            A[d_abs:N] += (1.0 - h0[:n])
            A[:n] += -h0[d_abs:N]
        
        A_ub[r, :N] = A
        A_ub[r, N] = -1.0
        b_ub[r] = -const_r
    
    # Equality constraint: sum(h) = N/2
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.array([N / 2.0])
    
    # Bounds
    bounds = [(0, 1)] * N + [(-np.inf, np.inf)]
    
    from scipy.optimize import linprog
    result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                     bounds=bounds, method='highs')
    
    if result.status != 0:
        return None
    
    h_new = result.x[:N]
    return h_new
```

```python
def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def build_lp(h0, N):
    """Build LP matrices for one SCA iteration."""
    n_cons = 2 * N - 1
    n_vars = N + 1
    
    c_obj = np.zeros(n_vars)
    c_obj[N] = 1.0
    
    A_ub = np.zeros((n_cons, n_vars))
    b_ub = np.zeros(n_cons)
    
    for r in range(n_cons):
        d = r - (N - 1)
        A = np.zeros(N)
        if d >= 0:
            n = N - d
            const_r = np.dot(h0[:n], h0[d:d+n])
            A[:n] += (1.0 - h0[d:d+n])
            A[d:N] -= h0[:n]
        else:
            d_abs = -d
            n = N - d_abs
            const_r = np.dot(h0[d_abs:n+d_abs], h0[:n])
            A[d_abs:N] += (1.0 - h0[:n])
            A[:n] -= h0[d_abs:N]

                         bounds=bounds, method='highs',

        else:
```

np.correlate(a, v, mode='full')[k] = sum_n a[n] * v[n - k + len(v) - 1]

So for a = h, v = 1-h:
corr[k] = sum_n h[n] * (1-h[n - k + N - 1])

Let l = k - (N-1) be the "lag" (can be negative):
corr[l + N - 1] = sum_n h[n] * (1-h[n - l])

This is the same as: sum over valid n of h[n] * (1-h[n-l])

For l ≥ 0: h[n] and h[n-l], so n ranges from l to N-1, giving sum_{n=l}^{N-1} h[n](1-h[n-l])
         = sum_{i=0}^{N-1-l} h[i+l](1-h[i])  (substituting i = n-l)

For l < 0: let m = -l > 0. sum_n h[n](1-h[n+m]), n ranges from 0 to N-1-m
         = sum_{i=0}^{N-1-m} h[i](1-h[i+m])

```python
import numpy as np
import time
from scipy.optimize import linprog


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_to_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += max(-1.0, min(1.0, residual))
    return h


def build_and_solve_lp(h0, N):
    n_cons = 2 * N - 1
    n_vars = N + 1

    c_obj = np.zeros(n_vars)
    c_obj[N] = 1.0

    A_ub = np.zeros((n_cons, n_vars))
    b_ub = np.zeros(n_cons)

    for r in range(n_cons):
        d = r - (N - 1)
        A = np.zeros(N)
        if d >= 0:
            n = N - d
            const_r = np.dot(h0[:n], h0[d:d + n])
            A[:n] += (1.0 - h0[d:d + n])
            A[d:N] -= h0[:n]
        else:
            d_abs = -d
            n = N - d_abs
            const_r = np.dot(h0[d_abs:n + d_abs], h0[:n])
            A[d_abs:N] += (1.0 - h0[:n])
            A[:n] -= h0[d_abs:N]

        A_ub[r, :N] = A
        A_ub[r, N] = -1.0
        b_ub[r] = -const_r

    A_eq = np.zeros((1, n_vars))
    A_eq[0, :N] = 1.0
    b_eq = np.array([N / 2.0])

    bounds = [(0.0, 1.0)] * N + [(-np.inf, np.inf)]

    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs',
                         options={'time_limit': 8.0, 'presolve': True})
        if result.status == 0:
            return result.x[:N].copy(), float(result.x[N])
    except Exception:
        pass
    return None, None


def run_sca(h0, N, max_iters, deadline):
    h = h0.copy()
    target = N / 2.0
    c5_current = compute_c5(h)
    best_local_h = h.copy()
    best_local_c5 = c5_current
    no_improve_count = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        h_new, t_lp = build_and_solve_lp(h, N)
        if h_new is None:
            break

        h_new = np.clip(h_new, 0.0, 1.0)
        residual = target - np.sum(h_new)
        if abs(residual) > 1e-10:
            h_new = project_to_feasible(h_new, target)

        c5_new = compute_c5(h_new)

        improved = False
        if c5_new < c5_current - 1e-12:
            h = h_new
            c5_current = c5_new
            improved = True
        else:
            for alpha in [0.5, 0.25, 0.1, 0.05]:
                h_try = (1.0 - alpha) * h + alpha * h_new
                h_try = project_to_feasible(h_try, target)
                c5_try = compute_c5(h_try)
                if c5_try < c5_current - 1e-12:

                    improved = True

                best_local_c5 = c5_current
                best_local_h = h.copy()
        else:
            no_improve_count += 1
            if no_improve_count >= 5:
                break

def subgradient_polish(h, N, n_iters, deadline):
    h_curr = h.copy()
    target = N / 2.0
    best_h = h_curr.copy()
    best_c5 = compute_c5(h_curr)
    dx = 2.0 / N

    for it in range(n_iters):
        if time.time() > deadline:
            break

        corr = np.correlate(h_curr, 1.0 - h_curr, mode='full')
        c_vals = corr * dx
        k_star = np.argmax(c_vals)
        d = k_star - (N - 1)

        grad = np.zeros(N)
        if d >= 0:
            n = N - d
            grad[:n] += (1.0 - h_curr[d:d + n])
            grad[d:N] -= h_curr[:n]
        else:
            d_abs = -d
            n = N - d_abs
            grad[d_abs:N] += (1.0 - h_curr[:n])
            grad[:n] -= h_curr[d_abs:N]

        # Adaptive step size
        step = 0.002 / (1.0 + 0.01 * it)
        h_new = h_curr - step * dx * grad
        h_new = project_to_feasible(h_new, target)

        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new

    return best_h, best_c5
```
