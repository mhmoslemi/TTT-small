        return np.max(a * t + b)
    
    # Check endpoints
    f0 = f(0.0)
    f1 = f(1.0)
    
    # Binary search for the point where slope changes sign
    lo = 0.0
    hi = 1.0
    
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        max_val = np.max(vals)
        active = np.where(vals == max_val)[0]
        min_slope = np.min(a[active])
        max_slope = np.max(a[active])
        
        if min_slope >= 0:
            # Function is non-decreasing here, minimum is to the left
            hi = mid
        elif max_slope <= 0:
            # Function is non-increasing here, minimum is to the right
            lo = mid
        else:
            # 0 is in the subdifferential, we're at the minimum
            lo = mid
            hi = mid
            break

    # Compare with endpoints
    f_opt = f(t_opt)
    if f0 <= f_opt:
        return 0.0
    if f1 <= f_opt:
        return 1.0
    return t_opt

```python
import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(compute_corr(h)) * dx)

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-14:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            proj[mask] += adj
            # Re-clip in case of tiny violations
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def smooth_gradient(h, corr, tau):
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

def minimize_pl_1d(a, b):
    """Find t in [0,1] minimizing max_j (a[j]*t + b[j]).
    a, b are arrays of length m."""
    m = len(a)
    # Check if all slopes are the same
    # Binary search for minimum of convex PL function
    lo = 0.0
    hi = 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = a * mid + b
        max_val = np.max(vals)
        # Find active set (within tolerance)
        tol = 1e-10 * max(1.0, abs(max_val))
        active = vals >= max_val - tol
        if not np.any(active):
            active = vals >= max_val - 1e-8
        min_slope = np.min(a[active])
        max_slope = np.max(a[active])
        if min_slope >= -1e-12:
            hi = mid
        elif max_slope <= 1e-12:
            lo = mid
        else:
            # 0 in subdifferential
            break
    t_star = 0.5 * (lo + hi)
    # Compare with endpoints
    f_star = np.max(a * t_star + b)
    f_0 = np.max(b)
    f_1 = np.max(a + b)
    if f_0 <= f_star + 1e-12:
        return 0.0
    if f_1 <= f_star + 1e-12:
        return 1.0
    return t_star

def coord_descent(h, n, max_sweeps, deadline):
    """Coordinate descent: for each i, find optimal h[i] minimizing max correlation."""
    h = h.copy()
    best_c5 = compute_c5(h)
    
    for sweep in range(max_sweeps):
        if time.time() >= deadline:
            break
        improved = False
        for i in range(n):
            # Compute a_J and b_J for all lags J
            a = np.zeros(2 * n - 1)
            b = np.zeros(2 * n - 1)
            for J in range(2 * n - 1):
                l = J - (n - 1)
                a_J = 0.0
                b_J = 0.0
                # Case 1: h[i] appears as h[n+l] where n = i-l
                n1 = i - l
                if 0 <= n1 < n:
                    a_J += (1.0 - h[n1])
                # Case 2: h[i] appears as 1-h[n] where n = i, so h[n+l] = h[i+l]
                n2 = i + l
                if 0 <= n2 < n:
                    a_J += (-h[n2])
                    b_J += h[n2]
                a[J] = a_J
                b[J] = b_J
            
            t_opt = minimize_pl_1d(a, b)
            
            if abs(t_opt - h[i]) > 1e-10:
                old_h = h[i]
                h[i] = t_opt
                h = project_constraint(h, n)
                c5_new = compute_c5(h)
                if c5_new <= best_c5 + 1e-12:
                    best_c5 = max(c5_new, 0.0)
                    if c5_new < best_c5 - 1e-12:

                    h[i] = old_h

    return h, best_c5

def smooth_optimize(h0, time_limit):
    """Multiple-pass smooth gradient descent with momentum."""
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    v = np.zeros(n)  # momentum
    dx = 2.0 / n
    t0 = time.time()
    
    tau_schedule = [(20, 80), (80, 300), (300, 1200)]
    time_fractions = [0.30, 0.35, 0.35]
    
    step = 4.0 / n
    
    for pass_i, (tau_lo, tau_hi) in enumerate(tau_schedule):
        pass_deadline = t0 + time_limit * sum(time_fractions[:pass_i+1])
        it_count = 0
        
        while time.time() < pass_deadline - 2:
            it_count += 1
            frac = min(it_count / 100.0, 1.0)
            tau = tau_lo + (tau_hi - tau_lo) * frac
            
            corr = compute_corr(h)
            c5_val = float(np.max(corr) * dx)
            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()
            
            grad = smooth_gradient(h, corr, tau)
            gnorm = float(np.linalg.norm(grad))
            if gnorm < 1e-14:
                break
            
            # Nesterov momentum
            mu = 0.7
            v = mu * v - step * grad
            h_trial = h + v
            h_trial = project_constraint(h_trial, n)
            
            # Line search
            accepted = False
            for bt in range(8):
                c_trial = compute_c5(h_trial)
                if c_trial <= c5_val + 1e-12:
                    h = h_trial
                    if c_trial < c5_val - 1e-11:
                        step = min(step * 1.05, 20.0 / n)
                    else:
                        step *= 0.9
                    accepted = True
                    break
                h_trial = h - step * (v / step)  # backtrack
                # Actually, simpler: scale down
                v *= 0.5
                h_trial = project_constraint(h - step * grad, n)
            
            if not accepted:
                step *= 0.5
                if step < 0.1 / n:
                    step = 0.1 / n
                    h = project_constraint(h + np.random.randn(n) * 0.01, n)
            
    return best_h, best_c5
