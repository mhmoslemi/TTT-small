                    c5_new = evaluate_c5(h_new)
                    if c5_new < best_c5:
                        best_h = h_new
                        best_c5 = c5_new
                        found = True
            else:
                # Try negative
                t_neg = -t_max * 0.05
                h_neg = project_feasible(best_h + t_neg * d, target)
                c5_neg = evaluate_c5(h_neg)
                if c5_neg < best_c5 - 1e-9:
                    res = minimize_scalar(_line_obj, bounds=(-t_cap, 0.0), method='bounded',
                                          options={'xatol': 1e-9, 'maxiter': 60})
                    if res.success and res.fun < best_c5 - 1e-9:
                        h_new = project_feasible(best_h + res.x * d, target)
                        c5_new = evaluate_c5(h_new)
                        if c5_new < best_c5:
                            best_h = h_new
                            best_c5 = c5_new
                            found = True

        if not found:
            no_improve += 1
            if no_improve >= 5:
                rng2 = np.random.default_rng(it * 101 + 3)
                amp = 0.003 * (1 + no_improve * 0.5)
                h_pert = project_feasible(best_h + rng2.standard_normal(n) * amp, target)
                c5_pert = evaluate_c5(h_pert)
                if c5_pert < best_c5:
                    best_h = h_pert
                    best_c5 = c5_pert
                    no_improve = 0
                if no_improve >= 15:
                    break
        else:
            no_improve = 0

    return best_h, best_c5


def subspace_optimize(h, deadline, max_iter=5):
    """Optimize in the subspace spanned by active lag gradients."""
    n = len(h)
    dx = 2.0 / n
    target = n / 2.0
    global _G_H0_S, _G_B, _G_TARGET_S
    best_h = h.copy()
    best_c5 = evaluate_c5(h)

    for trial in range(max_iter):
        if time.time() > deadline - 10:
            break

        c = np.correlate(best_h, 1.0 - best_h, 'full') * dx
        cmax = c.max()
        spread = max(1e-8, cmax - c.mean())
        tol = max(1e-6, 0.2 * spread)
        active = np.where(c >= cmax - tol)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])
        # Take top K lags
        K = min(len(active), 6)
        if len(active) > K:
            order = np.argsort(-c[active])
            active = active[order[:K]]

        # Build basis from gradient directions
        basis = np.zeros((n, K))
        for k, ell in enumerate(active):
            g = compute_grad_lag(best_h, int(ell), dx)
            d = -(g - np.mean(g))
            norm = np.linalg.norm(d)
            if norm > 1e-14:
                basis[:, k] = d / norm

        # QR orthonormalization (in zero-sum subspace is automatic)
        if K > 1:
            Q, R = np.linalg.qr(basis, mode='reduced')
            # Check for zero columns in R
            valid = np.abs(np.diag(R)) > 1e-10
            Q = Q[:, valid]
            K_eff = int(valid.sum())
            if K_eff < 2:
                continue
        else:
            Q = basis
            K_eff = K

        if K_eff < 1:
            continue

        _G_H0_S = best_h
        _G_B = Q
        _G_TARGET_S = target

        # Nelder-Mead in K_eff dimensions
        x0 = np.zeros(K_eff)
        sim0 = x0.copy()
        simplex = [sim0]
        for j in range(K_eff):
            v = x0.copy()
            v[j] = 0.02
            simplex.append(v)
        simplex = np.array(simplex)

        best_val = evaluate_c5(best_h)
        rng_nm = np.random.default_rng(trial * 17 + 5)
        for j in range(K_eff):
            simplex[j + 1, j] = 0.02 * (1.0 + 0.5 * rng_nm.standard_normal())

        try:
            res = minimize(_subspace_obj, x0, method='Nelder-Mead',
                           options={'xatol': 1e-9, 'fatol': 1e-10, 'maxiter': 300,
                                    'initial_simplex': simplex})
            if res.success:
                h_new = project_feasible(best_h + Q @ res.x, target)
                c5_new = evaluate_c5(h_new)
                if c5_new < best_c5 - 1e-10:
                    best_h = h_new
                    best_c5 = c5_new
        except Exception:
            pass

    return best_h, best_c5


def safe_timestep(h, d, target):
    pos = d > 1e-12
    neg = d < -1e-12
    t_max = 5.0
    if pos.any():
        t_max = min(t_max, np.min((1.0 - h[pos]) / d[pos]))
    if neg.any():
        t_max = min(t_max, np.min(h[neg] / (-d[neg])))
    return max(t_max, 1e-10)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    # Get initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        n_par = 240
        parent = np.full(n_par, 0.5)

    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)
    par_c5 = evaluate_c5(parent)

    best_h = parent.copy()
    best_c5 = par_c5
    best_n = n_par

    # Phase 1: Active-lag optimization at parent resolution
    if time.time() < deadline - 30:
        h_opt, c5_opt = active_lag_optimize(parent, deadline)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
            best_n = n_par

    # Phase 2: Subspace refinement
    if time.time() < deadline - 30:
        h_opt, c5_opt = subspace_optimize(best_h, deadline)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 3: Try different resolutions
    n_candidates = [n_par]
    for mult in [2, 3, 4, 5, 6]:
        nc = n_par * mult
        if nc <= 2400 and nc not in n_candidates:
            n_candidates.append(nc)
    n_candidates.sort()

    for nc in n_candidates[1:]:
        if time.time() > deadline - 60:
            break

        h0 = interpolate_h(best_h, nc)
        # Quick active-lag pass
        if time.time() < deadline - 20:
            h_opt, c5_opt = active_lag_optimize(h0, deadline, max_outer=40)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = nc

        # Subspace on best
        if time.time() < deadline - 20 and len(best_h) <= 1200:
            h_opt, c5_opt = subspace_optimize(best_h, deadline, max_iter=3)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                best_n = len(best_h)

    # Phase 4: Final active-lag polish at best resolution
    if time.time() < deadline - 15:
        h_opt, c5_opt = active_lag_optimize(best_h, deadline, max_outer=60)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Phase 5: One more subspace pass
    if time.time() < deadline - 15 and len(best_h) <= 1200:
        h_opt, c5_opt = subspace_optimize(best_h, deadline, max_iter=3)
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final feasibility repair
    bn = len(best_h)
    target = bn / 2.0
    best_h = project_feasible(best_h, target)
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] += diff / cnt
            np.clip(best_h, 0.0, 1.0, out=best_h)

    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, len(best_h))
