    r = target - x.sum()
    if abs(r) > 1e-13:
        if r > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])]
                rem = r
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
                    rem -= inc
                    if rem <= 1e-12:
                        break
        else:
            idx = np.where(x > 1e-12)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])[::-1]]
                rem = -r
                for i in order:
                    dec = min(x[i], rem)
                    x[i] -= dec
                    rem -= dec
                    if rem <= 1e-12:
                        break
    return x


def compute_C(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, 'full') * dx


def exact_C5(h):
    if len(h) == 0:
        return 0.0
    return float(np.max(compute_C(h)))


def smoothed_obj(x, tau, target, mu):
    C = compute_C(x)
    M = float(np.max(C))
    a = (C - M) / tau
    a = np.clip(a, -400.0, 400.0)
    e = np.exp(a)
    s = float(np.sum(e))
    if s < 1e-300:
        val = M
    else:
        val = M + tau * np.log(s)
    val += mu * (float(np.sum(x)) - target) ** 2
    return val


def smoothed_grad(x, tau, target, mu):
    n = len(x)
    C = compute_C(x)
    M = float(np.max(C))
    a = (C - M) / tau
    a = np.clip(a, -400.0, 400.0)
    e = np.exp(a)
    s = float(np.sum(e))
    if s < 1e-300:
        alpha = np.zeros_like(C)
        alpha[int(np.argmax(C))] = 1.0
    else:
        alpha = e / s
    dx = 2.0 / n
    center = n - 1
    g = np.zeros(n)
    for j in range(len(C)):
        w = alpha[j] * dx
        if w < 1e-14:
            continue
        k = j - center
        if k >= 0:
            if k == 0:
                g += w * (1.0 - 2.0 * x)
            else:
                g[k:] += w * (1.0 - x[:n - k])
                g[:n - k] -= w * x[k:]
        else:
            d = -k
            g[:n - d] += w * (1.0 - x[d:])
            g[d:] -= w * x[:n - d]
    g += 2.0 * mu * (float(np.sum(x)) - target)
    return g


def _lbfgs_stage(h, tau, target, mu, deadline, maxiter):
    n = len(h)
    bounds = [(0.0, 1.0)] * n

    def objf(x):
        return smoothed_obj(x, tau, target, mu)

    def gradf(x):
        return smoothed_grad(x, tau, target, mu)

    try:
        res = minimize(objf, h, jac=gradf, method='L-BFGS-B',
                       bounds=bounds, options={'maxiter': maxiter,
                                               'ftol': 1e-15, 'gtol': 1e-9,
                                               'maxls': 20})
    except Exception:
        return h, exact_C5(h)
    x = np.asarray(res.x, dtype=float)
    x = make_feasible(x, target)
    return x, exact_C5(x)


def optimize_full(h, deadline, rng):
    n = len(h)
    target = 0.5 * float(n)
    h = make_feasible(h, target).copy()
    best_c = exact_C5(h)
    best_h = h.copy()
    mu = 30.0
    tau_schedule = [0.05, 0.03, 0.018, 0.010, 0.005, 0.0025, 0.0012, 0.0006, 0.0003]
    for idx, tau in enumerate(tau_schedule):
        if time.time() > deadline - 3.0:
            break
        frac = max(0.1, 1.0 - 0.08 * idx)
        remain = deadline - time.time()
        maxiter = int(max(50, min(1000, 4000.0 / max(n, 1) * 720.0 / max(1, n) * frac * 3)))
        maxiter = int(max(80, min(800, 250000.0 / max(n, 1) * frac)))
        x, c = _lbfgs_stage(h, tau, target, mu, deadline, maxiter)
        if c < best_c - 1e-12:
            best_c = c
            best_h = x.copy()
            h = x.copy()
        time_left = deadline - time.time()
        if time_left < 10.0:
            break
        if rng is not None and idx < len(tau_schedule) - 2:
            sig = 0.03 / (1.0 + 0.3 * idx)
            h_pert = make_feasible(h + rng.normal(0, sig, size=n), target)
            c_pert = exact_C5(h_pert)
            if c_pert < best_c:
                best_c = c_pert
                best_h = h_pert.copy()
                h = h_pert.copy()
    return best_h, best_c


def resample_h(h, N):
    n0 = len(h)
    target_N = 0.5 * float(N)
    if n0 == N:
        return make_feasible(h, target_N)
    xold = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))
    y = np.interp(xnew, xold, h)
    return make_feasible(y, target_N)


def block_shake(h, rng, n_trials=50):
    n = len(h)
    target = 0.5 * float(n)
    best_h = h.copy()
    best_c = exact_C5(h)
    block_sizes = [2, 3, 4, 5, 6, 8, 10]
    for _ in range(n_trials):
        bs = int(block_sizes[rng.integers(0, len(block_sizes))])
        start = int(rng.integers(0, max(1, n - bs)))
        flip_dir = rng.random() < 0.5
        h_try = h.copy()
        if flip_dir:
            h_try[start:start + bs] = 1.0 - h_try[start:start + bs]
        else:
            src_start = int(rng.integers(0, max(1, n - bs)))
            h_try[start:start + bs] = h[src_start:src_start + bs].copy()
        h_try = make_feasible(h_try, target)
        c_try = exact_C5(h_try)
        if c_try < best_c - 1e-12:
            best_c = c_try
            best_h = h_try.copy()
    return best_h, best_c


def _get_initial():
    val = globals().get('initial_h_values', None)
    if val is None:
        return None
    arr = np.asarray(val, dtype=float).ravel()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    return make_feasible(arr, 0.5 * float(arr.size))


def fallback_initial(N=360):
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    rng = np.random.default_rng(99)
    best = None
    bestc = float('inf')
    for f in [1, 2, 3, 4, 5, 6, 7]:
        for ph_off in range(4):
            ph = ph_off * 0.5 * np.pi
            y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph)
            h = make_feasible(y, 0.5 * float(N))
            c = exact_C5(h)
            if c < bestc:
                bestc = c
                best = h
    for _ in range(20):
        f = int(1 + rng.integers(0, 8))
        ph = float(rng.uniform(0, 2 * np.pi))
        y = 0.5 + 0.4 * np.cos(np.pi * f * x + ph) + 0.05 * rng.normal(0, 1, N)
        h = make_feasible(y, 0.5 * float(N))
        c = exact_C5(h)
        if c < bestc:
            bestc = c
            best = h
    return best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    deadline = t0 + total - 5.0
    rng = np.random.default_rng(int(seed) % (1 << 32))

    h0 = _get_initial()
    if h0 is None:
        h0 = fallback_initial(360)
    n0 = len(h0)
    best_h = h0.copy()
    best_c = exact_C5(h0)

    # Phase 1: Main optimization at initial resolution
    phase1_end = min(deadline, t0 + total * 0.45)
    h_opt, c_opt = optimize_full(best_h, phase1_end, rng)
    if c_opt < best_c:
        best_h, best_c = h_opt, c_opt

    # Phase 2: Block shake + re-optimize
    if time.time() < deadline - 60:
        h_shake, c_shake = block_shake(best_h, rng, n_trials=80)
        if c_shake < best_c:
            best_h, best_c = h_shake, c_shake
        shake_end = min(deadline - 30, time.time() + (deadline - time.time()) * 0.3)
        h_opt, c_opt = optimize_full(best_h, shake_end, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 3: Try lower resolution
    if time.time() < deadline - 40:
        N_low = max(200, int(n0 * 0.67))
        h_low = resample_h(best_h, N_low)
        low_end = min(deadline - 10, time.time() + (deadline - time.time()) * 0.35)
        h_opt_low, c_opt_low = optimize_full(h_low, low_end, rng)
        if c_opt_low < best_c:
            best_h, best_c = h_opt_low, c_opt_low
        else:
            h_up = resample_h(h_opt_low, len(best_h))
            c_up = exact_C5(h_up)
            if c_up < best_c:
                best_h, best_c = h_up, c_up

    # Phase 4: Second shake + optimize at current best
    if time.time() < deadline - 30:
        h_shake2, c_shake2 = block_shake(best_h, rng, n_trials=60)
        if c_shake2 < best_c:
            best_h, best_c = h_shake2, c_shake2
        final_end = min(deadline - 3, time.time() + (deadline - time.time()) * 0.5)
        h_opt, c_opt = optimize_full(best_h, final_end, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    # Phase 5: Final polish with very small tau
    if time.time() < deadline - 5:
        n = len(best_h)
        target = 0.5 * float(n)
        for tau in [0.0002, 0.0001]:
            if time.time() > deadline - 2:
                break
            x, c = _lbfgs_stage(best_h, tau, target, 30.0, deadline - 1, 200)
            if c < best_c:
                best_h, best_c = x, c

    # Final feasibility check
    best_h = make_feasible(best_h, 0.5 * float(len(best_h)))
    best_c = exact_C5(best_h)
    if not np.isfinite(best_c):
        best_h = make_feasible(np.full(n0, 0.5), 0.5 * float(n0))
        best_c = exact_C5(best_h)

    return best_h, float(best_c), int(len(best_h))
