import numpy as np
import time
from scipy.optimize import linprog
from scipy.optimize import minimize as sci_minimize

def compute_corr(h):
    """Compute raw correlation values (not yet multiplied by dx)."""
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    """Compute the C5 objective value."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def compute_c_norm(h):
    """Compute normalized correlation values c[k] * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return corr * dx

def project_to_box_sum(p, n):
    """Project p onto {0 <= x <= 1, sum(x) = n/2}."""
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def corr_gradient(h, m):
    """Compute gradient of c[k] (raw, not *dx) with respect to h, for lag m.
    lag m = k - (n-1), so m ranges from -(n-1) to (n-1).
    """
    n = len(h)
    grad = np.zeros(n)
    if m > 0:
        # c = sum_{j=0}^{n-1-m} h[j]*(1-h[j+m])
        # d/dh[j] = (1-h[j+m]) for j in [0, n-1-m]
        #          - h[j-m] for j in [m, n-1]
        js = np.arange(0, n - m)
        grad[js] += (1.0 - h[js + m])
        js2 = np.arange(m, n)
        grad[js2] -= h[js2 - m]
    elif m < 0:
        p = -m
        # c = sum_{j=p}^{n-1} h[j]*(1-h[j-p])
        # d/dh[j] = (1-h[j-p]) for j in [p, n-1]
        #          - h[j+p] for j in [0, n-1-p]
        js = np.arange(p, n)
        grad[js] += (1.0 - h[js - p])
        js2 = np.arange(0, n - p)
        grad[js2] -= h[js2 + p]
    else:
        # c = sum_j h[j]*(1-h[j]) = sum h[j] - sum h[j]^2
        # d/dh[j] = 1 - 2*h[j]
        grad[:] = 1.0 - 2.0 * h
    return grad

def solve_lp_direction(h, c_norm, f, radius, eta=0.01):
    """Solve the trust-region LP to get a direction.
    
    h: current point (n,)
    c_norm: normalized correlation values, shape (2n-1,)
    f: current objective = max(c_norm)
    radius: trust region radius
    eta: regularization parameter
    
    Returns: delta (n,) or None if LP infeasible
    """
    n = len(h)
    num_lags = 2 * n - 1
    
    # Variables: [delta_0, ..., delta_{n-1}, s_0, ..., s_{n-1}, t]
    # Total: 2n + 1 variables
    nv = 2 * n + 1
    
    # Objective: minimize t + eta * sum(s_i)
    c_obj = np.zeros(nv)
    c_obj[n:2*n] = eta  # s_i coefficients
    c_obj[2*n] = 1.0    # t coefficient
    
    # Inequality constraints: A_ub @ x <= b_ub
    # 1. delta_i <= s_i  =>  delta_i - s_i <= 0
    # 2. -delta_i <= s_i =>  -delta_i - s_i <= 0
    A_parts = []
    b_parts = []
    
    # |delta_i| <= s_i constraints
    # delta_i - s_i <= 0
    row = np.zeros(nv)
    row[0:n] = 1.0
    row[n:2*n] = -1.0
    # This is one row per i... let me use matrix form
    # Actually for efficiency, let's build the full matrix
    
    # |delta_i| <= s_i: 2n constraints
    A_id_s = np.zeros((2*n, nv))
    # delta_i - s_i <= 0 for each i
    idx = np.arange(n)
    A_id_s[np.arange(n), idx] = 1.0
    A_id_s[np.arange(n), n + idx] = -1.0
    # -delta_i - s_i <= 0 for each i
    A_id_s[n + np.arange(n), idx] = -1.0
    A_id_s[n + np.arange(n), n + idx] = -1.0
    b_id_s = np.zeros(2 * n)
    
    # Trust region: sum(s_i) <= radius
    # s_0 + ... + s_{n-1} <= radius
    A_tr = np.zeros((1, nv))
    A_tr[0, n:2*n] = 1.0
    b_tr = np.array([radius])
    
    # Correlation linearization: c_norm[k] + grad_k · delta <= f + t
    # => grad_k · delta - t <= f - c_norm[k]
    A_corr = np.zeros((num_lags, nv))
    b_corr = np.zeros(num_lags)
    for k in range(num_lags):
        m = k - (n - 1)
        g = corr_gradient(h, m) * (2.0 / n)  # normalized gradient
        A_corr[k, 0:n] = g
        A_corr[k, 2*n] = -1.0
        b_corr[k] = f - c_norm[k]
    
    # Box constraints on delta: -h[i] <= delta[i] <= 1-h[i]
    # delta[i] <= 1-h[i]
    A_box1 = np.zeros((n, nv))
    A_box1[np.arange(n), np.arange(n)] = 1.0
    b_box1 = 1.0 - h
    
    # -delta[i] <= h[i]
    A_box2 = np.zeros((n, nv))
    A_box2[np.arange(n), np.arange(n)] = -1.0
    b_box2 = h
    
    # Sum constraint: sum(delta) = 0 (handled as equality)
    A_eq = np.zeros((1, nv))
    A_eq[0, 0:n] = 1.0
    b_eq = np.array([0.0])
    
    # Assemble
    A_ub = np.vstack([A_id_s, A_tr, A_corr, A_box1, A_box2])
    b_ub = np.concatenate([b_id_s, b_tr, b_corr, b_box1, b_box2])
    
    # Bounds: s_i >= 0, t >= 0 (t is unbounded above in principle but >= 0 makes sense)
    bounds = [(None, None)] * n + [(0, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            delta = res.x[0:n].copy()
            if np.all(np.isfinite(delta)):
                return delta
        elif res.status == 2:  # infeasible
            pass
    except Exception:
        pass
    
    return None

def line_search_lp(h, delta, c_current, max_steps=8):
    """Line search for LP direction. Returns (h_new, c_new, alpha) or (h, c_current, 0)."""
    n = len(h)
    best_h = h.copy()
    best_c = c_current
    best_alpha = 0.0
    
    alphas = [1.0, 0.7, 0.5, 0.35, 0.2, 0.1, 0.05]
    for alpha in alphas:
        h_new = h + alpha * delta
        h_new = project_to_box_sum(h_new, n)
        c_new = compute_c5(h_new)
        if c_new < best_c - 1e-14:
            best_h = h_new
            best_c = c_new
            best_alpha = alpha
    
    return best_h, best_c, best_alpha

def smooth_max_gradient(h, tau):
    """Compute gradient of smooth max approximation."""
    n = len(h)
    c_norm = compute_c_norm(h)
    dx = 2.0 / n
    f = float(np.max(c_norm))
    
    # Softmax weights
    shifted = c_norm - f
    w = np.exp(tau * shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        return np.zeros(n)
    w = w / w_sum
    
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        if w[k] < 1e-14:
            continue
        m = k - (n - 1)
        g = corr_gradient(h, m) * dx
        grad += w[k] * g
    return grad

def local_optimize_lp(h0, time_limit, initial_radius=0.05, min_radius=0.002):
    """Main LP-based local optimizer."""
    n = len(h0)
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c = compute_c5(h)
    c_current = best_c
    
    radius = initial_radius
    stall_count = 0
    t0 = time.time()
    
    while time.time() - t0 < time_limit - 2:

        c_norm = compute_c_norm(h)
        f = float(np.max(c_norm))
        
        delta = solve_lp_direction(h, c_norm, f, radius, eta=0.01)
        
        if delta is None or np.linalg.norm(delta) < 1e-12:
            # Fallback: smooth gradient
            tau = 50.0 * (1.0 + stall_count * 0.5)
            grad = smooth_max_gradient(h, tau)
            if np.linalg.norm(grad) < 1e-12:
                # Escape: random perturbation
                rng = np.random.RandomState(int(time.time() * 1000) % 100000)
                h = project_to_box_sum(h + rng.randn(n) * 0.01, n)
                c_current = compute_c5(h)
                if c_current < best_c:
                    best_c = c_current
                    best_h = h.copy()
                stall_count += 1
                if stall_count > 20:
                    break
                continue

            step_size = radius / (np.linalg.norm(grad) + 1e-12)
            delta = -grad * step_size

import numpy as np
import time
from scipy.optimize import linprog

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = compute_corr(h)
    return float(np.max(corr) * dx)

def compute_c_norm(h):
    n = len(h)
    dx = 2.0 / n
    return compute_corr(h) * dx

def project_box_sum(p, n):
    target = n / 2.0
    p = np.clip(p, -2.0, 3.0)
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-11:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def corr_grad_component(h, m):
    n = len(h)
    grad = np.zeros(n)
    if m > 0:
        js = np.arange(0, n - m)
        grad[js] += (1.0 - h[js + m])
        js2 = np.arange(m, n)
        grad[js2] -= h[js2 - m]
    elif m < 0:
        p = -m
        js = np.arange(p, n)
        grad[js] += (1.0 - h[js - p])
        js2 = np.arange(0, n - p)
        grad[js2] -= h[js2 + p]
    else:
        grad[:] = 1.0 - 2.0 * h
    return grad

def all_normalized_gradients(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for k in range(2 * n - 1):
        m = k - (n - 1)
        grads[k] = corr_grad_component(h, m) * dx
    return grads

def solve_lp_direction(h, c_norm, radius):

    num_lags = 2 * n - 1
    nv = n + 1
    c_obj = np.zeros(nv)
    c_obj[n] = 1.0
    grads = all_normalized_gradients(h)
    A_ub = grads.copy()
    A_ub[:, n] = -1.0
    b_ub = -c_norm.copy()
    A_eq = np.zeros((1, nv))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = []
    for i in range(n):
        lo = max(-h[i], -radius)
        hi = min(1.0 - h[i], radius)
        if lo > hi:
            lo = hi = 0.0
        bounds.append((lo, hi))
    bounds.append((None, None))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'dual_feasibility_tolerance': 1e-9})
        if res.status == 0:
            delta = res.x[:n].copy()
            if np.all(np.isfinite(delta)) and float(np.linalg.norm(delta)) > 1e-13:
                return delta
    except Exception:
        pass
    return None

def smooth_max_grad(h, tau):
    n = len(h)
    c_norm = compute_c_norm(h)
    f = float(np.max(c_norm))
    shifted = c_norm - f
    max_shift = float(np.max(shifted))
    w = np.exp(tau * (shifted - max_shift))
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grads = all_normalized_gradients(h)
    return w @ grads

    return best_h, best_c

            h_test[i] -= mid
            h_test[j] += mid
            h_test = project_box_sum(h_test, n)
            c_test = compute_c5(h_test)
            if c_test < best_c_s - 1e-14:
                best_c_s = c_test
                best_s = mid
            if c_test <= c_cur:
                lo_s = mid
            else:
                hi_s = max_s - mid
        if best_s > 1e-8 and best_c_s < best_c - 1e-13:
            h[i] -= best_s
            h[j] += best_s
            h = project_box_sum(h, n)
            c_cur = best_c_s
            best_h = h.copy()
            best_c = c_cur
        if attempt > 50000:
            break
    return best_h, best_c

def make_wave_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = 0.5
    h += 0.25 * np.sin(2.0 * np.pi * x)
    h += 0.15 * np.sin(4.0 * np.pi * x + 0.7)
    h += 0.08 * np.sin(6.0 * np.pi * x + 1.3)
    h += 0.04 * np.sin(9.0 * np.pi * x + 2.0)
    h += rng.randn(n) * 0.02
    return project_box_sum(h, n)

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    h_new = np.interp(x_new, x_old, h_old)
    return project_box_sum(h_new, n_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    best_h = None
    best_c = float('inf')

    try:
        h0 = np.array(initial_h_values, dtype=float)
        n0 = len(h0)
    except (NameError, UnboundLocalError):
        n0 = 81
        h0 = make_wave_start(n0, seed)

    h0 = project_box_sum(h0, n0)
    c0 = compute_c5(h0)
    best_h = h0.copy()
    best_c = c0

    elapsed = time.time() - t_start
    remaining = budget_s - elapsed

    # Phase 1: LP optimization on parent
    alloc1 = min(remaining * 0.35, 400)
    if alloc1 > 20:
        h_opt, c_opt = local_optimize_lp(best_h, alloc1)
        if c_opt < best_c - 1e-13:
            best_h = h_opt
            best_c = c_opt

    # Phase 2: retry with different radius/escape from best
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 40:
        alloc2 = min(remaining * 0.2, 150)
        if alloc2 > 15:
            h_opt, c_opt = local_optimize_lp(best_h, alloc2)
            if c_opt < best_c - 1e-13:
                best_h = h_opt
                best_c = c_opt

    # Phase 3: different resolutions
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    n_parent = len(best_h)
    n_trials = []
    if n_parent <= 100:
        n_trials = [n_parent + 20, n_parent + 40]
    else:
        n_trials = [max(60, n_parent - 20), n_parent + 20]
    for n_try in n_trials:
        if n_try > 250 or n_try < 40:
            continue
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed
        if remaining < 30:
            break
        h_res = resample_h(best_h, n_try)
        alloc = min(remaining * 0.25, 80)
        if alloc < 15:
            alloc = max(alloc, 10)
        h_opt, c_opt = local_optimize_lp(h_res, alloc)
        if c_opt < best_c - 1e-13:
            best_h = h_opt
            best_c = c_opt

    # Phase 4: pair-swap polish
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 20:
        alloc4 = min(remaining * 0.3, 120)
        if alloc4 > 10:
            h_ps, c_ps = pair_swap_polish(best_h, alloc4)
            if c_ps < best_c - 1e-13:
                best_h = h_ps
                best_c = c_ps

    # Phase 5: final LP refinement
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 15:
        h_opt, c_opt = local_optimize_lp(best_h, remaining - 3)
        if c_opt < best_c - 1e-13:
            best_h = h_opt
            best_c = c_opt

    # Phase 6: final pair-swap
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed
    if remaining > 10:
        h_ps, c_ps = pair_swap_polish(best_h, remaining - 2)
        if c_ps < best_c - 1e-13:
            best_h = h_ps
            best_c = c_ps

    # Final validation
    best_h = project_box_sum(best_h, len(best_h))
    best_c = compute_c5(best_h)

    return best_h, best_c, len(best_h)
