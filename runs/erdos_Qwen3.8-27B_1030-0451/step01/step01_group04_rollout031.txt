import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_dx(h):
    """Return correlation values already scaled by dx."""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def active_gradient(h, c_dx, tau):
    """Compute weighted gradient of max(c_dx) using softmax weights."""
    N = len(h)
    cmax = np.max(c_dx)
    exponents = np.clip((c_dx - cmax) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    if total < 1e-30:
        return np.zeros(N)
    w = w / total
    
    g = np.zeros(N)
    for r in range(len(c_dx)):
        a = w[r]
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[:n] += a * (1.0 - h[d:d+n])
                g[d:] -= a * h[:n]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:dd+n])
                g[dd:] -= a * h[:n]
    return g - np.mean(g)

def feasible_interval(h, d):
    """Find the range of lambda for which h + lambda*d is in [0,1].^2"""
    pos = d > 1e-12
    neg = d < -1e-12
    lam_min = -1e10
    lam_max = 1e10
    
    if np.any(pos):
        lam_max = min(lam_max, np.min((1.0 - h[pos]) / d[pos]))
        lam_min = max(lam_min, np.max(-h[pos] / d[pos]))
    if np.any(neg):
        lam_max = min(lam_max, np.min(-h[neg] / d[neg]))
        lam_min = max(lam_min, np.max((1.0 - h[neg]) / d[neg]))
    
    return lam_min, lam_max

def line_search(h, d, n_samples=30):
    """Find best lambda along direction d."""
    lam_min, lam_max = feasible_interval(h, d)
    if lam_max <= lam_min:
        return 0.0, compute_c5(h)
    
    # Sample points in the interval
    if lam_max - lam_min < 1e-12:
        best_lam = (lam_min + lam_max) / 2
        best_val = compute_c5(h + best_lam * d)
        return best_lam, best_val
    
    # Use logarithmic spacing near the boundaries and linear in the middle
    candidates = np.linspace(lam_min, lam_max, n_samples)
    # Also add some points near the boundaries
    margin = 0.01 * (lam_max - lam_min)
    candidates = np.concatenate([
        candidates,
        [lam_min + margin, lam_max - margin,
         lam_min + 0.001 * margin, lam_max - 0.001 * margin]
    ])
    
    best_lam = 0.0
    best_val = compute_c5(h)
    for lam in candidates:
        h_try = h + lam * d
        val = compute_c5(h_try)
        if val < best_val:
            best_val = val
            best_lam = lam
    return best_lam, best_val

def project_feasible(h, target_sum):
    """Project h to [0,1]^N with sum = target_sum."""
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.argwhere(h < 1.0).flatten()
            if len(idx) > 0:
                for i in idx:
                    add = min(residual, 1.0 - h[i])
                    h[i] += add
                    residual -= add
                    if abs(residual) < 1e-12:
                        break
        else:
            idx = np.argwhere(h > 0.0).flatten()
            if len(idx) > 0:
                for i in idx:
                    sub = min(-residual, h[i])
                    h[i] -= sub
                    residual += sub
                    if abs(residual) < 1e-12:
                        break
    return h

def make_direction_gradient(h, c_dx, tau, N):
