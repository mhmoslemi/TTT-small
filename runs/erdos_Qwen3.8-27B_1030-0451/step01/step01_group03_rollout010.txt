```python
def solve_pair_1d(c, g, t_min, t_max, top_k=20):
    """Solve min_t max_{k in top} (c[k] + t*g[k]) for t in [t_min, t_max]"""
    # Find the top_k lags by current value
    idx = np.argsort(c)[::-1][:top_k]
    c_top = c[idx]
    g_top = g[idx]
    
    # The minimum of max of lines is at a breakpoint or boundary
    # Breakpoints: (c_i - c_j) / (g_j - g_i) for i != j with g_i != g_j
    candidates = [t_min, t_max]
    for a in range(len(c_top)):
        for b in range(a+1, len(c_top)):
            if abs(g_top[a] - g_top[b]) > 1e-15:
                t = (c_top[b] - c_top[a]) / (g_top[b] - g_top[a] - ... )
                # Wait: c[a] + t*g[a] = c[b] + t*g[b]
                # t*(g[a]-g[b]) = c[b] - c[a]
                # t = (c[b] - c[a]) / (g[a] - g[b])
                t = (c_top[b] - c_top[a]) / (g_top[a] - g_top[b])
                if t_min <= t <= t_max:
                    candidates.append(t)
    
    # Also add the point where the "knee" of the full max is
    # Evaluate at all candidates
    best_t = candidates[0]
    best_val = np.inf
    for t in candidates:
        val = np.max(c_top + t * g_top)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Verify against full max (not just top_k)
    # The true max might be larger than top_k approximation
    # But for small t changes, top_k should be sufficient
    
    return best_t, best_val
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(c.max())

def _c_array(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def _repair(h, target):
    n = len(h)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(h + mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _dc_dh_pair(h, n, i, j):
    """Compute g_k = dc_k/dh[i] - dc_k/dh[j] for all k=0..2n-2"""
    g = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        d = idx - (n - 1)
        if d >= 0:
            v_i = 0.0
            if 0 <= i <= n - 1 - d:
                v_i += (1.0 - h[i + d])
            if d <= i <= n - 1:
                v_i -= h[i - d]
            v_j = 0.0
            if 0 <= j <= n - 1 - d:
                v_j += (1.0 - h[j + d])
            if d <= j <= n - 1:
                v_j -= h[j - d]
        else:
            d2 = -d
            v_i = 0.0
            if d2 <= i <= n - 1:
                v_i += (1.0 - h[i - d2])
            if 0 <= i <= n - 1 - d2:
                v_i -= h[i + d2]
            v_j = 0.0
            if d2 <= j <= n - 1:
                v_j += (1.0 - h[j - d2])
            if 0 <= j <= n - 1 - d2:
                v_j -= h[j + d2]
        g[idx] = v_i - v_j
    return g

def _solve_1d(c, g, t_min, t_max, top_k=30):
    """Solve min_{t in [t_min,t_max]} max_k (c[k] + t*g[k])"""
    if t_max < t_min:
        return t_min, np.inf
    if t_max - t_min < 1e-15:
        t = t_min
        return t, float(np.max(c + t * g))
    idx = np.argsort(c)[::-1][:top_k]
    c_s = c[idx]
    g_s = g[idx]
    candidates = [t_min, t_max]
    for a in range(len(c_s)):
        for b in range(a + 1, len(c_s)):
            dg = g_s[a] - g_s[b]
            if abs(dg) > 1e-15:
                t = (c_s[b] - c_s[a]) / dg
                if t_min - 1e-9 <= t <= t_max + 1e-9:
                    t = np.clip(t, t_min, t_max)
                    candidates.append(t)
    best_t = t_min
    best_val = np.inf
    for t in candidates:
        val = float(np.max(c + t * g))
        if val < best_val:
            best_val = val
            best_t = t
    return best_t, best_val

def _subgrad_from_active(c, h, n, active_idx):
    """Compute average gradient over active lags"""
    grad = np.zeros(n)
    ka = len(active_idx)
    if ka == 0:
        return grad
    for k_idx in active_idx:
        d = k_idx - (n - 1)
        if d >= 0:
            lo1, hi1 = 0, n - 1 - d
            lo2, hi2 = d, n - 1
            if hi1 >= lo1:
                grad[lo1:hi1 + 1] += (1.0 - h[lo1 + d:hi1 + 1 + d])
            if hi2 >= lo2:
                grad[lo2:hi2 + 1] -= h[lo2 - d:hi2 + 1 - d]
        else:
            d2 = -d
            lo1, hi1 = d2, n - 1
            lo2, hi2 = 0, n - 1 - d2
            if hi1 >= lo1:
                grad[lo1:hi1 + 1] += (1.0 - h[lo1 - d2:hi1 + 1 - d2])
            if hi2 >= lo2:
                grad[lo2:hi2 + 1] -= h[lo2 + d2:hi2 + 1 + d2]
    return grad / ka

def _pair_descent(h, n, target, deadline, max_iters=5000):
    """Exact pair coordinate descent"""
    best_c5 = _eval_c5(h)
    c = _c_array(h)
    current = h.copy()
    no_imp_count = 0
    rng = np.random.default_rng(12345)

    for it in range(max_iters):
        if time.time() > deadline - 10:
            break
        c = _c_array(current)
        cmax = c.max()
        if cmax < best_c5:
            best_c5 = cmax
            no_imp_count = 0
        else:
            no_imp_count += 1
            if no_imp_count > 500:
                # Try random perturbation
                pert = rng.standard_normal(n) * 0.002
                candidate = _repair(current + pert, target)
                cc = _eval_c5(candidate)
                if cc < best_c5:
                    current = candidate
                    best_c5 = cc
                    no_imp_count = 0
            if no_imp_count > 2000:
                break

        # Find active lags
        rel = c - cmax
        active = np.where(rel > -0.001)[0]
        if len(active) == 0:
            active = np.array([np.argmax(c)])

        # Compute subgradient
        sg = _subgrad_from_active(c, current, n, active)
        sg_proj = sg - np.mean(sg)

        # Select candidate pairs
        # Decrease: positions with high positive sg_proj
        # Increase: positions with low (negative) sg_proj
        desc_idx = np.argsort(sg_proj)[::-1][:10]
        inc_idx = np.argsort(sg_proj)[:10]

        best_t = 0.0
        best_improve = 0.0
        best_pair = None

        for ii in desc_idx:
            for jj in inc_idx:
                if ii == jj:
                    continue
                gi = current[ii]
                gj = current[jj]
                t_min = max(-gi, gj - 1.0)
                t_max = min(1.0 - gi, gj)
                if t_max < t_min:
                    continue
                g = _dc_dh_pair(current, n, ii, jj)
                t_opt, val = _solve_1d(c, g, t_min, t_max, top_k=25)
                # Check actual improvement using full c
                new_c_val = float(np.max(c + t_opt * g))
                improve = cmax - new_c_val
                if improve > best_improve:
                    best_improve = improve
                    best_t = t_opt
                    best_pair = (ii, jj)

        # Also try a few random pairs
        for _ in range(4):
            ii = rng.integers(0, n)
            jj = rng.integers(0, n)
            if ii == jj:
                continue
            gi = current[ii]
            gj = current[jj]
            t_min = max(-gi, gj - 1.0)
            t_max = min(1.0 - gi, gj)
            if t_max < t_min:
                continue
            g = _dc_dh_pair(current, n, ii, jj)
            t_opt, val = _solve_1d(c, g, t_min, t_max, top_k=25)
            new_c_val = float(np.max(c + t_opt * g))
            improve = cmax - new_c_val
            if improve > best_improve:
                best_improve = improve
                best_t = t_opt
                best_pair = (ii, jj)

        if best_pair is not None and best_improve > 1e-12:
            ii, jj = best_pair
            current[ii] += best_t
            current[jj] -= best_t
            np.clip(current, 0.0, 1.0, out=current)
            current = _repair(current, target)

    return current, best_c5

def _anneal_pair(h, n, target, deadline, rng):
    """Simulated annealing with pair moves"""
    c = _c_array(h)
    cmax = c.max()
    best_c5 = cmax
    current = h.copy()
    best_h = h.copy()

    T0 = 0.003
    Tend = 0.0001
    n_total = max(200, int((deadline - time.time()) * 30))
    n_total = min(n_total, 20000)

    for it in range(n_total):
        if time.time() > deadline - 10:
            break
        T = T0 * (Tend / T0) ** (it / max(1, n_total))

        ii = rng.integers(0, n)
        jj = rng.integers(0, n)
        if ii == jj:
            continue
        gi = current[ii]
        gj = current[jj]
        t_min = max(-gi, gj - 1.0)
        t_max = min(1.0 - gi, gj)
        if t_max < t_min:
            continue
        t = t_min + rng.random() * (t_max - t_min)
        if abs(t) < 1e-8:
            continue

        new_ii = np.clip(gi + t, 0.0, 1.0)
        new_jj = np.clip(gj - t, 0.0, 1.0)
        # Check bounds are actually satisfied
        if new_ii != gi + t or new_jj != gj - t:
            continue

        g = _dc_dh_pair(current, n, ii, jj)
        new_cmax = float(np.max(c + t * g))
        delta = new_cmax - cmax

        if delta < 0 or rng.random() < np.exp(-delta / max(T, 1e-9)):
            current[ii] = new_ii
            current[jj] = new_jj
            c = c + t * g
            cmax = new_cmax
            if cmax < best_c5 - 1e-12:
                best_c5 = cmax
                best_h = current.copy()
        else:
            # Try a smaller step in the same direction
            pass

    return best_h, best_c5

def _smooth_slsqp(h, n, target, deadline):
    """Smooth SLSQP optimization with tighter schedule"""
    c = _c_array(h)
    spread = max(1e-6, c.max() - c.mean())
    tau0 = max(0.01, 0.4 * spread)
    best_c5 = _eval_c5(h)
    best_h = h.copy()

    n_tau = 18
    ratio = 0.5
    h_curr = h.copy()

    for i in range(n_tau):
        if time.time() > deadline - 5:
            break
        tau = tau0 * (ratio ** i)
        if tau < 0.002:
            tau = 0.002

        def obj(hx):
            cc = _c_array(hx)
            m = cc.max()
            s = np.clip((cc - m) / tau, -400.0, 400.0)
            e = np.exp(s)
            return float(m + tau * np.log(e.sum()))

        def grad(hx):
            cc = _c_array(hx)
            m = cc.max()
            s = np.clip((cc - m) / tau, -400.0, 400.0)
            e = np.exp(s)
            alpha = e / e.sum()
            gr = np.zeros(n)
            omh = 1.0 - hx
            active = np.where(alpha > 1e-10)[0]
            for k in active:
                a = alpha[k]
                d = k - (n - 1)
                lo = max(0, d)
                hi = min(n - 1, n - 1 + d)
                if lo <= hi:
                    gr[lo:hi + 1] += a * omh[lo - d:hi + 1 - d]
                lo2 = max(0, -d)
                hi2 = min(n - 1, n - 1 - d)
                if lo2 <= hi2:
                    gr[lo2:hi2 + 1] -= a * hx[lo2 + d:hi2 + 1 + d]
            return gr * (2.0 / n)

        def cfun(hx):
            return float(np.sum(hx) - target)

        def cjac(hx):
            return np.ones(n)

        constraints = [{'type': 'eq', 'fun': cfun, 'jac': cjac}]
        bounds = [(0.0, 1.0)] * n
        maxiter = 30 if tau > 0.01 else 50

        try:
            res = minimize(obj, h_curr, method='SLSQP', jac=grad,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-10})
            h_curr = _repair(res.x, target)
        except Exception:
            pass

        c5e = _eval_c5(h_curr)
        if c5e < best_c5 - 1e-10:
            best_c5 = c5e
            best_h = h_curr.copy()

    return best_h, best_c5

def _interp_to(h_src, n_new):
    """Interpolate h_src to n_new points and repair"""
    n_src = len(h_src)
    target = n_new / 2.0
    xo = np.linspace(0.0, 2.0, n_src, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(xn, xo, h_src)
    return _repair(h_new, target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8

    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h_parent = np.array(initial_h_values, dtype=np.float64)
    else:
        h_parent = np.full(240, 0.5)

    n = len(h_parent)
    target = n / 2.0
    h_parent = _repair(h_parent, target)
    best_h = h_parent.copy()
    best_c5 = _eval_c5(h_parent)

    rng = np.random.default_rng(seed)

    # Phase 0: Try simple structural transforms
    transforms = [
        np.roll(h_parent, 1),
        np.roll(h_parent, -1),
        np.roll(h_parent, 2),
        h_parent[::-1].copy(),
        np.sort(h_parent),
        np.flip(np.sort(h_parent)).copy(),
    ]
    for tf in transforms:
        tf = _repair(tf.copy(), target)
        cv = _eval_c5(tf)
        if cv < best_c5:
            best_c5 = cv
            best_h = tf.copy()

    # Phase 1: Pair coordinate descent from best
    if time.time() < deadline - 120:
        h_p1, c5_p1 = _pair_descent(best_h, n, target, deadline - 80)
        if c5_p1 < best_c5:
            best_c5 = c5_p1
            best_h = h_p1

    # Phase 2: SLSQP from best
    if time.time() < deadline - 60:
        h_p2, c5_p2 = _smooth_slsqp(best_h, n, target, deadline - 30)
        if c5_p2 < best_c5:
            best_c5 = c5_p2
            best_h = h_p2

    # Phase 3: Pair descent again from updated best
    if time.time() < deadline - 40:
        h_p3, c5_p3 = _pair_descent(best_h, n, target, deadline - 15)
        if c5_p3 < best_c5:
            best_c5 = c5_p3
            best_h = h_p3

    # Phase 4: Annealing from best
    if time.time() < deadline - 30:
        h_p4, c5_p4 = _anneal_pair(best_h, n, target, deadline - 15, rng)
        if c5_p4 < best_c5:
            best_c5 = c5_p4
            best_h = h_p4

    # Phase 5: Try higher resolution
    if time.time() < deadline - 50:
        n2 = min(n * 2, 2400)
        if n2 != n:
            h_hi = _interp_to(best_h, n2)
            c5_hi = _eval_c5(h_hi)
            if c5_hi < best_c5:
                best_c5 = c5_hi
                best_h = h_hi
                n = n2
                target = n / 2.0
            # Try pair descent at higher res
            if time.time() < deadline - 30:
                h_p5, c5_p5 = _pair_descent(h_hi, n2, n2 / 2.0, deadline - 10)
                if c5_p5 < best_c5:
                    best_c5 = c5_p5
                    best_h = h_p5
                    n = n2

    # Phase 6: Final pair descent polish
    if time.time() < deadline - 10:
        n_final = len(best_h)
        tgt_final = len(best_h) / 2.0
        h_fin, c5_fin = _pair_descent(best_h, n_final, tgt_final, deadline - 3, max_iters=2000)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final repair and validation
    n_final = len(best_h)
    tgt = n_final / 2.0
    best_h = _repair(best_h, tgt)
    best_c5 = _eval_c5(best_h)

    assert best_h.shape == (n_final,)
    assert np.all(best_h >= -1e-12) and np.all(best_h <= 1.0 + 1e-12)
    assert abs(best_h.sum() - tgt) < 1e-8

    return (best_h, best_c5, n_final)
```
