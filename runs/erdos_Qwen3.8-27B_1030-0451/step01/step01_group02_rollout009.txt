```python
def compute_B(h, i, j, dx):
    """Compute B_d = dC_d/dh[i] - dC_d/dh[j] for all d."""
    n = len(h)
    ctr = n - 1
    B = np.zeros(2*n - 1)
    for d in range(-(n-1), n):
        # dC_d/dh[i] = dx * [(1-h[i-d])_{valid} - h[i+d]_{valid}]
        v_i = 0.0
        if 0 <= i-d <= n-1:
            v_i += (1.0 - h[i-d])
        if 0 <= i+d <= n-1:
            v_i -= h[i+d]
        # dC_d/dh[j] = dx * [(1-h[j-d])_{valid} - h[j+d]_{valid}]
        v_j = 0.0
        if 0 <= j-d <= n-1:
            v_j += (1.0 - h[j-d])
        if 0 <= j+d <= n-1:
            v_j -= h[j+d]
        B[d + ctr] = dx * (v_i - v_j)
    return B
```

This is O(n) per pair (looping over d). For n=720, that's 1439 iterations per pair. With 720 pairs per sweep, that's ~1M iterations. In Python, this might take ~0.1s per sweep. That's acceptable.

```python
def compute_B_vec(h, i, j, dx):
    n = len(h)
    ctr = n - 1
    B = np.zeros(2*n - 1)
    
    # Term 1: (1-h[i-d]) for d where 0 <= i-d <= n-1
    # d ranges from i-(n-1) to i
    d_lo1 = max(-(n-1), i-(n-1))
    d_hi1 = min(n-1, i)
    if d_lo1 <= d_hi1:
        d_range = np.arange(d_lo1, d_hi1+1)
        idx = i - d_range  # indices into h, from i-d_lo1 to i-d_hi1
        B[d_range + ctr] += (1.0 - h[idx])
    
    # Term 2: -h[i+d] for d where 0 <= i+d <= n-1
    d_lo2 = max(-(n-1), -i)
    d_hi2 = min(n-1, n-1-i)
    if d_lo2 <= d_hi2:
        d_range = np.arange(d_lo2, d_hi2+1)
        idx = i + d_range
        B[d_range + ctr] -= h[idx]
    
    # Term 3: -(1-h[j-d]) for d where 0 <= j-d <= n-1
    d_lo3 = max(-(n-1), j-(n-1))
    d_hi3 = min(n-1, j)
    if d_lo3 <= d_hi3:
        d_range = np.arange(d_lo3, d_hi3+1)
        idx = j - d_range
        B[d_range + ctr] -= (1.0 - h[idx])
    
    # Term 4: +h[j+d] for d where 0 <= j+d <= n-1
    d_lo4 = max(-(n-1), -j)
    d_hi4 = min(n-1, n-1-j)

        d_range = np.arange(d_lo4, d_hi4+1)
        idx = j + d_range
        B[d_range + ctr] += h[idx]
    
    B *= dx
    return B
```

```python
def solve_2d(C, B, hi, hj):
    """
    Solve: minimize max_d (C[d] + delta * B[d])
    subject to: max(-hi, hj-1) <= delta <= min(1-hi, hj)
    Returns optimal delta.
    """
    lo = max(-hi, hj - 1.0)
    hi_bound = min(1.0 - hi, hj)
    if lo >= hi_bound - 1e-14:
        return 0.0
    
    # Bisection
    for _ in range(60):
        mid = 0.5 * (lo + hi_bound)
        vals = C + mid * B
        max_val = vals.max()
        active = vals >= max_val - 1e-10
        if not active.any():
            break
        b_active = B[active]

            hi_bound = mid  # f is increasing at mid, move left
        elif b_active.max() < -1e-12:
            lo = mid  # f is decreasing at mid, move right
        else:
            break  # optimal
    
    opt_delta = 0.5 * (lo + hi_bound)
    # Clamp
    opt_delta = max(lo, min(hi_bound, opt_delta))
    return opt_delta
```

def _project(h, target):

    n = len(h)
    lo = -np.max(h) - 0.1
    hi = 1.0 - np.min(h) + 0.1
    for _ in range(80):
