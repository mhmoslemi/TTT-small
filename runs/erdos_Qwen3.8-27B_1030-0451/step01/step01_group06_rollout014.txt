def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')

    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0

        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(10):
        res = h.sum() - target_sum
        if abs(res) < 1e-14:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0:
                    break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0:
                    break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def compute_los_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    Q = np.correlate(h, 1.0 - h, mode='full') * dx
    M = np.max(Q)
    exp_args = (Q - M) / tau
    exp_vals = np.exp(np.minimum(exp_args, 0.0))
    Z = np.sum(exp_vals)
    if Z < 1e-300:
        top_k = min(5, len(Q))
        top_idx = np.argsort(Q)[-top_k:]
        w = np.zeros(len(Q))
        w[top_idx] = 1.0 / top_k
    else:
        w = exp_vals / Z
    obj = M + tau * np.log(max(Z, 1e-300))
    grad = np.zeros(n)
    n_lags = 2 * n - 1

        wi = w[idx]
        if wi < 1e-14:
            continue
        ell = idx - (n - 1)
        if ell == 0:
            grad += wi * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            nv = n - ell
            grad[ell:] += wi * dx * (1.0 - h[:nv])

            nv = n - m_val
            grad[:nv] += wi * dx * (1.0 - h[m_val:])
            grad[m_val:] += wi * (-dx) * h[:nv]
    return obj, grad

    return x.sum() - _current_target

def _eqjac(x):
    return np.ones(len(x))

    global _current_tau, _current_target
    _current_tau = tau

    _current_target = n / 2.0
    bounds = [(0.0, 1.0)] * n

    try:
        res = minimize(_obj, h_start, jac=_grad, method='SLSQP',
                      bounds=bounds, constraints=cons,
                      options={'maxiter': maxiter, 'ftol': 1e-11})
        return np.clip(res.x, 0.0, 1.0)

        return h_start.copy()
