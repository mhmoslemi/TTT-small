import time
from scipy.optimize import minimize_scalar

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    # Bisection to find shift that makes sum exactly target_sum
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -max(h), 1.0 - min(h)
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h

    N = len(h)
    grad = np.zeros(N)
    d = r - (N - 1)

        grad[i_min:i_max+1] = dx * (1.0 - 2.0 * h[j])
    return grad

def line_search_step(h, a, A, L, Q):
    """Find best delta on the line h + delta*a"""
    N = len(h)
    # Find feasible interval
    pos_mask = a > 1e-15
    neg_mask = a < -1e-15
    
    delta_min = 0.0
    delta_max = 0.0

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    # Use bisection to find the shift
    lo, hi = -max(h.min(), 0.0) - 0.1, 1.0 - min(h.max(), 1.0) + 0.1
    # Actually, we need a range that definitely brackets the solution
    # If we add t to all of h and clip to [0,1], the sum is monotonically increasing in t
    # At t = -min(h), all values >= 0, sum >= 0
    # At t = 1-max(h), all values <= 1, sum <= N
    lo = -h.min() - 0.1
    hi = 1.0 - h.max() + 0.1
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h

def get_correlations(h, a=None):
    N = len(h)
    dx = 2.0 / N
    A = np.correlate(h, 1.0 - h, mode='full') * dx
    if a is not None:
        P_a1 = np.correlate(a, np.ones(N), mode='full') * dx
        P_ha = np.correlate(h, a, mode='full') * dx
        P_ah = np.correlate(a, h, mode='full') * dx
        P_aa = np.correlate(a, a, mode='full') * dx
        L = P_a1 - P_ha - P_ah
        Q = -P_aa
        return A, L, Q
    return A, None, None

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo = -h.min() - 0.01
    hi = 1.0 - h.max() + 0.01
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    res = target_sum - h.sum()
    if abs(res) > 1e-13:
        idx = np.where((h < 1.0) if res > 0 else (h > 0.0))[0]
        if len(idx):
            h[idx[0]] += res * np.sign(res) if abs(res) < 1 else res

def line_search(h, a, A, L, Q):
    """Exact line search along h + delta*a. Returns (best_delta, best_value)."""
    N = len(h)
    pos = a > 1e-14
    neg = a < -1e-14
    dlo, dhi = 0.0, 0.0
    if pos.any():

        dlo = max(dlo, np.max(-h[pos]/a[pos]))
    if neg.any():
        dhi = min(dhi, np.max(-h[neg]/a[neg]))
        dlo = max(dlo, np.min((1-h[neg])/a[neg]))
    dlo += 1e-12
    dhi -= 1e-12
    if dhi - dlo < 1e-14:
        return 0.0, float(A.max())
    curr = A.max()
    
    def F(d):
        return float(np.max(A + d*L + d*d*Q))
    
    # Try a few strategic points first
    candidates = [dlo, dhi, (dlo+dhi)*0.5, (dlo+dhi)*0.25, (dlo+dhi)*0.75]

        if valid.any():
            dmin_quad = dmin_quad[valid]
            # Add a few

            candidates.extend(dmin_quad.tolist())
    
    best_d = 0.0
    best_v = curr
    for d in candidates:
        d = np.clip(d, dlo, dhi)
        v = F(d)
        if v < best_v:
            best_v = v
            best_d = d
    
    # Refine with bounded scalar minimization
    if best_v < curr:
        try:
            res = minimize_scalar(F, bounds=(dlo, dhi), method='bounded',
                                options={'xatol': 1e-9, 'maxiter': 200})
            if res.fun < best_v:
                best_v = res.fun
                best_d = res.x
        except Exception:
            pass
    
    best_v = F(best_d)
    return best_d, best_v

def gradient_direction(h, dx, A, tau):
    """Compute gradient-informed zero-sum direction."""
    N = len(h)
    c_max = A.max()
    exps = np.clip((A - c_max) / tau, -500, 500)
    w = np.exp(exps)
    w /= w.sum()
    
    # g[i] = dx * sum_j w[i+j] * (1-2h[j]) for j=0..N-1
    v = 1.0 - 2.0 * h
    g = np.zeros(N)
    for i in range(N):
        g[i] = dx * np.dot(w[i:i+N], v)
    
    # Descent direction, zero-sum
    a = -g - (-g).mean()  # = -g + g.mean()
    a = -g + g.mean()
    norm = np.linalg.norm(a)
    if norm > 1e-15:
        a /= norm
    return a

def random_direction(rng, N):
    a = rng.standard_normal(N)
    a -= a.mean()
    norm = np.linalg.norm(a)
    if norm > 1e-15:
        a /= norm
    return a

def pair_direction(rng, N, h):
    """Pick a random pair and transfer mass."""
    i = rng.integers(N)
    j = rng.integers(N)
    while j == i:
        j = rng.integers(N)
    a = np.zeros(N)
    a[i] = 1.0
    a[j] = -1.0
    # Already zero-sum
    return a

import numpy as np
import time
from scipy.optimize import minimize_scalar


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo = -h.min() - 0.01
    hi = 1.0 - h.max() + 0.01
    for _ in range(60):
        mid = (lo + hi) * 0.5
        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    res = target_sum - h.sum()
    if abs(res) > 1e-13:
        if res > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx):
                h[idx[0]] = min(1.0, h[idx[0]] + res)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx):
                h[idx[0]] = max(0.0, h[idx[0]] + res)
    return h


def do_line_search(h, a, A, L, Q):
    N = len(h)
    pos = a > 1e-14
    neg = a < -1e-14
    dlo, dhi = 0.0, 0.0
    if pos.any():
        dhi = min(dhi, np.min((1.0 - h[pos]) / a[pos]))
        dlo = max(dlo, np.max(-h[pos] / a[pos]))
    if neg.any():
        dhi = min(dhi, np.max(-h[neg] / a[neg]))
        dlo = max(dlo, np.min((1.0 - h[neg]) / a[neg]))
    dlo += 1e-12
    dhi -= 1e-12
    if dhi - dlo < 1e-14:
        return 0.0, float(np.max(A))

    curr = float(np.max(A))

    def F(d):
        return float(np.max(A + d * L + d * d * Q))

    candidates = [dlo, dhi, 0.5 * (dlo + dhi), 0.25 * (dlo + dhi), 0.75 * (dlo + dhi)]
    neg_Q = Q < -1e-15
    if neg_Q.any():
        dmin_q = -L[neg_Q] / (2.0 * Q[neg_Q])
        valid = (dmin_q > dlo) & (dmin_q < dhi)
        if valid.any():
            dmin_q = dmin_q[valid][:30]
            candidates.extend(dmin_q.tolist())

    best_d = 0.0
    best_v = curr
    for d in candidates:
        d = float(np.clip(d, dlo, dhi))
        v = F(d)
        if v < best_v:
            best_v = v
            best_d = d

    if best_v < curr - 1e-14:
        try:
            res = minimize_scalar(F, bounds=(dlo, dhi), method='bounded',
                                  options={'xatol': 1e-10, 'maxiter': 300})
            if res.fun < best_v:
                best_v = res.fun
                best_d = res.x
        except Exception:
            pass

    best_v = F(best_d)
    return best_d, best_v


def gradient_direction(h, dx, A, tau):
    N = len(h)
    c_max = A.max()
    exps = np.clip((A - c_max) / tau, -500.0, 500.0)
    w = np.exp(exps)
    w /= w.sum()
    v = 1.0 - 2.0 * h
    g = np.zeros(N)
    for i in range(N):
        g[i] = dx * np.dot(w[i:i + N], v)
    a = -g + g.mean()
    norm = np.linalg.norm(a)
    if norm > 1e-15:
        a /= norm
    return a


def random_direction(rng, N):
    a = rng.standard_normal(N)
    a -= a.mean()
    norm = np.linalg.norm(a)
    if norm > 1e-15:
        a /= norm
    return a


def pair_direction(rng, N):
    i = int(rng.integers(N))
    j = int(rng.integers(N))
    while j == i:
        j = int(rng.integers(N))
    a = np.zeros(N)
    a[i] = 1.0
    a[j] = -1.0
    return a


def density_direction(h, N):
    impurity = h * (1.0 - h)
    a = impurity - impurity.mean()
    norm = np.linalg.norm(a)
    if norm > 1e-15:
        a /= norm
    return a

        P_ah = np.correlate(a, h, mode='full') * dx

        Q = -P_aa

def downsample(h, N_old, N_new):
