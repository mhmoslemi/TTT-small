import numpy as np

def make_feasible(y, target):
    y = np.asarray(y, dtype=float).ravel()
    n = y.size
    if n == 0:
        return y
    target = float(target)
    y = np.clip(y, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) < 1e-12:
        return y
    if r > 0:
        idx = np.where(y < 1.0)[0]
        if idx.size:
            order = idx[np.argsort(y[idx])]
            rem = r
            for i in order:
                inc = min(1.0 - y[i], rem)
                y[i] += inc
                rem -= inc
                if rem <= 1e-14:
                    break
    else:
        rpos = -r
        idx = np.where(y > 0.0)[0]
        if idx.size:
            order = idx[np.argsort(y[idx])[::-1]]
            rem = rpos
            for i in order:
                dec = min(y[i], rem)
                y[i] -= dec
                rem -= dec
                if rem <= 1e-14:
                    break
    return np.clip(y, 0.0, 1.0)

    n = len(h)
    dx = 2.0 / float(n)
    C = np.correlate(h, 1.0 - h, mode='full')
    C *= dx
    return C
