    n = len(h)

def smooth_max_grad(h, tau):
    """
    Gradient of the smooth max of c[k] where c = correlate(h, 1-h, 'full') * dx.
    Returns (smooth_max_value, gradient).
    """
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)

    alpha = e / e.sum()
    obj = m + tau * np.log(e.sum())

def aug_lag_optimize(h0, n, deadline, rho=10.0, n_outer=20, n_inner=100):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h = h0.copy()
    lam = 0.0
    tau = 0.03
    
    for outer in range(n_outer):
        if time.time() > deadline - 10:
            break
        # Update tau (anneal)
        tau = max(0.005, tau * 0.8)
        
        # Inner: minimize augmented Lagrangian with L-BFGS-B
        # L(h) = smooth_max(h, tau) + lam*(sum(h)-target) + (rho/2)*(sum(h)-target)^2
        # grad L = grad smooth_max + (lam + rho*(sum(h)-target)) * ones
        
        def aug_obj(h):
            obj, g = smooth_max_grad(h, tau)
            s = h.sum() - target
            return obj + lam * s + 0.5 * rho * s * s
        
        def aug_grad(h):
            obj, g = smooth_max_grad(h, tau)
            s = h.sum() - target
            g = g + (lam + rho * s)
            return g
        
        from scipy.optimize import minimize
        res = minimize(aug_obj, h, method='L-BFGS-B', jac=aug_grad,
                      bounds=[(0,1)]*n, options={'maxiter': n_inner, 'ftol': 1e-14})
        h = res.x
        h = project_feasible(h, target)  # Ensure feasibility
        
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update multiplier
        s = h.sum() - target
        lam += rho * s
    
    return best_h, best_c5

For coordinate i, the 1D problem is:
minimize max{ -v² + v + gamma,  max_{k≠0} (alpha_k * v + beta_k) }  over v ∈ [0,1]

I can solve this by:
1. Collect all "breakpoints" (intersections of pairs of functions) in [0,1]
2. Evaluate f(v) = max of all functions at each breakpoint and at 0, 1
3. Return the v with minimum f(v)

The number of functions is 2n-1 (about 479 for n=240). The number of breakpoints is O(n²) ≈ 230K. Evaluating f at each is O(n). Total: O(n³) ≈ 55M per coordinate. For 240 coordinates: 13B. Too slow.

Better: use bisection on the value t. For a given t, check if there exists v with max_k c[k](v) ≤ t. This is O(n) per check (compute the feasible interval for v). Binary search on t: O(n log(1/eps)) per coordinate. Total: O(n² log(1/eps)) per sweep. For n=240, that's about 57K * 30 ≈ 1.7M per sweep. Very fast!

def coord_1d_solve(h, i):
    """Solve the 1D subproblem for coordinate i. Returns optimal v."""
    n = len(h)
    dx = 2.0 / n
    v_old = h[i]
    
    # Compute alpha_k, beta_k for all k, and gamma for k=0
    # c[k] = S_k - R_k
    # S_k = sum of h[m] for m in [max(0,k), min(n-1, n-1+k)]
    # R_k = sum h[j+k]*h[j] for valid j
    
    # For the 1D problem, we need c[k] as a function of v = h[i]
    # c[k](v) = alpha_k * v + beta_k for k≠0
    # c[0](v) = -v^2 + v + gamma
    
    # Compute S_k for all k (independent of v, except the v contribution)
    # S_k = sum of h over positions [max(0,k), min(n-1, n-1+k)]
    # The contribution of h[i]=v to S_k is v if i is in that range
    
    # R_k = sum h[j+k]*h[j]
    # Contribution of v: v*h[i-k] (if 0≤i-k<n) + v*h[i+k] (if 0≤i+k<n) [for k≠0]
    # For k=0: v^2
    
    # Let me compute the "base" values (with h[i] set to 0) and the coefficients.
    
    # Base c[k] with h[i]=0:
    # S_k_base = sum of h[m] for m in range, m≠i
    # R_k_base = sum h[j+k]*h[j] where neither j nor j+k equals i
    
    # alpha_k = [i in S_k range] - h[i+k]*[0≤i+k<n] - h[i-k]*[0≤i-k<n]  (k≠0)
    # beta_k = c[k] - v_old * alpha_k  (the base value)
    
    # For k=0: gamma = c[0] - v_old + v_old^2 = (S_0 - v_old) - (R_0 - v_old^2) = S_0' - R_0'
    # where S_0' = sum h[j] for j≠i, R_0' = sum h[j]^2 for j≠i
    # c[0](v) = (S_0' + v) - (R_0' + v^2) = -v^2 + v + (S_0' - R_0')
    # gamma = S_0' - R_0'
    
    # Let me compute these:
    S0_prime = h.sum() - v_old
    R0_prime = (h**2).sum() - v_old**2
    gamma = S0_prime - R0_prime
    
    # For k≠0:
    # alpha_k = inS_k - h[i+k]*valid_right - h[i-k]*valid_left
    # where inS_k = 1 if max(0,k) ≤ i ≤ min(n-1, n-1+k)
    # valid_right = 1 if 0 ≤ i+k < n
    # valid_left = 1 if 0 ≤ i-k < n
    
    # beta_k = c[k]_current - v_old * alpha_k
    # But c[k]_current includes the contribution of h[i]=v_old already.
    # c[k] = alpha_k * v_old + beta_k, so beta_k = c[k] - alpha_k * v_old
    
    # Compute c for all k:
    c_all = np.correlate(h, 1.0 - h, 'full') * dx
    # c_all has length 2n-1, indexed by k_idx = k + (n-1) where k is the lag
    
    # For each k_idx, compute alpha:
    alphas = np.zeros(2*n-1)
    for k_idx in range(2*n-1):
        d = k_idx - (n - 1)  # actual lag
        inS = 1.0 if (max(0,d) <= i <= min(n-1, n-1+d)) else 0.0
        vr = h[i+d] if (0 <= i+d < n) else 0.0
        vl = h[i-d] if (0 <= i-d < n) else 0.0
        if d == 0:
            alphas[k_idx] = 0  # handled separately (quadratic)
        else:
            alphas[k_idx] = inS - vr - vl
    
    # beta for k≠0:
    betas = c_all.copy()
    for k_idx in range(2*n-1):
        if k_idx != n - 1:  # k≠0
            betas[k_idx] = c_all[k_idx] - alphas[k_idx] * v_old
    
    # Now solve: min_v max{ -v²+v+gamma,  max_{k≠0} (alpha_k*v + beta_k) }
    # Use bisection on t.
    
    def feasible(t):
        """Check if there exists v in [0,1] with all c[k](v) <= t."""
        # Constraint from k=0: -v² + v + gamma <= t
        #   v² - v + (t - gamma) >= 0
        #   (v - 1/2)² >= 1/4 - (t - gamma) = 1/4 - t + gamma
        #   If 1/4 - t + gamma <= 0, i.e., t >= gamma + 1/4: no constraint
        #   Otherwise: v <= (1 - sqrt(1-4(t-gamma)))/2 or v >= (1 + sqrt(1-4(t-gamma)))/2
        #              (using sqrt(1-4(t-gamma)) = sqrt(4*(1/4 - t + gamma)))
        
        # Constraint from k≠0: alpha_k * v + beta_k <= t
        #   If alpha_k > 0: v <= (t - beta_k) / alpha_k
        #   If alpha_k < 0: v >= (t - beta_k) / alpha_k
        #   If alpha_k = 0: need beta_k <= t
        
        lo = 0.0
        hi = 1.0
        
        # From k≠0 constraints:
        neg_alpha = alphas[(alphas < 0) & (np.arange(2*n-1) != n-1)]
        pos_alpha = alphas[(alphas > 0) & (np.arange(2*n-1) != n-1)]
        zero_alpha = alphas[(np.abs(alphas) < 1e-15) & (np.arange(2*n-1) != n-1)]
        
        if len(zero_alpha) > 0:
            if np.max(betas[zero_alpha]) > t:
                return False, None, None
        
        if len(neg_alpha) > 0:
            # v >= (t - beta_k) / alpha_k for each negative alpha_k
            lower = (t - betas[neg_alpha]) / neg_alpha
            lo = max(lo, np.max(lower))
        
        if len(pos_alpha) > 0:
            # v <= (t - beta_k) / alpha_k for each positive alpha_k
            upper = (t - betas[pos_alpha]) / pos_alpha
            hi = min(hi, np.min(upper))
        
        if lo > hi:
            return False, None, None
        
        # From k=0 constraint: -v²+v+gamma <= t
        # v² - v + (t-gamma) >= 0
        disc = 1.0 - 4.0*(t - gamma)
        if disc <= 0:
            # t >= gamma + 1/4: no constraint from k=0
            pass
        else:
            sqrt_d = np.sqrt(disc)
            v_lo_quad = (1.0 - sqrt_d) / 2.0
            v_hi_quad = (1.0 + sqrt_d) / 2.0
            # Need v <= v_lo_quad OR v >= v_hi_quad
            # Combined with lo <= v <= hi:
            # Option 1: lo <= v <= min(hi, v_lo_quad)
            # Option 2: max(lo, v_hi_quad) <= v <= hi
            
            opt1_hi = min(hi, v_lo_quad)
            opt2_lo = max(lo, v_hi_quad)
            
            if lo <= opt1_hi:
                return True, lo, opt1_hi  # feasible in [lo, opt1_hi]
            elif opt2_lo <= hi:
                return True, opt2_lo, hi  # feasible in [opt2_lo, hi]
            else:
                return False, None, None
        
        return True, lo, hi
    
    # Bisection on t
    # Lower bound: min of all c[k](v) over v, k (very negative)
    # Upper bound: max of c[k](0) and c[k](1)
    t_lo = -1.0
    t_hi = np.max(c_all) + 1.0
    
    # Actually, t_lo should be below the minimum possible value of max_k c[k](v)
    # and t_hi should be above it.
    # A safe upper bound: max of c[k](v_old) which is the current C5 value
    t_hi = max(c_all.max(), 0.5)
    # Lower bound: 0 (C5 can't be negative... actually it can if c[k] < 0)
    # c[k] = sum h[j+k]*(1-h[j]) ≥ 0 since h ∈ [0,1]! So c[k] ≥ 0 always.
    t_lo = 0.0
    
    for _ in range(60):
        t_mid = (t_lo + t_hi) * 0.5
        ok, _, _ = feasible(t_mid)
        if ok:
            t_hi = t_mid
        else:
            t_lo = t_mid
    
    # At t = t_hi (approximately optimal), find the best v
    # Try v = t_hi (no, that's not right)
    # Find a feasible v at t = t_hi * (1 + small)
    t_test = t_hi * 1.001 + 1e-10
    ok, lo_v, hi_v = feasible(t_test)
    if ok:
        v_opt = (lo_v + hi_v) * 0.5
    else:
        v_opt = v_old
    
    return v_opt, t_hi
