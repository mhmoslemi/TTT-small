import time

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            g[i] += 1.0 - h[i + lag]
            g[i + lag] -= h[i]
    else:
        l = -lag
        for i in range(n - l):
            g[i + l] += 1.0 - h[i]
            g[i] -= h[i + l]
    return g

This means the problem is non-convex, which explains why it's hard.

Given non-convexity, we need to be careful. The LP trust region method works because it replaces the non-convex quadratic with a linear approximation (which is always valid as a local model).

def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[:n-lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n-lag]
    else:
        l = -lag
        g[l:] += 1.0 - h[:n-l]
        g[:n-l] -= h[l:]
    return g

import numpy as np
import time
from scipy.optimize import linprog


def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))


def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def grad_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        g[:n - lag] += 1.0 - h[lag:]
        g[lag:] -= h[:n - lag]
    else:
        l = -lag
        g[l:] += 1.0 - h[:n - l]
        g[:n - l] -= h[l:]
    return g


def lp_trust(h0, deadline, max_iters=800, r0=0.04, rmax=0.15):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = r0
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = v.max()
        gap = max(0.002 * vmax, 0.0005)
        active = np.where(v >= vmax - gap)[0]
        if len(active) == 0:
            active = np.array([np.argmax(v)])
        if len(active) > 120:
            order = np.argsort(v[active])
            active = active[order[-120:]]
        A = np.zeros((len(active), n))
        b_lin = np.zeros(len(active))
        for i, k in enumerate(active):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)
            b_lin[i] = A[i] @ h - v[k]
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.0]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None
        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                radius = min(radius * 1.04, rmax)
            else:
                no_improve += 1
                if no_improve >= 4:
                    radius = max(radius * 0.6, 0.003)
                    no_improve = 0
                    if radius <= 0.003:
                        break
        else:
            no_improve += 1
            if no_improve >= 8:
                radius = max(radius * 0.5, 0.003)
                no_improve = 0
                if radius <= 0.003:
                    break
    return best_h, best_c

    best_h = h0.copy()

    stall = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        c_cur = v.max()
        if c_cur < best_c - 1e-12:
            best_c = c_cur
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 300:
                break
        k_star = np.argmax(v)
        lag = k_star - (n - 1)
        g = grad_lag(h, lag)
        gn = np.linalg.norm(g)
        if gn < 1e-14:

        h_new = h - step * g / gn
        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)

            step = min(step * 1.02, 0.04)
        else:
            step *= 0.88
            if step < 1e-7:
                break
    return best_h, best_c


def gold_initial(n):
    phi = (1.0 + np.sqrt(5.0)) / 2.0
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if (i * phi) % 1.0 < 0.5 else 0.0
    n_ones = int(h.sum())
    if n_ones > n // 2:

        for idx in ones[n // 2:]:
            h[idx] = 0.0
    elif n_ones < n // 2:
        zeros = np.where(h == 0)[0]
        for idx in zeros[:n // 2 - n_ones]:
            h[idx] = 1.0
    return h


def transfer_res(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_ext = np.concatenate([h0, h0[:1]])
    x_ext = np.concatenate([x_old, np.array([x_old[-1] + 2.0 / n_old])])

    h_new = np.interp(x_new_ext, x_ext, h_ext)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

                    c_tr = evaluate_c5(h_tr)
                    if c_tr < best_c:
                        best_h = h_tr.copy()
                        best_c = c_tr
                    sub_dl = min(deadline - 10, time.time() + max(20, (deadline - time.time()) * 0.4))
                    h_tr2, c_tr2 = lp_trust(best_h, sub_dl, max_iters=200)
                    if c_tr2 < best_c:
                        best_h, best_c = h_tr2, c_tr2
                    sub_dl2 = min(deadline - 5, time.time() + max(10, (deadline - time.time()) * 0.25))
                    h_tr3, c_tr3 = subgrad_polish(best_h, sub_dl2, max_iters=10000)
                    if c_tr3 < best_c:
                        best_h, best_c = h_tr3, c_tr3
                except Exception:
                    pass

    # Phase 7: Final LP + subgradient
    if time.time() < deadline - 5:
        try:
            h7, c7 = lp_trust(best_h, deadline - 2, max_iters=150)
            if c7 < best_c:
                best_h, best_c = h7, c7
            h8, c8 = subgrad_polish(best_h, deadline - 1, max_iters=5000)
            if c8 < best_c:
                best_h, best_c = h8, c8
        except Exception:
            pass

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
