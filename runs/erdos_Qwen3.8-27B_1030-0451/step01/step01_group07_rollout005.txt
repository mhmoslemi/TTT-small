```python
import time
import math
import numpy as np

def project_exact(h, target):
    """Project h onto {0<=h<=1, sum(h)=target} via bisection on threshold."""
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            room = (1.0 - out).sum()
            if room > 1e-12:
                out -= (d / room) * (1.0 - out)
                out = np.clip(out, 0.0, 1.0)
        else:
            room = out.sum()
            if room > 1e-12:
                out += (-d / room) * out
                out = np.clip(out, 0.0, 1.0)
    return out

def c5_val(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)

def sturmian(n, phase, ones_count):
    """Generate a Sturmian (mechanical) word with exactly ones_count ones."""
    rho = ones_count / n
    out = np.zeros(n)
    for i in range(n):
        if math.floor((i + 1) * rho + phase) > math.floor(i * rho + phase):
            out[i] = 1.0
    # Adjust to get exactly ones_count ones
    actual = int(out.sum())
    if actual > ones_count:
        # Flip some 1s to 0s (preferably spread out)
        ones_pos = np.where(out == 1)[0]
        to_flip = ones_pos[len(ones_pos) - (actual - ones_count):]
        out[to_flip] = 0
    elif actual < ones_count:
        zeros_pos = np.where(out == 0)[0]
        to_flip = zeros_pos[:ones_count - actual]
        out[to_flip] = 1
    return out

def de_optimize(h0, n, budget, seed, pop_size=30, F=0.8, CR=0.7):
    """Differential Evolution optimizer."""
    rng = np.random.default_rng(seed)
    target = n / 2.0
    t_start = time.monotonic()
    
    # Initialize population
    pop = np.zeros((pop_size, n))
    pop[0] = h0
    for i in range(1, pop_size):
        r = rng.random(n)
        pop[i] = project_exact(r, target)
    
    costs = np.array([c5_val(pop[i]) for i in range(pop_size)])
    order = np.argsort(costs)
    pop = pop[order]
    costs = costs[order]
    
    best_h = pop[0].copy()
    best_c = costs[0]
    
    gen = 0
    while time.monotonic() - t_start < budget:
        for i in range(pop_size):
            if time.monotonic() - t_start >= budget:
                break
            # Select r1, r2 different from i
            r1, r2 = rng.choice(pop_size, size=2, replace=False)
            while r1 == i:
                r1 = rng.integers(pop_size)
            while r2 == i or r2 == r1:
                r2 = rng.integers(pop_size)
            
            # Mutation
            mutant = pop[0] + F * (pop[r1] - pop[r2])
            mutant = project_exact(mutant, target)
            
            # Crossover
            trial = pop[i].copy()
            jrand = rng.integers(n)
            for j in range(n):
                if rng.random() < CR or j == jrand:
                    trial[j] = mutant[j]
            trial = project_exact(trial, target)
            
            # Selection
            ct = c5_val(trial)
            if ct <= costs[i]:
                pop[i] = trial
                costs[i] = ct
                if ct < best_c:
                    best_c = ct
                    best_h = trial.copy()
            # Re-sort for "best/2" variant (optional, expensive)
        
        gen += 1
        # Periodic re-sort
        if gen % 10 == 0:
            order = np.argsort(costs)
            pop = pop[order]
            costs = costs[order]
    
    return best_h, best_c
```

import time
import math
import numpy as np


def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def project_exact(h, target):
    h = np.asarray(h, dtype=float)
    lo, hi = -1.5, 1.5
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            room = (1.0 - out).sum()
            if room > 1e-12:
                out -= (d / room) * (1.0 - out)
        else:
            room = out.sum()
            if room > 1e-12:
                out += (-d / room) * out
        out = np.clip(out, 0.0, 1.0)
    return out


def c5_val(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def lse_value(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    val = m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
    return dx * val


def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp <= 1e-18 and wm <= 1e-18:
            continue
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * (h[p:])
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * (h[:-p])
    grad *= dx
    return grad


def lse_polish(h0, rng, time_limit, sigma, target, deadline):
    h = np.asarray(h0, dtype=float).copy()
    cur_lse = lse_value(h, sigma)
    best_h = h.copy()
    best_c5 = c5_val(h)
    eta = 0.1
    t0 = time.monotonic()
    stall = 0
    while (time.monotonic() - t0) < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = lse_gradient(h, sigma)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        improved = False
        e = eta
        for _ in range(30):
            cand = project_exact(h - e * g, target)
            cv = lse_value(cand, sigma)
            if cv < cur_lse - 1e-14:
                h = cand
                cur_lse = cv
                c5 = c5_val(h)
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            e *= 0.5
        if not improved:
            h = project_exact(h + rng.normal(0.0, 0.02, len(h)), target)
            cur_lse = lse_value(h, sigma)
            c5 = c5_val(h)
            if c5 < best_c5 - 1e-14:
                best_c5 = c5
                best_h = h.copy()
            stall += 1
            if stall >= 5:
                eta *= 0.5
                stall = 0
                if eta < 1e-7:
                    break
    c5 = c5_val(h)
    if c5 < best_c5 - 1e-14:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def sturmian_word(n, slope, phase):
    out = np.zeros(n)
    for i in range(n):
        if math.floor((i + 1) * slope + phase) > math.floor(i * slope + phase):
            out[i] = 1.0
    actual = int(round(out.sum()))
    target = n // 2
    if actual > target:
        ones_pos = np.where(out == 1)[0]
        idx = ones_pos[len(ones_pos) - (actual - target):]
        out[idx] = 0.0
    elif actual < target:
        zeros_pos = np.where(out == 0)[0]
        idx = zeros_pos[:target - actual]
        out[idx] = 1.0
    return out


def de_search(h0, n, budget, seed):
    rng = np.random.default_rng(seed)
    target = n / 2.0
    t_start = time.monotonic()
    pop_size = 40
    pop = np.zeros((pop_size, n))
    pop[0] = h0.copy()
    for i in range(1, min(8, pop_size)):
        slope = 0.3 + 0.4 * (i / max(1, pop_size - 8))
        phase = rng.random()
        pop[i] = project_exact(sturmian_word(n, slope, phase), target)
    for i in range(8, pop_size):
        pop[i] = project_exact(rng.random(n), target)
    costs = np.array([c5_val(pop[i]) for i in range(pop_size)])
    order = np.argsort(costs)
    pop = pop[order]
    costs = costs[order]
    best_h = pop[0].copy()
    best_c = costs[0]
    gen = 0
    stagnant = 0
    F = 0.8
    CR = 0.7
    while time.monotonic() - t_start < budget:
        gen += 1
        if gen % 50 == 0:
            F = max(0.3, F - 0.005)
            if costs[0] >= best_c:
                stagnant += 1
            else:
                stagnant = 0
            if stagnant > 4:
                idx = rng.integers(pop_size)
                pop[idx] = project_exact(rng.random(n), target)
                costs[idx] = c5_val(pop[idx])
                order = np.argsort(costs)
                pop = pop[order]
                costs = costs[order]
                stagnant = 0
        for i in range(pop_size):
            if time.monotonic() - t_start >= budget:
                break
            r1 = rng.integers(pop_size)
            r2 = rng.integers(pop_size)
            r3 = rng.integers(pop_size)
            r4 = rng.integers(pop_size)
            while r1 == i:
                r1 = rng.integers(pop_size)
            while r2 == i or r2 == r1:
                r2 = rng.integers(pop_size)
            while r3 == i or r3 == r1 or r3 == r2:
                r3 = rng.integers(pop_size)
            while r4 == i or r4 == r1 or r4 == r2 or r4 == r3:
                r4 = rng.integers(pop_size)
            F2 = F * 0.5
            mutant = pop[0] + F * (pop[r1] - pop[r2]) + F2 * (pop[r3] - pop[r4])
            mutant = project_exact(mutant, target)
            jrand = rng.integers(n)
            mask = rng.random(n) < CR
            mask[jrand] = True
            trial = np.where(mask, mutant, pop[i])
            trial = project_exact(trial, target)
            ct = c5_val(trial)
            if ct <= costs[i]:
                pop[i] = trial
                costs[i] = ct
                if ct < best_c:

                    best_h = trial.copy()
                    stagnant = 0

                    if cc < best_c:
                        best_c = cc
                        best_h = hh.copy()
        order = np.argsort(costs)
        pop = pop[order]
        costs = costs[order]
    return best_h, best_c


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def multi_sigma_polish(h0, rng, budget, target, deadline):
    best_h = np.asarray(h0, dtype=float).copy()
    best_c5 = c5_val(best_h)
    t_start = time.monotonic()
    sigmas = [300.0, 150.0, 80.0, 40.0, 20.0, 10.0, 5.0, 3.0]
    for sigma in sigmas:
        elapsed = time.monotonic() - t_start
        remaining = budget - elapsed
        if remaining < 0.5:
            break
        alloc = min(remaining, max(1.0, 0.15 * budget))
        alloc = min(alloc, deadline - time.monotonic()) if deadline else alloc
        if alloc < 0.3:
            break
        ch, cc = lse_polish(best_h, rng, alloc, sigma, target, deadline)
        if cc < best_c5 - 1e-14:
            best_c5 = cc
            best_h = ch.copy()
    return best_h, best_c5


def active_lag_polish(h0, rng, budget, target, deadline):
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = c5_val(h)
    t_start = time.monotonic()
    step_size = 0.05
    while time.monotonic() - t_start < budget:
        if deadline is not None and time.monotonic() > deadline:
            break
        c = np.correlate(h, 1.0 - h, mode='full')
        c_scaled = c * dx
        mx = c_scaled.max()
        threshold = mx - 0.002
        active = c_scaled > threshold
        if not active.any():
            break
        w = np.where(active, (c_scaled - threshold), 0.0)
        w = w / max(w.sum(), 1e-12)
        center = n - 1
        grad = np.zeros(n)
        grad += w[center] * (1.0 - 2.0 * h)
        for p in range(1, n):
            wp = w[center + p]
            wm = w[center - p]
            if wp <= 1e-18 and wm <= 1e-18:
                continue
            if wp > 1e-18:
                grad[p:] += wp * (1.0 - h[:-p])
                grad[:-p] -= wp * (h[p:])
            if wm > 1e-18:
                grad[:-p] += wm * (1.0 - h[p:])
                grad[p:] -= wm * (h[:-p])
        grad *= dx
        improved = False
        e = step_size
        for _ in range(25):
            cand = project_exact(h - e * grad, target)
            cv = c5_val(cand)
            if cv < best_c5 - 1e-14:

                best_c5 = cv
                best_h = h.copy()
                improved = True

        if not improved:
            h = project_exact(h + rng.normal(0.0, 0.01, n), target)
            step_size *= 0.7
            if step_size < 1e-6:

    return best_h, best_c5


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 10.0)
    deadline = t0 + effective

    parent = get_parent()

        parent = project_exact(rng.random(131), 65.5)

    best_h = h0.copy()

        ch, cc = de_search(h0, n0, de_budget, seed)

            best_h = ch.copy()

    # Phase 1b: Second DE with different seed

                best_h = ch.copy()

    best_h = project_exact(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_c5 = c5_val(best_h)
    n_pts = len(best_h)
    return best_h, float(best_c5), int(n_pts)
