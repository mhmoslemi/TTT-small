def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def grad_at_lag(h, k_idx, n):
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[lag:] += (1.0 - h[:m])
        grad[:m] -= h[lag:]
    else:
        L = -lag
        m = n - L
        grad[:m] += (1.0 - h[L:])
        grad[L:] -= h[:m]
    return grad

def minmax_grad(h, corr, top_k, sharpness):
    n = len(h)
    max_c = np.max(corr)
    top_idx = np.argsort(corr)[-top_k:]
    gaps = max_c - corr[top_idx]
    w = np.exp(-gaps * sharpness * 2.0)  # scale sharpness by domain
    ws = w.sum()
    if ws < 1e-30:
        return grad_at_lag(h, top_idx[0], n)
    w = w / ws
    g = np.zeros(n)
    for i in range(len(top_idx)):
        g += w[i] * grad_at_lag(h, top_idx[i], n)
    return g

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
    return proj

def grad_at_lag(h, k_idx, n):
    lag = k_idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        m = n - lag
        grad[lag:] += (1.0 - h[:m])
        grad[:m] -= h[lag:]
    else:
        L = -lag
        m = n - L
        grad[:m] += (1.0 - h[L:])
        grad[L:] -= h[:m]
    return grad

def minmax_grad(h, corr, top_k, sharpness):
    n = len(h)
    max_c = float(np.max(corr))
    top_idx = np.argsort(corr)[-top_k:]
    gaps = max_c - corr[top_idx]
    w = np.exp(-gaps * sharpness * 2.0)
    ws = float(w.sum())
    if ws < 1e-30:
        return grad_at_lag(h, int(top_idx[0]), n)
    w = w / ws
    g = np.zeros(n)
    for i in range(len(top_idx)):
        g += w[i] * grad_at_lag(h, int(top_idx[i]), n)
    return g

def optimize_momentum(h0, time_limit, seed=42):
    n = len(h0)
    dx = 2.0 / n
    h = h0.astype(float).copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    vel = np.zeros(n)
    beta = 0.85
    step = 10.0 / n
    rng = np.random.RandomState(seed)
    no_improve = 0
    threshold = 60

    while time.time() - t0 < time_limit:
        corr = compute_corr(h)
        c5_val = float(np.max(corr)) * dx
        if c5_val < best_c5 - 1e-15:
            best_c5 = c5_val
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        frac = min((time.time() - t0) / max(time_limit, 1.0), 1.0)
        sharpness = 20.0 + 400.0 * frac
        top_k = 3 if frac < 0.3 else (2 if frac < 0.65 else 1)

        g = minmax_grad(h, corr, top_k, sharpness)
        vel = beta * vel + g
        h_new = project_constraint(h - step * vel, n)
        corr_new = compute_corr(h_new)
        c5_new = float(np.max(corr_new)) * dx

        if c5_new > c5_val + 1e-13:
            step = max(step * 0.75, 0.3 / n)
        else:
            h = h_new
            if c5_new < c5_val - 1e-13:
                step = min(step * 1.04, 40.0 / n)
                beta = min(beta + 0.00005, 0.96)

        if no_improve > threshold:
            noise_scale = 0.02 + 0.04 * (no_improve - threshold) / threshold
            h = project_constraint(h + rng.randn(n) * noise_scale, n)
            vel *= 0.2
            no_improve = 0
            threshold = min(int(threshold * 1.4), 600)

    return best_h, best_c5
