Let me also consider: what does the optimal h look like? Based on the theory, it should be a smooth function that's close to 1/2 everywhere but with some oscillation. The oscillation helps make the autocorrelation more uniform.

A good parameterization: h(x) = 1/2 + a·f(x) where f is some periodic function. The constraint ∫h = 1 means ∫f = 0. The constraint 0 ≤ h ≤ 1 means |f| ≤ 1/2.

Let me try a different approach: parametrize h as h = 1/2 + Σ a_k cos(πkx/2 + φ_k) (Fourier series on [0,2]) and optimize the coefficients. This automatically satisfies the box constraint if |Σ a_k cos(...)| ≤ 1/2, and the integral constraint is automatic (no constant term).

But this restricts the solution to be smooth and periodic-ish, which might not be optimal.

    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_feasible_interval(h, a):
    """Find δ range such that h + δ*a stays in [0,1]."""
    pos = a > 1e-14
    neg = a < -1e-14
    
    delta_min = -np.inf
    delta_max = np.inf
    
    if np.any(pos):
        delta_min = max(delta_min, np.min(-h[pos] / a[pos]))
        delta_max = min(delta_max, np.min((1.0 - h[pos]) / a[pos]))

    """Minimize F(δ) = max_r(A_r + δ L_r + δ² Q_r) over [delta_min, delta_max]."""
    if delta_min >= delta_max:
        return 0.0, A.max()
    
    # Coarse grid
    deltas = np.linspace(delta_min, delta_max, n_samples)
    # F(δ) = max_r(A + δ L + δ² Q) for all r
    # Shape: (n_samples, n_r)
    vals = A[None, :] + deltas[:, None] * L[None, :] + deltas[:, None]**2 * Q[None, :]
    F_vals = vals.max(axis=1)
    
    best_idx = np.argmin(F_vals)
    best_delta = deltas[best_idx]
    best_F = F_vals[best_idx]
    
    # Refine around best
    if 0 < best_idx < n_samples - 1:
        local_range = deltas[best_idx] - (deltas[1] - deltas[0]) * 2, deltas[best_idx] + (deltas[1] - deltas[0]) * 2
        local_range = (max(local_range[0], delta_min), min(local_range[1], delta_max))
        if local_range[1] > local_range[0]:

            vals_fine = A[None, :] + deltas_fine[:, None] * L[None, :] + deltas_fine[:, None]**2 * Q[None, :]
            F_fine = vals_fine.max(axis=1)
            best_idx_f = np.argmin(F_fine)
            if F_fine[best_idx_f] < best_F:
                best_delta = deltas_fine[best_idx_f]
                best_F = F_fine[best_idx_f]
    
    return best_delta, best_F

def compute_LQ(h, a):
    """Compute L_r and Q_r for direction a."""
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h
    corr_ab = np.correlate(a, b, mode='full')
    corr_ha = np.correlate(h, a, mode='full')
    corr_aa = np.correlate(a, a, mode='full')
    L = dx * (corr_ab - corr_ha)
    Q = -dx * corr_aa
    return L, Q

def project_feasible(h, target_sum):
    """Project h to [0,1]^N with sum = target_sum."""
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:

        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            h[idx[0]] += np.clip(residual, -(h[idx[0]]), 1.0 - h[idx[0]])
    return h

    """Compute softmax-weighted average gradient of the max constraint."""
    N = len(h)
    dx = 2.0 / N
    corr_vals = compute_corr_vals(h)
    m = corr_vals.max()
    exponents = np.clip((corr_vals - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w = w / w.sum()

    # c_r = dx * sum_n h[n] * (1-h[n-d]) where d = r - (N-1)
    # ∂c_r/∂h[k] = dx * [(1-h[k-d]) - h[k+d]] (with boundary handling)
    # But it's easier to compute numerically or use the structure of correlate
    
    # Actually, let's use the fact that:
    # np.correlate(h, 1-h) at lag d = sum_n h[n] * (1-h[n-d])
    # ∂/∂h[k] = (1-h[k-d]) - h[k+d] (times dx)
    # where h[k-d] is 0 if k-d outside [0,N-1], etc.
    
    # More efficiently: the gradient of the full correlation vector is:
    # grad[r] = dx * (convolution-like operations)
    # Let me just compute it using the structure
    
    # For the full correlation corr[r] = sum_n h[n] * (1-h[n-d(r)]) where d(r) = r - (N-1)
    # ∂corr[r]/∂h[k] = dx * [(1 - h[k-d(r)]) * 1_{valid(k-d(r))} + h[k] * (-1) * 1_{valid(k+d(r))} ... ]
    
    # This is getting complex. Let me use a simpler approach:
    # The gradient of corr_full w.r.t. h is related to convolutions.
    # Actually: corr_full = correlate(h, 1-h) 
    # ∂/∂h: the derivative of correlate(h, g) w.r.t. h is correlate(δ, g) shifted...
    
    # Let me just use the fact that for a weighted sum:
    # F = sum_r w_r * corr_r
    # ∂F/∂h[k] = dx * sum_r w_r * [(1-h[k-d(r)]) - h[k+d(r)]]
    #           = dx * [(sum_r w_r * (1-h[k-d(r)])) - (sum_r w_r * h[k+d(r)])]
    
    # The first term: sum_r w_r * (1-h[k-d(r)]) = sum_r w_r - sum_r w_r * h[k-d(r)]
    #             = sum(w) - (w conv h)[k] (approximately)
    # The second term: sum_r w_r * h[k+d(r)] = (w_flip conv h)[k]
    
    # Let me just compute the gradient numerically using the correlation structure.
    # For efficiency, I'll compute it as:
    # grad = dx * (correlate(w, 1-h, mode='same' + offsets) - correlate(w_flip, h, ...))
    
    # Actually the cleanest way: 
    # corr_r = sum_n h[n]*(1-h[n-d]) where d = r-(N-1)
    # = sum_n h[n] - sum_n h[n]*h[n-d]
    # 
    # ∂corr_r/∂h[k] = 1 - [h[k-d] + h[k+d]] (for valid indices, times dx)
    # Wait no: ∂/∂h[k] of sum_n h[n]*(1-h[n-d])
    # = (1-h[k-d]) + sum_n h[n]*(-1)*1_{n-d=k}
    # = (1-h[k-d]) - h[k+d]
    # where we interpret h[out of range] = 0
    
    # So: grad_h[r, k] = dx * ((1-h[k-d_r]) - h[k+d_r])
    # And the weighted gradient is:
    # grad[k] = dx * sum_r w_r * ((1-h[k-d_r]) - h[k+d_r])
    #         = dx * (sum_r w_r*(1-h[k-d_r]) - sum_r w_r*h[k+d_r])
    
    # Let me define: for a given k, d_r = r - (N-1)
    # sum_r w_r * h[k - d_r] = sum_r w_r * h[k - r + (N-1)]
    # Let j = k - r + (N-1), so r = k + (N-1) - j
    # = sum_j w_{k+(N-1)-j} * h[j] (where the index k+(N-1)-j must be in [0, 2N-2])
    # This is a cross-correlation of w and h!
    
    # Similarly: sum_r w_r * h[k + d_r] = sum_r w_r * h[k + r - (N-1)]
    # Let j = k + r - (N-1), so r = j - k + (N-1)
    # = sum_j w_{j-k+(N-1)} * h[j]
    # This is also a cross-correlation (reversed).
    
    # Let me just compute it numerically for clarity. For N up to 500, a loop over k is fine.
    
    grad = np.zeros(N)
    # For each k, grad[k] = dx * sum_r w_r * ((1 - h[k - d_r]) - h[k + d_r])
    # where d_r = r - (N-1), and h[index] = 0 if index out of [0, N-1]

    # sum_r w_r * h[k - d_r] = cross-correlation of w and h at a specific offset
    
    # Let me define: h_ext = zeros(2N-1), h_ext[N-1 : 2N-1] = h  (centered)
    # Then h[k - d_r] = h_ext[k - d_r + (N-1)] = h_ext[k + N - 1 - d_r] = h_ext[2(k - r + ...)]
    
    # This is getting too complicated analytically. Let me just use a vectorized approach.
    
    # Alternative: use the fact that the gradient of the full correlation can be computed as:
    # If corr_full = np.correlate(h, g, mode='full') where g = 1-h
    # Then ∂corr_full[r]/∂h[k] relates to g and h at specific positions.
    
    # For np.correlate(h, g, mode='full')[r]:
    # This is sum_n h[n] * g[n - (r - (N-1))]  (for the 'full' mode convention)
    # Actually numpy's correlate is: (a * v)[k] = sum_n a[n+k] * conj(v[n])
    # For real: (a * v)[k] = sum_n a[n+k] * v[n]
    # mode='full' gives k from -(M-1) to N-1 where N=len(a), M=len(v)
    # With N=M: k from -(N-1) to N-1, output length 2N-1
    
    # So np.correlate(h, 1-h, mode='full')[r] where r = 0..2N-2:
    # Let k = r - (N-1) (this is the lag)
    # value = sum_n h[n+k] * (1-h)[n]  (for valid n, n+k)
    #       = sum_n h[n+k] * (1-h[n])
    
    # ∂/∂h[j]: nonzero when j = n+k (from h[n+k]) or j = n (from (1-h[n]))
    # = sum_n 1_{j=n+k} * (1-h[n]) + sum_n h[n+k] * (-1_{j=n})
    # = (1-h[j-k]) (if j-k valid) - h[j-k] (if j-k valid for the second term... wait)
    
    # First term: j = n+k → n = j-k. Valid if 0 ≤ j-k ≤ N-1.
    #   Contribution: (1 - h[j-k])
    # Second term: j = n. Valid if 0 ≤ j ≤ N-1 (always true).
    #   Contribution: -h[j-k] (since n=j, n+k = j+k, valid if 0 ≤ j+k ≤ N-1)
    #   Wait, we need j+k to be valid for h[j+k].
    
    # So: ∂corr_full[r]/∂h[j] = (1-h[j-k]) * 1_{0≤j-k≤N-1} - h[j+k] * 1_{0≤j+k≤N-1}
    # where k = r - (N-1)
    
    # Therefore: grad[j] = dx * sum_r w_r * [(1-h[j-k_r]) * 1_{valid} - h[j+k_r] * 1_{valid}]
    
    # Let me compute this using numpy operations.
    # The key insight: for each r, the gradient contribution to all j is a shifted version of h.
    
    # I'll compute it by building the gradient directly.
    # For N=500, the loop over r (2N-1 ≈ 1000 iterations) with vectorized inner ops should be fast enough.
    
    for r in range(2 * N - 1):
        if w[r] < 1e-15:
            continue
        d = r - (N - 1)
        # Contribution to grad[j] from lag d:
        # dx * w[r] * [(1-h[j-d]) * 1_{0≤j-d≤N-1} - h[j+d] * 1_{0≤j+d≤N-1}]
        # Term 1: (1-h[j-d]) for j-d in [0,N-1], i.e., j in [d, d+N-1]
        #   If d >= 0: j in [d, N-1], index j-d in [0, N-1-d]
        #   If d < 0: j in [0, N-1+d], index j-d in [-d, N-1]
        # Term 2: h[j+d] for j+d in [0,N-1], i.e., j in [-d, N-1-d]
        #   If d >= 0: j in [0, N-1-d], index j+d in [d, N-1]
        #   If d < 0: j in [-d, N-1], index j+d in [0, N-1+d]
        
        a = w[r] * dx
        if d >= 0:
            n1 = N - d  # length of valid range for term 1
            if n1 > 0:
                grad[d:d+n1] += a * (1.0 - h[:n1])
            n2 = N - d  # length of valid range for term 2
            if n2 > 0:
                grad[:n2] -= a * h[d:d+n2]
        else:
            dd = -d
            n1 = N - dd
            if n1 > 0:
                grad[:n1] += a * (1.0 - h[dd:])
            n2 = N - dd
            if n2 > 0:
                grad[dd:] -= a * h[:n2]
    
    # Project to zero-sum (orthogonal to all-ones vector)
    grad -= np.mean(grad)
    return grad

Hmm, this gradient computation is O(N²) per call (loop over 2N-1 r values, each doing O(N) work). For N=500, that's about 500K operations per gradient call. Should be fast enough.

Now let me think about the overall algorithm more carefully.

The key insight for improvement: instead of using SLSQP (which is a general-purpose constrained optimizer), use a custom active-set method with exact line search.

Algorithm:
1. Start from initial_h_values (or interpolate to different N)
2. At each iteration:
   a. Compute current correlation values A
   b. Identify active constraints (those with A_r close to max)
   c. Generate candidate directions:
      - Direction 1: negative smooth gradient (projected to zero-sum)
      - Direction 2: best pairwise swap (sample random pairs, pick best)
      - Direction 3: random direction
   d. For each direction, do exact line search
   e. Accept the best improvement
   f. If no improvement, try harder (more random pairs, smaller steps)
3. Periodically try different N values

- Line search with 2000 samples: O(2000 * 700) ≈ 1.4M ops

Per iteration: maybe 5-10M ops total. At ~100M ops/sec (conservative for numpy), that's 0.05-0.1 sec per iteration.
In 1000 seconds, we can do ~10000-20000 iterations.

That should be more than enough. The question is whether the algorithm converges to a better local minimum.

Let me also think about a different strategy: **coordinate descent with exact line search**.

At each step, pick a pair (i, j) and move mass from j to i (or vice versa). The optimal step size is found by exact line search. This is like a Frank-Wolfe / conditional gradient method on the hypersimplex.

The key advantage: each step is guaranteed to improve (or we stop at a local optimum), and the line search is exact.

For the pairwise direction a = e_i - e_j:
- Feasible δ range: [max(-h_i, h_j - 1), min(1 - h_i, h_j)]
  Actually from our formula:
  - a_i = 1 > 0: lower bound -h_i/1 = -h_i, upper bound (1-h_i)/1 = 1-h_i
  - a_j = -1 < 0: lower bound (1-h_j)/(-1) = h_j-1, upper bound -h_j/(-1) = h_j
  - δ_min = max(-h_i, h_j-1)
  - δ_max = min(1-h_i, h_j)

The line search for this direction:
F(δ) = max_r (A_r + δ L_r + δ² Q_r)

For a = e_i - e_j, L_r and Q_r are sparse (only O(N) non-zero entries). But computing them still requires the full correlation, so it's O(N²) per direction.

To speed up: we can precompute which r values are affected. For a = e_i - e_j:
- corr_r(a, b) is non-zero only when the support of a overlaps with a shifted version of b
- Since a has support {i, j}, corr_r(a, b) is non-zero only for r values where i or j aligns with some position in b
- This gives at most 2N-1 non-zero values... actually all of them could be non-zero since b = 1-h is dense.

So the sparsity doesn't help much for the dense case. Let me just compute L and Q directly.

Let me also consider: should I try to use CVXPY? The problem is non-convex (max of non-convex functions), so CVXPY won't directly help. But I could solve a sequence of convex relaxations... probably not worth the complexity.

What about using the fact that the problem is equivalent to:
minimize t
subject to: c_r(h) ≤ t for all r
            0 ≤ h_i ≤ 1
            sum(h) = N/2

This is a non-convex problem because c_r(h) = dx * corr_r(h, 1-h) is not convex in h (it's bilinear/quadratic).

Specifically, c_r(h) = dx * (sum_n h[n] - sum_n h[n]h[n-d]) = dx * (sum_n h[n] - autocorr_r(h))

The autocorrelation is a quadratic form, so c_r is a quadratic (concave in the quadratic part). The constraint c_r ≤ t is therefore not a convex constraint.

So we're definitely in non-convex territory.

import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_vals(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_feasible_interval(h, a):
    pos = a > 1e-14
    neg = a < -1e-14
    delta_min = -np.inf
    delta_max = np.inf
    if np.any(pos):
        delta_min = max(delta_min, float(np.min(-h[pos] / a[pos])))
        delta_max = min(delta_max, float(np.min((1.0 - h[pos]) / a[pos])))
    if np.any(neg):
        delta_min = max(delta_min, float(np.max((1.0 - h[neg]) / a[neg])))
        delta_max = min(delta_max, float(np.max(-h[neg] / a[neg])))
    delta_min = max(delta_min, -2.0)
    delta_max = min(delta_max, 2.0)
    if delta_min >= delta_max:
        return 0.0, 0.0
    return delta_min, delta_max

def line_search(A, L, Q, d_min, d_max, n_coarse=1000, n_fine=500):
    if d_min >= d_max:
        return 0.0, float(A.max())
    deltas = np.linspace(d_min, d_max, n_coarse)
    # Compute F(delta) for all deltas: max over r of A[r] + delta*L[r] + delta^2*Q[r]
    # Vectorized: shape (n_coarse, len(A))
    F_vals = A[None, :] + deltas[:, None] * L[None, :] + (deltas[:, None] ** 2) * Q[None, :]
    F_vals = F_vals.max(axis=1)
    best_i = int(np.argmin(F_vals))
    best_d = float(deltas[best_i])
    best_F = float(F_vals[best_i])
    
    # Refine
    if 0 < best_i < n_coarse - 1:
        step = deltas[1] - deltas[0]
        lo = max(d_min, best_d - 2 * step)
        hi = min(d_max, best_d + 2 * step)
        if hi > lo:
            deltas_f = np.linspace(lo, hi, n_fine)
            F_f = A[None, :] + deltas_f[:, None] * L[None, :] + (deltas_f[:, None] ** 2) * Q[None, :]
            F_f = F_f.max(axis=1)
            best_i_f = int(np.argmin(F_f))
            if F_f[best_i_f] < best_F:
                best_d = float(deltas_f[best_i_f])
                best_F = float(F_f[best_i_f])
    return best_d, best_F

def compute_LQ(h, a):
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h
    L = dx * (np.correlate(a, b, mode='full') - np.correlate(h, a, mode='full'))
    Q = -dx * np.correlate(a, a, mode='full')
    return L, Q

def smooth_grad(h, tau=0.01):
    N = len(h)
    dx = 2.0 / N
    cv = compute_corr_vals(h)
    m = cv.max()
    exponents = np.clip((cv - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w = w / w.sum()
    
    grad = np.zeros(2 * N - 1)  # Will be the gradient of weighted sum
    # For each r, contribution to grad is:
    # w[r] * dx * [(1-h[j-d_r]) - h[j+d_r]] for valid j
    # This is equivalent to:
    # grad[j] = dx * sum_r w[r] * [(1-h[j-d_r])*1_{valid} - h[j+d_r])*1_{valid}]
    
    # Vectorized approach:
    # Term 1: sum_r w[r] * (1-h[j-d_r]) for valid j-d_r
    #   = sum_r w[r] - sum_r w[r]*h[j-d_r]
    #   The second sum is a cross-correlation of w and h
    # Term 2: sum_r w[r] * h[j+d_r] for valid j+d_r
    #   This is also a cross-correlation
    
    # Let me use the loop approach for correctness
    for r in range(2 * N - 1):
        if w[r] < 1e-16:
            continue
        d = r - (N - 1)
        a_coef = w[r] * dx
        if d >= 0:
            n = N - d
            if n > 0:
                # Term 1: grad[d:d+n] += a_coef * (1 - h[:n])
                grad_tmp = a_coef * (1.0 - h[:n])
                # This adds to positions d..d+n-1 in h-index space
                # But our grad is indexed 0..N-1 (for h positions)
                pass
        # This approach is getting confusing. Let me use a different method.
    
    # Actually, let me just compute the gradient of the smoothed objective directly.
    # The smoothed objective is: S(h) = m + tau * log(sum_r exp((c_r - m)/tau))
    # Its gradient is: sum_r w_r * grad(c_r)

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr_vals(h):
    N = len(h)
    dx = 2.0 / N
    return np.correlate(h, 1.0 - h, mode='full') * dx

def get_feasible_interval(h, a):
    pos = a > 1e-14
    neg = a < -1e-14
    delta_min = -np.inf
    delta_max = np.inf
    if np.any(pos):
        delta_min = max(delta_min, float(np.min(-h[pos] / a[pos])))
        delta_max = min(delta_max, float(np.min((1.0 - h[pos]) / a[pos])))
    if np.any(neg):
        delta_min = max(delta_min, float(np.max((1.0 - h[neg]) / a[neg])))
        delta_max = min(delta_max, float(np.max(-h[neg] / a[neg])))
    delta_min = max(delta_min, -2.0)
    delta_max = min(delta_max, 2.0)
    if delta_min >= delta_max:
        return 0.0, 0.0
    return delta_min, delta_max

def line_search(A, L, Q, d_min, d_max, n_coarse=1000, n_fine=500):
    if d_min >= d_max:
        return 0.0, float(A.max())
    deltas = np.linspace(d_min, d_max, n_coarse)
    F_vals = A[None, :] + deltas[:, None] * L[None, :] + (deltas[:, None] ** 2) * Q[None, :]
    F_vals = F_vals.max(axis=1)
    best_i = int(np.argmin(F_vals))
    best_d = float(deltas[best_i])
    best_F = float(F_vals[best_i])
    if 0 < best_i < n_coarse - 1:
        step = deltas[1] - deltas[0]
        lo = max(d_min, best_d - 3 * step)
        hi = min(d_max, best_d + 3 * step)
        if hi > lo:
            deltas_f = np.linspace(lo, hi, n_fine)
            F_f = (A[None, :] + deltas_f[:, None] * L[None, :]
                   + (deltas_f[:, None] ** 2) * Q[None, :]).max(axis=1)
            bi = int(np.argmin(F_f))
            if F_f[bi] < best_F:
                best_d = float(deltas_f[bi])
                best_F = float(F_f[bi])
    return best_d, best_F

def compute_LQ(h, a):
    N = len(h)
    dx = 2.0 / N
    b = 1.0 - h
    L = dx * (np.correlate(a, b, mode='full') - np.correlate(h, a, mode='full'))
    Q = -dx * np.correlate(a, a, mode='full')
    return L, Q

def smooth_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    cv = np.correlate(h, 1.0 - h, mode='full') * dx
    m = cv.max()
    exponents = np.clip((cv - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w = w / w.sum()
    g = np.zeros(N)
    for r in range(2 * N - 1):
        if w[r] < 1e-16:
            continue
        d = r - (N - 1)
        c = w[r] * dx
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:d+n] += c * (1.0 - h[:n])
                g[:n] -= c * h[d:d+n]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += c * (1.0 - h[dd:])
                g[dd:] -= c * h[:n]
    g -= g.mean()
    return g

    h_new = np.interp(x_new, x_old, h_source)

    n_trials = len(N_list)
    time_per = (deadline - start_time) / (n_trials + 1)

    for idx, N_target in enumerate(N_list):
        remaining = deadline - time.time()
        if remaining < 60:
            break
        time_limit = start_time + budget_s - 20 - (n_trials - idx - 1) * time_per * 0.3
        time_limit = min(time_limit, deadline - 30)
        if time_limit - time.time() < 30:
            time_limit = deadline - 30

        h_try = interpolate_h(h_init, N_target)
        c5_try = compute_c5(h_try)

        h_opt, c5_opt = optimize_at_N(h_try, N_target, time_limit, rng, tau_list)

        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_N = N_target

    # Final polish on the best solution
    remaining = deadline - time.time()
    if remaining > 30:
        h_work = best_h.copy()
        N_work = best_N
        target_sum = N_work / 2.0
        tau_final = [0.001, 0.0005, 0.0003, 0.0002, 0.0001, 0.00005]

        A = compute_corr_vals(h_work)
        cur_c5 = float(A.max())

        for tau in tau_final:
            if time.time() > deadline - 10:
                break
            for _ in range(50):
                if time.time() > deadline - 10:
                    break
                g = smooth_grad(h_work, tau)
                g_norm = np.linalg.norm(g)
                if g_norm < 1e-14:
                    break
                a = -g / g_norm
                a = a - a.mean()
                d_min, d_max = get_feasible_interval(h_work, a)
                if d_min >= d_max:
                    continue
                L, Q = compute_LQ(h_work, a)
                bd, bF = line_search(A, L, Q, d_min, d_max)
                new_h = np.clip(h_work + bd * a, 0.0, 1.0)
                new_h = project_to_sum(new_h, target_sum)
                new_c5 = float(compute_corr_vals(new_h).max())
                if new_c5 < cur_c5:
                    h_work = new_h
                    A = compute_corr_vals(h_work)
                    cur_c5 = new_c5
                    if cur_c5 < best_c5:
                        best_c5 = cur_c5
                        best_h = new_h.copy()
                else:
                    break

        # Additional pairwise refinement
        for _ in range(200):
            if time.time() > deadline - 5:
                break
            improved = False
            for _ in range(30):
                if time.time() > deadline - 5:
                    break
                i, j = rng.integers(0, N_work, 2)
                if i == j:
                    continue
                a = np.zeros(N_work)
                a[i] = 1.0
                a[j] = -1.0
                d_min, d_max = get_feasible_interval(h_work, a)
                if d_min >= d_max:
                    continue
                L, Q = compute_LQ(h_work, a)
                bd, bF = line_search(A, L, Q, d_min, d_max)
                if bF < float(A.max()) - 1e-12:
                    new_h = np.clip(h_work + bd * a, 0.0, 1.0)
                    new_h = project_to_sum(new_h, target_sum)
                    new_c5 = float(compute_corr_vals(new_h).max())
                    if new_c5 < float(A.max()):
                        h_work = new_h
                        A = compute_corr_vals(h_work)
                        improved = True
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = new_h.copy()
            if not improved:
                break

    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
