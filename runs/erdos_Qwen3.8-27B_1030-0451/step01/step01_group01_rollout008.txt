                x0 = best_h + np.random.randn(n) * 0.005
                x0 = project_sum_box(x0, n / 2.0)
            else:
                x0 = best_h + np.random.randn(n) * 0.01
                x0 = project_sum_box(x0, n / 2.0)

            def obj_func(x):
                return smooth_obj_and_grad(x, beta)[0]

            def grad_func(x):
                return smooth_obj_and_grad(x, beta)[1]

            try:
                res = minimize(obj_func, x0, jac=grad_func, method='SLSQP',
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 300, 'ftol': 1e-15})
            except Exception:
                continue
            if res is not None and res.x is not None:
                x_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_c5(x_new)
                if c_new < best_c - 1e-12:
                    best_h = x_new.copy()
                    best_c = c_new
    return best_h, best_c

def interpolate_h(h0, n_new):
    n_old = len(h0)
    x_old = np.linspace(0, 1, n_old, endpoint=False)
    x_new = np.linspace(0, 1, n_new, endpoint=False)
    h_ext = np.concatenate([h0, [h0[0]]])
    h_new = np.interp(x_new, x_old, h_ext, period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def random_perturb_optimize(h0, n_perturb, deadline, perturb_scale=0.05):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    for i in range(n_perturb):
        if time.time() > deadline:
            break
        h_pert = h0 + np.random.randn(n) * perturb_scale
        h_pert = project_sum_box(h_pert, n / 2.0)
        c_pert = evaluate_c5(h_pert)
        dl = min(deadline, time.time() + 3.0)
        h_opt, c_opt = lp_active_optimize(h_pert, dl, max_iters=200, init_radius=0.05)
        if c_opt < best_c:
            best_h = h_opt.copy()
            best_c = c_opt
    return best_h, best_c

def cutting_plane_lp(h0, deadline, max_rounds=50):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = v.max()
        worst_idx = np.argmax(v)
        lag_worst = worst_idx - (n - 1)
        g_k = grad_lag(h, lag_worst)
        dt = v[worst_idx] - vmax + 1e-8
        A_ub = np.array([g_k])
        b_ub = np.array([dt])
        n_var = n
        c_obj = np.zeros(n)
        A_eq = np.ones((1, n))
        b_eq = np.array([n / 2.0])
        bounds = [(0.0, 1.0)] * n
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs', options={'maxiter': 3000})
        except Exception:
            break
        if res is not None and res.success:
            h_new = project_sum_box(res.x, n / 2.0)
            c_new = evaluate_c5(h_new)
            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()
                h = h_new.copy()
            else:
                break
        else:
            break
    return best_h, best_c

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 5.0
    deadline = start_time + budget_s - safety

    def time_left():
        return deadline - time.time()

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        n = 100
        h_init = np.full(n, 0.5)
        h_init = project_sum_box(h_init, n / 2.0)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)

    # Phase 1: LP optimization at initial resolution
    n_init = len(best_h)
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.25)
        try:
            h1, c1 = lp_active_optimize(best_h, dl, max_iters=3000, init_radius=0.18)
            if c1 < best_c:
                best_h, best_c = h1, c1
        except Exception:
            pass

    # Phase 2: SLSQP refinement at initial resolution
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.38)
        try:
            h2, c2 = slsqp_refine(best_h, dl, max_betas=8)
            if c2 < best_c:
                best_h, best_c = h2, c2
        except Exception:
            pass

    # Phase 3: Another LP round
    if time_left() > 40:
        dl = min(deadline, start_time + budget_s * 0.48)
        try:
            h3, c3 = lp_active_optimize(best_h, dl, max_iters=2000, init_radius=0.1)
            if c3 < best_c:
                best_h, best_c = h3, c3
        except Exception:
            pass

    # Phase 4: Increase resolution progressively
    n_cur = len(best_h)
    target_list = []
    for f in [1.5, 2.0, 2.5, 3.0, 3.5, 4.0]:
        t = int(round(n_cur * f))
        if t > n_cur and t <= 500:
            target_list.append(t)
    target_list = sorted(set(target_list))

    for n_t in target_list:
        if time_left() < 80:
            break
        try:
            h_tr = interpolate_h(best_h, n_t)
            c_tr = evaluate_c5(h_tr)
            if c_tr < best_c:
                best_h, best_c = h_tr.copy(), c_tr
            else:
                best_h = h_tr.copy()
            # Optimize at new resolution
            sub_dl = min(deadline - 30, time.time() + min(120, time_left() * 0.35))
            h_a, c_a = lp_active_optimize(best_h, sub_dl, max_iters=1500, init_radius=0.12)
            if c_a < best_c:
                best_h, best_c = h_a, c_a
            # SLSQP at new resolution
            if time_left() > 60:
                sub_dl2 = min(deadline - 20, time.time() + min(60, time_left() * 0.2))
                h_b, c_b = slsqp_refine(best_h, sub_dl2, max_betas=5)
                if c_b < best_c:
                    best_h, best_c = h_b, c_b
            n_cur = len(best_h)
        except Exception:
            pass

    # Phase 5: Random perturbations at current resolution
    if time_left() > 60:
        n_now = len(best_h)
        n_pert = max(5, int(time_left() / 8))
        dl = min(deadline - 20, time.time() + min(200, time_left() * 0.5))
        try:
            h5, c5 = random_perturb_optimize(best_h, n_pert, dl, perturb_scale=0.04)
            if c5 < best_c:
                best_h, best_c = h5, c5
        except Exception:
            pass

    # Phase 6: Final LP + subgradient polish
    if time_left() > 30:
        dl = min(deadline - 10, time.time() + min(100, time_left() * 0.4))
        try:
            h6, c6 = lp_active_optimize(best_h, dl, max_iters=2000, init_radius=0.08)
            if c6 < best_c:
                best_h, best_c = h6, c6
        except Exception:
            pass

    if time_left() > 20:
        try:
            h7, c7 = subgradient_polish(best_h, deadline - 2, max_iters=20000)
            if c7 < best_c:
                best_h, best_c = h7, c7
        except Exception:
            pass

    # Phase 7: Cutting plane as final touch
    if time_left() > 15:
        try:
            h8, c8 = cutting_plane_lp(best_h, deadline - 1, max_rounds=30)
            if c8 < best_c:
                best_h, best_c = h8, c8
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
