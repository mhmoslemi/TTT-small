    g_fd = np.zeros(n)
    for i in range(min(n, 20)):

def polish_exact(h, best_val, max_time, step_sizes=None):
    """Polish using exact max with small steps toward lowering the max correlation."""
    t_start = time.time()
    n = len(h)
    target = n / 2.0
    if step_sizes is None:
        step_sizes = [0.02, 0.01, 0.005, 0.002, 0.001]
    h = h.copy()
    best = best_val
    stagnant = 0

    while time.time() - t_start < max_time - 1.0 and stagnant < 80:
        corr = correlation_array(h)
        cur_max = float(np.max(corr))
        if cur_max < best:
            best = cur_max
            stagnant = 0
        else:
            stagnant += 1

        active_idx = np.where(corr > cur_max - 0.005)[0]
        # Compute subgradient of max (averaged over active lags)
        w = np.zeros(len(corr))
        w[active_idx] = 1.0 / len(active_idx)
        g = lse_grad(w, h)
        # Normalize
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        g = g / gn

        improved = False
        for ss in step_sizes:
            h_try = project_feasible(h - ss * g, target)
            v_try = evaluate_exact(h_try)
            if v_try < best - 1e-12:
                best = v_try
                h = h_try.copy()
                improved = True
                break
        if not improved:
            # Try random perturbation
            if stagnant > 20:
                h_try = project_feasible(h + 0.03 * np.random.default_rng(stagnant).standard_normal(n), target)
                v_try = evaluate_exact(h_try)
                if v_try < best:
                    best = v_try
                    h = h_try.copy()
                    stagnant = 0
    return h, best
