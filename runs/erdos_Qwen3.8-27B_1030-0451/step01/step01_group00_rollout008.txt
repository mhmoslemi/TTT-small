import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_constraint(p, n):
    target = n / 2.0
    p = np.asarray(p, dtype=float).copy()
    p[~np.isfinite(p)] = 0.0
    p = np.clip(p, -2.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(p - mid, 0.0, 1.0)))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj = np.clip(proj, 0.0, 1.0)
            diff2 = target - float(np.sum(proj))
            if abs(diff2) > 1e-10:
                mask2 = (proj > 1e-8) & (proj < 1.0 - 1e-8)
                nm2 = int(np.sum(mask2))
                if nm2 > 0:
                    proj[mask2] = np.clip(proj[mask2] + diff2 / nm2, 0.0, 1.0)
    return proj

def full_grad_matrix(h, n, dx):
    m = 2 * n - 1
    G = np.zeros((m, n))
    for k in range(m):
        lag = k - (n - 1)
        if lag >= 0:
            idx = np.arange(n - lag)
            G[k, idx] += (1.0 - h[idx + lag])
            G[k, idx + lag] -= h[idx]
        else:
            L = -lag
            idx = np.arange(L, n)
            G[k, idx] += (1.0 - h[idx - L])
            G[k, idx - L] -= h[idx]
    G *= dx
    return G

def active_set_gradient(h, corr, n, dx, K=15):
    c = corr * dx
    f = float(np.max(c))
    order = np.argsort(-c)
    active = order[:K]
    G = full_grad_matrix(h, n, dx)
    weights = c[active] - c[active].min()
    weights = np.exp(5.0 * weights / (f + 1e-15))
    weights /= weights.sum()
    return (G[active].T @ weights).astype(float)

def lp_direction(h, n, dx, radius, eta):
    m = 2 * n - 1
    corr = np.correlate(h, 1.0 - h, mode="full")
    c = corr * dx
    f = float(np.max(c))
    G = full_grad_matrix(h, n, dx)
    nvar = 2 * n + 1
    c_obj = np.zeros(nvar)
    c_obj[:n] = eta
    c_obj[2 * n] = 1.0
    A1 = np.column_stack([G, np.zeros((m, n)), -np.ones(m)])
    b1 = f - c
    A2 = np.zeros((n, nvar))
    for i in range(n):
        A2[i, i] = 1.0
        A2[i, n + i] = -1.0
    b2 = np.zeros(n)
    A3 = np.zeros((n, nvar))
    for i in range(n):
        A3[i, i] = -1.0
        A3[i, n + i] = -1.0
    b3 = np.zeros(n)
    A_ub = np.vstack([A1, A2, A3])
    b_ub = np.concatenate([b1, b2, b3])
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    bounds = []
    for i in range(n):
        lo_d = max(-radius, -h[i] + 1e-12)
        hi_d = min(radius, 1.0 - h[i] + 1e-12)
        bounds.append((lo_d, hi_d))
    for _ in range(n):
        bounds.append((0.0, radius))
    bounds.append((0.0, 2.0))
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            delta = res.x[:n].copy()
            delta -= delta.mean()
            for i in range(n):
                lo = max(-radius, -h[i])
                hi = min(radius, 1.0 - h[i])
                delta[i] = max(lo, min(hi, delta[i]))
            delta -= delta.mean()
            if np.all(np.isfinite(delta)) and np.linalg.norm(delta) > 1e-13:
                return delta
    except Exception:
        pass
    return None

def line_search(h, direction, n, dx, max_iters=20):
    best_h = h.copy()
    best_c5 = compute_c5(h)
    h_try = h.copy()
    for alpha in np.concatenate([[1.0], 0.5 ** np.arange(1, max_iters)]):
        h_try = project_constraint(h + alpha * direction, n)
        c5_try = compute_c5(h_try)
        if c5_try < best_c5 - 1e-14:
            best_h = h_try.copy()
            best_c5 = c5_try
        else:
            break
    return best_h, best_c5

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    freqs = [1, 2, 3, 5, 7]
    amps = [0.22, 0.12, 0.08, 0.04, 0.02]
    phases = [0.0, 0.5, 1.2, 2.1, 3.3]
    for f, a, p in zip(freqs, amps, phases):
        h += a * np.sin(2.0 * np.pi * f * x + p)
    h += rng.randn(n) * 0.015
    return project_constraint(h, n)

def local_optimize(h_start, time_limit, use_lp=True, lp_eta=0.002, lp_radius=0.08):
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    radius = lp_radius
    stall = 0
    iter_count = 0
    while time.time() - t0 < time_limit - 2:
        iter_count += 1
        corr = np.correlate(h, 1.0 - h, mode="full")
        c5_cur = float(np.max(corr) * dx)
        if c5_cur < best_c5 - 1e-13:
            best_c5 = c5_cur
            best_h = h.copy()
            stall = 0
        direction = None
        method = None
        if use_lp and iter_count % 3 == 1:
            direction = lp_direction(h, n, dx, radius, lp_eta)
            if direction is not None:
                method = 'lp'
        if direction is None:
            direction = active_set_gradient(h, corr, n, dx, K=20)
            gn = np.linalg.norm(direction)
            if gn < 1e-14:
                direction = None
                method = None
            else:
                direction = -direction / gn
                method = 'grad'
        if direction is None:
            stall += 1
            scale = 0.01 + 0.01 * min(stall, 5)
            rng_local = np.random.RandomState(iter_count * 7919 + int(n))
            direction = rng_local.randn(n) * scale
            direction -= direction.mean()
            method = 'random'
        if method in ('lp', 'grad'):
            direction = -direction if method == 'grad' else direction
            if method == 'lp':

            else:

        else:
            h_new = project_constraint(h + direction, n)
            c5_new = compute_c5(h_new)
        if c5_new < c5_cur - 1e-14:
            h = h_new
            if c5_new < best_c5 - 1e-13:
                stall = 0
                if method == 'lp':
                    radius = min(radius * 1.1, 0.15)
                elif method == 'grad':
                    radius = min(radius * 1.05, 0.12)
            else:

        else:
            stall += 1
            if stall > 6:

                h = project_constraint(h + rng_local.randn(n) * 0.02, n)
                radius = max(0.04, radius * 0.7)
                stall = 0
        if iter_count % 500 == 0 and time.time() - t0 > time_limit * 0.4:
            radius = max(0.03, 0.05)
    return best_h, best_c5

def resample_h(h_old, n_new):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old * 2.0
    x_new = (np.arange(n_new) + 0.5) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    return project_constraint(h_new, n_new)
