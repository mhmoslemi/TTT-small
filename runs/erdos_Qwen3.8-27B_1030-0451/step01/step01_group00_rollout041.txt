import numpy as np
import time
from scipy.optimize import linprog

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_h(p, n):
    target = n / 2.0
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            proj[mask] += adj
            proj = np.clip(proj, 0.0, 1.0)
            # If still off, do a second pass
            diff2 = target - float(np.sum(proj))
            if abs(diff2) > 1e-12:
                mask2 = (proj > 1e-10) & (proj < 1.0 - 1e-10)
                nm2 = int(np.sum(mask2))
                if nm2 > 0:
                    proj[mask2] += diff2 / nm2
    return proj

def soft_gradient(h, corr, tau):
    n = len(h)
    dx = 2.0 / n
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k] * dx
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            L = n - lag
            # h[i+lag]*(1-h[i]) for i=0..L-1
            # d/dh[j]: from i=j-lag term: dx*(1-h[j-lag]) if j>=lag
            #          from i=j term: -dx*h[j+lag] if j+lag < n
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                # j from lag to n-1: contribution from i=j-lag
                grad[lag:] += wk * (1.0 - h[:L])
                # j from 0 to L-1: contribution from i=j (h[j]*(1-h[j+lag]))
                grad[:L] -= wk * h[lag:]
        else:
            L = -lag
            M = n - L
            # h[i]*(1-h[i+L]) for i=0..M-1
            # d/dh[j]: from i=j term: dx*(1-h[j+L]) if j < M
            #          from i+L=j term (i=j-L): -dx*h[j-L] if j>=L
            grad[:M] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:M]
    return grad

def lp_direction(h, corr, dx, radius, n_rows=None):
    """Find LP direction to reduce max correlation."""
    n = len(h)
    c = corr * dx
    f = np.max(c)
    
    # Select rows: all if small, else top-K + margin
    if n_rows is None or len(c) <= n_rows:
        active_idx = np.arange(len(c))
    else:
        sorted_idx = np.argsort(c)[::-1]
        active_idx = sorted_idx[:n_rows]
    
    M = len(active_idx)
    # Build constraint matrix A for: A @ [delta; t] <= b
    # For each active k: grad_k · delta - t <= f - c[k]
    # Also: sum(delta) = 0
    # Bounds: 0 <= h[j] + delta[j] <= 1
    
    # Objective: minimize t + eta * ||delta||_1 (approx via L-inf or just t)
    # Let's use: minimize t with trust region on delta
    
    # Variables: delta[0..n-1], t (index n)
    # c_obj: minimize t => c_obj[n] = 1, rest 0
    # Or: minimize t + eta*sum|delta| -> need auxiliary vars, skip for simplicity
    # Just minimize t
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    A_ub = np.zeros((M * 2 + 2 + n, n + 1))
    b_ub = np.zeros(M * 2 + 2 + n)
    
    # Trust region: |delta[j]| <= radius
    for j in range(n):
        A_ub[0, j] = 1.0
        b_ub[0] = radius
        A_ub[1, j] = -1.0
        b_ub[1] = radius
        0  # placeholder
    
    # Actually let me restructure. Use scipy linprog properly.
    # A_ub @ x <= b_ub
    # x = [delta_0, ..., delta_{n-1}, t]
    
    row = 0
    A_ub = np.zeros((M + 2 * n, n + 1))
    b_ub = np.zeros(M + 2 * n)
    
    # Trust region bounds
    for j in range(n):
        A_ub[row, j] = 1.0
        b_ub[row] = radius
        row += 1
        A_ub[row, j] = -1.0
        b_ub[row] = radius
        row += 1
    
    # Correlation constraints: grad_k · delta - t <= f - c[k]
    for idx in range(M):
        k = active_idx[idx]
        lag = k - (n - 1)
        grad_k = np.zeros(n)
        if lag >= 0:
            L = n - lag
            if lag == 0:
                grad_k[:] = dx * (1.0 - 2.0 * h)
            else:
                grad_k[lag:] += dx * (1.0 - h[:L])
                grad_k[:L] -= dx * h[lag:]
        else:
            L = -lag
            M2 = n - L
            grad_k[:M2] += dx * (1.0 - h[L:])
            grad_k[L:] -= dx * h[:M2]
        
        A_ub[row, :n] = grad_k
        A_ub[row, n] = -1.0
        b_ub[row] = f - c[idx]
        row += 1
    
    # A_eq @ x = b_eq: sum(delta) = 0, t >= 0 (handled by bound)
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds on x: delta free (but bounded by trust region), t >= 0
    bounds = [(None, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            delta = res.x[:n].copy()
            t_new = res.x[n]
            # Clamp delta to respect box constraints
            delta = np.clip(delta, -h, 1.0 - h)
            return delta, float(t_new)
    except Exception:
        pass
    return None, None

def pair_descent(h, time_limit, seed=42):
    """Coordinate descent: swap mass between pairs."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = compute_c5(h)
    t0 = time.time()
    rng = np.random.RandomState(seed)

        # Try random pairs
        for _ in range(200):

            i, j = rng.randint(0, n, 2)
            if i == j:
                continue
            max_transfer = min(h[i], 1.0 - h[j])
            if max_transfer < 1e-6:

                if c_new < best_c5 - 1e-12:
                    best_h = h_new
                    best_c5 = c_new
                    improved = True
                    break
        
        if not improved:
            # Try larger random perturbation
            break
    
    return best_h, best_c5

def resample(h_old, n_new):

def project_h(p, n):

def gradient_row(h, k, n):
    """Compute gradient of correlation at index k w.r.t. h, times dx."""
    dx = 2.0 / n
    lag = k - (n - 1)

    else:

        M = n - L
        grad[:M] += dx * (1.0 - h[L:])

    return grad
