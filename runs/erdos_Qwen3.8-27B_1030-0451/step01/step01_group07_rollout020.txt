        for i in range(n - 1, -1, -1):
            if deadline is not None and time.monotonic() > deadline:
                return best_h, best_val
            old = h[i]
            h[i] = 0.0
            C0 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = 1.0
            C1 = np.correlate(h, 1.0 - h, mode='full')
            h[i] = old
            D = C1 - C0
            h[i] = solve_1d_minimax(C0, D)
        cur_val = objective(h)
        if cur_val < best_val - 1e-15:
            best_val = cur_val
            best_h = h.copy()
        elif abs(cur_val - best_val) < 1e-13:
            break
    cur_val = objective(h)
    if cur_val < best_val:
        return h, cur_val
    return best_h, best_val


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def make_balanced_binary(n, rng, pattern='random'):
    target = n // 2
    if pattern == 'random':
        h = np.zeros(n)
        ones = rng.choice(n, size=target, replace=False)
        h[ones] = 1.0
    elif pattern == 'sturmian':
        rho = target / n
        phase = rng.random()
        h = np.zeros(n)
        for i in range(n):
            if math.floor((i + 1) * rho + phase) > math.floor(i * rho + phase):
                h[i] = 1.0
        cnt = int(h.sum())
        if cnt > target:
            excess = np.where(h == 1.0)[0]
            to_remove = rng.choice(excess, size=cnt - target, replace=False)
            h[to_remove] = 0.0
        elif cnt < target:
            deficit = np.where(h == 0.0)[0]
            to_add = rng.choice(deficit, size=target - cnt, replace=False)
            h[to_add] = 1.0
    elif pattern == 'periodic':
        period = int(math.sqrt(n))
        if period < 2:
            period = 2
        block_ones = max(1, period // 2)
        block = np.zeros(period)
        block[:block_ones] = 1.0
        h = np.tile(block, n // period + 1)[:n]
        cnt = int(h.sum())
        if cnt > target:
            excess = np.where(h == 1.0)[0]
            to_remove = rng.choice(excess, size=cnt - target, replace=False)
            h[to_remove] = 0.0
        elif cnt < target:
            deficit = np.where(h == 0.0)[0]
            to_add = rng.choice(deficit, size=target - cnt, replace=False)
            h[to_add] = 1.0
    else:
        h = np.zeros(n)
        h[:target] = 1.0
    return h


def pair_moves(h, rng, n_tries, deadline=None):
    n = len(h)
    best_h = h.copy()
    best_val = objective(h)
    for _ in range(n_tries):
        if deadline is not None and time.monotonic() > deadline:
            break
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        op = rng.random()
        c = h.copy()
        if op < 0.33:
            c[i], c[j] = h[j], h[i]
        elif op < 0.66:
            c[i] = 1.0 - h[i]
            c[j] = 1.0 - h[j]
        else:
            d = rng.random() * 0.3
            if d <= h[i] and d <= 1.0 - h[j]:
                c[i] -= d
                c[j] += d
            elif d <= h[j] and d <= 1.0 - h[i]:
                c[j] -= d
                c[i] += d
            else:
                continue
        cv = objective(c)
        if cv < best_val - 1e-14:
            best_val = cv
            best_h = c.copy()
    return best_h, best_val


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 10.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        parent = make_balanced_binary(131, rng, 'random')
    n0 = len(parent)
    h0 = project_to_simplex(parent, n0 / 2.0)

    best_h = h0.copy()
    best_c5 = objective(best_h)

    def consider(cand):
        nonlocal best_h, best_c5
        try:
            c = objective(cand)
            if np.all(np.isfinite(cand)) and c < best_c5 - 1e-13:
                best_c5 = c
                best_h = np.array(cand, dtype=float, copy=True)
        except Exception:
            pass

    # Phase 1: Coordinate descent from parent
    t1 = time.monotonic()
    elapsed1 = t1 - t0
    alloc1 = min(0.30 * effective, deadline - t1)
    if alloc1 > 2.0:
        ch, cv = coord_descent(h0, alloc1, deadline)
        consider(ch)

    # Phase 2: Structured starts at various n
    t2 = time.monotonic()
    alloc2 = min(0.20 * effective, deadline - t2)
    if alloc2 > 2.0:
        patterns = ['sturmian', 'periodic', 'random']
        phases = [0.0, 0.3, 0.7, 0.15, 0.55]
        n_starts = len(patterns) * min(len(phases), 3)
        per_start = alloc2 / n_starts
        for pi, pat in enumerate(patterns):
            for ph_idx in range(min(3, len(phases))):
                if time.monotonic() > deadline:
                    break
                ph = phases[ph_idx]
                for n_try in [n0]:
                    h_s = make_balanced_binary(n_try, rng, pat)
                    h_s = project_to_simplex(h_s, n_try / 2.0)
                    a_s = min(per_start, deadline - time.monotonic())
                    if a_s < 1.0:
                        break
                    ch, cv = coord_descent(h_s, a_s, deadline)
                    consider(ch)

    # Phase 3: Perturbation + restart from best (multiple noise levels)
    t3 = time.monotonic()
    alloc3 = min(0.30 * effective, deadline - t3)
    if alloc3 > 3.0:
        noise_levels = [0.02, 0.05, 0.10, 0.03, 0.08, 0.15]
        per_restart = alloc3 / len(noise_levels)
        for nl in noise_levels:
            if time.monotonic() > deadline:
                break
            a_r = min(per_restart, deadline - time.monotonic())
            if a_r < 1.0:
                break
            h_p = np.clip(best_h + rng.normal(0, nl, len(best_h)), 0, 1)
            h_p = project_to_simplex(h_p, len(best_h) / 2.0)
            ch, cv = coord_descent(h_p, a_r * 0.7, deadline)
            if time.monotonic() < deadline - 1.0:
                ch2, cv2 = pair_moves(ch, rng, int(20000), deadline)
                consider(ch2)
            consider(ch)

    # Phase 4: Different n values
    t4 = time.monotonic()
    alloc4 = min(0.15 * effective, deadline - t4)
    if alloc4 > 3.0:
        n_candidates = [n0 - 2, n0 + 2, n0 - 4, n0 + 4, n0 + 8, n0 - 8,
                        int(n0 * 1.5), int(n0 * 0.75), n0 + 16]
        n_candidates = sorted(set(x for x in n_candidates if 60 <= x <= 500))
        per_n = alloc4 / max(len(n_candidates), 1)
        for nn in n_candidates:
            if time.monotonic() > deadline:
                break
            a_n = min(per_n, deadline - time.monotonic())
            if a_n < 1.0:
                break
            h_r = resample(best_h, nn)
            h_r = project_to_simplex(h_r, nn / 2.0)
            ch, cv = coord_descent(h_r, a_n * 0.8, deadline)
            if time.monotonic() < deadline - 1.0:
                ch2, cv2 = pair_moves(ch, rng, int(10000), deadline)
                consider(ch2)
            consider(ch)

    # Phase 5: Final polish and pair moves on global best
    t5 = time.monotonic()
    remaining = max(0.0, deadline - t5)
    if remaining > 1.0:
        a_pm = remaining * 0.6
        ch, cv = pair_moves(best_h, rng, int(a_pm * 5000), deadline)
        consider(ch)
        a_cd = max(0.5, remaining * 0.3)
        if time.monotonic() < deadline - 0.5:
            ch, cv = coord_descent(best_h, a_cd, deadline)
            consider(ch)

    # Final feasibility enforcement
    best_h = project_to_simplex(best_h, len(best_h) / 2.0)
    best_c5 = objective(best_h)
    return best_h, float(best_c5), int(len(best_h))
