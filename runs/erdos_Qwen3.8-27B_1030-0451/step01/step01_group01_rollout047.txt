import numpy as np
import time

def project_sum_box(h, target):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(h + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            mask = y < 1.0
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.clip(y[mask] + r / cnt, 0.0, 1.0)
        else:
            mask = y > 0.0
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.clip(y[mask] + r / cnt, 0.0, 1.0)
    return y

def compute_full(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_c5(h):
    return float(np.max(compute_full(h)))

def d_row(h, i):
    n = len(h)
    center = n - 1
    d = np.zeros(2 * n - 1)
    # Vectorized computation
    # For each lag idx (0 to 2n-2), L = idx - center
    # d[idx] = (1-h[i+L]) if 0<=i+L<n else 0, minus h[i-L] if 0<=i-L<n else 0
    L = np.arange(2 * n - 1) - center  # shape (2n-1,)
    # Forward: i+L in [0, n-1]
    fwd_mask = (i + L >= 0) & (i + L < n)
    # Backward: i-L in [0, n-1]
    bwd_mask = (i - L >= 0) & (i - L < n)
    d[fwd_mask] += 1.0 - h[i + L[fwd_mask]]
    d[bwd_mask] -= h[i - L[bwd_mask]]
    return d

def d_matrix(h):
    n = len(h)
    mat = np.zeros((n, 2 * n - 1))
    for i in range(n):
        mat[i] = d_row(h, i)
    return mat

def line_search(full, c, dmin, dmax):
    if dmax - dmin < 1e-14:
        return dmin
    lo, hi = dmin, dmax
    mid = 0.5 * (lo + hi)
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        vals = full + mid * c
        mx = vals.max()
        tol = 1e-13
        active = vals >= mx - tol
        ca = c[active]
        if ca.size == 0:
            break
        if ca.min() > 1e-12:
            hi = mid
        elif ca.max() < -1e-12:
            lo = mid
        else:
            break
    # Final candidate check
    cands = np.array([dmin, dmax, mid, 0.5 * (lo + hi)])
    best_d = dmin
    best_v = np.inf
    for d in cands:
        v = (full + d * c).max()
        if v < best_v:
            best_v = v
            best_d = d
    return best_d

def coord_descent(h, max_sweeps=15, deadline=None, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    n = len(h)
    h = h.copy()
    full = compute_full(h)
    dmat = d_matrix(h)
    best_c = float(full.max())
    best_h = h.copy()
    
    total_pairs = n * (n - 1) // 2
    pair_indices = np.array([(i, j) for i in range(n) for j in range(i + 1, n)])
    
    for sweep in range(max_sweeps):
        if deadline is not None and time.time() > deadline:
            break
        order = rng.permutation(total_pairs)
        improved = 0
        for pidx in order:
            i, j = pair_indices[pidx]
            dmin = max(-h[i], 0.0)
            dmax = min(1.0 - h[i], h[j])
            if dmax - dmin < 1e-12:
                continue
            c = dmat[i] - dmat[j]
            delta = line_search(full, c, dmin, dmax)
            new_max = (full + delta * c).max()
            old_max = full.max()
            if new_max < old_max - 1e-12:
                full += delta * c
                h[i] += delta
                h[j] -= delta
                dmat[i] = d_row(h, i)
                dmat[j] = d_row(h, j)
                improved += 1
        cur_c = float(full.max())
        if cur_c < best_c - 1e-13:
            best_c = cur_c
            best_h = h.copy()
        if improved == 0:
            break
    
    return best_h, best_c

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False) + 2.0 / (2 * n_old)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 2.0 / (2 * n_new)
    # Simple linear interpolation
    h_new = np.interp(x_new, x_old, h_old)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def make_binary(n, rng):
    """Create a random balanced binary sequence."""
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    t0 = time.time()
    safety = 5.0
    deadline = t0 + budget_s - safety
    
    # Get initial values
    h_init = None
    n_init = 84
    ig = globals().get('initial_h_values', None)

        try:
            h_try = np.array(ig, dtype=float).copy()
            n_init = len(h_try)
            if n_init >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_init / 2.0)
                h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(n_init, 0.5)
    
    best_h = h_init.copy()
    best_c = eval_c5(best_h)
    
    def time_ok():
        return time.time() < deadline
    
    # Phase 1: Coordinate descent at original resolution
    if time_ok():
        dl1 = min(deadline - 60, t0 + budget_s * 0.25)
        h1, c1 = coord_descent(best_h, max_sweeps=30, deadline=dl1, rng=rng)
        if c1 < best_c:
            best_h, best_c = h1, c1
    
    # Phase 2: Try binary initialization at same resolution
    if time_ok():
        n_cur = len(best_h)
        for attempt in range(3):
            if not time_ok():

            h_bin = make_binary(n_cur, rng)
            h_bin = project_sum_box(h_bin, n_cur / 2.0)

            h_try, c_try = coord_descent(h_bin, max_sweeps=15, deadline=dl, rng=rng)
            if c_try < best_c:

    # Phase 3: Higher resolution
    target_ns = [168, 336, 500, 672]
    for n_t in target_ns:
        if not time_ok():
            break
        if n_t <= len(best_h):
            continue
        h_interp = interpolate_h(best_h, n_t)
        c_interp = eval_c5(h_interp)
        # How much time for this resolution?
        # Estimated time per sweep ~ 50 * n^3 / 1e9 seconds
        est_per_sweep = 50.0 * (n_t ** 3) / 1e9
        max_sw = max(3, min(20, int((deadline - time.time() - 60) / max(est_per_sweep, 1.0))))
        dl = min(deadline - 30, time.time() + max_sw * est_per_sweep * 1.5 + 30)

        dl4 = deadline - safety

        max_sw = max(3, min(30, int((dl4 - time.time()) / max(est_per_sweep, 0.5))))
        h_pol, c_pol = coord_descent(best_h, max_sweeps=max_sw, deadline=dl4, rng=rng)
        if c_pol < best_c:
            best_h, best_c = h_pol, c_pol
    
    # Final: ensure feasibility
    n_final = len(best_h)
    best_h = project_sum_box(best_h, n_final / 2.0)
    final_c = eval_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    
    return best_h, float(best_c), n_final
