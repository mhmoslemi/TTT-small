def _smooth_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _smooth_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h
    active = np.where(alpha > 1e-10)[0]
    for k in active:
        a = alpha[k]
        d = k - (n - 1)
        if d >= 0:
            hi = n - d
            grad[:hi] += a * omh[d:]
            grad[d:] -= a * h[:hi]
        else:
            dd = -d
            hi = n - dd
            grad[dd:] += a * omh[:hi]
            grad[:hi] -= a * h[dd:]
    return grad * dx

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def _smooth_optimize(h0, n, deadline, n_tau=16, tau_ratio=0.5):
    global _G_T, _G_N, _G_TAU
    _G_T = n / 2.0
    _G_N = n
    target = n / 2.0
    c5_init = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_init
    dx = 2.0 / n
    c_vals = np.correlate(h0, 1.0 - h0, 'full') * dx
    spread = max(1e-6, c_vals.max() - c_vals.mean())
    tau_start = max(0.015, 0.4 * spread)
    tau_sched = tau_start * (tau_ratio ** np.arange(n_tau))
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n
    h_curr = h0.copy()
    stall = 0
    for i in range(len(tau_sched)):
        if time.time() > deadline - 5:
            break
        _G_TAU = tau_sched[i]
        maxiter = 50 if tau_sched[i] > 0.005 else 90
        try:
            res = minimize(_smooth_obj, h_curr, method='SLSQP', jac=_smooth_grad,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_curr = project_feasible(res.x, target)
        except Exception:
            pass
        c5e = evaluate_c5(h_curr)
        if c5e < best_c5 - 1e-13:
            best_c5 = c5e
            best_h = h_curr.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 2 and i < len(tau_sched) - 1:
                rng = np.random.default_rng((n * 1000 + i) & 0xFFFFFFFF)
                h_curr = project_feasible(h_curr + rng.standard_normal(n) * 0.004, target)
                stall = 0
    return best_h, best_c5

def _compute_subgrad(h, eps=1e-5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    c_max = c.max()
    active = np.where(c >= c_max - eps)[0]
    grad = np.zeros(n)
    omh = 1.0 - h
    for k in active:
        d = k - (n - 1)
        if d >= 0:
            hi = n - d
            grad[:hi] += omh[d:]
            grad[d:] -= h[:hi]
        else:
            dd = -d
            hi = n - dd
            grad[dd:] += omh[:hi]
            grad[:hi] -= h[dd:]
    grad /= len(active)
    grad *= dx
    grad -= grad.mean()
    return -grad, c_max

def _exact_polish(h, n_iters=3000, deadline=None, beta=0.85, lr0=0.015, eps=1e-5):
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = evaluate_c5(h)
    v = np.zeros(n)
    lr = lr0
    stall = 0
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline - 2:
            break
        direction, c_cur = _compute_subgrad(h, eps)
        v = beta * v + direction
        norm_v = np.linalg.norm(v)
        if norm_v < 1e-12:
            break
        v = v / norm_v
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            t = lr * factor
            h_try = h + t * v
            h_try = project_feasible(h_try, target)
            c_try = evaluate_c5(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                stall = 0
                found = True
                break
        if not found:
            lr *= 0.75
            stall += 1
            if stall > 40:
                break
    return best_h, best_c

def _fixed_lag_descent(h, active_lags, n_iters=2000, deadline=None, lr0=0.01):
    n = len(h)
    target = n / 2.0
    dx = 2.0 / n
    best_h = h.copy()
    best_c = evaluate_c5(h)
    lr = lr0
    stall = 0
    for it in range(n_iters):
        if deadline is not None and time.time() > deadline - 2:
            break
        grad = np.zeros(n)
        omh = 1.0 - h
        for k in active_lags:
            d = k - (n - 1)
            if d >= 0:
                hi = n - d
                grad[:hi] += omh[d:]
                grad[d:] -= h[:hi]
            else:
                dd = -d
                hi = n - dd
                grad[dd:] += omh[:hi]
                grad[:hi] -= h[dd:]
        grad /= len(active_lags)
        grad *= dx
        grad -= grad.mean()
        found = False
        for factor in [1.0, 0.5, 0.25, 0.1, 0.05]:
            t = lr * factor
            h_try = h - t * grad
            h_try = project_feasible(h_try, target)
            c_try = evaluate_c5(h_try)
            if c_try < best_c - 1e-14:
                h = h_try
                best_c = c_try
                best_h = h.copy()
                stall = 0
                found = True
                break
        if not found:
            lr *= 0.7
            stall += 1
            if stall > 30:
                break
    return best_h, best_c

def _identify_active_lags(h, k=5):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    idx = np.argsort(c)[-k:]
    return sorted(idx.tolist())

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    try:
        parent = np.array(initial_h_values, dtype=np.float64)
    except NameError:
        parent = np.full(240, 0.5)
    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)
    best_h = parent.copy()
    best_c5 = evaluate_c5(parent)
    rng = np.random.default_rng(seed)

    # Phase 1: Smooth SLSQP at parent resolution
    if time.time() < deadline - 120:
        h_opt, c5_opt = _smooth_optimize(parent, n_par, deadline, n_tau=18, tau_ratio=0.48)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Exact subgradient polish with momentum
    if time.time() < deadline - 60:
        n_max_polish = 960
        if len(best_h) <= n_max_polish:
            n_iters = max(500, min(5000, int((deadline - time.time() - 10) * 30)))
            h_opt, c5_opt = _exact_polish(best_h, n_iters=n_iters, deadline=deadline,
                                          beta=0.85, lr0=0.02, eps=1e-5)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt

    # Phase 3: Fixed active-lag descent
    if time.time() < deadline - 30:
        active = _identify_active_lags(best_h, k=5)
        n_iters = max(500, min(3000, int((deadline - time.time() - 5) * 25)))
        h_opt, c5_opt = _fixed_lag_descent(best_h, active, n_iters=n_iters, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Try higher resolution
    for mult in [2, 3, 4, 6]:
        n_new = n_par * mult
        if n_new > 2400:
            break
        if time.time() > deadline - 100:
            break
        h0 = interp_h(best_h, n_new)
        n_tau = max(10, 18 - mult)
        h_opt, c5_opt = _smooth_optimize(h0, n_new, deadline, n_tau=n_tau, tau_ratio=0.55)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
        # Quick exact polish at higher res
        if time.time() < deadline - 30 and n_new <= 960:
            n_iters_p = max(300, int((deadline - time.time() - 10) * 15))
            h_opt2, c5_opt2 = _exact_polish(best_h, n_iters=n_iters_p, deadline=deadline,
                                            beta=0.85, lr0=0.015, eps=1e-5)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2

    # Phase 5: Multi-start perturbations on best
    if time.time() < deadline - 15:
        n_cur = len(best_h)
        for amp in [0.003, 0.008, 0.015]:
            for _ in range(2):
                if time.time() > deadline - 5:
                    break
                h_try = project_feasible(best_h + rng.standard_normal(n_cur) * amp, n_cur / 2.0)
                n_iters_small = max(200, int((deadline - time.time() - 3) * 20))
                h_opt, c5_opt = _exact_polish(h_try, n_iters=n_iters_small,
                                              deadline=deadline, beta=0.8, lr0=0.01, eps=1e-5)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt

    # Phase 6: Final fixed-lag polish
    if time.time() < deadline - 5:
        active = _identify_active_lags(best_h, k=3)
        n_iters_f = max(200, int((deadline - time.time() - 2) * 30))
        h_opt, c5_opt = _fixed_lag_descent(best_h, active, n_iters=n_iters_f, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Final feasibility repair
    n_final = len(best_h)
    target = n_final / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] -= diff / cnt
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] += diff / cnt
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, n_final)
