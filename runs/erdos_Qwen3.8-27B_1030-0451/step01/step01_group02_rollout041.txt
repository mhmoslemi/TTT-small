def compute_gradient(h, alpha, dx, n):
    """Compute gradient of smoothed max objective.
    alpha: weights for each lag, indexed 0..2n-2 corresponding to lag = -(n-1) .. n-1
    """
    grad = np.zeros(n)
    # lag = 0 (index n-1 in alpha)
    a0 = alpha[n-1]
    if abs(a0) > 1e-15:
        grad += a0 * dx * (1 - 2*h)
    
    # positive lags k = 1, ..., n-1 (indices n .. 2n-2 in alpha)
    for k in range(1, n):
        ak = alpha[n-1+k]
        if abs(ak) < 1e-15:
            continue
        # dC(k)/dh[j] = dx * [(1-h[j-k]) * I[j>=k] - h[j+k] * I[j<=n-1-k]]
        term1 = np.zeros(n)
        term1[k:] = (1 - h[:-k])  # h[j-k] for j >= k, which is h[0..n-k-1]
        term2 = np.zeros(n)
        term2[:n-k] = h[k:]  # h[j+k] for j <= n-1-k, which is h[k..n-1]
        grad += ak * dx * (term1 - term2)
    
    # negative lags k = -1, ..., -(n-1) (indices 0 .. n-2 in alpha)
    for k in range(1, n):
        ak = alpha[n-1-k]  # lag = -k
        if abs(ak) < 1e-15:
            continue
        # dC(-k)/dh[j] = dx * [(1-h[j+k]) * I[j<=n-1-k] - h[j-k] * I[j>=k]]
        term1 = np.zeros(n)
        term1[:n-k] = (1 - h[k:])  # h[j+k] for j <= n-1-k
        term2 = np.zeros(n)
        term2[k:] = h[:-k]  # h[j-k] for j >= k
        grad += ak * dx * (term1 - term2)
    
    return grad
