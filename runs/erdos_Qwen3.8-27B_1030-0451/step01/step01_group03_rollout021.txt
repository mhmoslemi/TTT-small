    c5_cur = best_c5

def _project(y, target):

    for _ in range(70):

        s = np.clip(y + mid, 0.0, 1.0).sum()

        else:

def _interp_to_n(h, n):

    if np_ == n:
        return h.copy()

def _grad_weighted(h, weights):
    n = len(h)
    grad = np.zeros(n)
    active = np.where(weights > 1e-14)[0]
    for k in active:
        w = weights[k]
        s = k - (n - 1)
        if s >= 0:
            grad[:n-s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n-s]
        else:
            ss = -s
            grad[ss:] += w * (1.0 - h[:n-ss])
            grad[:n-ss] -= w * h[ss:]
    return grad

        else:

            w_raw = np.maximum(w_raw, 0.0)

            mask_active[active_idx] = True
            w_raw[~mask_active] = 0.0
            w_sum = w_raw.sum()
            if w_sum < 1e-14:
                weights_full[active_idx] = 1.0 / K
            else:

        g = _grad_weighted(h, weights_full)

def _pair_swap_polish(h0, n, target, deadline, max_rounds=500):
    best_h = h0.copy()
    best_c5 = _eval_c5(h0)
    h = h0.copy()
    c5_cur = best_c5
    dx = 2.0 / n
    for rnd in range(max_rounds):

            break

        cmax = cvec.max()
        top_k = np.argsort(cvec)[-8:]
        # Find candidate pairs: indices with high gradient contribution
        grad = np.zeros(n)
        for k in top_k:
            s = k - (n - 1)
            if s >= 0:
                grad[:n-s] += (1.0 - h[s:])
                grad[s:] -= h[:n-s]
            else:
                ss = -s
                grad[ss:] += (1.0 - h[:n-ss])
                grad[:n-ss] -= h[ss:]
        # Select top |gradient| indices
        top_grad = np.argsort(np.abs(grad))[-20:]
        n_tried = 0
        improved = False
        for i_idx in range(len(top_grad)):
            i = int(top_grad[i_idx])
            for j_idx in range(i_idx + 1, len(top_grad)):
                j = int(top_grad[j_idx])
                if i == j:
                    continue
                # Try swapping portions

                    h_try = h.copy()
                    delta = (h[j] - h[i]) * frac
                    h_try[i] += delta
                    h_try[j] -= delta
                    if h_try[i] < -1e-9 or h_try[i] > 1.0 + 1e-9:

                    if h_try[j] < -1e-9 or h_try[j] > 1.0 + 1e-9:

def _slsqp_polish(h0, n, target, deadline):

    best_c5 = _eval_c5(h)
    best_h = h.copy()

    bounds = [(0.0, 1.0)] * n

    def obj_func(hv):
        c = np.correlate(hv, 1.0 - hv, 'full') * dx
        m = c.max()
        s = np.clip((c - m) / tau, -400.0, 400.0)
        e = np.exp(s)
        return float(m + tau * np.log(np.sum(e)))

    def obj_grad(hv):
        c = np.correlate(hv, 1.0 - hv, 'full') * dx
        m = c.max()
        s = np.clip((c - m) / tau, -400.0, 400.0)
        e = np.exp(s)
        alpha = e / e.sum()
        g = _grad_weighted(hv, alpha)
        return g * dx

    def eq_fun(hv):
        return float(np.sum(hv) - target)

    def eq_jac(hv):
        return np.ones(n)

    constraints.append({'type': 'eq', 'fun': eq_fun, 'jac': eq_jac})
    try:
        res = minimize(obj_func, h, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 100, 'ftol': 1e-12})
        h_new = _project(res.x, target)
        c5_new = _eval_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new

            best_n = n_child
        # Pair polish at higher res
        if time.time() < deadline - 30:
            h_ps2, c5_ps2 = _pair_swap_polish(best_h, best_n, best_n / 2.0, deadline, max_rounds=200)
            if c5_ps2 < best_c5:
                best_c5 = c5_ps2
                best_h = h_ps2

    # Phase 6: Final active descent with smaller K for precision
    if time.time() < deadline - 20:
        h_fin, c5_fin = _active_descent(best_h, best_n, best_n / 2.0, deadline, K=6, max_iter=2000, mu=0.5)
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin

    # Final cleanup
    n_final = len(best_h)
    diff = n_final / 2.0 - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] -= diff / cnt
    np.clip(best_h, 0.0, 1.0, out=best_h)
    # Re-fix sum after clip
    diff2 = n_final / 2.0 - best_h.sum()
    if abs(diff2) > 1e-12:
        mask2 = (best_h > 0) & (best_h < 1.0)
        cnt2 = int(mask2.sum())
        if cnt2 > 0:
            best_h[mask2] += diff2 / cnt2
    best_c5 = _eval_c5(best_h)

    return (best_h, best_c5, n_final)
