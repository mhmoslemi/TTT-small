def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max())

def _project(y, target):
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _interp(hp, nn):
    np_ = len(hp)
    if nn == np_:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return _project(h, nn / 2.0)

def _smooth_obj(h):
    tau = _G['tau']
    dx = _G['dx']
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    ss = e.sum()
    if ss < 1e-300:
        return float(m)
    return float(m + tau * np.log(ss))

def _smooth_grad(h):
    tau = _G['tau']
    dx = _G['dx']
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    m = c.max()
    s = np.clip((c - m) / tau, -500.0, 500.0)
    e = np.exp(s)
    ss = e.sum()
    if ss < 1e-300:
        return np.zeros(n)
    alpha = e / ss
    grad = np.zeros(n)
    nc = len(c)
    for k in range(nc):
        if alpha[k] < 1e-12:
            continue
        a = alpha[k]
        d = k - (n - 1)
        omh = 1.0 - h
        if d >= 0:
            hi = n - d
            if hi > 0:
                grad[:hi] += a * omh[d:]
                grad[:hi] -= a * h[:hi] * 0.0
                # d(c_k)/d(h_i) = (1-h_{i+d}) for i in [0, n-d-1]
                # plus contribution when h_i appears as the second factor: -h_{i-d}
                # For lag d>=0: c_k = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
                # dc/dh[i] = (1-h[i+d]) + (-h[i-d]) if i-d valid
                # For i in [0, n-1-d]: dc/dh[i] = (1-h[i+d])
                # For i in [n-d, n-1]: h[i] appears as second factor in term i-d
                #   term is h[i-d]*(1-h[i]), dc/dh[i] = -h[i-d]
                grad[d:] += -a * h[:n - d]  # second factor contribution for i >= d
                # Wait this double counts. Let me redo.
                # c_k = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
                # dc/dh[j] for j in [0, n-1-d]: (1-h[j+d])  [first factor]
                # dc/dh[j] for j in [d, n-1]: -h[j-d]       [second factor: h[j-d]*(1-h[j])]
                # For j in [d, n-1-d]: both apply: (1-h[j+d]) - h[j-d]
                pass
        else:
            dd = -d
            hi = n - dd
            if hi > 0:
                # c_k = sum_{i=dd}^{n-1} h[i]*(1-h[i-dd])
                pass

    # Let me redo the gradient computation more carefully
    grad = np.zeros(n)
    for k in range(nc):
        a = alpha[k]
        if a < 1e-14:
            continue
        d = k - (n - 1)
        if d >= 0:
            # c_k = sum_{i=0}^{n-1-d} h[i] * (1 - h[i+d])
            lo_i = 0
            hi_i = n - 1 - d
            if hi_i < lo_i:
                continue
            # First factor: d/dh[i] of h[i] = 1, times (1-h[i+d])
            grad[lo_i:hi_i+1] += a * (1.0 - h[lo_i + d:hi_i + 1 + d])
            # Second factor: d/dh[j] of (1-h[j]) = -1, times h[j-d]
            # j = i + d, so j ranges from d to n-1
            lo_j = d
            hi_j = n - 1
            grad[lo_j:hi_j+1] -= a * h[lo_j - d:hi_j + 1 - d]
        else:
            dd = -d
            # c_k = sum_{i=dd}^{n-1} h[i] * (1 - h[i-dd])
            lo_i = dd
            hi_i = n - 1
            # First factor
            grad[lo_i:hi_i+1] += a * (1.0 - h[lo_i - dd:hi_i + 1 - dd])
            # Second factor: j = i - dd, j ranges from 0 to n-1-dd
            lo_j = 0
            hi_j = n - 1 - dd
            grad[lo_j:hi_j+1] -= a * h[lo_j + dd:hi_j + 1 + dd]
    return grad * dx

def _c_eq(h):
    return float(h.sum() - _G['target'])

def _c_eq_jac(h):
    return np.ones(_G['n'])

def _slsqp_optimize(h0, n, deadline, tau_ratio=0.58, n_tau=14):
    global _G
    _G['n'] = n
    _G['target'] = n / 2.0
    _G['dx'] = 2.0 / n

    best_h = h0.copy()
    best_c5 = _eval_c5(h0)
    c_vals = np.correlate(h0, 1.0 - h0, mode='full') * (2.0/n)
    spread = max(1e-6, c_vals.max() - c_vals.mean())
    tau0 = max(0.012, 0.25 * spread)
    tau_sched = [tau0 * (tau_ratio ** i) for i in range(n_tau)]

    constraints = [{'type': 'eq', 'fun': _c_eq, 'jac': _c_eq_jac}]
    bounds = [(0.0, 1.0)] * n
    h_curr = h0.copy()
    stall = 0

    for tau in tau_sched:
        if time.time() > deadline - 5:
            break
        _G['tau'] = tau
        maxiter = 30 if tau > 0.01 else 60
        try:
            res = minimize(_smooth_obj, h_curr, method='SLSQP', jac=_smooth_grad,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14})
            h_curr = _project(res.x, n / 2.0)
        except Exception:
            pass
        c5e = _eval_c5(h_curr)
        if c5e < best_c5 - 1e-10:
            best_c5 = c5e
            best_h = h_curr.copy()
            stall = 0
        else:
            stall += 1
            if stall >= 3:
                break  # Move to next resolution or method

    return best_h, best_c5

def _pgd_optimize(h0, n, deadline, max_iter=2000):
    """Projected gradient with adaptive step (Barzilai-Borwein-like)."""
    best_h = h0.copy()
    best_c5 = _eval_c5(h0)
    h = h0.copy()
    dx = 2.0 / n
    n_tau_stages = 8
    tau_start = max(0.02, 0.15)
    tau_end = 0.003

    for stage in range(n_tau_stages):
        if time.time() > deadline - 10:
            break
        tau = tau_start * (tau_end / tau_start) ** (stage / max(1, n_tau_stages - 1))
        iters_this_stage = max_iter // n_tau_stages
        alpha = 0.01
        h_prev = h.copy()
        g_prev = None

        for it in range(iters_this_stage):
            if time.time() > deadline - 10:
                break
            c = np.correlate(h, 1.0 - h, mode='full') * dx
            m = c.max()
            s = np.clip((c - m) / tau, -500.0, 500.0)
            e = np.exp(s)
            ss = e.sum()
            if ss < 1e-300:
                break
            alpha_w = e / ss

            # Compute gradient
            grad = np.zeros(n)
            nc = len(c)
            for k in range(nc):
                a = alpha_w[k]
                if a < 1e-14:
                    continue
                d = k - (n - 1)
                if d >= 0:
                    hi = n - d
                    if hi > 0:
                        grad[:hi] += a * (1.0 - h[d:d+hi])
                        grad[d:] -= a * h[:hi]
                else:
                    dd = -d
                    hi = n - dd
                    if hi > 0:
                        grad[dd:] += a * (1.0 - h[:hi])
                        grad[:hi] -= a * h[dd:]
            grad *= dx

            # BB step size
            if g_prev is not None:
                s_vec = h - h_prev
                y_vec = grad - g_prev
                sy = s_vec @ y_vec
                yy = y_vec @ y_vec
                if yy > 1e-14 and sy > 1e-14:
                    alpha_new = sy / yy
                    alpha = np.clip(0.5 * alpha + 0.5 * alpha_new, 1e-6, 0.5)

            # Gradient step
            y = h - alpha * grad
            h_new = _project(y, n / 2.0)

            # Objective check
            c5_new = _eval_c5(h_new)
            if c5_new < best_c5 - 1e-10:
                best_c5 = c5_new
                best_h = h_new.copy()

            h_prev = h
            h = h_new
            g_prev = grad

    return best_h, best_c5

def _pair_moves(h, n, n_iter=200, deadline=None):
    """Local pair-move search to reduce max correlation."""
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    dx = 2.0 / n
    h = h.copy()
    c5 = best_c5
    no_improve = 0

    for it in range(n_iter):
        if deadline is not None and time.time() > deadline - 5:
            break
        if no_improve > 30:
            break

        c_raw = np.correlate(h, 1.0 - h, mode='full')
        c = c_raw * dx
        cmax = c.max()
        # Focus on lags near the max
        threshold = cmax - 0.005
        active = np.where(c >= threshold)[0]
        if len(active) == 0:
            active = np.array([int(np.argmax(c))])

        improved = False
        rng_local = np.random.default_rng(it + 12345)
        n_tries = min(8, len(active))
        for _ in range(n_tries):
            k = int(active[rng_local.integers(len(active))])
            d = k - (n - 1)
            # Find best pair for this lag
            # c_k = sum h[i]*(1-h[i+d]) for valid i
            # To reduce c_k, move mass from high h[i] to high h[i+d]
            # or equivalently, increase h where h[i+d] is high and h[i] is low
            if d >= 0:
                valid_i = np.arange(0, n - d)
            else:
                dd = -d
                valid_i = np.arange(dd, n)

            # Score: for each i in valid_i, "benefit" of moving mass from i to i+d
            # or from i+d to i (whichever reduces c_k more)
            # Moving delta from i to i+d:
            #   dc_k = -delta*(1-h[i+d]) + delta*(1-h[i+d]) ... need full analysis
            # Simplify: just try moving small amounts at random nearby positions
            n_pairs = min(4, len(valid_i))
            for _ in range(n_pairs):
                idx = int(valid_i[rng_local.integers(len(valid_i))])
                if d >= 0:
                    partner = idx + d
                else:
                    partner = idx + d  # d is negative, so idx + d < idx
                if partner < 0 or partner >= n:
                    continue
                # Try moving from higher value to lower value
                deltas = [0.01, 0.02, 0.04, 0.005, 0.002]
                for delta in deltas:
                    if h[idx] - delta < 0 or h[partner] + delta > 1:
                        continue
                    h_try = h.copy()
                    h_try[idx] -= delta
                    h_try[partner] += delta
                    c5_try = _eval_c5(h_try)
                    if c5_try < c5 - 1e-9:
                        h = h_try
                        c5 = c5_try
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                        improved = True
                        break
                    # Also try reverse
                    if h[partner] - delta < 0 or h[idx] + delta > 1:
                        continue
                    h_try2 = h.copy()
                    h_try2[partner] -= delta
                    h_try2[idx] += delta
                    c5_try2 = _eval_c5(h_try2)
                    if c5_try2 < c5 - 1e-9:
                        h = h_try2
                        c5 = c5_try2
                        if c5 < best_c5:
                            best_c5 = c5
                            best_h = h.copy()
                        improved = True
                        break

        if not improved:
            no_improve += 1
        else:
            no_improve = 0

    return best_h, best_c5

def _make_init_osc(n, freq, amp, phase):
    """Create an oscillatory initialization."""
    x = np.arange(n, dtype=np.float64)
    h = 0.5 + amp * np.sin(2.0 * np.pi * freq * x / n + phase)
    return _project(h, n / 2.0)

def _make_init_two_level(n, a_val, pattern_rng):
    """Two-level initialization with a binary pattern."""
    h = np.full(n, a_val)
    # Set half the positions to 1-a_val
    n_high = n // 2
    idx = pattern_rng.permutation(n)[:n_high]
    h[idx] = 1.0 - a_val
    return _project(h, n / 2.0)

def _gradient_ascent_min_ac(h, n, deadline, max_iter=500):
    """
    Alternative: maximize min autocorrelation via gradient ascent on
    the smooth-min of autocorrelations.
    AC(d) = sum h[i]*h[i+d]
    min_d AC(d) is what we want to maximize.
    C5 = (n/2 - min_d AC(d)) * dx
    So minimize C5 = maximize -min AC = maximize smooth_min(-AC)
    """
    best_h = h.copy()
    best_c5 = _eval_c5(h)
    h = h.copy()
    dx = 2.0 / n

    # Compute autocorrelations: ac[d] = sum_i h[i]*h[i+d]
    # This is related to correlate(h, h, 'full')
    hh = np.correlate(h, h, mode='full')
    # hh[k] = sum h[i]*h[i-d] where d = k-(n-1)
    # AC(d) = hh[n-1-d] ... let me verify
    # np.correlate(h, h, 'full')[k] = sum_i h[i] * h[i - k + n - 1]
    # Let d = k - (n-1): = sum_i h[i] * h[i-d]
    # So autocorrelation at lag d is hh[n-1+d]... 

    # Actually: c_k with a=h, v=h: c[k] = sum_n h[n]*h[n-k+(n-1)]
    # d = k-(n-1): c[k] = sum_n h[n]*h[n - (k-(n-1))] = sum_n h[n]*h[n-d]
    # So autocorr at lag d is at index k = n-1+d

    # Our objective: c5_k = sum h[i]*(1-h[i+d]) = (n-|d|) - sum h[i]*h[i+d] (approximately, with bounds)
    # Actually c5_k = sum_{valid} h[i] - sum_{valid} h[i]*h[i+d]
    # The "sum of h" part varies with d (boundary effects), so this isn't exactly n/2 - AC(d)

    # Let me just use the smooth-max on the original correlations but approach from
    # a different angle: use a convex combination with emphasis on reducing the max.

    # Instead, let's do a simple projected gradient descent on smooth max with
    # different step size strategy (Polyak step).
    tau_start = 0.02
    tau_end = 0.003
    alpha = 0.02

    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        tau = tau_start * (tau_end / tau_start) ** (it / max(1, max_iter - 1))
        c = np.correlate(h, 1.0 - h, mode='full') * dx
        m = c.max()
        s = np.clip((c - m) / tau, -500.0, 500.0)
        e = np.exp(s)
        ss = e.sum()
        if ss < 1e-300:
            break
        w = e / ss

        # Compute gradient (same as before)
        grad = np.zeros(n)
        nc = len(c)
        for k in range(nc):
            a = w[k]
            if a < 1e-14:
                continue
            d = k - (n - 1)
            if d >= 0:
                hi = n - d
                if hi > 0:
                    grad[:hi] += a * (1.0 - h[d:d+hi])
                    grad[d:] -= a * h[:hi]
            else:
                dd = -d
                hi = n - dd
                if hi > 0:
                    grad[dd:] += a * (1.0 - h[:hi])
                    grad[:hi] -= a * h[dd:]
        grad *= dx

        # Polyak-like step: alpha = f_current / ||grad||^2
        gnorm2 = grad @ grad
        if gnorm2 > 1e-14:
            f_val = _smooth_obj(h) - m  # excess over the hard max (approx)
            alpha_p = max(0.001, min(0.1, f_val / gnorm2)) if f_val > 0 else 0.01
            alpha = 0.7 * alpha + 0.3 * alpha_p

        y = h - alpha * grad
        h = _project(y, n / 2.0)
        c5 = _eval_c5(h)
        if c5 < best_c5 - 1e-10:
            best_c5 = c5
            best_h = h.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15
    rng = np.random.default_rng(seed)

    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)

    n_par = len(parent)
    if n_par % 2 == 1:
        # Make even
        n_par = n_par - 1
        parent = parent[:n_par]
    parent = _project(parent, n_par / 2.0)

    best_h = parent.copy()
    best_c5 = _eval_c5(parent)
    best_n = n_par

    # Phase 1: Optimize at parent resolution with SLSQP
    if time.time() < deadline - 60:
        h_opt, c5_opt = _slsqp_optimize(parent, n_par, deadline, tau_ratio=0.58, n_tau=14)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n_par

        # Try different tau ratios
        if time.time() < deadline - 60:
            h0 = best_h.copy()
            # Perturb and retry with different parameters
            h_pert = _project(h0 + rng.standard_normal(n_par) * 0.02, n_par / 2.0)
            h_opt2, c5_opt2 = _slsqp_optimize(h_pert, n_par, deadline, tau_ratio=0.5, n_tau=16)
            if c5_opt2 < best_c5:
                best_c5 = c5_opt2
                best_h = h_opt2
                best_n = n_par

    # Phase 2: PGD at parent resolution
    if time.time() < deadline - 60:
        h_opt, c5_opt = _pgd_optimize(best_h, best_n, deadline, max_iter=1500)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = best_n

    # Phase 3: Scale up to 2x
    n2 = min(best_n * 2, 960)
    if n2 > best_n and time.time() < deadline - 80:
        h0 = _interp(best_h, n2)
        h_opt, c5_opt = _slsqp_optimize(h0, n2, deadline, tau_ratio=0.58, n_tau=12)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = n2
        # PGD refinement
        if time.time() < deadline - 30:
            h_opt, c5_opt = _pgd_optimize(best_h, best_n, deadline, max_iter=1000)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = best_n

    # Phase 4: Alternative initializations
    if time.time() < deadline - 100:
        # Try oscillatory initializations
        n_try = best_n
        inits = []
        for freq in [1, 2, 3, 5, 7]:
            for amp in [0.3, 0.45, 0.6]:
                for phase in [0, np.pi/2, np.pi]:
                    h_init = _make_init_osc(n_try, freq, amp, phase)
                    inits.append(h_init)
        # Two-level
        for a_val in [0.1, 0.15, 0.2, 0.3]:
            h_init = _make_init_two_level(n_try, a_val, rng)
            inits.append(h_init)
        # Shifted versions of current best
        for shift in [n_try//8, n_try//4, n_try//3, n_try//2, 3*n_try//4]:
            h_init = _project(np.roll(best_h, shift), n_try / 2.0)
            inits.append(h_init)
        # Reversed
        inits.append(_project(best_h[::-1], n_try / 2.0))
        # Complemented
        if abs((1.0 - best_h).sum() - n_try/2.0) < 0.1:
            inits.append(_project(1.0 - best_h, n_try / 2.0))

        # Evaluate and pick best few
        scored = []
        for hd in inits:
            c5v = _eval_c5(hd)
            scored.append((c5v, hd))
        scored.sort(key=lambda x: x[0])

        # Optimize top candidates
        for c5v, hd in scored[:5]:
            if time.time() > deadline - 40:
                break
            if c5v >= best_c5:
                break
            h_opt, c5_opt = _slsqp_optimize(hd, n_try, deadline, tau_ratio=0.6, n_tau=10)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = n_try

    # Phase 5: Gradient ascent on min-AC reformulation
    if time.time() < deadline - 50:
        h_opt, c5_opt = _gradient_ascent_min_ac(best_h, best_n, deadline, max_iter=800)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = best_n

    # Phase 6: Pair moves for fine-tuning
    if time.time() < deadline - 20:
        h_opt, c5_opt = _pair_moves(best_h, best_n, n_iter=300, deadline=deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = best_n
        # Another round of pair moves
        if time.time() < deadline - 10:
            h_opt, c5_opt = _pair_moves(best_h, best_n, n_iter=200, deadline=deadline)
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt
                best_n = best_n

    # Phase 7: Final SLSQP polish at current best resolution
    if time.time() < deadline - 15:
        h_opt, c5_opt = _slsqp_optimize(best_h, best_n, deadline, tau_ratio=0.55, n_tau=8)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
            best_n = best_n

    # Final feasibility check
    n_final = len(best_h)
    target = n_final / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-6) & (best_h < 1.0 - 1e-6)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] += diff / cnt
    np.clip(best_h, 0.0, 1.0, out=best_h)
    # Re-fix sum after clip
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        cnt = int(mask.sum())
        if cnt > 0:
            best_h[mask] += diff / cnt
    best_c5 = _eval_c5(best_h)

    return (best_h, best_c5, n_final)
