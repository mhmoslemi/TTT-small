```python
import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_T = 0.0
_G_TAU = 1.0
_G_K = 20

def _eq_fun(h):
    return float(np.sum(h) - _G_T)

def _eq_jac(h):
    return np.ones(_G_N)

def _get_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def _obj_active(h):
    n = len(h)
    c = _get_c(h)
    k = min(_G_K, len(c))
    idx = np.argpartition(c, len(c) - k)[-k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip((c_top - m) / _G_TAU, -100.0, 100.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum() / k))

def _grad_active(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    k = min(_G_K, len(c))
    idx = np.argpartition(c, len(c) - k)[-k:]
    c_top = c[idx]
    m = c_top.max()
    s = np.clip((c_top - m) / _G_TAU, -100.0, 100.0)
    e = np.exp(s)
    alpha = e / e.sum()
    
    grad = np.zeros(n)
    for ii, m_idx in enumerate(idx):
        a = alpha[ii]
        d = int(m_idx) - (n - 1)
        if d == 0:
            grad[:] += a * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += a * (1.0 - h[:n-d])
            grad[:n-d] -= a * h[d:]
        else:
            s_ = -d
            grad[:n-s_] += a * (1.0 - h[s_:])
            grad[s_:] -= a * h[:n-s_]
    return grad * dx

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = mask.sum()
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full')
    return float(np.max(c) * dx)

def interp_h(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)

def smooth_optimize_active(h0, n, deadline):
    global _G_T, _G_N, _G_TAU, _G_K
    _G_T = n / 2.0
    _G_N = n
    
    target = n / 2.0
    c5_init = evaluate_c5(h0)
    best_h = h0.copy()
    best_c5 = c5_init
    
    dx = 2.0 / n
    c_vals = _get_c(h0)
    m = c_vals.max()
    sorted_c = np.sort(c_vals)[::-1]
    spread = max(1e-6, m - sorted_c[min(19, len(sorted_c)-1)])
    tau0 = max(0.008, 0.35 * spread)
    
    K_schedule = [6, 10, 15, 20, 30, len(c_vals)]
    constraints = [{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}]
    bounds = [(0.0, 1.0)] * n
    
    h_curr = h0.copy()
    
    for K in K_schedule:
        if time.time() > deadline - 3:
            break
        _G_K = min(K, len(c_vals))
        n_tau = 10
        tau = tau0
        for _ in range(n_tau):
            if time.time() > deadline - 3:
                break
            _G_TAU = tau
            maxiter = 80 if tau > 0.003 else 120
            try:
                res = minimize(_obj_active, h_curr, method='SLSQP', jac=_grad_active,
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': maxiter, 'ftol': 1e-14})
                h_curr = project_feasible(res.x, target)
            except Exception:
                pass
            c5e = evaluate_c5(h_curr)
            if c5e < best_c5 - 1e-12:
                best_c5 = c5e
                best_h = h_curr.copy()
            tau *= 0.5
    
    return best_h, best_c5

def pair_search(h0, n, deadline, max_evals=3000):
    target = n / 2.0
    h = h0.copy()
    c5 = evaluate_c5(h)
    best_h = h0.copy()
    best_c5 = c5
    n_evals = 0
    
    # Get top lags to identify important coordinates
    c_vals = np.correlate(h, 1.0 - h, 'full') * (2.0/n)
    cmax = c_vals.max()
    
    # Identify important coordinates via gradient magnitude
    dx = 2.0 / n
    grad_mag = np.zeros(n)
    top_idx = np.argsort(c_vals)[-15:]
    for m_idx in top_idx:
        d = int(m_idx) - (n - 1)
        if d == 0:
            grad_mag[:] += np.abs(1.0 - 2.0 * h)
        elif d > 0:
            grad_mag[d:] += np.abs(1.0 - h[:n-d])
            grad_mag[:n-d] += np.abs(h[d:])
        else:
            s_ = -d
            grad_mag[:n-s_] += np.abs(1.0 - h[s_:])
            grad_mag[s_:] += np.abs(h[:n-s_])
    
    # Pick coordinates with high gradient magnitude and not at boundary
    mask = (h > 0.02) & (h < 0.98)
    scores = grad_mag * mask
    top_coords = np.argsort(-scores)[:50]
    top_coords = top_coords[h[top_coords] > 0.02]
    top_coords = top_coords[h[top_coords] < 0.98]
    top_coords = top_coords[:50]
    
    deltas = [0.02, 0.04, 0.06, 0.1, 0.01, 0.15]
    
    for i_idx in range(len(top_coords)):
        if time.time() > deadline - 1:
            break
        i = top_coords[i_idx]
        for j_idx in range(len(top_coords)):
            if i_idx == j_idx:
                continue
            if time.time() > deadline - 1:
                break
            if n_evals >= max_evals:
                break
            j = top_coords[j_idx]
            for d in deltas:
                if n_evals >= max_evals:
                    break
                # h[i] += d, h[j] -= d
                if h[i] + d <= 1.0 and h[j] - d >= 0.0:
                    h_try = h.copy()
                    h_try[i] += d
                    h_try[j] -= d
                    c5_try = evaluate_c5(h_try)
                    n_evals += 1
                    if c5_try < best_c5 - 1e-12:
                        best_c5 = c5_try
                        best_h = h_try.copy()
                        h = h_try.copy()
                        break
    
    return best_h, best_c5

def subgradient_optimize(h0, n, deadline):
    """Projected subgradient method with backtracking"""
    target = n / 2.0
    h = h0.copy()
    c5 = evaluate_c5(h)
    best_h = h0.copy()
    best_c5 = c5
    
    # Initial step size
    dx = 2.0 / n
    c_vals = _get_c(h)
    spread = max(1e-6, c_vals.max() - c_vals.mean())
    lr = 0.1 * spread / (2.0 / n)  # scale appropriately
    
    n_iter = 0
    stuck = 0
    
    while time.time() < deadline - 2:
        n_iter += 1
        # Compute subgradient (softmax-weighted gradient of top lags)
        c = _get_c(h)
        k = min(20, len(c))
        idx = np.argpartition(c, len(c) - k)[-k:]
        c_top = c[idx]
        m = c_top.max()
        tau = max(0.005, 0.1 * (m - c_top.min()) + 0.002)
        s = np.clip((c_top - m) / tau, -100.0, 100.0)
        e = np.exp(s)
        alpha = e / e.sum()
        
        grad = np.zeros(n)
        for ii, m_idx in enumerate(idx):
            a = alpha[ii]
            d = int(m_idx) - (n - 1)
            if d == 0:
                grad[:] += a * (1.0 - 2.0 * h)
            elif d > 0:
                grad[d:] += a * (1.0 - h[:n-d])
                grad[:n-d] -= a * h[d:]
            else:
                s_ = -d
                grad[:n-s_] += a * (1.0 - h[s_:])
                grad[s_:] -= a * h[:n-s_]
        grad *= dx
        
        # Project gradient onto sum=0 subspace
        grad -= grad.mean()
        
        # Line search
        found = False
        for factor in [1.0, 0.5, 0.25, 0.125, 0.0625]:
            step = lr * factor
            h_try = h - step * grad
            h_try = project_feasible(h_try, target)
            c5_try = evaluate_c5(h_try)
            if c5_try < c5 - 1e-14:
                h = h_try
                c5 = c5_try
                found = True
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                stuck = 0
                break
        if not found:
            stuck += 1
            lr *= 0.7
            if stuck > 50:
                break
    
    return best_h, best_c5

    return 1.0 - h

def make_reversed(h):
    return h[::-1].copy()

def make_sorted_bits(h, seed=0):

    low = v[:n//2]
```
