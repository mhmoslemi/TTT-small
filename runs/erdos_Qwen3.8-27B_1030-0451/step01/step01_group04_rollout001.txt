import numpy as np
import time


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            adj = min(abs(residual), 1.0 - h[idx[0]]) if residual > 0 else max(abs(residual), h[idx[0]])
            h[idx[0]] += residual if residual > 0 else -h[idx[0]] if h[idx[0]] > abs(residual) else residual
    return h


def compute_gradient_smooth(A, h, w, N, dx):
    """Compute gradient of weighted sum sum_r w[r]*c_r with respect to h."""
    Nc = 2 * N - 1
    grad = np.zeros(N)
    for r in range(Nc):
        wr = w[r] * dx
        if wr < 1e-12:
            continue
        m_lag = r - (N - 1)
        # T1: (1-h[i-m_lag]) for valid i where 0<=i-m_lag<N
        i_lo1 = max(0, m_lag)
        i_hi1 = min(N, N + m_lag)
        if i_hi1 > i_lo1:
            idx1 = np.arange(i_lo1, i_hi1)
            n_idx = idx1 - m_lag
            grad[idx1] += wr * (1.0 - h[n_idx])
        # T2: -h[i+m_lag] for valid i where 0<=i+m_lag<N
        i_lo2 = max(0, -m_lag)
        i_hi2 = min(N, N - m_lag)
        if i_hi2 > i_lo2:
            idx2 = np.arange(i_lo2, i_hi2)
            n_idx = idx2 + m_lag
            grad[idx2] -= wr * h[n_idx]
    return grad


def exact_line_search(h, a, A, dx, n_grid=300):
    """Minimize F(delta)=max_r(A_r + delta*L_r + delta^2*Q_r) over feasible [dmin,dmax]."""
    N = len(h)
    b = 1.0 - h

    L = dx * (np.correlate(a, b, 'full') - np.correlate(h, a, 'full'))
    Q = -dx * np.correlate(a, a, 'full')

    pos_mask = a > 1e-14
    neg_mask = a < -1e-14

    dmin = 0.0
    dmax = 0.0
    if np.any(pos_mask):
        dmin = max(dmin, float(np.min(-h[pos_mask] / a[pos_mask])))
    if np.any(neg_mask):
        dmax = min(dmax, float(np.min((1.0 - h[neg_mask]) / a[neg_mask])))

    margin = 1e-11
    dmin += margin
    dmax -= margin

    if dmax - dmin < 1e-13:
        return 0.0

    # Coarse grid
    deltas = np.linspace(dmin, dmax, n_grid)
    F_vals = A[None, :] + deltas[:, None] * L[None, :] + (deltas[None, :] ** 2) * Q[None, :]
    F_max = np.max(F_vals, axis=1)

    best_idx = int(np.argmin(F_max))
    best_delta = deltas[best_idx]

    # Refine
    if 0 < best_idx < n_grid - 1:
        lo_d = deltas[best_idx - 1]
        hi_d = deltas[best_idx + 1]
        for _ in range(3):
            deltas2 = np.linspace(lo_d, hi_d, 50)
            F2 = A[None, :] + deltas2[:, None] * L[None, :] + (deltas2[None, :] ** 2) * Q[None, :]
            F2_max = np.max(F2, axis=1)
            bi = int(np.argmin(F2_max))
            if bi == 0:
                hi_d = deltas2[bi + 1]
            elif bi == len(deltas2) - 1:
                lo_d = deltas2[bi - 1]
            else:
                lo_d = deltas2[bi - 1]
                hi_d = deltas2[bi + 1]
        best_delta = (lo_d + hi_d) * 0.5

    # Check endpoints
    F_left = float(np.max(A + dmin * L + dmin ** 2 * Q))
    F_right = float(np.max(A + dmax * L + dmax ** 2 * Q))
    F_best = float(np.max(A + best_delta * L + best_delta ** 2 * Q))
    if F_left < F_best:
        best_delta = dmin
    if F_right < F_best:
        best_delta = dmax

    return best_delta


def try_binarize(h, N, target_sum):
    """Try rounding to {0,1} and repairing."""
    h_bin = (h > 0.5).astype(float)
    h_bin = project_feasible(h_bin, target_sum)
    return h_bin


def upscale_func(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)
