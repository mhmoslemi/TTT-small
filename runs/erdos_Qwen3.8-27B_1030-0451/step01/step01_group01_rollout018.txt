import numpy as np

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
        else:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r / len(idx)
    return np.clip(y, 0.0, 1.0)


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))


def grad_corr_lag(h, lag):
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:
        if lag < n:
            idx1 = np.arange(0, n - lag)
            if len(idx1) > 0:
                g[idx1] += 1.0 - h[idx1 + lag]
            idx2 = np.arange(lag, n)
            if len(idx2) > 0:
                g[idx2] -= h[idx2 - lag]
    else:
        l = -lag
        if l < n:
            idx1 = np.arange(l, n)
            if len(idx1) > 0:
                g[idx1] += 1.0 - h[idx1 - l]
            idx2 = np.arange(0, n - l)
            if len(idx2) > 0:
                g[idx2] -= h[idx2 + l]
    return g


def eval_pair_move(h, i, j, t):
    orig_i = h[i]
    orig_j = h[j]
    s = orig_i + orig_j
    h[i] = t
    h[j] = s - t
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    c = float(np.max(res))
    h[i] = orig_i
    h[j] = orig_j
    return c


def pairwise_line_search(h, i, j):
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return h[i]
    current = h[i]
    best_t = current
    best_f = eval_pair_move(h, i, j, current)
    n_grid = 60
    grid = np.linspace(lo, hi, n_grid)
    for t in grid:
        f = eval_pair_move(h, i, j, t)

        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new <= c_cur + 1e-14:
            h = h_new
            step = min(step * 1.02, 0.08)
        else:
            step *= 0.7
            if step < 1e-7:
                break
    return best_h, best_c


def transfer_resolution(h0, n_new):
    n_old = len(h0)
    x_old = np.arange(n_old) / float(n_old)
    x_new = np.arange(n_new) / float(n_new)
    h_ext = np.concatenate([h0, h0[:1]])
    h_new = np.interp(x_new, x_old, h_ext, period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new


def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    safety = 4.0
    deadline = start_time + budget_s - safety

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n = len(best_h)

    def time_left():
        return deadline - time.time()

    # Phase 1: LP Trust Region
    if time_left() > 30:
        dl1 = min(deadline, start_time + budget_s * 0.20)
        try:
            h1, c1 = lp_trust_region(best_h, best_c, dl1, max_iters=3000)
            if c1 < best_c:
                best_h = h1
                best_c = c1
        except Exception:
            pass

    # Phase 2: Pairwise Coordinate Descent (main workhorse)
    if time_left() > 60:
        dl2 = min(deadline, start_time + budget_s * 0.65)
        try:
            h2, c2 = pairwise_cd(best_h, best_c, dl2, max_sweeps=2000)
            if c2 < best_c:
                best_h = h2
                best_c = c2
        except Exception:
            pass

    # Phase 3: Another LP round from improved point
    if time_left() > 40:
        dl3 = min(deadline, start_time + budget_s * 0.75)
        try:
            h3, c3 = lp_trust_region(best_h, best_c, dl3, max_iters=1500)
            if c3 < best_c:
                best_h = h3
                best_c = c3
        except Exception:
            pass

    # Phase 4: More pairwise CD
    if time_left() > 30:
        dl4 = min(deadline, start_time + budget_s * 0.88)
        try:
            h4, c4 = pairwise_cd(best_h, best_c, dl4, max_sweeps=2000)
            if c4 < best_c:
                best_h = h4
                best_c = c4
        except Exception:
            pass

    # Phase 5: Subgradient polish
    if time_left() > 20:
        dl5 = min(deadline, start_time + budget_s * 0.94)
        try:
            h5, c5 = subgradient_polish(best_h, best_c, dl5, max_iters=20000)
            if c5 < best_c:
                best_h = h5
                best_c = c5
        except Exception:
            pass

    # Phase 6: Final pairwise polish
    if time_left() > 10:
        try:
            h6, c6 = pairwise_cd(best_h, best_c, deadline, max_sweeps=500)
            if c6 < best_c:
                best_h = h6
                best_c = c6
        except Exception:
            pass

    # Optional: try resolution upgrade if time permits
    if time_left() > 80:
        n_cur = len(best_h)
        for n_target in [n_cur + 12, n_cur + 24, n_cur + 36]:
            if time_left() < 50:
                break
            try:
                h_tr = transfer_resolution(best_h, n_target)
                c_tr = evaluate_c5(h_tr)
                sub_h = h_tr
                sub_c = c_tr
                if sub_c < best_c:
                    best_h = sub_h.copy()
                    best_c = sub_c
                dl_sub = min(deadline, time.time() + min(40, time_left() * 0.4))
                h_sub, c_sub = pairwise_cd(sub_h, sub_c, dl_sub, max_sweeps=300)
                if c_sub < best_c:
                    best_h = h_sub
                    best_c = c_sub
            except Exception:
                pass

    # Final verification and projection
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
