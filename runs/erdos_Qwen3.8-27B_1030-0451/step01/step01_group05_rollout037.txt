import numpy as np
from scipy.optimize import minimize
import time


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)


def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h


def compute_line_coeffs(h, d):
    """Compute linear and quadratic coefficients for corr(h+t*d, 1-h-t*d)."""
    n = len(h)
    L_full = 2 * n - 1
    one_h = 1.0 - h
    Lin = np.zeros(L_full)
    Quad = np.zeros(L_full)
    for k in range(L_full):
        m_lo = max(0, k)
        m_hi = min(n - 1, n - 1 + k)
        if m_lo > m_hi:
            continue
        m = np.arange(m_lo, m_hi + 1)
        j = m - k
        Lin[k] = np.sum(d[m] * one_h[j] - h[m] * d[j])
        Quad[k] = -np.sum(d[m] * d[j])
    return Lin, Quad


def smooth_gradient(h, tau, n_active=8):
    """Compute smoothed gradient biased toward active (max) lags."""
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    c = R * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -400, 400)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw

    g = np.zeros(n)
    for k in range(2 * n - 1):
        w = weights[k]
        if w < 1e-14:
            continue
        m_lo = max(0, k)
        m_hi = min(n - 1, n - 1 + k)
        if m_lo > m_hi:
            continue
        m = np.arange(m_lo, m_hi + 1)
        j = m - k
        g[m] += w * (1.0 - h[j])
        g[j] += w * (-h[m])

    g -= np.mean(g)
    return g


def zero_sum_normalize(d):
    d = d - np.mean(d)
    norm = np.linalg.norm(d)
    if norm < 1e-15:
        return np.zeros_like(d)
    return d / norm


def fw_direction(h, g):
    """Frank-Wolfe: direction toward best vertex in feasible polytope."""
    n = len(h)
    S = n / 2.0
    order = np.argsort(g)
    u = np.zeros(n)
    S_int = int(np.floor(S))
    S_frac = S - S_int
    for i in range(S_int):
        u[order[i]] = 1.0
    if S_frac > 1e-12:
        u[order[S_int]] = S_frac
    d = u - h
    return zero_sum_normalize(d)


def line_search(h, d, n_grid=60):
    """Exact quadratic line search from h along direction d (zero-sum)."""
    n = len(h)
    dx = 2.0 / n
    cur_score = exact_score(h)
    R = np.correlate(h, 1.0 - h, mode='full')

    # Feasible interval
    d_abs = np.abs(d)
    mask = d_abs > 1e-14
    if not np.any(mask):
        return None

    idx = np.where(mask)[0]
    t_lo = np.inf
    t_hi = -np.inf
    for i in idx:
        if d[i] > 0:
            t_lo = max(t_lo, -h[i] / d[i])
            t_hi = min(t_hi, (1.0 - h[i]) / d[i])
        else:
            t_lo = max(t_lo, (1.0 - h[i]) / d[i])
            t_hi = min(t_hi, -h[i] / d[i])

    if t_hi <= t_lo + 1e-14:
        return None

    margin = 1e-12
    t_lo += margin
    t_hi -= margin
    if t_hi <= t_lo:
        return None

    # Compute line coefficients
    Lin, Quad = compute_line_coeffs(h, d)

    # Candidate t values
    candidates = [t_lo, t_hi, 0.5 * (t_lo + t_hi)]

    # Critical points of individual quadratics
    for K in range(len(Lin)):
        a = Quad[K]
        b = Lin[K]
        if abs(a) > 1e-14:
            t_star = -b / (2.0 * a)
            if t_lo < t_star < t_hi:
                candidates.append(t_star)

    # Grid
    grid = np.linspace(t_lo, t_hi, n_grid)
    candidates.extend(grid.tolist())

    best_t = 0.0
    best_s = cur_score
    seen = set()

    for t in candidates:
        t = float(np.clip(t, t_lo, t_hi))
        key = round(t, 10)
        if key in seen:
            continue
        seen.add(key)
        vals = R + t * Lin + t * t * Quad
        s = float(np.max(vals) * dx)
        if s < best_s - 1e-13:
            best_s = s
            best_t = t

    if best_t == 0.0:
        return None

    h_new = h + best_t * d
    actual = exact_score(h_new)
    if actual < cur_score - 1e-12:
        return best_t, actual, h_new
    return None


def make_block_direction(h, n, rng):
    """Create a direction that redistributes mass between blocks."""
    n_blocks = int(rng.integers(2, max(3, n // 8)))
    block_size = n // n_blocks
    d = np.zeros(n)
    for b in range(n_blocks):
        start = b * block_size
        end = start + block_size if b < n_blocks - 1 else n
        d[start:end] = rng.standard_normal(end - start)
    return zero_sum_normalize(d)


def make_sinusoidal_direction(h, n, rng):
    """Create a sinusoidal direction."""
    freq = int(rng.integers(1, max(2, n // 5)))
    phase = rng.random() * 2.0 * np.pi
    x = np.arange(n)
    d = np.sin(2.0 * np.pi * freq * x / n + phase)
    return zero_sum_normalize(d)


def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)


def optimize_from(h0, n, tau_init, deadline, rng, max_rounds=200):
    """Main optimization loop using exact line search."""
    h_cur = h0.copy()
    s_cur = exact_score(h_cur)
    best_local = h_cur.copy()
    best_local_s = s_cur
    tau = tau_init
    no_improve = 0
    elite = [h_cur.copy()]

    for rnd in range(max_rounds):
        if time.time() > deadline:
            break

        n_dirs = 20 + 10 * (rnd % 5)
        improved = False

        for didx in range(n_dirs):
            if time.time() > deadline:
                break

            dtype = didx % 6

            if dtype == 0:
                g = smooth_gradient(h_cur, tau)
                d = zero_sum_normalize(g)
            elif dtype == 1:
                g = smooth_gradient(h_cur, tau)
                d = fw_direction(h_cur, g)
            elif dtype == 2:
                d = zero_sum_normalize(rng.standard_normal(n) * 0.5)
            elif dtype == 3:
                d = make_block_direction(h_cur, n, rng)
            elif dtype == 4:
                d = make_sinusoidal_direction(h_cur, n, rng)
            else:
                if len(elite) > 1:
                    j = int(rng.integers(0, len(elite)))
                    d = zero_sum_normalize(elite[j] - h_cur)
                else:
                    d = zero_sum_normalize(rng.standard_normal(n))

            if np.linalg.norm(d) < 1e-14:
                continue

            result = line_search(h_cur, d, n_grid=50)
            if result is not None:
                _, s_new, h_new = result
                h_cur = h_new
                s_cur = s_new
                improved = True

        if improved:
            no_improve = 0
            h_cur = project_feasible(h_cur, n)
            s_cur = exact_score(h_cur)
            if s_cur < best_local_s - 1e-13:
                best_local = h_cur.copy()
                best_local_s = s_cur
                if len(elite) < 8:
                    elite.append(best_local.copy())
                elif best_local_s < exact_score(elite[-1]):
                    elite[-1] = best_local.copy()
        else:
            no_improve += 1
            tau *= 0.8
            if tau < 0.002:
                tau = 0.002

            if no_improve > 30:
                # Restart from best with perturbation
                h_cur = best_local + 0.01 * rng.standard_normal(n)
                h_cur = project_feasible(h_cur, n)
                s_cur = exact_score(h_cur)
                no_improve = 15

            if no_improve > 80:
                break

    return best_local, best_local_s
