    # This is the direction that most decreases the sum (or max) of active correlations

    dx = 2.0 / n

    c = np.correlate(h, v, 'full') * dx

    n = len(h)
    d = k - (n - 1)

The key challenge is that the objective is a max of bilinear functions. At the optimum, typically 2-5 lags are "active" (at the max value), and the solution balances these. The difficulty is that moving to reduce one active lag might increase another.

A powerful technique for this is the "alternating minimization" or "round-robin" approach:
1. Pick the worst lag k*
2. Compute gradient of c_{k*}
3. Take a step in the -gradient direction (projected to tangent space)
4. This reduces c_{k*} but might increase other lags
5. Repeat, cycling through the active lags

This is different from the "average all active gradients" approach because it focuses on one lag at a time, which can be more effective when the active set is small.

Let me also think about a "dual" approach. The problem can be written as:
min max_k c_k(h)
= min min_t { t : c_k(h) ≤ t ∀k }

The dual of the inner constraint (at fixed t) is:
max_{λ_k ≥ 0, sum λ_k = 1} min_h { t + sum λ_k (c_k(h) - t) }
= max_λ min_h { sum λ_k c_k(h) }  (since sum λ_k = 1, the t terms cancel)

So the dual is:
max_{λ ≥ 0, sum λ = 1} min_{h feasible} sum_k λ_k c_k(h)

This is a max-min problem. By Sion's minimax theorem (or just the structure), at the optimum the primal and dual objectives are equal (strong duality holds for this bilinear problem? Actually, bilinear problems don't generally have strong duality, but at a local optimum of a particular structure, the duality gap might be zero).

In practice, we can use this to guide the optimization:
- Maintain a distribution λ over lags
- The subproblem min_h sum λ_k c_k(h) is a quadratic minimization (since each c_k is bilinear, the weighted sum is quadratic)
- Update λ based on which lags are most violated

This is essentially the "multiplicative weights" or "follow-the-regularized-leader" approach for minimax problems.

            d = tangent_direction(g)
            if d is None:
                continue
            t, c5_new = line_search_c5(h_curr, d)
            if c5_new < best_c5 - 1e-10:
                h_curr = project_feasible(h_curr + t * d)
                best_c5 = c5_new
                best_h = h_curr.copy()
                stall = 0
                improved = True
                break

        if not improved:
            # Random weighted direction
            w_rand = rng.dirichlet(np.ones(len(active)))
            g = grad_weighted_lags(active, w_rand, h_curr, dx)
            d = tangent_direction(g)
            if d is not None:
                t, c5_new = line_search_c5(h_curr, d)
                if c5_new < best_c5 - 1e-10:
                    h_curr = project_feasible(h_curr + t * d)
                    best_c5 = c5_new
                    best_h = h_curr.copy()
                    stall = 0
                    improved = True

        if not improved:
            stall += 1
            if stall >= 12:
                h_curr = project_feasible(h_curr + rng.standard_normal(n) * 0.012)
                stall = 0

    return best_h, best_c5

def interp_to_resolution(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    xo = np.linspace(0.0, 2.0, n_old, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(xn, xo, h_old)
    return project_feasible(h_new, n_new / 2.0)

def quadratic_subproblem(h0, lam, c_indices, n, maxiter=100):
    """Solve min_h sum lam_k * c_k(h) via SLSQP on the quadratic."""
    dx = 2.0 / n
    target = n / 2.0

    def obj_quad(h):
        val = 0.0
        for i, k in enumerate(c_indices):
            d = k - (n - 1)
            if d >= 0:
                m = n - 1 - d
                if m >= 0:
                    val += lam[i] * dx * np.sum(h[0:m + 1] * (1.0 - h[d:m + 1 + d]))
            else:
                a = -d
                m = n - 1 - a
                if m >= 0:
                    val += lam[i] * dx * np.sum(h[a:m + 1 + a] * (1.0 - h[0:m + 1]))
        return val

    def grad_quad(h):
        return grad_weighted_lags(c_indices, lam, h, dx)

    constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - target, 'jac': lambda h: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj_quad, h0, method='SLSQP', jac=grad_quad,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-12})
        return project_feasible(res.x, target)
    except Exception:
        return project_feasible(h0, target)

def dual_guided_optimize(h, max_rounds=50, deadline=None):
    """Use dual weights to guide quadratic subproblem solves."""
    n = len(h)
    dx = 2.0 / n
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    h_curr = h.copy()
    n_c = 2 * n - 1
    lam = np.ones(n_c) / n_c

    for rnd in range(max_rounds):
        if deadline is not None and time.time() > deadline - 5:
            break
        # Update dual: multiplicative weights on violated constraints
        c = compute_correlations(h_curr)
        cmax = c.max()
        spread = max(1e-6, cmax - c.min())
        lam = lam * np.exp(0.5 * (c - c.mean()) / spread)
        lam /= lam.sum()

        # Solve quadratic subproblem with top-weighted lags
        top_k = 8
        top_idx = np.argsort(-lam)[:top_k]
        lam_top = lam[top_idx]
        lam_top /= lam_top.sum()
        h_new = quadratic_subproblem(h_curr, lam_top, top_idx, n, maxiter=50)

        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-10:
            best_c5 = c5_new
            best_h = h_new.copy()
            h_curr = h_new
        else:
            # Try from current with line search
            g = grad_weighted_lags(top_idx, lam_top, h_curr, dx)
            d = tangent_direction(g)
            if d is not None:
                t, c5_ls = line_search_c5(h_curr, d)
                if c5_ls < best_c5 - 1e-10:
                    h_curr = project_feasible(h_curr + t * d)
                    best_c5 = c5_ls
                    best_h = h_curr.copy()

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)

    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)

    best_h = parent.copy()
    best_c5 = evaluate_c5(parent)
    best_n = n_par

    # Phase 1: Active-lag line search at parent resolution
    h1, c5_1 = active_lag_line_search(parent, max_iters=200, deadline=deadline)
    if c5_1 < best_c5:
        best_c5 = c5_1
        best_h = h1
        best_n = len(h1)

    # Phase 2: SORO with multiple weightings
    if time.time() < deadline - 60:
        h2, c5_2 = soro_optimize(best_h, max_iters=200, deadline=deadline)
        if c5_2 < best_c5:
            best_c5 = c5_2
            best_h = h2
            best_n = len(h2)

    # Phase 3: Dual-guided quadratic subproblem
    if time.time() < deadline - 80:
        h3, c5_3 = dual_guided_optimize(best_h, max_rounds=30, deadline=deadline)
        if c5_3 < best_c5:
            best_c5 = c5_3
            best_h = h3
            best_n = len(h3)

    # Phase 4: More active-lag refinement
    if time.time() < deadline - 60:
        h4, c5_4 = active_lag_line_search(best_h, max_iters=300, deadline=deadline)
        if c5_4 < best_c5:
            best_c5 = c5_4
            best_h = h4
            best_n = len(h4)

    # Phase 5: Try finer resolution
    if time.time() < deadline - 120:
        for mult in [2, 3, 4]:
            if time.time() > deadline - 60:
                break
            n_new = n_par * mult
            if n_new > 2400:
                break
            h_interp = interp_to_resolution(best_h, n_new)
            h_fine, c5_fine = active_lag_line_search(h_interp, max_iters=150, deadline=deadline)
            if c5_fine < best_c5:
                best_c5 = c5_fine
                best_h = h_fine
                best_n = n_new

    # Phase 6: Final polish with SLSQP smooth objective
    if time.time() < deadline - 30:
        n = best_n
        dx = 2.0 / n
        target = n / 2.0

        def smooth_obj(h):
            c = np.correlate(h, 1.0 - h, 'full') * dx
            m = c.max()
            tau = 0.03
            s = np.clip((c - m) / tau, -500, 500)
            return m + tau * np.log(np.exp(s).sum())

        def smooth_jac(h):
            c = np.correlate(h, 1.0 - h, 'full') * dx
            m = c.max()
            tau = 0.03
            s = np.clip((c - m) / tau, -500, 500)
            e = np.exp(s)
            alpha = e / e.sum()
            g = np.zeros(n)
            for k in range(len(alpha)):
                if alpha[k] < 1e-12:
                    continue
                g += alpha[k] * grad_single_lag(k, h, dx)
            return g

        constraints = [{'type': 'eq', 'fun': lambda h: h.sum() - target, 'jac': lambda h: np.ones(n)}]
        bounds = [(0.0, 1.0)] * n
        try:
            res = minimize(smooth_obj, best_h, method='SLSQP', jac=smooth_jac,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': 200, 'ftol': 1e-13})
            h_slq = project_feasible(res.x, target)
            c5_slq = evaluate_c5(h_slq)
            if c5_slq < best_c5:
                best_c5 = c5_slq
                best_h = h_slq
                best_n = n
        except Exception:
            pass

    # Final feasibility enforcement
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
