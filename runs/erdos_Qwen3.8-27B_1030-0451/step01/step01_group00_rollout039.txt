def weighted_gradient(h, w):

    # w length 2n-1, k=0 corresponds lag=-(n-1)

def project_h(p, n):
    p = np.asarray(p, dtype=float)
    if n <= 0: return p
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        q = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(q))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    q = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(q))
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(-(1.0 - q))
            rem = diff
            for t in range(n):
                room = 1.0 - q[idx[t]]
                if room <= 0: continue
                add = room if room < rem else rem
                q[idx[t]] += add
                rem -= add
                if rem <= 1e-12: break
        else:
            idx = np.argsort(q)
            rem = -diff
            for t in range(n):
                take = q[idx[t]] if q[idx[t]] < rem else rem
                q[idx[t]] -= take
                rem -= take
                if rem <= 1e-12: break
    return q
```

import numpy as np
import time
from scipy.optimize import linprog


def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")


def compute_c5(h):
    n = len(h)
    corr = compute_corr(h)
    return float(np.max(corr) * (2.0 / n))


def project_h(p, n):
    p = np.asarray(p, dtype=float).copy()
    if p.ndim != 1:
        p = p.reshape(-1)
    if len(p) < n:
        p = np.pad(p, (0, n - len(p)), mode="constant", constant_values=0.5)
    elif len(p) > n:
        p = p[:n]

    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0

    for _ in range(72):
        mid = 0.5 * (lo + hi)
        q = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(q))
        if s > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    q = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(q))

    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.argsort(-(1.0 - q))
            rem = diff
            for t in range(n):
                room = 1.0 - q[idx[t]]
                if room <= 0.0:
                    continue
                add = room if room < rem else rem
                q[idx[t]] += add
                rem -= add
                if rem <= 1e-12:
                    break
        else:
            idx = np.argsort(q)
            rem = -diff
            for t in range(n):
                take = q[idx[t]] if q[idx[t]] < rem else rem
                q[idx[t]] -= take
                rem -= take
                if rem <= 1e-12:
                    break

    return np.clip(q, 0.0, 1.0)


def make_start(n, seed):
    rng = np.random.RandomState(seed)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)

    freqs = [1.0, 2.0, 3.0, 5.0, 7.0, 11.0]
    amps = [0.18, 0.11, 0.06, 0.035, 0.02, 0.012]
    phases = rng.uniform(0.0, 2.0 * np.pi, len(freqs))

    for a, f, ph in zip(amps, freqs, phases):
        h += a * np.sin(2.0 * np.pi * f * x + ph)

    h += 0.08 * np.sin(np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
    h += 0.05 * np.cos(np.pi * x + rng.uniform(0.0, 2.0 * np.pi))
    h += rng.randn(n) * 0.01
    return project_h(h, n)


def resample_h(h, n_new):
    n = len(h)
    if n_new <= 0:
        return h.copy()
    if n_new == n:
        return project_h(h, n)

    x_old = (np.arange(n) + 0.5) * (2.0 / n)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h)
    return project_h(y, n_new)


def soft_weights(corr, tau):
    m = len(corr)
    if tau <= 1e-8:
        return np.full(m, 1.0 / m)

    maxc = float(np.max(corr))
    z = tau * (corr - maxc)
    w = np.exp(z)
    s = float(np.sum(w))

    if s <= 1e-300:
        w = np.zeros(m)
        mask = corr >= (maxc - 1e-9)
        w[mask] = 1.0
        s = float(np.sum(w))
        if s <= 0.0:
            w[int(np.argmax(corr))] = 1.0
            s = 1.0

    return w / s


def weighted_gradient(h, w):
    n = len(h)
    off = n - 1
    grad = np.zeros(n)

    for k in range(len(w)):
        wk = float(w[k])
        if abs(wk) < 1e-15:
            continue
        lag = k - off
        if lag >= 0:
            if lag == 0:
                grad += wk * (1.0 - 2.0 * h)
            else:
                grad[lag:] += wk * (1.0 - h[:n - lag])
                grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]

    return grad


def smooth_descent(h0, time_limit, seed=0):
    n = len(h0)
    h = project_h(h0, n).copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    if time_limit <= 1.0:
        return best_h, best_c

    t0 = time.time()
    rng = np.random.RandomState(seed)

    pass_defs = [
        (80.0, 0.15),
        (300.0, 0.35),
        (1200.0, 0.50),
    ]

    for p, (tau_max, frac) in enumerate(pass_defs):
        elapsed = time.time() - t0
        if time_limit * frac - elapsed < 1.0:
            break

        pass_time = time_limit * frac - elapsed
        if pass_time < 1.0:
            pass_time = max(0.5, time.time() - t0)
        pt0 = time.time()
        alpha = 0.05
        stall = 0

        while time.time() - pt0 < pass_time and time.time() - t0 < time_limit - 0.5:
            prog = min(1.0, (time.time() - pt0) / max(pass_time, 1e-6))
            tau = 8.0 + (tau_max - 8.0) * prog

            corr = compute_corr(h)
            w = soft_weights(corr, tau)
            g = weighted_gradient(h, w)
            c = compute_c5(h)

            if c < best_c - 1e-13:
                best_c = c
                best_h = h.copy()

            norm_g = float(np.linalg.norm(g))
            if norm_g < 1e-12:
                h = project_h(h + rng.randn(n) * 0.01, n)
                c = compute_c5(h)
                continue

            d = -g / norm_g
            a = alpha
            old_c = c
            accepted = False
            progress = False

            for _ in range(14):
                cand = project_h(h + a * d, n)
                cand_c = compute_c5(cand)
                if cand_c <= c + 1e-12:
                    h = cand
                    c = cand_c
                    accepted = True
                    if c < old_c - 1e-12:
                        progress = True
                    break
                a *= 0.5
                if a < 1e-5:
                    break

            if accepted:
                stall = 0
                if progress:
                    alpha = min(0.15, alpha * 1.18)
                else:
                    alpha = max(0.005, alpha * 0.88)
            else:
                stall += 1
                alpha = max(0.002, alpha * 0.55)
                if stall >= 18:
                    scale = 0.01 + 0.03 * prog
                    h = project_h(h + rng.randn(n) * scale, n)
                    c = compute_c5(h)
                    alpha = 0.05
                    stall = 0

        if p < len(pass_defs) - 1 and time.time() - t0 < time_limit - 5:
            h = project_h(h + rng.randn(n) * 0.025, n)
            c = compute_c5(h)

    if c < best_c:
        best_h = h.copy()
        best_c = c

    return best_h, best_c


def select_lags(corr, max_all=560, top_count=140, tol_frac=0.004):
    m = len(corr)
    if m <= max_all:
        return list(range(m))

    maxc = float(np.max(corr))
    thresh = maxc - max(1e-6, tol_frac * maxc)
    order = np.argsort(-corr)
    idx = order[:top_count]
    active = np.where(corr >= thresh)[0]
    idx = np.unique(np.concatenate((idx, active)))

    if len(idx) > top_count + 70:
        idx = idx[:top_count + 70]

    return list(idx)


def gradient_rows(h, indices):
    n = len(h)
    off = n - 1
    G = np.zeros((len(indices), n))

    for r, k in enumerate(indices):
        lag = int(k) - off
        row = G[r]
        if lag >= 0:
            if lag == 0:
                row[:] = 1.0 - 2.0 * h
            else:
                row[lag:] += 1.0 - h[:n - lag]
                row[:n - lag] -= h[lag:]
        else:
            L = -lag
            row[:n - L] += 1.0 - h[L:]
            row[L:] -= h[:n - L]

    return G


def lp_active_step(h, R, lambda_pen=0.01):
    n = len(h)
    corr = compute_corr(h)
    idx = select_lags(corr)
    if len(idx) == 0:
        return None

    G = gradient_rows(h, idx)
    m = len(idx)

    A1 = np.hstack((G, -G, -np.ones((m, 1))))
    b1 = corr[idx].astype(float)

    Abox = np.zeros((3 * n, 2 * n + 1))
    Abox[0:n, 0:n] = 1.0
    Abox[0:n, n:2 * n] = -1.0
    b1box = 1.0 - h

    Abox[n:2 * n, 0:n] = -1.0
    Abox[n:2 * n, n:2 * n] = 1.0
    b2box = h

    Abox[2 * n:3 * n, 0:n] = 1.0
    Abox[2 * n:3 * n, n:2 * n] = 1.0
    b3box = np.full(n, R)

    A_ub = np.vstack((A1, Abox))
    b_ub = np.concatenate((b1, b1box, b2box, b3box))

    A_eq = np.zeros((1, 2 * n + 1))
    A_eq[0, 0:n] = 1.0
    A_eq[0, n:2 * n] = -1.0
    b_eq = np.array([0.0])

    cvec = np.empty(2 * n + 1)
    cvec[:2 * n] = lambda_pen
    cvec[-1] = 1.0

    bounds = [(0.0, None)] * (2 * n)
    bounds.append((0.0, 1.0e9))

    res = None
    try:
        res = linprog(
            cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
            bounds=bounds, method="highs", options={"time_limit": 2.0}
        )
    except Exception:
        res = None

    if res is None:
        try:
            res = linprog(
                cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                bounds=bounds, method="highs"
            )
        except Exception:
            try:
                res = linprog(cvec, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq, bounds=bounds)
            except Exception:
                res = None

    if res is None or int(getattr(res, "status", -1)) != 0:
        return None

    x = np.asarray(res.x, dtype=float)
    if x.size < 2 * n + 1:
        return None

    d = x[:n] - x[n:2 * n]
    if not np.all(np.isfinite(d)):
        return None
    if float(np.max(np.abs(d))) < 1e-12:
        return None

    return d

def coord_rows(h):
    n = len(h)
    off = n - 1
    M = 2 * n - 1
    rows = np.zeros((n, M))

    for i in range(n):
        Lmin = i - (n - 1)
        Lmax = n - 1 - i
        row = rows[i]

        for L in range(Lmin, Lmax + 1):
            v = 0.0

            j = i - L
            if 0 <= j < n:
                v += 1.0 - h[j]

            r = i + L
            if 0 <= r < n:
                v -= h[r]

            if v != 0.0:
                row[off + L] = v

    return rows


def pair_interval(h, i, j):
    hi = float(h[i])
    hj = float(h[j])
    smax = min(1.0 - hi, hj)
    smin = max(-hj, hi - 1.0)
    if smax - smin <= 1e-12:
        return None
    return float(smin), float(smax)


def pair_evaluate(corr, diff, idx0, idx1, idx2, s):
    c = corr + s * diff
    if s != 0.0:
        c[idx0] -= 2.0 * s * s
        c[idx1] += s * s
        if idx2 != idx1:
            c[idx2] += s * s
    return float(np.max(c))


def pair_grid_min(h, corr, rows, i, j, n, current_raw):
    iv = pair_interval(h, i, j)
    if iv is None:
        return 0.0, current_raw

    smin, smax = iv
    diff = rows[i] - rows[j]
    off = n - 1
    idx0 = off
    idx1 = off + i - j
    idx2 = off + j - i

    best_s = 0.0
    best_f = current_raw

    main = np.linspace(smin, smax, 31)
    for s in main:
        f = pair_evaluate(corr, diff, idx0, idx1, idx2, float(s))
        if f < best_f - 1e-13:
            best_f = f
            best_s = float(s)

    fine = np.linspace(-0.1, 0.1, 21)
    for s in fine:
        if smin <= s <= smax:
            f = pair_evaluate(corr, diff, idx0, idx1, idx2, float(s))
            if f < best_f - 1e-13:
                best_f = f
                best_s = float(s)

    span = smax - smin
    step = max(span / 24.0, 0.004)
    refine = np.linspace(best_s - step, best_s + step, 11)
    for s in refine:
        if smin <= s <= smax:
            f = pair_evaluate(corr, diff, idx0, idx1, idx2, float(s))
            if f < best_f - 1e-13:
                best_f = f
                best_s = float(s)

    return best_s, best_f


def add_candidate(out, seen, i, j, n, limit):
    i = int(i)
    j = int(j)
    if len(out) >= limit:
        return
    if i == j:
        return
    if 0 <= i < n and 0 <= j < n:
        key = i * n + j
        if key not in seen:
            seen.add(key)
            out.append((i, j))


def top_pair_candidates(h, corr, n, max_pairs=120, top_lags=10, per_lag=6, seed=0):
    off = n - 1
    order = np.argsort(-corr)[:top_lags]
    out = []
    seen = set()

    for k in order:
        L = int(k) - off

        if L == 0:
            idx = np.argsort(h)
            lows = idx[:per_lag]
            highs = idx[-per_lag:][::-1]
            for hi in highs:
                for lo in lows:
                    add_candidate(out, seen, hi, lo, n, max_pairs)

        elif L > 0:
            if L >= n:
                continue
            i1 = np.arange(L, n)
            i2 = i1 - L
            if len(i1) == 0:
                continue
            w = h[i1] * (1.0 - h[i2])
            top = np.argsort(-w)[:per_lag]
            for t in top:
                add_candidate(out, seen, i1[t], i2[t], n, max_pairs)

        else:
            Lb = -L
            if Lb >= n:
                continue
            i1 = np.arange(0, n - Lb)
            i2 = i1 + Lb
            if len(i1) == 0:
                continue
            w = h[i1] * (1.0 - h[i2])
            top = np.argsort(-w)[:per_lag]
            for t in top:
                add_candidate(out, seen, i1[t], i2[t], n, max_pairs)

    rng = np.random.RandomState(seed)
    for _ in range(max_pairs * 20):
        i = int(rng.randint(n))
        j = int(rng.randint(n))
        add_candidate(out, seen, i, j, n, max_pairs)
        if len(out) >= max_pairs:
            break

    return out


def coord_refine(h0, time_limit, seed=0):
    n = len(h0)
    h = project_h(h0, n).copy()
    dx = 2.0 / n

    corr = compute_corr(h)
    c_raw = float(np.max(corr))
    c5 = c_raw * dx
    best_h = h.copy()
    best_c5 = c5
    rows = coord_rows(h)

    if time_limit <= 1.0:
        return best_h, best_c5

    t0 = time.time()
    rng = np.random.RandomState(seed)
    cycle = 0
    noimp = 0

    while time.time() - t0 < time_limit - 0.5:
        if c5 < best_c5 - 1e-13:
            best_c5 = c5
            best_h = h.copy()

        if time.time() - t0 >= time_limit - 1.5:
            break

        max_pairs = 120 if n <= 120 else 80
        pairs = top_pair_candidates(h, corr, n, max_pairs=max_pairs, seed=seed + cycle)
        improved = False

        for i, j in pairs:
            if time.time() - t0 >= time_limit - 0.5:
                break

            s, fmin = pair_grid_min(h, corr, rows, i, j, n, c_raw)
            if fmin < c_raw - 1e-12:
                h[i] += s
                h[j] -= s

                if h[i] < 0.0:
                    h[i] = 0.0
                elif h[i] > 1.0:
                    h[i] = 1.0
                if h[j] < 0.0:
                    h[j] = 0.0
                elif h[j] > 1.0:
                    h[j] = 1.0

                if abs(float(np.sum(h)) - n / 2.0) > 1e-9:
                    h = project_h(h, n)

                corr = compute_corr(h)
                c_raw = float(np.max(corr))
                c5 = c_raw * dx
                rows = coord_rows(h)

                improved = True
                noimp = 0
                break

        if not improved:
            noimp += 1
            if noimp >= 3 and c5 > best_c5 + 0.0005:
                h = best_h.copy()
                corr = compute_corr(h)
                c_raw = float(np.max(corr))
                c5 = c_raw * dx
                rows = coord_rows(h)
                noimp = 0

        if not improved and time.time() - t0 < time_limit - 4:
            if cycle < 80:
                scale = 0.004 + 0.004 * min(float(cycle), 8.0)
                if noimp > 3:
                    scale *= 2.0

                hp = project_h(h + rng.randn(n) * scale, n)
                hpcorr = compute_corr(hp)
                hpraw = float(np.max(hpcorr))
                hpc5 = hpraw * dx

                if hpc5 <= best_c5 + 0.0015 and hpc5 <= c5 + 0.002:
                    h = hp
                    corr = hpcorr
                    c_raw = hpraw
                    c5 = hpc5
                    rows = coord_rows(h)
            else:
                break

        cycle += 1

    if c5 < best_c5:
        best_h = h.copy()
        best_c5 = c5

    return best_h, best_c5


def local_pipeline(h0, time_limit, seed, profile="deep"):
    n = len(h0)
    h = project_h(h0, n).copy()
    c = compute_c5(h)
    best_h = h.copy()
    best_c = c

    if time_limit <= 2.0:
        return best_h, best_c

    t0 = time.time()

    if profile == "quick":
        phases = [("lp", 0.55), ("coord", 0.45)]
    elif profile == "medium":
        if n <= 110:
            phases = [("smooth", 0.10), ("lp", 0.42), ("coord", 0.48)]
        else:
            phases = [("smooth", 0.06), ("lp", 0.40), ("coord", 0.54)]
    elif profile == "final":
        phases = [("coord", 0.65), ("lp", 0.35)]
    else:
        if n <= 100:
            phases = [("smooth", 0.12), ("lp", 0.34), ("coord", 0.34), ("lp", 0.20)]
        else:
            phases = [("smooth", 0.08), ("lp", 0.40), ("coord", 0.52)]

    for ph_idx, (name, weight) in enumerate(phases):
        elapsed = time.time() - t0
        alloc = time_limit * weight - elapsed
        if alloc < 5.0:
            continue

        if name == "smooth":
            h2, c2 = smooth_descent(h, alloc, seed + 1)
        elif name == "lp":
            h2, c2 = lp_local_search(h, alloc, seed + 2)
        else:
            h2, c2 = coord_refine(h, alloc, seed + 3)

        if c2 < c - 1e-15:
            h = h2
            c = c2
            if c < best_c:
                best_c = c
                best_h = h.copy()

    if c < best_c:
        best_c = c
        best_h = h.copy()

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8.0
    np.random.seed(seed)

    try:
        h0 = np.asarray(initial_h_values, dtype=float).reshape(-1)
        if len(h0) == 0:
            raise ValueError("empty parent")
    except Exception:
        h0 = make_start(81, seed)

    n0 = len(h0)
    h0 = project_h(h0, n0)
    best_h = h0.copy()
    best_c = compute_c5(best_h)

    for mult in [0.70, 0.85, 1.15, 1.35, 1.70, 2.10]:
        hc = 0.5 + mult * (best_h - 0.5)
        hc = project_h(hc, len(best_h))
        cc = compute_c5(hc)
        if cc < best_c - 1e-12:
            best_c = cc
            best_h = hc.copy()

    rem = deadline - time.time()
    if rem > 60.0:
        alloc = min(420.0, rem * 0.38)
        if alloc > 20.0:
            h1, c1 = local_pipeline(best_h, alloc, seed + 101, profile="deep")
            if c1 < best_c - 1e-12:
                best_c = c1
                best_h = h1.copy()

    if budget_s >= 1000:
        max_n = 241
    elif budget_s >= 750:
        max_n = 221
    elif budget_s >= 500:
        max_n = 201
    else:
        max_n = 161

    incs = [20, 40, 70, 100, 130, 160]
    targets = []
    for inc in incs:
        nt = n0 + inc
        if nt > n0 and nt <= max_n:
            if not any(abs(nt - t) <= 3 for t in targets):
                targets.append(nt)
    targets.sort()

    chain_h = best_h.copy()
    chain_c = best_c

    for tidx, nt in enumerate(targets):
        rem = deadline - time.time()
        remaining = len(targets) - tidx
        if rem < 35 + 25 * remaining:
            break

        if best_c < chain_c:
            chain_h = best_h.copy()
            chain_c = best_c

        alloc = min(170.0, max(25.0, rem / (remaining + 0.5)))
        if alloc < 20.0:
            continue

        h_trial = resample_h(chain_h, nt)
        c_trial = compute_c5(h_trial)

        if c_trial <= best_c + 0.004:
            rng_t = np.random.RandomState(seed + 2000 + nt)
            h_trial = project_h(h_trial + rng_t.randn(nt) * 0.004, nt)

            h_new, c_new = local_pipeline(h_trial, alloc, seed + 3000 + nt, profile="medium")

            if c_new < best_c - 1e-12:
                best_c = c_new
                best_h = h_new.copy()

            if c_new < chain_c:
                chain_c = c_new
                chain_h = h_new.copy()
        else:
            if c_trial < best_c - 1e-12:
                best_c = c_trial
                best_h = h_trial.copy()
            if c_trial < chain_c:
                chain_c = c_trial
                chain_h = h_trial.copy()

    rem = deadline - time.time()
    if rem > 40.0:
        alloc = min(150.0, rem - 3.0)
        if alloc > 10.0:
            h1, c1 = local_pipeline(best_h, alloc, seed + 55555, profile="final")
            if c1 < best_c - 1e-12:
                best_c = c1
                best_h = h1.copy()

    best_h = project_h(best_h, len(best_h))
    best_c = compute_c5(best_h)

    return best_h, best_c, int(len(best_h))
