def line_minimize(h, d, n, num_samples=200):
    """Minimize F(t) = dx * max(R + t*Lin + t^2*Quad) on feasible interval."""
    dx = 2.0 / n
    R0 = np.correlate(h, 1.0 - h, mode='full')
    
    # Compute Lin and Quad using finite differences
    # R(t) = R0 + t*Lin + t^2*Quad
    # R(1) = R0 + Lin + Quad
    # R(2) = R0 + 2*Lin + 4*Quad
    # Lin = 2*R(1) - R(2) - R0
    # Quad = R(2) - 2*R(1) + R0
    
    h1 = h + d
    R1 = np.correlate(h1, 1.0 - h1, mode='full')
    h2 = h + 2.0 * d
    R2 = np.correlate(h2, 1.0 - h2, mode='full')
    
    Lin = 2.0 * R1 - R2 - R0
    Quad = R2 - 2.0 * R1 + R0
    
    t_lo, t_hi = feasible_interval(h, d)
    if t_hi <= t_lo:
        return 0.0, exact_score(h)
    
    # Ensure t=0 is in the interval (it should be since h is feasible)
    t_lo = min(t_lo, 0.0)
    t_hi = max(t_hi, 0.0)
    
    # Sample points
    t_samples = np.linspace(t_lo, t_hi, num_samples)
    # Add t=0
    t_samples = np.concatenate([t_samples, [0.0]])
    
    # Evaluate F(t) = dx * max(R0 + t*Lin + t^2*Quad) for each t
    # Vectorized: for each t, compute max over K
    # Shape: (num_t, L)
    L = len(R0)
    tt = t_samples[:, np.newaxis]
    values = R0[np.newaxis, :] + tt * Lin[np.newaxis, :] + tt**2 * Quad[np.newaxis, :]
    scores = np.max(values, axis=1) * dx
    
    best_idx = np.argmin(scores)
    t_best = t_samples[best_idx]
    score_best = scores[best_idx]
    
    # Refine: check vertex of active quadratic
    active = np.argmax(R0 + t_best * Lin + t_best**2 * Quad)
    a = Quad[active]
    b = Lin[active]
    if abs(a) > 1e-15:
        t_vertex = -b / (2.0 * a)
        if t_lo < t_vertex < t_hi:
            s_v = (R0[active] + t_vertex * b + t_vertex**2 * a) * dx
            if s_v < score_best:
                t_best = t_vertex
                score_best = s_v
    
    # Also check a few nearby crossings
    # For the active quadratic, find where it crosses its neighbors
    active_val = R0[active] + t_best * Lin[active] + t_best**2 * Quad[active]
    for k in range(L):
        if k == active:
            continue
        # Solve: (Quad[k]-Quad[active])t^2 + (Lin[k]-Lin[active])t + (R0[k]-R0[active]) = 0
        a_diff = Quad[k] - Quad[active]
        b_diff = Lin[k] - Lin[active]
        c_diff = R0[k] - R0[active]
        if abs(a_diff) < 1e-15:
            if abs(b_diff) > 1e-15:
                t_cross = -c_diff / b_diff
                if t_lo < t_cross < t_hi:
                    s_c = dx * np.max(R0 + t_cross * Lin + t_cross**2 * Quad)
                    if s_c < score_best:
                        t_best = t_cross
                        score_best = s_c
        else:
            disc = b_diff**2 - 4.0 * a_diff * c_diff
            if disc >= 0:
                sq = np.sqrt(disc)
                for t_c in [(-b_diff + sq) / (2.0 * a_diff), (-b_diff - sq) / (2.0 * a_diff)]:
                    if t_lo < t_c < t_hi:
                        s_c = dx * np.max(R0 + t_c * Lin + t_c**2 * Quad)
                        if s_c < score_best:
                            t_best = t_c
                            score_best = s_c
    
    # Clamp to feasible interval
    t_best = max(t_lo, min(t_hi, t_best))
    
    return t_best, score_best
