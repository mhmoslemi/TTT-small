        if abs(res) < 1e-12:

        h_opt, v_opt = optimize_smooth(h_smooth, tau_fine, n_cur / 2.0, remaining, maxiter=150)
        if v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()

        # Sharpened version
        h_sharp = project_feasible(1.5 * best_h - 0.25, n_cur / 2.0)
        h_opt, v_opt = optimize_smooth(h_sharp, tau_fine, n_cur / 2.0, remaining, maxiter=120)
        if v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()

    # Phase 6: Try 4x upscale if time and resolution allow
    if remaining() > 180 and n_cur <= 200:
        factor = 2 if n_cur >= 180 else 4
        h_up4 = upscale(best_h, factor)
        n_up4 = len(h_up4)
        if n_up4 <= 600:
            h_up4 = project_feasible(h_up4, n_up4 / 2.0)
            h_opt, v_opt = optimize_smooth(h_up4, tau_main[:8], n_up4 / 2.0, remaining, maxiter=100)
            if v_opt < best_val:
                best_val = v_opt
                best_h = h_opt.copy()

    # Phase 7: Fine-tuning with very small tau
    if remaining() > 20:
        h_fin = best_h.copy()
        n_fin = len(h_fin)
        tau_very_fine = [0.005, 0.003, 0.0015, 0.0008]
        h_opt, v_opt = optimize_smooth(h_fin, tau_very_fine, n_fin / 2.0, remaining, maxiter=300, ftol=1e-10)
        if v_opt < best_val:
            best_val = v_opt
            best_h = h_opt.copy()

    # Phase 8: More restarts if plenty of time
    if remaining() > 60:
        for ri in range(6):
            if remaining() < 15:
                break
            sigma = 0.05 + 0.05 * (ri % 4)
            h_pert = project_feasible(best_h + sigma * rng.standard_normal(n_cur), n_cur / 2.0)
            h_opt, v_opt = optimize_smooth(h_pert, tau_fine, n_cur / 2.0, remaining, maxiter=100)
            if v_opt < best_val:
                best_val = v_opt
                best_h = h_opt.copy()
            n_cur = len(best_h)

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)

    return best_h, float(best_val), len(best_h)
