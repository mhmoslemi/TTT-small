        v = eval_f(t)
        if v < best_val:
            best_val = v
            best_t = t
    
    # Local refinement
    step = 1.0 / 500.0
    for _ in range(25):
        found = False
        for dt in (-step, step):
            tt = best_t + dt
            if 0.0 <= tt <= 1.0:
                v = eval_f(tt)
                if v < best_val - 1e-16:
                    best_val = v
                    best_t = tt
                    found = True
        if not found:
            break
        step *= 0.6
    
    return best_t


def coord_descent(h, N, n_full_sweeps=1):
    """Run coordinate descent on h, in-place. Returns updated h."""
    target_sum = N / 2.0
    for _ in range(n_full_sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        for i in range(N):
            old = h[i]
            new = find_optimal_t(corr, h, i, N)
            h[i] = new
            delta = new - old
            if abs(delta) > 1e-15:
                s = np.arange(2 * N - 1) - (N - 1)
                is_valid = (s >= -i) & (s <= N - 1 - i) & (s != 0)
                ims_valid = (s >= i - N + 1) & (s <= i) & (s != 0)
                if np.any(is_valid):
                    idx = i + s[is_valid]
                    corr[is_valid] += delta * (1.0 - h[idx])
                if np.any(ims_valid):
                    idx2 = i - s[ims_valid]
                    corr[ims_valid] -= delta * h[idx2]
                corr[N - 1] += delta * (1.0 - new - old)
    return h


def coord_descent_random(h, N, rng, n_full_sweeps=1):
    """Coordinate descent with random ordering."""
    for _ in range(n_full_sweeps):
        corr = np.correlate(h, 1.0 - h, mode='full')
        order = rng.permutation(N)
        for i in order:
            old = h[i]
            new = find_optimal_t(corr, h, i, N)
            h[i] = new
            delta = new - old
            if abs(delta) > 1e-15:
                s = np.arange(2 * N - 1) - (N - 1)
                is_valid = (s >= -i) & (s <= N - 1 - i) & (s != 0)
                ims_valid = (s >= i - N + 1) & (s <= i) & (s != 0)
                if np.any(is_valid):
                    idx = i + s[is_valid]
                    corr[is_valid] += delta * (1.0 - h[idx])
                if np.any(ims_valid):
                    idx2 = i - s[ims_valid]
                    corr[ims_valid] -= delta * h[idx2]
                corr[N - 1] += delta * (1.0 - new - old)
    return h


def random_binary(N, rng):
    """Create a random binary sequence with exactly N/2 ones."""
    h = np.zeros(N)
    ones = rng.choice(N, N // 2, replace=False)
    h[ones] = 1.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 15.0
    
    rng = np.random.default_rng(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    N = len(h_init)
    target_sum = N / 2.0
    h_init = project_feasible(h_init, target_sum)
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Phase 1: Coordinate descent from initial point
    h_work = h_init.copy()
    n_sweeps = 0
    prev_c5 = best_c5
    stuck_count = 0
    
    while time.time() < deadline - 60.0:
        h_work = coord_descent_random(h_work, N, rng, n_full_sweeps=1)
        c5 = compute_c5(h_work)
        n_sweeps += 1
        
        if c5 < best_c5 - 1e-14:
            best_c5 = c5
            best_h = h_work.copy()
            stuck_count = 0
        elif c5 < prev_c5 + 1e-14:
            stuck_count += 1
        else:
            stuck_count = 0
        
        prev_c5 = c5
        
        # If stuck, try perturbation
        if stuck_count >= 8:
            pert_scale = min(0.02 * (stuck_count // 4 + 1), 0.15)
            h_pert = best_h + rng.normal(0, pert_scale, N)
            h_pert = project_feasible(h_pert, target_sum)
            h_pert = coord_descent_random(h_pert, N, rng, n_full_sweeps=3)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
                h_work = h_pert.copy()
                stuck_count = 0
            else:
                h_work = best_h.copy()
                stuck_count = 0
    
    # Phase 2: Try random binary starting points
    remaining = deadline - time.time()
    if remaining > 40.0:
        n_starts = min(15, int(remaining / 25))
        for _ in range(n_starts):
            if time.time() > deadline - 30.0:
                break
            h_bin = random_binary(N, rng)
            # Smooth slightly then optimize
            h_bin = project_feasible(h_bin, target_sum)
            h_bin = coord_descent_random(h_bin, N, rng, n_full_sweeps=5)
            c5_bin = compute_c5(h_bin)
            if c5_bin < best_c5:
                best_c5 = c5_bin
                best_h = h_bin.copy()
    
    # Phase 3: More perturbation attempts from best
    remaining = deadline - time.time()
    if remaining > 20.0:
        for scale in [0.03, 0.05, 0.08, 0.04, 0.06, 0.1, 0.02, 0.07]:
            if time.time() > deadline - 10.0:
                break
            h_pert = best_h + rng.normal(0, scale, N)
            h_pert = project_feasible(h_pert, target_sum)
            h_pert = coord_descent_random(h_pert, N, rng, n_full_sweeps=4)
            c5_pert = compute_c5(h_pert)
            if c5_pert < best_c5:
                best_c5 = c5_pert
                best_h = h_pert.copy()
    
    # Phase 4: Final polish with more sweeps
    remaining = deadline - time.time()
    if remaining > 5.0:
        h_work = best_h.copy()
        n_extra = min(5, max(1, int(remaining / 3)))
        h_work = coord_descent_random(h_work, N, rng, n_full_sweeps=n_extra)
        c5_final = compute_c5(h_work)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_work.copy()
    
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, len(best_h)
