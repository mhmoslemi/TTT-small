def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)

def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target} via bisection on shift."""
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-9:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
        r2 = target - y.sum()
        if r2 > 0:
            idx = np.where(y < 1.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
        elif r2 < 0:
            idx = np.where(y > 0.0)[0]
            if len(idx) > 0:
                y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n

    return float(np.max(res))

    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def grad_lag(h, Lag):
    n = len(h)
    g = np.zeros(n)
    if Lag >= 0:
        if Lag < n:
            m1 = np.arange(Lag, n)
            g[m1] += 1.0 - h[m1 - Lag]
            m2 = np.arange(0, n - Lag)
            g[m2] -= h[m2 + Lag]
    else:
        l = -Lag
        if l < n:

            g[m1] += 1.0 - h[m1 + l]
            m2 = np.arange(l, n)
            g[m2] -= h[m2 - l]
    return g

def lp_trust(h0, deadline, max_iters=500, init_radius=0.05):
    """LP trust region with adaptive radius."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = init_radius
    no_improve = 0
    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        if vmax < best_c:
            best_c = vmax
            best_h = h.copy()
        # Active set: lags within gap of max
        gap = max(0.005 * vmax, 0.003)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > 60:
            # Keep only the top 60
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-60:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-10:]
        
        # Build LP
        n_act = len(active_idx)
        A = np.zeros((n_act, n))
        b_lin = np.zeros(n_act)
        for i, k in enumerate(active_idx):
            Lag = k - (n - 1)
            A[i] = grad_lag(h, Lag)
            b_lin[i] = A[i] @ h - v[k]
        
        # LP: minimize t s.t. A*d - t <= -b_lin, sum(d)=0, |d| <= radius
        # Variable: d (n) + t (1)
        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0  # minimize t
        
        A_ub = np.zeros((n_act, n_var))
        A_ub[:, :n] = A
        A_ub[:, n] = -1.0
        b_ub = -b_lin  # A*d - t <= -b_lin means corr_k(h) + grad_k·d - t <= 0
        
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-1.0]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [1.0]])
        
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:

        if res is not None and res.success and res.x is not None:
            h_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(h_new)

        else:

    return best_h, best_c

def projected_grad(h0, deadline, max_iters=3000):
    """Projected subgradient method with exponential weighting."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(h0)
    step = 0.01
    stall = 0

        v = corr_all(h)

            stall = 0
        else:
            stall += 1
        if stall > 300:
            break
        # Compute weighted gradient
        m = c_cur
        beta = 50.0
        expo = np.clip(beta * (v - m), None, 400)
        w = np.exp(expo)
        Z = w.sum()
        w = w / Z
        g = np.zeros(n)
        for k in range(len(v)):
            if w[k] > 1e-10:
                Lag = k - (n - 1)
                g += w[k] * grad_lag(h, Lag)

        if g_norm < 1e-14:
            break

        h_new = project_sum_box(h_new, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < c_cur - 1e-14:
            h = h_new
            step = min(step * 1.02, 0.05)
        else:
            step *= 0.85
            if step < 1e-7:
                break
    return best_h, best_c

def project_sum_box(x, target):

    x = np.array(x, dtype=float).copy()

    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - y
            ts = float(slack.sum())
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = float(y.sum())
            if ty > 1e-12:

    r2 = target - y.sum()

            g[m1] += 1.0 - h[m1 - Lag]
            m2 = np.arange(0, n - Lag)
            g[m2] -= h[m2 + Lag]

            g[m1] += 1.0 - h[m1 + l]

    # Phase 3: Fine LP at current resolution
    dl3 = t0 + budget_s * 0.48
    if deadline > dl3:
        try:
            h3, c3 = lp_trust(best_h, dl3, max_iters=400, init_radius=0.02, gap_frac=0.008)
            if c3 < best_c:
                best_h, best_c = h3, c3
        except Exception:
            pass

    # Phase 4: Multi-resolution exploration
    n_cur = len(best_h)
    res_list = []
    for f in [1.2, 1.4, 1.7, 2.1, 2.5]:
        nn = int(round(n_cur * f))
        if 50 <= nn <= 350:
            res_list.append(nn)
    res_list = sorted(set(res_list))
    res_list = [r for r in res_list if r != n_cur]

    for n_t in res_list:
        if time.time() > deadline - 50:
            break
        try:
            h_tr = transfer_res(best_h, n_t)
            c_tr = evaluate_c5(h_tr)
            if c_tr < best_c:
                best_h = h_tr.copy()
                best_c = c_tr
            sub_dl = min(deadline - 30, time.time() + min(80, budget_s * 0.10))
            h_opt, c_opt = lp_trust(h_tr, sub_dl, max_iters=120, init_radius=0.04, gap_frac=0.015)
            if c_opt < best_c:
                best_h, best_c = h_opt, c_opt
            if time.time() < deadline - 15:
                sub_dl2 = min(deadline - 10, time.time() + 25)
                h_pol, c_pol = subgrad_method(h_opt, sub_dl2, max_iters=800)
                if c_pol < best_c:
                    best_h, best_c = h_pol, c_pol
        except Exception:
            pass

    # Phase 5: Multi-start restarts at best resolution
    if time.time() < deadline - 50:
        n_ms = len(best_h)
        for attempt in range(8):
            if time.time() > deadline - 10:
                break
            noise = np.random.randn(n_ms) * 0.025
            h_pert = project_sum_box(best_h + noise, n_ms / 2.0)
            sub_dl = min(deadline - 3, time.time() + 18)
            try:
                h_ms_res, c_ms_res = lp_trust(h_pert, sub_dl, max_iters=100, init_radius=0.10, gap_frac=0.02)
                if c_ms_res < best_c:
                    best_h, best_c = h_ms_res, c_ms_res
            except Exception:
                pass

    # Phase 6: Final fine-tuning LP
    if time.time() < deadline - 3:
        try:
            h_f, c_f = lp_trust(best_h, deadline, max_iters=300, init_radius=0.01, gap_frac=0.005)
            if c_f < best_c:
                best_h, best_c = h_f, c_f
        except Exception:
            pass

    # Final subgradient polish
    if time.time() < deadline:
        try:
            h_sp, c_sp = subgrad_method(best_h, deadline, max_iters=3000)
            if c_sp < best_c:
                best_h, best_c = h_sp, c_sp
        except Exception:
            pass

    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
