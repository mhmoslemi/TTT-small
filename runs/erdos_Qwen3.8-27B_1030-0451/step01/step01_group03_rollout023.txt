import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))

def project_feasible(y, target):
    """Project y onto {h: 0<=h<=1, sum(h)=target}"""
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = mask.sum()
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_to(h, nn):
    """Interpolate h to nn points"""
    np_ = len(h)
    if nn == np_:
        return h.copy()
    xo = np.linspace(0.0, 2.0, np_, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h_new = np.interp(xn, xo, h)
    return project_feasible(h_new, nn / 2.0)

def active_gradient(h, top_k=1):
    """Compute gradient focusing on top_k correlation lags"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    # Find top k lags
    idx = np.argsort(c)[::-1][:top_k]
    c_max = c[idx[0]]
    # Weight: softmax over top k
    s = (c[idx] - c_max) / max(c_max * 0.01, 1e-10)
    w = np.exp(s)
    w /= w.sum()
    
    grad = np.zeros(n)
    for ki, k in enumerate(idx):
        d = k - (n - 1)  # actual shift
        a = w[ki] * dx
        if d >= 0:
            # c[d] = sum_{i=0}^{n-1-d} h[i]*(1-h[i+d])
            # dc/dd_h[i] = (1-h[i+d]) for i in [0, n-1-d]
            # dc/dd_h[i+d] = -h[i] for i in [0, n-1-d]
            m = n - d
            grad[:m] += a * (1.0 - h[d:m+d])
            grad[d:] -= a * h[:m]
        else:
            # d < 0, let d' = -d
            dp = -d
            # c[-dp] = sum_{i=dp}^{n-1} h[i]*(1-h[i-dp])
            m = n - dp
            grad[dp:] += a * (1.0 - h[:m])
            grad[:m] -= a * h[dp:]
    
    return grad, c.max()

def gradient_step(h, step_size, top_k=1):
    """Take one gradient step with projection"""
    n = len(h)
    target = n / 2.0
    grad, cmax = active_gradient(h, top_k)
    # Project to zero-mean (tangent to sum constraint)
    grad = grad - grad.mean()
    # Step
    y = h - step_size * grad
    h_new = project_feasible(y, target)
    return h_new, cmax

def pair_transfer_optimize(h, n_trials=500, deadline=0):
    """Try random pair transfers to improve"""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    for t in range(n_trials):
        if time.time() > deadline:
            break
        # Pick two random indices
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # Try moving delta from j to i
        for delta in [0.001, 0.005, 0.01, 0.02, -0.001, -0.005, -0.01, -0.02]:
            h_try = best_h.copy()
            h_try[i] += delta
            h_try[j] -= delta
            if h_try[i] < -1e-10 or h_try[i] > 1.0 + 1e-10:
                continue
            if h_try[j] < -1e-10 or h_try[j] > 1.0 + 1e-10:
                continue
            c5_try = eval_c5(h_try)
            if c5_try < best_c5 - 1e-12:
                best_h = h_try
                best_c5 = c5_try
                break  # Try different pair
    
    return best_h, best_c5

c[m] = sum_i h[i+m] * (1-h[i])

If I change h from h to h' where h'[j] = h[j] + δ and h'[i] = h[i] for i≠j:

c'[m] = c[m] + δ * (1-h[j-m]) [if j-m in [0,n)) - δ * h[j+m] [if j+m in [0,n)) - δ^2 [if m=0 and j in range]

c'[0] = c[0] + δ*(1-h[j]) - (h[j]+δ)*δ = c[0] + δ - 2*h[j]*δ - δ^2

For m≠0: c'[m] = c[m] + δ*(1-h[j-m]) - δ*h[j+m] (where valid indices are used, and terms are 0 if index out of range)

So for m≠0, c'[m] is linear in δ. For m=0, it's quadratic in δ.

The new objective is C5' = max(c_unaffected_max, max_{m in S_j} c'[m])

Since c'[m] for m≠0 is linear in δ, and c'[0] is quadratic (concave, opening downward), the max is a piecewise function. We want to minimize it over δ ∈ [-h[j], 1-h[j]].

This is a 1D optimization problem that can be solved by:
1. Finding all breakpoints (where the max switches from one c'[m] to another, or where the quadratic switches dominance)
2. Evaluating at breakpoints and at the critical point of the quadratic

This is doable but a bit involved. Let me simplify: just do a small grid search over δ for each j.

import numpy as np
from scipy.optimize import minimize
import time

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))

def get_cvec(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return c

def project_feasible(y, target):
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-14:
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def interp_to(h, nn):
    np_ = len(h)
    if nn == np_:
        return h.copy()
    xo = np.linspace(0.0, 2.0, np_, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h_new = np.interp(xn, xo, h)
    return project_feasible(h_new, nn / 2.0)

def grad_single_lag(h, k_idx):
    """Gradient of c[k_idx] with respect to h"""
    n = len(h)
    dx = 2.0 / n
    d = k_idx - (n - 1)  # actual shift
    grad = np.zeros(n)
    if d >= 0:
        m = n - d
        if m > 0:
            grad[:m] += (1.0 - h[d:d+m]) * dx
            grad[d:d+m] -= h[:m] * dx
    else:
        dp = -d
        m = n - dp
        if m > 0:
            grad[dp:dp+m] += (1.0 - h[:m]) * dx
            grad[:m] -= h[dp:dp+m] * dx
    return grad

def optimize_gradient(h, deadline, top_k=3, verbose=False):
    """Optimize using projected gradient on top-k lags"""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    step = 0.005
    stall = 0
    
    while time.time() < deadline:
        c = get_cvec(h)
        cmax = c.max()
        
        # Top k lags
        idx = np.argsort(c)[::-1][:top_k]
        
        # Compute weighted gradient
        # Weights: focus on the top, with exponential decay
        c_top = c[idx]
        w = np.exp(-(c_top - c_top[0]) / max(0.01 * c_top[0], 1e-10))
        w /= w.sum()
        
        grad = np.zeros(n)
        for ki in range(len(idx)):
            grad += w[ki] * grad_single_lag(h, idx[ki])
        
        # Project to zero-mean
        grad -= grad.mean()
        
        # Backtracking line search
        alpha = step
        improved = False
        for _ in range(30):
            y = h - alpha * grad
            h_new = project_feasible(y, target)
            c5_new = eval_c5(h_new)
            if c5_new < best_c5 - 1e-13:
                h = h_new
                best_h = h_new.copy()
                best_c5 = c5_new
                step = min(0.05, step * 1.1)
                stall = 0
                improved = True
                break
            alpha *= 0.5
        
        if not improved:
            step *= 0.7
            stall += 1
            if stall > 20 or step < 1e-9:
                break
            # Perturbation
            if stall > 10:
                rng_p = np.random.default_rng(int(time.time() * 1000) % 100000)
                h = project_feasible(h + rng_p.standard_normal(n) * 0.002, target)
                stall = 0
    
    return best_h, best_c5

def coordinate_descent(h, max_passes=20, deadline=0):
    """Coordinate descent: for each j, find optimal h[j]"""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for j in range(n):
            # Try a few values for h[j]
            h_orig = best_h[j]
            # Candidate values: current ± small, 0, 1
            candidates = [h_orig]
            for delta in [0.01, 0.005, 0.002, 0.001, -0.01, -0.005, -0.002, -0.001]:
                v = h_orig + delta
                if 0 <= v <= 1:
                    candidates.append(v)
            candidates.extend([0.0, 1.0])
            
            best_val = h_orig
            best_obj = best_c5
            
            for v in candidates:
                if v == h_orig:
                    continue
                best_h[j] = v
                c5_try = eval_c5(best_h)
                if c5_try < best_obj - 1e-14:
                    best_obj = c5_try
                    best_val = v
            
            best_h[j] = best_val
            if best_val != h_orig:
                improved = True
                best_c5 = best_obj
        
        if not improved:
            break
    
    return best_h, best_c5

def pair_swap(h, n_trials=1000, deadline=0, rng=None):
    """Try pair swaps/transfers"""
    n = len(h)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    for t in range(n_trials):
        if time.time() > deadline:
            break
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            continue
        for delta in [0.005, 0.01, 0.02, -0.005, -0.01, -0.02]:
            if best_h[i] + delta < 0 or best_h[i] + delta > 1:
                continue
            if best_h[j] - delta < 0 or best_h[j] - delta > 1:
                continue
            old_i, old_j = best_h[i], best_h[j]
            best_h[i] += delta
            best_h[j] -= delta
            c5_try = eval_c5(best_h)
            if c5_try < best_c5 - 1e-13:
                best_c5 = c5_try
            else:
                best_h[i] = old_i
                best_h[j] = old_j
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5
    
    # Get parent
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)

    best_h = parent.copy()

        if time.time() > deadline - 500:

        shift = rng.integers(1, n_par)
        s = np.roll(parent, int(shift))
        c5_s = eval_c5(s)

        # Also try reversal + rotation

        c5_s2 = eval_c5(s2)

            best_c5 = c5_s2
            best_h = s2.copy()
    
    # Phase 2: Gradient optimization at base resolution
    if time.time() < deadline - 300:

    # Phase 3: Try different resolutions
    resolutions = []
    for mult in [2, 3, 4, 6, 8]:
        n_new = n_par * mult
        if n_new <= 2400:
            resolutions.append(n_new)
    
    for n_new in resolutions:
        if time.time() > deadline - 100:
            break
        h_interp = interp_to(best_h, n_new)
        h_opt, c5_opt = optimize_gradient(h_interp, deadline - 80, top_k=3)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    # Phase 4: Coordinate descent on best
    if time.time() < deadline - 50:

    # Phase 5: Pair swaps
    if time.time() < deadline - 10:
        best_h, best_c5 = pair_swap(best_h, n_trials=3000, deadline=deadline - 5, rng=rng)
    
    # Final
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    
    return (best_h, best_c5, len(best_h))

def grad_of_c(k_idx, h):
    """Gradient of np.correlate(h, 1-h, 'full')[k_idx] with respect to h."""
    n = len(h)
    dx = 2.0 / n
    d = k_idx - (n - 1)
    grad = np.zeros(n)
    
    if d >= 0:
        # c = sum_{i=0}^{n-1-d} h[i+d]*(1-h[i])
        m = n - d
        if m > 0:
            # d(h[i+d])/dh[j] is 1 when j = i+d, i.e., i = j-d
            # So for j in [d, n-1]: contributes (1-h[j-d])
            grad[d:] += (1.0 - h[:m]) * dx
            # d(1-h[i])/dh[j] is -1 when j = i
            # So for j in [0, m-1]: contributes -h[j+d]
            grad[:m] -= h[d:] * dx
    else:

        # c = sum_{i=p}^{n-1} h[i-p]*(1-h[i])

            # d(h[i-p])/dh[j] is 1 when j = i-p, i.e., i = j+p
