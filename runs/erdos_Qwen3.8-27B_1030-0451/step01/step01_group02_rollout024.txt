def binarize(h, target):

    order = np.argsort(h, kind='stable')[::-1][:target]

    b[order] = 1.0

    max_C = max(C).item()
    best_h = h.copy()
    best_max_C = max_C

    n = y.size
    target = float(target)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(y+mid, 0, 1))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(y + 0.5*(lo+hi), 0, 1)
    # Fix any small sum error
    diff = target - x.sum()
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(x < 1)[0]
            if len(idx):
                x[idx[np.argmin(x[idx])]] += min(1-x[idx].min(), diff)
        else:
            idx = np.where(x > 0)[0]
            if len(idx):
                x[idx[np.argmax(x[idx])]] += max(-x[idx].max(), diff)
    return np.clip(x, 0, 1)

def exact_C5(h):
    n = len(h)
    dx = 2.0/n
    C = np.correlate(h, 1-h, 'full') * dx
    return C.max()

def binarize(h):
    n = len(h)
    target = n // 2
    order = np.argsort(h, kind='stable')[::-1][:target]
    b = np.zeros(n)
    b[order] = 1.0
    return b

def init_vdc(n, alpha=None):
    """Van der Corput based initialization."""
    if alpha is None:
        alpha = (np.sqrt(5)-1)/2  # golden ratio conjugate
    frac = (np.arange(n) * alpha) % 1.0
    h = (frac < 0.5).astype(float)
    # Adjust to have exactly n/2 ones
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        # Remove excess ones: remove the ones with highest frac
        one_idx = np.where(h == 1)[0]
        remove = one_idx[np.argsort(frac[one_idx])[:n_ones - target]]
        h[remove] = 0
    elif n_ones < target:
        zero_idx = np.where(h == 0)[0]
        add = zero_idx[np.argsort(frac[zero_idx][::-1])[:target - n_ones]]
        h[add] = 1
    return h

def init_bea(n):
    """Beatty sequence initialization."""
    # h[i] = 1 if floor(i*phi) - floor((i-1)*phi) = 1 (Beatty sequence)
    phi = (1+np.sqrt(5))/2
    h = np.zeros(n)
    # Upper Beatty sequence: positions of 1s are floor(k*phi) for k=1,2,...
    k_max = int(n / phi) + 2
    positions = np.floor(np.arange(1, k_max+1) * phi).astype(int)
    positions = positions[positions < n]
    h[positions] = 1
    # Adjust count
    n_ones = int(h.sum())
    target = n // 2
    if n_ones > target:
        h[np.where(h==1)[0][n_ones-target:]] = 0
    elif n_ones < target:
        zero_idx = np.where(h==0)[0]
        h[zero_idx[:target-n_ones]] = 1
    return h

def binary_SA(h, dx, n_iters, rng, deadline, T0=0.05, cool=0.999):
    """Simulated annealing on binary sequences with incremental C5 update."""
    n = len(h)
    h = h.copy().astype(float)
    C = np.correlate(h, 1-h, 'full') * dx
    max_C = C.max()
    best_h = h.copy()
    best_max_C = max_C
    
    h_pad = np.zeros(3*n - 2)
    h_pad[n-1:2*n-1] = h
    d_arr = np.arange(-(n-1), n)
    offset = n - 1
    
    T = T0
    ones_list = list(np.where(h > 0.5)[0])
    zeros_list = list(np.where(h < 0.5)[0])
    
    step = 0
    while step < n_iters and time.time() < deadline:
        step += 1
        if step % 1000 == 0:
            T *= cool
        
        # Random swap
        p = ones_list[rng.integers(len(ones_list))]
        q = zeros_list[rng.integers(len(zeros_list))]
        if p == q:
            continue
        
        # Compute delta
        idx1 = offset + p - d_arr
        idx2 = offset + q - d_arr
        idx3 = offset + p + d_arr
        idx4 = offset + q + d_arr
        delta = (-h_pad[idx1] + h_pad[idx2] + h_pad[idx3] - h_pad[idx4]) * dx
        
        new_max = (C + delta).max()
        
        if new_max <= max_C:
            accept = True
        else:
            delta_C = new_max - max_C
            if T < 1e-10:
                accept = False
            else:
                accept = rng.random() < np.exp(-delta_C / T)
        
        if accept:
            h[p] = 0
            h[q] = 1
            h_pad[offset+p] = 0
            h_pad[offset+q] = 1
            C = C + delta
            max_C = new_max
            ones_list.remove(p)
            ones_list.append(q)
            zeros_list.remove(q)
            zeros_list.append(p)
            if max_C < best_max_C:
                best_max_C = max_C
                best_h = h.copy()
        
        # Periodic recompute to fix drift
        if step % 10000 == 0:
            C = np.correlate(h, 1-h, 'full') * dx
            max_C = C.max()
    
    return best_h, best_max_C
