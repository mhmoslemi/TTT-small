def grad_c_m(h, m):
    n = len(h)
    dx = 2.0 / n
    omh = 1.0 - h
    grad = np.zeros(n)
    d = m - (n - 1)

    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
