import time

def score(h):
    """Exact C5 evaluation."""
    n = len(h)
    dx = 2.0 / n
    C = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(C))

def project(y, target):
    """Project y onto {x : 0≤x≤1, sum(x)=target}."""
    n = len(y)
    # Bisection on lambda
    lo = min(y) - 1.0 - 1.0  # ensure sum(clip(y-lo,0,1)) ≥ target
    hi = max(y) + 1.0        # ensure sum(clip(y-hi,0,1)) ≤ target
    for _ in range(60):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(y - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5*(lo+hi)
    x = np.clip(y - lam, 0.0, 1.0)
    # Fix sum exactly
    x = fix_sum(x, target)
    return x

def fix_sum(x, target):
    """Fix sum to target with minimal changes."""
    x = np.clip(x.astype(float), 0.0, 1.0)
    r = target - x.sum()
    if abs(r) < 1e-12:
        return x
    if r > 0:
        # Add r, distribute to elements with most room
        order = np.argsort(1.0 - x)  # most room first
        rem = r
        for j in order:
            inc = min(1.0 - x[j], rem)
            x[j] += inc
            rem -= inc
            if rem < 1e-15:
                break
    else:
        r = -r
        order = np.argsort(x)  # least first
        rem = r
        for j in order:
            dec = min(x[j], rem)
            x[j] -= dec
            rem -= dec
            if rem < 1e-15:
                break
    return np.clip(x, 0.0, 1.0)

def smooth_obj_and_grad(h, tau, dx):
    """Compute smooth objective and its gradient."""
    C = np.correlate(h, 1.0 - h, 'full') * dx
    m = np.max(C)
    if tau < 1e-14:
        # Exact max gradient (subgradient = average of active)
        act = C >= m - 1e-12
        alpha = np.zeros_like(C)
        alpha[act] = 1.0 / np.sum(act)
    else:
        a = (C - m) / tau
        e = np.exp(a)
        s = e.sum()
        alpha = e / s
        obj = m + tau * np.log(s)
    # Compute gradient
    n = len(h)
    G = np.zeros(n)
    ctr = n - 1
    for k in range(len(C)):
        ak = alpha[k] * dx
        if ak < 1e-15:
            continue
        d = k - ctr
        if d >= 0:
            G[:n-d] += ak * (1.0 - h[d:])
            G[d:] -= ak * h[:n-d]
        else:
            dd = -d
            G[dd:] += ak * (1.0 - h[:n-dd])
            G[:n-dd] -= ak * h[dd:]
    return obj, G

def proj_grad_descend(h, target, taudata, deadline):
    """Projected gradient descent with log-sum-exp."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_c = score(h)

        if time.time() > deadline - 10:
            break
        alpha = alpha_init
        for it in range(n_iter):
            if time.time() > deadline - 5:
                break
            obj, G = smooth_obj_and_grad(h, tau, dx)
            # Backtracking line search

                x_new = project(y, target)
                c_new = score(x_new)
                if c_new < best_c - 1e-14:
                    h = x_new
                    best_c = c_new
                    break
                alpha *= 0.5
            else:

                pass
            if alpha < 1e-10:
                break
        # Decay alpha
