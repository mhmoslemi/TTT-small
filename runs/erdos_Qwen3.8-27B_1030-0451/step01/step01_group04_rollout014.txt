**Custom algorithm: Focused Subgradient Descent**

The key insight: at any point h, the "bottleneck" shifts are those k where g_k(h) is close to the maximum. We should focus our optimization on reducing these.

5. Project to the feasible set
6. Repeat

def focused_subgradient(h, N, n_iter, epsilon, alpha_schedule):
    dx = 2.0 / N
    target_sum = N / 2.0
    h = h.copy()
    
    for it in range(n_iter):
        # Compute overlaps
        corr = np.correlate(h, 1.0 - h, mode='full')
        c = corr * dx
        cmax = np.max(c)
        
        # Active set
        active = c > (cmax - epsilon)
        n_active = active.sum()
        
        # Compute gradient (average over active)
        grad = np.zeros(N)
        for r in np.where(active)[0]:
            d = r - (N - 1)
            if d >= 0:
                n = N - d
                grad[:n] += (1.0 - h[d:]) * dx
                if d > 0:
                    grad[d:] -= h[:n] * dx
            else:
                dd = -d
                n = N - dd
                grad[dd:] += (1.0 - h[:n]) * dx
                if dd > 0:
                    grad[:n] -= h[dd:] * dx
        grad /= n_active
        
        # Step size
        alpha = alpha_schedule(it)
        
        # Take step
        h_new = h - alpha * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Check if it improved
        c5_new = compute_c5(h_new)
        if c5_new <= cmax + 1e-10:  # should always be true at first order
            h = h_new
        else:
            # Halve step size
            alpha /= 2
            h_new = h - alpha * grad
            h_new = project_feasible(h_new, target_sum)
            h = h_new
    
    return h

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            for i in idx:
                add = min(residual, 1.0 - h[i])
                h[i] += add
                residual -= add
                if abs(residual) < 1e-13:
                    break
        else:
            idx = np.where(h > 0.0)[0]
            for i in idx:
                rem = max(residual, -(1.0 - h[i]))
                h[i] += rem
                residual -= rem
                if abs(residual) < 1e-13:
                    break
    return h


def symmetrize_project(h, N):
    h = 0.5 * (h + h[::-1].copy())
    return project_feasible(h, N / 2.0)


def smooth_obj(h, tau, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = np.clip((c - m) / tau, -500.0, 500.0)
    logsum = np.log(np.sum(np.exp(shifted)))
    obj = m + tau * logsum
    s = np.sum(h) - N / 2.0
    obj += penalty * s * s
    return obj


def smooth_grad(h, tau, penalty):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    shifted = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(shifted)
    w = w / w.sum()

    grad = np.zeros(N)
    R = 2 * N - 1
    for r in range(R):
        if w[r] < 1e-14:
            continue
        a = w[r] * dx
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            grad[:n] += a * (1.0 - h[d:])
            if d > 0:
                grad[d:] -= a * h[:n]
        else:
            dd = -d
            n = N - dd
            grad[dd:] += a * (1.0 - h[:n])
            if dd > 0:
                grad[:n] -= a * h[dd:]

    s = np.sum(h) - N / 2.0
    grad += 2.0 * penalty * s
    return grad


def subgradient_step(h, N, epsilon, alpha):
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    cmax = np.max(c)
    active = c > (cmax - epsilon)
    n_active = max(active.sum(), 1)

    grad = np.zeros(N)
    active_idx = np.where(active)[0]
    for r in active_idx:
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            grad[:n] += (1.0 - h[d:]) * dx
            if d > 0:
                grad[d:] -= h[:n] * dx
        else:
            dd = -d
            n = N - dd
            grad[dd:] += (1.0 - h[:n]) * dx
            if dd > 0:
                grad[:n] -= h[dd:] * dx
    grad /= n_active

    h_new = h - alpha * grad
    h_new = project_feasible(h_new, N / 2.0)
    return h_new, cmax


def lbgfbsb_run(h, N, tau, penalty, maxiter):
    bounds = [(0.0, 1.0)] * N
    res = minimize(
        smooth_obj, h.copy(),
        jac=smooth_grad,
        args=(tau, penalty),
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': maxiter, 'ftol': 1e-16, 'gtol': 1e-8}
    )
    h_new = res.x
    h_new = project_feasible(h_new, N / 2.0)
    return h_new


def upscale(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)
