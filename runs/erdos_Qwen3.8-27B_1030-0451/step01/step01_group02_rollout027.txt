    C = np.correlate(h, 1.0 - h, "full") * dx
    return C, dx


def C5_of(h):
    C, _ = corr_vals(h)
    return float(np.max(C))


def smooth_obj(h, tau):
    C, _ = corr_vals(h)
    m = float(np.max(C))
    if tau <= 1e-15:
        return m
    a = (C - m) / tau
    a = a - a.max()
    s = float(np.sum(np.exp(a)))
    return m + tau * float(np.log(s))


def smooth_grad(h, C, tau, n):
    dx = 2.0 / float(n)
    m = float(np.max(C))
    if tau <= 1e-15:
        act = C >= m - 1e-12
        cnt = float(np.count_nonzero(act))
        if cnt <= 0:
            alpha = np.zeros_like(C)
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha = np.zeros_like(C)
            alpha[act] = 1.0 / cnt
    else:
        a = (C - m) / tau
        a = a - a.max()
        e = np.exp(a)
        s = float(np.sum(e))
        if s < 1e-300:
            alpha = np.zeros_like(C)
            alpha[int(np.argmax(C))] = 1.0
        else:
            alpha = e / s
    grad = np.zeros(n)
    ctr = n - 1
    dirf = 1
    if _CORR_DIR_SET:
        dirf = _CORR_DIR
    thresh = 1e-12
    active = np.where(alpha > thresh)[0]
    for k in active:
        ak = float(alpha[k])
        d = dirf * (int(k) - ctr)
        cd = ak * dx
        if d == 0:
            grad += cd * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += cd * (1.0 - h[:n - d])
            grad[:n - d] += -cd * h[d:]
        else:
            dd = -d
            grad[:n - dd] += cd * (1.0 - h[dd:])
            grad[dd:] += -cd * h[:n - dd]
    return grad


def proj_grad_step(h, grad, alpha_init, target, n):
    g = grad - np.mean(grad)
    alpha = alpha_init
    best_h, best_f = h.copy(), smooth_obj(h, 0.01)
    for _ in range(30):
        h_new = make_feasible(h - alpha * g, target)
        f_new = smooth_obj(h_new, 0.01)
        if f_new < best_f - 1e-14:
            best_h, best_f = h_new, f_new
            break
        alpha *= 0.5
        if alpha < 1e-10:
            break
    return best_h, best_f, alpha


def grad_descent(h, deadline, rng, tau_start=0.05, tau_end=0.001, min_iter=50):
    n = len(h)
    target = 0.5 * float(n)
    h = h.copy()
    c_best = C5_of(h)
    h_best = h.copy()
    
    n_taus = 12
    taus = tau_start * (tau_end / tau_start) ** (np.arange(n_taus) / (n_taus - 1))
    
    total_iters = 0
    for tau in taus:
        if time.time() > deadline - 10:
            break
        iters = max(min_iter, int(20000.0 / n))
        alpha = 1.0
        for it in range(iters):
            if time.time() > deadline - 10:
                break
            C, dx = corr_vals(h)
            grad = smooth_grad(h, C, tau, n)
            gn = np.linalg.norm(grad)
            if gn < 1e-14:
                break
            alpha = min(alpha, 1.0 / (gn + 1e-6))
            h_new, f_new, alpha = proj_grad_step(h, grad, alpha * 0.9, target, n)
            c_new = C5_of(h_new)
            if c_new < c_best - 1e-14:
                h = h_new
                c_best = c_new
                h_best = h_new.copy()
                total_iters += 1
                alpha = min(alpha * 1.2, 2.0 / (gn + 1e-6))
            else:
                alpha *= 0.7
        if time.time() > deadline - 20:
            break
    
    return h_best, c_best


def binarize(h, target):
    n = len(h)
    cnt = int(round(target))
    if cnt <= 0 or cnt >= n:
        return h.copy()
    order = np.argsort(h, kind="stable")[::-1][:cnt]
    b = np.zeros(n)
    b[order] = 1.0
    return b


def binary_greedy(h, deadline, rng, max_rounds=500):
    n = len(h)
    target = 0.5 * float(n)
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    if not ones or not zeros:
        return h, C5_of(h)
    
    C, dx = corr_vals(h)
    c_best = float(np.max(C))
    
    dirf = 1
    if _CORR_DIR_SET:
        dirf = _CORR_DIR
    ctr = n - 1
    
    stall = 0
    for rnd in range(max_rounds):
        if time.time() > deadline - 5:
            break
        if stall > 30:
            break
        max_k = int(np.argmax(C))
        d_star = dirf * (max_k - ctr)
        
        best_score = 0
        best_i, best_j = -1, -1
        
        sample_ones = ones if len(ones) <= 60 else rng.choice(ones, 60, replace=False).tolist()
        sample_zeros = zeros if len(zeros) <= 60 else rng.choice(zeros, 60, replace=False).tolist()
        
        for i in sample_ones:
            for j in sample_zeros:
                if i == j:
                    continue
                delta = 0
                if 0 <= i - d_star < n:
                    delta += h[i - d_star]
                if 0 <= i + d_star < n:
                    delta += h[i + d_star]
                if 0 <= j - d_star < n:
                    delta -= h[j - d_star]
                if 0 <= j + d_star < n:
                    delta -= h[j + d_star]
                score = dx * delta
                if score < best_score:
                    best_score = score
                    best_i, best_j = i, j
        
        if best_i < 0 or best_score >= -1e-14:
            stall += 1
            # Try random swap
            i = int(rng.choice(ones))
            j = int(rng.choice(zeros))
            h[i], h[j] = h[j], h[i]
            ones.remove(i)
            zeros.remove(j)
            zeros.append(i)
            ones.append(j)
            C_new, _ = corr_vals(h)
            c_new = float(np.max(C_new))
            if c_new < c_best:
                c_best = c_new
                C = C_new
                stall = 0
            else:
                h[i], h[j] = h[j], h[i]
                zeros.remove(i)
                ones.remove(j)
                ones.append(i)
                zeros.append(j)
            continue
        
        # Apply the swap
        i, j = best_i, best_j
        h[i], h[j] = h[j], h[i]
        ones.remove(i)
        zeros.remove(j)
        zeros.append(i)
        ones.append(j)
        C, _ = corr_vals(h)
        c_new = float(np.max(C))
        if c_new < c_best:
            c_best = c_new
            stall = 0
        else:
            h[i], h[j] = h[j], h[i]
            zeros.remove(i)
            ones.remove(j)
            ones.append(i)
            zeros.append(j)
            stall += 1
    
    return h, c_best


def binary_anneal(h, deadline, rng, T0=0.02, T_end=0.001, max_iters=3000):
    n = len(h)
    target = 0.5 * float(n)
    h = h.copy()
    ones = np.where(h > 0.5)[0].tolist()
    zeros = np.where(h <= 0.5)[0].tolist()
    if not ones or not zeros:
        return h, C5_of(h)
    
    C, dx = corr_vals(h)
    c_cur = float(np.max(C))
    c_best = c_cur
    h_best = h.copy()
    
    n_mix = int(0.3 * len(ones)) if len(ones) > 10 else len(ones)
    n_swap = max(1, n_mix)
    
    for it in range(max_iters):
        if time.time() > deadline - 5:
            break
        T = T0 * (T_end / T0) ** (it / max(max_iters, 1))
        
        i = int(rng.choice(ones))
        j = int(rng.choice(zeros))
        h[i], h[j] = h[j], h[i]
        ones.remove(i)
        zeros.remove(j)
        zeros.append(i)
        ones.append(j)
        C, _ = corr_vals(h)
        c_new = float(np.max(C))
        dc = c_new - c_cur
        
        if dc < 0 or rng.random() < np.exp(-dc / max(T, 1e-10)):
            c_cur = c_new
            if c_new < c_best:
                c_best = c_new
                h_best = h.copy()
        else:
            h[i], h[j] = h[j], h[i]
            zeros.remove(i)
            ones.remove(j)
            ones.append(i)
            zeros.append(j)
    
    return h_best, c_best


def resample_to(h, N):
    n0 = len(h)
    target = 0.5 * float(N)
    if n0 == N:
        return make_feasible(h, target)
    xold = (np.arange(n0) + 0.5) * (2.0 / n0)
    xnew = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(xnew, xold, h)
    return make_feasible(y, target)


def fourier_init(N, rng):
    x = (np.arange(N) + 0.5) * (2.0 / N)
    best_h, best_c = None, np.inf
    for f in (1, 2, 3, 4, 5, 6):
        for ph in np.linspace(0, 2 * np.pi, 6, endpoint=False):
            y = 0.5 + 0.4 * np.cos(np.pi * f * x + ph)
            h = make_feasible(y, 0.5 * N)
            c = C5_of(h)
            if c < best_c:
                best_c, best_h = c, h
    for _ in range(12):
        f1 = int(rng.integers(1, 7))
        f2 = int(rng.integers(1, 7))
        ph1 = rng.uniform(0, 2 * np.pi)
        ph2 = rng.uniform(0, 2 * np.pi)
        a1 = rng.uniform(0.2, 0.45)
        a2 = rng.uniform(0.1, 0.3)
        y = 0.5 + a1 * np.cos(np.pi * f1 * x + ph1) + a2 * np.cos(np.pi * f2 * x + ph2)
        h = make_feasible(y, 0.5 * N)
        c = C5_of(h)
        if c < best_c:
            best_c, best_h = c, h
    return best_h, best_c


def get_initial():
    val = globals().get("initial_h_values", None)
    if val is not None:
        arr = np.asarray(val, dtype=float).ravel()
        if arr.size > 0 and np.all(np.isfinite(arr)):
            return make_feasible(arr, 0.5 * arr.size)
    return None


def polish(h, deadline, rng):
    """Final fine-polish with very small tau."""
    n = len(h)
    target = 0.5 * float(n)
    h = h.copy()
    c_best = C5_of(h)
    taus = [0.003, 0.0015, 0.0008, 0.0004, 0.0002]
    for tau in taus:
        if time.time() > deadline - 3:
            break
        for it in range(200):
            if time.time() > deadline - 3:
                break
            C, dx = corr_vals(h)
            grad = smooth_grad(h, C, tau, n)
            gn = np.linalg.norm(grad)
            if gn < 1e-15:
                break
            g = grad - np.mean(grad)
            for alpha in [1.0 / (gn * gn + 1e-8), 0.5 / (gn + 1e-6), 0.1, 0.01, 0.001]:
                h2 = make_feasible(h - alpha * g, target)
                c2 = C5_of(h2)
                if c2 < c_best - 1e-14:
                    h = h2
                    c_best = c2
                    break
    return h, c_best


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    hardcoded = t0 + max(10.0, total) - 8.0
    
    _calibrate_dir()
    rng = np.random.default_rng(int(seed) % (1 << 32))
    
    # Get initial
    h_init = get_initial()
    if h_init is None:
        h_init, _ = fourier_init(480, rng)
    n0 = len(h_init)
    
    best_h = h_init.copy()
    best_c = C5_of(best_h)
    best_n = n0
    
    # Phase 1: Gradient descent at original resolution
    if time.time() < hardcoded - 60:
        dl = min(hardcoded - 30, t0 + 0.35 * total)
        h1, c1 = grad_descent(best_h, dl, rng)
        if c1 < best_c:
            best_h, best_c, best_n = h1, c1, len(h1)
    
    # Phase 2: Binarize + binary local search at original resolution
    if time.time() < hardcoded - 40:
        h_bin = binarize(best_h, 0.5 * best_n)
        dl = min(hardcoded - 20, t0 + 0.50 * total)
        h_b2, c_b2 = binary_greedy(h_bin, dl, rng)
        if c_b2 < best_c:
            best_h, best_c, best_n = h_b2, c_b2, len(h_b2)
    
    # Phase 3: Binary annealing
    if time.time() < hardcoded - 30:
        h_bin = binarize(best_h, 0.5 * best_n)
        dl = min(hardcoded - 15, t0 + 0.62 * total)
        h_a, c_a = binary_anneal(h_bin, dl, rng)
        if c_a < best_c:
            best_h, best_c, best_n = h_a, c_a, len(h_a)
    
    # Phase 4: Multi-resolution
    Ns = []
    for fac in [0.5, 1.5, 2.0, 2.5, 3.0]:
        N = int(round(n0 * fac))
        if 120 <= N <= 1500 and N != n0 and N not in Ns:
            Ns.append(N)
    Ns.sort(key=lambda x: (abs(x - n0), x))
    
    for N in Ns[:4]:
        if time.time() > hardcoded - 50:
            break
        if best_c <= 0.3808 - 1e-9:
            break
        h_r = resample_to(best_h, N)
        # Quick gradient descent at this resolution
        dl = min(hardcoded - 25, time.time() + max(5, 0.15 * total / max(len(Ns), 1)))
        h_g, c_g = grad_descent(h_r, dl, rng, tau_start=0.03, tau_end=0.0005, min_iter=30)
        if c_g < best_c:
            best_h, best_c, best_n = h_g, c_g, len(h_g)
        # Binary at this resolution
        if time.time() < hardcoded - 25:
            h_b = binarize(best_h, 0.5 * best_n)
            dl2 = min(hardcoded - 15, time.time() + max(3, 0.08 * total / max(len(Ns), 1)))
            h_b2, c_b2 = binary_greedy(h_b, dl2, rng)
            if c_b2 < best_c:
                best_h, best_c, best_n = h_b2, c_b2, len(h_b2)
    
    # Phase 5: Final polish
    if time.time() < hardcoded - 5:
        h_p, c_p = polish(best_h, hardcoded - 2, rng)
        if c_p < best_c:
            best_h, best_c, best_n = h_p, c_p, len(h_p)
    
    # Final validation
    best_h = make_feasible(best_h, 0.5 * float(len(best_h)))
    best_c = C5_of(best_h)
    best_n = int(len(best_h))
    
    if not np.isfinite(best_c):
        best_h = make_feasible(np.full(best_n, 0.5), 0.5 * best_n)
        best_c = C5_of(best_h)
    
    return best_h, float(best_c), best_n
