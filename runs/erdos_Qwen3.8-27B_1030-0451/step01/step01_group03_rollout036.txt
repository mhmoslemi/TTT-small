    n = len(h)

    g = np.zeros(n)
    if 0 <= lag < n:
        g[lag:] += dx * (1.0 - h[:n-lag])
        g[:n-lag] -= dx * h[lag:]
    elif -n < lag < 0:
        a = -lag
        g[:n-a] += dx * (1.0 - h[a:])
        g[a:] -= dx * h[:n-a]
    return g

# --- Line search ---

def do_line_search(h_curr, d, c5_curr):
    t_max = 2.0
    p = d > 1e-14
    q = d < -1e-14
    if p.any():
        t_max = min(t_max, float(np.min((1.0 - h_curr[p]) / d[p])))
    if q.any():
        t_max = min(t_max, float(np.min(h_curr[q] / (-d[q]))))
    t_max = max(t_max, 1e-10)
    
    best_t = 0.0
    best_c5 = c5_curr
    best_h = h_curr.copy()
    
    # Coarse grid
    n_g = 20
    for i in range(1, n_g + 1):
        t = i * t_max / n_g
        ht = project_feasible(h_curr + t * d)
        ct = evaluate_c5(ht)
        if ct < best_c5:
            best_c5 = ct
            best_t = t
            best_h = ht
    
    # Ternary refinement
    if 0 < best_t < t_max:
        frac = best_t / t_max
        lo = max(0.0, frac * 0.5)
        hi = min(1.0, frac * 1.5)
        for _ in range(25):
            t1 = lo + (hi - lo) / 3.0
            t2 = hi - (hi - lo) / 3.0
            c1 = evaluate_c5(project_feasible(h_curr + t1 * t_max * d))
            c2 = evaluate_c5(project_feasible(h_curr + t2 * t_max * d))
            if c1 < c2:
                hi = t2
                if c1 < best_c5:
                    best_c5 = c1; best_t = t1 * t_max; best_h = project_feasible(h_curr + best_t * d)
            else:
                lo = t1
                if c2 < best_c5:
                    best_c5 = c2; best_t = t2 * t_max; best_h = project_feasible(h_curr + best_t * d)
    
    return best_h, best_c5

# --- Main optimizer ---

def optimize(h0, deadline):
    n = len(h0)
    dx = 2.0 / n
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h_curr = h0.copy()
    
    max_iters = 800
    for it in range(max_iters):
        if time.time() > deadline - 2:
            break
        
        c = np.correlate(h_curr, 1.0 - h_curr, 'full') * dx
        cmax = float(c.max())
        
        # Active lags
        spread = max(1e-8, cmax - float(c.mean()))
        tol = 0.15 * spread
        active = np.where(c >= cmax - tol)[0]
        if len(active) < 3:
            active = np.argsort(c)[-min(5, len(c)):]
        if len(active) > 20:
            active = active[np.argsort(c[active])[-20:]]
        
        c_act = c[active]
        
        improved = False
        
        # Try several weightings
        weightings = []
        # Uniform
        weightings.append(np.ones(len(active)) / len(active))
        # Softmax
        for t in [0.5, 1.0, 2.0]:
            s = (c_act - c_act.max()) / (t * spread)
            e = np.exp(np.clip(s, -50, 50))
            weightings.append(e / e.sum())
        # Top-3 equal
        if len(active) >= 3:
            w = np.zeros(len(active))
            top3 = np.argsort(c_act)[::-1][:3]
            w[top3] = 1.0 / 3
            weightings.append(w)
        # Top-5 equal
        if len(active) >= 5:
            w = np.zeros(len(active))
            top5 = np.argsort(c_act)[::-1][:5]
            w[top5] = 1.0 / 5
            weightings.append(w)
        
        for w in weightings:
            if time.time() > deadline - 2:
                break
            
            g = np.zeros(n)
            for k in range(len(active)):
                if w[k] > 1e-12:
                    g += w[k] * lag_grad(h_curr, active[k], dx)
            
            d = -(g - g.mean())
            dn = float(np.max(np.abs(d)))
            if dn < 1e-14:
                continue
            d /= dn
            
            h_new, c5_new = do_line_search(h_curr, d, evaluate_c5(h_curr))
            
            if c5_new < best_c5 - 1e-12:
                best_c5 = c5_new
                best_h = h_new.copy()
                h_curr = h_new.copy()
                improved = True
            elif c5_new < evaluate_c5(h_curr) - 1e-12:
                h_curr = h_new.copy()
        
        if not improved and it > 10:
            # Try random perturbation
            rng = np.random.default_rng(hash((it, n)) & 0xFFFFFFFF)
            h_try = project_feasible(h_curr + rng.standard_normal(n) * 0.01)
            c5_try = evaluate_c5(h_try)
            if c5_try < best_c5:
                best_c5 = c5_try
                best_h = h_try.copy()
            h_curr = h_try.copy()
    
    return best_h, best_c5

def interp_to_n(h_src, n_dst):
    n_src = len(h_src)
    if n_dst == n_src:
        return project_feasible(h_src, n_dst * 0.5)
    xo = np.linspace(0, 2, n_src, endpoint=False)
    xn = np.linspace(0, 2, n_dst, endpoint=False)
    h = np.interp(xn, xo, h_src)
    return project_feasible(h, n_dst * 0.5)

def run(seed=42, budget_s=1100, **kwargs):

    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, 'full')) * dx)

def project_feasible(y, target=None):
    y = np.asarray(y, dtype=np.float64)
    y = np.nan_to_num(y, nan=0.0, posinf=3.0, neginf=-3.0)
    n = len(y)
