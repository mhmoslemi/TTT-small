        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    r = target - y.sum()

        if r > 0:
            mask = y < 1.0
            if mask.sum() > 0:
                y[mask] += r / mask.sum()
        else:
            mask = y > 0.0
            if mask.sum() > 0:
                y[mask] += r / mask.sum()
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def corr_vector(h):
    """Return full correlation vector (scaled by dx)."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def pair_exchange_search(h, active_lags, n_samples=30):
    """For each active lag, try pair exchanges at that distance."""
    n = len(h)
    best_h = h.copy()

        if abs(lag) >= n:
            continue
        dist = abs(lag)

            if s < 1e-10 or s > 2 - 1e-10:

                pass

            ts = np.linspace(lo_t, hi_t, n_samples)
            # Add current value

                if t < -1e-10 or t > 1 + 1e-10 or s - t < -1e-10 or s - t > 1 + 1e-10:

                t = np.clip(t, 0.0, 1.0)
                st = np.clip(s - t, 0.0, 1.0)
                h_new = h.copy()
                h_new[i] = t
                h_new[j] = st
                c_new = evaluate_c5(h_new)
                if c_new < best_c - 1e-10:
                    best_c = c_new
                    best_h = h_new.copy()

                for k in active_idx:
                    l = k - (n - 1)
                    if l not in new_lags:
                        new_lags.append(l)
                active_lags = new_lags
    
    return best_h, best_c

The objective is f(h) = max_k g_k(h) where g_k(h) = sum_i h[i]*(1-h[i+k])*dx.

Each g_k is bilinear (not linear! because h appears in both factors). So this is actually a max of bilinear functions, which is a non-convex problem.

The subgradient of f at h (when the max is achieved at a unique k*) is just the gradient of g_{k*}:

Let me be more careful:
g_k = dx * sum_{i: 0≤i<n, 0≤i+k<n} h[i] * (1 - h[i+k])

∂g_k/∂h[i] = dx * [(1-h[i+k]) * 1_{0≤i+k<n} - h[i-k] * 1_{0≤i-k<n}]

The first term: h[i] appears as the first factor in the i-term (if i+k < n).
The second term: h[i] appears as the second factor (inside 1-h) in the (i-k)-term (if i-k ≥ 0).

For the max, if multiple k* achieve the max, the subgradient is the average (or any convex combination).

This gives us a projected subgradient method. But since the problem is non-convex, this might get stuck.

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0

            hi = mid
        else:

            mask = y > 1e-12
            cnt = mask.sum()
            if cnt > 0:
                y[mask] = np.maximum(y[mask] + r / cnt, 0.0)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def corr_vec(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def get_active_lags(h, gap=0.05, maxlen=30):
    v = corr_vec(h)
    n = len(h)
    vmax = v.max()
    idx = np.where(v >= vmax - gap * vmax)[0]
    if len(idx) > maxlen:
        # Keep the top maxlen
        order = np.argsort(v[idx])
        idx = idx[order[-maxlen:]]
    lags = sorted(set(int(k - (n-1)) for k in idx))
    return lags, float(vmax)

def golden_pair(h, i, j, max_iter=40):
    """Minimize C5 over t = h[i], h[j] = s-t. Returns (new_c, best_t)."""
    n = len(h)
    s = h[i] + h[j]
    lo = max(0.0, s - 1.0)
    hi = min(1.0, s)
    if hi - lo < 1e-12:
        return evaluate_c5(h), h[i]
    
    gr = (np.sqrt(5) - 1) / 2
    h_save = h.copy()
    
    def f(t):
        h[i] = t
        h[j] = s - t
        return evaluate_c5(h)
    
    a, b = lo, hi
    c = b - gr * (b - a)
    d = a + gr * (b - a)
    fc = f(c)
    fd = f(d)
    
    for _ in range(max_iter):
        if b - a < 1e-9:
            break
        if fc < fd:
            b = d
            d = c
            fd = fc
            c = b - gr * (b - a)
            fc = f(c)
        else:
            a = c
            c = d
            fc = fd
            d = a + gr * (b - a)
            fd = f(d)
    
    t_opt = (a + b) / 2
    h[i] = t_opt
    h[j] = s - t_opt
    f_opt = evaluate_c5(h)
    
    # Restore if no improvement
    # (Caller decides whether to keep)
    return f_opt, t_opt

def pair_descent(h, lags, deadline, max_passes=10):
    """Coordinate descent over pairs at active lag distances."""
    n = len(h)
    best_h = h.copy()
    best_c = evaluate_c5(h)
    
    for p in range(max_passes):
        if time.time() > deadline:
            break
        improved = False
        for lag in lags:
            dist = abs(lag)
            if dist == 0 or dist >= n:
                continue
            for i in range(n - dist):
                if time.time() > deadline:
                    return best_h, best_c
                j = i + dist
                s = best_h[i] + best_h[j]
                lo = max(0.0, s - 1.0)
                hi = min(1.0, s)
                if hi - lo < 1e-8:
                    continue
                # Try golden section
                c_new, t_opt = golden_pair(best_h, i, j, max_iter=30)
                if c_new < best_c - 1e-10:
                    best_c = c_new
                    improved = True
        if not improved:
            break
    return best_h, best_c

def perturbation(h, rng, magnitude=0.05):
    """Random perturbation preserving sum."""
    n = len(h)
    k = rng.randint(3, min(15, n))
    idx = rng.choice(n, size=k, replace=False)
    delta = rng.randn(k) * magnitude
    delta -= delta.mean()
    h_new = h.copy()
    h_new[idx] += delta
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n / 2.0)
    return h_new

def transfer_res(h, n_new):
    """Transfer to new resolution."""
    n_old = len(h)
    x_old = np.arange(n_old, dtype=float)
    x_new = np.linspace(0, n_old - 1, n_new)
    h_new = np.interp(x_new, x_old, h)
    h_new = project_sum_box(np.clip(h_new, 0, 1), n_new / 2.0)
    return h_new
