    v = A[None, :] + deltas[:, None] * L[None, :] + d2[:, None] * Q[None, :]
    F = np.max(v, axis=1)
    bi = int(np.argmin(F))
    if 0 < bi < nc - 1:
        lo = deltas[max(0, bi - 3)]
        hi = deltas[min(nc - 1, bi + 3)]
    else:
        lo, hi = dmin, dmax
    fd = np.linspace(lo, hi, nf)
    fd2 = fd * fd
    v2 = A[None, :] + fd[:, None] * L[None, :] + fd2[:, None] * Q[None, :]
    F2 = np.max(v2, axis=1)
    fi = int(np.argmin(F2))
    return float(fd[fi]), float(F2[fi])


def wgrad(h, w, dx, N):
    g = np.zeros(N)
    omh = 1.0 - h
    half = N - 1
    for k in range(2 * N - 1):
        wk = w[k]
        if wk < 1e-10:
            continue
        d = k - half
        ww = wk * dx
        if d >= 0:
            n = N - d
            if n > 0:
                g[:n] += ww * omh[d:]
                g[d:] += -ww * h[:n]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[dd:] += ww * omh[:n]
                g[:n] += -ww * h[dd:]
    return g


def try_direction(h, a_dir, c, dx, N):
    dmin, dmax = get_feasible_range(h, a_dir)
    if dmax - dmin < 1e-12:
        return None, None
    L = (np.correlate(a_dir, 1.0 - h, mode='full') - np.correlate(h, a_dir, mode='full')) * dx
    Q = -np.correlate(a_dir, a_dir, mode='full') * dx
    delta, F = ls_quadratic(c, L, Q, dmin, dmax)
    return delta, F


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 12.0

    h = np.array(initial_h_values, dtype=float)
    N = len(h)
    tsum = N / 2.0
    h = project_feasible(h, tsum)

    best_h = h.copy()
    best_c5 = compute_c5(h)
    dx = 2.0 / N
    rng = np.random.default_rng(seed)

    tau = 0.008
    stalled = 0
    it = 0

    while time.time() < deadline:
        it += 1

        # Current correlation values
        omh = 1.0 - h
        c = np.correlate(h, omh, mode='full') * dx
        m = float(np.max(c))

        # Softmax weights for direction finding
        exps = np.exp((c - m) / tau)
        wsum = exps.sum()
        if wsum > 1e-30:
            w = exps / wsum
        else:
            w = np.zeros(2 * N - 1)
            w[int(np.argmax(c))] = 1.0

        # Weighted gradient
        g = wgrad(h, w, dx, N)
        g_mean = np.mean(g)

        # Build candidate directions
        # Dir 1: negative gradient (zero-sum)
        a1 = -(g - g_mean)
        nrm1 = np.linalg.norm(a1)
        if nrm1 > 1e-10:
            a1 /= nrm1
        else:
            a1 = None

        # Dir 2: Frank-Wolfe (steepest fractional step)
        imin = int(np.argmin(g))
        imax = int(np.argmax(g))
        if imin != imax:
            a2 = np.zeros(N)
            a2[imin] = 1.0
            a2[imax] = -1.0
        else:
            a2 = None

        # Dir 3: top-2 FW (if different from Dir 2)
        g_sorted = np.sort(g)
        a3 = None
        if len(g_sorted) >= 4 and g_sorted[-1] - g_sorted[0] > 1e-10:
            i_lo = int(np.argmin(g))
            # Find second highest
            g2 = g.copy()
            g2[int(np.argmax(g))] = -np.inf
            i_hi2 = int(np.argmax(g2))
            g2 = g.copy()
            g2[i_lo] = np.inf
            i_lo2 = int(np.argmin(g2))
            a3 = np.zeros(N)
            a3[i_lo] = 0.5
            a3[i_lo2] = 0.5
            a3[int(np.argmax(g))] = -0.5
            a3[i_hi2] = -0.5

        # Dir 4: random (if stalled)
        a4 = None
        if stalled > 10:
            a4 = rng.normal(0, 1, N)
            a4 -= np.mean(a4)
            nr = np.linalg.norm(a4)
            if nr > 1e-10:
                a4 /= nr

        # Try all directions, keep best
        best_delta = 0.0
        best_F = m
        best_a = None

        for a_dir in [a1, a2, a3, a4]:
            if a_dir is None:
                continue
            delta, F = try_direction(h, a_dir, c, dx, N)
            if F is not None and F < best_F:
                best_F = F
                best_delta = delta
                best_a = a_dir

        # Accept if improved
        if best_a is not None and best_F < best_c5 - 5e-13:
            h_new = np.clip(h + best_delta * best_a, 0.0, 1.0)
            h_new = project_feasible(h_new, tsum)
            c5_new = compute_c5(h_new)
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                h = h_new
                stalled = 0
            else:
                stalled += 1
        else:
            stalled += 1

        # Adaptive strategies
        if stalled % 20 == 19:
            # Decrease tau to focus on hottest constraints
            tau = max(tau * 0.75, 0.0003)

        if stalled >= 50:
            # Increase tau for broader view + perturbation
            tau = 0.015
            stalled = 0
            scale = 0.03 + 0.04 * (stalled % 3)
            h_pert = h + rng.normal(0, scale, N)
            h_pert = project_feasible(h_pert, tsum)
            c5_pert = compute_c5(h_pert)
            h = h_pert
            if c5_pert < best_c5:
                best_h = h_pert.copy()
                best_c5 = c5_pert

        # Periodic random exploration
        if it % 100 == 99:
            for _ in range(8):
                if time.time() > deadline - 5:
                    break
                ar = rng.normal(0, 1, N)
                ar -= np.mean(ar)
                nr = np.linalg.norm(ar)
                if nr < 1e-10:
                    continue
                ar /= nr
                delta_r, F_r = try_direction(h, ar, c, dx, N)
                if F_r is not None and F_r < best_c5 - 5e-13:
                    h_try = np.clip(h + delta_r * ar, 0.0, 1.0)
                    h_try = project_feasible(h_try, tsum)
                    c5_try = compute_c5(h_try)
                    if c5_try < best_c5:
                        best_h = h_try.copy()
                        best_c5 = c5_try
                        h = h_try
                        stalled = 0

        # Every 500 iterations, try concentrated moves on hot constraints
        if it % 500 == 499 and stalled < 30:
            # Find top-3 hottest shifts
            top3 = np.argsort(c)[-3:]
            for k_hot in top3:
                d = k_hot - (N - 1)
                # Compute gradient of just this constraint
                gh = np.zeros(N)
                if d >= 0:
                    n = N - d
                    if n > 0:
                        gh[:n] = dx * (1.0 - h[d:])
                        gh[d:] = -dx * h[:n]
                else:
                    dd = -d
                    n = N - dd
                    if n > 0:
                        gh[dd:] = dx * (1.0 - h[:n])
                        gh[:n] = -dx * h[dd:]
                # Zero-sum projection
                a_hot = -(gh - np.mean(gh))
                nh = np.linalg.norm(a_hot)
                if nh > 1e-10:
                    a_hot /= nh
                    delta_h, F_h = try_direction(h, a_hot, c, dx, N)
                    if F_h is not None and F_h < best_c5 - 5e-13:
                        h_try = np.clip(h + delta_h * a_hot, 0.0, 1.0)
                        h_try = project_feasible(h_try, tsum)
                        c5_try = compute_c5(h_try)
                        if c5_try < best_c5:
                            best_h = h_try.copy()
                            best_c5 = c5_try
                            h = h_try
                            stalled = 0

    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, N
