                else:
                    # direction: increase i, decrease j
                    best_alpha = 0.0
                    best_c = compute_c5(h)
                    for trial in range(5):
                        alpha = max_move * (1.0 - trial / 5.0)
                        if alpha < 1e-10:
                            break
                        h_try = h.copy()
                        h_try[i] += alpha
                        h_try[j] -= alpha
                        c_try = compute_c5(h_try)
                        if c_try < best_c:
                            best_c = c_try
                            best_alpha = alpha
                    if best_alpha > 1e-10:
                        h[i] += best_alpha
                        h[j] -= best_alpha
                        h = np.clip(h, 0.0, 1.0)
                        if best_c < best_c5 - 1e-14:
                            best_c5 = best_c
                            best_h = h.copy()
                        found_improve = True
                        break
        if not found_improve:
            break

    return best_h, best_c5

def interpolate_h(h_old, n_new):
    n_old = len(h_old)
    x_old = np.arange(n_old) / n_old * 2.0
    x_new = np.arange(n_new) / n_new * 2.0
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.4)
    h += 0.06 * np.sin(6.0 * np.pi * x + 1.1)
    h += 0.03 * np.sin(8.0 * np.pi * x + 2.0)
    h += 0.02 * np.sin(12.0 * np.pi * x + 3.1)
    h += rng.randn(n) * 0.015
    return project_constraint(h, n)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    np.random.seed(seed)
    deadline = t_start + budget_s
    reserve = max(20, 0.03 * budget_s)

    best_h = None
    best_c5 = float('inf')
    best_n = 0

    def try_update(h_cand, c5_cand):
        nonlocal best_h, best_c5, best_n
        n_c = len(h_cand)
        if not np.all(np.isfinite(h_cand)):
            return
        if np.min(h_cand) < -1e-6 or np.max(h_cand) > 1.0 + 1e-6:
            return
        if c5_cand < best_c5 - 1e-12:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            best_n = n_c
        elif abs(c5_cand - best_c5) <= 1e-12 and n_c < best_n:
            best_h = h_cand.copy()
            best_c5 = c5_cand
            best_n = n_c

    # Phase 0: Get initial
    n_init = 81
    try:
        h0 = np.array(initial_h_values, dtype=float).copy()
        n_init = len(h0)
        if n_init < 2 or not np.all(np.isfinite(h0)):
            h0 = make_start(n_init, seed)
    except (NameError, UnboundLocalError):
        h0 = make_start(n_init, seed)
    h0 = project_constraint(h0, n_init)
    c5_0 = compute_c5(h0)
    try_update(h0, c5_0)

    # Phase 1: Main smooth descent from initial
    elapsed = time.time() - t_start
    alloc = min(budget_s - elapsed - 200, 350)
    if alloc > 30:
        h_opt, c5_opt = smooth_descent(h0.copy(), alloc, tau_start=5.0, tau_end=500.0, num_passes=5)
        try_update(h_opt, c5_opt)

    # Phase 2: Coordinate descent on best
    elapsed = time.time() - t_start
    if best_h is not None and budget_s - elapsed > 80:
        alloc_cd = min(budget_s - elapsed - 100, 60)
        if alloc_cd > 20:
            h_cd, c5_cd = coordinate_descent(best_h.copy(), alloc_cd)
            try_update(h_cd, c5_cd)

    # Phase 3: Try different n values
    n_list = [n_init]
    for dn in [10, 20, 40, 60]:
        nc = n_init + dn
        if nc <= 201:
            n_list.append(nc)

    for n_try in n_list:
        elapsed = time.time() - t_start
        remaining = budget_s - elapsed - reserve - 30
        if remaining < 40:
            break
        if n_try == n_init:
            h_start = best_h
        else:
            h_start = interpolate_h(best_h, n_try)
            h_start = project_constraint(h_start, n_try)
            c5_start = compute_c5(h_start)
            if c5_start > best_c5 + 0.01:
                h_start = make_start(n_try, seed + n_try)
        alloc_sm = min(remaining // 2, 80)
        if alloc_sm > 15:
            h_opt, c5_opt = smooth_descent(h_start, alloc_sm, tau_start=10.0, tau_end=400.0, num_passes=3)
            try_update(h_opt, c5_opt)
        # Quick coord descent
        elapsed = time.time() - t_start
        if budget_s - elapsed > reserve + 20:
            h_cd, c5_cd = coordinate_descent(h_opt, min(15, budget_s - elapsed - reserve - 10))
            try_update(h_cd, c5_cd)

    # Phase 4: Re-optimization from best with different parameters
    elapsed = time.time() - t_start
    if best_h is not None and budget_s - elapsed > 80:
        rng_r = np.random.RandomState(seed + 7777)
        h_pert = best_h + rng_r.randn(len(best_h)) * 0.015
        h_pert = project_constraint(h_pert, len(best_h))
        alloc2 = min(budget_s - elapsed - reserve - 30, 120)
        if alloc2 > 30:
            h_opt, c5_opt = smooth_descent(h_pert, alloc2, tau_start=20.0, tau_end=800.0, num_passes=4)
            try_update(h_opt, c5_opt)
            h_cd, c5_cd = coordinate_descent(h_opt, min(20, budget_s - time.time() - reserve - 10))
            try_update(h_cd, c5_cd)

    # Phase 5: Final pair sweep on best
    elapsed = time.time() - t_start
    if best_h is not None and budget_s - elapsed > 30:
        alloc_ps = min(budget_s - elapsed - reserve, 30)
        if alloc_ps > 10:
            h_ps, c5_ps = pair_coord_sweep(best_h.copy(), alloc_ps)
            try_update(h_ps, c5_ps)

    # Final: ensure feasibility
    if best_h is not None:
        best_h = project_constraint(best_h, len(best_h))
        best_c5 = compute_c5(best_h)

    if best_h is None:
        best_h = np.full(n_init, 0.5)
        best_c5 = 0.5

    return best_h, best_c5, len(best_h)
