import time
from scipy.optimize import minimize

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj_and_weights(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    w = w / total
    obj = m + tau * np.log(total)
    return obj, w

def smooth_grad(h, weights):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = weights[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    return g

def slsqp_f(h, tau):
    obj, _ = smooth_obj_and_weights(h, tau)
    return obj

def slsqp_g(h, tau):
    _, w = smooth_obj_and_weights(h, tau)
    return smooth_grad(h, w)

def sum_eq(h, target):
    return np.sum(h) - target

def sum_eq_jac(h, target):
    return np.ones_like(h)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h

def upscale_func(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)

def compute_sensitivity(h):
    N = len(h)
    dx = 2.0 / N
    S = np.zeros((N, 2*N-1))
    for r in range(2*N-1):
        d = r - (N-1)
        j_lo1 = max(0, -d)
        j_hi1 = min(N, N-d)
        if j_hi1 > j_lo1:
            S[j_lo1:j_hi1, r] += (1.0 - h[j_lo1+d:j_hi1+d])
        j_lo2 = max(0, d)
        j_hi2 = min(N, N+d)
        if j_hi2 > j_lo2:
            S[j_lo2:j_hi2, r] -= h[j_lo2-d:j_hi2-d]
    S *= dx
    return S

def pair_swap_local_search(h, rng, n_sweeps=100, deadline=None):
    N = len(h)
    c5 = compute_c5(h)
    for sweep in range(n_sweeps):
        if deadline and time.time() > deadline:
            break
        S = compute_sensitivity(h)
        C = np.correlate(h, 1.0 - h, 'full') * (2.0 / N)
        current_max = C.max()
        improved = False
        n_trials = 200
        for _ in range(n_trials):
            i, j = rng.choice(N, 2, replace=False)
            d_lo = max(-h[i], h[j] - 1.0)
            d_hi = min(1.0 - h[i], h[j])
            if d_lo >= d_hi:
                continue
            diff = S[i] - S[j]
            best_delta = 0.0
            best_val = current_max
            # Try several fractions
            for frac in [0.1, 0.25, 0.5, 0.75, 0.9]:
                delta = d_lo + frac * (d_hi - d_lo)
                new_max = (C + delta * diff).max()
                if new_max < best_val - 1e-12:
                    best_val = new_max
                    best_delta = delta
            if best_delta != 0.0:
                h[i] += best_delta
                h[j] -= best_delta
                C = C + best_delta * diff
                current_max = C.max()
                improved = True
        if not improved:
            break
    c5_final = compute_c5(h)
    return h, c5_final

Wait, I realize there's an issue with using stale C and S in the pair swap. When I update h[i] and h[j], the sensitivity S changes for ALL positions, and C changes too. But I'm only updating C correctly (C + delta*diff gives the new C for the old S, but S itself has changed).

Actually, C + delta * (S[i] - S[j]) IS the correct new C, because the change in C due to changing h[i] by delta and h[j] by -delta is exactly delta * (S[i] - S[j]) where S is the OLD sensitivity. This is because the sensitivity is the derivative, and for a small change, the linear approximation is exact for the correlation (which is bilinear in h).

Wait, is the correlation bilinear in h? C[d] = sum_n h[n]*(1-h[n+d]). If I change h[i] by delta, the change in C[d] is:
- h[i] appears in h[n] at n=i: change is delta*(1-h[i+d])
- h[i] appears in h[n+d] at n=i-d: change is -h[i-d]*delta (since d(1-h)/dh = -1)

So the change is delta * [(1-h[i+d]) - h[i-d]] (with validity checks).

This is EXACT (not just linear approximation) because C[d] is linear in each h[j] when all other h values are held fixed. So C + delta*(S[i]-S[j]) is EXACTLY the new C!

But the sensitivity S itself changes because it depends on h. So for the NEXT move, I should use the updated S. But within a single sweep, using the old S is an approximation.

For the first move in a sweep, S is exact. After the first move, S is slightly stale. But for small moves, the error is second-order, so it should be fine.
