import time

def evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_box_sum(x, target):
    # Project x to [0,1]^n with sum = target
    x = np.clip(np.nan_to_num(x), 0.0, 1.0)
    # Binary search for lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Fix residual
    res = h.sum() - target
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
                if abs(res) < 1e-14:
                    break
        else:
            order = np.argsort(h)
            for idx in order:
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
                if abs(res) < 1e-14:
                    break
    return h

def to_binary(h, n):

    # Sort by value, take top target as 1s
    order = np.argsort(h)[::-1]
    b = np.zeros(n)
    b[order[:target]] = 1.0
    return b

def binary_optimize(h, max_sweeps=50, time_limit=30):

    n = len(h)
    start_time = time.time()
    h = h.copy()
    
    for sweep in range(max_sweeps):
        if time.time() - start_time > time_limit:
            break
        improved = False
        ones = np.where(h == 1.0)[0]
        zeros = np.where(h == 0.0)[0]
        best_gain = 0
        best_pair = None
        cur_max = np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n)

                h[j] = 1.0
                new_max = np.max(np.correlate(h, 1.0-h, mode='full')) * (2.0/n)
                gain = cur_max - new_max
                if gain > best_gain:
                    best_gain = gain
                    best_pair = (i, j)
                h[i] = 1.0
                h[j] = 0.0
        
        if best_pair is not None and best_gain > 1e-12:
            i, j = best_pair
            h[i] = 0.0
            h[j] = 1.0
            improved = True
            cur_max -= best_gain

    return h
