import numpy as np
import time

def project_sum_box(x, target):
    n = len(x)
    x = np.asarray(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-10:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-12:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-12:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    r2 = target - y.sum()
    if abs(r2) > 1e-12:
        if r2 > 0:
            idx = np.where(y < 1.0 - 1e-10)[0]
        else:
            idx = np.where(y > 1e-10)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(res))

def hill_climb(h, deadline, min_diff=0.01):
    n = len(h)
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    h = h.copy()
    stall = 0
    while time.time() < deadline:
        improved = False
        order = np.argsort(h)
        for a in range(n):
            i = order[a]
            for b in range(a + 1, n):
                j = order[b]
                diff = h[j] - h[i]
                if diff < min_diff:
                    break
                delta = diff * 0.5
                h[i] += delta
                h[j] -= delta
                c_new = evaluate_c5(h)
                if c_new < best_c - 1e-14:
                    best_c = c_new
                    best_h = h.copy()
                    improved = True
                    break
                else:
                    h[i] -= delta
                    h[j] += delta
            if improved:
                break
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c

def hill_climb_multi_frac(h, deadline, min_diff=0.015):
    n = len(h)
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    h = h.copy()
    stall = 0
    fracs = np.array([0.3, 0.5, 0.7, 0.9])
    while time.time() < deadline:
        improved = False
        order = np.argsort(h)
        for a in range(n):
            i = order[a]
            for b in range(a + 1, n):
                j = order[b]
                diff = h[j] - h[i]
                if diff < min_diff:
                    break
                best_d = None
                best_lc = best_c
                for fr in fracs:
                    d = diff * fr
                    h[i] += d
                    h[j] -= d
                    c_try = evaluate_c5(h)
                    h[i] -= d
                    h[j] += d
                    if c_try < best_lc:
                        best_lc = c_try
                        best_d = d
                if best_d is not None:
                    h[i] += best_d
                    h[j] -= best_d
                    best_c = best_lc
                    best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        if not improved:
            stall += 1
            if stall >= 3:
                break
        else:
            stall = 0
    return best_h, best_c

def anneal(h, deadline, seed=0, T0=0.004):
    n = len(h)
    rng = np.random.RandomState(seed)
    best_h = h.copy()
    best_c = evaluate_c5(best_h)
    h = h.copy()
    c_cur = best_c
    start_t = time.time()
    total_dur = max(deadline - start_t, 1.0)
    while time.time() < deadline:
        elapsed = time.time() - start_t
        frac = min(elapsed / total_dur, 1.0)
        T = T0 * (1.0 - frac) ** 2.5
        if T < 1e-9:
            T = 1e-9
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        if rng.random() < 0.5:
            src, dst = i, j
        else:
            src, dst = j, i
        max_d = min(h[src], 1.0 - h[dst])
        if max_d < 0.003:
            continue
        delta = rng.uniform(0.003, max_d)
        h[src] -= delta
        h[dst] += delta
        c_new = evaluate_c5(h)
        dC = c_new - c_cur
        if dC <= 0 or rng.random() < np.exp(-dC / T):
            c_cur = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
        else:
            h[src] += delta
            h[dst] -= delta
    return best_h, best_c

def transfer(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    h_new = np.zeros(n_new)
    for i in range(n_new):
        pos = i * n_old / n_new
        i0 = int(pos)
        frac = pos - i0
        i1 = (i0 + 1) % n_old
        h_new[i] = (1.0 - frac) * h[i0 % n_old] + frac * h[i1]
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def structured_perturb(h, rng, scale=0.15):
    n = len(h)
    h_new = h.copy()
    n_shifts = rng.randint(2, 6)
    for _ in range(n_shifts):
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        d = (h[j] - h[i]) * rng.uniform(0.3, 1.0) * 0.5
        h_new[i] += d
        h_new[j] -= d
    h_new = project_sum_box(h_new, n / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:

            n = len(h_try)

                h_try = project_sum_box(h_try, n / 2.0)
                c = evaluate_c5(h_try)
                if np.isfinite(c):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n0 = len(best_h)

    # Phase 1: Quick hill climb at original resolution (15% time)
    t1 = start + budget_s * 0.15

    if c < best_c:

    # Multi-frac refinement

    h, c = hill_climb_multi_frac(best_h, t1b, min_diff=0.01)
    if c < best_c:

    # Phase 2: Annealing with multiple restarts (45% time)
    t2 = start + budget_s * 0.70

        anneal_dl = time.time() + anneal_dur

        elif c_hc < best_c:
