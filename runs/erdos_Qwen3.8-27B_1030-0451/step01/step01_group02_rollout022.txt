def binary_sa(b, target_c, n_steps, t_start, t_end, rng):
    """Simulated annealing on binary sequence b (sum = n/2).
    target_c: current best C5 value we want to beat (for reference)
    """
    n = len(b)
    dx = 2.0 / n
    b_work = b.copy()
    C = np.correlate(b_work, 1 - b_work, "full")
    max_c = float(C.max())
    best_b = b_work.copy()
    best_max_c = max_c
    
    ones = np.where(b_work == 1)[0]
    zeros = np.where(b_work == 0)[0]
    
    for step in range(n_steps):
        # Temperature schedule
        t = t_start * (t_end / t_start) ** (step / max(n_steps - 1, 1))
        
        # Pick random one and zero
        pi = rng.integers(len(ones))
        qi = rng.integers(len(zeros))
        p = ones[pi]
        q = zeros[qi]
        
        if p == q:
            continue
        
        # Swap
        b_work[p] = 0
        b_work[q] = 1
        # Update ones/zeros
        ones[pi] = q
        zeros[qi] = p
        # (multiple of same value possible, that's fine)
        
        C_new = np.correlate(b_work, 1 - b_work, "full")
        max_new = float(C_new.max())
        
        delta = max_new - max_c
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-10)):
            # Accept
            max_c = max_new
            C = C_new
            if max_new < best_max_c - 1e-12:
                best_max_c = max_new
                best_b = b_work.copy()
        else:
            # Reject - undo
            b_work[p] = 1
            b_work[q] = 0
            ones[pi] = p
            zeros[qi] = q
    
    return best_b, best_max_c * dx

def make_feasible(x, target):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    target = float(target)
    x = np.clip(x, 0.0, 1.0)
    # Binary search for additive shift
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    if diff > 0:
        lo, hi = 0.0, max(1.0 - float(x.min()), 0.0)
    else:
        lo, hi = min(float(x.max()) - 1.0, 0.0), 0.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    # Fine adjustment
    diff = target - float(x.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size:
                order = idx[np.argsort(x[idx])]
                rem = diff
                for i in order:
                    inc = min(float(1.0 - x[i]), rem)
                    x[i] += inc
                    rem -= inc
                    if rem <= 1e-14:

        else:

        delta = new_max - cur_max
        if delta < 0 or rng.random() < np.exp(-delta / max(t, 1e-12)):
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a_work.copy()
        else:
            a_work[p] = 1
            a_work[q] = 0
            ones[pi] = p
            zeros[qi] = q
    
    return best_a, best_max

def resample_to(h, N):
    """Resample h to N points."""
    n0 = len(h)
    if n0 == N:
        return make_feasible(h, 0.5 * N)
    x_old = (np.arange(n0) + 0.5) * (2.0 / n0)
    x_new = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(x_new, x_old, h)
    return make_feasible(y, 0.5 * N)
