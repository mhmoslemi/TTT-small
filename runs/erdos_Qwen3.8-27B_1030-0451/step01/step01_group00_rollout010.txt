def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def smooth_gradient(h, corr, tau):
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

def topk_gradient(h, corr, k, tau):
    """Gradient focusing on top-k correlations with log-sum-exp smoothing."""
    n = len(h)
    m = len(corr)
    # Get top k indices
    top_idx = np.argsort(corr)[-k:]
    # Compute weights within top k using log-sum-exp
    top_vals = corr[top_idx]
    cs = top_vals - np.max(top_vals)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for i, k_idx in enumerate(top_idx):
        wk = w[i]
        if wk < 1e-14:
            continue
        lag = k_idx - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

def frank_wolfe_vertex(grad, n):
    """Find the vertex of {h: 0<=h<=1, sum(h)=n/2} minimizing grad·h."""
    target = n / 2.0
    sorted_idx = np.argsort(grad)
    s = np.zeros(n)
    k = int(target)
    s[sorted_idx[:k]] = 1.0
    frac = target - k
    if frac > 1e-12:

    return s

import numpy as np

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def smooth_gradient(h, corr, tau):
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

def topk_gradient(h, corr, k, tau):
    n = len(h)
    m = len(corr)
    top_idx = np.argsort(corr)[-k:]
    top_vals = corr[top_idx]
    cs = top_vals - np.max(top_vals)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)

def fw_vertex(grad, n):
    target = n / 2.0

    s = np.zeros(n)
    k = int(target)
    s[sorted_idx[:k]] = 1.0
    frac = target - k
    if frac > 1e-12:

    return s
