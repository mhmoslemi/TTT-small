import time

def corr_values(h):
    """Compute all C_k values (aligned with np.correlate(h, 1-h, 'full'))"""
    n = len(h)
    dx = 2.0 / n
    # np.correlate(h, 1-h, 'full') gives 2n-1 values
    return np.correlate(h, 1.0 - h, 'full') * dx

def exact_C5(h):
    n = len(h)
    if n == 0:
        return 0.0
    c = np.correlate(h, 1.0 - h, 'full') * (2.0 / n)
    return float(np.max(c))

def make_feasible(h, target):
    """Force h to satisfy 0 <= h <= 1 and sum(h) = target"""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to add 'diff' total
        # Add proportionally to (1-h)
        room = 1.0 - h
        if room.sum() < 1e-15:
            return h
        h += diff * room / room.sum()
    else:
        # Need to subtract |diff| total
        if h.sum() < 1e-15:
            return h
        h += diff * h / h.sum()
    h = np.clip(h, 0.0, 1.0)
    # Fix any remaining sum error
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                i = idx[np.argmin(h[idx])]
                h[i] = min(1.0, h[i] + diff)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                i = idx[np.argmax(h[idx])]
                h[i] = max(0.0, h[i] + diff)
    return h

def make_feasible(h, target):
    """Ensure 0<=h<=1 and sum(h)=target"""
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h

        else:
            # All at 1, can't fix
            pass
    else:

    h = np.clip(h, 0.0, 1.0)

    diff = target - h.sum()
    if abs(diff) > 1e-12:

        if sign > 0:

```python
import numpy as np
import time

def compute_C(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx

def make_feasible(h, target):
    h = np.clip(h.astype(float).copy(), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        room = 1.0 - h
        if room.sum() > 1e-14:
            h += diff * room / room.sum()
    else:
        if h.sum() > 1e-14:
            h += diff * h / h.sum()
    h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        for _ in range(20):
            diff = target - h.sum()
            if abs(diff) < 1e-14:
                break
            if diff > 0:
                idx = np.where(h < 1.0)[0]
                if len(idx) == 0: break
                i = idx[np.argmin(h[idx])]
                h[i] = min(1.0, h[i] + diff)
            else:
                idx = np.where(h > 0.0)[0]
                if len(idx) == 0: break
                i = idx[np.argmax(h[idx])]
                h[i] = max(0.0, h[i] + diff)
    return h
```

    L = np.zeros(2*n-1)
