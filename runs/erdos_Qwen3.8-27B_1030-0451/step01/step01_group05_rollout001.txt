import numpy as np
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_lin_quad(h, a, b, n):
    """Compute lin and quad arrays for pair (a,b) transfer."""
    L = 2 * n - 1
    center = n - 1
    lin = np.zeros(L)
    quad = np.zeros(L)
    
    # For each K, shift = K - center
    # lin[K] = -h[a+shift] (if valid) + (1-h[a-shift]) (if valid) 
    #          + h[b+shift] (if valid) - (1-h[b-shift]) (if valid)
    
    # Efficient computation using vectorized approach
    shifts = np.arange(L) - center  # shape (L,)
    
    # For coordinate a (increased by d):
    # -h[a+shift] where 0 <= a+shift < n
    idx_a_plus = a + shifts
    valid_a_plus = (idx_a_plus >= 0) & (idx_a_plus < n)
    lin[valid_a_plus] -= h[idx_a_plus[valid_a_plus]]
    
    # (1-h[a-shift]) where 0 <= a-shift < n
    idx_a_minus = a - shifts
    valid_a_minus = (idx_a_minus >= 0) & (idx_a_minus < n)
    lin[valid_a_minus] += (1.0 - h[idx_a_minus[valid_a_minus]])
    
    # For coordinate b (decreased by d):
    # +h[b+shift] where 0 <= b+shift < n
    idx_b_plus = b + shifts
    valid_b_plus = (idx_b_plus >= 0) & (idx_b_plus < n)
    lin[valid_b_plus] += h[idx_b_plus[valid_b_plus]]
    
    # -(1-h[b-shift]) where 0 <= b-shift < n
    idx_b_minus = b - shifts
    valid_b_minus = (idx_b_minus >= 0) & (idx_b_minus < n)
    lin[valid_b_minus] -= (1.0 - h[idx_b_minus[valid_b_minus]])
    
    # Quadratic terms
    quad[center] += -2.0  # self terms for both a and b at shift=0
    
    # Cross terms
    K_ab = center + (a - b)
    if 0 <= K_ab < L:
        quad[K_ab] += 1.0
    K_ba = center + (b - a)
    if 0 <= K_ba < L:
        quad[K_ba] += 1.0
    
    return lin, quad

def line_search(raw, lin, quad, d_lo, d_hi, dx, n_grid=30):
    """Find d in [d_lo, d_hi] minimizing dx * max(raw + d*lin + d^2*quad)."""
    if d_hi - d_lo < 1e-14:
        return 0.0
    
    # Grid search
    d_vals = np.linspace(d_lo, d_hi, n_grid)
    best_d = 0.0
    best_score = np.inf
    
    for d in d_vals:
        new_raw = raw + d * lin + d * d * quad
        s = np.max(new_raw) * dx
        if s < best_score:
            best_score = s
            best_d = d
    
    # Local refinement around best
    d_range = (d_hi - d_lo) / n_grid
    d_center = best_d
    d_lo2 = max(d_lo, d_center - d_range)
    d_hi2 = min(d_hi, d_center + d_range)
    if d_hi2 > d_lo2:
        d_vals2 = np.linspace(d_lo2, d_hi2, 20)
        for d in d_vals2:
            new_raw = raw + d * lin + d * d * quad
            s = np.max(new_raw) * dx
            if s < best_score:
                best_score = s
                best_d = d
    
    return best_d

def pair_gradient_score(h, tau):
    """Compute a gradient-like score for selecting pairs."""
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    
    # Smooth max gradient approximation
    exponents = np.clip((c - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        weights = np.ones_like(weights) / len(weights)
    else:
        weights /= sw
    
    # Compute gradient: for each coordinate i, sum over K of weight_K * col_i(K)
    # col_i(K) for h[i] (increasing h[i]):
    #   + (1-h[j]) where j = i + shift, valid
    #   - h[j] where j = i - shift, valid
    # But since we're doing transfer, the "gradient" for increasing h[i] is:
    #   sum_K w_K * [(1 - h[i+shift]) if valid - h[i-shift] if valid]
    
    grad = np.zeros(n)
    center = n - 1
    
    for K in range(len(weights)):
        w = weights[K]
        if w < 1e-12:
            continue
        shift = K - center
        if shift >= 0:
            # h[i+shift]*(1-h[i]) for i in [0, n-shift)
            # d/dh[i] for increasing i: -(h[i+shift]) for the first part... 
            # Actually for transfer we want the sensitivity of the objective to h[i]
            # Increasing h[i] by delta changes:
            #   terms where i is the "second" factor: h[j]*(1-h[i]) with j=i+shift → -delta*h[j]
            #   terms where i is the "first" factor: h[i]*(1-h[j]) with j=i-shift → +delta*(1-h[j])
            if shift == 0:
                # h[i]*(1-h[i]), derivative = 1-2h[i]
                grad += w * (1.0 - 2.0 * h)
            else:
                # shift > 0: i in [0, n-shift) for h[i+shift]*(1-h[i])
                # i in [shift, n) for h[i]*(1-h[i-shift])
                grad[:n-shift] += w * (-h[shift:])  # from first type
                grad[shift:] += w * (1.0 - h[:n-shift])  # from second type
        else:
            m = -shift
            # h[i-m]*(1-h[i]) for i in [m, n)
            # h[i]*(1-h[i+m]) for i in [0, n-m)
            grad[m:] += w * (-h[:n-m])  # from first type (h[i-m]*(1-h[i]))
            grad[:n-m] += w * (1.0 - h[m:])  # from second type (h[i]*(1-h[i+m]))
    
    return grad

def exact_pair_move(h, a, b, raw, n, dx):
    """Make an exact pair move and return new h, new raw, new score."""
    L = 2 * n - 1
    center = n - 1
    
    # Compute domain
    d_lo = max(-h[a], h[b] - 1.0)
    d_hi = min(1.0 - h[a], h[b])

    lin, quad = compute_lin_quad(h, a, b, n)
    
    # Line search
    best_d = line_search(raw, lin, quad, d_lo, d_hi, dx)
    
    if abs(best_d) < 1e-14:
        return h, raw, np.max(raw) * dx
    
    # Apply move
    h_new = h.copy()
    h_new[a] += best_d
    h_new[b] -= best_d
    h_new = np.clip(h_new, 0.0, 1.0)
    
    # Compute new raw using formula
    raw_new = raw + best_d * lin + best_d * best_d * quad
    score = np.max(raw_new) * dx
    
    return h_new, raw_new, score

            for _ in range(npairs_random):
                a = int(rng.integers(0, n_opt))
                b = int(rng.integers(0, n_opt))
                if a != b:
                    candidates.append((a, b))
            if n_opt > 8:
                for _ in range(npairs_near):
                    a = int(rng.integers(0, n_opt))
                    off = int(rng.integers(1, max(2, n_opt // 5)))
                    b = (a + off) % n_opt
                    if a != b:
                        candidates.append((a, b))

            best_move = None
            best_move_score = cur_score
            best_move_d_delta = 0.0

            for (a, b) in candidates:
                result = try_pair(h, raw, a, b, n_opt, dx, n_grid)
                if result is None:
                    continue
                d, new_raw, new_score = result
                delta = new_score - cur_score
                if delta < -1e-12:
                    accept = True
                elif delta < temp and temp > 1e-8:
                    accept = (rng.random() < np.exp(-delta / temp))
                else:
                    accept = False

                if accept:
                    if new_score < best_move_score or (best_move is None):
                        best_move_score = new_score
                        best_move = (a, b, d, new_raw)
                        best_move_d_delta = delta

            if best_move is not None:
                a, b, d, new_raw = best_move
                h[a] = np.clip(h[a] + d, 0.0, 1.0)
                h[b] = np.clip(h[b] - d, 0.0, 1.0)
                raw = new_raw
                cur_score = np.max(raw) * dx
                no_improve = 0
                moves_count += 1
                if cur_score < best_score - 1e-13:
                    best_score = cur_score
                    best_h = h.copy()
            else:
                no_improve += 1

            if moves_count % recompute_every == 0:
                raw = exact_raw(h)
                cur_score = np.max(raw) * dx
                if np.max(np.abs(raw - exact_raw(h))) > 1e-8:
                    raw = exact_raw(h)

        return h, cur_score

    h_opt, s_opt = do_optimize(parent, n, deadline - 60.0)
    if s_opt < best_score:
        best_score = s_opt
        best_h = h_opt

    # Second pass with different parameters from improved solution
    if time.time() < deadline - 40.0:
        h_opt2, s_opt2 = do_optimize(best_h, n, deadline - 30.0)
        if s_opt2 < best_score:
            best_score = s_opt2
            best_h = h_opt2

    # Try different dimensions
    if time.time() < deadline - 60.0:
        dim_factors = [1.2, 0.85, 1.5, 0.7, 2.0]
        for factor in dim_factors:
            if time.time() > deadline - 30.0:
                break
            dn = int(round(n * factor))
            if dn < 30 or dn > 300 or dn == n:
                continue
            h_d = interpolate_h(best_h, dn)
            s_d = exact_score(h_d)
            if s_d < best_score + 0.02:
                h_d, s_d = do_optimize(h_d, dn, deadline - 25.0)
                if s_d < best_score - 1e-13:
                    best_score = s_d
                    best_h = h_d
                    n = dn

    # Final polish
    if time.time() < deadline - 15.0:
        h_final, s_final = do_optimize(best_h, len(best_h), deadline - 5.0)
        if s_final < best_score:
            best_score = s_final
            best_h = h_final

    best_h = project_feasible(best_h, len(best_h))
    final_score = exact_score(best_h)
    if final_score < best_score:
        best_score = final_score
    else:
        best_score = final_score

    return best_h, float(best_score), int(len(best_h))
