        # (since 1-s+hj = 1-hi-hj+hj = 1-hi, and 1-s+t - (1-hj) = t-s+hj = t-hi)
        a[idx[mask_not_i]] += dx * h[right_pos[mask_not_i]]
        b[idx[mask_not_i]] += -dx * h[right_pos[mask_not_i]] * hi

            quad[i_idx] += dx
