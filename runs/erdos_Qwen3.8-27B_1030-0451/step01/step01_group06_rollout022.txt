def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_box_sum(x, target):
    x = np.asarray(x, dtype=np.float64).flatten().copy()
    n = len(x)
    if n == 0:
        return x
    x[~np.isfinite(x)] = 0.5

        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:

    lam = 0.5 * (lo + hi)

    n = len(h)

    g = np.zeros(n)

            g[:n - d] += dx * wn * (1.0 - h[d:])
            g[d:] += -dx * wn * h[:n - d]
    return g

        return val

        _, w = lse_value_weights(x, tau)
        return gradient_weighted(x, w)

        return x.sum() - target

        return np.ones(n)

    return h
