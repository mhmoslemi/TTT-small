def optimize(h, deadline):
    n = len(h)
    target = n/2
    dx = 2.0/n
    
    # Adam state
    m = zeros(n)  # first moment
    v = zeros(n)  # second moment
    t = 0
    
    tau = 0.05  # initial smoothing
    lr = 0.1  # initial learning rate
    
    while time.time() < deadline:
        C = compute_corr(h) * dx
        M = max(C)
        alpha = softmax((C - M) / tau)
        
        grad = compute_grad(h, alpha)
        # Center gradient (projection onto sum=0 hyperplane)
        grad -= mean(grad)
        
        # Adam update
        t += 1
        m = 0.9*m + 0.1*grad
        v = 0.99*v + 0.01*grad^2
        m_hat = m / (1 - 0.9^t)
