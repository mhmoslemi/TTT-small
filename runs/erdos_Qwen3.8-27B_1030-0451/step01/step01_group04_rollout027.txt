def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h

def feasible_interval(h, a):
    """Find the range of delta such that 0 <= h + delta*a <= 1 for all i."""
    N = len(h)
    delta_lo = -np.inf
    delta_hi = np.inf
    
    pos = a > 1e-15
    neg = a < -1e-15
    
    if np.any(pos):
        # h[i] + delta*a[i] >= 0 => delta >= -h[i]/a[i]
        # h[i] + delta*a[i] <= 1 => delta <= (1-h[i])/a[i]
        delta_lo = max(delta_lo, np.max(-h[pos] / a[pos]))
        delta_hi = min(delta_hi, np.min((1.0 - h[pos]) / a[pos]))
    
    if np.any(neg):
        # h[i] + delta*a[i] >= 0 => delta <= -h[i]/a[i] (since a[i] < 0)
        # h[i] + delta*a[i] <= 1 => delta >= (1-h[i])/a[i] (since a[i] < 0)
        delta_lo = max(delta_lo, np.max((1.0 - h[neg]) / a[neg]))
        delta_hi = min(delta_hi, np.min(-h[neg] / a[neg]))
    
    # Add small margin
    delta_lo += 1e-12
    delta_hi -= 1e-12
    return delta_lo, delta_hi

    for i, d in enumerate(deltas):

    best_idx = np.argmin(vals)
    
    # Refine around best with golden section

                h[i] += add
                residual -= add
                if abs(residual) < 1e-12:
                    break
        else:
            idx = np.where(h > 1e-14)[0]
            for i in idx:
                sub = min(-residual, h[i])
                h[i] -= sub
                residual += sub
                if abs(residual) < 1e-12:
                    break
    return h


def feasible_interval(h, a):
    N = len(h)
    eps = 1e-15
    delta_lo = -1e10
    delta_hi = 1e10
    pos = a > eps
    neg = a < -eps
    if np.any(pos):
        delta_lo = max(delta_lo, float(np.max(-h[pos] / a[pos])))
        delta_hi = min(delta_hi, float(np.min((1.0 - h[pos]) / a[pos])))
    if np.any(neg):
        delta_lo = max(delta_lo, float(np.max((1.0 - h[neg]) / a[neg])))
        delta_hi = min(delta_hi, float(np.min(-h[neg] / a[neg])))
    if delta_lo < -1e10:
        delta_lo = -1e10
    if delta_hi > 1e10:
        delta_hi = 1e10
    return delta_lo, delta_hi


def line_search_max_quad(c0, L, Q, delta_lo, delta_hi):
    if delta_hi - delta_lo < 1e-12:
        return 0.0, float(np.max(c0))
    n1 = 60
    grid1 = np.linspace(delta_lo, delta_hi, n1)
    vals1 = np.empty(n1)
    for i in range(n1):
        d = grid1[i]
        vals1[i] = np.max(c0 + d * L + d * d * Q)
    best_i = int(np.argmin(vals1))
    if 1 <= best_i < n1 - 2:
        lo2 = grid1[best_i - 1]
        hi2 = grid1[best_i + 1]
    else:
        lo2, hi2 = delta_lo, delta_hi
    n2 = 60
    grid2 = np.linspace(lo2, hi2, n2)
    vals2 = np.empty(n2)
    for i in range(n2):
        d = grid2[i]
        vals2[i] = np.max(c0 + d * L + d * d * Q)
    best_i2 = int(np.argmin(vals2))
    best_delta = float(grid2[best_i2])
    best_val = float(vals2[best_i2])
    val0 = float(np.max(c0))
    if delta_lo <= 0.0 <= delta_hi and val0 < best_val:
        best_delta, best_val = 0.0, val0
    return best_delta, best_val


def get_line_coefficients(h, a, dx):
    c0 = np.correlate(h, 1.0 - h, mode='full') * dx
    L = (np.correlate(a, 1.0 - h, mode='full') - np.correlate(h, a, mode='full')) * dx
    Q = -np.correlate(a, a, mode='full') * dx
    return c0, L, Q


def compute_subgradient(h, active_indices, N, c_vals, c_max):
    grad = np.zeros(N)
    for r in active_indices:
        w = (c_vals[r] / max(c_max, 1e-30)) ** 4
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                grad[d:] += w * (1.0 - h[:n])
                grad[:n] -= w * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                grad[:n] += w * (1.0 - h[dd:])
                grad[dd:] -= w * h[:n]
    grad -= np.mean(grad)
    return grad


def get_constraint_gradient(h, r, N):
    d = r - (N - 1)
    g = np.zeros(N)
    if d >= 0:
        n = N - d
        if n > 0:
            g[d:] = 1.0 - h[:n]
            g[:n] = -h[d:]
    else:
        dd = -d
        n = N - dd
        if n > 0:
            g[:n] = 1.0 - h[dd:]
            g[dd:] = -h[:n]
    g -= np.mean(g)
    return g


def upscale_func(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    deadline = start_time + budget_s - 20.0

    h = np.array(initial_h_values, dtype=float).copy()
    N = len(h)
    dx = 2.0 / N
    target_sum = N / 2.0

    best_h = h.copy()
    best_c5 = compute_c5(h)

    rng = np.random.default_rng(seed)

    active_tol = 0.008
    no_improve = 0
    iter_count = 0

    while time.time() < deadline:
        iter_count += 1
        c_vals = np.correlate(h, 1.0 - h, mode='full') * dx
        c_max = float(np.max(c_vals))

        if c_max < best_c5 - 1e-13:
            best_c5 = c_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1

        if no_improve > 80:
            scale = 0.02 + 0.04 * min(1.0, no_improve / 200.0)
            noise = rng.normal(0, scale, N)
            noise -= np.mean(noise)
            h = project_feasible(np.clip(h + noise, 0.0, 1.0), target_sum)
            no_improve = 0
            continue

        active_mask = c_vals > c_max - active_tol
        active_indices = np.where(active_mask)[0]

        if len(active_indices) > 30:
            active_tol *= 0.6
        elif len(active_indices) < 3:
            active_tol *= 1.5
        active_tol = max(0.002, min(0.04, active_tol))

        # Build direction pool
        directions = []

        # Direction type 1: weighted subgradient
        if len(active_indices) > 0:
            grad = compute_subgradient(h, active_indices, N, c_vals, c_max)
            norm = np.linalg.norm(grad)
            if norm > 1e-12:
                directions.append(grad / norm)
                directions.append(-grad / norm)

        # Direction type 2: individual worst constraints (top 3)
        worst_idx = np.argsort(c_vals)[-min(3, N):][::-1]
        for r in worst_idx:
            g = get_constraint_gradient(h, int(r), N)
            ng = np.linalg.norm(g)
            if ng > 1e-12:
                directions.append(g / ng)
                directions.append(-g / ng)

        # Direction type 3: targeted pairwise from worst constraint
        worst_r = int(np.argmax(c_vals))
        d_w = worst_r - (N - 1)
        if d_w > 0:
            n_w = N - d_w
            if n_w > 1:
                terms = h[d_w:d_w + n_w] * (1.0 - h[:n_w])
                top_pairs = np.argsort(terms)[-3:][::-1]
                for tp in top_pairs:
                    i_dst = int(tp)
                    i_src = int(tp) + d_w
                    if i_src != i_dst and 0 <= i_src < N:
                        a_pair = np.zeros(N)
                        a_pair[i_src] = -1.0
                        a_pair[i_dst] = 1.0
                        directions.append(a_pair)
                        a_pair2 = np.zeros(N)
                        a_pair2[i_dst] = -1.0
                        a_pair2[i_src] = 1.0
                        directions.append(a_pair2)
        elif d_w < 0:
            dd = -d_w
            n_w = N - dd
            if n_w > 1:
                terms = h[:n_w] * (1.0 - h[dd:dd + n_w])
                top_pairs = np.argsort(terms)[-3:][::-1]
                for tp in top_pairs:
                    i_dst = int(tp) + dd
                    i_src = int(tp)
                    if i_src != i_dst and 0 <= i_src < N:
                        a_pair = np.zeros(N)
                        a_pair[i_src] = -1.0
                        a_pair[i_dst] = 1.0
                        directions.append(a_pair)

        # Direction type 4: random directions
        n_rand = 4
        for _ in range(n_rand):
            rd = rng.standard_normal(N)
            rd -= np.mean(rd)
            nrd = np.linalg.norm(rd)
            if nrd > 1e-12:
                directions.append(rd / nrd)

        # Direction type 5: random sparse (2-4 pairs)
        if N > 10:
            n_pairs = rng.integers(2, 5)
            a_sparse = np.zeros(N)
            for _ in range(n_pairs):
                i = int(rng.integers(0, N))
                j = int(rng.integers(0, N))
                if i != j:
                    a_sparse[i] -= 1.0
                    a_sparse[j] += 1.0
            ns = np.linalg.norm(a_sparse)
            if ns > 1e-12:
                directions.append(a_sparse / ns)

        # Try all directions, keep best
        best_improvement = 1e-13
        best_new_h = None
        c0_base = c_vals  # reuse

        for a_dir in directions:
            delta_lo, delta_hi = feasible_interval(h, a_dir)
            if delta_hi - delta_lo < 1e-10:
                continue

            L_dir = (np.correlate(a_dir, 1.0 - h, mode='full') -
                     np.correlate(h, a_dir, mode='full')) * dx
            Q_dir = -np.correlate(a_dir, a_dir, mode='full') * dx

            bd, bv = line_search_max_quad(c0_base, L_dir, Q_dir, delta_lo, delta_hi)

            if bv < c_max - best_improvement:
                best_improvement = c_max - bv
                h_try = h + bd * a_dir
                h_try = np.clip(h_try, 0.0, 1.0)
                h_try = project_feasible(h_try, target_sum)
                # Verify
                c_verify = compute_c5(h_try)
                if c_verify < c_max - best_improvement * 0.5:
                    best_new_h = h_try
                    best_improvement = c_max - c_verify

        if best_new_h is not None:
            h = best_new_h
        else:
            # No direction improved; try small random nudge
            nudge = rng.normal(0, 0.003, N)
            nudge -= np.mean(nudge)
            h_try = np.clip(h + nudge, 0.0, 1.0)
            h_try = project_feasible(h_try, target_sum)
            c_try = compute_c5(h_try)
            if c_try < c_max - 1e-12:
                h = h_try

        # Every 500 iterations, try a structured restart
        if no_improve > 150 and no_improve % 50 == 0:
            scale = 0.05
            noise = rng.normal(0, scale, N)
            noise -= np.mean(noise)
            h_restart = project_feasible(np.clip(h + noise, 0.0, 1.0), target_sum)
            # Quick local optimization from restart
            for _ in range(20):
                if time.time() > deadline:
                    break
                c_vals_r = np.correlate(h_restart, 1.0 - h_restart, mode='full') * dx
                c_max_r = float(np.max(c_vals_r))
                worst_r_r = int(np.argmax(c_vals_r))
                g_r = get_constraint_gradient(h_restart, worst_r_r, N)
                ng_r = np.linalg.norm(g_r)
                if ng_r < 1e-12:
                    break
                g_r = g_r / ng_r
                d_lo, d_hi = feasible_interval(h_restart, g_r)
                if d_hi - d_lo < 1e-10:
                    d_lo, d_hi = feasible_interval(h_restart, -g_r)
                    if d_hi - d_lo < 1e-10:
                        break
                    g_r = -g_r
                L_r = (np.correlate(g_r, 1.0 - h_restart, mode='full') -
                       np.correlate(h_restart, g_r, mode='full')) * dx
                Q_r = -np.correlate(g_r, g_r, mode='full') * dx
                bd_r, bv_r = line_search_max_quad(c_vals_r, L_r, Q_r, d_lo, d_hi)
                if bv_r >= c_max_r - 1e-12:
                    break
                h_restart = np.clip(h_restart + bd_r * g_r, 0.0, 1.0)
                h_restart = project_feasible(h_restart, target_sum)
            c_restart = compute_c5(h_restart)
            if c_restart < c_max:
                h = h_restart

    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    return best_h, best_c5, len(best_h)
