        grad[k, lo:hi+1] = 1.0 - 2.0 * h[k - indices]
    return grad

         -min(R, h[i]) ≤ δ[i] ≤ min(R, 1-h[i])

    c_obj[-1] = 1.0

    # Row: [G[k,0], G[k,1], ..., G[k,n-1], -1] ≤ -corr[k]
    A_ub = np.zeros((len(active_indices), n + 1))
    b_ub = np.zeros(len(active_indices))
    for idx, k in enumerate(active_indices):
        A_ub[idx, :n] = G[k, :]
        A_ub[idx, n] = -1.0
        b_ub[idx] = -corr[k]
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds: -min(R, h[i]) ≤ δ[i] ≤ min(R, 1-h[i]), t free
    bounds = []
    for i in range(n):
        lo = -min(R, h[i])
        hi = min(R, 1.0 - h[i])
        bounds.append((lo, hi))
    bounds.append((None, None))  # t is free
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        return res.x[:n]  # the δ direction
    return None

def project_constraint(p, n):

        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))

def corr_gradient(h):

        hi = min(n - 1, k)
        if lo <= hi:
            idx = np.arange(lo, hi + 1)
            G[k, idx] = 1.0 - 2.0 * h[k - idx]
    return G
