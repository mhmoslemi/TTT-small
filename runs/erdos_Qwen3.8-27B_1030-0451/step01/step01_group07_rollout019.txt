def get_parent():

def project_feasible(h, target):
    h = np.asarray(h, dtype=float).copy()
    lo, hi = -1.0, 1.0
    for _ in range(100):

        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    resid = out.sum() - target
    if abs(resid) > 1e-10:
        if resid > 0:
            mask = out > 1e-10

def c5_score(h):
    n = len(h)

def corr_scaled(h):

    return np.correlate(h, 1.0 - h, mode='full') * dx


def lse_obj(h, sigma):
    c = corr_scaled(h)
    m = c.max()
    if sigma < 1e-9:
        return m
    s = np.exp(sigma * (c - m))

def lse_grad(h, sigma):
    n = len(h)
    c = corr_scaled(h)
    m = c.max()
    s = np.exp(sigma * (c - m))
    w = s / s.sum()
    grad = np.zeros(n)
    center = n - 1
    for p in range(-(n - 1), n):

        if wp < 1e-14:
            continue

            grad += wp * (1.0 - 2.0 * h)

            grad[p:] += wp * (1.0 - h[:n - p])
            grad[:n - p] -= wp * h[p:]
        else:
            q = -p
            grad[:n - q] += wp * (1.0 - h[q:])
            grad[q:] -= wp * h[:n - q]
    return grad

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)
