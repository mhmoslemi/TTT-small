    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    for _ in range(30):
        diff = S - h.sum()
        if abs(diff) < 1e-14:
            break
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(np.sum(interior))
        if cnt == 0:
            break
        h[interior] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    return h

def softmax_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -300, 300)
    return c_max + tau * np.log(np.sum(np.exp(exponents)))

def softmax_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -300, 300)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights = weights / sw
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        s = K - (n - 1)
        if s >= 0:
            g[s:] += w * (1.0 - h[:n - s])
            g[:n - s] += w * (-h[s:])
        else:
            m = -s
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n - m])
    return g

class SoftObj:
    def __init__(self, tau):
        self.tau = tau
    def __call__(self, h):
        return softmax_obj(h, self.tau)
    def grad(self, h):
        return softmax_grad(h, self.tau)

def run_slsqp(h0, tau, n, maxiter):
    obj = lambda h: softmax_obj(h, tau)
    grad = lambda h: softmax_grad(h, tau)
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - n/2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

def greedy_binary(n, rng):
    """Construct a binary (or near-binary) initial solution using greedy placement."""
    h = np.zeros(n)
    target = n / 2.0
    if target == int(target):
        # Even n: fully binary
        ones = np.zeros(n, dtype=bool)
        remaining = int(target)
        # Start with random
        idx = rng.permutation(n)
        positions = list(idx[:remaining])
        ones[positions] = True
        h = ones.astype(float)
    else:
        # Odd n: one fractional value
        int_part = int(target)  # floor
        frac_part = target - int_part  # 0.5
        # Place int_part ones and one frac_part
        idx = rng.permutation(n)
        h[idx[:int_part]] = 1.0
        h[idx[int_part]] = frac_part
    return h

def greedy_improve_binary(h, n, rng, max_iters, deadline):
    """Improve a near-binary solution using greedy swaps."""
    cur = h.copy()
    cur_score = exact_score(cur)
    improved = True
    iters = 0
    while improved and time.time() < deadline and iters < max_iters:
        improved = False
        iters += 1
        # Find the lag with maximum contribution
        dx = 2.0 / n
        raw = np.correlate(cur, 1.0 - cur, mode='full') * dx
        active_k = int(np.argmax(raw))
        s = active_k - (n - 1)
        # Find pairs to swap: look for i where cur[i] is high and cur[i-s] is low (for the active lag)
        # Try random swaps
        for attempt in range(50):
            if time.time() > deadline:
                return cur, cur_score
            a = rng.integers(0, n)
            b = rng.integers(0, n)
            if a == b:
                continue
            # Swap
            cur[a], cur[b] = cur[b], cur[a]
            s_new = exact_score(cur)
            if s_new < cur_score - 1e-14:
                cur_score = s_new
                improved = True
                break
            else:
                cur[a], cur[b] = cur[b], cur[a]
    return cur, cur_score

def sa_optimize(h0, n, deadline, rng, t_start=None, t_end=None):
    """Simulated annealing with multi-point moves."""
    if t_start is None:
        t_start = 0.005
    if t_end is None:
        t_end = 1e-7
    
    cur = h0.copy()
    cur_score = exact_score(cur)
    best = cur.copy()
    best_score = cur_score
    
    n_lags = 2 * n - 1
   冷却 = 1000  # total iterations target (approx)
    # Actually let's just run until deadline
    
    T = t_start
    patience = 0
    max_patience = 2000
    
    while time.time() < deadline:
        # Choose a move type
        move_type = rng.integers(0, 5)
        
        if move_type == 0:
            # Random pair swap
            a = rng.integers(0, n)
            b = rng.integers(0, n)
            if a == b:
                continue
            h_new = cur.copy()
            h_new[a], h_new[b] = h_new[b], h_new[a]
        elif move_type == 1:
            # Random pair transfer
            a = rng.integers(0, n)
            b = rng.integers(0, n)
            if a == b:
                continue
            d = 0.01 + 0.3 * rng.random()
            d = min(d, cur[b], 1.0 - cur[a])
            if d < 1e-6:
                continue
            h_new = cur.copy()
            h_new[a] += d
            h_new[b] -= d
            h_new = np.clip(h_new, 0, 1)
            # Fix sum (clipping might have changed it)
            h_new = project_feasible(h_new, n)
        elif move_type == 2:
            # Multi-point transfer (3-5 pairs)
            n_pairs = rng.integers(2, 6)
            h_new = cur.copy()
            total_d = 0.0
            for _ in range(n_pairs):
                a = rng.integers(0, n)
                b = rng.integers(0, n)
                if a == b:
                    continue
                d = 0.005 + 0.05 * rng.random()
                d = min(d, h_new[b], 1.0 - h_new[a])
                if d < 1e-6:
                    continue
                h_new[a] += d
                h_new[b] -= d
                total_d += d
            if total_d < 1e-6:
                continue
            h_new = np.clip(h_new, 0, 1)
            h_new = project_feasible(h_new, n)
        elif move_type == 3:
            # Smooth perturbation (low frequency sine)
            freq = rng.integers(1, max(2, n // 10))
            phase = 2.0 * np.pi * rng.random()
            amp = 0.05 + 0.2 * rng.random()
            x = np.arange(n) / n
            pert = amp * np.sin(2.0 * np.pi * freq * x + phase)
            h_new = project_feasible(cur + pert, n)
        else:
            # Block perturbation
            blen = rng.integers(max(3, n // 8), max(4, n // 3))
            start = rng.integers(0, n - blen + 1)
            noise = 0.1 * (rng.random(blen) - 0.5)
            h_new = cur.copy()
            h_new[start:start+blen] = h_new[start:start+blen] + noise
            h_new = project_feasible(h_new, n)
        
        new_score = exact_score(h_new)
        
        # Acceptance criterion
        if new_score < cur_score:
            cur = h_new
            cur_score = new_score
            if cur_score < best_score:
                best = cur.copy()
                best_score = cur_score
            patience = 0
        else:
            delta = new_score - cur_score
            if delta < T:  # Avoid log of negative
                prob = math.exp(-delta / T)
                if rng.random() < prob:
                    cur = h_new
                    cur_score = new_score
            patience += 1
        
        # Cool down
        if patience % 100 == 0:
            T *= 0.98
            if T < t_end:
                T = t_end
        
        if patience > max_patience:
            # Restart with new temperature
            T = t_start * (0.5 + 0.5 * rng.random())
            patience = 0
            # Optionally reheat: add random perturbation
            if rng.random() < 0.3:
                cur = project_feasible(cur + 0.05 * (rng.random(n) - 0.5), n)
                cur_score = exact_score(cur)
    
    return best, best_score

def directed_cd(h0, n, deadline, rng):
    """Directed coordinate descent targeting active lags."""
    cur = h0.copy()
    cur_score = exact_score(cur)
    dx = 2.0 / n
    
    while time.time() < deadline:
        # Compute active lags
        raw = np.correlate(cur, 1.0 - cur, mode='full') * dx
        # Get top 5 lags
        top_ks = np.argsort(raw)[-5:]
        
        improved = False
        for K in top_ks:
            if time.time() > deadline:
                break
            s = int(K) - (n - 1)
            # Gradient of raw[K] w.r.t. h
            # ∂raw[K]/∂h[j] = (1-h[j-s]) * 1_{j-s in [0,n-1]} - h[j+s] * 1_{j+s in [0,n-1]}
            g = np.zeros(n)
            if s >= 0:
                # j-s >= 0 means j >= s
                g[s:] += (1.0 - cur[:n-s])
                # j+s < n means j < n-s
                g[:n-s] -= cur[s:]
            else:
                m = -s
                # j-m >= 0 means j >= m
                g[m:] += (1.0 - cur[:n-m])
                # j+m < n means j < n-m
                g[:n-m] -= cur[m:]
            
            # Project gradient to sum-preserving direction
            g = g - np.mean(g)
            
            # Pair selection: increase where g is most negative, decrease where g is most positive
            imin = int(np.argmin(g))
            imax = int(np.argmax(g))
            if imin == imax:
                continue
            
            # Line search
            # d range: h[imin] + d ∈ [0,1], h[imax] - d ∈ [0,1]
            d_lo = max(-cur[imin], -(1.0 - cur[imax]))  # Wait, this is wrong
            # h[imin] + d >= 0 => d >= -cur[imin]
            # h[imin] + d <= 1 => d <= 1 - cur[imin]
            # h[imax] - d >= 0 => d <= cur[imax]
            # h[imax] - d <= 1 => d >= cur[imax] - 1
            d_lo = max(-cur[imin], cur[imax] - 1.0)
            d_hi = min(1.0 - cur[imin], cur[imax])
            if d_hi - d_lo < 1e-10:
                continue
            
            # Try a few step sizes
            for frac in [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]:
                d = d_lo + frac * (d_hi - d_lo)
                if abs(d) < 1e-10:
                    continue
                cur[imin] += d
                cur[imax] -= d
                new_s = exact_score(cur)
                if new_s < cur_score - 1e-14:
                    cur_score = new_s
                    improved = True
                    break
                else:
                    cur[imin] -= d
                    cur[imax] += d
        
        if not improved:
            # Try random pairs as fallback
            found = False
            for _ in range(20):
                a = rng.integers(0, n)
                b = rng.integers(0, n)
                if a == b:
                    continue
                d = 0.01 * rng.random()
                if d >= cur[b] or d > 1 - cur[a]:
                    d = min(cur[b], 1 - cur[a]) * 0.5
                if d < 1e-8:
                    continue
                cur[a] += d
                cur[b] -= d
                new_s = exact_score(cur)
                if new_s < cur_score - 1e-14:
                    cur_score = new_s
                    improved = True
                    found = True
                    break
                cur[a] -= d
                cur[b] += d
            if not found:
                break
    
    return cur, cur_score

def interpolate_to_n(h, new_n):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h)
    return project_feasible(h_interp, new_n)
