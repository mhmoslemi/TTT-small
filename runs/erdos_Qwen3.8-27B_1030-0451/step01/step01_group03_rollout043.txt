_G_T = 0.0
_G_N = 0
_G_TAU = 1.0

    return float(np.sum(h) - _G_T)

    return np.ones(_G_N)

    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()

    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    omh = 1.0 - h

        lo2 = max(0, -d)

                h_curr = project_feasible(h_curr + rng_esc.standard_normal(n) * 0.008, target)
                stall_count = 0

    return best_h, best_c5

def subgradient_polish(h0, deadline, max_iter=2000):
    """Projected subgradient method for the max of c_k."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n

    c = compute_all_c(h)
    step = 0.005
    patience = 200
    no_improve = 0

    for it in range(max_iter):
        if time.time() > deadline - 3:
            break

        c = compute_all_c(h)
        current_max = c.max()

        if current_max < best_c5 - 1e-12:
            best_c5 = current_max
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > patience:
                step *= 0.7
                no_improve = 0
                if step < 1e-6:
                    break

        # Find top 3 lags
        top_idx = np.argsort(c)[-3:][::-1]
        # Compute combined gradient (weighted by proximity to max)
        weights = np.zeros(len(c))
        for ti in top_idx:
            weights[ti] = np.exp(-(c.max() - c[ti]) / max(1e-4, (c.max() - c.min()) * 0.1))
        weights /= weights.sum()

        grad = np.zeros(n)
        for ti in top_idx:
            d = ti - (n - 1)
            grad += weights[top_idx.tolist().index(ti)] * _grad_single_lag(h, d)

        # Projected gradient descent step
        h_new = h - step * grad
        h_new = project_feasible(h_new, target)

        # Check if it improved
        c_new = compute_all_c(h_new)
        if c_new.max() < current_max:
            h = h_new
        else:
            # Try smaller step
            h_new = h - 0.1 * step * grad
            h_new = project_feasible(h_new, target)
            c_new = compute_all_c(h_new)
            if c_new.max() < current_max:
                h = h_new
                step *= 0.8
            else:
                step *= 0.5

    return best_h, best_c5

def pair_swap_search(h0, deadline, max_iters=50000):
    """Local search via pair swaps for the active lag."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    dx = 2.0 / n
    rng = np.random.default_rng(12345)

    c = compute_all_c(h)
    k_star = int(np.argmax(c))
    d_star = k_star - (n - 1)

    for it in range(max_iters):
        if time.time() > deadline - 3:
            break

        # Occasionally recompute active lag
        if it % 50 == 0:
            c = compute_all_c(h)
            k_star = int(np.argmax(c))
            d_star = k_star - (n - 1)

        # Pick random pair
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            continue

        # Determine direction: which swap reduces c_{d_star}?
        # c_d = dx * sum_i h[i+d]*(1-h[i])
        # If we swap h[i] and h[j] (or transfer eps), we want to reduce c_d.
        # Transfer eps from h[i] to h[j]: h[i]-=eps, h[j]+=eps
        # Change in c_d ≈ -eps * [(1-h[i+d])*δ(i in range for h[i+d] term)
        #                            - h[j+d]*δ(j in range)
        #                            -(1-h[j+d])*δ(j+d in range for h[j+d] term from i side)
        #                            + h[i+d]*δ(i+d in range from j side)]
        # This is complex, so let's just try it and check.

        eps = 0.01
        if h[i] < eps or h[j] > 1.0 - eps:
            continue

        h[i] -= eps
        h[j] += eps
        c_new = compute_all_c(h)
        if c_new.max() < best_c5:
            best_c5 = c_new.max()
            best_h = h.copy()
        else:
            h[i] += eps
            h[j] -= eps

    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 5

    # Get parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        n_par = 120
        parent = np.full(n_par, 0.5)

    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)
    par_c5 = evaluate_c5(parent)

    best_h = parent.copy()
    best_c5 = par_c5
    rng = np.random.default_rng(seed)

    # Phase 1: Optimize at native resolution with diverse starts
    starts = [parent.copy()]
    # Add shifted versions
    for shift in [n_par // 4, n_par // 3, n_par // 5]:
        starts.append(np.roll(parent, shift))
        starts.append(np.roll(parent, -shift))
    # Add small random perturbations
    for amp in [0.01, 0.03]:
        starts.append(parent + rng.standard_normal(n_par) * amp)

    # Also try a sinusoidal starting point
    x = np.linspace(0, 2 * np.pi, n_par, endpoint=False)
    sin_start = 0.5 + 0.3 * np.sin(x) + 0.1 * np.sin(2 * x)
    sin_start = project_feasible(sin_start, n_par / 2.0)
    starts.append(sin_start)

    # Optimize each start (limited time each)
    time_per_start = 80
    for si, s in enumerate(starts):
        if time.time() > deadline - 150:
            break
        s = project_feasible(s, n_par / 2.0)
        s_c5 = evaluate_c5(s)
        # Skip if already worse than best by a lot
        if s_c5 > best_c5 + 0.01:
            continue
        start_time = time.time()
        local_deadline = min(deadline - 150, start_time + time_per_start)
        h_opt, c5_opt = smooth_optimize(s, n_par, local_deadline)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 2: Subgradient polish at native resolution
    if time.time() < deadline - 80:
        h_pol, c5_pol = subgradient_polish(best_h, min(deadline - 60, time.time() + 60), max_iter=3000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 3: Try at 2x resolution
    if time.time() < deadline - 60:
        n2 = min(2 * n_par, 960)
        h0_2x = interp_h(best_h, n2)
        h_opt, c5_opt = smooth_optimize(h0_2x, n2, min(deadline - 30, time.time() + 120))
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt

    # Phase 4: Subgradient polish on the best (possibly 2x) solution
    if time.time() < deadline - 20:
        h_pol, c5_pol = subgradient_polish(best_h, deadline, max_iter=5000)
        if c5_pol < best_c5:
            best_c5 = c5_pol
            best_h = h_pol

    # Phase 5: Pair swap local search if time allows
    if time.time() < deadline - 10:
        n_best = len(best_h)
        if n_best <= 960:
            h_swap, c5_swap = pair_swap_search(best_h, deadline, max_iters=50000)
            if c5_swap < best_c5:
                best_c5 = c5_swap
                best_h = h_swap

    # Final feasibility check
    best_n = len(best_h)
    best_h = project_feasible(best_h, best_n / 2.0)
    best_c5 = evaluate_c5(best_h)

    return (best_h, best_c5, best_n)
