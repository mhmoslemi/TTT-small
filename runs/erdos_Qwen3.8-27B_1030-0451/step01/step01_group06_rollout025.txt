        if k == 0:
            g = dx * (1.0 - 2.0*h)
        elif k > 0:
            g[k:] += dx * (1.0 - h[:n-k])
            g[:n-k] += -dx * h[k:]
        else:
            t = -k
            g[:n-t] += dx * (1.0 - h[t:])
            g[t:] += -dx * h[:n-t]
        grads[i] = g
    
    nvars = n + 1
    c_obj = np.zeros(nvars); c_obj[n] = 1.0
    A_ub = np.zeros((k_act, nvars))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -corr[top_idx]
    lower = np.maximum(-h, -trust)
    upper = np.minimum(1.0 - h, trust)
    A_eq = np.zeros((1, nvars)); A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = list(zip(lower, upper)) + [(None, None)]
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.status == 0:
            d = res.x[:n]
            if np.all(h + d >= -1e-9) and np.all(h + d <= 1 + 1e-9):
                return d
    except Exception:
        pass
    return None

def slsqp_lse(h0, tau, maxiter=100):
    """SLSQP optimization of log-sum-exp objective."""
    n = len(h0)
    target = n / 2.0
    
    def obj(x):
        corr = correlation_array(x)
        M = np.max(corr)
        u = np.clip((corr - M) / tau, -745, 745)
        return M + tau * np.log(np.exp(u).sum())
    
    def grad(x):
        corr = correlation_array(x)
        w = lse_weights(corr, tau)
        return gradient_weighted(x, w)
    
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
                    'jac': lambda x: np.ones(n)}]
    bounds = [(0, 1)] * n
    try:
        res = minimize(obj, h0, jac=grad, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or res.fun < obj(h0):
            return project_box_sum(res.x, target)
    except Exception:
        pass
    return project_box_sum(h0, target)

def lowpass(h, size=5):
    """Simple low-pass filter (convolution with triangular kernel)."""
    n = len(h)
    kernel = np.ones(size) / size
    padded = np.pad(h, size//2, mode='edge')
    return np.convolve(padded, kernel, mode='valid')[:n]

def upsample(h, factor):
    return np.repeat(h, factor)

def downsample(h, factor):
    n = len(h)
    new_n = n // factor
    return h[:new_n*factor].reshape(new_n, factor).mean(axis=1)
