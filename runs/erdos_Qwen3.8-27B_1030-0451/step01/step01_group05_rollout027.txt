    """Compute col_i: the K-th element is ∂c[K]/∂h[i]"""
    L = 2*n - 1
    center = n - 1
    s_col = np.zeros(L)
    for K in range(L):

        # h[i] appears as h[n+s] when n = i-s
        n_val = i - s
        if 0 <= n_val <= n-1:

        # h[i] appears in (1-h[n]) when n = i
        n2 = i + s
        if 0 <= n2 <= n-1:

def exact_pair_descent(h, n, rng, deadline, max_rounds=100000):
    """Exact pair-exchange descent. Returns improved h and score."""
    L = 2*n - 1
    dx = 2.0 / n
    raw = np.correlate(h, 1.0-h, mode='full')
    cur_score = float(np.max(raw) * dx)
    
    all_cols = np.zeros((n, L))
    for i in range(n):
        all_cols[i] = compute_col(h, i, n)
    
    # Precompute col as a 3D array: all_cols[i, K] = col_i[K]
    
    improved = True
    round_count = 0
    while improved and round_count < max_rounds:
        round_count += 1
        if time.time() > deadline:
            break
        improved = False
        
        # Compute gradient-like signal
        raw_max = np.max(raw)
        tau = 0.02 * raw_max if raw_max > 0 else 0.02
        weights = np.exp((raw - raw_max) / tau)
        weights /= np.sum(weights)
        
        # Gradient: grad[i] = sum_K weights[K] * all_cols[i, K]
        grad = all_cols @ weights  # shape (n,)
        
        # Select candidate pairs
        # We want to increase low-grad and decrease high-grad
        order = np.argsort(grad)
        best_improvement = -1e-18
        best_pair = None
        best_d = 0
        
        n_candidates = min(20, n)  # try up to 20 candidate pairs
        for idx in range(n_candidates):
            a = order[idx]  # low grad, increase
            for jdx in range(n - n_candidates, n):
                b = order[jdx]  # high grad, decrease
                if a == b:
                    continue
                # Try this pair
                lin = all_cols[a] - all_cols[b]
                quad = np.zeros(L)
                center = n - 1
                quad[center] = -2.0
                quad[center + a - b] += 1.0
                quad[center + b - a] += 1.0
                
                d_lo = max(-h[a], h[b] - 1.0)
                d_hi = min(1.0 - h[a], h[b])
                if d_hi <= d_lo:
                    continue
                
                # Grid search
                M = 40
                d_grid = np.linspace(d_lo, d_hi, M)
                new_raw = raw[None,:] + d_grid[:,None]*lin[None,:] + (d_grid**2)[:,None]*quad[None,:]
                scores = np.max(new_raw, axis=1) * dx
                best_local = np.min(scores)
                best_local_d = d_grid[np.argmin(scores)]
                
                if best_local < cur_score - 1e-12:
                    if best_local < cur_score - best_improvement:
                        best_improvement = cur_score - best_local
                        best_pair = (int(a), int(b))
                        best_d = float(best_local_d)
        
        # Also try some random pairs
        for _ in range(5):
            a = int(rng.integers(0, n))
            b = int(rng.integers(0, n))
            if a == b:
                continue
            lin = all_cols[a] - all_cols[b]
            quad = np.zeros(L)
            center = n - 1
            quad[center] = -2.0
            quad[center + a - b] += 1.0
            quad[center + b - a] += 1.0
            
            d_lo = max(-h[a], h[b] - 1.0)
            d_hi = min(1.0 - h[a], h[b])
            if d_hi <= d_lo:
                continue
            
            M = 30
            d_grid = np.linspace(d_lo, d_hi, M)
            new_raw = raw[None,:] + d_grid[:,None]*lin[None,:] + (d_grid**2)[:,None]*quad[None,:]
            scores = np.max(new_raw, axis=1) * dx
            best_local = np.min(scores)
            best_local_d = d_grid[np.argmin(scores)]
            
            if best_local < cur_score - 1e-12:
                if best_local < cur_score - best_improvement:
                    best_improvement = cur_score - best_local
                    best_pair = (int(a), int(b))
                    best_d = float(best_local_d)
        
        if best_pair is not None:
            a, b = best_pair
            h[a] += best_d
            h[b] -= best_d
            # Update raw exactly
            lin = all_cols[a] - all_cols[b]
            quad = np.zeros(L)
            center = n - 1
            quad[center] = -2.0
            quad[center + a - b] += 1.0
            quad[center + b - a] += 1.0
            raw = raw + best_d * lin + best_d**2 * quad
            cur_score = float(np.max(raw) * dx)
            improved = True
            
            # Update cols for a and b
            all_cols[a] = compute_col(h, a, n)
            all_cols[b] = compute_col(h, b, n)
    
    return h, cur_score

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -10.0, 10.0
    for _ in range(60):

        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:

        else:
            hi = mid
    mu = (lo + hi) / 2.0

            h = np.clip(h, 0.0, 1.0)

        tau_use = max(0.001, tau)
        exponents = (raw - raw_max_val) / tau_use
        exponents = np.clip(exponents, -500, 500)
        weights = np.exp(exponents)
        wsum = np.sum(weights)
        if wsum < 1e-300:
            weights = np.ones(L) / L
        else:
            weights /= wsum

        grad = cols @ weights
        order = np.argsort(grad)

        best_delta = 0.0
        best_a = -1
        best_b = -1
        best_d = 0.0

        # Structured pairs
        ns = min(n_struct, n)
        for ii in range(ns):
            a = int(order[ii])
            for jj in range(n - ns, n):
                b = int(order[jj])
                if a == b:
                    continue
                lin = cols[a] - cols[b]
                quad = np.zeros(L)
                quad[center] = -2.0
                da = a - b
                quad[center + da] += 1.0
                quad[center - da] += 1.0

                d_lo = max(-h[a], h[b] - 1.0)
                d_hi = min(1.0 - h[a], h[b])
                if d_hi - d_lo < 1e-12:
                    continue

                M = grid_size
                d_grid = np.linspace(d_lo, d_hi, M)
                new_raw = raw[None, :] + d_grid[:, None] * lin[None, :] + (d_grid ** 2)[:, None] * quad[None, :]
                scores = np.max(new_raw, axis=1) * dx
                best_local = float(np.min(scores))
                best_local_d = float(d_grid[np.argmin(scores)])

                # Refine
                half_step = (d_hi - d_lo) / max(1, M - 1)
                r_lo = max(d_lo, best_local_d - 2 * half_step)
                r_hi = min(d_hi, best_local_d + 2 * half_step)
                if r_hi > r_lo + 1e-14:
                    d_fine = np.linspace(r_lo, r_hi, 30)
                    nf = raw[None, :] + d_fine[:, None] * lin[None, :] + (d_fine ** 2)[:, None] * quad[None, :]
                    sf = np.max(nf, axis=1) * dx
                    if float(np.min(sf)) < best_local:
                        best_local = float(np.min(sf))
                        best_local_d = float(d_fine[np.argmin(sf)])

                delta = cur_score - best_local
                if delta > best_delta:
                    best_delta = delta
                    best_a = a
                    best_b = b
                    best_d = best_local_d

        # Random pairs
        for _ in range(n_random):
            a = int(rng_local.integers(0, n))
            b = int(rng_local.integers(0, n))
            if a == b:
                continue
            lin = cols[a] - cols[b]
            quad = np.zeros(L)
            quad[center] = -2.0
            da = a - b
            quad[center + da] += 1.0
            quad[center - da] += 1.0

            d_lo = max(-h[a], h[b] - 1.0)
            d_hi = min(1.0 - h[a], h[b])
            if d_hi - d_lo < 1e-12:
                continue

            M = grid_size
            d_grid = np.linspace(d_lo, d_hi, M)
            new_raw = raw[None, :] + d_grid[:, None] * lin[None, :] + (d_grid ** 2)[:, None] * quad[None, :]
            scores = np.max(new_raw, axis=1) * dx
            best_local = float(np.min(scores))
            best_local_d = float(d_grid[np.argmin(scores)])

            delta = cur_score - best_local
            if delta > best_delta:
                best_delta = delta
                best_a = a
                best_b = b
                best_d = best_local_d

        if best_delta > 1e-14:
            h[best_a] += best_d
            h[best_b] -= best_d
            lin = cols[best_a] - cols[best_b]
            quad = np.zeros(L)
            quad[center] = -2.0
            da = best_a - best_b
            quad[center + da] += 1.0
            quad[center - da] += 1.0
            raw = raw + best_d * lin + best_d ** 2 * quad
            cols = compute_all_cols(h, n)
            cur_score = float(np.max(raw) * dx)
            no_improve = 0
        else:
            no_improve += 1
            tau *= 0.96
            if tau < 0.002:
                tau = 0.002
            if no_improve > 400:
                break

    return h, cur_score


def sa_phase(h, deadline, max_iters=100000, T_start=None, T_end=0.001):
    n = len(h)
    L = 2 * n - 1
    dx = 2.0 / n
    center = n - 1
    raw = np.correlate(h, 1.0 - h, mode='full')
    cur_score = float(np.max(raw) * dx)
    cols = compute_all_cols(h, n)

    if T_start is None:
        T_start = 0.05 * cur_score
    T = T_start
    rng_sa = np.random.default_rng(99999)

    best_h = h.copy()
    best_score = cur_score

    for it in range(max_iters):
        if time.time() > deadline:
            break
        T = T_start * (T_end / T_start) ** (it / max(1, max_iters))

        a = int(rng_sa.integers(0, n))
        b = int(rng_sa.integers(0, n))
        if a == b:
            continue

        lin = cols[a] - cols[b]
        quad = np.zeros(L)
        quad[center] = -2.0
        da = a - b
        quad[center + da] += 1.0
        quad[center - da] += 1.0

        d_lo = max(-h[a], h[b] - 1.0)
        d_hi = min(1.0 - h[a], h[b])
        if d_hi - d_lo < 1e-12:
            continue

        # Try optimal d for min, and also random d
        M = 40
        d_grid = np.linspace(d_lo, d_hi, M)
        new_raw = raw[None, :] + d_grid[:, None] * lin[None, :] + (d_grid ** 2)[:, None] * quad[None, :]
        scores = np.max(new_raw, axis=1) * dx
        best_d = float(d_grid[np.argmin(scores)])
        new_score = float(np.min(scores))

        # Sometimes use a random d for exploration
        if rng_sa.random() < 0.3:
            best_d = float(d_lo + rng_sa.random() * (d_hi - d_lo))
            nr = raw + best_d * lin + best_d ** 2 * quad
            new_score = float(np.max(nr) * dx)

        delta = new_score - cur_score
        if delta < 0 or rng_sa.random() < np.exp(-delta / max(1e-10, T)):
            h[a] += best_d
            h[b] -= best_d
            raw = raw + best_d * lin + best_d ** 2 * quad
            cols = compute_all_cols(h, n)
            cur_score = new_score
            if cur_score < best_score:
                best_h = h.copy()
                best_score = cur_score

    return best_h, best_score


def interpolate_to(h, new_n):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_n)


def make_periodic(n, period, duty, phase):
    x = (np.arange(n) + 0.5) / n * 2.0
    t = ((x + 2.0 * phase / period) % (2.0 / period)) / (2.0 / period)
    h = (t < duty).astype(float)
    return project_feasible(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 15.0
    rng = np.random.default_rng(seed)

    # Get initial construction
    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None:
        n_p = len(parent)
        if n_p < 10 or not np.all(np.isfinite(parent)):
            parent = None

    if parent is not None:
        parent = project_feasible(parent, n_p)
        score_p = exact_score(parent)
    else:
        n_p = 80
        parent = make_periodic(n_p, 3, 0.5, 0.0)
        score_p = exact_score(parent)

    best_h = parent.copy()
    best_score = score_p
    best_n = n_p

    # Phase 1: Main pair descent on parent
    t_deadline_1 = min(deadline, t0 + budget_s * 0.55)
    h1, s1 = pair_descent(parent.copy(), t_deadline_1, max_rounds=300000, n_struct=15, n_random=10, grid_size=80)
    if s1 < best_score - 1e-13:
        best_h, best_score, best_n = h1, s1, n_p

    # Phase 1b: Continue with finer grid
    t_deadline_1b = min(deadline, t0 + budget_s * 0.70)
    if time.time() < t_deadline_1b:
        h1b, s1b = pair_descent(best_h.copy(), t_deadline_1b, max_rounds=200000, n_struct=10, n_random=5, grid_size=120)
        if s1b < best_score - 1e-13:
            best_h, best_score, best_n = h1b, s1b, best_n

    # Phase 2: SA escape
    t_deadline_2 = min(deadline, t0 + budget_s * 0.82)
    if time.time() < t_deadline_2:
        h2, s2 = sa_phase(best_h.copy(), t_deadline_2, max_iters=50000)
        if s2 < best_score - 1e-13:
            best_h, best_score, best_n = h2, s2, best_n
        # Follow up with pair descent from SA result
        if time.time() < t_deadline_2:
            h2b, s2b = pair_descent(best_h.copy(), t_deadline_2, max_rounds=100000, n_struct=12, n_random=8, grid_size=80)
            if s2b < best_score - 1e-13:
                best_h, best_score, best_n = h2b, s2b, best_n

    # Phase 3: Multi-resolution
    t_deadline_3 = min(deadline, t0 + budget_s * 0.92)
    if time.time() < t_deadline_3 - 30:
        for factor in [0.9, 1.1, 0.85, 1.15, 0.95, 1.05]:
            if time.time() > t_deadline_3 - 10:
                break
            dn = int(round(best_n * factor))
            if dn < 40 or dn > 300 or dn == best_n:
                continue
            h_d = interpolate_to(best_h, dn)
            s_d = exact_score(h_d)
            if s_d < best_score + 0.003:
                h_d, s_d = pair_descent(h_d, t_deadline_3, max_rounds=50000, n_struct=12, n_random=8, grid_size=70)
                if s_d < best_score - 1e-13:
                    best_h, best_score, best_n = h_d, s_d, dn

    # Phase 4: Try different starting structures
    t_deadline_4 = min(deadline - 10, t0 + budget_s * 0.96)
    if time.time() < t_deadline_4 - 20:
        structures = []
        for period in [2, 3, 4, 5, 7]:
            for duty in [0.3, 0.4, 0.5, 0.6]:
                for phase in [0.0, 0.25, 0.5, 0.75]:
                    structures.append((period, duty, phase))
        rng.shuffle(structures)
        for (period, duty, phase) in structures[:6]:
            if time.time() > t_deadline_4 - 5:
                break
            h_s = make_periodic(best_n, period, duty, phase)
            s_s = exact_score(h_s)
            if s_s < best_score + 0.01:
                h_s, s_s = pair_descent(h_s, t_deadline_4, max_rounds=30000, n_struct=12, n_random=8, grid_size=60)
                if s_s < best_score - 1e-13:
                    best_h, best_score, best_n = h_s, s_s, best_n

    # Phase 5: Final polish
    if time.time() < deadline - 5:
        h_final, s_final = pair_descent(best_h.copy(), deadline - 2, max_rounds=100000, n_struct=10, n_random=5, grid_size=150)
        if s_final < best_score - 1e-13:
            best_h, best_score, best_n = h_final, s_final, best_n

    # Final verification
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-10:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
