def sqp_step(h, n_num_active=30, margin=0.0):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    
    # Get active lags
    sorted_idx = np.argsort(c)[::-1]
    active = sorted_idx[:n_num_active]
    
    # Compute gradients for active lags
    center = n - 1
    A_rows = []
    b_rows = []
    
    for K in active:
        m = K - center
        grad = np.zeros(n)
        for j in range(n):
            g = 0.0
            jm = j + m
            if 0 <= jm < n:
                g += (1.0 - h[jm])
            jmm = j - m
            if 0 <= jmm < n:
                g -= h[jmm]
            grad[j] = g * dx
        # Constraint: grad · δ + C ≤ c[K]
        row = np.concatenate([grad, [1.0]])
        A_rows.append(row)
        b_rows.append(c[K])
    
    # Bound constraints on δ
    for j in range(n):
        # -δ[j] ≤ h[j]
        row = np.zeros(n + 1)
        row[j] = -1.0
        A_rows.append(row)
        b_rows.append(h[j])
        # δ[j] ≤ 1 - h[j]
        row = np.zeros(n + 1)
        row[j] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - h[j])
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    # Equality: sum(δ) = 0
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    # Objective: minimize C
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] * n + [(c_max - 0.1, None)],
                  method='highs')
    
    if res.success:
        delta = res.x[:n]
        h_new = h + delta
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new = project_feasible(h_new, n)
        return h_new, res.x[n]
    return None, None

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_lags(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = min(z) - 2.0, max(z) + 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-13:
        for _ in range(50):
            interior = (h > 1e-13) & (h < 1.0 - 1e-13)
            cnt = int(np.sum(interior))
            if cnt == 0:
                break
            adj = diff / cnt
            h[interior] += adj
            h = np.clip(h, 0.0, 1.0)
            diff = S - np.sum(h)
            if abs(diff) < 1e-14:
                break
    return h

def all_lag_gradients(h):
    """Compute gradients of all lags w.r.t. h. Returns array shape (2n-1, n)."""
    n = len(h)
    dx = 2.0 / n
    center = n - 1
    grads = np.zeros((2 * n - 1, n))
    lags = np.correlate(h, 1.0 - h, mode='full') * dx
    for K in range(2 * n - 1):
        m = K - center
        for j in range(n):
            g = 0.0
            jm = j + m
            if 0 <= jm < n:
                g += (1.0 - h[jm])
            jmm = j - m
            if 0 <= jmm < n:
                g -= h[jmm]
            grads[K, j] = g * dx
    return lags, grads

def sqp_step(h, max_step=0.2, n_active=None):
    """One SQP step. Returns (h_new, success)."""
    n = len(h)
    lags, grads = all_lag_gradients(h)
    c_max = np.max(lags)
    
    if n_active is None:
        n_active = 2 * n - 1  # Use all lags
    
    # Select active lags (top n_active)
    order = np.argsort(lags)[::-1]
    active = order[:n_active]
    
    # Build LP
    # Variables: [δ_0, ..., δ_{n-1}, C]  (n+1 vars)
    # Min: C
    # s.t.: For K in active: lags[K] + grads[K,:]·δ ≤ C
    #       0 ≤ h[j] + δ[j] ≤ 1
    #       sum(δ) = 0
    #       |δ[j]| ≤ max_step
    
    n_vars = n + 1
    A_rows = []
    b_rows = []
    
    for K in active:
        row = np.zeros(n_vars)
        row[:n] = grads[K]
        row[n] = 1.0
        A_rows.append(row)
        b_rows.append(lags[K])
    
    for j in range(n):
        # δ[j] ≤ 1 - h[j]
        row = np.zeros(n_vars)
        row[j] = 1.0
        A_rows.append(row)
        b_rows.append(1.0 - h[j])
        # -δ[j] ≤ h[j]
        row = np.zeros(n_vars)
        row[j] = -1.0
        A_rows.append(row)
        b_rows.append(h[j])
        # δ[j] ≤ max_step
        if max_step < 1.0 - h[j]:
            row = np.zeros(n_vars)
            row[j] = 1.0
            A_rows.append(row)
            b_rows.append(max_step)
        # -δ[j] ≤ max_step
        if max_step < h[j]:
            row = np.zeros(n_vars)
            row[j] = -1.0
            A_rows.append(row)
            b_rows.append(max_step)
    
    A_ub = np.array(A_rows)
    b_ub = np.array(b_rows)
    
    A_eq = np.zeros((1, n_vars))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=[(None, None)] * n + [(0.0, c_max + 0.1)],
                  method='highs', options={'presolve': True})
    
    if not res.success:
        return None, False
    
    delta = res.x[:n]
    # Line search
    for alpha in [1.0, 0.5, 0.25, 0.125, 0.0625]:
        h_try = h + alpha * delta
        h_try = np.clip(h_try, 0.0, 1.0)
        h_try = project_feasible(h_try, n)
        s = exact_score(h_try)
        if s < c_max - 1e-10:
            return h_try, True
    return None, False

def smooth_gradient(h, tau):
    """Compute smooth (log-sum-exp) gradient."""
    n = len(h)
    dx = 2.0 / n
    lags = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(lags)
    exponents = np.clip((lags - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    weights /= sw
    
    grads = np.zeros(n)
    center = n - 1
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-15:
            continue
        m = K - center
        for j in range(n):
            g = 0.0
            jm = j + m
            if 0 <= jm < n:
                g += w * (1.0 - h[jm])
            jmm = j - m
            if 0 <= jmm < n:
                g -= w * h[jmm]
            grads[j] += g
    return grads, c_max

def pair_search(h, max_iters, tau, rng, deadline):
    """2-coordinate exact line search."""
    n = len(h)
    cur = h.copy()
    cur_score = exact_score(cur)
    best = cur.copy()
    best_score = cur_score
    no_improve = 0
    
    for it in range(max_iters):
        if time.time() > deadline:
            break
        
        g, _ = smooth_gradient(cur, tau)
        # Find best descent pair
        imin = int(np.argmin(g))
        imax = int(np.argmax(g))
        
        if imin == imax:
            if no_improve > 50:
                imin = int(rng.integers(0, n))
                imax = int(rng.integers(0, n))
                if imin == imax:
                    imax = (imax + 1) % n
            
            candidates = [(imin, imax)]
            for _ in range(2):
                a = int(rng.integers(0, n))
                b = int(rng.integers(0, n))
                if a != b:
                    candidates.append((a, b))
        else:
            candidates = [(imin, imax)]
        
        improved = False
        for (a, b) in candidates:
            max_d = min(1.0 - cur[a], cur[b])
            if max_d < 1e-13:
                continue
            for frac in [1.0, 0.6, 0.3, 0.15, 0.07, 0.03, 0.01]:
                d = max_d * frac
                if d < 1e-13:
                    continue
                cur[a] += d
                cur[b] -= d
                s = exact_score(cur)
                if s < cur_score - 1e-14:
                    cur_score = s
                    improved = True
                    break
                cur[a] -= d
                cur[b] += d
            
            if improved:
                if cur_score < best_score:
                    best = cur.copy()
                    best_score = cur_score
                break
        
        if improved:
            no_improve = 0
        else:
            no_improve += 1
            if no_improve > 100:
                break
    
    return best, best_score
