    return (m + math.log(np.exp(sigma * (c - m)).sum()) / sigma) * dx

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)

    return grad

def binary_project(h, target):
    """Project to nearest binary sequence with exactly target ones."""
    n = len(h)
    # Sort by value, take top 'target' as 1
    idx = np.argsort(h)[::-1]
    out = np.zeros(n)

    # If target is not integer, distribute fraction
    frac = target - int(target)

    return out

    h = np.asarray(h0, dtype=float).copy()
    best_h = h.copy()
    best_c5 = objective(h)
    n = len(h)
    t0 = time.monotonic()
    step = 0.05
    t_star = 0
    while (time.monotonic() - t0) < time_limit and time.monotonic() < deadline:
        c = np.correlate(h, 1.0 - h, mode='full')
        m_idx = int(np.argmax(c))
        dx = 2.0 / n
        g = np.zeros(n)
        center = n - 1
        p = abs(m_idx - center)
        if p == 0:
            g += (1.0 - 2.0 * h)
        elif m_idx > center:
            g[p:] += (1.0 - h[:-p])
            g[:-p] -= h[p:]
        else:
            g[:-p] += (1.0 - h[p:])
            g[p:] -= h[:-p]
        g *= dx
        # Check if already at constrained optimum
        gn = np.linalg.norm(g)
        if gn < 1e-10:
            break
        improved = False
        s = step
        for _ in range(30):
            cand = project_exact(h - s * g, target)
            cv = objective(cand)
            if cv < best_c5 - 1e-15:
                h = cand
                best_c5 = cv
                best_h = cand.copy()
                improved = True
                break
            s *= 0.5
        if not improved:
            step *= 0.5
            if step < 1e-8:
                break
            h = project_exact(h + rng_local_normal(n, 0.02), target)
    return best_h, best_c5


def rng_local_normal(n, scale):
    return np.random.default_rng().normal(0, scale, n)


def make_sturmian(n, rho, phi):
    h = np.zeros(n)
    for i in range(n):
        h[i] = 1.0 if int((i + 1) * rho + phi) > int(i * rho + phi) else 0.0
    return h


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(1.0, float(budget_s) - 12.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        n0 = 132
        h0 = project_exact(rng.random(n0), n0 / 2.0)
    else:
        h0 = parent.copy()
        n0 = len(h0)
    h0 = project_exact(h0, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = objective(best_h)

    def consider(cand_h):
        nonlocal best_h, best_c5
        if cand_h is None:
            return
        try:
            arr = np.asarray(cand_h, dtype=float).ravel()
            if not arr.size:
                return
            arr = project_exact(arr, len(arr) / 2.0)
            c = objective(arr)
            if np.isfinite(c) and c < best_c5 - 1e-14:
                best_c5 = c
                best_h = arr.copy()
        except Exception:
            pass

    # Phase 1: Baseline LSE polish on parent with sigma ramp
    for sigma in [10.0, 30.0, 80.0, 200.0, 500.0]:
        if time.monotonic() > deadline - 5:
            break
        alloc = min(0.04 * effective, deadline - time.monotonic())
        if alloc < 0.5:
            break
        ch, cc = lse_polish(best_h, rng, alloc, sigma, len(best_h) / 2.0, deadline)
        consider(ch)

    # Phase 2: Multi-resolution exploration with LSE polish
    for dn in [0, 2, -2, 4, -4, 6, -6, 8, 10]:
        if time.monotonic() > deadline - 10:
            break
        nr = len(best_h) + dn
        if nr < 80 or nr > 220:
            continue
        alloc = min(0.03 * effective, deadline - time.monotonic())
        if alloc < 1.0:
            break
        rs = resample(best_h, nr)
        ch, cc = lse_polish(rs, rng, alloc, 50.0, nr / 2.0, deadline)
        if time.monotonic() < deadline - 3:
            alloc2 = min(0.015 * effective, deadline - time.monotonic())
            if alloc2 > 0.3:
                ch2, cc2 = lse_polish(ch, rng, alloc2, 300.0, nr / 2.0, deadline)
                consider(ch2)
        consider(ch)

    # Phase 3: Population DE at current best resolution
    if time.monotonic() < deadline - 20:
        alloc = min(0.18 * effective, deadline - time.monotonic())
        if alloc > 8.0:
            ch, cc = population_de(best_h, rng, alloc, len(best_h), deadline)
            consider(ch)
            if time.monotonic() < deadline - 5:
                alloc2 = min(0.04 * effective, deadline - time.monotonic())
                if alloc2 > 1.0:
                    ch2, cc2 = lse_polish(ch, rng, alloc2, 200.0, len(ch) / 2.0, deadline)
                    consider(ch2)

    # Phase 4: Binary annealing at even resolutions
    best_n = len(best_h)
    even_res = [best_n if best_n % 2 == 0 else best_n + 1]
    for k in [2, 4, 6, -2, -4]:
        r = even_res[0] + k
        if 80 <= r <= 220:
            even_res.append(r)
    for nr in even_res[:5]:
        if time.monotonic() > deadline - 12:
            break
        alloc = min(0.035 * effective, deadline - time.monotonic())
        if alloc < 2.0:
            break
        rs = resample(best_h, nr)
        ch, cc = anneal_binary(rs, rng, alloc, nr / 2.0, deadline)
        consider(ch)
        if time.monotonic() < deadline - 3:
            alloc2 = min(0.015 * effective, deadline - time.monotonic())
            if alloc2 > 0.3:
                ch2, cc2 = lse_polish(ch, rng, alloc2, 100.0, nr / 2.0, deadline)
                consider(ch2)

    # Phase 5: Sturmian starts + polish
    for trial in range(4):
        if time.monotonic() > deadline - 8:
            break
        nr = len(best_h)
        rho = 0.5 + rng.uniform(-0.04, 0.04)
        phi = rng.random()
        st = make_sturmian(nr, rho, phi)
        alpha = rng.uniform(0.3, 0.7)
        mixed = alpha * st + (1.0 - alpha) * best_h
        mixed = project_exact(mixed, nr / 2.0)
        alloc = min(0.025 * effective, deadline - time.monotonic())
        if alloc < 0.5:
            break
        ch, cc = lse_polish(mixed, rng, alloc, 60.0, nr / 2.0, deadline)
        consider(ch)

    # Phase 6: Symmetrization and transforms
    for transform_name in ['reverse', 'symmetric', 'roll1', 'roll_neg1']:
        if time.monotonic() > deadline - 6:
            break
        h_cur = best_h.copy()
        if transform_name == 'reverse':
            cand = h_cur[::-1].copy()
        elif transform_name == 'symmetric':
            cand = 0.5 * (h_cur + h_cur[::-1])
        elif transform_name == 'roll1':
            cand = np.roll(h_cur, 1)
        else:
            cand = np.roll(h_cur, -1)
        cand = project_exact(cand, len(cand) / 2.0)
        consider(cand)
        alloc = min(0.015 * effective, deadline - time.monotonic())
        if alloc > 0.3:
            ch, cc = lse_polish(cand, rng, alloc, 150.0, len(cand) / 2.0, deadline)
            consider(ch)

    # Phase 7: Final aggressive high-sigma polish
    for sigma in [300.0, 800.0, 2000.0]:
        if time.monotonic() > deadline - 1.5:
            break
        alloc = min(0.03 * effective, deadline - time.monotonic())
        if alloc < 0.3:
            break
        ch, cc = lse_polish(best_h, rng, alloc, sigma, len(best_h) / 2.0, deadline)
        consider(ch)

    # Phase 8: One last subgradient push
    if time.monotonic() < deadline - 0.5:
        alloc = min(0.01 * effective, deadline - time.monotonic())
        if alloc > 0.2:
            # Use a local subgradient approach
            h_cur = best_h.copy()
            n = len(h_cur)
            target = n / 2.0
            t_end = time.monotonic() + alloc
            step_sz = 0.03
            for _ in range(200):
                if time.monotonic() > t_end:
                    break
                c_raw = np.correlate(h_cur, 1.0 - h_cur, mode='full')
                m_idx = int(np.argmax(c_raw))
                dx = 2.0 / n
                g = np.zeros(n)
                center = n - 1
                p = abs(m_idx - center)
                if p == 0:
                    g += (1.0 - 2.0 * h_cur)
                elif m_idx > center:
                    g[p:] += (1.0 - h_cur[:-p])
                    g[:-p] -= h_cur[p:]
                else:
                    g[:-p] += (1.0 - h_cur[p:])
                    g[p:] -= h_cur[:-p]
                g *= dx
                gn = np.linalg.norm(g)
                if gn < 1e-12:
                    break
                s = step_sz
                for _ in range(15):
                    cand = project_exact(h_cur - s * g, target)
                    cv = objective(cand)
                    if cv < best_c5 - 1e-15:
                        h_cur = cand
                        best_c5 = cv
                        best_h = cand.copy()
                        break
                    s *= 0.5
                step_sz = max(step_sz * 0.95, 1e-6)

    # Final feasibility lock
    best_h = project_exact(best_h, len(best_h) / 2.0)
    best_c5 = objective(best_h)
    return best_h, float(best_c5), int(len(best_h))
