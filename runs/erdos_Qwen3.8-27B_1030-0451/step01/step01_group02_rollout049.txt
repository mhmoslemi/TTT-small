```python
def smoothed_gradient(h, tau, dx):
    n = len(h)
    C = np.correlate(h, 1-h, 'full') * dx
    M = np.max(C)
    a = tau * (C - M)
    # Clip to avoid underflow
    a = np.clip(a, -50, 50)
    w = np.exp(a)
    w_sum = np.sum(w)
    if w_sum < 1e-300:
        w = np.zeros_like(C)
        w[np.argmax(C)] = 1.0
    else:
        w /= w_sum
    
    # Compute gradient
    grad = np.zeros(n)
    ctr = n - 1  # index in C corresponding to d=0
    # Only process significant weights
    active = w > 1e-10
    idx = np.where(active)[0]
    for k in idx:
        d = k - ctr
        wk = w[k]
        if d == 0:
            grad += wk * dx * (1 - 2*h)
        elif d > 0:
            # (1-h[j+d]) for j in [0, n-d-1]
            grad[:n-d] += wk * dx * (1 - h[d:])
            # -h[j-d] for j in [d, n-1]
            grad[d:] -= wk * dx * h[:n-d]
        else:
            dd = -d
            # (1-h[j+(-dd)]) = (1-h[j-dd]) for j in [dd, n-1]
            grad[dd:] += wk * dx * (1 - h[:n-dd])
            # -h[j-(-dd)] = -h[j+dd] for j in [0, n-dd-1]
            grad[:n-dd] -= wk * dx * h[dd:]
    return grad
```
