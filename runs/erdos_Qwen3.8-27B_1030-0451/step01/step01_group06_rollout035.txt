    x = np.nan_to_num(x, nan=0.5, posinf=1.0, neginf=0.0)
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    for _ in range(20):
        res = h.sum() - target_sum
        if abs(res) < 1e-12:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0: break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0: break
                inc = min(-res, 1.0 - h[idx])
                h[idx] += inc
                res += inc
    return h

def smooth_obj_and_grad(h, tau):
    """Compute F_tau(h) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    M = float(np.max(corr))
    u = (corr - M) / tau
    u = np.clip(u, -700.0, 0.0)
    exp_u = np.exp(u)
    sum_exp = float(np.sum(exp_u))
    F = M + tau * np.log(max(sum_exp, 1e-300))
    
    w = exp_u / max(sum_exp, 1e-300)
    
    # Gradient
    g = np.zeros(n)
    # Zero lag
    g += dx * w[n-1] * (1.0 - 2.0 * h)
    
    for d in range(1, n):
        wp = w[n-1+d]  # weight for lag +d
        wm = w[n-1-d]  # weight for lag -d
        wtot = wp + wm
        if wtot < 1e-14:
            continue
        
        # For j in [0, min(d, n-d)-1]:
        #   from +d: -dx*wp*h[j+d]
        #   from -d: +dx*wm*(1-h[j+d])
        #   total: dx*(wm*(1-h[j+d]) - wp*h[j+d]) = dx*(wm - wtot*h[j+d])
        r1_end = min(d, n - d)
        if r1_end > 0:
            g[:r1_end] += dx * (wm - wtot * h[d:d+r1_end])
        
        # For j in [d, n-d-1] (if d < n-d):
        #   from +d: dx*wp*(1-h[j-d]) - dx*wp*h[j+d]
        #   from -d: dx*wm*(1-h[j+d]) - dx*wm*h[j-d]
        #   total: dx*wtot*(1 - h[j-d] - h[j+d])
        if d < n - d:
            L = n - d - d  # = n - 2d
            g[d:n-d] += dx * wtot * (1.0 - h[:L] - h[d:d+L])
        
        # For j in [n-d, n-1] (if n-d >= d, i.e., d <= n/2):
        #   from +d: dx*wp*(1-h[j-d])
        #   from -d: -dx*wm*h[j-d]
        #   total: dx*(wp*(1-h[j-d]) - wm*h[j-d]) = dx*(wp - wtot*h[j-d])
        # For j in [d, n-1] (if d > n-d, i.e., d > n/2):
        #   same formula: dx*(wp - wtot*h[j-d])
        r3_start = max(d, n - d)
        if r3_start < n:
            L = n - r3_start
            g[r3_start:n] += dx * (wp - wtot * h[r3_start-d:r3_start-d+L])
    
    return F, g

                trust = max(0.02, trust * 0.65)
                stagnant += 1
        else:
            corr_full = correlation_array(h)
            active = np.argmax(corr_full)
            d_lag = active - (n - 1)
            dx = 2.0 / n
            if d_lag == 0:
                grad = dx * (1.0 - 2.0 * h)
            elif d_lag > 0:
                grad = np.zeros(n)
                grad[d_lag:] += dx * (1.0 - h[:n - d_lag])
                grad[:n - d_lag] += -dx * h[d_lag:]
            else:
                t = -d_lag
                grad = np.zeros(n)
                grad[:n - t] += dx * (1.0 - h[t:])
                grad[t:] += -dx * h[:n - t]
            grad -= np.mean(grad)
            for alpha in [0.5, 0.25, 0.1]:
                h_new = project_feasible(h - alpha * trust * grad, target)
                val = evaluate_exact(h_new)
                if val < best_v - 1e-14:
                    best_v = val
                    best_h = h_new.copy()
                    h = h_new
                    stagnant = 0
                    break
            else:
                trust = max(0.02, trust * 0.7)
                stagnant += 1
        if stagnant >= max_stag:
            h = project_feasible(best_h + 0.08 * rng.standard_normal(n), target)
            trust = 0.15
            stagnant = 0
        if stagnant == 0:
            trust = min(0.4, trust * 1.12)
    return best_h, best_v

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 3.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).flatten()
        if h0.size > 0 and np.all(np.isfinite(h0)):
            h0 = project_feasible(h0, len(h0) / 2.0)
        else:
            raise ValueError("bad")
    except Exception:
        h0 = np.full(120, 0.5)

    best_h = h0.copy()
    best_val = evaluate_exact(best_h)
    n_base = len(h0)

    # Phase 1: SLSQP smoothed at base resolution
    for tau in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003]:
        if remaining() < 15:
            break
        h_try = slsqp_smooth(best_h, tau, maxiter=250)
        v = evaluate_exact(h_try)
        if v < best_val:
            best_val = v
            best_h = h_try

    # Phase 2: Try complement
    h_comp = project_feasible(1.0 - best_h, len(best_h) / 2.0)
    v_comp = evaluate_exact(h_comp)
    if v_comp < best_val:
        best_val = v_comp
        best_h = h_comp

    # Phase 3: SLSQP after complement
    for tau in [0.02, 0.01, 0.005, 0.002]:
        if remaining() < 15:
            break
        h_try = slsqp_smooth(best_h, tau, maxiter=200)
        v = evaluate_exact(h_try)
        if v < best_val:
            best_val = v
            best_h = h_try

    # Phase 4: Upscale to 2x and optimize
    if remaining() > 120:
        h_up = np.repeat(best_h, 2)
        h_up = project_feasible(h_up, len(h_up) / 2.0)
        for tau in [0.03, 0.015, 0.008, 0.004, 0.002]:
            if remaining() < 15:
                break
            h_try = slsqp_smooth(h_up, tau, maxiter=200)
            h_up = h_try
        v_up = evaluate_exact(h_up)
        if v_up < best_val:
            best_val = v_up
            best_h = h_up
        else:
            h_dn = h_up.reshape(len(best_h), 2).mean(axis=1)
            h_dn = project_feasible(h_dn, len(best_h) / 2.0)
            v_dn = evaluate_exact(h_dn)
            if v_dn < best_val:
                best_val = v_dn
                best_h = h_dn

    # Phase 5: LP-based local refinement at best resolution
    if remaining() > 60:
        h_lp, v_lp = optimize_lp(best_h.copy(), best_val, remaining, rng)
        if v_lp < best_val:
            best_val = v_lp
            best_h = h_lp

    # Phase 6: Momentum subgradient on exact objective
    if remaining() > 40:
        h = best_h.copy()
        n_cur = len(h)
        target_cur = n_cur / 2.0
        momentum = 0.3
        lr = 0.03
        for it in range(300):
            if remaining() < 10:
                break
            h, grad, cur_max, corr = subgradient_exact(h, momentum, lr)
            if cur_max < best_val:
                best_val = cur_max
                best_h = h.copy()
            h_new = project_feasible(h - lr * grad, target_cur)
            v_new = evaluate_exact(h_new)
            if v_new < best_val - 1e-14:
                h = h_new
                best_val = v_new
                best_h = h_new.copy()
                lr = min(0.08, lr * 1.05)
            else:
                if it % 20 == 0:
                    h = project_feasible(best_h + 0.03 * rng.standard_normal(n_cur), target_cur)
                    lr = max(0.01, lr * 0.8)

    # Phase 7: Final LP refinement
    if remaining() > 30:
        h_lp, v_lp = optimize_lp(best_h.copy(), best_val, remaining, rng)
        if v_lp < best_val:
            best_val = v_lp
            best_h = h_lp

    # Phase 8: Try a few diverse restarts
    if remaining() > 20:
        n_cur = len(best_h)
        target_cur = n_cur / 2.0
        restarts = [
            project_feasible(best_h + 0.1 * rng.standard_normal(n_cur), target_cur),
            project_feasible(1.0 - best_h + 0.05 * rng.standard_normal(n_cur), target_cur),
            project_feasible(best_h + 0.2 * rng.standard_normal(n_cur), target_cur),
        ]
        for hr in restarts:
            if remaining() < 15:
                break
            h_r, v_r = optimize_lp(hr, best_val, remaining, rng)
            if v_r < best_val:
                best_val = v_r
                best_h = h_r

    # Final
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)
    return best_h, float(best_val), len(best_h)
