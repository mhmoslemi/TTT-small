def lse_weights(corr, tau):
    M = np.max(corr)
    u = (corr - M) / tau
    u = np.clip(u, -700.0, 700.0)
    exp_u = np.exp(u)
    Z = np.sum(exp_u)
    if Z < 1e-300:
        return np.full(len(corr), 1.0 / len(corr)), M
    return exp_u / Z, M

def gradient_from_weights(h, w):
    n = len(h)
    dx = 2.0 / n
    m = 2 * n - 1
    center = n - 1
    grad = np.zeros(n)
    for r in range(m):
        wr = w[r]
        if wr < 1e-12:
            continue
        k = r - center
        if k == 0:
            grad += wr * dx * (1.0 - 2.0 * h)
        elif k > 0:
            grad[k:] += wr * dx * (1.0 - h[:n-k])
            grad[:n-k] -= wr * dx * h[k:]
        else:
            d = -k
            grad[d:] += wr * dx * (1.0 - h[:n-d])
            grad[:n-d] -= wr * dx * h[d:]
    return grad

def slsqp_lse(h0, tau, maxiter=100):

    target = n / 2.0
    
    def fun_and_jac(x):

        w, M = lse_weights(corr, tau)
        F = M + tau * np.log(np.sum(np.exp(np.clip((corr - M) / tau, -700, 700))))
        g = gradient_from_weights(x, w)
        return F, g
    
    def fun(x):
        return fun_and_jac(x)[0]
    
    def jac(x):
        return fun_and_jac(x)[1]
    
    cons = [{'type': 'eq', 'fun': lambda x: x.sum() - target,
             'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    
    try:
        res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': maxiter, 'ftol': 1e-12})
        if res.x is not None and np.min(res.x) >= -0.01 and np.max(res.x) <= 1.01:
            return project_feasible(res.x, target)
    except Exception:
        pass
    return None

def lp_refine(h, max_iter=50, trust_max=0.3):
    """Active-set LP refinement."""
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_v = evaluate(h)
    trust = 0.15
    stagnant = 0
    
    for it in range(max_iter):
        corr = correlation_array(best_h)
        cur_max = float(np.max(corr))
        if cur_max < best_v:
            best_v = cur_max
        
        # Select active lags (top 30%)
        m = 2 * n - 1
        k_active = max(1, int(m * 0.3))
        top_idx = np.argpartition(corr, -k_active)[-k_active:]
        
        # Solve LP: minimize t s.t. grad_k · d ≤ -corr_k + t, |d| ≤ trust, sum(d) = 0, 0≤h+d≤1
        nvars = n + 1
        c_obj = np.zeros(nvars)
        c_obj[n] = 1.0
        
        center = n - 1
        A_ub = np.zeros((k_active, nvars))
        for i, r in enumerate(top_idx):
            k = r - center
            if k == 0:
                A_ub[i, :n] = (1.0 - 2.0 * best_h) * (2.0 / n)
            elif k > 0:
                A_ub[i, k:] -= best_h[:n-k] * (2.0 / n)
                A_ub[i, :n-k] += (1.0 - best_h[k:]) * (2.0 / n)
            else:
                d_ = -k
                A_ub[i, d_:] -= best_h[:n-d_] * (2.0 / n)
                A_ub[i, :n-d_] += (1.0 - best_h[d_:]) * (2.0 / n)
            A_ub[i, n] = -1.0
        b_ub = -corr[top_idx]
        
        lower = np.maximum(-best_h, -trust)
        upper = np.minimum(1.0 - best_h, trust)
        
        A_eq = np.zeros((1, nvars))
        A_eq[0, :n] = 1.0
        b_eq = np.array([0.0])
        
        bounds = list(zip(lower, upper)) + [(None, None)]
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if res.status == 0:
                d = res.x[:n]
                if np.max(np.abs(d)) > 1e-12:
                    accepted = False
                    for alpha in [1.0, 0.7, 0.5, 0.3, 0.1]:
                        h_new = best_h + alpha * d
                        if np.any(h_new < -1e-9) or np.any(h_new > 1 + 1e-9):
                            continue
                        val = evaluate(h_new)
                        if val < best_v - 1e-14:
                            best_v = val
                            best_h = h_new.copy()
                            accepted = True
                            stagnant = 0
                            break
                    if not accepted:
                        trust = max(0.01, trust * 0.7)
                        stagnant += 1
                else:
                    stagnant += 1
            else:
                stagnant += 1
        except Exception:
            stagnant += 1
        
        if stagnant >= 15:
            break
        if stagnant == 0:
            trust = min(trust_max, trust * 1.2)
    
    return best_h, best_v

            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                          bounds=bounds, method='highs')
            if res.status == 0:
                d = res.x[:n]
                if np.max(np.abs(d)) > 1e-13:
                    accepted = False
                    for alpha in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1]:
                        h_new = best_h + alpha * d
                        if np.any(h_new < -1e-9) or np.any(h_new > 1 + 1e-9):
                            continue
                        val = evaluate(h_new)
                        if val < best_v - 1e-14:
                            best_v = val
                            best_h = h_new.copy()
                            accepted = True
                            stagnant = 0
                            break
                    if not accepted:
                        trust = max(0.005, trust * 0.7)
                        stagnant += 1
                else:
                    stagnant += 1
            else:
                stagnant += 1
        except Exception:
            stagnant += 1
        if stagnant >= 20:
            break
        if stagnant == 0:
            trust = min(trust_max, trust * 1.15)
    return best_h, best_v

def binary_greedy(n, rng, max_steps=5000):
    h = np.zeros(n)
    ones = rng.choice(n, size=n // 2, replace=False)
    h[ones] = 1.0
    best_v = evaluate(h)
    stalls = 0
    for step in range(max_steps):
        improved = False
        n_zeros = np.where(h < 0.5)[0]
        n_ones = np.where(h > 0.5)[0]
        if len(n_zeros) == 0 or len(n_ones) == 0:
            break
        best_gain = best_v
        best_pair = None
        n_try = min(200, len(n_zeros) * len(n_ones))
        for _ in range(n_try):
            i = n_zeros[rng.integers(len(n_zeros))]
            j = n_ones[rng.integers(len(n_ones))]
            h[i], h[j] = 1.0, 0.0
            v = evaluate(h)
            h[i], h[j] = 0.0, 1.0
            if v < best_gain - 1e-14:
                best_gain = v
                best_pair = (i, j)
        if best_pair is not None:
            i, j = best_pair
            h[i], h[j] = 1.0, 0.0
            best_v = best_gain
            stalls = 0
        else:
            stalls += 1
            if stalls > 200:
                break
    return h, best_v

def multi_resolve(h, factor, budget_fn, rng):
    n = len(h)
    if n * factor > 2500:
        return h
    h_up = np.repeat(h, factor)
    h_up = project_feasible(h_up, len(h_up) / 2.0)
    h_ref, v_ref = lp_refine(h_up, max_iter=40, trust_start=0.1, trust_max=0.2)
    new_n = len(h_ref)
    if new_n % factor == 0:
        h_dn = h_ref[:n * factor].reshape(n, factor).mean(axis=1)
        h_dn = project_feasible(h_dn, n / 2.0)
        return h_dn
    return h

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    deadline = t_start + budget_s - 3.0
    rng = np.random.default_rng(seed)

    def remaining():
        return deadline - time.time()

    h0 = None
    try:
        h0 = np.asarray(initial_h_values, dtype=np.float64).flatten().copy()
        if h0.size > 0 and np.all(np.isfinite(h0)):
            h0 = project_feasible(h0, len(h0) / 2.0)
        else:
            h0 = None
    except Exception:
        h0 = None

    if h0 is None:
        h0 = np.full(120, 0.5)

    best_h = h0.copy()
    best_val = evaluate(best_h)

    # Phase 1: SLSQP continuation from parent
    if remaining() > 30:
        h_curr = best_h.copy()
        for tau in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002]:
            if remaining() < 5:
                break
            result = slsqp_lse(h_curr, tau, maxiter=150)
            if result is not None:
                v = evaluate(result)
                if v < best_val:
                    best_val = v
                    best_h = result.copy()
                h_curr = result.copy()

    # Phase 2: LP refinement
    if remaining() > 15:
        h_ref, v_ref = lp_refine(best_h.copy(), max_iter=80, trust_start=0.15, trust_max=0.3)
        if v_ref < best_val:
            best_val = v_ref
            best_h = h_ref.copy()

    # Phase 3: Diverse SLSQP starts
    n = len(best_h)
    target = n / 2.0
    if remaining() > 60:
        starts = [project_feasible(1.0 - best_h, target),
                  best_h[::-1].copy(),
                  project_feasible(best_h + 0.2 * rng.standard_normal(n), target),
                  project_feasible(best_h + 0.3 * rng.standard_normal(n), target),
                  project_feasible(best_h + 0.1 * rng.standard_normal(n), target)]
        for s in starts:
            if remaining() < 10:
                break
            r = slsqp_lse(s, 0.01, maxiter=120)
            if r is not None:
                v = evaluate(r)
                if v < best_val:
                    best_val = v
                    best_h = r.copy()
            h_ref, v_ref = lp_refine(s, max_iter=40, trust_start=0.1, trust_max=0.2)
            if v_ref < best_val:
                best_val = v_ref
                best_h = h_ref.copy()

    # Phase 4: Binary greedy
    if remaining() > 60:
        for trial in range(4):
            if remaining() < 10:
                break
            if trial == 0:
                hb = (best_h > 0.5).astype(float)
                hb = project_feasible(hb, target)
            else:
                hb, vb = binary_greedy(n, rng, max_steps=2000)
                if vb < best_val + 0.01:
                    h_cont = project_feasible(hb + 0.5 * rng.standard_normal(n) * 0.1, target)
                    h_ref, v_ref = lp_refine(h_cont, max_iter=40, trust_start=0.1, trust_max=0.2)
                    if v_ref < best_val:
                        best_val = v_ref
                        best_h = h_ref.copy()
                    r = slsqp_lse(hb, 0.02, maxiter=100)
                    if r is not None:
                        v = evaluate(r)
                        if v < best_val:
                            best_val = v
                            best_h = r.copy()

    # Phase 5: Multi-resolution
    if remaining() > 60:
        h_mr = multi_resolve(best_h, 2, remaining, rng)
        if len(h_mr) == n:
            h_mr = project_feasible(h_mr, target)
            h_ref, v_ref = lp_refine(h_mr, max_iter=30, trust_start=0.05, trust_max=0.1)
            if v_ref < best_val:
                best_val = v_ref
                best_h = h_ref.copy()

    # Phase 6: Final fine LP
    if remaining() > 10:
        h_ref, v_ref = lp_refine(best_h.copy(), max_iter=60, trust_start=0.05, trust_max=0.1)
        if v_ref < best_val:
            best_val = v_ref
            best_h = h_ref.copy()

    # Phase 7: Very fine LP
    if remaining() > 5:
        h_ref, v_ref = lp_refine(best_h.copy(), max_iter=40, trust_start=0.02, trust_max=0.05)
        if v_ref < best_val:
            best_val = v_ref
            best_h = h_ref.copy()

    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate(best_h)
    return best_h, float(best_val), len(best_h)
