def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        k = K - (n - 1)
        if k >= 0:
            g[k:] += w * (1.0 - h[:n - k])
            g[:n - k] += w * (-h[k:])
        else:
            m = -k
            g[:n - m] += w * (1.0 - h[m:])
            g[m:] += w * (-h[:n - m])
    return g

For h = [a, b, c], 1-h = [1-a, 1-b, 1-c]:
np.correlate(h, 1-h, 'full') should give:
- Index 0: h[0]*(1-h[2]) = a*(1-c) ... wait no.

Let me look at the numpy docs more carefully.

np.correlate(a, v) = sum_n a[n+k] * conj(v[n])

For mode='full', k ranges from -(len(v)-1) to len(a)-1.

So for a=h=[a0,a1,a2], v=1-h=[v0,v1,v2]:
- k=-2: sum_n h[n-2]*v[n], valid n: n=2. Result: h[0]*v[2] = a0*(1-c)
- k=-1: sum_n h[n-1]*v[n], valid n: n=1,2. Result: h[0]*v[1] + h[1]*v[2] = a0*(1-b) + a1*(1-c)
- k=0: sum_n h[n]*v[n], valid n: n=0,1,2. Result: a0*v0 + a1*v1 + a2*v2 = a0(1-a0) + a1(1-a1) + a2(1-a2)
- k=1: sum_n h[n+1]*v[n], valid n: n=0,1. Result: h[1]*v[0] + h[2]*v[1] = a1*(1-a0) + a2*(1-a1)
- k=2: sum_n h[n+2]*v[n], valid n: n=0. Result: h[2]*v[0] = a2*(1-a0)

The full result has 2*3-1 = 5 elements, indexed 0..4, corresponding to k=-2,-1,0,1,2.

So the mapping is: index i in the output corresponds to k = i - (len(v)-1) = i - 2.

In general for two arrays of length n:
index i corresponds to k = i - (n-1)
output[i] = sum over valid n' of h[n'+k] * (1-h[n'])

where k = i - (n-1).

For k >= 0: output[i] = sum_{n'=0}^{n-1-k} h[n'+k] * (1-h[n'])
For k < 0 (let m = -k): output[i] = sum_{n'=m}^{n-1} h[n'-m] * (1-h[n']) = sum_{n'=0}^{n-1-m} h[n'] * (1-h[n'+m])

OK so my earlier derivation was correct. Let me verify the perturbation formula for the k>=0 case:

output[i](t) = sum_{n'=0}^{n-1-k} (h[n'+k] + t*d[n'+k]) * (1 - h[n'] - t*d[n'])
= sum (h[n'+k](1-h[n']) + t*(d[n'+k](1-h[n']) - h[n'+k]*d[n']) + t²*(-d[n'+k]*d[n']))

So Lin[i] = sum_{n'=0}^{n-1-k} (d[n'+k]*(1-h[n']) - h[n'+k]*d[n'])
   Quad[i] = -sum_{n'+k -k}^{n-1-k... } Let me just say: Quad[i] = -sum_{n'=0}^{n-1-k} d[n'+k]*d[n']

For k < 0 (m = -k):
output[i](t) = sum_{n'=0}^{n-1-m} (h[n'] + t*d[n']) * (1 - h[n'+m] - t*d[n'+m])
= sum (h[n'](1-h[n'+m]) + t*(d[n'](1-h[n'+m]) - h[n']*d[n'+m]) + t²*(-d[n']*d[n'+m]))

So Lin[i] = sum_{n'=0}^{n-1-m} (d[n']*(1-h[n'+m]) - h[n']*d[n'+m])
   Quad[i] = -sum_{n'=0}^{n-1-m} d[n']*d[n'+m]

import numpy as np
from scipy.optimize import minimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def project_feasible(z, n):
    S = n / 2.0
    lo, hi = -20.0, 20.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-12:
        interior = (h > 1e-12) & (h < 1.0 - 1e-12)
        cnt = int(np.sum(interior))
        if cnt > 0:
            h[interior] += diff / cnt
            h = np.clip(h, 0.0, 1.0)
    return h

def compute_line_coefs(h, d):
    n = len(h)
    L = 2 * n - 1
    R0 = np.correlate(h, 1.0 - h, mode='full')
    Lin = np.zeros(L)
    Quad = np.zeros(L)
    center = n - 1
    for s in range(-(n-1), n):
        K = s + center
        if s >= 0:
            m = n - s
            idx = np.arange(m)
            Lin[K] = np.sum(d[idx + s] * (1.0 - h[idx]) - h[idx + s] * d[idx])
            Quad[K] = -np.sum(d[idx + s] * d[idx])
        else:
            m = n + s  # n - |s| = n - (-s) = n + s
            idx = np.arange(m)
            Lin[K] = np.sum(d[idx] * (1.0 - h[idx + (-s)]) - h[idx] * d[idx + (-s)])
            Quad[K] = -np.sum(d[idx] * d[idx + (-s)])
    return R0, Lin, Quad

def feasible_interval(h, d):
    n = len(h)
    t_lo = -np.inf
    t_hi = np.inf
    pos = d > 1e-14
    neg = d < -1e-14
    if np.any(pos):
        t_lo = max(t_lo, np.max(-h[pos] / d[pos]))
        t_hi = min(t_hi, np.min((1.0 - h[pos]) / d[pos]))
    if np.any(neg):
        t_lo = max(t_lo, np.max((1.0 - h[neg]) / d[neg]))
        t_hi = min(t_hi, np.min(-h[neg] / d[neg]))
    return t_lo, t_hi

def line_search(h, d, n_samples=200):
    n = len(h)
    dx = 2.0 / n
    t_lo, t_hi = feasible_interval(h, d)
    if t_hi - t_lo < 1e-12:
        return None, None
    
    # Add small safety margin
    margin = 0.01 * (t_hi - t_lo) + 1e-10
    t_lo_s = t_lo + margin
    t_hi_s = t_hi - margin
    if t_hi_s <= t_lo_s:
        return None, None
    
    R0, Lin, Quad = compute_line_coefs(h, d)
    
    # Sample
    t_samples = np.linspace(t_lo_s, t_hi_s, n_samples)
    best_t = 0.0
    best_val = np.max(R0) * dx  # F(0)
    
    # Evaluate at samples
    for t in t_samples:
        vals = R0 + t * Lin + t * t * Quad
        f = np.max(vals) * dx
        if f < best_val:
            best_val = f
            best_t = t
    
    # Check vertices of top active lags
    active = np.argsort(R0)[::-1][:30]
    for K in active:
        if Quad[K] < -1e-14:
            tv = -Lin[K] / (2.0 * Quad[K])
            if t_lo_s < tv < t_hi_s:
                vals = R0 + tv * Lin + tv * tv * Quad
                f = np.max(vals) * dx
                if f < best_val:
                    best_val = f
                    best_t = tv
    
    # Refine around best_t with more samples
    if best_t != 0.0:
        span = max(0.1 * abs(best_t), 0.01 * (t_hi_s - t_lo_s))
        t_refine = np.linspace(best_t - span, best_t + span, 100)
        t_refine = t_refine[(t_refine >= t_lo_s) & (t_refine <= t_hi_s)]
        for t in t_refine:
            vals = R0 + t * Lin + t * t * Quad
            f = np.max(vals) * dx
            if f < best_val:
                best_val = f
                best_t = t
    
    if best_val < np.max(R0) * dx - 1e-14:
        h_new = h + best_t * d
        h_new = np.clip(h_new, 0.0, 1.0)
        # Fix sum
        h_new = project_feasible(h_new, n)
        return h_new, best_val
    return None, None

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    c = raw * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -500, 500)
    weights = np.exp(exponents)
    sw = np.sum(weights)
    if sw < 1e-300:
        return np.zeros(n)
    weights /= sw
    
    g = np.zeros(n)
    for K in range(2 * n - 1):
        w = weights[K] * dx
        if w < 1e-14:
            continue
        k = K - (n - 1)
        if k >= 0:
            idx = np.arange(n - k)
            g[idx + k] += w * (1.0 - h[idx])
            g[idx] += w * (-h[idx + k])
        else:
            m = -k
            idx = np.arange(n - m)
            g[idx] += w * (1.0 - h[idx + m])
            g[idx + m] += w * (-h[idx])
    return g

def fw_vertex(h, g, n):
    S = n / 2.0
    order = np.argsort(g)
    v = np.zeros(n)
    S_int = int(np.floor(S))
    S_frac = S - S_int
    for i in range(S_int):
        v[order[i]] = 1.0
    if S_frac > 1e-12:
        v[order[S_int]] = S_frac
    return v

def interpolate_h(h_old, new_n):
    n_old = len(h_old)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(new_n) + 0.5) / new_n
    h_interp = np.interp(x_new, x_old, h_old)

def make_structured(n, rng):
    """Create a structured initial guess."""
    x = np.arange(n) / n * 2.0
    # Try a few different structures
    type = rng.integers(0, 4)
    if type == 0:
        # Semi-periodic
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x * 3.0) + 0.1 * rng.standard_normal(n)
    elif type == 1:
        # Block structure
        h = np.zeros(n)
        block_size = n // 4
        for b in range(4):
            if b % 2 == 0:
                h[b*block_size:(b+1)*block_size] = 0.8
            else:
                h[b*block_size:(b+1)*block_size] = 0.2
        h += 0.05 * rng.standard_normal(n)
    elif type == 2:
        # Gaussian bump
        center = n // 2
        h = 0.3 + 0.5 * np.exp(-0.5 * ((np.arange(n) - center) / (n/6.0))**2)
        h += 0.05 * rng.standard_normal(n)
    else:
        # Uniform + noise
        h = np.full(n, 0.5) + 0.1 * rng.standard_normal(n)
    return project_feasible(h, n)
