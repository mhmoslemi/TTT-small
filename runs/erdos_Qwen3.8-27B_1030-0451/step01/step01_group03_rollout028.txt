import numpy as np
from scipy.optimize import minimize
import time

_GN = 0
_GTAU = 0.01

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))

def project(y, target):
    lo, hi = -3.0, 3.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def upsample(h, n_new):
    n_old = len(h)
    if n_old == n_new:
        return h.copy()
    xo = np.arange(n_old, dtype=np.float64)
    xn = np.linspace(0.0, n_old, n_new, endpoint=False)
    h_new = np.interp(xn, xo, h)
    return project(h_new, n_new / 2.0)

def lsp_obj(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _GTAU, -500, 500)
    e = np.exp(s)
    return m + _GTAU * np.log(e.sum())

def lsp_grad(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _GTAU, -500, 500)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    active = np.where(alpha > 1e-8)[0]
    for k in active:
        a = alpha[k]
        d = int(k - (n - 1))
        if d >= 0:
            idx = np.arange(0, n - d)
            grad[idx] += a * (1.0 - h[idx + d])
            grad[idx + d] -= a * h[idx]
        else:
            m2 = -d
            idx = np.arange(0, n - m2)
            grad[idx + m2] += a * (1.0 - h[idx])
            grad[idx] -= a * h[idx + m2]
    return grad * dx

def topk_obj_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    K = min(K, len(c))
    top_idx = np.argpartition(c, -K)[-K:]
    val = c[top_idx].sum() / K
    grad = np.zeros(n)
    for k in top_idx:
        d = int(k - (n - 1))
        if d >= 0:
            idx = np.arange(0, n - d)
            grad[idx] += (1.0 - h[idx + d])
            grad[idx + d] -= h[idx]
        else:
            m2 = -d
            idx = np.arange(0, n - m2)
            grad[idx + m2] += (1.0 - h[idx])
            grad[idx] -= h[idx + m2]
    return val, grad * dx / K

    tau_sched = [max(0.0001, t) for t in tau_sched]

    g_prev = None

                break
            g = lsp_grad(h)
            g_norm = np.linalg.norm(g)
            if g_norm < 1e-12:
                break

                s_vec = h - h_prev
                y_vec = g - g_prev
                sy = s_vec.dot(y_vec)
                if sy > 1e-14:
                    step = (s_vec.dot(s_vec)) / sy
                    step = min(max(step, 1e-8), 1.0 / g_norm + 1e-8)
                else:
                    step = 1.0 / g_norm
            else:

                if obj_try <= obj_cur:

                step *= 0.5

            g_prev = g.copy()
            h = h_try

def topk_grad(h, K):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    top_idx = np.argpartition(c, -K)[-K:]
    grad = np.zeros(n)
    for k in top_idx:
        d = int(k - (n - 1))
        if d >= 0:
            i1 = np.arange(d, n)
            grad[i1] += (1.0 - h[i1 - d])
            i2 = np.arange(0, n - d)
            grad[i2] -= h[i2 + d]
        else:
            m2 = -d
            i1 = np.arange(0, n - m2)
            grad[i1] += (1.0 - h[i1 + m2])
            i2 = np.arange(m2, n)
            grad[i2] -= h[i2 - m2]
    return grad * dx / K
