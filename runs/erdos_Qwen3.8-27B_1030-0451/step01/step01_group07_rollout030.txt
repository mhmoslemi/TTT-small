    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            slack = out.copy()
            total = slack.sum()
            if total > 1e-15:
                out = out - (d / total) * slack
        else:
            slack = 1.0 - out
            total = slack.sum()
            if total > 1e-15:
                out = out + (-d / total) * slack
    out = np.clip(out, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-10:
        idx = int(np.argmax(out)) if d > 0 else int(np.argmin(out))
        out = out.copy()
        out[idx] = float(np.clip(out[idx] - d, 0.0, 1.0))
    return out


def objective(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def corr_full(h):
    return np.correlate(h, 1.0 - h, mode='full')


def lse_val(h, sigma):
    c = corr_full(h)
    m = c.max()
    val = m + math.log(np.sum(np.exp(sigma * (c - m)))) / sigma
    n = len(h)
    return (2.0 / n) * val


def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    for p in range(n):
        wp = w[center + p] if center + p < len(w) else 0.0
        wm = w[center - p] if center - p >= 0 else 0.0
        if p == 0:
            if wp > 1e-18:
                grad += wp * (1.0 - 2.0 * h)
        else:
            if wp > 1e-18:
                grad[p:] += wp * (1.0 - h[:-p])
                grad[:-p] -= wp * h[p:]
            if wm > 1e-18:
                grad[:-p] += wm * (1.0 - h[p:])
                grad[p:] -= wm * h[:-p]
    grad *= dx
    return grad


def subgrad_direct(h, rel_tol=0.02):
    n = len(h)
    dx = 2.0 / n
    c = corr_full(h)
    M = c.max()
    if M < 1e-15:
        return np.zeros(n)
    active = np.where(c > M * (1.0 - rel_tol))[0]
    grad = np.zeros(n)
    center = n - 1
    for idx in active:
        p = idx - center
        if p == 0:
            grad += (1.0 - 2.0 * h)
        elif p > 0:
            grad[p:] += (1.0 - h[:-p])
            grad[:-p] -= h[p:]
        else:
            q = -p
            grad[:-q] += (1.0 - h[q:])
            grad[q:] -= h[:-q]
    return grad * (dx / len(active))


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def lse_optimize(h0, rng, time_limit, sigma, deadline):
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    cur = lse_val(h, sigma)
    best_h = h.copy()
    best_c = objective(h)
    eta = 0.05
    t0 = time.monotonic()
    stall = 0
    while time.monotonic() < deadline and (time.monotonic() - t0) < time_limit:
        g = lse_gradient(h, sigma)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        step = eta
        for _ in range(20):
            cand = project_exact(h - step * g, target)
            cv = lse_val(cand, sigma)
            if cv < cur - 1e-14:
                h = cand
                cur = cv
                c5 = objective(h)
                if c5 < best_c - 1e-13:
                    best_c = c5
                    best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            kick_size = 0.02 * (1.0 + stall * 0.3)
            h = project_exact(h + rng.normal(0.0, kick_size, n), target)
            cur = lse_val(h, sigma)
            c5 = objective(h)
            if c5 < best_c - 1e-13:
                best_c = c5
                best_h = h.copy()
            stall += 1
            if stall >= 8:
                eta *= 0.5
                stall = 0
                if eta < 1e-7:
                    break
    c5 = objective(h)
    if c5 < best_c - 1e-13:
        best_c = c5
        best_h = h.copy()
    return best_h, best_c


def subgrad_optimize(h0, rng, time_limit, deadline):
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = objective(h)
    eta = 0.03
    t0 = time.monotonic()
    stall = 0
    tol_schedule = [0.05, 0.03, 0.01, 0.005, 0.002]
    tol_idx = 0
    while time.monotonic() < deadline and (time.monotonic() - t0) < time_limit:
        rel_tol = tol_schedule[min(tol_idx, len(tol_schedule) - 1)]
        g = subgrad_direct(h, rel_tol)
        gn = float(np.linalg.norm(g))
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        step = eta
        for _ in range(20):
            cand = project_exact(h - step * g, target)
            c5 = objective(cand)
            if c5 < best_c - 1e-14 or c5 < objective(h) - 1e-14:
                h = cand
                if c5 < best_c - 1e-13:
                    best_c = c5
                    best_h = h.copy()
                improved = True
                break
            step *= 0.5
        if not improved:
            h = project_exact(h + rng.normal(0.0, 0.03, n), target)
            c5 = objective(h)
            if c5 < best_c - 1e-13:
                best_c = c5
                best_h = h.copy()
            stall += 1
            if stall >= 10:
                eta *= 0.6
                stall = 0
                tol_idx += 1
                if eta < 1e-7:
                    break
    return best_h, best_c


def targeted_local_search(h0, rng, time_limit, deadline):
    """Local search that targets the worst lags with directed swaps."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    best_h = h.copy()
    best_c = objective(h)
    t0 = time.monotonic()
    no_improve = 0
    while time.monotonic() < deadline and (time.monotonic() - t0) < time_limit:
        c = corr_full(h)
        cur_max = c.max()
        sorted_idx = np.argsort(c)[::-1]
        worst_k = sorted_idx[:5]
        improved = False
        for k_idx in worst_k:
            p = k_idx - (n - 1)
            if p == 0:
                # Try to make h more binary (increase sum of squares)
                for trial in range(50):
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    cand = h.copy()
                    # Push i and j apart (one more 0, other more 1)
                    if h[i] < h[j]:
                        d = min(h[j] - h[i], 0.5) * rng.random()
                        if d < 1e-6:
                            continue
                        cand[i] -= d
                        cand[j] += d
                    else:
                        d = min(h[i] - h[j], 0.5) * rng.random()
                        if d < 1e-6:
                            continue
                        cand[i] += d
                        cand[j] -= d
                    cand = project_exact(cand, target)
                    nc = np.correlate(cand, 1.0 - cand, mode='full').max() * (2.0 / n)
                    if nc < best_c - 1e-12:
                        h = cand
                        best_c = nc
                        best_h = cand.copy()
                        improved = True
                        break
            elif p > 0:
                # c at lag p: sum h[i+p](1-h[i])
                # Want to reduce: find i where h[i] small, h[i+p] large
                for trial in range(80):
                    i = int(rng.integers(n - p))
                    if h[i] < 0.15 and h[i + p] > 0.85:
                        cand = h.copy()
                        amt = min(0.5, (h[i+p] - h[i]) * 0.5)
                        cand[i] += amt
                        cand[i + p] -= amt
                        cand = project_exact(cand, target)
                        nc = np.correlate(cand, 1.0 - cand, mode='full').max() * (2.0 / n)
                        if nc < best_c - 1e-12:
                            h = cand
                            best_c = nc
                            best_h = cand.copy()
                            improved = True
                            break
                    elif h[i] > 0.85 and h[i + p] < 0.15:
                        cand = h.copy()
                        amt = min(0.3, (h[i] - h[i+p]) * 0.3)
                        cand[i] -= amt
                        cand[i + p] += amt
                        cand = project_exact(cand, target)
                        nc = np.correlate(cand, 1.0 - cand, mode='full').max() * (2.0 / n)
                        if nc < best_c - 1e-12:
                            h = cand
                            best_c = nc
                            best_h = cand.copy()
                            improved = True
                            break
            else:
                q = -p
                # c at lag -q: sum h[j](1-h[j+q])
                for trial in range(80):
                    j = int(rng.integers(n - q))
                    if h[j] > 0.85 and h[j + q] < 0.15:
                        cand = h.copy()
                        amt = min(0.5, (h[j] - h[j+q]) * 0.5)
                        cand[j] -= amt
                        cand[j + q] += amt
                        cand = project_exact(cand, target)
                        nc = np.correlate(cand, 1.0 - cand, mode='full').max() * (2.0 / n)
                        if nc < best_c - 1e-12:
                            h = cand
                            best_c = nc
                            best_h = cand.copy()
                            improved = True
                            break
        if not improved:
            no_improve += 1
            kick = 0.03 * np.log1p(no_improve * 0.5)
            h = project_exact(h + rng.normal(0.0, kick, n), target)
            c5 = objective(h)
            if c5 < best_c - 1e-13:
                best_c = c5
                best_h = h.copy()
            if no_improve > 200:
                break
        else:
            no_improve = 0
    return best_h, best_c


def iterated_local_search(h0, rng, time_limit, deadline):
    """ILS: optimize, kick, re-optimize, keep best."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    best_h = h.copy()
    best_c = objective(h)
    kick_sizes = [0.05, 0.08, 0.12, 0.18, 0.25, 0.35, 0.5]
    ki = 0
    t0 = time.monotonic()
    while time.monotonic() < deadline and (time.monotonic() - t0) < time_limit:
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break
        # Optimize current
        alloc = min(remaining * 0.4, 60.0)
        if alloc > 1.0:
            h2, c2 = lse_optimize(h, rng, alloc, 80.0, time.monotonic() + alloc)
            if c2 < best_c - 1e-13:
                best_c = c2
                best_h = h2.copy()
                h = h2.copy()
        # Kick
        if time.monotonic() >= deadline - 2.0:
            break
        ks = kick_sizes[min(ki, len(kick_sizes) - 1)]
        h = project_exact(h + rng.normal(0.0, ks, n), n / 2.0)
        # Re-optimize after kick
        remaining = deadline - time.monotonic()
        if remaining < 2.0:
            break
        alloc = min(remaining * 0.4, 60.0)
        if alloc > 1.0:
            h2, c2 = lse_optimize(h, rng, alloc, 80.0, time.monotonic() + alloc)
            if c2 < best_c - 1e-13:
                best_c = c2
                best_h = h2.copy()
        ki += 1
    return best_h, best_c


def anneal_local(h0, rng, time_limit, deadline):
    """Fast simulated annealing with swap-based moves."""
    h = np.asarray(h0, dtype=float).copy()
    n = len(h)
    target = n / 2.0
    cur = objective(h)
    best_h = h.copy()
    best_c = cur
    T = 5e-5
    t0 = time.monotonic()
    total = max(time_limit, 1e-9)
    while time.monotonic() < deadline and (time.monotonic() - t0) < time_limit:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        cand = h.copy()
        if rng.random() < 0.5:
            # Transfer
            d = rng.random() * 0.4 * (1.0 if h[i] >= h[j] else -1.0)
            if h[i] + d < 0 or h[j] - d < 0 or h[i] + d > 1 or h[j] - d > 1:
                d = 0.5 * np.sign(h[i] - h[j]) * min(abs(h[i] - h[j]), rng.random() * 0.3)
            cand[i] += d
            cand[j] -= d
        else:
            # Swap
            cand[i], cand[j] = cand[j], cand[i]
        cand = project_exact(cand, target)
        nc = objective(cand)
        delta = nc - cur
        frac = min(1.0, (time.monotonic() - t0) / total)
        T = 1e-9 + (5e-5 - 1e-9) * (1.0 - frac) ** 2
        if delta < -1e-14:
            h = cand
            cur = nc
            if nc < best_c - 1e-13:
                best_c = nc
                best_h = cand.copy()
        elif rng.random() < math.exp(-delta / max(T, 1e-15)):
            h = cand
            cur = nc
    return best_h, best_c


def make_fallback(n, rng):
    h = rng.random(n)
    return project_exact(h, n / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    rng = np.random.default_rng(seed)
    effective = max(2.0, float(budget_s) - 5.0)
    deadline = t0 + effective

    parent = get_parent()
    if parent is None:
        parent = make_fallback(131, rng)
    n0 = len(parent)
    h0 = project_exact(parent, n0 / 2.0)
    best_h = h0.copy()
    best_c5 = objective(best_h)

    # Phase 1: LSE polish with increasing sigma
    sigma_schedule = [(60.0, 0.12), (150.0, 0.10), (400.0, 0.08), (1000.0, 0.05)]
    for sigma, fr in sigma_schedule:
        remaining = deadline - time.monotonic()
        alloc = fr * effective
        alloc = min(alloc, remaining * 0.8)
        if alloc < 1.0:
            break
        h1, c1 = lse_optimize(best_h, rng, alloc, sigma, t0 + effective)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 2: Subgradient phase targeting the max directly
    remaining = deadline - time.monotonic()
    alloc = min(0.10 * effective, remaining * 0.5)
    if alloc > 2.0:
        h1, c1 = subgrad_optimize(best_h, rng, alloc, t0 + effective)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 3: Iterated local search
    remaining = deadline - time.monotonic()
    alloc = min(0.25 * effective, remaining * 0.5)
    if alloc > 5.0:
        h1, c1 = iterated_local_search(best_h, rng, alloc, t0 + effective)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 4: Targeted local search on worst lags
    remaining = deadline - time.monotonic()
    alloc = min(0.15 * effective, remaining * 0.5)
    if alloc > 2.0:
        h1, c1 = targeted_local_search(best_h, rng, alloc, t0 + effective)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 5: Annealing
    remaining = deadline - time.monotonic()
    alloc = min(0.12 * effective, remaining * 0.5)
    if alloc > 2.0:
        h1, c1 = anneal_local(best_h, rng, alloc, t0 + effective)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 6: Multi-resolution
    remaining = deadline - time.monotonic()
    for dn in [0, -2, 2, 4, -4, 6]:
        nn = n0 + dn
        if nn < 80 or nn > 300:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 3.0:
            break
        alloc = min(0.06 * effective, remaining * 0.3)
        if alloc < 1.0:
            break
        rs = project_exact(resample(best_h, nn), nn / 2.0)
        h1, c1 = lse_optimize(rs, rng, alloc, 100.0, t0 + effective)
        if c1 < best_c5 - 1e-13:
            best_c5 = c1
            best_h = h1.copy()

    # Phase 7: Final squeeze - combine methods on the best
    remaining = deadline - time.monotonic()
    if remaining > 3.0:
        alloc = min(0.08 * effective, remaining * 0.5)
        if alloc > 1.0:
            h1, c1 = targeted_local_search(best_h, rng, alloc, t0 + effective)
            if c1 < best_c5 - 1e-13:
                best_c5 = c1
                best_h = h1.copy()
    remaining = deadline - time.monotonic()
    if remaining > 2.0:
        alloc = min(0.05 * effective, remaining * 0.5)
        if alloc > 0.5:
            h1, c1 = lse_optimize(best_h, rng, alloc, 500.0, t0 + effective)
            if c1 < best_c5 - 1e-13:
                best_c5 = c1
                best_h = h1.copy()

    # Final feasibility check
    best_h = project_exact(np.asarray(best_h, dtype=float), len(best_h) / 2.0)
    best_c5 = objective(best_h)
    return best_h, float(best_c5), int(len(best_h))
