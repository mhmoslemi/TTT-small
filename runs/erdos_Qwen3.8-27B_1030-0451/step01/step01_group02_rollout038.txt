```python
import numpy as np
import time

def exact_C5(h):
    n = len(h)
    if n == 0:
        return 0.0
    dx = 2.0 / float(n)
    c = np.correlate(h, 1.0 - h, "full") * dx
    return float(np.max(c))

def make_feasible(x, target):
    """Project x to [0,1]^n with sum = target"""
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    n = len(x)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    # Binary search for shift
    lo, hi = -1.0 - x.max() + 1e-15, 1.0 - x.min() - 1e-15
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.clip(x + mid, 0, 1).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + (lo + hi) / 2, 0, 1)
    # Fix residual
    diff = target - x.sum()
    if diff > 0:
        idx = np.argsort(x)  # smallest first (most room)
        for i in idx:
            add = min(1.0 - x[i], diff)
            x[i] += add
            diff -= add
            if diff < 1e-12:
                break
    else:
        idx = np.argsort(x)[::-1]  # largest first
        for i in idx:
            rem = min(x[i], -diff)
            x[i] -= rem
            diff += rem
            if diff > -1e-12:
                break
    return np.clip(x, 0.0, 1.0)

def resample(h, N):
    """Resample h to length N"""
    n0 = len(h)
    if n0 == N:
        return make_feasible(h.copy(), 0.5 * N)
    x_old = (np.arange(n0) + 0.5) * (2.0 / n0)
    x_new = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(x_new, x_old, h)
    return make_feasible(y, 0.5 * N)
```

```python
import numpy as np
import time

def _c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, "full") * dx
    return float(np.max(c))

def _feasible(x, target):
    x = np.array(x, dtype=float, copy=True)
    x = np.clip(x, 0.0, 1.0)
    n = len(x)
    diff = target - x.sum()
    if abs(diff) < 1e-10:
        return x
    # Binary search shift
    lo, hi = -1.0, 1.0
    for _ in range(50):
        mid = (lo + hi) * 0.5
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    x = np.clip(x + (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if diff > 0:
        order = np.argsort(x)
        for i in order:
            add = min(1.0 - x[i], diff)
            x[i] += add
            diff -= add
            if diff < 1e-12:
                break
    elif diff < 0:
        order = np.argsort(x)[::-1]
        for i in order:
            rem = min(x[i], -diff)
            x[i] -= rem
            diff += rem
            if diff > -1e-12:
                break
    return x

def _resample(h, N):
    n0 = len(h)
    target = 0.5 * N
    if n0 == N:
        return _feasible(h.copy(), target)
    x0 = (np.arange(n0) + 0.5) / n0 * 2.0
    x1 = (np.arange(N) + 0.5) / N * 2.0
    y = np.interp(x1, x0, h)
    return _feasible(y, target)

def _antipodal_h(bits):
    h = np.empty(2 * len(bits), dtype=float)
    h[:len(bits)] = bits
    h[len(bits):] = 1.0 - bits[::-1]
    return h

def _sa_antipodal(init_bits, N, deadline, rng, T0=None, Tcool=None):
    half = N // 2
    bits = np.array(init_bits, dtype=float, copy=True)
    h = _antipodal_h(bits)
    ones_h = 1.0 - h
    c = _c5(h)
    best_c = c
    best_bits = bits.copy()
    
    dx = 2.0 / N
    if T0 is None:
        T0 = 5.0 * dx
    if Tcool is None:
        Tcool = 0.9997
    T_min = 0.15 * dx
    
    T = T0
    stall = 0
    
    while time.time() < deadline:
        i = int(rng.integers(half))
        bits[i] = 1.0 - bits[i]
        h = _antipodal_h(bits)
        c_new = _c5(h)
        
        if c_new <= c or rng.random() < np.exp(-(c_new - c) / max(T, 1e-12)):
            h_cur = h
            c = c_new
            if c_new < best_c:
                best_c = c_new
                best_bits = bits.copy()
                stall = 0
            else:
                stall += 1
        else:
            bits[i] = 1.0 - bits[i]  # revert
            stall += 1
        
        T *= Tcool
        if T < T_min:
            T = T_min
        
        if stall > 80000:
            T = max(T, 2.0 * dx)
            stall = 0
    
    return _antipodal_h(best_bits), best_c

def _sa_general(init_h, N, deadline, rng):
    target = 0.5 * N
    h = (init_h >= 0.5).astype(float)
    # Fix count
    n_ones = int(h.sum())
    target_ones = N // 2
    ones_idx = np.where(h == 1.0)[0]
    zeros_idx = np.where(h == 0.0)[0]
    if n_ones > target_ones:
        to_remove = n_ones - target_ones
        order = ones_idx[np.argsort(init_h[ones_idx])]
        for i in order[:to_remove]:
            h[i] = 0.0
    elif n_ones < target_ones:
        to_add = target_ones - n_ones
        order = zeros_idx[np.argsort(init_h[zeros_idx])[::-1]]
        for i in order[:to_add]:
            h[i] = 1.0
    
    c = _c5(h)
    best_c = c
    best_h = h.copy()
    
    dx = 2.0 / N
    T = 5.0 * dx
    Tcool = 0.9997
    T_min = 0.15 * dx
    
    # Precompute sets
    ones_set = set(np.where(h == 1.0)[0].tolist())
    zeros_set = set(np.where(h == 0.0)[0].tolist())
    
    stall = 0
    while time.time() < deadline:
        if not ones_set or not zeros_set:
            break
        i = list(ones_set)[int(rng.integers(len(ones_set)))]
        j = list(zeros_set)[int(rng.integers(len(zeros_set)))]
        
        h[i], h[j] = h[j], h[i]
        c_new = _c5(h)
        
        if c_new <= c or rng.random() < np.exp(-(c_new - c) / max(T, 1e-12)):
            ones_set.remove(i)
            ones_set.add(j)
            zeros_set.remove(j)
            zeros_set.add(i)
            c = c_new
            if c_new < best_c:
                best_c = c_new
                best_h = h.copy()
                stall = 0
            else:
                stall += 1
        else:
            h[i], h[j] = h[j], h[i]  # revert
            stall += 1
        
        T *= Tcool
        if T < T_min:
            T = T_min
        
        if stall > 100000:
            T = max(T, 2.0 * dx)
            stall = 0
    
    return best_h, best_c
```
