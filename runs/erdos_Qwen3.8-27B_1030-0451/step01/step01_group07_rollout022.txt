def lse_gradient_correct(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    center = n - 1
    grad = np.zeros(n)
    
    # Zero lag (index center)
    grad += w[center] * (1.0 - 2.0 * h)
    
    for p in range(1, n):
        # Positive lag p: index center+p
        # dF_p/dh[j] = [p<=j<=n-1]*(1-h[j-p]) - [0<=j<=n-1-p]*h[j+p]
        wp = w[center + p]
        if wp > 1e-18:
            # [p<=j<=n-1]*(1-h[j-p]): for j in [p, n-1], value is (1-h[j-p]) = (1-h[0:n-p])
            g1 = np.zeros(n)
            g1[p:] = 1.0 - h[:n-p]
            # [0<=j<=n-1-p]*h[j+p]: for j in [0, n-1-p], value is h[j+p] = h[p:n]
            g2 = np.zeros(n)
            g2[:n-p] = h[p:]
            grad += wp * (g1 - g2)
        
        # Negative lag -p: index center-p
        # dF_{-p}/dh[j] = [0<=j<=n-1-p]*(1-h[j+p]) - [p<=j<=n-1]*h[j-p]
        wm = w[center - p]
        if wm > 1e-18:
            g1 = np.zeros(n)
            g1[:n-p] = 1.0 - h[p:]
            g2 = np.zeros(n)
            g2[p:] = h[:n-p]
            grad += wm * (g1 - g2)
    
    grad *= dx
    return grad

def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    lo, hi = -2.0, 2.0

        if y.sum() > target:

        else:

    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return dx * (m + math.log(np.exp(sigma * (c - m)).sum()) / sigma)

def topk_grad(h, k=5):
    """Gradient that emphasizes the top-k correlations equally"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    center = n - 1
    top_idx = np.argsort(c)[-k:]
    grad = np.zeros(n)

        if p == 0:

            grad[:n-p] -= h[p:]
        else:
            q = -p
            grad[:n-q] += (1.0 - h[q:])
            grad[q:] -= h[:n-q]

        if not improved:
            # Perturb and restart
            h = project_exact(h + rng.normal(0, 0.02, len(h)), target)
            cur_val = lse_val(h, sigma)
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            stall += 1
            if stall > 10:
                eta *= 0.5
                stall = 0
                if eta < 1e-8:
                    break
    return best_h, best_c5

def topk_descent(h0, k, target, time_limit, deadline, rng):
    h = project_exact(h0, target)
    best_h = h.copy()
    best_c5 = objective(h)
    eta = 0.03
    t0 = time.monotonic()
    stall = 0
    while (time.monotonic() - t0) < time_limit and time.monotonic() < deadline:
        g = topk_grad(h, k)
        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break
        g = g / gn
        improved = False
        step = eta
        for _ in range(30):
            cand = project_exact(h - step * g, target)
            c5 = objective(cand)
            if c5 < best_c5 - 1e-14:
                h = cand
                improved = True
                break
            step *= 0.5
        if not improved:
            # Try to reduce the single worst lag
            c = compute_corr(h)
            worst = int(np.argmax(c))
            p = worst - (len(h) - 1)
            n = len(h)
            g2 = np.zeros(n)
            if p == 0:
                g2[:] = 1.0 - 2.0 * h
            elif p > 0:
                g2[p:] = (1.0 - h[:n-p])
                g2[:n-p] -= h[p:]
            else:
                q = -p
                g2[:n-q] = (1.0 - h[q:])
                g2[q:] -= h[:n-q]
            g2n = np.linalg.norm(g2)
            if g2n > 1e-14:
                g2 /= g2n
                step = eta
                for _ in range(30):
                    cand = project_exact(h - step * g2, target)
                    c5 = objective(cand)
                    if c5 < best_c5 - 1e-14:
                        h = cand
                        improved = True
                        break
                    step *= 0.5
            if not improved:
                h = project_exact(h + rng.normal(0, 0.01, len(h)), target)
                stall += 1
                if stall > 8:
                    eta *= 0.5
                    stall = 0
                    if eta < 1e-8:
                        break
        else:
            stall = 0
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    return best_h, best_c5

def pair_swap_search(h0, target, time_limit, deadline, rng):
    """Local search via pair swaps of mass between positions"""
    h = project_exact(h0, target)
    n = len(h)
    cur = objective(h)
    best_h = h.copy()
    best_c = cur
    t0 = time.monotonic()
    while (time.monotonic() - t0) < time_limit and time.monotonic() < deadline:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        # Try several step sizes
        best_step = None
        best_val = cur
        for alpha in [0.05, 0.1, 0.2, 0.3, 0.5]:
            for direction in [1, -1]:
                d = alpha * direction
                if d > 0:
                    if h[i] < d or h[j] > 1.0 - d:
                        continue
                else:
                    if h[j] < -d or h[i] > 1.0 - (-d):
                        continue
                cand = h.copy()
                cand[i] -= d
                cand[j] += d
                cv = objective(cand)
                if cv < best_val:
                    best_val = cv
                    best_step = d
        if best_step is not None:
            h[i] -= best_step
            h[j] += best_step
            cur = best_val
            if cur < best_c:
                best_c = cur
                best_h = h.copy()
    return best_h, best_c

def sturmian(n, ones_count, phase):
    """Generate a Sturmian (mechanical) word of length n with ones_count ones"""
    rho = ones_count / n
    h = np.zeros(n)
    for i in range(n):
        if math.floor((i+1)*rho + phase) > math.floor(i*rho + phase):
            h[i] = 1.0
    # Fix count
    while h.sum() > ones_count:
        # Remove a 1 that's most "surplus"
        idx = np.where(h == 1)[0]
        h[idx[0]] = 0
    while h.sum() < ones_count:
        idx = np.where(h == 0)[0]
        h[idx[0]] = 1
    return h

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

def random_binary(n, rng, target):
    h = np.zeros(n)
    perm = rng.permutation(n)
    h[perm[:int(target)]] = 1.0
    return h

def gaussian_perturb(h, rng, scale=0.1):
    return h + rng.normal(0, scale, len(h))
