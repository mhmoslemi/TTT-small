    t = (x - offset) % (2.0 / period)
    h = (t < duty * 2.0 / period).astype(float)
    return project_feasible(h, n)

def smooth_surrogate_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    c_max = np.max(c)
    exponents = np.clip((c - c_max) / tau, -500, 500)
    return c_max + tau * np.log(np.sum(np.exp(exponents)))

def run_slsqp(h0, tau, n, maxiter, deadline):
    dx = 2.0 / n
    def obj(x):
        if time.time() > deadline:
            return 1e10
        return smooth_surrogate_obj(x, tau)
    def obj_grad(x):
        c = np.correlate(x, 1.0 - x, mode='full') * dx
        c_max = np.max(c)
        exponents = np.clip((c - c_max) / tau, -500, 500)
        weights = np.exp(exponents)
        sw = np.sum(weights)
        if sw < 1e-300:
            return np.zeros(n)
        weights /= sw
        g = np.zeros(n)
        for K in range(2 * n - 1):
            w = weights[K]
            if w < 1e-14:
                continue
            k = K - (n - 1)
            if k >= 0:
                g[k:] += w * (1.0 - x[:n - k])
                g[:n - k] += w * (-x[k:])
            else:
                m = -k
                g[:n - m] += w * (1.0 - x[m:])
                g[m:] += w * (-x[:n - m])
        return g
    cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n/2.0, 'jac': lambda x: np.ones(n)}]
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(fun=obj, x0=h0, jac=obj_grad, method='SLSQP',
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False})
        if res.x is not None and np.all(np.isfinite(res.x)):
            return project_feasible(res.x, n)
    except Exception:
        pass
    return None

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    deadline = t0 + budget_s - 15.0

    try:
        parent = np.array(initial_h_values, dtype=float).copy()
    except Exception:
        parent = None

    if parent is not None:
        n_p = len(parent)
        if not (n_p >= 2 and np.all(np.isfinite(parent))):
            parent = None

    if parent is not None:
        parent = project_feasible(parent, n_p)
    else:
        n_p = 81
        parent = project_feasible(np.full(n_p, 0.5), n_p)

    best_h = parent.copy()
    best_score = exact_score(parent)
    best_n = n_p

    def check_update(h, score):
        nonlocal best_h, best_score, best_n
        if score < best_score - 1e-14:
            best_h = h.copy()
            best_score = score

    # Phase 1: SLSQP annealing on parent dimension
    h_work = best_h.copy()
    s_work = best_score
    for tau in [0.05, 0.03, 0.02, 0.01, 0.005, 0.003, 0.001]:
        if time.time() > deadline - 60:
            break
        rem = deadline - time.time()
        if rem < 30:
            break
        mi = min(150, max(30, int(rem * 0.5)))
        h_new = run_slsqp(h_work, tau, best_n, mi, deadline)
        if h_new is not None:
            s = exact_score(h_new)
            if s < s_work - 1e-12:
                h_work = h_new
                s_work = s
                check_update(h_work, s)

    # Phase 2: Exact pair sweep (key improvement)
    h_work = best_h.copy()
    if time.time() < deadline - 80:
        h_ps, s_ps = pair_sweep(h_work, deadline - 60, max_rounds=40, top_k=50)
        if s_ps < best_score - 1e-14:
            best_h = h_ps
            best_score = s_ps
            check_update(best_h, best_score)

    # Phase 3: SA for global exploration
    if time.time() < deadline - 100:
        h_sa, s_sa = sa_large_moves(best_h, best_n, deadline - 80, rng, n_iter=3000, T0=0.3, T1=0.02)
        if s_sa < best_score - 1e-14:
            s_ps2, s_ps2_v = pair_sweep(h_sa, deadline - 40, max_rounds=30, top_k=40)
            if s_ps2_v < best_score - 1e-14:
                best_h = s_ps2
                best_score = s_ps2_v
                check_update(best_h, best_score)

    # Phase 4: More SLSQP + pair polish
    if time.time() < deadline - 50:
        for tau in [0.002, 0.001, 0.0005]:
            if time.time() > deadline - 30:
                break
            h_new = run_slsqp(best_h, tau, best_n, 100, deadline)
            if h_new is not None:
                s = exact_score(h_new)
                if s < best_score - 1e-14:
                    best_h = h_new
                    best_score = s
                    check_update(best_h, best_score)
        if time.time() < deadline - 20:
            h_ps, s_ps = pair_sweep(best_h, deadline - 10, max_rounds=25, top_k=50)
            if s_ps < best_score - 1e-14:
                best_h = h_ps
                best_score = s_ps
                check_update(best_h, best_score)

    # Phase 5: Dimension hopping
    dim_options = []
    pn = best_n
    for factor in [1.1, 0.9, 1.25, 0.8, 1.5, 0.75, 1.33, 1.2]:
        cn = int(round(pn * factor))
        if 40 <= cn <= 400 and cn != pn:
            dim_options.append(cn)
    dim_options = sorted(set(dim_options))

    for dn in dim_options:
        if time.time() > deadline - 25:
            break
        h_d = interpolate_h(best_h, dn)
        s_d = exact_score(h_d)
        if s_d < best_score + 0.01:
            for tau in [0.02, 0.01, 0.005]:
                if time.time() > deadline - 15:
                    break
                h_new = run_slsqp(h_d, tau, dn, 80, deadline)
                if h_new is not None:
                    s_new = exact_score(h_new)
                    if s_new < s_d:
                        h_d = h_new
                        s_d = s_new
            if time.time() < deadline - 10:
                h_d, s_d = pair_sweep(h_d, deadline - 5, max_rounds=15, top_k=30)
            if s_d < best_score - 1e-14:
                best_h = h_d
                best_score = s_d
                best_n = dn
                check_update(best_h, best_score)

    # Phase 6: Alternative initializations
    if time.time() < deadline - 120:
        alts = []
        alts.append(make_sinusoidal(best_n, 3, 0.3, 0.5, rng))
        alts.append(make_sinusoidal(best_n, 5, 0.2, 1.0, rng))
        alts.append(make_periodic(best_n, 4, 0.5, 0.1))
        alts.append(make_periodic(best_n, 5, 0.4, 0.05))
        x = (np.arange(best_n) + 0.5) / best_n
        alts.append(project_feasible(0.5 + 0.25*np.cos(2*np.pi*2*x) + 0.15*np.sin(2*np.pi*7*x), best_n))
        alts.append(project_feasible(0.5 + 0.3*np.sin(np.pi*x) + 0.1*np.sin(3*np.pi*x), best_n))

        for alt in alts:
            if time.time() > deadline - 30:
                break
            h_alt = alt
            s_alt = exact_score(h_alt)
            if s_alt < best_score + 0.02:
                h_alt = run_slsqp(h_alt, 0.03, best_n, 80, deadline)
                if h_alt is not None:
                    s_alt = exact_score(h_alt)
                if s_alt < best_score + 0.01:
                    h_alt, s_alt = pair_sweep(h_alt, deadline - 15, max_rounds=20, top_k=35)
                    if s_alt < best_score - 1e-14:
                        best_h = h_alt
                        best_score = s_alt
                        check_update(best_h, best_score)

    # Phase 7: Final polish with multiple techniques
    if time.time() < deadline - 10:
        h_pol = best_h.copy()
        h_pol, s_pol = pair_sweep(h_pol, deadline - 5, max_rounds=20, top_k=60)
        if s_pol < best_score - 1e-14:
            best_h = h_pol
            best_score = s_pol
            check_update(best_h, best_score)
        h_new = run_slsqp(best_h, 0.001, best_n, 80, deadline)
        if h_new is not None:
            s_new = exact_score(h_new)
            if s_new < best_score - 1e-14:
                best_h = h_new
                best_score = s_new
                check_update(best_h, best_score)

    # Final: ensure feasibility
    best_h = project_feasible(best_h, best_n)
    final_score = exact_score(best_h)
    if final_score > best_score + 1e-10:
        # Fallback to parent if something went wrong
        best_h = parent.copy()
        best_score = exact_score(parent)
        best_n = n_p
    else:
        best_score = final_score

    return best_h, float(best_score), int(best_n)
