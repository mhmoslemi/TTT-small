    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    c = np.correlate(h, one_minus_h, 'full') * dx
    cs = c - c.max()
    alpha = np.exp(cs / tau)
    alpha /= alpha.sum()
    
    n_c = 2 * n - 1
    grad = np.zeros(n)
    for k in range(n_c):
        a = alpha[k]
        if a < 1e-15:
            continue

    return grad * dx, c
