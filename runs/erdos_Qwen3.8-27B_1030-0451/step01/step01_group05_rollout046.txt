def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    raw = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(raw) * dx)

def project_feasible(z, n):
    S = n / 2.0
    lo = z.min() - 2.0
    hi = z.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(z - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    h = np.clip(z - mu, 0.0, 1.0)
    diff = S - np.sum(h)
    if abs(diff) > 1e-14:
        interior = np.where((h > 1e-10) & (h < 1.0 - 1e-10))[0]
        if len(interior) > 0:
            h[interior] += diff / len(interior)
            h = np.clip(h, 0.0, 1.0)
    return h

def get_lin(h, a, b, n):
    L = 2 * n - 1
    center = n - 1
    lin = np.zeros(L)
    for K in range(L):
        s = K - center
        v = 0.0
        ia = a - s
        if 0 <= ia < n:
            v += 1.0 - h[ia]
        ia = a + s
        if 0 <= ia < n:
            v -= h[ia]
        ib = b - s
        if 0 <= ib < n:
            v -= 1.0 - h[ib]
        ib = b + s
        if 0 <= ib < n:
            v += h[ib]
        lin[K] = v
    return lin

def get_quad(a, b, n):
    L = 2 * n - 1
    center = n - 1
    quad = np.zeros(L)
    quad[center] = -2.0
    quad[center + (b - a)] += 1.0
    quad[center + (a - b)] += 1.0
    return quad

def line_optimize(raw, lin, quad, d_lo, d_hi, dx):
    if d_hi - d_lo < 1e-14:
        return 0.0, raw.max() * dx
    
    best_d = 0.0
    best_s = raw.max() * dx  # score at d=0
    
    # Coarse grid
    n_pts = 31
    for d in np.linspace(d_lo, d_hi, n_pts):
        s = (raw + d * lin + d * d * quad).max() * dx
        if s < best_s:
            best_s = s
            best_d = d
    
    # Refine
    span = (d_hi - d_lo) / (n_pts - 1)
    for _ in range(4):
        lo = max(d_lo, best_d - 3 * span)
        hi = min(d_hi, best_d + 3 * span)
        if hi - lo < 1e-13:

            if s < best_s:
                best_s = s
                best_d = d
        if abs(best_s - prev_best) < 1e-15:
            break
        span *= 0.3
    
    return best_d, best_s

def compute_weighted_grad(h, raw, tau, n):
    L = 2 * n - 1
    center = n - 1
    raw_max = raw.max()
    exps = np.clip((raw - raw_max) / tau, -300, 300)
    w = np.exp(exps)
    sw = w.sum()
    if sw < 1e-200:
        w = np.ones(L) / L
    else:
        w /= sw
    
    grad = np.zeros(n)
    for K in range(L):
        wk = w[K]
        if wk < 1e-14:
            continue
        s = K - center
        if s >= 0:
            grad[s:] += wk * (1.0 - h[:n - s])
            grad[:n - s] += wk * (-h[s:])
        else:
            m = -s
            grad[:n - m] += wk * (1.0 - h[m:])
            grad[m:] += wk * (-h[:n - m])
    return grad
