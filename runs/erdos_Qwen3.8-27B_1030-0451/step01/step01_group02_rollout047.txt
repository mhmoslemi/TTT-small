def exact_C5(h):

def make_feasible(x, target):
    """Project x onto {0<=x<=1, sum(x)=target} via bisection on shift."""
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return x
    if not np.all(np.isfinite(x)):
        x = np.full(n, 0.5)
    
    # Bisection to find shift s such that sum(clip(x+s, 0, 1)) = target
    lo = -1.0 - float(np.max(x))
    hi = 1.0 - float(np.min(x))
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    
    x = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    
    # Fine-tune sum
    diff = target - float(x.sum())
    if abs(diff) > 1e-12:
        if diff > 0:
            idx = np.where(x < 1.0)[0]
            if idx.size > 0:
                # Spread the needed amount to the positions with most room
                order = idx[np.argsort(1.0 - x[idx])]
                rem = diff
                for i in order:
                    inc = min(1.0 - x[i], rem)
                    x[i] += inc
                    rem -= inc
                    if rem < 1e-14:

        else:

def make_feasible(x, target):

            hi = mid

        if diff > 0:
            idx = np.where(x < 1.0)[0]

    n = len(h)
    M = C.size
    ctr = n - 1
    dx = 2.0 / float(n)
    maxC = float(np.max(C))
    a = (C - maxC) / tau

        else:
            dd = -d
            grad[:n-dd] += dx * wi * (1.0 - h[dd:])
            grad[dd:] += -dx * wi * h[:n-dd]
    return grad

    return make_feasible(h_new, target)

            else:
                stall += 1
                if stall >= 10:
                    alpha *= 0.5
                    stall = 0
            if alpha < 1e-8:
                break

    return best_h, best_c


def guided_pair_refine(h, n_moves, deadline, rng, tau=0.01):
    n = len(h)
    target = 0.5 * float(n)
    best_c = exact_C5(h)
    C = corr_values(h)
    grad = subgrad_smoothed(h, C, tau)

    for mv in range(n_moves):
        if time.time() > deadline:
            break
        pos = np.where(grad > 0.005)[0]
        neg = np.where(grad < -0.005)[0]
        if pos.size == 0 or neg.size == 0:
            grad = subgrad_smoothed(h, C, tau)
            pos = np.where(grad > 0)[0]
            neg = np.where(grad < 0)[0]
            if pos.size == 0 or neg.size == 0:
                break

        i = int(pos[int(rng.integers(pos.size))])
        j = int(neg[int(rng.integers(neg.size))])
        max_eps = min(h[i], 1.0 - h[j])
        if max_eps < 1e-7:
            continue

        for frac in (0.15, 0.08, 0.04, 0.02):
            eps = max_eps * frac
            if eps < 1e-7:
                continue
            h2 = h.copy()
            h2[i] -= eps
            h2[j] += eps
            c2 = exact_C5(h2)
            if c2 < best_c - 1e-13:
                h = h2
                best_c = c2
                C = corr_values(h)
                grad = subgrad_smoothed(h, C, tau)
                break

    return h, best_c


def random_perturb_refine(h, n_trials, deadline, rng):
    n = len(h)
    target = 0.5 * float(n)
    best_c = exact_C5(h)
    sigmas = (0.03, 0.02, 0.01, 0.005)

    for t in range(n_trials):
        if time.time() > deadline:
            break
        sig = sigmas[t % len(sigmas)]
        h2 = make_feasible(h + rng.normal(0.0, sig, size=n), target)
        c2 = exact_C5(h2)
        if c2 < best_c - 1e-13:
            h = h2
            best_c = c2

    return h, best_c


def resample_h(h, N):
    n0 = len(h)
    if n0 == N:
        return make_feasible(h, 0.5 * float(N))
    xold = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))
    y = np.interp(xnew, xold, h)
    return make_feasible(y, 0.5 * float(N))


def fourier_init(N, rng):
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    best_h = None
    best_c = float("inf")
    for f in range(1, 6):
        for k in range(4):
            ph = k * 0.5 * np.pi + float(rng.uniform(-0.3, 0.3))
            y = 0.5 + 0.35 * np.cos(np.pi * f * x + ph)
            h = make_feasible(y, 0.5 * float(N))
            c = exact_C5(h)
            if c < best_c:
                best_c = c
                best_h = h
    for _ in range(6):
        f1 = int(rng.integers(1, 5))
        f2 = int(rng.integers(2, 8))
        ph1 = float(rng.uniform(0, 2 * np.pi))
        ph2 = float(rng.uniform(0, 2 * np.pi))
        a1 = float(rng.uniform(0.1, 0.3))
        a2 = float(rng.uniform(0.05, 0.15))
        y = 0.5 + a1 * np.cos(np.pi * f1 * x + ph1) + a2 * np.cos(np.pi * f2 * x + ph2)
        h = make_feasible(y, 0.5 * float(N))
        c = exact_C5(h)
        if c < best_c:
            best_c = c
            best_h = h
    return best_h


def optimize_full(h, deadline, rng):
    n = len(h)
    target = 0.5 * float(n)
    best_h = h.copy()
    best_c = exact_C5(best_h)

    if n > 50:
        iters = max(100, min(600, 200000 // n))
        tau1 = [0.06, 0.04, 0.025, 0.015, 0.008]
        h_opt, c_opt = optimize_subgrad(best_h, min(deadline, time.time() + 0.35 * (deadline - time.time())), tau1, iters, 0.8, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    if time.time() < deadline:
        tau2 = [0.005, 0.003, 0.002, 0.001, 0.0005]
        iters2 = max(100, min(400, 100000 // n))
        h_opt, c_opt = optimize_subgrad(best_h, min(deadline, time.time() + 0.3 * (deadline - time.time())), tau2, iters2, 0.4, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    if time.time() < deadline - 5.0:
        n_mov = max(500, min(5000, 2000000 // n))
        h_opt, c_opt = guided_pair_refine(best_h, n_mov, deadline, rng, tau=0.008)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt

    if time.time() < deadline - 5.0:
        h_opt, c_opt = random_perturb_refine(best_h, 100, deadline, rng)
        if c_opt < best_c:
            best_h, best_c = h_opt, c_opt
        if c_opt < best_c + 0.01 and time.time() < deadline - 3.0:
            h_opt2, c_opt2 = guided_pair_refine(best_h, 500, deadline, rng, tau=0.005)
            if c_opt2 < best_c:
                best_h, best_c = h_opt2, c_opt2

    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    deadline = t0 + total - 8.0

    rng = np.random.default_rng(int(seed) % (1 << 32))

    init = globals().get("initial_h_values", None)
    if init is not None:
        arr = np.asarray(init, dtype=float).ravel()
        if arr.size > 10 and np.all(np.isfinite(arr)):
            h_parent = make_feasible(arr, 0.5 * float(arr.size))
        else:
            h_parent = None
    else:
        h_parent = None

    if h_parent is None:
        h_parent = fourier_init(300, rng)

    n_parent = int(h_parent.size)
    best_h = h_parent.copy()
    best_c = exact_C5(best_h)
    best_n = n_parent

    resolutions = []
    if n_parent <= 1200:
        resolutions.append(n_parent)
    for N in (480, 360, 420):
        if N <= 1200 and N not in resolutions:
            resolutions.append(N)

    resolutions = sorted(resolutions, reverse=True)
    n_res = len(resolutions)
    time_per_res = (deadline - time.time() - 20.0) / max(n_res, 1)

    for ridx, Nr in enumerate(resolutions):
        if time.time() > deadline - 15.0:
            break
        res_deadline = min(deadline - 10.0, time.time() + time_per_res * 0.9)

        if Nr == n_parent:
            h_start = best_h.copy()
        else:
            h_start = resample_h(best_h, Nr)
            c_start = exact_C5(h_start)
            h_f = fourier_init(Nr, rng)
            c_f = exact_C5(h_f)
            if c_f < c_start:
                h_start = h_f

        h_opt, c_opt = optimize_full(h_start, res_deadline, rng)
        if c_opt < best_c - 1e-14:
            best_h = h_opt
            best_c = c_opt
            best_n = len(h_opt)

    if time.time() < deadline - 5.0:
        h_opt, c_opt = guided_pair_refine(best_h, 3000, deadline - 2.0, rng, tau=0.004)
        if c_opt < best_c:
            best_h, best_c, best_n = h_opt, c_opt, len(h_opt)

    best_n = int(best_h.size)
    best_h = make_feasible(best_h, 0.5 * float(best_n))
    best_c = exact_C5(best_h)

    if not np.isfinite(best_c):
        best_h = make_feasible(np.full(best_n, 0.5), 0.5 * float(best_n))
        best_c = exact_C5(best_h)

    return best_h, float(best_c), int(best_n)
