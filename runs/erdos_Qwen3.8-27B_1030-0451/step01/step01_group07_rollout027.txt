def compute_subgradient(h):
    """Compute subgradient of F(h) = max_m c_m(h) * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    cmax = c.max()
    
    # Active set: lags where c is within tolerance of max
    tol = 1e-8
    active = np.where(c >= cmax - tol)[0]
    
    if len(active) == 0:
        active = np.array([np.argmax(c)])
    
    # Compute gradient of each active lag and average
    grad = np.zeros(n)
    center = n - 1
    
    for m in active:
        k = m - center  # lag offset
        if k == 0:
            # c_center = sum h[j](1-h[j]), grad = 1-2h
            grad += (1.0 - 2.0 * h)
        elif k > 0:
            # c_{n-1+k} = sum_{j=k}^{n-1} h[j](1-h[j-k])
            # grad[p] = (1 if k<=p<=n-1 else 0) - h[p-k] (if k<=p) - h[p+k] (if p<=n-1-k)
            for p in range(n):
                g = 0.0
                if p >= k:
                    g += (1.0 - h[p-k])
                if p <= n-1-k:
                    g -= h[p+k]
                grad[p] += g
        else:
            k = -k  # now k > 0, negative lag
            # c_{n-1-k} = sum_{j=0}^{n-k-1} h[j](1-h[j+k])
            # grad[p] = (1 if 0<=p<=n-k-1 else 0) - h[p+k] (if p<=n-k-1) - h[p-k] (if p>=k)
            for p in range(n):
                g = 0.0
                if p <= n-1-k:
                    g += (1.0 - h[p+k])
                if p >= k:
                    g -= h[p-k]
                grad[p] += g
    
    grad *= dx / len(active)
    return grad

        best_h = h.copy()

def anneal(h, target, rng, time_limit, deadline):
    """Simulated annealing with pair moves"""
    n = len(h)
    cur = eval_c5(h)
    best_h = h.copy()
    best_val = cur
    t0 = time.monotonic()
    T = 5e-5
    T_end = 1e-8
    total = max(time_limit, 1e-9)
    
    while time.monotonic() < min(t0 + time_limit, deadline):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        
        hi_val = h[i]
        hj_val = h[j]
        lo = max(-hj_val, hi_val - 1.0)
        hi = min(hi_val, 1.0 - hj_val)
        if hi <= lo + 1e-13:
            continue
        
        t = rng.uniform(lo, hi)
        h[i] = hi_val - t
        h[j] = hj_val + t
        new_val = eval_c5(h)
        delta = new_val - cur
        
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-12)):
            cur = new_val
            if cur < best_val - 1e-13:
                best_val = cur
                best_h = h.copy()
        else:
            h[i] = hi_val
            h[j] = hj_val
        
        frac = min(1.0, (time.monotonic() - t0) / total)
        T = T_end + (T - T_end) * (1.0 - frac) ** 2
    
    return best_h, best_val

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()

    best_c5 = eval_c5(best_h)
    
    # Phase 1: Greedy coordinate descent (50% time)
    t1 = deadline - 0.50 * (deadline - t0)
    ch, cc = greedy_cd(best_h, target, rng, t1 - t0, t1)
    if cc < best_c5:
        best_c5 = cc
        best_h = ch
    if time.monotonic() > deadline:
        return best_h, best_c5, len(best_h)
    
    # Phase 2: Multi-start with transforms (20% time)
    transforms = [
        best_h[::-1].copy(),
        0.5 * (best_h + best_h[::-1]),
        np.roll(best_h, 1).copy(),
        np.roll(best_h, -1).copy(),
    ]
    t2 = deadline - 0.75 * (deadline - t0)
    per_transform = max(0.5, (t2 - time.monotonic()) / max(1, len(transforms)))
    for tf in transforms:
        if time.monotonic() > t2:
            break
        tf = project_exact(tf, target)
        tf_c5 = eval_c5(tf)
        if tf_c5 < best_c5:
            best_c5 = tf_c5
            best_h = tf.copy()
        alloc = min(per_transform, t2 - time.monotonic())
        if alloc > 0.5:
            ch, cc = greedy_cd(tf, target, rng, alloc, t2)
            if cc < best_c5:
                best_c5 = cc
                best_h = ch
    if time.monotonic() > deadline:
        return best_h, best_c5, len(best_h)
    
    # Phase 3: Annealing (15% time)
    t3 = deadline - 0.88 * (deadline - t0)
    alloc = max(0.5, t3 - time.monotonic())
    if alloc > 1.0:
        ch, cc = anneal(best_h, target, rng, alloc, t3)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch
    if time.monotonic() > deadline:
        return best_h, best_c5, len(best_h)
    
    # Phase 4: Multi-resolution (10% time)
    t4 = deadline - 0.97 * (deadline - t0)
    for dn in [-2, 2, -4, 4, -6, 6, -8, 8]:
        if time.monotonic() > t4:
            break
        nn = n + dn
        if nn < 50 or nn > 200:
            continue
        alloc = min(3.0, t4 - time.monotonic())
        if alloc < 0.5:
            break
        rs = project_exact(resample(best_h, nn), nn / 2.0)
        rs_c5 = eval_c5(rs)
        if rs_c5 < best_c5:
            best_c5 = rs_c5
            best_h = rs.copy()
        ch, cc = greedy_cd(rs, nn / 2.0, rng, alloc, t4)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch
    
    # Phase 5: Final polish (3% time)
    alloc = min(1.0, deadline - time.monotonic())
    if alloc > 0.3:
        target = len(best_h) / 2.0
        ch, cc = greedy_cd(best_h, target, rng, alloc, deadline - 0.1)
        if cc < best_c5:
            best_c5 = cc
            best_h = ch
    
    # Final repair
    best_h = project_exact(best_h, len(best_h) / 2.0)
    best_c5 = eval_c5(best_h)
    
    return best_h, best_c5, len(best_h)

import time

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None

def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, -3.0, 3.0)
    lo, hi = -3.0, 3.0
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if d > 1e-10:
        mask = out > 1e-12
        if mask.sum() > 0:
            out[mask] -= d / mask.sum()
    elif d < -1e-10:
        mask = out < 1.0 - 1e-12
        if mask.sum() > 0:
            out[mask] += (-d) / mask.sum()
    out = np.clip(out, 0.0, 1.0)
    # Final micro-fix
    d = out.sum() - target
    if abs(d) > 1e-14 and len(out) > 0:
        if d > 0:
            idx = int(np.argmax(out))
            out = out.copy()
            out[idx] = max(0.0, out[idx] - d)
        else:
            idx = int(np.argmin(out))
            out = out.copy()
            out[idx] = min(1.0, out[idx] - d)
