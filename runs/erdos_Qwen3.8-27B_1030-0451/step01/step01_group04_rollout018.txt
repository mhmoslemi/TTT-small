import numpy as np
import time

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_corr(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add mass
            free = 1.0 - h
            free[h >= 1.0 - 1e-15] = 0
            if free.sum() > 0:
                h += residual * free / free.sum()
                h = np.clip(h, 0, 1)
        else:
            # Need to remove mass
            free = h.copy()
            free[h <= 1e-15] = 0
            if free.sum() > 0:
                h += residual * free / free.sum()
                h = np.clip(h, 0, 1)
    # Final exact fix
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] = min(1.0, h[idx[0]] + residual)
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] = max(0.0, h[idx[0]] + residual)
    return h

def compute_gradient(h, weights):
    """Compute gradient of sum_r weights[r] * corr[r] * dx"""
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        w = weights[r] * dx
        if abs(w) < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                # corr[r] = sum_{i=0}^{n-1} h[i+d]*(1-h[i])
                # d/dh[j]: (1-h[j-d]) for j>=d, -h[j+d] for j<d
                g[d:] += w * (1.0 - h[:n])  # from h[i+d] term, j=i+d, i goes 0..n-1, j goes d..N-1
                g[:n] -= w * h[d:]            # from (1-h[i]) term, j=i, i goes 0..n-1
        else:
            dd = -d
            n = N - dd
            if n > 0:
                # corr[r] = sum_{i=dd}^{N-1} h[i]*(1-h[i-dd])
                # d/dh[j]: (1-h[j-dd]) for j>=dd, -h[j+dd] for j<N-dd
                g[dd:] += w * (1.0 - h[:n])   # from h[i] term, j=i, i goes dd..N-1
                g[:n] -= w * h[dd:]           # from (1-h[i-dd]) term, j=i-dd, i goes dd..N-1, j goes 0..N-1-dd
    return g

def softmax_weights(corr, cmax, tau):
    """Compute softmax weights focused on active constraints"""
    exponents = np.clip((corr - cmax) / tau, -100.0, 0.0)
    w = np.exp(exponents)
    total = w.sum()
    if total > 0:
        w = w / total
    else:
        w = np.ones(len(corr)) / len(corr)
    return w

def interpolate_h(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    target = N_new / 2.0
    return project_feasible(h_new, target)

def line_search(h, direction, current_c5, target_sum):
    """Backtracking line search along direction"""
    N = len(h)
    # Normalize direction
    dir_norm = np.linalg.norm(direction)
    if dir_norm < 1e-15:
        return h, current_c5
    direction = direction / dir_norm
    
    # Determine max step
    # h + t*d must stay in [0,1]
    d = direction
    pos_mask = d > 1e-15
    neg_mask = d < -1e-15
    t_max = 1.0
    if np.any(pos_mask):
        t_max = min(t_max, (1.0 - h[pos_mask]).min() / d[pos_mask].max())
    if np.any(neg_mask):
        t_max = min(t_max, h[neg_mask].min() / (-d[neg_mask].max()))
    t_max = max(t_max, 0.0)
    
    # Try different step sizes
    best_h = h.copy()
    best_c5 = current_c5
    base_step = 0.1  # Will be scaled by how we use it
    
    for k in range(20):
        t = base_step * (0.7 ** k)
        if t > t_max:
            t = t_max * 0.9
        if t < 1e-12:
            break
        h_new = h + t * direction
        h_new = project_feasible(h_new, target_sum)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5 - 1e-14:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    return best_h, best_c5

def local_search(h, tau_start, max_iters, target_sum, rng, time_limit=None, start_time=None):
    """Main local search loop"""
    N = len(h)
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    tau = tau_start
    
    for it in range(max_iters):
        if time_limit is not None and time.time() - start_time > time_limit:
            break
        
        # Compute correlations and gradient
        corr = compute_corr(h)
        cmax = np.max(corr)
        weights = softmax_weights(corr, cmax, tau)
        grad = compute_gradient(h, weights)
        
        # Remove mean to maintain sum constraint
        grad = grad - np.mean(grad)
        
        # Scale gradient
        g_norm = np.linalg.norm(grad)
        if g_norm < 1e-15:
            break
        direction = -grad / g_norm
        
        # Line search
        h_new, c5_new = line_search(h, direction, current_c5, target_sum)
        
        if c5_new < current_c5 - 1e-14:
            h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            # Reduce tau slowly
            tau = max(tau * 0.95, 0.001)
        else:
            # No improvement: reduce tau or try perturbation
            tau = max(tau * 0.8, 0.0005)
            if tau <= 0.0006:
                # Try random perturbation
                noise = rng.normal(0, 0.02, N)
                h_pert = project_feasible(h + noise, target_sum)
                c5_pert = compute_c5(h_pert)
                if c5_pert < current_c5:
                    h = h_pert
                    current_c5 = c5_pert
                tau = tau_start * 0.5
    return best_h, best_c5, tau

def gradient_weighted(h, weights):
    N = len(h)
    dx = 2.0 / N
    g = np.zeros(N)
    for r in range(2 * N - 1):
        w = weights[r] * dx
        if abs(w) < 1e-16:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n <= 0:
                continue
            g[d:] += w * (1.0 - h[:n])
            g[:n] -= w * h[d:]
        else:
            dd = -d
            n = N - dd
            if n <= 0:
                continue
            g[dd:] += w * (1.0 - h[:n])
            g[:n] -= w * h[dd:]
    return g

    h_new = np.interp(x_new, x_old, h)

def project_feasible(h, target_sum):

            mask = slack > 1e-10
            if mask.sum() > 0:

        else:

    h = np.clip(h, 0.0, 1.0)

    if abs(residual) > 1e-14:

        else:

            if len(idx) > 0:
                h[idx[0]] = float(np.clip(h[idx[0]] + residual, 0, 1))
