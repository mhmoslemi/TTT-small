import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TAU = 1.0
_G_TOPK = 10

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(c.max())

def _project_fey(y, target):
    """Project y onto {0<=h<=1, sum(h)=target} via bisection on offset"""
    lo, hi = -2.0, 3.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-8) & (h < 1.0 - 1e-8)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _obj_topk(h):
    n = _G_N
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    k = min(_G_TOPK, len(c))
    topk = np.sort(c)[-k:]
    s = np.clip((topk - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()) - _G_TAU * np.log(float(k)))

def _grad_topk(h):
    n = _G_N
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()

    e = np.exp(s)

    omh = 1.0 - h
    for t in range(k):
        a = alpha[t]
        d = int(topk_idx[t]) - (n - 1)
        lo = max(0, d)

def _interp(parent, n_child):

    if n_child == n_par:

    xo = np.linspace(0.0, 2.0, n_par, endpoint=False)

    h = np.interp(xn, xo, parent)
    return _project_fey(h, n_child / 2.0)

def _slsqp_topk(h0, n, deadline, k_start=20, k_end=3, tau_start=0.04, tau_end=0.003):
    """SLSQP with top-K weighted objective, adaptive K and tau"""
    global _G_N, _G_TOPK, _G_TAU
    _G_N = n
    target = n / 2.0
    
    best_h = h0.copy()
    best_c5 = _eval_c5(h0)
    
    # Schedule: 16 iterations, K goes from k_start to k_end, tau from tau_start to tau_end
    n_steps = 16
    for i in range(n_steps):
        if time.time() > deadline - 5:
            break
        frac = i / max(1, n_steps - 1)
        # K schedule: exponential decay
        _G_TOPK = max(k_end, int(round(k_start * (k_end / k_start) ** frac)))
        # Tau schedule
        _G_TAU = tau_start * (tau_end / tau_start) ** frac
        
        h_curr = best_h.copy()
        maxiter = 60 if _G_TAU > 0.01 else 100
        try:
            res = minimize(_obj_topk, h_curr, method='SLSQP', jac=_grad_topk,
                          bounds=[(0.0, 1.0)] * n,
                          constraints=[{'type': 'eq', 'fun': _eq_fun, 'jac': _eq_jac}],
                          options={'maxiter': maxiter, 'ftol': 1e-10})
            h_curr = _project_fey(res.x, target)
        except Exception:
            pass
        
        c5 = _eval_c5(h_curr)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h_curr.copy()

def _active_descent(h0, n, deadline, n_iter=2000):

    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _eval_c5(h0)
    
    dx = 2.0 / n
    
    for it in range(n_iter):

        c = np.correlate(h, 1.0 - h, 'full') * dx

            active = np.array([np.argmax(c)])

        grad = np.zeros(n)
        for k in active:

def _pair_refine(h0, n, deadline, n_passes=20):

    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _eval_c5(h0)
    c = np.correlate(best_h, 1.0 - best_h, 'full') * (2.0/n)
    active_thresh = c.max() - 0.003
    active = np.where(c >= active_thresh)[0]
    
    for p in range(n_passes):
        if time.time() > deadline - 5:
            break
        
        # Recompute gradient for direction guidance
        c = np.correlate(best_h, 1.0 - best_h, 'full') * (2.0/n)
        cmax = c.max()
        active_thresh = cmax - 0.003
        active = np.where(c >= active_thresh)[0]
        
        # Compute gradient of max (averaged over active)
        grad = np.zeros(n)
        for k in active:
            d = int(k) - (n - 1)
            lo = max(0, d)
            hi = min(n - 1, n - 1 + d)
            if lo <= hi:
                grad[lo:hi+1] += (1 - best_h[lo-d:hi+1-d])
            lo2 = max(0, -d)
            hi2 = min(n - 1, n - 1 - d)
            if lo2 <= hi2:
                grad[lo2:hi2+1] -= best_h[lo2+d:hi2+1+d]
        grad /= max(1, len(active))
        
        # Positions to decrease (high gradient) and increase (low gradient)
        dec_positions = np.where(grad > np.percentile(grad, 80))[0]
        inc_positions = np.where(grad < np.percentile(grad, 20))[0]
        
        if len(dec_positions) == 0 or len(inc_positions) == 0:
            continue
        
        # Try random pairs
        rng = np.random.default_rng(p * 1000 + n)
        n_trials = 500

        for _ in range(n_trials):
            if time.time() > deadline - 3:
                break
            
            i = rng.integers(len(dec_positions))
            j = rng.integers(len(inc_positions))
            i = int(dec_positions[i])
            j = int(inc_positions[j])
            if i == j:

            for delta in [0.005, 0.01, 0.02, 0.002]:

                h_try[i] -= delta

                    best_h = h_try
                    best_c5 = c5_try
                    improved = True
                    break

    return best_h, best_c5

import numpy as np
from scipy.optimize import minimize
import time

_G_N = 0
_G_TAU = 1.0
_G_TOPK = 10

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(c.max())

def _project_fey(y, target):
    lo, hi = -3.0, 4.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        cnt = int(mask.sum())
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def _grad_at_lag(h, d):
    n = len(h)
    g = np.zeros(n)
    omh = 1.0 - h
    lo = max(0, d)
    hi = min(n - 1, n - 1 + d)
    if lo <= hi:
        g[lo:hi+1] += omh[lo-d:hi+1-d]
    lo2 = max(0, -d)
    hi2 = min(n - 1, n - 1 - d)
    if lo2 <= hi2:
        g[lo2:hi2+1] -= h[lo2+d:hi2+1+d]
    return g

def _obj_topk(h):
    n = _G_N
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    k = min(_G_TOPK, len(c))
    topk = np.sort(c)[-k:]
    s = np.clip((topk - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()) - _G_TAU * np.log(float(k)))

def _grad_topk(h):
    n = _G_N
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    k = min(_G_TOPK, len(c))
    topk_idx = np.argsort(c)[-k:]
    topk_vals = c[topk_idx]
    s = np.clip((topk_vals - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()
    grad = np.zeros(n)
    for t in range(k):
        a = alpha[t]
        d = int(topk_idx[t]) - (n - 1)
        grad += a * _grad_at_lag(h, d)
    return grad * dx

def _obj_global(h):
    n = _G_N
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    return float(m + _G_TAU * np.log(e.sum()))

def _grad_global(h):
    n = _G_N
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    m = c.max()
    s = np.clip((c - m) / _G_TAU, -500.0, 500.0)
    e = np.exp(s)
    alpha = e / e.sum()

        a = alpha[k]
        if a < 1e-14:

        d = k - (n - 1)
        grad += a * _grad_at_lag(h, d)
    return grad * dx

def _eq_fun(h):
    return float(np.sum(h) - _G_N / 2.0)

def _eq_jac(h):
    return np.ones(_G_N)

def _interp(parent, n_child):
    n_par = len(parent)
    if n_child == n_par:
        return parent.copy()
    xo = np.linspace(0.0, 2.0, n_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, n_child, endpoint=False)
    h = np.interp(xn, xo, parent)
    return _project_fey(h, n_child / 2.0)

def _slsqp_optimize(h0, n, deadline, use_topk=True):

    _G_N = n
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = _eval_c5(h0)

        else:

            jac_fn = _grad_global

def _active_descent(h0, n, deadline):

    best_h = h0.copy()
    best_c5 = _eval_c5(h0)
    dx = 2.0 / n
    momentum = np.zeros(n)
    beta = 0.3

            break

        if len(active) == 0:
            active = np.array([np.argmax(c)])

        for k in active:
            d = int(k) - (n - 1)
            grad += _grad_at_lag(h, d)
        grad /= float(len(active))
        grad *= dx
        grad_tan = grad - np.mean(grad)

        momentum = beta * momentum + (1 - beta) * grad_tan
        direction = -momentum - 0.5 * grad_tan

        norm_dir = np.linalg.norm(direction)
        if norm_dir < 1e-14:

                best_c5 = c5_try
                accepted = True

                break

    return best_h, best_c5

def _pair_refine(h0, n, deadline):
    target = n / 2.0
    best_h = h0.copy()

    dx = 2.0 / n

    deltas_coarse = [0.03, 0.015, 0.008]

            for k in active:

                grad += _grad_at_lag(best_h, d)
            grad /= float(len(active))

            dec_pct = np.percentile(grad, 85)
            inc_pct = np.percentile(grad, 15)
            dec_pos = np.where(grad > dec_pct)[0]
            inc_pos = np.where(grad < inc_pct)[0]

            if len(dec_pos) == 0 or len(inc_pos) == 0:
                dec_pos = np.argsort(grad)[-n // 10:]
                inc_pos = np.argsort(grad)[:n // 10]

            rng = np.random.default_rng(p * 31337 + n + (0 if phase is deltas_coarse else 1))
            n_trials = 400
            improved_this_pass = False

            for _ in range(n_trials):
                if time.time() > deadline - 2:
                    break
                di = int(dec_pos[rng.integers(len(dec_pos))])
                ii = int(inc_pos[rng.integers(len(inc_pos))])
                if di == ii:
                    continue
                for delta in phase:
                    if best_h[di] < delta or best_h[ii] > 1.0 - delta:
                        continue
                    h_try = best_h.copy()
                    h_try[di] -= delta
                    h_try[ii] += delta
                    c5_try = _eval_c5(h_try)
                    if c5_try < best_c5 - 1e-12:

                        best_c5 = c5_try
                        improved_this_pass = True
                        break
