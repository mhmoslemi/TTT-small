def get_parent():
    g = globals()

def project_to_feasible(h, n):
    """Project h to {0<=h<=1, sum(h) = n/2}"""
    target = n / 2.0
    h = np.asarray(h, dtype=float)
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    # Bisection on threshold
    lo, hi = -1.5, 1.5
    for _ in range(60):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    # Fix residual
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d > 0:
            mask = out > 0
            if mask.sum() > 0:
                out[mask] -= d / mask.sum()
                out = np.clip(out, 0.0, 1.0)
        else:
            mask = out < 1
            if mask.sum() > 0:
                out[mask] += (-d) / mask.sum()
                out = np.clip(out, 0.0, 1.0)
    # Final exact fix
    d = out.sum() - target
    if abs(d) > 1e-14:
        if d > 0:
            idx = np.argmax(out)
        else:
            idx = np.argmin(out)
        out[idx] = np.clip(out[idx] - d, 0.0, 1.0)

def sturmian_binary(n, slope, phase):
    """Generate a Sturmian sequence of length n with given slope and phase."""
    h = np.zeros(n)
    for i in range(n):
        val = math.floor((i + 1) * slope + phase) - math.floor(i * slope + phase)
        h[i] = val
    return h

def make_binary_from_sturmian(n, slope, phase, rng):
    """Make a balanced binary sequence from Sturmian, adjusted to have exactly n//2 ones."""
    h = sturmian_binary(n, slope, phase)
    n_ones = int(h.sum())
    target = n // 2
    if n_ones == target:
        return h
    elif n_ones > target:
        # Remove some ones
        ones_idx = np.where(h == 1)[0]
        rng.shuffle(ones_idx)
        for i in ones_idx[:n_ones - target]:
            h[i] = 0
    else:
        # Add some ones
        zeros_idx = np.where(h == 0)[0]
        rng.shuffle(zeros_idx)
        for i in zeros_idx[:target - n_ones]:
            h[i] = 1
    return h

def lse_val(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    return dx * (m + math.log(np.exp(sigma * (c - m)).sum()) / sigma)

def lse_grad(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * h[p:]
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * h[:-p]
    grad *= dx
    return grad

def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

def differential_evolution_pop(pop, fitness, rng, n_gen, n, deadline=None):
    """Simple DE with mutation and local refinement."""
    NP = len(pop)
    best_idx = np.argmin(fitness)
    best_h = pop[best_idx].copy()
    best_c = fitness[best_idx]
    
    for gen in range(n_gen):
        if deadline and time.monotonic() > deadline:
            break
        for i in range(NP):
            if deadline and time.monotonic() > deadline:
                break
            # Mutation
            a, b, c = rng.choice(NP, 3, replace=False)
            # Ensure they're different from i
            while a == i or b == i or c == i:
                a, b, c = rng.choice(NP, 3, replace=False)
            
            F = 0.5 + 0.3 * rng.random()
            v = pop[a] + F * (pop[b] - pop[c])
            
            # Crossover (binomial)
            CR = 0.7
            j_rand = rng.integers(n)
            u = np.zeros(n)
            for j in range(n):
                if rng.random() < CR or j == j_rand:
                    u[j] = v[j]
                else:
                    u[j] = pop[i][j]
            
            # Repair
            u = project_to_feasible(u, n)
            new_c = c5_value(u)
            
            if new_c <= fitness[i]:
                pop[i] = u
                fitness[i] = new_c
                if new_c < best_c:
                    best_c = new_c
                    best_h = u.copy()
    
    return best_h, best_c

            m = out > 0
            if m.any():
                out[m] = np.clip(out[m] - d/m.sum(), 0, 1)
        else:
            m = out < 1
            if m.any():
                out[m] = np.clip(out[m] - d/m.sum(), 0, 1)
    d = out.sum() - target
    if abs(d) > 1e-14:
        idx = int(np.argmax(out)) if d > 0 else int(np.argmin(out))
        out[idx] = float(np.clip(out[idx] - d, 0, 1))
    return out

def c5val(h):
    return float(np.correlate(h, 1.0-h, mode='full').max() * 2.0 / len(h))

def sturmian(n, slope, phase):
    out = np.zeros(n)
    for i in range(n):
        out[i] = math.floor((i+1)*slope + phase) - math.floor(i*slope + phase)
    return out

def make_binary(n, rng, type_str, params):
    """Generate a binary sequence with exactly n//2 ones."""
    if type_str == 'sturmian':
        slope, phase = params
        h = sturmian(n, slope, phase)
    elif type_str == 'random':
        h = (rng.random(n) < 0.5).astype(float)
    elif type_str == 'parent_bin':
        h = (params >= 0.5).astype(float)
        h = h[:n] if len(h) >= n else np.pad(h, (0, n-len(h)))
    else:
        h = (rng.random(n) < 0.5).astype(float)
    
    # Balance to exactly n//2 ones
    n_ones = int(h.sum())
    target = n // 2
    h = h.copy()
    if n_ones > target:
        idx = np.where(h == 1)[0]
        rng.shuffle(idx)
        h[idx[:n_ones-target]] = 0
    elif n_ones < target:
        idx = np.where(h == 0)[0]
        rng.shuffle(idx)
        h[idx[:target-n_ones]] = 1
    return h
