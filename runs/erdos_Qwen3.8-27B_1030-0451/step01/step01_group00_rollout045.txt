import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    """Project p onto {0 <= h <= 1, sum(h) = n/2}"""
    target = n / 2.0
    p = np.clip(p, -2.0, 2.0)  # safe clip
    lo = float(np.min(p)) - 2.0
    hi = float(np.max(p)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    # Fix residual
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            proj[mask] = np.clip(proj[mask] + adj, 0.0, 1.0)
    # Final enforcement
    proj = np.clip(proj, 0.0, 1.0)
    return proj

def grad_c_k(h, lag):
    """Compute gradient of c[lag] w.r.t. h, where lag = k - (n-1)"""
    n = len(h)
    g = np.zeros(n)
    if lag >= 0:

            g[p] -= h[p + lag]
    else:

    w = np.exp(tau * c)
    ws = np.sum(w)
    if ws < 1e-300:
        return np.full(len(corr), 1.0 / len(corr))
    return w / ws

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    p = np.nan_to_num(p, nan=0.5, posinf=1.0, neginf=0.0)
    lo = float(np.min(p)) - 2.0
    hi = float(np.max(p)) + 2.0

        mid = 0.5 * (lo + hi)

        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))

    w = np.zeros(n_corr)

    c = top_vals - np.max(top_vals)
    exp_c = np.exp(tau * c)
    w[top_idxs] = exp_c / np.sum(exp_c)

            break
        seed_try = seed + n_try * 7
        h_opt, c5_opt = optimize_gradient(h_res, alloc_grad, seed_val=seed_try)
        remaining2 = budget_s - (time.time() - t_start) - 80
        if remaining2 > 10:
            h_cd, c5_cd = coordinate_polish(h_opt, min(remaining2, alloc_cd), seed_val=seed_try + 1)
            if c5_cd < c5_opt:
                c5_opt = c5_cd
                h_opt = h_cd
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt
            n_current = n_try

    # Phase 4: More aggressive gradient on best
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 30
    if remaining > 20:
        h_opt, c5_opt = optimize_gradient(best_h.copy(), remaining, tau_start=40, tau_end=900, topk_frac=0.2, seed_val=seed + 500)
        if c5_opt < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_opt

    # Phase 5: Final coordinate polish
    elapsed = time.time() - t_start
    remaining = budget_s - elapsed - 5
    if remaining > 8:
        h_cd, c5_cd = coordinate_polish(best_h.copy(), remaining, seed_val=seed + 999)
        if c5_cd < best_c5:
            best_h = h_cd.copy()
            best_c5 = c5_cd

    # Final verification and cleanup
    best_h = project_constraint(best_h, len(best_h))
    best_c5 = compute_c5(best_h)

    return best_h, best_c5, len(best_h)
