def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid

                c += 1.0 - h[j + lag]
            if 0 <= j - lag < n:
                c -= h[j - lag]

  = Σ h[i+k]*(1-h[i])*dx = corr_k(h) [by reindexing, for symmetric lags]
  Actually, corr_k(1-h) = corr_{-k}(h) by symmetry. So the max is the same.
  This means h and 1-h give the same C₅. So we can assume WLOG that h has some symmetry.

- Reversal: h_rev[i] = h[n-1-i]. This preserves sum and box constraints. The correlations are the same (just negated lags). So C₅ is preserved.

- Translation (circular): h_circ[i] = h[(i+s) mod n]. This preserves everything for the periodic case, but for the non-periodic case, it changes boundary effects.

These symmetries can be used to reduce the search space, but for a heuristic, they're not critical.

import numpy as np

def project_sum_box(x, target):
    n = len(x)
    x = np.array(x, dtype=float).copy()
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    return np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def corr_all(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_coef(h):
    n = len(h)
    n_lags = 2 * n - 1
    coef = np.zeros((n_lags, n))
    # For each position pos (0 to n_lags-1), lag = pos - (n-1)
    # coef[pos, j] = (1-h[j+lag])*1_{0<=j+lag<n} - h[j-lag]*1_{0<=j-lag<n}
    for pos in range(n_lags):
        lag = pos - (n - 1)
        # j+lag range: j from max(0,-lag) to min(n-1, n-1-lag)
        # j-lag range: j from max(0,lag) to min(n-1, n-1+lag)
        for j in range(n):
            c = 0.0
            if 0 <= j + lag < n:
                c += 1.0 - h[j + lag]
            if 0 <= j - lag < n:
                c -= h[j - lag]
            coef[pos, j] = c
    return coef

            coef[pos, idx_1] += 1.0 - h[idx_1 + lag]
        # Second term: -h[j-lag] for valid j
        j_start_2 = max(0, lag)
        j_end_2 = min(n, n + lag)  # exclusive... wait
        # j-lag in [0, n-1] means j in [lag, n-1+lag], intersect with [0, n-1]
        j_start_2 = max(0, lag)
        j_end_2 = min(n, n + lag)  # hmm, j < n and j-lag < n means j < n+lag
        # Actually j-lag >= 0 means j >= lag, j-lag <= n-1 means j <= n-1+lag
        # Combined with 0 <= j <= n-1: j in [max(0,lag), min(n-1, n-1+lag)]
        j_start_2 = max(0, lag) if lag >= 0 else 0
        j_end_2 = min(n, n + lag) if lag < 0 else n
        # Wait, if lag >= 0: j in [lag, n-1] (since j-lag >= 0 requires j >= lag, and j <= n-1)
        #   j-lag <= n-1 is automatic since j <= n-1 and lag >= 0
        # If lag < 0: j-lag = j+|lag|, need j+|lag| <= n-1, so j <= n-1-|lag| = n+lag-1
        #   Also j >= 0. So j in [0, n+lag-1]
        if lag >= 0:
            j_start_2 = lag
            j_end_2 = n
        else:
            j_start_2 = 0
            j_end_2 = n + lag  # = n - |lag|
        if j_start_2 < j_end_2:
            idx_2 = np.arange(j_start_2, j_end_2)
            coef[pos, idx_2] -= h[idx_2 - lag]
    return coef

def pairwise_descent(h, deadline, max_rounds=500, grid_size=80):
    n = len(h)
    dx = 2.0 / n
    n_lags = 2 * n - 1
    h = h.copy()
    best_c = evaluate_c5(h)
    best_h = h.copy()
    coef = compute_coef(h)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        c_cur = corr_all(h)
        current_max = np.max(c_cur)
        
        for i in range(n):
            if time.time() > deadline:
                break
            for j in range(i + 1, n):
                if abs(h[j] - h[i]) < 1e-4:
                    continue
                
                a = dx * (coef[:, j] - coef[:, i])
                
                eps_lo = max(-h[i], h[j] - 1.0)
                eps_hi = min(1.0 - h[i], h[j])
                
                if eps_hi - eps_lo < 1e-8:
                    continue
                
                # Quick upper bound: min possible value of max(c_cur + a*eps)
                # For a given eps, the value is max(c_cur + a*eps)
                # The minimum over eps is at least min(max(c_cur + a*eps_lo), max(c_cur + a*eps_hi))
                # But that's not a tight bound. Let's just do the grid.
                
                eps_grid = np.linspace(eps_lo, eps_hi, grid_size)
                # f_vals[p] = max over all lags of (c_cur + a * eps_grid[p])
                f_vals = np.max(c_cur[None, :] + a[None, :] * eps_grid[:, None], axis=1)
                best_idx = np.argmin(f_vals)
                
                if f_vals[best_idx] >= current_max - 1e-12:
                    continue
                
                best_eps = float(eps_grid[best_idx])
                
                # Refine
                lo_idx = max(0, best_idx - 1)
                hi_idx = min(grid_size - 1, best_idx + 1)
                eps_fine = np.linspace(eps_grid[lo_idx], eps_grid[hi_idx], 30)
                f_fine = np.max(c_cur[None, :] + a[None, :] * eps_fine[:, None], axis=1)
                best_eps = float(eps_fine[np.argmin(f_fine)])
                
                h_new = h.copy()
                h_new[i] += best_eps
                h_new[j] -= best_eps
                h_new = np.clip(h_new, 0.0, 1.0)
                
                c_new = evaluate_c5(h_new)
                if c_new < best_c - 1e-14:
                    # Update coef incrementally
                    old_hi, old_li = h[j], h[i]
                    h = h_new
                    best_h = h_new.copy()
                    best_c = c_new
                    improved = True
                    
                    # Update coef
                    eps_used = best_eps
                    # pos = n-1 (lag 0): coef[n-1,i] changes by -2*eps, coef[n-1,j] by +2*eps
                    coef[n-1, i] -= 2.0 * eps_used
                    coef[n-1, j] += 2.0 * eps_used
                    # pos = j-i+n-1 (lag = j-i)
                    p1 = j - i + n - 1
                    if 0 <= p1 < n_lags:
                        coef[p1, i] += eps_used
                        coef[p1, j] -= eps_used
                    # pos = i-j+n-1 (lag = i-j)
                    p2 = i - j + n - 1
                    if 0 <= p2 < n_lags:
                        coef[p2, i] += eps_used
                        coef[p2, j] -= eps_used
                    
                    c_cur = corr_all(h)
                    current_max = np.max(c_cur)
        
        if not improved:
            break
    
    return best_h, best_c
