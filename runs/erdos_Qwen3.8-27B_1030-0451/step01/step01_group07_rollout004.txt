    """Try moving delta mass from i to j or j to i."""
    h2 = h.copy()

        h2[j] += delta
        return h2

        h2[i] += delta
        return h2
    return None

    weights = 2.0 * excess
    grad = np.zeros(n)
    center = n - 1
    w0 = weights[center]
    if w0 > 1e-18:
        grad += w0 * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = weights[center + p]
        wm = weights[center - p]
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * h[p:]
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * h[:-p]
    return grad

def lse_grad(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * h[p:]
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * h[:-p]
    grad *= dx
    return grad

            tau = cur_c5_now - tau_margin

            m = c_raw.max()

            g = lse_grad(h, sigma)

def population_search(h_init, rng, target, time_limit, pop_size, deadline):
    n = len(h_init)
    pop = [h_init.copy()]
    for i in range(1, pop_size):
        mutant = project_exact(h_init + rng.normal(0.0, 0.15, n), target)
        pop.append(mutant)
    scores = [objective(p) for p in pop]
    best_h = pop[int(np.argmin(scores))].copy()
    best_c = float(min(scores))
    
    t_start = time.monotonic()
    
    while True:
        if time.monotonic() - t_start >= time_limit:
            break
        if deadline is not None and time.monotonic() > deadline:
            break
        
        i1 = int(rng.integers(pop_size))
        i2 = int(rng.integers(pop_size))
        if scores[i1] < scores[i2]:
            p1, p2 = pop[i1], pop[i2]
        else:
            p1, p2 = pop[i2], pop[i1]
        
        F = 0.5 + 0.5 * rng.random()
        mutant = project_exact(p1 + F * (p1 - p2), target)
        
        if rng.random() < 0.15:
            mutant = project_exact(p1 + rng.normal(0.0, 0.08, n), target)
        
        mc = objective(mutant)
        worst = int(np.argmax(scores))
        if mc < scores[worst]:
            pop[worst] = mutant
            scores[worst] = mc
            if mc < best_c - 1e-14:
                best_c = mc
                best_h = mutant.copy()
    
    return best_h, best_c

def coordinate_descent_round(h, rng, target, max_moves):
    n = len(h)
    cur = objective(h)
    best_h = h.copy()
    best_c = cur
    for _ in range(max_moves):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        for d in [0.05, 0.15, 0.3]:
            cand = h.copy()
            if d <= cand[i] and d <= 1.0 - cand[j]:
                cand[i] -= d
                cand[j] += d
            elif d <= cand[j] and d <= 1.0 - cand[i]:
                cand[j] -= d
                cand[i] += d
            else:
                continue
            c = objective(cand)
            if c < cur - 1e-14:
                h = cand
                cur = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                break
    return best_h, best_c

def resample_h(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)

def sturmian_h(n, rho, phi):
    h = np.zeros(n)
    for i in range(n):
        if math.floor((i + 1) * rho + phi) > math.floor(i * rho + phi):
            h[i] = 1.0
    target = n // 2
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0.0)[0]
        k = 0
        while current < target and k < len(zeros):
            h[zeros[k]] = 1.0
            current += 1
            k += 1
    elif current > target:
        ones = np.where(h == 1.0)[0]
        k = 0
        while current > target and k < len(ones):
            h[ones[k]] = 0.0
            current -= 1
            k += 1
    return h

import time

def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size > 0 and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def project_exact(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    lo = -1.5
    hi = 1.5
    for _ in range(80):
        t = 0.5 * (lo + hi)
        s = np.clip(h - t, 0.0, 1.0).sum()
        if s > target:
            lo = t
        else:
            hi = t
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-10:
        if d < 0:
            slack = 1.0 - out
            room = slack.sum()
            if room > 1e-15:
                out = np.clip(out + (-d / room) * slack, 0.0, 1.0)
        else:
            slack = out
            room = slack.sum()
            if room > 1e-15:
                out = np.clip(out - (d / room) * slack, 0.0, 1.0)
    d = out.sum() - target
    n = len(out)
    if n > 0 and abs(d) > 1e-14:
        if d > 0:
            idx = int(np.argmax(out))
        else:
            idx = int(np.argmax(1.0 - out))
        out = out.copy()
        out[idx] = float(np.clip(out[idx] - d, 0.0, 1.0))
    return out


def objective(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def corr_scaled(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx


def active_grad(h, tau):
    n = len(h)
    c = corr_scaled(h)
    excess = np.maximum(0.0, c - tau)
    weights = 2.0 * excess
    grad = np.zeros(n)
    center = n - 1
    w0 = weights[center]
    if w0 > 1e-18:
        grad += w0 * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = weights[center + p]
        wm = weights[center - p]
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * h[p:]
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * h[:-p]
    return grad


def lse_grad(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * h[p:]
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * h[:-p]
    grad *= dx
    return grad


def lse_val(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    return float((m + math.log(e.sum()) / sigma) * dx)


def grad_polish_active(h, rng, target, time_limit, deadline, tau_margin=0.001):
    n = len(h)
    h = h.copy()
    cur_c5 = objective(h)
    best_h = h.copy()
    best_c5 = cur_c5
    eta = 0.04
    t_start = time.monotonic()
    stall = 0
    while True:
        if time.monotonic() - t_start >= time_limit:
            break
        if deadline is not None and time.monotonic() > deadline:
            break
        cur_c5_now = objective(h)
        tau = max(cur_c5_now - tau_margin, 0.0)
        c = corr_scaled(h)
        excess = np.maximum(0.0, c - tau)
        val = float(np.sum(excess * excess))
        if val < 1e-20:
            break
        g = active_grad(h, tau)
        gnorm = float(np.linalg.norm(g))
        if gnorm < 1e-14:
            break
        improved = False
        e_step = eta

            if new_val < val - 1e-16:
                h = cand
                val = new_val
                if c5 < best_c5 - 1e-14:
                    best_c5 = c5
                    best_h = h.copy()
                improved = True

            e_step *= 0.5

            tau = max(c5 - tau_margin, 0.0)

def pop_search(h_init, rng, target, time_limit, pop_size, deadline):
    n = len(h_init)
    pop = [h_init.copy()]
    for i in range(1, pop_size):
        pop.append(project_exact(h_init + rng.normal(0.0, 0.12, n), target))
    scores = [objective(p) for p in pop]
    best_idx = int(np.argmin(scores))
    best_h = pop[best_idx].copy()
    best_c = scores[best_idx]
    t_start = time.monotonic()
    while True:
        if time.monotonic() - t_start >= time_limit:
            break
        if deadline is not None and time.monotonic() > deadline:
            break
        i1 = int(rng.integers(pop_size))
        i2 = int(rng.integers(pop_size))
        if scores[i1] <= scores[i2]:
            p1, p2 = pop[i1], pop[i2]
        else:
            p1, p2 = pop[i2], pop[i1]
        F = 0.4 + 0.6 * rng.random()
        mutant = project_exact(p1 + F * (p1 - p2), target)
        if rng.random() < 0.12:
            mutant = project_exact(p1 + rng.normal(0.0, 0.06, n), target)
        mc = objective(mutant)
        worst = int(np.argmax(scores))
        if mc < scores[worst] - 1e-15:
            pop[worst] = mutant
            scores[worst] = mc
            if mc < best_c - 1e-14:
                best_c = mc
                best_h = mutant.copy()
    return best_h, best_c


def coord_round(h, rng, target, max_moves):
    n = len(h)
    cur = objective(h)
    best_h = h.copy()
    best_c = cur
    deltas = [0.02, 0.05, 0.1, 0.2, 0.4]
    for _ in range(max_moves):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i == j:
            continue
        for d in deltas:
            cand = h.copy()
            applied = False
            if d <= cand[i] + 1e-15 and d <= 1.0 - cand[j] + 1e-15:
                cand[i] = max(0.0, cand[i] - d)
                cand[j] = min(1.0, cand[j] + d)
                applied = True
            elif d <= cand[j] + 1e-15 and d <= 1.0 - cand[i] + 1e-15:
                cand[j] = max(0.0, cand[j] - d)
                cand[i] = min(1.0, cand[i] + d)
                applied = True
            if not applied:
                continue
            c = objective(cand)
            if c < cur - 1e-14:
                h = cand
                cur = c
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
                break
    return best_h, best_c


def resample_h(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)


def sturmian_h(n, rho, phi):
    h = np.zeros(n)
    for i in range(n):
        if math.floor((i + 1) * rho + phi) > math.floor(i * rho + phi):
            h[i] = 1.0
    target = n // 2
    current = int(h.sum())
    if current < target:
        zeros = np.where(h == 0.0)[0]
        k = 0
        while current < target and k < len(zeros):
            h[zeros[k]] = 1.0
            current += 1
            k += 1
    elif current > target:
        ones = np.where(h == 1.0)[0]
        k = 0
        while current > target and k < len(ones):
            h[ones[k]] = 0.0
            current -= 1
            k += 1
    return h


def anneal_local(h, rng, target, time_limit, deadline):

    h = h.copy()
    cur = objective(h)
    best_h = h.copy()
    best_c = cur
    T = 1e-4

            cand[j] += d
        elif d <= cand[j] + 1e-15 and d <= 1.0 - cand[i] + 1e-15:

            cand[i] += d
        else:

        nc = objective(cand)
        delta = nc - cur
        frac = min(1.0, (time.monotonic() - t_start) / total)
