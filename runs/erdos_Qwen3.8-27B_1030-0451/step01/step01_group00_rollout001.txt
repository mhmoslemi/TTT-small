def project_feasible(p, n):
    """Project p onto {0<=x<=1, sum(x)=n/2}."""
    target = n / 2.0
    # Binary search for lambda
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-10) & (proj < 1.0 - 1e-10)
        nm = int(np.sum(mask))
        if nm > 0:
            adj = diff / nm
            # Make sure adjustment keeps within bounds
            for i in np.where(mask)[0]:
                proj[i] += adj
            proj = np.clip(proj, 0.0, 1.0)
            # Re-correct if clipping changed sum
            diff2 = target - float(np.sum(proj))
            if abs(diff2) > 1e-12:
                mask2 = (proj > 1e-10) & (proj < 1.0 - 1e-10)
                nm2 = int(np.sum(mask2))
                if nm2 > 0:
                    proj[mask2] += diff2 / nm2
    return proj
