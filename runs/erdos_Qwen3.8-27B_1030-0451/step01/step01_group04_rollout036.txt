def compute_active_gradient(h, active_lags, N):
    """Compute weighted gradient of active correlations.
    active_lags: array of lag values (correlation indices, 0 to 2N-2)
    """
    g = np.zeros(N)
    for k in active_lags:
        d = k - (N - 1)  # actual lag
        if d >= 0:
            # f_d = sum_{n=0}^{N-1-d} h[n+d] * (1-h[n])
            # grad[i] = (1-h[i-d]) for i >= d, minus h[i+d] for i <= N-1-d
            if d == 0:
                g += 1.0 - 2.0 * h
            else:
                # (1-h[i-d]) for i in [d, N-1]
                g[d:] += (1.0 - h[:N-d])
                # -h[i+d] for i in [0, N-1-d]
                g[:N-d] -= h[d:]
        else:
            e = -d
            # f_{-e} = sum_{m=0}^{N-1-e} h[m] * (1-h[m+e])
            # grad[i] = (1-h[i+e]) for i in [0, N-1-e], minus h[i-e] for i in [e, N-1]
            g[:N-e] += (1.0 - h[e:])
            g[e:] -= h[:N-e]
    return g

**Key insight about the problem structure:**

At the optimum, we'd want both f_k and f_{-k} to be small for all k. But they're not independent - reducing one might increase the other.

import numpy as np
import time
from scipy.optimize import minimize_scalar


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def ensure_feasible(h, target_sum):
    """Project h to satisfy sum(h)=target_sum and 0<=h<=1."""
    h = np.array(h, dtype=float, copy=True)
    h = np.clip(h, 0.0, 1.0)
    # Adjust sum using bisection
    current_sum = h.sum()
    if abs(current_sum - target_sum) < 1e-12:
        return h
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) / 2.0
    h = np.clip(h + t, 0.0, 1.0)
    # Final correction
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        if residual > 0:
            idx = np.argmin(h)  # smallest value, most room to increase
            h[idx] = min(1.0, h[idx] + residual)
        else:
            idx = np.argmax(h)  # largest value, most room to decrease
            h[idx] = max(0.0, h[idx] + residual)
    return h


def feasible_interval(h, d):
    """Find [t_min, t_max] such that 0 <= h[i] + t*d[i] <= 1 for all i."""
    mask = np.abs(d) > 1e-15
    if not np.any(mask):
        return -1.0, 1.0
    t_min = 0.0
    t_max = 0.0
    # For d[i] > 0: t >= -h[i]/d[i], t <= (1-h[i])/d[i]
    pos = mask & (d > 0)
    if np.any(pos):
        t_min = max(t_min, np.max(-h[pos] / d[pos]))
        t_max = min(t_max, np.min((1.0 - h[pos]) / d[pos]))
    # For d[i] < 0: t >= (1-h[i])/d[i], t <= -h[i]/d[i]
    neg = mask & (d < 0)
    if np.any(neg):
        t_min = max(t_min, np.max((1.0 - h[neg]) / d[neg]))
        t_max = min(t_max, np.min(-h[neg] / d[neg]))
    return t_min, t_max


def active_gradient(h, N, active_lags, weights):
    """Compute weighted gradient of active correlations."""
    g = np.zeros(N)
    for k, w in zip(active_lags, weights):
        d = k - (N - 1)
        if d >= 0:
            if d == 0:
                g += w * (1.0 - 2.0 * h)
            else:
                g[d:] += w * (1.0 - h[:N-d])
                g[:N-d] -= w * h[d:]
        else:
            e = -d
            g[:N-e] += w * (1.0 - h[e:])
            g[e:] -= w * h[:N-e]
    return g


def line_search(h, d, target_sum):
    """Find best t to minimize C5(h + t*d) over feasible interval."""
    t_min, t_max = feasible_interval(h, d)
    if t_max - t_min < 1e-14:
        return 0.0, compute_c5(h)
    
    # Add small margin
    margin = 1e-12
    t_min += margin
    t_max -= margin
    if t_max <= t_min:
        return 0.0, compute_c5(h)
    
    def obj(t):
        h_try = h + t * d
        h_try = np.clip(h_try, 0.0, 1.0)
        return compute_c5(h_try)
    
    try:
        result = minimize_scalar(obj, bounds=(t_min, t_max), method='bounded',
                                 options={'xatol': 1e-10})
        t_best = result.x
        val_best = result.fun
    except Exception:
        # Fallback: sample a few points
        t_vals = np.linspace(t_min, t_max, 20)
        best_t = 0.0
        best_val = compute_c5(h)
        for t in t_vals:
            h_try = np.clip(h + t * d, 0.0, 1.0)
            val = compute_c5(h_try)
            if val < best_val:
                best_val = val
                best_t = t
        t_best = best_t
        val_best = best_val
    
    return t_best, val_best

            e = -d

            g[e:] -= w * h[:N-e]
    return g
