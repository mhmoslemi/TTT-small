    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    # Get top m correlation indices
    top_idx = np.argsort(corr)[::-1][:top_m]
    
    # Build gradient matrix for these lags
    # A[i, j] = dc_{k_i}/dh[j] * dx
    A = np.zeros((len(top_idx), n))
    for row, idx in enumerate(top_idx):
        lag = idx - (n - 1)
        if lag >= 0:
            for j in range(lag, n):
                A[row, j] += dx * (1.0 - h[j - lag])
            for j in range(n - lag):
                A[row, j] -= dx * h[j + lag]
        else:
            L = -lag
            for j in range(n - L):
                A[row, j] += dx * (1.0 - h[j])
            for j in range(L, n):
                A[row, j] -= dx * h[j - L]
    
    # LP: minimize t
    # subject to: A @ d - t <= -c_active (i.e., c_active + A@d <= t)

    #             -eps <= d_i <= eps
    # Variables: [d_0, ..., d_{n-1}, t]
    # Objective: [0, ..., 0, 1]

    c_obj = np.zeros(n_vars)
    c_obj[n] = 1.0  # minimize t

    # i.e., A @ d + [-1]*t <= -c_active
    A_ub = np.hstack([A, -np.ones((A.shape[0], 1))])
    b_ub = -dx * corr[top_idx]  # -c_active (in normalized units)

        improved = False
        for i in perm:
            # Try moving h[i] up or down, compensating with a random other coordinate
            j = rng.randint(n)
            if j == i:
                continue
            # Find best delta
            h_i = h[i]
            h_j = h[j]
            # delta can range from -h_i to (1-h_i) for i, and from -(1-h_j) to h_j for j
            # We need h[i] + delta in [0,1] and h[j] - delta in [0,1]
            delta_max = min(h[i], 1.0 - h[j])  # positive delta: increase i, decrease j
            delta_min = max(-h[i], h[j] - 1.0)  # negative delta: decrease i, increase j
            
            # Try a few delta values
            best_delta = 0.0
            best_score = score_h(h)
            for delta in [delta_max * 0.5, delta_max * 0.25, delta_max * 0.75,
                         delta_min * 0.5, delta_min * 0.25, delta_min * 0.75]:
                if abs(delta) < 1e-10:
                    continue
                h_test = h.copy()
                h_test[i] += delta
                h_test[j] -= delta
                if 0 <= h_test[i] <= 1 and 0 <= h_test[j] <= 1:
                    c_test = score_h(h_test)
                    if c_test < best_score:
                        best_score = c_test
                        best_delta = delta
            if best_delta != 0.0:
                h[i] += best_delta
                h[j] -= best_delta
                improved = True
                if best_score < best_c5:
                    best_c5 = best_score
                    best_h = h.copy()
        if not improved:
            # Try random perturbations
            if sweep % 5 == 0:
                h_pert = project_feasible(h + rng.randn(n) * 0.01, n)
                c_pert = score_h(h_pert)
                if c_pert < best_c5:
                    h = h_pert
                    best_h = h_pert.copy()
                    best_c5 = c_pert
    
    return best_h, best_c5

def make_start(n, seed_val):
    rng = np.random.RandomState(seed_val)
    x = np.linspace(0.0, 2.0, n, endpoint=False)
    h = np.full(n, 0.5)
    h += 0.22 * np.sin(2.0 * np.pi * x)
    h += 0.10 * np.sin(4.0 * np.pi * x + 0.5)
    h += 0.05 * np.sin(6.0 * np.pi * x + 1.2)
    h += 0.02 * np.sin(10.0 * np.pi * x + 2.1)
    h += rng.randn(n) * 0.01
    return project_feasible(h, n)
