def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    res = h.sum() - target_sum
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(h)[::-1]
            rem = res
            for idx in order:
                if rem <= 1e-14:
                    break
                dec = min(rem, h[idx])
                h[idx] -= dec
                rem -= dec
        else:

            rem = -res
            for idx in order:
                if rem <= 1e-14:
                    break
                inc = min(rem, 1.0 - h[idx])
                h[idx] += inc
                rem -= inc

    s = np.sum(exp_vals)
    F = M + tau * np.log(s)
    weights = exp_vals / s
    return F, weights

        target_best = n_best / 2.0
        for _ in range(10):
            if time_left() < 8:
                break
            sigma = 0.03 + 0.05 * rng.random()
            h_try = project_feasible(best_h + sigma * rng.standard_normal(n_best), target_best)
            if verify_ok:
                h_try = solve_smooth_stage(h_try, 0.01, target_best, n_best, 80, time_left)
                h_try = project_feasible(h_try, target_best)
            v = evaluate_exact(h_try)
            if v < best_val:
                best_val = v
                best_h = h_try.copy()

    # Final projection and evaluation
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    # Ensure exact bounds
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)

    return best_h, float(best_val), len(best_h)
