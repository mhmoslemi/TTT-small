    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    val = m + math.log(np.exp(sigma * (c - m)).sum()) / sigma
    return dx * val

def lse_gradient(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    center = n - 1
    grad += w[center] * (1.0 - 2.0 * h)
    for p in range(1, n):
        wp = w[center + p]
        wm = w[center - p]
        if wp <= 1e-18 and wm <= 1e-18:
            continue
        if wp > 1e-18:
            grad[p:] += wp * (1.0 - h[:-p])
            grad[:-p] -= wp * (h[p:])
        if wm > 1e-18:
            grad[:-p] += wm * (1.0 - h[p:])
            grad[p:] -= wm * (h[:-p])
    grad *= dx
    return grad
