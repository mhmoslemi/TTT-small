    """Project x to [0,1]^n with sum = target."""
    n = len(x)
    x = np.clip(x.astype(float), 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-13:
        return x
    if diff > 0:
        # Need to add diff, spread to elements below 1
        room = 1.0 - x
        mask = room > 1e-12
        if mask.sum() == 0:
            return x
        add = diff / mask.sum()
        x[mask] = np.minimum(x[mask] + add, 1.0)

import numpy as np

def _fix_sum(x, target):
    # Quick sum fix
    x = np.clip(x, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) < 1e-12:
        return x
    n = len(x)
    if diff > 0:
        room = (1.0 - x)
        total_room = room.sum()
        if total_room < 1e-15:
            return x
        x = x + diff * room / total_room
        x = np.clip(x, 0.0, 1.0)
    else:
        total = x.sum()
        if total < 1e-15:
            return x
        x = x * (1.0 + diff/total)
        x = np.clip(x, 0.0, 1.0)
    # Final correction
    diff = target - x.sum()
    if abs(diff) > 1e-14:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-12)[0]
            if len(idx) > 0:
                x[idx[0]] = min(1.0, x[idx[0]] + diff)
        else:
            idx = np.where(x > 1e-12)[0]
            if len(idx) > 0:
                x[idx[0]] = max(0.0, x[idx[0]] + diff)
    return x

def _max_step(h, s):
    t_max = 1e10
    pos = s > 1e-15
    if pos.any():
        t_max = min(t_max, np.min((1.0 - h[pos]) / s[pos]))
    neg = s < -1e-15
    if neg.any():
        t_max = min(t_max, np.min(h[neg] / (-s[neg])))
    return max(t_max, 0.0)

    g = np.zeros(n)
    # alpha has length 2n-1

                g[:n-l] -= a * h[l:]
        else:
            d = -l
            g[:n-d] += a * (1.0 - h[d:])
            g[d:] -= a * h[:n-d]

            t_best, c_best = t_mid, c_mid
    
    # Also check t=0
    c0 = f(0.0)
    if c0 < c_best:
        t_best, c_best = 0.0, c0
    
    return t_best, c_best

def _binarize(h, n):
    if n % 2 != 0:
        return None
    target = n // 2
    order = np.argsort(h, kind='stable')[::-1][:target]
    b = np.zeros(n)
    b[order] = 1.0
    return b

def _resample(h, N):
    n0 = len(h)
    target = N / 2.0
    if n0 == N:
        return _fix_sum(h, target)
    x0 = (np.arange(n0) + 0.5) * (2.0 / n0)
    x1 = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(x1, x0, h)
    return _fix_sum(y, target)

def _pair_refine(h, dx, deadline, rng, max_rounds=50):
    n = len(h)
    target = n / 2.0
    c = float(np.max(np.correlate(h, 1-h, "full")) * dx)
    
    for rnd in range(max_rounds):
        if time.time() > deadline:
            break
        improved = False
        for _ in range(100):
            if time.time() > deadline:
                break
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            t_lo = max(0.0, -h[i])  # h[i]+t >= 0
            t_hi = min(1.0 - h[i], h[j])  # h[i]+t <= 1, h[j]-t >= 0
            if t_hi - t_lo < 1e-10:
                continue
            # Try a few t values
            t_mids = [t_lo + t_hi*0.5, t_lo + t_hi*0.25, t_lo + t_hi*0.75]
            for t in t_mids:
                if t_lo <= t <= t_hi:
                    h2 = h.copy()
                    h2[i] += t
                    h2[j] -= t
                    h2 = np.clip(h2, 0, 1)
                    c2 = float(np.max(np.correlate(h2, 1-h2, "full")) * dx)
                    if c2 < c - 1e-13:
                        h = h2
                        c = c2
                        improved = True
                        break
        if not improved:
            break
    return h, c

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except:
        total = 1100.0
    if not np.isfinite(total) or total <= 0:
        total = 1100.0
    hard_deadline = t0 + total - 5.0
    
    rng = np.random.default_rng(int(seed) % (1 << 32))
    
    # Get initial
    h_init_val = globals().get("initial_h_values", None)
    if h_init_val is not None:
        h0 = np.asarray(h_init_val, dtype=float).ravel()
        if h0.size > 0 and np.all(np.isfinite(h0)):
            h0 = _fix_sum(h0, h0.size / 2.0)
        else:
            h0 = None
    if h0 is None:
        # Fallback: simple cosine construction
        n = 480
        x = (np.arange(n) + 0.5) * (2.0 / n)
        h0 = 0.5 + 0.5*np.cos(np.pi * x)
        h0 = _fix_sum(h0, n/2.0)
    
    n = len(h0)
    dx = 2.0 / n
    target = n / 2.0
    
    best_h = h0.copy()
    best_c = float(np.max(np.correlate(h0, 1-h0, "full")) * dx)
    best_n = n
    
    # Phase 1: Subgradient at native resolution
    tau_schedule = [0.05, 0.03, 0.015, 0.008, 0.004, 0.002, 0.001, 0.0005, 0.0002]
    
    h = h0.copy()
    for tau in tau_schedule:
        if time.time() > hard_deadline - 30:
            break
        h, c = _subgradient_loop(h, tau, dx, hard_deadline, target)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    
    # Try binarization
    if n % 2 == 0:
        hb = _binarize(best_h, n)
        if hb is not None:
            cb = float(np.max(np.correlate(hb, 1-hb, "full")) * dx)
            if cb < best_c:
                best_c = cb
                best_h = hb
    
    # Phase 2: Pair refinement
    if time.time() < hard_deadline - 20:
        best_h, best_c = _pair_refine(best_h, dx, min(time.time() + 100, hard_deadline - 10), rng)
    
    # Phase 3: Try different resolutions
    if time.time() < hard_deadline - 30:
        for N_try in [n//2, n*3//2, 2*n]:
            if N_try < 100 or N_try > 2400:
                continue
            if time.time() > hard_deadline - 30:
                break
            h_try = _resample(best_h, N_try)
            dx_try = 2.0 / N_try
            # Quick subgradient
            for tau in [0.03, 0.01, 0.003, 0.001]:
                if time.time() > hard_deadline - 20:
                    break
                h_try, c_try = _subgradient_loop(h_try, tau, dx_try, hard_deadline - 15, N_try/2.0)
                if c_try < best_c:
                    best_c = c_try
                    best_h = h_try.copy()
                    best_n = N_try
