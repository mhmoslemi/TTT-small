  Wait: h[i] - t >= 0 means t <= h[i]
  h[i] - t <= 1 means t >= h[i] - 1
  h[j] + t >= 0 means t >= -h[j]
  h[j] + t <= 1 means t <= 1 - h[j]
  So: t in [max(h[i]-1, -h[j]), min(h[i], 1-h[j])]

For the golden section search, I'll evaluate the objective at a few points and find the best.

Actually, since the objective along the pair line is the max of quadratics, it's a unimodal-ish function (not strictly, but the max of quadratics is convex... wait, max of convex functions is convex, but these quadratics can be concave).

Actually, c[k] as a function of t for a fixed pair:
c[k](t) = c[k](0) + t * A_k + t² * B_k

where A_k and B_k depend on the pair and lag. The coefficient B_k is the "curvature" and can be positive, negative, or zero.

The max of (possibly concave) quadratics is not necessarily unimodal. However, in practice, for small t ranges, it should be close to unimodal.

Let me just use a simple approach: evaluate at ~20 points on the interval, pick the best, then do a few golden section iterations around it.

Actually, for robustness, let me just do a grid search with ~30 points, then refine the best region with a few more evaluations. This is simple and robust.

```python
import time
import math
import numpy as np


def get_parent():
    g = globals()
    if 'initial_h_values' in g:
        val = g['initial_h_values']
        try:
            a = np.asarray(val, dtype=float).ravel()
            if a.size and np.all(np.isfinite(a)):
                return a
        except Exception:
            pass
    return None


def repair(h, target):
    h = np.asarray(h, dtype=float).copy()
    h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    t = 0.5 * (lo + hi)
    out = np.clip(h - t, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-12:
        if d < 0:
            room = 1.0 - out
            tr = room.sum()
            if tr > 1e-15:
                out = out + (-d / tr) * room
        else:
            room = out
            tr = room.sum()
            if tr > 1e-15:
                out = out - (d / tr) * room
    out = np.clip(out, 0.0, 1.0)
    d = out.sum() - target
    if abs(d) > 1e-14:
        if d > 0:
            idx = int(np.argmax(out))
            out[idx] = max(0.0, out[idx] - d)
        else:
            idx = int(np.argmax(1.0 - out))
            out[idx] = min(1.0, out[idx] - d)
    return out


def obj_val(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(c.max() * dx)


def lse_grad(h, sigma):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    m = c.max()
    e = np.exp(sigma * (c - m))
    w = e / e.sum()
    grad = np.zeros(n)
    for k in range(2 * n - 1):
        wk = w[k]
        if wk < 1e-18:
            continue
        lag = k - (n - 1)
        i_lo = max(0, lag)
        i_hi = min(n - 1, lag + n - 1)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] += wk * (1.0 - h[idx - lag])
        i_lo = max(0, -lag)
        i_hi = min(n - 1, n - 1 - lag)
        if i_lo <= i_hi:
            idx = np.arange(i_lo, i_hi + 1)
            grad[idx] -= wk * h[idx + lag]
    grad *= dx
    return grad


def pair_minimize(h, i, j, max_evals=25):
    """Minimize objective along the pair transfer line h[i]-=t, h[j]+=t."""
    n = len(h)
    lo = max(h[i] - 1.0, -h[j])
    hi = min(h[i], 1.0 - h[j])
    if hi - lo < 1e-12:
        return h.copy(), obj_val(h)
    # Grid search
    ns = min(max_evals, 40)
    ts = np.linspace(lo, hi, ns)
    h_copy = h.copy()
    best_t = 0.0
    best_val = obj_val(h)
    for t in ts:
        if abs(t) < 1e-14:
            continue
        hc = h.copy()
        hc[i] -= t
        hc[j] += t
        v = obj_val(hc)
        if v < best_val:
            best_val = v
            best_t = t
    # Refine around best_t with golden section
    if abs(best_t) > 1e-12:
        # Narrow the interval
        step = (hi - lo) / ns
        a = max(lo, best_t - 3 * step)
        b = min(hi, best_t + 3 * step)
        if b - a < 1e-12:
            pass
        else:
            gr = (np.sqrt(5) - 1) / 2
            c1 = b - gr * (b - a)
            c2 = a + gr * (b - a)
            f1 = _pair_obj(h, i, j, c1)
            f2 = _pair_obj(h, i, j, c2)
            for _ in range(15):
                if f1 < f2:
                    b = c2; c2 = c1; f2 = f1
                    c1 = b - gr * (b - a)
                    f1 = _pair_obj(h, i, j, c1)
                else:
                    a = c1; c1 = c2; f1 = f2
                    c2 = a + gr * (b - a)
                    f2 = _pair_obj(h, i, j, c2)
            t_opt = 0.5 * (a + b)
            v_opt = _pair_obj(h, i, j, t_opt)
            if v_opt < best_val:
                best_val = v_opt
                best_t = t_opt
    h_new = h.copy()
    h_new[i] -= best_t
    h_new[j] += best_t
    h_new[i] = np.clip(h_new[i], 0.0, 1.0)
    h_new[j] = np.clip(h_new[j], 0.0, 1.0)
    return h_new, best_val


def _pair_obj(h, i, j, t):
    hc = h.copy()
    hc[i] = h[i] - t
    hc[j] = h[j] + t
    return obj_val(hc)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
```
