def project(h, target):

        # Backtracking line search
        improved = False
        s = step
        for _ in range(40):
            cand = project(h - s * g, target)
            cv = lse_val(cand, sigma)
            if cv < cur_val - 1e-14:
                h = cand
                cur_val = cv
                c5 = objective(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                improved = True
                break
            s *= 0.5
        if not improved:
            # Random perturbation
            h = project(h + rng.normal(0, 0.1, n), target)
            cur_val = lse_val(h, sigma)
            c5 = objective(h)
            if c5 < best_c5 - 1e-13:
                best_c5 = c5
                best_h = h.copy()
            step *= 0.5
            stall += 1
            if stall >= 10:
                step = max(step, 0.01)
                # Bigger perturbation
                h = project(h + rng.normal(0, 0.3, n), target)
                cur_val = lse_val(h, sigma)
                c5 = objective(h)
                if c5 < best_c5 - 1e-13:
                    best_c5 = c5
                    best_h = h.copy()
                stall = 0
                if step < 0.001:
                    break
    
    # Final check
    c5 = objective(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def subgrad_polish(h0, rng, time_limit, target, deadline=None):
    n = len(h0)
    h = project(h0, target).copy()
    cur = objective(h)
    best_h = h.copy()
    best_c5 = cur
    t0 = time.monotonic()
    alpha = 0.02
    momentum = np.zeros(n)
    beta = 0.9
    
    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        g = max_subgrad(h, top_k=5)
        momentum = beta * momentum + (1 - beta) * g
        cand = project(h - alpha * momentum, target)
        cv = objective(cand)
        if cv < cur - 1e-14:
            h = cand
            cur = cv
            if cur < best_c5 - 1e-13:
                best_c5 = cur
                best_h = h.copy()
            alpha = min(alpha * 1.01, 0.1)
        else:
            # Try smaller step
            accepted = False
            for _ in range(5):
                alpha *= 0.5
                cand = project(h - alpha * momentum, target)
                cv = objective(cand)
                if cv < cur - 1e-14:
                    h = cand
                    cur = cv
                    if cur < best_c5 - 1e-13:
                        best_c5 = cur
                        best_h = h.copy()
                    accepted = True
                    break
            if not accepted:
                # Random perturbation
                h = project(h + rng.normal(0, 0.05, n), target)
                cur = objective(h)
                if cur < best_c5 - 1e-13:
                    best_c5 = cur
                    best_h = h.copy()
    
    c5 = objective(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def local_search(h0, rng, time_limit, target, deadline=None):
    n = len(h0)
    h = project(h0, target).copy()
    cur = objective(h)
    best_h = h.copy()
    best_c5 = cur
    t0 = time.monotonic()
    
    while time.monotonic() - t0 < time_limit:
        if deadline is not None and time.monotonic() > deadline:
            break
        improved = False
        # Try random coordinate moves
        for _ in range(200):
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i == j:
                continue
            # Transfer move
            if rng.random() < 0.5:
                d = rng.random() * 0.3
                a = h.copy()
                if d <= h[i] and d <= 1 - h[j]:
                    a[i] -= d
                    a[j] += d
                elif d <= h[j] and d <= 1 - h[i]:
                    a[j] -= d
                    a[i] += d
                else:
                    continue
            else:
                a = h.copy()
                a[i], a[j] = a[j], a[i]
            na = objective(a)
            if na < cur - 1e-14:
                h = a
                cur = na
                improved = True
                if cur < best_c5 - 1e-13:
                    best_c5 = cur
                    best_h = h.copy()
                break  # restart the loop with new h
        if not improved:
            # Try coordinate-wise perturbation
            i = int(rng.integers(n))
            deltas = [0.01, -0.01, 0.05, -0.05, 0.1, -0.1]
            best_d = None
            best_a = None
            best_na = cur
            for d in deltas:
                a = h.copy()
                a[i] = float(np.clip(a[i] + d, 0, 1))
                a = project(a, target)
                na = objective(a)
                if na < best_na - 1e-14:
                    best_na = na
                    best_a = a
                    best_d = d
            if best_a is not None:
                h = best_a
                cur = best_na
                if cur < best_c5 - 1e-13:
                    best_c5 = cur
                    best_h = h.copy()
    
    c5 = objective(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    return best_h, best_c5


def make_thue_morse(n):
    """Generate Thue-Morse based binary pattern of length n."""
    # Thue-Morse: t(i) = popcount(i) mod 2
    tm = np.array([bin(i).count('1') % 2 for i in range(n)], dtype=float)
    # Map to [0,1] with 1 - 2*tm giving {-1, 1}, then shift to [0,1]
    # Actually let's use tm directly: 0s and 1s
    # We need sum = n/2
    # tm has roughly n/2 ones (slightly off)
    return project(tm, n / 2.0)


def make_random_binary(n, rng):
    """Random binary pattern."""
    h = (rng.random(n) < 0.5).astype(float)
    return project(h, n / 2.0)


def make_sinusoidal_binary(n, freq, phase, rng):
    """Sinusoidal pattern thresholded to binary."""
    x = np.arange(n) / n
    s = 0.5 + 0.5 * np.sin(2 * np.pi * freq * x + phase)
    h = (s > 0.5).astype(float)
    return project(h, n / 2.0)


def resample(h, n_new):
    n0 = len(h)
    if n0 == n_new:
        return h.copy()
    x0 = (np.arange(n0) + 0.5) / n0
    x1 = (np.arange(n_new) + 0.5) / n_new
    return np.interp(x1, x0, h)
