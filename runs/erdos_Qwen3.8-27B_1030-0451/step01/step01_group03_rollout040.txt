def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, 'full') * dx
    return float(np.max(c))


def get_corr(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, 'full') * dx


def project_feasible(y, target):
    lo, hi = -3.0, 4.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - 0.5 * (lo + hi), 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-10) & (h < 1.0 - 1e-10)
        c = int(mask.sum())
        if c > 0:
            h[mask] += diff / c
            np.clip(h, 0.0, 1.0, out=h)
    return h


def interp_to_n(hp, nn):
    np_par = len(hp)
    if nn == np_par:
        return hp.copy()
    xo = np.linspace(0.0, 2.0, np_par, endpoint=False)
    xn = np.linspace(0.0, 2.0, nn, endpoint=False)
    h = np.interp(xn, xo, hp)
    return project_feasible(h, nn / 2.0)


def smooth_obj(h, tau):
    c = get_corr(h)
    cm = c.max()
    s = (c - cm) / tau
    s = np.clip(s, -500.0, 500.0)
    e = np.exp(s)
    return cm + tau * float(np.log(e.sum()))


def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    cm = c.max()
    s = np.clip((c - cm) / tau, -500.0, 500.0)
    w = np.exp(s)
    w /= w.sum()
    grad = np.zeros(n)
    for m in range(len(c)):
        wm = w[m]
        if wm < 1e-14:
            continue
        d = m - (n - 1)
        jlo = max(0, d)
        jhi = min(n, n + d)
        if jlo < jhi:
            idx = np.arange(jlo, jhi) - d
            grad[jlo:jhi] += wm * (1.0 - h[idx]) * dx
        jlo2 = max(0, -d)
        jhi2 = min(n, n - d)
        if jlo2 < jhi2:
            idx = np.arange(jlo2, jhi2) + d
            grad[jlo2:jhi2] -= wm * h[idx] * dx
    return grad


def subgradient_true_max(h):
    n = len(h)
    dx = 2.0 / n
    c = get_corr(h)
    cm = c.max()
    tol = max(1e-8, 0.02 * (c.max() - c.min()))
    active = np.where(c >= cm - tol)[0]
    grad = np.zeros(n)
    for m in active:
        d = m - (n - 1)
        jlo = max(0, d)
        jhi = min(n, n + d)
        if jlo < jhi:
            idx = np.arange(jlo, jhi) - d
            grad[jlo:jhi] += (1.0 - h[idx]) * dx
        jlo2 = max(0, -d)
        jhi2 = min(n, n - d)
        if jlo2 < jhi2:
            idx = np.arange(jlo2, jhi2) + d
            grad[jlo2:jhi2] -= h[idx] * dx
    grad /= len(active)
    return grad


def smooth_optimize_lbfgs(h0, n, deadline, max_time_per_tau=15.0):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    c_init = get_corr(h0)
    spread = max(1e-4, c_init.max() - c_init.mean())
    
    tau_list = []
    tau = 0.5 * spread
    for _ in range(18):
        tau_list.append(tau)
        tau *= 0.62
    tau_list = [max(t, 1e-4) for t in tau_list]
    
    h_curr = h0.copy()
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: float(np.sum(x) - target),
                    'jac': lambda x: np.ones(n)}]
    
    for tau in tau_list:
        if time.time() > deadline - 5:
            break
        t_start = time.time()
        maxiter = 120 if tau > 0.01 else 200
        
        def obj_fn(x):
            return smooth_obj(x, tau)
        
        def grad_fn(x):
            return smooth_grad(x, tau)
        
        try:
            res = minimize(obj_fn, h_curr, method='L-BFGS-B', jac=grad_fn,
                           bounds=bounds, constraints=constraints,
                           options={'maxiter': maxiter, 'ftol': 1e-14, 'maxls': 40})
            h_new = project_feasible(res.x, target)
        except Exception:
            h_new = h_curr
        
        c5_new = evaluate_c5(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_curr = h_new
        
        if time.time() - t_start > max_time_per_tau:
            break
    
    return best_h, best_c5


def subgradient_polish(h0, n, n_steps=200, deadline=None):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    h_curr = h0.copy()
    lr0 = 0.05
    n_corr = 2 * n - 1
    
    for step in range(n_steps):
        if deadline is not None and time.time() > deadline - 3:
            break
        lr = lr0 / (1.0 + 0.02 * step)
        g = subgradient_true_max(h_curr)
        h_trial = project_feasible(h_curr - lr * g, target)
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < best_c5 - 1e-12:
            best_c5 = c5_trial
            best_h = h_trial.copy()
        h_curr = h_trial
    
    return best_h, best_c5


def coordinate_descent(h0, n, n_iter=300, deadline=None):
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    c_curr = get_corr(h0)
    rng = np.random.default_rng(12345)
    eps_vals = [0.01, 0.03, 0.05, 0.08, 0.12]
    
    for it in range(n_iter):
        if deadline is not None and time.time() > deadline - 3:
            break
        i = rng.integers(0, n)
        j = rng.integers(0, n)
        if i == j:
            j = (j + 1) % n
        eps = eps_vals[rng.integers(0, len(eps_vals))]
        
        for sign in [1, -1]:
            h_trial = h0.copy()
            hi_new = np.clip(h0[i] + sign * eps, 0.0, 1.0)
            hj_new = np.clip(h0[j] - sign * eps, 0.0, 1.0)
            actual_di = hi_new - h0[i]
            actual_dj = hj_new - h0[j]
            if abs(actual_di) < 1e-14 and abs(actual_dj) < 1e-14:
                continue
            h_trial[i] = hi_new
            h_trial[j] = hj_new
            h_trial = project_feasible(h_trial, target)
            c5_t = evaluate_c5(h_trial)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_trial.copy()
                h0 = h_trial.copy()
                break
    
    return best_h, best_c5


def water_fill(h0, n, deadline):
    """Try to equalize the top few correlation values."""
    target = n / 2.0
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    for topk in [3, 5, 8, 12]:
        if time.time() > deadline - 5:
            break
        c = get_corr(h0)
        order = np.argsort(-c)
        top_indices = order[:topk]
        top_mean = c[top_indices].mean()
        
        # Compute gradient that reduces top-k correlations equally
        dx = 2.0 / n
        grad = np.zeros(n)
        for m in top_indices:
            d = m - (n - 1)
            jlo = max(0, d)
            jhi = min(n, n + d)
            if jlo < jhi:
                idx = np.arange(jlo, jhi) - d
                grad[jlo:jhi] += (1.0 - h0[idx]) * dx
            jlo2 = max(0, -d)
            jhi2 = min(n, n - d)
            if jlo2 < jhi2:
                idx = np.arange(jlo2, jhi2) + d
                grad[jlo2:jhi2] -= h0[idx] * dx
        grad /= topk
        
        # Try small steps
        for lr in [0.005, 0.01, 0.02, 0.03]:
            h_trial = project_feasible(h0 - lr * grad, target)
            c5_t = evaluate_c5(h_trial)
            if c5_t < best_c5 - 1e-12:
                best_c5 = c5_t
                best_h = h_trial.copy()
                h0 = h_trial.copy()
                break
            elif c5_t < best_c5 + 1e-10:
                # Try opposite direction (increasing)
                h_trial = project_feasible(h0 + lr * grad, target)
                c5_t2 = evaluate_c5(h_trial)
                if c5_t2 < best_c5 - 1e-12:
                    best_c5 = c5_t2
                    best_h = h_trial.copy()
                    h0 = h_trial.copy()
                    break
    
    return best_h, best_c5


def diverse_starts(h_parent, n, rng, n_starts=6):
    """Generate diverse starting points at resolution n."""
    starts = [interp_to_n(h_parent, n)]
    
    # Shift in frequency domain
    h_shift = starts[0].copy()
    freq = np.fft.rfft(h_shift)
    phase = rng.uniform(-0.3, 0.3, size=len(freq))
    freq_mod = freq * np.exp(1j * phase)
    h_shift = np.real(np.fft.irfft(freq_mod, n=n))
    starts.append(project_feasible(h_shift, n / 2.0))
    
    # Smoothed perturbation
    noise = rng.standard_normal(n)
    noise = np.convolve(noise, np.ones(7) / 7.0, mode='same')
    noise /= (np.std(noise) + 1e-12)
    for amp in [0.05, 0.1]:
        starts.append(project_feasible(starts[0] + amp * noise, n / 2.0))
    
    # Binarize-like: push towards {0, 0.5, 1}
    h_bin = starts[0].copy()
    h_bin = np.where(h_bin > 0.75, 1.0, np.where(h_bin < 0.25, 0.0, 0.5))
    h_bin = project_feasible(h_bin, n / 2.0)
    starts.append(h_bin)
    
    # Another random smooth perturbation
    noise2 = rng.standard_normal(n)
    noise2 = np.convolve(noise2, np.ones(11) / 11.0, mode='same')
    noise2 /= (np.std(noise2) + 1e-12)
    starts.append(project_feasible(starts[0] + 0.15 * noise2, n / 2.0))
    
    return starts


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    deadline = t0 + budget_s - 8
    
    # Get parent construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        parent = np.array(initial_h_values, dtype=np.float64)
    else:
        parent = np.full(240, 0.5)
    
    n_par = len(parent)
    parent = project_feasible(parent, n_par / 2.0)
    par_c5 = evaluate_c5(parent)
    
    best_h = parent.copy()
    best_c5 = par_c5
    best_n = n_par
    
    rng = np.random.default_rng(seed)
    
    # Determine n values to try, keeping total under 2400
    n_list = []
    n_par_eff = n_par
    for mult in [1, 2, 3, 4, 5]:
        n = n_par_eff * mult
        if n <= 2400 and n not in n_list:
            n_list.append(n)
    n_list.sort()
    
    # Estimate time per n and allocate budget
    time_per_n_est = 80  # rough estimate
    max_n_targets = min(len(n_list), int((budget_s - 60) / time_per_n_est))
    
    for n in n_list[:max(2, max_n_targets)]:
        if time.time() > deadline - 30:
            break
        
        remaining = deadline - time.time()
        if remaining < 20:
            break
        
        # Generate diverse starts
        starts = diverse_starts(parent, n, rng, n_starts=5)
        
        # Optimize each start with time allocation
        time_per_start = min(remaining * 0.15, 60.0)
        
        for si, h_start in enumerate(starts):
            if time.time() > deadline - 10:
                break
            start_deadline = min(deadline - 5, time.time() + time_per_start)
            
            # Phase 1: Smooth optimization
            h_opt, c5_opt = smooth_optimize_lbfgs(h_start, n, start_deadline)
            
            # Phase 2: Subgradient polish
            h_opt, c5_opt = subgradient_polish(h_opt, n, n_steps=100,
                                               deadline=min(deadline - 5, time.time() + 20))
            
            # Phase 3: Coordinate descent
            h_opt, c5_opt = coordinate_descent(h_opt, n, n_iter=200,
                                               deadline=min(deadline - 5, time.time() + 20))
            
            # Phase 4: Water fill
            h_opt, c5_opt = water_fill(h_opt, n, deadline=min(deadline - 3, time.time() + 10))
            
            if c5_opt < best_c5 - 1e-12:
                best_c5 = c5_opt
                best_h = h_opt.copy()
                best_n = n
    
    # Final refinement on best solution
    if time.time() < deadline - 20:
        n = best_n
        # Another round of subgradient + coordinate descent
        h_ref, c5_ref = subgradient_polish(best_h, n, n_steps=150, deadline=deadline - 10)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
        
        h_ref, c5_ref = coordinate_descent(best_h, n, n_iter=300, deadline=deadline - 10)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
        
        h_ref, c5_ref = water_fill(best_h, n, deadline=deadline - 3)
        if c5_ref < best_c5:
            best_h, best_c5 = h_ref, c5_ref
    
    # Try one more resolution up if time allows
    if time.time() < deadline - 40:
        n_next = best_n + n_par
        if n_next <= 2400:
            h0 = interp_to_n(best_h, n_next)
            h_opt, c5_opt = smooth_optimize_lbfgs(h0, n_next, deadline - 10)
            h_opt, c5_opt = subgradient_polish(h_opt, n_next, n_steps=100, deadline=deadline - 5)
            if c5_opt < best_c5:
                best_h, best_c5, best_n = h_opt, c5_opt, n_next
    
    # Final feasibility cleanup
    best_n = len(best_h)
    target = best_n / 2.0
    diff = target - best_h.sum()
    if abs(diff) > 1e-10:
        mask = (best_h > 1e-8) & (best_h < 1.0 - 1e-8)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] -= diff / c
    np.clip(best_h, 0.0, 1.0, out=best_h)
    diff = target - best_h.sum()
    if abs(diff) > 1e-12:
        mask = (best_h > 0) & (best_h < 1.0)
        c = int(mask.sum())
        if c > 0:
            best_h[mask] += diff / c
    
    best_c5 = evaluate_c5(best_h)
    return (best_h, best_c5, best_n)
