def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')

def corr_values(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max_obj(h, tau):
    c = corr_values(h)
    m = np.max(c)
    shifted = (c - m) / tau
    shifted = np.clip(shifted, -500, 500)
    logsum = m / tau + np.log(np.sum(np.exp(shifted)))
    return logsum * tau

def smooth_max_grad(h, tau):

    dx = 2.0 / N
    c = corr_values(h)
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500, 500)
    w = np.exp(exponents)
    w = w / w.sum()
    # Gradient of sum w[r] * c[r] with respect to h

def project_h(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    # Binary search on the threshold
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) / 2
    h_proj = np.clip(h + t, 0, 1)
    # Fix residual
    res = target_sum - np.sum(h_proj)
    if abs(res) > 1e-12:
        if res > 0:
            idx = np.where(h_proj < 1.0)[0]
            if len(idx) > 0:
                h_proj[idx[0]] += min(res, 1.0 - h_proj[idx[0]])
        else:
            idx = np.where(h_proj > 0.0)[0]
            if len(idx) > 0:
                h_proj[idx[0]] += max(res, -h_proj[idx[0]])
    return h_proj

def projected_gradient_step(h, tau, step_size, target_sum):
    """One step of projected gradient descent"""
    grad = smooth_max_grad(h, tau)
    h_new = h - step_size * grad
    h_new = project_h(h_new, target_sum)
    return h_new
