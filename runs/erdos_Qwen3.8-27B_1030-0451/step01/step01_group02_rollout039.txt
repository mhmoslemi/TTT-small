def corr_scaled(h):
    n = h.size
    if n == 0:
        return np.empty(0, dtype=float)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / float(n))

def corr_for_d(h, d):
    n = len(h)
    if abs(d) >= n:
        return 0.0
    if d == 0:
        return float(np.sum(h * (1.0 - h)))
    if d > 0:
        return float(np.sum(h[d:] * (1.0 - h[:-d])))
    dd = -d
    return float(np.sum(h[:-dd] * (1.0 - h[dd:])))

def alpha_from_C(C, tau, cap, slack):

        return np.empty(0, dtype=float), np.empty(0, dtype=int)

    if near.size > cap:
        order = np.argsort(C, kind="stable")
        idx = order[::-1][:cap].astype(int)

            return alpha, idx
        alpha = np.zeros(idx.size, dtype=float)

        return alpha, idx

    y = (C[idx] - maxc) / tau

        cnt = max(1, len(order) // 4)
        alpha = np.zeros(idx.size, dtype=float)
        alpha[order[::-1][:cnt]] = 1.0 / cnt
        return alpha, idx
    alpha = e / s
    return alpha, idx


def grad_from_alpha(alpha, h, dx, idx):
    n = h.size
    G = np.zeros(n, dtype=float)
    if idx.size == 0:
        return G
    dirf = _get_corr_dir()
    ctr = n - 1
    for w, k in zip(alpha, idx):
        if float(w) <= 1e-14:
            continue
        d = int(dirf * (int(k) - ctr))
        coeff = float(w) * dx
        if d == 0:
            G += coeff * (1.0 - 2.0 * h)
        elif d > 0:
            G[d:] += coeff * (1.0 - h[:-d])
            if n - d > 0:
                G[:-d] += -coeff * h[d:]
        else:
            dd = -d
            if n - dd > 0:
                G[:n - dd] += coeff * (1.0 - h[dd:])
            if dd > 0:
                G[dd:] += -coeff * h[:-dd]
    return G


def _lbfgs_fg(x, tau, mu, target, dx):
    x = np.asarray(x, dtype=float).ravel()
    n = x.size
    if n == 0:
        return 0.0, np.empty(0, dtype=float)
    if not np.all(np.isfinite(x)):
        return 1e100, np.zeros(n, dtype=float)
    C = corr_scaled(x)
    if not np.all(np.isfinite(C)):
        return 1e100, np.zeros(n, dtype=float)
    m = float(np.max(C))

    if tau <= 1e-13:
        act = np.where(C >= m - 1e-10)[0]
        if act.size == 0:
            act = np.array([int(np.argmax(C))], dtype=int)
        alpha = np.full(act.size, 1.0 / float(act.size), dtype=float)
        idx = act
        f = m
    else:
        y = (C - m) / tau
        y = np.clip(y, -60.0, 60.0)
        e = np.exp(y)
        s = float(np.sum(e))
        if s <= 1e-300:
            act = np.where(C >= m - 1e-10)[0]
            if act.size == 0:
                act = np.array([int(np.argmax(C))], dtype=int)
            alpha = np.full(act.size, 1.0 / float(act.size), dtype=float)
            idx = act
            f = m
        else:
            alpha_all = e / s
            idx = np.where(alpha_all > 1e-10)[0]
            if idx.size == 0:
                idx = np.array([int(np.argmax(C))], dtype=int)
                alpha = np.array([1.0], dtype=float)
            else:
                alpha = alpha_all[idx]
            f = m + tau * log(s)

    g = grad_from_alpha(alpha, x, dx, idx)
    r = float(x.sum() - target)
    f += 0.5 * mu * r * r
    g = g + mu * r
    return float(f), g

def _pair_score(h, fr, to, lam):
    if lam <= 0.0:
        return exact_C5(h)
    v1 = float(h[fr]) - lam
    v2 = float(h[to]) + lam
    if v1 < -1e-12 or v2 > 1.0 + 1e-12:
        return float("inf")
    h2 = h.copy()
    h2[fr] = max(0.0, min(1.0, v1))
    h2[to] = max(0.0, min(1.0, v2))
    return exact_C5(h2)


def pair_improve(h, fr, to, c, max_eval=10):

        val = _pair_score(h, fr, to, lam)
        if np.isfinite(val) and val < best_c - 1e-13:
            best_c = val
            best_lam = lam

    phi = 0.6180339887498949
    a = 0.0
    b = hi
    c1 = b - phi * (b - a)
    c2 = a + phi * (b - a)

    f2 = _pair_score(h, fr, to, c2)

    if np.isfinite(f2) and f2 < best_c - 1e-13:

        if best_c >= c - 1e-13 and _ >= 5:

            c2 = c1
            f2 = f1
            c1 = b - phi * (b - a)
            f1 = _pair_score(h, fr, to, c1)
            if np.isfinite(f1) and f1 < best_c - 1e-13:
                best_c, best_lam = f1, c1
        else:
            a = c1
            c1 = c2
            f1 = f2
            c2 = a + phi * (b - a)
            f2 = _pair_score(h, fr, to, c2)

    if best_c < c - 1e-12 and best_lam > 0.0:

def add_raw_pair(raw, seen, i, j):
    if i == j:
        return
    i = int(i)
    j = int(j)
    if i > j:
        i, j = j, i
    key = (i, j)
    if key not in seen:
        seen.add(key)
        raw.append(key)


def pair_descent(h, deadline, rng):

            break
        if time.time() - last_C_t > 0.7:
            Ccarr = corr_scaled(cur_h)
            cur_c = exact_C5(cur_h)

                best_h, best_c = cur_h.copy(), cur_c

def binarize_h(h, target):
    n = h.size
    if abs(target - round(target)) > 1e-8:
        return None
    cnt = int(round(target))
    if cnt <= 0 or cnt >= n:
        return None
    order = np.argsort(h, kind="stable")[::-1][:cnt]
    b = np.zeros(n, dtype=float)
    b[order] = 1.0
    return b


def resample_h(h, N):
    h = np.asarray(h, dtype=float).ravel()
    n0 = h.size
    N = int(N)
    target = 0.5 * float(N)
    if n0 == N:
        return make_feasible(h, target)

    cands = []
    xold = (np.arange(n0) + 0.5) * (2.0 / float(n0))
    xnew = (np.arange(N) + 0.5) * (2.0 / float(N))
    y = np.interp(xnew, xold, h)
    cands.append(make_feasible(y, target))

    if N % n0 == 0:
        cands.append(make_feasible(np.tile(h, N // n0), target))
    if n0 % N == 0:
        cands.append(make_feasible(h.reshape(N, n0 // N).mean(axis=1), target))
    if N > n0:
        idx = np.floor(np.arange(N) * (n0 / float(N))).astype(int)
        idx = np.clip(idx, 0, n0 - 1)
        cands.append(make_feasible(h[idx], target))

    best = None
    bestc = float("inf")
    for c in cands:
        cv = exact_C5(c)
        if np.isfinite(cv) and cv < bestc:
            bestc = cv
            best = c
    if best is None:
        best = make_feasible(np.full(N, 0.5, dtype=float), target)
    return best


def build_basis(N, kind):
    N = int(max(16, N))
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    cols = []
    if kind == 0:
        for f in (1, 2, 3):
            cols.append(np.cos(np.pi * f * x))
            cols.append(np.sin(np.pi * f * x))
    else:
        t = x - 1.0
        cols.append(t)
        cols.append(t * t)
        cols.append(t ** 3 - t)
        cols.append(t ** 4 - t ** 2)
    if not cols:
        return np.empty((N, 0), dtype=float)
    basis = np.array(cols, dtype=float).T
    basis -= basis.mean(axis=0)
    for j in range(basis.shape[1]):
        nm = float(np.linalg.norm(basis[:, j]))
        if nm > 1e-12:
            basis[:, j] /= nm
    return basis

    if not np.all(np.isfinite(coeff)):
        return None
    return np.clip(coeff, -2.0, 2.0)


def param_optimize(N, kind, deadline, rng, hbase=None):
    N = int(max(16, N))
    if time.time() > deadline - 5.0:
        return None
    basis = build_basis(N, kind)
    D = basis.shape[1]
    if D == 0:
        return None

    btime = max(0.0, deadline - time.time())
    max_starts = 3 if btime > 60.0 else 2
    maxiter = int(max(30, min(110, int(55000.0 / float(max(N, 1))))))

    starts = []
    if hbase is not None:
        cs = basis_coeffs_from_h(hbase, N, basis)
        if cs is not None:
            starts.append(cs)
    for s in range(3):
        cs = rng.normal(0.0, 0.8, size=D)
        if s == 0 and D > 0:
            cs[0] = 1.0
        if s == 1 and D > 1:
            cs[1] = -1.0
        starts.append(cs)
    starts = starts[:max_starts]

    best_h = None
    best_c = float("inf")

    for cs in starts:
        if time.time() > deadline - 2.0:
            break
        hh = param_h_from_coeffs(N, basis, cs)
        cc = exact_C5(hh)
        if np.isfinite(cc) and cc < best_c:
            best_h, best_c = hh, cc

        try:
            res = minimize(
                _param_objectf,
                np.asarray(cs, dtype=float).copy(),
                args=(N, basis, kind),
                method="Nelder-Mead",
                options={"maxiter": maxiter, "xatol": 1e-5, "fatol": 1e-9},
            )
        except Exception:
            res = None
        if res is not None and getattr(res, "x", None) is not None:
            hh = param_h_from_coeffs(N, basis, res.x)
            cc = exact_C5(hh)
            if np.isfinite(cc) and cc < best_c:
                best_h, best_c = hh, cc

    return best_h


def _get_initial():
    val = globals().get("initial_h_values", None)
    if val is None:
        return None
    arr = np.asarray(val, dtype=float).ravel()
    if arr.size == 0 or not np.all(np.isfinite(arr)):
        return None
    return make_feasible(arr, 0.5 * float(arr.size))


def fallback_initial(N=320):
    N = int(N)
    x = (np.arange(N) + 0.5) * (2.0 / float(N))
    best = None
    bestc = float("inf")
    for f in (1, 2, 3, 4):
        for ph in (0.0, 0.5 * np.pi, np.pi, 1.5 * np.pi):
            y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph)
            h = make_feasible(y, 0.5 * float(N))
            c = exact_C5(h)
            if np.isfinite(c) and c < bestc:
                bestc = c
                best = h
    rng = np.random.default_rng(1234)
    for _ in range(14):
        f = int(1 + rng.integers(0, 4))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        y = 0.5 + 0.5 * np.cos(np.pi * f * x + ph) + 0.07 * rng.normal(0.0, 1.0, size=N)
        h = make_feasible(y, 0.5 * float(N))
        c = exact_C5(h)
        if np.isfinite(c) and c < bestc:
            bestc = c
            best = h
    if best is None:
        best = np.full(N, 0.5, dtype=float)
    return best


def add_candidate(cands, h):
    if h is None:
        return cands
    h = np.asarray(h, dtype=float).ravel()
    if h.size == 0:
        return cands
    n = h.size
    target = 0.5 * float(n)
    h = make_feasible(h, target)
    c = exact_C5(h)
    if np.isfinite(c):
        cands.append((h, float(c), int(n)))
    return cands


def cand_better(a, b):
    if b is None:
        return True
    if a is None:
        return False
    if a[1] < b[1] - 1e-12:
        return True
    if abs(a[1] - b[1]) <= 1e-10 and a[2] < b[2]:
        return True
    return False


def best_candidate(cands):
    best = None
    for it in cands:
        if cand_better(it, best):
            best = it
    return best


def top_candidates(cands, k):
    pool = list(cands)
    out = []
    k = int(k)
    for _ in range(min(k, len(pool))):
        bi = 0
        best = pool[0]
        for i in range(1, len(pool)):
            if cand_better(pool[i], best):
                best = pool[i]
                bi = i
        out.append(best)
        del pool[bi]
    return out


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    try:
        total = float(budget_s)
    except Exception:
        total = 1100.0
    if not np.isfinite(total) or total <= 0.0:
        total = 1100.0
    total = max(30.0, total)
    hard = t0 + total - 14.0

    rng = np.random.default_rng(int(seed) % (1 << 32))
    cands = []

    hinit = _get_initial()
    if hinit is None:
        hinit = fallback_initial(360)
    cands = add_candidate(cands, hinit)
    best = best_candidate(cands)
    if best is None:
        h = np.full(64, 0.5, dtype=float)
        return h, float(exact_C5(h)), int(h.size)

    ninit = int(best[2])

    extra_Ns = []
    for N in (
        int(max(120, ninit // 2)),
        int(min(960, ninit * 2)),
        240,
        320,
        480,
        640,
    ):
        if 16 <= N <= 1000 and N not in extra_Ns:
            extra_Ns.append(N)
        if len(extra_Ns) >= 5:
            break

    for N in extra_Ns:
        if time.time() > hard - 180.0:
            break
        add_candidate(cands, resample_h(best[0], N))
        hb = binarize_h(best[0], 0.5 * float(ninit))
        if hb is not None:
            add_candidate(cands, resample_h(hb, N))

    if time.time() < hard - 180.0:
        bh = best_candidate(cands)[0]
        for N in (320, 480):
            if time.time() > hard - 120.0:
                break
            pd = min(45.0, hard - time.time() - 60.0)
            if pd > 18.0:
                add_candidate(cands, param_optimize(N, 0, time.time() + pd, rng, bh))
            if time.time() < hard - 70.0:
                pd2 = min(25.0, hard - time.time() - 45.0)
                if pd2 > 12.0:
                    add_candidate(cands, param_optimize(N, 1, time.time() + pd2, rng, bh))

    if time.time() < hard - 120.0:
        best = best_candidate(cands)
        if best[2] > 600:
            add_candidate(cands, resample_h(best[0], 480))

    top = top_candidates(cands, 3)
    for it in top:
        if time.time() > hard - 60.0:
            break
        rem = hard - time.time() - 30.0
        if rem <= 15.0:
            break
        b = max(12.0, min(55.0, rem / float(max(1, len(top)))))
        h2, c2 = pair_descent(it[0], time.time() + b, rng)
        add_candidate(cands, h2)

    top = top_candidates(cands, 2)
    lbfs_done = 0
    for it in top:
        if time.time() > hard - 80.0 or lbfs_done >= 2:
            break
        if it[2] > 700:
            continue
        rem = hard - time.time() - 50.0
        if rem <= 25.0:
            break
        b = max(25.0, min(170.0, rem / 2.0))
        h2, c2 = lbfgs_smooth(it[0], time.time() + b, rng)
        add_candidate(cands, h2)
        lbfs_done += 1

    best = best_candidate(cands)
    if time.time() < hard - 45.0:
        rem = hard - time.time() - 25.0
        if rem > 25.0:
            h2, c2 = pair_descent(best[0], time.time() + rem, rng)
            add_candidate(cands, h2)

    best = best_candidate(cands)
    if time.time() < hard - 70.0:
        best_h = best[0]
        target = 0.5 * float(best[2])
        for sig in (0.016, 0.009, 0.005, 0.003):
            if time.time() > hard - 30.0:
                break
            hp = make_feasible(best_h + rng.normal(0.0, sig, size=best_h.size), target)
            rem = hard - time.time() - 20.0
            if rem <= 15.0:
                break
            b = max(15.0, min(45.0, rem))
            h2, c2 = pair_descent(hp, time.time() + b, rng)
            add_candidate(cands, h2)
            best = best_candidate(cands)

    best = best_candidate(cands)
    if time.time() < hard - 8.0:
        rem = hard - time.time() - 4.0
        b = max(5.0, min(45.0, rem))
        h2, c2 = pair_descent(best[0], time.time() + b, rng)
        add_candidate(cands, h2)

    best = best_candidate(cands)
    if best is None:
        h = np.full(64, 0.5, dtype=float)
        return h, float(exact_C5(h)), int(h.size)

    h = make_feasible(best[0], 0.5 * float(best[2]))
    c = exact_C5(h)
    if not np.isfinite(c):
        h = np.full(best[2], 0.5, dtype=float)
        c = exact_C5(h)
    return h, float(c), int(h.size)
