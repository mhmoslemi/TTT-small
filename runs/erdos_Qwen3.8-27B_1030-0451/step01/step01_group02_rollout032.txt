def make_feasible(x, target):
    n = len(x)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    s = x.sum()
    diff = target - s
    if abs(diff) < 1e-12:
        return x
    # Distribute the correction
    if diff > 0:
        # Need to increase sum
        room = 1.0 - x
        if room.sum() < diff:
            return np.clip(x, 0.0, 1.0)
        frac = diff / room.sum()
        x = x + frac * room
    else:
        # Need to decrease sum
        if x.sum() < -diff:
            return np.clip(x, 0.0, 1.0)
        frac = (-diff) / x.sum()
        x = x - frac * x
    return np.clip(x, 0.0, 1.0)

    grad_x = grad_h[idx] - grad_h[jdx]
    return grad_x, maxC

def subgrad_optimize_antipodal(x0, max_iter, deadline, rng):
    """Subgradient descent with Nesterov momentum in antipodal space"""
    m = len(x0)
    x = x0.copy()
    v = np.zeros(m, dtype=float)
    best_x = x.copy()
    best_c = score_h(antipodal_from_half(x))
    c = best_c
    tau = 0.01
    beta = 0.9  # momentum
    
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it % 50 == 0:
            tau = max(0.001, 0.01 * (0.95 ** (it // 50)))
        
        g, _ = subgrad_antipodal(x, tau)
        # Nesterov: x_{t+1} = x_t - alpha * (g(x_t) + beta * momentum)
        step = 0.01 / (1.0 + 0.01 * it)
        x_new = x - step * (g + beta * v)
        v = beta * v + (1 - beta) * g
        x = np.clip(x_new, 0.0, 1.0)
        
        h = antipodal_from_half(x)
        c = score_h(h)
        if c < best_c - 1e-12:
            best_c = c
            best_x = x.copy()
    
    return best_x, best_c

def binary_sa(h0, n_steps, deadline, rng):
    """Binary simulated annealing"""
    n = len(h0)
    target = n // 2
    # Start from thresholded version
    order = np.argsort(h0)[::-1][:target]
    h = np.zeros(n, dtype=float)
    h[order] = 1.0
    
    c = score_h(h)
    best_h = h.copy()
    best_c = c
    T = 0.02
    cool = 0.97
    
    for it in range(n_steps):
        if time.time() > deadline:
            break
        # Pick two positions to swap
        p = int(rng.integers(0, n))
        q = int(rng.integers(0, n))
        if p == q or h[p] == h[q]:
            continue
        
        # Swap
        h[p], h[q] = h[q], h[p]
        c_new = score_h(h)
        delta = c_new - c
        if delta < 0 or rng.random() < math.exp(-delta / max(T, 1e-8)):
            c = c_new
            if c < best_c - 1e-14:
                best_c = c
                best_h = h.copy()
        else:
            h[p], h[q] = h[q], h[p]  # Revert
        
        T *= cool
    
    return best_h, best_c

def resample(h, N):
    """Resample h to N points"""
    n0 = len(h)
    if n0 == N:
        return h.copy()
    xold = (np.arange(n0) + 0.5) * (2.0 / n0)
    xnew = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(xnew, xold, h)
    return make_feasible(y, N / 2)
