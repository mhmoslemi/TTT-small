            proj[mask] += diff / nm
            proj[mask] = np.clip(proj[mask], 0.0, 1.0)
    return proj

def grad_smooth(h, corr, tau, n):
    cs = corr - np.max(corr)
    # Numerically stable softmax
    cs_clipped = np.clip(cs, -50.0/tau, 0.0)
    w = np.exp(tau * cs_clipped)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            # corr_k = sum_{j=lag}^{n-1} h[j]*(1-h[j-lag])
            # d/dh[j] for j in [lag, n-1]: +(1-h[j-lag])
            # d/dh[j] for j in [0, n-1-lag): -h[j+lag]
            idx1 = np.arange(lag, n)
            grad[idx1] += wk * (1.0 - h[idx1 - lag])
            idx2 = np.arange(0, n - lag)
            grad[idx2] -= wk * h[idx2 + lag]
        else:
            L = -lag
            # corr_k = sum_{j=0}^{n-1-L} h[j]*(1-h[j+L])
            # d/dh[j] for j in [0, n-1-L]: +(1-h[j+L])
            # d/dh[j] for j in [L, n-1]: -h[j-L]
            idx1 = np.arange(0, n - L)
            grad[idx1] += wk * (1.0 - h[idx1 + L])
            idx2 = np.arange(L, n)
            grad[idx2] -= wk * h[idx2 - L]
    return grad

import numpy as np
import time

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def score_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(p, n):
    target = 0.5 * n
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        if float(np.sum(proj)) > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-10:
        mask = (proj > 1e-9) & (proj < 1.0 - 1e-9)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
            proj[mask] = np.clip(proj[mask], 0.0, 1.0)
    return proj

def grad_smooth(h, corr, tau, n):
    cs = corr - np.max(corr)
    cs_clip = np.clip(cs, -50.0 / max(tau, 1.0), 0.0)
    w = np.exp(tau * cs_clip)
    wsum = w.sum()
    if wsum < 1e-300:
        return np.zeros(n)
    w = w / wsum
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] -= wk * h[lag:n]
        else:
            L = -lag
            grad[0:n - L] += wk * (1.0 - h[L:n])
            grad[L:n] -= wk * h[0:n - L]
    return grad

def grad_topk(h, corr, n, ktop=8):
    vals = corr.copy()
    maxv = np.max(vals)
    idx = np.argsort(vals)[::-1][:ktop]
    active = vals[idx]
    cs = active - maxv
    w = np.exp(80.0 * cs)
    wsum = w.sum()
    if wsum < 1e-300:
        w = np.ones(ktop) / ktop
    else:
        w = w / wsum
    grad = np.zeros(n)
    for ii, k in enumerate(idx):
        wk = w[ii]
        if wk < 1e-15:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:n] += wk * (1.0 - h[0:n - lag])
            grad[0:n - lag] -= wk * h[lag:n]
        else:
            L = -lag
            grad[0:n - L] += wk * (1.0 - h[L:n])
            grad[L:n] -= wk * h[0:n - L]
    return grad

def smooth_descent(h, t_start, t_end, tau_lo, tau_hi, n, momentum=False):
    best_h = h.copy()
    best_c5 = score_c5(h)
    cur_c5 = best_c5
    v = np.zeros(n)
    step = 4.0 / n
    n_iter = 0
    stall_count = 0
    while time.time() < t_end:
        frac = (time.time() - t_start) / max(t_end - t_start, 1e-6)
        frac = min(max(frac, 0.0), 1.0)
        tau = tau_lo + (tau_hi - tau_lo) * frac
        corr = compute_corr(h)
        cur_c5 = float(np.max(corr) * 2.0 / n)
        if cur_c5 < best_c5 - 1e-14:
            best_c5 = cur_c5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        if stall_count > 200:
            if tau < 50:
                h = project_box_sum(h + 0.02 * np.random.randn(n), n)
                stall_count = 0
                continue
            break
        if momentum:
            v = 0.85 * v + 0.15 * grad_smooth(h, corr, tau, n)
            g = v
        else:
            g = grad_smooth(h, corr, tau, n)
        gnorm = np.linalg.norm(g)
        if gnorm < 1e-14:
            break
        h_new = project_box_sum(h - step * g, n)
        c_new = score_c5(h_new)
        if c_new <= cur_c5 + 1e-13:
            if c_new < best_c5 - 1e-14:
                best_c5 = c_new
                best_h = h_new.copy()
                stall_count = 0
            h = h_new
            cur_c5 = c_new
            step = min(step * 1.04, 8.0 / n)

            step *= 0.5
            if step < 0.1 / n:
                step = 0.5 / n
                h = project_box_sum(h + 0.01 * np.random.randn(n), n)
                stall_count = 0
            elif step < 2.0 / n:
                h = project_box_sum(h - step * g, n)
                cur_c5 = score_c5(h)
        n_iter += 1
    return best_h, best_c5

    dx = 2.0 / n
    accepted = 0
    total = 0
    while time.time() < t_end:

        for _ in range(batch):

            i = rng.randint(n)
            j = rng.randint(n)
            if i == j:
                continue
            direction = 1 if rng.rand() < 0.5 else -1
            if direction == 1:

            else:
                max_d = min(h[j], 1.0 - h[i])
            if max_d < 1e-6:

            for dfrac in [1.0, 0.5, 0.25, 0.75]:

                if d < 1e-8:

                h_new = h.copy()
                if direction == 1:
                    h_new[i] -= d
                    h_new[j] += d
                else:
                    h_new[j] -= d
                    h_new[i] += d
                h_new = project_box_sum(h_new, n)
                c_new = score_c5(h_new)
                total += 1
                if c_new < cur_c5 - 1e-14:
                    h = h_new
                    cur_c5 = c_new
                    if c_new < best_c5 - 1e-14:
                        best_c5 = c_new
                        best_h = h_new.copy()
                    accepted += 1
                    break
                elif c_new < best_c5 - 1e-14:
                    break
    return best_h, best_c5
