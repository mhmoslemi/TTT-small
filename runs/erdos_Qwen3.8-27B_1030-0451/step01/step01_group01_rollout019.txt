def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    x = np.array(x, dtype=float).copy()
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-15:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-15:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    # Final fix
    r2 = target - y.sum()
    if abs(r2) > 1e-14:

            idx = np.where(y < 1.0)[0]
        else:
            idx = np.where(y > 0.0)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def project_sum_box(x, target):
    """Project x onto {y: 0<=y<=1, sum(y)=target}."""
    x = np.array(x, dtype=float).copy()
    n = len(x)
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        y = np.clip(x + mid, 0.0, 1.0)
        if y.sum() > target:
            hi = mid
        else:
            lo = mid
    y = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    r = target - y.sum()
    if abs(r) > 1e-12:
        if r > 0:
            slack = 1.0 - y
            ts = slack.sum()
            if ts > 1e-15:
                y = np.clip(y + r * slack / ts, 0.0, 1.0)
        else:
            ty = y.sum()
            if ty > 1e-15:
                y = np.clip(y + r * y / ty, 0.0, 1.0)
    r2 = target - y.sum()
    if abs(r2) > 1e-14:
        if r2 > 0:
            idx = np.where(y < 1.0 - 1e-15)[0]
        else:
            idx = np.where(y > 1e-15)[0]
        if len(idx) > 0:
            y[idx] += r2 / len(idx)
    return np.clip(y, 0.0, 1.0)

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))

def solve_pair_minimax(h, i, j, corr_raw, n):
    """Solve 1D minimax for pair (i,j). Returns (V_opt_raw, u_opt) or None."""
    a = h[i]
    b = h[j]
    u_lo = max(-a, b - 1.0)
    u_hi = min(1.0 - a, b)
    if u_hi - u_lo < 1e-12:
        return None
    
    # Padded h: h_pad[m] = h[m-n] for n<=m<2n, else 0
    h_pad = np.zeros(3 * n)
    h_pad[n:2*n] = h
    
    d = np.arange(2 * n - 1, dtype=np.int64) - (n - 1)
    
    ip = i + d + n
    im = i - d + n
    jp = j + d + n
    jm = j - d + n
    
    ip = np.clip(ip, 0, 3*n - 1)
    im = np.clip(im, 0, 3*n - 1)
    jp = np.clip(jp, 0, 3*n - 1)
    jm = np.clip(jm, 0, 3*n - 1)
    
    C = (h_pad[jp] - h_pad[ip]) + (h_pad[im] - h_pad[jm])
    
    # Minimize max_k (corr_raw[k] + u * C[k]) for u in [u_lo, u_hi]
    V_hi = float(np.max(corr_raw + u_hi * C))
    V_lo = 0.0
    
    for _ in range(45):
        V_mid = (V_lo + V_hi) / 2.0
        pos = C > 1e-14
        neg = C < -1e-14
        u_min_req = u_lo
        u_max_req = u_hi
        if np.any(pos):
            vals = (V_mid - corr_raw[pos]) / C[pos]
            u_max_req = min(u_max_req, float(np.min(vals)))
        if np.any(neg):
            vals = (V_mid - corr_raw[neg]) / C[neg]
            u_min_req = max(u_min_req, float(np.max(vals)))
        if np.any(~pos & ~neg):
            if float(np.max(corr_raw[~pos & ~neg])) > V_mid:
                V_lo = V_mid
                continue
        if u_min_req <= u_max_req:
            V_hi = V_mid
        else:
            V_lo = V_mid
    
    V_opt = (V_lo + V_hi) / 2.0
    
    # Find u_opt
    pos = C > 1e-14
    neg = C < -1e-14
    u_min_req = u_lo
    u_max_req = u_hi
    if np.any(pos):
        u_max_req = min(u_max_req, float(np.min((V_opt - corr_raw[pos]) / C[pos])))
    if np.any(neg):
        u_min_req = max(u_min_req, float(np.max((V_opt - corr_raw[neg]) / C[neg])))
    if u_min_req > u_max_req:
        u_opt = (u_lo + u_hi) / 2.0
    else:
        u_opt = (u_min_req + u_max_req) / 2.0
    u_opt = float(np.clip(u_opt, u_lo, u_hi))
    
    return V_opt, u_opt

def pair_exchange_optimize(h0, deadline, max_steps_per_iter=400, no_improve_limit=150):
    """Iterative best-pair exchange optimization."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    no_improve = 0
    iter_count = 0
    
    while time.time() < deadline:
        iter_count += 1
        corr_raw = np.correlate(h, 1.0 - h, mode="full")
        corr_s = corr_raw * (2.0 / n)
        c_max = float(np.max(corr_s))
        
        # Active lags
        gap = max(0.03 * c_max, 0.0005)
        active_k = np.where(corr_s >= c_max - gap)[0]
        active_lags = active_k - (n - 1)
        
        # Build candidate pairs
        pairs = set()
        for d in active_lags:
            d = int(d)
            if d == 0 or abs(d) >= n:
                continue
            for i in range(max(0, -d), min(n, n - d)):
                j = i + d
                pairs.add((min(i,j), max(i,j)))
        
        # Add random pairs for exploration
        n_rand = min(50, n * n // 4)
        for _ in range(n_rand):
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j:
                pairs.add((min(i,j), max(i,j)))
        
        pair_list = list(pairs)
        np.random.shuffle(pair_list)
        pair_list = pair_list[:max_steps_per_iter]
        
        # Find best
        best_delta = 0.0
        best_pair = None
        best_u = 0.0
        current_c_raw = float(np.max(corr_raw))
        
        for (i, j) in pair_list:
            res = solve_pair_minimax(h, i, j, corr_raw, n)
            if res is not None:
                V_opt, u_opt = res
                V_opt_s = V_opt * (2.0 / n)
                delta = best_c - V_opt_s
                if delta > best_delta:
                    best_delta = delta
                    best_pair = (i, j)
                    best_u = u_opt
        
        if best_pair is not None and best_delta > 1e-12:
            i, j = best_pair
            s = h[i] + h[j]
            h[i] += best_u
            h[j] -= best_u
            h = project_sum_box(h, n / 2.0)
            c_cur = evaluate_c5(h)
            if c_cur < best_c - 1e-14:
                best_h = h.copy()
                best_c = c_cur
                no_improve = 0
            else:
                no_improve += 1
        else:
            no_improve += 1
        
        if no_improve >= no_improve_limit:
            break
    
    return best_h, best_c

def make_binary_seed(n, rng):
    """Create balanced binary sequence."""
    if n % 2 != 0:
        n = n + 1
    half = n // 2
    signs = np.zeros(n)
    signs[:half] = 0.0
    signs[half:] = 1.0
    rng.shuffle(signs)
    return signs.copy()

def make_periodic_seed(n, period, fill, rng):
    """Create periodic seed with given pattern length and fill count."""
    if n % 2 != 0:
        n = n + 1
    base = np.zeros(period)
    base[:min(fill, period)] = 1.0
    reps = (n + period - 1) // period
    seq = np.tile(base, reps)[:n]
    # Adjust for balance
    half = n // 2
    excess = int(np.sum(seq)) - half
    if excess > 0:
        ones = np.where(seq == 1.0)[0]
        rng.shuffle(ones)
        seq[ones[:excess]] = 0.0
    elif excess < 0:
        zeros = np.where(seq == 0.0)[0]
        rng.shuffle(zeros)
        seq[zeros[:(-excess)]] = 1.0
    return seq

def relax_binary(h_bin, deadline):
    """Relax a binary seed using pair exchange."""
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    h = project_sum_box(h, n / 2.0)
    return pair_exchange_optimize(h, deadline, max_steps_per_iter=300, no_improve_limit=80)

def transfer_resolution(h0, n_new, rng=None):
    """Interpolate h0 to n_new points."""
    n_old = len(h0)
    x_old = np.linspace(0.0, 2.0, n_old, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    # Periodic interpolation
    h_ext = np.concatenate([h0, h0[:1]])
    x_ext = np.concatenate([x_old, [x_old[-1] + 2.0/n_old]])
    h_new = np.interp(x_new, x_old, h0, period=2.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    rng = np.random.RandomState(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0
    
    # Get initial values
    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = np.full(84, 0.5)
    
    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n_best = len(best_h)
    
    def time_left():
        return deadline - time.time()
    
    # Phase 1: Polish initial with pair exchange
    if time_left() > 60:
        dl = min(deadline, start_time + budget_s * 0.25)
        try:
            h1, c1 = pair_exchange_optimize(best_h, dl, max_steps_per_iter=400, no_improve_limit=200)
            if c1 < best_c:
                best_h, best_c, n_best = h1, c1, len(h1)
        except Exception:
            pass
    
    # Phase 2: Try binary seeds at various sizes
    binary_configs = [
        (84, 'random'), (84, 'periodic4'), (84, 'periodic6'),
        (88, 'random'), (88, 'periodic4'), (92, 'random'),
        (96, 'random'), (96, 'periodic4'), (100, 'random'),
        (100, 'periodic6'), (108, 'random'), (112, 'random'),
        (104, 'periodic4'), (92, 'periodic6'), (88, 'periodic6'),
    ]
    
    for (bn, btype) in binary_configs:
        if time_left() < 40:
            break
        try:
            if btype == 'random':
                h_bin = make_binary_seed(bn, rng)
            elif btype == 'periodic4':
                h_bin = make_periodic_seed(bn, 4, 2, rng)
            elif btype == 'periodic6':
                h_bin = make_periodic_seed(bn, 6, 3, rng)
            else:
                h_bin = make_binary_seed(bn, rng)
            
            bl_dl = min(deadline, time.time() + min(60, time_left() * 0.4))
            h_r, c_r = relax_binary(h_bin, bl_dl)
            if c_r < best_c:
                best_h, best_c, n_best = h_r, c_r, len(h_r)
        except Exception:
            pass
    
    # Phase 3: More pair exchange on best
    if time_left() > 60:
        dl3 = min(deadline, start_time + budget_s * 0.65)
        try:
            h3, c3 = pair_exchange_optimize(best_h, dl3, max_steps_per_iter=500, no_improve_limit=300)
            if c3 < best_c:
                best_h, best_c, n_best = h3, c3, len(h3)
        except Exception:
            pass
    
    # Phase 4: Resolution transfer and re-optimize
    if time_left() > 80:
        n_cur = len(best_h)
        target_sizes = []
        for f in [1.15, 1.25, 1.4, 1.6]:
            t = int(round(n_cur * f))
            if t > n_cur and t <= 200 and t not in target_sizes:
                target_sizes.append(t)
        # Also try slightly smaller
        for f in [0.85, 0.9]:
            t = int(round(n_cur * f))
            if t < n_cur and t >= 60 and t not in target_sizes:
                target_sizes.append(t)
        
        for n_t in sorted(target_sizes):
            if time_left() < 50:
                break
            try:
                h_tr = transfer_resolution(best_h, n_t, rng)
                c_tr = evaluate_c5(h_tr)
                tr_dl = min(deadline, time.time() + min(80, time_left() * 0.5))
                h_opt, c_opt = pair_exchange_optimize(h_tr, tr_dl, max_steps_per_iter=300, no_improve_limit=100)
                if c_opt < best_c:
                    best_h, best_c, n_best = h_opt, c_opt, len(h_opt)
            except Exception:
                pass
    
    # Phase 5: Final intensive polish
    if time_left() > 20:
        try:
            h5, c5 = pair_exchange_optimize(best_h, deadline - 1.0, max_steps_per_iter=500, no_improve_limit=500)
            if c5 < best_c:
                best_h, best_c, n_best = h5, c5, len(h5)
        except Exception:
            pass
    
    # Final verification and fix
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c
    
    return best_h, float(best_c), len(best_h)
