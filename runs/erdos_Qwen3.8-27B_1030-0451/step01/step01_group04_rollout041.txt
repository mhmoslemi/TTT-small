import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, tau, mu, target_sum):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    obj = m + tau * np.log(total) + mu * (np.sum(h) - target_sum) ** 2
    return obj

def smooth_grad(h, tau, mu, target_sum):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    total = w.sum()
    w = w / total
    
    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]
    
    g += 2.0 * mu * (np.sum(h) - target_sum)
    return g

def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(80):

        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    t = (lo + hi) * 0.5
    h = np.clip(h + t, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-13:

        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])

    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)

def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def smooth_obj(h, tau, mu, target_sum):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    obj = m + tau * np.log(np.exp(exponents).sum()) + mu * (np.sum(h) - target_sum) ** 2
    return obj

def smooth_grad(h, tau, mu, target_sum):

    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)

import numpy as np
import time
from scipy.optimize import minimize


def compute_c5(h):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)


def smooth_obj(h, tau, mu, target_sum):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    obj = m + tau * np.log(np.exp(exponents).sum())
    if mu > 0:
        obj += mu * (np.sum(h) - target_sum) ** 2
    return obj


def smooth_grad(h, tau, mu, target_sum):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    c = corr * dx
    m = np.max(c)
    exponents = np.clip((c - m) / tau, -500.0, 500.0)
    w = np.exp(exponents)
    w = w / w.sum()

    g = np.zeros(N)
    for r in range(2 * N - 1):
        a = w[r] * dx
        if a < 1e-15:
            continue
        d = r - (N - 1)
        if d >= 0:
            n = N - d
            if n > 0:
                g[d:] += a * (1.0 - h[:n])
                g[:n] -= a * h[d:]
        else:
            dd = -d
            n = N - dd
            if n > 0:
                g[:n] += a * (1.0 - h[dd:])
                g[dd:] -= a * h[:n]

    if mu > 0:
        g += 2.0 * mu * (np.sum(h) - target_sum)
    return g


def project_feasible(h, target_sum):
    h = np.array(h, dtype=float, copy=True)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5

        if s > target_sum:
            hi = mid
        else:

    t = (lo + hi) * 0.5

        if residual > 0:
            idx = np.where(h < 1.0)[0]
            if len(idx) > 0:
                h[idx[0]] += min(residual, 1.0 - h[idx[0]])
        else:
            idx = np.where(h > 0.0)[0]
            if len(idx) > 0:
                h[idx[0]] += max(residual, -h[idx[0]])
    return h


def upscale_func(h, N_new):
    N_old = len(h)
    if N_old == N_new:
        return h.copy()
    x_old = (np.arange(N_old) + 0.5) * (2.0 / N_old)
    x_new = (np.arange(N_new) + 0.5) * (2.0 / N_new)
    h_new = np.interp(x_new, x_old, h)
    return project_feasible(h_new, N_new / 2.0)


def solve_coord_newton(a, b, tau):
    t = 0.5
    for _ in range(30):
        vals = a * t + b
        m = np.max(vals)
        w = np.exp((vals - m) / tau)
        sw = np.sum(w)
        if sw < 1e-30:
            break
        f1 = np.sum(a * w) / sw
        f2 = np.sum(a * a * w) / sw - f1 * f1
        if f2 < 1e-15:
            break
        t_new = t - f1 / f2
        if t_new < 0 or t_new > 1:
            t_new = np.clip(t_new, 0.01, 0.99)
        if abs(t_new - t) < 1e-12:
            t = t_new
            break
        t = t_new
    return np.clip(t, 0.0, 1.0)


def coordinate_descent_pass(h, tau, n_sweeps, time_limit):
    N = len(h)
    dx = 2.0 / N
    h = h.copy()

    for sweep in range(n_sweeps):
        if time.time() > time_limit:

        for j in range(N):
            if time.time() > time_limit and j % 50 == 0:
                break
            a = np.zeros(2 * N - 1)

                    b[r] += 0.0

                    a[r] -= h[j - s] * dx
                    b[r] += h[j - s] * dx
            t_new = solve_coord_newton(a, b, tau)
            h[j] = t_new

    return h
