import numpy as np
import time

def project_feasible(x, target):
    n = x.size
    target = float(target)
    x = np.clip(x, -2.0, 2.0)  # prevent extreme values
    lo = -1.0 - float(x.min())
    hi = 1.0 - float(x.max())
    if hi - lo < 1e-15:
        return np.full(n, target / n)
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = float(np.sum(np.clip(x + mid, 0.0, 1.0)))
        if s > target:
            hi = mid
        else:
            lo = mid
    result = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix tiny sum errors
    diff = target - float(result.sum())
    if abs(diff) > 1e-10:
        if diff > 0:
            order = np.argsort(result)
            rem = diff
            for i in order:
                add = min(1.0 - result[i], rem)
                result[i] += add
                rem -= add
                if rem < 1e-14:
                    break
        else:
            rem = -diff
            order = np.argsort(result)[::-1]
            for i in order:
                sub = min(result[i], rem)
                result[i] -= sub
                rem -= sub
                if rem < 1e-14:
                    break
    return result

def compute_C(h):
    n = h.size
    dx = 2.0 / float(n)
    return np.correlate(h, 1.0 - h, "full") * dx

def C_max(h):
    return float(np.max(compute_C(h)))

def smooth_obj(h, tau):
    C = compute_C(h)
    m = float(np.max(C))
    a = (C - m) / tau
    a = np.clip(a, -100.0, 0.0)
    s = float(np.sum(np.exp(a)))
    return m + tau * float(np.log(max(s, 1e-300)))

def smooth_grad(h, tau):
    n = h.size
    dx = 2.0 / float(n)
    C = compute_C(h)
    m = float(np.max(C))
    a = (C - m) / tau
    a = np.clip(a, -100.0, 0.0)
    e = np.exp(a)
    s = float(np.sum(e))
    if s < 1e-300:
        alpha = np.zeros_like(C)
        alpha[np.argmax(C)] = 1.0
    else:
        alpha = e / s
    
    G = np.zeros(n)
    ctr = n - 1
    
    # For efficiency, only process lags with significant weight
    significant = alpha > 1e-8
    for k in np.where(significant)[0]:
        w = alpha[k] * dx
        d = int(k) - ctr
        if d >= 0:
            # grad[j] = (1-h[j-d]) for j in [d, n-1]
            #          - h[j+d]  for j in [0, n-d-1]
            if d == 0:
                G += w * (1.0 - 2.0 * h)
            else:
                G[d:] += w * (1.0 - h[:-d])
                G[:n-d] += -w * h[d:]
        else:
            e2 = -d
            # grad[j] = (1-h[j+e2]) for j in [0, n-e2-1]
            #          - h[j-e2]    for j in [e2, n-1]
            G[:n-e2] += w * (1.0 - h[e2:])
            G[e2:] += -w * h[:n-e2]
    
    return G

def subgrad_descent(h, target, tau, deadline, rng):
    n = h.size
    h = h.copy()
    best_c = C_max(h)
    alpha = 0.01
    no_improve = 0
    
    while time.time() < deadline:
        G = smooth_grad(h, tau)
        # Center the gradient to maintain sum constraint
        G = G - float(np.mean(G))
        
        # Try a few step sizes
        improved = False
        for step_factor in (1.0, 0.5, 0.25, 0.1, 0.05, 0.01):
            step = alpha * step_factor
            h_new = project_feasible(h - step * G, target)
            c_new = C_max(h_new)
            if c_new < best_c - 1e-12:
                h = h_new
                best_c = c_new
                no_improve = 0
                improved = True
                break
        
        if not improved:
            no_improve += 1
            alpha *= 0.5
            if alpha < 1e-8:
                break
        
        # Occasionally reschedule tau
        if no_improve % 10 == 5:
            tau *= 0.5
            if tau < 1e-4:
                tau = 1e-4
    
    return h, best_c

def pair_local_search(h, target, deadline, n_pairs=200):
    """For selected pairs (i,j), try moving mass between them."""
    n = h.size
    best_c = C_max(h)
    h = h.copy()

    C = compute_C(h)
    dx = 2.0 / float(n)
    target = float(target)
    
    # Find pairs that might help: ones where moving mass from high to low
    # reduces the max correlation
    # Simple heuristic: try pairs where h[i] > h[j] (move mass from high to low)

    rng = np.random.default_rng(12345)

            break

        pairs = []
        for _ in range(n_pairs):
            i = rng.integers(0, n)
            j = rng.integers(0, n)
            if i != j:
                pairs.append((i, j))
        
        for (i, j) in pairs:
            if time.time() > deadline:
                break
            
            # Try moving delta from j to i (if h[j] > h[i])
            # or from i to j (if h[i] > h[j])
            direction = 1 if h[j] > h[i] else -1
            max_delta = min(abs(h[i] - h[j]), 0.5)
            
            # Try a few delta values
            for frac in (0.1, 0.3, 0.5, 0.7):
                delta = direction * max_delta * frac
                if h[i] + delta < 0 or h[i] + delta > 1:
                    continue
                if h[j] - delta < 0 or h[j] - delta > 1:
                    continue
                
                h2 = h.copy()
                h2[i] += delta
                h2[j] -= delta
                # Fix sum (should be exact with this move)
                c2 = C_max(h2)
                if c2 < best_c - 1e-12:
                    h = h2
                    best_c = c2
                    break
    
    return h, best_c

    xold = (np.arange(n0) + 0.5) * (2.0 / n0)

                best = h
    return best
