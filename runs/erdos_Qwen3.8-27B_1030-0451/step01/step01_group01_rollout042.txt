Let me also think about a different kind of optimization: **paired swaps**.

For binary-ish solutions, swapping two values (one up, one down) is a natural move. More generally, for continuous h, we can do:
- Pick i, j
- Move ε from h[j] to h[i] (maintaining sum)
- Accept if it reduces C₅

This is like coordinate descent but with paired moves. The problem is that n^2 pairs is expensive. But we can be smart about which pairs to try:
- Pairs where h[i] and h[j] have large difference
- Pairs at specific relative positions (related to active lags)

Actually, for n=84, we have 84*83/2 = 3486 pairs. For each pair, evaluating C₅ takes O(n^2) time (for the full correlation). That's about 3486 * 7056 ≈ 24.6 million operations per round. With n=84, the correlation is 167 values, each computed in O(n) = O(84), so total is 167*84 = 14028 per evaluation. With 3486 pairs, that's about 49 million per round. That's feasible for a few rounds.

But actually, for the paired swap, we can compute the change in C₅ more efficiently. If we change h[i] += ε and h[j] -= ε, the change in c_k is:
Δc_k = ε * (grad_k[i] - grad_k[j])

So the change in the maximum is bounded by max_k ε * (grad_k[i] - grad_k[j]). But the actual change in the max depends on which k is active.

This gives us a fast way to evaluate the effect of a swap without recomputing the full correlation!

        m1 = np.arange(0, n - l)
        g[m1] += 1.0 - h[m1 + l]
        m2 = np.arange(l, n)
        g[m2] -= h[m2 - l]
    return g

def smooth_obj_and_grad(h, beta):
    v = corr_all(h)
    m = np.max(v)
    expo = np.clip(beta * (v - m), None, 500)
    u = np.exp(expo)
    Z = u.sum()
    f = m - np.log(Z) / beta
    w = u / Z
    n = len(h)
    g = np.zeros(n)
    for k in range(len(v)):
        if w[k] > 1e-14:
            lag = k - (n - 1)
            g += w[k] * grad_lag(h, lag)
    return f, g

def lp_trust(h0, deadline, max_iters=400, init_radius=0.06, max_radius=0.10, max_active=120):
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    radius = init_radius
    no_improve = 0
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        v = corr_all(h)
        vmax = np.max(v)
        gap = max(0.004 * vmax, 0.002)
        active_idx = np.where(v >= vmax - gap)[0]
        if len(active_idx) > max_active:
            order = np.argsort(v[active_idx])
            active_idx = active_idx[order[-max_active:]]
        if len(active_idx) == 0:
            active_idx = np.argsort(v)[-20:]

        A = np.zeros((len(active_idx), n))
        b_lin = np.zeros(len(active_idx))
        for i, k in enumerate(active_idx):
            lag = k - (n - 1)
            A[i] = grad_lag(h, lag)
            b_lin[i] = A[i] @ h - v[k]

        n_var = n + 1
        c_obj = np.zeros(n_var)
        c_obj[n] = 1.0
        A_ub = np.hstack([A, -np.ones((len(active_idx), 1))])
        lb = np.concatenate([np.maximum(h - radius, 0.0), [-0.5]])
        ub = np.concatenate([np.minimum(h + radius, 1.0), [vmax]])
        A_eq = np.zeros((1, n_var))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])

        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_lin,
                          A_eq=A_eq, b_eq=b_eq,
                          bounds=list(zip(lb, ub)), method='highs')
        except Exception:
            res = None

        if res is not None and res.success and res.x is not None:
            x_new = project_sum_box(res.x[:n], n / 2.0)
            c_new = evaluate_c5(x_new)
            if c_new < best_c - 1e-12:
                best_h = x_new.copy()
                best_c = c_new
                h = x_new.copy()
                no_improve = 0
                stall = 0
                radius = min(radius * 1.08, max_radius)
            else:
                no_improve += 1
                stall += 1
        else:
            no_improve += 1
            stall += 1

        if no_improve >= 4:
            radius = max(radius * 0.55, 0.004)
            no_improve = 0
            if radius <= 0.004:
                break
        if stall > 50:
            h = h + np.random.randn(n) * 0.008
            h = project_sum_box(h, n / 2.0)
            radius = max(init_radius, 0.04)
            stall = 0

    return best_h, best_c

def smooth_momentum(h0, deadline, max_iters=4000):
    n = len(h0)
    best_h = h0.copy()
    best_c = evaluate_c5(best_h)
    h = h0.copy()
    v_mom = np.zeros(n)
    step = 0.004
    stall = 0

    for it in range(max_iters):
        if time.time() > deadline:
            break
        beta = 25.0 + 5.0 * it / max_iters
        f, g = smooth_obj_and_grad(h, beta)
        v_mom = 0.88 * v_mom - step * g
        h_new = project_sum_box(h + v_mom, n / 2.0)
        c_new = evaluate_c5(h_new)
        if c_new < best_c - 1e-13:
            best_c = c_new
            best_h = h_new.copy()
            stall = 0
            step = min(step * 1.02, 0.015)
        else:
            stall += 1
            step *= 0.96
        h = h_new
        if stall > 300 or step < 1e-7:
            h = best_h.copy()
            v_mom = np.zeros(n)
            if step < 1e-7:
                break
            stall = 0

    return best_h, best_c

def transfer_res(h0, n_new):
    n_old = len(h0)
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, np.concatenate([h0, h0[:1]]), period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def fuse_res(h0, n_new):
    """Interpolate h0 down to n_new points."""
    n_old = len(h0)
    if n_old == n_new:
        return h0.copy()
    x_old = np.arange(n_old) / n_old
    x_new = np.arange(n_new) / n_new
    h_new = np.interp(x_new, x_old, np.concatenate([h0, h0[:1]]), period=1.0)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def make_initial(n=84):
    # Construct a structured initial guess: blocks of varying size
    h = np.zeros(n)
    # Create a pattern with period roughly n/3
    p = max(2, n // 3)
    fill = max(1, p // 2)
    for i in range(0, n, p):
        h[i:i+fill] = 1.0
    h = project_sum_box(h, n / 2.0)
    return h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0

    # Get initial values
    h_init = None
    ig = globals().get('initial_h_values', None)
    if ig is not None:
        try:
            h_try = np.array(ig, dtype=float).copy()
            n_try = len(h_try)
            if n_try >= 4 and np.all(np.isfinite(h_try)):
                h_try = project_sum_box(h_try, n_try / 2.0)
                c_try = evaluate_c5(h_try)
                if np.isfinite(c_try):
                    h_init = h_try
        except Exception:
            pass
    if h_init is None:
        h_init = make_initial(84)

    best_h = h_init.copy()
    best_c = evaluate_c5(best_h)
    n0 = len(best_h)

    # Phase 1: LP trust region (primary optimization)
    dl1 = min(deadline, start_time + budget_s * 0.22)
    try:
        h1, c1 = lp_trust(best_h, dl1, max_iters=500, init_radius=0.07, max_radius=0.12, max_active=140)
        if c1 < best_c:
            best_h, best_c = h1, c1
    except Exception:
        pass

    # Phase 2: Smooth gradient polish
    dl2 = min(deadline, start_time + budget_s * 0.33)
    try:
        h2, c2 = smooth_momentum(best_h, dl2, max_iters=3000)
        if c2 < best_c:
            best_h, best_c = h2, c2
    except Exception:
        pass

    # Phase 3: LP with finer radius
    dl3 = min(deadline, start_time + budget_s * 0.48)
    try:
        h3, c3 = lp_trust(best_h, dl3, max_iters=400, init_radius=0.04, max_radius=0.07, max_active=160)
        if c3 < best_c:
            best_h, best_c = h3, c3
    except Exception:
        pass

    # Phase 4: Resolution cycling (up and back)
    if time.time() < deadline - 120:
        n_cur = len(best_h)
        n_up = min(n_cur * 2, 280)
        if n_up > n_cur:
            h_up = transfer_res(best_h, n_up)
            dl4 = min(deadline - 80, start_time + budget_s * 0.68)
            try:
                h4, c4 = lp_trust(h_up, dl4, max_iters=200, init_radius=0.05, max_radius=0.08, max_active=100)
                if c4 < best_c:
                    best_h, best_c = h4, c4
                # Cycle back down
                h_back = fuse_res(h4 if c4 < best_c else best_h, n_cur)
                dl4b = min(deadline - 60, start_time + budget_s * 0.75)
                h4b, c4b = lp_trust(h_back, dl4b, max_iters=200, init_radius=0.03, max_radius=0.05, max_active=140)
                if c4b < best_c:
                    best_h, best_c = h4b, c4b
            except Exception:
                pass

    # Phase 5: Try another resolution (3x if time permits)
    if time.time() < deadline - 100:
        n_cur = len(best_h)
        n_up2 = min(n_cur * 3, 400)
        if n_up2 > n_cur:
            h_up2 = transfer_res(best_h, n_up2)
            dl5 = min(deadline - 70, start_time + budget_s * 0.82)
            try:
                h5, c5 = lp_trust(h_up2, dl5, max_iters=150, init_radius=0.04, max_radius=0.06, max_active=80)
                if c5 < best_c:
                    best_h, best_c = h5, c5
                    h_back2 = fuse_res(h5, n_cur)
                    dl5b = min(deadline - 40, start_time + budget_s * 0.88)
                    h5b, c5b = lp_trust(h_back2, dl5b, max_iters=200, init_radius=0.03, max_radius=0.05, max_active=140)
                    if c5b < best_c:
                        best_h, best_c = h5b, c5b
            except Exception:
                pass

    # Phase 6: Final fine polish - very small LP steps
    dl6 = min(deadline, start_time + budget_s * 0.95)
    try:
        h6, c6 = lp_trust(best_h, dl6, max_iters=600, init_radius=0.02, max_radius=0.035, max_active=180)
        if c6 < best_c:
            best_h, best_c = h6, c6
    except Exception:
        pass

    # Phase 7: Final smooth gradient
    if time.time() < deadline - 5:
        try:
            h7, c7 = smooth_momentum(best_h, deadline - 1, max_iters=2000)
            if c7 < best_c:
                best_h, best_c = h7, c7
        except Exception:
            pass

    # Final verification
    best_h = project_sum_box(best_h, len(best_h) / 2.0)
    final_c = evaluate_c5(best_h)
    if final_c < best_c:
        best_c = final_c

    return best_h, float(best_c), len(best_h)
