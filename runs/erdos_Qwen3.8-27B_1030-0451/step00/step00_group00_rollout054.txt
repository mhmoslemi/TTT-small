- To minimize, we want to move in the direction opposite to the subgradient
- The minimum is where 0 is in the subgradient, i.e., there exist k1, k2 achieving the max with s_{k1} <= 0 <= s_{k2}

        h_temp[a] = val
        h_temp[n-1-a] = 1.0 - val
        return exact_c5(h_temp)

    lo, hi = 0.0, 1.0
    for _ in range(20):  # 20 iterations gives precision ~1e-6
        mid = (lo + hi) / 2

        eps = 1e-4

            # At left boundary
            hi = mid
            continue
        if mid + eps > 1:
            lo = mid
            continue
        f_left = obj(mid - eps)
        f_right = obj(mid + eps)
        if f_right > f_left:
            hi = mid
        else:
            lo = mid
    
    best = (lo + hi) / 2
    return best, obj(best)
