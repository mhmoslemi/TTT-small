def subgradient_step(h, N):
    dx = 2.0 / N
    c = np.correlate(h, 1-h, mode='full') * dx
    cmax = np.max(c)
    
    # Find active set (within tolerance)
    tol = 1e-4 * cmax  # relative tolerance for "active"
    active_idx = np.where(c >= cmax - tol)[0]
    
    # If only one active, use its gradient
    # If multiple, average their gradients
    grad = np.zeros(N)
    for r in active_idx:
        d = r - (N - 1)
        if d == 0:
            grad += (1 - 2*h)
        elif d > 0:
            grad[d:] += (1 - h[:N-d])
            grad[:N-d] -= h[d:]
        else:  # d < 0, d' = -d
            dprime = -d
            grad[:N-dprime] += (1 - h[dprime:])
            grad[dprime:] -= h[:N-dprime]
    
    grad /= len(active_idx)
    return grad * dx, cmax, active_idx
