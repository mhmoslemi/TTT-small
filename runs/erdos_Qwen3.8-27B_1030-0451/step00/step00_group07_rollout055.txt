def mass_project(x, target):
    """Project x onto {y: 0<=y_i<=1, sum(y)=target}"""
    x = np.asarray(x, dtype=np.float64).copy()
    N = len(x)
    if N == 0:
        return x
    # Replace non-finite
    x[~np.isfinite(x)] = 0.5
    # Bisection on lambda: find lambda such that sum(clip(x+lambda, 0, 1)) = target
    lo = -max(x) - 2.0
    hi = 1.0 - min(x) + 2.0
    # Ensure bracket
    for _ in range(20):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    y = np.clip(x + lam, 0, 1)
    # Fine-tune
    residual = math.fsum(y.tolist()) - target
    if abs(residual) > 1e-10:
        if residual > 0:
            # Too much mass, reduce from largest values
            order = np.argsort(-y)
            for i in order:
                dec = min(residual, y[i])
                y[i] -= dec
                residual -= dec
                if abs(residual) < 1e-12:

        else:

            # Actually add to those with most room (smallest y)
            order = np.argsort(y)
            for i in order:
                inc = min(-residual, 1.0 - y[i])
                y[i] += inc
                residual += inc
                if abs(residual) < 1e-12:

def transfer(h_src, N_src, N_dst):
    """Transfer from h_src (length N_src) to N_dst using area average"""
    if N_src == N_dst:
        return h_src.copy()
    # Create piecewise constant function on [0,2]
    xs_src = (np.arange(N_src) + 0.5) * (2.0 / N_src)
    # For each target cell, compute the average value
    h_dst = np.zeros(N_dst, dtype=np.float64)
    dx_src = 2.0 / N_src
    dx_dst = 2.0 / N_dst
    for j in range(N_dst):
        x0 = j * dx_dst
        x1 = (j + 1) * dx_dst
        # Find source cells that overlap [x0, x1]
        # Source cell i covers [(i)*dx_src, (i+1)*dx_src]
        i_start = max(0, math.floor(x0 / dx_src))
        i_end = min(N_src - 1, math.ceil(x1 / dx_src) - 1)
        if i_start > i_end:
            h_dst[j] = 0.0
            continue
        total_area = 0.0
        for i in range(i_start, i_end + 1):
            s0 = i * dx_src
            s1 = (i + 1) * dx_src
            overlap = max(0.0, min(x1, s1) - max(x0, s0))
            total_area += overlap * h_src[i]
        h_dst[j] = total_area / dx_dst
    return h_dst
