where "valid first index" means 0 ≤ j ≤ n-1-k (for k≥0) and "valid second index" means k ≤ j ≤ n-1 (for k≥0).

- j is a valid "second index" (appears as h[i+lag] with i+lag=j) iff 0 ≤ j-lag and 0 ≤ j < n

∂corr[k]/∂h[j] = [1-h[j+lag] if j+lag in [0,n-1] else 0] - [h[j-lag] if j-lag in [0,n-1] else 0]

- second_grad[j] = h[j-lag] if 0 ≤ j-lag < n, else 0
- ∂corr[k]/∂h[j] = first_grad[j] - second_grad[j]

So G[k, :] = first_grad - second_grad, which can be computed using sliced arrays.

                res -= add

    dx = 2.0 / n

        else:
            first[-lag:] = 1.0 - h[:n+lag]
            second[:n+lag] = h[-lag:]

    return corr, G

def solve_lp_step(h, n, top_k=20, max_t=None):
    """Solve LP to find best perturbation."""
    corr, G = compute_gradients(h, n)
    current_max = np.max(corr) * (2.0/n)
    
    if max_t is None:
        max_t = current_max
    
    # Select top_k lags
    idx = np.argsort(corr)[-top_k:]
    
    # Setup LP
    # Variables: delta[0..n-1], t
    c_obj = np.zeros(n+1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: G[k,:] @ delta - t <= -corr[k]
    A_ub = np.zeros((len(idx), n+1))
    b_ub = np.zeros(len(idx))
    for i, k in enumerate(idx):
        A_ub[i, :n] = G[k]
        A_ub[i, n] = -1.0
        b_ub[i] = -corr[k]

    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    
    # Bounds
    bounds = [(-h[j], 1.0-h[j]) for j in range(n)]
    bounds.append((0.0, max_t))
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[0],
                  bounds=bounds, method='highs')

        new_h = h + delta
        new_h = project_feasible(new_h, n)
        new_c5 = evaluate_c5(new_h)
        return new_h, new_c5
