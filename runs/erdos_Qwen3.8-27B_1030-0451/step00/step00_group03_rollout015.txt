import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, target_sum):
    n = len(y)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    alpha = (lo + hi) / 2.0
    h = np.clip(y + alpha, 0.0, 1.0)
    # Fix tiny residual
    residual = target_sum - np.sum(h)
    if residual > 1e-12:
        # distribute to elements with room
        for _ in range(n):
            idx = np.argmax(1.0 - h)
            add = min(1.0 - h[idx], residual)
            h[idx] += add
            residual -= add
            if residual < 1e-14:
                break
    elif residual < -1e-12:
        for _ in range(n):
            idx = np.argmax(h)
            sub = min(h[idx], -residual)
            h[idx] -= sub
            residual += sub
            if -residual < 1e-14:
                break
    return h

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return corr

def compute_grad_lag(h, dx, k):
    """Gradient of f_k * dx where f_k = sum h[n+k]*(1-h[n])"""
    n = len(h)
    grad = np.zeros(n)
    if k >= 0:
        grad[k:] += dx * (1.0 - h[:n-k])
        grad[:n-k] -= dx * h[k:]
    else:
        r = -k
        grad[:n-r] += dx * (1.0 - h[r:])
        grad[r:] -= dx * h[:n-r]
    return grad

def compute_grad_weighted(h, dx, lags, weights):
    """Weighted gradient of sum of weighted f_k"""
    n = len(h)
    grad = np.zeros(n)
    for k, w in zip(lags, weights):
        if k >= 0:
            grad[k:] += w * dx * (1.0 - h[:n-k])
            grad[:n-k] -= w * dx * h[k:]
        else:
            r = -k
            grad[:n-r] += w * dx * (1.0 - h[r:])
            grad[r:] -= w * dx * h[:n-r]
    return grad

def upsample(h_old, n_new):
    """Upsample h from n_old to n_new using linear interpolation, then project"""
    n_old = len(h_old)
    target_sum_new = 0.5 * n_new
    target_sum_old = 0.5 * n_old
    # Create interpolation
    x_old = np.linspace(0, 2, n_old + 1)  # cell centers? or edges?
    # Actually, for step functions, the samples represent values at cell centers
    # x_i = (i + 0.5) * dx_old
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_interp = np.interp(x_new, x_old, h_old)
    h_new = project_feasible(h_interp, target_sum_new)
    return h_new

def projected_gradient_descent(h, max_iter=500, time_limit=60.0, n_active=40):
    """Project gradient descent on smoothed max of correlations"""
    n = len(h)
    dx = 2.0 / n
    target_sum = 0.5 * n
    h = h.astype(float).copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    tau = max(1e-3, 0.003 * best_c5)
    eta = 0.01
    start_time = time.time()
    no_improve = 0
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        if time.time() - start_time > 0:
            pass  # just for timing
        
        # Compute correlations
        corr = compute_correlations(h)
        c5_current = float(np.max(corr))
        
        if c5_current < best_c5 - 1e-12:
            best_c5 = c5_current
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        # Select active lags
        n_lags = len(corr)
        if n_active < n_lags:
            top_idx = np.argsort(corr)[::-1][:n_active]
            active_vals = corr[top_idx]
        else:
            top_idx = np.arange(n_lags)
            active_vals = corr
        
        # Compute weights (softmax)
        max_val = np.max(active_vals)
        exponents = (active_vals - max_val) / tau
        np.clip(exponents, -50, 50, out=exponents)
        weights = np.exp(exponents)
        wsum = np.sum(weights)
        weights = weights / wsum
        
        # Compute gradient
        grad = compute_grad_weighted(h, dx, top_idx, weights)
        
        # Step
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        h_trial = project_feasible(h - eta * grad / gnorm * max(c5_current, 0.1), target_sum)
        
        # Evaluate trial
        c5_trial = evaluate_c5(h_trial)
        if c5_trial < c5_current - 1e-10:
            h = h_trial
            no_improve = 0
        else:
            # Try smaller step
            h_trial = project_feasible(h - eta * 0.5 * grad / gnorm * max(c5_current, 0.1), target_sum)
            c5_trial = evaluate_c5(h_trial)
            if c5_trial < c5_current - 1e-10:
                h = h_trial
                eta *= 0.7
            else:
                eta *= 0.5
                no_improve += 1
        
        # Anneal tau
        if no_improve > 50:
            tau *= 1.5
            eta *= 0.5
            no_improve = 0
        elif it % 50 == 0 and it > 0:
            tau *= 0.8
        
        if eta < 1e-6:
            break
    
    return best_h, best_c5

def slsqp_polish(h, max_iter=30, n_active=30, time_limit=30.0):
    """SLSQP polish with active lag constraints"""
    n = len(h)
    dx = 2.0 / n
    target_sum = 0.5 * n
    h0 = h.astype(float).copy()
    
    # Get active lags
    corr = compute_correlations(h0)
    c5_0 = float(np.max(corr))
    top_idx = np.argsort(corr)[::-1][:n_active]
    top_idx_sorted = sorted(top_idx)
    
    start_time = time.time()
    
    best_h = h0.copy()
    best_c5 = c5_0
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    # Convert to lag values (k = idx - (n-1))
    lag_values = top_idx_sorted - (n - 1)
    
    def constraint_f(x):
        h_x = x[:n]
        # Compute f_k for active lags
        corr_x = np.correlate(h_x, 1.0 - h_x, mode="full") * dx
        return corr_x[top_idx_sorted] - x[-1]
    
    def constraint_jac(x):
        h_x = x[:n]
        n_active = len(lag_values)
        jac = np.zeros((n_active, n + 1))
        for i, k in enumerate(lag_values):
            if k >= 0:
                jac[i, k:] += dx * (1.0 - h_x[:n-k])
                jac[i, :n-k] -= dx * h_x[k:]
            else:
                r = -k
                jac[i, :n-r] += dx * (1.0 - h_x[r:])
                jac[i, r:] -= dx * h_x[:n-r]
            # t column
            jac[i, n] = -1.0
        return jac
    
    # Equality constraint: sum(h) = target_sum
    def eq_constraint(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, c5_0 + 0.01)]
    
    cons = [
        {'type': 'ineq', 'fun': constraint_f, 'jac': constraint_jac},
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jac}
    ]
    
    x0 = np.append(h0, c5_0 + 0.001)
    
    try:
        result = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                         bounds=bounds, constraints=cons,
                         options={'maxiter': max_iter, 'ftol': 1e-8})

        z = (vals - vmax) / tau
        z = np.clip(z, -100, 100)
        w = np.exp(z)
        wsum = np.sum(w)
        if wsum < 1e-30:
            w = np.ones(K) / K
        else:
            w /= wsum
        grad = compute_active_grad(h, dx, top_idx, w)
        gn = np.linalg.norm(grad)
        if gn < 1e-14:
            break
        cur_c = float(np.max(corr))
        accepted = False
        for shrink in range(10):
            step = eta_base * (0.5 ** shrink)
            h_trial = project_feasible(h - step * grad / gn, target_sum)
            c_trial = eval_c5(h_trial)
            if c_trial < cur_c - 1e-12:
                h = h_trial
                corr = get_corr(h)
                if c_trial < best_c:
                    best_c = c_trial
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                accepted = True
                break
        if not accepted:
            eta_base *= 0.5
            stagnant += 1
            if eta_base < 1e-8:
                break
        if stagnant > 80:
            tau = min(tau * 2.0, 0.05)
            stagnant = 0
        elif it % 100 == 50:
            tau = max(tau_init, tau * 0.85)
    return best_h, best_c

def slsqp_optimize(h, target_sum, n_active=25, max_iter=50, time_end=None):
    n = len(h)
    dx = 2.0 / n
    h0 = h.astype(np.float64).copy()
    c5_0 = eval_c5(h0)
    corr = get_corr(h0)
    top_idx = np.argsort(corr)[::-1][:n_active]
    top_idx = np.sort(top_idx)
    lag_vals = top_idx - (n - 1)
    n_act = len(lag_vals)
    
    def obj(x):
        return x[n]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g
    
    def ineq_fun(x):
        h_x = x[:n]
        c = np.correlate(h_x, 1.0 - h_x, mode='full') * dx
        return c[top_idx] - x[n]
    
    def ineq_jac(x):
        h_x = x[:n]
        J = np.zeros((n_act, n + 1))
        for i, k in enumerate(lag_vals):
            J[i, :n] = grad_lag_vector(h_x, dx, k)
            J[i, n] = -1.0
        return J
    
    def eq_fun(x):
        return np.sum(x[:n]) - target_sum
    
    def eq_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, c5_0 + 0.005)]
    cons = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    x0 = np.concatenate([h0, [c5_0 + 0.001]])
    best_h = h0.copy()
    best_c = c5_0
    try:
        res = minimize(obj, x0, method='SLSQP', jac=obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': max_iter, 'ftol': 1e-10})
        if time_end is not None and time.time() > time_end:
            return best_h, best_c
        if hasattr(res, 'x') and res.x is not None:
            h_r = project_feasible(res.x[:n], target_sum)
            c_r = eval_c5(h_r)
            if c_r < best_c:
                best_h = h_r
                best_c = c_r
    except Exception:
        pass
    return best_h, best_c

def local_search(h, target_sum, n_iters=2000, time_end=None, delta=0.02):
    n = len(h)
    h = h.astype(np.float64).copy()
    c = eval_c5(h)
    rng = np.random.default_rng(12345)
    for it in range(n_iters):
        if time_end is not None and time.time() > time_end:
            break
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        d = delta * (1.0 if rng.random() < 0.5 else -1.0)
        h_trial = h.copy()
        h_trial[i] = np.clip(h[i] + d, 0.0, 1.0)
        h_trial[j] = np.clip(h[j] - d, 0.0, 1.0)
        h_trial = project_feasible(h_trial, target_sum)
        c_trial = eval_c5(h_trial)
        if c_trial < c - 1e-12:
            h = h_trial
            c = c_trial
    return h, c

def upsample(h_old, n_new):
    n_old = len(h_old)
    if n_new == n_old:
        return h_old.copy()
    dx_old = 2.0 / n_old
    x_old = (np.arange(n_old) + 0.5) * dx_old
    dx_new = 2.0 / n_new
    x_new = (np.arange(n_new) + 0.5) * dx_new
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, 0.5 * n_new)

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t_start = time.time()
    hard_deadline = t_start + budget_s * 0.90
    
    # Get initial values
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).flatten()
    except Exception:
        h_init = np.full(64, 0.5, dtype=np.float64)
    
    n_init = len(h_init)
    h_init = project_feasible(h_init, 0.5 * n_init)
    c5_init = eval_c5(h_init)
    
    best_h = h_init.copy()
    best_c5 = c5_init
    
    # Try transforms
    for t in [h_init[::-1], 1.0 - h_init, 1.0 - h_init[::-1]]:
        t = project_feasible(t, 0.5 * n_init)
        c_t = eval_c5(t)
        if c_t < best_c5:
            best_c5 = c_t
            best_h = t.copy()
    
    # Optimization at current resolution
    curr_h = best_h.copy()
    curr_n = n_init
    target_sum = 0.5 * curr_n
    
    # PGD round 1
    t_end = min(hard_deadline, time.time() + budget_s * 0.15)
    curr_h, c_curr = pgd_optimize(curr_h, target_sum, max_iter=3000,
                                   time_end=t_end, n_active=50)
    if c_curr < best_c5:
        best_c5 = c_curr
        best_h = curr_h.copy()
    
    # SLSQP
    if time.time() < hard_deadline - 30:
        t_end_sl = min(hard_deadline, time.time() + budget_s * 0.10)
        curr_h, c_sl = slsqp_optimize(curr_h, target_sum, n_active=30,
                                       max_iter=60, time_end=t_end_sl)
        if c_sl < best_c5:
            best_c5 = c_sl
            best_h = curr_h.copy()
    
    # PGD round 2
    if time.time() < hard_deadline - 30:
        t_end2 = min(hard_deadline, time.time() + budget_s * 0.10)
        curr_h, c_curr2 = pgd_optimize(best_h, 0.5 * len(best_h),
                                        max_iter=2000, time_end=t_end2, n_active=60)
        if c_curr2 < best_c5:
            best_c5 = c_curr2
            best_h = curr_h.copy()
    
    # Local search
    if time.time() < hard_deadline - 20:
        t_end_ls = min(hard_deadline, time.time() + budget_s * 0.05)
        curr_h, c_ls = local_search(best_h, 0.5 * len(best_h),
                                     n_iters=5000, time_end=t_end_ls, delta=0.01)
        if c_ls < best_c5:
            best_c5 = c_ls
            best_h = curr_h.copy()
    
    # Try higher resolutions
    resolutions = [96, 128, 160, 192, 256]
    for n_res in resolutions:
        if time.time() > hard_deadline - 60:
            break
        if n_res <= len(best_h):
            continue
        
        # Upscale
        h_up = upsample(best_h, n_res)
        ts = 0.5 * n_res
        c_up = eval_c5(h_up)
        
        # PGD at new resolution
        t_end_pg = min(hard_deadline - 30, time.time() + max(20, budget_s * 0.08))
        h_pg, c_pg = pgd_optimize(h_up, ts, max_iter=2000,
                                   time_end=t_end_pg, n_active=50)
        
        # SLSQP at new resolution
        h_final_r = h_pg
        c_final_r = c_pg
        if time.time() < hard_deadline - 40:
            t_end_sl = min(hard_deadline - 20, time.time() + max(15, budget_s * 0.06))
            h_sl, c_sl = slsqp_optimize(h_final_r, ts, n_active=30,
                                         max_iter=50, time_end=t_end_sl)
            if c_sl < c_final_r:
                h_final_r = h_sl
                c_final_r = c_sl
        
        # Another PGD round
        if time.time() < hard_deadline - 30:
            t_end_pg2 = min(hard_deadline - 20, time.time() + max(10, budget_s * 0.05))
            h_pg2, c_pg2 = pgd_optimize(h_final_r, ts, max_iter=1000,
                                         time_end=t_end_pg2, n_active=60)
            if c_pg2 < c_final_r:
                h_final_r = h_pg2
                c_final_r = c_pg2
        
        if c_final_r < best_c5:
            best_c5 = c_final_r
            best_h = h_final_r.copy()
    
    # Final local search on best
    if time.time() < hard_deadline - 10:
        ts = 0.5 * len(best_h)
        t_end = min(hard_deadline, time.time() + budget_s * 0.03)
        h_ls, c_ls = local_search(best_h, ts, n_iters=3000,
                                   time_end=t_end, delta=0.005)
        if c_ls < best_c5:
            best_c5 = c_ls
            best_h = h_ls.copy()
    
    # Final verification
    best_h = project_feasible(best_h, 0.5 * len(best_h))
    final_c5 = eval_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Ensure clean output
    best_h = np.clip(best_h, 0.0, 1.0)
    if abs(np.sum(best_h) - 0.5 * len(best_h)) > 1e-10:
        best_h = project_feasible(best_h, 0.5 * len(best_h))
    
    return best_h, float(best_c5), int(len(best_h))
