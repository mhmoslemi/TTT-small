import numpy as np
import scipy.optimize as opt
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def ensure_feasible(h, S):
    n = len(h)
    h = np.array(h, dtype=float).copy()
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -3.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > S:
            lo = mid
        else:
            hi = mid
    mu = (lo + hi) / 2.0
    result = np.clip(h - mu, 0.0, 1.0)
    diff = S - np.sum(result)
    if abs(diff) > 1e-12:
        interior = (result > 1e-8) & (result < 1.0 - 1e-8)
        n_int = int(np.sum(interior))
        if n_int > 0:
            result[interior] += diff / n_int
            result = np.clip(result, 0.0, 1.0)
    return result

def smooth_obj_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx
    c_max = np.max(c)
    val = c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))
    exp_vals = np.exp((c - c_max) / tau)
    weights = exp_vals / np.sum(exp_vals)
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w = weights[m] * dx
        if k >= 0:
            grad[:n-k] += w * (1.0 - h[k:])
            grad[k:] -= w * h[:n-k]
        else:
            kk = -k
            grad[kk:] += w * (1.0 - h[:n-kk])
            grad[:n-kk] -= w * h[kk:]
    return val
