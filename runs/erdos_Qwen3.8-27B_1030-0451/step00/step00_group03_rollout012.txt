```python
def project_to_feasible(y, n):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}."""
    target = 0.5 * n
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    # Bisection for alpha
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-13:
        if residual > 0:
            room = 1.0 - h
            for i in np.argsort(-room):
                step = min(room[i], residual)
                h[i] += step
                residual -= step
                if residual <= 1e-14:
                    break
        else:
            for i in np.argsort(h):
                step = min(h[i], -residual)
                h[i] -= step
                residual += step
                if residual >= -1e-14:
                    break
    return h
```

```python
def lse_value_and_grad(h, tau, active_lags):
    n = len(h)
    dx = 2.0 / n
    corr = get_correlations(h)
    corr_sel = corr[active_lags]
    fmax = np.max(corr_sel)
    z = np.clip((corr_sel - fmax) / tau, -50.0, 50.0)
    exp_z = np.exp(z)
    norm = np.sum(exp_z)
    weights = exp_z / norm
    lse_val = fmax + tau * np.log(norm)
    
    grad = np.zeros(n)
    for idx, m in enumerate(active_lags):
        w = weights[idx]
        if w < 1e-12:
            continue

            grad[k:] += w * (1.0 - h[:n-k])
            grad[:n-k] += w * (-h[k:])
        else:
            r = -k
            grad[:n-r] += w * (1.0 - h[r:])
            grad[r:] += w * (-h[:n-r])
    grad *= dx
    return lse_val, grad, fmax
```

Now for SLSQP. The formulation is:
minimize t
subject to:
  f_m(h) - t ≤ 0 for m in active_set
  sum(h) - n/2 = 0
  0 ≤ h_i ≤ 1

Variables: x = [h_0, h_1, ..., h_{n-1}, t]

Objective: x[-1] (i.e., t)
Jacobian of objective: [0, 0, ..., 0, 1]

Inequality constraints (f_m - t ≤ 0):
  value: f_m(x[:n]) - x[-1]
  Jacobian: [∂f_m/∂h_0, ..., ∂f_m/∂h_{n-1}, -1]

Equality constraint (sum(h) - n/2 = 0):
  value: sum(x[:n]) - n/2
  Jacobian: [1, 1, ..., 1, 0]

Bounds: 0 ≤ x[i] ≤ 1 for i < n, t is free (or we can bound it)

Let me code this:

```python
def slsqp_polish(h_start, n, deadline, active_lags):
    target = 0.5 * n
    dx = 2.0 / n
    
    # Get active set
    corr = get_correlations(h_start)
    fmax = np.max(corr)
    K = min(len(corr), max(30, n // 3))
    if active_lags is None:
        active_lags = np.sort(np.argsort(corr)[::-1][:K])
    
    c0 = evaluate_c5(h_start)
    
    def objective(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def make_ineq(m):
        k = m - (n - 1)
        def fun(x):
            h = x[:n]
            return f_m_value(h, k, n, dx) - x[-1]
        def jac(x):
            h = x[:n]
            g = np.zeros(n + 1)
            # df_m/dh
            if k >= 0:
                g[k:n] += (1.0 - h[0:n-k])
                g[0:n-k] += (-h[k:n])
            else:
                r = -k
                g[0:n-r] += (1.0 - h[r:n])
                g[r:n] += (-h[0:n-r])
            g[:n] *= dx
            g[-1] = -1.0
            return g
        return {'type': 'ineq', 'fun': fun, 'jac': jac}
    
    def eq_con_fun(x):
        return np.sum(x[:n]) - target
    
    def eq_con_jac(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    # Build constraint list
    constraints = []
    for m in active_lags:
        constraints.append(make_ineq(m))
    constraints.append({'type': 'eq', 'fun': eq_con_fun, 'jac': eq_con_jac})
```
