def mass_project(x, target):
    """Project x onto {0 <= h_i <= 1, sum(h) = target}"""
    x = np.array(x, dtype=np.float64)
    N = len(x)
    # Find lambda such that sum(clip(x_i + lambda, 0, 1)) = target
    # F(lambda) = sum(clip(x_i + lambda, 0, 1)) is continuous and nondecreasing in lambda
    lo = -1.0 - np.max(x)  # F(lo) = 0
    hi = 1.0 - np.min(x)   # F(hi) = N
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2
        val = np.sum(np.clip(x + mid, 0, 1))
        if val < target:
            lo = mid
        else:
            hi = mid

    # Fine-tune for exact mass
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual to elements not at bounds
        if residual > 0:
            idx = np.where(result < 1.0)[0]
        else:
            idx = np.where(result > 0.0)[0]
        per = residual / len(idx) if len(idx) > 0 else 0
        result[idx] = np.clip(result[idx] + per, 0, 1)
    
    return result

def smooth_objective_and_grad(h, beta=50.0):
    """Smooth approximation of max using log-sum-exp, returns (value, gradient)"""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full') * dx  # shape (2N-1,)
    
    # Log-sum-exp for numerical stability
    max_corr = np.max(corr)
    exp_corr = np.exp(beta * (corr - max_corr))
    sum_exp = np.sum(exp_corr)
    value = max_corr + np.log(sum_exp) / beta

    weights = exp_corr / sum_exp  # shape (2N-1,)

    grad = np.zeros(N)
    for k in range(2 * N - 1):
        if weights[k] < 1e-10:
            continue
        grad += weights[k] * gradient_overlap(h, k) * dx
    
    return value, grad

def optimize_smooth(h0, beta, max_iter=5000, tol=1e-12):
    """Projected gradient descent with backtracking line search."""
    h = h0.copy()
    N = len(h)
    target = N / 2.0
    best_val = evaluate_exact(h)
    best_h = h.copy()
    
    lr = 0.1  # initial learning rate
    prev_val = smooth_value_grad(h, beta)[0]
    
    for it in range(max_iter):
        val, grad = smooth_value_grad(h, beta)
        
        # Check convergence
        if it > 0 and abs(prev_val - val) < tol:
            break
        prev_val = val
        
        # Gradient step
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = mass_project(h_new, target)
        
        # Backtracking line search
        new_val = evaluate_exact(h_new)
        while new_val > val + 1e-15 and lr > 1e-10:
            lr *= 0.5
            h_new = h - lr * grad
            h_new = mass_project(h_new, target)
            new_val = evaluate_exact(h_new)
        
        if new_val < best_val - 1e-15:
            best_val = new_val
            best_h = h_new.copy()
        
        h = h_new
        if abs(lr - 0.1) < 0.01:  # if lr didn't decrease much, try increasing
            lr = min(lr * 1.1, 0.5)
    
    return best_h, best_val

def transfer(h_old, N_new):
    """Transfer a solution from N_old points to N_new points by averaging."""
    N_old = len(h_old)
    dx_old = 2.0 / N_old
    dx_new = 2.0 / N_new

```python
def transfer(h_old, N_new):
    N_old = len(h_old)
    # Use cumulative sum for efficient area computation
    # Old grid: cell j covers [2j/N_old, 2(j+1)/N_old)
    # New grid: cell i covers [2i/N_new, 2(i+1)/N_new)
    # Area of new cell i = integral of step function over [2i/N_new, 2(i+1)/N_new)
    
    # Precompute cumulative areas of old function
    # cumsum[j] = sum_{k=0}^{j-1} h_old[k] * (2/N_old) for j=0,...,N_old
    dx_old = 2.0 / N_old
    cum = np.zeros(N_old + 1)
    cum[1:] = np.cumsum(h_old * dx_old)
    
    dx_new = 2.0 / N_new
    h_new = np.zeros(N_new)
    
    for i in range(N_new):
        x_start = i * dx_new
        x_end = (i + 1) * dx_new
        # Area = integral from x_start to x_end of step function
        # = sum of h_old[j] * dx_old for old cells j that overlap [x_start, x_end]
        # Find which old cells overlap
        j_start = int(np.floor(x_start / dx_old))
        j_end = int(np.ceil(x_end / dx_old)) - 1
        j_start = max(0, j_start)
        j_end = min(N_old - 1, j_end)
        
        # For the boundary cells, we need partial overlaps
        area = 0.0
        for j in range(j_start, j_end + 1):
            ox_start = max(x_start, j * dx_old)
            ox_end = min(x_end, (j + 1) * dx_old)
            area += h_old[j] * (ox_end - ox_start)
        h_new[i] = area / dx_new
    
    return mass_project(h_new, N_new / 2.0)
```

Actually, this loop over N_new * (average overlap) is fine for reasonable N values. For N_old=120 and N_new=240, each new cell overlaps at most 2 old cells, so it's 240*2 = 480 operations. Very fast.
