2. **Evaluation**: np.max(dx * np.correlate(h, 1-h, mode="full"))

3. **Gradient of individual lag f_m**: For lag k = m - (n-1):
   - f_m = dx * sum_j h[j+k] * (1-h[j]) over valid j
   - ∂f_m/∂h[p] = dx * (contribution of p in the product)

   - For p in [0, n-1-k]: appears as (1-h[p]) in term j=p → contribution -dx*h[p+k]
   - For p in [k+k, n-1] (i.e., p >= k): appears as h[p] in term j=p-k → contribution dx*(1-h[p-k])
   
   If k < 0, let r = -k: f_m = dx * sum_{j=r}^{n-1} h[j](1-h[j-r])

   - For p in [0, n-1-r]: appears as (1-h[p]) in term j=p+r → contribution -dx*h[p+r]

import time

def project_feasible(y, n):
    """Project y onto {h: 0<=h_i<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on alpha
    lo, hi = -np.max(y) - 2, np.max(y) + 2 - target
    # Actually lo should make all 0, hi should make all 1
    lo = 0 - np.max(y) - 1e-10  # all clip to 0
    hi = 1 - np.min(y) + 1e-10   # all clip to 1
    
    target_sum = n / 2.0
    
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(y + mid, 0, 1).sum()
        if s > target_sum:
            hi = mid
        else:
            lo = mid
    
    alpha = (lo + hi) / 2
    h = np.clip(y + alpha, 0, 1)
    
    # Fix residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-10:
        # Distribute residual
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                step = min(room[i], residual)
                h[i] += step
                residual -= step
                if abs(residual) < 1e-14:

            for i in order:
                step = min(h[i], residual)
                h[i] -= step
                residual -= step
                if abs(residual) < 1e-14:
                    break
    return h

def evaluate_c5(h):
    """Compute C5 = max(k) dx * correlate(h, 1-h, full)[k]"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    """Compute full correlation vector"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def compute_lag_gradient(h, k, n):
    """Compute gradient of f_m for lag k (where m = k + n - 1)"""
    dx = 2.0 / n
    grad = np.zeros(n)
    if k >= 0:
        L = n - k
        # f_m = dx * sum_{j=0}^{L-1} h[j+k] * (1-h[j])
        # d/dh[p]: for p in [0, L-1]: -dx * h[p+k]
        #          for p in [k, n-1]:  dx * (1-h[p-k])
        grad[:L] += -dx * h[k:k+L]
        grad[k:] += dx * (1.0 - h[:L])
    else:
        r = -k
        L = n - r
        # f_m = dx * sum_{j=r}^{n-1} h[j] * (1-h[j-r])
        # d/dh[p]: for p in [r, n-1]: dx * (1-h[p-r])
        #          for p in [0, L-1]: -dx * h[p+r]
        grad[r:] += dx * (1.0 - h[:L])
        grad[:L] += -dx * h[r:]
    return grad

        return x[-1]
