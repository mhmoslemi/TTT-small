    n = len(h)
    m = n // 2
    return h[:m].copy()

def c5_exact(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')
    return dx * np.max(R)

def c5_smoothed(p, beta):
    """Compute smoothed C5 objective."""
    h = build_h(p)
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode='full')

            p[i] = best_val
            if abs(best_val - base[i]) > tol:
                improved = True
            p[i] = best_val
        h = build_h(p)
        new_c5 = c5_from_h(h)
        if new_c5 < cur_c5:
            cur_c5 = new_c5
            improved = True
        else:
            break
    h = build_h(p)
    return p, c5_from_h(h)

def interpolate_parent(parent, new_n):
    old_n = len(parent)
    old_x = np.linspace(0, 2.0, old_n, endpoint=False) + 1.0 / old_n
    new_x = np.linspace(0, 2.0, new_n, endpoint=False) + 1.0 / new_n
    interp = np.interp(new_x, old_x, parent)
    return interp

def lsbfgs_optimize(p0, beta, maxiter=200, gtol=1e-10, time_limit=300):
    p0 = np.clip(p0, 1e-10, 1.0 - 1e-10)
    t_start = time.perf_counter()
    
    def fun(x):
        v, g = obj_grad(x, beta)
        if time.perf_counter() - t_start > time_limit:
            raise RuntimeError("time limit")
        return v
    
    def jac(x):
        v, g = obj_grad(x, beta)
        return g
    
    try:
        res = minimize(fun, p0, jac=jac, method='L-BFGS-B',
                       bounds=[(1e-10, 1.0 - 1e-10)] * len(p0),
                       options={'maxiter': maxiter, 'gtol': gtol, 'maxfun': maxiter * 3})
    except RuntimeError:
        pass
    except Exception:
        res = None
    
    if res is not None and res.x is not None:
        p_res = np.clip(res.x, 0.0, 1.0)
    else:
        p_res = p0.copy()
    return p_res

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.perf_counter()
    np.random.seed(seed)
    
    def elapsed():
        return time.perf_counter() - t_start
    
    def time_left():
        return budget_s - elapsed()
    
    # Get parent
    try:
        parent = initial_h_values.copy()
    except NameError:
        parent = None
    
    best_h = None
    best_c5 = np.inf
    
    if parent is not None and len(parent) > 0:
        n_par = len(parent)
        parent = np.clip(parent, 0.0, 1.0)
        c5_par = c5_from_h(parent)
        if np.isfinite(c5_par):
            best_c5 = c5_par
            best_h = parent.copy()
    
    # Antisymmetric optimization at parent resolution
    if parent is not None and time_left() > 60:
        n0 = len(parent)
        if n0 % 2 == 0:
            m0 = n0 // 2
            # Project to antisymmetric
            h_anti = np.empty(n0)
            h_anti[:m0] = 0.5 * (parent[:m0] + 1.0 - parent[m0-1::-1][::-1])
            # Actually: anti[i] = 0.5*(h[i] + 1-h[n-1-i]) for i < m
            # h_anti[i] = 0.5*(parent[i] + 1 - parent[n0-1-i])
            p0 = np.empty(m0)
            for i in range(m0):
                p0[i] = 0.5 * (parent[i] + 1.0 - parent[n0 - 1 - i])
            p0 = np.clip(p0, 0.0, 1.0)
            h_anti = build_h(p0)
            c5_anti = c5_from_h(h_anti)
            
            if c5_anti <= best_c5 + 0.01:
                # L-BFGS with increasing beta
                for beta in [50.0, 200.0, 500.0, 1000.0]:
                    if time_left() < 30:
                        break
                    tl = min(120, time_left() - 20)
                    p_new = lsbfgs_optimize(p0, beta, maxiter=150, time_limit=tl)
                    h_new = build_h(p_new)
                    c5_new = c5_from_h(h_new)
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    p0 = p_new
                
                # Coordinate descent polish
                if time_left() > 30:
                    p_cd, c5_cd = coord_descent(p0, max_rounds=30)
                    h_cd = build_h(p_cd)
                    c5_cd_final = c5_from_h(h_cd)
                    if c5_cd_final < best_c5:
                        best_c5 = c5_cd_final
                        best_h = h_cd.copy()
        
        # Also try direct parent without antisymmetric constraint
        # by clipping and checking
        if time_left() > 50:
            # Try perturbing the parent
            for sigma in [0.01, 0.05, 0.1]:
                if time_left() < 20:
                    break
                p_pert = np.clip(parent[:len(parent)//2] + np.random.randn(len(parent)//2) * sigma, 0, 1)
                if len(parent) % 2 == 0:
                    h_pert = build_h(p_pert)
                    c5_pert = c5_from_h(h_pert)
                    if c5_pert < best_c5 and time_left() > 30:
                        p_new = lsbfgs_optimize(p_pert, 500.0, maxiter=100, time_limit=min(60, time_left()-10))
                        h_new = build_h(p_new)
                        c5_new = c5_from_h(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
    
    # Try other resolutions
    if parent is not None and time_left() > 80:
        for new_n in [320, 340, 360, 400]:
            if time_left() < 40:
                break
            if new_n % 2 != 0:
                new_n += 1
            n_new = new_n
            m_new = n_new // 2
            h_interp = interpolate_parent(parent, n_new)
            p_new = np.empty(m_new)
            for i in range(m_new):
                p_new[i] = 0.5 * (h_interp[i] + 1.0 - h_interp[n_new - 1 - i])
            p_new = np.clip(p_new, 0.0, 1.0)
            h_check = build_h(p_new)
            c5_check = c5_from_h(h_check)
            
            if c5_check < best_c5 + 0.02 and time_left() > 40:
                for beta in [200.0, 800.0]:
                    if time_left() < 20:
                        break
                    tl = min(80, time_left() - 15)
                    p_opt = lsbfgs_optimize(p_new, beta, maxiter=100, time_limit=tl)
                    h_opt = build_h(p_opt)
                    c5_opt = c5_from_h(h_opt)
                    if c5_opt < best_c5:
                        best_c5 = c5_opt
                        best_h = h_opt.copy()
                        p_new = p_opt
                # Quick coordinate descent
                if time_left() > 20:
                    p_cd, c5_cd = coord_descent(p_new, max_rounds=15)
                    h_cd = build_h(p_cd)
                    c5_cd_f = c5_from_h(h_cd)
                    if c5_cd_f < best_c5:
                        best_c5 = c5_cd_f
                        best_h = h_cd.copy()
    
    # Final safety: ensure best_h is valid
    if best_h is None:
        # Fallback: create a simple valid solution
        n_fb = 200
        m_fb = n_fb // 2
        p_fb = np.full(m_fb, 0.5)
        best_h = build_h(p_fb)
        best_c5 = c5_from_h(best_h)
    
    # Final evaluation
    final_c5 = c5_from_h(best_h)
    return best_h, final_c5, len(best_h)
