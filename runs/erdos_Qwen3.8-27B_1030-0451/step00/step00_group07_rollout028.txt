def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    """Project x onto {h: 0<=h_i<=1, sum(h_i)=target_sum}"""
    x = np.asarray(x, dtype=np.float64)
    n = len(x)
    # Bisect on lambda: sum(clip(x_i - lambda, 0, 1)) = target_sum
    lo, hi = float(np.max(x)) - target_sum - 2.0, float(np.min(x)) - target_sum + 2.0
    # Ensure f(lo) >= target_sum >= f(hi)
    # f(lambda) = sum(clip(x - lambda, 0, 1))
    # f is decreasing in lambda
    # At lambda very negative: clip(x - lambda, 0, 1) -> 1 for all, sum = n
    # At lambda very positive: clip(x - lambda, 0, 1) -> 0 for all, sum = 0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix residual
    residual = h.sum() - target_sum
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to decrease some values
            order = np.argsort(h)[::-1]  # largest first
            for idx in order:
                if residual <= 0:
                    break
                dec = min(residual, h[idx])
                h[idx] -= dec
                residual -= dec
        else:
            # Need to increase some values
            order = np.argsort(h)  # smallest first
            for idx in order:
                if residual >= 0:
                    break
                inc = min(-residual, 1.0 - h[idx])
                h[idx] += inc
                residual += inc
    return h

def compute_subgradient(h, corr, active_indices, n):
    """Compute subgradient of max over active correlation values."""
    dx = 2.0 / n
    grad = np.zeros(n)
    for k in active_indices:
        # k is index into correlation output, shift s = k - (n-1)
        s = k - (n - 1)
        if s == 0:
            grad += dx * (1.0 - 2.0 * h)
        elif s > 0:

            # d/dh[j]: if j in [s, n-1]: (1-h[j-s]); if j in [0, n-1-s]: -h[j+s]
            for j in range(s, n):
                grad[j] += dx * (1.0 - h[j - s])
            for j in range(0, n - s):

            for j in range(0, n - t):
                grad[j] += dx * (1.0 - h[j + t])
            for j in range(t, n):
                grad[j] += -dx * h[j - t]
    return grad / len(active_indices)

import numpy as np
import time
from scipy.optimize import linprog

def evaluate_exact(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))

def correlation_array(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_feasible(x, target_sum):
    x = np.asarray(x, dtype=np.float64).copy()
    n = len(x)
    if n == 0:
        return x
    # Bisect on lambda
    lo = float(np.min(x)) - target_sum / n - 2.0
    hi = float(np.max(x)) - target_sum / n + 2.0
    # Ensure f(lo) >= target >= f(hi) where f(lam) = sum(clip(x-lam,0,1))
    for _ in range(80):
        mid = 0.5*(lo+hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5*(lo+hi)
    h = np.clip(x - lam, 0.0, 1.0)
    # Fix residual
    for _ in range(3):
        res = h.sum() - target_sum
        if abs(res) < 1e-12:
            break
        if res > 0:
            order = np.argsort(h)[::-1]
            for idx in order:
                if res <= 0: break
                dec = min(res, h[idx])
                h[idx] -= dec
                res -= dec
        else:
            order = np.argsort(h)
            for idx in order:
                if res >= 0: break
                inc = min(-res, 1.0-h[idx])
                h[idx] += inc
                res += inc
    return h

    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -corr
    lower = np.maximum(-h, -trust)
    upper = np.minimum(1.0-h, trust)
    A_eq = np.zeros((1, n+1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    bounds = list(zip(lower, upper)) + [(None, None)]
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')

        # Verify feasibility
        if np.all(h + d >= -1e-10) and np.all(h + d <= 1+1e-10) and abs(d.sum()) < 1e-8:

        return None
    return None

def upscale(h, factor):
    """Upscale by repeating each element 'factor' times."""
    return np.repeat(h, factor)

def downscale(h, factor):
    """Downscale by averaging groups of 'factor' elements."""
    n = len(h)
    assert n % factor == 0
    return h.reshape(n//factor, factor).mean(axis=1)

def random_feasible(n, rng):

            elif restart == 2:

                from scipy.ndimage import uniform_filter1d
                smoothed = uniform_filter1d(best_h, size=5)
                h_try = project_feasible(smoothed + 0.05 * rng.standard_normal(n), target)
            elif restart == 3:
                # Binary-like perturbation
                binary = (best_h > 0.5).astype(float)
                binary = project_feasible(binary, target)
                h_try = project_feasible(0.7 * best_h + 0.3 * binary, target)
            else:
                h_try = project_feasible(best_h + 0.2 * rng.standard_normal(n), target)

            h_try, v_try = optimize_local(h_try, best_val, remaining, rng, use_active=False)
            if v_try < best_val:
                best_val = v_try
                best_h = h_try.copy()

    # Phase 5: Fine-tuning with smaller trust on best solution
    if remaining() > 15:
        h = best_h.copy()
        n = len(h)
        target = n / 2.0
        trust = 0.05
        for _ in range(50):
            if remaining() < 5:
                break
            corr = correlation_array(h)
            grads = compute_gradients(h)
            d = solve_lp_active(h, corr, grads, trust, 0.5)
            if d is not None:
                for alpha in [1.0, 0.5, 0.2]:
                    h_new = h + alpha * d
                    if np.any(h_new < -1e-9) or np.any(h_new > 1 + 1e-9):
                        continue
                    val = evaluate_exact(h_new)
                    if val < best_val - 1e-15:
                        best_val = val
                        best_h = h_new.copy()
                        h = h_new.copy()
                        break
            trust = min(0.1, trust * 1.1)

    # Final: ensure feasibility and compute exact value
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    best_val = evaluate_exact(best_h)

    return best_h, float(best_val), len(best_h)
