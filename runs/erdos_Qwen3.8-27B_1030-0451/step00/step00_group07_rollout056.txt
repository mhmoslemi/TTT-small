def compute_smoothed_objective_and_gradient(h, tau, dx):
    N = len(h)
    C = np.correlate(h, 1-h, mode="full")  # length 2N-1
    m = np.max(C)
    # softmax weights
    shifted = (C - m) / tau
    # clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    w = exp_shifted / np.sum(exp_shifted)
    
    S = (m + tau * np.log(np.sum(exp_shifted))) * dx
    
    # Gradient
    g = np.zeros(N)
    for r in range(2*N - 1):
        if w[r] < 1e-15:
            continue
        d = r - N + 1
        if d >= 0:
            # First term: (1-h[j-d]) for j in [d, N)
            js = np.arange(d, N)
            g[js] += w[r] * (1 - h[js - d])
            # Second term: -h[j+d] for j in [0, N-d)
            js2 = np.arange(0, N - d)
            g[js2] -= w[r] * h[js2 + d]
        else:
            # d < 0, let ad = -d = |d|
            ad = -d
            # First term: (1-h[j-d]) = (1-h[j+ad]) for j in [0, N-d) = [0, N+ad)
            js = np.arange(0, N + d)  # N + d = N - ad
            g[js] += w[r] * (1 - h[js - d])
            # Second term: -h[j+d] = -h[j-ad] for j in [-d, N) = [ad, N)
            js2 = np.arange(ad, N)
            g[js2] -= w[r] * h[js2 + d]
    
    g *= dx
    return S, g

import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_feasible(y, target_sum):
    N = len(y)
    y = np.clip(y.astype(np.float64), -2.0, 2.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    resid = target_sum - float(np.sum(h))
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = np.where(h < 1.0)[0]
            if len(mask) > 0:
                h[mask[0]] = min(1.0, h[mask[0]] + resid)
        else:
            mask = np.where(h > 0.0)[0]
            if len(mask) > 0:
                h[mask[0]] = max(0.0, h[mask[0]] + resid)
    return h

def smooth_obj_and_grad(h, tau, dx):
    N = len(h)
    C = np.correlate(h, 1.0 - h, mode="full")
    m = float(np.max(C))
    shifted = (C - m) / tau
    shifted = np.clip(shifted, -80.0, 80.0)
    exp_s = np.exp(shifted)
    s_exp = float(np.sum(exp_s))
    w = exp_s / s_exp
    obj = (m + tau * np.log(s_exp)) * dx
    g = np.zeros(N, dtype=np.float64)
    for r in range(2 * N - 1):
        wr = float(w[r])
        if wr < 1e-14:
            continue
        d = r - N + 1
        if d >= 0:
            g[d:N] += wr * (1.0 - h[0:N - d])
            if d > 0:
                g[0:N - d] -= wr * h[d:N]
        else:
            ad = -d
            g[0:N - ad] += wr * (1.0 - h[ad:N])
            g[ad:N] -= wr * h[0:N - ad]
    g *= dx
    return obj, g

    x_new = (np.arange(N_target, dtype=np.float64) + 0.5) / N_target
    return np.interp(x_new, x_old, h_init)

        self.dx = dx

    def jac(self, x):
        return smooth_obj_and_grad(x, self.tau, self.dx)[1]
    def cfun(self, x):
        return np.sum(x) - self.target_sum
    def cjac(self, x):
        return np.ones(len(x))
