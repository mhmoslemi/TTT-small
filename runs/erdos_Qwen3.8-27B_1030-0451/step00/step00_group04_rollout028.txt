import numpy as np
import time

def exact_score(h):
    """Compute C5 value exactly."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(x, n):
    """Project x onto {0 <= h <= 1, sum(h) = n/2}."""
    h = np.clip(x, 0.0, 1.0)
    target = n / 2.0
    diff = h.sum() - target
    if abs(diff) < 1e-12:
        return h
    # Adjust: if sum too large, decrease; if too small, increase
    if diff > 0:
        # Need to decrease sum by diff
        # Find coordinates where h < 1 (can decrease)
        mask = h < 1.0 - 1e-12
        if mask.sum() == 0:
            return h
        # Distribute decrease proportional to available space
        space = (1.0 - h) * mask
        total_space = space.sum()
        if total_space < diff:
            h = np.where(mask, 0.0, h)  # This shouldn't happen if we started in [0,1]
            # Actually this means we can't reach target, which would be a problem
            # But if x was already feasible-ish, this shouldn't happen
            # Fallback: uniform
            h = np.full(n, 0.5)
            return h
        h -= (diff / total_space) * space
    else:
        # Need to increase sum by -diff
        mask = h > 1e-12
        if mask.sum() == 0:
            h = np.full(n, 0.5)
            return h
        space = h * mask
        total_space = space.sum()
        if total_space < -diff:
            h = np.full(n, 0.5)
            return h
        h += (-diff / total_space) * space
    return np.clip(h, 0.0, 1.0)

def exact_score(h):
    n = len(h)

def compute_grad(h, corr, n, K=10):
    """Compute the gradient of the max correlation (approximate, using top K lags)."""
    active_idx = np.argsort(corr)[-K:]
    g = np.zeros(n)
    for idx in active_idx:
        lag = abs(int(idx) - (n - 1))
        w = corr[idx]
        if lag == 0:
            g += w * (1.0 - 2.0 * h)
        else:
            g[lag:] += w * (1.0 - h[:-lag])
            g[:n - lag] -= w * h[lag:]
    return g
