def project_sum_box(x, target):
    """Project x onto {0 <= y <= 1, sum(y) = target}"""
    n = len(x)
    lo, hi = -x.max(), 1 - x.min()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0, 1).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + 0.5*(lo+hi), 0, 1)
    # Fix residual
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = y < 1 - 1e-12
            if mask.sum() > 0:
                y[mask] += residual / mask.sum()
                y = np.clip(y, 0, 1)
        else:
            mask = y > 1e-12
            if mask.sum() > 0:
                y[mask] += residual / mask.sum()
                y = np.clip(y, 0, 1)
    return y

def evaluate_c5(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def corr_values(h):
    """Compute all lag values v_k = dx * correlate_k"""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def logsumexp_obj(h, beta):
    """Smoothed objective: logsumexp of corr values"""
    v = corr_values(h)
    m = v.max()
    shifted = beta * (v - m)
    # Clip to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def logsumexp_grad(h, beta):
    """Gradient of logsumexp objective"""
    n = len(h)
    dx = 2.0 / n
    v = corr_values(h)
    m = v.max()
    shifted = np.clip(beta * (v - m), -500, 500)
    w = np.exp(shifted)
    w = w / w.sum()  # normalize
    
    # For each lag k, compute gradient

    # Element index p corresponds to lag k = p - (n-1)

    grad = np.zeros(n)
    for p in range(2 * n - 1):

        lo1 = max(0, k)  # i such that i-k >= 0

            idx = np.arange(lo1, hi1)
            grad[idx] += w[p] * dx * (1.0 - h[idx - k])
        # Contribution from (1-h[n]) term: for index i, need n = i, valid if 0 <= i+k < n
        lo2 = max(0, -k)  # i such that i+k >= 0
        hi2 = min(n, n - k)  # i such that i+k < n
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= w[p] * dx * h[idx + k]
    
    return grad

def transfer_resolution(h, n_new):
    """Interpolate h from its current length to n_new"""
    n_old = len(h)
    pos_old = np.arange(n_old) / n_old
    pos_new = np.arange(n_new) / n_new
    h_new = np.interp(pos_new, pos_old, h)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new

def lp_active_refine(h, radius, max_active=80):
    """Trust-region LP on active constraints"""
    n = len(h)
    v = corr_values(h)
    # Select active lags (top values)
    k = min(max_active, 2 * n - 1)
    top_idx = np.argsort(v)[::-1][:k]
    
    # For each active lag, compute gradient row
    # grad_k for lag at position p (k_lag = p - (n-1))
    A_rows = []
    b_rows = []
    for p in top_idx:
        k_lag = p - (n - 1)
        row = np.zeros(n)
        lo1 = max(0, k_lag)
        hi1 = min(n, n + k_lag)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            row[idx] += (1.0 - h[idx - k_lag])
        lo2 = max(0, -k_lag)
        hi2 = min(n, n - k_lag)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            row[idx] -= h[idx + k_lag]
        row *= 2.0 / n  # dx factor
        
        # Linearized constraint: row · h <= t + (row · h0 - v[p])
        # i.e., row · h - t <= row · h0 - v[p]
        A_rows.append(row)
        b_rows.append(row @ h - v[p])
    
    A = np.array(A_rows)
    b = np.array(b_rows)
    
    # Build LP: minimize t
    # Variables: [h_0, ..., h_{n-1}, t]
    # Objective: minimize t => c = [0,...,0, 1]
    c = np.zeros(n + 1)
    c[-1] = 1.0
    
    # Inequality constraints: A @ h - t <= b  =>  [A | -1] @ [h, t] <= b
    A_ub = np.column_stack([A, -np.ones(len(A_rows))])
    
    # Equality: sum(h) = n/2  =>  [1,...,1, 0] @ [h, t] = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds: [h0_i - r, h0_i + r] intersect [0, 1] for h; t in [-0.5, 1.0]
    lo = np.maximum(0, h - radius)
    hi = np.minimum(1, h + radius)
    bounds = list(zip(lo, hi))
    bounds.append((-0.5, 1.0))
    
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs', options={'time_limit': 10})
        if res.success and res.x is not None:
            h_candidate = res.x[:n]
            h_candidate = project_sum_box(h_candidate, n / 2.0)
            return h_candidate
    except Exception:
        pass
    return None

    n = len(h)
    target = n / 2.0
    
    def objective(x):
        return logsumexp_obj(x, beta)

        return logsumexp_grad(x, beta)

import numpy as np
import time
from scipy.optimize import minimize, linprog

_lse_beta = 10
_sum_target = 84.0

def project_sum_box(x, target):
    n = len(x)
    lo, hi = -x.max(), 1.0 - x.min()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s < target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - y.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = y < 1.0 - 1e-10
            cnt = mask.sum()
            if cnt > 0:
                y[mask] += residual / cnt
        else:
            mask = y > 1e-10
            cnt = mask.sum()
            if cnt > 0:
                y[mask] += residual / cnt
        y = np.clip(y, 0.0, 1.0)
    return y

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def lse_objective(h, beta):
    v = corr_values(h)
    m = v.max()
    shifted = np.clip(beta * (v - m), -500.0, 500.0)
    return m + np.log(np.sum(np.exp(shifted))) / beta

def lse_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    v = corr_values(h)
    m = v.max()
    shifted = np.clip(beta * (v - m), -500.0, 500.0)
    w = np.exp(shifted)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros_like(v)
        w[v.argmax()] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n)
    for p in range(2 * n - 1):
        if w[p] < 1e-15:
            continue
        k = p - (n - 1)
        lo1, hi1 = max(0, k), min(n, n + k)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            grad[idx] += w[p] * (1.0 - h[idx - k])
        lo2, hi2 = max(0, -k), min(n, n - k)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            grad[idx] -= w[p] * h[idx + k]
    grad *= dx
    return grad

def _slsqp_obj(x):
    return lse_objective(x, _lse_beta)

def _slsqp_jac(x):
    return lse_grad(x, _lse_beta)

def _sum_fun(x):
    return np.sum(x) - _sum_target

def _sum_jac(x):
    return np.ones(len(x))

def lp_refine(h, radius, max_active=60, time_limit=5):
    n = len(h)
    target = n / 2.0
    v = corr_values(h)
    k_use = min(max_active, 2 * n - 1)
    top_idx = np.argsort(v)[::-1][:k_use]
    A_rows = []
    b_rows = []
    for p in top_idx:
        k_lag = p - (n - 1)
        row = np.zeros(n)
        lo1, hi1 = max(0, k_lag), min(n, n + k_lag)
        if hi1 > lo1:
            idx = np.arange(lo1, hi1)
            row[idx] += (1.0 - h[idx - k_lag])
        lo2, hi2 = max(0, -k_lag), min(n, n - k_lag)
        if hi2 > lo2:
            idx = np.arange(lo2, hi2)
            row[idx] -= h[idx + k_lag]
        row *= 2.0 / n
        A_rows.append(row)
        b_rows.append(row @ h - v[p])
    A = np.array(A_rows)
    b = np.array(b_rows)
    c = np.zeros(n + 1)
    c[-1] = 1.0
    A_ub = np.column_stack([A, -np.ones(len(A_rows))])
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([target])
    lo = np.maximum(0.0, h - radius)
    hi = np.minimum(1.0, h + radius)
    bounds = list(zip(lo, hi)) + [(-0.5, 1.0)]
    try:
        res = linprog(c, A_ub=A_ub, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs',
                      options={'time_limit': time_limit})
        if res.success and res.x is not None:
            return project_sum_box(res.x[:n], target)
    except Exception:
        pass
    return None

def slsqp_refine(h, beta, max_iter=100):
    global _lse_beta, _sum_target
    n = len(h)
    _lse_beta = beta
    _sum_target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    try:
        res = minimize(_slsqp_obj, h.copy(), method='SLSQP', jac=_slsqp_jac,
                       bounds=bounds,
                       constraints=[{'type': 'eq', 'fun': _sum_fun, 'jac': _sum_jac}],
                       options={'maxiter': max_iter, 'ftol': 1e-12})
        if res.x is not None and np.isfinite(res.x).all():
            return project_sum_box(res.x, n / 2.0)
    except Exception:
        pass
    return None
