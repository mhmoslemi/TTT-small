def project_feasible(h):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = len(x)/2} using shift-clip."""
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    
    # If already feasible, return
    if abs(np.sum(h) - target) < 1e-12:
        return h
    
    # Use bisection on shift parameter
    # x(alpha) = clip(h + alpha, 0, 1)
    # Find alpha such that sum(x(alpha)) = target
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        x = np.clip(h + mid, 0.0, 1.0)
        s = np.sum(x)
        if s > target:
            hi = mid
        else:
            lo = mid
    
    x = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    
    # Final correction for any residual
    residual = target - np.sum(x)
    if abs(residual) > 1e-15:
        # Distribute residual to coordinates with room
        if residual > 0:
            # Need to add mass: find coords where x < 1
            room = 1.0 - x
            order = np.argsort(-room)
            for i in order:
                add = min(residual, room[i])
                x[i] += add
                residual -= add
                if abs(residual) < 1e-15:
                    break
        else:
            # Need to remove mass: find coords where x > 0
            room = x
            order = np.argsort(room)
            for i in order:
                rem = min(-residual, room[i])
                x[i] -= rem
                residual += rem
                if abs(residual) < 1e-15:
                    break
    
    return x

import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max_k ∫ h(x)(1-h(x+k)) dx using discrete formula."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h):
    """Project h onto feasible set {x: 0<=x_i<=1, sum(x)=n/2}."""
    h = np.asarray(h, dtype=np.float64).copy()
    h = np.clip(h, 0.0, 1.0)
    n = len(h)
    target = n / 2.0
    
    diff = target - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    
    # Bisection on alpha: x(alpha) = clip(h + alpha, 0, 1)
    lo = -h.max() - 1.0
    hi = 1.0 - h.min() + 1.0
    
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s > target:
            hi = mid
        else:
            lo = mid
    
    alpha = 0.5 * (lo + hi)
    h = np.clip(h + alpha, 0.0, 1.0)
    
    # Final residual correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            order = np.argsort(1.0 - h)  # most room first
        else:
            order = np.argsort(h)  # least mass first
        for idx in order:
            if residual > 0:
                add = min(residual, 1.0 - h[idx])
                h[idx] += add
                residual -= add
            else:
                rem = min(-residual, h[idx])
                h[idx] -= rem
                residual += rem
            if abs(residual) < 1e-14:
                break
    
    return h

def resample_vector(h, target_n):
    """Resample vector h from len(h) to target_n points, preserving feasibility."""
    n = len(h)
    target_n = int(target_n)
    
    if target_n == n:
        return h.copy()
    
    if target_n % n == 0:
        # Upsampling by repetition
        k = target_n // n
        h_new = np.repeat(h, k)
    elif n % target_n == 0:
        # Downsampling by block averaging
        k = n // target_n
        h_new = h.reshape(target_n, k).mean(axis=1)
    else:
        # General interpolation
        x_old = (np.arange(n) + 0.5) / n
        x_new = (np.arange(target_n) + 0.5) / target_n
        h_new = np.interp(x_new, x_old, h)
    
    return project_feasible(h_new)

def coordinate_descent(h, rng, time_limit, max_sweeps=10000):
    """Run coordinate descent with pairwise mass exchange."""
    n = len(h)
    current_c5 = compute_c5(h)
    h = project_feasible(h)
    current_c5 = compute_c5(h)
    
    start = time.time()
    no_improve_sweeps = 0
    max_no_improve = 50
    
    for sweep in range(max_sweeps):
        if time.time() > time_limit:
            break
        
        improved = False
        # Shuffle pair order for diversity
        pairs = []
        for i in range(n):
            for j in range(i+1, n):
                pairs.append((i, j))
        rng.shuffle(pairs)
        
        for (i, j) in pairs:
            if time.time() > time_limit:
                break
            
            # Compute feasible range for d
            d_min = max(-h[i], h[j] - 1.0)
            d_max = min(1.0 - h[i], h[j])
            
            if d_max - d_min < 1e-10:
                continue
            
            # Try several candidate d values
            width = d_max - d_min
            mid = 0.5 * (d_min + d_max)
            
            candidates = [mid]
            # Add fraction-based candidates
            for frac in [0.25, 0.5, 0.75]:
                candidates.append(d_min + frac * width)
                candidates.append(d_max - frac * width)
            # Add some random candidates
            for _ in range(3):
                candidates.append(d_min + rng.random() * width)
            # Add endpoints (slightly inside)
            candidates.append(d_min + 1e-6)
            candidates.append(d_max - 1e-6)
            # Add 0 (no change)
            candidates.append(0.0)
            
            best_d = 0.0
            best_local_c5 = current_c5
            
            for d in candidates:
                if d < d_min - 1e-12 or d > d_max + 1e-12:
                    continue
                d = max(d_min, min(d_max, d))
                if abs(d) < 1e-12:
                    continue
                
                h[i] += d
                h[j] -= d
                
                # Quick feasibility check
                if h[i] < -1e-12 or h[i] > 1 + 1e-12 or h[j] < -1e-12 or h[j] > 1 + 1e-12:
                    h[i] -= d
                    h[j] += d
                    continue
                
                new_c5 = compute_c5(h)
                
                if new_c5 < best_local_c5 - 1e-12:
                    best_local_c5 = new_c5
                    best_d = d
                else:
                    # Revert
                    h[i] -= d
                    h[j] += d
            
            if best_d != 0.0:
                h[i] += best_d
                h[j] -= best_d
                current_c5 = best_local_c5
                improved = True
        
        if improved:
            no_improve_sweeps = 0
        else:
            no_improve_sweeps += 1
            if no_improve_sweeps >= max_no_improve:
                break
    
    h = project_feasible(h)
    final_c5 = compute_c5(h)
    return h, final_c5

def compute_sensitivity(h):
    """Compute S[i][m] = d(corrm)/d(h[i]) for all i, m."""
    n = len(h)
    L = 2*n - 1
    S = np.zeros((n, L))
    
    for r in range(0, n):
        m = n - 1 + r
        # S[i][m] = (1-h[i-r])*[r<=i<=n-1] - h[i+r]*[0<=i<=n-1-r]
        for i in range(r, n):
            S[i, m] += (1.0 - h[i-r])
        for i in range(0, n-r):
            S[i, m] -= h[i+r]
    
    for s in range(1, n):
        m = n - 1 - s
        # S[i][m] = (1-h[i+s])*[0<=i<=n-1-s] - h[i-s]*[s<=i<=n-1]
        for i in range(0, n-s):
            S[i, m] += (1.0 - h[i+s])
        for i in range(s, n):
            S[i, m] -= h[i-s]
    
    return S

    target = 0.5 * n
    s = np.sum(h)
    if abs(s - target) < 1e-12:

    # Bisection

        if np.sum(np.clip(h + mid, 0.0, 1.0)) > target:
            hi = mid
        else:
            lo = mid
    h = np.clip(h + 0.5*(lo+hi), 0.0, 1.0)
    # Final fix
    res = target - np.sum(h)
    if abs(res) > 1e-14:
        if res > 0:
            for idx in np.argsort(1.0 - h):
                a = min(res, 1.0 - h[idx])
                h[idx] += a
                res -= a
                if abs(res) < 1e-14: break
        else:
            for idx in np.argsort(h):
                a = min(-res, h[idx])
                h[idx] -= a
                res += a
                if abs(res) < 1e-14: break

def compute_sensitivity(h):

    L = 2 * n - 1
    S = np.zeros((n, L))
    # For shift r >= 0 (lag m = n-1+r):
    for r in range(0, n):
        m = n - 1 + r
        # S[i][m] += (1-h[i-r]) for i in [r, n-1]
        i_range = np.arange(r, n)
        S[i_range, m] += (1.0 - h[i_range - r])
        # S[i][m] -= h[i+r] for i in [0, n-1-r]
        i_range2 = np.arange(0, n - r)
        S[i_range2, m] -= h[i_range2 + r]
    # For shift r < 0, s = -r > 0 (lag m = n-1-s):
    for s in range(1, n):
        m = n - 1 - s
        # S[i][m] += (1-h[i+s]) for i in [0, n-1-s]
        i_range = np.arange(0, n - s)
        S[i_range, m] += (1.0 - h[i_range + s])
        # S[i][m] -= h[i-s] for i in [s, n-1]
        i_range2 = np.arange(s, n)
        S[i_range2, m] -= h[i_range2 - s]
    return S

def ternary_search_min(corr, slope, a, b, iters=35):
    """Find d in [a,b] minimizing max(corr + d*slope)."""
    if b - a < 1e-14:
        return 0.5 * (a + b)
    for _ in range(iters):
        m1 = a + (b - a) / 3.0
        m2 = b - (b - a) / 3.0
        f1 = np.max(corr + m1 * slope)
        f2 = np.max(corr + m2 * slope)
        if f1 < f2:
            b = m2
        else:
            a = m1
    return 0.5 * (a + b)

def resample_vector(h, target_n):
    n = len(h)
    if target_n == n:
        return h.copy()
    if target_n % n == 0:
        k = target_n // n
        return project_feasible(np.repeat(h, k))
    if n % target_n == 0:
        k = n // target_n
        return project_feasible(h.reshape(target_n, k).mean(axis=1))
    x_old = (np.arange(n) + 0.5) / n
    x_new = (np.arange(target_n) + 0.5) / target_n
    return project_feasible(np.interp(x_new, x_old, h))

Wait, one more consideration. The `compute_sensitivity` function I wrote earlier computes S using a loop over shifts. Let me make sure it's vectorized properly for speed.

For shift r >= 0 (m = n-1+r):
S[i][m] = (1-h[i-r]) * [r <= i <= n-1] - h[i+r] * [0 <= i <= n-1-r]

Vectorized:
```python
for r in range(n):
    m = n-1+r
    # First term: (1-h[i-r]) for i in [r, n-1]
    S[r:, m] += (1.0 - h[:n-r])  # h[i-r] for i=r..n-1 is h[0..n-1-r]
    # Second term: -h[i+r] for i in [0, n-1-r]
    S[:n-r, m] -= h[r:]  # h[i+r] for i=0..n-1-r is h[r..n-1]
```

For shift r < 0, s = -r (m = n-1-s):
S[i][m] = (1-h[i+s]) * [0 <= i <= n-1-s] - h[i-s] * [s <= i <= n-1]

Vectorized:
```python
for s in range(1, n):
    m = n-1-s
    # First term: (1-h[i+s]) for i in [0, n-1-s]
    S[:n-s, m] += (1.0 - h[s:])  # h[i+s] for i=0..n-1-s is h[s..n-1]
    # Second term: -h[i-s] for i in [s, n-1]
    S[s:, m] -= h[:n-s]  # h[i-s] for i=s..n-1 is h[0..n-1-s]
```

This is O(n²) total (n shifts, each O(n) work). For n=192: 192*192 = 37K operations. Fast.
