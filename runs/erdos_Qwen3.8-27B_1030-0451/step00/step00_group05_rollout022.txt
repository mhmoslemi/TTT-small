```python
import numpy as np
from scipy.optimize import linprog
import time

def _eval_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return np.max(corr)

def _fix_sum(h, n):
    """Adjust h to make sum exactly n/2 while keeping in [0,1]."""
    target = n / 2.0
    s = np.sum(h)
    if abs(s - target) < 1e-10:
        return h
    diff = target - s
    # Distribute evenly among non-boundary elements
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    if np.any(interior):
        h[interior] += diff / np.sum(interior)
        h = np.clip(h, 0, 1)
    else:
        # All at bounds, need to adjust
        # This shouldn't happen if we're careful
        h += diff / n
        h = np.clip(h, 0, 1)
    return h

def _compute_correlation_values(h, n):
    """Compute all 2n-1 correlation values (scaled by dx)."""
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return corr

def _compute_gradients(h, n):
    """Compute gradient of each correlation value wrt h.
    Returns array of shape (2n-1, n)."""
    dx = 2.0 / n
    G = np.zeros((2*n-1, n))
    for d in range(-(n-1), n):
        k = d + n - 1
        if d >= 0:
            # valid i: 0 to n-1-d
            # grad[m] = (1 if m <= n-1-d else 0) - (h[m+d] if m <= n-1-d else 0) - (h[m-d] if m >= d else 0)
            G[k, :n-d] = 1.0 - h[d:n]
            G[k, d:] -= h[:n-d]
        else:
            dp = -d
            # valid i: dp to n-1
            # grad[m] = (1 if m >= dp else 0) - (h[m-dp] if m >= dp else 0) - (h[m+dp] if m <= n-1-dp else 0)
            G[k, dp:] = 1.0 - h[:n-dp]
            G[k, :n-dp] -= h[dp:]
    G *= dx
    return G

def _lp_step(h, n, rho):
    """Solve the trust-region LP to find a descent direction.
    Returns (d, tau) or (None, None) if infeasible."""
    corr_vals = _compute_correlation_values(h, n)
    G = _compute_gradients(h, n)
    
    # LP: minimize tau
    # s.t. for each j: -tau + G[j,:] @ d <= -corr_vals[j]
    #      -h[i] <= d[i] <= 1-h[i]  (box)
    #      |d[i]| <= rho  (trust region)
    #      sum(d) = 0
    
    n_lags = 2*n - 1
    
    # Variables: [d_0, ..., d_{n-1}, tau] → n+1 variables
    nvar = n + 1
    
    # Objective: minimize tau → c = [0, ..., 0, 1]
    c_obj = np.zeros(nvar)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each lag j: G[j,:] @ d - tau <= -corr_vals[j]
    A_ub = np.zeros((n_lags, nvar))
    A_ub[:, :n] = G
    A_ub[:, n] = -1.0  # -tau
    b_ub = -corr_vals
    
    # Box + trust region: -min(h[i], rho) <= d[i] <= min(1-h[i], rho)
    # As inequalities: d[i] <= min(1-h[i], rho) and -d[i] <= min(h[i], rho)
    lo = np.minimum(h, rho)
    hi = np.minimum(1.0 - h, rho)
    
    A_box = np.zeros((2*n, nvar))
    A_box[:n, :n] = np.eye(n)      # d[i] <= hi[i]
    A_box[n:, :n] = -np.eye(n)     # -d[i] <= lo[i]
    b_box = np.concatenate([hi, lo])
    
    A_ub_full = np.vstack([A_ub, A_box])
    b_ub_full = np.concatenate([b_ub, b_box])
    
    # Equality: sum(d) = 0
    A_eq = np.zeros((1, nvar))
    A_eq[0, :n] = 1.0
    b_eq = np.zeros(1)
    
    # Bounds for tau: unbounded (or large)
    bounds = [(None, None)] * n + [(0, None)]
    
    try:
        res = linprog(c_obj, A_ub=A_ub_full, b_ub=b_ub_full, 
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
        if res.success:
            d = res.x[:n]
            tau = res.x[n]
            return d, tau
        else:
            return None, None
    except Exception:
        return None, None
```

```python
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
```

```python
def _grads(h, n):
    dx = 2.0 / n
    G = np.zeros((2 * n - 1, n))
    for d in range(-(n - 1), n):
        k = d + n - 1
        if d >= 0:
            G[k, :n - d] = 1.0 - h[d:n]
            if d > 0:
                G[k, d:] -= h[:n - d]
            else:
                G[k, :] -= h[:]
        else:
            dp = -d
            G[k, dp:] = 1.0 - h[:n - dp]
            G[k, :n - dp] -= h[dp:]
    return G * dx

def _fix_sum(h, n):

    A_ub[:, n] = -1.0

    hi = np.minimum(1.0 - h, rho)

    A_box[n:, :n] = -np.eye(n)
```
