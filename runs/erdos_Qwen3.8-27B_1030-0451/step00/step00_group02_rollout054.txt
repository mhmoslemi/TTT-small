import numpy as np
import time

def compute_c5(h):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full") * dx
    return float(np.max(corr))

def project_feasible(h, target_sum):
    """Project h onto {0<=h_i<=1, sum(h)=target_sum} using bisection on Lagrange multiplier."""
    h = np.asarray(h, dtype=float)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smooth_max_value(values, alpha):
    m = np.max(values)
    return m + np.log(np.sum(np.exp(alpha * (values - m)))) / alpha

def smooth_max_weights(values, alpha):
    m = np.max(values)
    ev = np.exp(alpha * (values - m))
    return ev / np.sum(ev)

def grad_smooth_max_corr(h, alpha):
    """
    Gradient of smooth_max(corr * dx) with respect to h.
    corr = np.correlate(h, 1-h, mode='full')
    """
    n = len(h)
    dx = 2.0 / n
    comp = 1.0 - h
    corr = np.correlate(h, comp, mode="full") * dx
    w = smooth_max_weights(corr, alpha)  # shape (2n-1,)
    
    # w[k] corresponds to lag = k - (n-1)
    # c[k] = sum_i h[i] * (1-h[i + k - (n-1)])
    # dc[k]/dh[m] = (1-h[m+k-(n-1)]) * 1{0<=m+k-(n-1)<n} 
    #              - h[m-k+(n-1)] * 1{0<=m-k+(n-1)<n}
    
    g = np.zeros(n)
    for m in range(n):
        # Valid k range: need 0<=m+k-(n-1)<n or 0<=m-k+(n-1)<n
        # k-(n-1) ranges from -(n-1) to n-1
        # For term 1: 0<=m+k-(n-1)<n => n-1-m <= k <= n-1+m, intersect [0, 2n-2]
        # For term 2: 0<=m-k+(n-1)<n => m-n+1 <= k <= m+n-1, intersect [0, 2n-2]
        k_lo = max(0, n - 1 - m, m - n + 1)
        k_hi = min(2 * n - 2, n - 1 + m, m + n - 1)
        
        if k_lo > k_hi:
            continue
        
        lags = np.arange(k_lo, k_hi + 1) - (n - 1)
        w_sub = w[k_lo:k_hi + 1]
        
        # Term 1: (1-h[m+lag]) where 0<=m+lag<n
        idx1 = m + lags
        valid1 = (idx1 >= 0) & (idx1 < n)
        t1 = 0.0
        if np.any(valid1):
            t1 = np.dot(w_sub[valid1], 1.0 - h[idx1[valid1]])
        
        # Term 2: h[m-lag] where 0<=m-lag<n
        idx2 = m - lags
        valid2 = (idx2 >= 0) & (idx2 < n)
        t2 = 0.0
        if np.any(valid2):
            t2 = np.dot(w_sub[valid2], h[idx2[valid2]])
        
        g[m] = dx * (t1 - t2)
    
    return g

def optimize_smooth(h, target_sum, alpha_start=5.0, alpha_end=200.0, 
                    max_iters=3000, time_limit=60.0, verbose=False):
    """Projected gradient descent with annealed smooth max."""
    n = len(h)
    h = project_feasible(h.copy(), target_sum)
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    t0 = time.time()
    lr = 1.0 / (n * 0.1)  # initial learning rate
    
    for it in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        
        frac = it / max(max_iters - 1, 1)
        alpha = alpha_start * (alpha_end / alpha_start) ** frac
        
        # Compute gradient
        g = grad_smooth_max_corr(h, alpha)
        
        # Backtracking line search
        f_curr = smooth_max_value(np.correlate(h, 1-h, mode="full") * (2.0/n), alpha)
        step_accepted = False
        for _ in range(30):
            h_new = project_feasible(h - lr * g, target_sum)
            f_new = smooth_max_value(np.correlate(h_new, 1-h_new, mode="full") * (2.0/n), alpha)
            if f_new < f_curr:
                h = h_new
                step_accepted = True
                break
            lr *= 0.5
        
        if not step_accepted:
            lr *= 0.5
            if lr < 1e-12:
                lr = 1e-3
        else:
            lr *= 1.05  # increase slightly if step was accepted
        
        # Check actual C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return best_h, best_c5

where i + k - (m-1) must be in [0, m-1].

So ∂corr[k]/∂h[j]:
- h[j] appears directly when i=j: contributes comp[j + k - (m-1)] if 0 ≤ j+k-(m-1) < m
- h[j] appears through comp when i + k - (m-1) = j, i.e., i = j - k + (m-1): 
  contributes -h[j - k + (m-1)] if 0 ≤ j-k+(m-1) < m (since comp = 1-h, d(comp[j])/d(h[j]) = -1)

So ∂corr[k]/∂h[j] = comp[j+k-(m-1)] * 1{0≤j+k-(m-1)<m} - h[j-k+(m-1)] * 1{0≤j-k+(m-1)<m}
