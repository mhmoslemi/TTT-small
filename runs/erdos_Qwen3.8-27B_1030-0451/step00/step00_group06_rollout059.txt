import numpy as np

def project_to_box_sum(h, S):
    """Project h onto {x: 0≤x_i≤1, sum(x)=S}."""
    n = len(h)
    # Bisection on tau
    lo, hi = h.min() - S, h.max() - S + 1
    for _ in range(100):
        tau = (lo + hi) / 2
        x = np.clip(h - tau, 0, 1)
        s = x.sum()
        if s > S:
            lo = tau
        else:
            hi = tau
    x = np.clip(h - (lo+hi)/2, 0, 1)
    # Fix rounding
    diff = S - x.sum()
    if abs(diff) > 1e-12:
        # Distribute difference
        # Find indices where we can adjust
        idx = np.argsort(-x) if diff > 0 else np.argsort(x)
        step = diff / n
        x += step
        x = np.clip(x, 0, 1)
        # If still off, adjust individual elements
        # This should be rare with good bisection
        residual = S - x.sum()
        if abs(residual) > 1e-10:
            # Add/residual to one element that has room
            if residual > 0:
                for i in range(n):
                    if x[i] < 1 - 1e-15:
                        x[i] += residual
                        break
            else:
                for i in range(n):
                    if x[i] > 1e-15:
                        x[i] += residual
                        break
    return x
