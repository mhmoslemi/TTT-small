from scipy.optimize import minimize
import time

def compute_c5(h):

    dx = 2.0 / n

def project_onto_simplex_box(p, target_sum):
    """Project p onto {x: 0<=x<=1, sum(x)=target_sum}."""
    n = len(p)
    lo, hi = 0.0, 1.0
    # Bisect for lambda
    for _ in range(100):
        mid = (lo + hi) / 2
        proj = np.clip(p - mid, 0.0, 1.0)
        s = np.sum(proj)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    proj = np.clip(p - (lo+hi)/2, 0.0, 1.0)
    # Fix rounding
    diff = target_sum - np.sum(proj)
    if abs(diff) > 1e-10:
        # Distribute the difference
        idx = np.argsort(p - (lo+hi)/2)
        if diff > 0:
            for i in range(int(abs(diff)*n) + 1):
                proj[idx[i % n]] += diff / (int(abs(diff)*n) + 1)
        else:
            for i in range(int(abs(diff)*n) + 1):
                proj[idx[i % n]] += diff / (int(abs(diff)*n) + 1)
        proj = np.clip(proj, 0.0, 1.0)
    return proj

def project_constraint(p, n):

    target = n / 2.0
    # Bisection for lambda
    lo = np.min(p) - 1.0

        if s > target:
            lo = mid
        else:

    proj = np.clip(p - lam, 0.0, 1.0)

    n = len(h)

    corr_shifted = corr - np.max(corr)

import numpy as np
import time
from scipy.optimize import minimize as sci_minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_constraint(p, n):
    target = n / 2.0
    lo = float(np.min(p)) - 1.0
    hi = float(np.max(p)) + 1.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        proj = np.clip(p - mid, 0.0, 1.0)
        s = float(np.sum(proj))
        if s > target:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    proj = np.clip(p - lam, 0.0, 1.0)
    diff = target - float(np.sum(proj))
    if abs(diff) > 1e-12:
        mask = (proj > 1e-8) & (proj < 1.0 - 1e-8)
        nm = int(np.sum(mask))
        if nm > 0:
            proj[mask] += diff / nm
    return proj

def smooth_gradient(h, corr, tau):
    n = len(h)
    cs = corr - np.max(corr)
    w = np.exp(tau * cs)
    ws = w.sum()
    if ws < 1e-300:
        return np.zeros(n)
    w = w / ws
    grad = np.zeros(n)
    for k in range(len(corr)):
        wk = w[k]
        if wk < 1e-14:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            grad[lag:] += wk * (1.0 - h[:n - lag])
            grad[:n - lag] -= wk * h[lag:]
        else:
            L = -lag
            grad[:n - L] += wk * (1.0 - h[L:])
            grad[L:] -= wk * h[:n - L]
    return grad

        tau_max = tau_maxes[pass_num]

        pass_t0 = time.time()

        while time.time() - pass_t0 < pass_time and time.time() - t0 < time_limit - 5:
            elapsed_in_pass = max(time.time() - pass_t0, 1e-6)
            frac = min(elapsed_in_pass / max(pass_time, 1.0), 1.0)
            tau = 3.0 + (tau_max - 3.0) * (frac ** 0.5)

            c5_val = float(np.max(corr) * dx)

            if c5_val < best_c5 - 1e-13:
                best_c5 = c5_val
                best_h = h.copy()

            grad = smooth_gradient(h, corr, tau)
            gnorm = float(np.linalg.norm(grad))

            if gnorm < 1e-12:
                h = project_constraint(h + np.random.randn(n) * 0.02, n)
                continue

            made_progress = False
            for bt in range(10):
