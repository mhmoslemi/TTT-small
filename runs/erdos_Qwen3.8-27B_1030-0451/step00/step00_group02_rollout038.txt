def make_random_balanced(n, kind, rng):
    idx = np.arange(n, dtype=float)
    if kind == 0:
        scores = rng.random(n)
    elif kind == 1:
        alpha = 0.45339765151942366  # (sqrt5-1)/4? Actually compute math.
        scores = np.abs(np.sin((idx + rng.random()) * 2.0 * math.pi * alpha))
    elif kind == 2:
        alpha = math.sqrt(2.0)
        scores = (idx * alpha + rng.random()) % 1.0
    elif kind == 3:
        a = rng.uniform(0.1, 5.0)
        b = rng.uniform(0.1, 5.0)
        scores = 0.6*np.sin(idx*a + rng.random()) + 0.4*np.cos(idx*b + rng.random())
    else:
        scores = rng.random(n)
    if not np.all(np.isfinite(scores)):
        scores = rng.random(n)

import numpy as np

class SmoothCtx:
    def __init__(self, n, tau):
        self.n = int(n)
        self.tau = float(tau)
        self.dx = 2.0 / float(self.n)
    def __call__(self, x):
        h = np.asarray(x, dtype=float)
        q = 1.0 - h
        C = np.correlate(h, q, mode='full')
        G = C * self.dx
        if not np.all(np.isfinite(G)):
            return 1e100, np.zeros_like(h)
        m = float(G.max())
        tau = self.tau
        if tau <= 1e-15:
            w = np.zeros_like(G)
            idx = int(np.argmax(G))
            w[idx] = 1.0
            logsum = m
        else:
            z = (G - m) / tau
            expz = np.exp(z)
            s = float(expz.sum())
            if s <= 0.0 or not np.isfinite(s):

import math
import time
import numpy as np
import scipy.optimize as sopt


class SmoothCtx:
    def __init__(self, n, tau):
        self.n = int(n)
        self.tau = float(tau)
        self.dx = 2.0 / float(self.n)

    def __call__(self, x):
        h = np.asarray(x, dtype=float)
        q = 1.0 - h
        C = np.correlate(h, q, mode="full")
        G = C * self.dx
        if not np.all(np.isfinite(G)):
            return 1e100, np.zeros_like(h)

        m = float(G.max())
        tau = self.tau
        if tau <= 1e-15:
            w = np.zeros_like(G)
            idx = int(np.argmax(G))
            w[idx] = 1.0
            logsum = m
        else:
            z = (G - m) / tau
            expz = np.exp(z)
            s = float(expz.sum())
            if s <= 0.0 or not math.isfinite(s):
                w = np.zeros_like(G)
                idx = int(np.argmax(G))
                w[idx] = 1.0
                logsum = m
            else:
                logsum = m + tau * math.log(s)
                w = expz / s

        coeff = build_coeff(h, q, self.n)
        grad = self.dx * (coeff @ w)
        if not np.all(np.isfinite(grad)):
            grad = np.zeros_like(h)
        return float(logsum), grad


class SumEq:
    def __init__(self, target):
        self.target = float(target)

    def __call__(self, x):
        return float(np.sum(np.asarray(x, dtype=float)) - self.target)


class SumJac:
    def __call__(self, x):
        return np.ones_like(np.asarray(x, dtype=float))


def project_sum(z, S):
    h = np.asarray(z, dtype=float).copy()
    if h.size == 0:
        return h
    h = np.clip(h, 0.0, 1.0)
    if abs(float(h.sum()) - float(S)) < 1e-13:
        return h

    lo = -1.0
    hi = 1.0
    for _ in range(70):
        mid = 0.5 * (lo + hi)
        val = float(np.clip(h - mid, 0.0, 1.0).sum())
        if val > float(S):
            lo = mid
        else:
            hi = mid
    mu = 0.5 * (lo + hi)
    y = np.clip(h - mu, 0.0, 1.0)

    err = float(S) - float(y.sum())
    if abs(err) > 1e-14:
        if err > 0.0:
            idx = np.argsort(1.0 - y)[::-1]
            for i in idx:
                inc = min(float(1.0 - y[i]), err)
                if inc <= 0.0:
                    break
                y[i] += inc
                err -= inc
        else:
            idx = np.argsort(y)[::-1]
            for i in idx:
                dec = min(float(y[i]), -err)
                if dec <= 0.0:
                    break
                y[i] -= dec
                err += dec
    return y


def repair_vector(z, n):
    return project_sum(z, 0.5 * int(n))


def compute_C(h, n):
    h = np.asarray(h, dtype=float)
    dx = 2.0 / float(n)
    C = np.correlate(h, 1.0 - h, mode="full")
    return C, float(C.max() * dx)


def exact_score(h, n):
    return compute_C(h, n)[1]


def build_coeff(h, q, n):
    h = np.asarray(h, dtype=float)
    q = np.asarray(q, dtype=float)
    L = 2 * int(n) - 1
    coeff = np.zeros((int(n), L), dtype=float)
    q_rev = q[::-1]
    for i in range(int(n)):
        coeff[i, i:i + int(n)] += q_rev
        coeff[i, int(n) - 1 - i:2 * int(n) - 1 - i] -= h
    return coeff


def pair_eval(C, D, a, r1, r2, rc, dx):
    arr = C + a * D
    aa = a * a
    if aa != 0.0:
        arr[r1] += aa
        arr[r2] += aa
        arr[rc] -= 2.0 * aa
    return float(arr.max() * dx)


def pair_refine(C, D, r1, r2, rc, dx, lo, hi, center, best_a, best_val, zoom_pts):
    radius = max(1e-6, (hi - lo) * 0.12)
    for _ in range(3):
        aL = center - radius
        if aL < lo:
            aL = lo
        aH = center + radius
        if aH > hi:
            aH = hi
        if aH <= aL + 1e-13:
            break
        changed = False
        denom = float(max(1, int(zoom_pts) - 1))
        for k in range(int(zoom_pts)):
            a = aL + (aH - aL) * (float(k) / denom)
            v = pair_eval(C, D, a, r1, r2, rc, dx)
            if v < best_val:
                best_val = v
                best_a = a
                changed = True
        if not changed:
            break
        center = best_a
        radius *= 0.35
    return best_a, best_val


def pair_line_search(h, C, coeff, dx, n, p, q, score_cur):
    hp = float(h[p])
    hq = float(h[q])
    lo = max(-hp, hq - 1.0)
    hi = min(1.0 - hp, hq)
    if (not math.isfinite(lo)) or (not math.isfinite(hi)) or hi <= lo + 1e-12:
        return 0.0, score_cur

    D = coeff[p] - coeff[q]
    r1 = int(n) - 1 + p - q
    r2 = int(n) - 1 + q - p
    rc = int(n) - 1

    b1a = 0.0
    b1v = score_cur
    b2a = 0.0
    b2v = score_cur

    v0 = pair_eval(C, D, 0.0, r1, r2, rc, dx)
    if v0 < b1v:
        b1v = v0
        b1a = 0.0
    b2v = max(b2v, b1v)

    w = hi - lo
    coarse = 33 if int(n) <= 120 else 21
    denom = float(max(1, coarse - 1))
    for k in range(1, coarse):
        a = lo + w * (float(k) / denom)
        v = pair_eval(C, D, a, r1, r2, rc, dx)
        if v < b1v:
            b2a, b2v = b1a, b1v
            b1a, b1v = a, v
        elif v < b2v and abs(a - b1a) > 1e-12:
            b2a, b2v = a, v

    if b1v < score_cur - 1e-12:
        zoom_pts = 41 if int(n) <= 120 else 25
        b1a, b1v = pair_refine(C, D, r1, r2, rc, dx, lo, hi, b1a, b1v, zoom_pts)
        if b2v < score_cur - 1e-12 and abs(b2a - b1a) > 1e-7:
            b2a, b2v = pair_refine(C, D, r1, r2, rc, dx, lo, hi, b2a, b2v, zoom_pts)
            if b2v < b1v:
                b1a, b1v = b2a, b2v
        return b1a, b1v

    return b1a, b1v


def pair_descent(h, n, deadline, max_passes=2):
    n = int(n)
    h = project_sum(np.asarray(h, dtype=float), 0.5 * n)
    if n < 2:
        return h, exact_score(h, n)

    C, score = compute_C(h, n)
    coeff = build_coeff(h, 1.0 - h, n)
    dx = 2.0 / float(n)
    passes = 0

    while passes < int(max_passes) and time.monotonic() < float(deadline):
        improved = False
        pair_count = 0
        for p in range(n - 1):
            for qidx in range(p + 1, n):
                pair_count += 1
                if pair_count % 96 == 0 and time.monotonic() >= float(deadline):
                    return h, score

                a, bv = pair_line_search(h, C, coeff, dx, n, p, qidx, score)
                if abs(a) > 1e-14 and bv < score - 1e-12:
                    h2 = h.copy()
                    h2[p] += a
                    h2[qidx] -= a
                    h2 = project_sum(h2, 0.5 * n)
                    C2, s2 = compute_C(h2, n)
                    if s2 < score - 1e-13:
                        h = h2
                        score = s2
                        C = C2
                        coeff = build_coeff(h, 1.0 - h, n)
                        improved = True

            if time.monotonic() >= float(deadline):
                break

        passes += 1
        if not improved:
            break

    return h, score


def slsqp_refine(h, n, deadline, tau_list=None):

def rank_fill(n, scores):
    n = int(n)
    scores = np.asarray(scores, dtype=float)
    if scores.size != n or not np.all(np.isfinite(scores)):
        scores = np.arange(n, dtype=float)

    order = np.argsort(scores, kind="stable")[::-1]
    S = 0.5 * n
    m = int(math.floor(S + 1e-12))
    frac = S - float(m)
    h = np.zeros(n, dtype=float)
    if m > 0:
        h[order[:m]] = 1.0
    if frac > 1e-12 and m < n:
        h[order[m]] = frac
    return project_sum(h, S)


def make_random_balanced(n, kind, rng):
    n = int(n)
    idx = np.arange(n, dtype=float)
    if kind == 0:
        scores = rng.random(n)
    elif kind == 1:
        alpha = (math.sqrt(5.0) - 1.0) / 2.0
        phase = float(rng.random())
        scores = np.abs(np.sin(2.0 * math.pi * (idx * alpha + phase)))
    elif kind == 2:
        alpha = math.sqrt(2.0)
        phase = float(rng.random())
        scores = (idx * alpha + phase) % 1.0
    elif kind == 3:
        a = float(rng.uniform(0.1, 4.0))
        b = float(rng.uniform(0.1, 4.0))
        p1 = float(rng.random())
        p2 = float(rng.random())
        scores = 0.6 * np.sin(idx * a + p1) + 0.4 * np.cos(idx * b + p2)
    else:
        scores = rng.random(n)

    if not np.all(np.isfinite(scores)):
        scores = rng.random(n)
    return rank_fill(n, scores)


def perturb_vector(h, n, sigma, rng):
    n = int(n)
    z = np.asarray(h, dtype=float) + float(sigma) * rng.normal(0.0, 1.0, n)
    return project_sum(z, 0.5 * n)


def transfer_vector(h, n_old, n_new):
    n_old = int(n_old)
    n_new = int(n_new)
    h = np.asarray(h, dtype=float)
    if n_old <= 0 or n_new <= 0:
        return None
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    y = np.interp(x_new, x_old, h, left=float(h[0]), right=float(h[-1]))
    return project_sum(y, 0.5 * n_new)

            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, h2, s2, best_n)

        d3 = min(work_deadline, t0 + min(90.0, 0.46 * budget))
        if time.monotonic() < d3:
            h3, s3 = pair_descent(best_h, best_n, d3, 2)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, h3, s3, best_n)

        base = best_h.copy()
        if time.monotonic() < work_deadline:
            for sh in (1, -1, 2, -2):
                htr = np.roll(base, sh)
                str_ = exact_score(htr, best_n)
                best_h, best_score, best_n = consider_update(best_h, best_score, best_n, htr, str_, best_n)
            htr = base[::-1].copy()
            str_ = exact_score(htr, best_n)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, htr, str_, best_n)

            htr = rank_fill(best_n, base)
            str_ = exact_score(htr, best_n)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, htr, str_, best_n)

        dp = min(work_deadline, t0 + min(210.0, 0.67 * budget))
        for sigma in (0.01, 0.025, 0.05):
            if time.monotonic() >= dp:
                break
            sub = min(dp, time.monotonic() + max(12.0, 0.05 * budget))
            hp = perturb_vector(best_h, best_n, sigma, rng)
            hpo, spo = slsqp_refine(hp, best_n, sub)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hpo, spo, best_n)
            if spo < best_score + 0.003:
                pd = min(dp, time.monotonic() + max(6.0, 0.025 * budget))
                hpo2, spo2 = pair_descent(hpo, best_n, pd, 1)
                best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hpo2, spo2, best_n)

        hr = make_random_balanced(best_n, 3, rng)
        sr = exact_score(hr, best_n)
        if sr < best_score + 0.03 and time.monotonic() < dp:
            sub = min(dp, time.monotonic() + max(10.0, 0.04 * budget))
            hro, sro = slsqp_refine(hr, best_n, sub)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hro, sro, best_n)
            if sro < best_score + 0.004:
                pd = min(dp, time.monotonic() + max(4.0, 0.02 * budget))
                hro2, sro2 = pair_descent(hro, best_n, pd, 1)
                best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hro2, sro2, best_n)

        dd = min(work_deadline, t0 + min(260.0, 0.85 * budget))
        if time.monotonic() < dd and best_n > 0:
            raw_dims = [
                int(best_n) * 2,
                96,
                128,
                max(16, int(best_n) // 2),
                160,
                64,
            ]
            dims = []
            for nd in raw_dims:
                if 16 <= nd <= 2500 and nd != int(best_n) and nd not in dims:
                    dims.append(nd)
            for nd in dims:
                if time.monotonic() >= dd:
                    break
                try:
                    ht = transfer_vector(best_h, best_n, nd)
                    if ht is None:
                        continue
                    st = exact_score(ht, nd)
                    if st < best_score + 0.025:
                        sub = min(dd, time.monotonic() + max(25.0, 0.07 * budget))
                        hto, sto = slsqp_refine(ht, nd, sub)
                        best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hto, sto, nd)
                        if sto < best_score + 0.005:
                            pd = min(dd, time.monotonic() + max(6.0, 0.025 * budget))
                            hto2, sto2 = pair_descent(hto, nd, pd, 1)
                            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hto2, sto2, nd)
                except Exception:
                    pass

        if time.monotonic() < work_deadline:
            hfin, sfin = pair_descent(best_h, best_n, work_deadline, 1)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hfin, sfin, best_n)

        if time.monotonic() < work_deadline - 10.0:
            hfin, sfin = slsqp_refine(best_h, best_n, work_deadline)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hfin, sfin, best_n)

        if time.monotonic() < work_deadline - 2.0:
            hfin, sfin = pair_descent(best_h, best_n, min(work_deadline, time.monotonic() + 5.0), 1)
            best_h, best_score, best_n = consider_update(best_h, best_score, best_n, hfin, sfin, best_n)

    if best_h is None:
        nfinal = 81
        best_h = project_sum(np.full(nfinal, 0.5, dtype=float), 0.5 * nfinal)
        best_score = exact_score(best_h, nfinal)
        best_n = nfinal

    best_h = project_sum(best_h, 0.5 * int(best_n))
    best_score = exact_score(best_h, int(best_n))
    if not math.isfinite(best_score) or best_score > 1e100:
        nfinal = 81
        best_h = project_sum(np.full(nfinal, 0.5, dtype=float), 0.5 * nfinal)
        best_score = exact_score(best_h, nfinal)
        best_n = nfinal

    return np.asarray(best_h, dtype=float), float(best_score), int(best_n)
