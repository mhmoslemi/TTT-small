def compute_grad_f(h, beta=100.0):
    """Compute gradient of smoothed max using log-sum-exp trick."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    # softmax weighting
    c_max = np.max(c)
    weights = np.exp(beta * (c - c_max) / n)  # scale by n for stability
    w_sum = np.sum(weights)
    # Gradient of c[m] with respect to h[j]
    # c[m] = sum_i h[i] * (1-h[i+k]) where k = m - (n-1)
    # ∂c[m]/∂h[j] = (1-h[j+k])*[valid] - h[j-k]*[valid]
    grad = np.zeros(n)
    for m in range(2*n - 1):
        k = m - (n-1)
        w = weights[m] / w_sum * beta * dx
        if w < 1e-10:
            continue
        # First term: (1-h[j+k]) for j where 0<=j<n and 0<=j+k<n
        if k >= 0:
            # j ranges from 0 to n-1-k
            js = np.arange(0, n-k)
            grad[js] += w * (1 - h[js + k])
        else:
            # j ranges from -k to n-1
            js = np.arange(-k, n)
            grad[js] += w * (1 - h[js + k])
        # Second term: -h[j-k] for j where 0<=j<n and 0<=j-k<n
        if k >= 0:
            # j-k >= 0 means j >= k, j ranges from k to n-1
            js = np.arange(k, n)
            grad[js] -= w * h[js - k]
        else:

            js = np.arange(0, n+k)
