import math

def exact_metric(h):
    """Compute C5 = max(correlate(h, 1-h, 'full')) * dx"""
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_feasible(h, target):
    """Project h to [0,1]^N with sum = target"""
    h = np.clip(h, 0.0, 1.0)
    # Bisection to find shift that makes sum = target
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    h = np.clip(h + (lo + hi) / 2, 0.0, 1.0)
    # Final correction
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            idx = np.where(h < 1.0)[0]
        else:
            idx = np.where(h > 0.0)[0]
        if len(idx) > 0:
            step = residual / len(idx)
            h[idx] = np.clip(h[idx] + step, 0.0, 1.0)
