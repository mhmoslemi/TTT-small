import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return np.max(corr) * dx

def project_box_sum(h, target_sum):
    """Project h onto {x: 0<=x<=1, sum(x)=target_sum}"""
    # Bisection on lambda
    # x = clip(h - lambda, 0, 1), find lambda s.t. sum(x) = target_sum
    lo = h.min() - 1.0  # lambda such that all x = 1
    hi = h.max() + 1.0  # lambda such that all x = 0
    
    for _ in range(60):
        mid = (lo + hi) / 2
        x = np.clip(h - mid, 0, 1)
        s = x.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_subgrad(h):
    """Compute subgradient of C5 at h"""
    n = len(h)
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    max_corr = np.max(corr)
    dx = 2.0 / n
    
    # Active lags (within tolerance of max)
    tol = 0.01 * max_corr + 1e-10
    active = np.where(corr >= max_corr - tol)[0]
    
    subgrad = np.zeros(n)
    for k in active:
        # Term 1: b[m + k - (n-1)] for valid m
        # Valid when 0 <= m + k - (n-1) < n
        # => n-1-k <= m <= n-1 (intersected with [0, n-1])
        m_lo = max(0, n - 1 - k)
        m_hi = n - 1
        if m_lo <= m_hi:
            idx = m_lo + k - (n - 1)
            # b[idx : idx + (m_hi - m_lo + 1)]
            subgrad[m_lo:m_hi+1] += b[idx:idx + (m_hi - m_lo + 1)]
        
        # Term 2: -h[m - k + (n-1)] for valid m
        # Valid when 0 <= m - k + (n-1) < n
        # => k-(n-1) <= m <= k+n-2 (intersected with [0, n-1])
        m_lo2 = max(0, k - (n - 1))
        m_hi2 = min(n - 1, k + n - 2)
        if m_lo2 <= m_hi2:
            idx2 = m_lo2 - k + (n - 1)
            subgrad[m_lo2:m_hi2+1] -= h[idx2:idx2 + (m_hi2 - m_lo2 + 1)]
    
    subgrad /= len(active)
    return max_corr * dx, subgrad

Alternatively, I can just use the gradient of a smoothed max (like the softmax/log-sum-exp) which is smooth and easy to compute.

Let me use the log-sum-exp smoothing:

∂f/∂h[m] = (1/dx) * ∑_k exp(tau * c_k/dx) * ∂c_k/∂h[m] / ∑_k exp(tau * c_k/dx)

This is a weighted average of the individual gradients, with weights proportional to exp(tau * c_k/dx).

But computing all individual gradients is still O(n * n_lags) = O(n²).

Actually, there's a smarter way. Note that:
∂c_k/∂h[m] = b[m+k-(n-1)] - h[m-k+(n-1)] (with bounds)

So the weighted sum is:
∑_k w_k * b[m+k-(n-1)] - ∑_k w_k * h[m-k+(n-1)]

where w_k = exp(tau * c_k / dx) / Z.

The first sum is a convolution of w with b (reversed), and the second is a convolution of w with h. Both can be computed in O(n log n) using FFT!

Let me define:

- sum1[m] = ∑_k w[k] * b[m+k-(n-1)] (with bounds checking)
- sum2[m] = ∑_k w[k] * h[m-k+(n-1)] (with bounds checking)

sum1[m] = ∑_j w[j-m+(n-1)] * b[j] (where j and j-m+(n-1) are valid)

This is a cross-correlation of w and b, which can be done with FFT.

Similarly for sum2.

import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_box_sum(h, target_sum):
    n = len(h)
    if target_sum <= 0 or target_sum >= float(n):
        return np.clip(h, 0.0, 1.0)
    lo = -2.0
    hi = 2.0
    for _ in range(50):
        mid = (lo + hi) / 2.0
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def smoothed_grad(h, tau=10.0):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode="full")
    max_c = float(np.max(corr))
