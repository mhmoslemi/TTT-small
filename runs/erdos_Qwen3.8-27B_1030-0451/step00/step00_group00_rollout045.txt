import numpy as np
from scipy.optimize import minimize
import time

def project_hypersimplex(v, target):
    """Project v onto {h : 0 <= h[i] <= 1, sum(h) = target}"""
    n = len(v)
    lo = v.min() - 2.0
    hi = v.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(v - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2.0
    result = np.clip(v - t, 0.0, 1.0)
    # Small correction for floating point
    diff = target - result.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference among elements not at bounds
        free = (result > 1e-10) & (result < 1.0 - 1e-10)
        if free.sum() > 0:
            result[free] += diff / free.sum()
            result = np.clip(result, 0.0, 1.0)
    return result

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (corr * dx).max()

def logsumexp(vals, beta):
    x = vals * beta
    x_max = x.max()
    return (x_max + np.log(np.exp(x - x_max).sum())) / beta

def compute_smoothed_grad(h, dx, beta):
    """Compute gradient of logsumexp(corr*dx, beta)"""
    n_pts = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights
    x = corr * beta
    x -= x.max()
    w = np.exp(x)
    w /= w.sum()
    
    # Compute gradient
    grad = np.zeros(n_pts)
    omh = 1.0 - h
    n_lags = 2 * n_pts - 1
    for i in range(n_lags):
        wi = w[i]
        if wi < 1e-15:
            continue
        l = (n_pts - 1) - i  # lag for index i
        # Term 1: (1-h)[j+l] for valid j
        if l >= 0:
            valid = n_pts - l
            if valid > 0:
                grad[:valid] += wi * omh[l:]
        else:
            valid = n_pts + l
            if valid > 0:
                grad[-l:] += wi * omh[:valid]
        # Term 2: -h[j-l] for valid j
        if l >= 0:
            if l < n_pts:
                grad[l:] -= wi * h[:n_pts - l]
        else:
            valid = n_pts + l
            if valid > 0:
                grad[:valid] -= wi * h[-l:]
    
    return grad

def projected_gradient_optimize(h_init, dx, target, n_iter=5000, beta_start=50, beta_end=500, lr_start=0.05, lr_end=0.001):
    h = h_init.copy()
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for it in range(n_iter):
        # Schedule
        frac = it / max(n_iter - 1, 1)
        beta = beta_start + (beta_end - beta_start) * frac
        lr = lr_start * (lr_end / lr_start) ** frac
