import numpy as np
import time

def project_to_constraint(g, n):
    """Project g onto {0 <= x <= 1, sum(x) = n/2}"""
    target = n / 2.0
    
    def sum_at_tau(tau):
        h = np.clip(g - tau, 0, 1)
        return h.sum()
    
    # Bisection to find tau
    # sum_at_tau is decreasing in tau
    lo, hi = g.min() - 1, g.max() + 1
    for _ in range(60):
        mid = (lo + hi) / 2
        if sum_at_tau(mid) > target:
            lo = mid
        else:
            hi = mid
    tau = (lo + hi) / 2
    h = np.clip(g - tau, 0, 1)
    
    # Handle rounding to get exact sum
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Adjust the values closest to the boundary
        residual = diff
        # Find indices where 0 < h < 1 (interior points)
        interior = np.where((h > 1e-10) & (h < 1 - 1e-10))[0]
        if len(interior) > 0:
            # Distribute the residual
            per = residual / len(interior)
            h[interior] += per
            h = np.clip(h, 0, 1)
    
    return h

def compute_correlations(h):
    """Compute all correlation values c[m] * dx"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full")
    return c * dx, dx

def compute_subgradient(h, m_indices):
    """Compute subgradient at h for the given correlation indices m_indices"""
    n = len(h)
    g = np.zeros(n)
    for m in m_indices:
        s = m - (n - 1)  # shift
        for j in range(n):
            grad = 0.0
            js = j - s
            jp = j + s
            if 0 <= js < n:
                grad += (1 - h[js])
            if 0 <= jp < n:
                grad -= h[jp]
            g[j] += grad
    g /= len(m_indices)
    return g

        c5 = np.max(c5_vals)
        if c5 < best_c5:

            best_h = h.copy()

            if t_pol > dl - 0.5:
                t_pol = dl - 0.5
            h_p, c_p = grad_opt(bh, t_pol, lr=lr_try, pool=20, decay=0.9995)
            if c_p < bc:
                bc, bh = c_p, h_p.copy()

    return bh, bc, len(bh)
