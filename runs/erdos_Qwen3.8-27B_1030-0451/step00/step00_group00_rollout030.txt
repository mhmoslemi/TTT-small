def build_h(p):
    m = len(p)
    n = 2 * m
    h = np.empty(n)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h

def p_from_h(h):
    n = len(h)
    m = n // 2
    pf = h[:m].copy()
    pb = 1.0 - h[n-1:m-1:-1].copy()
    return 0.5 * (pf + pb)

def exact_c5(h):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(R) * 2.0 / n)

def obj_func(p, eta):
    h = build_h(p)
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = np.max(R)
    shifted = eta * (R - R_max)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    return 2.0 / n * (R_max + log_sum_exp) / eta

def obj_grad(p, eta):

    n = len(h)

    S = np.correlate(w, C_vec, mode='valid')
    return (2.0 / n) * (S[::-1][:m] - S[:m])

import numpy as np
from scipy.optimize import minimize
import time


def build_h(p):
    m = len(p)
    n = 2 * m
    h = np.empty(n)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h


def p_from_h(h):
    n = len(h)
    m = n // 2
    pf = h[:m].copy()
    pb = 1.0 - h[n-1:m-1:-1].copy()
    return 0.5 * (pf + pb)


def exact_c5(h):
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(R) * 2.0 / n)


def obj_func(p, eta):
    h = build_h(p)
    n = len(h)
    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = np.max(R)
    shifted = eta * (R - R_max)
    log_sum_exp = np.log(np.sum(np.exp(shifted)))
    return 2.0 / n * (R_max + log_sum_exp) / eta


def obj_grad(p, eta):
    h = build_h(p)
    n = len(h)
    m = len(p)
    R = np.correlate(h, 1.0 - h, mode='full')
    R_max = np.max(R)
    shifted = eta * (R - R_max)
    w = np.exp(shifted)
    w /= np.sum(w)
    C_vec = 1.0 - h + h[::-1]
    S = np.correlate(w, C_vec, mode='valid')
    return (2.0 / n) * (S[::-1][:m] - S[:m])


def coord_descent(p, max_time=60.0, t_start=None):
    if t_start is None:
        t_start = time.time()
    m = len(p)
    h = build_h(p)
    cur = exact_c5(h)
    step = 0.04
    min_step = 5e-6

    while step > min_step:
        improved = False
        for i in range(m):
            if time.time() - t_start > max_time:
                return p, cur
            best = cur
            best_val = p[i]
            for frac in [1.0, 0.5, 0.25]:
                d = step * frac
                for sign in [1, -1]:
                    nv = p[i] + sign * d
                    if nv < 0 or nv > 1:
                        continue
                    p[i] = nv
                    h2 = build_h(p)
                    c2 = exact_c5(h2)
                    if c2 < best - 1e-13:
                        best = c2
                        best_val = nv
            p[i] = best_val
            if best < cur - 1e-13:
                cur = best
                improved = True
        if not improved:
            step *= 0.5
    return p, cur


def lbgfb_optimize(p, eta, maxiter=300,
                   gtol=1e-10):
    m = len(p)
    try:
        res = minimize(obj_func, p, jac=obj_grad, args=(eta,),
                       method='L-BFGS-B', bounds=[(0.0, 1.0)] * m,
                       options={'maxiter': maxiter, 'gtol': gtol,
                                'ftol': 1e-12})
        return res.x
    except Exception:
        return p


def interpolate_to_resolution(p, n_new):
    h_old = build_h(p)
    n_old = len(h_old)
    x_old = np.arange(n_old) * 2.0 / n_old
    x_new = np.arange(n_new) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    p_new = p_from_h(h_new)
    np.clip(p_new, 0, 1, out=p_new)
    return p_new
