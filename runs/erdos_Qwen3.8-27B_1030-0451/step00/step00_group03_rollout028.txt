def exact_c5(h):

def project_mean_box(y):
    """Project y to {0<=x<=1, sum(x)=n/2}"""
    n = len(y)

    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add mass; find indices with room
            room = 1.0 - x
            idx = np.argsort(-room)
            add = residual
            for i in idx:
                if add <= 0:
                    break
                d = min(add, room[i])
                x[i] += d
                add -= d
        else:
            # Need to remove mass
            idx = np.argsort(x)
            remove = -residual
            for i in idx:
                if remove <= 0:
                    break

                remove -= d
    
    return x

            v = np.zeros(n)

def resample(h, target_n, rng):
    """Resample h to target_n points using linear interpolation"""
    n = len(h)

        return np.full(n_new, 0.5)
    x_old = (0.5 + np.arange(n)) * 2.0 / n
    x_new = (0.5 + np.arange(n_new)) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h, left=0.0, right=0.0)
    return project_mean_box(h_new)

def _slsqp_obj(x):
    if time.monotonic() > _S['deadline']:
        raise RuntimeError("timeout")
    return float(x[-1])

def _slsqp_obj_jac(x):
    if time.monotonic() > _S['deadline']:
        raise RuntimeError("timeout")
    n = _S['n']
    g = np.zeros(n + 1)
    g[n] = 1.0
    return g

def _slsqp_ineq(x):
    if time.monotonic() > _S['deadline']:
        raise RuntimeError("timeout")
    h_x = x[:-1]
    T = x[-1]
    corr = full_corr(h_x)
    return T - corr[_S['top_k']]

def _slsqp_eq(x):
    if time.monotonic() > _S['deadline']:
        raise RuntimeError("timeout")
    n = _S['n']
    return float(np.dot(x[:n], _S['eq_vec']) - n / 2.0)

def _slsqp_eq_jac(x):
    if time.monotonic() > _S['deadline']:
        raise RuntimeError("timeout")
    n = _S['n']
    g = np.zeros(n + 1)
    g[:n] = _S['eq_vec']
    return g

def slsqp_polish(h, max_time=25.0):
    n = len(h)
    if n > 160:
        return None, None
    _S['n'] = n
    _S['deadline'] = time.monotonic() + max_time
    _S['eq_vec'] = np.ones(n)
    corr = full_corr(h)
    num_c = min(n, 36)
    _S['top_k'] = np.argsort(corr)[-num_c:]
    x0 = np.zeros(n + 1)
    x0[:n] = h
    x0[n] = exact_c5(h) + 1e-4
    bounds = [(0.0, 1.0)] * n + [(0.0, 2.0)]
    constraints = [
        {'type': 'ineq', 'fun': _slsqp_ineq},
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac}
    ]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_jac,
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': 200, 'ftol': 1e-9})
        if np.all(np.isfinite(res.x)):
            h_new = project_mean_box(res.x[:n])
            c_new = exact_c5(h_new)
            if np.isfinite(c_new):
                return h_new, c_new
    except Exception:
        pass
    return None, None

def run(seed=42, budget_s=1100, **kwargs):
    _G['t_start'] = time.monotonic()
    _G['budget'] = budget_s
    _G['best_c'] = np.inf
    _G['best_h'] = None
    _G['last_improve'] = _G['t_start']
    rng = np.random.default_rng(seed)

    # Get initial
    h = None
    try:
        h = np.asarray(initial_h_values, dtype=np.float64).reshape(-1).copy()
    except (NameError, TypeError, AttributeError):
        pass
    if h is None or not np.all(np.isfinite(h)) or len(h) < 16:
        h = np.full(64, 0.5)
    h = project_mean_box(h)
    submit(h)
    n0 = len(h)

    # Try some initial transforms
    transforms = [h]
    transforms.append(h[::-1].copy())
    transforms.append(1.0 - h)
    transforms.append(1.0 - h[::-1].copy())
    for t in range(1, min(5, n0 // 4) + 1):
        transforms.append(np.roll(h, t).copy())
        transforms.append(np.roll(h, -t).copy())
    for tr in transforms:
        submit(tr)

    # Phase 1: Intensive local search on best (45% budget)
    t_p1 = _G['t_start'] + 0.45 * budget_s
    h_work = _G['best_h'].copy() if _G['best_h'] is not None else h
    h_work = local_search(h_work, t_p1, rng)

    # Phase 2: Resolution changes (20%)
    t_p2 = _G['t_start'] + 0.65 * budget_s
    dims = sorted(set([n0, max(32, n0 - 16), min(128, n0 + 16), min(96, n0 * 2)]))
    dims = [d for d in dims if d >= 16 and d <= 2500]
    for dim in dims:
        if time.monotonic() > t_p2:
            break
        if _G['best_h'] is not None and dim == len(_G['best_h']):
            continue
        h_try = resample(_G['best_h'], dim)
        remaining = max(1.0, (t_p2 - time.monotonic()))
        sub_end = time.monotonic() + min(remaining, 60.0)
        h_try = local_search(h_try, sub_end, rng)
        submit(h_try)

    # Phase 3: Focus on current best (20%)
    t_p3 = _G['t_start'] + 0.85 * budget_s
    h_work = _G['best_h'].copy() if _G['best_h'] is not None else h
    h_work = local_search(h_work, t_p3, rng)

    # Phase 4: SLSQP polish (10%)
    t_p4 = _G['t_start'] + 0.96 * budget_s
    if _G['best_h'] is not None and time.monotonic() < t_p4:
        rem = t_p4 - time.monotonic()
        if rem > 8.0:
            h_new, c_new = slsqp_polish(_G['best_h'].copy(), max_time=rem * 0.7)
            if h_new is not None:
                submit(h_new)
            # Second polish attempt if time
            if time.monotonic() < t_p4 - 5.0 and _G['best_h'] is not None:
                rem2 = min(15.0, t_p4 - time.monotonic() - 5.0)
                if rem2 > 5.0:
                    h_new2, _ = slsqp_polish(_G['best_h'].copy(), max_time=rem2)
                    if h_new2 is not None:
                        submit(h_new2)

    # Final validation
    if _G['best_h'] is None:
        _G['best_h'] = h.copy()
        _G['best_c'] = exact_c5(_G['best_h'])

    h_out = _G['best_h'].copy()
    c_out = exact_c5(h_out)
    return h_out, float(c_out), len(h_out)
