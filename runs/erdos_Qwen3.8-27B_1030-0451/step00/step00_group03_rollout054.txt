def project_feasible(y, target_sum):
    """Project y onto {h: 0<=h_i<=1, sum(h)=target_sum}"""
    n = len(y)
    # Bisection on alpha: find alpha such that sum(clip(y+alpha, 0, 1)) = target_sum
    lo = -y.max() - 1
    hi = 1 - y.min() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(y + mid, 0, 1).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2
    h = np.clip(y + alpha, 0, 1)
    # Fix residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            rooms = 1.0 - h
            idx = np.argsort(-rooms)
            for i in idx:
                step = min(rooms[i], residual)
                if step <= 0:
                    break
                h[i] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break

            idx = np.argsort(rooms)

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, target_sum):
    n = len(y)
    y = np.asarray(y, dtype=np.float64)
    if not np.all(np.isfinite(y)):
        y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo = -y.max() - 2.0
    hi = 1.0 - y.min() + 2.0
    for _ in range(70):
        mid = (lo + hi) * 0.5
        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            rooms = 1.0 - h
            idx = np.argsort(-rooms)
        else:
            residual = -residual
            rooms = h.copy()
            idx = np.argsort(rooms)
        for i in idx:
            step = min(rooms[i], residual)
            if step <= 1e-15:
                break
            h[i] += step if (target_sum - h.sum() + step > 0 or residual > 0) else -step
            residual -= step
            if abs(residual) < 1e-14:
                break
    return h

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr * dx))

def compute_correlations(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_single_lag(h, m, n, dx):
    grad = np.zeros(n, dtype=np.float64)
    if m <= n - 1:
        r = n - 1 - m
        L = n - r
        if L > 0:
            grad[:L] += (1.0 - h[r:r + L])
            if r > 0:
                grad[r:] += -h[:L]
    else:
        s = m - (n - 1)
        L = n - s
        if L > 0:
            grad[s:] += (1.0 - h[:L])
            grad[:L] += -h[s:s + L]
    return dx * grad

def compute_smoothed_grad(h, active_lags, tau, dx):
    n = len(h)
    corr = compute_correlations(h)
    f_vals = dx * corr[active_lags]
    f_max = np.max(f_vals)
    z = (f_vals - f_max) / tau
    z -= z.max()
    w = np.exp(z)
    w_sum = w.sum()
    if w_sum < 1e-300:
        w = np.zeros(len(active_lags))
        w[np.argmax(f_vals)] = 1.0
    else:
        w /= w_sum
    grad = np.zeros(n, dtype=np.float64)
    for i, m in enumerate(active_lags):
        if w[i] > 1e-10:
            grad += w[i] * grad_single_lag(h, m, n, dx)
    return grad, f_max + tau * np.log(w_sum)

def slsqp_polish(h0, active_lags, n, dx, target_sum, maxiter=200):
    t0 = eval_c5(h0)
    x0 = np.append(h0.copy(), t0 + 0.005)
    eq = EqCon(n, target_sum)
    batch = BatchIneq(active_lags, n, dx)
    constraints = [
        {'type': 'eq', 'fun': eq.fun, 'jac': eq.jac},
        {'type': 'ineq', 'fun': batch.fun, 'jac': batch.jac}
    ]
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    from functools import partial
    obj_fun = partial(slsqp_obj_fun, n=n)
    obj_jac = partial(slsqp_obj_jac, n=n)
    try:
        res = minimize(obj_fun, x0, jac=obj_jac, method='SLSQP',
                       bounds=bounds, constraints=constraints,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or res.status == 4:
            h_new = project_feasible(res.x[:n], target_sum)
            c_new = eval_c5(h_new)
            if c_new < t0 - 1e-10:
                return h_new, c_new

        pass
    return None, None

    x_new = np.linspace(0, old_n - 1, new_n)
    h_interp = np.interp(x_new, x_old, h_old)
    return project_feasible(h_interp, new_target)

    gamma = 0.01
    stall_count = 0
    t_start = time.time()

    for it in range(max_iter):
        if time.time() - t_start > time_limit:

        if it == 0:
            corr = compute_correlations(h)
            f_vals = dx * corr
            sorted_idx = np.argsort(-f_vals)[:n_active]
            active_lags = sorted_idx
        else:
            corr = compute_correlations(h)

            f_max = np.max(f_vals)
            active_lags = np.argsort(-f_vals)[:n_active]

        grad, smooth_val = compute_smoothed_grad(h, active_lags, tau, dx)
        gnorm = np.linalg.norm(grad)

        step_accepted = False

            step_size = gamma / (gnorm + 1e-15) * (0.5 ** trial)
            y = h - step_size * gnorm * grad

            stall_count = 0

                tau *= 1.3
            gamma *= 0.7

def active_slqp_iterate(h0, n, dx, target_sum, max_rounds=5, maxiter=100):
