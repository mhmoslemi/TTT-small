```python
import numpy as np
import time
from scipy.optimize import minimize

def build_h(p, n):
    """Build h from antisymmetric parameters p."""
    m = n // 2
    h = np.empty(n)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h

def objective_exact(h):
    """Compute C5 = dx * max(correlate(h, 1-h, 'full'))."""
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(R)

def corr_values(h):
    """Return the correlation vector R."""
    return np.correlate(h, 1.0 - h, mode="full")

def smooth_obj_and_grad(p, eta, n):
    """Smoothed log-sum-exp objective and its gradient."""
    m = len(p)
    h = build_h(p, n)
    dx = 2.0 / n
    R = corr_values(h)
    x = eta * dx * R
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    S = exp_x.sum()
    obj = (x_max + np.log(S)) / eta
    w = exp_x / S
    
    # Gradient via analytical formula
    h_pad = np.pad(h, (n-1, n-1), mode='constant')
    i_arr = np.arange(m)
    grad = np.zeros(m)
    for r in range(2 * n - 1):
        wr = w[r]
        if wr < 1e-15:
            continue
        grad += wr * ((1.0 - h_pad[i_arr + r]) 
                     - (1.0 - h_pad[(n-1) + r - i_arr]) 
                     - h_pad[2*(n-1) + i_arr - r] 
                     + h_pad[3*(n-1) - i_arr - r])
    grad *= dx
    return obj, grad

def coordinate_descent(p, n, max_rounds=20, tolerance=1e-10):
    """Exact coordinate descent exploiting linearity in each p[i]."""
    m = len(p)
    h = build_h(p, n)
    dx = 2.0 / n
    R = corr_values(h)
    current_c5 = dx * R.max()
    
    for round_idx in range(max_rounds):
        improved = False
        for i in range(m):
            # Compute R with p[i] = 0 and p[i] = 1
            p_save = p[i]
            p[i] = 0.0
            h0 = build_h(p, n)
            R0 = corr_values(h0)
            p[i] = 1.0
            h1 = build_h(p, n)
            R1 = corr_values(h1)
            dR = R1 - R0
            
            # Minimize max_k (R0[k] + t * dR[k]) over t in [0,1]
            # Grid search
            ts = np.linspace(0, 1, 101)
            vals = np.max(R0[None, :] + ts[:, None] * dR[None, :], axis=1)
            best_t = ts[vals.argmin()]
            best_val = vals.min()
            
            if best_val * dx < current_c5 - tolerance:
                p[i] = best_t
                # Update R
                R = R0 + best_t * dR
                current_c5 = dx * R.max()
                improved = True
            else:
                p[i] = p_save
        
        if not improved:
            break
    
    return p, current_c5

def run(seed=42, budget_s=1100, **kwargs):
```

```python
def build_h(p, n):
    m = n // 2

def corr_values(h):

    m = len(p)
    h = build_h(p, n)
    dx = 2.0 / n
    R = corr_values(h)
    x = eta * dx * R
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    S = exp_x.sum()
    obj = (x_max + np.log(S)) / eta
    w = exp_x / S
    h_pad = np.pad(h, (n - 1, n - 1), mode='constant')
    i_arr = np.arange(m)
    grad = np.zeros(m)
    nm1 = n - 1

                      - h_pad[2 * nm1 + i_arr - r]
                      + h_pad[3 * nm1 - i_arr - r])
    grad *= dx
    return obj, grad
```
