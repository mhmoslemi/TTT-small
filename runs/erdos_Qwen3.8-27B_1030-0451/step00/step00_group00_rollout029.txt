import time

def compute_c5(h):

    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

    target = n / 2.0
    lo = float(np.min(h)) - 1.0
    hi = float(np.max(h)) + 1.0
    for _ in range(80):

            lo = mid
        else:

    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

    W = np.zeros(n)
    W[0] = weights[n - 1]
    for m in range(1, n):
        W[m] = weights[n - 1 + m] + weights[n - 1 - m]
    return W

    h_rev = h[::-1]
    first = np.convolve(W, one_minus_h, mode='full')[:n]
    second = np.convolve(W, h_rev, mode='full')[:n][::-1]

    first = np.convolve(W, one_minus_h, mode='full')[:n]
