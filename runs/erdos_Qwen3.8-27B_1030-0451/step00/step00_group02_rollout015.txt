            h[idx] -= residual
        else:
            idx = int(np.argmin(h))
            h[idx] -= residual

def line_search(h, d, c_cur, max_alpha=1.0):
    """Line search along direction d from h, maintaining feasibility"""
    n = len(h)
    # Find max alpha for feasibility
    alpha_max = max_alpha
    pos_mask = d > 1e-12
    neg_mask = d < -1e-12
    if np.any(pos_mask):
        alpha_max = min(alpha_max, (1.0 - h[pos_mask]) / d[pos_mask]).min()
    if np.any(neg_mask):
        alpha_max = min(alpha_max, h[neg_mask] / (-d[neg_mask])).min()
    alpha_max = max(0, min(alpha_max, max_alpha))
    
    if alpha_max < 1e-10:
        return h, c_cur, 0.0
    
    # Try a set of alpha values
    alphas = [alpha_max * f for f in [1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.08, 0.04, 0.02]]
    alphas = [min(a, alpha_max) for a in alphas if a > 1e-10]
    
    best_h = h.copy()
    best_c = c_cur
    best_alpha = 0.0
    
    for a in alphas:
        h_try = h + a * d
        if np.any(h_try < -1e-12) or np.any(h_try > 1 + 1e-12):
            continue
        h_try = np.clip(h_try, 0, 1)
        c_try = exact_score(h_try)
        if c_try < best_c - 1e-12:
            best_c = c_try
            best_h = h_try.copy()
            best_alpha = a
    return best_h, best_c, best_alpha

def exact_score(h):

def ensure_feasible(h):
    n = len(h)
    target = n / 2.0
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    s = float(np.sum(h))
    for _ in range(500):
        diff = s - target
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h > 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(abs(diff) / cnt, 0.49 * float(np.min(h[mask])))
            h[mask] -= step
        else:
            mask = h < 1.0 - 1e-15
            cnt = int(np.sum(mask))
            if cnt == 0:
                break
            step = min(abs(diff) / cnt, 0.49 * float(1.0 - np.max(h[mask])))
            h[mask] += step
        s = float(np.sum(h))
    h = np.clip(h, 0.0, 1.0)
    # Final fix
    resid = float(np.sum(h)) - target

            h[idx] = max(0.0, h[idx] - resid)
        else:

    return h


def lse_value(h, tau):

    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = float(np.max(vals))
    return m + tau * float(np.log(np.sum(np.exp((vals - m) / tau))))


def lse_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    vals = corr * dx
    m = float(np.max(vals))

    grad = np.zeros(n)

            grad += w[k] * dx * (1.0 - 2.0 * h)

            grad[:n - ell] += w[k] * dx * (1.0 - h[ell:])
            grad[ell:] -= w[k] * dx * h[:n - ell]
        else:
            e = -ell
            grad[e:] += w[k] * dx * (1.0 - h[:n - e])
