Let me also think about what makes this problem interesting. The Erdős problem is about finding a set A subset [0,1] of measure 1/2 such that the maximum overlap of A with its translates is minimized. The step function h represents the indicator of such a set (relaxed to [0,1] values).

For a binary (0/1) function with exactly n/2 ones, the overlap for lag k is:
z_k = (number of positions where h=1 and h shifted by k = 0) * dx

This is related to autocorrelation. The problem is equivalent to finding a binary sequence of length n with exactly n/2 ones that minimizes the maximum "cross-correlation" with its complement.

Actually, for binary sequences, h(1-h) is zero whenever h=h', so z_k counts the positions where h and h' differ (one is 1, other is 0). This is related to the discrepancy of the sequence at different shifts.

import numpy as np
import time
from scipy.optimize import minimize

def evaluate(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_onto_box_sum(x, target):
    """Project x onto {0 <= y <= 1, sum(y) = target}."""
    n = len(x)
    # Bisection on lambda: y = clip(x - lambda, 0, 1), find lambda s.t. sum(y) = target
    lo, hi = x.min() - target - 1.0, x.max() - target + 1.0
    # Actually, sum(clip(x-lambda, 0, 1)) is decreasing in lambda
    # At lambda = min(x) - 1 (very negative), sum = n
    # At lambda = max(x) (very positive), sum = 0
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(200):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Final repair
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # Distribute evenly among non-boundary elements
        free = (y > 1e-10) & (y < 1.0 - 1e-10)
        if free.sum() > 0:
            y[free] += diff / free.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def smooth_max_and_grad(h, tau):
    """Compute smooth-max objective and its gradient."""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full") * dx
    m = len(z)  # 2n-1
    
    # Stable softmax
    z_max = z.max()
    shifted = (z - z_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    exp_shifted = np.exp(shifted)
    p = exp_shifted / exp_shifted.sum()
    
    F = z[0]  # placeholder, we'll compute properly
    F = z_max + tau * np.log(exp_shifted.sum())
    
    # Compute gradient
    # grad[j] = dx * sum_k p[k] * [(1-h[j-k]) * I(0<=j-k<n) - h[j+k] * I(0<=j+k<n)]
    # where k = idx - (n-1) is the shift for index idx in z
    
    grad = np.zeros(n)
    for idx in range(m):
        k = idx - (n - 1)  # shift
        if abs(p[idx]) < 1e-15:
            continue
        w = p[idx]
        if k == 0:
            grad += w * (1.0 - 2.0 * h)
        elif k > 0:
            # (1-h[j-k]) for j in [k, n-1]: contributes (1-h[0:n-k]) to positions [k:n]
            grad[k:] += w * (1.0 - h[:n-k])
            # -h[j+k] for j in [0, n-1-k]: contributes -h[k:n] to positions [0:n-k]
            grad[:n-k] -= w * h[k:]
        else:
            a = -k
            # (1-h[j-a]) for j in [a, n-1]: contributes (1-h[a:n]) to positions [a:n]... 
            # Wait: (1-h[j-k]) = (1-h[j+a]) for j in [0, n-1-a]
            # So this contributes to positions [0, n-1-a]
            grad[:n-a] += w * (1.0 - h[a:])
            # -h[j+k] = -h[j-a] for j in [a, n-1]
            grad[a:] -= w * h[:n-a]
    
    grad *= dx
    return F, grad

def project_box_sum(x, target):
    """Project x onto {0 <= y <= 1, sum(y) = target} using bisection."""
    n = len(x)
    # sum(clip(x - lambda, 0, 1)) is decreasing in lambda
    lo = min(0.0)  # lambda very negative -> all y=1 -> sum=n
    hi = max(2.0)  # lambda very positive -> all y=0 -> sum=0

        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

            y = np.clip(y, 0.0, 1.0)

    """Compute the log-sum-exp smooth-max and its gradient."""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full") * dx
    z_max = z.max()
    shifted = (z - z_max) / tau

    ess = e.sum()
