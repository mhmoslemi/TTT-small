def exact_score(h):

    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def project_box_sum(h, target_sum, tol=1e-10, max_iter=200):
    """Project h onto {x : 0 <= x <= 1, sum(x) = target_sum}"""
    n = len(h)
    x = np.clip(h.astype(np.float64), 0.0, 1.0)
    # Water-filling: find alpha such that sum(clip(x + alpha, 0, 1)) = target_sum
    current_sum = x.sum()
    if abs(current_sum - target_sum) < tol:
        return x
    # Bisection for alpha
    lo = -max(x) - 1.0
    hi = 1.0 - min(x) + 1.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        trial = np.clip(x + mid, 0.0, 1.0)
        s = trial.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fine correction
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        # Distribute residual among non-boundary elements
        free = (x > 1e-10) & (x < 1.0 - 1e-10)
        if free.any():
            adj = residual / free.sum()
            x[free] += adj
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_for_lag(h, ell):
    """Compute gradient of c[k] (where lag = ell) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        L = n - ell
        if L > 0:
            g[:L] += (1.0 - h[ell:])
            g[ell:] -= h[:L]
    else:
        a = -ell
        L = n - a
        if L > 0:
            g[a:] += (1.0 - h[:L])
            g[:L] -= h[a:]
    return g

def compute_subgradient(h, active_lags, weights=None):
    """Compute weighted sum of gradients for active lags."""
    n = len(h)
    g = np.zeros(n)
    if weights is None:
        weights = np.ones(len(active_lags)) / len(active_lags)
    for w, ell in zip(weights, active_lags):
        g += w * grad_for_lag(h, ell)
    return g

def find_active_lags(h, K=20, gap_frac=0.01):
    """Find lags near the maximum."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode="full")
    z_max = np.max(c)
    gap = max(1e-10, gap_frac * z_max)
    threshold = z_max - gap
    active = np.where(c >= threshold)[0]
    if len(active) > K:
        # Take top K by value
        idx = np.argsort(-c[active])[:K]
        active = active[idx]
    # Always include the argmax
    argmax_idx = np.argmax(c)
    if argmax_idx not in active:
        active = np.append(active, argmax_idx)
    # Convert to lags
    lags = (active - (n - 1)).astype(int)
    return lags, c, z_max

def slp_step(h, rho, K_active=20, gap_frac=0.01):
    """One SLP iteration: linearize active constraints, solve LP, return candidate."""
    n = len(h)
    c_full = np.correlate(h, 1.0 - h, mode="full")
    z_max = np.max(c_full)
    gap = max(1e-10, gap_frac * z_max)
    threshold = z_max - gap
    
    active_indices = np.where(c_full >= threshold)[0]
    if len(active_indices) > K_active:
        idx = np.argsort(-c_full[active_indices])[:K_active]
        active_indices = active_indices[idx]
    
    # Build LP
    # Variables: y[0..n-1], t
    # Minimize t
    # Subject to:
    #   sum(y) = n/2
    #   h[i]-rho <= y[i] <= h[i]+rho (and 0<=y[i]<=1)
    #   g_k · y - t <= g_k · h - c[k] for k in active
    #   t >= 0
    
    K = len(active_indices)
    
    # Objective
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag k: g_k · y - t <= g_k · h - c[k]
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for i, q in enumerate(active_indices):
        ell = q - (n - 1)
        g = grad_for_lag(h, ell)
        A_ub[i, :n] = g
        A_ub[i, n] = -1.0
        b_ub[i] = g @ h - c_full[q]
    
    # Equality constraint: sum(y) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lower = np.maximum(0.0, h - rho)
    upper = np.minimum(1.0, h + rho)
    lower = np.append(lower, 0.0)
    upper = np.append(upper, z_max + 1.0)  # t upper bound (loose)
    
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=list(zip(lower, upper)), method='highs')
        if res.success:
            y = res.x[:n]
            # Project to ensure exact feasibility
            y = project_box_sum(y, n / 2.0)
            return y, res.fun
        else:
            return None, None
    except:
        return None, None

def line_search(h, direction, current_score, max_alpha=1.0):
    """Line search along h + alpha * direction."""
    best_alpha = 0.0
    best_score = current_score
    alphas = [0.9, 0.7, 0.5, 0.35, 0.25, 0.15, 0.1, 0.05, 0.02, 0.01]

        s = exact_score(h_new)
        if s < best_score - 1e-12:
            best_score = s
            best_alpha = alpha
    if best_alpha > 0:
        return h + best_alpha * direction, best_score
    return None, None

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c)) * dx

def project_box_sum(h, target_sum, max_iter=300):
    n = len(h)
    x = np.clip(h.astype(np.float64), 0.0, 1.0)
    current_sum = x.sum()
    if abs(current_sum - target_sum) < 1e-12:
        return x
    lo = -max(x.max(), 0.0) - 2.0
    hi = 1.0 - min(x.min(), 1.0) + 2.0
    for _ in range(max_iter):
        mid = (lo + hi) / 2.0
        trial = np.clip(x + mid, 0.0, 1.0)
        if trial.sum() < target_sum:
            lo = mid
        else:
            hi = mid
    x = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - x.sum()
    if abs(residual) > 1e-12:
        free = (x > 1e-12) & (x < 1.0 - 1e-12)
        if free.any():
            x[free] += residual / free.sum()
            x = np.clip(x, 0.0, 1.0)
    return x

def grad_for_lag(h, ell):
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        L = n - ell
        if L > 0:
            g[:L] += (1.0 - h[ell:])
            g[ell:] -= h[:L]
    else:
        a = -ell
        L = n - a
        if L > 0:
            g[a:] += (1.0 - h[:L])
            g[:L] -= h[a:]
    return g

    h_new = np.repeat(h, factor)
    h_new = project_box_sum(h_new, n_new / 2.0)
    return h_new

def downsample(h, factor):
    n_old = len(h)
    if n_old % factor != 0:
        return h
    n_new = n_old // factor
    h_blocks = h.reshape(n_new, factor)
    h_new = h_blocks.mean(axis=1)
    h_new = project_box_sum(h_new, n_new / 2.0)
    return h_new

def random_perturbation(h, sigma, rng):
    n = len(h)
    noise = rng.normal(0, sigma, size=n)
    h_new = h + noise
    h_new = project_box_sum(h_new, n / 2.0)
    return h_new

def run(seed=42, budget_s=1100, **kwargs):
