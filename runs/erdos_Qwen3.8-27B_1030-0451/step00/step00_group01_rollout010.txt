import numpy as np
import time

def _project_feasible(z, target_sum):
    """Project z onto {h: 0<=h_i<=1, sum(h_i)=target_sum} via bisection on clipping offset."""
    n = len(z)
    target_sum = float(target_sum)
    # Find lambda such that sum(clip(z - lambda, 0, 1)) = target_sum
    lo = z.min() - 1.5
    hi = z.max() + 1.5
    # f(lam) = sum(clip(z - lam, 0, 1)) is decreasing in lam
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.clip(z - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(z - lam, 0.0, 1.0)
    # Fix residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-14:
        # Find coordinates with room
        if residual > 0:
            mask = h < 1.0 - 1e-12
        else:
            mask = h > 1e-12
        if mask.sum() > 0:
            idxs = np.where(mask)[0]
            per = residual / len(idxs)
            h[idxs] += per
            h = np.clip(h, 0.0, 1.0)
            # Second pass for tiny residual
            residual2 = target_sum - h.sum()
            if abs(residual2) > 1e-14 and idxs.size > 0:
                h[idxs[0]] += residual2
                h = np.clip(h, 0.0, 1.0)
    return h

def _exact_score(h):
    """Compute C5 = max_k sum(h_i*(1-h_{i+k})) * dx for all valid shifts k."""
    n = len(h)
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(one_minus_h, h, mode='full')
    # The correlation gives sum over valid i of one_minus_h[i]*h[i+k] for various k
    # We need max of sum h[i]*(1-h[i+k]) which is same as correlate(h, 1-h)
    # np.correlate(a, b, 'full')[k] = sum_i a[i] * b[i + k - (len(b)-1)]
    # Let's use the correct orientation: we want sum_i h[i] * (1-h[i+k])
    # This equals np.correlate(h, 1-h, mode='full') with appropriate indexing
    corr2 = np.correlate(h, one_minus_h, mode='full')
    # corr2[m] = sum_i h[i] * (1-h)[i - (m - (n-1))] -- let me just take max of both
    val = max(corr.max(), corr2.max()) * dx
    return val

def _exact_score_direct(h):
    """Direct exact scoring using np.correlate as specified in the problem."""
    n = len(h)
    dx = 2.0 / n
    res = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(res) * dx)

def _build_basis(n, d):
    """Build d Fourier modes (sin/cos pairs) over n points, excluding constant."""
    x = (np.arange(n) + 0.5) / n  # centered positions in [0,1]
    B = np.zeros((n, d))
    cols = 0
    freq = 1
    while cols < d and freq <= max(d, n // 2):
        for fn in (np.sin, np.cos):
            if cols >= d:
                break
            B[:, cols] = fn(2.0 * np.pi * freq * x)
            cols += 1
        freq += 1
    # Normalize columns
    for j in range(min(cols, d)):
        amax = np.abs(B[:, j]).max()
        if amax > 1e-12:
            B[:, j] /= amax
    return B[:, :min(cols, d)]

def _decode(anchor, basis, coeffs, scale):
    """Decode latent coefficients to feasible vector."""
    delta = (basis @ coeffs) * scale
    z = anchor + delta
    target_sum = len(anchor) / 2.0
    return _project_feasible(z, target_sum)

def _resample_h(h, n_new):
    """Resample h from its native resolution to n_new points via linear interpolation."""
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(x_new, x_old, h)
    target_sum = n_new / 2.0
    h_new = _project_feasible(h_new, target_sum)
    return h_new
