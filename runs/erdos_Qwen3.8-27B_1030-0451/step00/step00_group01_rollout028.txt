    """Find best α for transferring mass from p to q."""
    hp = h[p]
    hq = h[q]
    
    # Feasible range for α
    alpha_lo = max(hp - 1.0, -hq)
    alpha_hi = min(hp, 1.0 - hq)

        return 0.0, evaluate_c5(h, n)
    
    d = np.zeros(n)
    d[p] = -1.0
    d[q] = 1.0

    b_1h = np.correlate(d, 1.0 - h, mode="full")
    h_d = np.correlate(h, d, mode="full")
    B = b_1h - h_d
    d_d = np.correlate(d, d, mode="full")
    C = -d_d

            best_alpha = a

        vals = A + a * B + a * a * C

    """Find α in [alpha_lo, alpha_hi] minimizing max_l (A_l + B_l*α + C_l*α²)."""
    candidates = [alpha_lo, alpha_hi, 0.0]
    
    # Vertex points (where derivative of individual quadratic = 0)
    for l in range(len(A)):
        if abs(C[l]) > 1e-15:
            av = -B[l] / (2.0 * C[l])
            if alpha_lo <= av <= alpha_hi:
                candidates.append(av)
    
    # Top K functions at α=0
    vals0 = A  # value at α=0
    # Rank by value (highest first)
    top_idx = np.argsort(-vals0)[:K]
    
    # Intersections of top K functions
    for i in range(len(top_idx)):
        for j in range(i+1, len(top_idx)):
            li, lj = top_idx[i], top_idx[j]
            # Solve (A_i + B_i α + C_i α²) = (A_j + B_j α + C_j α²)
            # (C_i - C_j) α² + (B_i - B_j) α + (A_i - A_j) = 0
            a2 = C[li] - C[lj]
            a1 = B[li] - B[lj]
            a0 = A[li] - A[lj]
            
            if abs(a2) < 1e-15:
                # Linear: a1 α + a0 = 0
                if abs(a1) > 1e-15:
                    root = -a0 / a1
                    if alpha_lo <= root <= alpha_hi:
                        candidates.append(root)
            else:
                disc = a1*a1 - 4*a2*a0
                if disc >= 0:
                    sqrt_disc = np.sqrt(disc)
                    r1 = (-a1 + sqrt_disc) / (2*a2)
                    r2 = (-a1 - sqrt_disc) / (2*a2)
                    if alpha_lo <= r1 <= alpha_hi:
                        candidates.append(r1)
                    if alpha_lo <= r2 <= alpha_hi:
                        candidates.append(r2)
    
    # Also add some grid points for safety
    ngrid = 11
    grid = np.linspace(alpha_lo, alpha_hi, ngrid)
    candidates.extend(grid.tolist())
    
    # Evaluate all candidates
    best_a = 0.0
    best_val = np.max(A)
    for a in candidates:
        a = np.clip(a, alpha_lo, alpha_hi)
        val = np.max(A + a*B + a*a*C)
        if val < best_val:
            best_val = val
            best_a = a

def evaluate_c5(h, n):
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode="full"))) * dx


def fix_feasibility(h, n):
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-14:
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        # Second pass for any residual
        diff2 = target - np.sum(h)
        if abs(diff2) > 1e-15:
            h += diff2 / n
            h = np.clip(h, 0.0, 1.0)
    return h


def upsample(h, n_old, n_new):
    x_old = np.linspace(0.0, 2.0, n_old)
    x_new = np.linspace(0.0, 2.0, n_new)
    h_new = np.interp(x_new, x_old, h)
    return fix_feasibility(h_new, n_new)


def line_search(A, B, C, alpha_lo, alpha_hi, K=25):
    if alpha_hi - alpha_lo < 1e-14:
        mid = (alpha_lo + alpha_hi) / 2.0
        return mid, float(np.max(A + mid * B + mid * mid * C))

    candidates = [alpha_lo, alpha_hi]
    if alpha_lo < 0.0 < alpha_hi:
        candidates.append(0.0)

    # Quadratic vertices where C is nonzero
    nz_c = np.where(np.abs(C) > 1e-14)[0]
    for l in nz_c:
        av = -B[l] / (2.0 * C[l])
        if alpha_lo - 1e-12 <= av <= alpha_hi + 1e-12:
            candidates.append(float(np.clip(av, alpha_lo, alpha_hi)))

    # Top-K intersections
    K_use = min(K, len(A))
    top_idx = np.argsort(-A)[:K_use]

        li = int(top_idx[i])
        for j in range(i + 1, len(top_idx)):
            lj = int(top_idx[j])
            a2 = C[li] - C[lj]

            a0 = A[li] - A[lj]

    # Grid points for safety
