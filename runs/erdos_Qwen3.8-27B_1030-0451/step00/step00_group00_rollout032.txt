def project_to_simplex_box(h, target_sum):
    """Project h onto {x: 0 <= x <= 1, sum(x) = target_sum}"""
    # Binary search on lambda
    def sum_clamped(lam):
        return np.clip(h - lam, 0, 1).sum()
    
    lo, hi = h.min() - target_sum, h.max() + target_sum
    for _ in range(100):
        mid = (lo + hi) / 2
        if sum_clamped(mid) > target_sum:

        else:

import numpy as np
import time

def project_to_constraint(h, target_sum):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = target_sum}"""
    # Use bisection on the Lagrange multiplier
    def s(lam):
        return np.clip(h - lam, 0.0, 1.0).sum()
    
    # Find range
    lo = -10.0
    hi = 10.0
    while s(lo) < target_sum:
        lo -= 1.0
    while s(hi) > target_sum:
        hi += 1.0
    
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if s(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5_values(h):
    """Compute all c[k] * dx values."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def gradient_component(h, m):
    """
    Gradient of (c[k] * dx) where k corresponds to shift m = k - (n-1).
    c[k] = sum_i h[i] * (1 - h[i+m]) for valid i.
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    # For each j, compute ∂c[k]/∂h[j]
    # = (1 - h[j+m]) if 0 <= j+m < n, minus h[j-m] if 0 <= j-m < n
    # All times dx
    
    # First part: (1 - h[j+m]) for valid j+m
    valid1 = (m >= 0)
    if m >= 0:
        # j ranges from 0 to n-1-m
        grad[:n-m] += (1.0 - h[m:n])
    elif m < 0:
        # j+m >= 0 means j >= -m, j ranges from -m to n-1
        grad[-m:] += (1.0 - h[:n+m])
    
    # Second part: -h[j-m] for valid j-m
    if m >= 0:
        # j-m >= 0 means j >= m, j ranges from m to n-1
        grad[m:] -= h[:n-m]
    elif m < 0:
        # j-m < n means j < n+m, j ranges from 0 to n+m-1
        grad[:n+m] -= h[-m:]
    
    return grad * dx

def optimize_subgradient(h0, max_iters=5000, initial_lr=0.05, verbose=False):
    n = len(h0)
    target_sum = n / 2.0
    h = h0.astype(float).copy()
    # Ensure feasibility
    h = project_to_constraint(h, target_sum)
    
    best_h = h.copy()
    best_c5 = np.max(compute_c5_values(h))
    
    lr = initial_lr
    patience = 0
    
    for it in range(max_iters):
        c_vals = compute_c5_values(h)
        c5 = c_vals.max()
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0

            patience += 1

            break

        k_star = np.argmax(c_vals)
        m = k_star - (n - 1)
        grad = gradient_component(h, m)
        
        # Normalize gradient to avoid scale issues
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:

        h_new = h - lr * grad
        h_new = project_to_constraint(h_new, target_sum)
        
        c5_new = np.max(compute_c5_values(h_new))
        if c5_new < c5:
            h = h_new
            lr = min(lr * 1.02, 1.0)
        else:
            lr *= 0.7

    n = len(h)
    def s(lam):
        return np.clip(h - lam, 0.0, 1.0).sum()
    lo, hi = -5.0, 5.0
    for _ in range(30):
        if s(lo) > target_sum: break
        lo -= 1.0
    for _ in range(30):
        if s(hi) < target_sum: break
        hi += 1.0
    for _ in range(80):

        if s(mid) > target_sum: lo = mid
        else: hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def get_c5_values(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if m >= 0:
        g[:n-m] += (1.0 - h[m:])
        g[m:] -= h[:n-m]
    else:
        p = -m
        g[p:] += (1.0 - h[:n-p])
        g[:n-p] -= h[p:]
    return g * dx
