def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(corr))

def project_constraint(h, target_sum):
    n = len(h)
    if np.sum(h) > target_sum + 0.5:
        lo, hi = -2.0, float(h.max())
    elif np.sum(h) < target_sum - 0.5:
        lo, hi = float(h.min()), 2.0
    else:
        lo, hi = float(h.min()) - 1.0, float(h.max()) + 1.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = 0.5 * (lo + hi)
    return np.clip(h - lam, 0.0, 1.0)

def soft_max_weights(corr, tau):
    m = np.max(corr)
    x = tau * (corr - m)
    x = np.clip(x, -400.0, 0.0)
    ex = np.exp(x)
    sx = ex.sum()
    w = ex / sx
    val = m + np.log(sx) / tau
    return val, w

def compute_gradient(h, weights):

    center = n - 1
    grad = np.zeros(n)
    omh = 1.0 - h
    j = np.arange(n)

            continue

            grad += wk * dx * (1.0 - 2.0 * h)

            msk1 = j >= lag
            msk2 = j <= n - 1 - lag
            grad[msk1] += wk * dx * omh[j[msk1] - lag]
            grad[msk2] -= wk * dx * h[j[msk2] + lag]
        else:

            msk2 = j >= m
            grad[msk1] += wk * dx * omh[j[msk1] + m]
