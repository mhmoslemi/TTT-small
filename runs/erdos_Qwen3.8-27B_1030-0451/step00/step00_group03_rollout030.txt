def exact_c5(h):
    """Compute exact C5 value for a step function h."""
    n = len(h)
    if n == 0:
        return np.inf
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    return float(np.max(c))

def project_mean_box(y, n_target=None):
    """Project y to {x: 0<=x_i<=1, sum(x)=n/2} where n=len(y) or n_target."""
    y = np.asarray(y, dtype=np.float64).copy()
    n = len(y)
    if not np.isfinite(y).all():
        y[~np.isfinite(y)] = 0.5
    target = n / 2.0
    
    # Find alpha such that sum(clip(y + alpha, 0, 1)) = target
    # phi(alpha) = sum(clip(y_i + alpha, 0, 1)) is monotone nondecreasing
    lo = -1.0 - np.max(y)
    hi = 1.0 - np.min(y)
    
    # Bisection
    for _ in range(100):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(y + mid, 0.0, 1.0))
        if val < target:
            lo = mid
        else:
            hi = mid
    
    x = np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(x)
    if abs(residual) > 1e-12:
        if residual > 0:
            # Need to add residual; find indices with room
            for i in range(n):
                room = 1.0 - x[i]
                if room > 1e-15:
                    add = min(room, residual)
                    x[i] += add
                    residual -= add
                    if abs(residual) < 1e-15:
                        break
        else:
            for i in range(n):
                if x[i] > 1e-15:
                    rem = min(x[i], -residual)
                    x[i] -= rem
                    residual += rem
                    if abs(residual) < 1e-15:
                        break
    return x

def resample_h(h, n_new):
    """Resample h to length n_new using interpolation, then project."""
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    # Center of each bin
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_interp = np.interp(x_new, x_old, h)
    return project_mean_box(h_interp)

def line_search_pair(h, src, dst, n_samples=16):
    """Line search along pair transfer direction src→dst.
    Returns (best_h, best_c) or (None, current_c) if no improvement."""
    n = len(h)
    # Direction: v[src] = +1, v[dst] = -1
    # h_new[t] = h + t * v
    # Feasible: 0 <= h[src]+t <= 1, 0 <= h[dst]-t <= 1
    # => -h[src] <= t <= 1-h[src] and h[dst]-1 <= t <= h[dst]
    t_lo = max(-h[src], h[dst] - 1.0)
    t_hi = min(1.0 - h[src], h[dst])
    if t_hi - t_lo < 1e-12:
        return None, exact_c5(h)
    
    current_c = exact_c5(h)
    best_c = current_c
    best_t = 0.0
    
    # Evaluate at multiple points
    if n_samples <= 2:
        t_vals = np.array([t_lo, t_hi])
    else:
        t_vals = np.linspace(t_lo, t_hi, n_samples)
    
    for t in t_vals:
        h_try = h.copy()
        h_try[src] += t
        h_try[dst] -= t
        # Clip for safety (shouldn't be needed if interval is correct)
        h_try = np.clip(h_try, 0.0, 1.0)
        c_try = exact_c5(h_try)
        if c_try < best_c:
            best_c = c_try
            best_t = t
    
    if best_t == 0.0:
        return None, current_c
    
    h_best = h.copy()
    h_best[src] += best_t
    h_best[dst] -= best_t
    h_best = np.clip(h_best, 0.0, 1.0)
    return h_best, best_c

    """Fast line search using analytical correlation change."""
    n = len(h)
    dx = 2.0 / n
    t_lo = max(-h[src], h[dst] - 1.0)
    t_hi = min(1.0 - h[src], h[dst])
    if t_hi - t_lo < 1e-12:
        return None, None
    
    # Compute current correlations
    R = np.correlate(h, 1.0 - h, mode="full")  # length 2n-1
    
    # Compute slope for each lag
    # s[m] where m = 0..2n-2, k = m - (n-1)
    k_vals = np.arange(2*n-1) - (n-1)
    # s[m] = h[dst+k] + h[dst-k] - h[src+k] - h[src-k] (out of bounds = 0)
    s = np.zeros(2*n-1)
    for m in range(2*n-1):
        k = k_vals[m]
        val = 0.0
        # h[dst+k]
        idx = dst + k
        if 0 <= idx < n:
            val += h[idx]
        # h[dst-k]
        idx = dst - k
        if 0 <= idx < n:
            val += h[idx]
        # -h[src+k]
        idx = src + k
        if 0 <= idx < n:
            val -= h[idx]
        # -h[src-k]
        idx = src - k
        if 0 <= idx < n:
            val -= h[idx]
        s[m] = val
    
    # Now f(t) = max(R + t*s) * dx
    # Minimize over [t_lo, t_hi]
    # f is convex in t
    best_c = np.max(R) * dx  # current value
    best_t = 0.0
    
    # Try candidate t values
    t_candidates = np.linspace(t_lo, t_hi, n_eval)
    for t in t_candidates:
        c_try = np.max(R + t * s) * dx
        if c_try < best_c:
            best_c = c_try
            best_t = t
    
    # Refine near best_t
    if 0 < best_t < t_hi - t_lo:
        margin = (t_hi - t_lo) / n_eval
        t_fine = np.linspace(max(t_lo, best_t - margin), min(t_hi, best_t + margin), 10)

            if c_try < best_c:

                best_t = t

    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    resid = target - np.sum(x)
    if resid > 0:
        for i in range(n):
            if resid <= 0: break
            add = min(1.0 - x[i], resid)
            x[i] += add
            resid -= add
    elif resid < 0:
        for i in range(n):
            if resid >= 0: break
            rem = min(x[i], -resid)
            x[i] -= rem
            resid += rem
    return x

def resample_h(h, n_new):
    n_old = len(h)
    if n_new == n_old:
        return h.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    return project_mean_box(np.interp(x_new, x_old, h))

def compute_slope(h, src, dst):
    """Compute s[m] for pair transfer src->dst."""
    n = len(h)
    s = np.zeros(2 * n - 1)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        val = 0.0
        idx = dst + k
        if 0 <= idx < n:
            val += h[idx]
        idx = dst - k
        if 0 <= idx < n:
            val += h[idx]
        idx = src + k
        if 0 <= idx < n:
            val -= h[idx]
        idx = src - k
        if 0 <= idx < n:
            val -= h[idx]
        s[m] = val
    return s

def fast_pair_search(h, src, dst, C, n_eval=20):
    """Fast line search using precomputed correlation C."""
    n = len(h)
    t_lo = max(-h[src], h[dst] - 1.0)
    t_hi = min(1.0 - h[src], h[dst])
    if t_hi - t_lo < 1e-14:
        return None
    
    s = compute_slope(h, src, dst)
    
    # Vectorized evaluation
    t_vals = np.linspace(t_lo, t_hi, n_eval + 2)[1:-1]  # exclude endpoints initially
    # Include endpoints
    t_vals = np.concatenate([[t_lo, t_hi], t_vals])
    # Actually just use linspace with enough points
    t_vals = np.linspace(t_lo, t_hi, n_eval)
    
    # C[None, :] + t_vals[:, None] * s[None, :] → shape (n_eval, 2n-1)
    c_vals = np.max(C[None, :] + t_vals[:, None] * s[None, :], axis=1)
    
    best_idx = np.argmin(c_vals)
    best_t = t_vals[best_idx]
    best_c = c_vals[best_idx]
    
    current_c = float(np.max(C))
    if best_c >= current_c:
        return None
    
    # Refine
    margin = (t_hi - t_lo) / (n_eval - 1)
    t_lo_r = max(t_lo, best_t - margin)
    t_hi_r = min(t_hi, best_t + margin)
    if t_hi_r > t_lo_r:
        t_fine = np.linspace(t_lo_r, t_hi_r, 15)
        c_fine = np.max(C[None, :] + t_fine[:, None] * s[None, :], axis=1)
        fine_idx = np.argmin(c_fine)
        if c_fine[fine_idx] < best_c:
            best_t = t_fine[fine_idx]
            best_c = c_fine[fine_idx]
    
    h_new = h.copy()
    h_new[src] += best_t
    h_new[dst] -= best_t
    return h_new

    dx = _g_slsqp[1]
    active = _g_slsqp[3]
    vals = np.zeros(len(active))
    for ii, m in enumerate(active):
        k = m - (n - 1)
        if k >= 0:
            hi = n - k
            s = np.sum(h_x[:hi] * (1.0 - h_x[k:]))
        else:
            hi = n + k
            s = np.sum(h_x[-k:] * (1.0 - h_x[:hi]))

    return vals

        for i in range(n):
            # d(R[m])/dh[i] = (1-h[i+k]) - h[i-k] (out of bounds = 0)
            dval = 0.0

            idx = i - k
            if 0 <= idx < n:
                dval -= h_x[idx]
            G[ii, i] = -dx * dval
    return G
