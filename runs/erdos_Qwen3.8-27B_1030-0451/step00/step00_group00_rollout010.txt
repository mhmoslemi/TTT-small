def smooth_obj_and_grad(p, eta):
    # Returns (F, grad) where F is the smooth approximation and grad is w.r.t. p
    m = len(p)
    n = 2 * m
    h = build_h(p)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    nR = len(R)
    Rmax = np.max(R)
    
    if Rmax < 1e-15:
        return 0.0, np.zeros(m)
    
    # Softmax weights
    scaled = eta * (R - Rmax) / dx
    scaled = np.clip(scaled, -700, 0)  # clip to avoid overflow (all <= 0)
    w = np.exp(scaled)
    wsum = w.sum()
    F = Rmax + (dx / eta) * np.log(wsum)
    w /= wsum
    
    # Gradient computation
    # dF/dh[j] = sum_i w[i] * dR[i]/dh[j]
    # dR[i]/dh[j] = (1-h[j-d_i])*I(0<=j-d_i<n) - h[j+d_i]*I(0<=j+d_i<n)
    # where d_i = i - (n-1)
    
    grad_h = np.zeros(n)
    v = 1.0 - h
    for i in range(nR):
        wi = w[i]
        if wi < 1e-8:
            continue
        d = i - (n - 1)
        # Term 1: (1-h[j-d]) for valid j-d
        js = max(0, d)
        je = min(n, d + n)
        if js < je:
            ms = js - d
            me = je - d
            grad_h[js:je] += wi * v[ms:me]
        # Term 2: -h[j+d] for valid j+d
        js2 = max(0, -d)
        je2 = min(n, n - d)
        if js2 < je2:
            ms2 = js2 + d
            me2 = je2 + d
            grad_h[js2:je2] -= wi * h[ms2:me2]
    
    # Chain rule: dF/dp[s] = dF/dh[s] - dF/dh[n-1-s]
    grad_p = grad_h[:m] - grad_h[n-1 : n-1-m : -1]
    
    return F, grad_p

import numpy as np
import time
from scipy.optimize import minimize

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    return dx * np.max(R)

def build_h(p):
    m = len(p)
    n = 2 * m
    h = np.empty(n)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h

def smooth_obj_and_grad(p, eta):
    m = len(p)
    n = 2 * m
    h = build_h(p)
    dx = 2.0 / n
    R = np.correlate(h, 1.0 - h, mode="full")
    nR = len(R)
    Rmax = np.max(R)
    if Rmax < 1e-15:
        return 0.0, np.zeros(m)
    scaled = eta * (R - Rmax) / dx
    scaled = np.clip(scaled, -700, 0.0)
    w = np.exp(scaled)
    wsum = w.sum()
    F = Rmax + (dx / eta) * np.log(wsum)
    w /= wsum
    grad_h = np.zeros(n)
    v = 1.0 - h
    for i in range(nR):
        wi = w[i]
        if wi < 1e-8:
            continue
        d = i - (n - 1)
        js = max(0, d)
        je = min(n, d + n)
        if js < je:
            ms = js - d
            me = je - d
            grad_h[js:je] += wi * v[ms:me]
        js2 = max(0, -d)
        je2 = min(n, n - d)
        if js2 < je2:
            ms2 = js2 + d
            me2 = je2 + d
            grad_h[js2:je2] -= wi * h[ms2:me2]
    grad_p = grad_h[:m] - grad_h[n - 1: n - 1 - m: -1]
    return F, grad_p

def lbfgsb_optimize(p0, eta, maxiter=300):
    m = len(p0)
    bounds = [(0.0, 1.0)] * m
    def fg(x):
        return smooth_obj_and_grad(x, eta)
    res = minimize(fg, p0, method='L-BFGS-B', bounds=bounds,
                   options={'maxiter': maxiter, 'ftol': 1e-14, 'gtol': 1e-10})
    return res.x

def grid_coord_polish(p, n_grid=30, max_rounds=8, time_budget=None, t_start=None):
    m = len(p)
    h = build_h(p)
    cur = exact_c5(h)
    grid = np.linspace(0.0, 1.0, n_grid)
    for rnd in range(max_rounds):
        if time_budget is not None and time.perf_counter() - t_start > time_budget:
            break
        improved = False
        for s in range(m):
            if time_budget is not None and time.perf_counter() - t_start > time_budget:
                return p, cur
            old_val = p[s]
            best_local = old_val
            best_local_c5 = cur
            for gv in grid:
                if abs(gv - old_val) < 1e-6:
                    continue
                p[s] = gv
                h = build_h(p)
                val = exact_c5(h)
                if val < best_local_c5 - 1e-14:
                    best_local_c5 = val
                    best_local = gv
            if best_local != old_val:
                p[s] = best_local
                cur = best_local_c5
                improved = True
        if not improved:
            break
    return p, cur

def coarse_to_fine_polish(p, time_budget=None, t_start=None):
    """Multi-resolution coordinate polish: coarse grid first, then refine."""
    for n_grid in [10, 20, 40, 60]:
        if time_budget is not None and time.perf_counter() - t_start > time_budget:
            break
        p, cur = grid_coord_polish(p, n_grid=n_grid, max_rounds=3,
                                   time_budget=time_budget, t_start=t_start)
    return p, cur

def random_perturb(p, rng, scale):
    m = len(p)
    p_try = p + rng.randn(m) * scale
    return np.clip(p_try, 0.0, 1.0)

def make_antisymmetric_from_h(h):
    n = len(h)
    m = n // 2
    p = np.empty(m)
    for i in range(m):
        p[i] = np.clip(h[i], 0.0, 1.0)
