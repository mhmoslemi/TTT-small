        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.85:
            break
        remaining = budget_s - elapsed - 10
        if remaining < 25:
            break
        h_interp = interp_to_n(best_h, best_n, n_try)
        t_limit_sub = min(remaining * 0.55, 80.0)
        h_try, c5_try = optimize_subgrad(h_interp, n_try, time.time(), t_limit_sub)
        if c5_try < best_c5:
            best_c5 = c5_try
            best_h = h_try.copy()
            best_n = n_try
        remaining2 = budget_s - (time.time() - t_start) - 10
        if remaining2 < 20:
            continue
        h_bin_try = make_binary(h_try, n_try)
        t_limit_bin = min(remaining2 * 0.5, 50.0)
        h_bin_try, c5_bin = binary_local_search(h_bin_try, n_try, time.time(), t_limit_bin)
        if c5_bin < best_c5:
            best_c5 = c5_bin
            best_h = h_bin_try.copy()
            best_n = n_try

    # Phase 5: Random restarts with binary search
    for _ in range(5):
        elapsed = time.time() - t_start
        if elapsed > budget_s * 0.92:
            break
        remaining = budget_s - elapsed - 5
        if remaining < 40:
            break
        for n_try in [n0, 100, 200]:
            remaining = budget_s - (time.time() - t_start) - 5
            if remaining < 20:
                break
            h_rand = random_binary(n_try)
            t_limit = min(remaining * 0.4, 40.0)
            h_rs, c5_rs = binary_local_search(h_rand, n_try, time.time(), t_limit)
            if c5_rs < best_c5:
                best_c5 = c5_rs
                best_h = h_rs.copy()
                best_n = n_try
            remaining2 = budget_s - (time.time() - t_start) - 5
            if remaining2 < 15:
                continue
            h_sub_rs, c5_sub_rs = optimize_subgrad(h_rs.copy(), n_try, time.time(), min(remaining2 * 0.5, 30.0))
            if c5_sub_rs < best_c5:
                best_c5 = c5_sub_rs
                best_h = h_sub_rs.copy()
                best_n = n_try

    # Final verification
    final_c5 = compute_c5(best_h, best_n)
    best_c5 = final_c5

    return best_h, best_c5, best_n
