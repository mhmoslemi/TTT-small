import numpy as np
import time
from scipy.optimize import minimize

def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    val = float(np.max(corr))
    return val if np.isfinite(val) else np.inf

def build_h_from_p(p, n):
    p = np.clip(p, 0.0, 1.0)
    if n % 2 == 0:
        m = n // 2
        h = np.empty(n)
        h[:m] = p
        h[m:] = 1.0 - p[::-1]
    else:
        m = n // 2
        h = np.empty(n)
        h[:m] = p
        h[m] = 0.5
        h[m+1:] = 1.0 - p[::-1]
    return h

def p_from_h(h):
    n = len(h)
    if n % 2 == 0:
        m = n // 2
        p = 0.5 * (h[:m] + 1.0 - h[m:][::-1])
    else:
        m = n // 2
        p = 0.5 * (h[:m] + 1.0 - h[m+1:][::-1])
    return np.clip(p, 0.0, 1.0)

def interp_p(p_old, m_new):
    m_old = len(p_old)
    if m_old == m_new:
        return p_old.copy()
    if m_old <= 1:
        return np.full(m_new, p_old[0] if m_old == 1 else 0.5)
    x_old = (np.arange(m_old) + 0.5) / m_old
    x_new = (np.arange(m_new) + 0.5) / m_new
    return np.clip(np.interp(x_new, x_old, p_old), 0.0, 1.0)

def project_full(y, n):
    target = n / 2.0
    lo, hi = -2.0, 3.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    return np.clip(y + (lo + hi) / 2.0, 0.0, 1.0)

def lse_obj(p, tau, n):
    p_clip = np.clip(p, 0.0, 1.0)
    h = build_h_from_p(p_clip, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = float(np.max(corr))
    if tau < 1e-12:
        return cm
    shifted = (corr - cm) / tau
    shifted = np.clip(shifted, -400.0, 400.0)
    return cm + tau * float(np.log(np.sum(np.exp(shifted))))

def full_lse_obj(y, tau, n):
    h = project_full(y, n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    cm = float(np.max(corr))
    if tau < 1e-12:

def binary_local_search(h, n, max_iters=200):
    h = h.copy()
    ones = np.where(h > 0.5)[0]
    zeros = np.where(h <= 0.5)[0]
    h[ones] = 1.0
    h[zeros] = 0.0
    dx = 2.0 / n
    curr = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
    for _ in range(max_iters):
        if len(ones) < 1 or len(zeros) < 1:
            break
        best_gain = 0.0
        best_move = None
        sample_i = np.random.choice(ones, size=min(len(ones), 30), replace=False)
        sample_j = np.random.choice(zeros, size=min(len(zeros), 30), replace=False)
        for i in sample_i:
            for j in sample_j:
                h[i] = 0.0
                h[j] = 1.0
                val = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
                h[i] = 1.0
                h[j] = 0.0
                gain = curr - val
                if gain > best_gain:
                    best_gain = gain
                    best_move = (i, j)
        if best_move is None:
            break
        i, j = best_move
        h[i] = 0.0
        h[j] = 1.0
        ones = list(set(ones) - {i} | {j})
        zeros = list(set(zeros) - {j} | {i})
        curr = float(np.max(np.correlate(h, 1.0 - h, mode="full") * dx))
    return h, curr

    # Cosine

    for freq in [1, 2, 3, 4]:
        for amp in [0.3, 0.5, 0.7]:
            p = 0.5 + amp * np.cos(2.0 * np.pi * freq * x)
            cands.append(np.clip(p, 0.0, 1.0))
    # Triangle

    # Binary-like from random

        cands.append(np.clip(p, 0.0, 1.0))
    return cands

                    h_bin[to_remove] = 0.0
                else:
                    zero_idx = np.where(h_bin == 0.0)[0]
                    to_add = zero_idx[:target_ones - n_ones]
                    h_bin[to_add] = 1.0
            h_res, c5_bin = binary_local_search(h_bin, n_try, max_iters=150)
            if np.isfinite(c5_bin):
                try_update(h_res)

    # Phase 6: Full non-symmetric optimization (if time allows)
    if time.monotonic() < deadline - 20 and best_h is not None:
        for n_full in [len(best_h), 128]:
            if time.monotonic() > deadline - 10:
                break
            if len(best_h) == n_full:
                h_start = best_h.copy()
            else:
                p_sym = p_from_h(best_h)
                p_new = interp_p(p_sym, n_full // 2)
                h_start = build_h_from_p(p_new, n_full)
            for tau in [0.003, 0.001]:
                if time.monotonic() > deadline - 5:
                    break
                try:
                    res = minimize(
                        full_lse_obj, h_start, args=(tau, n_full),
                        method='L-BFGS-B',
                        bounds=[(-0.1, 1.1)] * n_full,
                        options={'maxiter': 150, 'ftol': 1e-12, 'maxfun': 3000}
                    )
                    if res.x is not None:
                        h_new = project_full(res.x, n_full)
                        c5_new = exact_c5(h_new)
                        if np.isfinite(c5_new):
                            try_update(h_new)
                except Exception:
                    pass

    # Final
    if best_h is None:
        best_h = np.full(64, 0.5)
        best_c5 = exact_c5(best_h)

    best_h = np.clip(best_h, 0.0, 1.0)
    n_final = len(best_h)
    target = n_final / 2.0
    if abs(np.sum(best_h) - target) > 1e-8:
        best_h = project_full(best_h, n_final)

    # Compare with original initial
    if initial is not None:
        h_init = np.clip(initial, 0.0, 1.0)
        c5_init = exact_c5(h_init)
        if c5_init < best_c5:
            best_h = h_init
            best_c5 = c5_init
            n_final = len(best_h)

    c5_final = exact_c5(best_h)
    return best_h, c5_final, int(n_final)
