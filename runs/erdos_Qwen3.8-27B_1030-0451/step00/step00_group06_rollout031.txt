```python
def min_convex_pl(c, a, lo, hi):
    """Find min of max_i(c_i + a_i * x) over [lo, hi]."""
    # Binary search on the derivative
    for _ in range(60):  # ~1e-18 precision
        mid = (lo + hi) / 2
        vals = c + a * mid
        max_val = np.max(vals)
        active = np.isclose(vals, max_val, atol=1e-14)
        # Subgradient at mid: {a_i : active}
        if np.max(a[active]) <= 0:
            # Function is decreasing (or flat) at mid, move right
            lo = mid
        elif np.min(a[active]) >= 0:
            # Function is increasing (or flat) at mid, move left
            hi = mid
        else:
            # 0 is in the subgradient, mid is optimal
            return mid, max_val
    mid = (lo + hi) / 2
    return mid, np.max(c + a * mid)
```

```python
def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def min_convex_pl(c, a, lo, hi):
    """Minimize max_i(c_i + a_i * x) over [lo, hi] using binary search on derivative."""
    if hi - lo < 1e-14:
        x = (lo + hi) * 0.5
        return x, float(np.max(c + a * x))
    for _ in range(50):
        mid = (lo + hi) * 0.5
        vals = c + a * mid
        max_val = float(np.max(vals))
        # Find range of active slopes
        # Use a tolerance relative to the scale
        tol = 1e-12 * max(1.0, max_val)
        active = (max_val - vals) <= tol
        if not np.any(active):
            active = vals == max_val
        a_active_min = float(np.min(a[active]))
        a_active_max = float(np.max(a[active]))
        if a_active_max <= 1e-14:
            lo = mid
        elif a_active_min >= -1e-14:
            hi = mid
        else:
            return mid, max_val
    mid = (lo + hi) * 0.5
    return mid, float(np.max(c + a * mid))

def project_sum(h, target_sum):
    """Project h into [0,1]^n with sum = target_sum."""
    h = h.copy()
    for _ in range(200):
        diff = target_sum - float(np.sum(h))
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        if not np.any(mask):
            break

        np.clip(h, 0.0, 1.0, out=h)
    return h

def resample_h(h, new_n):
    """Resample step function from len(h) bins to new_n bins."""
    old_n = len(h)
    x_old = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    x_new = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(x_new, x_old, h, left=h[0], right=h[-1])
    h_new = project_sum(h_new, new_n / 2.0)

    best_local = float(np.max(c5_vals))

        m = rng.randint(0, n)
        if j == m:

        a = np.zeros(2*n - 1)

        vj2 = (j - lags >= 0) & (j - lags < n)
        vm1 = (m + lags >= 0) & (m + lags < n)
        vm2 = (m - lags >= 0) & (m - lags < n)
        
        a[vj1] += (1.0 - h[j + lags[vj1]])
        a[vj2] -= h[j - lags[vj2]]
        a[vm1] -= (1.0 - h[m + lags[vm1]])
        a[vm2] += h[m - lags[vm2]]
        
        delta_lo = max(-h[j], h[m] - 1.0)
        delta_hi = min(1.0 - h[j], h[m])
        margin = 1e-12
        if delta_hi - delta_lo < margin:
            continue
        
        delta_lo += margin
        delta_hi -= margin
        
        opt_delta, new_max = min_convex_pl(c5_vals, a, delta_lo, delta_hi)

            h[m] -= opt_delta

            elif h[j] > 1: h[j] = 1.0
            if h[m] < 0: h[m] = 0.0
            elif h[m] > 1: h[m] = 1.0

            best_local = float(np.max(c5_vals))
            no_improve = 0
        else:
            no_improve += 1

        if iter_count % 1000 == 0:
            c5_vals = np.correlate(h, 1.0 - h, mode='full') * dx
            best_local = float(np.max(c5_vals))

                best_c5 = c5_final
```

```python
import numpy as np
import time

def compute_c5(h, n):
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def min_convex_pl(c, a, lo, hi):
    """Minimize max_i(c_i + a_i*x) over [lo,hi] via bisection on subgradient."""
    if hi - lo < 1e-13:
        x = (lo + hi) * 0.5
        return x, float(np.max(c + a * x))
    for _ in range(45):
        mid = (lo + hi) * 0.5
        vals = c + a * mid
        max_val = float(np.max(vals))
        tol = 1e-11 * max(1.0, abs(max_val))
        active = (vals >= max_val - tol)
        if not np.any(active):
            active = vals == max_val
        amin = float(np.min(a[active]))
        amax = float(np.max(a[active]))
        if amax <= 1e-12:
            lo = mid
        elif amin >= -1e-12:
            hi = mid
        else:
            return mid, max_val
    mid = (lo + hi) * 0.5
    return mid, float(np.max(c + a * mid))

def project_sum(h, target):
    """Project h to [0,1]^n with sum=target."""
    h = h.copy()
    for _ in range(300):
        diff = target - float(np.sum(h))
        if abs(diff) < 1e-14:
            break
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        cnt = int(np.sum(mask))
        if cnt == 0:
            break
        h[mask] += diff / cnt
        np.clip(h, 0.0, 1.0, out=h)
    return h

def resample_h(h, new_n):
    """Resample step function to new_n bins."""
    old_n = len(h)
    xo = (np.arange(old_n) + 0.5) * (2.0 / old_n)
    xn = (np.arange(new_n) + 0.5) * (2.0 / new_n)
    h_new = np.interp(xn, xo, h, left=h[0], right=h[-1])
    return project_sum(h_new, new_n / 2.0)

def coord_descent(h, n, dur, rng):
    """Coordinate descent with exact line search for dur seconds."""
    t0 = time.time()
    dx = 2.0 / n
    c5v = np.correlate(h, 1.0 - h, mode='full') * dx
    best = float(np.max(c5v))
    lags = np.arange(-(n - 1), n)
    two_n_minus_1 = 2 * n - 1
    stalls = 0

    while time.time() - t0 < dur:
        # Occasionally use guided selection based on worst lag
        if rng.random() < 0.3:
            worst_k = int(np.argmax(c5v))
            worst_lag = worst_k - (n - 1)
            # Find positions contributing most to worst lag overlap
            valid = (np.arange(n) + worst_lag >= 0) & (np.arange(n) + worst_lag < n)
            contrib = np.zeros(n)
            contrib[valid] = h[valid] * (1.0 - h[valid + worst_lag])
            # Pick j from high-contrib positions, m from low-position
            j_candidates = np.where(valid & (h > 0.3))[0]
            m_candidates = np.where(h < 0.7)[0]
            if len(j_candidates) > 0 and len(m_candidates) > 0:
                j = int(j_candidates[rng.randint(len(j_candidates))])
                m = int(m_candidates[rng.randint(len(m_candidates))])
            else:
                j = int(rng.randint(0, n))
                m = int(rng.randint(0, n))
        else:
            j = int(rng.randint(0, n))
            m = int(rng.randint(0, n))

        if j == m:
            continue

        a = np.zeros(two_n_minus_1)
        vj1 = (j + lags >= 0) & (j + lags < n)
        vj2 = (j - lags >= 0) & (j - lags < n)
        vm1 = (m + lags >= 0) & (m + lags < n)
        vm2 = (m - lags >= 0) & (m - lags < n)

        a[vj1] += (1.0 - h[j + lags[vj1]])
        a[vj2] -= h[j - lags[vj2]]
        a[vm1] -= (1.0 - h[m + lags[vm1]])
        a[vm2] += h[m - lags[vm2]]

        dlo = max(-h[j], h[m] - 1.0) + 1e-13
        dhi = min(1.0 - h[j], h[m]) - 1e-13
        if dhi - dlo < 1e-12:

        if new_max < best - 1e-14:
            h[j] += opt_d
            h[m] -= opt_d
            if h[j] < 0.0:
                h[j] = 0.0
            elif h[j] > 1.0:
                h[j] = 1.0
            if h[m] < 0.0:
                h[m] = 0.0
            elif h[m] > 1.0:
                h[m] = 1.0
            c5v = np.correlate(h, 1.0 - h, mode='full') * dx
            best = float(np.max(c5v))
            stalls = 0
        else:
            stalls += 1

        if stalls > 5000:

            for _ in range(20):
                p = int(rng.randint(0, n))
                q = int(rng.randint(0, n))
                if p == q:

                d = (rng.random() - 0.5) * 0.1
                h[p] += d
                h[q] -= d
            np.clip(h, 0.0, 1.0, out=h)
            c5v = np.correlate(h, 1.0 - h, mode='full') * dx
            best = float(np.max(c5v))
            stalls = 0

    return h, best
```
