This is a 1D problem! The optimal v is where the two "tightest" lines cross, or at a boundary.

This is much more efficient than the subgradient approach for local optimization!

Now for the 1D minimization of max_k (a_k * v + b_k) over v ∈ [0,1]:
Since this is a convex piecewise-linear function, I can find the minimum by:
1. Computing f(0) and f(1)
2. If f'(0+) < 0 and f'(1-) > 0 (in the subdifferential sense), the minimum is in the interior
3. The interior minimum is at a point where the subdifferential contains 0

For a piecewise linear convex function, the minimum is at a "kink" point. The kink points are where two lines are equal: a_k1 * v + b_k1 = a_k2 * v + b_k2, i.e., v = (b_k2 - b_k1) / (a_k1 - a_k2).

But finding all kink points is O(n²). Instead, I can use the fact that f(v) = max_k (a_k * v + b_k) and use a binary search / bisection approach:

f is convex, so I can use ternary search or bisection on the derivative.

The "derivative" of f at v (right derivative) is max_{k: a_k*v+b_k = f(v)} a_k.
The "derivative" (left derivative) is min_{k: a_k*v+b_k = f(v)} a_k.

f is minimized at v* where the left derivative ≤ 0 ≤ the right derivative.

Algorithm:
1. Start with lo=0, hi=1
2. At each step, compute v_mid = (lo+hi)/2
3. Compute f(v_mid) and the set of maximizing k's
4. If all maximizing k's have a_k ≤ 0: f is decreasing at v_mid, so minimum is to the right → lo = v_mid
5. If all maximizing k's have a_k ≥ 0: f is increasing at v_mid, so minimum is to the left → hi = v_mid
6. Otherwise (mixed signs): v_mid is the minimum (or very close)

This converges in O(log(1/ε)) iterations, each taking O(n) time.

Let me code this up:

```python
def _coord_optimal(h, corr_raw, j, n):
    """Find optimal value of h[j] given others fixed. Returns new h[j]."""
    # Compute a vector (gradient of corr w.r.t. h[j] for each lag)
    a = np.zeros(2 * n - 1)
    a[n-1-j : 2*n-1-j] += (1.0 - h)
    a[j : j+n] += -h[::-1]
    
    # b_k = corr_raw[k] - h[j] * a[k]
    h_j = h[j]
    b = corr_raw - h_j * a
    
    # Minimize max_k (a_k * v + b_k) for v in [0,1]
    # Bisection on sign of "derivative"
    lo, hi = 0.0, 1.0
    for _ in range(50):
        v_mid = 0.5 * (lo + hi)
        vals = a * v_mid + b
        max_val = vals.max()
        active = vals >= max_val - 1e-14
        a_active = a[active]
        if a_active.min() > 1e-14:
            # f' > 0, minimum is to the left
            hi = v_mid
        elif a_active.max() < -1e-14:
            # f' < 0, minimum is to the right
            lo = v_mid
        else:
            # 0 in subdifferential, v_mid is optimal
            break
    
    v_star = 0.5 * (lo + hi)
    return v_star
```

```python
import numpy as np
import time

def _project_simplex_box(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        if np.clip(h - mid, 0.0, 1.0).sum() > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def _corr_raw(h):
    return np.correlate(h, 1.0 - h, mode="full")

def _coord_update(h, corr_raw, j, n, lam):
    """Optimize h[j] with Lagrange penalty lam*(sum - target)."""
    nf = 2 * n - 1
    a = np.zeros(nf)
    a[n-1-j : 2*n-1-j] += (1.0 - h)
    a[j : j+n] += -h[::-1]
    
    h_j = h[j]
    b = corr_raw - h_j * a
    
    # Minimize max_k(a_k*v + b_k) + lam*v for v in [0,1]
    # = max_k((a_k+lam)*v + b_k) ... no, that's wrong.
    # = max_k(a_k*v + b_k) + lam*v
    # = max_k(a_k*v + b_k) + max(v) * lam  ... no.
    # The + lam*v is not inside the max.
    
    # So f(v) = max_k(a_k*v + b_k) + lam*v
    # This is still convex piecewise linear.
    # f'(v) = a_{k*} + lam where k* is a maximizer (subgradient)
    
    lo, hi = 0.0, 1.0
    for _ in range(40):
        v_mid = 0.5 * (lo + hi)
        vals = a * v_mid + b
        max_val = vals.max()
        active = vals >= max_val - 1e-12
        a_min = a[active].min()
        a_max = a[active].max()
        # Subgradient of f at v_mid: [a_min + lam, a_max + lam]
        if a_min + lam > 1e-14:
            hi = v_mid
        elif a_max + lam < -1e-14:
            lo = v_mid
        else:
            break
    return 0.5 * (lo + hi)

def _coord_descent(h, n, dx, lam_init=0.0, max_sweeps=200, tol=1e-10):
    """Lagrangian coordinate descent."""
    target_sum = n / 2.0
    lam = lam_init
    h = h.copy()
    
    for sweep in range(max_sweeps):
        corr_raw = _corr_raw(h)
        c5 = corr_raw.max() * dx
        
        # Coordinate updates
        max_change = 0.0
        for j in range(n):
            old_v = h[j]
            new_v = _coord_update(h, corr_raw, j, n, lam)
            h[j] = new_v
            max_change = max(max_change, abs(new_v - old_v))
        
        # Update Lagrange multiplier
        sum_h = h.sum()
        lam += 0.1 * (sum_h - target_sum)
        lam = np.clip(lam, -10, 10)
        
        if max_change < tol:
            break
    
    # Final projection to satisfy constraint exactly
    h = _project_simplex_box(h, target_sum)
    corr_raw = _corr_raw(h)
    c5 = corr_raw.max() * dx
    return h, c5

def _subgradient_optimize(h, n, dx, deadline, best_state):
    """Subgradient descent with backtracking."""
    target_sum = n / 2.0
    h = _project_simplex_box(h, target_sum)
    corr = _corr_raw(h) * dx
    c5 = corr.max()
    if c5 < best_state[1]:
        best_state[1] = c5
        best_state[0] = h.copy()
    
    lr = 0.005
    top_k = min(30, 2*n-1)
    stall = 0
    
    for it in range(20000):
        if time.time() > deadline:
            break
        max_c = corr.max()
        idx = np.argpartition(-corr, top_k)[:top_k]
        active = idx[corr[idx] >= 0.97 * max_c]
        if len(active) == 0:
            active = np.array([np.argmax(corr)])
        
        sg = np.zeros(n)
        for k in active:
            m = int(k) - (n - 1)
            omh = 1.0 - h
            if m >= 0:
                sg[:n-m] += omh[m:]
                sg[m:] -= h[:n-m]
            else:
                sg[-m:] += omh[:n+m]
                sg[:n+m] -= h[-m:]
        sg /= len(active)
        
        alpha = lr
        improved = False
        for _ in range(10):
            h_new = _project_simplex_box(h - alpha * sg, target_sum)
            corr_new = _corr_raw(h_new) * dx
            c5_new = corr_new.max()
            if c5_new < c5 - 1e-13:
                h = h_new
                corr = corr_new
                c5 = c5_new
                if c5 < best_state[1]:
                    best_state[1] = c5
                    best_state[0] = h.copy()
                improved = True
                break
            alpha *= 0.6
        if not improved:
            stall += 1
            lr *= 0.93
            if lr < 1e-10 or stall > 150:
                break
        else:
            stall = max(0, stall - 1)
    return h
```
