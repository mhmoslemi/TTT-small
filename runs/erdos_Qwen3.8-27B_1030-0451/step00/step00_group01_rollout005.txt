```python
    n = len(h)
```

```python
import numpy as np
import time
from scipy.optimize import linprog, minimize

def project_sum_box(x, target):
    """Project x onto {x : 0<=x_i<=1, sum(x)=target}."""
    n = len(x)
    x = np.clip(x, 0, 1)
    # Bisection on lambda
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(x + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + (lo+hi)/2, 0, 1)
    # Fix any residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        count = np.sum(mask)
        if count > 0:
            result[mask] += residual / count
            result = np.clip(result, 0, 1)
    return result

def corr_values(h):
    """Compute all correlation values v_k = dx * correlate(h, 1-h) at lag k."""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx

def evaluate_h(h):
    """Compute C5 for given h."""
    return np.max(corr_values(h))

def grad_single_lag(h, lag):
    """Gradient of v_lag with respect to h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        for i in range(n - lag):
            g[i + lag] += (1.0 - h[i]) * dx
            g[i] -= h[i + lag] * dx
    else:
        d = -lag
        for i in range(d, n):
            g[i] += (1.0 - h[i - d]) * dx
            g[i - d] -= h[i] * dx
    return g

def smooth_objective(h, beta):
    """Log-sum-exp smooth approximation of max correlation."""
    v = corr_values(h)
    m = np.max(v)
    s = beta * (v - m)
    # Clip for numerical stability
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    Z = np.sum(e)
    f = m + np.log(Z) / beta
    return f

def smooth_gradient(h, beta):
    """Gradient of log-sum-exp smooth objective."""
    v = corr_values(h)
    m = np.max(v)
    s = beta * (v - m)
    s = np.clip(s, -500, 500)
    e = np.exp(s)
    Z = np.sum(e)
    w = e / Z  # weights
    
    g = np.zeros(len(h))
    n = len(h)
    # Only compute gradients for lags with significant weight
    for k, wk in enumerate(w):
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        g += wk * grad_single_lag(h, lag)
    return g

def lp_optimize(h0, n_active, radius, deadline):
    """Trust-region LP optimization on active constraints."""
    n = len(h0)
    v0 = corr_values(h0)
    
    # Select top n_active lags
    order = np.argsort(v0)[::-1]
    active_lags = order[:n_active]
    
    # Build LP
    A_ub_rows = []
    b_ub = []
    for k in active_lags:
        lag = k - (n - 1)
        g = grad_single_lag(h0, lag)
        # grad_lag · h - t <= v0[k] - grad_lag · h0
        A_ub_rows.append(np.concatenate([g, [-1.0]]))
        b_ub.append(v0[k] - np.dot(g, h0))
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub)
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    # Bounds
    lo = np.maximum(0, h0 - radius)
    hi = np.minimum(1, h0 + radius)
    bounds = list(zip(lo, hi))
    bounds.append((None, None))  # t is free
    
    # Minimize t
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.success:
        h_candidate = res.x[:n]
        h_candidate = project_sum_box(h_candidate, n / 2.0)
        return h_candidate
    return None

def slsqp_optimize(h0, betas, deadline):

    n = len(h0)
    target = n / 2.0
    
    best_h = h0.copy()
    best_c = evaluate_h(h0)
    
    def obj_func(x):

    def obj_grad(x):

        try:

                    best_c = c_candidate
                    best_h = h_candidate.copy()
                    h0 = h_candidate.copy()
        except Exception:
            pass
    
    return best_h, best_c
```

```python
_slsqp_target = 42.0

def project_sum_box(x, target):
    """Project x onto {x : 0<=x_i<=1, sum(x)=target}."""
    n = len(x)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-8:
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += residual / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def corr_values(h):

    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def evaluate_h(h):
    return float(np.max(corr_values(h)))

def grad_single_lag(h, lag):
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        g[idx + lag] += (1.0 - h[idx]) * dx
        g[idx] -= h[idx + lag] * dx
    else:
        d = -lag
        idx = np.arange(d, n)
        g[idx] += (1.0 - h[idx - d]) * dx
        g[idx - d] -= h[idx] * dx
    return g

    v = corr_values(x)
    m = np.max(v)
    s = np.clip(beta * (v - m), -500.0, 500.0)
    e = np.exp(s)
    Z = np.sum(e)
    return m + np.log(Z) / beta

def smooth_gradient(x, beta):
    v = corr_values(x)
    m = np.max(v)
    s = np.clip(beta * (v - m), -500.0, 500.0)
    e = np.exp(s)
    Z = np.sum(e)

    n = len(x)
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-9:
            continue
        lag = k - (n - 1)
        g += w[k] * grad_single_lag(x, lag)
    return g

    return smooth_objective(x, beta)

def _slsqp_grad(x, beta):
    return smooth_gradient(x, beta)

def _slsqp_cons_fun(x):
    return np.sum(x) - _slsqp_target

def _slsqp_cons_jac(x):
    return np.ones(len(x))

def lp_step(h0, n_active, radius):
    """One LP trust-region step."""
    n = len(h0)
    v0 = corr_values(h0)
    order = np.argsort(v0)[::-1]
    active = order[:min(n_active, len(order))]
    
    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for row, k in enumerate(active):
        lag = k - (n - 1)
        g = grad_single_lag(h0, lag)
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = v0[k] - np.dot(g, h0)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = list(zip(lo, hi))

                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_candidate = project_sum_box(res.x[:n], n / 2.0)
            return h_candidate
    except Exception:
        pass
    return None

def slsqp_phase(h0, betas, time_limit):

    n = len(h0)
    _slsqp_target = n / 2.0

                    best_h = h_new.copy()
                    h0 = h_new.copy()
        except Exception:

    return best_h, best_c

def projected_gradient_phase(h0, beta, n_iters, time_limit):
    """Projected gradient descent with backtracking on true objective."""
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()

        c_current = evaluate_h(h0)

            if c_trial < best_c:
                best_c = c_trial
                best_h = h0.copy()
        else:
            step *= 0.5
            if step < 1e-8:
                break
    
    return best_h, best_c

    h_new = np.clip(h_new, 0.0, 1.0)
```

```python
import numpy as np
import time
from scipy.optimize import linprog, minimize

_slsqp_target = 42.0


def project_sum_box(x, target):
    """Project x onto {x : 0<=x_i<=1, sum(x)=target}."""
    n = len(x)
    x = np.clip(np.asarray(x, dtype=float), 0.0, 1.0)
    lo, hi = -1.0, 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        cnt = np.sum(mask)
        if cnt > 0:
            result[mask] += residual / cnt
            result = np.clip(result, 0.0, 1.0)
    return result


def corr_values(h):
    """Compute v_k = dx * correlate(h, 1-h) for all lags."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx


def evaluate_h(h):
    """Compute C5 score for h."""
    return float(np.max(corr_values(h)))


def grad_single_lag(h, lag):
    """Gradient of v_lag with respect to h."""
    n = len(h)
    dx = 2.0 / n
    g = np.zeros(n)
    if lag >= 0:
        idx = np.arange(n - lag)
        g[idx + lag] += (1.0 - h[idx]) * dx
        g[idx] -= h[idx + lag] * dx
    else:
        d = -lag
        idx1 = np.arange(n - d)
        g[idx1] += (1.0 - h[idx1 + d]) * dx
        idx2 = np.arange(d, n)
        g[idx2] -= h[idx2 - d] * dx
    return g


def smooth_objective(x, beta):
    """Log-sum-exp smooth upper bound on max correlation."""
    v = corr_values(x)
    m = np.max(v)
    s = np.clip(beta * (v - m), -500.0, 500.0)
    e = np.exp(s)
    Z = np.sum(e)
    return m + np.log(Z) / beta


def smooth_gradient(x, beta):
    """Gradient of log-sum-exp smooth objective."""
    v = corr_values(x)
    m = np.max(v)
    s = np.clip(beta * (v - m), -500.0, 500.0)
    e = np.exp(s)
    Z = np.sum(e)
    w = e / Z
    n = len(x)
    g = np.zeros(n)
    for k in range(len(w)):
        if w[k] < 1e-10:
            continue
        lag = k - (n - 1)
        g += w[k] * grad_single_lag(x, lag)
    return g


def _slsqp_obj(x, beta):
    return smooth_objective(x, beta)


def _slsqp_grad(x, beta):
    return smooth_gradient(x, beta)


def _slsqp_cons_fun(x):
    return np.sum(x) - _slsqp_target


def _slsqp_cons_jac(x):
    return np.ones(len(x))


def lp_step(h0, n_active, radius):
    """Trust-region LP step on active constraints."""
    n = len(h0)
    v0 = corr_values(h0)
    order = np.argsort(v0)[::-1]
    active = order[:min(n_active, len(order))]

    A_ub = np.zeros((len(active), n + 1))
    b_ub = np.zeros(len(active))
    for row, k in enumerate(active):
        lag = k - (n - 1)
        g = grad_single_lag(h0, lag)
        A_ub[row, :n] = g
        A_ub[row, n] = -1.0
        b_ub[row] = v0[k] - np.dot(g, h0)

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    lo = np.maximum(0.0, h0 - radius)
    hi = np.minimum(1.0, h0 + radius)
    bounds = list(zip(lo, hi))
    bounds.append((-0.5, 1.0))

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
        if res.success and res.x is not None:
            h_candidate = project_sum_box(res.x[:n], n / 2.0)
            return h_candidate
    except Exception:
        pass
    return None


def slsqp_phase(h0, betas, time_limit):
    """SLSQP optimization with increasing beta schedule."""
    global _slsqp_target
    n = len(h0)
    _slsqp_target = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_h(h0)

    for beta in betas:
        if time.time() > time_limit:
            break
        try:
            cons = [{'type': 'eq', 'fun': _slsqp_cons_fun, 'jac': _slsqp_cons_jac}]
            res = minimize(_slsqp_obj, h0, args=(beta,), jac=_slsqp_grad,
                           method='SLSQP', bounds=[(0.0, 1.0)] * n,
                           constraints=cons,
                           options={'maxiter': 400, 'ftol': 1e-12})
            if res.x is not None:
                h_new = project_sum_box(res.x, n / 2.0)
                c_new = evaluate_h(h_new)
                if c_new < best_c - 1e-12:
                    best_c = c_new
                    best_h = h_new.copy()
                    h0 = h_new.copy()
        except Exception:
            pass
    return best_h, best_c


def projected_gradient_phase(h0, beta, n_iters, time_limit):
    """Projected gradient with backtracking on true objective."""
    n = len(h0)
    target = n / 2.0
    best_h = h0.copy()
    best_c = evaluate_h(h0)
    h_cur = h0.copy()
    step = 0.05

    for it in range(n_iters):
        if time.time() > time_limit:
            break
        g = smooth_gradient(h_cur, beta)
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        h_trial = h_cur - step * g
        h_trial = project_sum_box(h_trial, target)

        c_trial = evaluate_h(h_trial)
        c_current = evaluate_h(h_cur)

        if c_trial < c_current - 1e-13:
            h_cur = h_trial
            if c_trial < best_c:
                best_c = c_trial
                best_h = h_cur.copy()
            step = min(step * 1.2, 0.2)
        else:
            step *= 0.5
            if step < 1e-10:
                break
    return best_h, best_c


def transfer_resolution(h_old, n_new):
    """Interpolate h from old grid to new grid and project."""
    n_old = len(h_old)
    old_pos = (np.arange(n_old) + 0.5) / n_old
    new_pos = (np.arange(n_new) + 0.5) / n_new
    h_new = np.interp(new_pos, old_pos, h_old, left=h_old[0], right=h_old[-1])
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = project_sum_box(h_new, n_new / 2.0)
    return h_new


def random_restart(n, target, rng, n_trials):
    """Generate random feasible points and return the best."""
    best_h = None
    best_c = np.inf
    for _ in range(n_trials):
        h = rng.uniform(0.0, 1.0, size=n)
        h = project_sum_box(h, target)
        c = evaluate_h(h)
        if c < best_c:
            best_c = c
            best_h = h.copy()
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    global _slsqp_target
    start_time = time.time()
    deadline = start_time + budget_s - 10

    # Get initial solution
```
