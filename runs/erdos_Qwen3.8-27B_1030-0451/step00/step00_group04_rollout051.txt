  Wait, h_{j-a} for j in [a, n-1] is h[0] to h[n-1-a] = h[:n-a]. Yes!
- For j in [0, n-1-a]: contributes -h_{j+a}. So grad[:n-a] -= h[a:]
  h_{j+a} for j in [0, n-1-a] is h[a] to h[n-1] = h[a:]. Yes!

```python
def corr_grad(h, weights):
    """Compute gradient of weighted sum of correlations.
    h: array of length n
    weights: array of length 2n-1 (weights for each lag in np.correlate order)
    Returns: gradient array of length n
    """
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    
    for q in range(2*n-1):
        w = weights[q]
        if abs(w) < 1e-15:
            continue
        ell = q - (n-1)
        if ell == 0:
            grad += w * dx * (1 - 2*h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += w * dx * (1 - h[:L])
            grad[:L] -= w * dx * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[a:] += w * dx * (1 - h[:L])
            grad[:L] -= w * dx * h[a:]
    
    return grad
```

This is O(n²) per call which is fine for n=250-500.

```python
import numpy as np
import time
from scipy.optimize import minimize

def corr_values(h):
    """Compute all correlation values z_q = correlate(h, 1-h, 'full')[q] * dx"""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    return z

def evaluate_c5(h):
    """Compute C5 = max of correlation values"""
    return float(np.max(corr_values(h)))

def project_box_sum(h, T):
    """Project h onto {x: 0<=x_i<=1, sum(x)=T} using bisection on lambda"""
    x = h.copy()
    # Find lambda such that sum(clip(x - lambda, 0, 1)) = T
    lo, hi = x.min() - 1, x.max() + 1
    # Actually lambda can be any real; clip(x-lambda, 0, 1)
    # When lambda is very negative, clip gives all 1s, sum = n
    # When lambda is very positive, clip gives all 0s, sum = 0
    # We need sum = T
    
    # Ensure lo and hi bracket the solution
    for _ in range(200):
        mid = 0.5 * (lo + hi)
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    # Fix any tiny sum error
    diff = T - y.sum()
    if abs(diff) > 1e-12:
        # Distribute difference
        mask = (y > 1e-10) & (y < 1 - 1e-10)
        if mask.sum() > 0 and abs(diff) < 0.5:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def corr_gradient(h, weights, dx=None):
    """Compute gradient of sum_q weights[q] * z_q where z = correlate(h,1-h,'full')*dx"""
    n = len(h)
    if dx is None:
        dx = 2.0 / n
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = weights[q]
        if abs(w) < 1e-14:
            continue
        ell = q - (n - 1)
        if ell == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += w * dx * (1.0 - h[:L])
            grad[:L] -= w * dx * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[a:] += w * dx * (1.0 - h[:L])
            grad[:L] -= w * dx * h[a:]
    return grad

def smooth_max_and_grad(h, tau):
    """Compute smooth max surrogate F_tau and its gradient"""
    z = corr_values(h)
    z_max = z.max()
    shifted = (z - z_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 0)
    exp_shifted = np.exp(shifted)
    s = exp_shifted.sum()
    p = exp_shifted / s
    F = z_max + tau * np.log(s)
    grad = corr_gradient(h, p)
    return F, grad, z

def pgd_optimize(h0, max_iter=5000, tau_start=None, tau_end=1e-4,
                 tau_decay=0.97, alpha_start=0.05, time_limit=60.0,
                 deadline=None, verbose=False):
    """PGD optimization with smooth max surrogate"""
    h = project_box_sum(h0.copy(), len(h0) / 2.0)
    n = len(h)
    T = n / 2.0
    
    if tau_start is None:
        z = corr_values(h)
        spread = z.max() - z.min()
        tau_start = max(0.01 * spread, 0.002)
    
    tau = tau_start
    alpha = alpha_start
    best_hard = evaluate_c5(h)
    best_h = h.copy()
    F_prev = smooth_max_and_grad(h, tau)[0]
    
    t0 = time.time()
    no_improve_count = 0
    
    for it in range(max_iter):
        # Time check
        elapsed = time.time() - t0
        if elapsed > time_limit:
            break
        if deadline is not None and time.time() > deadline:
            break
        
        # Compute smooth max and gradient
        F, grad, z = smooth_max_and_grad(h, tau)
        
        # Normalize gradient
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        
        direction = -grad / gnorm
        
        # Line search
        accepted = False
        for _ in range(20):
            h_new = project_box_sum(h + alpha * direction, T)
            F_new = smooth_max_and_grad(h_new, tau)[0]
            if F_new < F - 1e-12:
                h = h_new
                F = F_new
                accepted = True
                break
            alpha *= 0.5
        
        if not accepted:
            # Try smaller alpha
            alpha *= 0.5
            if alpha < 1e-12:
                break
            continue
        
        # Check hard objective periodically
        if it % 20 == 0:
            hard = evaluate_c5(h)
            if hard < best_hard - 1e-10:
                best_hard = hard
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 20
        
        # Decay temperature
        tau = max(tau * tau_decay, tau_end)
        
        # Occasionally restore alpha
        if it % 50 == 0:
            alpha = max(alpha, alpha_start * 0.3)
        
        # Early stopping if no improvement for a long time
        if no_improve_count > 2000:
            break
    
    # Final hard evaluation
    final_hard = evaluate_c5(h)
    if final_hard < best_hard:
        best_hard = final_hard
        best_h = h.copy()
    
    return best_h, best_hard


class EpiProblem:
    """SLSQP epigraph problem: min t s.t. z_q - t <= 0, 0<=h<=1, sum(h)=T"""
    def __init__(self, h0):
        self.n = len(h0)
        self.T = self.n / 2.0
        self.dx = 2.0 / self.n
        self.h0 = h0.copy()
        self.call_count = 0
        self.deadline = None
    
    def objective(self, x):
        self.call_count += 1
        if self.deadline is not None and time.time() > self.deadline:
            raise KeyboardInterrupt
        return x[-1]
    
    def objective_grad(self, x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    def constraints(self, x):
        h = x[:-1]
        z = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return z - x[-1]
    
    def constraint_jac(self, x):
        n = self.n
        m = 2 * n - 1
        h = x[:-1]
        jac = np.zeros((m, n + 1))
        # Last column is -1 for all rows
        jac[:, -1] = -1.0
        # Compute gradient of each z_q w.r.t. h
        for q in range(m):
            ell = q - (n - 1)
            if ell == 0:
                jac[q, :n] = self.dx * (1.0 - 2.0 * h)
            elif ell > 0:
                L = n - ell
                jac[q, ell:n] += self.dx * (1.0 - h[:L])
                jac[q, :L] -= self.dx * h[ell:]
            else:
                a = -ell
                L = n - a
                jac[q, a:n] += self.dx * (1.0 - h[:L])
                jac[q, :L] -= self.dx * h[a:]
        return jac
    
    def solve(self, t_init, max_iter=200, time_limit=30.0):
        n = self.n
        h = project_box_sum(self.h0, self.T)
        c5 = evaluate_c5(h)
        t0 = max(c5 + 0.01, 0.05)
        x0 = np.concatenate([h, [t0]])
        
        bounds = [(0.0, 1.0)] * n + [(0.0, 0.6)]
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([self.T])
        
        self.deadline = time.time() + time_limit
        try:
            res = minimize(
                self.objective, x0,
                method='SLSQP',
                jac=self.objective_grad,
                bounds=bounds,
                constraints=[
                    {'type': 'ineq', 'fun': self.constraints, 'jac': self.constraint_jac},
                    {'type': 'eq', 'fun': lambda x: x[:n].sum() - self.T,
                     'jac': lambda x: A_eq.ravel()}
                ],
                options={'maxiter': max_iter, 'ftol': 1e-12, 'maxls': 10}
            )
        except (KeyboardInterrupt, Exception):
            self.deadline = None
            return None, None
        
        self.deadline = None
        
        if res.success or res.status == 0:
            h_new = project_box_sum(res.x[:-1], self.T)
            c5_new = evaluate_c5(h_new)
            return h_new, c5_new
        else:
            # Try the result anyway
            h_new = project_box_sum(res.x[:-1], self.T)
            c5_new = evaluate_c5(h_new)
            return h_new, c5_new


def upsample(h, factor):
    """Upsample by repeating each value 'factor' times"""
    return np.repeat(h, factor)

def downsample(h, factor):
    """Downsample by averaging blocks of 'factor'"""
    n = len(h)
    if n % factor != 0:
        # Pad if needed
        pad = factor - (n % factor)
        h = np.concatenate([h, np.zeros(pad)])
    n2 = len(h) // factor
    h2 = h.reshape(factor, n2).mean(axis=0)
    return h2

def make_feasible_random(rng, n):
    """Create a random feasible h"""
    T = n / 2.0
    h = rng.uniform(0, 1, n)
    return project_box_sum(h, T)

    shifted = (z - z_max) / tau
    shifted = np.clip(shifted, -60.0, 0.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    if s > 1e-300:
        p = exp_s / s
    else:
        p = np.zeros_like(z)
        p[int(np.argmax(z))] = 1.0
    F = z_max + tau * np.log(max(s, 1e-300))
    grad = corr_gradient(h, p)
    return F, grad
```

import numpy as np
import time
from scipy.optimize import minimize

def corr_values(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def evaluate_c5(h):
    return float(np.max(corr_values(h)))

def project_box_sum(h, T):
    x = np.asarray(h, dtype=np.float64).copy()
    lo = x.min() - 2.0
    hi = x.max() + 2.0
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.any():
            y[mask] += diff / mask.sum()
    return np.clip(y, 0.0, 1.0)

def corr_gradient(h, weights):
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = weights[q]
        if abs(w) < 1e-15:
            continue
        ell = q - (n - 1)
        if ell == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += w * dx * (1.0 - h[:L])
            grad[:L] -= w * dx * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[:L] += w * dx * (1.0 - h[a:])
            grad[a:] -= w * dx * h[:L]
    return grad

def smooth_max_grad(h, tau, top_k=None):
    z = corr_values(h)
    if top_k is not None and top_k < len(z) and top_k > 0:
        idx = np.argpartition(z, -top_k)[-top_k:]
        zw = np.zeros_like(z)
        zw[idx] = z[idx]
        z = zw
    z_max = z.max()
    shifted = (z - z_max) / tau
    shifted = np.clip(shifted, -60.0, 0.0)
    exp_s = np.exp(shifted)
    s = exp_s.sum()
    if s > 1e-300:
        p = exp_s / s
    else:
        p = np.zeros_like(z)
        p[int(np.argmax(z))] = 1.0
    F = z_max + tau * np.log(max(s, 1e-300))
    grad = corr_gradient(h, p)
    return F, grad

def pgd_optimize(h0, max_iter=3000, tau0=None, tau_end=5e-5,
                 tau_decay=0.985, alpha0=0.04, time_limit=60.0,
                 deadline=None, top_k=None, use_momentum=False):
    h = project_box_sum(np.asarray(h0, dtype=np.float64), len(h0) / 2.0)
    n = len(h)
    if tau0 is None:
        z = corr_values(h)
        tau0 = max(0.01 * (z.max() - z.min() + 1e-6), 0.004)
    tau = tau0
    alpha = alpha0
    best_hard = evaluate_c5(h)
    best_h = h.copy()
    F_prev, _ = smooth_max_grad(h, tau, top_k)
    v = np.zeros(n) if use_momentum else None
    beta = 0.9 if use_momentum else 0.0
    t0 = time.time()
    stuck = 0
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if deadline is not None and time.time() > deadline:
            break
        F, grad = smooth_max_grad(h, tau, top_k)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        if use_momentum:
            v = beta * v - grad / gnorm
            direction = v / (np.linalg.norm(v) + 1e-15)
        else:
            direction = -grad / gnorm
        accepted = False
        a = alpha
        for _ in range(25):
            h_new = project_box_sum(h + a * direction, n / 2.0)
            F_new, _ = smooth_max_grad(h_new, tau, top_k)
            if F_new <= F - 1e-13:
                h = h_new
                F = F_new
                alpha = min(a * 1.2, alpha0 * 2)
                accepted = True
                break
            a *= 0.5
        if not accepted:
            alpha *= 0.5
            if alpha < 1e-14:
                break
            continue
        if it % 10 == 0:
            hard = evaluate_c5(h)
            if hard < best_hard - 1e-11:
                best_hard = hard
                best_h = h.copy()
                stuck = 0
            else:
                stuck += 10
        tau = max(tau * tau_decay, tau_end)
        if it % 100 == 0:
            alpha = max(alpha, alpha0 * 0.4)
        if stuck > 1000:
            break
    final_hard = evaluate_c5(h)
    if final_hard < best_hard:
        best_hard = final_hard
        best_h = h.copy()
    return best_h, best_hard

class EpiSolver:
    def __init__(self, h0, top_k=None):
        self.n = len(h0)
        self.nvar = self.n + 1
        self.T = self.n / 2.0
        self.dx = 2.0 / self.n
        self.h0 = h0.copy()
        self.top_k = top_k
        self.deadline = None

    def _active_lags(self, h, t):
        z = np.correlate(h, 1.0 - h, mode='full') * self.dx
        if self.top_k is not None and self.top_k < len(z):
            idx = np.argsort(z)[::-1][:self.top_k]
        else:
            idx = np.arange(len(z))
        return idx

    def solve(self, max_iter=100, time_limit=40.0):
        n = self.n
        h = project_box_sum(self.h0, self.T)
        c5 = evaluate_c5(h)
        t0 = max(c5 + 0.02, 0.05)
        x0 = np.concatenate([h, [t0]])
        k = self.top_k if self.top_k else 2 * n - 1
        m = min(k, 2 * n - 1)
        bounds = [(0.0, 1.0)] * n + [(0.0, 0.7)]
        self.deadline = time.time() + time_limit

        def obj(x):
            if self.deadline and time.time() > self.deadline:
                raise KeyboardInterrupt
            return x[-1]

        def obj_grad(x):
            g = np.zeros(self.nvar)
            g[-1] = 1.0
            return g

        def con_fun(x):
            if self.deadline and time.time() > self.deadline:
                raise KeyboardInterrupt
            h = x[:-1]
            z = np.correlate(h, 1.0 - h, mode='full') * self.dx
            if self.top_k is not None and self.top_k < len(z):
                idx = np.argsort(z)[::-1][:self.top_k]
                return z[idx] - x[-1]
            return z - x[-1]

        def con_jac(x):
            if self.deadline and time.time() > self.deadline:
                raise KeyboardInterrupt
            h = x[:-1]
            z = np.correlate(h, 1.0 - h, mode='full') * self.dx
            if self.top_k is not None and self.top_k < len(z):
                idx = np.argsort(z)[::-1][:self.top_k]
            else:
                idx = np.arange(len(z))
            mm = len(idx)
            jac = np.zeros((mm, self.nvar))
            jac[:, -1] = -1.0
            for row_i, q in enumerate(idx):
                ell = q - (n - 1)
                dx = self.dx
                if ell == 0:
                    jac[row_i, :n] = dx * (1.0 - 2.0 * h)
                elif ell > 0:
                    L = n - ell
                    jac[row_i, ell:n] += dx * (1.0 - h[:L])
                    jac[row_i, :L] -= dx * h[ell:]
                else:
                    a = -ell
                    L = n - a
                    jac[row_i, :L] += dx * (1.0 - h[a:])
                    jac[row_i, a:n] -= dx * h[:L]
            return jac

        A_eq = np.zeros((1, self.nvar))
        A_eq[0, :n] = 1.0

        def eq_fun(x):
            return x[:n].sum() - self.T

        def eq_jac(x):
            return A_eq.ravel()

        try:
            res = minimize(
                obj, x0, method='SLSQP', jac=obj_grad,
                bounds=bounds,
                constraints=[
                    {'type': 'ineq', 'fun': con_fun, 'jac': con_jac},
                    {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
                ],
                options={'maxiter': max_iter, 'ftol': 1e-12, 'maxls': 5}
            )
        except (KeyboardInterrupt, Exception):
            self.deadline = None
            return None, None
        self.deadline = None
        h_new = project_box_sum(res.x[:-1], self.T)
        c5_new = evaluate_c5(h_new)
        return h_new, c5_new

def upsample(h, factor):
    return np.repeat(h, factor)

def downsample(h, factor):
    n = len(h)
    if n % factor != 0:
        pad = factor - (n % factor)
        h = np.concatenate([h, np.zeros(pad)])
    return h.reshape(factor, -1).mean(axis=0)

def make_feasible_random(rng, n):
    h = rng.uniform(0, 1, n)
    return project_box_sum(h, n / 2.0)
