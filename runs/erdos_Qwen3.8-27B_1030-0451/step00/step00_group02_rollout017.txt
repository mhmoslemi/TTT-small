        grad_sum += _grad_f(h, ell, n, dx)

    # Decrease h where grad > 0 (reduces active F)
    # Increase h where grad < 0 (reduces active F)

    low_grad = np.where(grad_sum < -0.01)[0]  # candidates to increase

    low_grad = low_grad[h[low_grad] < 0.99]

def _evaluate(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max())


def _evaluate_full(h):
    n = len(h)
    dx = 2.0 / n
    c = correlate(h, 1.0 - h, mode='full') * dx
    return float(c.max()), c


def _project(h, n):
    h = np.clip(h.copy(), 0.0, 1.0)
    target = n / 2.0
    for _ in range(50):
        diff = target - h.sum()
        if abs(diff) < 1e-13:
            break
        if diff > 0:
            room = 1.0 - h
            total = room.sum()
            if total < 1e-15:
                break
            h += room * diff / total
        else:
            total = h.sum()
            if total < 1e-15:
                break
            h -= h * (-diff) / total
        h = np.clip(h, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h


def _grad_f(h, ell, n, dx):
    g = np.zeros(n)
    for p in range(n):
        val = 0.0
        ip = p + ell
        im = p - ell
        if 0 <= ip < n:
            val += (1.0 - h[ip])
        if 0 <= im < n:
            val -= h[im]
        g[p] = dx * val
    return g


def _lp_direction(h, n, c_vec):
    """Find optimal direction using LP. Returns direction or None."""
    C = c_vec.max()
    dx = 2.0 / n
    
    gap = max(0.003, 0.015 * C)
    active = np.where(c_vec >= C - gap)[0]
    k = min(len(active), 15)
    # Sort by value (highest first)
    active_sorted = active[np.argsort(-c_vec[active])][:k]
    
    if len(active_sorted) == 0:
        return None
    
    # Compute gradients
    grads = []
    for r in active_sorted:
        ell = r - (n - 1)
        g = _grad_f(h, ell, n, dx)
        gn = np.linalg.norm(g, 1)
        if gn > 1e-12:
            g = g / gn
        grads.append(g)
    
    # Build LP: maximize delta
    # x = [d_0, ..., d_{n-1}, delta]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = -1.0  # minimize -delta
    
    A_ub_rows = []
    b_ub_rows = []
    
    for g in grads:
        row = np.zeros(n + 1)
        row[:n] = g
        row[-1] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(0.0)
    
    for i in range(n):
        row = np.zeros(n + 1)
        row[i] = 1.0
        A_ub_rows.append(row)
        b_ub_rows.append(1.0)
        
        row = np.zeros(n + 1)
        row[i] = -1.0
        A_ub_rows.append(row)
        b_ub_rows.append(1.0)
    
    A_ub = np.array(A_ub_rows)
    b_ub = np.array(b_ub_rows)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([0.0])
    
    bounds = [(None, None)] * n + [(-1.0, 10.0)]
    
    try:
        result = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,

    except Exception:
        return None
    
    if result is None or not result.success:
        return None
    
    d = result.x[:n]
    delta = result.x[-1]
    
    if delta < 1e-6:
        return None
    
    return d


def _line_search(h, d, n, c_current):
    """Find best step alpha. Returns (alpha, h_new) or (0, None)."""
    # Max alpha
    alpha_max = 1.0
    for i in range(n):
        if d[i] > 1e-12:
            a = (1.0 - h[i]) / d[i]
            if a < alpha_max:
                alpha_max = a
        elif d[i] < -1e-12:
            a = h[i] / (-d[i])
            if a < alpha_max:
                alpha_max = a
    
    alpha_max = max(alpha_max, 1e-8)
    alpha_max = min(alpha_max, 2.0)
    
    best_alpha = 0.0
    best_c = c_current
    best_h = None
    
    fracs = [1.0, 0.95, 0.85, 0.75, 0.65, 0.55, 0.45, 0.35, 0.25, 0.15, 0.08, 0.04, 0.02]

        j = inc_candidates[rng.randint(len(inc_candidates))]
        if i == j:
            continue

            h_trial[j] += a
            c_trial = _evaluate(h_trial)
            if c_trial < best_c - 1e-11:
                best_c = c_trial
                best_swap = (i, j, a)
    
    if best_swap is not None:
        i, j, a = best_swap
        h[i] -= a
        h[j] += a
        return True
    
    return False
