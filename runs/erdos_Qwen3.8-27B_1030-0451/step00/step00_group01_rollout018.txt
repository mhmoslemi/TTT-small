import numpy as np

def _score(h, dx):
    return float(np.max(np.correlate(h, 1.0 - h, mode="full")) * dx)


def _fix_sum(h, n):
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    # Distribute adjustment among entries with room
    if diff > 0:
        order = np.argsort(1.0 - h)
    else:
        order = np.argsort(h)
    rem = diff
    for idx in order:
        if rem > 0:
            room = 1.0 - h[idx]
            add = min(room, rem)
            if add > 1e-15:
                h[idx] += add
                rem -= add
        else:
            room = h[idx]
            sub = min(room, -rem)
            if sub > 1e-15:
                h[idx] -= sub
                rem += sub
        if abs(rem) < 1e-15:
            break
    return h


def _supergradient(h, corr, dx):
    n = len(h)
    M = 2 * n - 1
    center = n - 1
    max_corr = float(np.max(corr))
    if max_corr < 1e-15:
        return np.zeros(n)
    # Find active lags (within 5% of max)
    active_mask = corr >= max_corr * 0.92
    grad = np.zeros(n)
    for lag_idx in range(M):
        if not active_mask[lag_idx]:
            continue
        w = (corr[lag_idx] - max_corr * 0.5) / (max_corr * 0.5 + 1e-30)
        lag = lag_idx - center
        start = max(0, -lag)
        stop = min(n, n - lag)
        if stop <= start:
            continue
        i_idx = np.arange(start, stop)
        j_idx = i_idx + lag
        grad[i_idx] += w * (1.0 - h[j_idx])
        grad[j_idx] -= w * h[i_idx]
    return grad * dx


def _select_pairs(grad, h, n, rng, n_pairs=60):

        return []
    # Score each pair by grad[p] - grad[q]
    # To be efficient, pick top grad[p] and bottom grad[q]

    q_sorted = q_candidates[np.argsort(grad[q_candidates])]

    candidates.sort(key=lambda x: -x[2])
    pairs = [(p, q) for p, q, _ in candidates[:n_pairs]]
    # Add some random pairs for diversity

        p = rng.choice(p_candidates)
        q = rng.choice(q_candidates)
        if p != q and (p, q) not in pairs:
            pairs.append((p, q))
    return pairs


def _line_search(h, p, q, direction, dx, current_score):
    n = len(h)
    if direction > 0:
        alpha_max = min(h[p], 1.0 - h[q])
    else:
        alpha_max = min(h[q], 1.0 - h[p])
    if alpha_max < 1e-12:
        return None

    d = np.zeros(n)
    if direction > 0:
        d[p] = -1.0
        d[q] = 1.0
    else:
        d[p] = 1.0
        d[q] = -1.0

    corr_h = np.correlate(h, 1.0 - h, mode="full")
    bd = np.correlate(d, 1.0 - h, mode="full")
    hd = np.correlate(h, d, mode="full")
    dc = -np.correlate(d, d, mode="full")
    bcoef = bd - hd

    def obj(a):
        vals = corr_h + bcoef * a + dc * a * a
        return float(np.max(vals)) * dx

    # Scan
    n_scan = 24
    alphas = np.linspace(0.0, alpha_max, n_scan + 1)
    # Evaluate at scan points (skip alpha=0)
    scores_arr = np.empty(n_scan)
    for i in range(n_scan):
        scores_arr[i] = obj(alphas[i + 1])
    best_local = int(np.argmin(scores_arr))
    if scores_arr[best_local] >= current_score - 1e-13:
        return None
    best_a = alphas[best_local + 1]
    best_s = scores_arr[best_local]

    # Refine with bounded scalar minimization

    hi = min(alphas[min(n_scan, best_local + 2)], alpha_max)

        hi = alpha_max

                              options={'xatol': 1e-10, 'maxiter': 50})
        if res.fun < best_s:
            best_a = res.x
            best_s = res.fun
    except Exception:
        pass

    if best_s < current_score - 1e-14:
        return (best_a, best_s)
    return None

    n_old = len(h_old)
    dx_old = 2.0 / n_old
    dx_new = 2.0 / new_n
    x_old = (np.arange(n_old) + 0.5) * dx_old
    x_new = (np.arange(new_n) + 0.5) * dx_new
    h_new = np.interp(x_new, x_old, h_old, left=h_old[0], right=h_old[-1])
    # Fix sum
    target = new_n / 2.0
    diff = target - np.sum(h_new)
    if diff > 0:
        h_new += diff / new_n
    else:
        h_new += diff / new_n
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = _fix_sum(h_new, new_n)
    return h_new
