def line_search_exact(F, G, t_min, t_max, n_lags):
    """Find t in [t_min, t_max] minimizing max(F + t*G)."""
    # Evaluate at a grid
    best_t = t_min
    best_val = np.max(F + t_min * G)
    
    # Check t_max
    val_end = np.max(F + t_max * G)
    if val_end < best_val:
        best_val = val_end
        best_t = t_max
    
    # Grid evaluation
    for i in range(1, 20):
        t = t_min + (t_max - t_min) * i / 20.0
        val = np.max(F + t * G)
        if val < best_val:
            best_val = val
            best_t = t
    
    # Refine: find the two dominant lags and solve exactly
    # At the optimal t, the max is achieved by at least two lags
    # (unless it's at a boundary)
    if t_min < best_t < t_max:
        # Find dominant lags
        vals = F + best_t * G
        max_val = np.max(vals)
        dominant = np.where(vals >= max_val - 1e-10)[0]
        if len(dominant) >= 2:
            for i in range(len(dominant)):
                for j in range(i+1, len(dominant)):
                    li, lj = dominant[i], dominant[j]
                    if G[li] != G[lj]:
                        t_exact = (F[lj] - F[li]) / (G[li] - G[lj])
                        if t_min <= t_exact <= t_max:
                            val_exact = np.max(F + t_exact * G)
                            if val_exact < best_val:
                                best_val = val_exact
                                best_t = t_exact
    
    return best_t, best_val

import numpy as np
import time
import math
import random

def exact_score(h, n):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return float(np.max(corr) * dx)

def correlation_vector(h, n):
    """Compute all lag values F(l) = sum h[i+l]*(1-h[i]) * dx"""
    dx = 2.0 / n
    q = 1.0 - h
    corr = np.correlate(h, q, mode="full")
    return corr * dx  # length 2n-1

def compute_coeff(h, n, p, l_offset):
    """Compute coeff(p, l) for lag l. l_offset is the index into the correlation array."""
    # lag l = l_offset - (n-1)
    l = l_offset - (n - 1)
    i_lo = max(0, -l)
    i_hi = min(n - 1, n - 1 - l)
    
    coeff = 0.0
    a = p - l
    if i_lo <= a <= i_hi:
        coeff += (1.0 - h[a])
    if i_lo <= p <= i_hi:

                            if val_exact < best_val:
                                best_val = val_exact
                                best_t = t_exact

def line_search(h, n, p, q, F):
    """Exact line search for pair (p,q). Returns (best_t, best_score)."""
    t_lo = max(-h[p], h[q] - 1.0)
    t_hi = min(1.0 - h[p], h[q])
    if t_hi - t_lo < 1e-12:
        return 0.0, np.max(F)
    
    G = coeff_vec(h, n, p) - coeff_vec(h, n, q)
    
    # Grid + refinement
    best_t = 0.0
    best_v = np.max(F)
    
    # Include t=0 check
    n_pts = 25
    ts = np.linspace(t_lo, t_hi, n_pts + 1)
    for t in ts:
        v = np.max(F + t * G)
        if v < best_v:
            best_v = v
            best_t = t
    
    # Refine with dominant lags
    if t_lo < best_t < t_hi:
        vals = F + best_t * G
        mx = np.max(vals)
        dom = np.where(vals >= mx - 1e-13)[0]
        if len(dom) >= 2:
            d = dom[:8]
            for i in range(len(d)):
                for j in range(i+1, len(d)):
                    dG = G[d[i]] - G[d[j]]
                    if abs(dG) > 1e-13:
                        t_ex = (F[d[j]] - F[d[i]]) / dG
                        if t_lo <= t_ex <= t_hi:
                            v_ex = np.max(F + t_ex * G)
                            if v_ex < best_v:
                                best_v = v_ex
                                best_t = t_ex
    return best_t, best_v

def pair_descent(h, n, t_start, t_end, rng):

    best_h = h.copy()

    F = corr_vector(h, n)
    maxF = np.max(F)
    stall = 0
