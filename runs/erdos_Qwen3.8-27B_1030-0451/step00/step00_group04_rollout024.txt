```python
def project_box_sum(h, T):
    h = np.clip(h, 0, 1)
    # Find alpha such that sum(clip(h + alpha, 0, 1)) = T
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0, 1))
        if s < T:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2
    y = np.clip(h + alpha, 0, 1)
    # Fine correction
    err = T - np.sum(y)
    if abs(err) > 1e-10:
        interior = (y > 1e-10) & (y < 1 - 1e-10)
        n_int = np.sum(interior)
        if n_int > 0:
            y[interior] += err / n_int
            y = np.clip(y, 0, 1)
    return y
```

```python
import numpy as np
import scipy.optimize
import time

def exact_score(h):
    """Compute C(h) = max_k integral h(x)(1-h(x+k)) dx using discrete correlation."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def exact_z_vector(h):
    """Compute all z_k * dx values."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_z_lag(h, ell):
    """Compute gradient of z_ell (unscaled) where ell = q - (n-1) for full correlation index q."""
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        a = ell
        # z_a = sum_{i=0}^{n-1-a} h[i+a]*(1-h[i])
        # grad[j] = [a<=j]*(1-h[j-a]) - [j<=n-1-a]*h[j+a]
        if a < n:
            L = n - a
            # For j in [a, n-1]: += (1 - h[j-a])
            g[a:] += (1.0 - h[:L])
            # For j in [0, L-1]: -= h[j+a]
            g[:L] -= h[a:]
    else:
        a = -ell
        if a < n:
            L = n - a
            # z_{-a} = sum_{i=0}^{L-1} h[i]*(1-h[i+a])
            # grad[j] = [j<=L-1]*(1-h[j+a]) - [j>=a]*h[j-a]
            g[:L] += (1.0 - h[a:])
            g[a:] -= h[:L]
    return g

def project_box_sum(h, T):
    """Project h to [0,1]^n with sum = T using bisection on shift."""
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    y = np.clip(h + alpha, 0.0, 1.0)
    # Fine correction
    err = T - np.sum(y)
    if abs(err) > 1e-12:
        interior = (y > 1e-12) & (y < 1.0 - 1e-12)
        n_int = np.sum(interior)
        if n_int > 0:
            y[interior] += err / n_int
            np.clip(y, 0.0, 1.0, out=y)
    return y

def solve_slp_step(h, z_vec, active_indices, rho):
    """Solve one SLP LP step. Returns (y, t) or (None, None)."""
    n = len(h)
    dx = 2.0 / n
    T = n / 2.0
    K = len(active_indices)
    
    c = np.zeros(n + 1)
    c[n] = 1.0
    
    A_ub = np.zeros((K, n + 1))
    b_ub = np.zeros(K)
    
    for idx, q in enumerate(active_indices):
        ell = q - (n - 1)
        g = grad_z_lag(h, ell) * dx
        A_ub[idx, :n] = g
        A_ub[idx, n] = -1.0
        b_ub[idx] = g @ h - z_vec[q]
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([T])
    
    bounds = [(max(0.0, h[i] - rho), min(1.0, h[i] + rho)) for i in range(n)]
    bounds.append((0.0, 1.0))
    
    try:
        res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                                      bounds=bounds, method='highs')
        if res.success:
            y = res.x[:n].copy()
            t = float(res.x[n])
            # Fix tiny numerical issues
            y = np.clip(y, 0.0, 1.0)
            err = T - np.sum(y)
            if abs(err) > 1e-10:
                interior = (y > 1e-10) & (y < 1.0 - 1e-10)
                if np.sum(interior) > 0:
                    y[interior] += err / np.sum(interior)
                    np.clip(y, 0.0, 1.0, out=y)
            return y, t
    except Exception:
        pass
    return None, None

def slp_optimize(h, time_limit):
    """Run SLP optimization from starting point h."""
    n = len(h)
    T = n / 2.0
    h = project_box_sum(h.copy(), T)
    score = exact_score(h)
    best_h = h.copy()
    best_score = score
    rho = 0.03
    stall = 0
    start = time.time()
    max_iters = 500
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        z_vec = exact_z_vector(h)
        score = float(np.max(z_vec))
        
        # Select active lags (top K by z value)
        K = min(60, 2 * n - 1)
        order = np.argsort(-z_vec)
        active = order[:K].tolist()
        # Ensure argmax is included
        argmax = int(np.argmax(z_vec))
        if argmax not in active:
            active[0] = argmax
        
        # Solve LP
        y, t_opt = solve_slp_step(h, z_vec, active, rho)
        if y is None:
            rho *= 0.5
            stall += 1
            if stall > 30:
                break
            continue
        
        # Line search
        improved = False
        for alpha in [1.0, 0.9, 0.7, 0.5, 0.3, 0.1]:
            h_try = (1.0 - alpha) * h + alpha * y
            h_try = np.clip(h_try, 0.0, 1.0)
            # Fix sum (should already be correct due to convex combo of feasible points)
            # But floating point might introduce tiny errors
            err = T - np.sum(h_try)
            if abs(err) > 1e-10:
                h_try = project_box_sum(h_try, T)
            s_try = exact_score(h_try)
            if s_try < score - 1e-12:
                h = h_try
                score = s_try
                improved = True
                if score < best_score:
                    best_score = score
                    best_h = h.copy()
                break
        
        if improved:
            stall = 0
            rho = min(rho * 1.1, 0.08)
        else:
            stall += 1
            rho *= 0.7
            if rho < 1e-6:
                break
    
    return best_h, best_score

def pair_polish(h, time_limit, max_rounds=50):
    """Polish using pairwise transfers."""
    n = len(h)
    T = n / 2.0
    dx = 2.0 / n
    h = h.copy()
    score = exact_score(h)
    best_h = h.copy()
    best_score = score
    start = time.time()
    
    for rnd in range(max_rounds):
        if time.time() - start > time_limit:
            break
```

import scipy.optimize
import time

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def exact_z_vector(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def grad_z_lag(h, ell):
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        a = ell
        if a < n:
            L = n - a
            g[a:] += (1.0 - h[:L])
            g[:L] -= h[a:]
    else:
        a = -ell
        if a < n:
            L = n - a
            g[:L] += (1.0 - h[a:])
            g[a:] -= h[:L]
    return g
