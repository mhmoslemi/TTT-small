def repair_profile(arr):
    """Ensure 0 <= arr <= 1 and sum(arr) = len(arr)/2"""
    arr = np.array(arr, dtype=np.float64, copy=True)
    n = len(arr)
    target = n / 2.0
    arr = np.clip(arr, 0.0, 1.0)
    s = arr.sum()
    diff = s - target
    if abs(diff) < 1e-12:
        return arr, True
    # Adjust: if diff > 0, need to remove; if diff < 0, need to add
    if diff > 0:
        # Remove from largest values first (but keep >= 0)
        order = np.argsort(-arr)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            take = min(arr[idx], remaining)
            arr[idx] -= take
            remaining -= take
    else:
        diff = -diff
        order = np.argsort(arr)
        remaining = diff
        for idx in order:
            if remaining <= 1e-14:
                break
            take = min(1.0 - arr[idx], remaining)
            arr[idx] += take
            remaining -= take
    # Final clip for safety
    arr = np.clip(arr, 0.0, 1.0)
    # Fix residual
    residual = target - arr.sum()
    if abs(residual) > 1e-14:
        # Spread residual across all elements
        arr += residual / n
        arr = np.clip(arr, 0.0, 1.0)

            if time_left() < 10:
                break
            t_limit = min(60, time_left() * 0.15)
            h_new, c5_new = smooth_max_descent(best_h, [c], max_iters=3000,
                                               time_limit=t_limit, start_time=time.monotonic())
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 2: Coordinate descent
    if time_left() > 10:
        for _ in range(3):
            if time_left() < 10:
                break
            t_limit = min(80, time_left() * 0.2)
            h_new, c5_new = coordinate_descent(best_h, max_iters=20000,
                                               time_limit=t_limit, start_time=time.monotonic(), rng=rng)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 3: Pairwise local search with annealing
    if time_left() > 15:
        h_new, c5_new = pairwise_local_search_clean(best_h, max_iters=500000,
                                                    time_limit=min(300, time_left() * 0.4),
                                                    start_time=time.monotonic(), rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 4: More gradient descent at higher temperatures
    if time_left() > 10:
        for c in [150.0, 300.0, 500.0]:
            if time_left() < 10:
                break
            t_limit = min(40, time_left() * 0.15)
            h_new, c5_new = smooth_max_descent(best_h, [c], max_iters=2000,
                                               time_limit=t_limit, start_time=time.monotonic())
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 5: Another round of coordinate descent + local search
    if time_left() > 10:
        t_limit = min(60, time_left() * 0.2)
        h_new, c5_new = coordinate_descent(best_h, max_iters=30000,
                                           time_limit=t_limit, start_time=time.monotonic(), rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    if time_left() > 15:
        h_new, c5_new = pairwise_local_search_clean(best_h, max_iters=300000,
                                                    time_limit=min(200, time_left() * 0.3),
                                                    start_time=time.monotonic(), rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Phase 6: Try different resolutions
    if time_left() > 30:
        for target_n in [129, 133, 135, 127, 137]:
            if time_left() < 20:
                break
            h_resample = resample_to_n(best_h, target_n)
            # Quick gradient descent
            t_limit = min(30, time_left() * 0.15)
            h_new, c5_new = smooth_max_descent(h_resample, [80.0, 150.0], max_iters=1500,
                                               time_limit=t_limit, start_time=time.monotonic())
            if time_left() > 10:
                h_new2, c5_new2 = coordinate_descent(h_new, max_iters=10000,
                                                     time_limit=min(30, time_left() * 0.15),
                                                     start_time=time.monotonic(), rng=rng)
                if c5_new2 < c5_new:
                    c5_new, h_new = c5_new2, h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Phase 7: Final polishing
    if time_left() > 10:
        for c in [100.0, 200.0]:
            if time_left() < 5:
                break
            t_limit = min(30, time_left() * 0.3)
            h_new, c5_new = smooth_max_descent(best_h, [c], max_iters=3000,
                                               time_limit=t_limit, start_time=time.monotonic())
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new
    
    # Final local search
    if time_left() > 10:
        h_new, c5_new = pairwise_local_search_clean(best_h, max_iters=200000,
                                                    time_limit=min(100, time_left() * 0.5),
                                                    start_time=time.monotonic(), rng=rng)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    # Final repair and evaluation
    best_h = repair_profile(best_h)
    final_c5, _ = evaluate_c5(best_h)
    
    return best_h, final_c5, len(best_h)
