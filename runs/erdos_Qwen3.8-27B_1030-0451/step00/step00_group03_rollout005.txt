def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def build_h_sym(p, n):
    m = n // 2
    if n % 2 == 0:
        return np.concatenate([p, 1.0 - p[::-1]])
    else:
        return np.concatenate([p, np.array([0.5]), 1.0 - p[::-1]])

def sym_p_from_h(h, n):
    m = n // 2
    off = n % 2
    p1 = h[:m].copy()
    p2 = (1.0 - h[m+off:n])[::-1]
    return np.clip(0.5 * (p1 + p2), 0.0, 1.0)

def interp_p(p_old, m_new):
    m_old = len(p_old)
    if m_old == m_new:
        return p_old.copy()
    x_old = (np.arange(m_old) + 0.5) / m_old
    x_new = (np.arange(m_new) + 0.5) / m_new
    return np.clip(np.interp(x_new, x_old, p_old), 0.0, 1.0)

    if c < state['best_val']:
        state['best_val'] = c
        state['best_h'] = h_new.copy()
    return c
