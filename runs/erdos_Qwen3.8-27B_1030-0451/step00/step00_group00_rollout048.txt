def project_to_constraint(x, target_sum, n):
    """Project x onto {y: 0<=y_i<=1, sum y_i = target_sum}."""
    # Bisection on lambda: sum clip(x_i - lambda, 0, 1) = target_sum
    lo, hi = 0.0, 1.0
    
    def f(lam):
        return np.sum(np.clip(x - lam, 0, 1)) - target_sum
    
    # f(lo) = sum(x) - target_sum (if all x in [0,1])
    # f(hi) = sum(clip(x-1, 0, 1)) - target_sum = -target_sum <= 0
    
    # Actually, we need:
    # When lambda is very negative, clip(x-lambda, 0, 1) → 1 for all x>0, so sum → n (for x_i > 0)
    # When lambda is very positive, clip(x-lambda, 0, 1) → 0, so sum → 0
    
    # Let's adjust bounds
    lo = x.min() - 1  # ensures sum > target
    hi = x.max() + 1  # ensures sum < target
    
    for _ in range(60):
        mid = (lo + hi) / 2
        if f(mid) > 0:
            lo = mid
        else:
            hi = mid
    
    lam = (lo + hi) / 2
    return np.clip(x - lam, 0, 1)

    top_idx = np.argsort(h)[::-1][:n // 2]
    b[top_idx] = 1.0
    cur_c5 = _eval(b)
    if cur_c5 < best_c5:
        best_c5 = cur_c5
        best_h = b.copy()
    temp = 0.0001
    cool = 0.99998
    t_start = time.time()
    ones_list = list(top_idx)
    while time.time() - t_start < budget_s:
        if budget_s - (time.time() - t_start) < 3:
            break
        # Pick a 1-position and a 0-position
        i = ones_list[np.random.randint(len(ones_list))]
        zeros = np.setdiff1d(np.arange(n), top_idx, assume_unique=False)
        if len(zeros) == 0:
            break
        j = zeros[np.random.randint(len(zeros))]
        # Swap: b[i]=0, b[j]=1
        b[i], b[j] = 0.0, 1.0
        new_c5 = _eval(b)
        dc = new_c5 - cur_c5
        if dc < 0 or (temp > 1e-12 and np.random.random() < np.exp(-max(0.0, dc) / temp)):
            cur_c5 = new_c5
            ones_list[ones_list.index(i)] = j
            top_idx = np.array(ones_list)
            if cur_c5 < best_c5:
                best_c5 = cur_c5
                best_h = b.copy()
        else:
            b[i], b[j] = 1.0, 0.0
        temp *= cool
        if temp < 1e-12:
            temp = 0.0001
    return best_c5, best_h

def run(seed=42, budget_s=1100, **kwargs):
    np.random.seed(seed)
    t0 = time.time()

    def _remain():
        return budget_s - (time.time() - t0)

    # Get initial values
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
    except (NameError, Exception):
        pass

    best_h = None
    best_c5 = 1.0

    if h_init is not None:
        best_h = h_init.copy().astype(float)
        best_c5 = _eval(best_h)

    # Determine n values to try (even, diverse)
    n_list = []
    if h_init is not None:
        n_list.append(len(h_init))
    for n in [338, 256, 384, 512, 448, 676]:
        if n % 2 == 0 and n not in n_list:
            n_list.append(n)

    for n in n_list:
        if _remain() < 60:
            break

        # Initialize h for this n
        if h_init is not None and n == len(h_init):
            h = h_init.copy().astype(float)
        else:
            h = np.zeros(n)
            for i in range(n // 2):
                h[2 * i] = 1.0
            # Shuffle a bit for diversity
            for _ in range(50):
                a, b_ = np.random.randint(0, n, 2)
                h[a], h[b_] = h[b_], h[a]

        target = n / 2.0
        h = project_sum(h, target).copy()
        cur_c5 = _eval(h)
        if cur_c5 < best_c5:
            best_c5 = cur_c5
            best_h = h.copy()

        # Phase 1: Gradient descent
        if _remain() > 40:
            g_budget = min(100, _remain() * 0.25)
            h, cur_c5, best_c5, best_h = _gradient_phase(h, n, target, g_budget, best_c5, best_h)

        # Try binarization
        b = np.zeros(n)
        top_idx = np.argsort(h)[::-1][:n // 2]
        b[top_idx] = 1.0
        b_c5 = _eval(b)
        if b_c5 < best_c5:
            best_c5 = b_c5
            best_h = b.copy()

        # Phase 2: Local search (continuous)
        if _remain() > 30:
            ls_budget = min(60, _remain() * 0.3)
            h_ls = project_sum(h, target).copy()
            if b_c5 < cur_c5:
                h_ls = b.copy()
            best_c5, best_h = _local_search(h_ls, n, target, ls_budget, best_c5, best_h)

        # Phase 3: Binary local search
        if _remain() > 20:
            bs_budget = min(40, _remain() * 0.25)
            best_c5, best_h = _binary_local_search(h, n, bs_budget, best_c5, best_h)

    return best_h, best_c5, len(best_h)
