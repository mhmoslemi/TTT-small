def exact_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def build_h_from_p(p, n):
    """Build full h from half-vector p using mirror complement. n must be even."""
    p = np.clip(np.asarray(p, dtype=np.float64), 0.0, 1.0)
    m = n // 2
    h = np.empty(n)
    h[:m] = p
    h[m:] = 1.0 - p[::-1]
    return h

def project_feasible(y):
    """Project y onto {0<=x<=1, sum(x)=n/2} via bisection on shift."""
    n = len(y)
    target = n / 2.0
    y = np.asarray(y, dtype=np.float64)
    lo, hi = -1.0, 1.0
    for _ in range(80):

        s = np.clip(y + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    alpha = 0.5 * (lo + hi)
    h = np.clip(y + alpha, 0.0, 1.0)
    # Fine correction
    diff = target - h.sum()
    if abs(diff) > 1e-12:

        nf = free.sum()
        if nf > 0:
