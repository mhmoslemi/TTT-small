def exact_c5(h):
    h=np.asarray(h,float)

import time

from scipy.optimize import minimize, linprog

_SMOOTH_TAU = 1e-3

GAMMA_GRID = (1.0, 0.8, 0.6, 0.4, 0.25, 0.15, 0.1, 0.06, 0.03, 0.015, 0.0075)
PAIR_FRACS = (1.0, 0.8, 0.6, 0.4, 0.25, 0.12, 0.06, 0.03)


def is_feasible(h, tol=1e-6):
    if h.ndim != 1 or h.size == 0:
        return False
    if not np.all(np.isfinite(h)):
        return False
    if float(np.min(h)) < -tol or float(np.max(h)) > 1.0 + tol:
        return False
    return abs(float(h.sum()) - 0.5 * h.size) <= tol * max(1, h.size)


def project_to_box_sum(h, target=None):
    h = np.asarray(h, dtype=float).copy()
    if h.ndim != 1:
        h = h.ravel()
    n = h.size
    if n == 0:
        return h
    if target is None:
        target = 0.5 * n
    target = float(target)

    if not np.all(np.isfinite(h)):
        h = np.where(np.isfinite(h), h, 0.0)
    h = np.clip(h, 0.0, 1.0)

    if abs(float(h.sum()) - target) <= 1e-11 * max(1, n):
        return h

    lo, hi = -1.0, 1.0
    for _ in range(60):
        mu = 0.5 * (lo + hi)
        val = np.clip(h + mu, 0.0, 1.0)
        s = float(val.sum())
        if s < target:
            lo = mu
        else:
            hi = mu

    val = np.clip(h + 0.5 * (lo + hi), 0.0, 1.0)

    for _ in range(30):
        res = target - float(val.sum())
        if abs(res) < 1e-13:
            break
        if res > 0:
            idx = np.where(1.0 - val > 1e-12)[0]
        else:
            idx = np.where(val > 1e-12)[0]
        if idx.size == 0:
            break
        if res > 0:
            cap = 1.0 - val[idx]
            tot = float(cap.sum())
            if tot <= 0.0:
                break
            add = min(res, tot)
            val[idx] += add * cap / tot
        else:
            cap = val[idx]
            tot = float(cap.sum())
            if tot <= 0.0:
                break
            add = min(-res, tot)
            val[idx] -= add * cap / tot

    res = target - float(val.sum())
    if abs(res) > 1e-15:
        if res > 0:
            idx = int(np.argmax(1.0 - val))
        else:
            idx = int(np.argmax(val))
        val[idx] = float(np.clip(val[idx] + res, 0.0, 1.0))
    return val


def exact_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 1e9
    dx = 2.0 / float(n)
    return float(dx * np.max(np.correlate(h, 1.0 - h, mode="full")))


def score_vec(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    dx = 2.0 / float(n)
    return dx * np.correlate(h, 1.0 - h, mode="full")


def transfer_step(h_src, N):
    n = int(len(h_src))
    N = int(N)
    if N <= 0:
        return np.array([], dtype=float)
    xs = (np.arange(N) + 0.5) * (2.0 / N)
    idx = np.floor(xs * (n / 2.0)).astype(int)
    idx = np.clip(idx, 0, n - 1)
    y = h_src[idx]
    return project_to_box_sum(y, 0.5 * N)


def transfer_linear(h_src, N):
    n = int(len(h_src))
    N = int(N)
    if N <= 0:
        return np.array([], dtype=float)
    xs_src = (np.arange(n) + 0.5) * (2.0 / n)
    xs_new = (np.arange(N) + 0.5) * (2.0 / N)
    y = np.interp(xs_new, xs_src, h_src)
    return project_to_box_sum(y, 0.5 * N)


def make_balanced_from_values(values):
    vals = np.asarray(values, dtype=float)
    if not np.all(np.isfinite(vals)):
        vals = np.where(np.isfinite(vals), vals, 0.0)
    n = vals.size
    out = np.zeros(n, dtype=float)
    if n == 0:
        return out
    k = n // 2
    order = np.argsort(vals, kind="stable")
    if k > 0:
        out[order[:k]] = 1.0
    if n % 2 == 1 and k < n:
        out[order[k]] = 0.5
    return out


def random_h(rng, n):
    n = int(n)
    out = np.zeros(n, dtype=float)
    if n == 0:
        return out
    k = n // 2
    if k > 0:
        idx = rng.choice(n, k, replace=False)
        out[idx] = 1.0
    if n % 2 == 1:
        zeros = np.where(out == 0.0)[0]
        if zeros.size > 0:
            out[int(rng.choice(zeros))] = 0.5
    return out


def to_binary(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    out = np.zeros(n, dtype=float)
    if n == 0:
        return out
    k = n // 2
    top_start = n - k
    order = np.argsort(h, kind="stable")
    if top_start < n:
        out[order[top_start:]] = 1.0
    if n % 2 == 1 and top_start - 1 >= 0:
        out[order[top_start - 1]] = 0.5
    return out


def generate_shifts(h, rng, max_candidates=12):
    h = np.asarray(h, dtype=float)
    n = h.size
    res = []
    if n <= 1:
        return res
    if n <= 250:
        shifts = np.arange(1, n)
    else:
        m = min(int(max_candidates), n - 1)
        shifts = rng.choice(n - 1, m, replace=False) + 1
    for s in shifts:
        cand = np.roll(h, int(s))
        c = exact_c5(cand)
        if np.isfinite(c):
            res.append((cand, float(c), int(n)))
    return res


def generate_structured(n):
    n = int(n)
    idx = np.arange(n)
    res = []
    if n == 0:
        return res
    thetas = [
        math.sqrt(2.0) - 1.0,
        (math.sqrt(5.0) - 1.0) / 2.0,
        2.0 - math.sqrt(3.0),
        (3.0 - math.sqrt(5.0)) / 2.0,
    ]
    phases = [0.0, 0.25, 0.5, 0.75]
    for th in thetas:
        for ph in phases:
            vals = np.mod(idx * th + ph, 1.0)
            res.append(make_balanced_from_values(vals))
            res.append(make_balanced_from_values(-vals))
    for p in (1, 2, 3, 5):
        for ph in phases:
            vals = np.sin(2.0 * math.pi * (idx * float(p) / float(n) + ph))
            res.append(make_balanced_from_values(vals))
            res.append(make_balanced_from_values(-vals))
    return res


def add_candidate(queue, h):
    if h is None:
        return
    h = np.asarray(h, dtype=float)
    if h.ndim != 1:
        h = h.ravel()
    if h.size == 0:
        return
    h = project_to_box_sum(h)
    try:
        c = exact_c5(h)
    except Exception:
        return
    if np.isfinite(c):
        queue.append((h, float(c), int(h.size)))


def pop_best(queue):
    best_idx = -1
    best_key = (1e9, 1e9)
    for idx, item in enumerate(queue):
        key = (float(item[1]), int(item[2]))
        if key < best_key:
            best_key = key
            best_idx = idx
    if best_idx < 0:
        return None
    return queue.pop(best_idx)


def lse_weights(F, tau):
    F = np.asarray(F, dtype=float)
    w = np.zeros_like(F, dtype=float)
    if F.size == 0:
        return w
    Fmax = float(np.max(F))
    if tau <= 1e-15:
        w[int(np.argmax(F))] = 1.0
        return w
    adv = (F - Fmax) / tau
    adv = np.clip(adv, -50.0, 0.0)
    z = np.exp(adv)
    s = float(z.sum())
    if s < 1e-300:
        w[int(np.argmax(F))] = 1.0
    else:
        w = z / s
    return w


def active_weights(F, gap_frac=0.005, min_count=3):
    F = np.asarray(F, dtype=float)
    if F.size == 0:
        return np.zeros_like(F, dtype=float), 0.0
    C = float(np.max(F))
    gap = max(1e-12, gap_frac * C)
    w = (F >= C - gap).astype(float)
    if float(w.sum()) < min_count:
        m = min(int(min_count), F.size)
        idx = np.argsort(F, kind="stable")[-m:]
        w[idx] = 1.0
    s = float(w.sum())
    if s > 0.0:
        w = w / s
    return w, C


def lag_gradient(h, lag):
    h = np.asarray(h, dtype=float)
    n = h.size
    g = np.zeros(n, dtype=float)
    if n == 0:
        return g
    dx = 2.0 / float(n)
    i_lo = max(0, -lag)
    i_hi = min(n, n - lag)
    if i_lo >= i_hi:
        return g
    idx_i = np.arange(i_lo, i_hi)
    idx_p = idx_i + lag
    g[idx_p] += dx * (1.0 - h[idx_i])
    g[idx_i] -= dx * h[idx_i + lag]
    return g


def weighted_gradient(h, w):
    h = np.asarray(h, dtype=float)
    w = np.asarray(w, dtype=float)
    n = h.size
    grad = np.zeros(n, dtype=float)
    if n == 0 or w.size == 0:
        return grad
    dx = 2.0 / float(n)
    lags = np.arange(w.size) - (n - 1)
    nz = np.flatnonzero(np.abs(w) > 1e-16)
    for pos in nz:
        pos = int(pos)
        wk = float(w[pos])
        lag = int(lags[pos])
        i_lo = max(0, -lag)
        i_hi = min(n, n - lag)
        if i_lo >= i_hi:
            continue
        idx_i = np.arange(i_lo, i_hi)
        idx_p = idx_i + lag
        grad[idx_p] += wk * dx * (1.0 - h[idx_i])
        grad[idx_i] -= wk * dx * h[idx_i + lag]
    return grad


def lse_obj_grad(x, tau):
    x = np.asarray(x, dtype=float)
    F = score_vec(x)
    Fmax = float(np.max(F))
    if tau <= 1e-15:
        idx = int(np.argmax(F))
        w = np.zeros_like(F, dtype=float)
        w[idx] = 1.0
        obj = Fmax
    else:
        adv = (F - Fmax) / tau
        adv = np.clip(adv, -50.0, 0.0)
        z = np.exp(adv)
        s = float(z.sum())
        if s < 1e-300:
            idx = int(np.argmax(F))
            w = np.zeros_like(F, dtype=float)
            w[idx] = 1.0
            obj = Fmax
        else:
            obj = Fmax + tau * float(np.log(s / float(F.size)))
            w = z / s
    grad = weighted_gradient(x, w)
    return obj, grad


def extreme_point(g, target):
    g = np.asarray(g, dtype=float)
    n = g.size
    v = np.zeros(n, dtype=float)
    if n == 0:
        return v
    target = float(target)
    k = int(np.floor(target + 1e-12))
    if k > n:
        k = n
    order = np.argsort(g, kind="stable")
    if k > 0:
        v[order[:k]] = 1.0
    frac = target - float(k)
    if frac > 1e-12 and k < n:
        v[order[k]] = frac
    return v


def direction_line_search(h, d, C, target):
    h = np.asarray(h, dtype=float)
    d = np.asarray(d, dtype=float)
    if h.size == 0 or d.size != h.size:
        return None, float(C)
    norm = float(np.sum(np.abs(d)))
    if norm <= 1e-12:
        return None, float(C)
    d = d / norm

    max_step = 1e300
    pos = d > 1e-12
    if np.any(pos):
        vals = (1.0 - h[pos]) / d[pos]
        ms = float(np.min(vals))
        if ms < max_step:
            max_step = ms
    neg = d < -1e-12
    if np.any(neg):
        vals = h[neg] / (-d[neg])
        ms = float(np.min(vals))
        if ms < max_step:
            max_step = ms

    if not np.isfinite(max_step) or max_step <= 1e-12:
        return None, float(C)

    best_h = None
    best_C = float(C)
    for gamma in GAMMA_GRID:
        beta = gamma * max_step
        if beta <= 1e-12:
            continue
        cand = h + beta * d
        if cand.min() < -1e-9 or cand.max() > 1.0 + 1e-9:
            cand = np.clip(cand, 0.0, 1.0)
        if abs(float(cand.sum()) - float(target)) > 1e-8:
            cand = project_to_box_sum(cand, float(target))
        c = exact_c5(cand)
        if c < best_C - 1e-12:
            best_C = c
            best_h = cand.copy()
    return best_h, best_C


def fw_line_search(h, v, C, target):
    h = np.asarray(h, dtype=float)
    v = np.asarray(v, dtype=float)
    if h.size == 0 or h.size != v.size:
        return None, float(C)
    best_h = None
    best_C = float(C)
    for gamma in GAMMA_GRID:
        cand = h + gamma * (v - h)
        if cand.min() < -1e-9 or cand.max() > 1.0 + 1e-9:
            cand = np.clip(cand, 0.0, 1.0)
        if abs(float(cand.sum()) - float(target)) > 1e-8:
            cand = project_to_box_sum(cand, float(target))
        c = exact_c5(cand)
        if c < best_C - 1e-12:
            best_C = c
            best_h = cand.copy()
    return best_h, best_C


def pair_moves(h, C, g, limit_pairs=80, time_stop=None):
    h = np.asarray(h, dtype=float)
    g = np.asarray(g, dtype=float)
    n = h.size
    target = 0.5 * n
    best_h = None
    best_C = float(C)
    if n == 0:
        return best_h, best_C

    B = 18 if n > 220 else (28 if n > 120 else min(n, 35))
    order = np.argsort(g, kind="stable")
    low = order[:B]
    high = order[-B:][::-1]

    pairs = []
    for ii in range(B):
        i = int(high[ii])
        hi = float(h[i])
        if hi <= 1e-10:
            continue
        gi = float(g[i])
        for jj in range(B):
            j = int(low[jj])
            if i == j:
                continue
            hj = float(h[j])
            if hj >= 1.0 - 1e-10:
                continue
            diff = gi - float(g[j])
            if diff > 1e-13:
                pairs.append((diff, i, j))

    pairs.sort(reverse=True)
    fracs = PAIR_FRACS if n <= 220 else PAIR_FRACS[:5]
    limit = min(int(limit_pairs), len(pairs))

    for p in range(limit):
        if time_stop is not None and time.monotonic() > time_stop:
            break
        _diff, i, j = pairs[p]
        max_t = min(float(h[i]), 1.0 - float(h[j]))
        if max_t < 1e-8:
            continue
        for frac in fracs:
            t = frac * max_t
            cand = h.copy()
            cand[i] -= t
            cand[j] += t
            if cand[i] < -1e-12 or cand[j] > 1.0 + 1e-12:
                cand = np.clip(cand, 0.0, 1.0)
                if abs(float(cand.sum()) - target) > 1e-8:
                    cand = project_to_box_sum(cand, target)
            c = exact_c5(cand)
            if c < best_C - 1e-12:
                best_C = c
                best_h = cand.copy()
    return best_h, best_C


def lp_direction(h, F, active_indices, theta=0.01, rho=1.5):
    h = np.asarray(h, dtype=float)
    F = np.asarray(F, dtype=float)
    n = h.size
    if n == 0 or len(active_indices) == 0:
        return None
    try:
        valid = []
        for x in active_indices:
            xi = int(x)
            if 0 <= xi < len(F):
                valid.append(xi)
        M = len(valid)
        if M == 0:
            return None

        nvar = 2 * n + 1 + M
        c = np.zeros(nvar, dtype=float)
        c[2 * n] = -1.0
        c[2 * n + 1:] = rho

        R = 3 * n + 2 + M
        A = np.zeros((R, nvar), dtype=float)
        b = np.zeros(R, dtype=float)
        r = 0

        for i in range(n):
            A[r, i] = 1.0
            A[r, n + i] = -1.0
            b[r] = 0.0
            r += 1
        for i in range(n):
            A[r, i] = -1.0
            A[r, n + i] = -1.0
            b[r] = 0.0
            r += 1

        A[r, n:2 * n] = 1.0
        b[r] = 1.0
        r += 1

        for i in range(n):
            A[r, n + i] = 1.0
            b[r] = 1.0
            r += 1

        A[r, 2 * n] = 1.0
        b[r] = float(theta)
        r += 1

        for mpos, idx in enumerate(valid):
            lag = int(idx) - (n - 1)
            g = lag_gradient(h, lag)
            A[r, :n] = g
            A[r, 2 * n] = 1.0
            A[r, 2 * n + 1 + mpos] = -1.0
            b[r] = 0.0
            r += 1

        A_eq = np.zeros((1, nvar), dtype=float)
        A_eq[0, :n] = 1.0
        b_eq = np.zeros(1, dtype=float)

        bounds = [(-1.0, 1.0)] * n + [(0.0, 1.0)] * n
        bounds.append((0.0, float(theta)))
        bounds += [(0.0, 10.0)] * M

        res = None
        for meth in ("highs", "interior-point"):
            try:
                if meth == "highs":
                    res = linprog(
                        c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method=meth, options={"presolve": True}
                    )
                else:
                    res = linprog(
                        c, A_ub=A, b_ub=b, A_eq=A_eq, b_eq=b_eq,
                        bounds=bounds, method=meth
                    )
                if res is not None and res.success:
                    break
                res = None
            except Exception:
                res = None

        if res is not None and getattr(res, "x", None) is not None:
            d = np.asarray(res.x[:n], dtype=float)
            delta = float(res.x[2 * n]) if 2 * n < len(res.x) else 0.0
            if np.all(np.isfinite(d)) and float(np.max(np.abs(d))) > 1e-10 and delta > 1e-10:
                d[np.abs(d) < 1e-12] = 0.0
                return d
    except Exception:
        pass
    return None


def _smooth_fun(x):
    global _SMOOTH_LAST_X, _SMOOTH_LAST_OBJ, _SMOOTH_LAST_GRAD
    obj, grad = lse_obj_grad(x, _SMOOTH_TAU)
    _SMOOTH_LAST_X = np.asarray(x, dtype=float).copy()
    _SMOOTH_LAST_OBJ = float(obj)
    _SMOOTH_LAST_GRAD = grad
    return _SMOOTH_LAST_OBJ


def _smooth_jac(x):
    global _SMOOTH_LAST_X, _SMOOTH_LAST_GRAD
    if _SMOOTH_LAST_X is not None and _SMOOTH_LAST_X.size == x.size:
        try:
            if np.array_equal(_SMOOTH_LAST_X, x):
                return _SMOOTH_LAST_GRAD
        except Exception:
            pass
    obj, grad = lse_obj_grad(x, _SMOOTH_TAU)
    _SMOOTH_LAST_X = np.asarray(x, dtype=float).copy()
    _SMOOTH_LAST_GRAD = grad
    return grad


def _smooth_con_fun(x):
    return float(np.sum(x) - 0.5 * len(x))


def _smooth_con_jac(x):
    return np.ones_like(x, dtype=float)


def smooth_optimize(h0, tau=0.02, maxiter=40):
    global _SMOOTH_TAU, _SMOOTH_N
    h0 = np.asarray(h0, dtype=float)
    n = h0.size
    if n == 0:
        return h0
    h0 = project_to_box_sum(h0, 0.5 * n)
    _SMOOTH_TAU = float(tau)
    _SMOOTH_N = int(n)
    bounds = [(0.0, 1.0)] * n
    cons = [{"type": "eq", "fun": _smooth_con_fun, "jac": _smooth_con_jac}]
    try:
        res = minimize(
            _smooth_fun, h0, jac=_smooth_jac, method="SLSQP",
            bounds=bounds, constraints=cons,
            options={"maxiter": int(maxiter), "ftol": 1e-7}
        )
        h = np.asarray(res.x, dtype=float)
    except Exception:
        h = h0
    if h.size != n or not np.all(np.isfinite(h)):
        h = h0
    return project_to_box_sum(h, 0.5 * n)


def improve_candidate(h, time_limit, rng, use_lp=True, use_smooth=True):
    h = np.asarray(h, dtype=float)
    if h.ndim != 1:
        h = h.ravel()
    n = int(h.size)
    if n == 0:
        return h, 1e9
    target = 0.5 * n
    h = project_to_box_sum(h, target)

    cur_h = h.copy()
    cur_c = exact_c5(h)
    best_h = h.copy()
    best_c = cur_c

    if time_limit is None or float(time_limit) < 2.0:
        best_h = project_to_box_sum(best_h, target)
        best_c = exact_c5(best_h)
        return best_h, best_c

    end = time.monotonic() + max(1.0, float(time_limit))
    iter_count = 0
    stall = 0
    T = 0.004

    while time.monotonic() < end - 1.0:
        if cur_c > best_c + 0.05:
            cur_h = best_h.copy()
            cur_c = best_c

        F = score_vec(cur_h)
        cur_c = float(np.max(F))
        if cur_c < best_c - 1e-12:
            best_c = cur_c
            best_h = cur_h.copy()

        moved = False

        tau = max(1e-4, 0.025 * cur_c if (iter_count % 2 == 0) else 0.008 * cur_c)
        w_l = lse_weights(F, tau)
        w_a, _ = active_weights(F, 0.01, 4)
        w = 0.55 * w_l + 0.45 * w_a
        ws = float(w.sum())
        if ws > 1e-15:
            w = w / ws
            g = weighted_gradient(cur_h, w)
        else:
            g = np.zeros(n, dtype=float)

        gn = float(np.sum(np.abs(g)))
        if gn > 1e-13:
            cand, cc = direction_line_search(cur_h, -g, cur_c, target)
            if cand is not None and cc < cur_c - 1e-12:
                cur_h = cand
                cur_c = cc
                moved = True
                stall = 0
                if cc < best_c - 1e-12:
                    best_c = cc
                    best_h = cur_h.copy()

        if not moved and use_lp and n <= 220 and (iter_count % 6 == 0) and time.monotonic() < end - 15.0:
            M = 5 if n <= 120 else 3
            act_idx = np.argsort(F, kind="stable")[-M:]
            d = lp_direction(cur_h, F, act_idx, theta=max(0.004, 0.015 * cur_c), rho=1.5)
            if d is not None:
                cand, cc = direction_line_search(cur_h, d, cur_c, target)
                if cand is not None and cc < cur_c - 1e-12:
                    cur_h = cand
                    cur_c = cc
                    moved = True
                    stall = 0
                    if cc < best_c - 1e-12:
                        best_c = cc
                        best_h = cur_h.copy()

        if not moved:
            v = extreme_point(g, target)
            if float(np.sum(np.abs(v - cur_h))) > 1e-10:
                cand, cc = fw_line_search(cur_h, v, cur_c, target)
                if cand is not None and cc < cur_c - 1e-12:
                    cur_h = cand
                    cur_c = cc
                    moved = True
                    stall = 0
                    if cc < best_c - 1e-12:
                        best_c = cc
                        best_h = cur_h.copy()

        if not moved and time.monotonic() < end - 5.0:
            limit_pairs = 25 if n > 220 else (70 if n > 120 else 120)
            cand, cc = pair_moves(cur_h, cur_c, g, limit_pairs, end - 5.0)
            if cand is not None and cc < cur_c - 1e-12:
                cur_h = cand
                cur_c = cc
                moved = True
                stall = 0
                if cc < best_c - 1e-12:
                    best_c = cc
                    best_h = cur_h.copy()

        if moved:
            iter_count += 1
            T = max(1e-5, T * 0.985)
            continue

        iter_count += 1
        stall += 1

        if use_smooth and n <= 260 and time.monotonic() < end - 60.0 and (iter_count % 15 == 0):
            tau_sm = max(0.01, 0.03 * cur_c)
            if n <= 100:
                mi = 50
            elif n <= 180:
                mi = 35
            else:
                mi = 25
            cand = smooth_optimize(cur_h, tau_sm, mi)
            cc = exact_c5(cand)
            if cc < cur_c - 1e-12:
                cur_h = cand
                cur_c = cc
                moved = True
                stall = 0
                if cc < best_c - 1e-12:
                    best_c = cc
                    best_h = cur_h.copy()
                continue
            if cc < best_c - 1e-12:
                best_h = cand.copy()
                best_c = cc
                cur_h = cand
                cur_c = cc
                continue

        if stall >= 3 and time.monotonic() < end - 3.0:
            source = best_h if cur_c > best_c + 0.005 else cur_h
            accepted = False
            for _ in range(15):
                if time.monotonic() > end - 3.0:
                    break
                if rng.random() < 0.6:
                    i = int(rng.integers(n))
                    j = int(rng.integers(n))
                    if i == j:
                        continue
                    ai = float(source[i])
                    aj = float(source[j])
                    if ai < aj:
                        i, j = j, i
                        ai, aj = aj, ai
                    mt = min(ai, 1.0 - aj)
                    if mt < 1e-7:
                        continue
                    tmove = (0.3 + 0.7 * rng.random()) * mt
                    cand = source.copy()
                    cand[i] -= tmove
                    cand[j] += tmove
                    cand = np.clip(cand, 0.0, 1.0)
                    if abs(float(cand.sum()) - target) > 1e-8:
                        cand = project_to_box_sum(cand, target)
                    cc = exact_c5(cand)
                else:
                    scale = 0.03 if n <= 160 else 0.015
                    cand = source + scale * rng.standard_normal(n)
                    cand = project_to_box_sum(cand, target)
                    cc = exact_c5(cand)

                delta = cc - cur_c
                accept = False
                if delta <= 1e-12:
                    accept = True
                elif cur_c <= best_c + 0.02:
                    prob = math.exp(max(-60.0, -delta / T))
                    if rng.random() < prob:
                        accept = True

                if accept:
                    cur_h = cand
                    cur_c = cc
                    accepted = True
                    if cc < best_c - 1e-12:
                        best_c = cc
                        best_h = cur_h.copy()
                    break

            if not accepted and cur_c > best_c + 0.01:
                cur_h = best_h.copy()
                cur_c = best_c
                stall = 0

        if stall > 30 and time.monotonic() < end - 20.0:
            scale = 0.08 if n <= 160 else 0.04
            cand = best_h + scale * rng.standard_normal(n)
            cand = project_to_box_sum(cand, target)
            cc = exact_c5(cand)
            if cc < cur_c + 0.02:
                cur_h = cand
                cur_c = cc
                stall = 0
            else:
                cur_h = best_h.copy()
                cur_c = best_c
                stall = 0

    best_h = project_to_box_sum(best_h, 0.5 * len(best_h))
    best_c = exact_c5(best_h)
    return best_h, best_c


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.monotonic()
    try:
        budget = max(10.0, float(budget_s))
    except Exception:
        budget = 1100.0
    deadline = t0 + budget * 0.95
    rng = np.random.default_rng(seed)

    best_h = None
    best_c = 1e9
    best_n = 0
    queue = []

    parent_h = None
    parent_c = 1e9
    p = globals().get("initial_h_values", None)
    if p is not None:
        try:
            ph = np.asarray(p, dtype=float)
            if ph.ndim == 1 and ph.size > 0 and np.all(np.isfinite(ph)):
                ph = project_to_box_sum(ph, 0.5 * ph.size)
                pc = exact_c5(ph)
                if np.isfinite(pc):
                    parent_h = ph
                    parent_c = float(pc)
        except Exception:
            pass

    if parent_h is None:
        parent_h = random_h(rng, 128)
        parent_c = exact_c5(parent_h)

    add_candidate(queue, parent_h)
    best_h = parent_h.copy()
    best_c = float(parent_c)
    best_n = int(parent_h.size)

    if budget > 60.0 and time.monotonic() < deadline - 20.0:
        for sh in generate_shifts(parent_h, rng, 30):
            add_candidate(queue, sh[0])

    if budget > 120.0 and time.monotonic() < deadline - 30.0:
        transfer_ns = (80, 96, 112, 128, 160, 192, 224, 256)
        for N in transfer_ns:
            if time.monotonic() > deadline - 30.0:
                break
            if N == int(parent_h.size):
                continue
            add_candidate(queue, transfer_step(parent_h, N))
            if time.monotonic() > deadline - 15.0:
                break
            h_lin = transfer_linear(parent_h, N)
            add_candidate(queue, h_lin)
            if N % 2 == 0:
                add_candidate(queue, to_binary(h_lin))

    if (parent_c > 0.39) and time.monotonic() < deadline - 120.0:
        for N in (128, 192):
            if time.monotonic() > deadline - 60.0:
                break
            for h in generate_structured(N):
                if time.monotonic() > deadline - 30.0:
                    break
                try:
                    c = exact_c5(h)
                    if np.isfinite(c) and c < 0.45:
                        queue.append((h, float(c), int(h.size)))
                except Exception:
                    pass

    if time.monotonic() < deadline - 60.0 and (parent_c > 0.385 or best_n > 200):
        for N in (128, 160):
            for _ in range(2):
                if time.monotonic() > deadline - 30.0:
                    break
                add_candidate(queue, random_h(rng, N))

    for item in queue:
        ic = float(item[1])
        inum = int(item[2])
        if ic < best_c - 1e-12 or (abs(ic - best_c) <= 1e-10 and inum < best_n):
            best_c = ic
            best_h = item[0].copy()
            best_n = inum

    while time.monotonic() < deadline - 120.0:
        item = pop_best(queue)
        if item is None:
            break
        h, c, n = item
        if c > best_c + 0.10:
            continue

        remaining = deadline - time.monotonic() - 120.0
        if remaining < 20.0:
            break

        if n <= 120:
            limit = min(150.0, max(30.0, remaining * 0.25))
        elif n <= 200:
            limit = min(100.0, max(25.0, remaining * 0.20))
        elif n <= 300:
            limit = min(60.0, max(15.0, remaining * 0.15))
        else:
            limit = min(35.0, max(10.0, remaining * 0.10))

        if c > best_c + 0.04:
            limit *= 0.6
        if limit < 5.0:
            limit = 5.0

        h2, c2 = improve_candidate(h, limit, rng, use_lp=(n <= 220), use_smooth=(n <= 260))
        if h2 is not None:
            h2 = project_to_box_sum(h2, 0.5 * len(h2))
            c2 = exact_c5(h2)
            if np.isfinite(c2):
                if c2 < best_c - 1e-12 or (abs(c2 - best_c) <= 1e-10 and len(h2) < best_n):
                    best_c = float(c2)
                    best_h = h2.copy()
                    best_n = int(len(h2))

        if time.monotonic() < deadline - 180.0 and best_h is not None and best_n <= 320:
            src = best_h
            src_n = int(len(src))
            added = 0
            factors = (0.75, 0.9, 1.1, 1.3, 1.6)
            for fac in factors:
                if added >= 3:
                    break
                N2 = int(2.0 * round((src_n * fac) / 2.0))
                if N2 < 64 or N2 > 512 or N2 == src_n:
                    continue
                add_candidate(queue, transfer_step(src, N2))
                added += 1
                if added < 3 and time.monotonic() < deadline - 120.0:
                    add_candidate(queue, transfer_linear(src, N2))
                    added += 1

            for sh in generate_shifts(best_h, rng, 4):
                add_candidate(queue, sh[0])

            if time.monotonic() < deadline - 90.0 and best_n > 16:
                scale = 0.02 if best_n <= 160 else 0.01
                hp = best_h + scale * rng.standard_normal(best_n)
                add_candidate(queue, project_to_box_sum(hp, 0.5 * best_n))

    remaining = deadline - time.monotonic() - 5.0
    if best_h is not None and remaining > 20.0:
        final_limit = max(5.0, min(remaining, 120.0 if best_n <= 200 else 60.0))
        h2, c2 = improve_candidate(best_h, final_limit, rng, use_lp=(best_n <= 220), use_smooth=(best_n <= 260))
        if h2 is not None:
            h2 = project_to_box_sum(h2, 0.5 * len(h2))
            c2 = exact_c5(h2)
            if np.isfinite(c2) and (c2 < best_c - 1e-12 or (abs(c2 - best_c) <= 1e-10 and len(h2) < best_n)):
                best_c = float(c2)
                best_h = h2.copy()
                best_n = int(len(h2))

    if best_h is not None and time.monotonic() < deadline - 5.0:
        for N in (80, 96, 112, 128):
            if time.monotonic() > deadline - 2.0:
                break
            if N < best_n:
                h2 = transfer_step(best_h, N)
                c2 = exact_c5(h2)
                if np.isfinite(c2) and (c2 < best_c - 1e-12 or (abs(c2 - best_c) <= 1e-9 and N < best_n)):
                    best_c = float(c2)
                    best_h = h2.copy()
                    best_n = int(N)
                h2 = transfer_linear(best_h, N)
                c2 = exact_c5(h2)
                if np.isfinite(c2) and (c2 < best_c - 1e-12 or (abs(c2 - best_c) <= 1e-9 and N < best_n)):
                    best_c = float(c2)
                    best_h = h2.copy()
                    best_n = int(N)

    if best_h is None:
        best_h = random_h(rng, 128)
        best_c = exact_c5(best_h)
        best_n = int(len(best_h))

    best_h = project_to_box_sum(best_h, 0.5 * len(best_h))
    best_c = exact_c5(best_h)
    return (best_h, float(best_c), int(best_n))
