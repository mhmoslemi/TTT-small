```python
import numpy as np

def project_feasible(v, n):
    """Project v onto {h: 0<=h[i]<=1, sum(h)=n/2}"""
    target = n / 2.0
    
    def _sum_at_t(t):
        return np.sum(np.clip(v - t, 0.0, 1.0))
    
    lo, hi = -v.max() - 1, -v.min() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        if _sum_at_t(mid) > target:
            lo = mid
        else:
            hi = mid
    t = (lo + hi) / 2
    h = np.clip(v - t, 0.0, 1.0)
    # Fix any tiny numerical issues
    diff = target - h.sum()
    if abs(diff) > 1e-10:
        # Distribute the difference
        mask = (h > 0) & (h < 1)
        if mask.sum() > 0:
            h[mask] += diff / mask.sum()
            h = np.clip(h, 0.0, 1.0)
    return h
```

```python
def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return np.max(corr) * dx

def active_weights(corr_values, top_frac=0.15, temp=1e-4):
    """Get weights for top active correlation terms."""
    M = np.max(corr_values)
    # Select terms within some fraction of max
    threshold = M * (1.0 - top_frac)
    active = corr_values >= threshold
    n_active = np.sum(active)
    if n_active == 0:
        idx = np.argmax(corr_values)
        return np.array([idx]), np.array([1.0])
    
    active_vals = corr_values[active]
    # Exponential weighting based on distance from max
    z = (M - active_vals) / max(temp, 1e-10)
    w = np.exp(-z)
    w = w / w.sum()
    indices = np.where(active)[0]

            grad[:n-R] += w * dx * (-h[R:])

            grad[R:] += w * dx * (-h[:n-R])
```
