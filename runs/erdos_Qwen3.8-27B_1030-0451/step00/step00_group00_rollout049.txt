**Faster approach: Gradient-based with line search**

Use the subgradient at k* and do a backtracking line search:
1. Compute grad = ∇g_{k*}(h)
2. Try step = α * grad
3. If f(h - step * t) < f(h) for some t, accept; otherwise reduce t.

This is the "subgradient method with line search" which has convergence guarantees for convex problems (though ours is non-convex, it often works well in practice).

```python
def subgradient_linesearch(h, dx, target_sum, max_iters=100000):
    n = len(h)
    current_f = None
    h = h.copy()
    
    for it in range(max_iters):
        c = np.correlate(h, 1-h, mode="full") * dx
        k_star = np.argmax(c)
        current_f = c[k_star]
        
        # Compute subgradient
        lag = k_star - (n-1)
        d = abs(lag)
        grad = np.zeros(n)
        if d > 0 and d < n:
            grad[:n-d] += (1 - h[d:])
            grad[d:] -= h[:n-d]
        elif d == 0:
            grad[:] = 1 - 2*h  # ∂/∂h[j] of h[j]*(1-h[j]) = 1-2*h[j]
        
        # Line search: try decreasing step sizes
        for step in [0.5, 0.25, 0.1, 0.05, 0.01, 0.005]:
            h_new = h - step * grad * dx
            h_new = project_to_box_sum(h_new, target_sum)
            
            c_new = np.correlate(h_new, 1-h_new, mode="full") * dx
            new_f = c_new.max()
            
            if new_f < current_f:
                h = h_new
                break
    
    return h
```

```python
import numpy as np
import time

def project_to_box_sum(x, target_sum):
    lo = -1.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)

def get_subgradient(h, k_star, n):
    lag = k_star - (n - 1)
    d = abs(lag)
    grad = np.zeros(n)
    if d == 0:
        grad[:] = 1.0 - 2.0 * h
    elif d < n:
        grad[d:] += (1.0 - h[:n - d])
        grad[:n - d] -= h[d:]
    return grad

def get_smooth_gradient(h, c_vals, n, beta=30.0):
    c_max = c_vals.max()
    w = np.exp(beta * (c_vals - c_max))
    ws = w.sum()
    if ws < 1e-300:
        w = np.full(len(c_vals), 1.0 / len(c_vals))
    else:
        w /= ws
    grad = np.zeros(n)
    for k in range(len(c_vals)):
        wk = w[k]
        if wk < 1e-10:
            continue
        lag = k - (n - 1)
        d = abs(lag)
        if d == 0:
            grad += wk * (1.0 - 2.0 * h)
        elif d < n:
            grad[d:] += wk * (1.0 - h[:n - d])
            grad[:n - d] -= wk * h[d:]
    return grad

def optimize_subgradient(h, dx, target_sum, n, time_limit, rng, init_step=0.03, top_k=3, use_smooth=False):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    best_c5 = float(c.max())
    best_h = h.copy()
    step = init_step
    no_improve = 0
    start = time.time()
    vel = np.zeros(n)
    momentum = 0.6
    
    while time.time() - start < time_limit:
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = int(np.argmax(c))
        current_f = c[k_star]
        
        if current_f < best_c5 - 1e-13:
            best_c5 = current_f
            best_h = h.copy()
            no_improve = 0
        else:
            no_improve += 1
        
        if use_smooth:
            grad = get_smooth_gradient(h, c / dx, n, beta=30.0)
        else:
            ntop = min(top_k, len(c))
            idx = np.argpartition(c, -ntop)[-ntop:]
            c_top = c[idx]
            w = np.exp(40.0 * (c_top - c_top.max()))
            w /= w.sum()
            grad = np.zeros(n)
            for wk, k in zip(w, idx):
                lag = k - (n - 1)
                d = abs(lag)
                if d == 0:
                    grad += wk * (1.0 - 2.0 * h)
                elif d < n:
                    grad[d:] += wk * (1.0 - h[:n - d])
                    grad[:n - d] -= wk * h[d:]
        
        vel = momentum * vel - step * grad * dx
        h_new = h + vel
        h_new = project_to_box_sum(h_new, target_sum)

def coordinate_descent_refine(h, dx, target_sum, n, time_limit, rng):
    h = h.copy()
    c = np.correlate(h, 1.0 - h, mode="full") * dx
    best_c5 = float(c.max())
    best_h = h.copy()
    start = time.time()
    
    while time.time() - start < time_limit:
        c = np.correlate(h, 1.0 - h, mode="full") * dx
        k_star = int(np.argmax(c))
        current_f = c[k_star]
        
        if current_f < best_c5 - 1e-13:
            best_c5 = current_f
            best_h = h.copy()
        
        # Try moving mass from a high-h position to a low-h position
        # Focus on positions that affect the active correlation
        lag = k_star - (n - 1)
        d = abs(lag)
        
        # Candidate positions to modify
        if d == 0:
            cands = list(range(n))
        else:
            cands = list(range(d, n)) + list(range(n - d))
        
        rng.shuffle(cands)
        improved = False
        for j in cands[:20]:
            # Try decreasing h[j] slightly and increasing another position
            for j2 in cands[:8]:
                if j2 == j or j2 <= j:
                    continue
                # Move mass from j to j2: h[j]-eps, h[j2]+eps
                eps_max = min(h[j], 1.0 - h[j2])
                if eps_max < 1e-6:
                    continue
                best_eps = 0.0
                best_val = current_f
                for _ in range(20):
                    eps = rng.uniform(0, eps_max)
                    h[j] -= eps
                    h[j2] += eps
                    c2 = np.correlate(h, 1.0 - h, mode="full") * dx
                    v2 = c2.max()
                    if v2 < best_val:
                        best_val = v2
                        best_eps = eps
                    h[j] += eps
                    h[j2] -= eps
                if best_eps > 1e-8:
                    h[j] -= best_eps
                    h[j2] += best_eps
                    h = project_to_box_sum(h, target_sum)
                    improved = True
                    break
            if improved:
                break
    
    return best_h, best_c5
```
