        # where k ranges from -(n-1) to n-1, mapped to index k+(n-1)

        # h[m] appears as h[i+k] when i = m-k, contributing (1-h[m-k]) to ∂/∂h[m]
        # h[m] appears in (1-h[i]) when i = m, contributing -h[m+k] to ∂/∂h[m]
        
        # So ∂corr[k]/∂h[m] = (1-h[m-k]) * [0<=m-k<n] - h[m+k] * [0<=m+k<n]
        
        lag = k - (n - 1)
        valid_m = np.zeros(n, dtype=bool)
        # First term: m-k in [0, n-1] => m in [k, k+n-1] intersect [0, n-1]
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)  # m ranges from lo1 to hi1-1
        # Actually: m - lag >= 0 => m >= lag; m - lag < n => m < n + lag
        # Combined with 0 <= m < n: m in [max(0,lag), min(n, n+lag))
        
        # Second term: m+lag in [0, n-1] => m in [-lag, n-lag-1] intersect [0, n-1]
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)  # m in [lo2, hi2)

        if lo1 < hi1:
            m_indices = np.arange(lo1, hi1)
            sg[m_indices] += weights[k] * (1.0 - h[m_indices - lag])
        if lo2 < hi2:
            m_indices = np.arange(lo2, hi2)
            sg[m_indices] -= weights[k] * h[m_indices + lag]
    
    return sg * dx

    """Line search for h[p] -= alpha, h[q] += alpha."""
    n = len(h)
    alpha_hi = min(h[p], 1.0 - h[q])
    if alpha_hi < 1e-14:
        return None, 0.0

    a = np.correlate(h, b, mode='full')

    d[p] = -1.0
    d[q] = 1.0

    corr_hd = np.correlate(h, d, mode='full')
    corr_dd = np.correlate(d, d, mode='full')
    
    bc = corr_db - corr_hd
    c = -corr_dd
    
    # Minimize max_l (a_l + bc_l*alpha + c_l*alpha^2) over [0, alpha_hi]
    # Grid scan
    n_grid = 33
    alphas = np.linspace(0.0, alpha_hi, n_grid)
    V = a[:, None] + bc[:, None] * alphas[None, :] + c[:, None] * alphas[None, :]**2
    scores = V.max(axis=0) * dx
    
    best_i = np.argmin(scores)
    best_alpha = alphas[best_i]
    best_score = scores[best_i]
    
    # Refine with bounded scalar minimization
    if 0 < best_i < n_grid - 1:
        lo = alphas[best_i - 1]
        hi = alphas[best_i + 1]
    elif best_i == 0:
        lo, hi = 0.0, alphas[1]
    else:
        lo, hi = alphas[-2], alpha_hi
    
    def obj(alpha):
        return np.max(a + bc * alpha + c * alpha**2) * dx
    
    try:
        res = minimize_scalar(obj, bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
        if res.x > 0 and res.fun < best_score:
            best_alpha = res.x
            best_score = res.fun
    except:
        pass
    
    h_new = h.copy()
    h_new[p] -= best_alpha
    h_new[q] += best_alpha
    h_new = np.clip(h_new, 0.0, 1.0)

    if abs(h_new.sum() - n/2.0) > 1e-8:
        return None, best_score  # shouldn't happen but safety
    
    actual = eval_c5(h_new)
    return h_new, actual

import numpy as np
from scipy.optimize import minimize_scalar
import time

def _eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full'))) * dx

def _upsample(h, factor):
    return np.repeat(h, factor)

def _supergradient(h, dx, n_active=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    # Weight by proximity to max
    max_v = corr.max()
    if max_v <= 0:
        return np.zeros(n)
    
    # Use all lags with value > 50% of max as "active"
    active = corr > 0.5 * max_v
    
    sg = np.zeros(n)
    indices = np.where(active)[0]
    
    for idx in indices:
        w = corr[idx]
        lag = idx - (n - 1)
        
        # h[m] in h[i+lag] when i = m-lag: weight (1-h[m-lag])
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            m_idx = np.arange(lo1, hi1)
            sg[m_idx] += w * (1.0 - h[m_idx - lag])
        
        # h[m] in (1-h[i]) when i = m: weight -h[m+lag]
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            m_idx = np.arange(lo2, hi2)
            sg[m_idx] -= w * h[m_idx + lag]
    
    return sg * dx

def _line_search_pair(h, p, q, dx):
    n = len(h)
    alpha_max = min(h[p], 1.0 - h[q])
    if alpha_max < 1e-13:
        return None
    
    b = 1.0 - h
    a = np.correlate(h, b, mode='full')
    
    d = np.zeros(n)
    d[p] = -1.0
    d[q] = 1.0
    
    corr_db = np.correlate(d, b, mode='full')
    corr_hd = np.correlate(h, d, mode='full')
    corr_dd = np.correlate(d, d, mode='full')
    
    bc = corr_db - corr_hd
    c = -corr_dd
    
    # Grid scan
    n_grid = 25
    alphas = np.linspace(0.0, alpha_max, n_grid)
    V = a[:, None] + bc[:, None] * alphas + c[:, None] * alphas**2
    scores = V.max(axis=0) * dx
    
    best_i = np.argmin(scores)
    current_score = a.max() * dx
    if scores[best_i] >= current_score - 1e-14:
        return None
    
    best_alpha = alphas[best_i]
    
    # Refine
    if 0 < best_i < n_grid - 1:
        lo, hi = alphas[best_i - 1], alphas[best_i + 1]
    elif best_i == 0:
        lo, hi = 0.0, alphas[1]
    else:
        lo, hi = alphas[-2], alpha_max
    
    def _obj(alpha):
        return np.max(a + bc * alpha + c * alpha**2) * dx
    
    try:
        res = minimize_scalar(_obj, bounds=(lo, hi), method='bounded', options={'xatol': 1e-10})
        if res.fun < scores[best_i] * dx:  # wait, scores already has dx
            pass
        if res.fun < scores[best_i]:
            best_alpha = res.x
    except:
        pass
    
    h_new = h.copy()
    h_new[p] -= best_alpha
    h_new[q] += best_alpha
    h_new = np.clip(h_new, 0.0, 1.0)
    return h_new

def _quad_obj(a, bc, c, alpha, dx):
    """Objective: max of (a + bc*alpha + c*alpha^2) * dx"""
    return float(np.max(a + bc * alpha + c * alpha**2)) * dx

def _minimize_max_quad(a, bc, c, dx, lo, hi):
    """Minimize max_l (a_l + bc_l*alpha + c_l*alpha^2) * dx over [lo, hi] using golden section."""
    invgold = (np.sqrt(5.0) - 1.0) / 2.0
    a_gs = lo
    b_gs = hi
    
    c1 = b_gs - invgold * (b_gs - a_gs)
    c2 = a_gs + invgold * (b_gs - a_gs)
    f1 = _quad_obj(a, bc, c, c1, dx)
    f2 = _quad_obj(a, bc, c, c2, dx)
    
    tol = 1e-10
    while (b_gs - a_gs) > tol:
        if f1 < f2:
            b_gs = c2
            c2 = c1
            f2 = f1
            c1 = b_gs - invgold * (b_gs - a_gs)
            f1 = _quad_obj(a, bc, c, c1, dx)
        else:
            a_gs = c1
            c1 = c2
            f1 = f2
            c2 = a_gs + invgold * (b_gs - a_gs)
            f2 = _quad_obj(a, bc, c, c2, dx)
    
    return (a_gs + b_gs) / 2.0

This is a standard golden-section search. It assumes the function is unimodal, which the max of quadratics may not be, but in practice it works well for the local minimum we're looking for.

Actually, for the max of (mostly) linear functions with a few quadratic terms, the function is piecewise linear (or piecewise quadratic) and convex (max of convex is convex). Wait, is the max of quadratics convex? Only if each quadratic is convex (c >= 0). In our case, c = -corr(d,d), and for pair moves, corr(d,d) has at most 3 non-zero entries, so c_l = 0 for most l, and c_l = -2 at lag 0, c_l = +1 at the two other lags. So some c_l > 0 (convex) and some c_l < 0 (concave).

The max of a mix of convex and concave quadratics is not necessarily convex. So golden section might not find the global minimum. But for a local search (finding the best α in a small neighborhood), it should work well enough.

def _supergradient(h, dx, n_active=50):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    max_v = corr.max()
    if max_v <= 0:
        return np.zeros(n)
    
    sg = np.zeros(n)
    # Use top n_active lags
    top = np.argsort(corr)[-n_active:]
    
    for idx in top:
        w = corr[idx]
        lag = idx - (n - 1)
        
        lo1 = max(0, lag)
        hi1 = min(n, n + lag)
        if lo1 < hi1:
            m_idx = np.arange(lo1, hi1)
            sg[m_idx] += w * (1.0 - h[m_idx - lag])
        
        lo2 = max(0, -lag)
        hi2 = min(n, n - lag)
        if lo2 < hi2:
            m_idx = np.arange(lo2, hi2)
            sg[m_idx] -= w * h[m_idx + lag]
    
    return sg * dx

def _quad_eval(a, bc, c, alpha, dx):
    return float(np.max(a + bc * alpha + c * alpha**2)) * dx

    """Golden section minimization of _quad_eval over [lo, hi]."""
    if hi - lo < 1e-12:
        return (lo + hi) / 2.0
    
    invgold = (np.sqrt(5.0) - 1.0) / 2.0
    c1 = hi - invgold * (hi - lo)
    c2 = lo + invgold * (hi - lo)
    f1 = _quad_eval(a, bc, c, c1, dx)
    f2 = _quad_eval(a, bc, c, c2, dx)

            f1 = f2
