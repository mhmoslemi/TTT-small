import numpy as np
import time

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_01(h, n):
    """Round h to 0-1 valued with exactly n//2 ones."""
    target = n // 2
    # Sort indices by h value descending
    idx = np.argsort(h)[::-1]
    result = np.zeros(n)
    result[idx[:target]] = 1.0
    return result

def random_01(n, seed=0):
    """Generate a random 0-1 vector with exactly n//2 ones."""
    rng = np.random.RandomState(seed)
    idx = rng.permutation(n)
    h = np.zeros(n)
    h[idx[:n//2]] = 1.0
    return h

def local_search_01(h, n, dx, max_sweeps=50, time_limit=300):
    """Perform local search on 0-1 valued h using swaps."""
    start = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    ones = np.where(h == 1)[0]
    zeros = np.where(h == 0)[0]
    
    for sweep in range(max_sweeps):
        if time.time() - start > time_limit:
            break
        improved = False
        # Try all pairs (optional: random subset)
        best_delta = 0
        best_swap = None
        
        # For efficiency, check a random subset of swaps
        n_check = min(len(ones) * len(zeros), 5000)
        for _ in range(n_check):
            i = ones[np.random.randint(len(ones))]
            j = zeros[np.random.randint(len(zeros))]
            if i == j:
                continue
            # Perform swap
            h[i] = 0
            h[j] = 1
            c5_new = compute_c5(h, dx)
            # Undo
            h[i] = 1
            h[j] = 0
            if c5_new < best_c5:
                best_c5 = c5_new
                best_swap = (i, j)
                improved = True
                # Accept immediately and update
                h[i] = 0
                h[j] = 1
                ones = np.where(h == 1)[0]
                zeros = np.where(h == 0)[0]
                break
        
        if not improved:
            break
    
    return h, best_c5
