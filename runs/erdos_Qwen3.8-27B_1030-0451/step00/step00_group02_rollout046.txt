    corrs = compute_corrs(h)
    mc = corrs.max()

    t_start = time.time()
    
    for it in range(n_iters):
        if time.time() - t_start > time_limit:
            break
        t_frac = it / max(1, n_iters - 1)
        beta = beta_start + (beta_end - beta_start) * t_frac
        grad = grad_smoothed(h, beta)
        obj_now = smoothed_obj(h, beta)
        
        accepted = False
        lr_ls = lr
        for _ in range(20):
            h_trial = project_to_constraint(h - lr_ls * grad, n)
            obj_trial = smoothed_obj(h_trial, beta)
            if obj_trial < obj_now - 1e-13:
                h = h_trial
                accepted = True
                break
            lr_ls *= 0.5
        if not accepted:
            lr = lr_ls * 0.5
            if lr < 1e-9:
                lr = lr_init * 0.001
            continue
        lr = min(lr * 1.03, lr_init * 20)
        
        c5_now = c5_value(h)
        if c5_now < best_c5:
            best_c5 = c5_now
            best_h = h.copy()
    
    return best_h, best_c5

def try_snap(h, n):
    """Try to snap near-0 or near-1 values, check if it improves C5."""
    h2 = h.copy()
    changed = False
    for tol in [0.01, 0.05, 0.1]:
        h2 = h.copy()
        h2[h2 < tol] = 0.0
        h2[h2 > 1.0 - tol] = 1.0
        # Re-project to fix sum
        h2 = project_to_constraint(h2, n)
        c5_test = c5_value(h2)
        if c5_test < c5_value(h):
            return h2, c5_test
    return h, c5_value(h)

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values, dtype=float)
    else:
        n0 = 81
        x0 = np.linspace(0, 2, n0, endpoint=False)
        h0 = 0.5 + 0.4 * np.sin(2 * np.pi * x0)
        h0 = np.clip(h0, 0, 1)
        h0 = project_to_constraint(h0, n0)
    
    best_h = h0.copy()
    best_c5 = c5_value(h0)
    best_n = len(h0)
    
    # Phase 1: Optimize at original and upsized n
    configs = [
        (len(h0), 50.0, 200.0, 3000, 0.05, 40.0),
        (128, 40.0, 150.0, 2000, 0.03, 35.0),
        (256, 30.0, 120.0, 1500, 0.02, 40.0),
        (512, 25.0, 100.0, 1000, 0.01, 45.0),
        (1024, 20.0, 80.0, 600, 0.005, 45.0),
        (2048, 15.0, 60.0, 400, 0.003, 40.0),
    ]
    
    for n_t, bs, be, n_it, lr_i, tlim in configs:
        if time.time() - start_time > budget_s * 0.75:
            break
        h_c, c5_c = optimize_pg(h0, n_t, bs, be, n_it, lr_i, seed, tlim)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
            best_n = n_t
    
    # Phase 2: Polish with high beta
    if time.time() - start_time < budget_s * 0.7:
        h_c, c5_c = optimize_pg(best_h, best_n, 100.0, 500.0, 3000, 0.005, seed + 1, 50.0)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
        # Try snap
        h_s, c5_s = try_snap(best_h, best_n)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    
    # Phase 3: Alternative starting points
    if time.time() - start_time < budget_s * 0.5:
        rng = np.random.RandomState(seed)
        for trial in range(4):
            if time.time() - start_time > budget_s * 0.78:
                break
            n_t = 256 + trial * 192
            x = np.linspace(0, 2, n_t, endpoint=False)
            f1 = rng.randint(1, 6)
            f2 = rng.randint(2, 10)
            a1 = rng.uniform(0.1, 0.4)
            a2 = rng.uniform(0.05, 0.25)
            p1 = rng.uniform(0, 2 * np.pi)
            p2 = rng.uniform(0, 2 * np.pi)
            h_rand = 0.5 + a1 * np.sin(2 * np.pi * f1 * x + p1) + a2 * np.sin(2 * np.pi * f2 * x + p2)
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = project_to_constraint(h_rand, n_t)
            h_c, c5_c = optimize_pg(h_rand, n_t, 20.0, 80.0, 800, 0.01, seed + 50 + trial, 30.0)
            if c5_c < best_c5:
                best_c5 = c5_c
                best_h = h_c
                best_n = n_t
    
    # Phase 4: Final polish on best
    if time.time() - start_time < budget_s * 0.85:
        h_c, c5_c = optimize_pg(best_h, best_n, 200.0, 1000.0, 5000, 0.002, seed + 200, 30.0)
        if c5_c < best_c5:
            best_c5 = c5_c
            best_h = h_c
        h_s, c5_s = try_snap(best_h, best_n)
        if c5_s < best_c5:
            best_c5 = c5_s
            best_h = h_s
    
    return best_h, best_c5, best_n
