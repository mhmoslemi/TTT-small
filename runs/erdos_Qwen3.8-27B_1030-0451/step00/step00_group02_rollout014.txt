import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr_vec(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_of_corr_idx(h, idx, n, dx):
    """Gradient of corr[idx] with respect to h."""
    lag = idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        k = lag
        m = n - k
        if m > 0:
            grad[:m] += dx * (1.0 - h[k:k+m])
            grad[k:k+m] -= dx * h[:m]
    else:
        k = -lag
        m = n - k
        if m > 0:
            grad[k:] += dx * (1.0 - h[:m])
            grad[:m] -= dx * h[k:]
    return grad

def active_subgradient(h, corr, n, dx, tol_frac=0.01):
    """Compute weighted average gradient of active (near-max) constraints."""
    cmax = np.max(corr)
    threshold = cmax - tol_frac * cmax
    active = np.where(corr >= threshold)[0]
    
    # Weight by how close to max
    weights = corr[active] - (cmax - tol_frac * cmax) + 1e-10
    weights = weights / np.sum(weights)
    
    sg = np.zeros(n)
    for idx in active:
        sg += weights[list(active).index(idx)] * grad_of_corr_idx(h, idx, n, dx)
    
    return sg

def line_search_step(h, direction, n, max_alpha=None):
    """Find max alpha such that h + alpha*direction stays in [0,1]."""
    # sum(direction) should be ~0
    alpha = 1.0
    for i in range(n):
        if direction[i] > 1e-15:
            alpha = min(alpha, (1.0 - h[i]) / direction[i])
        elif direction[i] < -1e-15:
            alpha = min(alpha, h[i] / (-direction[i]))
    
    if max_alpha is not None:
        alpha = min(alpha, max_alpha)
    
    return max(alpha, 0.0)

    dx = 2.0 / n
    curr = h.copy()

        curr_c5 = cmax * dx

        sg = active_subgradient(curr, corr, n, dx, tol_frac=0.02)

        sg_mean = np.mean(sg)
        sg = sg - sg_mean  # zero-sum direction

        if norm_sg < 1e-12:
            break
        
        direction = -sg / norm_sg  # unit direction
        
        # Line search
        max_alpha = line_search_step(curr, direction, n)
        if max_alpha < 1e-10:
            step_mult *= 0.5
            continue
        
        # Try a few step sizes
        best_alpha = 0
        best_c5 = curr_c5
        for frac in [1.0, 0.8, 0.6, 0.4, 0.2, 0.1, 0.05, 0.02]:

            if np.min(h_new) < -1e-10 or np.max(h_new) > 1.0 + 1e-10:

            curr_c5 = best_c5

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def grad_corr_idx(h, idx, n, dx):
    """Gradient of corr[idx] with respect to h."""
    lag = idx - (n - 1)
    grad = np.zeros(n)
    if lag >= 0:
        k = lag
        m = n - k
        if m > 0:
            grad[:m] += dx * (1.0 - h[k:k+m])
            grad[k:k+m] -= dx * h[:m]
    else:
        k = -lag
        m = n - k
        if m > 0:
            grad[k:] += dx * (1.0 - h[:m])
            grad[:m] -= dx * h[k:]
    return grad
