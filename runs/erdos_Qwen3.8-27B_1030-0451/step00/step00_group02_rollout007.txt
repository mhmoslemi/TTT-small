import numpy as np
import time

def exact_score(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def ensure_feasible(h, n=None):
    """Ensure h is feasible: 0 <= h <= 1, sum(h) = n/2"""
    if n is None:
        n = len(h)
    S = n / 2.0
    h = np.clip(h.astype(float), 0, 1)
    # Project onto sum constraint
    diff = S - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference over non-saturated entries
        mask = (h > 1e-10) & (h < 1 - 1e-10)
        if np.sum(mask) > 0:
            h[mask] += diff / np.sum(mask)
            h = np.clip(h, 0, 1)
            # Final correction
            diff2 = S - np.sum(h)
            if abs(diff2) > 1e-12:
                mask2 = (h > 0) & (h < 1)
                if np.sum(mask2) > 0:
                    h[mask2] += diff2 / np.sum(mask2)
                    h = np.clip(h, 0, 1)
    # Final exact fix
    diff3 = S - np.sum(h)
    if abs(diff3) > 1e-14 and np.sum(h) > 0:
        # Add to interior points
        mask3 = (h > 1e-12) & (h < 1 - 1e-12)
        if np.sum(mask3) > 0:
            h[mask3] += diff3 / np.sum(mask3)
    return h

def project_to_box_sum(z, n, S):
    """Project z onto {0 <= x <= 1, sum(x) = S}"""
    # Bisection on the dual variable
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        x = np.clip(z - mid, 0, 1)
        s = np.sum(x)
        if s > S:
            lo = mid
        else:
            hi = mid
    x = np.clip(z - (lo + hi) / 2.0, 0, 1)
    # Fix small residual
    diff = S - np.sum(x)
    if abs(diff) > 1e-12:
        mask = (x > 1e-12) & (x < 1 - 1e-12)
        if np.sum(mask) > 0:
            x[mask] += diff / np.sum(mask)
            x = np.clip(x, 0, 1)
    return x

def smooth_obj_grad(h, tau):
    """Compute smooth objective and gradient for log-sum-exp of correlations"""
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    c = np.correlate(h, q, mode='full') * dx  # shape (2n-1,)
    
    # Log-sum-exp for numerical stability
    c_max = np.max(c)
    log_sum = c_max + tau * np.log(np.sum(np.exp((c - c_max) / tau)))
    
    # Weights (softmax)
    w = np.exp((c - c_max) / tau)  # shape (2n-1,)
    w /= np.sum(w)
    
    # Gradient: for each correlation index m (corresponding to k = m - (n-1))
    # c_k = sum_{n=n_start}^{n_end-1} h_{n+k} * (1 - h_n) * dx
    # grad_c_k[h_j] = dx * [(1-h_{j-k}) * I[j-k valid as n] - h_{j+k} * I[j valid as n]]
    
    grad = np.zeros(n)
    for m in range(2 * n - 1):
        k = m - (n - 1)
        w_m = w[m] * dx  # weight including dx
        
        n_start = max(0, -k)
        n_end = min(n, n - k)
        
        if n_start >= n_end:
            continue
        
        # For each valid n in [n_start, n_end):
        # h_{n+k} * (1-h_n)
        # d/dh_j: (1-h_{n+k-k})= (1-h_n) when j = n+k, and -h_{n+k} when j = n
        
        # j = n + k for n in [n_start, n_end):
        #   j ranges from n_start + k to n_end - 1 + k
        #   contribution: w_m * (1 - h_n) = w_m * q_n
        j_indices = np.arange(n_start + k, n_end + k)
        n_indices = np.arange(n_start, n_end)
        grad[j_indices] += w_m * q[n_indices]
        
        # j = n for n in [n_start, n_end):
        #   contribution: w_m * (-h_{n+k}) = -w_m * h_{n+k}
        grad[n_indices] -= w_m * h[j_indices]
    
    return log_sum, grad

def pair_move_optimize(h, n_iter=500, seed=42, time_limit=None, t_start=None):
    """Local optimization using pair moves"""
    n = len(h)
    rng = np.random.RandomState(seed)
    h = h.copy()
    S = n / 2.0
    best_score = exact_score(h)
    patience = 0
    max_patience = 100
    
    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        # Compute gradient (use smooth with small tau)
        tau = max(0.001, 0.02 * best_score)
        _, grad = smooth_obj_grad(h, tau)
        
        # Pick a pair: one to increase (lowest gradient) and one to decrease (highest gradient)
        # We want to decrease the objective, so move mass from high-gradient to low-gradient
        # Actually: if we increase h_j, the change in smooth obj is ~ grad_j
        # To decrease, we want to increase h at low-grad positions and decrease at high-grad positions
        
        i_dec = int(np.argmin(grad))  # increase here
        i_inc = int(np.argmax(grad))  # decrease here
        
        if i_dec == i_inc:
            # Use random pair
            i_dec = rng.randint(n)
            i_inc = rng.randint(n)
            if i_dec == i_inc:
                continue
        
        # Direction: h[i_dec] increases, h[i_inc] decreases
        # Feasible range:
        d_max_inc = 1.0 - h[i_dec]
        d_max_dec = h[i_inc]
        d_max = min(d_max_inc, d_max_dec)
        
        if d_max < 1e-10:
            # Try random pair
            for _ in range(5):
                i_dec = rng.randint(n)
                i_inc = rng.randint(n)
                if i_dec == i_inc:
                    continue
                d_max = min(1.0 - h[i_dec], h[i_inc])
                if d_max > 1e-10:
                    break
            else:
                continue
        
        # Line search over several step sizes
        best_d = 0
        best_new_score = best_score
        for frac in [0.5, 0.3, 0.1, 0.05, 0.01]:
            d = frac * d_max
            new_h = h.copy()
            new_h[i_dec] += d
            new_h[i_inc] -= d
            new_score = exact_score(new_h)
            if new_score < best_new_score:
                best_new_score = new_score
                best_d = d
        
        if best_d > 0 and best_new_score < best_score - 1e-14:
            h[i_dec] += best_d
            h[i_inc] -= best_d
            best_score = best_new_score
            patience = 0
        else:
            patience += 1
            # Try random pairs sometimes
            if patience % 10 == 0:
                for _ in range(10):
                    i_dec = rng.randint(n)
                    i_inc = rng.randint(n)
                    if i_dec == i_inc:
                        continue
                    d_max = min(1.0 - h[i_dec], h[i_inc])
                    if d_max < 1e-10:
                        continue
                    d = rng.random() * 0.5 * d_max
                    new_h = h.copy()
                    new_h[i_dec] += d
                    new_h[i_inc] -= d
                    new_score = exact_score(new_h)
                    if new_score < best_score - 1e-14:
                        h[i_dec] += d
                        h[i_inc] -= d
                        best_score = new_score
                        patience = 0
                        break
        
        if patience >= max_patience:
            break
    
    return h, best_score

        for step in range(n_steps):
            _, grad = compute_smooth_grad(h, tau)
            gnorm = np.max(np.abs(grad))
            if gnorm < 1e-12:
                break
            lr = 1.0 / (gnorm + 1e-10)
            for _ in range(5):
                new_h = h - lr * grad
                new_h = project_box_sum(new_h, S)
                s = exact_score(new_h)
                if s < best - 1e-15:
                    h = new_h
                    best = s
                    break
                lr *= 0.5
            else:
                # Try random perturbation
                ip = rng.randint(n)
                im = rng.randint(n)
                if ip != im:
                    d = 0.1 * min(1 - h[ip], h[im])
                    h[ip] += d
                    h[im] -= d
                    s = exact_score(h)
                    if s < best:
                        best = s
                    else:
                        h[ip] -= d
                        h[im] += d
    
    return h, best

def slsqp_optimize(h0, maxiter=200, time_budget=30):
    n = len(h0)
    S = n / 2.0
    t0 = time.time()
    h = h0.copy()
    best = exact_score(h)
    
    for tau in [0.05, 0.02, 0.01, 0.005, 0.002]:
        if time.time() - t0 > time_budget - 3:
            break
        
        obj = SmoothObj()
        obj.tau = tau
        obj.n = n
        con = SumCon(S)
        bounds = [(0.0, 1.0)] * n
        cons = [{'type': 'eq', 'fun': con, 'jac': con.grad}]
        
        try:
            res = scipy.optimize.minimize(
                obj, h, method='SLSQP', jac=obj.grad,
                bounds=bounds, constraints=cons,
                maxiter=maxiter, ftol=1e-12
            )
            if res.x is not None and np.all(np.isfinite(res.x)):
                new_h = project_box_sum(res.x, S)
                s = exact_score(new_h)
                if s < best - 1e-15:
                    h = new_h
                    best = s
        except Exception:
            pass
    
    return h, best

def make_random_balanced(n, rng, method='uniform'):
    S = n / 2.0
    if method == 'uniform':
        z = rng.random(n)
    elif method == 'bimodal':
        z = np.where(rng.random(n) < 0.5, 0.3, 0.7)
        z += 0.2 * rng.randn(n)
    else:  # 'spiky'
        z = np.zeros(n)
        n_spikes = int(n * 0.3)
        spike_idx = rng.choice(n, n_spikes, replace=False)
        z[spike_idx] = 1.0
    return project_box_sum(z, S)

def interpolate_to_dim(h_src, n_dst):
    """Interpolate h_src to n_dst points and project."""
    n_src = len(h_src)
    S_dst = n_dst / 2.0
    x_src = (np.arange(n_src) + 0.5) / n_src
    x_dst = (np.arange(n_dst) + 0.5) / n_dst
    z = np.interp(x_dst, x_src, h_src)
    return project_box_sum(z, S_dst)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    t_deadline = t_start + budget_s * 0.95  # Leave 5% safety margin
    
    def time_left():
        return t_deadline - time.time()
    
    def deadline_hit():
        return time.time() >= t_deadline
    
    # Get parent
    best_h = None
    best_score = 1.0
    best_n = 0
    
    try:
        parent = np.asarray(initial_h_values, dtype=float)
        if parent.ndim == 1 and parent.size > 1 and np.all(np.isfinite(parent)):
            n_p = parent.size
            S_p = n_p / 2.0
            # Check feasibility
            if np.all(parent >= -1e-6) and np.all(parent <= 1 + 1e-6) and abs(np.sum(parent) - S_p) < 0.01:
                parent = project_box_sum(parent, S_p)
                s = exact_score(parent)
                best_h = parent.copy()
                best_score = s
                best_n = n_p
    except (NameError, AttributeError, Exception):
        parent = None
        n_p = 81
    
    if best_h is None:
        # Fallback: uniform
        n_p = 81
        best_h = np.full(n_p, 0.5)
        best_score = exact_score(best_h)
        best_n = n_p
    
    # Candidate pool
    candidates = [(best_h.copy(), best_score, best_n)]
    
    if not deadline_hit():
        # Generate variants
        rng = np.random.RandomState(seed)
        n_p_local = best_n
        
        # Complement (same score by symmetry, but include for completeness)
        comp = project_box_sum(1.0 - best_h, n_p_local / 2.0)
        s = exact_score(comp)
        candidates.append((comp, s, n_p_local))
        
        # Thresholded versions
        for q in [0.3, 0.4, 0.5, 0.6, 0.7]:
            if deadline_hit():
                break
            thr = np.quantile(best_h, 1 - q)
            z = np.where(best_h > thr, 1.0, 0.0).astype(float)
            tv = project_box_sum(z, n_p_local / 2.0)
            s = exact_score(tv)
            candidates.append((tv, s, n_p_local))
        
        # Random balanced
        for i in range(10):
            if deadline_hit():
                break
            method = ['uniform', 'bimodal', 'spiky'][i % 3]
            rv = make_random_balanced(n_p_local, np.random.RandomState(seed + i + 100), method)
            s = exact_score(rv)
            candidates.append((rv, s, n_p_local))
        
        # Noisy variants of parent
        for i in range(5):
            if deadline_hit():
                break
