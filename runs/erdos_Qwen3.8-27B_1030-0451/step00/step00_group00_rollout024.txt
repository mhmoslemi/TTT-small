import numpy as np
from scipy.optimize import minimize
import time

def compute_overlaps(h, dx):
    return np.correlate(h, 1.0 - h, mode="full") * dx

def smooth_gradient(h, dx, T=50.0):
    """Compute smooth gradient of logsumexp approx to max overlap."""
    n = len(h)
    overlaps = np.correlate(h, 1.0 - h, mode="full") * dx
    # Softmax weights
    max_ov = np.max(overlaps)
    exp_vals = np.exp(T * (overlaps - max_ov))
    weights = exp_vals / np.sum(exp_vals)
    
    # Compute weighted gradient
    grad = np.zeros(n)
    for k in range(len(overlaps)):
        if weights[k] < 1e-10:
            continue
        lag = k - (n - 1)
        if lag >= 0:
            if n - lag > 0:
                grad[0:n-lag] += weights[k] * (1.0 - h[lag:n]) * dx
                grad[lag:n] += weights[k] * (-h[0:n-lag]) * dx
        else:
            m = -lag
            if n - m > 0:
                grad[m:n] += weights[k] * (1.0 - h[0:n-m]) * dx
                grad[0:n-m] += weights[k] * (-h[m:n]) * dx
    return grad

def project_feasible(v, target_sum):
    """Project v onto {0 ≤ x ≤ 1, sum(x) = target_sum}."""
    n = len(v)
    # Find lambda such that sum(clip(v - lambda, 0, 1)) = target_sum
    lo, hi = v.min() - 1.0, v.max() + 1.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(v - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2
    return np.clip(v - lam, 0.0, 1.0)

    best_h = h.copy()

    lr = 0.001  # learning rate

        for sweep in range(200):
            if elapsed() > budget_s - 15:
                break
            any_improve = False
            for i in range(0, n, 3):
                if elapsed() > budget_s - 15:
                    break
                j = (i + 17) % n
                for delta in [step, -step]:
                    if h[i] + delta < 0 or h[i] + delta > 1:
                        continue
                    if h[j] - delta < 0 or h[j] - delta > 1:
                        continue
                    h[i] += delta
                    h[j] -= delta
                    c5_try = get_max_c5(h, dx)
                    h[i] -= delta
                    h[j] += delta
                    if c5_try < c5_cur:
                        h[i] += delta
                        h[j] -= delta
                        c5_cur = c5_try
                        any_improve = True
            if c5_cur < best_c5:
                best_c5 = c5_cur
                best_h = h.copy()
            if not any_improve:
                break

    return best_h, best_c5, n
