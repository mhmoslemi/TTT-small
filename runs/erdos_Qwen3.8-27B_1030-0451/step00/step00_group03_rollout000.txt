import numpy as np
import time
from scipy.optimize import minimize

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def project_feasible(y, target_sum):
    """Project y onto {0 <= x <= 1, sum(x) = target_sum}"""
    n = len(y)
    y = np.clip(np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)
    # Use bisection to find alpha such that sum(clip(y + alpha, 0, 1)) = target_sum
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    h = np.clip(y + alpha, 0.0, 1.0)
    # Fix residual
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-14:
        # Distribute residual
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                step = min(room[idx], residual)
                h[idx] += step
                residual -= step
                if abs(residual) < 1e-15:
                    break
        else:
            room = h
            order = np.argsort(room)
            for idx in order:
                step = min(room[idx], -residual)
                h[idx] -= step
                residual += step
                if abs(residual) < 1e-15:
                    break
    return h

def compute_gradient(h, n_active, target_sum):
    """Compute gradient of smoothed max objective using top n_active lags"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    fmax = np.max(f)
    
    # Select top lags
    n_sel = min(n_active, len(f))
    top_idx = np.argsort(f)[-n_sel:]
    
    # Compute smooth weights
    tau = 0.005
    selected_f = f[top_idx]
    z = (selected_f - fmax) / tau
    w = np.exp(z - np.max(z))
    w /= np.sum(w)
    
    grad = np.zeros(n)
    for i, m in enumerate(top_idx):
        k = m - (n - 1)  # lag
        if k >= 0:
            L = n - k
            # f_m = dx * sum_{j=0}^{L-1} h[j] * (1-h[j+k])
            # d/dh[j] for j in [0, L-1]: dx * (1 - h[j+k])
            # d/dh[j] for j in [k, k+L-1]: dx * (-h[j-k])
            grad[:L] += w[i] * dx * (1.0 - h[k:k+L])
            grad[k:k+L] += w[i] * dx * (-h[:L])
        else:
            r = -k
            L = n - r
            # f_m = dx * sum_{j=r}^{n-1} h[j] * (1-h[j-r])
            # Let's reindex: sum_{j=0}^{L-1} h[j+r]*(1-h[j])
            # d/dh[j] for j in [0, L-1]: dx * (-h[j+r])
            # d/dh[j] for j in [r, n-1]: dx * (1-h[j-r])
            grad[:L] += w[i] * dx * (-h[r:n])
            grad[r:n] += w[i] * dx * (1.0 - h[:L])
    
    return grad

def smooth_descent(h0, max_iter, target_sum, time_limit, start_time):
    """Projected gradient descent with log-sum-exp smoothing"""
    n = len(h0)
    h = h0.copy()
    best_c = evaluate_c5(h)
    best_h = h.copy()
    
    tau = 0.005
    eta = 0.01
    n_active = min(120, 2 * n - 1)
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        grad = compute_gradient(h, n_active, target_sum)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-10:
            break
        
        h_new = h - eta / gnorm * grad
        h_new = project_feasible(h_new, target_sum)
        c_new = evaluate_c5(h_new)
        
        if c_new < best_c:
            best_c = c_new
            best_h = h_new.copy()
            h = h_new
            eta = min(eta * 1.05, 0.1)
        else:
            h = h_new  # still move, but don't update best
            if c_new > best_c + 0.001:
                eta *= 0.5
            else:
                eta = max(eta * 0.7, 0.001)
        
        # Adapt tau
        if it % 100 == 0 and it > 0:
            tau *= 0.8
    
    return best_h, best_c

def try_slsqp(h0, target_sum, n_active, max_iter_slsqp, time_limit, start_time):
    """Use SLSQP to polish a candidate"""
    n = len(h0)
    
    # Get active lags
    corr = np.correlate(h0, 1.0 - h0, mode="full")
    dx = 2.0 / n
    f = corr * dx
    fmax = np.max(f)
    
    n_sel = min(n_active, len(f))
    top_idx = np.argsort(f)[-n_sel:]
    
    def objective(x):
        h = x[:n]
        t = x[n]
        corr = np.correlate(h, 1.0 - h, mode="full")
        return t + 0.01 * sum(max(0, corr[m] * dx - t) for m in top_idx)
    
    def constraint_eq(x):
        h = x[:n]
        return np.sum(h) - target_sum
    
    def constraint_jac_eq(x):
        h = x[:n]
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j
    
    x0 = np.concatenate([h0, [fmax + 0.01]])
    bounds = [(0, 1)] * n + [(0, 2)]
    
    try:
        result = minimize(objective, x0, method='SLSQP',
                         bounds=bounds,
                         constraints=[{'type': 'eq', 'fun': constraint_eq, 'jac': constraint_jac_eq}],
                         options={'maxiter': max_iter_slsqp})
        if result.success or result.fun < fmax:
            h_new = project_feasible(result.x[:n], target_sum)
            c_new = evaluate_c5(h_new)
            return h_new, c_new
    except Exception:
        pass
    return h0, evaluate_c5(h0)

def resample(h, n_new):
    """Resample h from its current length to n_new using linear interpolation"""
    n_old = len(h)
    x_old = np.linspace(0, 2, n_old, endpoint=False)
    x_new = np.linspace(0, 2, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, np.concatenate([h, [h[0]]]))
    # Handle wrap-around better
    x_old_ext = np.concatenate([x_old, [2.0]])
    h_old_ext = np.concatenate([h, [h[0]]])
    h_new = np.interp(x_new, x_old_ext, h_old_ext)
    target = n_new / 2.0
    return project_feasible(h_new, target)

def project_feasible(y, target_sum):
    n = len(y)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    lo, hi = -1.0, 1.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(y + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    h = np.clip(y + alpha, 0.0, 1.0)
    residual = target_sum - np.sum(h)
    if abs(residual) > 1e-14:
        if residual > 0:
            for idx in np.argsort(-(1.0 - h)):
                step = min(1.0 - h[idx], residual)
                h[idx] += step
                residual -= step
                if abs(residual) < 1e-15:

    return h
