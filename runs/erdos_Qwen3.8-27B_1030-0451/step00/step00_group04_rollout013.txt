import numpy as np
from scipy.optimize import minimize
import time

def project_box_sum(x, T):
    """Project x onto {0 <= h <= 1, sum(h) = T}."""
    n = len(x)
    x = np.clip(x, 0, 1)
    # Bisection on lambda
    lo, hi = -1.0, 2.0  # lambda range
    for _ in range(80):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        s = np.sum(y)
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - mid, 0, 1)
    # Final adjustment
    diff = T - np.sum(y)
    if abs(diff) > 1e-12:
        # Distribute the difference
        # Find elements not at bounds
        mask = (y > 1e-10) & (y < 1 - 1e-10)
        if np.sum(mask) > 0:
            y[mask] += diff / np.sum(mask)
            y = np.clip(y, 0, 1)
    return y

def eval_c5(h):
    """Compute exact C5 value."""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(z))

def smooth_max_grad(h, tau):
    """Compute smooth max surrogate F_tau and its gradient."""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    z = np.correlate(h, b, mode='full') * dx  # length 2n-1
    
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    # For numerical stability, clip
    shifted = np.clip(shifted, -50, 50)
    exp_z = np.exp(shifted)
    sum_exp = np.sum(exp_z)
    p = exp_z / sum_exp
    F = z_max + tau * np.log(sum_exp)
    
    # Gradient
    grad = np.zeros(n)
    # Lag 0 (index n-1 in z)
    w0 = p[n-1]
    grad += w0 * dx * (1.0 - 2.0 * h)
    
    # Positive lags: idx n to 2n-2 correspond to ell = n-1-n = -1? No...
    # idx = n-1+ell where ell = n-1-idx... wait
    # Let me reclarify: z[idx] = dx * sum_i h[i+ell] * (1-h[i]) where ell = n-1-idx
    # So for idx = n: ell = n-1-n = -1 → negative lag
    # For idx = n-2: ell = n-1-(n-2) = 1 → positive lag
    
    # Let me just iterate over all lags
    for idx in range(2*n - 1):
        w = p[idx]
        if w < 1e-14:

        ell = (n - 1) - idx  # the shift
        if ell > 0:

            grad[ell:] += w * dx * b[:L]
            grad[:L] -= w * dx * h[ell:]
        elif ell < 0:
            a = -ell
            L = n - a
            grad[:L] += w * dx * b[a:]
            grad[a:] -= w * dx * h[:L]
        # ell == 0 already handled above
    
    return F, grad, z

def pgd_optimize(h0, tau_init, tau_final, n_iter, alpha_init, deadline, min_alpha=1e-10):
    """PGD optimization with smooth-max surrogate."""
    h = h0.copy()
    T = len(h) / 2.0
    h = project_box_sum(h, T)
    
    best_h = h.copy()
    best_score = eval_c5(h)
    
    tau = tau_init
    alpha = alpha_init
    tau_decay = (tau_final / tau_init) ** (1.0 / max(n_iter, 1))
    
    for it in range(n_iter):
        if time.time() > deadline:
            break
        
        F, grad, z = smooth_max_grad(h, tau)
        
        # Normalize gradient
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-15:
            break
        direction = -grad / gnorm
        
        # Line search
        F_best = F
        best_step = None
        for trial in range(10):
            a = alpha * (0.5 ** trial)
            if a < min_alpha:
                break
            h_new = project_box_sum(h + a * direction, T)
            F_new = smooth_max_grad(h_new, tau)[0]
            if F_new < F_best:
                F_best = F_new
                best_step = a
                h = h_new
                break
            alpha = a * 0.5
        
        if best_step is None:
            alpha *= 0.5
            if alpha < min_alpha:
                break
        else:
            # Slightly increase alpha on success
            alpha = min(alpha * 1.2, 1.0)
            # Check exact score periodically
            if it % 20 == 0:
                score = eval_c5(h)
                if score < best_score - 1e-12:
                    best_score = score
                    best_h = h.copy()
        
        # Decay tau
        tau = max(tau * tau_decay, tau_final)
        
        # Periodic exact check
        if it % 50 == 0:
            score = eval_c5(h)
            if score < best_score - 1e-12:
                best_score = score
                best_h = h.copy()
    
    # Final exact check
    score = eval_c5(h)
    if score < best_score - 1e-12:
        best_score = score
        best_h = h.copy()
    
    return best_h, best_score

import numpy as np
from scipy.optimize import minimize
import time

def project_box_sum(x, T):
    """Project x onto {0 <= h <= 1, sum(h) = T} using bisection."""
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    lo, hi = -2.0, 3.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - mid, 0.0, 1.0)
    # Tiny repair
    diff = T - np.sum(y)
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = np.sum(mask)
        if cnt > 0:
            y[mask] += diff / cnt
            y = np.clip(y, 0.0, 1.0)
    return y

def eval_c5(h):
    """Compute exact C5 = max_k integral h(x)(1-h(x+k)) dx."""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(z))

def get_z(h):
    """Compute all correlation values."""
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def smooth_max_grad(h, tau):
    """Compute F_tau(h) and its gradient."""
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    z = np.correlate(h, b, mode='full') * dx
    m = len(z)  # 2n-1
    
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_z = np.exp(shifted)
    s_exp = np.sum(exp_z)
    p = exp_z / s_exp
    F = z_max + tau * np.log(s_exp)
    
    grad = np.zeros(n)
    # Only process active lags
    active = np.where(p > 1e-12)[0]
    for idx in active:
        w = p[idx]
        ell = (n - 1) - idx
        if ell > 0:
            L = n - ell
            grad[ell:] += w * dx * b[:L]
            grad[:L] -= w * dx * h[ell:]
        elif ell < 0:
            a = -ell
            L = n - a
            grad[:L] += w * dx * b[a:]
            grad[a:] -= w * dx * h[:L]
        else:
            grad += w * dx * (1.0 - 2.0 * h)
    
    return F, grad, z

def pgd_optimize(h0, deadline, tau_init=0.015, tau_final=5e-5,
                 alpha_init=0.1, tau_decay=0.95, tau_decay_period=100):
    """PGD with smooth-max surrogate. Returns (best_h, best_score)."""
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.copy(), T)
    
    best_h = h.copy()
    best_score = eval_c5(h)
    
    tau = tau_init
    alpha = alpha_init
    it = 0
    n_iter = 50000  # max iterations, will stop at deadline
    
    while it < n_iter:
        if time.time() >= deadline:
            break
        
        F, grad, z = smooth_max_grad(h, tau)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-14:
            break
        
        direction = -grad / gnorm
        
        # Line search (at most 5 trials)
        accepted = False
        for trial in range(5):
            a = alpha * (0.5 ** trial)
            if a < 1e-10:
                break
            h_new = project_box_sum(h + a * direction, T)
            F_new = smooth_max_grad(h_new, tau)[0]
            if F_new <= F + 1e-14:
                h = h_new
                alpha = min(a * 1.15, 0.5)
                accepted = True
                break
        if not accepted:
            alpha *= 0.4
            if alpha < 1e-10:
                break
        
        it += 1
        
        # Decay tau
        if it % tau_decay_period == 0:
            tau = max(tau * tau_decay, tau_final)
        
        # Periodic exact evaluation
        if it % 50 == 0:
            score = eval_c5(h)
            if score < best_score - 1e-13:
                best_score = score
                best_h = h.copy()
    
    # Final evaluation
    score = eval_c5(h)
    if score < best_score - 1e-13:
        best_score = score
        best_h = h.copy()
    
    return best_h, best_score

class SlsqpEpi:
    """SLSQP epigraph state."""
    def __init__(self, n, T, dx):
        self.n = n
        self.T = T
        self.dx = dx
        self.call_count = 0
        self.max_calls = 2000
    
    def objective(self, x):
        self.call_count += 1
        return x[-1]
    
    def jac_objective(self, x):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g
    
    def constraint(self, x):
        h = x[:self.n]
        t = x[-1]
        z = np.correlate(h, 1.0 - h, mode='full') * self.dx
        return t - z
    
    def jacobian(self, x):
        h = x[:self.n]
        n = self.n
        dx = self.dx
        b = 1.0 - h
        m = 2 * n - 1
        J = np.zeros((m, n + 1))
        J[:, -1] = 1.0
        for idx in range(m):
            ell = (n - 1) - idx
            if ell > 0:
                L = n - ell
                J[idx, ell:] += -dx * b[:L]
                J[idx, :L] += dx * h[ell:]
            elif ell < 0:
                a = -ell
                L = n - a
                J[idx, :L] += -dx * b[a:]
                J[idx, a:] += dx * h[:L]
            else:
                J[idx, :] -= dx * (1.0 - 2.0 * h)
        return J
    
    def eq_constraint(self, x):
        return np.sum(x[:self.n]) - self.T
    
    def eq_jacobian(self, x):
        J = np.zeros(self.n + 1)
        J[:self.n] = 1.0
        return J


def slsqp_polish(h0, deadline, max_iter=300):
    """SLSQP epigraph polish."""
    n = len(h0)
    T = n / 2.0
    dx = 2.0 / n
    c5 = eval_c5(h0)
    t0 = max(0.01, c5 + 1e-6)
    
    x0 = np.zeros(n + 1)
    x0[:n] = h0
    x0[-1] = t0
    
    state = SlsqpEpi(n, T, dx)
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.8)]
    
    cons = [
        {'type': 'ineq', 'fun': state.constraint, 'jac': state.jacobian},
        {'type': 'eq', 'fun': state.eq_constraint, 'jac': state.eq_jacobian}
    ]
    
    try:
        res = minimize(
            state.objective, x0, jac=state.jac_objective,
            method='SLSQP', bounds=bounds, constraints=cons,
            options={'maxiter': max_iter, 'ftol': 1e-12}
        )
    except Exception:
        return h0.copy(), c5
    
    if res.success or res.fun < t0:
        h_sol = project_box_sum(res.x[:n], T)
        score = eval_c5(h_sol)
        if score < c5 - 1e-13:
            return h_sol, score
    
    return h0.copy(), c5

def upsample(h, factor):
    return np.repeat(h, factor)

def downsample(h, factor):
    n = len(h)
    new_n = n // factor
    h_trim = h[:new_n * factor]
    return h_trim.reshape(new_n, factor).mean(axis=1)
