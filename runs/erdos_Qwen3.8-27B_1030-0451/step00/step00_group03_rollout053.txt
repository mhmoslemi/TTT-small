    if rng is None:
        rng = np.random.default_rng(0)
    active = np.where((h > 1e-6) & (h < 1.0 - 1e-6))[0]
    for p in range(max_passes):
        if (time.monotonic() - t_start) > timeout:
            break
        improved = False
        n_tries = min(200, len(active) * 3)
        for _ in range(n_tries):
            if len(active) < 2:
                break
            i = active[rng.integers(len(active))]
            j = active[rng.integers(len(active))]
            if i == j:
                continue
            t_lo = max(-h[i], h[j] - 1.0)
            t_hi = min(1.0 - h[i], h[j])
            if t_hi - t_lo < 1e-12:
                continue
            oi, oj = h[i], h[j]
            best_t, best_cv = 0.0, c
            ts = np.linspace(t_lo, t_hi, n_grid)
            for t in ts:
                h[i] = oi + t; h[j] = oj - t
                cv = exact_c5(h)
                if cv < best_cv:
                    best_cv = cv
                    best_t = t
            if best_t == 0.0:
                h[i], h[j] = oi, oj
                continue
            step = (t_hi - t_lo) / n_grid
            a = max(t_lo, best_t - step)
            b = min(t_hi, best_t + step)
            for _ in range(14):
                m1 = a + 0.381966 * (b - a)
                m2 = a + 0.618034 * (b - a)
                h[i] = oi + m1; h[j] = oj - m1
                c1 = exact_c5(h)
                h[i] = oi + m2; h[j] = oj - m2
                c2 = exact_c5(h)
                if c1 < c2:
                    b = m2
                else:
                    a = m1
            ts = (a + b) / 2.0
            h[i] = oi + ts; h[j] = oj - ts
            cv = exact_c5(h)
            if cv < c - 1e-11:
                c = cv
                improved = True
            else:
                h[i], h[j] = oi, oj
        if not improved:
            break
    return h, c

def _slsqp_obj(x, args):
    return x[-1]

def _slsqp_obj_grad(x, args):
    n = args['n']
    g = np.zeros(n + 1)
    g[-1] = 1.0
    return g

def _slsqp_constraints(x, args):
    h = x[:-1]
    T = x[-1]
    n = args['n']
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = args['active_idx']
    vals = T - cor[idx]
    return vals

def _slsqp_constraints_grad(x, args):
    h = x[:-1]
    T = x[-1]
    n = args['n']
    dx = 2.0 / n
    idx = args['active_idx']
    m = len(idx)
    J = np.zeros((m, n + 1))
    for r, m_idx in enumerate(idx):
        k = m_idx - (n - 1)
        if k >= 0:
            valid_i = np.arange(0, n - k)
            J[r, valid_i] += dx * (1.0 - h[valid_i + k])
        else:
            valid_i = np.arange(-k, n)
            J[r, valid_i] += dx * (1.0 - h[valid_i + k])
        if k >= 0:
            J[r, valid_i + k] += -dx * h[valid_i]
        else:
            J[r, valid_i - k] += -dx * h[valid_i]
        J[r, -1] = -1.0
    return J

def slsqp_polish(h, c, timeout=60.0, t_start=None, max_rounds=3):
    if t_start is None:
        t_start = time.monotonic()
    n = len(h)
    h = h.copy()
    c0 = exact_c5(h)
    if (time.monotonic() - t_start) > timeout:
        return h, c0
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full") * dx
    max_idx = np.argmax(cor)
    active = np.argsort(cor)[-min(40, 2 * n - 1):][::-1]
    _SLSQP_STATE['n'] = n
    _SLSQP_STATE['active_idx'] = active
    _SLSQP_STATE['deadline'] = t_start + timeout
    args = {'n': n, 'active_idx': active}
    x0 = np.concatenate([h, [c0 + 1e-4]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    con_eq = {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_grad, 'args': args}
    con_ineq = {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_grad, 'args': args}
    best_h, best_c = h.copy(), c0
    for rnd in range(max_rounds):
        if (time.monotonic() - t_start) > timeout * 0.85:
            break
        try:
            res = minimize(_slsqp_obj, x0, jac=_slsqp_obj_grad, args=args,
                          method='SLSQP', bounds=bounds,
                          constraints=[con_eq, con_ineq],
                          options={'maxiter': 200, 'ftol': 1e-9})
            if res.x is not None and np.all(np.isfinite(res.x)):
                h_cand = np.clip(res.x[:-1], 0.0, 1.0)
                h_cand = project_mean_box(h_cand)
                c_cand = exact_c5(h_cand)
                if c_cand < best_c:
                    best_h = h_cand
                    best_c = c_cand
                    h = h_cand
                    cor_new = np.correlate(h, 1.0 - h, mode="full") * dx
                    active = np.argsort(cor_new)[-min(40, 2 * n - 1):][::-1]
                    args['active_idx'] = active
                    x0 = np.concatenate([h, [best_c + 1e-4]])
        except Exception:
            break
    return best_h, best_c

def _slsqp_eq(x, args):
    h = x[:-1]
    n = args['n']
    return np.sum(h) - n / 2.0

def _slsqp_eq_grad(x, args):
    n = args['n']
    g = np.ones(n)
    g = np.concatenate([g, [0.0]])
    return g

def _slsqp_ineq(x, args):
    h = x[:-1]
    T = x[-1]
    n = args['n']
    dx = 2.0 / n
    cor = np.correlate(h, 1.0 - h, mode="full") * dx
    idx = args['active_idx']
    return T - cor[idx]

def _slsqp_ineq_grad(x, args):
    h = x[:-1]
    n = args['n']
    dx = 2.0 / n
    idx = args['active_idx']
    m = len(idx)
    J = np.zeros((m, n + 1))
    for r in range(m):
        m_idx = idx[r]
        k = m_idx - (n - 1)
        if k >= 0:
            valid_i = np.arange(0, n - k)
            J[r, valid_i] += dx * (1.0 - h[valid_i + k])
            J[r, valid_i + k] += -dx * h[valid_i]
        else:
            valid_i = np.arange(-k, n)
            J[r, valid_i] += dx * (1.0 - h[valid_i + k])
            J[r, valid_i - k] += -dx * h[valid_i]
        J[r, -1] = -1.0
    return J

def random_perturb(h, sigma, rng):
    h_new = h + sigma * rng.standard_normal(len(h))
    return project_mean_box(h_new)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.monotonic()
    deadline = t_start + budget_s
    rng = np.random.default_rng(seed)

    # Get initial values
    h_best = None
    c_best = np.inf
    n_best = 0

    def submit(h_cand):
        nonlocal h_best, c_best, n_best
        if h_cand is None:
            return
        h_cand = np.asarray(h_cand, dtype=np.float64).copy()
        if not np.all(np.isfinite(h_cand)):
            return
        h_feas = project_mean_box(h_cand)
        c_val = exact_c5(h_feas)
        if c_val < c_best - 1e-12:
            c_best = c_val
            h_best = h_feas
            n_best = len(h_feas)
        elif abs(c_val - c_best) < 1e-12 and len(h_feas) < n_best:
            h_best = h_feas
            c_best = c_val
            n_best = len(h_feas)

    # Initialize from parent
    try:
        parent = np.asarray(initial_h_values, dtype=np.float64).copy()
    except NameError:
        parent = np.full(64, 0.5)
    except Exception:
        parent = np.full(64, 0.5)

    parent = project_mean_box(parent)
    n_parent = len(parent)
    submit(parent)

    # Try simple transforms of parent
    transforms = [parent.copy(), parent[::-1].copy(), 1.0 - parent.copy(), 1.0 - parent[::-1].copy()]
    for t in transforms:
        submit(t)

    # Also try small shifts
    for s in range(1, n_parent // 2):
        shifted = np.roll(parent, s)
        submit(shifted)

    # Phase 1: Pair sweep at parent dimension
    n_work = n_parent
    h_work = h_best.copy() if h_best is not None else parent.copy()
    c_work = exact_c5(h_work)

    remaining = deadline - time.monotonic()
    if remaining > 30:
        t1 = time.monotonic()
        h_work, c_work = pair_sweep(h_work, c_work, timeout=min(remaining * 0.5, 200), t_start=t1)
        submit(h_work)

    # Phase 2: Multi-pair sweep with randomness
    remaining = deadline - time.monotonic()
    if remaining > 30:
        t2 = time.monotonic()
        h_work, c_work = multi_pair_sweep(h_work, c_work, timeout=min(remaining * 0.3, 120), t_start=t2, rng=rng)
        submit(h_work)

    # Phase 3: Try different dimensions
    dims_to_try = []
    if n_parent <= 96:
        dims_to_try = [n_parent, 2 * n_parent]
    else:
        dims_to_try = [n_parent]
    for d in [64, 96, 128, 160, 192]:
        if d not in dims_to_try and d <= 2500:
            dims_to_try.append(d)
    dims_to_try = sorted(set(d for d in dims_to_try if d <= 2500))

    for target_n in dims_to_try:
        if target_n == n_best:
            continue
        remaining = deadline - time.monotonic()
        if remaining < 40:
            break
        h_dim = resample(h_best, target_n)
        c_dim = exact_c5(h_dim)
        # Pair sweep
        t3 = time.monotonic()
        timeout_dim = min(remaining * 0.35, 150)
        if target_n <= 96:
            h_dim, c_dim = pair_sweep(h_dim, c_dim, timeout=timeout_dim, t_start=t3, n_grid=16)
        else:
            h_dim, c_dim = pair_sweep(h_dim, c_dim, timeout=timeout_dim, t_start=t3, n_grid=12, max_passes=20)
        submit(h_dim)
        # Multi-pair
        remaining = deadline - time.monotonic()
        if remaining > 30:
            t4 = time.monotonic()
            h_dim, c_dim = multi_pair_sweep(h_dim, c_dim, timeout=min(remaining * 0.3, 100), t_start=t4, rng=rng)
            submit(h_dim)

    # Phase 4: Random restarts
    remaining = deadline - time.monotonic()
    n_restarts = 0
    while remaining > 60 and n_restarts < 20:
        n_restarts += 1
        h_rand = random_perturb(h_best, 0.02, rng)
        c_rand = exact_c5(h_rand)
        if c_rand < c_best:
            submit(h_rand)
        t5 = time.monotonic()
        h_r, c_r = pair_sweep(h_rand, min(c_rand, c_best), timeout=min(remaining * 0.15, 60), t_start=t5, n_grid=14)
        submit(h_r)
        remaining = deadline - time.monotonic()

    # Phase 5: SLSQP polish
    remaining = deadline - time.monotonic()
    if remaining > 45 and n_best <= 192:
        t6 = time.monotonic()
        h_pol, c_pol = slsqp_polish(h_best, c_best, timeout=min(remaining * 0.6, 300), t_start=t6)
        submit(h_pol)

    # Phase 6: Final pair sweep on best
    remaining = deadline - time.monotonic()
    if remaining > 20:
        t7 = time.monotonic()
        h_fin, c_fin = pair_sweep(h_best, c_best, timeout=min(remaining * 0.7, 120), t_start=t7, n_grid=24, max_passes=30)
        submit(h_fin)

    # Final validation
    if h_best is None:
        h_best = np.full(64, 0.5)
        c_best = exact_c5(h_best)
        n_best = 64

    h_best = project_mean_box(h_best)
    c_final = exact_c5(h_best)
    n_final = len(h_best)

    # Ensure consistency: if final eval is slightly different, use it
    if c_final < c_best:
        c_best = c_final

    return h_best, c_best, n_final
