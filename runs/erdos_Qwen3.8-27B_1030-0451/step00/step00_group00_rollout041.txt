    b = 1 - h

    k_max = np.argmax(corr)

from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    corr = correlate(h, 1-h, mode='full') * dx
    return np.max(corr)

def gradient_at_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if n > lag:
            grad[lag:] = (1.0 - h[:n-lag])
            grad[:n-lag] -= h[lag:]
    else:
        m = -lag
        if n > m:
            grad[:n-m] = (1.0 - h[m:])
            grad[m:] -= h[:n-m]
    return grad

def smoothed_grad(h, n, dx, epsilon=0.03):
    corr_raw = correlate(h, 1-h, mode='full')
    max_c = np.max(corr_raw)
    exp_c = np.exp((corr_raw - max_c) / epsilon)
    w = exp_c / exp_c.sum()
    # Top lags
    threshold = w.max() * 0.01
    active = np.where(w > threshold)[0]
    if len(active) == 0:
        active = np.array([np.argmax(w)])
    grad = np.zeros(n)
    for idx in active:

            continue
        lag = idx - (n - 1)
        grad += w[idx] * gradient_at_lag(h, n, lag)

import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def _compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full') * dx
    return float(np.max(corr))

def _grad_at_lag(h, n, lag):
    grad = np.zeros(n)
    if lag >= 0:
        if n > lag:
            grad[lag:] = (1.0 - h[:n - lag])
            grad[:n - lag] -= h[lag:]
    else:
        m = -lag
        if n > m:
            grad[:n - m] = (1.0 - h[m:])
            grad[m:] -= h[:n - m]
    return grad

def _smoothed_obj_grad(h, n, dx, eps=0.02):
    corr = correlate(h, 1 - h, mode='full') * dx
    mc = np.max(corr)
    exp_c = np.exp((corr - mc) / eps)
    w = exp_c / exp_c.sum()
    obj = mc + eps * np.log(exp_c.sum() / len(corr))
    active = np.where(w > w.max() * 0.005)[0]
    if len(active) == 0:
        active = np.array([np.argmax(w)])
    g = np.zeros(n)
    for idx in active:
        if w[idx] < 1e-12:
            continue
        lag = int(idx) - (n - 1)
        g += w[idx] * _grad_at_lag(h, n, lag)
    return obj, g * dx

def _project(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def _optimize_slsqp(h0, n, dx, eps=0.02, max_iter=150):
    h = _project(h0.astype(float), n / 2.0)
    def obj(x):
        return _smoothed_obj_grad(x, n, dx, eps)[0]

        return _smoothed_obj_grad(x, n, dx, eps)[1]
    def cons_fun(x):
        return np.sum(x) - n / 2.0
    res = minimize(obj, h, jac=grad, method='SLSQP',
                   bounds=[(0.0, 1.0)] * n,
                   constraints=[{'type': 'eq', 'fun': cons_fun}],
                   options={'maxiter': max_iter, 'ftol': 1e-10})
    return _project(res.x, n / 2.0)

def _optimize_lbfgs(h0, n, dx, eps=0.02, max_iter=300):
    h = _project(h0.astype(float), n / 2.0)
    pen = 500.0
    def pobj(x):
        c5 = _compute_c5(x, dx)
        return c5 + pen * (np.sum(x) - n / 2.0) ** 2
    def pgrad(x):
        g = _smoothed_obj_grad(x, n, dx, eps)[1]
        g = g + 2.0 * pen * (np.sum(x) - n / 2.0) * np.ones(n)
        return g
    res = minimize(pobj, h, jac=pgrad, method='L-BFGS-B',
                   bounds=[(0.0, 1.0)] * n,
                   options={'maxiter': max_iter, 'ftol': 1e-12})
    return _project(res.x, n / 2.0)

def _projected_subgrad(h0, n, dx, max_iter=3000, lr0=0.02, eps=0.02, seed=0):

    best_h = h.copy()

def _binary_round_optimize(h0, n, dx, max_iter=200, seed=0):
    rng = np.random.default_rng(seed)
    target = n // 2
    h = _project(h0, n / 2.0)

            best_h = h_bin.copy()
        else:

            h_bin[i2] = 0.0
    return best_h, best_c5

    h = rng.random(n)
    target = n / 2.0
    s = h.sum()
    h = h * (target / s) if s > 0 else h
