        i0 = zeros[rng.integers(len(zeros))]
        h[i1] = 0.0
        h[i0] = 1.0
        c5_new, _ = compute_c5(h, dx)
        if c5_new < best_val:
            best_val = c5_new
            best = h.copy()
        else:
            h[i1] = 1.0
            h[i0] = 0.0
    return best, best_val

def make_initial_h(n, seed=42):
    rng = np.random.default_rng(seed)
    if n % 2 == 0:
        ones = rng.choice(n, size=n // 2, replace=False)
        h = np.zeros(n)
        h[ones] = 1.0
        return h
    else:
        h = np.zeros(n)
        h[:n // 2] = 1.0
        h[n // 2] = 0.5
        return h

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    h_init = np.array(initial_h_values, dtype=float)
    n_init = len(h_init)
    dx_init = 2.0 / n_init
    best_h = h_init.copy()
    best_c5, _ = compute_c5(best_h, dx_init)
    best_n = n_init
    
    def try_optimize(h_start, n, dx, s, t_limit):
        nonlocal best_h, best_c5, best_n
        h_opt, c5_opt = optimize_gradient(h_start, n, dx, s, time_limit=t_limit)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            best_n = n
        return h_opt
    
    # Phase 1: Optimize from initial (n_init)
    t_left = budget_s - (time.time() - start_time) - 30
    if t_left > 15:
        try_optimize(h_init, n_init, dx_init, n_init / 2.0, t_left * 0.25)
    
    # Phase 2: Try various n values
    n_options = []
    for n_try in [80, 100, 120, 128, 150, 160, 200, 250]:
        if n_try != n_init and n_try >= 40:
            n_options.append(n_try)
    
    t_left = budget_s - (time.time() - start_time) - 20
    if len(n_options) > 0 and t_left > 20:
        time_per_n = t_left / len(n_options)
        rng = np.random.default_rng(seed + 1)
        
        for n in n_options:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 25:
                break
            t_avail = min(time_per_n, budget_s - elapsed - 15)
            if t_avail < 8:
                break
            
            dx = 2.0 / n
            s = n / 2.0
            
            # Interpolate from initial
            h_interp = np.interp(np.linspace(0, 2, n, endpoint=False),
                                 np.linspace(0, 2, n_init, endpoint=False), h_init)
            h_interp = project_feasible(h_interp, s)
            
            h_opt = try_optimize(h_interp, n, dx, s, t_avail * 0.5)
            
            # Binary local search for even n
            if n % 2 == 0 and time.time() - start_time < budget_s - 20:
                t_left_bin = min(t_avail * 0.3, budget_s - (time.time() - start_time) - 15)
                if t_left_bin > 5:
                    h_bin, c5_bin = local_search_binary(h_opt, n, dx, t_left_bin, rng)
                    if c5_bin < best_c5:
                        best_c5 = c5_bin
                        best_h = h_bin.copy()
                        best_n = n
            
            # Random restart
            if time.time() - start_time < budget_s - 30:
                t_left_rand = min(t_avail * 0.2, budget_s - (time.time() - start_time) - 15)
                if t_left_rand > 5:
                    h_rand = make_initial_h(n, seed=seed + n)
                    h_rand = project_feasible(h_rand, s)
                    h_opt2 = try_optimize(h_rand, n, dx, s, t_left_rand * 0.6)
                    if n % 2 == 0 and time.time() - start_time < budget_s - 20:
                        t_lb = min(t_left_rand * 0.3, budget_s - (time.time() - start_time) - 15)
                        if t_lb > 3:
                            h_bin2, c5_bin2 = local_search_binary(h_opt2, n, dx, t_lb, rng)
                            if c5_bin2 < best_c5:
                                best_c5 = c5_bin2
                                best_h = h_bin2.copy()
                                best_n = n
    
    # Phase 3: Fine-tune best solution with longer gradient run
    t_left = budget_s - (time.time() - start_time) - 5
    if t_left > 10:
        dx_best = 2.0 / best_n
        s_best = best_n / 2.0
        h_final, c5_final = optimize_gradient(best_h, best_n, dx_best, s_best, time_limit=t_left)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Final verification
    final_c5, _ = compute_c5(best_h, 2.0 / best_n)
    return best_h, final_c5, best_n
