    def sum_clipped(mu):
        return np.sum(np.clip(z - mu, 0.0, 1.0))
    # Bisection on mu
    lo, hi = -1.0 - np.max(np.abs(z)), 1.0 + np.max(np.abs(z))

def is_feasible(h, tol=1e-9):
    n = len(h)
    target = n / 2.0
    return (np.all(h >= -tol) and np.all(h <= 1+tol) and abs(np.sum(h) - target) < 1e-7)

def repair(h):
    n = len(h)
    target = n / 2.0
    y = np.clip(h, 0.0, 1.0)
    # Bisection on threshold
    for _ in range(100):
        s = np.sum(y)
        err = target - s
        if abs(err) < 1e-14:

        if err > 0:

            tr = np.sum(room)
            if tr < 1e-15:

            y += (err / tr) * room
        else:

            tr = np.sum(room)
            if tr < 1e-15:

            y += (err / tr) * room

    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    max_c = np.max(c)
    shifted = tau * (c - max_c)
    exp_c = np.exp(shifted)
    sum_exp = np.sum(exp_c)
    obj = (2.0 / n) * (max_c + np.log(sum_exp) / tau)

def _exact_score(h):

    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    return float(np.max(c) * dx)

def _repair(h):
    n = len(h)
    target = n / 2.0
    y = np.clip(h, 0.0, 1.0)
    # Bisection
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    err = target - np.sum(y)
    if abs(err) > 1e-13:
        interior = (y > 1e-10) & (y < 1.0 - 1e-10)
        if np.any(interior):
            y[interior] += err / np.sum(interior)
            y = np.clip(y, 0.0, 1.0)

    if abs(err) > 1e-13:
        # Find the element with most room
        if err > 0:
            idx = np.argmax(1.0 - y)
        else:
            idx = np.argmax(y)
        y[idx] += err
        y = np.clip(y, 0.0, 1.0)
    return y

def _lse_obj_grad(h, tau):
    n = len(h)
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    max_c = np.max(c)
    exp_c = np.exp(tau * (c - max_c))
    sum_exp = np.sum(exp_c)
    obj = (2.0 / n) * (max_c + math.log(sum_exp) / tau)
    w = exp_c / sum_exp
    grad = np.zeros(n)
    for i in range(len(c)):
        k = i - (n - 1)
        wi = w[i]
        if wi < 1e-16:
            continue
        # (1-h[j-k]) for j-k in [0,n)
        js = max(0, k)
        je = min(n, n + k)
        if je > js:
            grad[js:je] += wi * (1.0 - h[js-k:je-k])
        # -h[j+k] for j+k in [0,n)
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if je2 > js2:
            grad[js2:je2] -= wi * h[js2+k:je2+k]
    grad *= (2.0 / n)
    return obj, grad

def _lse_descent(h0, tau, max_iter=300, time_limit=30.0):
    h = h0.copy()
    t0 = time.monotonic()
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        obj, grad = _lse_obj_grad(h, tau)
        # Projected gradient step with backtracking
        improved = False
        for alpha in [0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = _repair(h - alpha * grad)
            obj_trial, _ = _lse_obj_grad(h_trial, tau)
            if obj_trial < obj - 1e-14:
                h = h_trial
                improved = True
                break
        if not improved:
            break
    return h

            d = delta_min + frac * (delta_max - delta_min)

            h[i] += d

            s = _exact_score(h)

def _optimal_delta(h, i, j, c):
    n = len(h)
    # Compute A_k for all k
    # A_k = -h[i+k] - h[i-k] + h[j+k] + h[j-k] (with boundary = 0)
    A = np.zeros(2*n-1)
    for idx in range(2*n-1):
        k = idx - (n-1)
        a = 0.0
        if 0 <= i+k < n: a -= h[i+k]
        if 0 <= i-k < n: a -= h[i-k]
        if 0 <= j+k < n: a += h[j+k]
        if 0 <= j-k < n: a += h[j-k]
        A[idx] = a
    
    # f(δ) = max_k (c[k] + δ*A[k])
    # δ range: [delta_min, delta_max]
    delta_min = max(-h[i], h[j]-1.0)
    delta_max = min(1.0-h[i], h[j])
    
    # Evaluate at boundaries
    f_min = np.max(c + delta_min * A)
    f_max = np.max(c + delta_max * A)
    
    # Find breakpoints among top lags
    # For efficiency, only consider lags where c[k] is near the max
    top_idx = np.argsort(c)[-min(20, 2*n-1):]
    breakpoints = set()
    breakpoints.add(delta_min)
    breakpoints.add(delta_max)
    c_top = c[top_idx]
    A_top = A[top_idx]
    for a in range(len(top_idx)):
        for b in range(a+1, len(top_idx)):
            if abs(A_top[b] - A_top[a]) > 1e-15:
                db = (c_top[a] - c_top[b]) / (A_top[b] - A_top[a])
                if delta_min <= db <= delta_max:
                    breakpoints.add(db)
    
    # Evaluate f at all breakpoints and find the minimum
    best_d = delta_min
    best_f = f_min
    for d in breakpoints:
        f = np.max(c + d * A)
        if f < best_f:
            best_f = f
            best_d = d
    
    return best_d, best_f

c[k] = number of i where h[i]=1 and h[i+k]=0

c[k>0] = # of 10 patterns at distance k
c[-k] = # of 01 patterns at distance k

And c[k] + c[-k] = # of "transitions" (either 10 or 01) at distance k.

import numpy as np

def _exact_score(h):
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    return float(np.max(c) * dx)

def _repair(h):
    n = len(h)
    target = n / 2.0
    lo, hi = -3.0, 3.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    err = target - np.sum(y)
    if abs(err) > 1e-13:
        interior = (y > 1e-12) & (y < 1.0 - 1e-12)
        cnt = np.sum(interior)
        if cnt > 0:
            y[interior] += err / cnt
            y = np.clip(y, 0.0, 1.0)
    err = target - np.sum(y)
    if abs(err) > 1e-13:
        if err > 0:
            idx = int(np.argmax(1.0 - y))
        else:
            idx = int(np.argmax(y))
        y[idx] = float(np.clip(y[idx] + err, 0.0, 1.0))
    return y

def _lse_obj_grad(h, tau):
    n = len(h)
    g = 1.0 - h
    c = np.correlate(h, g, mode='full')
    max_c = float(np.max(c))
    exp_c = np.exp(tau * (c - max_c))
    sum_exp = float(np.sum(exp_c))
    obj = (2.0 / n) * (max_c + math.log(sum_exp) / tau)
    w = exp_c / sum_exp
    grad = np.zeros(n)
    nc = len(c)
    for i in range(nc):
        k = i - (n - 1)
        wi = w[i]
        if wi < 1e-14:
            continue
        js = max(0, k)
        je = min(n, n + k)
        if je > js:
            grad[js:je] += wi * (1.0 - h[js - k:je - k])
        js2 = max(0, -k)
        je2 = min(n, n - k)
        if je2 > js2:
            grad[js2:je2] -= wi * h[js2 + k:je2 + k]
    grad *= (2.0 / n)
    return obj, grad

def _lse_descent(h0, tau, max_iter=500, time_limit=20.0):
    h = h0.copy()
    t0 = time.monotonic()
    prev_obj = _lse_obj_grad(h, tau)[0]
    for it in range(max_iter):
        if time.monotonic() - t0 > time_limit:
            break
        obj, grad = _lse_obj_grad(h, tau)
        improved = False
        for alpha in [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]:
            h_trial = _repair(h - alpha * grad)
            obj_trial = _lse_obj_grad(h_trial, tau)[0]
            if obj_trial < obj - 1e-12:
                h = h_trial
                prev_obj = obj_trial
                improved = True
                break
        if not improved:
            break
    return h

def _compute_A(h, i, j):
    n = len(h)
    A = np.zeros(2 * n - 1)
    for idx in range(2 * n - 1):
        k = idx - (n - 1)
        a = 0.0
        if 0 <= i + k < n:
            a -= h[i + k]
        if 0 <= i - k < n:
            a -= h[i - k]
        if 0 <= j + k < n:
            a += h[j + k]
        if 0 <= j - k < n:
            a += h[j - k]
        A[idx] = a
    return A

def _optimal_delta(c, i, j, h):
    n = len(h)
    delta_min = max(-h[i], h[j] - 1.0)
    delta_max = min(1.0 - h[i], h[j])
    if delta_max - delta_min < 1e-12:
        return 0.0
    A = _compute_A(h, i, j)
    fmin = float(np.max(c + delta_min * A))
    fmax = float(np.max(c + delta_max * A))
    top_k = min(30, 2 * n - 1)
    top_idx = np.argsort(c)[-top_k:]
    best_d = 0.0
    best_f = min(fmin, fmax)
    for d in [delta_min, delta_max, 0.0]:
        f = float(np.max(c + d * A))
        if f < best_f:
            best_f = f
            best_d = d
    c_top = c[top_idx]
    A_top = A[top_idx]
    for a in range(len(top_idx)):
        for b in range(a + 1, len(top_idx)):
            da = A_top[b] - A_top[a]
            if abs(da) > 1e-14:
                db = (c_top[a] - c_top[b]) / da
                if delta_min - 1e-12 <= db <= delta_max + 1e.12:
                    db = float(np.clip(db, delta_min, delta_max))
                    f = float(np.max(c + db * A))
                    if f < best_f:
                        best_f = f
                        best_d = db
    return best_d
