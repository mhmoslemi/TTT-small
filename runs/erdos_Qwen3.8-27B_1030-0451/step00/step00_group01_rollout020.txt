    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    res = target - h.sum()
    if abs(res) > 1e-12:
        if res > 0:
            order = np.argsort(1.0 - h)
        else:
            order = np.argsort(h)
        for idx in order:
            if res > 0:
                d = min(res, 1.0 - h[idx])
            else:
                d = max(res, -h[idx])
            h[idx] += d
            res -= d
            if abs(res) < 1e-12:
                break
    return h

def _score(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def _fourier_basis(n, dim):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    B = np.zeros((n, dim))
    col, k = 0, 1
    while col < dim:
        B[:, col] = np.cos(2.0 * np.pi * k * x)
        col += 1
        if col < dim:
            B[:, col] = np.sin(2.0 * np.pi * k * x)
            col += 1
        k += 1
    for j in range(dim):
        m = np.abs(B[:, j]).max()
        if m > 1e-12:
            B[:, j] /= m
    return B

def _resample(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) / n_old
    x_new = (np.arange(n_new) + 0.5) / n_new
    return _project_feasible(np.interp(x_new, x_old, h), n_new)
