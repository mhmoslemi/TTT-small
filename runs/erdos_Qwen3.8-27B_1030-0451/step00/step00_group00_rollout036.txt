```python
def compute_gradient(h, p, dx):
    n = len(h)
    # p has length 2n-1
    conv = np.convolve(h, p, 'full')  # length 3n-2
    corr = np.correlate(h, p, 'full')  # length 3n-2
    A = conv[n-1 : 2*n-1]  # n elements
    B = corr[n-1 : 2*n-1]  # n elements
    return dx * (1.0 - A - B)
```

Now for the full optimization:

```python
def smooth_objective_and_grad(h, T, dx):
    n = len(h)
    c5_vals = np.correlate(h, 1-h, 'full') * dx  # length 2n-1
    
    # Log-sum-exp
    max_val = np.max(c5_vals)
    exp_vals = np.exp(T * (c5_vals - max_val))
    sum_exp = np.sum(exp_vals)
    obj = max_val + np.log(sum_exp) / T
    
    # Softmax weights
    p = exp_vals / sum_exp
    
    # Gradient
    conv = np.convolve(h, p, 'full')
    corr = np.correlate(h, p, 'full')
    A = conv[n-1 : 2*n-1]
    B = corr[n-1 : 2*n-1]
    grad = dx * (1.0 - A - B)
    
    return obj, grad
```

```python
def project(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    n = len(h)
    # Binary search for the threshold
    # We want: sum(clamp(h[i] - tau, 0, 1)) = target_sum
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        projected = np.clip(h - mid, 0, 1)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo+hi)/2, 0, 1)
    # Adjust for numerical error
    diff = target_sum - np.sum(projected)
    if abs(diff) < 1e-10:
        return projected
    # Small adjustment
    # Add diff to elements that are strictly between 0 and 1
    mask = (projected > 1e-10) & (projected < 1 - 1e-10)
    if np.sum(mask) > 0:
        projected[mask] += diff / np.sum(mask)
        projected = np.clip(projected, 0, 1)
    return projected
```

```python
def greedy_improve(h, n, dx, max_iter=10000):
    target_sum = n / 2
    f = np.correlate(h, 1-h, 'full') * dx  # current f values
    
    for it in range(max_iter):
        # Find worst lag
        k_star = np.argmax(f)
        k_star_val = k_star - (n-1)  # actual lag
        
        if abs(k_star_val) == n-1:
            # Edge case: only one valid term, can't improve much
            pass
        
        # Find best i (h[i]=1) to turn off
        ones = np.where(h == 1)[0]
        zeros = np.where(h == 0)[0]
        
        if len(ones) == 0 or len(zeros) == 0:
            break
        
        # Score for each i: h[i-k] + h[i+k] (lower is better)
        i_scores = np.zeros(len(ones))
        for idx, i in enumerate(ones):
            s = 0
            ik = i - k_star_val
            if 0 <= ik < n:
                s += h[ik]
            ik = i + k_star_val
            if 0 <= ik < n:
                s += h[ik]
            i_scores[idx] = s
        
        # Score for each j: h[j-k] + h[j+k] (higher is better)
        j_scores = np.zeros(len(zeros))
        for idx, j in enumerate(zeros):
            s = 0
            jk = j - k_star_val
            if 0 <= jk < n:
                s += h[jk]
            jk = j + k_star_val
            if 0 <= jk < n:
                s += h[jk]
            j_scores[idx] = s
        
        i_best = ones[np.argmin(i_scores)]
        j_best = zeros[np.argmax(j_scores)]
        
        if i_best == j_best:
            break
        
        # Apply swap
        h[i_best] = 0
        h[j_best] = 1
        
        # Recompute f
        f_new = np.correlate(h, 1-h, 'full') * dx
        
        # Accept only if max improved
        if np.max(f_new) < np.max(f) - 1e-12:
            f = f_new
        else:
            # Revert
            h[i_best] = 1
            h[j_best] = 0
            break
    
    return h
```

def fix_sum_binary(h, n):

    target = n // 2
    num_ones = int(h.sum())
    if num_ones > target:
        ones = np.where(h == 1)[0]
