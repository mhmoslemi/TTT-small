def sa_block(h0, n, cur_score, t_start, wall, rng, n_rounds=2000):
    """Simulated annealing block."""
    h = h0.copy()
    best_h = h0.copy()
    best_score = cur_score
    S = n / 2.0
    
    # Temperature schedule
    T_start = 0.005
    T_end = 0.0001
    ratio = (T_end / T_start) ** (1.0 / max(1, n_rounds))
    
    for rnd in range(n_rounds):
        if time.monotonic() - t_start > wall:
            break
        T = T_start * (ratio ** rnd)
        
        # Random pair
        i = rng.randint(0, n - 1)
        j = rng.randint(0, n - 1)
        if i == j:
            continue
        
        # Random delta
        orig_i = h[i]
        orig_j = h[j]
        t_lo = max(-orig_i, orig_j - 1.0)
        t_hi = min(1.0 - orig_i, orig_j)
        if t_hi - t_lo < 1e-10:
            continue
        
        t = t_lo + rng.random() * (t_hi - t_lo)
        h[i] = orig_i + t
        h[j] = orig_j - t
        
        new_score = exact_score(h, n)
        delta = new_score - cur_score

def lse_objective(h, n, tau):
    """Log-sum-exp approximation of max correlation."""
    dx = 2.0 / n
    q = 1.0 - h
    full = np.correlate(h, q, mode='full') * dx
    max_f = np.max(full)
    if max_f < -1e15:
        return 0.0
    return max_f + tau * math.log(np.sum(np.exp((full - max_f) / tau)))
