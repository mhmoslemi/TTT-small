                temp = min(temp * 1.5, 0.1)
                h = project_feasible(h + rng.randn(n) * 0.003, n)
    
    return best_h, best_c5

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.RandomState(seed)
    t_start = time.time()
    
    def elapsed():
        return time.time() - t_start
    
    def time_left():
        return budget_s - elapsed()
    
    # Get initial construction
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
    except:
        h_init = np.full(131, 0.5)
    
    n = len(h_init)
    h = project_feasible(h_init, n)
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Phase 1: Main gradient search
    best_h, best_c5 = local_search(h, n, time_left, 200000, seed)
    
    # Phase 2: Try multiple random restarts from best
    n_restarts = 5
    for ri in range(n_restarts):
        if time_left() < 20:
            break
        scale = rng.uniform(0.05, 0.2)
        h_r = project_feasible(best_h + rng.randn(n) * scale, n)
        h_r, c5_r = local_search(h_r, n, time_left, 30000, seed + 1000 + ri)
        if c5_r < best_c5:
            best_h = h_r
            best_c5 = c5_r
    
    # Phase 3: Try nearby resolutions
    for dn in [1, -1, 2, -2, 3, -3]:
        if time_left() < 15:
            break
        n_new = n + dn
        if n_new < 50 or n_new > 400:
            continue
        x_old = np.linspace(0.0, 2.0, n, endpoint=False) + 1.0 / n
        x_new = np.linspace(0.0, 2.0, n_new, endpoint=False) + 1.0 / n_new
        h_res = np.interp(x_new, x_old, best_h)
        h_new = project_feasible(h_res, n_new)
        h_new, c5_new = local_search(h_new, n_new, time_left, 10000, seed + 5000 + dn)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
            n = n_new
    
    # Phase 4: Intensive final polish
    n = len(best_h)
    best_h, best_c5 = local_search(best_h, n, time_left, 50000, seed + 9999)
    
    # Verify and return
    best_h = project_feasible(best_h, len(best_h))
    best_c5 = compute_c5(best_h)
    return (best_h, best_c5, len(best_h))
