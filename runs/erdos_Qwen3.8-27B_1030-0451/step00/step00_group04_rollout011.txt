def project(x, n):
    # Project x to [0,1]^n with sum = n/2
    T = n / 2.0
    x = np.clip(x, 0, 1)
    # Use bisection on shift
    # x + shift, clipped to [0,1], should sum to T
    def sum_at_shift(shift):
        return np.clip(x + shift, 0, 1).sum()
    
    # Initial: current sum
    s = x.sum()
    if abs(s - T) < 1e-10:
        return x
    
    # Bisect
    lo = -1.0 - x.max()
    hi = 1.0 + 1.0 - x.min()
    # Actually, shift ranges from -max(x) to 1-min(x)
    lo = -x.max() - 0.1
    hi = 1.0 - x.min() + 0.1
    
    for _ in range(60):
        mid = (lo + hi) / 2
        s = sum_at_shift(mid)
        if s > T:
            hi = mid
        else:
            lo = mid
    
    result = np.clip(x + (lo+hi)/2, 0, 1)
    # Fix any residual
    residual = T - result.sum()
    if abs(residual) > 1e-10:
        # Distribute to elements not at boundary
        free = (result > 1e-8) & (result < 1 - 1e-8)
        if free.sum() > 0:
            adj = residual / free.sum()
            result[free] += adj
            result = np.clip(result, 0, 1)
            # Re-project if needed
            result = np.clip(result, 0, 1)
    return result

def compute_gradient(h, active_indices, dx):
    # Compute gradient of sum of active correlations
    n = len(h)
    g = np.zeros(n)
    # For each active lag index k (0 to 2n-2), lag = k - (n-1)
    for k in active_indices:
        ell = k - (n - 1)
        if ell == 0:
            # r(0) = sum h[i]*(1-h[i]), grad = 2*h[i]-1... wait
            # r(0) = sum h[i] * (1-h[i]) = sum h[i] - sum h[i]^2
