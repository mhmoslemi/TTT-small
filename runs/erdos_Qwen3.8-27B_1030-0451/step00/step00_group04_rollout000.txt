    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full") * dx
    z_max = np.max(z)
    exponents = np.clip((z - z_max) / tau, -50.0, 0.0)
    exp_vals = np.exp(exponents)
    p = exp_vals / np.sum(exp_vals)
    F = z_max + tau * np.log(np.sum(exp_vals))
    grad = np.zeros(n)
    for q in range(len(z)):
        wk = p[q]
        if wk < 1e-14:
            continue
        ell = q - (n - 1)
        if ell == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += wk * dx * (1.0 - h[:L])
            grad[:L] -= wk * dx * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[:L] += wk * dx * (1.0 - h[a:])
            grad[a:] -= wk * dx * h[:L]
    return F, grad, z

def pgd_optimize(h0, tau_init, tau_min, max_iter, deadline):
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.copy(), T)
    best_h = h.copy()
    best_score = corr_exact(h)
    tau = tau_init
    alpha = 0.05
    for it in range(max_iter):
        if time.time() > deadline:
            break
        if it > 0 and it % 35 == 34:
            tau = max(tau * 0.82, tau_min)
        F, g, z = smooth_max_and_grad(h, tau)
        gnorm = np.max(np.abs(g))
        if gnorm < 1e-12:
            break
        d = -g / gnorm
        found = False
        a = alpha
        for _ in range(18):
            h_new = project_box_sum(h + a * d, T)
            F_new, _, _ = smooth_max_and_grad(h_new, tau)
            if F_new < F - 1e-14:
                h = h_new
                F = F_new
                found = True
                break
            a *= 0.5
        if found:
            alpha = max(a * 1.3, 0.01)
            if it % 12 == 0:
                s = corr_exact(h)
                if s < best_score - 1e-13:
                    best_score = s
                    best_h = h.copy()
        else:
            if alpha < 1e-11:
                break
            alpha *= 0.5
    s = corr_exact(h)
    if s < best_score:
        best_score = s
        best_h = h.copy()
    return best_h, best_score

def slsqp_constraint_factory(n, dx, max_lags):
    def cfunc(x):
        h = x[:n]
        z = np.correlate(h, 1.0 - h, mode="full") * dx
        sel = np.argsort(z)[-max_lags:]
        return z[sel] - x[-1]
    def cjac(x):
        h = x[:n]
        z = np.correlate(h, 1.0 - h, mode="full") * dx
        sel = np.argsort(z)[-max_lags:]
        m = len(sel)
        J = np.zeros((m, n + 1))
        J[:, -1] = -1.0
        for i, q in enumerate(sel):
            ell = q - (n - 1)
            if ell == 0:
                J[i, :n] = dx * (1.0 - 2.0 * h)
            elif ell > 0:
                L = n - ell
                J[i, ell:] += dx * (1.0 - h[:L])
                J[i, :L] -= dx * h[ell:]
            else:
                a = -ell
                L = n - a
                J[i, :L] += dx * (1.0 - h[a:])
                J[i, a:] -= dx * h[:L]
        return J
    return cfunc, cjac

def slsqp_polish(h0, deadline, max_lags=80):
    n = len(h0)
    T = n / 2.0
    dx = 2.0 / n
    c0 = corr_exact(h0)
    SF, SJ = slsqp_constraint_factory(n, dx, max_lags)
    x0 = np.concatenate([h0, [c0 + 0.005]])
    def obj(x):
        return x[-1]
    def obj_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    cons = [{'type': 'ineq', 'fun': lambda x: -SF(x), 'jac': lambda x: -SJ(x)}]
    bounds = [(0.0, 1.0)] * n + [(0.0, 0.95)]
    mi = 80 if n <= 300 else 40
    try:
        res = minimize(obj, x0, jac=obj_grad, method='SLSQP', bounds=bounds,
                       constraints=cons, options={'maxiter': mi, 'ftol': 1e-14})
    except Exception:
        return h0, c0
    if time.time() > deadline:
        return h0, c0
    h_new = project_box_sum(np.clip(res.x[:n], 0.0, 1.0), T)
    s = corr_exact(h_new)
    return h_new, s

def make_feasible(h):
    return project_box_sum(np.clip(h, 0.0, 1.0), len(h) / 2.0)

def get_parent():
    try:
        pv = np.array(initial_h_values, dtype=np.float64).ravel()
        if len(pv) >= 50 and np.all(np.isfinite(pv)):
            return make_feasible(pv), len(pv)
    except Exception:
        pass
    return None, 0

def build_random(n, rng):
    h = rng.uniform(0.3, 0.7, n)
    return make_feasible(h)

def build_smooth(n, rng):
    w = np.sort(rng.uniform(0, n, n))
    h = np.zeros(n)
    h += 0.5
    f = rng.integers(2, 8)
    a = rng.uniform(0.05, 0.25)
    ph = rng.uniform(0, 2 * np.pi)
    x = np.arange(n)
    h += a * np.sin(2 * np.pi * f * x / n + ph)
    return make_feasible(h)

def build_block(n, rng):
    k = rng.integers(n // 5, n // 2)
    start = rng.integers(0, n - k)
    h = np.full(n, 0.4)
    h[start:start + k] = 0.9
    return make_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    rng = np.random.default_rng(seed)
    t0 = time.time()
    hard_deadline = t0 + budget_s - 10.0

    best_h, best_score = None, 1.0
    parent_n = 0

    pv, pn = get_parent()
    if pv is not None:
        best_h = pv.copy()
        best_score = corr_exact(pv)
        parent_n = pn

    if best_h is None:
        n_init = 250
        best_h = build_smooth(n_init, rng)
        best_score = corr_exact(best_h)
        parent_n = n_init

    def try_candidate(h, deadline):
        nonlocal best_h, best_score
        s = corr_exact(h)
        if s < best_score - 1e-13:
            best_h = h.copy()
            best_score = s
            return True
        return False

    def try_pgd(h, deadline, tau_init=0.02, tau_min=5e-6):
        hl, sl = pgd_optimize(h, tau_init, tau_min, 500, deadline)
        try_candidate(hl, deadline)

    def try_slsqp(h, deadline):
        hl, sl = slsqp_polish(h, deadline, max_lags=60)
        try_candidate(hl, deadline)

    # Phase 1: Direct PGD on parent
    dl = min(t0 + budget_s * 0.25, hard_deadline)
    try_pgd(best_h.copy(), dl, tau_init=0.02, tau_min=1e-5)

    # Phase 2: Transformations
    if time.time() < t0 + budget_s * 0.35:
        n = len(best_h)
        transforms = [best_h.copy(), best_h[::-1].copy(), (1.0 - best_h).copy(),
                      (1.0 - best_h[::-1]).copy()]
        for i, tr in enumerate(transforms):
            if time.time() > hard_deadline:
                break
            dl = min(t0 + budget_s * (0.35 + 0.08 * i), hard_deadline)
            try_pgd(tr, dl, tau_init=0.015, tau_min=5e-6)

    # Phase 3: SLSQP polish on best
    if time.time() < t0 + budget_s * 0.55:
        dl = min(t0 + budget_s * 0.6, hard_deadline)
        try_slsqp(best_h.copy(), dl)

    # Phase 4: Perturbation restarts
    if time.time() < t0 + budget_s * 0.75:
        n = len(best_h)
        for trial in range(6):
            if time.time() > t0 + budget_s * 0.7:
                break
            sigma = 0.02 + 0.01 * trial
            h_pert = make_feasible(best_h + sigma * rng.standard_normal(n))
            dl = min(t0 + budget_s * (0.65 + 0.03 * trial), hard_deadline)
            try_pgd(h_pert, dl, tau_init=0.01, tau_min=5e-6)

    # Phase 5: Random diverse starts
    if time.time() < t0 + budget_s * 0.85:
        n = len(best_h)
        generators = [
            lambda: build_smooth(n, rng),
            lambda: build_random(n, rng),
            lambda: build_block(n, rng),
        ]
        for ti in range(4):
            if time.time() > t0 + budget_s * 0.82:
                break
            h_rand = generators[ti % len(generators)]()
            s_rand = corr_exact(h_rand)
            if s_rand < best_score + 0.05:
                dl = min(t0 + budget_s * (0.80 + 0.03 * ti), hard_deadline)
                try_pgd(h_rand, dl, tau_init=0.03, tau_min=1e-5)

    # Phase 6: Higher resolution upsampling
    if time.time() < t0 + budget_s * 0.90 and parent_n <= 300:
        n_up = min(parent_n * 2, 500)
        from scipy.ndimage import zoom
        h_up = zoom(best_h, n_up / len(best_h), order=1)
        h_up = make_feasible(h_up)
        s_up = corr_exact(h_up)
        if s_up < best_score + 0.01:
            dl = min(t0 + budget_s * 0.88, hard_deadline)
            h_opt, _ = pgd_optimize(h_up, 0.01, 3e-6, 200, dl)
            # Downsample back
            factor = len(h_opt) // len(best_h)
            if factor >= 1 and len(h_opt) % factor == 0:
                h_down = h_opt.reshape(factor, len(best_h)).mean(axis=0)
                h_down = make_feasible(h_down)
                try_candidate(h_down, hard_deadline)
            try_candidate(h_opt, hard_deadline)

    # Phase 7: Final SLSQP polish
    if time.time() < hard_deadline:
        dl = hard_deadline
        try_slsqp(best_h.copy(), dl)
        # Another PGD round
        if time.time() < hard_deadline - 2:
            try_pgd(best_h.copy(), hard_deadline, tau_init=0.005, tau_min=2e-6)

    # Final validation
    best_h = make_feasible(best_h)
    best_score = corr_exact(best_h)
    return (best_h, best_score, len(best_h))
