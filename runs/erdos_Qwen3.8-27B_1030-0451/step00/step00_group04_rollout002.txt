def project_box_sum(h, T):
    """Project h onto {0<=x<=1, sum(x)=T} using bisection on water level."""
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    # Find alpha such that sum(clip(h + alpha, 0, 1)) = T
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2
    result = np.clip(h + alpha, 0.0, 1.0)
    # Fix any remaining numerical error
    err = T - np.sum(result)
    if abs(err) > 1e-10:
        # Distribute error among non-boundary elements
        free = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(free):
            result[free] += err / np.sum(free)
            result = np.clip(result, 0.0, 1.0)
    return result

def slp_step(h, K, rho, dx):
    """One SLP step. Returns (new_h, improved)."""
    n = len(h)
    T = n / 2.0
    rc = raw_corr(h)
    scaled = rc * dx
    max_val = np.max(scaled)
    
    # Select active constraints
    order = np.argsort(-scaled)
    gap = max(1e-6, 0.005 * max_val)
    active = [q for q in order if scaled[q] >= max_val - gap]
    active = active[:K]
    if not active:
        active = [order[0]]
    # Always include the argmax
    if order[0] not in active:
        active.append(order[0])
    
    # Build LP
    # Variables: y_0..y_{n-1}, t
    # Minimize t
    # Subject to:
    #   dx*grad_q(h)•y - t <= dx*grad_q(h)•h - scaled_q  (for each active q)
    #   sum(y) = T
    #   max(0, h-rho) <= y <= min(1, h+rho)
    #   t >= 0
    
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0  # minimize t
    
    A_ub = []
    b_ub = []
    for q in active:
        g = grad_raw(h, q)
        row = np.zeros(n + 1)
        row[:n] = dx * g
        row[n] = -1.0
        rhs = dx * np.dot(g, h) - scaled[q]
        A_ub.append(row)
        b_ub.append(rhs)
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([T])
    
    bounds = []
    for i in range(n):
        lo = max(0.0, h[i] - rho)
        hi = min(1.0, h[i] + rho)
        bounds.append((lo, hi))
    bounds.append((0.0, max_val + 0.1))  # bound on t
    
    try:
        res = linprog(c_obj, A_ub=np.array(A_ub), b_ub=np.array(b_ub),
                      A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
    except Exception:
        return h, False
    
    if res.status != 0:
        return h, False
    
    y = res.x[:n]
    y = np.clip(y, 0.0, 1.0)
    
    # Check and repair
    y = project_box_sum(y, T)
    
    new_score = exact_score(y)
    if new_score < max_val - 1e-12:
        return y, True
    return h, False

def pair_transfer(h, max_iters=200, dx=None):
    """Pairwise transfer local search."""
    n = len(h)
    T = n / 2.0
    if dx is None:
        dx = 2.0 / n
    current = exact_score(h)
    h = h.copy()
    
    for it in range(max_iters):
        # Compute gradients of top active lags
        rc = raw_corr(h)
        scaled = rc * dx
        max_val = np.max(scaled)
        if max_val <= current - 1e-12:
            current = max_val
            h = h.copy()  # keep
        
        order = np.argsort(-scaled)
        gap = max(1e-6, 0.01 * max_val)
        active = [q for q in order if scaled[q] >= max_val - gap][:10]
        
        # Compute weighted gradient
        weights = np.exp((scaled[active] - max_val) / (0.001 * max_val + 1e-10))
        weights /= weights.sum()
        g_mix = np.zeros(n)
        for w, q in zip(weights, active):
            g_mix += w * grad_raw(h, q)
        
        # Find best pair transfer
        # Transfer from high gradient to low gradient
        # g[i] > 0 means increasing h[i] increases the objective
        # g[j] < 0 means increasing h[j] decreases the objective
        # So transfer from i (high h, high positive g) to j (low h, negative g)
        
        # Sort by gradient
        pos_idx = np.argsort(-g_mix)[:30]  # top 30 positive
        neg_idx = np.argsort(g_mix)[:30]   # bottom 30 negative
        
        best_improvement = 0
        best_move = None
        
        for i in pos_idx:
            if h[i] < 1e-6:
                continue
            for j in neg_idx:
                if i == j:
                    continue
                if h[j] > 1.0 - 1e-6:
                    continue
                # Try transferring delta from i to j
                max_delta = min(h[i], 1.0 - h[j])
                if max_delta < 1e-8:
                    continue
                # Try a few fractions
                for frac in [0.01, 0.05, 0.1, 0.3, 0.5, 0.8, 1.0]:
                    delta = frac * max_delta
                    if delta < 1e-10:
                        continue
                    h[i] -= delta
                    h[j] += delta
                    s = exact_score(h)
                    h[i] += delta
                    h[j] -= delta
                    if s < current - 1e-12:
                        improvement = current - s
                        if improvement > best_improvement:
                            best_improvement = improvement
                            best_move = (i, j, delta)
        
        if best_move is None:
            break
        
        i, j, delta = best_move
        h[i] -= delta
        h[j] += delta
        current = exact_score(h)
    
    return h, current

import numpy as np
import time
from scipy.optimize import linprog

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def raw_corr(h):
    return np.correlate(h, 1.0 - h, mode='full')

def grad_raw(h, q):
    n = len(h)
    k = q - (n - 1)
    g = np.zeros(n)
    if k >= 0:
        g[k:] += (1.0 - h[:n - k])
        g[:n - k] -= h[k:]
    else:
        a = -k
        g[:n - a] += (1.0 - h[a:])
        g[a:] -= h[:n - a]
    return g

def grad_matrix(h):
    n = len(h)
    m = 2 * n - 1
    G = np.zeros((m, n))
    for q in range(m):
        G[q] = grad_raw(h, q)
    return G

def project_box_sum(h, T):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < T:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    result = np.clip(h + alpha, 0.0, 1.0)
    err = T - np.sum(result)
    if abs(err) > 1e-10:
        free = (result > 1e-8) & (result < 1.0 - 1e-8)
        cnt = np.sum(free)
        if cnt > 0:
            result[free] += err / cnt
            result = np.clip(result, 0.0, 1.0)
            err2 = T - np.sum(result)
            if abs(err2) > 1e-10:
                result[free[:min(cnt, 100)]] += err2 / min(cnt, 100)
                result = np.clip(result, 0.0, 1.0)
    return result

        elapsed = time.time() - start_time
        if elapsed > time_limit - 30:
            break

            K = min(K, max_K)
            rc = raw_corr(h)
            scaled = rc * dx
            max_val = np.max(scaled)

            active = active[:K]
            if not active:

            if int(order[0]) not in active:
                active.append(int(order[0]))
            
            c_obj = np.zeros(n + 1)
            c_obj[n] = 1.0

            b_ub_list = []

def pair_polish(h, max_rounds=100, dx=None, start_time=None, time_limit=None):
    n = len(h)
    T = n / 2.0
    if dx is None:
        dx = 2.0 / n
    h = h.copy()
    current = exact_score(h)
    
    for rnd in range(max_rounds):
        if start_time is not None and time.time() - start_time > time_limit - 20:
            break

        max_val = np.max(scaled)
        if max_val < current - 1e-12:
            current = max_val

        gap = max(1e-6, 0.008 * max_val)

        if not active_qs:
            active_qs = [int(order[0])]
        
        G_act = np.array([grad_raw(h, q) for q in active_qs])
        w = np.exp((scaled[active_qs] - max_val) / (0.002 * max_val + 1e-12))
        w /= w.sum()
        g_mix = np.dot(w, G_act)
        
        pos_idx = np.argsort(-g_mix)[:25]
        neg_idx = np.argsort(g_mix)[:25]

        for i in pos_idx:
            if h[int(i)] < 1e-7:
                continue
            for j in neg_idx:
                j = int(j)
                if i == j:
                    continue
                if h[j] > 1.0 - 1e-7:
                    continue
                max_delta = min(h[int(i)], 1.0 - h[j])

                    if delta < 1e-10:

                    h[int(i)] -= delta
                    h[j] += delta
                    s = exact_score(h)
                    if s < current - 1e-11:
                        current = s
                        improved = True
                        break
                    else:
                        h[int(i)] += delta
                        h[j] -= delta
                if improved:
                    break
            if improved:
                break
        if not improved:
            break
    
    return h, current
