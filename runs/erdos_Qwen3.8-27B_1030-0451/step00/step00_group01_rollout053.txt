def project_sum_box(x, target_sum):
    n = len(x)

        else:

    lam = (lo + hi) / 2.0
    result = np.clip(x + lam, 0.0, 1.0)

    Z = np.sum(exps)

    def obj_func(x):

        return o

        _, g = lse_obj_and_grad(x, beta)
        return g

def project_sum_box(x, target_sum):
    n = len(x)
    x = np.asarray(x, dtype=np.float64).copy()
    lo, hi = -1.5, 1.5
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x + lam, 0.0, 1.0)
    diff = target_sum - np.sum(result)
    if abs(diff) > 1e-12:
        if diff > 0:
            mask = result < 0.999
        else:
            mask = result > 0.001
        if np.any(mask):
            result[mask] += diff / np.sum(mask)
            result = np.clip(result, 0.0, 1.0)

        k = idx - (n - 1)
        wt = wi * dx

        je1 = min(n, n + k)

        js2 = max(0, -k)
        je2 = min(n, n - k)
