def compute_c5(h):
    """Compute the C5 value for a feasible h."""

def is_feasible(h, tol=1e-8):
    n = len(h)
    target = n / 2.0
    if not np.all(np.isfinite(h)):
        return False
    if np.any(h < -tol) or np.any(h > 1 + tol):
        return False
    if abs(np.sum(h) - target) > tol:
        return False
    return True

def project_feasible(h, target_sum=None):
    """Project h onto {0 <= h <= 1, sum(h) = target_sum}."""
    n = len(h)
    if target_sum is None:
        target_sum = n / 2.0
    h = np.clip(h, 0.0, 1.0)
    
    # Binary search for the offset
    lo, hi = -2.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        projected = np.clip(h - mid, 0.0, 1.0)
        s = np.sum(projected)
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    projected = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    
    # Final adjustment: distribute remaining mass
    diff = target_sum - np.sum(projected)
    if abs(diff) < 1e-15:
        return projected
    # Adjust non-saturated entries
    mask = (projected > 1e-10) & (projected < 1 - 1e-10)
    if np.sum(mask) > 0:
        projected[mask] += diff / np.sum(mask)
        projected = np.clip(projected, 0.0, 1.0)
        # Might need another round
        diff2 = target_sum - np.sum(projected)
        if abs(diff2) > 1e-12 and np.sum(mask) > 0:
            projected[mask] += diff2 / np.sum(mask)
            projected = np.clip(projected, 0.0, 1.0)
    return projected

def smooth_obj_and_grad(h, tau):
    """Compute smooth approximation of C5 and its gradient."""
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1.0 - h, mode='full')
    F = corr * dx
    
    F_max = np.max(F)
    exp_F = np.exp((F - F_max) / tau)
    Z = np.sum(exp_F)
    w = exp_F / Z
    obj = F_max + tau * np.log(Z)

def local_search(h, budget_left, rng, max_iters=5000):
    """Local search via pairwise exchanges."""
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    cur_c5 = compute_c5(h)
    
    for it in range(max_iters):
        if time.time() > budget_left:
            break
        # Pick two random indices
        i, j = rng.integers(0, n, 2)
        if i == j:
            continue
        # Move mass from j to i (or try both directions)
        # Max transfer: min(h[j], 1-h[i]) for i gets, min(h[i], 1-h[j]) for j gets
        max_to_i = min(h[j], 1.0 - h[i])
        max_to_j = min(h[i], 1.0 - h[j])
        
        if max_to_i < 1e-10 and max_to_j < 1e-10:
            continue
        
        # Try several fractions of the max transfer
        best_new_c5 = cur_c5
        best_delta = 0
        best_dir = 0  # 1: to i, -1: to j
        
        fractions = [0.01, 0.05, 0.1, 0.3, 0.5, 0.8, 1.0]
        
        for f in fractions:
            # Try moving to i
            d = f * max_to_i
            if d > 1e-12:
                h[i] += d
                h[j] -= d
                new_c5 = compute_c5(h)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_delta = d
                    best_dir = 1
                h[i] -= d
                h[j] += d
            
            # Try moving to j
            d = f * max_to_j
            if d > 1e-12:
                h[i] -= d
                h[j] += d
                new_c5 = compute_c5(h)
                if new_c5 < best_new_c5:
                    best_new_c5 = new_c5
                    best_delta = d
                    best_dir = -1
                h[i] += d
                h[j] -= d
        
        if best_delta < 1e-12:
            continue
        
        if best_dir == 1:
            h[i] += best_delta
            h[j] -= best_delta
        else:
            h[i] -= best_delta
            h[j] += best_delta
        
        cur_c5 = best_new_c5
    
    return h, cur_c5

def smooth_optimize(h, tau, maxiter=200):
    """Optimize using SLSQP on the smooth objective."""
    n = len(h)
    target_sum = n / 2.0
    
    def obj(x):
        o, _ = smooth_obj_and_grad(x, tau)
        return o
    
    def grad(x):
        _, g = smooth_obj_and_grad(x, tau)
        return g
    
    constraints = [
        {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum,

    ]
    bounds = [(0.0, 1.0)] * n
    
    result = minimize(obj, h, jac=grad, method='SLSQP',
                      bounds=bounds, constraints=constraints,
                      options={'maxiter': maxiter, 'ftol': 1e-10})
    
    if result.success or result.fun < obj(h):
        h_new = project_feasible(result.x, target_sum)
        return h_new
    return h

def transfer_dimension(h_from, n_to):
    """Transfer h from one dimension to another via interpolation."""
    n_from = len(h_from)
    # Midpoint rule: x_i = (i + 0.5) * 2/n_from for i in [0, n_from)
    x_from = (np.arange(n_from) + 0.5) * 2.0 / n_from
    x_to = (np.arange(n_to) + 0.5) * 2.0 / n_to

    h_to = np.clip(h_to, 0.0, 1.0)

    return h_to
