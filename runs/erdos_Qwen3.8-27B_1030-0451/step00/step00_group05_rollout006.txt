def project_box_simplex(x, target_sum):
    """Project x onto {h: 0<=h_i<=1, sum(h)=target_sum}."""
    n = len(x)
    # Bisection on lambda
    # h_i = clip(x_i - lam, 0, 1)
    # sum(h) = target_sum
    
    def sum_at_lam(lam):
        return np.sum(np.clip(x - lam, 0.0, 1.0))
    
    # Binary search
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0

        s = sum_at_lam(mid)
        if s > target_sum:
            lo = mid
        else:
