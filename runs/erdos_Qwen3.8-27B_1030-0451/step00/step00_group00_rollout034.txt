def compute_c(h):
    """Compute all correlation values c[k'] for k' = -(n-1) to n-1"""
    n = len(h)
    result = np.correlate(h, 1.0 - h, mode='full')
    return result  # length 2n-1, index k+n-1 corresponds to lag k

def compute_gradient(h, p):
    """
    Compute gradient of sum_k' p[k'] * c[k'] w.r.t. h
    where p is the softmax weights (length 2n-1)
    """
    n = len(h)
    grad = np.zeros(n)
    # p is indexed 0 to 2n-2, corresponding to lags -(n-1) to n-1
    for k_idx in range(2*n-1):
        k = k_idx - (n-1)  # actual lag
        if p[k_idx] < 1e-15:
            continue
        if k >= 0:
            # c[k] = sum_{i=0}^{n-1-k} h[i+k]*(1-h[i])
            # d/dh[j]: (1-h[j-k]) if k <= j <= n-1, minus h[j+k] if 0 <= j <= n-1-k
            valid_first = np.arange(k, n)  # j values where h[j] is the first factor
            valid_second = np.arange(0, n-k)  # j values where h[j] is in (1-h[j])
            grad[valid_first] += p[k_idx] * (1.0 - h[valid_first - k])
            grad[valid_second] -= p[k_idx] * h[valid_second + k]
        else:
            # k < 0, let j_abs = -k
            j_abs = -k
            # c[-j_abs] = sum_{i=j_abs}^{n-1} h[i-j_abs]*(1-h[i])
            # d/dh[j]: (1-h[j+j_abs]) if 0 <= j <= n-1-j_abs, minus h[j-j_abs] if j_abs <= j <= n-1
            valid_first = np.arange(0, n-j_abs)  # j values where h[j] is the first factor
            valid_second = np.arange(j_abs, n)  # j values where h[j] is in (1-h[j])
            grad[valid_first] += p[k_idx] * (1.0 - h[valid_first + j_abs])
            grad[valid_second] -= p[k_idx] * h[valid_second - j_abs]
    return grad

import numpy as np
import time

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x) = target_sum, 0 <= x <= 1}"""
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def compute_c5(h):
    """Compute C5 value"""
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return np.max(c) * dx

def compute_gradient_active(h, active_indices):
    """Compute average gradient over active lags"""
    n = len(h)
    grad = np.zeros(n)
    for k_idx in active_indices:
        L = k_idx - (n - 1)
        if L >= 0:
            g1 = np.zeros(n)
            g1[L:] = 1.0 - h[:n-L]
            g2 = np.zeros(n)
            g2[:n-L] = h[L:]
            grad += g1 - g2
        else:
            M = -L
            g1 = np.zeros(n)
            g1[:n-M] = 1.0 - h[M:]
            g2 = np.zeros(n)
            g2[M:] = h[:n-M]
            grad += g1 - g2
    grad /= len(active_indices)
    return grad

def subgradient_descent(h, max_time, target_sum, initial_lr=0.1):
    """Run subgradient descent"""
    n = len(h)
    h = h.copy()
    lr = initial_lr
    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
    no_improve = 0
    start = time.time()
    
    while time.time() - start < max_time:
        c = np.correlate(h, 1.0 - h, mode='full')
        c_max = np.max(c)
        
        if c_max < best_c:
            best_c = c_max
            best_h = h.copy()
            best_c5 = c_max * 2.0 / n
            no_improve = 0
        else:
            no_improve += 1
        
        # Active set: lags within 2% of max
        threshold = c_max * 0.98
        active = np.where(c >= threshold)[0]
        
        # Compute gradient
        grad = compute_gradient_active(h, active)
        
        # Step
        h_new = h - lr * grad
        h_new = project_feasible(h_new, target_sum)
        
        # Check improvement
        c_new = np.max(np.correlate(h_new, 1.0 - h_new, mode='full'))
        if c_new < c_max:
            h = h_new
            lr = min(lr * 1.05, 2.0)
        else:
            lr *= 0.7
            if lr < 0.001:
                # Restart
                h = best_h + 0.03 * np.random.randn(n)
                h = project_feasible(h, target_sum)
                lr = 0.1
            no_improve += 1
        
        if no_improve > 200:
            h = best_h + 0.05 * np.random.randn(n)
            h = project_feasible(h, target_sum)
            lr = 0.1
            no_improve = 0
    
    return best_h, best_c5

def local_search_binary(h_binary, target_sum, max_flips=1000):
    """Local search: try flipping pairs of bits to maintain sum constraint"""
    n = len(h_binary)
    h = h_binary.copy().astype(float)
    best_c = np.max(np.correlate(h, 1.0 - h, mode='full'))
    
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    improved = True
    while improved and max_flips > 0:
        improved = False
        max_flips -= 1
        
        # Try random pairs: move a 1 to a 0 position
        for _ in range(200):
            i = np.random.choice(ones)
            j = np.random.choice(zeros)
            if i == j:
                continue
            h[i] = 0.0
            h[j] = 1.0
            c_new = np.max(np.correlate(h, 1.0 - h, mode='full'))
            if c_new < best_c:
                best_c = c_new
                # Update ones and zeros
                ones = ones[ones != i]
                zeros = zeros[zeros != j]
                ones = np.append(ones, j)
                zeros = np.append(zeros, i)
                improved = True
            else:
                h[i] = 1.0
                h[j] = 0.0
    
    return h, best_c

def project_feasible(h, target_sum):
    lo = -2.0
    hi = 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_value(h):
    n = len(h)
    return np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n)

def soft_gradient(h, tau=0.1):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    c_n = c / tau
    c_n -= np.max(c_n)
    p = np.exp(c_n)
    p /= p.sum()
    
    # Compute gradient: sum_k p[k] * dc_k/dh
    grad = np.zeros(n)
    # For each lag index k_idx (0 to 2n-2), Lag L = k_idx - (n-1)
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-10:

        L = k_idx - (n - 1)
        if L >= 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            M = -L
            grad[:n - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:n - M]
    return grad

import numpy as np
import time

def project_feasible(h, target_sum):
    """Project h onto {x: sum(x)=target_sum, 0<=x<=1} via bisection on lambda."""
    lo, hi = -2.0, 2.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        proj = np.clip(h - mid, 0.0, 1.0)
        if np.sum(proj) > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - 0.5 * (lo + hi), 0.0, 1.0)

def c5_value(h):
    n = len(h)
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * (2.0 / n))

def soft_gradient(h, tau):
    """Gradient of tau*log(sum exp(c[k]/tau)) w.r.t. h."""
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cn = c / tau
    cn -= np.max(cn)
    p = np.exp(cn)
    p /= p.sum()
    grad = np.zeros(n)
    for k_idx in range(2 * n - 1):
        w = p[k_idx]
        if w < 1e-12:
            continue
        L = k_idx - (n - 1)
        if L >= 0:
            grad[L:] += w * (1.0 - h[:n - L])
            grad[:n - L] -= w * h[L:]
        else:
            M = -L
            grad[:n - M] += w * (1.0 - h[M:])
            grad[M:] -= w * h[:n - M]
    return grad

def optimize_continuous(h0, max_time, target_sum, seed=42):
    """Soft-gradient descent with annealing and restarts."""
    rng = np.random.RandomState(seed)
    n = len(h0)
    h = h0.copy()
    best_h = h.copy()
    best_c5 = c5_value(h)
    lr = 1.0
    tau = 1.0
    stall = 0
    t_start = time.time()
    iter_count = 0

    while time.time() - t_start < max_time:
        grad = soft_gradient(h, tau)
        h_new = project_feasible(h - lr * grad, target_sum)
        c5_new = c5_value(h_new)

        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new
            lr = min(lr * 1.05, 10.0)
            stall = 0
        else:
            lr *= 0.75
            stall += 1

        tau = max(tau * 0.995, 0.02)

        if stall > 150:
            h = best_h + 0.08 * rng.randn(n)
            h = project_feasible(h, target_sum)
            lr = 1.0
            tau = max(tau, 0.3)
            stall = 0

        iter_count += 1
        if iter_count % 5000 == 0 and time.time() - t_start < max_time:
            pass  # checkpoint (no-op, just for timing awareness)

    return best_h, best_c5

def binary_local_search(h_bin, max_time, seed=0):
    """Local search on binary h: swap a 1 with a 0 to reduce max correlation."""
    rng = np.random.RandomState(seed)
    n = len(h_bin)
    h = h_bin.copy().astype(float)
    best_c = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
    ones = list(np.where(h == 1.0)[0])
    zeros = list(np.where(h == 0.0)[0])
    t_start = time.time()
    no_improve = 0

    while time.time() - t_start < max_time and no_improve < 3000:
        improved = False
        batch = min(1000, len(ones) * len(zeros))
        for _ in range(batch):
            i = ones[rng.randint(len(ones))]
            j = zeros[rng.randint(len(zeros))]
            h[i] = 0.0
            h[j] = 1.0
            c_new = float(np.max(np.correlate(h, 1.0 - h, mode='full')))
            if c_new < best_c - 1e-12:
                best_c = c_new
                ones.remove(i)
                zeros.remove(j)
                ones.append(j)
                zeros.append(i)
                improved = True
                no_improve = 0
            else:
                h[i] = 1.0
                h[j] = 0.0
                no_improve += 1
        if not improved:
            no_improve += 500

    return h, best_c * 2.0 / n

def make_binary_close(h_frac, target_sum, seed=0):
    """Convert fractional h to binary with exact sum constraint."""
    rng = np.random.RandomState(seed)
    n = len(h_frac)
    h_bin = (h_frac > 0.5).astype(float)
    diff = target_sum - np.sum(h_bin)
    if diff > 0:
        n_flip = int(np.ceil(diff))
        zeros_idx = np.where(h_bin == 0.0)[0]
        if len(zeros_idx) >= n_flip:
            # Flip the ones with highest fractional value
            scores = h_frac[zeros_idx]
            flip = zeros_idx[np.argsort(scores)[::-1][:n_flip]]
            h_bin[flip] = 1.0
    elif diff < 0:
        n_flip = int(np.ceil(-diff))
        ones_idx = np.where(h_bin == 1.0)[0]
        if len(ones_idx) >= n_flip:
            scores = h_frac[ones_idx]
            flip = ones_idx[np.argsort(scores)[:n_flip]]
            h_bin[flip] = 0.0
    return h_bin

        ttl = min(budget_s - _elapsed() - 30, 200)
        if ttl > 5 and np.sum(h_bin) == ts:
            h_b, c5_b = binary_local_search(h_bin, ttl, seed=seed + 1)
