import numpy as np

def project_feasible(g, n):
    """Project g onto {h: 0 <= h[i] <= 1, sum(h) = n/2}."""
    target = n / 2.0
    lo, hi = -1.0, 1.0
    
    # First, find the range of valid lambda
    # If sum(clamp(g - λ, 0, 1)) = target
    # λ is decreasing in the sum
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        h = np.clip(g - mid, 0.0, 1.0)
        s = h.sum()
        if s > target:
            lo = mid  # need larger λ to reduce sum
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    h = np.clip(g - lam, 0.0, 1.0)
    
    # Final adjustment: distribute remaining budget
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        # Adjust elements that are strictly inside (0,1)
        interior = (h > 1e-10) & (h < 1.0 - 1e-10)
        if interior.sum() > 0:
            adj = diff / interior.sum()
            h[interior] += adj
            h = np.clip(h, 0.0, 1.0)
        else:
            # Edge case: distribute among all
            h += diff / n
            h = np.clip(h, 0.0, 1.0)
    
    return h

    """Compute weights for smooth max approximation.
    Returns weights p[i] proportional to exp(C[i]/tau)."""
    # Use log-sum-exp trick for numerical stability
    C_max = C.max()
    exp_C = np.exp((C - C_max) / tau)
    p = exp_C / exp_C.sum()
    return p

import time

def project_feasible(g, n):
    """Project g onto {h: 0<=h[i]<=1, sum(h)=n/2}."""
    target = n * 0.5
    lo = g.min() - 2.0
    hi = g.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(g - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(g - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - h.sum()
    if abs(diff) > 1e-12:
        mask = (h > 1e-9) & (h < 1.0 - 1e-9)
        cnt = mask.sum()
        if cnt > 0:
            h[mask] += diff / cnt
            np.clip(h, 0.0, 1.0, out=h)
    return h

def get_corrs(h):
    """Compute all 2n-1 correlation values scaled by dx."""
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def get_c5(h):
    """Compute C5 = max of correlation values."""
    return get_corrs(h).max()

def grad_corrs(h, C, tau):
    """
    Compute gradient of smooth max: tau * log(sum(exp(C/tau)))
    with respect to h.
    """
    n = len(h)
    dx = 2.0 / n
    c0 = n - 1
    # Softmax weights
    shifted = (C - C.max()) / tau
    p = np.exp(shifted)
    psum = p.sum()
    if psum < 1e-300:
        return np.zeros(n)
    p /= psum
    
    grad = np.zeros(n)
    for i in range(2 * n - 1):
        if p[i] < 1e-14:
            continue
        R = i - c0
        if R == 0:
            grad += p[i] * dx * (1.0 - 2.0 * h)
        elif R > 0:
            m = n - R
            grad[:m] -= p[i] * dx * h[R:]
            grad[R:] += p[i] * dx * (1.0 - h[:m])
        else:
            Ra = -R
            m = n - Ra
            grad[:m] += p[i] * dx * (1.0 -
