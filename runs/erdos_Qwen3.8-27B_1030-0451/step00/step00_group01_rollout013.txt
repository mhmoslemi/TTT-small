        else:

def _latent_nm(h0, dim=30, scale=0.5, max_iter=2000, deadline=None):
    """Nelder-Mead in latent Fourier space."""
    n = len(h0)
    x = np.arange(n)
    B = np.zeros((n, dim))
    for k in range(dim):
        freq = k + 1
        if k % 2 == 0:
            B[:, k] = np.cos(2 * np.pi * freq * x / n)
        else:
            B[:, k] = np.sin(2 * np.pi * freq * x / n)
    
    def obj(b):
        z = h0 + scale * (B @ b)
        h = _project_feasible(z, n)
        return _exact_score(h)
    
    b0 = np.zeros(dim)
    res = minimize(obj, b0, method='Nelder-Mead', 
                   options={'maxiter': max_iter, 'xatol': 1e-8, 'fatol': 1e-8})
    
    h_new = _project_feasible(h0 + scale * (B @ res.x), n)
    return h_new, _exact_score(h_new)
