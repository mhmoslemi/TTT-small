from scipy.optimize import minimize
import time

def project_feasible(h):
    """Exact Euclidean projection onto {x: 0 <= x <= 1, sum(x) = n/2}"""
    m = len(h)
    target = m / 2.0
    
    # Bisection on lambda
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)

def compute_c5(h):
    m = len(h)
    dx = 2.0 / m
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr)) * dx

def compute_all_overlaps(h):
    m = len(h)
    dx = 2.0 / m
    corr = np.correlate(h, 1.0 - h, mode="full")
    return corr * dx

def gradient_of_overlap(h, k):
    """Gradient of overlap at full-correlation index k with respect to h."""
    m = len(h)
    dx = 2.0 / m
    L = k - (m - 1)  # lag
    grad = np.zeros(m)
    # Term 1: (1-h[i-L]) for valid i-L
    # h[i] appears as h[n+L] when n = i-L, valid if 0 <= i-L < m
    # Contribution: (1-h[i-L])
    valid1 = np.arange(m) - L
    mask1 = (valid1 >= 0) & (valid1 < m)
    grad[mask1] += (1.0 - h[valid1[mask1]])
    # Term 2: -h[i+L] for valid i+L
    valid2 = np.arange(m) + L
    mask2 = (valid2 >= 0) & (valid2 < m)
    grad[mask2] -= h[valid2[mask2]]
    return dx * grad

    """Smooth approximation to max using log-sum-exp."""
    overlaps = compute_all_overlaps(h)
    max_val = np.max(overlaps)
    # Shift for numerical stability
    shifted = overlaps - max_val
    return max_val + np.log(np.sum(np.exp(beta * shifted))) / beta

    m = len(h)
    overlaps = compute_all_overlaps(h)
    max_val = np.max(overlaps)
    shifted = overlaps - max_val
    weights = np.exp(beta * shifted)

def project_feasible(h):
