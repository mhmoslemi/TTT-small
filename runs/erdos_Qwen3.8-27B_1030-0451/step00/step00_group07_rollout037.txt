import numpy as np
import time

def evaluate_c5(h, n):
    """Compute C5 for a given h vector."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    return np.max(corr) * dx

def project_onto_constraint(h, target_sum):
    """Project h onto {0≤x≤1, sum(x)=target_sum} using bisection on lambda."""
    h = np.clip(h, 0.0, 1.0)
    # Bisection on lambda
    lo, hi = -h.max() - 1, -h.min() + 1  # lambda range
    # Actually, h_proj[i] = clip(h[i] - lambda, 0, 1)
    # We need sum(h_proj) = target_sum
    # As lambda increases, sum decreases
    lo, hi = -2.0, 2.0
    for _ in range(60):
        mid = (lo + hi) / 2
        h_proj = np.clip(h - mid, 0.0, 1.0)
        s = h_proj.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    h_proj = np.clip(h - (lo+hi)/2, 0.0, 1.0)
    return h_proj

def compute_gradient_max_c5(h, n):
    """Compute subgradient of max correlation value."""
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr = np.correlate(h, one_minus_h, mode='full')
    max_idx = np.argmax(corr)
    lag = max_idx - (n - 1)
    
    # Gradient of c_lag with respect to h
    # c_lag = sum_{valid i} h[i] * (1 - h[i+lag])
    grad = np.zeros(n)
    if lag >= 0:
        # i ranges from 0 to n-1-lag
        for i in range(n - lag):
            grad[i] += (1 - h[i + lag])
            grad[i + lag] += (-h[i])
    else:
        # lag < 0, let m = -lag
        m = -lag
        # c = sum_{i=m}^{n-1} h[i] * (1 - h[i-m])
        for i in range(m, n):
            grad[i] += (1 - h[i - m])
            grad[i - m] += (-h[i])

def projected_gradient_descent(h, n, max_iter, initial_lr):
    target_sum = n / 2.0
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n)
    
    for it in range(max_iter):
        grad = compute_gradient_max_c5(h, n)
        # Gradient descent step
        h_new = h - initial_lr * grad
        # Project
        h_new = project_onto_constraint(h_new, target_sum)
        # Check improvement
        c5 = evaluate_c5(h_new, n)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        h = h_new
        # Decay learning rate
        initial_lr *= 0.999
    
    return best_h, best_c5

The soft max of values v_0, ..., v_{2n-2} is:
(1/β) * log(∑_j exp(β * v_j))

Its gradient w.r.t. h is:

= ∑_j w_j * ∇v_j
where w_j = exp(β*v_j) / ∑_k exp(β*v_k) (softmax weights)
