import numpy as np
from scipy.optimize import minimize
from scipy.interpolate import interp1d
import time

def project_to_feasible(y, target_sum):
    """Project y onto {h: 0<=h[i]<=1, sum(h)=target_sum} using bisection on alpha shift."""
    n = len(y)
    y = np.asarray(y, dtype=np.float64)
    # Bisection on alpha: find alpha such that sum(clip(y+alpha, 0, 1)) = target_sum
    lo, hi = -y.max() - 1, 1 - y.min() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        s = np.clip(y + mid, 0, 1).sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    h = np.clip(y + (lo+hi)/2, 0, 1)
    # Fix residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            # Add to positions with most room
            room = 1 - h
            order = np.argsort(-room)
            for idx in order:
                step = min(room[idx], residual)
                h[idx] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            # Remove from positions with most value
            order = np.argsort(-h)
            for idx in order:
                step = min(h[idx], -residual)
                h[idx] -= step
                residual += step
                if abs(residual) < 1e-14:
                    break
    return h

def evaluate_c5(h):
    """Compute C5 = max correlation * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def compute_correlations(h):
    """Compute all correlation values * dx."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def gradient_c(h, lag_idx):
    """Gradient of c_k (times dx) with respect to h, where lag_idx is the index in full correlation."""
    n = len(h)
    dx = 2.0 / n
    k = lag_idx - (n - 1)  # actual lag
    D = np.zeros(n)
    one_minus_h = 1.0 - h
    if k >= 0:
        L = n - k
        # h[n+k] term: contributes (1-h[n]) for n in [0, L)
        # i.e., for i in [k, n): D[i] += (1-h[i-k])
        D[k:] += one_minus_h[:L]
        # (1-h[n]) term: contributes -h[n+k] for n in [0, L)
        # i.e., for i in [0, L): D[i] += -h[i+k]
        D[:L] -= h[k:]
    else:
        r = -k
        L = n - r
        # h[n-r] term: contributes (1-h[n]) for n in [r, n)
        # i.e., for i in [0, L): D[i] += (1-h[i+r])
        D[:L] += one_minus_h[r:]
        # (1-h[n]) term: contributes -h[n-r] for n in [r, n)
        # i.e., for i in [r, n): D[i] += -h[i-r]
        D[r:] -= h[:L]
    return D * dx

def lse_smoothed(corr, tau):
    """Log-sum-exp smoothed max of corr values."""
    cmax = np.max(corr)
    if tau < 1e-15:
        return cmax
    shifted = (corr - cmax) / tau
    # Clamp to avoid overflow
    shifted = np.clip(shifted, -500, 500)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return cmax + tau * log_sum

def projected_descent(h0, max_iter=2000, tau_init=0.005, tau_min=0.0005, 
                       n_active=30, eta_init=0.05, time_limit=None, t_start=None):
    """Projected gradient descent with LSE smoothing."""
    n = len(h0)
    h = h0.copy()
    best_h = h0.copy()
    best_c5 = evaluate_c5(h0)
    
    tau = tau_init
    eta = eta_init
    stall_count = 0
    
    for it in range(max_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        corr = compute_correlations(h)
        c5_current = float(np.max(corr))
        
        if c5_current < best_c5 - 1e-12:
            best_c5 = c5_current
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Select active lags (top n_active by value)
        if len(corr) <= n_active:
            active = np.arange(len(corr))
        else:
            active = np.argsort(corr)[-n_active:]
        
        corr_active = corr[active]
        
        # Compute smoothed objective and gradient
        cmax_active = np.max(corr_active)
        shifted = np.clip((corr_active - cmax_active) / tau, -500, 0)
        weights = np.exp(shifted)
        wsum = weights.sum()
        if wsum < 1e-30:
            weights = np.ones(len(active)) / len(active)
        else:
            weights /= wsum
        
        # Gradient of smoothed objective
        grad = np.zeros(n)
        for w, idx in zip(weights, active):
            if abs(w) > 1e-10:
                grad += w * gradient_c(h, int(idx))
        
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            break
        
        # Line search
        accepted = False
        for scale in [eta, eta*0.5, eta*0.25, eta*0.1, eta*0.05]:
            if scale < 1e-10:
                break
            y = h - scale * grad / gnorm * np.max(corr)
            h_trial = project_to_feasible(y, n * 0.5)
            corr_trial = compute_correlations(h_trial)
            c5_trial = float(np.max(corr_trial))
            if c5_trial < c5_current - 1e-14:
                h = h_trial
                accepted = True
                break
        
        if not accepted:
            if stall_count > 50:
                tau = max(tau_min, tau * 0.5)
                eta = max(0.001, eta * 0.7)
                if tau <= tau_min and eta <= 0.002:
                    break
            elif stall_count > 20:
                tau = max(tau_min, tau * 0.7)
        else:
            if stall_count == 0:
                eta = min(0.5, eta * 1.1)
    
    return best_h, best_c5

def slsqp_polish(h0, n_active=40, max_iter=200):
    """Polish with SLSQP epigraph formulation."""
    n = len(h0)
    target = n * 0.5
    dx = 2.0 / n
    
    # Get active lags
    corr = compute_correlations(h0)
    if len(corr) <= n_active:
        active = np.arange(len(corr))
    else:
        active = np.argsort(corr)[-n_active:]
    
    c5_init = evaluate_c5(h0)
    t_init = c5_init + 0.001
    
    def objective(x):
        return x[-1]  # t
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def eq_constraint(x):
        return np.sum(x[:n]) - target
    
    def eq_jacobian(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    def ineq_constraints(x):
        h = x[:n]
        t = x[-1]
        # t - c_k >= 0 for each active lag
        corr = np.correlate(h, 1.0 - h, mode='full') * dx
        return t - corr[active]
    
    def ineq_jacobian(x):
        h = x[:n]
        # Build Jacobian: shape (len(active), n+1)
        # Row i: d(t - c_ki)/dt = 1, d(t - c_ki)/dh = -d(c_ki)/dh
        jac = np.zeros((len(active), n + 1))
        jac[:, -1] = 1.0
        for i, idx in enumerate(active):
            jac[i, :n] = -gradient_c(h, int(idx))
        return jac
    
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0, 1)] * n + [(0, 2)]
    
    try:
        result = minimize(objective, x0, method='SLSQP',
                          jac=objective_grad,
                          bounds=bounds,
                          constraints=[
                              {'type': 'eq', 'fun': eq_constraint, 'jac': eq_jacobian},
                              {'type': 'ineq', 'fun': ineq_constraints, 'jac': ineq_jacobian}
                          ],
                          options={'maxiter': max_iter, 'ftol': 1e-10})
        
        if result.x is not None:
            h_sol = project_to_feasible(result.x[:n], target)
            c5_sol = evaluate_c5(h_sol)
            if c5_sol < c5_init:
                return h_sol, c5_sol
    except Exception:
        pass
    
    return h0, c5_init

def interpolate_to_resolution(h, n_new):
    """Interpolate h from its current resolution to n_new."""
    n = len(h)
    if n == n_new:
        return h.copy()
    x_old = np.arange(n) * (2.0 / n)
    x_new = np.arange(n_new) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h, period=2.0)  # No, not periodic
    # Actually, for linear interpolation on [0,2]:
    h_new = np.interp(x_new, x_old, h)
    h_new = project_to_feasible(h_new, n_new * 0.5)
    return h_new

def add_noise(h, rng, sigma=0.05):
    """Add small Gaussian noise and project."""
    n = len(h)
    noisy = h + rng.normal(0, sigma, n)
    return project_to_feasible(noisy, n * 0.5)

        if r.x is not None:
            hs = project(r.x[:n], target)
            cs = c5_eval(hs)
            if cs < c0 - 1e-14:
                return hs, cs
    except Exception:
        pass
    return h0, c0


def proj_grad(h0, max_iter=1500, n_act=25, tau=0.008, time_limit=30.0, t0=None):
    """Projected gradient with LSE smoothing, accepting on exact C5 improvement."""
    n = len(h0)
    target = n / 2.0
    h = h0.copy()
    best_h = h0.copy()
    best_c = c5_eval(h0)
    alpha = 0.015
    stall = 0
    tau_cur = tau

    for it in range(max_iter):
        if t0 is not None and time.time() - t0 > time_limit:
            break
        cv = corr_vals(h)
        c = float(cv.max())
        if c < best_c - 1e-13:
            best_c = c
            best_h = h.copy()
            stall = 0
        else:
            stall += 1

        if len(cv) <= n_act:
            act = np.arange(len(cv))
        else:
            act = np.argsort(cv)[-n_act:]

        cv_a = cv[act]
        cmax = cv_a.max()
        z = np.clip((cv_a - cmax) / tau_cur, -80.0, 0.0)
        w = np.exp(z)
        ws = w.sum()
        if ws < 1e-30:
            w = np.ones(len(act)) / len(act)
        else:
            w /= ws

        g = np.zeros(n)
        for wi, mi in zip(w, act):
            if wi > 1e-9:
                g += wi * grad_lag(h, int(mi))

        gn = np.linalg.norm(g)
        if gn < 1e-14:
            break

        accepted = False
        for a in [alpha, alpha * 2.0, alpha * 0.5, alpha * 0.25, alpha * 0.1]:
            if a < 1e-6:
                break
            ht = project(h - a * g / gn, target)
            ct = c5_eval(ht)
            if ct < c - 1e-14:
                h = ht
                accepted = True
                break

        if not accepted:
            stall += 1
            if stall > 40:
                alpha *= 0.5
                tau_cur = max(0.002, tau_cur * 0.7)
                if alpha < 1e-5:
                    break
            elif stall > 15:
                tau_cur = max(0.002, tau_cur * 0.8)
        elif stall == 0:
            alpha = min(0.3, alpha * 1.05)

    return best_h, best_c


def interp_res(h, n_new):
    n = len(h)
    if n == n_new:
        return h.copy()
    xo = np.arange(n) * 2.0 / n
    xn = np.arange(n_new) * 2.0 / n_new
    hn = np.interp(xn, xo, h)
    return project(hn, n_new / 2.0)


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    hard_end = t0 + budget_s * 0.92
    rng = np.random.RandomState(seed)

    # Get parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64).flatten()
    except Exception:
        parent = np.full(64, 0.5, dtype=np.float64)

    np_ = len(parent)
    parent = project(parent, np_ / 2.0)

    # Evaluate parent and symmetries
    best_h = parent.copy()
    best_c = c5_eval(parent)
    variants = [parent[::-1].copy(), (1.0 - parent).copy(), (1.0 - parent[::-1]).copy()]
    for v in variants:
        vh = project(v, np_ / 2.0)
        vc = c5_eval(vh)
        if vc < best_c:
            best_c = vc
            best_h = vh

    # Resolution list (prefer smaller)
    res_list = [np_]
    for r in [80, 96, 128, 160, 192, 256]:
        if r > np_:
            res_list.append(r)

    for nr in res_list:
        if time.time() > hard_end - 40:
            break
        remaining = hard_end - time.time()

        # Interpolate current best to nr
        if nr == np_:
            hr = best_h.copy()
        else:
            hr = interp_res(best_h, nr)
            hrp = interp_res(parent, nr)
            if c5_eval(hrp) < c5_eval(hr):
                hr = hrp

        cr = c5_eval(hr)
        if cr < best_c - 1e-12:
            best_c = cr
            best_h = hr.copy()

        # Projected gradient phase
        remaining = hard_end - time.time()
        if remaining > 15:
            tl = min(12.0, remaining * 0.35)
            h1, c1 = proj_grad(hr, max_iter=int(tl * 80), n_act=min(30, nr // 2 + 1),
                               time_limit=tl, t0=time.time())
            if c1 < best_c - 1e-12:
                best_c = c1
                best_h = h1.copy()
            hr = h1  # continue from here

        # SLSQP pass 1
        remaining = hard_end - time.time()
        if remaining > 12:
            ha = min(35, nr)
            hs, cs = slsqp_solve(hr, n_act=ha, maxiter=80)
            if cs < best_c - 1e-12:
                best_c = cs
                best_h = hs.copy()
                hr = hs

        # SLSQP pass 2 (more constraints)
        remaining = hard_end - time.time()
        if remaining > 10:
            ha2 = min(55, int(nr * 1.5))
            hs2, cs2 = slsqp_solve(hr, n_act=ha2, maxiter=100)
            if cs2 < best_c - 1e-12:
                best_c = cs2
                best_h = hs2.copy()
                hr = hs2

        # SLSQP pass 3 (even more if time allows)
        remaining = hard_end - time.time()
        if remaining > 8 and nr <= 192:
            ha3 = min(70, int(nr * 2))
            hs3, cs3 = slsqp_solve(hr, n_act=ha3, maxiter=120)
            if cs3 < best_c - 1e-12:
                best_c = cs3
                best_h = hs3.copy()

        # Noisy perturbations + quick SLSQP
        remaining = hard_end - time.time()
        if remaining > 10:
            for sig in [0.02, 0.05, 0.1]:
                if time.time() > hard_end - 8:
                    break
                hn = project(best_h + rng.normal(0, sig, len(best_h)), len(best_h) / 2.0)
                cn = c5_eval(hn)
                if cn < best_c - 1e-12:
                    best_c = cn
                    best_h = hn.copy()
                    hr = hn
                if time.time() > hard_end - 6:
                    break
                hsn, csn = slsqp_solve(hn, n_act=min(30, len(hn)), maxiter=60)
                if csn < best_c - 1e-12:
                    best_c = csn
                    best_h = hsn.copy()
                    hr = hsn

    # Final polish on best
    remaining = hard_end - time.time()
    if remaining > 5:
        hf1, cf1 = slsqp_solve(best_h, n_act=min(45, len(best_h)), maxiter=120)
        if cf1 < best_c - 1e-12:
            best_c = cf1
            best_h = hf1.copy()
        remaining = hard_end - time.time()
        if remaining > 4:
            hf2, cf2 = slsqp_solve(best_h, n_act=min(60, len(best_h) * 2), maxiter=100)
            if cf2 < best_c - 1e-12:
                best_c = cf2
                best_h = hf2.copy()

    # Final verification
    best_h = project(best_h, len(best_h) / 2.0)
    best_c = c5_eval(best_h)

    assert np.all(np.isfinite(best_h))
    assert len(best_h) > 0

    return (best_h.astype(np.float64), float(best_c), len(best_h))
