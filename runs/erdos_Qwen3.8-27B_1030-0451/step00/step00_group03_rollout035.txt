    n = len(y)
    y = np.clip(np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0), 0.0, 1.0)

    hi = 1.0 - y.min() + 0.1

def make_mixed_pairs(h, rng, n, n_pairs=None):
    if n_pairs is None:
        n_pairs = min(n * n // 3, 3000)
    pairs = []
    order = np.argsort(h)
    n_half = n // 2
    for _ in range(min(n // 2, n_pairs // 3)):
        i = int(order[rng.integers(n_half)])
        j = int(order[n - 1 - rng.integers(n_half)])
        if i != j:
            pairs.append((min(i, j), max(i, j)))
    for _ in range(min(n, n_pairs // 2)):
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j:
            pairs.append((min(i, j), max(i, j)))
    while len(pairs) < n_pairs:
        i = int(rng.integers(n))
        j = int(rng.integers(n))
        if i != j:
            pairs.append((min(i, j), max(i, j)))
    return pairs[:n_pairs]

def try_pair(h, i, j, n_cand, rng):
    hi_v = h[i]
    hj_v = h[j]
    d_min = max(-hi_v, hj_v - 1.0)
    d_max = min(1.0 - hi_v, hj_v)
    width = d_max - d_min
    if width < 1e-10:
        return 0.0, 0.0  # no change, same c
    fracs = np.linspace(0.0, 1.0, n_cand + 2)[1:-1]
    fracs = np.concatenate([fracs, [rng.random()] * 2])
    best_d = 0.0
    best_c = exact_c5(h)
    old_i, old_j = h[i], h[j]
    for frac in fracs:
        d = d_min + float(frac) * width
        h[i] = hi_v + d
        h[j] = hj_v - d
        c_try = exact_c5(h)
        if c_try < best_c - 1e-13:
            best_c = c_try
            best_d = d
    h[i], h[j] = old_i, old_j
    if best_d != 0.0:
        h[i] = hi_v + best_d
        h[j] = hj_v - best_d
    return best_d, best_c

def descent(h, rng, time_limit, n_cand=10, max_no_improve=8):
    n = len(h)
    h = h.copy()
    c = exact_c5(h)
    best_h = h.copy()
    best_c = c
    no_imp = 0
    all_pairs = make_all_pairs(n)
    sweep = 0
    while time.time() < time_limit:
        sweep += 1
        improved = False
        if n <= 80 and (sweep % 3 == 1 or no_imp >= 2):
            pairs = all_pairs
        else:
            pairs = make_mixed_pairs(h, rng, n, n_pairs=min(n * n, 2000))
        rng.shuffle(pairs)
        for (i, j) in pairs:
            if time.time() > time_limit:
                break
            d, c_new = try_pair(h, i, j, n_cand, rng)
            if d != 0.0 and c_new < c:
                c = c_new
                improved = True
                if c < best_c:
                    best_c = c
                    best_h = h.copy()
        if not improved:
            no_imp += 1
            if no_imp >= max_no_improve:
                break
        else:
            no_imp = 0
    return best_h, best_c

def jitter_h(h, rng, sigma):
    n = len(h)
    y = h + rng.normal(0.0, sigma, size=n)
    return project_feasible(y, n)

def expand_h(h, n_target):
    n_src = len(h)
    if n_target == n_src:
        return h.copy()
    x_src = (0.5 + np.arange(n_src)) * 2.0 / n_src
    x_tgt = (0.5 + np.arange(n_target)) * 2.0 / n_target
    y = np.interp(x_tgt, x_src, h, left=0.5, right=0.5)
    return project_feasible(y, n_target)

def shrink_h(h, n_target):
    n_src = len(h)
    if n_target == n_src:
        return h.copy()
    if n_src % n_target == 0:
        k = n_src // n_target
        y = h.reshape(n_target, k).mean(axis=1)
    else:
        x_src = (0.5 + np.arange(n_src)) * 2.0 / n_src
        x_tgt = (0.5 + np.arange(n_target)) * 2.0 / n_target
        y = np.interp(x_tgt, x_src, h, left=0.5, right=0.5)
    return project_feasible(y, n_target)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    rng = np.random.default_rng(seed)
    hard_stop = t0 + budget_s * 0.94
    safe_stop = t0 + budget_s * 0.97

    # Load parent
    try:
        parent = np.array(initial_h_values, dtype=np.float64).flatten()
        n_par = len(parent)
        if n_par < 4 or not np.all(np.isfinite(parent)):
            raise ValueError("bad parent")
    except Exception:
        n_par = 64
        parent = np.full(n_par, 0.5)

    parent = project_feasible(parent, n_par)
    best_h = parent.copy()
    best_c = exact_c5(best_h)
    best_n = n_par

    # Phase 1: Descent at parent resolution (65% of time)
    phase1_end = t0 + budget_s * 0.65
    n_cand_base = 10

    # Multiple descent rounds with restarts
    n_rounds = 5
    for rnd in range(n_rounds):
        round_time = min(t0 + (budget_s * 0.65) * ((rnd + 1) / n_rounds), phase1_end)
        if time.time() > phase1_end:
            break
        # Jitter for restart (except first round)
        if rnd > 0:
            sigma = 0.05 / (rnd + 1)
            h_start = jitter_h(best_h, rng, sigma)
            c_start = exact_c5(h_start)
            # Only use jittered start if it's not too bad
            if c_start > best_c + 0.02:
                h_start = best_h.copy()
        else:
            h_start = best_h.copy()

        h_cur, c_cur = descent(h_start, rng, round_time, n_cand=n_cand_base, max_no_improve=6)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur
            best_n = len(h_cur)

    # Increase candidates for refinement
    if time.time() < phase1_end:
        h_cur, c_cur = descent(best_h, rng, phase1_end, n_cand=20, max_no_improve=12)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur
            best_n = len(h_cur)

    # Phase 2: Higher resolutions (25% of time)
    phase2_end = t0 + budget_s * 0.90
    child_dims = []
    for nd in [96, 128, 160, 192]:
        if nd > 2400:
            break
        child_dims.append(nd)

    for nd in child_dims:
        if time.time() > phase2_end:
            break
        # Expand best to child dim
        h_child = expand_h(best_h, nd)
        c_child = exact_c5(h_child)
        # Also try expanding from a jittered version
        h_jit = jitter_h(best_h, rng, 0.03)
        h_child2 = expand_h(h_jit, nd)
        c_child2 = exact_c5(h_child2)
        h_start = h_child if c_child <= c_child2 else h_child2
        c_start = min(c_child, c_child2)

        # Give this child a time budget
        child_time = min(time.time() + budget_s * 0.05, phase2_end)
        h_cur, c_cur = descent(h_start, rng, child_time, n_cand=8, max_no_improve=5)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur
            best_n = len(h_cur)

    # Phase 3: Final polish (remaining time until safe_stop)
    if time.time() < safe_stop:
        # Return to best resolution for final polish
        if best_n != n_par and best_n != len(best_h):
            pass  # already consistent
        # Try shrinking best to parent dim if it's at higher res
        if best_n > n_par:
            h_shrunk = shrink_h(best_h, n_par)
            c_shrunk = exact_c5(h_shrunk)
            if c_shrunk < best_c:
                best_c = c_shrunk
                best_h = h_shrunk
                best_n = n_par

        # Final fine descent
        h_cur, c_cur = descent(best_h, rng, safe_stop, n_cand=15, max_no_improve=15)
        if c_cur < best_c:
            best_c = c_cur
            best_h = h_cur
            best_n = len(h_cur)

    # Final feasibility check and evaluation
    best_h = project_feasible(best_h, best_n)
    best_c = exact_c5(best_h)

    # Safety: ensure we always have a valid result
    if not np.all(np.isfinite(best_h)) or len(best_h) < 2:
        best_h = np.full(n_par, 0.5)
        best_n = n_par
        best_c = exact_c5(best_h)

    return (best_h, best_c, best_n)
