def exact_c5(h):
    """Compute the exact C5 value for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def feasible_interval(h, a, b):
    """Find the feasible interval for tau in h[a] += tau, h[b] -= tau."""
    # h[a] + tau in [0, 1] and h[b] - tau in [0, 1]
    tau_min = max(-h[a], h[b] - 1.0)
    tau_max = min(1.0 - h[a], h[b])
    if tau_min > tau_max:
        return None, None
    return tau_min, tau_max

def evaluate_pair_transfer(h, a, b, n_samples=30):
    """Try transfer between a and b, find best tau, return (best_h, best_c5, best_tau)."""
    tau_min, tau_max = feasible_interval(h, a, b)
    if tau_min is None:
        return None, None, None
    
    # Also try the reverse: h[a] -= tau, h[b] += tau
    # This is equivalent to negating tau, so just consider [tau_min, tau_max]
    
    if tau_max - tau_min < 1e-15:
        return None, None, None
    
    # Sample points
    tau_vals = np.linspace(tau_min, tau_max, n_samples)
    
    best_tau = 0.0
    best_c5 = exact_c5(h)  # current value
    best_h = h.copy()
    
    h_new = h.copy()
    for tau in tau_vals:
        h_new[a] += tau
        h_new[b] -= tau
        c5 = exact_c5(h_new)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_tau = tau
            best_h = h_new.copy()
    
    return best_h, best_c5, best_tau

def project_feasible(h, T):
    """Project h onto {x: 0 <= x_i <= 1, sum(x) = T}."""
    h = np.clip(h, 0.0, 1.0)
    s = np.sum(h)
    if abs(s - T) < 1e-10:
        return h
    # Use bisection to find offset
    # We need sum(clip(h + delta, 0, 1)) = T
    lo, hi = -1.0, 1.0

        trial = np.clip(h + mid, 0.0, 1.0)
        if np.sum(trial) > T:
            hi = mid
        else:
            lo = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)

    diff = T - np.sum(result)
    if abs(diff) > 1e-10:
        # Distribute among non-bound entries
        mask = (result > 1e-8) & (result < 1.0 - 1e-8)
        if np.any(mask):
