import numpy as np
import time

def compute_c(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def eval_c5(h):
    return float(np.max(compute_c(h)))

def active_grad(h, T, K):
    n = len(h)
    dx = 2.0 / n
    c = compute_c(h)
    center = n - 1
    m = len(c)
    if K >= m:
        active = np.arange(m)
    else:
        active = np.argpartition(c, -K)[-K:]
    c_a = c[active]
    c_max = c_a.max()
    exps = np.exp((c_a - c_max) / T)
    p = exps / exps.sum()
    grad = np.zeros(n)
    for idx in range(len(active)):
        k = int(active[idx])
        w = p[idx] * dx
        r = k - center
        if r == 0:
            grad += w * (1.0 - 2.0 * h)
        elif r > 0:
            grad[r:] += w * (1.0 - h[:n - r])
            grad[:n - r] -= w * h[r:]
        else:
            s = -r
            grad[:n - s] += w * (1.0 - h[s:])
            grad[s:] -= w * h[:n - s]
    grad -= grad.mean()
    return grad, c

def project(y, n):
    target = n / 2.0
    lo = y.min() - 2.0
    hi = y.max() + 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(y - mid, 0.0, 1.0).sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    x = np.clip(y - (lo + hi) * 0.5, 0.0, 1.0)
    diff = target - x.sum()
    if abs(diff) > 1e-13:
        if diff > 0:
            idx = np.where(x < 1.0 - 1e-13)[0]
        else:
            idx = np.where(x > 1e-13)[0]
        if len(idx) > 0:
            x[idx] += diff / len(idx)
            x = np.clip(x, 0.0, 1.0)
    return x

def resample_h(h, n_new):
    n_old = len(h)
    x_old = (np.arange(n_old) + 0.5) * 2.0 / n_old
    x_new = (np.arange(n_new) + 0.5) * 2.0 / n_new
    h_new = np.interp(x_new, x_old, h)
    return project(h_new, n_new)

def gradient_descent(h, n, budget_s, T_init=0.005, K_init=25, lr_init=0.5):
    t0 = time.time()
    best_h = h.copy()
    best_c5 = eval_c5(best_h)
    T = T_init
    K = K_init
    lr = lr_init
    stuck = 0
    
    while budget_s - (time.time() - t0) > 10:
        grad, c = active_grad(h, T, K)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-15:
            stuck += 1
            if stuck > 50:
                scale = max(0.001, 0.01 * np.exp(-stuck / 200.0))
                p = np.random.randn(n) * scale
                p -= p.mean()
                h = project(best_h + p, n)
                stuck = 0
            continue
        
        f_curr = eval_c5(h)
        alpha = lr
        accepted = False
        for _ in range(25):
            h_try = project(h - alpha * grad, n)
            f_try = eval_c5(h_try)
            if f_try < f_curr - 1e-15:
                h = h_try
                accepted = True
                if f_try < best_c5:
                    best_h = h_try.copy()
                    best_c5 = f_try
                    stuck = 0
                break
            alpha *= 0.5
            if alpha < 1e-5:
                break
        
        if not accepted:
            stuck += 1
            if stuck > 100:
                for scale in [0.05, 0.02, 0.01, 0.005, 0.002]:
                    p = np.random.randn(n) * scale
                    p -= p.mean()
                    h_try = project(best_h + p, n)
                    if eval_c5(h_try) < best_c5:
                        best_h = h_try.copy()
                        best_c5 = eval_c5(best_h)
                h = best_h.copy()
                stuck = 0
                lr = lr_init
                T = max(0.001, T * 2.0)
        
        if (stuck + 1) % 50 == 0:
            T = max(0.0005, T * 0.95)
    
    return best_h, best_c5
