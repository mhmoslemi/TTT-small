    inits.append(np.concatenate([np.full(g // 2, 1.0), np.full(g - g // 2, 0.0)]))
    inits.append(np.concatenate([np.full(g // 3, 0.0), np.full(g - 2 * g // 3, 1.0), np.full(g // 3, 0.0)]))
    inits.append(1.0 - 2.0 * np.abs(x - 0.5))
    for _ in range(12):
        inits.append(np.clip(rng.random(g) * 0.8 + 0.1, 0, 1))
    for _ in range(6):
        inits.append(np.clip(0.5 + 0.35 * rng.standard_normal(g), 0, 1))
    for _ in range(4):
        p = np.clip(0.5 + 0.4 * np.cos(2 * np.pi * rng.uniform(0.5, 4.0) * x + rng.uniform(0, 2 * np.pi)), 0, 1)
        inits.append(p)
    return inits

def run_lse_bfgs(p0, n, tau_list, state, deadline):
    g = len(p0)
    bounds = [(0.0, 1.0)] * g
    _LSE['n'] = n
    _LSE['state'] = state
    _LSE['deadline'] = deadline
    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        _LSE['tau'] = tau
        maxfun = 3000 if g <= 20 else 2000 if g <= 40 else 1500
        res = minimize(_lse_obj, p0, method='L-BFGS-B', bounds=bounds,
                       options={'maxfun': maxfun, 'ftol': 1e-12, 'gtol': 1e-8})
        if res.x is not None and np.all(np.isfinite(res.x)):
            p0 = np.clip(res.x, 0.0, 1.0)
    return p0

def run_slq_epigraph(p0, n, state, deadline, top_k=None):
    g = n // 2
    h = build_h(p0, n)
    f = correlate_values(h)
    c = float(np.max(f))
    num_lags = len(f)
    if top_k is None:
        top_k = min(20, num_lags)
    order = np.argsort(f)[::-1]
    idxs = order[:top_k]
    _SLQ['g'] = g
    _SLQ['n'] = n
    _SLQ['idxs'] = idxs
    x0 = np.concatenate([np.clip(p0, 0, 1), [c + 1e-8]])
    bounds = [(0.0, 1.0)] * g + [(0.0, 1.0)]
    constraints = [{'type': 'ineq', 'fun': _slq_ineq}]
    maxiter = 100 if g <= 20 else 70 if g <= 40 else 50
    res = minimize(_slq_obj, x0, method='SLSQP', bounds=bounds,
                   constraints=constraints,
                   options={'maxiter': maxiter, 'ftol': 1e-10, 'eps': 1e-7})
    if res.x is not None and np.all(np.isfinite(res.x)):
        p_new = np.clip(res.x[:g], 0.0, 1.0)
        h_new = build_h(p_new, n)
        c_new = exact_c5(h_new)
        if c_new < state['best_val'] - 1e-12:
            state['best_val'] = c_new
            state['best_h'] = h_new.copy()
    return p0

def run_full_lse(y0, n, tau_list, state, deadline):
    _LSE['n'] = n
    _LSE['state'] = state
    _LSE['deadline'] = deadline
    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        _LSE['tau'] = tau
        maxfun = 1000 if n <= 128 else 600
        res = minimize(_full_obj, y0, method='L-BFGS-B',
                       options={'maxfun': maxfun, 'ftol': 1e-12})
        if res.x is not None and np.all(np.isfinite(res.x)):
            y0 = project_full(res.x)
    return y0

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.monotonic()
    deadline = start_time + budget_s * 0.95
    rng = np.random.default_rng(seed)

    state = {'best_val': 1.0, 'best_h': None}

    # Load parent
    parent_h = None
    try:
        if 'initial_h_values' in globals() and initial_h_values is not None:
            parent_h = np.array(initial_h_values, dtype=np.float64).copy()
            if parent_h.ndim != 1 or len(parent_h) < 4:
                parent_h = None
    except Exception:
        parent_h = None

    if parent_h is not None:
        parent_h = np.clip(parent_h, 0.0, 1.0)
        n_par = len(parent_h)
        if abs(np.sum(parent_h) - n_par / 2.0) > 1.0:
            parent_h = project_full(parent_h)
        c_par = exact_c5(parent_h)
        state['best_val'] = min(state['best_val'], c_par)
        state['best_h'] = parent_h.copy()
    else:
        n_par = 64
        parent_h = np.full(n_par, 0.5)
        state['best_val'] = exact_c5(parent_h)
        state['best_h'] = parent_h.copy()

    # Determine dimensions to try
    base_n = len(parent_h)
    dims = sorted(set([base_n, 96, 128, 160, 192, 256]))
    dims = [d for d in dims if d >= 32 and d <= 320]
    if base_n not in dims:
        dims.append(base_n)
        dims.sort()

    # Extract symmetric half from parent
    p_par = sym_p_from_h(parent_h)

    # Phase 1: L-BFGS-B with LSE on multiple dims and inits
    phase1_end = start_time + budget_s * 0.45
    candidates = []

    for n in dims:
        if time.monotonic() > phase1_end:
            break
        g = n // 2
        # Interpolate parent to this dimension
        p_init_par = interp_p(p_par, g)
        tau_list = [0.02, 0.01, 0.005, 0.002, 0.001]
        inits = generate_initials(n, seed + n)
        inits.insert(0, p_init_par)

        # Evaluate all inits, sort
        scored = []
        for p_init in inits:
            h_tmp = build_h(p_init, n)
            c_tmp = exact_c5(h_tmp)
            scored.append((c_tmp, p_init))
        scored.sort(key=lambda x: x[0])

        # Optimize top candidates
        top_count = min(8, len(scored))
        for i in range(top_count):
            if time.monotonic() > deadline:
                break
            c_sub, p_sub = scored[i]
            p_opt = run_lse_bfgs(p_sub, n, tau_list, state, deadline)
            if p_opt is not None:
                h_opt = build_h(p_opt, n)
                c_opt = exact_c5(h_opt)
                if c_opt < state['best_val'] - 1e-12:
                    state['best_val'] = c_opt
                    state['best_h'] = h_opt.copy()
                candidates.append((c_opt, p_opt, n))

    # Phase 2: SLSQP epigraph polish on best candidates
    phase2_end = start_time + budget_s * 0.75
    candidates.sort(key=lambda x: x[0])
    polished = 0
    for c_sub, p_sub, n_sub in candidates[:12]:
        if time.monotonic() > phase2_end:
            break
        if c_sub > state['best_val'] + 0.01:
            continue
        p_new = run_slq_epigraph(p_sub, n_sub, state, deadline, top_k=15)
        polished += 1
        if polished >= 15:
            break

    # Phase 3: Full-space optimization (non-symmetric)
    phase3_end = start_time + budget_s * 0.90
    if time.monotonic() < phase3_end:
        n_full = min(256, max(128, base_n))
        tau_full = [0.02, 0.01, 0.005]
        h_best = state['best_h']
        y0 = project_full(h_best[:n_full] if len(h_best) >= n_full else np.interp(
            np.linspace(0, len(h_best) - 1, n_full), np.arange(len(h_best)), h_best))
        y_opt = run_full_lse(y0, n_full, tau_full, state, deadline)
        if y_opt is not None:
            h_full = project_full(y_opt)
            c_full = exact_c5(h_full)
            if c_full < state['best_val'] - 1e-12:
                state['best_val'] = c_full
                state['best_h'] = h_full.copy()

    # Phase 4: Final polish - try SLSQP with more constraints on best
    if time.monotonic() < deadline - 5.0 and state['best_h'] is not None:
        h_best = state['best_h']
        n_best = len(h_best)
        p_best = sym_p_from_h(h_best)
        # Try with more active constraints
        _SLQ_test = None
        g = n_best // 2
        h_chk = build_h(p_best, n_best)
        f_chk = correlate_values(h_chk)
        top_k_final = min(30, len(f_chk))
        order = np.argsort(f_chk)[::-1]
        idxs = order[:top_k_final]
        _SLQ['g'] = g
        _SLQ['n'] = n_best
        _SLQ['idxs'] = idxs
        x0 = np.concatenate([p_best, [state['best_val'] + 1e-8]])
        bounds = [(0.0, 1.0)] * g + [(0.0, 1.0)]
        constraints = [{'type': 'ineq', 'fun': _slq_ineq}]
        res = minimize(_slq_obj, x0, method='SLSQP', bounds=bounds,
                       constraints=constraints,
                       options={'maxiter': 80, 'ftol': 1e-11, 'eps': 1e-7})
        if res.x is not None and np.all(np.isfinite(res.x)):
            p_new = np.clip(res.x[:g], 0.0, 1.0)
            h_new = build_h(p_new, n_best)
            c_new = exact_c5(h_new)
            if c_new < state['best_val'] - 1e-12:
                state['best_val'] = c_new
                state['best_h'] = h_new.copy()

    # Final validation
    if state['best_h'] is None:
        state['best_h'] = np.full(64, 0.5)
    
    h_out = np.clip(state['best_h'].copy(), 0.0, 1.0)
    n_out = len(h_out)
    # Ensure feasibility
    target = n_out / 2.0
    if abs(np.sum(h_out) - target) > 1e-6:
        h_out = project_full(h_out)
    c5_final = exact_c5(h_out)

    return (h_out, c5_final, n_out)
