def local_improve(h0, max_secs, rng, exact_max_M=200):

    n=len(h)

    no_improve=0
    temp_c5=0.0
    stage=0
    deadline=start+max_secs
    while time.time() < deadline:
        stage += 1
        # temp
        if no_improve <=0: temp_c5=0.0
        elif no_improve ==1: temp_c5=5.0e-5
        elif no_improve ==2: temp_c5=1.0e-4

def project_onto_box_sum(z, target):
    z = np.asarray(z, dtype=float)
    if z.size == 0:
        return z
    if not np.all(np.isfinite(z)):
        z = np.where(np.isfinite(z), z, 0.5)

    lo = float(np.min(z)) - 1.0
    hi = float(np.max(z)) + 1.0
    if lo > hi:
        lo, hi = hi, lo

    for _ in range(80):
        mid = 0.5 * (lo + hi)
        val = np.clip(z - mid, 0.0, 1.0)
        s = float(val.sum())
        if s > target:
            lo = mid
        else:
            hi = mid

    lam = 0.5 * (lo + hi)
    h = np.clip(z - lam, 0.0, 1.0)
    err = target - float(h.sum())
    if abs(err) > 1e-13:
        if err > 0.0:
            mask = (1.0 - h) > 1e-12
        else:
            mask = h > 1e-12
        if int(np.sum(mask)) > 0:
            h[mask] += err / float(np.sum(mask))
        h = np.clip(h, 0.0, 1.0)
    return h


def make_feasible(h):

    if not np.all(np.isfinite(h)):

def evaluate_c5(h):

    c = np.correlate(h, 1.0 - h, mode="full")

def zero_mean_norm(v):
    v = np.asarray(v, dtype=float)
    if v.size == 0 or not np.all(np.isfinite(v)):
        return None
    v = v.copy()
    v -= float(np.mean(v))
    m = float(np.max(np.abs(v)))
    if not np.isfinite(m) or m <= 1e-14:
        return None
    return v / m


def box_interval(h, v, limit=1.0):
    lo = -float(limit)
    hi = float(limit)

    pos = v > 0.0
    if np.any(pos):
        lo = max(lo, float(np.min(-h[pos] / v[pos])))
        hi = min(hi, float(np.min((1.0 - h[pos]) / v[pos])))

    neg = v < 0.0
    if np.any(neg):
        lo = max(lo, float(np.min((1.0 - h[neg]) / v[neg])))
        hi = min(hi, float(np.min(-h[neg] / v[neg])))

    if not (np.isfinite(lo) and np.isfinite(hi)):
        return None
    if hi - lo <= 1e-12:
        return None
    return lo, hi


def compute_coeffs(c0, h, v):

    c1 = np.correlate(h1, 1.0 - h1, mode="full")
    h2 = h - v
    c2 = np.correlate(h2, 1.0 - h2, mode="full")

    b = 0.5 * (2.0 * c0 - c1 - c2)

def line_search_exact(c0, a, b, lo, hi):

    mask = b < -1e-14

def eval_max_quad(t, c0, a, b):
    vals = c0 + a * t - b * (t * t)

    return nh, nc, float(t)


def lag_gradient(h, d):
    n = len(h)
    g = np.zeros(n, dtype=float)
    if d >= 0:
        m = n - d
        if m <= 0:
            return g
        g[d:] += 1.0
        g[d:] -= h[:m]
        g[:m] -= h[d:]
    else:
        e = -d
        m = n - e
        if m <= 0:
            return g
        g[:m] += 1.0
        g[:m] -= h[e:]
        g[e:] -= h[:m]
    return g


def gradient_directions(h, c, k_values):
    n = len(h)
    m = len(c)
    order = np.argsort(c)[::-1]
    dirs = []
    for k in k_values:
        kk = min(int(k), m)
        idx = order[:kk]
        g = np.zeros(n, dtype=float)
        for r in idx:
            d = int(r) - (n - 1)
            g += lag_gradient(h, d)

        v = zero_mean_norm(-g)
        if v is not None:
            dirs.append(v)

        if kk > 1:
            w = zero_mean_norm(g)
            if w is not None:
                dirs.append(w)

        vals = c[idx]
        base = float(vals[-1]) if kk > 0 else 0.0
        weights = vals - base + 1e-12
        gw = np.zeros(n, dtype=float)
        for wt, r in zip(weights, idx):
            d = int(r) - (n - 1)
            gw += float(wt) * lag_gradient(h, d)
        v = zero_mean_norm(-gw)
        if v is not None:
            dirs.append(v)
    return dirs


def pair_lag_directions(h, c, top_lags=5, pair_counts=(1, 2, 3, 5, 8)):
    n = len(h)
    order = np.argsort(c)[::-1]
    dirs = []
    for r in order[:top_lags]:
        d = int(r) - (n - 1)
        if d == 0:
            continue
        if d < 0:
            e = -d
            m = n - e
            if m <= 0:
                continue
            left = np.arange(m)
            right = left + e
            contrib = h[left] * (1.0 - h[right])
        else:
            m = n - d
            if m <= 0:
                continue
            right = np.arange(d, n)
            left = right - d
            contrib = h[right] * (1.0 - h[left])

        if contrib.max() <= 1e-15:
            continue
        order_c = np.argsort(contrib)[::-1]

        for k in pair_counts:
            kk = min(int(k), len(contrib))
            if kk <= 0:
                continue
            top = order_c[:kk]
            v = np.zeros(n, dtype=float)
            for idx in top:
                idx = int(idx)
                if d < 0:
                    v[right[idx]] += 1.0
                    v[left[idx]] -= 1.0
                else:
                    v[left[idx]] += 1.0
                    v[right[idx]] -= 1.0
            vn = zero_mean_norm(v)
            if vn is not None:
                dirs.append(vn)
                vn2 = zero_mean_norm(-vn)
                if vn2 is not None:
                    dirs.append(vn2)
    return dirs


def fourier_directions(n, modes=None, phases=None):
    if n < 4:
        return []
    if modes is None:
        modes = range(1, n // 2 + 1)
    if phases is None:
        phases = (0.0, 0.5 * np.pi)

    j = np.arange(n, dtype=float)
    dirs = []
    for m in modes:
        val = 2.0 * np.pi * float(int(m)) * j / float(n)
        for ph in phases:
            v = np.cos(val + float(ph))
            vn = zero_mean_norm(v)
            if vn is not None:
                dirs.append(vn)
    return dirs


def random_fourier_directions(n, rng, count):
    if n < 4 or count <= 0 or rng is None:
        return []
    j = np.arange(n, dtype=float)
    dirs = []
    for _ in range(count):
        m = int(rng.integers(1, n // 2 + 1))
        ph = float(rng.uniform(0.0, 2.0 * np.pi))
        v = np.cos(2.0 * np.pi * m * j / float(n) + ph)
        vn = zero_mean_norm(v)
        if vn is not None:
            dirs.append(vn)
    return dirs


def random_smooth_directions(n, rng, count):
    if n < 4 or count <= 0 or rng is None:
        return []
    freqs = np.arange(n // 2 + 1, dtype=float)
    dirs = []
    for _ in range(count):
        x = rng.normal(size=n)
        alpha = float(10.0 ** rng.uniform(0.0, 2.0))
        fx = np.fft.rfft(x)
        filt = 1.0 / (1.0 + (freqs / alpha) ** 2)
        v = np.fft.irfft(fx * filt, n=n)
        vn = zero_mean_norm(v)
        if vn is not None:
            dirs.append(vn)
    return dirs


def random_pair_directions(n, rng, count):
    if n < 2 or count <= 0 or rng is None:
        return []
    dirs = []
    for _ in range(count):
        i = int(rng.integers(0, n))
        j = int(rng.integers(0, n))
        if i == j:
            continue
        v = np.zeros(n, dtype=float)
        v[i] = 1.0
        v[j] = -1.0
        vn = zero_mean_norm(v)
        if vn is not None:
            dirs.append(vn)
    return dirs


def batch_directions(h, c, rng, stage, no_improve, n):
    if rng is None:
        rng = np.random.default_rng(0)

    dirs = []
    dirs.extend(pair_lag_directions(h, c, top_lags=5, pair_counts=(1, 2, 3, 5, 8)))
    dirs.extend(gradient_directions(h, c, k_values=(1, 2, 3, 5, 10)))

    if stage % 2 == 0:
        modes = range(1, min(11, n // 2 + 1))
        dirs.extend(fourier_directions(n, modes=modes, phases=(0.0, 0.5 * np.pi)))
    else:
        dirs.extend(random_fourier_directions(n, rng, 30))

    extra = 40 + 20 * min(int(no_improve), 10)
    dirs.extend(random_smooth_directions(n, rng, extra))
    dirs.extend(random_pair_directions(n, rng, extra))

    cap = 300 if n > 150 else 600
    if len(dirs) > cap:
        keep = 150
        targeted = dirs[:keep]
        rest = dirs[keep:]
        if len(rest) > 0:
            perm = rng.permutation(len(rest))
            shuffled = []
            for p in perm:
                shuffled.append(rest[int(p)])
            rest = shuffled
        dirs = targeted + rest
        if len(dirs) > cap:
            dirs = dirs[:cap]
    return dirs


def orthonormalize(vecs, max_d):
    basis = []
    for raw in vecs:
        v = zero_mean_norm(raw)
        if v is None:
            continue
        w = v.astype(float).copy()
        for b in basis:
            w = w - (float(np.dot(w, b)) / float(np.dot(b, b))) * b
        nrm = float(np.linalg.norm(w))
        if nrm > 1e-8:
            w /= nrm
            basis.append(w)
            if len(basis) >= max_d:
                break
    return basis


def objective_projected(y, h0, v_basis):
    y = np.asarray(y, dtype=float).reshape(-1)
    z = h0 + y @ v_basis
    hp = project_onto_box_sum(z, 0.5 * float(len(h0)))
    c = np.correlate(hp, 1.0 - hp, mode="full")
    return float(np.max(c) * 2.0 / float(len(h0)))


def multivariate_search(h0, max_secs, rng, d=4):
    start = time.time()
    h = make_feasible(h0)
    n = len(h)
    if n == 0 or rng is None:
        return h, evaluate_c5(h)

    best_h = h.copy()
    best5 = evaluate_c5(h)
    c = np.correlate(h, 1.0 - h, mode="full")

    raw = pair_lag_directions(h, c, top_lags=3, pair_counts=(1, 2, 3))
    raw.extend(gradient_directions(h, c, k_values=(1, 2, 3)))
    raw.extend(random_fourier_directions(n, rng, 3))
    raw.extend(random_smooth_directions(n, rng, 3))

    basis = orthonormalize(raw, d)
    if len(basis) == 0:
        return best_h, best5
    v_basis = np.array(basis, dtype=float)

    starts = [np.zeros(len(v_basis), dtype=float)]
    for _ in range(3):
        starts.append(rng.normal(scale=0.05, size=len(v_basis)))

    for y0 in starts:
        if time.time() - start > max_secs:
            break
        try:
            res = opt.minimize(
                objective_projected,
                y0,
                args=(h, v_basis),
                method="Nelder-Mead",
                options={"maxiter": 120, "xatol": 1e-7, "fatol": 1e-12}
            )
            eval_points = [y0, np.asarray(res.x, dtype=float)]
        except Exception:
            eval_points = [y0]

        for yp in eval_points:
            if time.time() - start > max_secs:
                break
            try:
                z = h + np.asarray(yp, dtype=float).reshape(-1) @ v_basis
                hc = project_onto_box_sum(z, 0.5 * float(n))
                c5c = evaluate_c5(hc)
                if c5c < best5:
                    best5 = c5c
                    best_h = hc.copy()
            except Exception:
                pass

    return best_h, best5


def periodic_projection(h):
    n = len(h)
    if n % 2 != 0:
        return h
    m = n // 2
    f = 0.5 * (h[:m] + h[m:])
    f = make_feasible(f)
    return np.tile(f, 2)


def upsample_h(h, new_n):
    h = np.asarray(h, dtype=float).flatten()
    old_n = len(h)
    if old_n == 0 or new_n <= old_n:
        return h
    pos = (np.arange(new_n, dtype=float) * 2.0 / float(new_n)) * float(old_n) / 2.0
    i0 = np.floor(pos).astype(int) % old_n
    frac = pos - np.floor(pos)
    i1 = (i0 + 1) % old_n
    f = (1.0 - frac) * h[i0] + frac * h[i1]
    return make_feasible(f)


def default_harmonic(n, phase=0.0):
    if int(n) % 2 != 0:
        n = int(n) + 1
    m = n // 2
    j = np.arange(m, dtype=float)
    f = 0.5 + 0.5 * np.cos(2.0 * np.pi * j / float(m) + float(phase))
    f = make_feasible(f)
    return np.tile(f, 2)


def local_improve(h0, max_secs, rng, exact_max_m=200):
    start = time.time()
    h = make_feasible(h0)
    n = len(h)
    if n == 0:
        return None, float("inf")
    if rng is None:
        rng = np.random.default_rng(0)

    c = np.correlate(h, 1.0 - h, mode="full")
    cur = float(np.max(c))
    best = h.copy()
    best5 = cur * 2.0 / float(n)

    no_improve = 0
    temp_c5 = 0.0
    stage = 0
    deadline = start + float(max_secs)

    while time.time() < deadline:
        stage += 1

        if no_improve <= 0:
            temp_c5 = 0.0
        elif no_improve == 1:
            temp_c5 = 5.0e-5
        elif no_improve == 2:
            temp_c5 = 1.0e-4
        elif no_improve == 3:
            temp_c5 = 2.0e-4
        else:
            temp_c5 = max(5.0e-6, 2.0e-4 * (0.7 ** (no_improve - 3)))

        temp_unscaled = temp_c5 * (float(n) / 2.0)
        dirs = batch_directions(h, c, rng, stage, no_improve, n)
        accepted = False

        for v in dirs:
            if time.time() > deadline:
                break
            res = try_direction(h, c, v, cur, temp_unscaled, exact_max_m)
            if res is None:
                continue
            nh, nc, t = res

            if nc < cur - 1e-12:
                new5 = nc * 2.0 / float(n)
                if new5 < best5 - 1e-14:
                    best = nh.copy()
                    best5 = new5
                h = nh
                c = np.correlate(h, 1.0 - h, mode="full")
                cur = nc
                no_improve = 0
                accepted = True
                break
            else:
                if (temp_unscaled > 0.0 and abs(t) > 1e-10 and
                        nc <= cur + temp_unscaled + 1e-12 and nc > cur + 1e-13):
                    h = nh
                    c = np.correlate(h, 1.0 - h, mode="full")
                    cur = nc
                    no_improve += 1
                    accepted = True
                    break

        if not accepted:
            no_improve += 1

        rem = deadline - time.time()

        if no_improve >= 4 and rem > 40.0:
            scale = 0.005
            if no_improve > 4:
                scale = 0.005 + 0.002 * float(no_improve - 4)
                if scale > 0.025:
                    scale = 0.025
            z = best + rng.normal(scale=scale, size=n)
            h = make_feasible(z)
            c = np.correlate(h, 1.0 - h, mode="full")
            cur = float(np.max(c))
            no_improve = 0
            temp_c5 = 0.0

        if no_improve >= 3 and stage % 3 == 0 and rem > 60.0:
            mv_h, mv5 = multivariate_search(best, min(40.0, rem * 0.5), rng, d=4)
            if mv_h is not None and mv5 < best5 - 1e-14:
                best = mv_h.copy()
                best5 = mv5
                h = best.copy()
                c = np.correlate(h, 1.0 - h, mode="full")
                cur = float(np.max(c))
                no_improve = 0

    return best, best5


def run(seed=42, budget_s=1100, **kwargs):
    total_start = time.time()
    budget = float(budget_s)
    rng = np.random.default_rng(seed)

    best_h = None
    best5 = float("inf")

    try:
        init = initial_h_values
    except NameError:
        init = None

    if init is not None:
        try:
            arr = np.asarray(init, dtype=float).flatten()
            if arr.size > 0:
                h0 = make_feasible(arr)
                if h0 is not None and len(h0) > 0:
                    best_h = h0.copy()
                    best5 = evaluate_c5(h0)
        except Exception:
            best_h = None
            best5 = float("inf")

    if best_h is None:
        best_h = default_harmonic(84, 0.0)
        best5 = evaluate_c5(best_h)

    rem = budget - (time.time() - total_start)
    if rem > 30.0:
        t1 = min(0.35 * budget, rem, 450.0)
        exact_m = 200 if len(best_h) <= 100 else 130
        try:
            h_loc, c5_loc = local_improve(best_h, t1, rng, exact_max_m=exact_m)
            if h_loc is not None and c5_loc < best5:
                best_h = h_loc.copy()
                best5 = c5_loc
        except Exception:
            pass

    rem = budget - (time.time() - total_start)
    if rem > 90.0 and len(best_h) % 2 == 0:
        try:
            hp = periodic_projection(best_h)
            if hp is not None and len(hp) > 0:
                t2 = min(0.20 * budget, rem, 180.0)
                exact_m = 200 if len(hp) <= 100 else 130
                h_loc, c5_loc = local_improve(hp, t2, rng, exact_max_m=exact_m)
                if h_loc is not None and c5_loc < best5:
                    best_h = h_loc.copy()
                    best5 = c5_loc
        except Exception:
            pass

    rem = budget - (time.time() - total_start)
    if rem > 120.0 and len(best_h) <= 120:
        new_n = 2 * len(best_h)
        if new_n <= 252:
            try:
                hu = upsample_h(best_h, new_n)
                if hu is not None and len(hu) > 0:
                    t3 = min(0.20 * budget, rem * 0.6, 200.0)
                    exact_m = 130 if len(hu) > 100 else 200
                    h_loc, c5_loc = local_improve(hu, t3, rng, exact_max_m=exact_m)
                    if h_loc is not None and c5_loc < best5:
                        best_h = h_loc.copy()
                        best5 = c5_loc
            except Exception:
                pass

    rem = budget - (time.time() - total_start)
    if rem > 60.0:
        phases = (0.0, np.pi / 3.0, 2.0 * np.pi / 3.0)
        for ph in phases:
            rem = budget - (time.time() - total_start)
            if rem < 25.0:
                break
            try:
                h0 = default_harmonic(84, ph)
                t4 = min(20.0, rem * 0.3, 25.0)
                h_loc, c5_loc = local_improve(h0, t4, rng, exact_max_m=200)
                if h_loc is not None and c5_loc < best5:
                    best_h = h_loc.copy()
                    best5 = c5_loc
            except Exception:
                pass

    rem = budget - (time.time() - total_start)
    if rem > 30.0:
        t5 = min(60.0, rem * 0.8, 80.0)
        exact_m = 200 if len(best_h) <= 100 else 130
        try:
            h_loc, c5_loc = local_improve(best_h, t5, rng, exact_max_m=exact_m)
            if h_loc is not None and c5_loc < best5:
                best_h = h_loc.copy()
                best5 = c5_loc
        except Exception:
            pass

    best_h = make_feasible(best_h)
    final5 = evaluate_c5(best_h)
    return best_h, float(final5), int(len(best_h))
