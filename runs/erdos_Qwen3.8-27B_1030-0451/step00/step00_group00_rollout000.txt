- u appears as p+d (so p=u-d): contribution is h[u]*(1-h[u-d]). With h[u] increased by s: (h[u]+s)*(1-h[u-d]) = h[u]*(1-h[u-d]) + s*(1-h[u-d]). Change: +s*(1-h[u-d]) (valid if 0<=u-d<n and 0<=u<n)

- v appears as p: contribution h[v+d]*(1-h[v]). With h[v] decreased by s: h[v+d]*(1-(h[v]-s)) = h[v+d]*(1-h[v]) + s*h[v+d]. Change: +s*h[v+d] (valid if 0<=v<n and 0<=v+d<n)

import numpy as np
import time

def partial_slope(t, h):
    n = len(h)
    L = 2 * n - 1
    A = np.zeros(L)
    r_lo1 = max(0, t)
    r_hi1 = min(L, t + n)
    if r_hi1 > r_lo1:
        td = t - np.arange(r_lo1, r_hi1) + n - 1
        A[r_lo1:r_hi1] += 1.0 - h[td]
    r_lo2 = max(0, n - 1 - t)
    r_hi2 = min(L, 2 * n - 1 - t)
    if r_hi2 > r_lo2:
        td2 = t + np.arange(r_lo2, r_hi2) - (n - 1)
        A[r_lo2:r_hi2] -= h[td2]

def _F_at(s, Q, A, g1, g2):
    base_max = np.max(Q + A * s)
    v1 = Q[g1] + A[g1] * s + s * s
    v2 = Q[g2] + A[g2] * s + s * s
    m = base_max
    if v1 > m:

    if v2 > m:

def fix_feasibility(h):
    n = len(h)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-12:
        if diff > 0:
            slack = 1.0 - h
            tot = np.sum(slack)
            if tot > 1e-12:
                h += (diff / tot) * slack
            else:
                h += diff / n
        else:
            slack = h.copy()
            tot = np.sum(slack)
            if tot > 1e-12:
                h += (diff / tot) * slack
