```python
def project_onto_feasible(h, target_sum):
    h = np.clip(h, 0, 1)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    if diff > 0:
        # Need to increase sum: add to entries with room
        # Water-filling: find lambda such that sum(min(1, h + lambda)) = target
        # This is equivalent to: for entries with h < 1, set to min(1, h + lambda)
        mask = h < 1.0
        if not np.any(mask):
            return h
        # Binary search for lambda
        lo, hi = 0.0, (target_sum - np.sum(h[mask])) / np.sum(mask) + 0.5
        for _ in range(50):
            mid = (lo + hi) / 2
            new_sum = np.sum(np.minimum(1.0, h[mask] + mid)) + np.sum(~mask)
            if new_sum < target_sum:
                lo = mid
            else:
                hi = mid
        h[mask] = np.minimum(1.0, h[mask] + (lo + hi) / 2)
    else:
        # Need to decrease sum
        mask = h > 0.0
        if not np.any(mask):
            return h
        lo, hi = 0.0, (-diff) / np.sum(mask) + 0.5
        for _ in range(50):
            mid = (lo + hi) / 2
            new_sum = np.sum(np.maximum(0.0, h[mask] - mid)) + np.sum(h[~mask])
            if new_sum > target_sum:
                lo = mid
            else:
                hi = mid
        h[mask] = np.maximum(0.0, h[mask] - (lo + hi) / 2)
    return h
```
