            if c5_0 < best_c5:

            if c5_b < best_c5:

                best_h = hb
                best_n = n_points

            best_n = n_final

    dx_best = 2.0 / best_n
    final_c5 = compute_c5(best_h, dx_best)
    
    return best_h, final_c5, best_n
