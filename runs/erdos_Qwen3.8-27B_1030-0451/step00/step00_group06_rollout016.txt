        if diff > 0:

            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
    return h

def interpolate_h(h_old, N_new):
    N_old = len(h_old)
    old_pos = (np.arange(N_old) + 0.5) * 2.0 / N_old
    new_pos = (np.arange(N_new) + 0.5) * 2.0 / N_new
    h_new = np.interp(new_pos, old_pos, h_old)
    return fix_sum(h_new, N_new / 2.0)

def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    safe_time = 30.0

    def tl():
        return budget_s - safe_time - (time.time() - t0)

    rng = np.random.default_rng(seed)

    h0 = np.array(initial_h_values, dtype=np.float64)
    N0 = len(h0)
    h0 = fix_sum(h0, N0 / 2.0)
    best_h = h0.copy()
    best_c5 = compute_c5(best_h)

    # Stage 1: Anti-palindromic Powell at original N
    N = N0
    half = N // 2
    x0 = project_to_anti(h0, N)

    def obj_anti(x, Nn=N):
        xc = np.clip(x, 0.0, 1.0)
        h = make_anti(xc, Nn)
        return compute_c5(h)

    if tl() > 60:
        res = minimize(obj_anti, x0, method='Powell',
                       options={'maxfev': 50000, 'xtol': 1e-10, 'ftol': 1e-12, 'disp': False})
        x_best = np.clip(res.x, 0.0, 1.0)
        h_test = make_anti(x_best, N)
        c5_test = compute_c5(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()

    x_curr = project_to_anti(best_h, N)

    # Stage 2: Multiple restarts with varied perturbations
    n_restarts = 15
    for r in range(n_restarts):
        if tl() < 60:
            break
        sigma = 0.02 + 0.03 * (r % 4)
        pert = rng.normal(0, sigma, size=half)
        x_start = np.clip(x_curr + pert, 0.0, 1.0)
        maxfev = 8000 if r < 5 else 4000
        res = minimize(obj_anti, x_start, method='Powell',
                       options={'maxfev': maxfev, 'xtol': 1e-9, 'ftol': 1e-11, 'disp': False})
        x_test = np.clip(res.x, 0.0, 1.0)
        h_test = make_anti(x_test, N)
        c5_test = compute_c5(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
            x_curr = x_test.copy()

    # Stage 3: Try different dimensions
    dim_options = [N - 30, N + 50, N + 150, N + 400, N + 800]
    dim_options = [d for d in dim_options if 100 < d < 2500]
    dim_options = sorted(set(dim_options))

    for N_new in dim_options:
        if tl() < 90:
            break
        h_interp = interpolate_h(best_h, N_new)
        half_new = N_new // 2
        x_start = project_to_anti(h_interp, N_new)

        def obj_anti_dim(x, Nn=N_new):
            xc = np.clip(x, 0.0, 1.0)
            h = make_anti(xc, Nn)
            return compute_c5(h)

        maxfev = min(20000, max(3000, int(120.0 / (N_new * 2.0 / N0) ** 2)))
        res = minimize(obj_anti_dim, x_start, method='Powell',
                       options={'maxfev': maxfev, 'xtol': 1e-9, 'ftol': 1e-11, 'disp': False})
        x_test = np.clip(res.x, 0.0, 1.0)
        h_test = make_anti(x_test, N_new)
        c5_test = compute_c5(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()

        # Quick restart at new dimension
        if tl() > 60 and c5_test < best_c5 + 0.005:
            for rr in range(3):
                if tl() < 45:
                    break
                pert = rng.normal(0, 0.03, size=half_new)
                xs = np.clip(x_test + pert, 0.0, 1.0)
                res2 = minimize(obj_anti_dim, xs, method='Powell',
                               options={'maxfev': 5000, 'xtol': 1e-9, 'ftol': 1e-11, 'disp': False})
                xt2 = np.clip(res2.x, 0.0, 1.0)
                ht2 = make_anti(xt2, N_new)
                c5_2 = compute_c5(ht2)
                if c5_2 < best_c5:
                    best_c5 = c5_2
                    best_h = ht2.copy()

    # Stage 4: Full (non-anti) optimization attempt at best dimension
    N_f = len(best_h)
    if N_f <= 1500 and tl() > 60:
        def obj_full(z, Nn=N_f):
            # Bisection to find shift for sum constraint
            lo = z.min() - 1.0
            hi = z.max()
            target_s = Nn / 2.0
            for _ in range(50):
                mid = (lo + hi) / 2.0
                h_mid = np.clip(z - mid, 0.0, 1.0)
                s = h_mid.sum()
                if s > target_s:
                    lo = mid
                else:
                    hi = mid
            h = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
            h = fix_sum(h, target_s)
            return compute_c5(h)

        z0 = best_h.astype(np.float64)
        maxfev_f = min(15000, max(2000, int(60.0 / (N_f * 2.0 / N0) ** 2)))
        res_f = minimize(obj_full, z0, method='Powell',
                         options={'maxfev': maxfev_f, 'xtol': 1e-9, 'ftol': 1e-11, 'disp': False})
        # Evaluate the result properly
        z_best = res_f.x.astype(np.float64)
        lo = z_best.min() - 1.0
        hi = z_best.max()
        for _ in range(50):
            mid = (lo + hi) / 2.0
            s = np.clip(z_best - mid, 0.0, 1.0).sum()
            if s > N_f / 2.0:
                lo = mid
            else:
                hi = mid
        h_test_f = fix_sum(np.clip(z_best - (lo + hi) / 2.0, 0.0, 1.0), N_f / 2.0)
        c5_f = compute_c5(h_test_f)
        if c5_f < best_c5:
            best_c5 = c5_f
            best_h = h_test_f.copy()

    # Stage 5: Final polish on best solution
    if tl() > 20:
        N_p = len(best_h)
        half_p = N_p // 2
        x_p = project_to_anti(best_h, N_p)

        def obj_polish(x, Nn=N_p):
            xc = np.clip(x, 0.0, 1.0)
            h = make_anti(xc, Nn)
            return compute_c5(h)

        res_p = minimize(obj_polish, x_p, method='Powell',
                         options={'maxfev': 30000, 'xtol': 1e-11, 'ftol': 1e-13, 'disp': False})
        x_pt = np.clip(res_p.x, 0.0, 1.0)
        h_pt = make_anti(x_pt, N_p)
        c5_pt = compute_c5(h_pt)
        if c5_pt < best_c5:
            best_c5 = c5_pt
            best_h = h_pt.copy()

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    return best_h, float(best_c5), len(best_h)
