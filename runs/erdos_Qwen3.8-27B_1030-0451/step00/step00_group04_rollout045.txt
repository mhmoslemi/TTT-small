    n = len(x)

import numpy as np
from scipy.optimize import minimize
import time

def corr_values(h):
    n = len(h)
    return np.correlate(h, 1.0 - h, mode="full") * (2.0 / n)

def c5_score(h):
    return float(np.max(corr_values(h)))

def project_box_sum(x, target):
    n = len(x)
    lo = float(np.min(x)) - 2.0
    hi = float(np.max(x)) + 2.0
    for _ in range(10):
        if np.sum(np.clip(x - lo, 0, 1)) < target:
            lo -= 4.0
        else:
            break
    for _ in range(10):
        if np.sum(np.clip(x - hi, 0, 1)) > target:
            hi += 4.0
        else:
            break
    for _ in range(60):
        mid = (lo + hi) * 0.5
        s = np.sum(np.clip(x - mid, 0, 1))
        if s > target:
            lo = mid
        else:
            hi = mid
    return np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)

def smooth_max_grad(h, tau):
    n = len(h)
    z = corr_values(h)
    zmax = float(np.max(z))
    exponents = (z - zmax) / tau
    exponents = np.clip(exponents, -40.0, 40.0)
    ev = np.exp(exponents)
    p = ev / np.sum(ev)
    F = zmax + tau * np.log(np.sum(ev))
    dx = 2.0 / n
    grad = np.zeros(n)
    n1 = n - 1
    for k in range(2 * n - 1):
        wk = p[k]
        if wk < 1e-14:
            continue
        ell = k - n1
        if ell == 0:
            grad += wk * dx * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[:L] += wk * dx * (1.0 - h[ell:])
            grad[ell:] -= wk * dx * h[:L]
        else:
            a = -ell
            L = n - a
            grad[a:] += wk * dx * (1.0 - h[:L])
            grad[:L] -= wk * dx * h[a:]
    return F, grad

def pgd_optimize(h0, tau_start, tau_end, max_iter, time_limit, target_sum):
    n = len(h0)
    h = project_box_sum(h0.copy(), target_sum)
    best_h = h.copy()
    best_score = c5_score(h)
    start_t = time.time()
    alpha = 0.3
    tau = tau_start
    stagnant = 0

    for it in range(max_iter):
        if time.time() - start_t > time_limit:
            break
        F, grad = smooth_max_grad(h, tau)
        gnorm = float(np.max(np.abs(grad)))
        if gnorm < 1e-14:
            break
        accepted = False
        a = alpha
        for _ in range(25):
            h_new = project_box_sum(h - a * grad / gnorm, target_sum)
            F_new, _ = smooth_max_grad(h_new, tau)
            if F_new < F - 1e-13:
                h = h_new
                accepted = True
                s = c5_score(h)
                if s < best_score:
                    best_score = s
                    best_h = h.copy()
                    stagnant = 0
                else:
                    stagnant += 1
                alpha = min(a * 1.1, 2.0)
                break
            a *= 0.4

        if not accepted:

            alpha *= 0.7

            if tau <= tau_end * 1.01 and stagnant > 50:
                break
        elif it % 40 == 39:
            tau = max(tau * 0.88, tau_end)
    return best_h, best_score

class SlsEpiState:
    def __init__(self, h0, target_sum, active_lags):
        self.n = len(h0)
        self.dx = 2.0 / self.n
        self.T = target_sum
        self.lags = active_lags
        self.n1 = self.n - 1
        self.h0 = h0.copy()

    def get_h(self, x):
        return x[:self.n]

    def objective(self, x):
        return x[-1]

    def objective_grad(self, x):
        g = np.zeros(self.n + 1)
        g[-1] = 1.0
        return g

    def constraints(self, x):
        h = x[:self.n]
        t = x[-1]
        z = corr_values(h)
        vals = t - z[self.lags]
        eq = np.sum(h) - self.T
        return np.append(vals, eq)

    def constraints_jac(self, x):
        h = x[:self.n]
        n = self.n
        m = len(self.lags)
        jac = np.zeros((m + 1, n + 1))
        dx = self.dx
        for idx, k in enumerate(self.lags):
            ell = k - self.n1
            if ell == 0:
                jac[idx, :n] = dx * (1.0 - 2.0 * h)
            elif ell > 0:
                L = n - ell
                jac[idx, :L] += dx * (1.0 - h[ell:])
                jac[idx, ell:] -= dx * h[:L]
            else:
                a = -ell
                L = n - a
                jac[idx, a:] += dx * (1.0 - h[:L])
                jac[idx, :L] -= dx * h[a:]
            jac[idx, -1] = 1.0
        jac[m, :n] = 1.0
        return jac

    n = len(h0)

        return h0.copy(), c5_score(h0)
    z = corr_values(h0)

    x0 = np.append(h0, t0)
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [{'type': 'ineq', 'fun': state.constraints, 'jac': state.constraints_jac}]
    
    best_h = h0.copy()
    best_score = c5_score(h0)
    
    try:
        res = minimize(
            state.objective, x0, method='SLSQP',
            jac=state.objective_grad, bounds=bounds,
            constraints=cons,
            options={'maxiter': 300, 'ftol': 1e-12}
        )
        if res.x is not None:
            h_slsq = project_box_sum(res.x[:n], target_sum)
            s_slsq = c5_score(h_slsq)
            if s_slsq < best_score:
                best_h = h_slsq
                best_score = s_slsq
    except Exception:
        pass
    return best_h, best_score
