def solve_lp(h0, n, rho, solver='ECOS'):
    """Solve the trust-region LP subproblem."""
    dx = 2.0 / n
    
    # Compute current correlations
    c = np.correlate(h0, 1 - h0, mode="full")  # length 2n-1
    c_scaled = c * dx
    
    # Compute gradients (Jacobian)
    # G[j, m] = dc_j/dh[m] at h0
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    
    for j in range(n_lags):
        k = j - (n - 1)  # offset
        if k >= 0:
            # Term 1: (1 - h0[m-k]) for m in [k, n-1]
            for m in range(k, n):
                G[j, m] += (1 - h0[m - k])
            # Term 2: -h0[m+k] for m in [0, n-1-k]
            for m in range(0, n - k):
                G[j, m] -= h0[m + k]
        else:
            kk = -k
            # Term 1: (1 - h0[m-k]) = (1 - h0[m+kk]) for m in [0, n-1-kk]
            for m in range(0, n - kk):
                G[j, m] += (1 - h0[m + kk])
            # Term 2: -h0[m+k] = -h0[m-kk] for m in [kk, n-1]
            for m in range(kk, n):
                G[j, m] -= h0[m - kk]
    
    # Set up LP
    d_var = cp.Variable(n)
    t_var = cp.Variable()
    
    constraints = [
        c_scaled[j] + G[j] @ d_var <= t_var for j in range(n_lags)

    obj = cp.Minimize(t_var)
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=solver)
    
    return d_var.value, t_var.value

import cvxpy as cp
import time

def compute_correlations(h, n):
    """Compute all correlation values and scale by dx."""
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return c * dx, c

def compute_gradients(h0, n):
    """Compute Jacobian of correlations with respect to h."""
    n_lags = 2 * n - 1
    G = np.zeros((n_lags, n))
    for j in range(n_lags):
        k = j - (n - 1)
        if k >= 0:
            G[j, k:] += (1.0 - h0[:n-k])
            G[j, :n-k] -= h0[k:]
        else:
            kk = -k
            G[j, :n-kk] += (1.0 - h0[kk:])
            G[j, kk:] -= h0[:n-kk]
    return G

def evaluate_c5(h):
    """Compute C5 value."""
    n = len(h)
    dx = 2.0 / n
    return np.max(np.correlate(h, 1.0 - h, mode="full") * dx)

def project_feasible(h, n):
    """Project h onto {0 <= h <= 1, sum(h) = n/2}."""
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint
    target = n / 2.0
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        # If we need to add, add to entries not at 1
        # If we need to subtract, subtract from entries not at 0
        if diff > 0:
            mask = h < 1.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
        else:
            mask = h > 0.0
            if np.sum(mask) > 0:
                h[mask] += diff / np.sum(mask)
                h = np.clip(h, 0, 1)
    return h

        constraints.append(c[j] + G[j] @ d_var <= t_var)
    # Box constraints
    constraints.append(d_var >= -h0)
    constraints.append(d_var <= 1.0 - h0)
    # Sum constraint
    constraints.append(cp.sum(d_var) == 0)
    # Trust region
    if rho > 0:
        constraints.append(d_var >= -rho)
        constraints.append(d_var <= rho)
    
    prob = cp.Problem(cp.Minimize(t_var), constraints)
    
    try:

        if d_var.value is not None:
