def solve_lp_trust_region(h0, rho):
    n = len(h0)
    corr0 = np.correlate(h0, 1-h0, mode="full")
    J = compute_nabla(h0)
    
    # Variables: x = [h[0], ..., h[n-1], tau]
    # Minimize tau
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Constraints:
    # 1. J @ h - tau <= corr0 - J @ h0  (for each k)
    #    [J, -I] @ x <= corr0 - J @ h0
    A1 = np.hstack([J, -np.eye(2*n-1)])
    b1 = corr0 - J @ h0
    
    # 2. -h[j] <= 0 → [-e_j] @ x <= 0
    # 3. h[j] <= 1 → [e_j] @ x <= 1
    A2 = np.hstack([-np.eye(n), np.zeros((n,1))])
    b2 = np.zeros(n)
    A3 = np.hstack([np.eye(n), np.zeros((n,1))])
    b3 = np.ones(n)
    
    # 4. sum(h) = n/2
    Aeq = np.hstack([np.ones(n), 0.0]).reshape(1, -1)
    beq = np.array([n/2.0])
    
    # 5. Trust region: h[j] - h0[j] <= rho → [e_j] @ x <= h0[j] + rho
    #    -(h[j] - h0[j]) <= rho → [-e_j] @ x <= -h0[j] + rho
    A4 = np.hstack([np.eye(n), np.zeros((n,1))])
    b4 = h0 + rho
    A5 = np.hstack([-np.eye(n), np.zeros((n,1))])
    b5 = -h0 + rho
    
    A_ub = np.vstack([A1, A2, A3, A4, A5])
    b_ub = np.concatenate([b1, b2, b3, b4, b5])
    
    bounds = [(0, 1)] * n + [(None, None)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=Aeq, b_eq=beq, 
                  bounds=bounds, method='highs')
    
    if res.status == 0:

        h_new = np.clip(h_new, 0, 1)
        # Fix sum
        diff = n/2.0 - np.sum(h_new)
        h_new += diff / n
        h_new = np.clip(h_new, 0, 1)
        return h_new
    return None

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_nabla(h0):
    n = len(h0)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for j in range(n):
        # Term 1: (1-h0[j-m]) for m where 0 <= j-m < n
        # m ranges from j-(n-1) to j
        m_vals = np.arange(j - (n-1), j + 1)
        k_idx = m_vals + (n - 1)
        J[k_idx, j] += (1.0 - h0[j - m_vals])
        # Term 2: -h0[j+m] for m where 0 <= j+m < n
        # m ranges from -j to n-1-j
        m_vals2 = np.arange(-j, n - j)
        k_idx2 = m_vals2 + (n - 1)
        J[k_idx2, j] += (-h0[j + m_vals2])
    return J

def solve_lp(h0, rho):
    n = len(h0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    J = compute_nabla(h0)
    
    # x = [h(0), ..., h(n-1), tau]
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Inequality constraints: A @ x <= b
    # (1) J @ h - tau <= corr0 - J @ h0
    A1 = np.hstack([J, -np.eye(2*n-1)])
    b1 = corr0 - J @ h0
    
    # (2) -h <= 0
    A2 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b2 = np.zeros(n)
    
    # (3) h <= 1
    A3 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b3 = np.ones(n)
    
    # (4) Trust region: h - h0 <= rho
    A4 = np.hstack([np.eye(n), np.zeros((n, 1))])
    b4 = h0 + rho
    
    # (5) Trust region: h0 - h <= rho  i.e., -h <= -h0 + rho
    A5 = np.hstack([-np.eye(n), np.zeros((n, 1))])
    b5 = -h0 + rho
    
    A_ub = np.vstack([A1, A2, A3, A4, A5])
    b_ub = np.concatenate([b1, b2, b3, b4, b5])
    
    # Equality: sum(h) = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    
    if res.status == 0:
        h_new = res.x[:n].copy()
        # Clip for numerical safety
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    return None

from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_nabla(h0):
    n = len(h0)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for j in range(n):
        # Term 1: (1-h0[j-m]) for m where 0 <= j-m < n
        # m ranges from j-(n-1) to j
        m1 = np.arange(j - (n-1), j + 1)
        k1 = m1 + (n - 1)
        J[k1, j] += (1.0 - h0[j - m1])
        # Term 2: -h0[j+m] for m where 0 <= j+m < n
        # m ranges from -j to n-1-j
        m2 = np.arange(-j, n - j)
        k2 = m2 + (n - 1)
        J[k2, j] += (-h0[j + m2])
    return J

def lp_solve(h0, rho):
    n = len(h0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    J = compute_nabla(h0)
    
    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0
    
    # Build constraints
    n_lags = 2 * n - 1
    A1 = np.hstack([J, -np.eye(n_lags)])
    b1 = corr0 - J @ h0
    
    I_n = np.eye(n)
    Z_n1 = np.zeros((n, 1))
    
    A_ub = np.vstack([
        A1,
        np.hstack([-I_n, Z_n1]),      # -h <= 0
        np.hstack([I_n, Z_n1]),        # h <= 1
        np.hstack([I_n, Z_n1]),        # h <= h0 + rho
        np.hstack([-I_n, Z_n1]),       # -h <= -h0 + rho
    ])
    b_ub = np.concatenate([
        b1,
        np.zeros(n),
        np.ones(n),
        h0 + rho,
        -h0 + rho,
    ])
    
    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])
    
    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]
    
    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        h_new = res.x[:n].copy()
        # Ensure exact feasibility (numerical safety)
        h_new = np.clip(h_new, 0.0, 1.0)
        return h_new
    return None

def lp_optimize(h0, max_iters=500, time_budget=1000.0, start_time=None):
    """LP trust region optimization."""
    n = len(h0)
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    rho = 0.25
    fails = 0
    
    for it in range(max_iters):
        if start_time is not None and time.time() - start_time > time_budget - 30:
            break
        
        h_new = lp_solve(h, rho)
        if h_new is None:
            rho *= 0.5
            fails += 1
            if fails > 10:
                break
            continue
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5 - 1e-10:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            fails = 0
            # Slightly increase rho for exploration
            if it > 0 and it % 15 == 0:
                rho = min(rho * 1.1, 0.4)
        else:
            fails += 1
            if fails >= 3:
                rho *= 0.5
                fails = 0
                if rho < 0.005:
                    break
    
    return best_h, best_c5

def fix_sum(h, target=None):
    """Ensure sum(h) = target (default n/2) while keeping h in [0,1]."""
    n = len(h)
    if target is None:
        target = n / 2.0
    h = h.copy()
    for _ in range(50):

        if abs(diff) < 1e-12:

        if mask.any():

        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    """Upsample h from len(h) to n_new points using linear interpolation."""
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n_new / 2.0)
    return h_new

            else:

def find_best_swap(h, m):
    """Find the swap (i,j) that most reduces c_m for 0-1 vector h."""
    n = len(h)
    ones = np.where(h == 1.0)[0]
    zeros = np.where(h == 0.0)[0]
    
    best_delta = 0
    best_pair = None
    
    # Score for each i (1→0): h[i+m] + h[i-m] (we want this to be small)
    # Score for each j (0→1): h[j+m] + h[j-m] (we want this to be large)
    
    i_scores = np.zeros(len(ones))
    for idx, i in enumerate(ones):
        s = 0.0
        if 0 <= i + m < n:
            s += h[i + m]
        if 0 <= i - m < n:
            s += h[i - m]
        i_scores[idx] = s
    
    j_scores = np.zeros(len(zeros))
    for idx, j in enumerate(zeros):
        s = 0.0
        if 0 <= j + m < n:
            s += h[j + m]
        if 0 <= j - m < n:
            s += h[j - m]
        j_scores[idx] = s
    
    # Best swap: minimize i_score - j_score
    # = minimize i_score + minimize (-j_score) = minimize i_score - j_score
    best_val = 0
    for idx_i in range(len(ones)):
        for idx_j in range(len(zeros)):
            val = i_scores[idx_i] - j_scores[idx_j]
            if val < best_val:
                best_val = val
                best_pair = (ones[idx_i], zeros[idx_j])
    
    if best_pair is not None:
        delta = best_val  # This is Δc_m
        return best_pair, delta

import numpy as np
from scipy.optimize import linprog
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_nabla(h0):
    n = len(h0)
    n_lags = 2 * n - 1
    J = np.zeros((n_lags, n))
    for j in range(n):
        m1 = np.arange(j - (n - 1), j + 1)
        k1 = m1 + (n - 1)
        J[k1, j] += (1.0 - h0[j - m1])
        m2 = np.arange(-j, n - j)
        k2 = m2 + (n - 1)
        J[k2, j] += (-h0[j + m2])
    return J

def lp_solve(h0, rho):
    n = len(h0)
    corr0 = np.correlate(h0, 1.0 - h0, mode="full")
    J = compute_nabla(h0)
    n_lags = 2 * n - 1

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    I_n = np.eye(n)
    Z_n1 = np.zeros((n, 1))

    A1 = np.hstack([J, -np.eye(n_lags)])
    b1 = corr0 - J @ h0

    A_ub = np.vstack([
        A1,
        np.hstack([-I_n, Z_n1]),
        np.hstack([I_n, Z_n1]),
        np.hstack([I_n, Z_n1]),
        np.hstack([-I_n, Z_n1]),
    ])
    b_ub = np.concatenate([
        b1,
        np.zeros(n),
        np.ones(n),
        h0 + rho,
        -h0 + rho,
    ])

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n / 2.0])

    bounds = [(0.0, 1.0)] * n + [(-np.inf, np.inf)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if res.status == 0:
        return np.clip(res.x[:n], 0.0, 1.0)
    return None

def lp_optimize(h0, max_iters=500, time_budget=1000.0, start_time=None):
    n = len(h0)
    h = h0.copy()
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    rho = 0.25
    fails = 0
    stagnation = 0

    for it in range(max_iters):
        if start_time is not None and time.time() - start_time > time_budget - 25:
            break
        h_new = lp_solve(h, rho)
        if h_new is None:
            rho *= 0.5
            fails += 1
            if fails > 15:
                break
            continue
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5 - 1e-12:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            fails = 0
            stagnation = 0
            if it > 0 and it % 20 == 0:
                rho = min(rho * 1.15, 0.4)
        else:
            fails += 1
            stagnation += 1
            if fails >= 4:
                rho *= 0.5
                fails = 0
                if rho < 0.003:
                    break
            if stagnation > 60:
                break
    return best_h, best_c5

def fix_sum(h, target=None):
    n = len(h)
    if target is None:
        target = n / 2.0
    h = h.copy()
    for _ in range(100):
        diff = target - np.sum(h)
        if abs(diff) < 1e-14:
            break
        mask = (h > 1e-12) & (h < 1.0 - 1e-12)
        if mask.any():
            h[mask] += diff / mask.sum()
        else:
            h += diff / n
        h = np.clip(h, 0.0, 1.0)
    return h

def upsample(h, n_new):
    old_n = len(h)
    x_old = np.linspace(0.0, 2.0, old_n, endpoint=False)
    x_new = np.linspace(0.0, 2.0, n_new, endpoint=False)
    h_new = np.interp(x_new, x_old, h)
    h_new = np.clip(h_new, 0.0, 1.0)
    h_new = fix_sum(h_new, n_new / 2.0)
    return h_new

def random_start(n, rng):
    h = np.zeros(n)
    half = n // 2
    if n % 2 == 0:
        ones = rng.choice(n, size=half, replace=False)
        h[ones] = 1.0
    else:
        ones = rng.choice(n, size=half, replace=False)
        h[ones] = 1.0
        h[rng.randint(n)] = 0.5

    return h
