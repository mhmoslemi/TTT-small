import numpy as np
import time
from scipy.optimize import minimize

def compute_z(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_eval(h):
    return float(np.max(compute_z(h)))

def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = compute_z(h)
    z_max = np.max(z)
    # Stable log-sum-exp
    shifted = (z - z_max) / tau
    # Clip to avoid overflow
    shifted = np.clip(shifted, -50, 50)
    exp_z = np.exp(shifted)
    p = exp_z / np.sum(exp_z)
    f = z_max + tau * np.log(np.sum(exp_z))
    
    # Gradient
    grad = np.zeros(n)
    b = 1.0 - h
    offset = n - 1
    
    for q in range(2 * n - 1):
        if p[q] < 1e-8:
            continue
        ell = q - offset
        w = p[q] * dx
        if ell == 0:
            grad += w * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += w * b[:L]
            grad[:L] -= w * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[:L] += w * b[a:]
            grad[a:] -= w * h[:L]
    
    return grad, f

def project_box_sum(x, T):
    n = len(x)
    # Bisection on lambda
    lo = np.min(x) - 1.0
    hi = np.max(x) + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = np.sum(y)
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix numerical error
    diff = T - np.sum(y)
    # Distribute diff among elements not at bounds
    free = (y > 1e-10) & (y < 1.0 - 1e-10)
    if np.any(free):
        y[free] += diff / np.sum(free)
        y = np.clip(y, 0.0, 1.0)
    return y

def pgd_optimize(h0, budget_start, time_limit, tau_init=None, tau_min=1e-5):
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.copy(), T)
    best_h = h.copy()
    best_score = corr_eval(h)
    
    z = compute_z(h)
    spread = z.max() - z.min()
    if tau_init is None:
        tau_init = max(0.01, 0.15 * spread)
    tau = tau_init
    
    alpha = 0.02
    decay = 0.85
    min_alpha = 1e-6
    
    while time.time() - budget_start < time_limit - 5:
        grad, f_smooth = smooth_max_and_grad(h, tau)
        
        # Line search
        improved = False
        for _ in range(20):
            h_new = project_box_sum(h - alpha * grad, T)
            score_new = corr_eval(h_new)
            if score_new < best_score - 1e-10:
                best_h = h_new.copy()
                best_score = score_new
            if score_new < corr_eval(h) - 1e-10 or f_smooth < corr_eval(h):
                # Accept if smooth objective decreased
                h = h_new
                improved = True
                break
            alpha *= 0.5
            if alpha < min_alpha:
                break
        
        if not improved:
            # Try smaller tau
            tau = max(tau * 0.7, tau_min)
            if tau <= tau_min * 1.01:
                break
            alpha = 0.02
        else:
            # Gradually reduce tau
            if np.random.random() < 0.1:
                tau = max(tau * 0.9, tau_min)
    
    return best_h, best_score

import time
from scipy.optimize import minimize

def compute_z(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_eval(h):
    return float(np.max(compute_z(h)))

def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = compute_z(h)
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    exp_z = np.exp(shifted)
    p = exp_z / np.sum(exp_z)
    f = z_max + tau * np.log(np.sum(exp_z))
    
    grad = np.zeros(n)
    b = 1.0 - h
    offset = n - 1
    
    for q in range(2 * n - 1):
        w = p[q]
        if w < 1e-8:
            continue
        ell = q - offset
        sw = w * dx
        if ell == 0:
            grad += sw * (1.0 - 2.0 * h)
        elif ell > 0:

            grad[ell:n] += sw * b[:L]
            grad[:L] -= sw * h[ell:n]

            L = n - a

            grad[a:n] -= sw * h[:L]

import numpy as np

def compute_z(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def corr_eval(h):
    return float(np.max(compute_z(h)))

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = compute_z(h)
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    shifted = np.clip(shifted, -60.0, 60.0)
    exp_z = np.exp(shifted)
    p = exp_z / np.sum(exp_z)
    f = z_max + tau * np.log(np.sum(exp_z))
    grad = np.zeros(n)
    b = 1.0 - h
    offset = n - 1
    for q in range(2 * n - 1):
        w = p[q]
        if w < 1e-7:
            continue
        ell = q - offset
        sw = w * dx
        if ell == 0:
            grad += sw * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:n] += sw * b[:L]
            grad[:L] -= sw * h[ell:n]
        else:
            a = -ell
            L = n - a
            grad[:L] += sw * b[a:n]
            grad[a:n] -= sw * h[:L]
    return grad, f

def project_box_sum(x, T):
    n = len(x)
    lo = float(np.min(x)) - 1.0
    hi = float(np.max(x)) + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x - mid, 0.0, 1.0))
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - np.sum(y)
    if abs(diff) > 1e-10:
        free = (y > 1e-9) & (y < 1.0 - 1e-9)
        nf = int(np.count_nonzero(free))
        if nf > 0:
            y[free] += diff / nf
            y = np.clip(y, 0.0, 1.0)
    return y

def ineq_jac_full(h, active_lags=None):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    offset = n - 1
    if active_lags is None:
        active_lags = list(range(2 * n - 1))
    m = len(active_lags)
    jac = np.zeros((m, n + 1))
    jac[:, -1] = 1.0
    for idx, q in enumerate(active_lags):
        ell = q - offset
        if ell == 0:
            jac[idx, :n] = -dx * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            jac[idx, :L] += dx * h[ell:n]
            jac[idx, ell:n] -= dx * b[:L]
        else:
            a = -ell
            L = n - a
            jac[idx, :L] -= dx * b[a:n]
            jac[idx, a:n] += dx * h[:L]
    return jac

def slsqp_polish(h0, t0, maxiter=120, K=None):
    n = len(h0)
    T = n / 2.0
    dx = 2.0 / n
    z = compute_z(h0)
    if K is None:
        K = min(80, 2 * n - 1)
    top_idx = np.argsort(z)[::-1][:K]
    top_idx = sorted(top_idx)

    def obj_fun(x):
        return x[-1]

    def obj_jac(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g

    def eq_fun(x):
        return np.sum(x[:n]) - T

    def eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    def ineq_fun(x):
        h = x[:n]
        t = x[-1]
        zv = compute_z(h)
        return t - zv[list(top_idx)]

    def ineq_jac(x):
        return ineq_jac_full(x[:n], top_idx)

    x0 = np.concatenate([h0, [t0]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    constraints = [
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac},
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac}

                       constraints=constraints, method='SLSQP',
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_res = project_box_sum(res.x[:n], T)
        sc = corr_eval(h_res)
        if sc < t0:

        pass
