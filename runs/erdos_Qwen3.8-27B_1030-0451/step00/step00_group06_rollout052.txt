def evaluate_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def overlaps(h):
    dx = 2.0 / len(h)
    return np.correlate(h, 1.0 - h, mode="full") * dx

def top_k_gradient(h, k, beta):
    n = len(h)
    dx = 2.0 / n
    f = np.correlate(h, 1.0 - h, mode="full") * dx
    top_idx = np.argpartition(f, -k)[-k:]
    f_top = f[top_idx]
    w = np.exp(beta * (f_top - f_top.max()))
    w = w / w.sum()
    grad = np.zeros(n)
    offset = n - 1
    for jj in range(k):
        d = int(top_idx[jj]) - offset
        wi = w[jj] * dx
        if d >= 0:
            m = n - d
            grad[:m] += wi * (1.0 - h[d:d + m])
            grad[d:] -= wi * h[:m]
        else:
            dd = -d
            m = n - dd
            grad[dd:] += wi * (1.0 - h[:m])
            grad[:m] -= wi * h[dd:dd + m]
    return grad, f

def project_feasible(h, n):
    h = h.astype(float).copy()
    np.clip(h, 0.0, 1.0, out=h)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-14:
        return h
    if diff > 0:
        mask = h < 1.0
    else:
        mask = h > 0.0
    cnt = mask.sum()
    if cnt == 0:
        return h
    h[mask] += diff / cnt
    np.clip(h, 0.0, 1.0, out=h)
    return h
