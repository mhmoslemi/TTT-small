                t_sl_r = min(25.0, (hard_deadline - time.time()) * 0.3)
                h_r, c5_r = slsqp_iterate(h_r, max_rounds=2, time_limit=t_sl_r)
            try_update(h_r, c5_r)

    # Step 7: Final SLSQP polish on best solution
    time_remaining = hard_deadline - time.time()
    if time_remaining > 15:
        h_final, c5_final = slsqp_iterate(best_h, max_rounds=3, time_limit=min(60.0, time_remaining * 0.7))
        try_update(h_final, c5_final)

    # Final verification
    n_final = len(best_h)
    target_final = 0.5 * n_final
    best_h = project_feasible(best_h, target_final)
    final_c5 = evaluate_c5(best_h)
    
    # Ensure feasibility
    if not np.all((best_h >= -1e-15) & (best_h <= 1.0 + 1e-15)):
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = project_feasible(best_h, target_final)
        final_c5 = evaluate_c5(best_h)
    
    if not np.isfinite(final_c5):
        best_h = np.full(n_final, 0.5)
        final_c5 = evaluate_c5(best_h)

    return (best_h, final_c5, n_final)
