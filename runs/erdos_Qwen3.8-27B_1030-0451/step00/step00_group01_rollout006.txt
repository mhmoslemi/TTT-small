            best_h = _feasible_project(parent, target_sum)
            best_score = _exact_score(best_h)
    except Exception:
        best_h = None

    if best_h is None:
        n_points = 84
        target_sum = n_points / 2.0
        best_h = np.full(n_points, 0.5)
        best_score = _exact_score(best_h)

    target_sum = n_points / 2.0

    # Phase 1: DE at multiple dims and scales
    configs = [
        (10, 0.3), (16, 0.4), (20, 0.5), (24, 0.6), (28, 0.8),
        (14, 1.2), (20, 1.0), (30, 1.0), (16, 1.5), (24, 0.3)
    ]

    n_configs = len(configs)
    time_per = max(20.0, (safe_deadline - time.monotonic()) * 0.55 / n_configs)

    for dim, ds in configs:
        if time.monotonic() > safe_deadline:
            break
        remaining = safe_deadline - time.monotonic()
        if remaining < 30:
            break
        t_alloc = min(time_per, remaining * 0.4)
        try:
            B = _make_basis(n_points, dim)
            h_new, s_new, _ = _de_solve(best_h, B, dim, ds, target_sum, rng, t_alloc)
            if s_new < best_score:
                best_h = h_new
                best_score = s_new
        except Exception:
            continue

    # Phase 2: Coordinate local search for fine-tuning
    if time.monotonic() < safe_deadline:
        remaining = safe_deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(remaining * 0.2, 120)
            try:
                h_new, s_new = _coordinate_local_search(
                    best_h, target_sum, rng, t_alloc, [0.05, 0.1, 0.2, 0.4]
                )
                if s_new < best_score:
                    best_h = h_new
                    best_score = s_new
            except Exception:
                pass

    # Phase 3: Nelder-Mead refinement
    if time.monotonic() < safe_deadline:
        remaining = safe_deadline - time.monotonic()
        if remaining > 40:
            for dim, ds in [(16, 0.4), (20, 0.6), (12, 1.0)]:
                if time.monotonic() > safe_deadline:
                    break
                remaining = safe_deadline - time.monotonic()
                if remaining < 20:
                    break
                t_alloc = min(remaining * 0.35, 60)
                try:
                    B = _make_basis(n_points, dim)
                    init_c = rng.uniform(-0.4, 0.4, size=dim)
                    hr, sr, _ = _nelder_refine(best_h, B, init_c, ds, target_sum, t_alloc)
                    if hr is not None and sr < best_score:
                        best_h = hr
                        best_score = sr
                except Exception:
                    continue

    # Phase 4: Try resampling to different grid sizes
    if time.monotonic() < safe_deadline:
        remaining = safe_deadline - time.monotonic()
        if remaining > 40:
            for n_new in [100, 120, 168, 200]:
                if time.monotonic() > safe_deadline:
                    break
                if n_new > 252:
                    continue
                if n_new <= n_points:
                    continue
                try:
                    h_res = _resample(best_h, n_new)
                    ts = n_new / 2.0
                    h_res = _feasible_project(h_res, ts)
                    s_res = _exact_score(h_res)
                    if s_res < best_score:
                        best_h = h_res
                        best_score = s_res
                        n_points = n_new
                        target_sum = ts
                        # Local refine on new grid
                        if time.monotonic() < safe_deadline:
                            h2, s2 = _coordinate_local_search(
                                best_h, target_sum, rng, 20, [0.05, 0.1, 0.2]
                            )
                            if s2 < best_score:
                                best_h = h2
                                best_score = s2
                except Exception:
                    continue

    # Phase 5: Final DE burst on best configuration
    if time.monotonic() < safe_deadline:
        remaining = safe_deadline - time.monotonic()
        if remaining > 30:
            t_alloc = min(remaining * 0.6, 150)
            for dim, ds in [(20, 0.5), (24, 0.7), (16, 0.9)]:
                if time.monotonic() > safe_deadline:
                    break
                remaining = safe_deadline - time.monotonic()
                if remaining < 20:
                    break
                ta = min(t_alloc / 3.0, remaining * 0.4)
                try:
                    B = _make_basis(n_points, dim)
                    h_new, s_new, _ = _de_solve(best_h, B, dim, ds, target_sum, rng, ta)
                    if s_new < best_score:
                        best_h = h_new
                        best_score = s_new
                except Exception:
                    continue

    # Final validation
    try:
        best_h = np.asarray(best_h, dtype=np.float64)
        best_h = _feasible_project(best_h, n_points / 2.0)
        final_score = _exact_score(best_h)
        if final_score < best_score:
            best_score = final_score
    except Exception:
        best_h = np.full(84, 0.5)
        best_score = _exact_score(best_h)
        n_points = 84

    if not np.all(np.isfinite(best_h)):
        best_h = np.full(84, 0.5)
        best_score = _exact_score(best_h)
        n_points = 84

    best_h = np.clip(best_h, 0.0, 1.0)
    final_c5 = _exact_score(best_h)

    return (best_h, final_c5, n_points)
