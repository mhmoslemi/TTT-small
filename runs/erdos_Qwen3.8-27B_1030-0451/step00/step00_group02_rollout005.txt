def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    q = 1.0 - h
    full = np.correlate(h, q, mode="full")
    return float(dx * np.max(full))

def repair_vector(z, n, S):
    """Project z onto [0,1]^n with sum = S using bisection + repair."""
    z = np.clip(np.asarray(z, dtype=float), 0.0, 1.0)
    err = S - np.sum(z)
    if abs(err) < 1e-12:
        return z
    # Bisection to find mu such that sum(clip(z - mu, 0, 1)) = S
    lo = z.min() - 1.0
    hi = z.max() + 1.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        val = np.sum(np.clip(z - mid, 0.0, 1.0))
        if val > S:
            lo = mid
        else:
            hi = mid
    y = np.clip(z - (lo + hi) / 2.0, 0.0, 1.0)
    err = S - np.sum(y)
    # Final repair
    if err > 0:
        idx = np.where(y < 1.0 - 1e-14)[0]
        if len(idx) > 0:
            # Distribute err among indices with most headroom
            headroom = 1.0 - y[idx]

            for i in order:
                add = min(remaining, headroom[i])
                y[idx[i]] += add
                remaining -= add

def rank_fill(n, scores):
    """Create a binary-like vector with sum = n/2 based on score ranking."""
    S = n / 2.0
    m = int(np.floor(S + 1e-13))
    frac = S - m

    h = np.zeros(n)
    h[order[:m]] = 1.0
    if frac > 1e-13 and m < n:
        h[order[m]] = frac
    return h

def make_score_vector(n, method, phase=0, seed_val=0):
    """Generate a score vector for rank_fill."""
    idx = np.arange(n, dtype=float)
    if method == 0:  # golden ratio
        phi = (sqrt(5.0) - 1.0) / 2.0
        return np.abs(np.sin(2.0 * pi * (idx * phi + phase)))
    elif method == 1:  # sqrt2
        return np.abs(np.sin(2.0 * pi * (idx * sqrt(2.0) + phase)))
    elif method == 2:  # sqrt3
        return np.abs(np.sin(2.0 * pi * (idx * sqrt(3.0) + phase)))
    elif method == 3:  # linear + noise
        rng = np.random.RandomState(seed_val)
        return idx / n + 0.3 * rng.randn(n)
    elif method == 4:  # random
        rng = np.random.RandomState(seed_val)
        return rng.randn(n)
    elif method == 5:  # quadratic
        return np.abs(idx - n/2.0) / n + 0.1 * np.sin(4.0 * pi * idx / n + phase)
    elif method == 6:  # two-scale
        phi = (sqrt(5.0) - 1.0) / 2.0
        return np.abs(np.sin(2.0 * pi * idx * phi)) + 0.5 * np.abs(np.sin(2.0 * pi * idx * phi * phi + phase))
    else:
        rng = np.random.RandomState(seed_val + method)
        return rng.randn(n)

def interpolate_between_dims(h_old, n_old, n_new):
