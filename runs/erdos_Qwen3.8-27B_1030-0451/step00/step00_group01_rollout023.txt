def _score(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full')) * dx)

        return lambda h: evaluate_erdos_solution(h)
    except (NameError, UnboundLocalError):

def _try_pair(h, p, q, c):
    n = len(h)
    amax = min(h[p], 1.0 - h[q])
    if amax < 1e-12:
        return 0.0, None, None

    d = np.zeros(n)
    d[p] = -1.0
    d[q] = 1.0
    b = 1.0 - h

    lin = np.correlate(d, b, mode='full') - np.correlate(h, d, mode='full')
    quad = -np.correlate(d, d, mode='full')

    cm = float(np.max(c))

    ng = 17
    best_a = 0.0
    best_v = cm
    da = amax / ng
    # Vectorized-ish grid eval
    for i in range(1, ng):
        ai = i * da
        v = float(np.max(c + ai * lin + ai * ai * quad))
        if v < best_v - 1e-12:
            best_v = v
            best_a = ai

    if best_a == 0.0:
        return 0.0, None, None

    # Golden section in neighborhood
    lo = max(0.0, best_a - da)
    hi = min(amax, best_a + da)
    gap = hi - lo
    if gap < 1e-13:
        ba, bv = best_a, best_v
    else:
        invgr = 1.618033988749895
        f1 = f2 = 0.0
        for _ in range(45):
            m1 = hi - gap / invgr
            m2 = lo + gap / invgr
            f1 = float(np.max(c + m1 * lin + m1 * m1 * quad))
            f2 = float(np.max(c + m2 * lin + m2 * m2 * quad))
            if f1 < f2:
                hi = m2
            else:
                lo = m1
            gap = hi - lo
            if gap < 1e-13:
                break
        ba = (lo + hi) * 0.5
        bv = float(np.max(c + ba * lin + ba * ba * quad))
        if bv > best_v:
            ba, bv = best_a, best_v

    if bv < cm - 1e-12:
        return ba, lin, quad
    return 0.0, None, None

def _optimize(h, rng, t_end):
    n = len(h)
    c = np.correlate(h, 1.0 - h, mode='full')
    cm = float(np.max(c))
    dx = 2.0 / n
    no_imp = 0

    while no_imp < 6:
        if time.time() > t_end:
            break
        perm = rng.permutation(n)
        improved = False
        for ii in range(n):
            if time.time() > t_end:
                return h, cm * dx
            p = perm[ii]
            hp = h[p]
            if hp < 1e-10:
                continue
            for jj in range(n):
                if jj == ii:
                    continue
                q = perm[jj]
                if 1.0 - h[q] < 1e-10:
                    continue
                a, lin, quad = _try_pair(h, p, q, c)
                if a > 0:
                    h[p] -= a
                    h[q] += a
                    c = c + a * lin + a * a * quad
                    nv = float(np.max(c))
                    if nv < cm:
                        cm = nv
                        improved = True
        no_imp = 0 if improved else no_imp + 1

    # Clamp to ensure feasibility
    h = np.clip(h, 0.0, 1.0)
    # Fix sum

    s = float(np.sum(h))
    if abs(s - ts) > 1e-10:
        diff = ts - s
        if diff > 0:

def _interp(h, new_n):
    on = len(h)

    ox = (np.arange(on) + 0.5) / on
    nx = (np.arange(new_n) + 0.5) / new_n
