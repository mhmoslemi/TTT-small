import time
from scipy.optimize import minimize

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    corr = np.correlate(h, u, mode="full") * dx
    return np.max(corr)

def compute_c5_full(h):
    n = len(h)
    dx = 2.0 / n
    u = 1.0 - h
    return np.correlate(h, u, mode="full") * dx

def project_onto_constraint(v, target_sum):
    """Euclidean projection onto {0<=h<=1, sum(h)=target_sum}"""
    n = len(v)
    if target_sum < 0 or target_sum > n:
        return np.zeros(n)
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        h = np.clip(v - mid, 0.0, 1.0)
        s = h.sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)

def get_top_active_full(C, c0, n, T, K_max):
    """Get top active indices and weights"""
    M = np.max(C)
    # Find indices within T of max
    active_mask = C >= (M - T)
    active_indices = np.where(active_mask)[0]
    
    if len(active_indices) > K_max:
        # Take top K_max
        sorted_active = active_indices[np.argsort(C[active_indices])[::-1][:K_max]]
    else:
        sorted_active = active_indices
    
    # Compute weights
    diffs = M - C[sorted_active]
    weights = np.exp(-diffs / T)
    weights = weights / weights.sum()
    
    return sorted_active, weights, M

    c0 = n - 1
    grad = np.zeros(n)
    
    for idx, w in zip(active_indices, weights):
        R = abs(idx - c0)
        if R == 0:
            # Zero lag: C = dx * sum h[i]*(1-h[i])
            # dC/dh[j] = dx * (1 - 2*h[j])
            grad += w * dx * (1.0 - 2.0 * h)

                # dC/dh[j] for j in [R, n-1]: -dx * h[j-R]
                length = n - R
                grad[:length] += w * dx * (1.0 - h[R:R+length])
                grad[R:] += w * (-dx) * h[:length]
            else:
                # Negative lag R: C[c0-R] = dx * sum_{i=0}^{n-1-R} h[i+R]*(1-h[i])
                # dC/dh[j] for j in [R, n-1]: dx * (1 - h[j-R])
                # dC/dh[j] for j in [0, n-1-R]: -dx * h[j+R]
                length = n - R
                grad[R:] += w * dx * (1.0 - h[:length])
                grad[:length] += w * (-dx) * h[R:R+length]
    
    return grad

            T = max(T * 0.7, 0.005)
        elif compute_c5(h) < c5_best - 0.005:

    final_c5 = compute_c5(h_best)
    if final_c5 < c5_best:
        c5_best = final_c5
    
    return (h_best, c5_best, len(h_best))

    M = C.max()

    mask = C >= (M - T)
    idx = np.where(mask)[0]

        order = np.argsort(C[idx])[::-1][:K]
        idx = idx[order]
    # Compute weights
    diffs = M - C[idx]
    w = np.exp(-diffs / T)
