def project_to_feasible(h, target_sum):
    """Project h onto {x : 0<=x[i]<=1, sum(x)=target_sum}"""
    def sum_fn(lam):
        return np.sum(np.clip(h - lam, 0.0, 1.0))
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if sum_fn(mid) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

    return float(np.max(corr)) * dx

def compute_grad_smooth(h, n, t):
    """Compute gradient of smooth max objective. Returns (obj, grad)."""
    dx = 2.0 / n

    corr_raw = np.correlate(h, one_minus_h, mode='full')
    corr_vals = corr_raw * dx
    
    m = np.max(corr_vals)
    exp_vals = np.exp((corr_vals - m) * t)
    Z = np.sum(exp_vals)
    obj = m + np.log(Z) / t
    weights = (exp_vals / Z) * dx  # includes the dx factor

def project_to_feasible(h, target_sum):
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h - mid, 0.0, 1.0))
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(h - lam, 0.0, 1.0)

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def compute_grad_smooth(h, n, t):
    dx = 2.0 / n
    one_minus_h = 1.0 - h
    corr_vals = np.correlate(h, one_minus_h, mode='full') * dx
    m = np.max(corr_vals)
    exp_vals = np.exp((corr_vals - m) * t)
    Z = np.sum(exp_vals)
    obj = m + np.log(Z) / t
    weights = (exp_vals / Z) * dx
    grad = np.zeros(n)
    for r in range(-(n-1), n):
        w = weights[r + n - 1]
        if w < 1e-15:
            continue
        a = max(0, r)
        b = min(n, n + r)
        if a < b:
            grad[a:b] += w * one_minus_h[a-r:b-r]
        c = max(0, -r)
        d = min(n, n - r)
        if c < d:
            grad[c:d] -= w * h[c+r:d+r]
    return obj, grad
