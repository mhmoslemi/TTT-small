import numpy as np
import time

def project_box_sum(x, T):
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(60):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5*(lo+hi), 0.0, 1.0)
    diff = T - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    z = corr * dx
    zmax = z.max()
    exp_args = np.clip((z - zmax) / tau, -60.0, 0.0)
    exp_vals = np.exp(exp_args)
    sum_exp = exp_vals.sum()
    F = zmax + tau * np.log(sum_exp)
    w = exp_vals / sum_exp
    
    # Compute gradient
    grad = np.zeros(n)
    # Lag 0 (middle index = n-1)
    w0 = w[n-1]
    if w0 > 1e-12:
        grad += w0 * dx * (1.0 - 2.0 * h)
    
    # Positive lags: ell = 1, ..., n-1, corresponding to index q = n-1+ell
    for ell in range(1, n):
        wq = w[n-1+ell]
        if wq < 1e-12:
            continue
        L = n - ell
        grad[:L] += -wq * dx * h[ell:]
        grad[ell:] += wq * dx * b[:L]
    
    # Negative lags: ell = -1, ..., -(n-1), a = 1,...,n-1, corresponding to index q = n-1-a
    for a in range(1, n):
        wq = w[n-1-a]
        if wq < 1e-12:
            continue
        L = n - a
        grad[:L] += wq * dx * b[a:]
        grad[a:] += -wq * dx * h[:L]
    
    return F, grad, z, w

    best_score = evaluate_c5(h)

    spread = z.max() - z.min()
    tau = max(0.02 * spread, 5e-4)
    tau_end = 1e-5
    
    alpha = 0.5
    it = 0
    stagnant = 0
    
    while time.time() - t0 < max_time and deadline_fn():
        F, grad, z, w = smooth_max_grad(h, tau)
        
        # Periodic hard evaluation
        if it % 20 == 0:
            score = evaluate_c5(h)
            if score < best_score - 1e-12:
                best_score = score
                best_h = h.copy()
                stagnant = 0
            else:
                stagnant += 1
        
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-10:

        direction = -grad / gnorm
        
        # Backtracking line search
        accepted = False
        for ls in range(10):
            step = alpha * gnorm
            h_new = project_box_sum(h + step * direction, T)

import numpy as np

def project_box_sum(x, T):
    n = len(x)
    lo = x.min() - 1.0
    hi = x.max() + 1.0
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)
    diff = T - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    return float(np.max(np.correlate(h, 1.0 - h, mode='full') * dx))

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    b = 1.0 - h
    corr = np.correlate(h, b, mode='full')
    z = corr * dx
    zmax = z.max()
    exp_args = np.clip((z - zmax) / tau, -60.0, 0.0)
    exp_vals = np.exp(exp_args)
    sum_exp = exp_vals.sum()
    F = zmax + tau * np.log(sum_exp)
    w = exp_vals / sum_exp

    grad = np.zeros(n)
    w0 = w[n - 1]
    if w0 > 1e-14:
        grad += w0 * dx * (1.0 - 2.0 * h)

    for ell in range(1, n):
        wq = w[n - 1 + ell]
        if wq < 1e-14:
            continue
        L = n - ell
        grad[:L] -= wq * dx * h[ell:n]
        grad[ell:n] += wq * dx * b[:L]

    for a in range(1, n):
        wq = w[n - 1 - a]
        if wq < 1e-14:
            continue
        L = n - a
        grad[:L] += wq * dx * b[a:n]
        grad[a:n] -= wq * dx * h[:L]

    return F, grad, z, w
