def evaluate_c5(h):
    """Compute C5 for a given h."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr) * dx)

def project_antipalindromic(h):
    """Project h onto the anti-palindromic constraint h[i] + h[N-1-i] = 1."""
    n = len(h)
