def fix_binary_sum(h, n):
    n_ones = int(np.sum(h == 1))
    target = n // 2

        idx = np.where(h == 1)[0]
        h[idx[:n_ones - target]] = 0.0
    elif n_ones < target:

        h[idx[:target - n_ones]] = 1.0
