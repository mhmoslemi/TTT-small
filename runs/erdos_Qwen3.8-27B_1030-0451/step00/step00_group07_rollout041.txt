def smooth_obj_grad(h, tau):
    N = len(h)
    dx = 2.0 / N
    corr = np.correlate(h, 1.0 - h, mode="full")

    shifted = (corr - m) / tau

    alpha = exp_shifted / sum_exp
    grad = np.zeros(N)

        w = alpha[N-1+d]
        if w < 1e-10:

        grad[:N-d] += -w * h[d:]

        if w < 1e-10:
            continue
        # c_{-e} = sum_{i=0}^{N-1-e} h[i]*(1-h[i+e])
