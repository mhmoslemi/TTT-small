    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')

    gamma = 0.01
    c_max = np.max(c)
    w = np.exp((c - c_max) / gamma)
    w /= np.sum(w)
    
    # Compute gradient
    grad = np.zeros(n)
    center = n - 1

        if wk < 1e-10:

def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def compute_gradient(h, gamma=0.01):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    c_max = np.max(c)
    # Softmax weights
    w = np.exp((c - c_max) / gamma)
    s = np.sum(w)
    if s < 1e-300:
        w = np.zeros_like(c)
        w[int(np.argmax(c))] = 1.0
    else:
        w /= s
    
    grad = np.zeros(n)
    center = n - 1
    
    # Zero lag (r=0): c_0 = sum h[i](1-h[i]), grad = 1-2h
    wk = w[center]
    if wk > 1e-12:
        grad += wk * (1.0 - 2.0 * h)
    
    # Non-zero lags
    for r in range(1, n):
        L = n - r
        wk_p = w[center + r]
        wk_n = w[center - r]
        
        if wk_p > 1e-12:
            # Positive lag r
            grad[r:] += wk_p * (1.0 - h[:L])
            grad[:L] -= wk_p * h[r:]
        
        if wk_n > 1e-12:
            # Negative lag r
            grad[:L] += wk_n * (1.0 - h[r:])
            grad[r:] -= wk_n * h[:L]
    
    return dx * grad

def project_feasible(h, n):
    """Project to [0,1]^n with sum = n/2"""
    h = np.array(h, dtype=np.float64)
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    diff = target - h.sum()
    if abs(diff) < 1e-10:
        return h
    # Distribute evenly among interior points
    interior = (h > 1e-6) & (h < 1.0 - 1e-6)
    cnt = interior.sum()
    if cnt > 0:
        h[interior] += diff / cnt
        h = np.clip(h, 0.0, 1.0)
    # Final fix
    diff2 = target - h.sum()
    if abs(diff2) > 1e-12:
        if diff2 > 0:
            # Need to add mass, find points with room
            for _ in range(50):
                room = 1.0 - h
                idx = np.argmax(room)
                add = min(room[idx], diff2)
                if add <= 0:
                    break
                h[idx] += add
                diff2 -= add
                if abs(diff2) < 1e-12:
                    break
        else:
            for _ in range(50):
                idx = np.argmax(h)
                sub = min(h[idx], -diff2)
                if sub <= 0:
                    break
                h[idx] -= sub
                diff2 += sub
                if abs(diff2) < 1e-12:
                    break
    return h

    n = len(h0)
    h = h0.copy()
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    
    temp = 5e-5
    temp_min = 1e-9
    cooling = 0.9995
    grad = compute_gradient(h, gamma)
    move_count = 0
    no_improve = 0
    
    while time.monotonic() < deadline:
        # Generate candidate pairs
        pairs = set()
        sorted_idx = np.argsort(grad)

        pairs.add((int(sorted_idx[0]), int(sorted_idx[-1])))
        pairs.add((int(sorted_idx[1]), int(sorted_idx[-2])))

            if a != b:

            a = int(rng.integers(n))
            b = int(rng.integers(n))

            # If ga > gb: delta < 0 (decrease h[a], increase h[b])
            # If ga < gb: delta > 0 (increase h[a], decrease h[b])
            sign = -1.0 if diff > 0 else 1.0
            
            # Bound check
            if sign > 0:
                max_d = min(1.0 - h[a], h[b])
            else:
                max_d = min(h[a], 1.0 - h[b])
            
            if max_d < 1e-12:

            best_move = None
            best_move_score = current_score

                delta = sign * frac * max_d
                nh = h.copy()
                nh[a] += delta
                nh[b] -= delta
                if nh[a] < -1e-12 or nh[a] > 1+1e-12 or nh[b] < -1e-12 or nh[b] > 1+1e-12:
                    continue
                ns = exact_score(nh)

                    best_move_score = ns
                    best_move = nh
            
            if best_move is not None:
                delta_f = best_move_score - current_score
                if delta_f < 0 or (temp > temp_min and rng.random() < math.exp(-delta_f / temp)):
                    h = best_move
                    current_score = best_move_score
                    move_count += 1
                    if current_score < best_score:
                        best_score = current_score
                        best_h = h.copy()
                        no_improve = 0
                    else:
                        no_improve += 1
                else:
                    no_improve += 1
            else:
                no_improve += 1
        
        # Recompute gradient periodically
        move_count += 1
        if move_count % recompute_every == 0:
            grad = compute_gradient(h, gamma)
            # Adapt gamma
            # If no improvement, increase gamma (smoother) or reset
            if no_improve > 200:
                if temp > temp_min * 10:
                    temp = max(temp * 0.5, temp_min)
                else:
                    # Reset to best and try different gamma
                    h = best_h.copy()
                    current_score = best_score
                    gamma *= 1.5
                    no_improve = 0
        
        # Anneal
        temp = max(temp * cooling, temp_min)
    
    return best_h, best_score
