def compute_gradient_smooth(h, tau):
    N = len(h)
    dx = 2.0 / N
    # Compute all lag values
    c = np.correlate(h, 1-h, 'full') * dx  # length 2N-1
    # Softmax weights
    c_shifted = c - c.max()  # for numerical stability
    w = np.exp(c_shifted / tau)
    w /= w.sum()
    # Now compute gradient
    grad = np.zeros(N)
    for m in range(2*N-1):
        ell = m - (N-1)  # actual lag
        wm = w[m] * dx
        if wm < 1e-15:
            continue
        # For each valid n in the correlation at this lag:
        # n ranges from max(0, -ell) to min(N-1, N-1-ell)
        n_start = max(0, -ell)
        n_end = min(N, N - ell)
        # Term 1: (1-h[n+ell]) for n in [n_start, n_end)
        # This contributes to grad[n+ell]
        idx_first = np.arange(n_start + ell, n_end + ell)
        idx_second = np.arange(n_start, n_end)
        grad[idx_first] += wm * (1 - h[idx_second])
        grad[idx_second] -= wm * h[idx_first]
    return grad
```

def project_feasible(z, N):
    target = N / 2.0
    # Binary search for mu such that sum(clip(z - mu, 0, 1)) = target
    lo, hi = z.min() - 2, z.max() + 2
    for _ in range(60):
        mu = (lo + hi) / 2
        s = np.sum(np.clip(z - mu, 0, 1))
        if s > target:
            lo = mu
        else:
            hi = mu
    mu = (lo + hi) / 2
    result = np.clip(z - mu, 0, 1)
    # Fix small numerical error in sum
    diff = target - result.sum()
    if abs(diff) > 1e-12:
        # Distribute among interior points
        interior = (result > 1e-10) & (result < 1 - 1e-10)
        if interior.sum() > 0:
            result[interior] += diff / interior.sum()
            result = np.clip(result, 0, 1)
    return result
```
