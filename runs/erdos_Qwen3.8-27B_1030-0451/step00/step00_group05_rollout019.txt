    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode="full")
    return np.max(corr) * dx

    g = 1.0 - h

def compute_gradients(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2*n-1, n))
    for L in range(-(n-1), n):
        idx = L + (n-1)
        t1 = np.zeros(n)
        if L >= 0:
            t1[:n-L] = 1.0 - h[L:]
        else:
            t1[-L:] = 1.0 - h[:n+L]
        t2 = np.zeros(n)
        if L >= 0:
            t2[L:] = -h[:n-L]
        else:
            t2[:n+L] = -h[-L:]
        grads[idx] = dx * (t1 + t2)
    return grads

def fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    free = (h > 1e-8) & (h < 1.0 - 1e-8)
    nfree = np.sum(free)
    if nfree == 0:
        if diff > 0:
            idx = np.argmin(h)
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = np.argmax(h)
            h[idx] = max(0.0, h[idx] + diff)
        return h
    h[free] += diff / nfree
    h = np.clip(h, 0.0, 1.0)
    # Second pass if needed
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(free) > 0:
            h[free] += diff / np.sum(free)
            h = np.clip(h, 0.0, 1.0)
    return h

    A_ub[:, n] = -1.0
    b_ub = -c_j

    A_eq[0, :n] = 1.0
    b_eq = [0.0]

    return h_new

import numpy as np
from scipy.optimize import linprog
import time

def c5_value(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_correlations(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def compute_gradients(h):
    n = len(h)
    dx = 2.0 / n
    grads = np.zeros((2 * n - 1, n))
    for L in range(-(n - 1), n):
        idx = L + (n - 1)
        if L >= 0:
            t1 = np.zeros(n)
            t1[:n - L] = 1.0 - h[L:]
            t2 = np.zeros(n)
            t2[L:] = -h[:n - L]
        else:
            t1 = np.zeros(n)
            t1[-L:] = 1.0 - h[:n + L]
            t2 = np.zeros(n)
            t2[:n + L] = -h[-L:]
        grads[idx] = dx * (t1 + t2)
    return grads

def fix_sum(h, target_sum):
    h = np.clip(h.copy(), 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-14:
        return h
    free = (h > 1e-9) & (h < 1.0 - 1e-9)
    nf = int(np.sum(free))
    if nf == 0:
        if diff > 0:
            idx = int(np.argmin(h))
            h[idx] = min(1.0, h[idx] + diff)
        else:
            idx = int(np.argmax(h))
            h[idx] = max(0.0, h[idx] + diff)
        return h
    h[free] += diff / nf
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) > 1e-12:
        free = (h > 1e-9) & (h < 1.0 - 1e-9)
        nf = int(np.sum(free))
        if nf > 0:
            h[free] += diff / nf
            h = np.clip(h, 0.0, 1.0)
    return h

def solve_lp(h):
    n = len(h)
    c_j = compute_correlations(h)
    grads = compute_gradients(h)

    c_obj = np.zeros(n + 1)
    c_obj[-1] = 1.0

    A_ub = np.zeros((2 * n - 1, n + 1))
    A_ub[:, :n] = grads
    A_ub[:, n] = -1.0
    b_ub = -c_j

    A_eq = np.zeros((1, n))
    A_eq[0, :] = 1.0
    b_eq = np.array([0.0])

    bounds = list(zip(-h, 1.0 - h)) + [(None, None)]

    res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                  bounds=bounds, method='highs')
    if not res.success:
        return None, float(np.max(c_j))
    return res.x[:n], float(res.x[n])

def optimize_lp(h0, max_iters=300, time_limit=60.0):
    n = len(h0)
    target_sum = n / 2.0
    h = fix_sum(h0.copy(), target_sum)
    best_c5 = c5_value(h)
    t0 = time.time()

    for _ in range(max_iters):
        if time.time() - t0 > time_limit:
            break
        d, t_opt = solve_lp(h)
        if d is None:
            break
        h_new = fix_sum(h + d, target_sum)
        c5_new = c5_value(h_new)
        if c5_new < best_c5 - 1e-12:
            best_c5 = c5_new
            h = h_new

                    best_c5 = c5_try
                    h = h_try
                    found = True
                    break
            if not found:

    return h, best_c5
