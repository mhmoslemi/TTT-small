def optimal_t(c, d, t_min, t_max):
    """Find t in [t_min, t_max] minimizing max_k (c[k] + t*d[k])"""
    # c and d are arrays of length 2n-1
    # Evaluate at endpoints
    val_min = np.max(c + t_min * d)
    val_max = np.max(c + t_max * d)
    
    # Subgradient at t: sum of d[k] for k in argmax(c + t*d)
    # Actually for minimization of max, we want the subgradient to contain 0
    # The subgradient of max_k f_k(t) at t is conv{f_k'(t) : k in argmax}
    # = [min_{k in argmax} d_k, max_{k in argmax} d_k]
    
    # Binary search for where subgradient contains 0
    lo, hi = t_min, t_max
    
    for _ in range(30):
        mid = (lo + hi) / 2
        vals = c + mid * d
        max_val = np.max(vals)
        # Active set: indices where vals is close to max
        active = np.where(vals >= max_val - 1e-12)[0]
        sub_lo = np.min(d[active])
        sub_hi = np.max(d[active])
        if sub_lo > 0:  # function is increasing, move left
            hi = mid
        elif sub_hi < 0:  # function is decreasing, move right
            lo = mid
        else:  # 0 is in subgradient, optimum
            break
    
    t_opt = (lo + hi) / 2
    return t_opt

    m1 = (ia >= 0) & (ia < n)
    m2 = (ja >= 0) & (ja < n)
    m3 = (ib >= 0) & (ib < n)
    m4 = (jb >= 0) & (jb < n)
    d[m1] += (1.0 - h[ia[m1]])
    d[m2] -= h[ja[m2]]
    d[m3] -= (1.0 - h[ib[m3]])
    d[m4] += h[jb[m4]]
    d *= dx
    lo, hi = t_min, t_max
    for _ in range(45):
        mid = (lo + hi) / 2.0
        vals = c + mid * d
        mv = np.max(vals)
        active = vals >= mv - 1e-15
        d_act = d[active]
        if np.all(d_act > 1e-16):
            hi = mid
        elif np.all(d_act < -1e-16):
            lo = mid
        else:
            break
    t_opt = (lo + hi) / 2.0
    new_max = np.max(c + t_opt * d)
    return t_opt, new_max

def local_search(h, time_limit, seed=42, max_iter=None):
    n = len(h)
    rng = np.random.default_rng(seed)
    h = h.copy().astype(np.float64)
    current_score = score_c5(h)
    c = compute_c(h)
    best_h = h.copy()
    best_score = current_score
    start_time = time.time()
    no_improve = 0
    iteration = 0
    if max_iter is None:
        max_iter = 1000000
    while time.time() - start_time < time_limit and iteration < max_iter:
        iteration += 1
        c_sorted = np.sort(c)[::-1]
        thresh = c_sorted[min(15, len(c_sorted) - 1)]
        active_mask = c >= thresh - 1e-14
        grad = np.zeros(n)
        k_active = np.where(active_mask)[0] - (n - 1)
        for k in k_active:
            j1lo = max(0, k)
            j1hi = min(n - 1, k + n - 1)
            j2lo = max(0, -k)
            j2hi = min(n - 1, n - 1 - k)
            if j1lo <= j1hi:
                idx1 = np.arange(j1lo, j1hi + 1)
                grad[idx1] += (1.0 - h[idx1 - k])
            if j2lo <= j2hi:
                idx2 = np.arange(j2lo, j2hi + 1)
                grad[idx2] -= h[idx2 + k]
        grad *= 2.0 / n
        sorted_g = np.argsort(grad)
        n_cand = min(300, n)
        best_new_score = current_score
        best_pair = None
        best_t = None
        n_grad_pairs = min(80, n_cand // 3)
        for i in range(n_grad_pairs):
            a = sorted_g[i]
            b = sorted_g[n - 1 - i]
            if a != b:
                t_opt, nm = optimal_t_for_pair(c, h, a, b)
                if t_opt is not None and nm < best_new_score:
                    best_new_score = nm
                    best_pair = (a, b)
                    best_t = t_opt
        n_rand = min(120, n_cand // 2)
        for i in range(n_rand):
            a = sorted_g[rng.integers(n)]
            b = sorted_g[rng.integers(n)]
            if a != b:
                t_opt, nm = optimal_t_for_pair(c, h, a, b)
                if t_opt is not None and nm < best_new_score:
                    best_new_score = nm
                    best_pair = (a, b)
                    best_t = t_opt
        n_adj = min(60, n_cand // 5)
        for i in range(n_adj):
            a = int(rng.integers(n - 1))
            b = a + 1
            t_opt, nm = optimal_t_for_pair(c, h, a, b)
            if t_opt is not None and nm < best_new_score:
                best_new_score = nm
                best_pair = (a, b)
                best_t = t_opt
            t_opt, nm = optimal_t_for_pair(c, h, b, a)
            if t_opt is not None and nm < best_new_score:
                best_new_score = nm
                best_pair = (b, a)
                best_t = t_opt
        n_fully_rand = min(40, n_cand // 7)
        for i in range(n_fully_rand):
            a, b = rng.integers(0, n, size=2)
            if a != b:
                t_opt, nm = optimal_t_for_pair(c, h, a, b)
                if t_opt is not None and nm < best_new_score:
                    best_new_score = nm
                    best_pair = (a, b)
                    best_t = t_opt
        if best_pair is not None and current_score - best_new_score > 1e-12:
            a, b = best_pair
            h[a] += best_t
            h[b] -= best_t
            h = np.clip(h, 0.0, 1.0)
            h = project_feasible(h)
            current_score = score_c5(h)
            c = compute_c(h)
            no_improve = 0
            if current_score < best_score:
                best_score = current_score
                best_h = h.copy()
        else:
            no_improve += 1
            if no_improve > 30:
                if time.time() - start_time < time_limit * 0.8:
                    h = project_feasible(h + rng.normal(0, 0.005, n))
                    c = compute_c(h)
                    current_score = score_c5(h)
                    no_improve = 0
                else:
                    break
    return best_h, best_score

def gen_fourier_start(n, seed):
    rng = np.random.default_rng(seed)
    m = min(10, n // 4)
    x = np.linspace(0, 2, n, endpoint=False)
    h = np.full(n, 0.5)
    for i in range(m):
        freq = int(rng.integers(1, 12))
        amp = rng.uniform(0.05, 0.35)
        phase = rng.uniform(0, 2 * np.pi)
        h += amp * np.cos(2 * np.pi * freq * x / 2.0 + phase)
    return project_feasible(h)

def gen_block_start(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    pos = 0
    while pos < n:
        blen = int(rng.integers(max(1, n // 16), n // 4 + 1))
        if rng.random() < 0.5:
            h[pos:pos + blen] = 1.0
        pos += blen
    return project_feasible(h)

def gen_sparse_start(n, seed):
    rng = np.random.default_rng(seed)
    h = np.zeros(n)
    n_ones = n // 2
    idx = rng.permutation(n)
    h[idx[:n_ones]] = 1.0
    return project_feasible(h)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng_main = np.random.default_rng(seed)
    try:
        h_init = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h_init = None
    if h_init is None or len(h_init) == 0 or not np.all(np.isfinite(h_init)):
        h_init = np.full(250, 0.5)
    n = len(h_init)
    h_init = project_feasible(h_init)
    best_h = h_init.copy()
    best_score = score_c5(h_init)
    def time_left():
        return budget_s - (time.time() - t_start)
    if time_left() < 10:
        return best_h, best_score, n
    # Phase 1: Primary local search from parent (45% budget)
    t1 = budget_s * 0.45 - (time.time() - t_start)
    if t1 > 5:
        h1, s1 = local_search(h_init, t1, seed=seed)
        if s1 < best_score:
            best_score = s1
            best_h = h1.copy()
    if time_left() < 10:
        return best_h, best_score, n
    # Phase 2: Transformed parent (20% budget)
    t2_total = budget_s * 0.20 - (time.time() - t_start)
    if t2_total > 5:
        transforms = []
        transforms.append(h_init[::-1].copy())
        transforms.append(1.0 - h_init.copy())
        transforms.append((1.0 - h_init)[::-1].copy())
        transforms.append(project_feasible(h_init + rng_main.normal(0, 0.02, n)))
        transforms.append(project_feasible(h_init + rng_main.normal(0, 0.05, n)))
        t_per = t2_total / len(transforms)
        for i, ht in enumerate(transforms):
            rem = min(t_per, time_left() - 5)
            if rem < 3:
                break
            ht = project_feasible(ht)
            h2, s2 = local_search(ht, rem, seed=seed + 1000 + i * 7)
            if s2 < best_score:
                best_score = s2
                best_h = h2.copy()
    if time_left() < 10:
        return best_h, best_score, n
    # Phase 3: Alternative constructions (20% budget)
    t3_total = budget_s * 0.20 - (time.time() - t_start)
    if t3_total > 10:
        starts = []
        starts.append(('fourier', gen_fourier_start(n, seed + 2000)))
        starts.append(('block', gen_block_start(n, seed + 3000)))
        starts.append(('sparse', gen_sparse_start(n, seed + 4000)))
        starts.append(('fourier2', gen_fourier_start(n, seed + 5000)))
        if n != 250:
            starts.append(('fourier_n250', gen_fourier_start(250, seed + 6000)))
        n_starts = len(starts)
        t_per = t3_total / n_starts
        for i, (name, hs) in enumerate(starts):
            rem = min(t_per, time_left() - 5)
            if rem < 3:
                break
            h3, s3 = local_search(hs, rem, seed=seed + 7000 + i * 13)
            if s3 < best_score:
                best_score = s3
                best_h = h3.copy()
    if time_left() < 10:
        return best_h, best_score, n
    # Phase 4: Final polish (15% budget)
    t4 = budget_s * 0.15 - (time.time() - t_start)
    if t4 > 5:
        h4, s4 = local_search(best_h, t4, seed=seed + 9999)
        if s4 < best_score:
            best_score = s4
            best_h = h4.copy()
    # Final validation
    best_h = project_feasible(best_h)
    final_score = score_c5(best_h)
    n_final = len(best_h)
    return best_h, final_score, n_final
