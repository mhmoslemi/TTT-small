import numpy as np

def exact_eval(h):
    """Compute C5 = max(np.correlate(h, 1-h, mode='full') * dx)"""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(z))

def project_box_sum(x, T):
    """Project x onto {h: 0<=h<=1, sum(h)=T}"""
    n = len(x)
    # Binary search for lambda such that sum(clip(x - lambda, 0, 1)) = T
    lo, hi = x.min() - 1, x.max() + 1
    for _ in range(80):
        mid = (lo + hi) / 2
        y = np.clip(x - mid, 0, 1)
        s = y.sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2, 0, 1)
    # Fix small numerical error
    diff = T - y.sum()
    if abs(diff) > 1e-12:
        # Spread the difference evenly among non-bound elements
        mask = (y > 1e-10) & (y < 1 - 1e-10)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0, 1)
    return y

def smooth_grad(h, tau):
    """Compute gradient of smooth max of z = np.correlate(h, 1-h, 'full') * dx"""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    # Softmax weights
    z_shifted = z - z.max()
    p = np.exp(z_shifted / tau)
    p /= p.sum()
    # Compute gradient
    grad = np.zeros(n)
    # For each lag d (from -(n-1) to n-1), index in z is d + (n-1)
    for d in range(-(n-1), n):
        w = p[d + n - 1]
        if w < 1e-15:
            continue
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            # First term: (1-h[m-d]) for m in [d, n-1]
            grad[d:] += w * dx * (1.0 - h[:n-d])
            # Second term: -h[m+d] for m in [0, n-1-d]
            grad[:n-d] -= w * dx * h[d:]
        else:
            a = -d
            # First term: (1-h[m+a]) for m in [0, n-1-a]
            grad[:n-a] += w * dx * (1.0 - h[a:])
            # Second term: -h[m-a] for m in [a, n-1]
            grad[a:] -= w * dx * h[:n-a]
    return grad

def pgd_optimize(h0, tau_start, tau_end, budget_fn, max_iter=2000):

            alpha = min(alpha * 1.1, 0.1)
        # Anneal temperature
        if it % 50 == 49:

from scipy.optimize import minimize
import time

def exact_eval(h):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(z))

def project_box_sum(x, T):
    n = len(x)
    lo, hi = -1.0, 2.0
    for _ in range(100):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    diff = T - y.sum()
    if abs(diff) > 1e-14:
        mask = (y > 1e-12) & (y < 1.0 - 1e-12)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def smooth_max_value(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    z_max = z.max()
    log_sum = np.log(np.exp((z - z_max) / tau).sum())
    return z_max + tau * log_sum

def smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    z_shifted = (z - z.max()) / tau
    p = np.exp(z_shifted)
    p /= p.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-14:
            continue
        d = idx - (n - 1)
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)

        else:

def slsqp_polish(h0, time_limit=20.0):
    """SLSQP epigraph polish"""
    n = len(h0)
    T = n / 2.0
    dx = 2.0 / n
    
    # Only use top K lags for constraints (for efficiency)
    z = np.correlate(h0, 1.0 - h0, mode='full') * dx
    c5_val = z.max()
    
    # Select active lags
    K = min(n, 2 * n - 1)
    if n <= 300:
        K = 2 * n - 1  # all lags
    else:
        K = min(200, 2 * n - 1)
    
    top_idx = np.argsort(z)[-K:]
    
    t_start = max(c5_val + 0.01, 0.05)
    
    # State: [h(0), ..., h(n-1), t]
    x0 = np.concatenate([h0, [t_start]])
    
    def objective(x):
        return x[-1]
    
    def objective_grad(x):
        g = np.zeros(n + 1)
        g[-1] = 1.0
        return g
    
    def constraints_func(x):
        h = x[:n]
        t = x[-1]
        z = np.correlate(h, 1.0 - h, mode='full') * dx
        # Return z[top_idx] - t <= 0
        return -(z[top_idx] - t)
    
    def constraints_grad(x):
        # Jacobian of -[z[top_idx] - t]
        # For each active lag, gradient of -z_d w.r.t. h is -grad_z_d (as computed in smooth_grad logic)
        h = x[:n]
        jac = np.zeros((K, n + 1))
        for i, idx in enumerate(top_idx):
            d = idx - (n - 1)
            if d == 0:
                jac[i, :n] = -dx * (1.0 - 2.0 * h)
            elif d > 0:
                jac[i, d:n] = -dx * (1.0 - h[:n - d])
                jac[i, 0:n - d] += -dx * h[d:]
            else:
                a = -d
                jac[i, 0:n - a] = -dx * (1.0 - h[a:])
                jac[i, a:n] += -dx * h[:n - a]
            jac[i, n] = 1.0  # derivative of -(z_d - t) w.r.t. t is +1
        return jac
    
    def eq_constraint(x):
        return x[:n].sum() - T
    
    def eq_constraint_grad(x):
        g = np.zeros(n + 1)
        g[:n] = 1.0
        return g
    
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    cons = [
        {'type': 'ineq', 'fun': constraints_func, 'jac': constraints_grad},
        {'type': 'eq', 'fun': eq_constraint, 'jac': eq_constraint_grad},
    ]
    
    maxiter = 100 if n <= 300 else 50
    
    t0 = time.time()
    try:
        res = minimize(objective, x0, method='SLSQP', jac=objective_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-10})
        if res.success or res.x[-1] < x0[-1]:

            sc = exact_eval(h_opt)
            if sc < exact_eval(h0) + 1e-8:
                return h_opt, sc
    except Exception:
        pass
    
    return h0, exact_eval(h0)

def _exact_eval(h):
    n = len(h)

def _project(x, T):
    n = len(x)
    lo, hi = float(x.min()) - 1.0, float(x.max()) + 1.0
    for _ in range(100):
        mid = (lo + hi) * 0.5
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) * 0.5, 0.0, 1.0)
    diff = T - y.sum()
    if abs(diff) > 1e-13:
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
    y = np.clip(y, 0.0, 1.0)
    return y

def _smooth_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode='full') * dx
    zs = (z - z.max()) / tau
    p = np.exp(zs - zs.max())
    p /= p.sum()
    grad = np.zeros(n)
    for idx in range(2 * n - 1):
        w = p[idx]
        if w < 1e-14:
            continue
        d = idx - (n - 1)
        if d == 0:
            grad += w * dx * (1.0 - 2.0 * h)
        elif d > 0:
            grad[d:] += w * dx * (1.0 - h[:n - d])
            grad[:n - d] -= w * dx * h[d:]
        else:
            a = -d
            grad[:n - a] += w * dx * (1.0 - h[a:])
            grad[a:] -= w * dx * h[:n - a]
    return grad
