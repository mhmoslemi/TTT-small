    """Exact C5 evaluation."""

def _smooth_max_value(corr, tau):
    """Log-sum-exp smooth max."""
    m = np.max(corr)
    if tau < 1e-14:
        return m, np.array([1.0 if c == m else 0.0 for c in corr]) / np.sum(corr == m)
    shifted = (corr - m) / tau
    shifted = np.clip(shifted, -40, 40)
    exp_shifted = np.exp(shifted)
    alpha = exp_shifted / np.sum(exp_shifted)
    return m + tau * np.log(np.sum(exp_shifted)), alpha

def _smooth_objective(h, tau, dx):
    corr = np.correlate(h, 1.0 - h, mode='full')
    val, _ = _smooth_max_value(corr, tau)
    return val * dx

    corr = np.correlate(h, 1.0 - h, mode='full')

        interior = (h > 1e-8) & (h < 1.0 - 1e-8)
        if np.sum(interior) > 0:
            h[interior] += residual / np.sum(interior)
        else:
            # Fallback: adjust one element
            idx = np.argmax(h) if residual < 0 else np.argmin(h)
            h[idx] += residual
    h = np.clip(h, 0.0, 1.0)
    return h

    """Upscale h from len(h) to new_n using linear interpolation."""
    old_n = len(h)
    old_x = np.linspace(0, 2, old_n+1)[:old_n]  # sample points
    # Actually samples are at (i+0.5)*dx for i=0..n-1, or i*dx for i=0..n-1?

        return _smooth_gradient(h, tau, dx)

def _slsqp_step(h0, tau, dx, maxiter=200):
    """One SLSQP optimization step."""
    n = len(h0)
    target = n / 2.0
    
    constraints = [
        {'type': 'eq', 'fun': lambda h: np.sum(h) - target,
         'jac': lambda h: np.ones(n)}
    ]
    bounds = [(0.0, 1.0)] * n
    
    obj = lambda h: _smooth_objective(h, tau, dx)
    grad = lambda h: _smooth_gradient(h, tau, dx)
    
    try:

            obj, h0, jac=grad,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': maxiter, 'ftol': 1e-12, 'disp': False}
        )

            h_new = _project_feasible(res.x, n)
            return h_new
    except Exception:
