import numpy as np
import time

def project_bounded_simplex(h, target_sum):
    n = len(h)
    lo, hi = -2.0, 2.0
    for _ in range(80):
        mid = (lo + hi) * 0.5
        s = np.clip(h - mid, 0.0, 1.0).sum()
        if s > target_sum:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) * 0.5, 0.0, 1.0)

def compute_c5(h, dx):
    corr = np.correlate(h, 1.0 - h, mode="full")
    return (corr * dx).max()

def smooth_max_grad(h, dx, beta):
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode="full")
    f = corr * dx
    f_max = f.max()
    exp_f = np.exp(beta * (f - f_max))

    p2 = np.correlate(h, w, mode="full")[n-1:2*n-1]
    grad = dx * (p1 - p2)
