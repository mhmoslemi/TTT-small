def project_to_feasible(y, n):
    """Project y onto {0 <= h <= 1, sum(h) = n/2}."""
    target = n / 2.0
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    
    def sum_at_alpha(alpha):
        return np.sum(np.clip(y + alpha, 0.0, 1.0))
    
    # Bisection for alpha
    lo = -y.max() - 1.0
    hi = 1.0 - y.min() + 1.0
    
    # Ensure bracket
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = sum_at_alpha(mid)
        if s > target:
            lo = mid
        else:
            hi = mid
    
    alpha = (lo + hi) / 2.0
    h = np.clip(y + alpha, 0.0, 1.0)
    
    # Fix residual
    residual = target - np.sum(h)
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                step = min(room[i], residual)
                if step > 0:
                    h[i] += step
                    residual -= step
                    if abs(residual) < 1e-14:
                        break
        else:
            room = h
            order = np.argsort(-room)
            for i in order:
                step = min(room[i], -residual)
                if step > 0:
                    h[i] -= step
                    residual += step
                    if abs(residual) < 1e-14:
                        break
    
    return h

def eval_c5(h):
    """Evaluate C5 exactly."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(corr) * dx)

def compute_f(h):
    """Compute all lag values f_m = dx * correlate(h, 1-h, 'full')[m]."""
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode="full")

def grad_lag(h, k, n):
    """Compute gradient of f_k with respect to h.
    k is the actual lag (can be negative or positive)."""
    g = np.zeros(n)
    r = abs(k)
    # For |k| = r:
    # g[i] = (1-h[i-r]) * I[i >= r] + (-h[i+r]) * I[i <= n-1-r]
    if r == 0:
        g[:] = (1.0 - 2.0 * h)
    else:
        # First term: (1-h[i-r]) for i >= r
        g[r:] += (1.0 - h[:n-r])
        # Second term: -h[i+r] for i <= n-1-r
        g[:n-r] += (-h[r:])
    return g

    start_time = time.time()
    h = project_to_feasible(h.copy(), n)
    best_h = h.copy()
    best_c5 = eval_c5(h)
    
    tau = tau_init
    n_active = max(30, min(n, int(0.15 * n)))
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        
        f = compute_f(h)  # shape (2n-1,)
        f_max = np.max(f)
        
        # Select top active lags
        idx = np.argsort(f)[::-1][:n_active]
        f_active = f[idx]
        
        # Log-sum-exp weights
        z = (f_active - f_active.max()) / tau
        exp_z = np.exp(z)
        w = exp_z / np.sum(exp_z)

        k_offsets = idx - (n - 1)  # actual lags

        g *= (2.0 / n)  # dx factor

def slsqp_polish(h, n, time_limit=60.0):

    h = project_to_feasible(h.copy(), n)
    c5_current = eval_c5(h)

        return x[-1]

    h_new = np.interp(x_new, x_old, h_old)

    best_h, best_c5 = try_transforms(best_h, n_par, best_c5)

    # Quick PGD at parent resolution
    t_end_phase1 = min(t0 + 80, hard_end - 200)
    if time.time() < t_end_phase1:
        rem = max(10, t_end_phase1 - time.time())
        h_pg, c5_pg = smooth_pgdescent(best_h, n_par, max_iter=1000, time_limit=rem, tau_init=0.01)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()
        # SLSQP at parent
        if time.time() < t_end_phase1:
            h_sl, c5_sl = slsqp_polish(best_h, n_par, time_limit=min(20, t_end_phase1 - time.time()))
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

    # Try random candidates at various resolutions for diversity
    resolutions = [96, 128, 160, 192, 256]
    resolutions = [r for r in resolutions if r <= 256]

    for n_res in resolutions:
        if time.time() > hard_end - 30:
            break

        # Upsample current best
        h_up = interp_to_resolution(best_h, len(best_h), n_res)
        h_up = project_to_feasible(h_up, n_res)
        c5_up = eval_c5(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()

        # PGD at this resolution
        alloc = max(15, min(150, (hard_end - time.time()) * 0.35))
        h_pg, c5_pg = smooth_pgdescent(h_up, n_res, max_iter=2000, time_limit=alloc, tau_init=0.008)
        if c5_pg < best_c5:
            best_c5 = c5_pg
            best_h = h_pg.copy()

        # SLSQP polish
        sl_rem = min(40, (hard_end - time.time()) * 0.25)
        if sl_rem > 5:
            h_sl, c5_sl = slsqp_polish(h_pg, n_res, time_limit=sl_rem)
            if c5_sl < best_c5:
                best_c5 = c5_sl
                best_h = h_sl.copy()

        # Second PGD with noise
        if time.time() < hard_end - 40:
            h_noisy = h_pg + rng.randn(n_res) * 0.03
            h_noisy = project_to_feasible(h_noisy, n_res)
            alloc2 = max(10, min(80, (hard_end - time.time()) * 0.2))
            h_pg2, c5_pg2 = smooth_pgdescent(h_noisy, n_res, max_iter=1000, time_limit=alloc2, tau_init=0.015)
            if c5_pg2 < best_c5:
                best_c5 = c5_pg2
                best_h = h_pg2.copy()
            if time.time() < hard_end - 10:
                h_sl2, c5_sl2 = slsqp_polish(h_pg2, n_res, time_limit=min(20, max(5, hard_end - time.time())))
                if c5_sl2 < best_c5:
                    best_c5 = c5_sl2
                    best_h = h_sl2.copy()

    # Random starts at medium resolution for diversity
    if time.time() < hard_end - 60:
        n_rand = min(192, 128)
        for trial in range(4):
            if time.time() > hard_end - 40:
                break
            mode = trial % 4
            h_rand = random_candidate(n_rand, rng, mode)
            alloc = max(8, min(40, (hard_end - time.time()) * 0.2))
            h_r, c5_r = smooth_pgdescent(h_rand, n_rand, max_iter=500, time_limit=alloc, tau_init=0.02)
            if c5_r < best_c5:
                best_c5 = c5_r
                best_h = h_r.copy()

    # Final SLSQP polish at best resolution
    if time.time() < hard_end - 5:
        h_fin, c5_fin = slsqp_polish(best_h, len(best_h), time_limit=min(25, max(5, hard_end - time.time())))
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_fin.copy()

    # Final validation
    best_h = project_to_feasible(best_h, len(best_h))
    best_c5 = eval_c5(best_h)

    return best_h, best_c5, len(best_h)
