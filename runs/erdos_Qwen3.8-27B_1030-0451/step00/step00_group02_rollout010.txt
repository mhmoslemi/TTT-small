def eval_c5(h):

def make_feasible(h, n):
    """Project h to [0,1]^n with sum = n/2 using bisection on the dual variable."""
    h = np.clip(h.astype(float), 0.0, 1.0)
    target = n / 2.0
    # Bisection on mu: find mu such that sum(clip(h - mu, 0, 1)) = target
    lo = -2.0
    hi = 2.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        val = np.clip(h - mid, 0.0, 1.0).sum()
        if val > target:
            lo = mid
        else:
            hi = mid
    h_new = np.clip(h - (lo + hi) / 2.0, 0.0, 1.0)
    # Fix tiny residual
    diff = target - h_new.sum()
    if abs(diff) > 1e-14:

        if mask.sum() > 0:

            h_new = np.clip(h_new, 0.0, 1.0)

        g_masked[~can_dec] = -np.inf

    """Optimize using SLSQP with log-sum-exp surrogate."""
    t0 = time.time()

    class Obj:
        def __init__(self, n, tau):
            self.n = n
            self.tau = tau
            self.dx = 2.0 / n
        def __call__(self, x):
            c = np.correlate(x, 1.0 - x, mode='full') * self.dx
            c_max = c.max()

            self.n = n
            self.tau = tau
        def __call__(self, x):
            return compute_grad(x, self.tau)

            self.target = n / 2.0
        def __call__(self, x):
            return x.sum() - self.target

        def __call__(self, x):
            return self.grad

                if new_c5 < c5 - 1e-12:

                    c5 = new_c5

            pass
        tau *= 0.5
