def soft_max_objective(h, dx, eps):
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    m = np.max(vals)
    return m + eps * np.log(np.sum(np.exp((vals - m) / eps)))

def soft_max_grad(h, dx, eps):
    n = len(h)
