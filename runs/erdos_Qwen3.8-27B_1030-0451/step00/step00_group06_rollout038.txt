import numpy as np
import time

def project_onto_simplex_with_bounds(h, S):
    """Project h onto {x: 0<=x<=1, sum(x)=S}"""
    n = len(h)
    # Binary search for lambda
    lo, hi = h.min() - 1, h.max() + 1
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.clip(h - mid, 0, 1).sum()
        if s > S:
            lo = mid
        else:
            hi = mid
    return np.clip(h - (lo + hi) / 2, 0, 1)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def gradient_at_lag(h, k):
    """Compute gradient of f(k) = dx * sum h[n+k]*(1-h[n]) with respect to h"""
    n = len(h)
    dx = 2.0 / n
    grad = np.zeros(n)
    # h[j] appears in h[n+k] when n=j-k: contribution dx*(1-h[j-k])

    # For the second term, h[j+k] must be valid: j+k < n
    # For the first term, h[j-k] must be valid: j-k >= 0

        j_start, j_end = k, n
