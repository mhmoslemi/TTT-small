def _project_box_sum(x, target):

    s = x.sum()
    if s == target: return x
    # bisection tau such that sum(clip(x - tau,0,1))=target

    lo = x.min() - 1.0
    hi = x.max() + 1.0
    # function monotone decreasing in tau
    for _ in range(60):

       val = np.clip(x - mid, 0,1).sum()
       if val > target: lo=mid
       else: hi=mid
    tau=(lo+hi)/2

import numpy as np
import time
import math


def _next_pow2(m):
    m = int(m)
    if m <= 1:
        return 1
    return 1 << (m - 1).bit_length()


def _project_box_sum(x, target):
    x = np.asarray(x, dtype=float)
    if x.size == 0:
        return x
    x = np.clip(x, 0.0, 1.0)
    s = float(x.sum())
    if abs(s - target) <= 1e-13:
        return x
    d = target - s
    if x.min() > 1e-9 and x.max() < 1.0 - 1e-9:
        cand = x + d / x.size
        if cand.min() >= 0.0 and cand.max() <= 1.0:
            return cand
    lo = float(x.min()) - 1.0
    hi = float(x.max()) + 1.0
    for _ in range(50):
        mid = 0.5 * (lo + hi)
        val = float(np.clip(x - mid, 0.0, 1.0).sum())
        if val > target:
            lo = mid
        else:
            hi = mid
    tau = 0.5 * (lo + hi)
    return np.clip(x - tau, 0.0, 1.0)


def _make_solver(n):
    n = int(n)
    Nt = _next_pow2(2 * n - 1)
    Ng = _next_pow2(3 * n - 2)
    ones = np.ones(n)
    return {"n": n, "Nt": Nt, "Ng": Ng, "fones": np.fft.rfft(ones, Ng)}


def _compute_t(h, solver):
    h = np.asarray(h, dtype=float)
    n = h.size
    N = solver["Nt"]
    f = np.fft.rfft(h, N)
    ac = np.fft.irfft(f * np.conj(f), N)[:n]
    pref = np.empty(n + 1)
    pref[0] = 0.0
    pref[1:] = np.cumsum(h)
    total = pref[-1]
    t = np.empty(2 * n - 1)
    base = n - 1
    if n > 1:
        l = np.arange(1, n)
        t[base - l] = pref[n - l] - ac[l]
        t[base + l] = total - pref[l] - ac[l]
    t[base] = total - ac[0]
    return t


def _fast_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    solver = _make_solver(n)
    return float(_compute_t(h, solver).max() * (2.0 / n))


def _direct_c5(h):
    h = np.asarray(h, dtype=float)
    n = h.size
    if n == 0:
        return 0.0
    return float(np.max(np.correlate(h, 1.0 - h, "full")) * (2.0 / n))


def _weights_soft(t, beta):
    m = t.size
    if m == 1:
        return np.ones(1)
    maxt = float(t.max())
    z = beta * (t - maxt)
    e = np.exp(z)
    s = float(e.sum())
    if s <= 0.0:
        e = np.zeros(m)
        e[int(np.argmax(t))] = 1.0
        s = 1.0
    return e / s


def _weights_active(t, k):
    m = t.size
    kk = int(k)
    if kk <= 0 or kk >= m:
        return np.ones(m) / float(m)
    idx = np.argpartition(t, m - kk)[m - kk:]
    w = np.zeros(m)
    w[idx] = 1.0 / float(kk)
    return w


def _weighted_gradient_direct(h, w_t):
    h = np.asarray(h, dtype=float)
    n = h.size
    base = n - 1
    G = np.zeros(n)
    nz = np.flatnonzero(w_t)
    for idx in nz:
        wk = float(w_t[idx])
        if wk == 0.0:
            continue
        k = base - idx
        if k > 0:
            m = n - k
            G[:m] += wk * (1.0 - h[k:])
            G[k:] -= wk * h[:m]
        elif k < 0:
            l = -k
            m = n - l
            G[l:] += wk * (1.0 - h[:m])
            G[:m] -= wk * h[l:]
        else:
            G += wk * (1.0 - 2.0 * h)
    G -= G.mean()
    return G


def _weighted_gradient_fft(h, w_t, solver):
    n = solver["n"]
    Ng = solver["Ng"]
    base = n - 1
    W = np.asarray(w_t, dtype=float)[::-1]
    fx = np.fft.rfft(h, Ng)
    fw = np.fft.rfft(W, Ng)
    fwr = np.fft.rfft(W[::-1], Ng)
    fones = solver["fones"]
    B = np.fft.irfft(fx * fwr, Ng)[base:base + n]
    C = np.fft.irfft(fx * fw, Ng)[base:base + n]
    A = np.fft.irfft(fones * fwr, Ng)[base:base + n]
    G = A - B - C
    G -= G.mean()
    return G


def _weighted_gradient(h, w, solver):
    w = np.asarray(w, dtype=float)
    nnz = int(np.count_nonzero(w))
    if nnz == 0:
        return np.zeros(h.size)
    if nnz <= max(6, w.size // 40):
        return _weighted_gradient_direct(h, w)
    return _weighted_gradient_fft(h, w, solver)

            no_improve = 0
            no_best = 0
            beta = min(800.0, beta * 1.5)

    c_best = _fast_c5(best_h)
    c_cur = _fast_c5(h)
    if c_cur < c_best:
        best_h = h
        best_c = c_cur
    else:
        best_c = c_best
    return best_h, best_c


def _to_binary(h, target):
    h = np.asarray(h, dtype=float)
    n = h.size
    k = int(round(target))
    k = max(0, min(n, k))
    b = np.zeros(n)
    if k > 0 and k < n:
        idx = np.argpartition(-h, k)[:k]
        b[idx] = 1.0
    elif k == n:
        b[:] = 1.0
    return b


def _is_binary(h):
    h = np.asarray(h, dtype=float)
    return bool(np.all(np.abs(h - np.round(h)) <= 1e-7))


def _binary_swap_delta(h, i, j, n, P, Nmask):
    dA = np.zeros(n)

    rlen = n - i
    if rlen > 1:
        dA[1:rlen] += h[i + 1:n]
    if i > 0:
        dA[1:i + 1] += h[i - 1::-1]

    rlen = n - j
    if rlen > 1:
        dA[1:rlen] -= h[j + 1:n]
    if j > 0:
        dA[1:j + 1] -= h[j - 1::-1]

    d = abs(i - j)
    if d >= 1:
        dA[d] -= 1.0

    dBp = P[i] - P[j]
    dBn = Nmask[j] - Nmask[i]
    return dA, dBp, dBn


def _binary_candidate_max(t, i, j, dA, dBp, dBn, base, pos_idx, neg_idx):
    m0 = float(t[base])
    m1 = float(np.max(t[pos_idx] + dBp - dA))
    m2 = float(np.max(t[neg_idx] + dBn - dA))
    return max(m0, m1, m2)


def _binary_apply_swap(h, t, i, j, dA, dBp, dBn, pos_idx, neg_idx):
    h[i], h[j] = h[j], h[i]
    t[pos_idx] += dBp - dA
    t[neg_idx] += dBn - dA


def _binary_delta_search(h, target, deadline, rng):
    n = int(h.size)
    h = _to_binary(h, target)
    h = _project_box_sum(h, target)
    solver = _make_solver(n)
    t = _compute_t(h, solver)
    t = np.asarray(t, dtype=float)
    current_t = float(t.max())
    best_h = h.copy()
    best_t = current_t
    base = n - 1
    l = np.arange(1, n)
    pos_idx = base - l
    neg_idx = base + l

    ar = np.arange(n)
    P = (l[None, :] < (n - ar)[:, None]).astype(np.int8)
    Nmask = (l[None, :] > ar[:, None]).astype(np.int8)

    remaining = max(0.0, deadline - time.time())
    if n <= 256 and remaining > 3.0:
        full_end = min(deadline, time.time() + min(60.0, max(2.0, 0.3 * remaining)))
    else:
        full_end = time.time()

    accepted = 0
    while time.time() < full_end:
        improved = False
        zeros = np.flatnonzero(h < 0.5)
        ones = np.flatnonzero(h > 0.5)
        if zeros.size == 0 or ones.size == 0:
            break
        zeros = zeros[rng.permutation(zeros.size)]
        for i in zeros:
            for j in ones:
                dA, dBp, dBn = _binary_swap_delta(h, i, j, n, P, Nmask)
                cm = _binary_candidate_max(t, i, j, dA, dBp, dBn, base, pos_idx, neg_idx)
                if cm < current_t - 1e-9:
                    _binary_apply_swap(h, t, i, j, dA, dBp, dBn, pos_idx, neg_idx)
                    current_t = cm
                    accepted += 1
                    if current_t < best_t - 1e-9:
                        best_h = h.copy()
                        best_t = current_t
                    improved = True
                    if accepted % 200 == 0:
                        t = _compute_t(h, solver)
                        current_t = float(t.max())
                    break
            if improved:
                break
        if not improved:
            break

    T = 0.0005 * n
    no_best = 0
    total_iters = 0

    while time.time() < deadline:
        total_iters += 1
        if total_iters % 2000 == 0:
            T = max(1e-7 * n, T * 0.93)
        if total_iters % 500 == 0:
            t = _compute_t(h, solver)
            current_t = float(t.max())

        zeros = np.flatnonzero(h < 0.5)
        ones = np.flatnonzero(h > 0.5)
        if zeros.size == 0 or ones.size == 0:
            break

        i = zeros[rng.integers(zeros.size)]
        j = ones[rng.integers(ones.size)]
        dA, dBp, dBn = _binary_swap_delta(h, i, j, n, P, Nmask)
        cm = _binary_candidate_max(t, i, j, dA, dBp, dBn, base, pos_idx, neg_idx)

        if cm < current_t - 1e-9:
            _binary_apply_swap(h, t, i, j, dA, dBp, dBn, pos_idx, neg_idx)
            current_t = cm
            if current_t < best_t - 1e-9:
                best_h = h.copy()
                best_t = current_t
                no_best = 0
            else:
                no_best += 1
        else:
            delta = cm - current_t
            if delta < T and rng.random() < math.exp(-delta / T):
                _binary_apply_swap(h, t, i, j, dA, dBp, dBn, pos_idx, neg_idx)
                current_t = cm
                if current_t < best_t - 1e-9:
                    best_h = h.copy()
                    best_t = current_t
                    no_best = 0
                else:
                    no_best += 1
            else:
                no_best += 1

        if total_iters % 1000 == 0:
            zeros2 = np.flatnonzero(h < 0.5)
            ones2 = np.flatnonzero(h > 0.5)
            r = int(rng.integers(2, 6))
            if zeros2.size >= r and ones2.size >= r:
                zi = zeros2[rng.choice(zeros2.size, r, replace=False)]
                oi = ones2[rng.choice(ones2.size, r, replace=False)]
                h_before = h.copy()
                t_before = t.copy()
                old_current = current_t
                h[zi] = 1.0
                h[oi] = 0.0
                t2 = _compute_t(h, solver)
                cm2 = float(t2.max())
                delta2 = cm2 - old_current
                accept = cm2 < old_current - 1e-9 or (
                    delta2 < T and rng.random() < math.exp(-delta2 / T)
                )
                if accept:
                    t = t2
                    current_t = cm2
                    if current_t < best_t - 1e-9:
                        best_h = h.copy()
                        best_t = current_t
                        no_best = 0
                    else:
                        no_best += 1
                else:
                    h[:] = h_before
                    t[:] = t_before
                    current_t = old_current
                    no_best += 1

        if no_best > 5000 or current_t > best_t + max(1.0, 10.0 * T):
            h = best_h.copy()
            t = _compute_t(h, solver)
            current_t = float(t.max())
            no_best = 0
            T = max(1e-7 * n, 0.0005 * n)

    c_best = _direct_c5(best_h)
    c_cur = _direct_c5(h)
    if c_cur < c_best:
        best_h = h
        c_best = c_cur
    return best_h, c_best


def _random_binary(n, rng):
    n = int(n)
    k = n // 2
    b = np.zeros(n)
    if k > 0:
        idx = rng.permutation(n)[:k]
        b[idx] = 1.0
    return b


def _smooth_start(n, rng):
    n = int(n)
    x = np.linspace(0.0, 1.0, n)
    a = rng.uniform(0.3, 0.8)
    b = rng.uniform(0.2, 0.7)
    f1 = int(rng.integers(2, 8))
    f2 = int(rng.integers(3, 12))
    p1 = rng.uniform(0.0, 2.0 * math.pi)
    p2 = rng.uniform(0.0, 2.0 * math.pi)
    h = 0.5 + a * np.sin(2.0 * math.pi * f1 * x + p1) + b * np.sin(2.0 * math.pi * f2 * x + p2)
    return _project_box_sum(h, n / 2.0)


def _upsample(h, n_new):
    h = np.asarray(h, dtype=float)
    n = h.size
    n_new = int(n_new)
    if n_new <= 1:
        return h.copy()[:1]
    if n_new == n:
        return h.copy()

    cands = []
    if n_new % n == 0:
        cands.append(np.repeat(h, n_new // n))
    else:
        rep = np.tile(h, n_new // n + 1)[:n_new]
        cands.append(rep)

    x_old = (np.arange(n) + 0.5) / float(n)
    x_new = (np.arange(n_new) + 0.5) / float(n_new)
    cands.append(np.interp(x_new, x_old, h))

    best = None
    bestc = np.inf
    for c in cands:
        c = np.asarray(c, dtype=float)
        c = _project_box_sum(c, n_new / 2.0)
        cc = _fast_c5(c)
        if cc < bestc:
            best = c
            bestc = cc
    return best


def _extract_initial(init):
    if init is None:
        return None
    try:
        if hasattr(init, "shape"):
            arr = np.asarray(init, dtype=float).ravel()
        elif isinstance(init, (tuple, list)) and len(init) > 0 and hasattr(init[0], "shape"):
            arr = np.asarray(init[0], dtype=float).ravel()
        else:
            arr = np.asarray(init, dtype=float).ravel()
        if arr.size >= 2:
            return arr
    except Exception:
        pass
    return None


def run(seed=42, budget_s=1100, **kwargs):
    t0 = time.time()
    end = t0 + max(1.0, float(budget_s) - 5.0)
    rng = np.random.default_rng(seed)

    h0 = _extract_initial(globals().get("initial_h_values"))
    if h0 is None or h0.size < 2:
        h0 = _random_binary(84, rng)
    else:
        h0 = np.asarray(h0, dtype=float).copy()
        h0 = np.clip(h0, 0.0, 1.0)
        if h0.size > 2400:
            h0 = _upsample(h0, 2000)
        if h0.size % 2 == 1:
            h0 = _upsample(h0, h0.size * 2)

    n0 = int(h0.size)
    target0 = n0 / 2.0
    h0 = _project_box_sum(h0, target0)

    best_h = h0.copy()
    best_c = _fast_c5(best_h)

    hb0 = _to_binary(h0, target0)
    cb0 = _fast_c5(hb0)
    if cb0 < best_c:
        best_h = hb0
        best_c = cb0

    if time.time() < end - 20.0:
        lim = min(end, time.time() + min(300.0, 0.35 * float(budget_s)))
        h1, c1 = _binary_delta_search(_to_binary(best_h, target0), target0, lim, rng)
        if c1 < best_c:
            best_h, best_c = h1, c1

    if time.time() < end - 30.0:
        lim = min(end, time.time() + min(250.0, 0.25 * float(budget_s)))
        h2, c2 = _local_optimize(h0, target0, lim, rng)
        if c2 < best_c:
            best_h, best_c = h2, c2

    if time.time() < end - 30.0 and best_h.size <= 512:
        lim = min(end, time.time() + min(120.0, 0.15 * float(budget_s)))
        hb2 = _to_binary(best_h, best_h.size / 2.0)
        h3, c3 = _binary_delta_search(hb2, best_h.size / 2.0, lim, rng)
        if c3 < best_c:
            best_h, best_c = h3, c3

    if time.time() < end - 60.0:
        for _ in range(2):
            if time.time() < end - 40.0:
                hr = _random_binary(n0, rng)
                hr, cr = _binary_delta_search(hr, target0, min(end, time.time() + 60.0), rng)
                if cr < best_c:
                    best_h, best_c = hr, cr

    if time.time() < end - 60.0:
        hs = _smooth_start(n0, rng)
        hs, cs = _local_optimize(hs, target0, min(end, time.time() + 60.0), rng)
        if cs < best_c:
            best_h, best_c = hs, cs
        if time.time() < end - 40.0:
            hb = _to_binary(hs, target0)
            hb, cb = _binary_delta_search(hb, target0, min(end, time.time() + 40.0), rng)
            if cb < best_c:
                best_h, best_c = hb, cb

    best_n = int(best_h.size)
    max_n = 1600
    for factor in (2, 4, 6, 8):
        n2 = best_n * factor
        if n2 <= best_n or n2 > max_n:
            continue
        if time.time() < end - 100.0:
            limit = min(end, time.time() + min(150.0, 0.2 * (end - time.time())))
            h_up = _upsample(best_h, n2)
            if _is_binary(best_h):
                h_up_bin = _to_binary(h_up, n2 / 2.0)
                if _fast_c5(h_up_bin) <= _fast_c5(h_up):
                    h_up = h_up_bin

            if _is_binary(h_up):
                h2, c2 = _binary_delta_search(h_up, n2 / 2.0, limit, rng)
            else:
                h2, c2 = _local_optimize(h_up, n2 / 2.0, limit, rng)
                if time.time() < end - 60.0:
                    hb2 = _to_binary(h2, n2 / 2.0)
                    hb2, cb2 = _binary_delta_search(hb2, n2 / 2.0, min(end, time.time() + 60.0), rng)
                    if cb2 < c2:
                        h2, c2 = hb2, cb2

            if c2 < best_c:
                best_h, best_c = h2, c2

    if time.time() < end - 5.0:
        target_best = best_h.size / 2.0
        if _is_binary(best_h):
            hfin, cfin = _binary_delta_search(
                _to_binary(best_h, target_best),
                target_best,
                min(end, time.time() + 60.0),
                rng
            )
        else:
            hfin, cfin = _local_optimize(best_h, target_best, min(end, time.time() + 60.0), rng)
            if time.time() < end - 20.0:
                hbd = _to_binary(hfin, hfin.size / 2.0)
                hbd, cbd = _binary_delta_search(
                    hbd,
                    hfin.size / 2.0,
                    min(end, time.time() + 20.0),
                    rng
                )
                if cbd < cfin:
                    hfin, cfin = hbd, cbd
        if cfin < best_c:
            best_h, best_c = hfin, cfin

    best_h = _project_box_sum(best_h, best_h.size / 2.0)
    c_final = _direct_c5(best_h)

    ev = globals().get("evaluate_erdos_solution")
    if callable(ev):
        try:
            cv = float(ev(best_h))
            if np.isfinite(cv):
                c_final = cv
        except Exception:
            pass

    if not np.isfinite(c_final):
        c_final = _direct_c5(best_h)

    return best_h, float(c_final), int(best_h.size)
