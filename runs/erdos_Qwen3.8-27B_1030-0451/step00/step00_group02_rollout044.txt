import numpy as np
import time

def compute_c5(h):
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1 - h, mode="full") * dx
    return float(np.max(c))

def project_onto_constraint(v, target_sum):
    """Project v onto {h: 0<=h<=1, sum(h)=target_sum}"""
    # Use bisection to find lambda
    lo, hi = -2.0, 2.0
    for _ in range(100):
        lam = (lo + hi) / 2
        p = np.clip(v - lam, 0, 1)
        s = np.sum(p)
        if s > target_sum:
            lo = lam
        else:
            hi = lam
    p = np.clip(v - (lo+hi)/2, 0, 1)
    # Adjust for any tiny numerical error
    diff = target_sum - np.sum(p)
    if abs(diff) < 1e-10:
        return p
    # Small correction
    mask = (p > 0) & (p < 1)
    if np.sum(mask) > 0:
        p[mask] += diff / np.sum(mask)
        p = np.clip(p, 0, 1)
    return p

def smoothed_objective_and_grad(h, eps, dx):
    n = len(h)
    c = np.correlate(h, 1 - h, mode="full") * dx
    scaled = c / eps
    m = np.max(scaled)
    weights = np.exp(scaled - m)
    Z = np.sum(weights)
    obj = eps * (m + np.log(Z))
    weights /= Z
    
    # Gradient
    # c_k = dx * sum_i h[i] * (1-h[i+s_k]) where s_k = k - (n-1)
    # dc_k/dh[j] = dx * [(1 - h[j+s_k]) * 1_{0<=j+s_k<n} - h[j-s_k] * 1_{0<=j-s_k<n}]
    
    grad = np.zeros(n)
    for k in range(2*n - 1):
        w = weights[k]
        if w < 1e-12:
            continue
        s = k - (n - 1)
        if s >= 0:
            # j in [0, n-1-s]: h[j+s] valid, contributes (1-h[j+s])
            # j in [s, n-1]: h[j-s] valid, contributes -h[j-s]
            grad[:n-s] += dx * w * (1 - h[s:])
            grad[s:] += -dx * w * h[:n-s]
        else:
            t = -s
            # j in [t, n-1]: h[j-t] valid (this is 1-h[j-t]), contributes (1-h[j-t])
            # j in [0, n-1-t]: h[j+t] valid (this is h[j+t]), contributes -h[j+t]
            grad[t:] += dx * w * (1 - h[:n-t])
            grad[:n-t] += -dx * w * h[t:]
    
    return obj, grad

def projected_gradient(h0, n, dx, eps_schedule, max_iter=10000, tol=1e-10):

    target_sum = n / 2.0
    best_h = h.copy()

    step_size = dx * 0.1  # initial step
    c = np.correlate(h, 1-h, mode="full") * dx

                    prev_max = c_new

                alpha *= 0.5
            else:

                pass
            
            if np.abs(alpha) < 1e-15:
                break

        if curr < best_obj - 1e-12:
            best_obj = curr
            best_h = h.copy()
    
    return best_h, best_obj
