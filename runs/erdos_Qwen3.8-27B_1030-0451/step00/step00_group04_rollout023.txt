def smooth_max_and_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    z = compute_z(h)
    z_max = np.max(z)
    shifted = (z - z_max) / tau
    shifted = np.clip(shifted, -50.0, 50.0)
    exps = np.exp(shifted)
    total = exps.sum()
    weights = exps / total
    F = z_max + tau * math.log(total)
    
    grad = np.zeros(n)
    for q in range(len(z)):
        wq = weights[q]
        if wq < 1e-8:
            continue
        ell = q - (n - 1)
        coeff = wq * dx
        if ell == 0:
            grad += coeff * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += coeff * (1.0 - h[:L])
            grad[:L] -= coeff * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[:L] += coeff * (1.0 - h[a:])
            grad[a:] -= coeff * h[:L]
    
    return F, grad

def upsample(h, factor):
    """Upsample by linear interpolation."""
    n = len(h)
    new_n = n * factor
    x_old = np.linspace(0, 2.0, n, endpoint=False)
    x_new = np.linspace(0, 2.0, new_n, endpoint=False)
    h_new = np.interp(x_new, x_old, h, period=2.0)
    return h_new

        ell = q - (n - 1)
        coeff = wq * dx
        if ell == 0:
            grad += coeff * (1.0 - 2.0 * h)
        elif ell > 0:
            L = n - ell
            grad[ell:] += coeff * (1.0 - h[:L])
            grad[:L] -= coeff * h[ell:]
        else:
            a = -ell
            L = n - a
            grad[:L] += coeff * (1.0 - h[a:])
            grad[a:] -= coeff * h[:L]
    return F, grad

def pgd_run(h0, budget_s, seed=42, tau_scale=0.08):
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.copy(), T)
    best_h = h.copy()
    best_score = eval_c5(h)
    z = compute_z(h)
    spread = z.max() - z.min()
    tau = max(0.003, tau_scale * spread)
    tau_min = 1e-5
    alpha = 3.0 / n
    alpha_max = 5.0 / n
    start_time = time.time()
    n_iter = 0
    stagnant = 0
    while time.time() - start_time < budget_s - 1.0:
        F, grad = smooth_max_and_grad(h, tau)
        gmax = np.max(np.abs(grad))
        if gmax < 1e-14:
            tau = max(tau * 0.6, tau_min)
            alpha = 3.0 / n
            stagnant += 1
            if stagnant > 50:
                break
            continue
        stagnant = 0
        d = -grad / gmax
        accepted = False
        for trial in range(14):
            a = alpha * (0.5 ** trial)
            h_new = project_box_sum(h + a * d, T)
            F_new, _ = smooth_max_and_grad(h_new, tau)
            if F_new <= F - 1e-15:
                h = h_new
                F = F_new
                accepted = True
                alpha = min(alpha * 1.05, alpha_max)
                break
        if not accepted:
            alpha *= 0.4
            if alpha < 1e-12:
                tau = max(tau * 0.6, tau_min)
                alpha = 3.0 / n
                continue
        n_iter += 1
        if n_iter % 6 == 0:
            score = eval_c5(h)
            if score < best_score - 1e-13:
                best_score = score
                best_h = h.copy()
        tau = max(tau * 0.996, tau_min)
        if tau <= tau_min * 1.01 and n_iter % 50 == 0:
            score = eval_c5(h)
            if score < best_score - 1e-13:
                best_score = score
                best_h = h.copy()
            if n_iter % 200 == 0:
                tau = max(0.003, tau_scale * 0.3 * spread)
    score = eval_c5(h)
    if score < best_score - 1e-13:
        best_score = score
        best_h = h.copy()
    return best_h, best_score

def upsample_periodic(h, factor):
    n = len(h)
    new_n = n * factor
    x = np.arange(new_n, dtype=np.float64) / factor
    x0 = np.floor(x).astype(int) % n
    x1 = (x0 + 1) % n
    frac = x - np.floor(x)
    h_new = h[x0] * (1.0 - frac) + h[x1] * frac
    return h_new

def downsample_avg(h, factor):
    n = len(h)
    old_n = n // factor
    h_r = h[:old_n * factor].reshape(old_n, factor)
    return h_r.mean(axis=1)

def slsqp_polish(h0, maxiter=300, K=50):
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.copy(), T)
    z = compute_z(h)
    c_cur = float(np.max(z))
    top_k = np.argsort(z)[-K:]
    dx = 2.0 / n

    slsqp_state = {'n': n, 'T': T, 'top_k': top_k, 'dx': dx, 'calls': 0, 'deadline': time.time() + 60}

    def _slsqp_obj(x):
        slsqp_state['calls'] += 1
        return x[n]

    def _slsqp_obj_grad(x):
        g = np.zeros(n + 1)
        g[n] = 1.0
        return g

    def _slsqp_ineq(x):
        h_x = x[:n]
        t_val = x[n]
        z_x = compute_z(h_x)
        return t_val - z_x[slsqp_state['top_k']]

    def _slsqp_ineq_jac(x):
        h_x = x[:n]
        J = np.zeros((K, n + 1))
        for i, q in enumerate(slsqp_state['top_k']):
            ell = q - (n - 1)
            if ell == 0:
                J[i, :n] = -slsqp_state['dx'] * (1.0 - 2.0 * h_x)
            elif ell > 0:
                L = n - ell
                J[i, ell:n] = -slsqp_state['dx'] * (1.0 - h_x[:L])
                J[i, :L] = -slsqp_state['dx'] * h_x[ell:]
            else:
                a = -ell
                L = n - a
                J[i, :L] = -slsqp_state['dx'] * (1.0 - h_x[a:])
                J[i, a:n] = -slsqp_state['dx'] * h_x[:L]
            J[i, n] = 1.0
        return J

    def _slsqp_eq(x):
        return np.sum(x[:n]) - slsqp_state['T']

    def _slsqp_eq_jac(x):
        j = np.zeros(n + 1)
        j[:n] = 1.0
        return j

    x0 = np.concatenate([h, [c_cur + 0.02]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    cons = [
        {'type': 'eq', 'fun': _slsqp_eq, 'jac': _slsqp_eq_jac},
        {'type': 'ineq', 'fun': _slsqp_ineq, 'jac': _slsqp_ineq_jac},
    ]
    try:
        res = minimize(_slsqp_obj, x0, method='SLSQP', jac=_slsqp_obj_grad,
                       bounds=bounds, constraints=cons,
                       options={'maxiter': maxiter, 'ftol': 1e-14})
        h_new = project_box_sum(res.x[:n], T)
        score = eval_c5(h_new)
        if score < c_cur:
            return h_new, score
    except Exception:
        pass
    return h, c_cur

def random_feasible(n, rng):
    T = n / 2.0
    h = rng.random(n) * 0.4 + 0.3
    return project_box_sum(h, T)

def binary_feasible(n, rng):
    T = n // 2
    idx = rng.permutation(n)[:T]
    h = np.zeros(n)
    h[idx] = 1.0
    return h * 0.7 + 0.15

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    try:
        h_init = np.array(initial_h_values, dtype=np.float64)
    except Exception:
        h_init = np.full(250, 0.5)
    n_init = len(h_init)
    T_init = n_init / 2.0
    h_init = project_box_sum(h_init, T_init)

    best_h = h_init.copy()
    best_score = eval_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start) - 8.0

    if time_left() <= 0:
        return (best_h, best_score, n_init)

    # Phase 1: PGD polish on parent
    alloc1 = min(time_left() * 0.35, 350.0)
    if alloc1 > 10:
        h1, s1 = pgd_run(best_h, alloc1, seed=seed)
        if s1 < best_score:
            best_h = h1
            best_score = s1

    # Phase 2: SLSQP polish
    if time_left() > 80:
        h2, s2 = slsqp_polish(best_h, maxiter=200, K=40)
        if s2 < best_score:
            best_h = h2
            best_score = s2

    # Phase 3: Upscale
    for factor in [2, 3]:
        if time_left() < 40:
            break
        n_up = len(best_h) * factor
        if n_up > 800:
            continue
        h_up = upsample_periodic(best_h, factor)
        T_up = n_up / 2.0
        h_up = project_box_sum(h_up, T_up)
        alloc_up = min(time_left() * 0.3, 250.0)
        if alloc_up < 20:
            continue
        h_up_opt, s_up = pgd_run(h_up, alloc_up, seed=seed + factor * 7)
        h_down = downsample_avg(h_up_opt, factor)
        T_orig = len(best_h) / 2.0
        h_down = project_box_sum(h_down, T_orig)
        s_down = eval_c5(h_down)
        if s_down < best_score:
            best_h = h_down
            best_score = s_down

    # Phase 4: More PGD at different tau scales
    if time_left() > 30:
        for ts in [0.03, 0.15]:
            if time_left() < 15:
                break
            h4, s4 = pgd_run(best_h, min(time_left() * 0.25, 120.0), seed=seed + 50 + int(ts * 100), tau_scale=ts)
            if s4 < best_score:
                best_h = h4
                best_score = s4

    # Phase 5: Random restarts
    rng = np.random.default_rng(seed + 777)
    if time_left() > 30:
        for attempt in range(4):
            if time_left() < 10:
                break
            noise = 0.03 * (rng.random(len(best_h)) - 0.5)
            h_pert = project_box_sum(best_h + noise, len(best_h) / 2.0)
            alloc_r = min(time_left() * 0.2, 80.0)
            if alloc_r < 5:
                break
            h_r, s_r = pgd_run(h_pert, alloc_r, seed=seed + 300 + attempt)
            if s_r < best_score:
                best_h = h_r
                best_score = s_r

    # Final: SLSQP on best
    if time_left() > 40 and len(best_h) <= 600:
        h_f, s_f = slsqp_polish(best_h, maxiter=300, K=60)
        if s_f < best_score:
            best_h = h_f
            best_score = s_f

    # Final repair
    T_final = len(best_h) / 2.0
    best_h = project_box_sum(best_h, T_final)
    best_score = eval_c5(best_h)

    return (best_h, best_score, len(best_h))
