```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_feasible(y, target_sum):
    """Project y onto {h: 0<=h_i<=1, sum(h)=target_sum}"""
    n = len(y)
    # Bisection for alpha
    lo = -y.max() - 1
    hi = 1 - y.min() + 1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        h = np.clip(y + mid, 0, 1)
        s = h.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(y + (lo + hi) / 2, 0, 1)
    # Fix any residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-12:
        # Distribute residual
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for i in order:
                step = min(room[i], residual)
                h[i] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            room = h.copy()
            order = np.argsort(-room)
            for i in order:
                step = min(room[i], -residual)
                h[i] -= step
                residual += step
                if abs(residual) < 1e-14:
                    break
    return h

def compute_correlation(h):
    """Compute full correlation of h with 1-h, times dx"""
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    """Compute C5 = max of correlation"""
    return float(np.max(compute_correlation(h)))

def gradient_for_lag(h, k, n, dx):
    """Compute gradient of f_{n-1+k} with respect to h"""
    # f_{n-1+k} = dx * sum_{j=0}^{n-1-k} h[j+k] * (1-h[j]) for k >= 0
    # f_{n-1-k} = dx * sum_{j=0}^{n-1-k} h[j] * (1-h[j+k]) for k > 0
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        # Role 1: h[j] appears as h[i+k] where i=j-k, for j in [k, n-1]
        g[k:] += (1.0 - h[:L])  # h[j] * (1-h[j-k]), j from k to n-1
        # Role 2: h[j] appears as (1-h[j]) for j in [0, n-1-k]
        g[:L] += -h[k:]
    else:

        L = n - r
        # f = dx * sum_{j=0}^{L-1} h[j] * (1-h[j+r])
        # Role 1: h[j] for j in [0, L-1]
        g[:L] += (1.0 - h[r:])
        # Role 2: h[j] appears as (1-h[j]) where j = i+r, for j in [r, n-1]
        g[r:] += -h[:L]
    return dx * g

    n = len(h)
    dx = 2.0 / n
    fmax = np.max(f[active_idx])
    z = (f[active_idx] - fmax) / tau

    z = np.maximum(z, -50)
    exp_z = np.exp(z)
    s = exp_z / exp_z.sum()

    for i, m in enumerate(active_idx):
        k = m - (n - 1)
        grad += s[i] * gradient_for_lag(h, k, n, dx)
    return grad

    """Run projected gradient descent on smoothed max objective"""
    if target_sum is None:
        target_sum = n / 2.0
    h = h0.copy()
    dx = 2.0 / n
    
    best_h = h.copy()
    best_c5 = compute_c5(h)

        current_c5 = float(np.max(f))

            h_trial = project_onto_feasible(y, target_sum)

            trial_c5 = float(np.max(f_trial))

                z_trial = (f_trial[active_idx] - fmax_trial) / tau
                z_trial = np.maximum(z_trial, -50)
                smooth_trial = fmax_trial + tau * np.log(np.exp(z_trial).sum())

                z_old = (f[active_idx] - fmax_old) / tau

                    break

                break
        else:
            step_size = min(step_size * 1.1, 2.0)

            tau *= 0.7
            no_improve = 0
        elif no_improve == 0:
            tau = max(0.0005, tau * 0.95)
    
    return best_h, best_c5
```

- p appears in h[i] when i = p, for 0 <= p <= n-1-r.

- p appears in (1-h[i+r]) when i+r = p, i.e., i = p-r, for r <= p <= n-1.

```python
def lag_gradient(h, k, n, dx):
    """Gradient of f_{n-1+k} w.r.t. h"""
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        # Term 1: dx * (1-h[p-k]) for p in [k, n-1]
        g[k:] += (1.0 - h[:L])
        # Term 2: dx * (-h[p+k]) for p in [0, L-1]
        g[:L] += -h[k:]
    else:
        r = -k
        L = n - r
        # Term 1: dx * (1-h[p+r]) for p in [0, L-1]
        g[:L] += (1.0 - h[r:])
        # Term 2: dx * (-h[p-r]) for p in [r, n-1]
        g[r:] += -h[:L]
    return dx * g
```

```python
import numpy as np
from scipy.optimize import minimize
import time

def project_onto_feasible(y, target_sum):
    """Project y onto {h: 0<=h_i<=1, sum(h)=target_sum} using bisection on alpha."""
    n = len(y)
    y = np.nan_to_num(y, nan=0.5, posinf=1.0, neginf=0.0)
    
    # Find alpha such that sum(clip(y + alpha, 0, 1)) = target_sum
    lo = -y.max() - 0.1
    hi = 1.0 - y.min() + 0.1
    
    for _ in range(80):
        mid = (lo + hi) * 0.5
        h = np.clip(y + mid, 0.0, 1.0)
        s = h.sum()
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    
    h = np.clip(y + (lo + hi) * 0.5, 0.0, 1.0)
    
    # Fix residual
    residual = target_sum - h.sum()
    if abs(residual) > 1e-13:
        if residual > 0:
            room = 1.0 - h
            order = np.argsort(-room)
            for idx in order:
                step = min(room[idx], residual)
                if step <= 0:
                    break
                h[idx] += step
                residual -= step
                if abs(residual) < 1e-14:
                    break
        else:
            room = h.copy()
            order = np.argsort(-room)
            for idx in order:
                step = min(room[idx], -residual)
                if step <= 0:
                    break
                h[idx] -= step
                residual += step
                if abs(residual) < 1e-14:
                    break
    
    return h

def compute_correlation(h):
    """Full correlation of h with (1-h), scaled by dx."""
    n = len(h)
    dx = 2.0 / n
    return dx * np.correlate(h, 1.0 - h, mode="full")

def compute_c5(h):
    return float(np.max(compute_correlation(h)))

def lag_gradient(h, k, n, dx):
    """
    Gradient of f_{n-1+k} w.r.t. h.
    k is the lag: k >= 0 means f = dx * sum h[j+k]*(1-h[j]),
    k < 0 means f = dx * sum h[j]*(1-h[j-k]).
    """
    g = np.zeros(n)
    if k >= 0:
        L = n - k
        if L <= 0:
            return g
        g[k:] += (1.0 - h[:L])
        g[:L] += (-h[k:])
    else:
        r = -k
        L = n - r
        if L <= 0:
            return g
        g[:L] += (1.0 - h[r:])
        g[r:] += (-h[:L])
    return dx * g

def interp_to_resolution(h_old, n_new):
    """Interpolate h from its resolution to n_new samples."""
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    # Linear interpolation
    x_old = np.linspace(0, 2, n_old, endpoint=False) + 1.0 / n_old  # cell centers
    x_new = np.linspace(0, 2, n_new, endpoint=False) + 1.0 / n_new
    h_new = np.interp(x_new, x_old, h_old)
    return h_new

def pgd_optimize(h0, n, target_sum, max_iters=3000, time_limit=60.0, verbose=False):
    """
    Projected gradient descent on the log-sum-exp smoothed max objective.
    Returns (best_h, best_c5).
    """
    dx = 2.0 / n
    h = h0.copy()
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    tau = 0.003
    step_size = 0.1
    no_improve_count = 0
    start = time.time()
    
    for it in range(max_iters):
        if time.time() - start > time_limit:
            break
        
        f = compute_correlation(h)
        cur_c5 = float(np.max(f))
        
        if cur_c5 < best_c5 - 1e-12:
            best_c5 = cur_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Select active lags
        fmax = cur_c5
        active_mask = f >= (fmax - max(0.002, 0.01 * tau))
        n_active = int(active_mask.sum())
        if n_active < 3:
            # At least top 3
            active_idx = np.argsort(f)[::-1][:3]
        elif n_active > 100:
            active_idx = np.argsort(f)[::-1][:100]
        else:
            active_idx = np.where(active_mask)[0]
        
        # Compute smoothed gradient
        z = (f[active_idx] - fmax) / tau
        z = np.clip(z, -50, 0)
        exp_z = np.exp(z)
        s = exp_z / exp_z.sum()
        
        grad = np.zeros(n)
        for i, m in enumerate(active_idx):
            k = m - (n - 1)
            grad += s[i] * lag_gradient(h, k, n, dx)
        
        # Project gradient onto feasible directions (orthogonal to constraint)
        grad_proj = grad - (grad.sum() / n) * np.ones(n)
        gnorm = np.linalg.norm(grad_proj)
        
        if gnorm < 1e-12:
            break
        
        direction = -grad_proj / gnorm
        
        # Backtracking line search
        accepted = False
        for trial in range(20):
            step = step_size / (2 ** trial)
            y = h + step * direction
            h_trial = project_onto_feasible(y, target_sum)
            
            f_trial = compute_correlation(h_trial)
            trial_c5 = float(np.max(f_trial))
            
            if trial_c5 < cur_c5 - 1e-13:
                h = h_trial
                accepted = True
                break
            
            # Check smoothed improvement
            if tau > 1e-4:
                fmax_t = float(np.max(f_trial[active_idx]))
                z_t = (f_trial[active_idx] - fmax_t) / tau
                z_t = np.clip(z_t, -50, 0)
                smooth_t = fmax_t + tau * np.log(np.exp(z_t).sum())
                
                z_o = (f[active_idx] - fmax) / tau
                z_o = np.clip(z_o, -50, 0)
                smooth_o = fmax + tau * np.log(np.exp(z_o).sum())
                
                if smooth_t < smooth_o - 1e-7:
                    h = h_trial
                    accepted = True
                    break
        
        if not accepted:
            step_size *= 0.5
            if step_size < 1e-10:
                break
        else:
            step_size = min(step_size * 1.3, 1.0)
        
        # Ada
        if no_improve_count > 200:
            tau *= 1.5
            no_improve_count = 0
            if tau > 0.02:
                tau = 0.003  # Reset
                step_size = 0.1
        elif no_improve_count > 50:
            tau *= 0.8
        
    return best_h, best_c5
```

```python
    start = time.time()
    
    for it in range(n_iters):
        if time.time() - start > time_limit:
            break
        
        # Pick random pair and direction
        i = rng.randint(n)
        j = rng.randint(n)
        if i == j:
            continue
        
        delta = rng.uniform(0.01, 0.05) * rng.choice([1, -1])
        
        # Try: h[i] += delta, h[j] -= delta
        h[i] += delta
        h[j] -= delta
        
        if h[i] < 0 or h[i] > 1 or h[j] < 0 or h[j] > 1:
            h[i] -= delta
            h[j] += delta
            continue
        
        c5 = compute_c5(h)
        if c5 < best_c5 - 1e-12:
            best_c5 = c5
            best_h = h.copy()
        else:
            # Revert
            h[i] -= delta
            h[j] += delta
    
    return best_h, best_c5
```
