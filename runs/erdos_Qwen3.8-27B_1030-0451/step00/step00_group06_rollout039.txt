    def f(lam):

import numpy as np
from scipy.optimize import minimize
import time


def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx


def project_onto_constraint(v, target_sum):
    n = len(v)
    if target_sum >= n:
        return np.ones(n)
    if target_sum <= 0:
        return np.zeros(n)
    lo = v.min() - 1.0
    hi = v.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(v - mid, 0, 1)) > target_sum:
            lo = mid
        else:
            hi = mid
    lam = (lo + hi) / 2.0
    return np.clip(v - lam, 0.0, 1.0)


def compute_subgradient(h, k, n):
    grad = np.zeros(n)
    if k == 0:
        grad[:] = 1.0 - 2.0 * h
    elif k > 0:
        grad[k:] += 1.0 - h[:-k]
        grad[:n - k] += -h[k:]
    else:
        j = -k
        grad[:n - j] += 1.0 - h[j:]
        grad[j:] += -h[:n - j]
    return grad


def soft_max_value(f, tau):
    m = np.max(f)
    return m + np.log(np.sum(np.exp(tau * (f - m)))) / tau


def soft_max_weights(f, tau):
    m = np.max(f)
    e = np.exp(tau * (f - m))
    return e / np.sum(e)


def obj_fn(h, tau, target_sum, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr * dx
    return soft_max_value(f, tau) + penalty * (np.sum(h) - target_sum) ** 2


def grad_fn(h, tau, target_sum, penalty):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    f = corr * dx
    w = soft_max_weights(f, tau)
    thresh = 1e-8
    active = np.where(w > thresh)[0]
    grad = np.zeros(n)
    for idx in active:
        k = idx - (n - 1)
        grad += w[idx] * compute_subgradient(h, int(k), n)
    grad += 2.0 * penalty * (np.sum(h) - target_sum)
    return grad


def make_obj(tau, target_sum, penalty):
    return lambda h: obj_fn(h, tau, target_sum, penalty)


def make_grad(tau, target_sum, penalty):
    return lambda h: grad_fn(h, tau, target_sum, penalty)


def l_bfgs_optimize(h, target_sum, max_time, tau=200, penalty=5000.0, maxiter=300):
    n = len(h)
    start = time.time()
    bounds = [(0.0, 1.0)] * n
    fn = make_obj(tau, target_sum, penalty)
    gfn = make_grad(tau, target_sum, penalty)
    try:
        res = minimize(fn, h, jac=gfn, method='L-BFGS-B', bounds=bounds,
                       options={'maxiter': maxiter, 'gtol': 1e-7, 'ftol': 1e-14,
                                'maxfun': maxiter * 2})
    except Exception:
        res = None
    if res is not None and hasattr(res, 'x'):
        return np.clip(res.x, 0, 1)
    return h


def subgradient_polish(h, target_sum, max_time, rng, alpha_base=0.01):
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start = time.time()
    for it in range(200000):
        if time.time() - start > max_time:
            break
        corr = np.correlate(h, 1 - h, mode='full')
        f = corr * dx
        c5_cur = np.max(f)
        if c5_cur < best_c5:
            best_c5 = c5_cur
            best_h = h.copy()
        thr = c5_cur - 0.001
        active = np.where(f >= thr)[0]
        if len(active) == 0:
            active = np.array([np.argmax(f)])
        idx = rng.randint(len(active))
        k = active[idx] - (n - 1)
        grad = compute_subgradient(h, int(k), n)
        t = it / max(1.0, max_time * 5.0)
        alpha = alpha_base / (1.0 + 0.05 * t)
        h_new = h - alpha * grad
        h_new = project_onto_constraint(h_new, target_sum)
        if rng.random() < 0.005:
            h_new = project_onto_constraint(h_new + rng.randn(n) * 0.03, target_sum)
        h = h_new
    return best_h, best_c5


def coord_slope(h, m, n):
    s = 0.0
    for k in range(-n + 1, n):
        if k == 0:
            a = 1.0 - 2.0 * h[m]
        elif k > 0:
            a = 0.0
            if k <= m < n:
                a += 1.0 - h[m - k]
            if 0 <= m < n - k:
                a -= h[m + k]
        else:
            j = -k
            a = 0.0
            if 0 <= m < n - j:
                a += 1.0 - h[m + j]
            if j <= m < n:
                a -= h[m - j]
        s += a
    return s


def coordinate_polish(h, max_time, rng, max_sweeps=200):
    n = len(h)
    target_sum = n / 2.0
    h = h.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    start = time.time()
    for sweep in range(max_sweeps):
        if time.time() - start > max_time:
            break
        if sweep < 50:
            order = np.arange(n)
        else:
            order = rng.permutation(n)
        improved = False
        for m in order:
            s = coord_slope(h, m, n)
            if s > 1e-10 and h[m] > 1e-10:
                h[m] = 0.0
                improved = True
            elif s < -1e-10 and h[m] < 1.0 - 1e-10:
                h[m] = 1.0
                improved = True
        if not improved:
            break
        h = project_onto_constraint(h, target_sum)
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    return best_h, best_c5


def random_restart_block(n, rng, mode):
    h = np.zeros(n)
    if mode == 0:
        h[:n // 2] = 1.0
    elif mode == 1:
        h[n // 2:] = 1.0
    elif mode == 2:
        h[:n // 4] = 1.0
        h[3 * n // 4:] = 1.0
        h[n // 4:3 * n // 4] = 0.5
    elif mode == 3:
        h[:] = 0.5
        h[:n // 2] = 1.0
    else:
        h[:] = 0.5
        h[:n // 3] = 1.0
        h[2 * n // 3:] = 1.0
    return h
