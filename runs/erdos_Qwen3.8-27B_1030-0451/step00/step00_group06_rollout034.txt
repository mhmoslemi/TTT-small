import numpy as np

def eval_c5(h, dx):
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def compute_corr(h):
    return np.correlate(h, 1.0 - h, mode="full")

def project_sum(h, target):
    """Adjust h to have exact sum = target, preserving [0,1] bounds as much as possible."""
    diff = target - h.sum()
    if abs(diff) < 1e-12:
        return h.copy()
    # Simple: add diff/n to all, then clip and redistribute
    h = h + diff / len(h)
    h = np.clip(h, 0.0, 1.0)
    # Fix remaining
    residual = target - h.sum()
    if abs(residual) > 1e-12:
        # Add/subtract from elements with room
        if residual > 0:
            room = 1.0 - h
        else:
            room = h
        total = room.sum()
        if total > 1e-12:
            h += (residual / total) * room
    return np.clip(h, 0.0, 1.0)

def grad_c5_active(h, corr, tol=1e-3):
    """Compute subgradient of C5 at active lags."""
    n = len(h)
    max_val = corr.max()

            grad[:n-sp] += (1.0 - h[sp:])

            grad[sp:] -= h[:n-sp]
