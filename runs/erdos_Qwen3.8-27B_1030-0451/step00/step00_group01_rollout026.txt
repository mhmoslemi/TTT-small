def compute_c5(h, n):
    """Compute C5 = max(correlate(h, 1-h, 'full') * dx)"""
    dx = 2.0 / n
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    # Bisection on lambda
    lo = -1.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        h = np.clip(y - mid, 0.0, 1.0)
        s = np.sum(h)
        if s > target:
            lo = mid
        else:
            hi = mid
    h = np.clip(y - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine adjustment
    diff = target - np.sum(h)
    if abs(diff) > 1e-10:
        # Distribute the difference
        if diff > 0:
            mask = h < 1.0
        else:
            mask = h > 0.0
        count = np.sum(mask)
        if count > 0:
            h[mask] += diff / count
            h = np.clip(h, 0.0, 1.0)
    return h
