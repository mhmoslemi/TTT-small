import numpy as np

def repair_profile(arr):
    arr = np.asarray(arr, dtype=np.float64).copy()
    n = len(arr)
    arr = np.clip(arr, 0.0, 1.0)
    target = n / 2.0
    diff = float(np.sum(arr)) - target
    if abs(diff) < 1e-10:
        return arr
    if diff > 0:
        order = np.argsort(-arr)
        remaining = diff
        for idx in order:
            if remaining <= 1e-12:
                break
            d = min(arr[idx], remaining)
            arr[idx] -= d
            remaining -= d
    else:
        order = np.argsort(arr)
        remaining = -diff
        for idx in order:
            if remaining <= 1e-12:
                break
            d = min(1.0 - arr[idx], remaining)
            arr[idx] += d
            remaining -= d
    arr = np.clip(arr, 0.0, 1.0)
    return arr


def exact_evaluate(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full') * dx
    return float(np.max(corr))


def compute_gradient(h, target):
    n = len(h)
    dx = 2.0 / n
    corr_raw = np.correlate(h, 1.0 - h, mode='full')
    vals = corr_raw * dx
    w = 2.0 * np.maximum(0.0, vals - target)
    if not np.any(w > 0):
        return np.zeros(n, dtype=np.float64)
    grad = np.zeros(n, dtype=np.float64)
    w0 = w[n - 1]
    if w0 > 0:
        grad += w0 * dx * (1.0 - 2.0 * h)
    for L in range(1, n):
        wp = w[n - 1 + L]
        wn = w[n - 1 - L]
        if wp > 0:
            grad[L:] += wp * dx * (1.0 - h[:n - L])
            grad[:n - L] -= wp * dx * h[L:]
        if wn > 0:
            grad[:n - L] += wn * dx * (1.0 - h[L:])
            grad[L:] -= wn * dx * h[:n - L]
    return grad


def compute_gradient_all(h, soft_width):

    return grad


def projected_step(h, direction, step_size):

    return repair_profile(h_new)


def gradient_descent(h0, target, max_iter, initial_lr, time_limit, start_time):
    h = h0.copy()
    best_h = h.copy()
    best_val = exact_evaluate(h)
    current_val = best_val
    lr = initial_lr
    stall_count = 0
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        grad = compute_gradient(h, target)
        gnorm = float(np.linalg.norm(grad))
        if gnorm < 1e-14:
            break
        improved = False
        for trial in range(10):
            step = lr * (0.5 ** trial)
            if step < 1e-12:
                break
            h_trial = projected_step(h, -grad, step)
            val = exact_evaluate(h_trial)

                improved = True
                break

            lr *= 0.4
            stall_count += 1
            if stall_count > 30:
                break
    return best_h, best_val

def swap_search(h0, n_iterations, time_limit, start_time, seed=42):

        h[i], h[j] = old_j, old_i
        new_val = exact_evaluate(h)
        delta = new_val - current_val
        if delta < -1e-13:
            current_val = new_val

            else:
                h[i], h[j] = old_i, old_j
        temp *= 0.998
    return best_h, best_val

    candidates.append(repair_profile(1.0 - h))
