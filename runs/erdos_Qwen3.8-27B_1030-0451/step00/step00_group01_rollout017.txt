import time

def project_sum_box(x, target):
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    lo, hi = -1.5, 1.5
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.clip(x + mid, 0.0, 1.0).sum()
        if s > target:
            hi = mid
        else:
            lo = mid
    lam = (lo + hi) / 2.0
    result = np.clip(x + lam, 0.0, 1.0)
    residual = target - result.sum()
    if abs(residual) > 1e-12:
        if residual > 0:
            mask = result < 1.0 - 1e-12
            slack = (1.0 - result[mask]).sum()
            if slack > 1e-12:
                result[mask] += residual / slack
        else:
            mask = result > 1e-12
            avail = result[mask].sum()
            if avail > 1e-12:
                result[mask] += residual / avail
    return np.clip(result, 0.0, 1.0)

def compute_overlaps(h):
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode="full") * dx

def c5_value(h):
    return float(np.max(compute_overlaps(h)))

def smooth_obj_and_grad(h, beta):
    n = len(h)
    dx = 2.0 / n
    overlaps = compute_overlaps(h)
    m = np.max(overlaps)
    shifted = beta * (overlaps - m)
    shifted = np.clip(shifted, -500, 0)
    exp_shifted = np.exp(shifted)
    Z = exp_shifted.sum()
    f = m + np.log(Z) / beta
    
    weights = exp_shifted / Z
    grad = np.zeros(n)
    
    n_ov = len(overlaps)
    for pos in range(n_ov):
        w = weights[pos]
        if w < 1e-14:
            continue
        k = pos - (n - 1)
        # Gradient of overlap at lag k
        if k >= 0:
            # m in [0, n-1-k]: (1-h[m+k])
            if n - k > 0:
                idx1 = np.arange(0, n - k)
                grad[idx1] += w * dx * (1.0 - h[idx1 + k])
            if k > 0:
                idx2 = np.arange(k, n)
                grad[idx2] -= w * dx * h[idx2 - k]
        else:
            ka = -k
            idx1 = np.arange(ka, n)
            grad[idx1] += w * dx * (1.0 - h[idx1 + k])
            if n + k > 0:
                idx2 = np.arange(0, n + k)
                grad[idx2] -= w * dx * h[idx2 - k]
    
    return f, grad

def slsqp_smooth_optimize(h0, beta, maxiter=200, deadline=None):
    n = len(h0)
    target = n / 2.0
    bounds = [(0.0, 1.0)] * n
    constraints = [{'type': 'eq', 'fun': lambda x: x.sum() - target, 'jac': lambda x: np.ones(n)}]
    
    def fun(x):
        return smooth_obj_and_grad(x, beta)[0]
    def jac(x):
        return smooth_obj_and_grad(x, beta)[1]
    
    res = minimize(fun, h0, jac=jac, method='SLSQP', bounds=bounds,
                   constraints=constraints, options={'maxiter': maxiter, 'ftol': 1e-12})
    
    if res.success or res.fun < smooth_obj_and_grad(h0, beta)[0]:
        result = project_sum_box(res.x, target)
        return result
    return h0

def transfer_resolution(h_old, n_new):
    """Interpolate h from n_old to n_new points"""
    n_old = len(h_old)
    x_old = np.arange(n_old, dtype=np.float64)

    h_new = np.interp(new_positions, old_positions, h_old, period=2.0)
    # Actually, the domain is [0, 2], not circular. Let me use linear interp.
    h_new = np.interp(new_positions, old_positions, h_old)
    target = n_new / 2.0
