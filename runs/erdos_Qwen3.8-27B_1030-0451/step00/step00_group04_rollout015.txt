import numpy as np

from scipy.optimize import linprog

def exact_score(h):
    n = len(h)
    if n == 0:
        return float('inf')
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_box_sum(h, n):
    """Project h to [0,1]^n with sum = n/2."""
    h = np.clip(h, 0.0, 1.0)
    target = n / 2.0
    # Water-filling: find alpha such that sum(clip(h + alpha, 0, 1)) = target
    s = np.sort(h)
    cumsum = np.cumsum(s)
    # Find the right alpha
    # sum(clip(h+alpha, 0, 1)) = sum_{h_i > -alpha} (h_i + alpha) - sum_{h_i < 1-alpha} ... 
    # Actually: clip(h_i + alpha, 0, 1) = max(0, min(1, h_i + alpha))
    # = (h_i + alpha) if 0 <= h_i + alpha <= 1
    # = 0 if h_i + alpha < 0
    # = 1 if h_i + alpha > 1
    
    # Binary search for alpha
    lo = -1.0 - h.max()
    hi = 1.0 - h.min()
    
    def f(alpha):
        return np.sum(np.clip(h + alpha, 0.0, 1.0))
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        if f(mid) < target:
            lo = mid
        else:
            hi = mid
    alpha = (lo + hi) / 2.0
    result = np.clip(h + alpha, 0.0, 1.0)
    
    # Fix any residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute residual
        if residual > 0:
            # Need to add more; find elements with room
            mask = result < 1.0
            n_add = mask.sum()
            if n_add > 0:
                result[mask] += residual / n_add
                result = np.minimum(result, 1.0)
        else:
            mask = result > 0.0
            n_sub = mask.sum()
            if n_sub > 0:
                result[mask] += residual / n_sub
                result = np.maximum(result, 0.0)
    return result

def grad_z(h, lag):
    """Gradient of correlation at given lag (lag can be negative, zero, or positive).
    lag=0 corresponds to index n-1 in the full correlation output."""
    n = len(h)
    g = np.zeros(n)
    
    if lag == 0:
        g = 1.0 - 2.0 * h
    elif lag > 0:
        L = n - lag
        # c[lag] = sum_{i=0}^{L-1} h[i] * (1-h[i+lag])
        # dc/dh[j] for j in [0, L-1]: += (1-h[j+lag])
        # dc/dh[j] for j in [lag, n-1]: -= h[j-lag]
        g[0:L] += (1.0 - h[lag:n])
        g[lag:n] -= h[0:L]
    else:
        a = -lag
        L = n - a
        # c[-a] = sum_{i=0}^{L-1} h[i+a] * (1-h[i])
        # dc/dh[j] for j in [a, n-1]: += (1-h[j-a])
        # dc/dh[j] for j in [0, L-1]: -= h[j+a]
        g[a:n] += (1.0 - h[0:L])
        g[0:L] -= h[a:n]
    
    return g

def slp_step(h, rho, K, target=0.3808):
    """One SLP step. Returns candidate y or None if failed."""
    n = len(h)
    dx = 2.0 / n
    
    # Compute full correlation
    c_full = np.correlate(h, 1.0 - h, mode="full")
    z_max = np.max(c_full)
    lag_of_max = np.argmax(c_full) - (n - 1)
    
    # Select active lags
    c_sorted = np.sort(c_full)[::-1]
    # Gap threshold
    gap = max(1e-4, 0.02 * z_max * dx)
    active_mask = c_full >= (z_max - gap / dx)
    active_indices = np.where(active_mask)[0]
    
    # Cap to K
    if len(active_indices) > K:
        # Take top K
        top_K = np.argsort(c_full)[::-1][:K]
        active_indices = top_K
    
    # Ensure the argmax is included
    if lag_of_max + (n-1) not in active_indices:
        active_indices = np.append(active_indices, lag_of_max + (n-1))
    
    # Build LP
    K_actual = len(active_indices)
    
    # Variables: y_0, ..., y_{n-1}, t  (total n+1 variables)
    # Objective: minimize t -> c_obj = [0, 0, ..., 0, 1]
    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0
    
    # Inequality constraints: A_ub @ x <= b_ub
    # For each active lag l: grad_l · y - t <= grad_l · h - z_l(h)
    # i.e.: grad_l · y + (-1)*t <= grad_l · h - z_l(h)
    
    A_ub = np.zeros((K_actual, n + 1))
    b_ub = np.zeros(K_actual)
    
    for idx, q in enumerate(active_indices):
        lag = q - (n - 1)
        g = grad_z(h, lag)
        A_ub[idx, 0:n] = g * dx  # scaled by dx since we're minimizing in same units
        A_ub[idx, n] = -1.0
        z_q = c_full[q] * dx
        b_ub[idx] = np.dot(g * dx, h) - z_q
    
    # Equality constraint: sum y_i = n/2
    A_eq = np.zeros((1, n + 1))
    A_eq[0, 0:n] = 1.0
    
    # Bounds on y_i: [max(0, h_i - rho), min(1, h_i + rho)]
    lo_bounds = np.maximum(0.0, h - rho)
    hi_bounds = np.minimum(1.0, h + rho)
    
    # Bounds on t: [0, 2.0] (generous upper bound)
    bounds = list(zip(lo_bounds, hi_bounds))
    bounds.append((0.0, 2.0))
    
    # Solve
    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=[n/2.0],
                      bounds=bounds, method='highs')
    except Exception:
        return None
    
    if not res.success:
        return None
    
    y = res.x[0:n]
    
    # Verify feasibility
    if np.any(y < -1e-9) or np.any(y > 1.0 + 1e-9):
        return None
    y = np.clip(y, 0.0, 1.0)
    
    # Check sum
    sum_diff = np.sum(y) - n / 2.0
    if abs(sum_diff) > 1e-6:
        y = project_box_sum(y, n)
    
    return y

def pairwise_transfer(h, n_iter=50, time_limit=None, t_start=None):
    """Pairwise transfer local search."""
    n = len(h)
    h = h.copy()
    current_score = exact_score(h)
    
    for it in range(n_iter):
        if time_limit is not None and t_start is not None:
            if time.time() - t_start > time_limit:
                break
        
        # Compute gradient of a smooth approximation of max
        c_full = np.correlate(h, 1.0 - h, mode="full")
        z_max = np.max(c_full)
        
        # Soft weights over lags
        scaled = (c_full - z_max) / (0.01 * z_max + 1e-10)
        weights = np.exp(scaled)
        weights /= weights.sum()
        
        # Compute composite gradient
        comp_grad = np.zeros(n)
        # Only compute for top lags to save time
        top_lags = np.argsort(c_full)[::-1][:20]
        for q in top_lags:
            lag = q - (n - 1)
            g = grad_z(h, lag)
            comp_grad += weights[q] * g
        
        # Find best pair transfer
        best_improvement = 0
        best_i, best_j, best_delta = -1, -1, 0
        
        # Get top 30 positive gradient and bottom 30 negative gradient
        pos_idx = np.argsort(comp_grad)[::-1][:30]
        neg_idx = np.argsort(comp_grad)[:30]
        
        for i in pos_idx:
            for j in neg_idx:
                if i == j:
                    continue
                if h[i] < 1e-6 or h[j] > 1.0 - 1e-6:
                    continue
                # Transfer delta from i to j
                max_delta = min(h[i], 1.0 - h[j])
                # Try a few fractions

                    delta = max_delta * frac
                    if delta < 1e-8:
                        continue
                    h_new = h.copy()
                    h_new[i] -= delta
                    h_new[j] += delta
                    new_score = exact_score(h_new)
                    if current_score - new_score > best_improvement:
                        best_improvement = current_score - new_score
                        best_i, best_j, best_delta = i, j, delta
        
        if best_i >= 0 and best_improvement > 1e-10:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            current_score -= best_improvement
        else:
            break
    
    return h, current_score

    if n == 0:

    c = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(c) * dx)

def project_box_sum(h, n):
    h = np.asarray(h, dtype=np.float64).flatten()
    target = n / 2.0
    # Binary search for alpha
    lo = -h.max() - 0.1
    hi = 1.0 - h.min() + 0.1
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    # Fix residual
    resid = target - np.sum(result)
    if abs(resid) > 1e-12:
        if resid > 0:
            mask = result < 1.0 - 1e-12
        else:
            mask = result > 1e-12
        cnt = mask.sum()
        if cnt > 0:
            result[mask] += resid / cnt
            result = np.clip(result, 0.0, 1.0)
    return result

def grad_correlate(h, lag):
    """Gradient of the correlation at the given lag (can be negative/zero/positive)."""
    n = len(h)
    g = np.zeros(n)
    if lag == 0:
        g[:] = 1.0 - 2.0 * h
    elif lag > 0:

        if L <= 0:

    else:
        a = -lag
        L = n - a
        if L <= 0:
            return g
        g[a:] += (1.0 - h[:L])
        g[:L] -= h[a:]
    return g

                    continue

            h[i] -= best_frac

        if abs(np.sum(y) - n / 2.0) > 1e-6:
            y = project_box_sum(y, n)

        new_score = exact_score(y)
        if new_score < current - 1e-12:
            h = y
            current = new_score
            stall = 0
            rho = max(0.003, rho * 0.9)
        else:
            stall += 1
            rho = max(0.001, rho * 0.5)

    return h, current


def pairwise_polish(h_init, n_iter=40, time_budget=30, t_start=None):
    n = len(h_init)
    h = h_init.copy()
    current = exact_score(h)
    deadline = (t_start or time.time()) + time_budget

    for it in range(n_iter):
        if time.time() > deadline:
            break

        c_full = np.correlate(h, 1.0 - h, mode="full")
        z_max = np.max(c_full)

        beta = 1.0 / max(0.01 * z_max, 1e-10)
        w = np.exp((c_full - z_max) * beta)
        w /= w.sum()

        top_count = min(12, len(c_full))
        top_idx = np.argsort(c_full)[::-1][:top_count]

        comp_grad = np.zeros(n)
        for q in top_idx:
            lag = q - (n - 1)
            comp_grad += w[q] * grad_correlate(h, lag)

        pos_c = np.where((comp_grad > 0) & (h > 0.03))[0]
        neg_c = np.where((comp_grad < 0) & (h < 0.97))[0]

        if len(pos_c) == 0 or len(neg_c) == 0:
            break

        pos_c = pos_c[np.argsort(comp_grad[pos_c])[::-1]][:15]
        neg_c = neg_c[np.argsort(comp_grad[neg_c])][:15]

        best_imp = 0.0
        best_i, best_j, best_d = -1, -1, 0.0

        for i in pos_c[:10]:
            for j in neg_c[:10]:
                if i == j:
                    continue
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-6:
                    continue
                for frac in (0.15, 0.4, 0.7, 1.0):
                    d = max_d * frac
                    if d < 1e-8:
                        continue
                    h[i] -= d
                    h[j] += d
                    s = exact_score(h)
                    if current - s > best_imp:
                        best_imp = current - s
                        best_i, best_j, best_d = i, j, d
                    h[i] += d
                    h[j] -= d

        if best_i >= 0 and best_imp > 1e-10:
            h[best_i] -= best_d
            h[best_j] += best_d
            current -= best_imp
        else:
            break

    return h, current


def make_perturbed_base(h0, scale, rng):
    n = len(h0)
    h = h0 + scale * rng.standard_normal(n)
    return project_box_sum(h, n)


def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    def remaining():
        return max(0.0, budget_s - (time.time() - t_start))

    try:
        initial = np.array(initial_h_values, dtype=np.float64).flatten()
    except (NameError, Exception):
        initial = np.full(250, 0.5)

    n = len(initial)
    h0 = project_box_sum(initial, n)
    best_h = h0.copy()
    best_score = exact_score(h0)

    # Phase 1: SLP from parent (majority of budget)
    t1 = remaining() * 0.45
    h_opt, sc = slp_optimize(h0, t1, t_start)
    if sc < best_score:
        best_h = h_opt.copy()
        best_score = sc

    # Phase 2: Pairwise polish on best
    if remaining() > 20:
        hp, scp = pairwise_polish(h_opt, n_iter=50,
                                  time_budget=min(25, remaining() * 0.2),
                                  t_start=t_start)
        if scp < best_score:
            best_h = hp.copy()
            best_score = scp

    # Phase 3: SLP from perturbed starts
    scales = [0.04, 0.08, 0.12]
    for trial in range(3):
        if remaining() < 40:
            break
        hs = make_perturbed_base(h0, scales[trial % 3], rng)
        time_alloc = min(remaining() * 0.15, 80)
        h2, sc2 = slp_optimize(hs, time_alloc, t_start)
        if sc2 < best_score:
            best_h = h2.copy()
            best_score = sc2

        if remaining() > 25:
            hp2, sc2p = pairwise_polish(h2, n_iter=25,
                                        time_budget=min(15, remaining() * 0.1),
                                        t_start=t_start)
            if sc2p < best_score:
                best_h = hp2.copy()
                best_score = sc2p

    # Phase 4: Try double resolution if time permits
    if remaining() > 80 and n <= 1000:
        n2 = min(2 * n, 1000)
        rep = max(1, n2 // n)
        h2 = np.repeat(h0, rep)[:n2]
        h2 = project_box_sum(h2, n2)
        time_alloc = min(remaining() * 0.3, 120)
        h3, sc3 = slp_optimize(h2, time_alloc, t_start)
        if sc3 < best_score:
            best_h = h3.copy()
            best_score = sc3

        if remaining() > 20:
            hp3, sc3p = pairwise_polish(h3, n_iter=20,
                                        time_budget=min(20, remaining() * 0.1),
                                        t_start=t_start)
            if sc3p < best_score:
                best_h = hp3.copy()
                best_score = sc3p

    # Final: polish best if time remains
    if remaining() > 15:
        hfin, scfin = pairwise_polish(best_h, n_iter=30,
                                      time_budget=min(15, remaining() * 0.5),
                                      t_start=t_start)
        if scfin < best_score:
            best_h = hfin.copy()
            best_score = scfin

    # Final validation
    best_h = project_box_sum(best_h, len(best_h))
    best_score = exact_score(best_h)

    return best_h, best_score, len(best_h)
