def pgd_step(h, tau, alpha, max_iters=2000, time_limit=60.0):
    """Run PGD optimization. Returns best h found."""
    n = len(h)
    T = n / 2.0
    h_cur = project_box_sum(h.copy(), T)
    best_h = h_cur.copy()
    best_score = corr_exact(h_cur)
    t0 = time.time()
    F, grad = smooth_max_and_grad(h_cur, tau)
    alpha_cur = alpha

            alpha_cur = min(alpha_cur * 1.05, 0.1)

    z = get_z(h)
    zmax = z.max()
    shifted = np.clip((z - zmax) / tau, -60.0, 0.0)
    exps = np.exp(shifted)
    total = exps.sum()
    F = zmax + tau * np.log(total)
    p = exps / total  # softmax weights
    dx = 2.0 / n
    b = 1.0 - h
    grad = np.zeros(n)
    for q in range(2 * n - 1):
        w = p[q]
        if w < 1e-10:
            continue
        ell = q - n + 1
        L = n - abs(ell)
        if L <= 0:
            continue
        if ell >= 0:
            grad[ell:] += dx * w * b[:L]
            grad[:L] -= dx * w * h[ell:]
        else:
            a = -ell
            grad[:L] += dx * w * b[a:]
            grad[a:] -= dx * w * h[:L]
    return F, grad

def project_box_sum(x, T):
    """Project onto {0<=h<=1, sum(h)=T} via bisection."""
    n = len(x)
    lo = min(x.min() - 1.0, -2.0)
    hi = max(x.max() + 1.0, 3.0)
    for _ in range(64):
        mid = 0.5 * (lo + hi)
        s = np.clip(x - mid, 0.0, 1.0).sum()
        if s > T:
            lo = mid
        else:
            hi = mid
    return np.clip(x - 0.5 * (lo + hi), 0.0, 1.0)

def pgd_optimize(h0, time_limit, tau_init=None, tau_final=0.0005):
    """Projected gradient descent with smooth-max surrogate."""
    n = len(h0)
    T = n / 2.0
    h = project_box_sum(h0.copy(), T)
    best_h = h.copy()
    best_score = corr_exact(h)
    
    z = get_z(h)
    z_spread = z.max() - z.min()
    if tau_init is None:
        tau = min(0.03, max(0.005, 0.3 * z_spread))
    else:
        tau = tau_init
    
    alpha = 0.05
    t_start = time.time()
    iter_count = 0
    best_iter = 0
    
    while time.time() - t_start < time_limit:
        F, grad = smooth_max_and_grad(h, tau)
        gnorm = np.max(np.abs(grad))
        if gnorm < 1e-10:
            break
        
        direction = -grad / gnorm
        h_new = project_box_sum(h + alpha * direction, T)
        F_new, _ = smooth_max_and_grad(h_new, tau)
        
        if F_new < F - 1e-12:
            h = h_new
            alpha = min(alpha * 1.03, 0.3)
            iter_count += 1
        else:
            alpha *= 0.5
            if alpha < 1e-8:
                alpha = 0.03
                tau = max(tau * 0.7, tau_final)
                if tau <= tau_final * 1.001:
                    break
        
        # Track best
        score = corr_exact(h)
        if score < best_score - 1e-14:
            best_score = score
            best_h = h.copy()
            best_iter = iter_count
        
        # Tau annealing
        if iter_count % 50 == 0:
            tau = max(tau * 0.95, tau_final)
    
    return best_h, best_score

def upsample(h, f=2):
    return np.repeat(h, f)

def downsample(h, f=2):
    n = len(h)
    if n % f != 0:
        return h
    return h.reshape(-1, f).mean(axis=1)

def slsqp_polish(h0, time_limit, K=80):
    """SLSQP epigraph minimax optimization."""
    n = len(h0)
    T = n / 2.0
    dx = 2.0 / n
    
    z = get_z(h0)
    K = min(K, 2 * n - 1)
    idx = np.argsort(z)[::-1][:K]
    
    def obj(x):
        return x[-1]
    
    def obj_grad(x):
        g = np.zeros(len(x))
        g[-1] = 1.0
        return g
    
    def ineq_fun(x):
        # t - z_q(h) >= 0
        h = x[:-1]
        return x[-1] - get_z(h)[idx]
    
    def ineq_jac(x):
        h = x[:-1]
        b = 1.0 - h
        m = len(idx)
        J = np.zeros((m, n + 1))
        for row in range(m):
            q = idx[row]
            ell = q - n + 1
            L = n - abs(ell)
            if ell >= 0:
                J[row, ell:] -= dx * b[:L]
                J[row, :L] += dx * h[ell:]
            else:
                a = -ell
                J[row, :L] -= dx * b[a:]
                J[row, a:] += dx * h[:L]
            J[row, -1] = 1.0
        return J
    
    def eq_fun(x):
        return x[:-1].sum() - T
    
    def eq_jac(x):
        J = np.ones(n + 1)
        J[-1] = 0.0
        return J
    
    t_init = max(corr_exact(h0) + 0.002, 0.1)
    x0 = np.concatenate([h0, [t_init]])
    bounds = [(0.0, 1.0)] * n + [(0.0, 1.0)]
    
    constraints = [
        {'type': 'ineq', 'fun': ineq_fun, 'jac': ineq_jac},
        {'type': 'eq', 'fun': eq_fun, 'jac': eq_jac}
    ]
    
    try:
        res = so.minimize(
            obj, x0, jac=obj_grad, bounds=bounds,
            constraints=constraints, method='SLSQP',
            options={'maxiter': 200, 'ftol': 1e-12}
        )
        h_res = project_box_sum(res.x[:-1], T)
        score = corr_exact(h_res)
        return h_res, score
    except Exception:
        return h0, corr_exact(h0)
