Now for the LSE objective with gradient:

The LSE objective is:
F(p) = f_max + tau * log(sum_k exp((c_k - f_max)/tau))

where c_k are the correlation values and f_max = max_k c_k.

The gradient with respect to p_j involves the derivative of each c_k with respect to p_j, weighted by the softmax weights.

For the mirror-complement h, the correlation at lag k (index in the full output) is:
c_k = sum_i h[i+k] * (1-h[i]) (for valid i, with appropriate index bounds)

The derivative dc_k/dh[j] is:
- If j appears in h[i+k] term: dc_k/dh[j] = -(1-h[i]) for the i where i+k=j
- If j appears in (1-h[i]) term: dc_k/dh[j] = -h[i+k] for the i where i=j
- If j appears in both: sum of both contributions

For the mirror complement, h[j] = p[j] for j < m, and h[j] = 1-p[n-1-j] for j >= m.

So dc_k/dp[j] = dc_k/dh[j] * dh[j]/dp[j] + dc_k/dh[n-1-j] * dh[n-1-j]/dp[j]
= dc_k/dh[j] * 1 + dc_k/dh[n-1-j] * (-1)
= dc_k/dh[j] - dc_k/dh[n-1-j]

        h[m+1:] = 1.0 - p[::-1]
    else:
        h[m:] = 1.0 - p[::-1]
    return h

def p_from_h(h):
    n, m = len(h), len(h) // 2
    return np.clip(0.5 * (h[:m] + 1.0 - h[n-1:n-m-1:-1]), 0, 1)

def interp_p(p, m_new):
    m_old = len(p)
    if m_old == m_new:
        return p.copy()
    if m_old == 1:
        return np.full(m_new, float(p[0]), dtype=np.float64)
    x_old = (np.arange(m_old, dtype=np.float64) + 0.5) / m_old
    x_new = (np.arange(m_new, dtype=np.float64) + 0.5) / m_new
    return np.clip(np.interp(x_new, x_old, p), 0, 1)

def make_lse_func(n, state):
    dx = 2.0 / n
    def _f(p, tau):
        p_c = np.clip(p, 0.0, 1.0)
        h = build_h(p_c, n)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        fmax = float(np.max(corr))
        if fmax < state['best'] - 1e-13:
            state['best'] = fmax
            state['h'] = h.copy()
            state['p'] = p_c.copy()
            state['n'] = n
        if tau <= 1e-12:
            return fmax
        s = (corr - fmax) / tau
        sm = float(np.max(s))
        return fmax + tau * (sm + np.log(float(np.sum(np.exp(s - sm)))))
    return _f

def make_full_lse_func(n, state):
    dx = 2.0 / n
    def _f(y, tau):
        h = project_full(y)
        corr = np.correlate(h, 1.0 - h, mode="full") * dx
        fmax = float(np.max(corr))
        if fmax < state['best'] - 1e-13:
            state['best'] = fmax
            state['h'] = h.copy()
            state['n'] = n
        if tau <= 1e-12:
            return fmax
        s = (corr - fmax) / tau
        sm = float(np.max(s))
        return fmax + tau * (sm + np.log(float(np.sum(np.exp(s - sm)))))
    return _f

def run_symmetric_lse(state, p_init, n, deadline):
    m = len(p_init)
    func = make_lse_func(n, state)
    bounds = [(0.0, 1.0)] * m
    c0 = exact_c5(build_h(np.clip(p_init, 0, 1), n))
    tau0 = max(0.002 * c0, 0.0005)
    tau_list = [tau0, tau0 * 0.4, tau0 * 0.15, tau0 * 0.05, max(tau0 * 0.01, 1e-6)]
    p_cur = np.clip(p_init, 0, 1).copy()
    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        try:
            res = opt.minimize(
                lambda x: func(x, tau),
                x0=p_cur,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 300, 'maxfun': 5000, 'ftol': 1e-14, 'gtol': 1e-9}
            )
            if res.x is not None:
                p_new = np.clip(res.x, 0, 1)
                h_new = build_h(p_new, n)
                v = exact_c5(h_new)
                if v < state['best'] - 1e-13:
                    state['best'] = v
                    state['h'] = h_new.copy()
                    state['p'] = p_new.copy()
                    state['n'] = n
                p_cur = p_new
        except Exception:
            pass
    return state

def run_full_lse(state, h_init, n, deadline):
    dx = 2.0 / n
    func = make_full_lse_func(n, state)
    bounds = [(0.0, 1.0)] * n
    c0 = exact_c5(h_init)
    tau0 = max(0.002 * c0, 0.0005)
    tau_list = [tau0, tau0 * 0.4, tau0 * 0.15, tau0 * 0.05, max(tau0 * 0.01, 1e-6)]
    y_cur = h_init.copy()
    for tau in tau_list:
        if time.monotonic() > deadline:
            break
        try:
            res = opt.minimize(
                lambda x: func(x, tau),
                x0=y_cur,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 200, 'maxfun': 3000, 'ftol': 1e-14, 'gtol': 1e-9}
            )
            if res.x is not None:
                h_new = project_full(res.x)
                v = exact_c5(h_new)
                if v < state['best'] - 1e-13:
                    state['best'] = v
                    state['h'] = h_new.copy()
                    state['n'] = n
                y_cur = h_new
        except Exception:
            pass
    return state

def run_slsqp(state, n, deadline):
    m = n // 2
    if 'p' not in state or len(state['p']) != m:
        h = state['h']
        p = p_from_h(h)
        if len(p) != m:
            return state
    else:
        p = state['p'].copy()
    h = build_h(np.clip(p, 0, 1), n)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode="full") * dx
    fmax = float(np.max(corr))
    top_k = min(30, 2 * n - 1)
    lag_idx = np.argsort(corr)[::-1][:top_k]

    def slq_obj(z):
        m_z = len(z) - 1
        p_z = np.clip(z[:m_z], 0, 1)
        return z[m_z]

    def mk_con(idx):
        def _con(z):
            m_z = len(z) - 1
            p_z = np.clip(z[:m_z], 0, 1)
            t_val = z[m_z]
            h_z = build_h(p_z, n)
            c_z = np.correlate(h_z, 1.0 - h_z, mode="full") * dx
            fv = float(np.max(c_z))
            if fv < state['best'] - 1e-13:
                state['best'] = fv
                state['h'] = h_z.copy()
                state['p'] = p_z.copy()
                state['n'] = n
            return t_val - c_z[idx]
        return _con

    constraints = [{'type': 'ineq', 'fun': mk_con(i)} for i in lag_idx]
    z0 = np.concatenate([np.clip(p, 0, 1), [fmax + 1e-5]])
    bnds = [(0.0, 1.0)] * m + [(0.0, 1.0)]
    try:
        res = opt.minimize(
            slq_obj, z0, method='SLSQP', bounds=bnds,
            constraints=constraints,
            options={'maxiter': 80, 'ftol': 1e-10, 'eps': 1e-6}
        )
        if res.x is not None:
            p_new = np.clip(res.x[:m], 0, 1)
            h_new = build_h(p_new, n)
            v = exact_c5(h_new)
            if v < state['best'] - 1e-13:
                state['best'] = v
                state['h'] = h_new.copy()
                state['p'] = p_new.copy()
                state['n'] = n
    except Exception:
        pass
    return state

def local_search(state, n, deadline, rng):
    h = state['h'].copy()
    dx = 2.0 / n
    current = state['best']
    steps = 0
    while time.monotonic() < deadline:
        steps += 1
        if steps > 50000:
            break
        idx = rng.integers(0, n, size=rng.integers(2, 8))
        delta = rng.normal(0, 0.05, size=len(idx))
        h_try = h.copy()
        h_try[idx] = np.clip(h_try[idx] + delta, 0, 1)
        h_try = project_full(h_try)
        v = exact_c5(h_try)
        if v < current - 1e-10:
            current = v
            h = h_try.copy()
            state['best'] = v
            state['h'] = h.copy()
            state['n'] = n
            state['p'] = p_from_h(h)
            steps = 0
    return state

def gen_seed_wave(m, freq, amp, phase):
    t = np.arange(m, dtype=np.float64) / m
    p = 0.5 + amp * np.sin(2.0 * np.pi * freq * t + phase)
    return np.clip(p, 0, 1)

def gen_seed_step(m, rng):
    p = (rng.random(m) > 0.5).astype(np.float64)
    return p

def run(seed=42, budget_s=1100, **kwargs):
    start_time = time.monotonic()
    deadline = start_time + budget_s * 0.93
    rng = np.random.default_rng(seed)

    state = {'best': 1.0, 'h': None, 'p': None, 'n': 0}

    # Load initial
    try:
        h_init = np.array(initial_h_values, dtype=np.float64).copy()
        h_init = np.clip(h_init, 0, 1)
        v_init = exact_c5(h_init)
        state['best'] = min(state['best'], v_init)
        state['h'] = h_init.copy()
        state['n'] = len(h_init)
        state['p'] = p_from_h(h_init)
    except Exception:
        n_fb = 128
        h_fb = np.full(n_fb, 0.5, dtype=np.float64)
        state['best'] = exact_c5(h_fb)
        state['h'] = h_fb
        state['n'] = n_fb
        state['p'] = np.full(n_fb // 2, 0.5)

    if state['h'] is None:
        n_fb = 128
        state['h'] = np.full(n_fb, 0.5)
        state['best'] = 0.5
        state['n'] = n_fb
        state['p'] = np.full(n_fb // 2, 0.5)

    base_n = state['n']
    if base_n < 16:
        base_n = 64

    # Target dimensions
    dims = sorted(set([base_n, 96, 128, 192] + [d for d in range(64, 257, 32) if d >= 48]))
    dims = [d for d in dims if 48 <= d <= 256]
    if base_n in dims:
        dims.remove(base_n)
    dims.insert(0, base_n)

    # Phase 1: Symmetric LSE at base dimension
    for n in dims:
        if time.monotonic() > deadline:
            break
        m = n // 2
        if n == state['n'] and 'p' in state and len(state['p']) == m:
            p_start = state['p'].copy()
        else:
            p_start = interp_p(state['p'], m)
        h_check = build_h(p_start, n)
        v_check = exact_c5(h_check)
        if v_check < state['best'] - 1e-13:
            state['best'] = v_check
            state['h'] = h_check
            state['n'] = n
            state['p'] = p_start.copy()
        # Also try complementary p
        p_comp = 1.0 - p_start
        h_comp = build_h(np.clip(p_comp, 0, 1), n)
        v_comp = exact_c5(h_comp)
        if v_comp < state['best'] - 1e-13:
            state['best'] = v_comp
            state['h'] = h_comp
            state['n'] = n
            state['p'] = np.clip(p_comp, 0, 1)
        # Try reversed p
        p_rev = p_start[::-1]
        h_rev = build_h(np.clip(p_rev, 0, 1), n)
        v_rev = exact_c5(h_rev)
        if v_rev < state['best'] - 1e-13:
            state['best'] = v_rev
            state['h'] = h_rev
            state['n'] = n
            state['p'] = np.clip(p_rev, 0, 1)

        # Run symmetric LSE
        run_symmetric_lse(state, p_start, n, deadline)

        # Generate additional seeds at this dimension
        n_seeds = min(4, max(1, int(5.0 * (deadline - time.monotonic()) / max(1, budget_s * 0.15))))
        for s in range(n_seeds):
            if time.monotonic() > deadline:
                break
            if s == 0:
                p_seed = gen_seed_wave(m, rng.integers(2, 8), float(rng.uniform(0.1, 0.4)), float(rng.uniform(0, 6.28)))
            elif s == 1:
                p_seed = gen_seed_step(m, rng)
            elif s == 2:
                p_seed = gen_seed_wave(m, 1, float(rng.uniform(0.2, 0.45)), float(rng.uniform(0, 3.14)))
                p_seed = 0.6 * p_seed + 0.4 * state['p'][:m] if len(state['p']) >= m else p_seed
                p_seed = np.clip(p_seed, 0, 1)
            else:
                p_seed = np.clip(state['p'][:m] + rng.normal(0, 0.1, m), 0, 1) if len(state['p']) >= m else np.full(m, 0.5)
            run_symmetric_lse(state, p_seed, n, deadline)

        # SLSQP refinement
        if time.monotonic() < deadline - 30:
            run_slsqp(state, n, deadline)

    # Phase 2: Full vector optimization (break symmetry)
    for n in [base_n, 128, 192]:
        if n > 256 or time.monotonic() > deadline - 60:
            break
        # Start from best (projected to this dimension if needed)
        h_cur = state['h']
        if len(h_cur) != n:
            x_old = (np.arange(len(h_cur)) + 0.5) / len(h_cur)
            x_new = (np.arange(n) + 0.5) / n
            h_cur = np.clip(np.interp(x_new, x_old, h_cur), 0, 1)
            h_cur = project_full(h_cur)
        v = exact_c5(h_cur)
        if v < state['best'] - 1e-13:
            state['best'] = v
            state['h'] = h_cur.copy()
            state['n'] = n
        run_full_lse(state, h_cur, n, deadline)

        # Also try a random full start
        if time.monotonic() < deadline - 40:
            h_rand = project_full(0.5 + rng.normal(0, 0.2, n))
            run_full_lse(state, h_rand, n, deadline)

    # Phase 3: Local search on best solution
    if time.monotonic() < deadline - 10:
        h_best = state['h'].copy()
        n_best = state['n']
        local_search(state, n_best, deadline - 5, rng)

    # Final validation
    h_out = np.array(state['h'], dtype=np.float64).copy()
    h_out = np.clip(h_out, 0, 1)
    # Ensure feasibility
    target_sum = len(h_out) * 0.5
    if abs(float(np.sum(h_out)) - target_sum) > 1e-6:
        h_out = project_full(h_out)
    c5_final = exact_c5(h_out)
    if c5_final < state['best'] - 1e-13:
        state['best'] = c5_final

    return h_out, c5_final, len(h_out)
