def project_feasible(z, n):
    S = n * 0.5

    lo = -3.0

            hi = mid

    resid = S - h.sum()

        mask = (h > 1e-14) & (h < 1.0 - 1e-14)
        cnt = int(mask.sum())
        if cnt > 0:

    h = np.clip(h, 0.0, 1.0)

def smooth_grad(h, tau, n):
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    cm = np.max(c)
    w = np.exp((c - cm) / tau)
    w /= w.sum()
    grad = np.zeros(n)

        s = p - (n - 1)
        wp = w[p] * dx

            continue
        # Term 1: (1-h[i-s])
        if s >= 0:
            ilo, ihi = s, n
        else:
            ilo, ihi = 0, n + s

            idx = np.arange(ilo, ihi)
            grad[idx] += wp * (1.0 - h[idx - s])
        # Term 2: -h[i+s]
        if s >= 0:
            ilo, ihi = 0, n - s
        else:
            ilo, ihi = -s, n
        if ihi > ilo:
            idx = np.arange(ilo, ihi)
            grad[idx] -= wp * h[idx + s]
    return grad
