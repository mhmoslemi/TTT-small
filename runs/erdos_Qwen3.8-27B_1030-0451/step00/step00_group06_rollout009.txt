def correlate_fft(a, b):
    """Compute full correlation using FFT."""
    n = len(a)
    m = len(b)
    # Pad to length n + m - 1 (or next power of 2)
    size = 1
    while size < n + m - 1:
        size *= 2
    fa = np.fft.rfft(a, size)
    fb = np.fft.rfft(b, size)
    # Correlation = ifft(conj(fa) * fb)
    corr = np.fft.irfft(np.conj(fa) * fb, size)
    return corr[:n+m-1]
