def line_eval(h, i, j, dx, alpha):
    """Evaluate the objective at a specific alpha for a pair transfer."""
    hi = h[i] - alpha
    hj = h[j] + alpha
    # Instead of copying the whole array, compute the correlation incrementally
    # For simplicity, just do a full copy (fast for n < 300)

    diff = target - np.sum(h)

        if count > 0:
            h[mask] += diff / float(count)
        else:
            h[:] += diff / len(h)

def evaluate(h, dx):

def line_eval(h, i, j, dx, alpha):

        cnt = int(np.sum(mask))
