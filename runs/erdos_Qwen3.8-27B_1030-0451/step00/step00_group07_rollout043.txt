        alpha[np.argmax(c)] = 1.0

        return alpha
    return e / s

        val = m

        shifted = np.clip(shifted, -50.0, 50.0)
        e = np.exp(shifted)
        s = np.sum(e)
        val = m + tau * np.log(max(s, 1e-300))

    # Positive lags d = 0,...,N-1 (r = N-1+d)

        if a < 1e-12:

        n_terms = N - d
        grad[d:] += a * dx * (1.0 - h[:n_terms])

    # Negative lags: d=-s, r=N-1-s, s=1,...,N-1
    for s in range(1, N):
        a = alpha[N - 1 - s]
        if a < 1e-12:
            continue
        n_terms = N - s
        grad[:n_terms] += a * dx * (1.0 - h[s:])

    """One subgradient step on the exact max (using smooth max as proxy)"""
    dx = 2.0 / N
    target = N / 2.0
    c = compute_c_all(h)
    alpha = smooth_max_alpha(c, tau)
    # Gradient of smooth max

        if a < 1e-12:

        n_terms = N - d
        grad[d:] += a * dx * (1.0 - h[:n_terms])
        grad[:n_terms] -= a * dx * h[d:]
    for s in range(1, N):
        a = alpha[N - 1 - s]
        if a < 1e-12:

        grad[:n_terms] += a * dx * (1.0 - h[s:])
        grad[s:] -= a * dx * h[:n_terms]

def generate_binary_pattern(N, seed_val, freq=1):

def generate_sinusoidal(N, freq, amp, phase):
    x = np.arange(N) * (2.0 / N)
    h = 0.5 + amp * np.sin(np.pi * freq * x + phase)
    return project_feasible(h, N)

def run(seed=42, budget_s=1100, **kwargs):
    t_start = time.time()
    rng = np.random.default_rng(seed)

    # Get initial values
    h_init = None
    try:
        h_init = np.array(initial_h_values, dtype=float)
        if h_init.shape[0] < 10:
            h_init = None
    except (NameError, AttributeError, IndexError):
        h_init = None

    if h_init is not None:
        best_h = h_init.copy()
        best_c = evaluate_c5(h_init)
        best_N = len(h_init)
    else:
        best_N = 240
        best_h = generate_sinusoidal(best_N, 3, 0.35, 0.0)
        best_c = evaluate_c5(best_h)

    def time_left():
        return budget_s - (time.time() - t_start)

    # Phase 1: Upscale and optimize at increasing resolutions
    resolutions = []
    parent_N = len(h_init) if h_init is not None else 120
    for mult in [1, 2, 3, 4, 6, 8]:
        n = parent_N * mult
        if 100 <= n <= 2400:
            resolutions.append(n)
    resolutions = sorted(set(resolutions))

    for N_new in resolutions:
        if time_left() < 45:
            break

        if N_new == len(best_h):
            h_work = best_h.copy()
        else:
            h_work = upscale_linear(best_h, N_new)

        c_work = evaluate_c5(h_work)
        if c_work < best_c:
            best_h = h_work
            best_c = c_work
            best_N = N_new

        # SLSQP optimization with tau annealing
        alloc = min(time_left() * 0.4, 120)
        if alloc > 10:
            h_opt, c_opt = run_slsqp(h_work, N_new, tau_start=0.03, tau_end=0.002,
                                     n_outer=8, time_limit=alloc)
            if c_opt < best_c:
                best_h = h_opt
                best_c = c_opt
                best_N = N_new

        # L-BFGS-B refinement
        if time_left() > 30 and best_N == N_new:
            alloc2 = min(time_left() * 0.2, 60)
            if alloc2 > 10:
                h_opt2, c_opt2 = run_slsqp_lbfgs(best_h, best_N, tau=0.005,
                                                  time_limit=alloc2, max_iter=300)
                if c_opt2 < best_c:
                    best_h = h_opt2
                    best_c = c_opt2

    # Phase 2: Subgradient refinement at best resolution
    if time_left() > 30:
        alloc = min(time_left() * 0.25, 90)
        h_sg, c_sg = run_subgradient(best_h, best_N, time_limit=alloc, n_iters=5000)
        if c_sg < best_c:
            best_h = h_sg
            best_c = c_sg

    # Phase 3: Multi-restart with different patterns
    if time_left() > 60:
        N_try = best_N
        n_restarts = min(8, int(time_left() / 20))
        for i in range(n_restarts):
            if time_left() < 20:
                break
            if i == 0:
                # Gaussian perturbation of best
                h_start = project_feasible(best_h + rng.normal(0, 0.03, N_try), N_try)
            elif i == 1:
                # Binary-like pattern
                h_start = generate_binary_pattern(N_try, seed + i * 137, freq=3)
            elif i == 2:
                # Sinusoidal with different freq
                h_start = generate_sinusoidal(N_try, 5, 0.4, rng.uniform(0, 6.28))
            elif i == 3:
                # Checkerboard-ish
                x = np.arange(N_try)
                h_start = np.where((x // 2) % 2 == 0, 0.8, 0.2)
                h_start = project_feasible(h_start, N_try)
            else:
                # Random smooth
                h_smooth = 0.5 + 0.3 * rng.normal(0, 1, N_try)
                h_smooth = scipy.ndimage.uniform_filter1d(h_smooth, size=10) if hasattr(scipy, 'ndimage') else h_smooth
                h_start = project_feasible(h_smooth, N_try)

            c_start = evaluate_c5(h_start)
            if c_start < best_c:
                best_h = h_start
                best_c = c_start
                best_N = N_try

            # Quick SLSQP from this start
            alloc_r = min(time_left() * 0.2, 30)
            if alloc_r > 8:
                h_opt, c_opt = run_slsqp(h_start, N_try, tau_start=0.02, tau_end=0.003,
                                         n_outer=5, time_limit=alloc_r)
                if c_opt < best_c:
                    best_h = h_opt
                    best_c = c_opt
                    best_N = N_try

    # Phase 4: Final fine-tuning at the best resolution
    if time_left() > 20:
        alloc_final = min(time_left() * 0.6, 120)
        h_final, c_final = run_slsqp(best_h, best_N, tau_start=0.003, tau_end=0.0003,
                                     n_outer=6, time_limit=alloc_final, ftol=1e-12)
        if c_final < best_c:
            best_h = h_final
            best_c = c_final

    # Final feasibility check
    best_h = project_feasible(best_h, len(best_h))
    best_c = evaluate_c5(best_h)

    return (best_h, best_c, len(best_h))
