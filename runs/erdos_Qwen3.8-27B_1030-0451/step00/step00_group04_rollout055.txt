from scipy.optimize import linprog

def exact_score(h):
    """Compute the exact C5 score for a feasible h."""
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(z)) * dx

def grad_z(h, ell):
    """Compute gradient of z_ell (unscaled) with respect to h."""
    n = len(h)
    g = np.zeros(n)
    if ell == 0:
        g[:] = 1.0 - 2.0 * h
    elif ell > 0:
        L = n - ell
        g[0:L] += (1.0 - h[ell:n])
        g[ell:n] -= h[0:L]
    else:
        a = -ell
        L = n - a
        g[a:n] += (1.0 - h[0:L])
        g[0:L] -= h[a:n]
    return g

def project_box_sum(x, target):
    """Project x onto {0 <= y <= 1, sum(y) = target} using bisection."""
    n = len(x)
    x = np.asarray(x, dtype=float64)
    # Bisection on alpha: sum(clip(x + alpha, 0, 1)) = target
    lo = -max(x.max(), 1.0) - 2.0
    hi = 1.0 - min(x.min(), 0.0) + 2.0
    for _ in range(100):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    # Fix residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:
        # Distribute residual among non-boundary elements
        interior = (result > 1e-8) & (result < 1.0 - 1e-8)
        n_int = interior.sum()
        if n_int > 0:
            per = residual / n_int
            result[interior] += per
            result = np.clip(result, 0.0, 1.0)
    return result

import numpy as np
import time
from scipy.optimize import linprog


def exact_score(h):
    n = len(h)
    dx = 2.0 / n
    z = np.correlate(h, 1.0 - h, mode="full")
    return float(np.max(z)) * dx


def z_vector(h):
    return np.correlate(h, 1.0 - h, mode="full")


def grad_z(h, k):
    """Gradient of c_k w.r.t. h, where c_k is the k-th correlation value."""
    n = len(h)
    g = np.zeros(n)
    if k == 0:
        g[:] = 1.0 - 2.0 * h
    elif k > 0:
        L = n - k
        g[k:n] += (1.0 - h[0:L])
        g[0:L] -= h[k:n]
    else:
        m = -k
        L = n - m
        g[0:L] += (1.0 - h[m:n])
        g[m:n] -= h[0:L]
    return g


def project_box_sum(x, target):
    n = len(x)
    x = np.asarray(x, dtype=np.float64)
    lo = -2.0 - x.max()
    hi = 3.0 - x.min()
    for _ in range(80):
        mid = 0.5 * (lo + hi)
        s = np.sum(np.clip(x + mid, 0.0, 1.0))
        if s < target:
            lo = mid
        else:
            hi = mid
    result = np.clip(x + 0.5 * (lo + hi), 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        interior = (result > 1e-8) & (result < 1.0 - 1e-8)
        n_int = interior.sum()
        if n_int > 0:
            result[interior] += residual / n_int
            result = np.clip(result, 0.0, 1.0)
    return result


def slp_step(h, K, rho):
    """One SLP step. Returns (new_h, new_score) or None."""
    n = len(h)
    dx = 2.0 / n
    z = z_vector(h)
    z_max = np.max(z)
    score_old = float(z_max) * dx

    order = np.argsort(-z)
    gap = max(1e-8, 0.03 * z_max)
    active = []
    for idx in order:
        if len(active) >= K:
            break
        if z[idx] >= z_max - gap:
            active.append(int(idx))
    # Always include argmax and center
    q_max = int(order[0])
    q_center = n - 1
    if q_max not in active:
        active.append(q_max)
    if q_center not in active:
        active.append(q_center)
    active = list(set(active))
    K_a = len(active)

    # Build LP
    # Variables: y[0..n-1], s (s = t/dx)
    # Minimize s
    # s.t. sum(y) = n/2
    #      grad_z_q . y - s <= grad_z_q . h - z_q  for each active q
    #      max(0, h_i - rho) <= y_i <= min(1, h_i + rho)
    #      s >= 0

    bounds = []
    for i in range(n):
        lo_b = max(0.0, h[i] - rho)
        hi_b = min(1.0, h[i] + rho)
        bounds.append((lo_b, hi_b))
    bounds.append((0.0, None))

    A_ub = np.zeros((K_a, n + 1))
    b_ub = np.zeros(K_a)
    for m_idx, q in enumerate(active):
        k = q - (n - 1)
        g = grad_z(h, k)
        A_ub[m_idx, :n] = g
        A_ub[m_idx, n] = -1.0
        b_ub[m_idx] = np.dot(g, h) - z[q]

    A_eq = np.zeros((1, n + 1))
    A_eq[0, :n] = 1.0
    b_eq = np.array([n * 0.5])

    c_obj = np.zeros(n + 1)
    c_obj[n] = 1.0

    try:
        res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                      bounds=bounds, method='highs')
    except Exception:
        return None

    if res.status != 0 or res.x is None:
        return None

    y_new = res.x[:n].copy()
    y_new = project_box_sum(y_new, n * 0.5)

    # Line search: try blending from h to y_new
    best_y = None
    best_s = score_old
    alphas = [1.0, 0.7, 0.5, 0.3, 0.15, 0.05]
    for a in alphas:
        yt = (1.0 - a) * h + a * y_new
        yt = project_box_sum(yt, n * 0.5)
        st = exact_score(yt)
        if st < best_s - 1e-12:
            best_s = st
            best_y = yt.copy()

    if best_y is not None:
        return best_y, best_s
    return None


def pair_transfer_polish(h, max_iter=100, time_limit=30.0):
    n = len(h)
    h = np.array(h, dtype=np.float64)
    current_score = exact_score(h)
    start_time = time.time()

    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break

        z = z_vector(h)
        z_max = float(np.max(z))
        current_score = z_max * (2.0 / n)

        order = np.argsort(-z)
        gap = max(1e-8, 0.02 * z_max)
        g_mix = np.zeros(n)
        w_sum = 0.0
        count = 0
        for idx in order:
            if z[idx] < z_max - gap:
                break
            k = idx - (n - 1)
            g = grad_z(h, k)
            w = float(z[idx])
            g_mix += w * g
            w_sum += w
            count += 1
            if count >= 40:
                break
        if w_sum > 0:
            g_mix /= w_sum
        else:
            break

        idx_sorted = np.argsort(-g_mix)
        n_s = min(25, n)
        sources = idx_sorted[:n_s]
        targets = idx_sorted[::-1][:n_s]

        best_delta = 0.0
        best_sc = current_score
        best_i = -1
        best_j = -1

        for i in sources:
            if h[i] < 1e-8:
                continue
            for j in targets:
                if i == j:
                    continue
                if h[j] > 1.0 - 1e-8:
                    continue
                max_d = min(h[i], 1.0 - h[j])
                if max_d < 1e-8:
                    continue
                for frac in [0.05, 0.15, 0.3, 0.5, 0.8, 1.0]:
                    d = frac * max_d
                    ht = h.copy()
                    ht[i] -= d
                    ht[j] += d
                    sc = exact_score(ht)
                    if sc < best_sc - 1e-12:
                        best_sc = sc
                        best_delta = d
                        best_i = i
                        best_j = j

        if best_delta > 0:
            h[best_i] -= best_delta
            h[best_j] += best_delta
            h = np.clip(h, 0.0, 1.0)
            current_score = best_sc
        else:
            break

    return h, current_score


def gen_random_feasible(n, rng, smooth_frac=1.0):
    h = rng.beta(2.0, 2.0, size=n)
    k = max(1, int(smooth_frac * 3))
    kernel = np.ones(2 * k + 1) / (2 * k + 1)
    h = np.convolve(h, kernel, mode='same')
    return project_box_sum(h, n * 0.5)
