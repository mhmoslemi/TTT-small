            best_h = h.copy()
        
        # Quick SLSQP
        tau = max(np.max(np.correlate(h, 1-h, mode='full')) * 0.08, 1e-3)
        h = do_slsqp(h, ntrial, tau, maxiter=15)
        h = project_feasible(h, ntrial / 2.0)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        ntrials += 1
    
    # Final
    best_h = project_feasible(best_h, len(best_h) / 2.0)
    c5_final = evaluate(best_h)
    return best_h, c5_final, len(best_h)
