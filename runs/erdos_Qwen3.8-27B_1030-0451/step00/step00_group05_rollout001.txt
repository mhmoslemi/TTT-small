import numpy as np

# Top-level helper: repair profile to be feasible
def repair_profile(arr, n_points=None):
    """Repair arr to satisfy 0<=h<=1 and sum(h)=n/2. Returns (h, ok)."""
    arr = np.asarray(arr, dtype=np.float64).ravel()
    n = len(arr)
    if n_points is not None:
        n = n_points
        if len(arr) != n:
            # truncate or pad
            if len(arr) > n:
                arr = arr[:n]
            else:
                arr = np.concatenate([arr, np.zeros(n - len(arr))])
    target = n / 2.0
    arr = np.clip(arr, 0.0, 1.0)
    # Fix sum
    s = np.sum(arr)
    diff = s - target
    if abs(diff) < 1e-12:
        return arr, True
    # If diff > 0, need to decrease; if diff < 0, need to increase
    if diff > 0:
        # Remove from largest values first
        order = np.argsort(-arr, kind='stable')
        remaining = diff
        for i in order:
            if remaining <= 1e-14:
                break
            to_remove = min(arr[i], remaining)
            arr[i] -= to_remove
            remaining -= to_remove
        # Final adjustment if still off
        if abs(np.sum(arr) - target) > 1e-12:
            idx = np.argmax(arr)
            arr[idx] -= (np.sum(arr) - target)
    else:
        # Add to smallest values first
        order = np.argsort(arr, kind='stable')
        remaining = -diff
        for i in order:
            if remaining <= 1e-14:
                break
            to_add = min(1.0 - arr[i], remaining)
            arr[i] += to_add
            remaining -= to_add
        if abs(np.sum(arr) - target) > 1e-12:
            idx = np.argmin(arr)
            arr[idx] += (target - np.sum(arr))
    arr = np.clip(arr, 0.0, 1.0)
    if abs(np.sum(arr) - target) > 1e-9:
        return arr, False
    return arr, True


def evaluate_profile(h):
    """Compute exact C5 bound. Returns (bound, ok)."""
    n = len(h)
    if n == 0:
        return float('inf'), False
    dx = 2.0 / n
    if np.any(np.isnan(h)) or np.any(h < -1e-10) or np.any(h > 1.0 + 1e-10):
        return float('inf'), False
    if abs(np.sum(h) - n / 2.0) > 1e-9:
        return float('inf'), False
    corr = np.correlate(h, 1.0 - h, mode='full')
    vals = corr * dx
    bound = float(np.max(vals))
    if not np.isfinite(bound):
        return float('inf'), False
    return bound, True


def compute_gradient_penalized(h, target, dx):
    """Compute gradient of sum_k max(0, c_k - target)^2 subject to constraints.
    c_k = dx * y_k where y_k from np.correlate(h, 1-h, 'full').
    Returns gradient array (before constraint projection)."""
    n = len(h)
    corr = np.correlate(h, 1.0 - h, mode='full')
    c_vals = corr * dx  # actual overlap values
    viol = np.maximum(0.0, c_vals - target)
    if np.sum(viol) < 1e-15:
        return None
    g = np.zeros(n)
    # Lag 0: index n-1
    v0 = viol[n - 1]
    if v0 > 0:
        g += dx * (1.0 - 2.0 * h) * v0
    # For R = 1 to n-1
    for R in range(1, n):
        v_plus = viol[n - 1 + R]  # lag +R
        v_minus = viol[n - 1 - R]  # lag -R
        if v_plus == 0 and v_minus == 0:
            continue
        # Pairs (u, u+R) for u = 0..n-1-R
        idx_low = np.arange(n - R)
        idx_high = idx_low + R
        # grad[low] += dx * (v_minus*(1-h[high]) - v_plus*h[high])
        # grad[high] += dx * (v_plus*(1-h[low]) - v_minus*h[low])
        if v_plus > 0:
            g[idx_low] += dx * (-h[idx_high]) * v_plus
            g[idx_high] += dx * (1.0 - h[idx_low]) * v_plus
        if v_minus > 0:
            g[idx_low] += dx * (1.0 - h[idx_high]) * v_minus
            g[idx_high] += dx * (-h[idx_low]) * v_minus
    return g


def penalty_descent(h_start, target, max_iter, time_limit, rng, verbose=False):
    """Projective gradient descent on penalty objective. Returns (best_h, best_val)."""
    start_time = time.monotonic()
    h = h_start.copy()
    cur_val, ok = evaluate_profile(h)
    if not ok:
        h, ok = repair_profile(h_start)
        cur_val, ok = evaluate_profile(h)
        if not ok:
            return h_start, cur_val
    best_h = h.copy()
    best_val = cur_val
    n = len(h)
    dx = 2.0 / n
    step = 0.3 / max(1.0, np.max(np.abs(h)))
    
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        # Adaptive target
        eff_target = max(target, cur_val - 0.01)
        g = compute_gradient_penalized(h, eff_target, dx)
        if g is None:
            break
        g_norm = np.linalg.norm(g)
        if g_norm < 1e-12:
            break
        # Line search
        accepted = False
        for trial in range(20):
            s = step / (trial + 1)
            h_new = h - s * g
            h_new = np.clip(h_new, 0.0, 1.0)
            h_new, ok = repair_profile(h_new, n)
            if not ok:
                continue
            val, ok = evaluate_profile(h_new)
            if not ok:
                continue
            # Accept if penalty improved (use soft criterion)
            corr = np.correlate(h_new, 1.0 - h_new, mode='full') * dx
            pen_new = np.sum(np.maximum(0, corr - eff_target)**2)
            corr_old = np.correlate(h, 1.0 - h, mode='full') * dx
            pen_old = np.sum(np.maximum(0, corr_old - eff_target)**2)
            if val < cur_val or pen_new < pen_old * 0.95:
                h = h_new
                cur_val = val
                if cur_val < best_val:
                    best_val = cur_val
                    best_h = h.copy()
                step = min(step * 1.5, 2.0)
                accepted = True
                break
        if not accepted:
            step *= 0.5
            if step < 1e-10:
                break
    return best_h, best_val


def simulated_annealing(h_start, max_iter, time_limit, rng, n_try_small=5):
    """Simulated annealing with various local moves. Returns (best_h, best_val)."""
    start_time = time.monotonic()
    n = len(h_start)
    dx = 2.0 / n
    h = h_start.copy()
    cur_val, ok = evaluate_profile(h)
    if not ok:
        return h_start, cur_val
    best_h = h.copy()
    best_val = cur_val
    
    temp = 0.01
    cool_rate = 0.995
    n_no_improve = 0
    
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        # Choose move type
        r = rng.random()
        h_new = h.copy()
        
        if r < 0.3:
            # Single position jitter
            i = rng.integers(n)
            delta = rng.uniform(-0.1, 0.1)
            h_new[i] += delta
        elif r < 0.5:
            # Pair swap
            i, j = rng.integers(0, n, size=2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        elif r < 0.7:
            # Block shift (short)
            blk = rng.integers(1, min(5, n // 4) + 1)
            start = rng.integers(0, n - blk)
            h_new[start:start+blk], h_new[start+blk:start+2*blk] = \
                h_new[start+blk:start+2*blk].copy(), h_new[start:start+blk].copy()
        elif r < 0.85:
            # Smoothing: blend neighbor
            i = rng.integers(1, n - 1)
            h_new[i] = 0.5 * (h_new[i-1] + h_new[i+1])
        else:
            # Random perturbation of a few positions
            nPerturb = rng.integers(1, 4)
            idxs = rng.choice(n, size=nPerturb, replace=False)
            h_new[idxs] += rng.uniform(-0.05, 0.05, size=len(idxs))
        
        h_new = np.clip(h_new, 0.0, 1.0)
        h_new, ok = repair_profile(h_new, n)
        if not ok:
            continue
        new_val, ok = evaluate_profile(h_new)
        if not ok:
            continue
        delta_val = new_val - cur_val
        if delta_val < 0 or rng.random() < math.exp(-delta_val / max(temp, 1e-10)):
            h = h_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
                n_no_improve = 0
            else:
                n_no_improve += 1
        else:
            n_no_improve += 1
        
        temp *= cool_rate
        if n_no_improve > 200:
            temp = max(temp, 0.005)
            n_no_improve = 0
    
    return best_h, best_val


def discrete_swap_anneal(h_start, max_iter, time_limit, rng):
    """Swap-based annealing focused on 0/1-like structure. Returns (best_h, best_val)."""
    start_time = time.monotonic()
    n = len(h_start)
    h = h_start.copy()
    cur_val, ok = evaluate_profile(h)
    if not ok:
        return h_start, cur_val
    best_h = h.copy()
    best_val = cur_val
    temp = 0.005
    
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        # Pick two positions and swap values
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        h_new = h.copy()
        h_new[i], h_new[j] = h_new[j], h_new[i]
        new_val, ok = evaluate_profile(h_new)
        if not ok:
            continue
        delta = new_val - cur_val
        if delta < 0 or rng.random() < math.exp(-delta / max(temp, 1e-10)):
            h = h_new
            cur_val = new_val
            if cur_val < best_val:
                best_val = cur_val
                best_h = h.copy()
        temp = max(temp * 0.998, 0.001)
    
    return best_h, best_val


def smooth_symmetry(h_start, rng):
    """Apply symmetry and smoothing variants. Returns list of candidates."""
    n = len(h_start)
    candidates = []
    # Reverse
    h_rev = h_start[::-1].copy()
    candidates.append(h_rev)
    # Circular shift by various amounts
    for shift in [1, 2, 3, n // 4, n // 2]:
        h_shift = np.roll(h_start, shift)
        candidates.append(h_shift)
    # Average with reverse (symmetrize)
    h_sym = 0.5 * (h_start + h_start[::-1])
    candidates.append(h_sym)
    # Light smoothing (3-point)
    h_smooth = h_start.copy()
    for _ in range(3):
        h_smooth = 0.5 * (h_smooth + np.concatenate([[h_smooth[-1]], h_smooth[:-1]]))
        h_smooth = 0.5 * (h_smooth + np.concatenate([h_smooth[1:], [h_smooth[0]]]))
        h_smooth = np.clip(h_smooth, 0, 1)
        h_smooth, _ = repair_profile(h_smooth, n)
    candidates.append(h_smooth)
    return candidates


def construct_seed(n, kind, rng):
    """Create an algorithmic seed profile of length n."""
    target = n / 2.0
    if kind == 0:
        # Constant at 0.5
        h = np.full(n, 0.5)
    elif kind == 1:
        # Cosine wave
        x = np.linspace(0, 2, n)
        h = 0.5 + 0.3 * np.cos(2 * np.pi * x)
    elif kind == 2:
        # Random with structure
        h = 0.5 + 0.2 * np.sin(2 * np.pi * 3.0 * np.linspace(0, 2, n))
        h += rng.normal(0, 0.05, n)
    elif kind == 3:
        # Binary-like: alternating blocks
        blk = max(1, n // 10)
        h = np.ones(n)
        for i in range(0, n, 2 * blk):
            h[i:i+blk] = 0.0
    elif kind == 4:
        # Random
        h = rng.uniform(0, 1, n)
    else:
        h = np.full(n, 0.5)
    h = np.clip(h, 0, 1)
    h, ok = repair_profile(h, n)
    return h if ok else np.full(n, 0.5)


def try_resolution(h_parent, n_target, rng, time_budget):
    """Resample parent to n_target and optimize. Returns (h, val) or None."""
    if time_budget <= 0:
        return None
    start_time = time.monotonic()
    n0 = len(h_parent)
    
    if n_target == n0:
        h = h_parent.copy()
    else:
        # Interpolate
        x_old = np.linspace(0, 2, n0)
        x_new = np.linspace(0, 2, n_target)
        h = np.interp(x_new, x_old, h_parent)
        h = np.clip(h, 0, 1)
    
    h, ok = repair_profile(h, n_target)
    if not ok:
        return None
    val, ok = evaluate_profile(h)
    if not ok:
        return None
    
    remaining = time_budget - (time.monotonic() - start_time)
    if remaining <= 1:
        return h, val
    
    # Penalty descent
    target = max(0.38, val - 0.005)
    h2, val2 = penalty_descent(h, target, max_iter=2000, time_limit=remaining * 0.6, rng=rng)
    if val2 < val:
        h, val = h2, val2
    
    remaining = remaining - (time.monotonic() - start_time)
    if remaining <= 1:
        return h, val
    
    # Simulated annealing
    h3, val3 = simulated_annealing(h, max_iter=5000, time_limit=remaining * 0.7, rng=rng)
    if val3 < val:
        h, val = h3, val3
    
    return h, val


def coordinate_descent(h_start, target, max_iter, time_limit, rng):
    """Coordinate-wise optimization: try adjusting each position. Returns (best_h, best_val)."""
    start_time = time.monotonic()
    n = len(h_start)
    h = h_start.copy()
    cur_val, _ = evaluate_profile(h)
    best_h = h.copy()
    best_val = cur_val
    
    for it in range(max_iter):
        if time.monotonic() - start_time > time_limit:
            break
        improved = False
        for i in range(n):
            if time.monotonic() - start_time > time_limit:
                break
            for delta in [0.01, -0.01, 0.02, -0.02, 0.05, -0.05]:
                h_new = h.copy()
                h_new[i] = h[i] + delta
                if h_new[i] < 0 or h_new[i] > 1:
                    continue
                h_new, ok = repair_profile(h_new, n)
                if not ok:
                    continue
                val_new, ok = evaluate_profile(h_new)
                if not ok:
                    continue
                if val_new < cur_val - 1e-10:
                    h = h_new
                    cur_val = val_new
                    improved = True
                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = h.copy()
                    break
        if not improved:
            # Try random coordinate with larger step
            i = rng.integers(n)
            delta = rng.choice([-0.1, -0.05, 0.05, 0.1])
            h_new = h.copy()
            h_new[i] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new, ok = repair_profile(h_new, n)
            if ok:
                val_new, ok = evaluate_profile(h_new)
                if ok and val_new < cur_val:
                    h = h_new
                    cur_val = val_new
                    if cur_val < best_val:
                        best_val = cur_val
                        best_h = h.copy()
    
    return best_h, best_val

        if 'initial_h_values' in globals() and initial_h_values is not None:
            parent = np.array(initial_h_values, dtype=np.float64).ravel()
            parent, ok = repair_profile(parent)
            if ok:
                val, ok2 = evaluate_profile(parent)
                if ok2:
                    best_h = parent.copy()
                    best_val = val
                    best_n = len(parent)
    except Exception:
        parent = None
    
    # Fallback: create a seed if no parent
    if best_h is None:
        n_seed = 131
        for kind in range(5):
            h_try = construct_seed(n_seed, kind, rng)
            val_try, ok = evaluate_profile(h_try)
            if ok and val_try < best_val:
                best_val = val_try
                best_h = h_try
                best_n = n_seed
    
    n_parent = best_n
    
    # Stage 1: Symmetry and smoothing variants on parent
    elapsed = time.monotonic() - start_time
    if elapsed < budget_s * 0.1:
        variants = smooth_symmetry(best_h, rng)
        for v in variants:
            v, ok = repair_profile(v, n_parent)
            if ok:
                val_v, ok2 = evaluate_profile(v)
                if ok2 and val_v < best_val:
                    best_val = val_v
                    best_h = v.copy()
    
    # Stage 2: Penalty descent on parent
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed
    if remaining > 5:
        target = max(0.38, best_val - 0.003)
        h_par, val_par = penalty_descent(best_h, target, max_iter=3000,
                                         time_limit=remaining * 0.25, rng=rng)
        if val_par < best_val:
            best_val = val_par
            best_h = h_par
    
    # Stage 3: Coordinate descent
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed
    if remaining > 10:
        target = max(0.38, best_val - 0.002)
        h_cd, val_cd = coordinate_descent(best_h, target, max_iter=200,
                                          time_limit=remaining * 0.2, rng=rng)
        if val_cd < best_val:
            best_val = val_cd
            best_h = h_cd
    
    # Stage 4: Simulated annealing on parent
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed
    if remaining > 10:
        h_sa, val_sa = simulated_annealing(best_h, max_iter=20000,
                                           time_limit=remaining * 0.3, rng=rng)
        if val_sa < best_val:
            best_val = val_sa
            best_h = h_sa
    
    # Stage 5: Try nearby resolutions
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed
    if remaining > 20:
        for n_try in [n_parent - 2, n_parent + 2, n_parent - 4, n_parent + 4]:
            if n_try < 50 or n_try > 200 or n_try <= 0:
                continue
            t_alloc = min(remaining * 0.15, 30.0)
            result = try_resolution(best_h, n_try, rng, t_alloc)
            if result is not None:
                h_try, val_try = result
                if val_try < best_val:
                    best_val = val_try
                    best_h = h_try
                    best_n = n_try
            elapsed = time.monotonic() - start_time
            remaining = budget_s - elapsed
            if remaining < 5:
                break
    
    # Stage 6: Discrete swap annealing
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed
    if remaining > 10:
        h_ds, val_ds = discrete_swap_anneal(best_h, max_iter=50000,
                                            time_limit=remaining * 0.3, rng=rng)
        if val_ds < best_val:
            best_val = val_ds
            best_h = h_ds
    
    # Stage 7: Final penalty polish
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed
    if remaining > 5:
        target = max(0.378, best_val - 0.002)
        h_final, val_final = penalty_descent(best_h, target, max_iter=5000,
                                             time_limit=remaining * 0.5, rng=rng)
        if val_final < best_val:
            best_val = val_final
            best_h = h_final
    
    # Stage 8: One more annealing pass
    elapsed = time.monotonic() - start_time
    remaining = budget_s - elapsed
    if remaining > 5:
        h_fin, val_fin = simulated_annealing(best_h, max_iter=10000,
                                             time_limit=remaining * 0.7, rng=rng)
        if val_fin < best_val:
            best_val = val_fin
            best_h = h_fin
    
    # Final validation
    final_h, ok = repair_profile(best_h, best_n)
    if not ok:
        final_h = np.full(best_n, 0.5)
        _, _ = repair_profile(final_h, best_n)
    final_val, ok = evaluate_profile(final_h)
    if not ok:
        final_h = np.full(best_n, 0.5)
        final_h, _ = repair_profile(final_h, best_n)
        final_val, _ = evaluate_profile(final_h)
    
    # Ensure time budget respected
    elapsed = time.monotonic() - start_time
    if elapsed > budget_s:
        pass  # We still return our best
    
    return final_h, final_val, len(final_h)
