import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def exact_z_vector(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full') * dx
    return c

def project_box_sum(h, target_sum):
    """Project h onto {x : 0 <= x_i <= 1, sum(x) = target_sum}"""
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    
    # Binary search for shift alpha such that sum(clip(h + alpha, 0, 1)) = target_sum
    lo = -1.0 - np.max(h)  # sum would be 0
    hi = 1.0 - np.min(h)  # sum would be n
    
    for _ in range(100):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    
    # Fix residual
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute residual among non-boundary elements
        idx = np.where((result > 1e-8) & (result < 1.0 - 1e-8))[0]
        if len(idx) > 0:
            adjust = residual / len(idx)
            result[idx] += adjust
            result = np.clip(result, 0.0, 1.0)
            # Final fix
            residual2 = target_sum - np.sum(result)
            if abs(residual2) > 1e-12:
                # Add to any element with room
                for i in range(n):
                    if residual2 > 0:
                        room = 1.0 - result[i]
                        add = min(room, residual2)
                        result[i] += add
                        residual2 -= add
                    else:
                        room = result[i]
                        sub = min(room, -residual2)
                        result[i] -= sub
                        residual2 += sub
                    if abs(residual2) < 1e-12:
                        break
    
    return result

def grad_z(h, q, n):
    """Gradient of z_q (the q-th element of full correlation * dx) w.r.t. h.
    q is the index in the full correlation output (0 to 2n-2).
    """
    dx = 2.0 / n
    ell = q - (n - 1)  # actual lag
    g = np.zeros(n)
    
    if ell == 0:
        g = (1.0 - 2.0 * h) * dx
    elif ell > 0:
        # z = sum_{i=0}^{n-1-ell} h[i+ell] * (1-h[i])
        L = n - ell
        g[ell:n] += (1.0 - h[0:L]) * dx
        g[:, ] -= 0  # placeholder
        # h[0:L] contributes: -h[0:L+ell]... wait
        # d/dh[j]: if j in [0, L-1], term is -h[j+ell], so g[j] += -h[j+ell]*dx
        # if j in [ell, n-1], term is (1-h[j-ell]), so g[j] += (1-h[j-ell])*dx
        g[:] = 0.0
        g[ell:n] += (1.0 - h[0:L]) * dx
        g[0:L] -= h[ell:n] * dx
    else:
        # ell < 0, let a = -ell > 0
        a = -ell
        # z = sum_{i=0}^{n-1-a} h[i] * (1-h[i+a])
        L = n - a
        g[:] = 0.0
        g[0:L] += (1.0 - h[a:n]) * dx
        g[a:n] -= h[0:L] * dx
    
    return g

def slp_optimize(h0, max_iters=200, rho_init=0.05, k_max=50, target=0.3808, time_limit=None, t_start=None):
    """Sequential Linear Programming optimization."""
    h = project_box_sum(h0, len(h0) / 2.0)
    n = len(h)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    rho = rho_init
    stall = 0
    
    if t_start is None:
        t_start = time.time()
    
    for it in range(max_iters):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        
        # Compute all z values
        z_vec = exact_z_vector(h)
        current_score = float(np.max(z_vec))
        
        if current_score < best_score:
            best_score = current_score
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 30:
            rho *= 0.5
            if rho < 1e-6:
                break
            stall = 0
        
        # Select active lags
        z_max = np.max(z_vec)
        gap = max(1e-4, 0.02 * z_max)
        active_mask = z_vec >= (z_max - gap)
        active_idx = np.where(active_mask)[0]
        
        # Cap active set
        if len(active_idx) > k_max:
            order = np.argsort(-z_vec[active_idx])
            active_idx = active_idx[order[:k_max]]
        
        # Ensure argmax is included
        argmax_idx = int(np.argmax(z_vec))
        if argmax_idx not in active_idx:
            active_idx = np.append(active_idx, argmax_idx)
        
        # Build LP
        K = len(active_idx)
        # Decision variable: [y_0, ..., y_{n-1}, t]
        # Objective: minimize t
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0  # minimize t
        
        # Inequality constraints: grad_q · y - t <= grad_q · h - z_q
        # i.e., grad_q · y - t <= rhs_q
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        
        for i, q in enumerate(active_idx):
            g = grad_z(h, q, n)
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = np.dot(g, h) - z_vec[q]
        
        # Equality constraint: sum(y) = n/2
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        # Bounds: [max(0, h_i - rho), min(1, h_i + rho)] for y_i, [0, z_max + 0.1] for t
        bounds = []
        for i in range(n):
            lo = max(0.0, h[i] - rho)
            hi = min(1.0, h[i] + rho)
            bounds.append((lo, hi))
        bounds.append((0.0, z_max + 0.5))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                rho *= 0.5
                continue
            
            y_new = res.x[:n]
            y_new = project_box_sum(y_new, n / 2.0)
            
            # Line search
            best_alpha = 0.0
            best_trial_score = current_score
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                candidate = (1.0 - alpha) * h + alpha * y_new
                candidate = project_box_sum(candidate, n / 2.0)
                trial_score = exact_score(candidate)
                if trial_score < best_trial_score - 1e-12:
                    best_trial_score = trial_score
                    best_alpha = alpha
                    best_candidate = candidate.copy()
            
            if best_alpha > 0:
                h = best_candidate
                current_score = best_trial_score
                if current_score < best_score:
                    best_score = current_score
                    best_h = h.copy()
                stall = 0
                # Slightly reduce rho on success
                rho = max(rho * 0.9, 1e-5)
            else:
                rho *= 0.5
        except Exception:
            rho *= 0.5
            if rho < 1e-6:
                break
    
    return best_h, best_score

def pairwise_transfer(h0, max_iters=200, n_top=20, time_limit=None, t_start=None):
    """Pairwise transfer local search."""
    h = project_box_sum(h0, len(h0) / 2.0)
    n = len(h)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    
    if t_start is None:
        t_start = time.time()
    
    for it in range(max_iters):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        
        # Compute gradient of squared max (surrogate)
        z_vec = exact_z_vector(h)
        z_max = np.max(z_vec)
        
        # Weight active lags by their proximity to max
        gap = max(1e-4, 0.03 * z_max)
        weights = np.maximum(0, z_vec - (z_max - gap))
        if np.max(weights) > 0:
            weights = weights / np.max(weights)
        else:
            weights = np.zeros(len(z_vec))
            weights[np.argmax(z_vec)] = 1.0
        
        # Weighted sum of gradients
        g_total = np.zeros(n)
        for q in range(len(z_vec)):
            if weights[q] > 1e-6:
                g_total += weights[q] * grad_z(h, q, n)
        
        # Find best pair transfer
        # Transfer δ from i to j: h[i] -= δ, h[j] += δ
        # Change in objective ≈ δ * (g_total[j] - g_total[i])
        # Want g_total[j] < g_total[i] for decrease
        
        # Select candidate positions
        sorted_g = np.argsort(g_total)
        sources = sorted_g[-n_top:]  # high gradient (decrease these)
        targets = sorted_g[:n_top]   # low gradient (increase these)
        
        improved = False
        for si in sources:
            if h[si] < 1e-8:
                continue
            for ti in targets:
                if ti == si:
                    continue
                if h[ti] > 1.0 - 1e-8:
                    continue
                
                max_delta = min(h[si], 1.0 - h[ti])
                if max_delta < 1e-8:
                    continue
                
                # Try several fractions
                best_delta = 0.0
                best_trial = current_score
                for frac in [0.5, 0.3, 0.1, 0.05, 0.01, 0.5]:
                    delta = frac * max_delta
                    if delta < 1e-10:
                        continue
                    candidate = h.copy()
                    candidate[si] -= delta
                    candidate[ti] += delta
                    trial = exact_score(candidate)
                    if trial < best_trial - 1e-12:
                        best_trial = trial
                        best_delta = delta
                
                if best_delta > 0:
                    h[si] -= best_delta
                    h[ti] += best_delta
                    h = np.clip(h, 0.0, 1.0)
                    h = project_box_sum(h, n / 2.0)
                    current_score = best_trial
                    if current_score < best_score:
                        best_score = current_score
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_h, best_score

import numpy as np
from scipy.optimize import linprog
import time

def exact_score(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    c = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(c) * dx)

def exact_z_vector(h):
    h = np.asarray(h, dtype=np.float64)
    n = len(h)
    dx = 2.0 / n
    return np.correlate(h, 1.0 - h, mode='full') * dx

def project_box_sum(h, target_sum):
    h = np.asarray(h, dtype=np.float64).copy()
    n = len(h)
    lo = -2.0
    hi = 2.0
    for _ in range(60):
        mid = (lo + hi) / 2.0
        s = np.sum(np.clip(h + mid, 0.0, 1.0))
        if s < target_sum:
            lo = mid
        else:
            hi = mid
    result = np.clip(h + (lo + hi) / 2.0, 0.0, 1.0)
    residual = target_sum - np.sum(result)
    if abs(residual) > 1e-10:
        idx = np.where((result > 1e-9) & (result < 1.0 - 1e-9))[0]
        if len(idx) > 0:
            result[idx] += residual / len(idx)
            result = np.clip(result, 0.0, 1.0)
    return result

def grad_z(h, q, n):
    dx = 2.0 / n
    ell = q - (n - 1)
    g = np.zeros(n)
    if ell == 0:
        g[:] = (1.0 - 2.0 * h) * dx
    elif ell > 0:
        L = n - ell
        g[ell:] += (1.0 - h[:L]) * dx
        g[:L] -= h[ell:] * dx
    else:
        a = -ell
        L = n - a
        g[:L] += (1.0 - h[a:]) * dx
        g[a:] -= h[:L] * dx
    return g

def smooth_max_descent(h0, max_iters=500, tau_init=0.001, lr_init=0.01, time_limit=None, t_start=None):
    """Smooth max-g-based gradient descent with projection."""
    h = project_box_sum(h0, len(h0) / 2.0)
    n = len(h)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    tau = tau_init
    lr = lr_init
    stall = 0
    
    if t_start is None:
        t_start = time.time()
    
    for it in range(max_iters):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        
        z_vec = exact_z_vector(h)
        z_max = np.max(z_vec)
        current_score = float(z_max)
        
        if current_score < best_score - 1e-10:
            best_score = current_score
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
            if stall > 50:
                break
        
        # Softmax weights
        z_shifted = (z_vec - z_max) / tau
        weights = np.exp(z_shifted)
        weights /= np.sum(weights)
        
        # Weighted gradient
        g_total = np.zeros(n)
        for q in range(len(z_vec)):
            if weights[q] > 1e-8:
                g_total += weights[q] * grad_z(h, q, n)
        
        # Gradient step
        h_new = h - lr * g_total
        h_new = project_box_sum(h_new, n / 2.0)
        
        # Line search
        best_alpha = 0.0
        best_trial = current_score
        for alpha in [1.0, 0.5, 0.25, 0.1, 0.05]:
            candidate = (1.0 - alpha) * h + alpha * h_new
            candidate = project_box_sum(candidate, n / 2.0)
            trial = exact_score(candidate)
            if trial < best_trial - 1e-12:
                best_trial = trial
                best_alpha = alpha
        
        if best_alpha > 0:
            h = (1.0 - best_alpha) * h + best_alpha * h_new
            h = project_box_sum(h, n / 2.0)
            current_score = best_trial
            lr = min(lr * 1.1, 0.1)
        else:
            lr *= 0.7
            tau *= 0.9
        
        # Decay tau to sharpen the max approximation
        tau = max(tau * 0.995, 1e-5)
    
    return best_h, best_score

def slp_optimize(h0, max_iters=300, rho_init=0.05, k_max=50, time_limit=None, t_start=None):
    """Sequential Linear Programming."""
    h = project_box_sum(h0, len(h0) / 2.0)
    n = len(h)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    rho = rho_init
    stall = 0
    
    if t_start is None:
        t_start = time.time()
    
    for it in range(max_iters):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        
        z_vec = exact_z_vector(h)
        z_max = float(np.max(z_vec))
        
        if z_max < best_score - 1e-12:
            best_score = z_max
            best_h = h.copy()
            stall = 0
        else:
            stall += 1
        
        if stall > 40:
            rho *= 0.5
            if rho < 1e-6:
                break
            stall = 0
        
        # Active set
        gap = max(2e-4, 0.015 * z_max)
        active_mask = z_vec >= (z_max - gap)
        active_idx = np.where(active_mask)[0]
        
        if len(active_idx) > k_max:
            order = np.argsort(-z_vec[active_idx])
            active_idx = active_idx[order[:k_max]]
        
        argmax_idx = int(np.argmax(z_vec))
        if argmax_idx not in active_idx:
            active_idx = np.append(active_idx, argmax_idx)
        
        K = len(active_idx)
        
        # Build LP
        c_obj = np.zeros(n + 1)
        c_obj[n] = 1.0
        
        A_ub = np.zeros((K, n + 1))
        b_ub = np.zeros(K)
        
        for i, q in enumerate(active_idx):
            g = grad_z(h, q, n)
            A_ub[i, :n] = g
            A_ub[i, n] = -1.0
            b_ub[i] = np.dot(g, h) - z_vec[q]
        
        A_eq = np.zeros((1, n + 1))
        A_eq[0, :n] = 1.0
        b_eq = np.array([n / 2.0])
        
        bounds = [(max(0.0, h[i] - rho), min(1.0, h[i] + rho)) for i in range(n)]
        bounds.append((0.0, z_max + 1.0))
        
        try:
            res = linprog(c_obj, A_ub=A_ub, b_ub=b_ub, A_eq=A_eq, b_eq=b_eq,
                         bounds=bounds, method='highs')
            if not res.success:
                rho *= 0.5
                continue
            
            y_new = res.x[:n]
            y_new = project_box_sum(y_new, n / 2.0)
            
            # Line search
            best_alpha = 0.0
            best_trial = z_max
            best_candidate = None
            for alpha in [1.0, 0.7, 0.5, 0.3, 0.1, 0.05]:
                candidate = (1.0 - alpha) * h + alpha * y_new

                if trial < best_trial - 1e-12:

                    best_alpha = alpha
                    best_candidate = candidate.copy()
            
            if best_candidate is not None:
                h = best_candidate
                current_score = best_trial
                if current_score < best_score:
                    best_score = current_score
                    best_h = h.copy()
                rho = max(rho * 0.95, 1e-5)
            else:
                rho *= 0.5
        except Exception:
            rho *= 0.5
            if rho < 1e-6:
                break
    
    return best_h, best_score

def pairwise_transfer(h0, max_iters=200, n_top=15, time_limit=None, t_start=None):
    """Pairwise transfer local search."""
    h = project_box_sum(h0, len(h0) / 2.0)
    n = len(h)
    current_score = exact_score(h)
    best_h = h.copy()
    best_score = current_score
    
    if t_start is None:
        t_start = time.time()
    
    for it in range(max_iters):
        if time_limit is not None and (time.time() - t_start) > time_limit:
            break
        
        z_vec = exact_z_vector(h)
        z_max = float(np.max(z_vec))
        
        gap = max(2e-4, 0.03 * z_max)
        weights = np.zeros(len(z_vec))
        active = z_vec >= (z_max - gap)
        weights[active] = (z_vec[active] - (z_max - gap))
        if np.max(weights) > 0:
            weights /= np.max(weights)
        else:
            weights[np.argmax(z_vec)] = 1.0
        
        g_total = np.zeros(n)
        for q in range(len(z_vec)):
            if weights[q] > 1e-6:
                g_total += weights[q] * grad_z(h, q, n)
        
        sorted_g = np.argsort(g_total)
        sources = sorted_g[-n_top:]
        targets = sorted_g[:n_top]
        
        improved = False
        for si in sources:
            if h[si] < 1e-8:
                continue
            for ti in targets:
                if ti == si:
                    continue
                if h[ti] > 1.0 - 1e-8:
                    continue
                max_delta = min(h[si], 1.0 - h[ti])
                if max_delta < 1e-8:
                    continue
                
                best_delta = 0.0
                best_trial = current_score
                for frac in [0.1, 0.3, 0.5, 0.1, 0.01]:
                    delta = frac * max_delta
                    if delta < 1e-10:
                        continue
                    candidate = h.copy()
                    candidate[si] -= delta
                    candidate[ti] += delta
                    trial = exact_score(candidate)
                    if trial < best_trial - 1e-12:
                        best_trial = trial
                        best_delta = delta
                
                if best_delta > 0:
                    h[si] -= best_delta
                    h[ti] += best_delta
                    current_score = best_trial
                    if current_score < best_score:
                        best_score = current_score
                        best_h = h.copy()
                    improved = True
                    break
            if improved:
                break
        
        if not improved:
            break
    
    return best_h, best_score
