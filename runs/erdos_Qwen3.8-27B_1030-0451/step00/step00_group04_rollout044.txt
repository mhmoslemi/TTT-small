import numpy as np
import time
from scipy.optimize import minimize

def eval_c5(h):
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(np.max(corr)) * dx

def project_box_sum(x, n):
    """Project x onto {h in [0,1]^n : sum(h) = n/2}"""
    target = n / 2.0
    # Binary search for lambda such that sum(clip(x - lambda, 0, 1)) = target
    lo, hi = x.min() - 1.0, x.max() + 1.0
    for _ in range(80):
        mid = (lo + hi) / 2.0
        y = np.clip(x - mid, 0.0, 1.0)
        s = y.sum()
        if s > target:
            lo = mid
        else:
            hi = mid
    y = np.clip(x - (lo + hi) / 2.0, 0.0, 1.0)
    # Fine repair for sum
    diff = target - y.sum()
    if abs(diff) > 1e-12:
        # Distribute evenly among non-bound entries
        mask = (y > 1e-10) & (y < 1.0 - 1e-10)
        if mask.sum() > 0:
            y[mask] += diff / mask.sum()
            y = np.clip(y, 0.0, 1.0)
    return y

def compute_z(h):
    """Compute all correlation lags scaled by dx"""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return corr * dx

def smooth_max_value(z, tau):
    z_max = z.max()
    shifted = (z - z_max) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -50, 50)
    log_sum = np.log(np.sum(np.exp(shifted)))
    return z_max + tau * log_sum

def pgd_optimize(h0, tau_start, tau_end, max_iter, time_limit):

    n = len(h0)
    h = project_box_sum(h0.copy(), n)
    F = eval_c5(h)  # track exact
    best_h = h.copy()
    best_F = F
    alpha = 0.02
    tau = tau_start
    
    t0 = time.time()
    for it in range(max_iter):
        if time.time() - t0 > time_limit:
            break
        if it % 20 == 0:
            tau = max(tau * 0.92, tau_end)
        
        F_smooth, grad = smooth_max_and_grad(h, tau)
        gnorm = np.linalg.norm(grad)
        if gnorm < 1e-12:
            break
        
        direction = -grad / gnorm
        
        # Backtracking line search
        accepted = False
        for trial in range(20):
            step = alpha * (0.5 ** trial)
            h_new_raw = h + step * direction
            h_new = project_box_sum(h_new_raw, n)
            F_new_smooth, _ = smooth_max_and_grad(h_new, tau)
            if F_new_smooth < F_smooth - 1e-10:
                h = h_new
                F_smooth = F_new_smooth
                accepted = True
                break

            if alpha < 1e-8:

                F_new = eval_c5(h)

                    best_F = F_new

    F_final = eval_c5(h)
    if F_final < best_F:
        best_F = F_final
        best_h = h.copy()
    
    return best_h, best_F
