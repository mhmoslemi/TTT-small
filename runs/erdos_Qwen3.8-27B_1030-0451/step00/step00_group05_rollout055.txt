import numpy as np
import time
from scipy.optimize import minimize

def compute_c5(h):
    """Compute exact C5."""
    n = len(h)
    dx = 2.0 / n
    corr = np.correlate(h, 1.0 - h, mode='full')
    return float(dx * np.max(corr))

def softmax_objective_and_grad(h, tau):
    """Compute soft-max objective value and gradient."""
    n = len(h)
    dx = 2.0 / n
    g = 1.0 - h
    corr = np.correlate(h, g, mode='full')
    scaled = corr / tau
    max_s = np.max(scaled)
    exps = np.exp(scaled - max_s)
    weights = exps / np.sum(exps)
    obj = tau * (np.log(np.sum(exps)) + max_s) * dx
    
    grad = np.zeros(n)
    center = n - 1
    for j in range(2 * n - 1):
        w = weights[j]
        if w < 1e-10:
            continue
        k = j - center
        if k >= 0:
            # grad[k:n] += w * (1 - h[0:n-k])
            # grad[0:n-k] += w * (-h[k:n])
            L = n - k
            if L > 0:
                grad[k:n] += w * (1.0 - h[:L])
                grad[:L] += -w * h[k:n]
        else:
            m = -k
            L = n - m
            if L > 0:
                grad[:L] += w * (1.0 - h[m:n])
                grad[m:n] += -w * h[:L]
    
    return obj, grad * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy 0<=h<=1 and sum(h)=target_sum."""
    h = np.clip(h, 0.0, 1.0)
    diff = target_sum - np.sum(h)
    if abs(diff) < 1e-12:
        return h
    # Add diff spread across all elements, respecting bounds
    # Simple approach: add diff/n to all, then clip, iterate
    for _ in range(50):
        h = h + diff / n
        h = np.clip(h, 0.0, 1.0)
        new_diff = target_sum - np.sum(h)
        if abs(new_diff) < 1e-14:
            return h
        diff = new_diff
    # If still not exact, adjust one element
    diff = target_sum - np.sum(h)
    if diff > 0:
        idx = np.argmin(h)
        h[idx] = min(1.0, h[idx] + diff)
    else:
        idx = np.argmax(h)
        h[idx] = max(0.0, h[idx] + diff)
    return h

    current_c5 = compute_c5(current)
    for tau in taus:
        if time.monotonic() > deadline:
            break
        h_new = do_slsqp(current, target_sum, tau)
        c_new = compute_c5(h_new)
        if c_new < current_c5 - 1e-14:
            current = h_new
            current_c5 = c_new
    return current, current_c5


def optimize_active_set(h, target_sum, deadline, max_iters=5000):
    n = len(h)
    current = h.copy()
    current_c5 = compute_c5(current)
    step_sizes = [0.05, 0.02, 0.01, 0.005, 0.002, 0.001, 0.0005]
    no_improve = 0
    for it in range(max_iters):
        if time.monotonic() > deadline:
            break
        grad = active_gradient(current, top_k=15)
        # Projected gradient step: try multiple step sizes
        improved = False
        for step in step_sizes:
            h_try = current - step * grad
            h_try = project_feasible(h_try, target_sum)
            c_try = compute_c5(h_try)
            if c_try < current_c5 - 1e-14:
                current = h_try
                current_c5 = c_try
                improved = True
                no_improve = 0
                break
        if not improved:
            no_improve += 1
            if no_improve > 50:
                # Try random kick
                kick = 0.01 * np.random.randn(n)
                h_try = project_feasible(current + kick, target_sum)
                c_try = compute_c5(h_try)
                if c_try < current_c5:
                    current = h_try
                    current_c5 = c_try
                    no_improve = 0
                elif no_improve > 200:
                    break
    return current, current_c5


def pairwise_guided(h, target_sum, deadline, rng, max_rounds=3000):
    n = len(h)
    current = h.copy()
    current_c5 = compute_c5(current)
    temperature = 5e-5
    for rnd in range(max_rounds):
        if time.monotonic() > deadline:
            break
        # Periodically cool
        if rnd % 200 == 199:
            temperature *= 0.85

        # Get gradient for guidance
        grad = active_gradient(current, top_k=10)

        # Select pairs: increase where grad is negative, decrease where positive
        order = np.argsort(grad)
        best_found = False
        for a in range(min(8, n)):
            for b in range(max(0, n - 8), n):
                i = int(order[a])
                j = int(order[b])
                if i == j:
                    continue
                max_d = min(current[j], 1.0 - current[i])
                if max_d < 1e-8:
                    continue
                for frac in [0.5, 0.25, 0.1, 0.05]:
                    d = max_d * frac
                    current[i] += d
                    current[j] -= d
                    c_try = compute_c5(current)
                    if c_try < current_c5 - 1e-15:
                        current_c5 = c_try
                        best_found = True
                        break
                    else:
                        # Metropolis acceptance
                        if c_try - current_c5 < temperature:
                            r = rng.random()
                            if np.log(r) < (current_c5 - c_try) / max(temperature, 1e-12):
                                best_found = True
                                current_c5 = c_try
                                break
                        current[i] -= d
                        current[j] += d
            if best_found:
                break
        if not best_found:
            # Random pair
            i = int(rng.integers(n))
            j = int(rng.integers(n))
            if i != j:
                max_d = min(current[j], 1.0 - current[i])
                if max_d > 1e-8:
                    d = max_d * rng.uniform(0.1, 0.4)
                    current[i] += d
                    current[j] -= d
                    c_try = compute_c5(current)
                    if c_try < current_c5 - 1e-15:
                        current_c5 = c_try
                    else:
                        if c_try - current_c5 < temperature:
                            r = rng.random()
                            if np.log(r) < (current_c5 - c_try) / max(temperature, 1e-12):
                                current_c5 = c_try
                            else:
                                current[i] -= d
                                current[j] += d
                        else:
                            current[i] -= d
                            current[j] += d
    return current, current_c5


def resample(h_old, n_new):
    n_old = len(h_old)
    if n_old == n_new:
        return h_old.copy()
    x_old = (np.arange(n_old) + 0.5) * (2.0 / n_old)
    x_new = (np.arange(n_new) + 0.5) * (2.0 / n_new)
    h_new = np.interp(x_new, x_old, h_old)
    target = n_new / 2.0
    return project_feasible(h_new, target)


def transform_candidates(h):
    """Generate simple transforms of h."""
    n = len(h)
    ts = n / 2.0
    candidates = [h.copy()]
    # Reverse
    candidates.append(project_feasible(h[::-1].copy(), ts))
    # Complement
    candidates.append(project_feasible(1.0 - h, ts))
    # Symmetrize
    sym = (h + h[::-1]) / 2.0
    candidates.append(project_feasible(sym, ts))
    # Reverse complement
    candidates.append(project_feasible(1.0 - h[::-1], ts))
    return candidates
