def smooth_max_obj(h, tau):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    C = r * dx
    # log-sum-exp for numerical stability
    m = np.max(C)
    return m + tau * np.log(np.sum(np.exp((C - m) / tau)))

def smooth_max_grad(h, tau):
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    C = r * dx
    m = np.max(C)
    w = np.exp((C - m) / tau)
    w /= np.sum(w)
    # Now compute gradient
    # Use the formula derived above
    g = np.zeros(n)
    for d in range(n):
        # Positive lag: k = n-1+d
        w_pos = w[n-1+d]
        if w_pos < 1e-15:
            continue
        # ∂r[n-1+d]/∂h[m] for m in [d, n-1]: (1-h[m-d])
        # ∂r[n-1+d]/∂h[m] for m in [0, n-1-d]: -h[m+d]
        if d <= n-1:
            g[d:] += w_pos * (1 - h[:n-d])
        if 0 <= n-1-d:
            g[:n-d] -= w_pos * h[d:]
        
        # Negative lag: k = n-1-d (only for d > 0 to avoid double counting d=0)
        if d > 0:
            w_neg = w[n-1-d]
            if w_neg < 1e-15:
                continue
            # ∂r[n-1-d]/∂h[m] for m in [0, n-1-d]: (1-h[m+d])

            if n-1-d >= 0:
                g[:n-d] += w_neg * (1 - h[d:])
            if d <= n-1:
                g[d:] -= w_neg * h[:n-d]

    return g

import numpy as np
from scipy.optimize import minimize
import time

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2}"""
    target = n / 2.0
    current_sum = np.sum(y)
    if abs(current_sum - target) < 1e-10:
        return np.clip(y, 0, 1)
    
    # Bisection to find t such that sum(clip(y+t, 0, 1)) = target
    lo = target - np.sum(np.clip(y, 0, 1)) - 1  # will be too low
    hi = 1 - y[np.argmax(y)] + (target - np.sum(np.clip(y, 0, 1))) + 1
    
    # Actually simpler: F(t) = sum(clip(y+t, 0, 1)) is monotone
    # F(-inf) = 0, F(inf) = n
    # We need F(t) = target
    lo = y.min() + target - y.max() - 1  # rough lower bound
    hi = y.max() - y.min() + 1  # rough upper bound
    
    # Better: lo such that all become 0, hi such that all become 1
    lo = -y.max() - 0.1
    hi = 1 - y.min() + 0.1
    
    for _ in range(100):
        mid = (lo + hi) / 2
        s = np.sum(np.clip(y + mid, 0, 1))
        if s < target:
            lo = mid
        else:
            hi = mid
    
    t = (lo + hi) / 2
    result = np.clip(y + t, 0, 1)
    
    # Fix any residual
    residual = target - np.sum(result)
    if abs(residual) > 1e-10:
        # Distribute residual to elements not at boundary
        if residual > 0:
            mask = result < 1.0
        else:
            mask = result > 0.0
        if np.any(mask):
            result[mask] += residual / np.sum(mask)
            result = np.clip(result, 0, 1)
    
    return result

def compute_c5(h):
    """Compute the exact C5 value"""
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    return np.max(r) * dx

def smooth_max_and_grad(h, tau):
    """Compute smooth max objective and gradient"""
    n = len(h)
    dx = 2.0 / n
    r = np.correlate(h, 1.0 - h, mode='full')
    C = r * dx
    m = np.max(C)
    shifted = (C - m) / tau
    # Clip for numerical stability
    shifted = np.clip(shifted, -100, 100)
    exp_shifted = np.exp(shifted)
    log_sum = np.log(np.sum(exp_shifted))
    obj = m + tau * log_sum
    
    w = exp_shifted / np.sum(exp_shifted)
    
    # Compute gradient
    g = np.zeros(n)
    for d in range(n):
        # Positive lag d: k = n-1+d
        wp = w[n - 1 + d]
        if wp > 1e-15:
            # For m in [d, n-1]: gradient contribution (1-h[m-d])
            g[d:] += wp * (1.0 - h[:n - d])
            # For m in [0, n-1-d]: gradient contribution -h[m+d]
            if n - 1 - d >= 0:
                g[:n - d] -= wp * h[d:]
        
        # Negative lag d: k = n-1-d (skip d=0, already counted)
        if d > 0:
            wn = w[n - 1 - d]
            if wn > 1e-15:
                # For m in [0, n-1-d]: gradient contribution (1-h[m+d])
                if n - 1 - d >= 0:
                    g[:n - d] += wn * (1.0 - h[d:])
                # For m in [d, n-1]: gradient contribution -h[m-d]
                g[d:] -= wn * h[:n - d]
    
    g *= dx
    return obj, g

def project_feasible(y, n):
    """Project y onto {h: 0<=h<=1, sum(h)=n/2} via bisection on additive shift."""
    target = n / 2.0
    y = np.asarray(y, dtype=float)
    lo = -y.max() - 0.01
    hi = 1.0 - y.min() + 0.01
    for _ in range(80):
        mid = (lo + hi) / 2.0
        if np.sum(np.clip(y + mid, 0.0, 1.0)) < target:
            lo = mid
        else:
            hi = mid

    result = np.clip(y + t, 0.0, 1.0)
    residual = target - np.sum(result)
    if abs(residual) > 1e-12:

        else:
            mask = result > 1e-9
